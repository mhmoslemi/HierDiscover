"""
Propose a step‑function (non‑negative vector) that minimizes the
evaluation function

    O(a) = 2·n·max(conv(a,a)) / (sum(a)²)

by a multi‑restart stochastic optimisation:

* Several random initialisations (uniform, Dirichlet, Beta‑shaped,
  random‑phase, previous best if available).
* Phase 1 – Adam on a smooth log‑sum‑exp surrogate of the max‑convolution.
* Phase 2 – deterministic sub‑gradient descent on the true max entry.
* Phase 3 – simulated‑annealing mass‑transfer hill climbing.
* Symmetry (a ≈ a[::-1]) is enforced after each update to reduce the
  effective search space.
* The whole search respects a 1000 s wall‑clock budget; the best sequence
  found is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x ≥ 0, Σx = 1}
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        # all non‑positive – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full self‑convolution (length 2·n‑1)
def conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution a * a using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
def objective(a: np.ndarray, n: int, L: int) -> float:
    """Exact objective for a simplex‑normalised vector."""
    conv = conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. a (averaged over all maximisers)."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    g /= len(ks)  # average if several maximisers
    return g


# ----------------------------------------------------------------------
def smooth_gradient(a: np.ndarray, conv: np.ndarray, alpha: float, L: int) -> np.ndarray:
    """
    Gradient of the smooth surrogate
        f_α(a) = (2·n/α)·log Σ_k exp(α·conv_k)
    Returns 2·corr(w, a) where w_k ∝ exp(α·conv_k).
    """
    # soft‑max weights (stable)
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / exp_scaled.sum()

    # correlation w ⋆ a   (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)   # length ≥ 3·n‑2
    n = a.size
    grad = 2.0 * corr_rev[n - 1 : n - 1 + n]          # central n entries
    return grad


# ----------------------------------------------------------------------
def generate_initial(n: int, L: int) -> np.ndarray:
    """
    Produce a random feasible start vector.
    Several strategies are mixed to increase diversity.
    """
    choice = np.random.rand()
    if 'height_sequence_1' in globals():
        # occasionally reuse the previous best (if it exists)
        if choice < 0.15:
            base = np.array(height_sequence_1, dtype=float)
            base = base[:n]                     # trim / pad if needed
            if base.size < n:
                # pad with tiny uniform mass
                pad = np.full(n - base.size, 1e-12)
                base = np.concatenate([base, pad])
            return project_to_simplex(base)

    if choice < 0.25:
        # uniform
        return np.full(n, 1.0 / n)

    if choice < 0.45:
        # Dirichlet with random concentration
        concentration = np.random.rand() * 5.0 + 0.1   # between 0.1 and 5.1
        a = np.random.gamma(concentration, 1.0, size=n)
        return project_to_simplex(a)

    if choice < 0.65:
        # Beta‑shaped (smooth) distribution
        alpha = np.random.uniform(0.2, 4.5)
        beta = np.random.uniform(0.2, 4.5)
        x = (np.arange(n) + 0.5) / n
        a = (x ** (alpha - 1)) * ((1.0 - x) ** (beta - 1))
        a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
        return project_to_simplex(a)

    if choice < 0.85:
        # Random‑phase Fourier construction
        # constant magnitude spectrum, random phases
        phases = np.random.rand(L // 2 + 1) * 2.0 * np.pi
        spectrum = np.exp(1j * phases)
        a_full = np.fft.irfft(spectrum, n=L)[:n]
        a_full = np.maximum(a_full, 0.0)          # clip negatives
        return project_to_simplex(a_full)

    # fallback: smooth random vector via Gaussian kernel
    raw = np.random.rand(n)
    sigma = np.random.uniform(2.0, max(4.0, n / 30.0))
    half = int(3 * sigma)
    xs = np.arange(-half, half + 1)
    gauss = np.exp(-0.5 * (xs / sigma) ** 2)
    gauss /= gauss.sum()
    smooth = np.convolve(raw, gauss, mode='same')
    return project_to_simplex(smooth)


# ----------------------------------------------------------------------
def optimise_one_run(a0: np.ndarray,
                     n: int,
                     L: int,
                     run_deadline: float) -> tuple[np.ndarray, float]:
    """
    Perform Phase‑1 (Adam on smooth surrogate), Phase‑2 (sub‑gradient)
    and Phase‑3 (simulated annealing) on a single starting vector.
    Returns the best vector seen in this run and its objective value.
    """
    # hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # ------------------------------------------------------------------
    # Phase 1 – Adam on smooth surrogate
    a = a0.copy()
    best_a = a.copy()
    best_obj = objective(a, n, L)

    max_iter_adam = 120_000                     # enough for a 60 s slice
    alpha_start, alpha_end = 5.0, 3000.0
    lr = 0.07
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(max_iter_adam):
        if time.time() >= run_deadline - 5.0:
            break

        conv = conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            print(f"[adam] it={it:,} O={best_obj:.6f}")

        # anneal α linearly
        frac = it / max_iter_adam
        alpha = alpha_start + (alpha_end - alpha_start) * frac

        # gradient of smooth surrogate
        grad = smooth_gradient(a, conv, alpha, L)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # projected symmetric update
        a_candidate = a - step
        a_candidate = (a_candidate + a_candidate[::-1]) * 0.5
        a = project_to_simplex(a_candidate)

    # ------------------------------------------------------------------
    # Phase 2 – deterministic sub‑gradient refinement
    max_iter_sub = 80_000
    for it in range(max_iter_sub):
        if time.time() >= run_deadline - 5.0:
            break

        conv = conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            print(f"[subgrad] it={it:,} O={best_obj:.6f}")

        g = subgradient_max_conv(a, conv)

        # linearly decaying step size (a bit larger than previous)
        eta = 0.15 * (1.0 - it / max_iter_sub)
        a = a - eta * g
        a = (a + a[::-1]) * 0.5
        a = project_to_simplex(a)

        # occasional tiny jitter to escape tiny plateaus
        if it % 2000 == 0 and it > 0:
            jitter = np.random.randn(n) * 4e-4
            a = project_to_simplex(a + jitter)

    # ------------------------------------------------------------------
    # Phase 3 – simulated‑annealing hill climbing (global refinement)
    local_a = best_a.copy()
    local_obj = 2.0 * n * conv_fft(local_a, L).max()
    total_hc = 0
    T0 = 0.04

    while time.time() < run_deadline - 1.0:
        T = T0 * (0.998 ** total_hc)

        # propose a random mass‑transfer move
        i, j = np.random.randint(n, size=2)
        if i == j or local_a[i] <= 0.0:
            total_hc += 1
            continue
        delta = local_a[i] * np.random.rand() * 0.15
        a_new = local_a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = (a_new + a_new[::-1]) * 0.5
        a_new = project_to_simplex(a_new)

        conv_new = conv_fft(a_new, L)
        obj_new = 2.0 * n * conv_new.max()

        # Metropolis acceptance
        if obj_new < local_obj or np.random.rand() < math.exp((local_obj - obj_new) / max(T, 1e-12)):
            local_a = a_new
            local_obj = obj_new
            if local_obj < best_obj:
                best_obj = local_obj
                best_a = local_a.copy()
                print(f"[anneal] iter={total_hc:,} O={best_obj:.6f}")

        total_hc += 1

        if total_hc % 5000 == 0:
            # Re‑anchor the local walk at the current global best
            local_a = best_a.copy()
            local_obj = 2.0 * n * conv_fft(local_a, L).max()

    return best_a, best_obj


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Multi‑restart optimisation for the step‑function minimisation problem.
    Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5       # safety margin

    # ------------------------------------------------------------------
    # Choose problem size.
    if 'height_sequence_1' in globals():
        base = np.array(height_sequence_1, dtype=float)
        n = int(base.size)
        print(f"[init] Using length from previous best: n={n}")
    else:
        n = 4096
        print(f"[init] No previous sequence – using n={n}")

    # FFT length – must accommodate full convolution (2n‑1) and correlation (3n‑2)
    L = 1 << ((3 * n - 2).bit_length())  # next power of two ≥ 3n‑2

    # Global best across all restarts
    global_best_a = None
    global_best_obj = float('inf')
    run_counter = 0

    # ------------------------------------------------------------------
    # Main optimisation loop – many short restarts until time runs out
    while time.time() < deadline - 15.0:       # keep ~15 s for finalisation
        run_counter += 1
        # allocate a slice of the remaining budget (≈ 60 s per run, but adapt)
        now = time.time()
        remaining = deadline - now
        slice_budget = min(65.0, remaining * 0.9)   # leave room for final checks
        run_deadline = now + slice_budget

        # random feasible start
        a0 = generate_initial(n, L)

        # run the three‑phase optimiser
        best_a_run, best_obj_run = optimise_one_run(a0, n, L, run_deadline)

        if best_obj_run < global_best_obj:
            global_best_obj = best_obj_run
            global_best_a = best_a_run.copy()
            print(f"[global] restart={run_counter} new best O={global_best_obj:.6f}")

    # ------------------------------------------------------------------
    # Final tiny local refinement (few Adam iterations with very high α)
    if global_best_a is not None:
        a = global_best_a.copy()
        # a quick Adam pass to polish
        max_iter_final = 10_000
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        lr = 0.03
        alpha_start, alpha_end = 2000.0, 8000.0
        for it in range(max_iter_final):
            if time.time() >= deadline - 1.0:
                break
            conv = conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < global_best_obj:
                global_best_obj = cur_obj
                global_best_a = a.copy()
                print(f"[final‑adam] it={it:,} O={global_best_obj:.6f}")

            alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iter_final)
            grad = smooth_gradient(a, conv, alpha, L)

            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** (it + 1))
            v_hat = v / (1.0 - beta2 ** (it + 1))
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            a_candidate = a - step
            a_candidate = (a_candidate + a_candidate[::-1]) * 0.5
            a = project_to_simplex(a_candidate)

        # assure final symmetry & normalisation
        global_best_a = project_to_simplex((global_best_a + global_best_a[::-1]) * 0.5)

    # ------------------------------------------------------------------
    # Sanity check with the provided `evaluate_sequence`
    try:
        final_score = evaluate_sequence(global_best_a.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence failed:", exc)

    return global_best_a.tolist()
