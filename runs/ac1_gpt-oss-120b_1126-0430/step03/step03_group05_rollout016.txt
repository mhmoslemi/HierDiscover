- Initialise with a power‑law sequence on the simplex.
- Optimise the objective `2 n max(conv) / (sum a)^2` (sum=1) using
  Adam on a smooth approximation of the `max` (log‑sum‑exp) of the
  autocorrelation.  Gradients are obtained analytically via FFT‑based
  convolutions, and each iterate is projected onto the probability

def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex (sum=1, v>=0)."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:

        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    print(f"Initial scan → best p = {best_p}, cheap score = {best_score:.6f}")

    # 2. Adam optimisation on a smooth approximation of max‑convolution
    # ---------------------------------------------------------------
    a = best_a.copy()                     # current iterate (Σa = 1)
    m = np.zeros_like(a)                  # first moment estimate
    v = np.zeros_like(a)                  # second moment estimate
    beta1 = 0.90
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05                               # base learning rate
    alpha = 5.0                             # temperature of the smooth‑max
    alpha_max = 50.0

    best_iter_a = a.copy()
    best_iter_score = best_score
    iters_without_improve = 0
    iteration = 0

    while time.time() < deadline and iteration < 300_000:
        iteration += 1

        # ----- full autocorrelation of a -----
        Fa = np.fft.fft(a, fft_len_a)
        conv = np.fft.ifft(Fa * Fa).real[: 2 * n - 1]
        maxb = conv.max()
        cheap_cur = 2.0 * n * maxb

        # ----- smooth‑max (log‑sum‑exp) -----
        shifted = conv - maxb                # stabilise exponentials
        exp_vals = np.exp(alpha * shifted)   # all ≤ 1
        w_sum = exp_vals.sum()
        if w_sum == 0 or np.isinf(w_sum):
            # numerical trouble – fall back to uniform weights
            weights = np.full_like(exp_vals, 1.0 / exp_vals.size)
        else:
            weights = exp_vals / w_sum

        # ----- gradient of the smooth max -----
        rev_a = a[::-1]
        Wf = np.fft.fft(weights, fft_len_w)
        RevAf = np.fft.fft(rev_a, fft_len_w)
        conv_wr = np.fft.ifft(Wf * RevAf).real
        grad = 2.0 * conv_wr[(n - 1) : (n - 1 + n)]

        # ----- Adam descent step -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a - step
        a_candidate = _simplex_projection(a_candidate)

        # ----- cheap score of the candidate -----
        maxb_cand, _ = _conv_max_fft(a_candidate, fft_len_a)
        cheap_cand = 2.0 * n * maxb_cand

        if cheap_cand < best_iter_score - 1e-12:
            # accept improvement
            a = a_candidate
            best_iter_score = cheap_cand
            best_iter_a = a.copy()
            iters_without_improve = 0
            lr = min(lr * 1.02, 0.20)        # gently increase LR when things improve
            print(f"[Iter {iteration:07d}] NEW BEST cheap = {best_iter_score:.6f} (lr={lr:.4f})")
            if best_iter_score <= 1.503:     # target reached – stop early
                print("Target threshold reached; terminating early.")
                break
        else:
            # no improvement – possibly shrink learning rate later
            iters_without_improve += 1
            if iters_without_improve % 2000 == 0:
                lr = max(lr * 0.5, 1e-4)
                # perturb around the best known vector to escape plateaus
                a = _simplex_projection(best_iter_a + 0.02 * np.random.rand(n))
                m.fill(0.0)
                v.fill(0.0)
                print(f"[Iter {iteration:07d}] stagnation → lr={lr:.5f}, perturbation applied")
                iters_without_improve = 0

        # ----- increase smooth‑max temperature slowly -----
        if iteration % 2000 == 0:
            alpha = min(alpha * 1.10, alpha_max)

        # ----- occasional logging -----
        if iteration % 5000 == 0:
            print(f"[Iter {iteration:07d}] cur cheap = {cheap_cur:.6f}, best = {best_iter_score:.6f}, α={alpha:.2f}, lr={lr:.5f}")

    # ---------------------------------------------------------------
    # 3. Final exact evaluation and return the best sequence found
    # ---------------------------------------------------------------
    final_seq = best_iter_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as e:
        final_score = float('inf')
        print(f"Final evaluation failed: {e}")

    return final_seq
