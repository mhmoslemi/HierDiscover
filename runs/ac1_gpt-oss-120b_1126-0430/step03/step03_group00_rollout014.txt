"""
Propose a non‑negative step‑function that (approximately) minimises

    O(a) = 2·n·max(conv(a,a)) / (sum(a)²)

The algorithm is a two‑phase heuristic:

* Phase 1 – Adam on a smooth log‑sum‑exp surrogate of the max‑convolution.
  The surrogate is
        f_α(a) = (2·n/α)·log Σ_k e^{α·conv_k(a)} .
  Its gradient is 2·corr(w,a) with w_k ∝ e^{α·conv_k}.  Adam adapts the
  learning rate and the result is projected onto the probability simplex.
  The smoothing parameter α is increased exponentially (5 → 8000).

* Phase 2 – Simulated‑annealing / stochastic local search.
  Small “mass‑transfer’’ moves (δ from index i to j) are proposed,
  the exact objective is evaluated (full FFT convolution) and a
  Metropolis rule with a slowly cooling temperature is applied.
  Every few iterations a sub‑gradient step for the actual max‑entry
  is taken (the same direction used in the reference implementation).
  Accepted moves update the current vector and its convolution; the
  best vector seen so far is stored.

Finally a tiny convex combination with the uniform distribution is
tested (λ∈[0,0.2]) – this often smoothes residual spikes and yields a
modest extra improvement.

All steps respect the 1000 s budget, use only NumPy (no external
solvers) and keep the sequence on the simplex, so the supplied
``evaluate_sequence`` routine never rejects the result.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:               # all entries non‑positive -> uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution via FFT (real input, real output)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return a*a (full linear convolution) using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    # length of full convolution = 2*len(a)-1
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Exact objective for a simplex‑normalised vector
def _objective_simplex(a: np.ndarray, n: int, L: int) -> float:
    """2·n·max(conv(a,a))  (a is assumed to sum to 1)."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Sub‑gradient of the maximal convolution entry (as in the reference code)
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Return a sub‑gradient of max(conv) w.r.t. a."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)   # possibly several maximisers
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        # derivative of Σ a_i a_j w.r.t. a_i is 2·a_j
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size > 0:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
# Gradient of the log‑sum‑exp (soft‑max) surrogate of max(conv)
def _smooth_gradient(a: np.ndarray, conv: np.ndarray,
                    alpha: float, L: int) -> np.ndarray:
    """
    ∇_a  ( (2·n/α)·log Σ_k e^{α·conv_k} )
    = 2·corr(w, a)   with  w_k ∝ e^{α·conv_k}.
    """
    # soft‑max weights (stable version)
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / exp_scaled.sum()            # shape (2n‑1,)

    # correlation w ⋆ a  (g_i = 2·Σ_j w_{i+j} a_j)
    a_rev = a[::-1]                               # reverse for correlation
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)   # length ≥ 3n‑2
    n_len = a.size
    grad = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]
    return grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that
    minimises the provided evaluation function.  Returns the best
    sequence found as a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin so we never overrun the 1000 s limit
    deadline = start_time + budget_s - 0.3

    # ------------------------------------------------------------------
    # Problem size ------------------------------------------------------
    # If a previous construction is available we keep the size,
    # otherwise we deliberately pick a larger size (≈8000) – more degrees
    # of freedom often give a lower constant c = n·max(conv).
    if 'height_sequence_1' in globals():
        base = np.array(height_sequence_1, dtype=float)
        n = base.size
        a = base / base.sum()
        print(f"[init] using supplied sequence, n={n}")
    else:
        n = 8000                       # larger than the previous 4587
        # start from a 50/50 blend of uniform and an exponential tail
        uniform = np.full(n, 1.0 / n)
        exp_decay = np.exp(-np.arange(n) / (n / 8.0))
        exp_decay /= exp_decay.sum()
        a = 0.5 * uniform + 0.5 * exp_decay
        a = _project_to_simplex(a)
        print(f"[init] no supplied sequence – using blended start, n={n}")

    # FFT length – next power of two ≥ 3·n‑2 (required for correlation)
    L = 1 << ((3 * n - 2).bit_length())

    # ------------------------------------------------------------------
    # Phase 1 – Adam on the soft‑max surrogate -------------------------
    best_seq = a.copy()
    best_score = _objective_simplex(best_seq, n, L)
    print(f"[phase‑1] start O={best_score:.6f}")

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    lr0 = 0.4                 # base learning rate (decays as 1/√t)
    alpha_start = 5.0
    alpha_end = 8000.0        # a stronger focus on the true maximum
    max_iter_phase1 = 250_000

    # allocate roughly 55 % of the total budget to phase 1
    phase1_deadline = start_time + budget_s * 0.55
    it = 0
    while time.time() < phase1_deadline and it < max_iter_phase1:
        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_score - 1e-12:
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[phase‑1] iter {it:6d} new best O={best_score:.6f}")

        # exponential schedule for α (smooth → sharp)
        frac = it / max_iter_phase1
        alpha = alpha_start * ((alpha_end / alpha_start) ** frac)

        # smooth gradient of the log‑sum‑exp surrogate
        grad = _smooth_gradient(a, conv, alpha, L)

        # Adam update (bias‑corrected moments)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        # step‑size decays like 1/√t
        lr = lr0 / math.sqrt(it + 1)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional tiny jitter to escape plateaus
        if np.random.rand() < 0.001:
            a_new += np.random.randn(n) * 4e-5

        a_new = _project_to_simplex(a_new)

        # gentle support pruning – drop the lowest 0.5 % after enough
        # iterations (helps speed‑up later phases)
        if it % 20000 == 0 and it > 0:
            thresh = np.percentile(a_new, 0.5)
            a_new[a_new < thresh] = 0.0
            a_new = _project_to_simplex(a_new)

        a = a_new
        it += 1

    # ------------------------------------------------------------------
    # Phase 2 – Simulated‑annealing + sub‑gradient refinements ----------
    # The best vector from Phase 1 is taken as a warm‑start.
    print(f"[phase‑2] entering local refinement, best O={best_score:.6f}")

    max_iter_local = 140_000          # inner loop budget per restart
    subgrad_interval = 12
    temperature0 = 0.06
    max_frac = 0.12                   # max relative mass transferred per move
    no_improve_limit = 6000

    restarts = 0
    while time.time() < deadline:
        # ---- initialise a new restart ----
        if np.random.rand() < 0.6:                # mostly start from current best
            a = best_seq.copy()
            a += np.random.randn(n) * 0.001      # small perturbation
            a = np.clip(a, 0, None)
            a = _project_to_simplex(a)
        else:
            a = np.random.rand(n)
            a = _project_to_simplex(a)

        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        it_loc = 0
        no_improve = 0
        T = temperature0

        while time.time() < deadline and it_loc < max_iter_local:
            # ---- mass‑transfer move ----
            # bias the source index towards larger entries
            src = np.random.choice(n, p=a)               # heavy indices more likely
            dst = np.random.randint(n)                   # destination uniformly random
            if src != dst and a[src] > 1e-15:
                delta = a[src] * np.random.rand() * max_frac
                a_new = a.copy()
                a_new[src] -= delta
                a_new[dst] += delta
                # projection is not needed – sum is preserved, only positivity
                conv_new = _conv_fft(a_new, L)
                new_obj = 2.0 * n * conv_new.max()

                # Metropolis acceptance
                if new_obj < cur_obj or np.random.rand() < math.exp((cur_obj - new_obj) / max(T, 1e-12)):
                    a = a_new
                    conv = conv_new
                    cur_obj = new_obj
                    no_improve = 0
                    if cur_obj < best_score:
                        best_score = cur_obj
                        best_seq = a.copy()
                        print(f"[{time.time():.1f}] new global best O={best_score:.6f} "
                              f"(restart={restarts}, iter={it_loc})")
                else:
                    no_improve += 1
            else:
                no_improve += 1

            # ---- sub‑gradient step on the current maximal entry ----
            if it_loc % subgrad_interval == 0:
                g = _subgradient_max_conv(a, conv)
                eta = 0.07 * (1.0 - it_loc / max_iter_local)   # linearly decaying step
                a = a - eta * g
                a = _project_to_simplex(a)
                conv = _conv_fft(a, L)        # recompute after the projection
                cur_obj = 2.0 * n * conv.max()
                if cur_obj < best_score:
                    best_score = cur_obj
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] sub‑grad improved O={best_score:.6f}")

            # temperature decay (slow)
            T *= 0.9996
            it_loc += 1

            # restart if we have been stuck for a while
            if no_improve >= no_improve_limit:
                break

        restarts += 1
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Final “smooth‑mix’’ with the uniform distribution -----------------
    # A tiny convex combination with the uniform vector often reduces the
    # remaining peak without breaking the good structure that the search
    # has found.
    uniform = np.full(n, 1.0 / n)
    best_mixed = best_seq.copy()
    best_mixed_score = best_score

    for lam in np.linspace(0.0, 0.20, 21):          # λ = 0,0.01,…,0.20
        cand = (1.0 - lam) * best_seq + lam * uniform
        cand = _project_to_simplex(cand)           # safeguard against tiny rounding errors
        score = _objective_simplex(cand, n, L)
        if score < best_mixed_score:
            best_mixed_score = score
            best_mixed = cand.copy()
            print(f"[mix] λ={lam:.3f} improved O to {best_mixed_score:.6f}")

    best_seq = best_mixed
    best_score = best_mixed_score

    # ------------------------------------------------------------------
    # Sanity check with the provided evaluator
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:
        print("[final] evaluate_sequence failed:", e)

    # Return the best found sequence (plain list of floats)
    return best_seq.tolist()
