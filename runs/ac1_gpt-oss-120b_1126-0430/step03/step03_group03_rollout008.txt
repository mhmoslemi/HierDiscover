"""
Propose a step‑function (non‑negative vector on the probability simplex) that
minimises

        O(a) = 2·n·max(conv(a,a))   with   Σ a_i = 1 ,

where conv(a,a) is the full (aperiodic) auto‑convolution.
The algorithm improves on the previous baseline by:

* representing the simplex point with unconstrained logits `z`
  (a = softmax(z)) – no explicit projection is needed;
* minimising a smooth log‑sum‑exp surrogate of the max‑convolution
  with a *geometric* schedule of the smoothing parameter α;
* using SciPy’s L‑BFGS‑B optimizer (exact gradient supplied) for fast
  quasi‑Newton steps;
* performing several random restarts (different initial logits);
* a final deterministic sub‑gradient polishing phase and a short
  simulated‑annealing hill‑climb for fine tuning.

All heavy work (FFT‑based convolution, gradient computation) is done with
numpy’s FFT on a power‑of‑two length to keep the per‑iteration cost low.
The routine respects the 1000 s budget and prints intermediate objective
values.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a vector on the simplex."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / np.sum(e)


def _logsumexp(x: np.ndarray) -> float:
    """Stable log‑sum‑exp."""
    x_max = np.max(x)
    return float(x_max + np.log(np.sum(np.exp(x - x_max))))


# ----------------------------------------------------------------------
# FFT‑based full self‑convolution (length 2·n‑1)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full convolution a * a using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Exact objective for a simplex‑normalised vector
def _objective_exact(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a))  (sum(a) is assumed to be 1)."""
    n = a.size
    L = 1 << ((3 * n - 2).bit_length())
    conv = _conv_fft(a, L)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Sub‑gradient of the maximal convolution entry (used in Phase 2)
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. a (a ≥ 0, Σa=1)."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)          # indices achieving the max
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    g /= len(ks)                                   # average if several maximisers
    return g


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient w.r.t. logits `z`
def _smooth_obj_and_grad(z: np.ndarray, n: int, L: int, alpha: float):
    """
    Compute the smoothed objective
        fα(a) = (2·n/α)·log Σ_k exp(α·conv_k)
    and its gradient w.r.t. the logits `z` (a = softmax(z)).
    Returns (f, grad_z).
    """
    # a on the simplex
    a = _softmax(z)

    # full convolution a*a
    conv = _conv_fft(a, L)

    # stable log‑sum‑exp of α·conv
    max_c = np.max(conv)
    # log Σ exp(α·conv) = α·max_c + log Σ exp(α·(conv-max_c))
    log_sum_exp = alpha * max_c + np.log(np.sum(np.exp(alpha * (conv - max_c))))
    f = (2.0 * n / alpha) * log_sum_exp

    # soft‑max weights over convolution entries
    exp_term = np.exp(alpha * (conv - max_c))          # = exp(α·conv) / exp(α·max_c)
    w = exp_term / np.sum(exp_term)                   # shape (2n‑1,)

    # gradient w.r.t. a : 2 * correlation(w, a)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)    # length ≥ 3·n‑2
    grad_a = 2.0 * corr_rev[n - 1 : n - 1 + n]        # central n entries

    # chain rule: ∂f/∂z = (diag(a) - a aᵀ) · ∂f/∂a
    mu = np.dot(a, grad_a)
    grad_z = a * (grad_a - mu)

    return f, grad_z
