    a = a0.copy()
    best = a.copy()
    best_obj = _objective(a)
    n = len(a)

    size = 1 << ((3*n-2).bit_length())
    # precompute indices for gradient extraction:
    idx_g = (2*n-2) - np.arange(n)   # shape (n,)
    alpha = 10.0
    alpha_max = 2000.0
    alpha_inc = (alpha_max - alpha) / max_iters
    eta0 = 0.2
    eps = 1e-15

    no_improve = 0
    patience = max(1000, max_iters // 20)

    for t in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        # compute FFT of a once
        fft_a = np.fft.fft(a, size)

        # compute self conv
        conv_full_self = np.fft.ifft(fft_a * fft_a).real
        conv_self = conv_full_self[:2*n-1]
        max_conv = conv_self.max()
        obj = 2.0 * n * max_conv

        # compute softmax weights
        w_exp = np.exp(alpha * (conv_self - max_conv))  # max at zero; w_exp <= 1
        w_norm = w_exp / w_exp.sum()  # shape 2n-1

        # compute gradient
        w_rev = w_norm[::-1]   # shape 2n-1
        fft_w_rev = np.fft.fft(w_rev, size)
        conv_full_grad = np.fft.ifft(fft_a * fft_w_rev).real[:3*n-2]
        g = 2.0 * conv_full_grad[idx_g]

        # step size adapt
        eta = eta0 * (1.0 - t / max_iters)  # linear decay

        # exponentiated gradient update
        log_a = np.log(a + eps) - eta * g

        log_a -= np.max(log_a)

"""
Non‑negative step‑function optimiser.

The objective to minimise is

    O(a) = 2·n·max(conv(a,a))    with  Σ a_i = 1 ,  a_i ≥ 0

where conv is the linear convolution.  The bound is scale‑invariant, so the
search is carried out on the probability simplex.

The algorithm proceeds in three phases:

1. **Parameter sweep** – a low‑dimensional family a_i ∝ (i+β)^{-p} is sampled
   (≈200 random (p,β) pairs) and the best shape is kept as an extra start.

2. **Continuation optimisation** – from each start we run a mirror‑descent
   (exponentiated‑gradient) iteration on a *smooth* approximation of
   max(·).  The smooth function is
        f_α(a) = (1/α)·log(∑_k e^{α·conv_k(a)}),
   whose gradient can be obtained by a single additional FFT per iteration.
   The temperature α is increased slowly (α≈5 → 5 000) so that the method
   first sees a smooth landscape and then converges to a true max‑objective.

3. **Final sub‑gradient polishing** – a short run of the classic sub‑gradient
   exponentiated‑gradient loop (as used in the previous state‑of‑the‑art) is
   applied to the best vector found.

All updates stay on the simplex via the usual normalisation of the
exponentiated variables.  Random jitter is injected when a run stagnates.
The routine respects the global time‑budget (default 1000 s) and returns the
best sequence found, ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact bound   O = 2·n·max(conv(a,a))   (a is already normalised)."""
    n = a.size
    size = 1 << ((2 * n - 2).bit_length())          # next power of two
    A = np.fft.fft(a, n=size)
    conv = np.fft.ifft(A * A).real[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
def _smooth_optimize(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the smoothed max‑convolution.
    Returns the best vector found and its exact (non‑smoothed) objective.
    """
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_obj = _objective(a)

    # FFT size large enough for the two convolutions we need:
    size = 1 << ((3 * n - 2).bit_length())        # ≥ 3 n‑2
    idx_g = (2 * n - 2) - np.arange(n)            # indices for gradient extraction

    # continuation parameters
    alpha = 5.0                                    # start temperature
    alpha_max = 5000.0
    # we plan for up to ~5·10⁵ inner steps; the increment will be tiny.
    alpha_inc = (alpha_max - alpha) / 500_000

    eta0 = 0.20                                    # initial learning‑rate
    eps = 1e-15                                    # avoid log(0)

    no_improve = 0
    patience = 8000

    t = 0
    while time.time() < deadline:
        # ---- 1) current convolution and soft‑max weights -----------------
        fft_a = np.fft.fft(a, size)

        conv_self_full = np.fft.ifft(fft_a * fft_a).real
        conv_self = conv_self_full[: 2 * n - 1]

        max_conv = conv_self.max()
        # exact objective for the *current* a (used for best‑track)
        cur_obj = 2.0 * n * max_conv

        # soft‑max over the convolution entries (smooth approx. of max)
        w_exp = np.exp(alpha * (conv_self - max_conv))   # shift = max avoids overflow
        w_norm = w_exp / w_exp.sum()                     # shape (2n‑1,)

        # ---- 2) gradient of the smooth objective -----------------------
        w_rev = w_norm[::-1]                              # reverse for correlation
        fft_w_rev = np.fft.fft(w_rev, size)

        conv_grad_full = np.fft.ifft(fft_a * fft_w_rev).real
        conv_grad = conv_grad_full[: 3 * n - 2]

        g = 2.0 * conv_grad[idx_g]                       # sub‑gradient of the smooth max

        # ---- 3) exponentiated‑gradient update -------------------------
        eta = eta0 * (1.0 - t / 500_000)                  # linear decay
        if eta < 1e-6:
            eta = 1e-6

        log_a = np.log(a + eps) - eta * g
        log_a -= np.max(log_a)                            # keep numbers moderate
        a = np.exp(log_a)
        a /= a.sum()

        # ---- 4) best‑track ------------------------------------------------
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best = a.copy()
            no_improve = 0
            print(f"[smooth] iter {t:6d} new best O={best_obj:.6f} α={alpha:.1f}")
        else:
            no_improve += 1
            if no_improve >= patience:
                # small random shake to escape flat regions
                jitter = np.random.randn(n) * 1e-4
                a = best + jitter
                a = np.clip(a, 0.0, None)
                a = _project_to_simplex(a)
                no_improve = 0

        # ---- 5) increase temperature ------------------------------------
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha + alpha_inc)

        t += 1

    # final check for the *last* a (might be better than the stored best)
    final_obj = _objective(a)
    if final_obj < best_obj:
        best_obj = final_obj
        best = a.copy()

    return best, best_obj


# ----------------------------------------------------------------------
def _subgradient_optimize(a0: np.ndarray, deadline: float,
                          max_iters: int = 30_000) -> tuple[np.ndarray, float]:
    """
    Classic exponentiated‑gradient on the true max‑convolution (sub‑gradient).
    Used as a short polishing phase after the smooth optimisation.
    """
    a = a0.copy()
    best = a.copy()
    best_obj = _objective(a)

    n = a.size
    size2 = 1 << ((2 * n - 2).bit_length())   # enough for a*a

    no_improve = 0
    patience = max_iters // 20

    for t in range(max_iters):
        if time.time() >= deadline:
            break

        # current convolution
        A = np.fft.fft(a, n=size2)
        conv_full = np.fft.ifft(A * A).real[: 2 * n - 1]

        max_val = conv_full.max()
        obj = 2.0 * n * max_val

        # indices where the maximum is attained (tiny tolerance)
        ks = np.where(np.abs(conv_full - max_val) < 1e-14)[0]

        # sub‑gradient of max(conv) w.r.t. a
        g = np.zeros_like(a)
        for k in ks:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_idx = np.arange(i_start, i_end + 1, dtype=int)
            j_idx = k - i_idx
            g[i_idx] += 2.0 * a[j_idx]
        if ks.size > 0:
            g /= ks.size

        # mirror‑descent update
        eta = 0.12 * (1.0 - t / max_iters)   # a small, linearly decaying step
        log_a = np.log(a + 1e-15) - eta * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()

        # best‑track
        if obj < best_obj - 1e-12:
            best_obj = obj
            best = a.copy()
            no_improve = 0
            print(f"[subgrad] iter {t:5d} new best O={best_obj:.6f}")
        else:
            no_improve += 1
            if no_improve >= patience:
                jitter = np.random.randn(n) * 1e-5
                a = best + jitter
                a = np.clip(a, 0.0, None)
                a = _project_to_simplex(a)
                no_improve = 0

    # final verification
    final_obj = _objective(best)
    if final_obj < best_obj:
        best_obj = final_obj
        best = best.copy()

    return best, best_obj
