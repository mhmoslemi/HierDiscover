"""
Propose a non‑negative step‑function that minimises the bound

    O = 2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm uses a smoothed version of the max‑operator (softmax) to obtain
a richer sub‑gradient, then performs an exponentiated‑gradient (mirror‑descent)
update on the simplex.  A linear schedule slowly sharpens the softmax (α
grows from 5 to 200) while the learning rate η decays linearly.  Several simple
initialisations (previous solution, 1/√ law, random) are tried; the best result
found before the time budget expires is returned.

The implementation is fully vectorised and relies on FFTs for fast convolution
and correlation, allowing thousands of iterations within the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a power‑of‑two FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 2).bit_length())
    A = np.fft.fft(a, n=size)
    conv = np.fft.ifft(A * A).real[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def _correlate_fft(p: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute   g_i = 2·∑_j a_j·p_{i+j}   (valid part, length n)
    using FFTs.  This is the gradient of the soft‑max approximation of
    max(conv(a,a)).
    """
    n = a.shape[0]
    size = 1 << ((3 * n - 2).bit_length())
    P = np.fft.fft(p, n=size)
    A_rev = np.fft.fft(a[::-1], n=size)
    full = np.fft.ifft(P * A_rev).real
    # slice the "valid" part: indices n-1 … 2·n-2 (inclusive)
    return full[n - 1 : n - 1 + n]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return the bound 2·n·max(conv(a,a)) for a already normalised (sum=1)."""
    n = a.size
    conv = _conv_fft(a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
def _exp_update(a: np.ndarray, g: np.ndarray, eta: float) -> np.ndarray:
    """
    Exponentiated‑gradient step A←A·exp(−η·g) with projection onto the simplex.
    """
    log_a = np.log(a + 1e-15) - eta * g
    log_a -= np.max(log_a)          # keep numbers moderate
    a_new = np.exp(log_a)
    a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
def _optimize(start: np.ndarray,
              deadline: float,
              eta0: float = 0.6,
              alpha_start: float = 5.0,
              alpha_end: float = 200.0,
              ramp_iters: int = 20000,
              patience: int = 4000) -> tuple[np.ndarray, float]:
    """
    Perform exponentiated‑gradient descent using a soft‑max smoothed max‑conv.
    Returns the best vector encountered and its objective value.
    """
    a = start.copy()
    best_a = a.copy()
    best_score = _objective(a)

    n = a.shape[0]
    no_improve = 0
    t = 0

    while time.time() < deadline:
        # ---- compute convolution and current score ---------------------------------
        conv = _conv_fft(a)
        cur_score = 2.0 * n * np.max(conv)

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            print(f"[opt] iter {t:6d}  new best O={best_score:.6f}")
        else:
            no_improve += 1
            if no_improve >= patience:
                # add a tiny random jitter to try to escape flat regions
                jitter = np.random.randn(n) * 1e-5
                a = best_a + jitter
                a = np.clip(a, 0.0, None)
                a = _project_to_simplex(a)
                no_improve = 0

        # ---- soft‑max weights -----------------------------------------------------
        max_c = np.max(conv)
        # shift to avoid overflow; max_c becomes exponent 0
        shifted = conv - max_c
        # linear increase of the temperature (α)
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, t / ramp_iters)
        w = np.exp(alpha * shifted)
        p = w / w.sum()                     # p is a probability vector on length 2n‑1

        # ---- gradient (correlation of p with a) ------------------------------------
        g = 2.0 * _correlate_fft(p, a)      # length n

        # ---- learning‑rate schedule (linear decay) --------------------------------
        eta = eta0 * max(0.0, 1.0 - t / (patience * 10 + 1))

        # ---- update ---------------------------------------------------------------
        a = _exp_update(a, g, eta)

        t += 1
        # safety exit if we have produced an unrealistically large number of iters
        if t > 2000000:
            break

    return best_a, best_score
