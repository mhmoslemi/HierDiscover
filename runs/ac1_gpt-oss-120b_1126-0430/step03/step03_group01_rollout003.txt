def compute_conv_and_grad(a, eps_factor):
    n = a.shape[0]
    L = _fft_len_for_corr(n)  # get length for correlation
    a_fft = np.fft.rfft(a, L)
    # convolution:
    conv = np.fft.irfft(a_fft * a_fft, L)[:2*n-1]
    max_conv = conv.max()
    # epsilon:
    eps = max(max_conv * eps_factor, 1e-12)
    # compute weights:
    shifted = conv - max_conv
    w = np.exp(shifted / eps)
    sum_w = w.sum()
    if sum_w == 0.0 or np.isnan(sum_w):
        w = np.full_like(w, 1.0/ w.size)
    else:
        w /= sum_w
    # correlation: compute w_fft and correlation with a.
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(w_fft * np.conj(a_fft), L)[:n]  # first n entries of linear correlation
    # gradient:
    grad_a = 4 * n * corr  # factor 4*n
    return max_conv, grad_a

def cheap_and_grad(a, eps_factor):
    n = a.shape[0]
    L = _fft_len_for_corr(n) # define function for length >= 3n-2
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[:2*n-1]
    max_conv = conv.max()
    eps = max(max_conv * eps_factor, 1e-12)
    # compute w:
    w_shifted = np.exp((conv - max_conv) / eps)
    w_sum = w_shifted.sum()
    if w_sum == 0.0:
        w = np.full_like(w_shifted, 1.0 / w_shifted.shape[0])
    else:
        w = w_shifted / w_sum
    # correlation:
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]  # correlation of a and w
    grad_a = 4 * n * corr
    return max_conv, grad_a

#!/usr/bin/env python
"""
Smoothed‑max gradient descent on the probability simplex.

The objective after normalising ‖a‖₁=1 is
    J(a) = 2·n·max_k (a*a)[k] .
The max is replaced by a smooth log‑sum‑exp:
    max ≈ ε·log∑_k exp(conv_k/ε)  with ε = α·max(conv).

For a fixed ε the gradient of J is
    ∇_a J = 4·n·corr(a, w),
where w_k ∝ exp(conv_k/ε) are the soft‑max weights and
corr(a,w) is the linear cross‑correlation of a and w.
The optimisation is performed on the log‑it (soft‑max) parametrisation:
    a = softmax(z) .
Adam updates are applied to the logits z, a back‑tracking line‑search
guarantees a non‑increasing cheap score 2·n·max(conv) and the best
candidate seen so far is finally evaluated with ``evaluate_sequence``.

The routine respects a ~1000 s time budget, supports progressive
support enlargement (up to 20 000 entries) and occasional random
perturbations to escape plateaux.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (must be top‑level – no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max; returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it representation that yields `a` after soft‑max."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """
    Smallest power‑of‑two ≥ 3·n‑2,
    which is enough for a linear convolution of length (2·n‑1) with a
    vector of length n (i.e. correlation a⋆w of size 3·n‑2).
    """
    return 1 << ((3 * n - 2).bit_length())


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    total = new_a.sum()
    if total == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= total
    return new_a


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """Normalised power‑law a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _conv_max_and_grad(a: np.ndarray, eps_factor: float) -> tuple[float, np.ndarray]:
    """
    Compute max entry of a*a and the gradient of the smoothed objective.

    Parameters
    ----------
    a : np.ndarray
        Probability vector (sum = 1, non‑negative).
    eps_factor : float
        Multiplicative factor for the smoothing constant:
        eps = eps_factor * max(conv).

    Returns
    -------
    max_conv : float
        Maximum of the self‑convolution a*a.
    grad_a : np.ndarray
        Gradient of 2·n·smooth_max w.r.t. a (already multiplied by 4·n).
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    # FFT of a (zero‑padded to length L)
    a_fft = np.fft.rfft(a, L)

    # --- self‑convolution a*a (linear, length 2n‑1) ---
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = conv.max()

    # --- smoothing weight vector w (soft‑max over conv) ---
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv          # for numerical stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or np.isnan(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # --- cross‑correlation of a and w (linear, length 3n‑2) ---
    w_fft = np.fft.rfft(w, L)
    # correlation = ifft( conj(a_fft) * w_fft )
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # gradient of the smoothed objective (4·n * correlation)
    grad_a = 4.0 * n * corr
    return float(max_conv), grad_a


def _conv_max(a: np.ndarray) -> float:
    """Return max entry of a*a (no gradient)."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())

# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) beats the target bound.
    The routine respects the given time budget (~1000 s) and uses a smoothed‑max
    Adam optimiser on the probability simplex.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0       # safety margin

    # ------------------------------------------------------------------
    # 0. Choose an initial support size and initial distribution.
    # ------------------------------------------------------------------
    try:
        init_global = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        n = max(5000, init_global.shape[0])
        a0 = np.maximum(init_global, 0.0)
        a0_sum = a0.sum()
        if a0_sum == 0.0:
            a0 = np.full_like(a0, 1.0 / a0.shape[0])
        else:
            a0 /= a0_sum
        print(f"Initial sequence taken from global variable, length = {n}")
    except Exception:

        n = 8000
        # Scan a few exponents quickly and keep the best cheap score.
        p_grid = np.linspace(0.55, 0.70, 31)
        best_a = None
        best_score = float("inf")
        for p in p_grid:

            max_c = _conv_max(cand)
            score = 2.0 * n * max_c
            if score < best_score:

                best_a = cand
        if best_a is None:
            best_a = _power_law_distribution(n, 0.60)
        a0 = best_a
        print(f"Power‑law start, length = {n}")

    # Small random jitter to avoid exact replicas of a previous local optimum
    z = _logits_from_distribution(a0)
    z += 0.01 * np.random.randn(n)

    # ------------------------------------------------------------------
    # 1. Adam state on logits
    # ------------------------------------------------------------------
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_alpha = 0.15                     # base learning‑rate for logits

    # ------------------------------------------------------------------
    # 2. Optimisation loop
    # ------------------------------------------------------------------
    best_seq = np.exp(z)                 # will be normalised via softmax each iter
    best_seq = _softmax(z)
    best_score = 2.0 * n * _conv_max(best_seq)

    it = 0
    last_improve_iter = 0
    eps_factor = 0.08                     # initial smoothing factor (ε = eps_factor·max)
    eps_factor_min = 5e-4
    eps_factor_decay = 0.9995
    report_every = 2000
    max_n = 20000
    n_growth_factor = 1.5

    while time.time() < deadline and it < 1_500_000:
        it += 1

        # ---- current point on the simplex ----
        a = _softmax(z)

        # ---- cheap evaluation + gradient (smooth max) ----
        max_conv, grad_a = _conv_max_and_grad(a, eps_factor)

        cheap_score = 2.0 * n * max_conv

        # ---- keep best ever seen ----
        if cheap_score < best_score:
            best_score = cheap_score
            best_seq = a.copy()
            last_improve_iter = it
            if it % report_every == 0:
                print(f"[Iter {it:07d}] NEW BEST cheap = {cheap_score:.6f}")

        # ---- gradient w.r.t. logits (mirror descent on simplex) ----
        mu = np.dot(grad_a, a)
        grad_z = a * (grad_a - mu)

        # ---- Adam moment updates (bias‑corrected) ----
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # ---- step size schedule (inverse‑sqrt decay) ----
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step_dir = - m_hat / (np.sqrt(v_hat) + eps_adam)
        trial_alpha = alpha_t

        # ---- back‑tracking line search (up to 4 halvings) ----
        accepted = False
        for _ in range(4):
            z_new = z + trial_alpha * step_dir
            z_new = np.clip(z_new, -50.0, 50.0)          # keep exp(z) safe
            a_new = _softmax(z_new)
            new_max = _conv_max(a_new)
            new_score = 2.0 * n * new_max
            if new_score <= cheap_score:
                z = z_new
                accepted = True
                break
            trial_alpha *= 0.5

        if not accepted:
            # no improvement – take a modest step anyway
            z += alpha_t * step_dir
            z = np.clip(z, -50.0, 50.0)

        # ---- occasional random perturbations to escape plateaus ----
        if it % 5000 == 0 and np.random.rand() < 0.15:
            z += 0.015 * np.random.randn(n)
            m *= 0.7
            v *= 0.7

        # ---- gradual reduction of the smoothing factor ----
        eps_factor = max(eps_factor_min, eps_factor * eps_factor_decay)

        # ---- progressive support enlargement (if time permits) ----
        if (it % 15000 == 0) and (n < max_n) and (time.time() + 30 < deadline):
            new_n = min(max_n, int(n * n_growth_factor) + 1)
            print(f"[Iter {it:07d}] Expanding support: {n} → {new_n}")
            # Interpolate the best sequence found so far to the larger size
            a_interp = _interpolate_distribution(best_seq, new_n)
            # Reset optimiser for the new problem size
            z = _logits_from_distribution(a_interp)
            m = np.zeros_like(z)
            v = np.zeros_like(z)
            n = new_n
            # Update best record for the new length
            best_seq = a_interp.copy()
            best_score = 2.0 * n * _conv_max(best_seq)
            # reset counters so the schedule starts afresh
            it = 0
            last_improve_iter = 0
            eps_factor = 0.08
            continue

        # ---- early stop if stagnation after reaching max size ----
        if n >= max_n and it - last_improve_iter > 300_000:
            print(f"[Iter {it:07d}] Stagnation detected – terminating early.")
            break

    # ------------------------------------------------------------------
    # 3. Final exact evaluation with the provided evaluate_sequence()
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:   # defensive – should never happen
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # ------------------------------------------------------------------
    # 4. Return the best sequence found.
    # ------------------------------------------------------------------
    return final_seq
