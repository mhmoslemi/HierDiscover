#!/usr/bin/env python
"""
A smooth‑gradient optimiser for the step‑function bound.

The evaluation metric is   2·n·max(a*a) / (sum(a)^2) .
Because scaling does not change the value (as long as no clipping occurs),
we restrict ourselves to probability vectors (∑a = 1).  The exact metric
then equals the cheap proxy   2·n·max(a*a).

The algorithm works on the simplex via a soft‑max parametrisation
(z‑logits → a).  At each iteration we minimise a smooth approximation of
the ℓ∞‑norm of the self‑convolution:

    f_p(a) = ( Σ_k (a*a)[k]^p )^{1/p} ,      p ≥ 2 .

For p=2 the objective is the ℓ2‑norm of the convolution; increasing p
focuses the gradient on the largest entries and approaches the true
ℓ∞‑norm.  The gradient is obtained analytically and efficiently by FFTs
(no O(n²) loops).  An Adam optimiser (mirrored‑descent on the simplex)
drives the logits; a simple back‑tracking line‑search guarantees
monotone decrease of the cheap proxy.

The support size starts modest (≈5000) and is enlarged progressively
(up to 20000) by linear interpolation of the best distribution found so
far.  Random perturbations and occasional restarts help escape plateaux.
The best sequence seen during the run is finally evaluated with the
provided ``evaluate_sequence`` and returned.

The routine respects a 1000 s time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max; returns a probability vector (sum=1)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it that yields ``a`` after soft‑max."""
    return np.log(a + eps)


def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two ≥ n (efficient length for real FFT)."""
    return 1 << ((n - 1).bit_length())


def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """
    Full self‑convolution a*a (length 2·n‑1) using an FFT of suitable size.
    Returns a 1‑D array of length 2*n-1.
    """
    n = a.shape[0]
    L = _fft_len_for_n(2 * n - 1)
    fa = np.fft.rfft(a, L)
    conv = np.fft.irfft(fa * fa, L)[: 2 * n - 1]
    return conv


def _conv_max_fft(a: np.ndarray) -> tuple[float, int]:
    """
    Returns (max_value, argmax) of the self‑convolution a*a.
    Uses a FFT of length power‑of‑two ≥ 2·n‑1.
    """
    conv = _conv_full_fft(a)
    idx = int(conv.argmax())
    return float(conv.max()), idx


def _smooth_gradient(a: np.ndarray, conv: np.ndarray, p: int) -> np.ndarray:
    """
    Gradient of   f_p(a) = ( Σ_k conv[k]^p )^{1/p}
    w.r.t. the probability vector ``a`` (∑a=1).

    Parameters
    ----------
    a : np.ndarray, shape (n,)
        Current probability vector.
    conv : np.ndarray, shape (2*n-1,)
        Full self‑convolution of ``a``.
    p : int, >=2
        Smoothing exponent.

    Returns
    -------
    g : np.ndarray, shape (n,)
        Gradient ∂f_p/∂a .
    """
    # weight w_k = conv[k]^{p-1}
    # when conv[k] == 0 the contribution is zero anyway.
    w = np.power(conv, p - 1, out=np.empty_like(conv))

    # reversed weight vector for the correlation
    w_rev = w[::-1]

    n = a.shape[0]
    L = _fft_len_for_n(3 * n - 2)          # length for a * w_rev convolution
    fa = np.fft.rfft(a, L)
    fw = np.fft.rfft(w_rev, L)
    corr = np.fft.irfft(fa * fw, L)[: 3 * n - 2]

    # slice corresponding to indices i+j (0 ≤ i+j ≤ 2n‑2)
    #   g_i = 2 * Σ_j a_j * w_{i+j}
    # slice = corr[n-1 : 2*n-1] (length n) but reversed to match i order
    slice_mid = corr[n - 1 : 2 * n - 1][::-1]

    # Apply the prefactor from the chain‑rule of f_p
    #   ∂f/∂a = (1 / (Σ conv^p)^{(p-1)/p}) * Σ_k p·conv^{p-1}·∂conv/∂a
    # The constant  p  is omitted (absorbed into step‑size) because
    # Adam is scale‑invariant.
    g = 2.0 * slice_mid
    return g


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    tot = new_a.sum()
    if tot == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= tot
    return new_a


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """Normalised power‑law a_i ∝ (i+1)^{-p},  p>0."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a
