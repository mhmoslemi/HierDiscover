"""
Propose a non‑negative step‑function sequence that (approximately) minimizes

    O(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The implementation follows two phases

* Phase 1 – Adam on a smooth log‑sum‑exp surrogate.
  * a is parametrised by a log‑vector y, a = softmax(y), which guarantees
    feasibility without an explicit projection.
  * The surrogate is      (2·n/α)·log∑_k exp(α·conv_k) .
  * Its gradient reduces to 2·corr(w,a) where w_k ∝ exp(α·conv_k).
    Both the convolution and the correlation are evaluated by FFTs.
  * α is increased exponentially from 10 to 5000.
  * Adam with bias‑corrected moments and a 1/√t stepsize is used.
  * The best vector seen (in terms of the exact objective) is stored.

* Phase 2 – stochastic local refinement.
  * Starts from the best candidate of phase 1; several restarts with
    Gaussian jitter are performed.
  * A cheap “mass‑transfer” Metropolis move (move a random fraction of the
    probability mass from one index to another) explores the neighbourhood.
  * Every few iterations a true sub‑gradient step for the exact
    max‑convolution entry is taken (the same sub‑gradient as in the reference
    code).
  * Temperature is slowly annealed; if no improvement is observed for a
    while a restart is triggered.

The routine respects the 1000 s time budget, prints occasional progress
messages and finally returns the best sequence found as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: softmax parametrisation (y -> a)
def _softmax(y: np.ndarray) -> np.ndarray:
    """Return softmax(y) – a probability vector that sums to 1."""
    y_max = np.max(y)
    e = np.exp(y - y_max)          # shift for numerical stability
    s = e.sum()
    return e / s

# ----------------------------------------------------------------------
# Full linear convolution via FFT (length 2·n‑1)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return a*a (full convolution) using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]

# ----------------------------------------------------------------------
# Gradient of the log‑sum‑exp surrogate
def _smooth_gradient(a: np.ndarray, conv: np.ndarray, alpha: float, L: int) -> np.ndarray:
    """
    Compute ∇_a ( (2·n/α)·log∑_k e^{α·conv_k} ).
    The closed form is 2·corr(w,a) with w_k ∝ e^{α·conv_k}.
    """
    # soft‑max weights over the convolution entries
    scaled = alpha * conv
    m = np.max(scaled)
    exp_scaled = np.exp(scaled - m)            # prevent overflow
    w = exp_scaled / exp_scaled.sum()

    # correlation w ⋆ a   (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)   # length ≥ 3·n‑2
    n = a.size
    grad = 2.0 * corr_rev[n - 1 : n - 1 + n]          # keep only the central n entries
    return grad

# ----------------------------------------------------------------------
# Exact objective for a simplex‑normalised vector (sum(a)=1)
def _objective(a: np.ndarray, n: int, L: int) -> float:
    """Return 2·n·max(conv(a,a)) with exact convolution."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()

# ----------------------------------------------------------------------
# Sub‑gradient of the maximal convolution entry (as used in the reference)
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) wrt a (averaged over all maximising shifts)."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)          # indices where the max is attained
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g

# ----------------------------------------------------------------------

    Search for a step‑function (non‑negative sequence) with a low evaluation
    score.  The function runs for at most ``budget_s`` seconds and returns the
    best list of floats it discovered.
    """

    m = np.zeros_like(y)                # Adam first moment
    v = np.zeros_like(y)                # Adam second moment
