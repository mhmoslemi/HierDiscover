def _weighted_subgradient(a, conv, topk, theta):
    n = a.shape[0]
    indices = np.argpartition(conv, -topk)[-topk:]
    # sort descending for stable weight exponent
    idx_sorted = indices[np.argsort(-conv[indices])]
    # compute weights via softmax with temperature 1/theta? Actually we want w_i = exp(theta * (conv_i - max_conv)).
    maxc = conv[idx_sorted[0]]
    exps = np.exp(theta * (conv[idx_sorted] - maxc))
    w = exps / exps.sum()
    g = np.zeros_like(a)
    for idx, weight in zip(idx_sorted, w):
        lo = max(0, idx - (n - 1))
        hi = min(idx, n - 1)
        # i indices as array
        i = np.arange(lo, hi+1)
        g[i] += weight * 2.0 * a[idx - i]
    return g

        # adjust theta based on elapsed time
        elapsed = time.time() - start
        theta = min(theta_max, theta0 + theta_inc * elapsed)

        # top K indices
        K = min(top_k, conv.size)
        idx_top = np.argpartition(conv, -K)[-K:]
        # sort descending
        order = np.argsort(-conv[idx_top])
        idx_top = idx_top[order]
        vals_top = conv[idx_top]
        # compute soft weights
        maxc = vals_top[0]
        exps = np.exp(theta * (vals_top - maxc))
        w = exps / exps.sum()

        # accumulate weighted subgradient
        g = np.zeros_like(a)
        lo_global = 0
        hi_global = n - 1
        for idx_k, wk in zip(idx_top, w):
            lo = max(lo_global, idx_k - (n - 1))
            hi = min(hi_global, idx_k)
            i = np.arange(lo, hi + 1, dtype=np.int64)
            g[i] += wk * 2.0 * a[idx_k - i]

        # gradient in logit space
        mu = np.dot(g, a)
        grad_z = a * (g - mu)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step_dir = - m_hat / (np.sqrt(v_hat) + eps_adam)

        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        # apply step
        z = z + alpha_t * step_dir
        # clip logits to avoid overflow
        np.clip(z, -80.0, 80.0, out=z)

        # occasional random perturbation
        if it % 5000 == 0 and np.random.rand() < 0.3:
            perturb = 0.02 * np.random.randn(n)
            z += perturb
            m *= 0.5
            v *= 0.5

        # occasional support expansion
        if it % 20000 == 0 and n < n_max and time.time() + 10 < deadline:
            new_n = min(n_max, n + 5000)
            print(f"[Iter {it}] expanding support to {new_n}")
            a_curr = best_seq  # we can interpolate best_seq into new size
            a = _interpolate_distribution(a_curr, new_n)
            z = _logits_from_distribution(a, eps)
            m = np.zeros_like(z)
            v = np.zeros_like(z)
            n = new_n
            # recompute best_score for new n
            fft_len = _fft_len_for_n(n)
            fa = np.fft.rfft(a, fft_len)
            conv = np.fft.irfft(fa * fa, fft_len)[:2*n-1]
            best_score = 2.0 * n * conv.max()
            best_seq = a.copy()
            # reset iteration counter
            it = 0
            last_improve_iter = 0
            continue

        # early stop if no improvement for long
        if it - last_improve_iter > 300000:
            print(f"Stagnation after {it} iters, stopping.")
            break

    # final evaluation
    final_seq = best_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score {final_score:.6f}")
    except Exception as e:
        print(f"Error evaluating: {e}")
        final_score = float('inf')
    return final_seq

#!/usr/bin/env python
"""
Optimisation of a non‑negative step‑function via a smoothed max‑convolution
gradient (Adam on the probability simplex).

* The vector is represented by logits `z`; `a = softmax(z)` guarantees
  a_i ≥ 0 and Σ a_i = 1.
* At each iteration we compute the full self‑convolution `c = a*a` with a
  real FFT (O(n log n)).
* Instead of using only the single active offset (the worst case) we
  build a *smoothed* sub‑gradient:
      – pick the K largest convolution entries,
      – weight them with a soft‑max (`θ` controls the smoothing),
      – sum the corresponding sub‑gradients.
  This approximates the gradient of `log‑sum‑exp` of the convolution,
  helping the optimiser escape plateaus.
* Adam with an inverse‑sqrt learning‑rate schedule performs mirror descent
  on the logits.
* Every few thousand iterations we inject a small random perturbation
  and, if enough time is left, we enlarge the support (interpolating the
  current best distribution) up to a maximum of 30 000 entries.
* The best probability vector (sum = 1) is returned; the external
  ``evaluate_sequence`` routine rescales it automatically.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max; returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it representation that yields `a` after soft‑max."""
    return np.log(a + eps)


def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two ≥ 2·n‑1 (efficient size for real FFT)."""
    return 1 << ((2 * n - 1).bit_length())


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full linear self‑convolution a*a using an FFT of length `fft_len`.
    Returns an array of length 2·len(a)‑1.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.size - 1]
    return conv


def _weighted_subgradient(a: np.ndarray,
                          conv: np.ndarray,
                          k: int,
                          theta: float) -> np.ndarray:
    """
    Returns a weighted sub‑gradient of `max(conv)` at the current vector `a`.

    Parameters
    ----------
    a : np.ndarray
        Current probability vector (sum = 1).
    conv : np.ndarray
        Full self‑convolution a*a (length 2·n‑1).
    k : int
        Number of largest convolution entries to use.
    theta : float
        Smoothing temperature – larger values concentrate the weight on the
        absolute maximum.

    Returns
    -------
    g : np.ndarray
        Weighted sub‑gradient w.r.t. `a`.
    """
    n = a.shape[0]
    # pick the k largest indices (order does not matter for the weighting)
    idx = np.argpartition(conv, -k)[-k:]
    # sort them so that the largest entry gets the largest weight (more stable)
    order = np.argsort(-conv[idx])
    idx = idx[order]

    # soft‑max style weights
    vals = conv[idx]
    max_val = vals[0]                     # the largest one
    # avoid overflow – subtract max before exponentiation
    exps = np.exp(theta * (vals - max_val))
    weights = exps / exps.sum()

    # accumulate the weighted sub‑gradients
    g = np.zeros_like(a)
    for off, w in zip(idx, weights):
        # contribution of offset `off`: ∂/∂a_i (a*a)[off] = 2·a_{off‑i}
        lo = max(0, off - (n - 1))
        hi = min(off, n - 1)
        i = np.arange(lo, hi + 1, dtype=np.int64)
        g[i] += w * 2.0 * a[off - i]
    return g


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)        # guard against tiny negatives
    total = new_a.sum()
    if total == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= total
    return new_a


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """Normalised power‑law a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) beats the target
    bound.  The routine respects the given time budget (≈1000 s) and uses a
    smoothed sub‑gradient Adam optimiser on the simplex.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0   # safety margin for final evaluation

    # ---------------------------------------------------------------
    # 0. Choose a working support size.
    # ---------------------------------------------------------------
    # Try to reuse the global variable if it exists; otherwise start from scratch.
    init_global = globals().get("height_sequence_1", None)
    if init_global is not None:
        init_global = np.array(init_global, dtype=np.float64)
        n = max(2000, init_global.shape[0])
        use_global = True
        print(f"[Init] Using supplied global sequence, length = {n}")
    else:
        init_global = None
        n = 20000                       # a decent default size
        use_global = False
        print(f"[Init] No global sequence – starting with length = {n}")

    n_max = 30000                       # hard cap on the support size

    # ---------------------------------------------------------------
    # 1. Initialise a probability vector.
    # ---------------------------------------------------------------
    if use_global and init_global is not None:
        a = np.maximum(init_global, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / a.shape[0])
        else:
            a /= s
    else:
        # A modest power‑law works well as a generic start.
        a = _power_law_distribution(n, 0.60)
        print("[Init] Power‑law start with p=0.60")

    # ---------------------------------------------------------------
    # 2. Log‑it parametrisation + Adam state.
    # ---------------------------------------------------------------
    eps = 1e-16
    z = _logits_from_distribution(a, eps)

    m = np.zeros_like(z)                # first‑moment estimate
    v = np.zeros_like(z)                # second‑moment estimate
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    base_alpha = 0.25                   # base learning‑rate

    # ---------------------------------------------------------------
    # 3. Main optimisation loop.
    # ---------------------------------------------------------------
    best_seq = a.copy()
    best_score = float("inf")
    it = 0
    last_improve = 0
    report_every = 2000

    # hyper‑parameters for the smoothed gradient
    K = 30                              # number of top convolution entries used
    theta_start = 0.5
    theta_end = 12.0
    # linear schedule for theta (per second)
    theta_slope = (theta_end - theta_start) / max(1.0, budget_s - 10.0)

    while time.time() < deadline and it < 1_500_000:
        it += 1

        # probability vector from the logits
        a = _softmax(z)

        # full convolution (FFT based)
        fft_len = _fft_len_for_n(n)
        conv = _conv_full_fft(a, fft_len)

        maxb = float(conv.max())
        cheap_score = 2.0 * n * maxb

        # record best ever seen (cheap proxy == exact score because Σa = 1)
        if cheap_score < best_score:
            best_score = cheap_score
            best_seq = a.copy()
            last_improve = it
            if it % report_every == 0:
                print(f"[Iter {it:07d}] NEW BEST cheap = {cheap_score:.6f}")

        # -----------------------------------------------------------
        # Smoothed sub‑gradient of the max‑convolution
        # -----------------------------------------------------------
        elapsed = time.time() - start
        theta = min(theta_end, theta_start + theta_slope * elapsed)

        g_a = _weighted_subgradient(a, conv, K, theta)

        # mirror‑descent conversion to log‑it space (KL geometry)
        mu = float(np.dot(g_a, a))
        grad_z = a * (g_a - mu)

        # Adam moment updates (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # direction (negative Adam step)
        step_dir = - m_hat / (np.sqrt(v_hat) + eps_adam)

        # basic step‑size schedule: inverse‑sqrt decay
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)

        # perform the update in log‑it space
        z = z + alpha_t * step_dir
        # keep logits in a safe numeric range
        np.clip(z, -80.0, 80.0, out=z)

        # -----------------------------------------------------------
        # Occasional random perturbations (helps escape plateaus)
        # -----------------------------------------------------------
        if it % 5000 == 0 and np.random.rand() < 0.25:
            perturb = 0.02 * np.random.randn(n)
            z += perturb
            m *= 0.5
            v *= 0.5

        # -----------------------------------------------------------
        # Progressive enlargement of the support (if time permits)
        # -----------------------------------------------------------
        if (it % 25000 == 0) and (n < n_max) and (time.time() + 25.0 < deadline):
            new_n = min(n_max, n + 5000)
            print(f"[Iter {it:07d}] Expanding support: {n} → {new_n}")
            # Interpolate the best distribution obtained so far
            a = _interpolate_distribution(best_seq, new_n)
            # Re‑initialise logits and Adam state for the new size
            z = _logits_from_distribution(a, eps)
            m = np.zeros_like(z)
            v = np.zeros_like(z)
            n = new_n
            # Reset counters for the new problem size
            it = 0
            last_improve = 0
            # Re‑evaluate the cheap score for the enlarged support
            fft_len = _fft_len_for_n(n)
            conv = _conv_full_fft(a, fft_len)
            best_score = 2.0 * n * conv.max()
            best_seq = a.copy()
            continue

        # -----------------------------------------------------------
        # Early stop if stagnation is observed after reaching max size
        # -----------------------------------------------------------
        if n >= n_max and it - last_improve > 300_000:
            print(f"[Iter {it:07d}] No improvement for {it-last_improve} iters – early stop.")
            break

    # ---------------------------------------------------------------
    # 4. Final exact evaluation (the platform will also evaluate)
    # ---------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[Result] Final exact score = {final_score:.6f}")
    except Exception as exc:
        final_score = float("inf")
        print(f"[Result] Evaluation error: {exc}")

    # ---------------------------------------------------------------
    # 5. Return the best sequence found.
    # ---------------------------------------------------------------
    return final_seq
