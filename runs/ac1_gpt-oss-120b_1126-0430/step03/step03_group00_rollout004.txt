"""
Propose a non‑negative step‑function (a probability vector) that minimizes

    O(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The search uses three ideas:

1. **Mirror descent on a smooth log‑sum‑exp surrogate** (exponentiated‑gradient
   updates).  The surrogate becomes tighter as the temperature α grows, so the
   iterates move toward the true max‑convolution objective.

2. **Deterministic importance‑driven mass transfer**.  After the surrogate
   phase we repeatedly move a small fraction of probability mass from the
   index that contributes most to the current maximal convolution entry to the
   index that contributes least (the sub‑gradient direction).  The move is
   accepted if it lowers the exact objective; otherwise it is accepted with a
   small Metropolis probability to escape plateaus.

3. **Occasional sub‑gradient refinements** (projected steps on the exact
   max‑convolution sub‑gradient).  These steps shrink the active set and
   polish the solution.

The algorithm runs for the whole time budget (default 1000 s) and returns the
best sequence found.  All operations respect the simplex constraint, i.e.
the sequence is non‑negative and has unit ℓ₁‑norm (scale‑invariant for the
objective).  The implementation is fully vectorised and relies on FFTs for
fast convolution and correlation.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        # Return the centre of the simplex (uniform distribution)
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full self‑convolution (length 2·n‑1)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return the full convolution a*a using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Exact objective for a simplex‑normalised vector (sum(a)=1)
def _objective_simplex(a: np.ndarray, n: int, L: int) -> float:
    """2·n·max(conv(a,a)) when sum(a)=1."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Gradient of the log‑sum‑exp (soft‑max) surrogate of the max
def _smooth_gradient(a: np.ndarray, conv: np.ndarray, alpha: float, L: int) -> np.ndarray:
    """
    Compute ∇_a ( (2·n/α)·log∑_k e^{α·conv_k} ).
    The result is 2·corr(w, a) where w_k ∝ e^{α·conv_k}.
    """
    # soft‑max weights over convolution entries
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)          # numeric stability
    w = exp_scaled / exp_scaled.sum()

    # correlation w ⋆ a   (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)   # length ≥ 3·n‑2
    n_len = a.size
    grad = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]
    return grad


# ----------------------------------------------------------------------
# Sub‑gradient of the maximal convolution entry (exact max)
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Return the sub‑gradient of max(conv) w.r.t. a.
    For each k attaining the maximum, the contribution of index i is
    2·∑_{j : i+j=k} a_j.
    """
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)  # all maximising positions
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size > 0:
        g /= ks.size
    return g

    Returns the sequence as a plain Python list of floats.

        else:
            a = a / a_sum

        # occasional pruning of the ultra‑small tail (keeps the support compact)
        if it % 5000 == 0 and it > 0:
            thresh = np.percentile(a, 0.5)    # drop the lowest 0.5 %
            a[a < thresh] = 0.0
            a = _project_to_simplex(a)

        it += 1

    # ------------------------------------------------------------------
    # Phase 2 – Deterministic importance‑driven refinement + occasional sub‑grad steps
    print(f"[phase‑2] entering local refinement, best O={best_score:.6f}")
    # Parameters for the refinement loop
    subgrad_interval = 11
    temperature0 = 0.025                     # Metropolis temperature for uphill moves
    max_local_iters = 200_000                # safety bound
    local_iter = 0

    while time.time() < deadline and local_iter < max_local_iters:
        # --------------------------------------------------------------
        # Initialise a restart (jittered best or fresh random)
        if np.random.rand() < 0.6:
            a = best_seq.copy()
            a += np.random.randn(n) * 0.0015
            a = np.clip(a, 0.0, None)
            a = _project_to_simplex(a)
        else:
            a = np.random.rand(n)
            a = _project_to_simplex(a)

        # Local optimisation on this restart
        T = temperature0
        no_improve = 0
        it_loc = 0
        while time.time() < deadline:
            # current convolution and objective
            conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max()

            # ----------------------------------------------------------
            # 1) Deterministic mass‑transfer move guided by the sub‑gradient
            g = _subgradient_max_conv(a, conv)
            i_max = np.argmax(g)
            i_min = np.argmin(g)
            if i_max != i_min and a[i_max] > 0:
                delta = a[i_max] * 0.07                # move 7 % of the source mass
                a_new = a.copy()
                a_new[i_max] -= delta
                a_new[i_min] += delta
                # evaluate the move
                conv_new = _conv_fft(a_new, L)
                new_obj = 2.0 * n * conv_new.max()
                # Metropolis acceptance (allows occasional uphill jumps)
                if new_obj < cur_obj or np.random.rand() < math.exp((cur_obj - new_obj) / max(T, 1e-12)):
                    a = a_new
                    cur_obj = new_obj
                    no_improve = 0
                    if cur_obj < best_score:
                        best_score = cur_obj
                        best_seq = a.copy()
                        print(f"[{time.time():.1f}] new global best O={best_score:.6f}")
                else:
                    no_improve += 1
            else:
                no_improve += 1

            # ----------------------------------------------------------
            # 2) Periodic exact sub‑gradient step (projected)
            if it_loc % subgrad_interval == 0:
                step = 0.04 * (1.0 - it_loc / max_local_iters)   # decaying step size
                a = a - step * g
                a = _project_to_simplex(a)

                # recompute objective after the step
                conv = _conv_fft(a, L)
                cur_obj = 2.0 * n * conv.max()
                if cur_obj < best_score:
                    best_score = cur_obj
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] sub‑grad improvement O={best_score:.6f}")

            # ----------------------------------------------------------
            # Temperature decay & iteration bookkeeping
            T *= 0.99997
            it_loc += 1
            local_iter += 1

            # Simple restart criterion: if we have been stuck too long, break
            if no_improve >= 6000:
                break

        # --------------------------------------------------------------
        # Exit if overall time budget is exhausted
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Final sanity check using the provided evaluator
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_val:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence failed:", exc)

    # Return the best sequence found (as a plain list of floats)
    return best_seq.tolist()
