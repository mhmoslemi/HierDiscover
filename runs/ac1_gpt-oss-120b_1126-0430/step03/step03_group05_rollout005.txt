def _smooth_max_and_gradient(a, beta, fft_len):
    # a: np.ndarray length n
    # returns (smooth_max, gradient array length n)
    # compute convolution a*a via FFT (full) length 2n-1

#!/usr/bin/env python
"""
A fast annealed projected gradient descent for the
step‑function optimisation problem.

The objective
    2·n·max_k (a * a)[k] / (∑ a_i)²
is scale‑invariant, therefore the algorithm works on
vectors normalised to the probability simplex (∑ a_i = 1,
a_i ≥ 0).  The non‑smooth max is approximated by a
soft‑max with a temperature β that is increased during the
run (simulated‑annealing).  In each iteration we

* compute the full convolution a*a with an FFT,
* build the soft‑max weights w_k ∝ exp(β·(conv_k‑max)),
* obtain the gradient of the smooth max:
        g_i = 2·∑_j a_j·w_{i+j}
  (implemented as an FFT‑based convolution of w with the
  reversed a),
* take a projected gradient step onto the simplex,
* periodically increase β and restart from the current best
  after long periods without improvement.

The algorithm is deliberately simple, uses only NumPy
(and the already‑imported ``evaluate_sequence``) and respects
the 1000 s time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (sum = 1, v ≥ 0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # All entries are non‑positive – fall back to the uniform point.
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float) -> np.ndarray:
    """Return a normalised power‑law vector a_i ∝ (i+1)^‑p (∑a_i = 1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = i ** (-p)
    seq /= seq.sum()
    return seq


def _conv_max_fft(a: np.ndarray, fft_len: int) -> tuple[float, int]:
    """Return (max value, index) of the full convolution a*a using FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa).real[: 2 * a.size - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx


def _smooth_max_and_gradient(
    a: np.ndarray,
    beta: float,
    fft_len: int,
) -> tuple[float, np.ndarray]:
    """
    Compute the actual maximum of the convolution (a*a) and the
    gradient of the β‑smooth max.

    Returns
        max_conv : float                 – max_k (a*a)[k]
        grad    : np.ndarray (len = n)   – ∇  ( (1/β)·log Σ e^{β·(a*a)[k]} )
    """
    n = a.shape[0]

    # ----- 1. full convolution a*a (FFT) -----
    fa = np.fft.fft(a, fft_len)
    conv_full = np.fft.ifft(fa * fa).real[: 2 * n - 1]

    max_conv = float(conv_full.max())

    # ----- 2. soft‑max weights w_k -----
    #   w_k = exp(β·(conv_k – max_conv)) / Σ exp(...)
    #   subtraction of max_conv guarantees numerical stability.
    exp_vals = np.exp(beta * (conv_full - max_conv))
    w = exp_vals / exp_vals.sum()               # length = 2n‑1

    # ----- 3. gradient g_i = 2·∑_j a_j·w_{i+j}  (valid i = 0..n‑1) -----
    #    This is a convolution of w with the reversed a.
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    farev = np.fft.fft(a_rev, fft_len)
    conv_w_rev = np.fft.ifft(fw * farev).real

    # The required slice is conv_w_rev[n‑1 : n‑1 + n]
    segment = conv_w_rev[n - 1 : n - 1 + n]
    grad = 2.0 * segment

    return max_conv, grad


# ----------------------------------------------------------------------
# Main entry point required by the evaluation platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the given time
    budget (a safety margin of 2 s is kept).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # --------------------------------------------------------------
    # 0. Choose the working length.
    #    Use the previous construction if it exists, otherwise fall back
    #    to a moderately large size (8000).  Lengths > 5000 give a better
    #    asymptotic bound while still allowing cheap FFTs.
    # --------------------------------------------------------------
    try:
        prev = height_sequence_1               # type: ignore[name-defined]
        n_prev = len(prev)
    except Exception:
        prev = None
        n_prev = 0

    n = max(8000, n_prev)                      # target length
    print(f"Working length n = {n}")

    # ----- FFT length – next power of two ≥ 3·n (needed for gradient) -----
    fft_len = 1 << ((3 * n).bit_length())

    # --------------------------------------------------------------
    # 1. Initialise with a power‑law scan (cheap proxy).
    # --------------------------------------------------------------
    p_grid = np.linspace(0.55, 0.68, 81)        # dense around the known optimum
    best_a = None
    best_score = float("inf")
    best_p = None

    for p in p_grid:
        if time.time() >= deadline:
            break

        maxb, _ = _conv_max_fft(a, fft_len)

    # fallback (should never happen)
    if best_a is None:

        best_p = 0.60

    # --------------------------------------------------------------
    # 2. Annealed projected gradient descent on the smoothed max.
    # --------------------------------------------------------------
    a = best_a.copy()                # always Σ a = 1
    best_iter_a = a.copy()

    beta_max = 300.0                 # final temperature – forces the true max
    beta_inc_factor = 1.02           # multiplicative increase

        eta = eta0 / math.sqrt(iter_count)          # classic 1/√t decay
        a_candidate = a - eta * grad
        a_candidate = _simplex_projection(a_candidate)

        a = a_candidate

        # ----- 2c. keep track of the best cheap (real) score -----
        if (iter_count % 200 == 0) or (time.time() > deadline - 5):
            cheap = 2.0 * n * maxb
            if cheap < best_iter_score:
                best_iter_score = cheap
                best_iter_a = a.copy()
                last_improve = iter_count
                print(f"[Iter {iter_count:07d}] NEW BEST cheap = {cheap:.6f}")

        # ----- 2d. annealing schedule for β -----
        if iter_count % inc_every == 0:
            beta = min(beta_max, beta * beta_inc_factor)

        # ----- 2e. occasional random perturbation to escape plateaus -----
        if iter_count - last_improve > 70_000:
            pert = 0.02 * np.random.rand(n)
            a = _simplex_projection(best_iter_a + pert)
            last_improve = iter_count
            print(f"[Iter {iter_count:07d}] Stagnation – random perturbation applied.")

    # --------------------------------------------------------------
    # 3. Final exact evaluation (the platform will also call it,
    #    but we do it here to report the achieved bound).
    # --------------------------------------------------------------
    final_seq = best_iter_a.tolist()
    try:

    except Exception as exc:       # defensive – should never happen
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # --------------------------------------------------------------
    # 4. Return the best sequence found.
    # --------------------------------------------------------------
    return final_seq
