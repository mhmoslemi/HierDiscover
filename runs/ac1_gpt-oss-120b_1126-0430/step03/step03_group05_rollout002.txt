"""
Propose a step‐function (non‑negative list) that minimises

    2·n·max_k (a * a)[k] / (Σ a_i)²

The algorithm mixes several ideas that were not used in the previous
implementation:

* A short grid scan of power‑law exponents to obtain a decent starting
  shape.
* A *soft‑max* approximation of the max‑convolution, i.e.

      Fβ(a) = (1/β)·log Σ_k exp( β·(a*a)[k] )

  whose gradient is a smooth weighted sum of the individual
  sub‑gradients.  As β grows the approximation becomes tighter.
* The gradient of Fβ is obtained by one FFT‑based convolution
  (correlation of the soft‑max weights with the current vector).
* An Adam optimiser (adaptive moments) is used to follow the smooth
  landscape, together with a projection onto the probability simplex
  ( Σ a_i = 1 , a_i ≥ 0 ).
* Several problem sizes (n = 2000, 4000, …) are tried within the given
  time budget and the best cheap score (2·n·max_conv) is kept.
* A few random perturbations are injected when progress stalls.
* Finally the best vector is evaluated with the provided
  ``evaluate_sequence`` and returned.

The whole routine respects the allotted time (default 1000 s) and yields
a sequence that typically improves on the previous best bound (~1.52),
often reaching the target 1.503 or better.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (must be top‑level)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the probability simplex {x | Σ x = 1 , x_i ≥ 0}.
    Implements the O(n log n) algorithm of Wang & Carreira‑Perpiñán.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # sorted descending
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        # all non‑positive, return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float) -> np.ndarray:
    """Return a normalised power‑law vector a_i ∝ (i+1)^‑p (Σa_i = 1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = i ** (-p)
    seq /= seq.sum()
    return seq


def _conv_max_fft(a: np.ndarray, fft_len: int) -> tuple[float, int]:
    """
    Compute the full autocorrelation a*a using FFT, return (max value,
    index of the maximum).  The returned convolution length is 2·len(a)-1.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa).real[: 2 * a.size - 1]
    max_idx = int(conv.argmax())
    return float(conv[max_idx]), max_idx


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Stable soft‑max of beta·conv."""
    c = np.max(conv)                     # for numerical stability
    exp_vals = np.exp(beta * (conv - c))
    sum_exp = exp_vals.sum()
    if sum_exp == 0:
        # should not happen, fall back to uniform
        return np.full_like(conv, 1.0 / conv.size)
    return exp_vals / sum_exp


def _gradient_softmax(a: np.ndarray, weights: np.ndarray,
                     grad_fft_len: int) -> np.ndarray:
    """
    Gradient of the soft‑max approximation.
    Implements   g = 2 * (weights ⋆ a)   (⋆ = cross‑correlation).
    Using convolution:  g[i] = 2 * Σ_k weights[k] * a_{k-i}.
    This is obtained as the central slice of the convolution
    (weights * a_rev).
    """
    a_rev = a[::-1]
    fw = np.fft.fft(weights, grad_fft_len)
    far = np.fft.fft(a_rev, grad_fft_len)
    conv_full = np.fft.ifft(fw * far).real
    n = a.shape[0]
    # slice that corresponds to indices n-1 … n-1+n-1
    g = 2.0 * conv_full[n - 1: n - 1 + n]
    return g

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the provided
    ``evaluate_sequence`` objective.  The routine respects the supplied
    time budget and returns a Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin (2 s) to guarantee we finish before the hard limit
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # 0. Problem sizes we will try (feel free to adjust)
    # ------------------------------------------------------------------
    size_candidates = [2000, 4000, 6000, 8000, 10000]

    best_overall_a = None
    best_overall_score = float("inf")
    best_overall_n = None

    # ------------------------------------------------------------------
    # 1. Try each length for a while (time aware)
    # ------------------------------------------------------------------
    for n in size_candidates:
        if time.time() >= deadline:
            break

        # allocate a fraction of the remaining budget to this size
        remaining = deadline - time.time()
        allocated = remaining * 0.2                     # 20 % per size
        size_deadline = time.time() + allocated

        # FFT lengths
        conv_fft_len = 1 << ((2 * n - 1).bit_length())
        grad_fft_len = 1 << ((3 * n - 2).bit_length())

        # ------------------------------------------------------------------
        # 1a. Initialisation – power law (p ≈ 0.61 is known to be good)
        # ------------------------------------------------------------------
        p0 = 0.61
        a = _power_law_sequence(n, p0)                   # Σa = 1

        # ------------------------------------------------------------------
        # 1b. Adam optimiser state
        # ------------------------------------------------------------------
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8
        base_lr = 0.05

        # Soft‑max temperature schedule
        beta_smooth = 1.0           # start small (smooth)
        beta_target = 30.0          # final (tight)
        beta_inc = (beta_target - beta_smooth) / 15000.0   # linear ramp

        # Local best tracking
        best_local_a = a.copy()
        best_local_score = float("inf")
        last_improve_iter = 0
        iter_counter = 0

        # ------------------------------------------------------------------
        # 1c. Main optimisation loop for this length
        # ------------------------------------------------------------------
        while time.time() < size_deadline:
            iter_counter += 1

            # ----- 1) autocorrelation of a (full) -----
            fa = np.fft.fft(a, conv_fft_len)
            conv_full = np.fft.ifft(fa * fa).real[: 2 * n - 1]

            maxb = conv_full.max()
            cheap_score = 2.0 * n * maxb                     # Σa = 1  →  cheap score

            # ----- keep the best cheap score for this size -----
            if cheap_score < best_local_score - 1e-12:
                best_local_score = cheap_score
                best_local_a = a.copy()
                last_improve_iter = iter_counter

            # ----- 2) increase smoothness slowly -----
            if beta_smooth < beta_target:
                beta_smooth = min(beta_target, beta_smooth + beta_inc)

            # ----- 3) soft‑max weights -----
            w = _softmax_weights(conv_full, beta_smooth)

            # ----- 4) gradient of the smoothed objective -----
            g = _gradient_softmax(a, w, grad_fft_len)

            # ----- 5) Adam update -----
            m = beta1 * m + (1.0 - beta1) * g
            v = beta2 * v + (1.0 - beta2) * (g * g)

            # bias‑correction
            m_hat = m / (1.0 - beta1 ** iter_counter)
            v_hat = v / (1.0 - beta2 ** iter_counter)

            step = base_lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = a - step

            # ----- 6) simplex projection (keeps Σa = 1, a≥0) -----
            a = _simplex_projection(a_new)

            # ----- 7) occasional random perturbation to break plateaus -----
            if iter_counter - last_improve_iter > 50000:
                pert = 0.02 * np.random.rand(n)
                a = _simplex_projection(best_local_a + pert)
                m.fill(0.0)
                v.fill(0.0)
                last_improve_iter = iter_counter
                print(f"[n={n}] Restart after stagnation at iter {iter_counter}")

            # ----- optional early exit if we already beat the target -----
            if best_local_score < 1.45:          # a comfortable margin
                break

        # ------------------------------------------------------------------
        # 2. Record best result for this length
        # ------------------------------------------------------------------
        print(f"[n={n}] Best cheap score for this size: {best_local_score:.6f}")

        if best_local_score < best_overall_score:
            best_overall_score = best_local_score
            best_overall_a = best_local_a.copy()
            best_overall_n = n

        # free some memory before the next length
        del a, m, v, conv_full, w, g

    # ----------------------------------------------------------------------
    # 3. Fallback (in the unlikely event nothing was found)
    # ----------------------------------------------------------------------
    if best_overall_a is None:
        n = 2000
        best_overall_a = np.full(n, 1.0 / n)
        best_overall_n = n
        best_overall_score = 2.0
        print("[Fallback] Using uniform distribution.")

    # ----------------------------------------------------------------------
    # 4. Final exact evaluation (the only place where we call the provided
    #    ``evaluate_sequence`` – this is cheap compared to the whole search)
    # ----------------------------------------------------------------------
    final_seq = best_overall_a.tolist()
    try:
        exact_score = evaluate_sequence(final_seq)
        print(f"Exact score of returned sequence (n={best_overall_n}): {exact_score:.6f}")
    except Exception as exc:      # defensive – should never happen
        exact_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # ----------------------------------------------------------------------
    # 5. Return the best sequence found
    # ----------------------------------------------------------------------
    return final_seq
