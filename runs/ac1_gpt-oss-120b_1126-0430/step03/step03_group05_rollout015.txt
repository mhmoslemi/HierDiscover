def propose_candidate(seed=42, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2

    # Determine sequence length n
    # Use previous if present and length >= 5000 else maybe 8000
    try:
        prev = height_sequence_1
        n = max(len(prev), 5000)
        prev_arr = np.asarray(prev, dtype=np.float64)
        prev_arr = prev_arr / prev_arr.sum() if prev_arr.sum() > 0 else None
    except Exception:
        prev_arr = None
        n = 8000

    # Precompute FFT lengths
    fft_len1 = 1 << ((2 * n - 1).bit_length())
    fft_len2 = 1 << ((3 * n - 2).bit_length())

    # Initial candidate: scan power law exponents
    p_grid = np.linspace(0.45, 0.70, 101)  # coarse then fine
    best_a = None
    best_score = np.inf
    best_p = None
    for p in p_grid:
        if time.time() >= deadline: break
        a = _power_law_sequence(n, p)
        conv = np.fft.ifft(np.fft.fft(a, fft_len1) * np.fft.fft(a, fft_len1)).real[:2*n - 1]

        cheap_score = 2.0 * n * maxb
        if cheap_score < best_score:
            best_score = cheap_score
            best_a = a.copy()
            best_p = p

    if prev_arr is not None and prev_arr.size == n:
        conv = np.fft.ifft(np.fft.fft(prev_arr, fft_len1) * np.fft.fft(prev_arr, fft_len1)).real[:2*n - 1]
        cheap_score = 2.0 * n * np.max(conv)
        if cheap_score < best_score:
            best_score = cheap_score
            best_a = prev_arr.copy()
            best_p = "prev"

    if best_a is None:

Alternatively use mirror descent: update a = a * np.exp(-lr * grad); then renormalize dividing by sum. This avoids sorting and ensures nonnegative. Might be faster. However multiplicative update may cause some values extremely small; but projection similarly may produce zeros.

We can attempt this: after gradient, compute log updates as new a_i = a_i * np.exp(-lr * grad_i). Then a = a / sum(a). This is equivalent to performing exponentiated gradient descent (mirror descent with KL prox). This ensures step size acts on relative log changes.

We need to bound step size to avoid underflow/overflow. Use lr ~ small, maybe 0.1. In practice, exponent may become large for large grad; but grad magnitude maybe up to 2 (max). So exponent values up to e^{-0.1*2}=~0.818 => fine.

Thus we can replace projection with exponentiated update. However exponentiated gradient may be slower to converge but cheaper computationally.

But we need to choose approach.

Given the algorithm may run many iterations, removing sorting may improve speed; but sorting may still be okay.

Time budgets: Each iteration currently approx 0.2ms for FFT + 0.2ms for second FFT + 0.02ms for sorting (maybe ~0.4ms). So 500k iterations -> 200 secs? Actually 0.4ms*500k=200k ms=200 seconds. Then plus overhead, still within 1000 sec.

Thus sorting not a bottleneck.

Thus keep projection.

#!/usr/bin/env python
"""
Optimiser for the step‑function objective.

The objective (with the sequence normalised to sum = 1) is
    2·n·max_k (a*a)[k] .
We therefore minimise the *maximum* entry of the self‑convolution.
The algorithm combines:

* a dense power‑law initialisation (grid search over the exponent);
* a smoothed stochastic gradient obtained from a soft‑max over the
  convolution entries (temperature τ is increased during the run);
* Adam‑style adaptive steps;
* projection on the probability simplex after each update;
* occasional random perturbations when progress stalls.

The implementation uses FFTs for O(n log n) convolutions and respects
the 1000 s time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Projects a vector onto the probability simplex {x | Σx = 1, x ≥ 0}.
    Uses the O(n log n) algorithm based on sorting.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:          # all non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float) -> np.ndarray:
    """
    Returns a normalised power‑law vector a_i ∝ (i+1)^{-p} with Σa = 1.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = i ** (-p)
    seq_sum = seq.sum()
    if seq_sum == 0:
        return np.full(n, 1.0 / n)
    return seq / seq_sum


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full convolution a*a (real part) using FFT of length fft_len."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa).real[: 2 * a.size - 1]
    return conv


def _gradient_smoothed(a: np.ndarray,
                       conv: np.ndarray,
                       maxb: float,
                       tau: float,
                       fft_len2: int) -> np.ndarray:
    """
    Computes the gradient of the soft‑max smoothed max‑convolution.
    Parameters
    ----------
    a       : current probability vector (length n)
    conv    : full convolution a*a (length 2n‑1)
    maxb    : maximum entry of conv
    tau     : temperature for the soft‑max (larger → sharper)
    fft_len2: length of the FFT used for the second convolution
    Returns
    -------
    grad    : ∇  ( Σ_k w_k (a*a)[k] )   where w_k ∝ exp(τ·(conv_k‑maxb))
    """
    # ----- soft‑max weights (stabilised) -----
    diff = conv - maxb                     # ≤ 0
    # Clip the exponent to avoid overflow/underflow (exp(-700) ≈ 5e-305)
    exp_args = np.clip(tau * diff, -700, 0)
    w = np.exp(exp_args)
    w_sum = w.sum()
    if w_sum == 0 or not np.isfinite(w_sum):
        # fall back to the pure max (single active index)
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        w_sum = 1.0
    w /= w_sum

    # ----- gradient = 2 * (a convolve w_rev) reversed slice -----
    w_rev = w[::-1]                        # length 2n‑1
    n = a.shape[0]
    # FFT of a and w_rev (both zero‑padded to fft_len2)
    fa = np.fft.fft(a, fft_len2)
    fw = np.fft.fft(w_rev, fft_len2)
    conv2 = np.fft.ifft(fa * fw).real      # length fft_len2 ≥ 3n‑2
    # indices that give the required slice
    idxs = (len(w_rev) - 1) - np.arange(n)   # 2n‑2 , … , n‑1
    grad = 2.0 * conv2[idxs]
    return grad


# ----------------------------------------------------------------------
# Main optimisation entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine runs for at most ``budget_s`` seconds
    and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    # ------------------------------------------------------------------
    # 0. Choose a working length.
    #    Re‑use a previous construction if it exists, otherwise start with
    #    a moderately large n (≈8000) which gives a good trade‑off
    #    between expressive power and runtime.
    # ------------------------------------------------------------------
    try:
        prev = height_sequence_1   # may be defined in the surrounding environment
        n = max(len(prev), 5000)
        prev_arr = np.asarray(prev, dtype=np.float64)
        s = prev_arr.sum()
        if s > 0:
            prev_arr = prev_arr / s
        else:
            prev_arr = None
    except Exception:
        prev_arr = None
        n = 8000                       # default length

    # FFT lengths for the two convolutions we need
    fft_len1 = _next_pow2(2 * n - 1)          # for a*a
    fft_len2 = _next_pow2(3 * n - 2)          # for a * w_rev

    # ------------------------------------------------------------------
    # 1. Scan a grid of power‑law exponents to obtain a good start.
    # ------------------------------------------------------------------
    p_grid = np.linspace(0.45, 0.70, 101)     # fairly dense scan
    best_a = None
    best_score = float("inf")
    best_p = None

    for p in p_grid:
        if time.time() >= deadline:
            break
        a = _power_law_sequence(n, p)
        conv = _conv_full_fft(a, fft_len1)
        maxb = float(conv.max())
        cheap_score = 2.0 * n * maxb
        if cheap_score < best_score:
            best_score = cheap_score
            best_a = a.copy()
            best_p = p

    # also try the previous construction (if any)
    if prev_arr is not None and prev_arr.size == n:
        conv = _conv_full_fft(prev_arr, fft_len1)
        maxb = float(conv.max())
        cheap_score = 2.0 * n * maxb
        if cheap_score < best_score:
            best_score = cheap_score
            best_a = prev_arr.copy()
            best_p = "prev"

    # fallback in the unlikely event that nothing was selected
    if best_a is None:
        best_a = _power_law_sequence(n, 0.55)
        conv = _conv_full_fft(best_a, fft_len1)
        best_score = 2.0 * n * float(conv.max())
        best_p = 0.55

    print(f"Initial scan finished: best p = {best_p}, cheap score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Refine using Adam + smoothed max‑convolution gradient.
    # ------------------------------------------------------------------
    a = best_a.copy()                # current iterate, Σa = 1
    m = np.zeros_like(a)             # Adam first moment
    v = np.zeros_like(a)             # Adam second moment

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 0.12                        # initial (pre‑decay) learning rate

    # Temperature schedule for the soft‑max (starts low → smooth,
    # grows to make the approximation tighter)
    tau = 300.0
    tau_max = 8000.0
    tau_inc = (tau_max - tau) / 500_000   # reach τ_max after ~5·10⁵ steps

    best_iter_a = a.copy()
    best_iter_score = best_score

    iter_count = 0
    last_improve = 0
    restart_after = 180_000            # perturb after this many idle steps

    while time.time() < deadline:
        iter_count += 1

        # ----- compute full self‑convolution -----
        conv = _conv_full_fft(a, fft_len1)
        max_idx = int(conv.argmax())
        maxb = float(conv[max_idx])
        cheap_score = 2.0 * n * maxb

        # ----- keep the best cheap score seen so far -----
        if cheap_score < best_iter_score - 1e-9:
            best_iter_score = cheap_score
            best_iter_a = a.copy()
            last_improve = iter_count
            print(f"[Iter {iter_count:07d}] NEW BEST cheap score = {cheap_score:.6f}")

        # ----- temperature update (slow linear increase) -----
        tau = min(tau_max, tau + tau_inc)

        # ----- smoothed gradient -----
        grad = _gradient_smoothed(a, conv, maxb, tau, fft_len2)

        # ----- Adam update (bias‑corrected) -----
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)

        bias_c1 = 1 - beta1 ** iter_count
        bias_c2 = 1 - beta2 ** iter_count
        m_hat = m / bias_c1
        v_hat = v / bias_c2

        # Decaying step size (helps later‑stage fine‑tuning)
        lr = lr0 / math.sqrt(iter_count)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        a_candidate = a - step
        a = _simplex_projection(a_candidate)

        # ----- occasional restart when progress stalls -----
        if iter_count - last_improve > restart_after:
            # small random bump around the current best
            bump = 0.02 * np.random.rand(n)
            a = _simplex_projection(best_iter_a + bump)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = iter_count
            print(f"[Iter {iter_count:07d}] Restarted after stagnation (perturb).")

        # ----- lightweight logging -----
        if iter_count % 5000 == 0:
            elapsed = time.time() - start_time
            print(f"[Iter {iter_count:07d}] elapsed {elapsed:.1f}s, cheap score {cheap_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Final exact evaluation with the provided ``evaluate_sequence``.
    # ------------------------------------------------------------------
    final_seq = best_iter_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # ------------------------------------------------------------------
    # 4. Return the best sequence discovered.
    # ------------------------------------------------------------------
    return final_seq
