"""
Propose a non‑negative step‑function (vector on the simplex) that minimises

        O(a) = 2·n·max(conv(a,a)) / (Σ a_i)²

The algorithm works in three stages:

1. **Large‑scale Adam optimisation** on a smooth log‑sum‑exp surrogate of
   the max‑convolution.  The surrogate becomes sharper (α grows) while Adam
   steers the vector on the probability simplex.

2. **Deterministic sub‑gradient descent** on the true maximal convolution
   entry.  The step size decays linearly.

3. **Guided simulated‑annealing** that repeatedly transfers a small
   fraction of mass from indices that contribute most to the current
   maximum (obtained from the sub‑gradient) to random low‑mass indices.
   Metropolis acceptance with an exponential cooling schedule helps to
   escape shallow plateaus.

The routine starts from a finer version of the previously found best
(`height_sequence_1` if present) by linear interpolation to a larger
length (default 8192).  All operations are vectorised and use FFTs for
full convolutions/correlations.  The total running time never exceeds the
provided budget (≈1000 s).  The final vector is returned as a plain
Python list; its quality is printed via `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, Σ x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full convolution via FFT (length 2·n‑1, real input)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return a * a (full convolution) using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Objective for a simplex‑normalised vector (sum = 1)
def _objective_simplex(a: np.ndarray, n: int, L: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for Σ a_i = 1."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Sub‑gradient of the maximal convolution entry
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a.
    Returns a vector of the same size as a.
    """
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)          # all maximising shifts
    g = np.zeros_like(a)

    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        # each pair contributes 2·a_j to the gradient w.r.t. a_i
        g[i_idx] += 2.0 * a[j_idx]

    if ks.size > 0:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
# Gradient of the smooth log‑sum‑exp surrogate
def _smooth_gradient(a: np.ndarray, conv: np.ndarray, alpha: float, L: int) -> np.ndarray:
    """
    ∇_a ( (2·n/α)·log∑_k e^{α·conv_k} ).
    The result is 2·corr(w, a) where w_k ∝ e^{α·conv_k}.
    """
    # soft‑max weights (stable)
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / exp_scaled.sum()

    # correlation w ⋆ a   (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)   # length ≥ 3·n‑2
    n = a.size
    grad = 2.0 * corr_rev[n - 1 : n - 1 + n]          # central n entries
    return grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000):
    """
    Search for a step‑function sequence that minimises the provided evaluation.
    Returns the best sequence (list of floats) found within the time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # 0.5 s safety margin

    # ------------------------------------------------------------------
    # Determine problem size and initialise a
    if 'height_sequence_1' in globals():
        base = np.array(height_sequence_1, dtype=float)
        base_n = base.size
        # enlarge the problem: double the length (or at least 8192)
        n = max(base_n * 2, 8192)
        # linear interpolation to the larger grid
        x_old = np.linspace(0, base_n - 1, base_n)
        x_new = np.linspace(0, base_n - 1, n)
        a = np.interp(x_new, x_old, base)
        a = np.clip(a, 0.0, None)
        a /= a.sum()
        print(f"[init] Upsampled previous solution: n={n}")
    else:
        n = 8192
        a = np.full(n, 1.0 / n)
        print(f"[init] No previous solution – start uniform, n={n}")

    # FFT length – must accommodate both convolution (2n‑1) and correlation (3n‑2)
    L = 1 << ((3 * n - 2).bit_length())            # next power of two

    # ------------------------------------------------------------------
    # Initial objective
    conv = _conv_fft(a, L)
    best_a = a.copy()
    best_obj = _objective_simplex(best_a, n, L)
    print(f"[init] O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam on a smoothed surrogate (≈45 % of the budget)
    phase1_deadline = start_time + budget_s * 0.45
    max_iter_phase1 = 400_000
    alpha_start, alpha_end = 5.0, 5000.0
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.2                     # modest learning rate
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    iter1 = 0

    while time.time() < phase1_deadline and iter1 < max_iter_phase1:
        # current convolution and objective
        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            print(f"[phase‑1] iter={iter1}, O={best_obj:.6f}")

        # linear schedule for α
        frac = iter1 / max_iter_phase1
        alpha = alpha_start + (alpha_end - alpha_start) * frac

        # gradient of the smooth surrogate
        grad = _smooth_gradient(a, conv, alpha, L)

        # Adam updates (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (iter1 + 1))
        v_hat = v / (1.0 - beta2 ** (iter1 + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # projected simplex step
        a = _project_to_simplex(a - step)

        iter1 += 1
        if time.time() >= deadline - 5.0:
            break

    # ------------------------------------------------------------------
    # Phase 2 – deterministic sub‑gradient descent (≈35 % of budget)
    phase2_deadline = start_time + budget_s * 0.80
    max_iter_subgrad = 200_000
    iter2 = 0

    while time.time() < phase2_deadline and iter2 < max_iter_subgrad:
        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            print(f"[subgrad] iter={iter2}, O={best_obj:.6f}")

        # sub‑gradient of the maximal convolution entry
        g = _subgradient_max_conv(a, conv)

        # linearly decaying step size
        eta = 0.15 * (1.0 - iter2 / max_iter_subgrad)
        a = a - eta * g
        a = _project_to_simplex(a)

        # occasional tiny Gaussian jitter to break symmetry
        if iter2 % 2000 == 0 and iter2 > 0:
            jitter = np.random.randn(n) * 1e-3
            a = _project_to_simplex(a + jitter)

        iter2 += 1
        if time.time() >= deadline - 5.0:
            break

    # ------------------------------------------------------------------
    # Phase 3 – guided simulated‑annealing hill‑climbing (remainder of time)
    temperature0 = 0.02
    total_hc_iters = 0
    local_a = best_a.copy()
    local_conv = _conv_fft(local_a, L)
    local_obj = 2.0 * n * local_conv.max()

    while time.time() < deadline:
        # exponential cooling
        T = temperature0 * (0.999 ** total_hc_iters)

        # sub‑gradient gives a convenient importance distribution
        g = _subgradient_max_conv(local_a, local_conv)
        prob = g / g.sum() if g.sum() > 0 else None

        # pick source index biased towards large contribution
        if prob is not None:
            i = np.random.choice(n, p=prob)
        else:
            i = np.random.randint(n)

        # pick a distinct destination uniformly
        j = np.random.randint(n)
        while j == i:
            j = np.random.randint(n)

        if local_a[i] == 0.0:
            total_hc_iters += 1
            continue

        # transfer a random fraction of the source mass
        delta = local_a[i] * np.random.rand() * 0.15
        a_new = local_a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = _project_to_simplex(a_new)

        conv_new = _conv_fft(a_new, L)
        obj_new = 2.0 * n * conv_new.max()

        # Metropolis acceptance
        if obj_new < local_obj or np.random.rand() < math.exp((local_obj - obj_new) / max(T, 1e-12)):
            local_a = a_new
            local_conv = conv_new
            local_obj = obj_new
            if local_obj < best_obj:
                best_obj = local_obj
                best_a = local_a.copy()
                print(f"[anneal] iter={total_hc_iters}, O={best_obj:.6f}")

        total_hc_iters += 1

        # periodically re‑anchor the local walk to the global best
        if total_hc_iters % 5000 == 0:
            local_a = best_a.copy()
            local_conv = _conv_fft(local_a, L)
            local_obj = 2.0 * n * local_conv.max()

        if time.time() >= deadline - 1.0:
            break

    # ------------------------------------------------------------------
    # Final sanity check using the supplied evaluation routine
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence failed:", exc)

    return best_a.tolist()
