"""
Propose a step‑function (non‑negative vector summing to 1) that
minimises the bound

    O = 2·n·max(conv(a,a))

where conv(a,a) is the full self‑convolution of the sequence.
The optimiser is a two‑phase method:

* Phase 1 – smooth optimisation: the max is approximated by a
  soft‑max (log‑sum‑exp) with an annealing parameter β that grows.
  A momentum term (exponential moving average of the gradient)
  is used and the iterate is kept on the probability simplex.

* Phase 2 – exact sub‑gradient optimisation: once β is large enough
  the soft‑max collapses to an indicator on the active set
  (the indices where the convolution attains its maximum).  The
  resulting exact sub‑gradient is used together with the same
  momentum scheme and a diminishing step‑size η∝1/√t.

Random jitter is injected each step to break symmetry and the best
solution so far is periodically used as a restart point.  After the
continuous optimisation a cheap local random‑swap refinement is run
until the time budget expires.  The final sequence is returned as a
plain list of floats; its quality is printed and also fed to the
provided ``evaluate_sequence`` function.
"""

import time
import numpy as np

# -------------------------------------------------------------
# Helper – projection onto the probability simplex {x ≥ 0, sum(x)=1}
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Projects a vector onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# -------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# -------------------------------------------------------------
def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv


# -------------------------------------------------------------
def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlate weight vector w (length 2n‑1) with a (length n):
        g[i] = Σ_t w[i+t]·a[t] ,   i=0…n‑1
    Implemented via FFT.
    """
    n = a.shape[0]
    m = w.shape[0]               # = 2n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


# -------------------------------------------------------------
def _max_conv(a: np.ndarray) -> float:
    """Return the maximum entry of the self‑convolution of a."""
    return _conv_self_fft(a).max()


# -------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats)
    minimising the ``evaluate_sequence`` bound.
    Returns the best sequence found (as a plain list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2   # small safety margin

    # ---------------------------------------------------------
    # 1) Choose an initial length and a starting point
    if 'height_sequence_1' in globals() and height_sequence_1:
        a0 = np.asarray(height_sequence_1, dtype=float)
        a0 = a0 / a0.sum()
        n = a0.size
        a = a0.copy()
    else:
        n = 4600                     # a reasonably large length
        a = np.random.rand(n)
        a = _project_to_simplex(a)

    best_seq = a.copy()
    best_score = 2.0 * n * _max_conv(best_seq)
    print(f"[init] n={n}, O={best_score:.6f}")

    # ---------------------------------------------------------
    # 2) Optimisation hyper‑parameters
    # Phase split – 30 % of the wall‑time for the smooth phase
    start_time = time.time()
    phase1_end = start_time + 0.30 * (deadline - start_time)

    beta_start = 2.0
    beta_end = 5000.0
    eta0 = 0.5                 # initial step‑size
    momentum_coeff = 0.9
    jitter0 = 0.005            # relative jitter magnitude
    active_eps = 1e-12         # tolerance for the active set

    momentum = np.zeros_like(a)
    it = 0

    # ---------------------------------------------------------
    # 3) Main optimisation loop
    while time.time() < deadline:
        it += 1

        # --- current objective and tracking of the best solution ---
        conv = _conv_self_fft(a)
        M = conv.max()
        cur_O = 2.0 * n * M
        if cur_O < best_score - 1e-12:
            best_score = cur_O
            best_seq = a.copy()
            print(f"[iter {it:06d}] new best O={best_score:.6f}")

        # --- decide which gradient to use (soft‑max vs indicator) ---
        if time.time() < phase1_end:                # smooth phase
            # linear schedule for β
            frac = (time.time() - start_time) / (phase1_end - start_time)
            beta = beta_start + (beta_end - beta_start) * frac
            shifted = conv - M                     # ≤ 0, max at 0
            # robust soft‑max (avoid overflow)
            exp_vals = np.exp(beta * shifted)
            w_sum = exp_vals.sum()
            if w_sum == 0.0:                       # numerical under‑flow
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w = exp_vals / w_sum
        else:                                        # exact sub‑gradient
            active = np.where(conv >= M - active_eps)[0]
            if active.size == 0:
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w = np.zeros_like(conv)
                w[active] = 1.0 / active.size

        # --- gradient of the (smoothed) max ---
        g = 2.0 * _correlation_fft(w, a)   # shape (n,)

        # --- momentum update ---
        momentum = momentum_coeff * momentum + (1.0 - momentum_coeff) * g

        # --- step‑size schedule (η ∝ 1/√t) ---
        eta = eta0 / np.sqrt(it)

        # --- gradient step + random jitter ---
        a_new = a - eta * momentum
        jitter = (jitter0 / np.sqrt(it)) * np.random.randn(n)
        a_new += jitter

        # --- projection back onto the simplex ---
        a = _project_to_simplex(a_new)

        # --- occasional restart from the current best (to escape plateaus) ---
        if it % 20000 == 0:
            a = best_seq.copy()
            a += 0.002 * np.random.randn(n)
            a = _project_to_simplex(a)
            print(f"[restart] iter {it}, re‑initialised from best.")

    # ---------------------------------------------------------
    # 4) Local random‑swap refinement (uses the cheap max‑conv objective)
    remaining = deadline - time.time()
    if remaining > 5.0:                    # leave a few seconds for the final call
        # Very cheap moves: transfer a small amount of mass between two bins
        loc_iters = int(remaining * 500)   # roughly 2 ms per iteration
        for _ in range(loc_iters):
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            # propose a transfer of a tiny fraction of the mass at i
            delta = 0.001 * best_seq[i] * np.random.rand()
            if delta <= 0.0:
                continue
            cand = best_seq.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = _project_to_simplex(cand)     # keep feasible
            new_O = 2.0 * n * _max_conv(cand)
            if new_O < best_score - 1e-12:
                best_score = new_O
                best_seq = cand
                print(f"[local] improvement O={best_score:.6f}")

    # ---------------------------------------------------------
    # 5) Final evaluation with the provided evaluate_sequence routine
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_seq.tolist()
