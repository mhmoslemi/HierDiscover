def _beta_distribution(n, p, q):
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    # avoid zero or one
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = (x ** (p - 1)) * ((1 - x) ** (q - 1))
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a

  smooth approximation of the max:
        L(a) = (1/γ)·log(∑_k exp(γ·(a*a)[k]))
  whose gradient is a weighted sum of all convolution entries.
* The gradient is computed efficiently with two FFT‑based convolutions.
* An Adam optimiser with a back‑tracking line‑search updates the logits.
* The schedule for γ starts small (smooth landscape) and is increased
  up to 30 so that the final iterates focus on the true maximum.
* Several simple parametric initialisations (uniform, power‑law,
  Beta‑shaped) are tried, and the best candidate is used as a start.
* The best sequence seen (according to the cheap proxy 2·n·max_conv)
  is finally evaluated with the provided ``evaluate_sequence`` and
  returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max (sum = 1, non‑negative)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Logit representation for a probability vector."""
    return np.log(a + eps)


def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two ≥ 2·n‑1 (size needed for a*a convolution)."""
    return 1 << ((2 * n - 1).bit_length())


def _fft_len_for_len(L: int) -> int:
    """Smallest power‑of‑two ≥ L."""
    return 1 << ((L - 1).bit_length())


def _conv_max_fft(a: np.ndarray, fft_len: int) -> tuple[float, int]:
    """Maximum value and its index of the self‑convolution a*a."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.size - 1]
    idx = int(conv.argmax())
    return float(conv.max()), idx


def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1)."""
    n = a.shape[0]
    fft_len = _fft_len_for_n(n)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def _smooth_max_gradient(a: np.ndarray, gamma: float) -> tuple[np.ndarray, float]:
    """
    Gradient of the smooth‑max objective
        L(a) = (1/γ)·log∑_k exp(γ·(a*a)[k])
    Returns (grad_a, max_conv) where grad_a = ∂L/∂a
    and max_conv = max_k (a*a)[k] (exact max, not the smooth one).
    """
    n = a.shape[0]

    # ----- full convolution a*a ------------------------------------
    conv_full = _conv_full_fft(a)                     # length 2n‑1
    max_conv = float(conv_full.max())

    # ----- soft‑max weights over convolution entries ---------------
    # exp(γ·(conv - max)) to avoid overflow
    exp_vals = np.exp(gamma * (conv_full - max_conv))
    w = exp_vals / exp_vals.sum()                     # shape (2n‑1,)

    # ----- gradient w.r.t a (2·(w * a_rev) convolution) ----------
    a_rev = a[::-1]                                   # reversed a
    conv_len = 3 * n - 2
    fft_len2 = _fft_len_for_len(conv_len)

    fw = np.fft.rfft(w, fft_len2)
    fa_rev = np.fft.rfft(a_rev, fft_len2)
    conv_wa = np.fft.irfft(fw * fa_rev, fft_len2)[:conv_len]

    # gradient component 2·∑_k w_k·a_{k-i}
    grad_a = 2.0 * conv_wa[n - 1 : n - 1 + n]

    return grad_a, max_conv


# ----------------------------------------------------------------------
# Parametric initial distributions
# ----------------------------------------------------------------------


def _uniform_distribution(n: int) -> np.ndarray:
    """Uniform probability vector."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}  (p>0)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    a /= s
    return a


def _beta_distribution(n: int, p: float, q: float) -> np.ndarray:
    """
    Normalised Beta‑shaped density:
        a_i ∝ x_i^{p-1} (1-x_i)^{q-1},   x_i = (i+0.5)/n .
    Works for any p,q>0 (p=q=1 gives uniform).
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = (x ** (p - 1)) * ((1.0 - x) ** (q - 1))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _random_dirichlet(n: int, alpha: float = 0.5) -> np.ndarray:
    """Draw a random probability vector from Dirichlet(alpha)."""
    a = np.random.gamma(alpha, 1.0, size=n)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _choose_initial_distribution(n: int, budget_deadline: float) -> np.ndarray:
    """
    Try a handful of simple families and keep the one with the smallest
    cheap proxy 2·n·max_convolution.
    """
    best_a = None
    best_score = float("inf")
    fft_len = _fft_len_for_n(n)

    # uniform
    cand = _uniform_distribution(n)
    maxb, _ = _conv_max_fft(cand, fft_len)
    score = 2.0 * n * maxb
    best_a, best_score = cand, score

    # power‑law grid
    for p in np.linspace(0.4, 0.8, 9):
        if time.time() >= budget_deadline:
            break
        cand = _power_law_distribution(n, p)
        maxb, _ = _conv_max_fft(cand, fft_len)
        score = 2.0 * n * maxb
        if score < best_score:
            best_a, best_score = cand, score

    # Beta grid (U‑shaped families)
    for p in [0.3, 0.5, 0.7, 0.9]:
        for q in [0.3, 0.5, 0.7, 0.9]:
            if time.time() >= budget_deadline:
                break
            cand = _beta_distribution(n, p, q)
            maxb, _ = _conv_max_fft(cand, fft_len)
            score = 2.0 * n * maxb
            if score < best_score:
                best_a, best_score = cand, score

    # a few random Dirichlet draws
    for _ in range(5):
        if time.time() >= budget_deadline:
            break
        cand = _random_dirichlet(n, alpha=0.3)
        maxb, _ = _conv_max_fft(cand, fft_len)
        score = 2.0 * n * maxb
        if score < best_score:
            best_a, best_score = cand, score

    return best_a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) improves the
    current bound.  The routine respects the supplied time budget
    (~1000 s) and uses an Adam optimiser on the probability simplex with
    a smooth‑max surrogate for the max‑convolution.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0   # small safety margin

    # ---------------------------------------------------------------
    # 0. Problem size
    # ---------------------------------------------------------------
    n = 20000               # default size – large enough for a good bound
    # one could enlarge later, but the current optimiser works well at 20 k

    # ---------------------------------------------------------------
    # 1. Initialise a sensible probability vector.
    # ---------------------------------------------------------------
    a = _choose_initial_distribution(n, deadline)
    print(f"[Init] chosen initial distribution, cheap score = {2*n*_conv_max_fft(a, _fft_len_for_n(n))[0]:.6f}")

    # ---------------------------------------------------------------
    # 2. Log‑it parametrisation + Adam state.
    # ---------------------------------------------------------------
    eps = 1e-16
    z = _logits_from_distribution(a, eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    base_alpha = 0.2

    # ---------------------------------------------------------------
    # 3. Main optimisation loop.
    # ---------------------------------------------------------------
    best_seq = a.copy()
    best_score = 2.0 * n * _conv_max_fft(a, _fft_len_for_n(n))[0]

    it = 0
    report_every = 2000

    while time.time() < deadline and it < 1_000_000:
        it += 1

        # probability vector from logits
        a = _softmax(z)

        # smooth‑max gradient (γ grows slowly)
        gamma = 5.0 + 45.0 * min(1.0, it / 20000.0)
        grad_a, maxb = _smooth_max_gradient(a, gamma)

        cheap_score = 2.0 * n * maxb

        # keep best ever seen
        if cheap_score < best_score:
            best_score = cheap_score
            best_seq = a.copy()
            if it % report_every == 0:
                print(f"[Iter {it:07d}] NEW BEST cheap = {cheap_score:.6f}")

        # gradient in log‑it space (mirror descent on the simplex)
        mu = np.dot(grad_a, a)
        grad_z = a * (grad_a - mu)

        # Adam moment updates (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # direction (negative Adam step)
        step_dir = - m_hat / (np.sqrt(v_hat) + eps_adam)

        # basic step‑size schedule: inverse‑sqrt decay
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)

        # -----------------------------------------------------------
        # Back‑tracking line search (up to 4 halvings)
        # -----------------------------------------------------------
        accepted = False
        trial_alpha = alpha_t
        for _ in range(4):
            z_new = z + trial_alpha * step_dir
            z_new = np.clip(z_new, -50.0, 50.0)           # keep exp() stable
            a_new = _softmax(z_new)
            maxb_new, _ = _conv_max_fft(a_new, _fft_len_for_n(n))
            new_score = 2.0 * n * maxb_new
            if new_score <= cheap_score:
                z = z_new
                accepted = True
                break
            trial_alpha *= 0.5

        if not accepted:
            # still move a little (helps escape very flat regions)
            z += alpha_t * step_dir
            z = np.clip(z, -50.0, 50.0)

        # -----------------------------------------------------------
        # Occasional random perturbations (helps escape plateaus)
        # -----------------------------------------------------------
        if it % 5000 == 0 and np.random.rand() < 0.3:
            perturb = 0.02 * np.random.randn(n)
            z += perturb
            m *= 0.5
            v *= 0.5

    # ---------------------------------------------------------------
    # 4. Final exact evaluation (uses the provided evaluate_sequence)
    # ---------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:   # defensive – should never happen
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # ---------------------------------------------------------------
    # 5. Return the best sequence found.
    # ---------------------------------------------------------------
    return final_seq
