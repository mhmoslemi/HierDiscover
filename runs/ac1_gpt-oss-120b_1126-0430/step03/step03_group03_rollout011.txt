"""
Propose a non‑negative step‑function sequence that (approximately) minimises

    O(a) = 2·n·max(conv(a,a))   with   Σ a_i = 1  (i.e. a lies on the probability simplex).

Algorithm – multi‑restart, projected Adam on a smooth log‑sum‑exp surrogate,
followed by sub‑gradient refinement and a final simulated‑annealing hill‑climb.
Each restart gets a share of the total time budget; the best vector over all
restarts is returned.

All helper functions are defined at top level – no closures or lambdas.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# 1.  Simplex projection ------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:                     # all entries ≤ 0  → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# 2.  FFT‑based full convolution ----------------------------------------
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# 3.  Exact objective (simplex‑normalised) -----------------------------
def _objective_simplex(a: np.ndarray, L: int) -> float:
    """2·n·max(conv(a,a)) – a is assumed to sum to 1."""
    n = a.size
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# 4.  Sub‑gradient of the maximal convolution entry --------------------
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. a."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)    # all maximising indices
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        # each ordered pair contributes a_j (or a_i) once;
        # the derivative w.r.t. a_i is sum_j a_j over all matching j's,
        # multiplied by 2 because of ordered counting.
        g[i_idx] += 2.0 * a[j_idx]
    g /= len(ks)                           # average if several maximisers
    return g


# ----------------------------------------------------------------------
# 5.  Gradient of the smooth log‑sum‑exp surrogate --------------------
def _smooth_gradient(a: np.ndarray, conv: np.ndarray, alpha: float, L: int) -> np.ndarray:
    """
    Gradient of (2·n/α)·log Σ_k exp(α·conv_k).
    Result = 2·corr(w, a) with w_k ∝ exp(α·conv_k).
    """
    # soft‑max weights, computed in a numerically stable way
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / exp_scaled.sum()

    # correlation w ⋆ a  (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)       # length ≥ 3·n‑2
    n = a.size
    grad = 2.0 * corr_rev[n - 1 : n - 1 + n]               # central n entries
    return grad


# ----------------------------------------------------------------------
# 6.  Optimisation of a single start vector ----------------------------
def _optimise_one_start(a_start: np.ndarray, deadline: float) -> tuple:
    """
    Run the three‑phase optimisation (Adam → sub‑gradient → annealing)
    on a single starting vector until *deadline* is reached.
    Returns (best_a, best_obj).
    """
    a = a_start.copy()
    n = a.size
    # FFT length – must accommodate both convolution (2n‑1) and correlation (3n‑2)
    L = 1 << ((3 * n - 2).bit_length())   # next power of two

    # ------------------------------------------------------------------
    # initial bookkeeping
    conv = _conv_fft(a, L)
    best_a = a.copy()
    best_obj = _objective_simplex(best_a, L)
    # print progress for this start
    print(f"[init] n={n}, O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam on the log‑sum‑exp surrogate
    alpha_start, alpha_end = 5.0, 2000.0
    max_iter_phase1 = 300_000               # large enough; will be cut by time
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.15                                 # a slightly higher base LR
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    phase1_deadline = min(deadline - 0.5, time.time() + (deadline - time.time()) * 0.45)

    iter1 = 0
    while time.time() < phase1_deadline and iter1 < max_iter_phase1:
        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            print(f"[phase‑1] iter={iter1:,}, O={best_obj:.6f}")

        # linear annealing of α
        frac = iter1 / max_iter_phase1
        alpha = alpha_start + (alpha_end - alpha_start) * frac

        # smooth gradient
        grad = _smooth_gradient(a, conv, alpha, L)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (iter1 + 1))
        v_hat = v / (1.0 - beta2 ** (iter1 + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a - step
        a = _project_to_simplex(a_candidate)

        iter1 += 1

    # ------------------------------------------------------------------
    # Phase 2 – deterministic sub‑gradient refinement
    max_iter_subgrad = 150_000
    phase2_deadline = min(deadline - 0.5, time.time() + (deadline - time.time()) * 0.25)

    iter2 = 0
    while time.time() < phase2_deadline and iter2 < max_iter_subgrad:
        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            print(f"[subgrad] iter={iter2:,}, O={best_obj:.6f}")

        # sub‑gradient of the maximal entry
        g = _subgradient_max_conv(a, conv)

        # linearly decaying step size – a bit more aggressive than before
        eta = 0.18 * (1.0 - iter2 / max_iter_subgrad)
        a = a - eta * g
        a = _project_to_simplex(a)

        # tiny jitter every 1000 steps
        if iter2 % 1000 == 0 and iter2 > 0:
            jitter = np.random.randn(n) * 4e-4
            a = _project_to_simplex(a + jitter)

        iter2 += 1

    # ------------------------------------------------------------------
    # Phase 3 – simulated‑annealing hill climbing (global refinement)
    total_hc_iters = 0
    # start from the best vector found so far
    local_a = best_a.copy()
    local_conv = _conv_fft(local_a, L)
    local_obj = 2.0 * n * local_conv.max()

    # simple exponential cooling
    temperature0 = 0.06
    while time.time() < deadline - 0.5:
        T = temperature0 * (0.996 ** total_hc_iters)

        i, j = np.random.randint(n, size=2)
        if i == j or local_a[i] == 0.0:
            total_hc_iters += 1
            continue

        # transfer a random fraction (up to 20 %) of mass from i → j
        delta = local_a[i] * np.random.rand() * 0.20
        a_new = local_a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = _project_to_simplex(a_new)

        conv_new = _conv_fft(a_new, L)
        obj_new = 2.0 * n * conv_new.max()

        # Metropolis acceptance
        if obj_new < local_obj or np.random.rand() < math.exp((local_obj - obj_new) / max(T, 1e-12)):
            local_a, local_conv, local_obj = a_new, conv_new, obj_new
            if local_obj < best_obj:
                best_obj = local_obj
                best_a = local_a.copy()
                print(f"[anneal] iter={total_hc_iters:,}, O={best_obj:.6f}")

        total_hc_iters += 1

        # Re‑seed every few thousand steps to avoid drifting too far
        if total_hc_iters % 4000 == 0:
            local_a = best_a.copy()
            local_conv = _conv_fft(local_a, L)
            local_obj = 2.0 * n * local_conv.max()

    return best_a, best_obj


# ----------------------------------------------------------------------
# 7.  Main entry point -------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence that minimises ``evaluate_sequence``.
    Returns the best sequence as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # small safety margin

    # ------------------------------------------------------------------
    # Choose a few problem sizes (the length of the step function).  For each
    # size we try a few different initialisations and keep the overall best.
    n_choices = [3000, 4096, 4800]               # modest sizes fitting comfortably
    best_global_obj = float("inf")
    best_global_seq = None

    remaining_sizes = len(n_choices)
    for n in n_choices:
        # allocate a share of the remaining wall‑clock time to this size
        now = time.time()
        time_left = deadline - now
        if time_left <= 0:
            break
        size_deadline = now + time_left / remaining_sizes - 0.2
        remaining_sizes -= 1

        # ------------------------------------------------------------------
        # Build a list of initial vectors for this size
        inits = []

        # a) use the provided sequence (if it exists and matches the size)
        if "height_sequence_1" in globals():
            base = np.array(height_sequence_1, dtype=float)
            if base.size == n:
                inits.append(base / base.sum())
            elif base.size > n:
                inits.append(base[:n] / base[:n].sum())
            else:   # pad with zeros
                pad = np.zeros(n, dtype=float)
                pad[: base.size] = base
                inits.append(pad / pad.sum())

        # b) uniform start
        inits.append(np.full(n, 1.0 / n))

        # c) random Dirichlet (exponential) start – gives a spikier shape
        rand_vec = np.random.exponential(1.0, size=n)
        inits.append(rand_vec / rand_vec.sum())

        # ------------------------------------------------------------------
        # Run optimisation from each start, respecting the per‑size deadline
        for a0 in inits:
            a_opt, obj_opt = _optimise_one_start(a0, size_deadline)
            if obj_opt < best_global_obj:
                best_global_obj = obj_opt
                best_global_seq = a_opt.copy()
                print(f"[global] new overall best O={best_global_obj:.6f} (n={n})")

    # ------------------------------------------------------------------
    # Final sanity check with the provided evaluation routine
    if best_global_seq is None:
        raise RuntimeError("No feasible sequence found within the time budget.")
    try:
        final_score = evaluate_sequence(best_global_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence failed:", exc)

    return best_global_seq.tolist()
