"""
Propose a step‑function (non‑negative sequence) that minimizes the bound

    O = 2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm starts from the analytically motivated “half‑binomial’’
distribution (coefficients of (1‑x)^‑½) which is known to give a nearly
flat self‑convolution.  It is then refined by an exponentiated‑gradient
descent on a smooth approximation of the max‑convolution (soft‑max with
an annealing parameter β).  The update

    a_i ← a_i·exp(‑η·g_i) ,   a ← a / sum(a)

keeps the vector on the probability simplex without an explicit
projection.  β is increased exponentially (from ≈0.1 to ≈2000) while the
step‑size η is decreased exponentially (from 0.5 to 0.001).  Small
multiplicative jitter is added occasionally and, if progress stalls,
the current best solution is perturbed and used as a new start.

The routine respects a 1000 s budget, prints progress and finally
returns the best sequence (as a plain list of floats) found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of a vector onto the simplex {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute g_i = Σ_t w[i+t]·a[t]   for i = 0,…,n‑1.
    (Linear correlation of w with a, no reversal of a.)
    Implemented via FFT.
    """
    n = a.shape[0]
    m = w.shape[0]                # = 2n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) (sum(a)=1)."""
    n = a.size
    conv = _conv_self_fft(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _half_binomial_seq(n: int) -> np.ndarray:
    """
    Return the first n coefficients of (1‑x)^‑½,
    i.e. a_i = C(2i,i) / 4**i  (unnormalised, all positive).
    Recurrence: a[0]=1, a[i] = a[i‑1] * (2i‑1) / (2i).
    """
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the
    evaluate_sequence bound.  Returns the best sequence as a list of floats.
    """

    deadline = time.time() + budget_s - 0.5   # small safety margin

        n = a.size
    else:
        # a moderately large length – the problem favours a few thousand.
        n = 4600
        a = _half_binomial_seq(n)          # analytic start
        a = a / a.sum()

    best_seq = a.copy()
    best_score = _objective(best_seq)
    print(f"[init] n={n}, initial O={best_score:.6f}")

    # ------------------------------------------------------------------
    # Schedules (exponential in the iteration fraction)
    max_iters_est = 300_000                # rough estimate for the schedule
    beta_start = 0.1
    beta_end = 2000.0
    eta_start = 0.5
    eta_end = 0.001

    # Momentum is not used – the update is exponentiated‑gradient.
    iter_cnt = 0
    while time.time() < deadline and iter_cnt < 2_000_000:
