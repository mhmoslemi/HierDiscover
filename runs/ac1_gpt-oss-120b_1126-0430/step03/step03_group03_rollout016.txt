def solve_lp(f, M):

    g = cp.Variable(n, nonneg=True)
    conv = cp.conv(f, g)  # length 2n-1 expression

    g_val = np.maximum(g.value, 0.0)  # ensures nonnegative

    a = softmax(x)
    conv = _conv_fft(a, L)

    scaled = alpha * conv

"""
Propose a non‑negative step‑function sequence that minimises

        O(a) = 2·n·max(conv(a,a)) ,   Σ a_i = 1

by a hybrid optimisation:
    1.  L‑BFGS on the smooth log‑sum‑exp surrogate (multiple α‑levels).
    2.  Exponentiated‑gradient (EG) sub‑gradient descent on the true max‑conv.
    3.  Classic Euclidean sub‑gradient descent (as in the reference code).
    4.  Simulated‑annealing hill‑climbing for a global polish.

The routine respects a 1000 s wall‑clock budget, prints progress
and finally returns the best sequence found (as a plain list of floats).
"""

import time
import math
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full self‑convolution (length 2·n‑1)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return a*a (full convolution) using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Exact objective for a simplex‑normalised vector
def _objective_simplex(a: np.ndarray, n: int, L: int) -> float:
    """2·n·max(conv(a,a)) – a is assumed to sum to 1."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Sub‑gradient of the maximal convolution entry (used in Phase 2/3)
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Return the sub‑gradient of max(conv) w.r.t. a."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)          # indices achieving the maximum
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]                # each pair contributes symmetrically
    g /= len(ks)
    return g


# ----------------------------------------------------------------------
# Gradient of the smooth log‑sum‑exp surrogate (used in L‑BFGS)
def _smooth_gradient(a: np.ndarray, conv: np.ndarray, alpha: float, L: int) -> np.ndarray:
    """
    Compute ∇_a ( (2·n/α)·log∑_k e^{α·conv_k} ).
    Result = 2·corr(w, a) where w_k ∝ e^{α·conv_k}.
    """
    # soft‑max weights (stable)
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / exp_scaled.sum()

    # correlation w ⋆ a   (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)   # length ≥ 3·n‑2
    n = a.size
    grad = 2.0 * corr_rev[n - 1 : n - 1 + n]          # central n entries
    return grad


# ----------------------------------------------------------------------
# Soft‑max based surrogate and its gradient (for L‑BFGS)
def _surrogate_and_grad(x: np.ndarray, n: int, L: int, alpha: float):
    """
    x    – log‑weights (unconstrained)
    returns (f, g) where
        f = (2·n/α)·log∑_k exp(α·conv_k)
        g = ∇_x f
    """
    # soft‑max to enforce simplex
    e = np.exp(x - np.max(x))
    a = e / e.sum()

    conv = _conv_fft(a, L)
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    sum_exp = exp_shifted.sum()
    f_val = (2.0 * n / alpha) * (np.log(sum_exp) + max_scaled)

    # gradient w.r.t. a
    grad_a = _smooth_gradient(a, conv, alpha, L)

    # chain rule: ∂f/∂x_i = Σ_j ∂f/∂a_j · ∂a_j/∂x_i
    # ∂a_j/∂x_i = a_j (δ_{ij} – a_i)
    inner = np.dot(a, grad_a)          # Σ_j a_j * ∂f/∂a_j
    grad_x = a * (grad_a - inner)      # vectorised Jacobian of soft‑max
    return f_val, grad_x


# ----------------------------------------------------------------------
# Exponentiated‑gradient step on the true max‑conv sub‑gradient
def _eg_step(a: np.ndarray, subgrad: np.ndarray, eta: float) -> np.ndarray:
    """EG update: a_i ← a_i·exp(‑η·g_i) then re‑normalise."""
    a_new = a * np.exp(-eta * subgrad)
    if a_new.sum() == 0.0:
        # fall back to uniform to avoid division by zero
        a_new = np.full_like(a, 1.0 / a.size)
    else:
        a_new = a_new / a_new.sum()
    return a_new

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """Search for a step‑function sequence that minimises the provided evaluation."""
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Problem size (use previous best if available)
    if 'height_sequence_1' in globals():
        base = np.array(height_sequence_1, dtype=float)
        n = base.size
        a = base / base.sum()
        print(f"[init] Using supplied sequence, n={n}")
    else:
        n = 4096
        a = np.full(n, 1.0 / n)
        print(f"[init] No prior sequence – start uniform, n={n}")

    # FFT length – must be ≥ 3·n‑2 for the smooth gradient routine
    L = 1 << ((3 * n - 2).bit_length())

    # ------------------------------------------------------------------
    # Best‑so‑far bookkeeping
    cur_obj = _objective_simplex(a, n, L)
    best_a = a.copy()
    best_obj = cur_obj
    print(f"[init] O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS on the smooth surrogate (multiple α‑levels)
    # ------------------------------------------------------------------
    # α schedule – start gentle, end sharp
    alpha_schedule = [5.0, 20.0, 100.0, 500.0, 2000.0, 8000.0]
    # limiting total L‑BFGS calls to stay inside the budget
    max_lbfgs_iters = 800   # per α (SciPy internal) – cheap for n≈4k

    for alpha in alpha_schedule:
        if time.time() >= deadline - 5.0:
            break

        # initialise log‑weights from the current best a
        x0 = np.log(best_a + 1e-12)

            return _surrogate_and_grad(x, n, L, alpha)

        # L‑BFGS‑B (no bound constraints needed)

                                      x0,
                                      method='L-BFGS-B',
                                      jac=True,
                                      options={'maxiter': max_lbfgs_iters,
                                               'ftol': 1e-12,
                                               'disp': False})
        # retrieve simplex‑normalised a
        a_opt = np.exp(res.x - np.max(res.x))
        a_opt = a_opt / a_opt.sum()

        cur_obj = _objective_simplex(a_opt, n, L)
        if cur_obj < best_obj:
            best_obj, best_a = cur_obj, a_opt.copy()

    # ------------------------------------------------------------------

            print(f"[subgrad] iter={it}, O={best_obj:.6f}")

        g = _subgradient_max_conv(a, conv)

        a = _project_to_simplex(a)

        if it % 1000 == 0 and it > 0:
            jitter = np.random.randn(n) * 5e-4
            a = _project_to_simplex(a + jitter)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    temperature0 = 0.04

        a_new = _project_to_simplex(a_new)

        conv_new = _conv_fft(a_new, L)
        obj_new = 2.0 * n * conv_new.max()

            local_a, local_conv, local_obj = a_new, conv_new, obj_new
            if local_obj < best_obj:
                best_obj, best_a = local_obj, local_a.copy()
                print(f"[anneal] iter={total_hc_iters}, O={best_obj:.6f}")

            local_conv = _conv_fft(local_a, L)
