def _smooth_obj_and_grad(w, n, beta):
    a = np.exp(w)
    a = a / np.sum(a)
    # compute convolution
    size1 = 1 << ((2 * n - 2).bit_length())
    A = np.fft.fft(a, n=size1)
    conv = np.fft.ifft(A * A).real[:2*n-1]

    maxc = conv.max()
    # compute softmax weights w_soft
    exps = np.exp(beta * (conv - maxc))
    sum_exps = np.sum(exps)
    w_soft = exps / sum_exps

    # smooth max
    smooth_max = maxc + np.log(sum_exps) / beta
    J = 2.0 * n * smooth_max

    # gradient w.r.t a:
    a_rev = a[::-1]
    size2 = 1 << ((2 * n - 1 + n - 1).bit_length())
    conv_rev = np.fft.ifft(np.fft.fft(w_soft, size2) * np.fft.fft(a_rev, size2)).real
    h = conv_rev[n-1 : n-1 + n]
    grad_a = 4.0 * n * h

    # gradient w.r.t w:
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)

    return J, grad_w

"""
Propose a candidate step‑function (non‑negative sequence) that minimises the
bound evaluated by ``evaluate_sequence``.

The search works in three stages:

1. **Exponentiated‑gradient descent** on the true max‑convolution
   (sub‑gradient).  This quickly finds a good region of the simplex.
2. **Refinement with L‑BFGS** on a smooth approximation of the max
   (log‑sum‑exp).  We optimise the logits of the soft‑max (a = softmax(w)),
   which guarantees the non‑negativity and the simplex constraint.
   Several increasing smoothing parameters β are used.
3. The best sequence found is finally evaluated with the supplied
   ``evaluate_sequence`` function.

The implementation respects the 1000 s time budget, prints progress, and
stores the best sequence in the global variable ``height_sequence_1`` for
future runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a power‑of‑two FFT."""
    n = a.size
    size = 1 << ((2 * n - 2).bit_length())
    A = np.fft.fft(a, n=size)
    conv = np.fft.ifft(A * A).real[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact bound 2·n·max(conv(a,a)) for a already normalised (Σa=1)."""
    n = a.size
    conv = _conv_fft(a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv(a,a)) w.r.t. a on the simplex."""
    n = a.size
    conv = _conv_fft(a)
    max_val = conv.max()
    ks = np.where(np.abs(conv - max_val) < 1e-14)[0]
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size > 0:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
def _exp_update(a: np.ndarray, g: np.ndarray, eta: float) -> np.ndarray:
    """
    Exponentiated‑gradient step on the simplex.
    a – current distribution (already normalised)
    g – sub‑gradient of the max‑convolution
    eta – learning rate
    """
    log_a = np.log(a + 1e-15)
    log_a -= eta * g
    log_a -= np.max(log_a)                # keep numbers moderate
    a_new = np.exp(log_a)
    a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
def _optimize(a0: np.ndarray, max_iters: int, eta0: float,
              deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple exponentiated‑gradient descent on the sub‑gradient of max(conv).
    Returns the best vector found and its objective value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)

    n = a.size
    no_improve = 0
    patience = max_iters // 20

    for t in range(max_iters):
        if time.time() >= deadline:
            break

        g = _subgradient_max_conv(a)
        eta = eta0 * (1.0 - t / max_iters)   # linearly decaying
        a = _exp_update(a, g, eta)

        cur_score = _objective(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            print(f"[EG] iter {t:5d}  new best O={best_score:.6f}")
        else:
            no_improve += 1
            if no_improve >= patience:
                # jitter to escape flat region
                jitter = np.random.randn(n) * 1e-4
                a = best_a + jitter
                a = np.clip(a, 0.0, None)
                a = _project_to_simplex(a)
                no_improve = 0

    return best_a, best_score


# ----------------------------------------------------------------------
def _obj_and_grad_w(w: np.ndarray, n: int, beta: float):
    """
    Smooth objective (log‑sum‑exp approximation of max) and its gradient
    w.r.t. the logits w (such that a = softmax(w)).
    Returns a tuple (value, grad_w) suitable for scipy.optimize.minimize.
    """
    # a = softmax(w)
    a = np.exp(w)
    a_sum = a.sum()
    a = a / a_sum

    # convolution a*a
    conv = _conv_fft(a)                     # length 2n‑1
    maxc = conv.max()

    # soft‑max weights over the convolution entries
    exps = np.exp(beta * (conv - maxc))     # stable formulation
    sum_exps = exps.sum()
    w_soft = exps / sum_exps                # Σ w_soft = 1

    # smooth approximation of the max
    smooth_max = maxc + np.log(sum_exps) / beta
    value = 2.0 * n * smooth_max

    # gradient w.r.t. a : 4·n·(w_soft * a)  (cross‑correlation)
    a_rev = a[::-1]                         # length n
    size2 = 1 << ((len(w_soft) + n - 1).bit_length())
    conv_rev = np.fft.ifft(np.fft.fft(w_soft, size2) *
                           np.fft.fft(a_rev, size2)).real
    # slice that yields Σ_k w_soft[k]·a_{k‑i}
    h = conv_rev[n - 1: n - 1 + n]          # shape (n,)
    grad_a = 4.0 * n * h

    # gradient w.r.t. logits w (chain rule through softmax)
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)

    return value, grad_w


# ----------------------------------------------------------------------
def _lbfgs_refine(a_start: np.ndarray, betas, maxiter, deadline):
    """
    Refine a distribution using L‑BFGS on the smooth objective for several
    smoothing parameters (betas).  Returns the best distribution found.
    """
    n = a_start.size
    # initialise logits
    w0 = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = _objective(best_a)

    # import lazily – SciPy may be unavailable in some environments
    try:
        from scipy.optimize import minimize
    except Exception as exc:           # pragma: no cover
        print("[L‑BFGS] SciPy not available, skipping refinement")
        return best_a

    for beta in betas:
        if time.time() >= deadline:
            break
        # wrapper required by scipy: returns (f, grad)
        def fun(w, n=n, beta=beta):
            return _obj_and_grad_w(w, n, beta)

        res = minimize(fun,
                       w0,
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': maxiter,
                                'ftol': 1e-12,
                                'disp': False})
        if not res.success:
            # continue anyway – we may still have a useful point
            print(f"[L‑BFGS] beta={beta:.1f} failed: {res.message}")

        a_candidate = np.exp(res.x)
        a_candidate /= a_candidate.sum()
        cand_score = _objective(a_candidate)

        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
            print(f"[L‑BFGS] beta={beta:.1f}  improved O={best_score:.6f}")

        # Use the last solution as start for the next (larger β)
        w0 = res.x.copy()

    return best_a


# ----------------------------------------------------------------------

    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Choose a working length.
    if 'height_sequence_1' in globals():
        a_prev = np.asarray(height_sequence_1, dtype=float)
        n = a_prev.size
        a_prev = a_prev / a_prev.sum()
    else:
        a_prev = None
        n = 4000                     # reasonable default

    # ------------------------------------------------------------------
    # Build a small pool of starting points.
    starts = []

    if a_prev is not None:
        starts.append(a_prev)

    # 1/√ law (theoretically good)
    idx = np.arange(1, n + 1, dtype=float)
    a_sqrt = 1.0 / np.sqrt(idx)
    a_sqrt /= a_sqrt.sum()
    starts.append(a_sqrt)

    # Random simplex point
    a_rand = np.random.rand(n)
    a_rand = _project_to_simplex(a_rand)
    starts.append(a_rand)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Parameters for the two optimisation phases
    eg_iters = 25000                 # more iterations than the previous script
    eg_eta0 = 0.45
    lbfgs_betas = [5.0, 20.0, 100.0]     # increasing smoothness
    lbfgs_maxiter = 120                 # limited inner iterations per β

    for i, a_start in enumerate(starts):
        if time.time() >= deadline:
            break
        init_score = _objective(a_start)
        print(f"[init {i}] start O={init_score:.6f}")

        # ---- Phase 1 : EG descent ---------------------------------
        a_opt, score_opt = _optimize(a_start, eg_iters, eg_eta0, deadline)
        print(f"[EG {i}] finished O={score_opt:.6f}")

        # ---- Phase 2 : L‑BFGS refinement ---------------------------
        a_refined = _lbfgs_refine(a_opt, lbfgs_betas, lbfgs_maxiter, deadline)
        final_score = _objective(a_refined)
        print(f"[L‑BFGS {i}] refined O={final_score:.6f}")

        if final_score < best_score:
            best_score = final_score
            best_seq = a_refined.copy()

    # ------------------------------------------------------------------
    # Fallback: if no sequence was found (e.g. time ran out) use the 1/√ shape.
    if best_seq is None:
        best_seq = a_sqrt.copy()
        best_score = _objective(best_seq)

    # Store for potential future runs

    height_sequence_1 = best_seq.tolist()

    # ------------------------------------------------------------------
    # Final exact evaluation using the supplied function
    try:

        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:  # pragma: no cover
        print("[final] evaluate_sequence call failed:", e)
        final_score = None

    return best_seq.tolist()
