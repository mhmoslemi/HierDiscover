def _adam_optimize(a, deadline, L, n):
    # Adam parameters
    lr = 0.05
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    best_seq = a.copy()
    best_metric = _fast_evaluation(a)
    start = time.time()
    # schedule parameters for softmax alpha
    alpha_min = 100.0
    alpha_max = 20000.0
    # estimate max_iterations based on time: we can compute roughly.
    max_iter_est = 20000
    while time.time() < deadline:
        # compute Fa once
        Fa = np.fft.rfft(a, L)
        # compute conv
        conv = np.fft.irfft(Fa * Fa, L)[: 2*n - 1]
        M = conv.max()
        # smoothing alpha schedule based on iteration t
        alpha = alpha_min + (alpha_max - alpha_min) * min(1.0, t / max_iter_est)
        # compute softmax weights
        # we compute w = exp(alpha*(conv - M)), no overflow cause exponent non-positive
        w = np.exp(alpha * (conv - M))
        w_sum = w.sum()
        if w_sum == 0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient
        Fw = np.fft.rfft(w_norm, L)
        corr = np.fft.irfft(np.conj(Fa) * Fw, L)
        grad = 2.0 * corr[:n]

        # Adam updates
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        # project onto simplex
        a = _project_simplex(a_new)

        # evaluate metric using fast evaluation
        cur_metric = _fast_evaluation(a)

        if cur_metric < best_metric - 1e-12:
            best_metric = cur_metric
            best_seq = a.copy()
            # print progress occasionally
            if t % 500 == 0:
                print(f"[adam] iter {t:6d} metric {best_metric:.6f}")

        # occasional random perturbation to escape
        if np.random.rand() < 0.001:
            idx = np.random.randint(n)
            a[idx] += np.random.randn() * 0.01
            a = _project_simplex(a)

    return best_seq, best_metric

"""
Propose a low‑score step‑function.

Algorithm
---------
1.  Random‑parameter search (Beta‑densities, sin‑powers, mixtures) on the
    full length (n≈5000).  This quickly discovers a promising shape.
2.  Refine the best candidate with a projected Adam optimizer.
    The objective is a smoothed version of the max‑convolution
    (soft‑max with a gradually increasing temperature).  The gradient
    of the smooth objective is obtained analytically via an FFT‑based
    cross‑correlation, then the iterate is projected onto the simplex
    (non‑negative, sum = 1).
3.  The final vector is returned as a plain Python list.  All values are
    non‑negative and sum to 1, so the supplied ``evaluate_sequence`` can
    be called safely by the harness.

The implementation respects the 1000 s budget, prints progress
information and makes no use of filesystem or external services.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast evaluation (identical to the supplied ``evaluate_sequence`` but
# without the security checks).  Used during the search.
# ----------------------------------------------------------------------
def _fast_evaluation(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) / (sum(seq)²) for a non‑negative seq."""
    n = seq.size
    if n == 0:
        return float("inf")
    # Length for linear convolution via FFT (next power of two)
    L = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, L)
    conv = np.fft.irfft(fa * fa, L)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean).  Projects onto {x ≥ 0, sum x = 1}.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # Find the largest rho satisfying the condition
    cond = u - cssv / ind > 0
    if not cond.any():
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Seed generators (Beta, sin‑power, Dirichlet, mixtures)
# ----------------------------------------------------------------------
def _beta_sequence(alpha: float, beta: float, length: int) -> np.ndarray:
    """Beta(α,β) density sampled on the mid‑points of a uniform grid."""
    xs = (np.arange(length, dtype=float) + 0.5) / length
    a = np.power(xs, alpha - 1.0, out=np.empty_like(xs))
    b = np.power(1.0 - xs, beta - 1.0, out=np.empty_like(xs))
    seq = a * b
    seq[np.isinf(seq)] = 0.0
    seq = np.maximum(seq, 0.0)
    total = seq.sum()
    if total == 0.0:
        seq = np.full(length, 1.0 / length)
    else:
        seq /= total
    return seq


def _sin_power_sequence(p: float, length: int) -> np.ndarray:
    """a_i ∝ sin(π·(i+0.5)/length)^p, p may be negative."""
    xs = (np.arange(length, dtype=float) + 0.5) / length
    vals = np.sin(np.pi * xs) ** p
    vals = np.maximum(vals, 0.0)
    total = vals.sum()
    if total == 0.0:
        vals = np.full(length, 1.0 / length)
    else:
        vals /= total
    return vals


def _random_dirichlet(length: int, concentration: float = 0.5) -> np.ndarray:
    """Draw a random point from Dirichlet(concentration,…,concentration)."""
    seq = np.random.gamma(concentration, 1.0, size=length)
    total = seq.sum()
    if total == 0.0:
        seq = np.full(length, 1.0 / length)
    else:
        seq /= total
    return seq


def _mixture_two_beta(length: int) -> np.ndarray:
    """Random mixture of two Beta densities."""
    a1 = 10 ** np.random.uniform(-1, 0.8)
    b1 = 10 ** np.random.uniform(-1, 0.8)
    a2 = 10 ** np.random.uniform(-1, 0.8)
    b2 = 10 ** np.random.uniform(-1, 0.8)
    w = np.random.rand()
    seq1 = _beta_sequence(a1, b1, length)
    seq2 = _beta_sequence(a2, b2, length)
    seq = w * seq1 + (1.0 - w) * seq2
    total = seq.sum()
    if total == 0.0:
        seq = np.full(length, 1.0 / length)
    else:
        seq /= total
    return seq


# ----------------------------------------------------------------------
# Adam‑style optimisation on the simplex, with a soft‑max
# approximation of the max‑convolution.
# ----------------------------------------------------------------------
def _adam_optimize(
    init_seq: np.ndarray,
    deadline: float,
    fft_len: int,
    n: int,
    *,
    lr: float = 0.07,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    alpha_start: float = 100.0,
    alpha_end: float = 20000.0,
) -> tuple[np.ndarray, float]:
    """
    Refine ``init_seq`` (sum=1, non‑negative) using Adam on the
    smooth max‑convolution.  Returns the best sequence seen and
    its fast‑evaluation score.
    """
    a = init_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    best_seq = a.copy()
    best_score = _fast_evaluation(a)

    start_opt = time.time()
    # maximum number of iterations is bounded only by the time budget
    while time.time() < deadline:
        # ----- convolution a*a (linear convolution) -----
        Fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(Fa * Fa, fft_len)[: 2 * n - 1]
        M = conv.max()

        # ----- soft‑max weights (smooth approximation of the max) -----
        elapsed = (time.time() - start_opt) / (deadline - start_opt)
        alpha = alpha_start + (alpha_end - alpha_start) * elapsed
        # exponent is always ≤ 0, so no overflow worries
        w = np.exp(alpha * (conv - M))
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # ----- gradient of the smooth objective -----
        # correlation of a with w_norm gives g_i = 2 * Σ_j a_j w_{j+i}
        Fw = np.fft.rfft(w_norm, fft_len)
        corr = np.fft.irfft(np.conj(Fa) * Fw, fft_len)
        grad = 2.0 * corr[:n]          # gradient of the smooth max

        # ----- Adam moment updates -----
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- projected update (simplex) -----
        a = _project_simplex(a - step)

        # ----- keep the best observed point -----
        cur_score = _fast_evaluation(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            if t % 500 == 0:
                print(f"[adam] iter {t:6d}  best {best_score:.6f}")

        # ----- occasional random jitter to escape shallow minima -----
        if np.random.rand() < 0.001:
            j = np.random.randint(n)
            a[j] = max(0.0, a[j] + np.random.randn() * 1e-3)
            a = _project_simplex(a)

    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:

    """

    start_time = time.time()
    # small safety margin before the harness times out
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    n = 5000

    # (n + (2n‑1) - 1 = 3n‑2).  Use the next power‑of‑two.
    fft_len = 1 << ((3 * n - 2).bit_length())

    # ------------------------------------------------------------------
    best_seq = None

            b = 10 ** np.random.uniform(-1, 0.8)

        elif r < 0.95:                    # sin‑power family
            p = np.random.uniform(-1.0, 1.0)

        else:                             # mixture of two Betas
            seq = _mixture_two_beta(n)

        # occasionally inject a raw Dirichlet draw (pure randomness)
        if np.random.rand() < 0.02:

        # evaluate quickly
        score = _fast_evaluation(seq)

        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[search] iter {iteration:5d}  new best {best_score:.6f}")

    # ------------------------------------------------------------------

                seq /= seq.sum()
            else:
                seq = hs.copy()
                seq = np.maximum(seq, 0.0)
                seq /= seq.sum()
            score = _fast_evaluation(seq)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"[seed] height_sequence_1 improves best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam refinement on the best seed.
    # ------------------------------------------------------------------
    if best_seq is None:

    # ensure a proper probability vector
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total == 0.0:
        best_seq = np.full(n, 1.0 / n)
    else:
        best_seq /= total

    # run the optimiser for the remaining time
    if time.time() < deadline:
        print("[final] starting Adam refinement on full size...")
        refined_seq, refined_score = _adam_optimize(
            best_seq,
            deadline,
            fft_len,
            n,
            lr=0.07,
            alpha_start=200.0,      # start a bit smoother than the search
            alpha_end=30000.0,      # finish with a very sharp soft‑max
        )
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq
            print(f"[final] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence as a plain Python list.
    # ------------------------------------------------------------------
    print(f"[summary] final metric {best_score:.6f} with length {best_seq.size}")
