def propose_candidate(seed=42,budget_s=1000):
    np.random.seed(seed)
    deadline=time.time() + budget_s - 2
    n=20000

    # Precompute FFT size
    size = 1 << ((3*n - 2).bit_length())
    offset = 2*n - 2

    def evaluate(a):
        conv = _fft_convolve(a)
        return 2.0 * n * conv.max()

    # Function to run exponential gradient from a0
    def run_exp_grad(a0):
        a = a0.copy()
        best_local = a.copy()
        best_val = evaluate(best_local)
        lam = 0.5
        lam_max = 200.0
        iter0 = 0
        while time.time() < deadline:
            t = iter0
            # compute a's FFT once
            fa = np.fft.rfft(a, size)
            # self-convolution via same FFT
            conv_full = np.fft.irfft(fa * fa, size)
            conv = conv_full[: 2*n - 1]
            max_c = conv.max()
            cur_val = 2.0 * n * max_c
            if cur_val < best_val:
                best_val = cur_val
                best_local = a.copy()
                print(f"[expgrad improve] iter {t:06d} val {best_val:.9f}")
            # smoothed weight
            # subtract max for stability
            max_conv = conv.max()
            exp_weights = np.exp(lam * (conv - max_conv))
            w = exp_weights / exp_weights.sum()
            # compute gradient via correlation
            w_rev = w[::-1]
            fw_rev = np.fft.rfft(w_rev, size)
            conv_rev = np.fft.irfft(fa * fw_rev, size)
            # extract gradient
            g = 2.0 * conv_rev[offset - np.arange(n)]
            # update step size
            eta = 0.5 / np.sqrt(t + 1)
            # multiplicative update
            a = a * np.exp(-eta * g)
            # renormalize
            sum_a = a.sum()
            if sum_a == 0:
                a = np.full_like(a, 1.0 / n)
            else:
                a = a / sum_a
            # schedule lambda
            lam = min(lam_max, lam * 1.001)
            # occasional random noise
            if t % 5000 == 0 and t>0:
                a += np.random.normal(scale=0.001, size=n)
                a = np.maximum(a,0)
                a = a / a.sum()
            iter0 +=1
        return best_local, best_val

"""
Propose a sequence of non‑negative heights that minimises the
``evaluate_sequence`` objective.

The algorithm combines two ideas:

1. **Catalan start** – the coefficients a_i = C(2i,i)/4^i (the Catalan‑type
   distribution) are known to be close to optimal for this problem.

2. **Exponential‑gradient descent with soft‑max smoothing** – we minimise a
   smooth approximation of the max‑convolution
   L_λ(a)= (1/λ)·log Σ_k exp(λ·(a∗a)_k).  Its gradient can be computed
   efficiently by a single FFT per iteration (plus one extra FFT for the
   soft‑max weights).  Using the mirror‑descent update
   a_i ← a_i·exp(−η·∇_i L_λ) and renormalising onto the simplex yields a
   fast, projection‑free optimiser that respects the non‑negativity and sum‑1
   constraints.

The temperature λ is slowly increased (annealed) so the optimiser first
explores the landscape on a smooth surface and then focuses on the true
max‑convolution.  Random perturbations and a few cheap random restarts are
added if time permits.  The best sequence seen during the run is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------
def _catalan_normalised(n: int) -> np.ndarray:
    """Return the first n Catalan‑type coefficients normalised to sum‑1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    a /= a.sum()
    return a


def _objective(a: np.ndarray, n: int) -> float:
    """
    Exact value returned by ``evaluate_sequence`` for a vector `a` whose
    entries already sum to 1.
    """
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, size)
    conv_full = np.fft.irfft(fa * fa, size)
    max_c = conv_full[: 2 * n - 1].max()
    return 2.0 * n * max_c


# ----------------------------------------------------------------------
# Exponential‑gradient optimiser (soft‑max smoothing of the max)
# ----------------------------------------------------------------------
def _expgrad_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    *,
    max_iters: int = 200_000,
    λ_start: float = 0.5,
    λ_max: float = 400.0,
    η0: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the smoothed max‑convolution objective.
    Returns the best vector encountered and its (unsmoothed) objective value.
    """
    a = a0.copy()
    best = a.copy()
    best_val = _objective(best, n)

    # FFT size large enough for the correlation a * w_rev (length 3n‑2)
    size = 1 << ((3 * n - 2).bit_length())
    offset = 2 * n - 2

    λ = λ_start
    t = 0

    while time.time() < deadline and t < max_iters:
        # ----- self‑convolution (full, using the large FFT size) -----
        fa = np.fft.rfft(a, size)
        conv_full = np.fft.irfft(fa * fa, size)
        conv = conv_full[: 2 * n - 1]               # a ∗ a
        max_c = conv.max()
        cur_val = 2.0 * n * max_c

        # keep the best point seen so far
        if cur_val < best_val:
            best_val = cur_val
            best = a.copy()
            print(f"[expgrad improve] iter {t:07d}  obj {best_val:.9f}")

        # ----- soft‑max weights for the current convolution -----
        # shift for numerical stability
        shift = conv.max()
        w = np.exp(λ * (conv - shift))
        w /= w.sum()                               # Σ w = 1

        # ----- gradient of the smoothed objective (correlation) -----
        w_rev = w[::-1]                            # reversed weights
        fw_rev = np.fft.rfft(w_rev, size)
        conv_rev = np.fft.irfft(fa * fw_rev, size)  # a ⋆ w_rev (full)
        # extract the n relevant entries: g_i = 2·(a ⋆ w_rev)_{offset‑i}
        g = 2.0 * conv_rev[offset - np.arange(n)]

        # ----- mirror‑descent (multiplicative) update -----
        η = η0 / np.sqrt(t + 1)                    # diminishing step size
        a = a * np.exp(-η * g)                     # element‑wise
        a_sum = a.sum()
        if a_sum == 0.0:                           # safety net
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / a_sum                           # re‑normalise to Σa = 1

        # ----- anneal λ (increase the “temperature”) -----
        λ = min(λ_max, λ * 1.001)

        # occasional tiny Gaussian perturbation to escape plateaus
        if t % 5_000 == 0 and t > 0:
            a += np.random.normal(scale=5e-4, size=n)
            a = np.maximum(a, 0.0)
            a /= a.sum()

        t += 1

    return best, best_val


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Return a list of non‑negative heights that (approximately) minimises
    ``evaluate_sequence`` within the allotted time budget.
    """
    np.random.seed(seed)
    # leave a small safety margin for the harness
    deadline = time.time() + budget_s - 2.0

    # --------------------------------------------------------------
    # Problem size – 20 000 gives a good trade‑off between quality
    #                and per‑iteration speed (FFT size 2²⁶ = 65 536).
    # --------------------------------------------------------------
    n = 20_000

    # --------------------------------------------------------------
    # 1) Catalan start (a very good initial point)
    # --------------------------------------------------------------
    a0 = _catalan_normalised(n)

    # --------------------------------------------------------------
    # 2) Main optimisation phase – exponential‑gradient descent
    # --------------------------------------------------------------
    best_seq, best_val = _expgrad_optimize(a0, n, deadline,
                                           max_iters=180_000,
                                           λ_start=0.5,
                                           λ_max=400.0,
                                           η0=0.25)

    # --------------------------------------------------------------
    # 3) If time remains, launch a few cheap random restarts.
    #    Each restart runs a short optimisation (≈30 000 iterations)
    #    and adopts any improvement.
    # --------------------------------------------------------------
    while time.time() < deadline - 5.0:
        a_rand = np.random.rand(n)
        a_rand /= a_rand.sum()
        a_opt, val = _expgrad_optimize(a_rand, n, deadline,
                                       max_iters=30_000,
                                       λ_start=0.5,
                                       λ_max=400.0,
                                       η0=0.25)
        if val < best_val:
            best_val = val
            best_seq = a_opt.copy()
            print(f"[restart] new best obj {best_val:.9f}")

    # --------------------------------------------------------------
    # 4) Final sanity check (the surrounding harness will call
    #    ``evaluate_sequence`` on the returned list).
    # --------------------------------------------------------------
    final_obj = _objective(best_seq, n)
    print(f"[final] best objective ≈ {final_obj:.9f} with n={n}")

    # Convert to plain Python list as required by the framework
    return best_seq.tolist()
