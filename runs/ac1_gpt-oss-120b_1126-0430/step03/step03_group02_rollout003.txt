def _conv_fft(a,b):
    m = a.size + b.size - 1
    size = 1 << ((m - 1).bit_length())
    A = np.fft.fft(a, n=size)

    return c

"""
Propose a non‑negative step function that minimises

    O = 2·n·max(conv(a,a)) / (sum(a)^2)

The bound is scale‑invariant, so we work on the probability simplex
{a_i ≥ 0, Σ a_i = 1}.  The optimisation proceeds in two stages:

1. **Parameter sweep** – generate a few sensible families
   a_i ∝ (i+α)^{-p} (including the classic 1/√i law) and keep the best
   as a start point.

2. **Smoothed mirror‑descent** – minimise a smooth approximation of the
   max‑convolution.  For a temperature γ the soft‑max
   w_k ∝ exp(γ·(conv_k−max(conv))) approximates the indicator of the
   maximal indices.  The gradient of the smoothed objective can be
   written as a cross‑correlation and is computed with two FFTs per
   iteration.  An exponentiated‑gradient (KL‑mirror) update with a
   linearly decaying learning rate is applied, followed by projection
   onto the simplex.  The schedule slowly raises γ (sharpening the
   approximation) while the learning rate decays, giving a stable
   optimisation that typically beats the previous 1.5205 bound.

The routine respects the 1000 s time limit, prints progress, and finally
calls the supplied ``evaluate_sequence`` for the exact score.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of real vectors a and b using a power‑of‑two FFT."""
    m = a.size + b.size - 1
    size = 1 << ((m - 1).bit_length())          # next power of two
    A = np.fft.fft(a, n=size)
    B = np.fft.fft(b, n=size)
    c = np.fft.ifft(A * B).real[:m]
    return c


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return the bound 2·n·max(conv(a,a)) for a already normalised (Σa=1)."""
    n = a.size
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, gamma: float) -> np.ndarray:
    """
    Gradient of the soft‑max approximation of max(conv(a,a)).
    grad_i = 2 * Σ_j a_j * w_{i+j},
    where w is the soft‑max distribution over the convolution entries.
    """
    n = a.size
    # conv = a * a
    conv = _conv_fft(a, a)                     # length 2n-1
    maxc = conv.max()
    # soft‑max weights (numerically stable)
    exparg = gamma * (conv - maxc)              # ≤ 0
    w = np.exp(exparg)
    w_sum = w.sum()
    if w_sum == 0:
        # should not happen (max entry always yields 1), but guard
        w = np.zeros_like(w)
        w[conv.argmax()] = 1.0
        w_sum = 1.0
    w /= w_sum

    # cross‑correlation a_rev ⋆ w gives the gradient (see analysis)
    a_rev = a[::-1]
    corr = _conv_fft(a_rev, w)                 # length 3n-2
    grad = 2.0 * corr[n - 1:n - 1 + n]         # slice of length n
    return grad


# ----------------------------------------------------------------------
def _optimize(
    a0: np.ndarray,
    max_iters: int,
    lr_start: float,
    lr_end: float,
    gamma_start: float,
    gamma_end: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Smoothed mirror‑descent on the simplex.
    Returns the best vector encountered and its (exact) objective value.
    """
    a = a0.copy()
    n = a.size
    best_a = a.copy()
    best_score = _objective(a)

    # early‑stop patience (in iterations without improvement)
    patience = max_iters // 20
    no_improve = 0

    for t in range(max_iters):
        if time.time() >= deadline:
            break

        # ---------- evaluate current point ----------
        conv = _conv_fft(a, a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            print(f"[opt] iter {t:5d}  new best O={best_score:.6f}")
        else:
            no_improve += 1
            if no_improve >= patience:
                # small random jitter to escape flat regions
                jitter = np.random.randn(n) * 1e-4
                a = best_a + jitter
                a = np.clip(a, 0.0, None)
                a = _project_to_simplex(a)
                no_improve = 0

        # ---------- compute smoothed gradient ----------
        gamma = gamma_start + (gamma_end - gamma_start) * (t / max_iters)
        grad = _smooth_gradient(a, gamma)

        # ---------- learning‑rate schedule ----------
        lr = lr_start + (lr_end - lr_start) * (t / max_iters)

        # ---------- exponentiated‑gradient (KL‑mirror) update ----------
        log_a = np.log(a + 1e-15)    # avoid log(0)
        log_a -= lr * grad
        log_a -= np.max(log_a)      # keep numbers moderate
        a = np.exp(log_a)
        a /= a.sum()                # renormalise onto simplex

    return best_a, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the provided
    ``evaluate_sequence`` bound.  Returns the sequence as a plain list of
    floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ---- Determine working length ------------------------------------
    if 'height_sequence_1' in globals():
        a_prev = np.asarray(height_sequence_1, dtype=float)
        n = a_prev.size
        # Normalise to the simplex
        a_prev = a_prev / a_prev.sum()
    else:
        a_prev = None
        n = 4000                     # default length (FFT‑friendly)

    # ---- Build a small pool of sensible starts -----------------------
    starts = []

    if a_prev is not None:
        starts.append(a_prev)

    # 1/√ law (theoretically good)
    idx = np.arange(1, n + 1, dtype=float)
    a_sqrt = 1.0 / np.sqrt(idx)
    a_sqrt /= a_sqrt.sum()
    starts.append(a_sqrt)

    # a few power‑law variants (p in [0.4,0.8], small shift α)
    for p in (0.4, 0.45, 0.5, 0.55, 0.6, 0.7):
        for alpha in (0.0, 0.5, 1.0):
            i = np.arange(1, n + 1, dtype=float)
            a_pl = 1.0 / np.power(i + alpha, p)
            a_pl /= a_pl.sum()
            starts.append(a_pl)

    # a random point – adds diversity
    a_rand = np.random.rand(n)
    a_rand = _project_to_simplex(a_rand)
    starts.append(a_rand)

    # ---- Pick the best initial point (fast evaluation) ---------------
    best_start = None
    best_start_score = float("inf")
    for i, a in enumerate(starts):
        sc = _objective(a)
        print(f"[init {i}] start O={sc:.6f}")
        if sc < best_start_score:
            best_start_score = sc
            best_start = a.copy()

    if best_start is None:                     # should never happen
        best_start = a_sqrt.copy()

    # ---- Optimisation parameters -------------------------------------
    max_iters = 25000
    lr_start = 0.6
    lr_end = 0.02
    gamma_start = 0.5
    gamma_end = 15.0

    # ---- Run the smoothed mirror‑descent -----------------------------
    print("[opt] starting smoothed mirror‑descent")
    a_opt, score_opt = _optimize(
        best_start,
        max_iters,
        lr_start,
        lr_end,
        gamma_start,
        gamma_end,
        deadline,
    )
    print(f"[result] optimized O={score_opt:.6f}")

    # ---- Fallback in case optimisation did not run -------------------
    if a_opt is None or a_opt.size == 0:
        a_opt = best_start.copy()
        score_opt = _objective(a_opt)

    # ---- Final exact evaluation using the supplied function -----------
    try:
        final_score = evaluate_sequence(a_opt.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:  # pragma: no cover
        print("[final] evaluate_sequence call failed:", e)
        final_score = None

    # store the result globally for possible reuse in later runs
    global height_sequence_1
    height_sequence_1 = a_opt.tolist()

    return a_opt.tolist()
