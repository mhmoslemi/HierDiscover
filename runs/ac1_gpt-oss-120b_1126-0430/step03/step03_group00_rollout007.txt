"""
Propose a non‑negative step‑function (probability vector) that minimises

    O(a) = 2·n·max(conv(a,a)) / (sum(a)²)

The objective is homogeneous, therefore the vector is kept on the
probability simplex (sum = 1, a_i ≥ 0).  The search consists of two
phases and runs on a single core (up to two CPUs are allowed by the
environment).

Phase 1 – Adam on a smooth log‑sum‑exp surrogate
-------------------------------------------------
A smoothed version of the max is
    f_α(a) = (2·n/α)·log Σ_k exp(α·conv_k(a)),
with α slowly increased (geometric schedule from 5 up to 2·10⁴).
The exact gradient of f_α can be written as a correlation,
which we evaluate with a single FFT.  Adam (β₁=0.9, β₂=0.999) with a
decaying stepsize performs projected updates on the simplex.
After every projection the vector is symmetrised
    a ← (a + a[::-1]) / 2,
which never harms optimality (the optimum can be chosen symmetric)
and reduces the effective dimensionality.

Phase 2 – Targeted stochastic refinement
----------------------------------------
From the best candidate of phase 1 we run a Simulated‑Annealing style
search.  In each iteration the sub‑gradient of the current maximal
convolution entry is computed; the index with the largest gradient
is used as a source of probability mass, the index with the smallest
gradient as a destination.  A small fraction of the source mass is
transferred, the result is projected onto the simplex and symmetrised.
Moves are accepted if they improve the exact objective, otherwise with
a Metropolis probability that cools down exponentially.
Every few iterations a pure sub‑gradient step
    a ← a – η·g
(with decreasing η) is performed.
Random restarts (jittered best or fresh draws from Dirichlet) keep the
search global.

Both phases are time‑driven: 60 % of the budget is spent on phase 1,
the rest on phase 2.  The final sequence is returned as a plain
list of floats; a sanity check with the provided ``evaluate_sequence``
is printed.

The algorithm is deliberately simple, fully vectorised and makes
extensive use of FFT‑based convolutions, allowing it to explore
sequences with several thousand entries within the 1000 s limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper 1 – projection onto the probability simplex {x ≥ 0, sum(x)=1}
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        # all entries non‑positive – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper 2 – enforce symmetry (a_i = a_{n-1-i})
def _make_symmetric(a: np.ndarray) -> np.ndarray:
    """Replace a by (a + rev(a))/2 – preserves sum and non‑negativity."""
    return (a + a[::-1]) * 0.5


# ----------------------------------------------------------------------
# Helper 3 – full convolution (length 2·n‑1) via FFT
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return a*a (full convolution) using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Helper 4 – exact objective for a vector already normalised (sum = 1)
def _objective_simplex(a: np.ndarray, n: int, L: int) -> float:
    """Return 2·n·max(conv(a,a)) (a is assumed to sum to 1)."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Helper 5 – sub‑gradient of max(conv) w.r.t. a
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) for the current a."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)   # could be several maximisers
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
# Helper 6 – gradient of the log‑sum‑exp surrogate
def _smooth_gradient(a: np.ndarray, conv: np.ndarray,
                    alpha: float, L: int) -> np.ndarray:
    """
    Compute ∇_a ( (2·n/α)·log Σ_k e^{α·conv_k} ).
    The gradient equals 2·corr(w, a) where w_k ∝ e^{α·conv_k}.
    """
    # soft‑max weights (numerically stable)
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / exp_scaled.sum()

    # correlation w ⋆ a   (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n_len = a.size
    grad = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]
    return grad

        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[phase‑1] iter {it:,} new best O={best_score:.6f}")

        # geometric annealing of α
        frac = it / max_iter_phase1
        alpha = alpha_start * (alpha_end / alpha_start) ** frac

        # smooth gradient of the log‑sum‑exp surrogate
        grad = _smooth_gradient(a, conv, alpha, L)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        # stepsize decays like 1/√t
        lr = 0.5 / math.sqrt(it + 1)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional tiny jitter to escape plateaus
        if np.random.rand() < 0.001:
            a_new += np.random.randn(n) * 1e-4

        # project onto the simplex and enforce symmetry
        a_new = _project_to_simplex(a_new)
        a_new = _make_symmetric(a_new)

        # prune extremely tiny entries every few thousand iterates
        if it % 4000 == 0 and it > 0:
            thresh = 1e-7
            a_new[a_new < thresh] = 0.0
            a_new = _project_to_simplex(a_new)
            a_new = _make_symmetric(a_new)

        a = a_new

    # --------------------------------------------------------------
    # Phase 2 – targeted stochastic hill‑climbing + sub‑gradient refinements
    print(f"[phase‑2] entering local refinement, best O={best_score:.6f}")
    best_seq_local = best_seq.copy()
    best_obj_local = best_score

    temperature = 0.05                     # initial Metropolis temperature
    subgrad_every = 5                     # compute sub‑gradient each few steps
    max_iter_local = 200_000              # bound for a single refinement run
    no_improve_limit = 5000

    while time.time() < deadline:
        # ---- (re)initialise a for the current restart -----------------
        if np.random.rand() < 0.5:
            # jittered version of the current global best
            a = best_seq_local.copy()
            a += np.random.randn(n) * 0.001
            a = np.clip(a, 0.0, None)
            a = _project_to_simplex(a)
            a = _make_symmetric(a)
        else:
            # fresh random point from Dirichlet(1) (i.e. uniform)
            a = np.random.rand(n)
            a = _project_to_simplex(a)
            a = _make_symmetric(a)

        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        it_loc = 0
        no_improve = 0

        while time.time() < deadline and it_loc < max_iter_local:
            # ---- directed mass‑transfer move based on sub‑gradient -----
            if it_loc % subgrad_every == 0:
                # recompute sub‑gradient of the current maximal entry
                conv = _conv_fft(a, L)
                cur_obj = 2.0 * n * conv.max()
                g = _subgradient_max_conv(a, conv)

                src = int(np.argmax(g))                     # source (large gradient)
                tgt = int(np.argmin(g))                     # target (small gradient)

                if src != tgt and a[src] > 1e-14:
                    delta = a[src] * np.random.rand() * 0.10   # up to 10 % of source mass
                    a_new = a.copy()
                    a_new[src] -= delta
                    a_new[tgt] += delta
                    a_new = _project_to_simplex(a_new)
                    a_new = _make_symmetric(a_new)

                    conv_new = _conv_fft(a_new, L)
                    new_obj = 2.0 * n * conv_new.max()

                    # Metropolis acceptance
                    if new_obj < cur_obj or np.random.rand() < math.exp((cur_obj - new_obj) / max(temperature, 1e-12)):
                        a = a_new
                        conv = conv_new
                        cur_obj = new_obj
                        no_improve = 0
                        if cur_obj < best_obj_local:
                            best_obj_local = cur_obj
                            best_seq_local = a.copy()
                            print(f"[{time.time():.1f}] new global best O={best_obj_local:.6f}")
                    else:
                        no_improve += 1
                else:
                    # fallback random tiny move
                    i, j = np.random.randint(n, size=2)
                    if i != j and a[i] > 1e-14:
                        delta = a[i] * np.random.rand() * 0.05
                        a_new = a.copy()
                        a_new[i] -= delta
                        a_new[j] += delta
                        a_new = _project_to_simplex(a_new)
                        a_new = _make_symmetric(a_new)

                        conv_new = _conv_fft(a_new, L)
                        new_obj = 2.0 * n * conv_new.max()
                        if new_obj < cur_obj or np.random.rand() < math.exp((cur_obj - new_obj) / max(temperature, 1e-12)):
                            a = a_new
                            conv = conv_new
                            cur_obj = new_obj
                            no_improve = 0
                            if cur_obj < best_obj_local:
                                best_obj_local = cur_obj
                                best_seq_local = a.copy()
                                print(f"[{time.time():.1f}] new global best O={best_obj_local:.6f}")
                        else:
                            no_improve += 1
                    else:
                        no_improve += 1

                # cool down the Metropolis temperature slowly
                temperature *= 0.9998

                # occasional pure sub‑gradient step (decreasing step size)
                if it_loc % (subgrad_every * 5) == 0:
                    eta = 0.07 * (1.0 - it_loc / max_iter_local)
                    if eta > 0:
                        a = a - eta * g
                        a = _project_to_simplex(a)
                        a = _make_symmetric(a)

                # break if we have been stuck for a while
                if no_improve >= no_improve_limit:
                    break

            else:
                # occasional random jitter (cheap cheap move)
                if np.random.rand() < 0.01:
                    i, j = np.random.randint(n, size=2)
                    if i != j and a[i] > 1e-14:
                        delta = a[i] * np.random.rand() * 0.03
                        a_new = a.copy()
                        a_new[i] -= delta
                        a_new[j] += delta
                        a_new = _project_to_simplex(a_new)
                        a_new = _make_symmetric(a_new)

                        conv_new = _conv_fft(a_new, L)
                        new_obj = 2.0 * n * conv_new.max()
                        if new_obj < cur_obj or np.random.rand() < math.exp((cur_obj - new_obj) / max(temperature, 1e-12)):
                            a = a_new
                            conv = conv_new
                            cur_obj = new_obj
                            no_improve = 0
                            if cur_obj < best_obj_local:
                                best_obj_local = cur_obj
                                best_seq_local = a.copy()
                                print(f"[{time.time():.1f}] new global best O={best_obj_local:.6f}")
                        else:
                            no_improve += 1
                else:
                    no_improve += 1

            it_loc += 1

        # small pause before the next restart – loop will exit when time exhausted
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Final sanity check with the provided evaluator
    try:
        final_score = evaluate_sequence(best_seq_local.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:
        print("[final] evaluate_sequence failed:", e)

    # Return the best vector found (as a plain Python list)
    return best_seq_local.tolist()
