def upsample(a_small, n_new):
    # a_small length n_small

    indices_small = np.linspace(0, 1, len(a_small), endpoint=False) # positions for small distribution
    indices_new = np.linspace(0, 1, n_new, endpoint=False)

    a_new = a_new / a_new.sum()

"""
Propose a step‑function (non‑negative vector on the probability simplex) that
minimises  O = 2·n·max(conv(a,a)) .  
The optimiser works in three stages for each problem size:
  1. Adam on a smooth log‑sum‑exp surrogate of the max‑convolution.
  2. Deterministic sub‑gradient descent on the true max‑convolution.
  3. Simulated‑annealing hill‑climbing with random mass transfers.
The procedure is repeated for a set of increasing lengths (1024, 2048, 4096,
8192).  An arcsine‑like “U‑shape’’ distribution is used as a good
initialisation; the best result found over all lengths (and the time budget)
is returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection ----------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:                       # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full self‑convolution ---------------------------------------
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return the full linear convolution a*a (length 2·n‑1) using FFT length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Objective for a simplex‑normalised vector -----------------------------
def _objective_simplex(a: np.ndarray, n: int, L: int) -> float:
    """Exact objective: 2·n·max(conv(a,a)) (a sums to 1)."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Sub‑gradient of the maximal convolution entry -------------------------
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. a (a ≥ 0, Σa = 1)."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)          # possibly several maxima
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]                # each pair contributes symmetrically
    g /= max(1, len(ks))                         # average if several maximisers
    return g


# ----------------------------------------------------------------------
# Gradient of the smooth log‑sum‑exp surrogate --------------------------
def _smooth_gradient(a: np.ndarray, conv: np.ndarray, alpha: float, L: int) -> np.ndarray:
    """
    Compute ∇_a ( (2·n/α)·log∑_k e^{α·conv_k} ).
    Result = 2·corr(w, a) where w_k ∝ e^{α·conv_k}.
    """
    # soft‑max weights (stable)
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / exp_scaled.sum()

    # correlation w ⋆ a   (central n entries)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr = np.fft.irfft(fft_w * fft_a_rev, n=L)   # length L ≥ 3·n‑2
    n = a.size
    grad = 2.0 * corr[n - 1 : n - 1 + n]          # extract the central block
    return grad


# ----------------------------------------------------------------------
# Upsampling from a smaller length to a larger one ----------------------
def _upsample(a_small: np.ndarray, n_new: int) -> np.ndarray:
    """Linear interpolation of a_small to length n_new and simplex projection."""
    n_old = a_small.size
    if n_old == n_new:
        return a_small.copy()
    x_old = np.linspace(0.0, 1.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n_new, endpoint=False)
    a_new = np.interp(x_new, x_old, a_small)
    a_new = a_new / a_new.sum()
    return a_new
