def _fft_correlation(p, a):
    # compute correlation of p (len 2n-1) and a (len n)
    size = 1 << ((len(p)+len(a)-1).bit_length())

    return conv[start:end]

"""
Propose a step‑function (non‑negative sequence) with a low self‑convolution peak.

Algorithm
---------
1. Choose a problem size `n` (here 20 000 – a good trade‑off between quality
   and speed).
2. Build a small portfolio of simple, analytically‑motivated candidates:
   * Catalan distribution (used in the reference solution)
   * Uniform distribution
   * Power‑law families a_i ∝ (i+1)^(-γ)
   * Symmetric beta‑type families a_i ∝ (i+1)^{α-1}·(n-i)^{α-1}
   * Convex mixtures of the above.
   The best of these (according to the exact objective 2·n·max(a*a))
   becomes the starting point.
3. Refine the start with a projected heavy‑ball / mirror‑descent scheme that
   uses **all** indices whose convolution value is (almost) maximal.
   The gradient of the max‑convolution is the correlation of the active‑mask
   with the current vector; this can be computed by a single FFT‑based
   convolution.  After each step we project back onto the probability simplex.
   Momentum (`beta = 0.9`) and a diminishing step‑size  η_t = η₀/√t
   accelerate convergence.
4. When the time budget permits we launch cheap random restarts
   (Dirichlet‑type starts) and run a short optimisation from each.
5. The best vector discovered is returned as a plain Python list.
   The harness will invoke `evaluate_sequence` on it.

The whole routine respects the 1000 s budget, prints occasional progress
messages and never returns an empty or invalid sequence.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers -----------------------------------------------------------
# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)-1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[: 2 * n - 1]


def _fft_correlation(p: np.ndarray, a_rev: np.ndarray, size: int) -> np.ndarray:
    """
    Correlation r[j] = Σ_k p_{j+k}·a_k   (0 ≤ j < len(a_rev))
    implemented as convolution of p with the reversed vector a_rev.
    Returns the slice of length len(a_rev) that corresponds to the correlation.
    """
    fp = np.fft.rfft(p, size)
    fa = np.fft.rfft(a_rev, size)
    full = np.fft.irfft(fp * fa, size)
    start = a_rev.shape[0] - 1          # = n-1
    return full[start:start + a_rev.shape[0]]


# ----------------------------------------------------------------------
# Simple feasibility utilities ------------------------------------------
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Candidate generators --------------------------------------------------
# ----------------------------------------------------------------------
def _catalan_normalised(n: int) -> np.ndarray:
    """Catalan‑type coefficients a_i = C(2i,i)/4**i, normalised to Σa=1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    a /= a.sum()
    return a


def _uniform_normalised(n: int) -> np.ndarray:
    """Uniform distribution on the simplex."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _powerlaw_normalised(n: int, gamma: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-γ) , Σa=1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-gamma)
    a /= a.sum()
    return a


def _beta_normalised(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta‑type shape a_i ∝ (i+1)^{α-1}·(n-i)^{α-1}, Σa=1."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1)) * ((n - i + 1) ** (alpha - 1))
    a /= a.sum()
    return a


def _mix_sequences(a: np.ndarray, b: np.ndarray, lam: float) -> np.ndarray:
    """Convex mixture lam·a + (1‑lam)·b.  Both a and b are already normalised."""
    return lam * a + (1.0 - lam) * b


# ----------------------------------------------------------------------
# Objective ------------------------------------------------------------
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value returned by evaluate_sequence for a normalised to Σa=1."""
    n = a.shape[0]
    return float(2.0 * n * np.max(_fft_convolve(a)))


# ----------------------------------------------------------------------
# Core optimiser – projected heavy‑ball with an active‑set gradient -----
# ----------------------------------------------------------------------
def _optimize_active(a0: np.ndarray,
                     deadline: float,
                     max_iters: int = 500_000,
                     verbose: bool = True) -> tuple[np.ndarray, float]:
    """
    Refine a starting vector a0 (already Σ=1) by projected heavy‑ball
    descent using the gradient of the *active* part of the self‑convolution.
    Returns the best vector ever seen and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]

    # FFT sizes (once, reused)
    size_conv = 1 << ((2 * n - 1).bit_length())
    size_corr = 1 << ((3 * n - 2).bit_length())

    # best bookkeeping
    conv = _fft_convolve(a)
    best_val = float(2.0 * n * conv.max())
    best_seq = a.copy()

    # heavy‑ball momentum
    beta = 0.9
    momentum = np.zeros_like(a)

    t = 0
    while time.time() < deadline and t < max_iters:
        # --- self‑convolution -------------------------------------------------
        fa = np.fft.rfft(a, size_conv)
        conv = np.fft.irfft(fa * fa, size_conv)[: 2 * n - 1]
        max_c = conv.max()
        cur_val = float(2.0 * n * max_c)

        # record improvement
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if verbose:
                print(f"[improve] iter {t:06d}  obj {best_val:.9f}")

        # --- active set (indices within a tiny relative tolerance) ------------
        eps = max_c * 1e-8
        active_mask = (conv >= max_c - eps).astype(np.float64)

        # --- gradient = 2 * correlation(active_mask, a) ----------------------
        a_rev = a[::-1]
        grad = 2.0 * _fft_correlation(active_mask, a_rev, size_corr)

        # --- heavy‑ball update -----------------------------------------------
        momentum = beta * momentum + (1.0 - beta) * grad

        # step‑size (diminishing)
        lr = 0.8 / np.sqrt(t + 1)

        # additive step + projection onto the simplex
        a = a - lr * momentum
        a = np.maximum(a, 0.0)           # non‑negative
        a = _project_simplex(a)           # enforce Σa = 1

        # occasional stochastic perturbation to escape plateaus
        if t % 5000 == 0 and t > 0:
            pert = np.random.normal(scale=0.005, size=n)
            a = np.maximum(a + pert, 0.0)
            a = _project_simplex(a)

        t += 1

    return best_seq, best_val

            old = np.asarray(height_sequence_1, dtype=np.float64)
            if old.size >= 2:
                # normalise to sum‑1 (the optimiser works on the simplex)
                old = old.copy()
                if old.sum() > 0:
                    old /= old.sum()
                else:
                    old = _uniform_normalised(n)
                # truncate / pad to the chosen n
                if old.size > n:
                    old = old[:n]
                elif old.size < n:
                    extra = np.full(n - old.size, 0.0)
                    old = np.concatenate([old, extra])
                candidates.append(old)
        except Exception:
            pass

    # --------------------------------------------------------------
    # 3) evaluate all candidates and pick the best start
    # --------------------------------------------------------------
    best_start = None
    best_val = float("inf")
    for idx, cand in enumerate(candidates):
        val = _objective(cand)
        if val < best_val:
            best_val = val
            best_start = cand.copy()
        if idx < 10:      # a few prints for early feedback
            print(f"[candidate] {idx:02d}  obj {val:.9f}")

    print(f"[selected] initial objective ≈ {best_val:.9f}")

    # --------------------------------------------------------------
    # 4) main optimisation from the best start
    # --------------------------------------------------------------
    best_seq, best_val = _optimize_active(best_start, deadline,
                                          max_iters=300_000,
                                          verbose=True)

    # --------------------------------------------------------------
    # 5) cheap random restarts if time permits
    # --------------------------------------------------------------
    while time.time() < deadline - 5.0:
        a_rand = np.random.rand(n)
        a_rand /= a_rand.sum()
        seq_opt, val = _optimize_active(a_rand, deadline,
                                        max_iters=60_000,
                                        verbose=False)
        if val < best_val:
            best_val = val
            best_seq = seq_opt.copy()
            print(f"[restart] new best obj {best_val:.9f}")

    # --------------------------------------------------------------
    # 6) final sanity check and return
    # --------------------------------------------------------------
    final_obj = _objective(best_seq)
    print(f"[final] best objective ≈ {final_obj:.9f} with n={n}")

    # `evaluate_sequence` will re‑normalise the sum, but we keep the sum = 1
    # to make the objective directly comparable.
    return best_seq.tolist()
