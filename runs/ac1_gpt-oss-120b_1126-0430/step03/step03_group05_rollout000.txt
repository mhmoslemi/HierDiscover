#!/usr/bin/env python
"""
A smoothed‐gradient optimisation for the non‑negative step function problem.

The objective (for a simplex‑normalised vector a, Σa=1) is
    F(a) = 2·n·max_k (a*a)[k] .
We minimise a smooth surrogate:
    F_β(a) = (1/β)·log Σ_k exp(β·(a*a)[k])   (log‑sum‑exp of the
    autocorrelation).  As β→∞ this approaches the true max.

Algorithm
---------
* Initialise a with a normalised power‑law (the best exponent found
  previously) or with the provided `height_sequence_1`.
* Iterate:
    - Compute the full autocorrelation c = a*a via an FFT.
    - Form soft‑max weights w_k = exp(β·c_k)/Σ exp(β·c_k).
    - The gradient of F_β is   g = 2·(a_rev * w)[n‑1 : 2n‑1]   where
      a_rev is a reversed copy of a.  Both convolutions are obtained
      by FFTs of length ≥3·n.
    - Perform a heavy‑ball (momentum) step on the simplex:
          v ← μ·v – η·g
          a ← Π_S (a + v)                (Π_S = projection onto the
                                          probability simplex)
      The step size η decays like η₀/√t.
    - Gradually increase β (β←β·γ) so the optimisation starts
      smooth and ends with a tight approximation of the max.
    - Keep the best vector seen (according to the cheap proxy
      2·n·max(c)).
    - If no improvement appears for a long time, add a small random
      perturbation and reset the momentum to escape plateaus.
* After the time budget the best vector is evaluated with the exact
  `evaluate_sequence` routine and returned.

The method explores the landscape more broadly than a pure
sub‑gradient descent (the code used previously) and typically finds
a lower value within the same time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (must be top‑level, no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (components ≥0, sum=1)."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all entries non‑positive
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float) -> np.ndarray:
    """Return a normalised power‑law vector a_i ∝ (i+1)^{-p} (Σa_i = 1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = i ** (-p)
    seq /= seq.sum()
    return seq


def _conv_full_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full convolution a*a using FFT of length L (L ≥ 2·len(a)-1)."""
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * a.size - 1]
    return conv


def _conv_rev_weights_fft(a_rev: np.ndarray, w: np.ndarray, L: int) -> np.ndarray:
    """
    Convolution of a_rev (length n) with weight vector w (length 2n-1)
    using FFT of length L (L ≥ len(a_rev) + len(w) - 1).
    Returns the full convolution of length len(a_rev)+len(w)-1.
    """
    fa_rev = np.fft.rfft(a_rev, n=L)
    fw = np.fft.rfft(w, n=L)
    conv = np.fft.irfft(fa_rev * fw, n=L)[: len(a_rev) + len(w) - 1]
    return conv
