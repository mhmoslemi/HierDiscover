"""
Candidate generator for the step‑function minimisation task.

The algorithm works directly on a probability vector a (∑a=1, a_i≥0) and
minimises the true objective

    O(a) = 2·n·max_k (a * a)[k]            (n = len(a))

by a smooth approximation of the max‑convolution term.
For a temperature α the smoothed objective is

    F_α(a) = (1/α)·log∑_k exp(α·(a*a)[k])

whose gradient is available in closed form:
    ∇F_α(a) = 2·(w ▷ a)   where w_k ∝ exp(α·(a*a)[k])
and “▷” is the linear correlation (cross‑correlation) of w with a.
The gradient is computed with FFTs in O(n log n).

The optimisation uses Adam (adaptive‑momentum) with a decreasing
learning‑rate schedule and an annealing of α from a very smooth regime
(α≈0.1) to a sharp regime (α≈5 000).  After each Adam step the vector is
projected onto the probability simplex.  Several restarts (Catalan
distribution, the previously best known sequence, and random Dirichlet
vectors) are performed while the time budget allows.

The best sequence encountered (as a plain Python list) is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Catalan‑type coefficients, normalised to unit L1‑norm
# ----------------------------------------------------------------------
def _catalan_normalised(n: int) -> np.ndarray:
    """First n Catalan numbers C_i = (2i choose i)/4**i, normalised to sum‑1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Core optimiser – smoothed max with Adam & simplex projection
# ----------------------------------------------------------------------
def _optimize_smooth_max(
    a0: np.ndarray,
    deadline: float,
    max_iters: int,
    alpha_start: float = 0.1,
    alpha_end: float = 5000.0,
    lr: float = 0.02,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the smoothed objective.
    Returns the best vector found and its *true* objective value
    2·n·max(conv).
    """
    a = a0.copy()
    n = a.shape[0]

    # length needed for a full linear convolution a*a (2n‑1)
    # and for the correlation a ▷ w (size ≥ n + (2n‑1) – 1 = 3n‑2)
    size = 1 << ((3 * n - 2).bit_length())

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_vec = a.copy()
    # true objective for the current best vector
    conv_full = np.convolve(a, a)
    best_val = float(2.0 * n * conv_full.max())

    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1

        # ---- anneal the temperature α ----
        # exponential schedule: α = α_start·(α_end/α_start)^{t/max_iters}
        factor = (alpha_end / alpha_start) ** (t / max_iters)
        alpha = alpha_start * factor

        # ---- FFT of the current vector ----
        A = np.fft.rfft(a, size)

        # ---- linear convolution a*a (full) ----
        conv = np.fft.irfft(A * A, size)[: 2 * n - 1]

        # true objective (max‑convolution term)
        cur_max = conv.max()
        cur_obj = 2.0 * n * cur_max
        if cur_obj < best_val:
            best_val = cur_obj
            best_vec = a.copy()
            print(f"[improve] iter {t:07d}  obj {best_val:.9f}")

        # ---- smoothed max weights w_k ∝ exp(α·conv_k) ----
        # shift by max to avoid overflow
        shift = cur_max
        w = np.exp(alpha * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:                     # safety fallback
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of the smoothed max (∇F_α) ----
        # correlation a ▷ w = Σ_k w_k a_{k‑i}
        W = np.fft.rfft(w, size)
        corr = np.fft.irfft(A * np.conj(W), size)[:n]
        grad = 2.0 * corr                     # ∇F_α(a)

        # ---- Adam update (learning rate is kept constant, we rely on Adam's
        #        adaptive scaling) ----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        # enforce feasibility
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        # occasional tiny random jitter to escape shallow plateaus
        if t % 20000 == 0:
            jitter = np.random.normal(scale=0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_simplex(a)

    return best_vec, best_val


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine runs for at most `budget_s` seconds and returns a plain
    Python list of heights.
    """
    np.random.seed(seed)
    # keep a small safety margin
    deadline = time.time() + budget_s - 2.0

    # --------------------------------------------------------------
    # Choose a problem size.
    # 25000 gives a good trade‑off between quality and speed.
    # --------------------------------------------------------------
    n = 25000

    # --------------------------------------------------------------
    # Assemble a few sensible starting points.
    # --------------------------------------------------------------
    starts = []

    # 1) Previously discovered construction, if available.
    prev = globals().get("height_sequence_1")
    if isinstance(prev, list) and len(prev) > 0:
        a = np.array(prev, dtype=np.float64)
        if a.sum() == 0.0:
            a = np.ones_like(a)
        a = a / a.sum()
        # resize/interpolate if its length differs from n
        if a.shape[0] != n:
            x_old = np.arange(a.shape[0])
            x_new = np.linspace(0, a.shape[0] - 1, n)
            a = np.interp(x_new, x_old, a)
        a = a / a.sum()
        starts.append(a)

    # 2) Catalan distribution (theoretical attractor)
    starts.append(_catalan_normalised(n))

    # 3) A random Dirichlet‑type start (uniform on the simplex)
    a_rand = np.random.rand(n)
    a_rand = a_rand / a_rand.sum()
    starts.append(a_rand)

    best_seq = None
    best_obj = float("inf")
    start_idx = 0

    # --------------------------------------------------------------
    # Main optimisation loop – keep launching restarts while time permits.
    # --------------------------------------------------------------
    while time.time() < deadline - 5.0:
        # pick a start (reuse the list cyclically, then random)
        if start_idx < len(starts):
            a0 = starts[start_idx]
        else:
            a0 = np.random.rand(n)
            a0 = a0 / a0.sum()

        # allocate a reasonable iteration budget based on remaining time
        remaining = deadline - time.time()
        # rough estimate: ~2000 iterations per second for the chosen n
        max_iters = int(remaining * 2000)
        max_iters = max(5_000, min(max_iters, 400_000))

        seq, val = _optimize_smooth_max(
            a0,
            deadline,
            max_iters,
            alpha_start=0.1,
            alpha_end=5000.0,
            lr=0.02,
        )

        if val < best_obj:
            best_obj = val
            best_seq = seq.copy()
            print(f"[global] new best objective {best_obj:.9f} (start {start_idx})")
            # early exit if we have already beaten the target
            if best_obj < 1.503:
                break

        start_idx += 1

    # --------------------------------------------------------------
    # Final sanity check – report the true objective as seen by the
    # evaluation function.
    # --------------------------------------------------------------
    if best_seq is None:
        # fallback – Catalan if something went wrong
        best_seq = _catalan_normalised(n)

    # The supplied `evaluate_sequence` performs the same scaling as our
    # objective (sum is forced to 1, values are tiny, so no clipping occurs).
    try:
        final_score = evaluate_sequence(best_seq.tolist())
    except Exception:          # the environment always provides it, but be safe
        final_score = float("inf")
    print(f"[final] evaluate_sequence = {final_score:.9f}  (n={n})")

    return best_seq.tolist()
