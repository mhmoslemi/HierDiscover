"""
Improved search for a non‑negative step‑function that minimises

    O = 2·n·max(conv(a,a))   with   sum(a)=1 .

The algorithm is a two‑phase projected exponential‑gradient descent:

* Phase 1 (≈80 % of the time) uses a smooth approximation of the max
  (log‑sum‑exp) with a temperature β that grows exponentially.
  The gradient of the smooth max is obtained by a weighted correlation,
  computed efficiently with FFTs.

* Phase 2 (last ≈20 %) switches to a true sub‑gradient of the max,
  i.e. the gradient of the single largest convolution entry.  This
  provides a sharp final refinement.

Updates are performed multiplicatively (exponentiated‑gradient /
mirror‑descent on the simplex), which avoids the costly Euclidean
projection each iteration.  A small momentum term accelerates convergence,
and occasional jitter/restarts prevent stagnation.

The routine respects a 1000 s budget, prints progress, and returns the
best sequence found as a plain list of floats.
"""

import time
import math
import numpy as np

# -------------------------------------------------------------
# Helper: next power of two (used for FFT sizes)
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# -------------------------------------------------------------
# Full self‑convolution using real FFTs
def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Return the full convolution a*a (length 2·n‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv


# -------------------------------------------------------------
# Correlation needed for the gradient of the smooth max
def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute g_i = Σ_t w[i+t]·a[t]   for i=0,…,n‑1.
    Implements the correlation via FFT.
    """
    n = a.shape[0]
    m = w.shape[0]               # = 2·n‑1
    L = _next_pow2(m + n - 1)    # length for linear convolution
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


# -------------------------------------------------------------
# Exact objective (used for bookkeeping)
def _objective(a: np.ndarray) -> float:
    """Exact O = 2·n·max(conv(a,a)) when sum(a)=1."""
    n = a.size
    conv = _conv_self_fft(a)
    return float(2.0 * n * conv.max())

    """
    Search for a step‑function with minimal bound.
    Returns the best sequence as a list of floats.
    """

    deadline = start_time + budget_s - 0.2         # small safety margin

    # -----------------------------------------------------------------
    # Initial sequence (use the previously found one if available)
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=float)
        a = np.clip(a, 0.0, None)
        if a.sum() == 0.0:
            a = np.random.rand(a.size)
        a = a / a.sum()
        n = a.size
    else:
        n = 4600                                   # a reasonably large length
        a = np.random.rand(n)
        a = a / a.sum()

    best_seq = a.copy()
    best_score = _objective(best_seq)
    print(f"[init] n={n}, O={best_score:.6f}")

    # -----------------------------------------------------------------
    # Hyper‑parameters
    beta_start = 2.0
    beta_end = 8000.0               # final temperature (exponential growth)
    lr_start = 0.5                  # initial learning‑rate
    lr_end = 0.005                  # final learning‑rate (exponential decay)
    momentum = 0.9                  # exponential moving average of the gradient
    max_no_imp_iters = 40000        # after this many iterations re‑initialise

    m = np.zeros_like(a)            # momentum buffer
    it_without_improvement = 0
    iter_cnt = 0

    # -----------------------------------------------------------------
    while time.time() < deadline:
        iter_cnt += 1

        # -------------------------------------------------------------

        M = conv.max()
        cur_score = 2.0 * n * M
        if cur_score < best_score - 1e-12:

            best_seq = a.copy()
            it_without_improvement = 0
            print(f"[iter {iter_cnt:07d}] new best O={best_score:.6f}")
        else:
            it_without_improvement += 1

        # -------------------------------------------------------------
        # 2) annealing / step‑size schedules (based on elapsed time)
        elapsed = time.time() - start_time
        fraction = min(elapsed / budget_s, 1.0)

        # exponential schedules (more stable than linear in this setting)
        beta = beta_start * (beta_end / beta_start) ** fraction
        lr = lr_start * (lr_end / lr_start) ** fraction

        if fraction < 0.80:                                 # Phase 1: smooth max
