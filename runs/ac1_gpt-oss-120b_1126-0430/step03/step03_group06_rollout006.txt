moves in a smooth landscape and finally focuses on the true maximum.

A Catalan‑type distribution provides a good deterministic start.  After the
main optimisation a few cheap random restarts are performed if time permits.
The best vector found (in terms of the objective) is returned as a plain
Python list.

The routine respects the 1000 s time budget, prints progress and the final
objective, and never produces invalid (negative or NaN) entries.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: next power of two
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
# Linear convolution using a real FFT (full mode)
# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * a (real, symmetric) using FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Linear cross‑correlation a ⋆ w (real, full mode) via FFT.
# Returns the first len(a) entries, i.e. (a ⋆ w)[i] = Σ_j a[j] w[j+i].
# ----------------------------------------------------------------------
def _fft_corr(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    m = w.shape[0]
    size = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w, size)
    corr = np.fft.irfft(fa * np.conj(fw), size)
    # Only the first n lags are needed (i = 0 … n‑1)
    return corr[:n]


# ----------------------------------------------------------------------
# Euclidean projection of a vector onto the probability simplex
# (non‑negative entries summing to 1).
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x ≥ 0, Σx = 1} (Euclidean projection)."""
    n = v.size
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)  # 1‑based index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # Numerical safety – fall back to uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Catalan‑type distribution (a_i = C(2i,i)/4**i) normalised to sum = 1.
# ----------------------------------------------------------------------
def _catalan_normalised(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)   # recurrence for Catalan numbers
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Exact objective for a normalised vector (sum = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    conv = _fft_convolve(a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Adam‑style optimiser with a smoothed max (soft‑max) surrogate.
# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
    lam_start: float = 10.0,
    lam_end: float = 100.0,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    base_lr: float = 0.2,
    verbose: bool = True,
) -> tuple[np.ndarray, float]:
    """
    Optimise a normalised vector a to reduce max(a*a) using Adam.
    Returns the best vector seen and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_val = _objective(best_a)

    # Adam moment estimates
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # pre‑compute λ schedule (linear increase)
    if max_iters > 1:
        lam_step = (lam_end - lam_start) / (max_iters - 1)
    else:
        lam_step = 0.0

    for t in range(max_iters):
        # time check
        if time.time() >= deadline:
            break

        # --- convolution and objective ---
        conv = _fft_convolve(a)
        max_conv = conv.max()
        cur_val = 2.0 * n * max_conv
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            if verbose:
                print(f"[improve] iter {t:06d}  obj {best_val:.9f}")

        # --- smooth max weights (soft‑max) ---
        lam = lam_start + lam_step * t
        # stability: subtract max before exponentiation
        w = np.exp(lam * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0.0 or np.isnan(w_sum):
            # fallback to hard max (single active index)
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum

        # --- gradient of the smoothed objective (cross‑correlation) ---
        grad = 2.0 * _fft_corr(a, w)   # shape (n,)

        # --- Adam update ---
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias‑corrected moments
        m_hat = m / (1.0 - beta1 ** (t + 1))
        v_hat = v / (1.0 - beta2 ** (t + 1))

        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        # keep on simplex
        a = _project_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Return a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` within the allotted time budget.

    The routine uses a smoothed‑max Adam optimiser starting from a
    Catalan‑type distribution and performs a few cheap random restarts
    if time permits.
    """
    np.random.seed(seed)
    # leave a small safety margin for the harness
    deadline = time.time() + budget_s - 2.0

    # ------------------------------------------------------------------
    # 1) Choose the problem size based on the time budget.
    # ------------------------------------------------------------------
    # For the full budget we can afford a fairly large n; for very short
    # budgets we fall back to a smaller problem.
    if budget_s < 200:
        n = 8_000
    elif budget_s < 400:
        n = 15_000
    else:
        n = 30_000

    # ------------------------------------------------------------------
    # 2) Initial candidate – Catalan distribution, normalised to sum‑1.
    # ------------------------------------------------------------------
    a0 = _catalan_normalised(n)

    # ------------------------------------------------------------------
    # 3) Main optimisation loop (Adam with smoothed max).
    # ------------------------------------------------------------------
    best_seq, best_val = _adam_optimize(
        a0,
        deadline,
        max_iters=200_000,
        lam_start=10.0,
        lam_end=100.0,
        verbose=True,
    )

    # ------------------------------------------------------------------
    # 4) Optional random restarts if time remains.
    # ------------------------------------------------------------------
    restart_counter = 0
    while time.time() < deadline - 2.0 and restart_counter < 3:
        # random Dirichlet‑type start on the simplex
        a_rand = np.random.rand(n)
        a_rand /= a_rand.sum()
        a_opt, val = _adam_optimize(
            a_rand,
            deadline,
            max_iters=50_000,
            lam_start=15.0,
            lam_end=120.0,
            verbose=False,
        )
        if val < best_val:
            best_val = val
            best_seq = a_opt.copy()
            print(f"[restart {restart_counter}] new best obj {best_val:.9f}")
        restart_counter += 1

    # ------------------------------------------------------------------
    # 5) Final sanity print; the harness will call ``evaluate_sequence``.
    # ------------------------------------------------------------------
    final_obj = _objective(best_seq)
    print(f"[final] best objective ≈ {final_obj:.9f} with n={n}")

    # Return as plain Python list (evaluate_sequence will clip/normalise itself)
    return best_seq.tolist()
