def _subgradient_multi(a, conv, topk):
    n = len(a)
    # get indices of topk entries
    if topk >= len(conv):
        idxs = np.arange(len(conv))
    else:
        idxs = np.argpartition(-conv, topk)[:topk]

    vals = conv[idxs]
    weights = vals / vals.sum()

        idx_range = np.arange(i_start, i_end+1)
        # a[k-idx] corresponds
        grad[idx_range] += 2 * w * a[k - idx_range]
    return grad

"""
Propose a step‑function (non‑negative vector) that approximately solves

    minimise   2·n·max_k (a * a)[k]      subject to   a_i ≥ 0 , Σ a_i = 1

The implementation improves on the previous sub‑gradient method by
 * using several of the largest convolution entries (top‑k) to build a
   smoother sub‑gradient,
 * employing Adam‑style adaptive step sizes,
 * adding occasional random perturbations to escape plateaus,
 * and performing a multi‑stage up‑sampling so that a relatively small
   optimisation problem is solved first (n≈20 000) and then refined on
   a larger grid (up to n≈80 000) while staying well within the 1000 s budget.
All updates keep the vector on the probability simplex.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based full linear convolution (length = 2·n‑1)
# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray) -> np.ndarray:
    """Return a*a (full linear convolution) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Projection onto the probability simplex {x ≥ 0, Σx = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Catalan (ballistic) distribution, normalised to unit ℓ¹‑norm
# ----------------------------------------------------------------------
def _catalan_normalised(n: int) -> np.ndarray:
    """First n Catalan‑type coefficients, Σ = 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Objective (exact value returned by evaluate_sequence when Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    return float(2.0 * n * np.max(_fft_convolve(a)))


# ----------------------------------------------------------------------
# Compute a multi‑active sub‑gradient (top‑k entries of the convolution)
# ----------------------------------------------------------------------
def _multi_subgradient(a: np.ndarray, conv: np.ndarray, topk: int) -> np.ndarray:
    """
    Return the (weighted) sub‑gradient of max(conv) using the top‑k entries.
    Weights are proportional to the corresponding convolution values.
    """
    n = a.shape[0]
    L = conv.size
    # Select the indices of the largest entries
    if topk >= L:
        idxs = np.arange(L)
    else:
        idxs = np.argpartition(-conv, topk)[:topk]
    vals = conv[idxs]
    # Guard against a pathological zero‑vector (should never happen)
    if vals.sum() <= 0:
        weights = np.full_like(vals, 1.0 / vals.size)
    else:
        weights = vals / vals.sum()

    grad = np.zeros_like(a)
    for w, k in zip(weights, idxs):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_range = np.arange(i_start, i_end + 1)
        # Contribution of the active quadratic form J_k a
        grad[idx_range] += 2.0 * w * a[k - idx_range]
    return grad


# ----------------------------------------------------------------------
# Adam‑style projected optimisation with multi‑active sub‑gradient
# ----------------------------------------------------------------------
def _optimize_multi(a0: np.ndarray,
                    deadline: float,
                    max_iters: int = 500_000,
                    topk: int = 5,
                    verbose: bool = True) -> tuple[np.ndarray, float]:
    """
    Optimise a starting vector a0 (already Σa = 1) by Adam with
    a multi‑active sub‑gradient.  The routine stops at the given deadline
    (or after max_iters) and returns the best vector seen.
    """
    a = a0.copy()
    n = a.shape[0]

    best = a.copy()
    best_val = _objective(best)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)   # 1st moment
    v = np.zeros_like(a)   # 2nd moment

    t = 1
    while time.time() < deadline and t <= max_iters:
        conv = _fft_convolve(a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best = a.copy()
            if verbose:
                print(f"[improve] iter {t:06d}  obj {best_val:.9f}")

        # sub‑gradient built from the top‑k convolution entries
        grad = _multi_subgradient(a, conv, topk)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias‑correction
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # learning‑rate schedule (slow decay)
        lr = 0.1 / np.sqrt(t)

        # Adam step
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # keep the vector feasible
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        # occasional jitter to shake out of plateaus
        if t % 5000 == 0:
            jitter = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_simplex(a)

        t += 1

    return best, best_val


# ----------------------------------------------------------------------
# Linear up‑sampling (piecewise‑linear interpolation) preserving Σ = 1
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate a to length new_n and renormalise to sum‑one."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    a_new /= a_new.sum()
    return a_new
