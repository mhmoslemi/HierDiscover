def _fft_convolve_generic(x, y):
    n = len(x)
    m = len(y)
    size = 1 << ((n + m - 1).bit_length())
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)
    return conv[: n + m - 1]

def _optimize_adam_variance(a0, deadline, verbose=True):

    a = a0.copy()

    n = a.shape[0]

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best = a.copy()
    best_val = _objective(best)
    t = 0

    while time.time() < deadline:
        conv = _fft_convolve(a)
        max_c = np.max(conv)
        cur_val = 2.0 * n * max_c
        if cur_val < best_val:
            best_val = cur_val; best = a.copy()
            if verbose: print(f"[improve] iter {t} obj {best_val}")

        # compute smooth max gradient
        gamma = min(2000.0, 5.0 + 0.01 * t)
        scaled = conv - np.max(conv)
        exp_scaled = np.exp(gamma * scaled)
        weights = exp_scaled / exp_scaled.sum()  # shape 2n-1

        # compute grad_max via convolution
        a_rev = a[::-1]
        grad_max_full = _fft_convolve_generic(weights, a_rev)
        start = n - 1
        grad_max = 2.0 * grad_max_full[start:start+n]   # shape n

        # compute variance gradient
        mu = conv.mean()
        diff = conv - mu
        grad_var_full = _fft_convolve_generic(diff, a_rev)
        grad_var = 4.0 * grad_var_full[start:start+n]

        # schedule alpha
        var_iters = 5000
        alpha = 0.0
        if t > var_iters:
            alpha = min(1.0, (t - var_iters) / 5000.0)  # linear from 0 to 1 over 5k steps
        else:
            alpha = 0.0
        grad = (1 - alpha) * grad_var + alpha * grad_max

        # Adam update
        beta1, beta2 = 0.9, 0.999
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        t1 = t + 1
        m_hat = m / (1 - beta1 ** t1)
        v_hat = v / (1 - beta2 ** t1)
        lr = 0.1 / np.sqrt(t1)  # diminishing
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        a = a - step
        a = _project_simplex(a)

        # occasional noise
        if t % 2000 == 0 and t > 0:
            a = a + np.random.normal(scale=0.001, size=n)
            a = _project_simplex(a)

        t += 1

    return best, best_val

"""
Propose a steep‑descent + variance‑reduction optimizer for the
self‑convolution max‑norm problem.

The evaluation function is
    O(a) = 2·n·max_k (a*a)[k] / (∑a)^2 .
Since the expression is homogeneous we may restrict ourselves to the
probability simplex (∑a = 1, a_i ≥ 0).  The optimiser works on the
simplex directly:

* The convolution a*a is computed with a real FFT (O(n log n)).
* To obtain a useful descent direction we minimise a *smooth* version
  of the max‑norm:
        φ_γ(a) = (1/γ) log Σ_i exp(γ·((a*a)[i]‑μ)),
  where μ = mean(a*a) = 1/(2n‑1) for a on the simplex.
  The gradient of φ_γ is a weighted sum of the “shift‑matrices’’ J_i a,
  the weights being the soft‑max of the convolution entries.
* Early in the run we also minimise the variance of the convolution
  (∑(a*a‑μ)^2) to flatten the profile.  The two gradients are blended
  with a schedule that moves from pure variance‑gradient to pure
  max‑gradient.
* Updates use the Adam adaptive‑step scheme and the result is projected
  back onto the simplex after every step.
* Several simple initialisations (Catalan, uniform, Beta‑shaped
  distributions and a random Dirichlet) are tried; the best sequence
  found before the time budget expires is returned.

The method stays fully vectorised, uses only NumPy, and respects the
budget of 1000 s.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based full linear convolution (real inputs)
# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT. Result length = 2*len(a)-1."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[: 2 * n - 1]


def _fft_convolve_generic(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two (real) vectors x and y using FFT."""
    n = x.shape[0]
    m = y.shape[0]
    size = 1 << ((n + m - 1).bit_length())
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)
    return conv[: n + m - 1]


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple families of candidate start vectors (all are normalised to sum‑1)
# ----------------------------------------------------------------------
def _catalan_normalised(n: int) -> np.ndarray:
    """Catalan coefficients a_i = C(2i,i)/4**i, normalised to sum 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    a /= a.sum()
    return a


def _uniform_vector(n: int) -> np.ndarray:
    """Uniform distribution on n points."""
    a = np.full(n, 1.0 / n, dtype=np.float64)
    return a


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta‑like shape: a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(n, dtype=np.float64) + 1.0
    a = (i ** (alpha - 1.0)) * ((n + 1 - i) ** (beta - 1.0))
    a_sum = a.sum()
    if a_sum == 0.0:
        # fallback to uniform if parameters are pathological
        return _uniform_vector(n)
    a /= a_sum
    return a


def _random_dirichlet(n: int) -> np.ndarray:
    """Random point on the simplex (vector of iid Exp(1) variables)."""
    a = np.random.rand(n)
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Objective for a vector already on the simplex (sum = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    return float(2.0 * n * np.max(_fft_convolve(a)))


# ----------------------------------------------------------------------
# Adam‑style optimiser that mixes variance and smoothed‑max gradients
# ----------------------------------------------------------------------
def _optimize_adam_variance(a0: np.ndarray,
                            deadline: float,
                            verbose: bool = True) -> tuple[np.ndarray, float]:
    """
    Refine a0 (already on the simplex) by a smoothed max‑gradient
    combined with a variance‑gradient.  Uses Adam updates and projects
    back onto the simplex each step.
    Returns the best vector encountered and its (true) objective value.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam moment vectors
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # Book‑keeping of the best solution
    best = a.copy()
    best_val = _objective(best)

    t = 0
    while time.time() < deadline:
        # --------------------------------------------------------------
        # current convolution and true objective
        # --------------------------------------------------------------
        conv = _fft_convolve(a)                     # length 2n‑1
        max_c = conv.max()
        cur_val = 2.0 * n * max_c
        if cur_val < best_val:
            best_val = cur_val
            best = a.copy()
            if verbose:
                print(f"[improve] iter {t:06d}  obj {best_val:.9f}")

        # --------------------------------------------------------------
        # smoothed max‑gradient (soft‑max weights)
        # --------------------------------------------------------------
        gamma = min(2000.0, 5.0 + 0.01 * t)          # temperature schedule
        scaled = conv - conv.max()                  # max at zero → safe exp
        w = np.exp(gamma * scaled)
        w /= w.sum()                                # soft‑max weights, sum = 1

        a_rev = a[::-1]
        grad_max_full = _fft_convolve_generic(w, a_rev)
        start = n - 1
        grad_max = 2.0 * grad_max_full[start:start + n]

        # --------------------------------------------------------------
        # variance‑gradient  (grad of Σ (conv‑μ)^2 )
        # --------------------------------------------------------------
        mu = conv.mean()
        diff = conv - mu
        grad_var_full = _fft_convolve_generic(diff, a_rev)
        grad_var = 4.0 * grad_var_full[start:start + n]

        # --------------------------------------------------------------
        # blending schedule: start with variance, then move to pure max
        # --------------------------------------------------------------
        var_iters = 5000
        if t <= var_iters:
            alpha = 0.0
        else:
            # linearly increase α from 0 to 1 over the next 5000 iterations
            alpha = min(1.0, (t - var_iters) / 5000.0)
        grad = (1.0 - alpha) * grad_var + alpha * grad_max

        # --------------------------------------------------------------
        # Adam update + projection onto simplex
        # --------------------------------------------------------------
        beta1, beta2 = 0.9, 0.999
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        t1 = t + 1
        m_hat = m / (1.0 - beta1 ** t1)
        v_hat = v / (1.0 - beta2 ** t1)

        lr = 0.1 / np.sqrt(t1)                     # diminishing step
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-12)

        a = a - step
        a = _project_simplex(a)

        # occasional tiny perturbation to escape plateaus
        if t % 2000 == 0 and t > 0:
            a = a + np.random.normal(scale=0.001, size=n)
            a = _project_simplex(a)

        t += 1

    return best, best_val


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Return a sequence of non‑negative heights that (approximately) minimises
    ``evaluate_sequence`` within the allotted time budget.  The routine tries
    a handful of analytically motivated initialisations and refines each of
    them with the Adam‑based optimiser defined above.  The best vector found
    before the deadline is returned as a plain Python list.
    """
    np.random.seed(seed)
    # leave a small margin for the harness
    deadline = time.time() + budget_s - 2.0

    # problem size – 20000 gives a good trade‑off between quality and speed
    n = 20_000

    # ------------------------------------------------------------------
    # 1) Prepare a pool of starting vectors
    # ------------------------------------------------------------------
    starts = []

    # Catalan (the previous state‑of‑the‑art)
    starts.append(_catalan_normalised(n))

    # Uniform
    starts.append(_uniform_vector(n))

    # A few Beta‑shaped families (U‑shaped & mildly peaked)
    starts.append(_beta_distribution(n, alpha=0.5, beta=0.5))   # strong U‑shape
    starts.append(_beta_distribution(n, alpha=0.8, beta=0.8))   # milder U‑shape
    starts.append(_beta_distribution(n, alpha=2.0, beta=2.0))   # centred peak

    # Random Dirichlet start (if time permits)
    starts.append(_random_dirichlet(n))

    global_best_seq = None
    global_best_val = float("inf")

    # ------------------------------------------------------------------
    # 2) Optimise each start while time permits
    # ------------------------------------------------------------------
    for idx, a0 in enumerate(starts):
        if time.time() >= deadline - 5.0:
            break      # not enough time for another run
        print(f"[run] starting optimisation #{idx+1} with initial obj "
              f"{_objective(a0):.9f}")
        seq, val = _optimize_adam_variance(a0, deadline, verbose=True)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[run] new global best = {global_best_val:.9f}")

    # ------------------------------------------------------------------
    # 3) Final polishing step (few thousand iterations of pure sub‑gradient)
    # ------------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline - 2.0:
        # reuse the simpler optimiser from the original baseline for polishing
        from math import sqrt

        def _simple_subgrad(a: np.ndarray, deadline_inner: float):
            """Very short heavy‑ball sub‑gradient refinement (max‑only)."""
            n_loc = a.shape[0]
            a_loc = a.copy()
            best_loc = a_loc.copy()
            best_val_loc = _objective(best_loc)
            momentum = np.zeros_like(a_loc)
            beta_hb = 0.9
            t_loc = 0
            while time.time() < deadline_inner:
                conv_loc = _fft_convolve(a_loc)
                k = int(conv_loc.argmax())
                max_c_loc = conv_loc[k]
                cur_val_loc = 2.0 * n_loc * max_c_loc
                if cur_val_loc < best_val_loc:
                    best_val_loc = cur_val_loc
                    best_loc = a_loc.copy()
                # sub‑gradient of the active term
                grad_loc = np.zeros_like(a_loc)
                i_start = max(0, k - (n_loc - 1))
                i_end = min(n_loc - 1, k)
                idxs = np.arange(i_start, i_end + 1, dtype=int)
                grad_loc[idxs] = 2.0 * a_loc[k - idxs]
                momentum = beta_hb * momentum + (1.0 - beta_hb) * grad_loc
                lr_loc = 0.5 / sqrt(t_loc + 1.0)
                a_loc = a_loc - lr_loc * momentum
                a_loc = _project_simplex(a_loc)
                t_loc += 1
                if t_loc % 2000 == 0:
                    # occasional tiny jitter
                    a_loc = a_loc + np.random.normal(scale=0.001, size=n_loc)
                    a_loc = _project_simplex(a_loc)
            return best_loc, best_val_loc

        polish_deadline = time.time() + min(30.0, deadline - time.time() - 1.0)
        polished_seq, polished_val = _simple_subgrad(global_best_seq, polish_deadline)
        if polished_val < global_best_val:
            global_best_val = polished_val
            global_best_seq = polished_seq
            print(f"[polish] improved objective = {global_best_val:.9f}")

    # ------------------------------------------------------------------
    # 4) Return the best sequence found
    # ------------------------------------------------------------------
    final_obj = _objective(global_best_seq)
    print(f"[final] best objective ≈ {final_obj:.9f} with n={n}")

    return global_best_seq.tolist()
