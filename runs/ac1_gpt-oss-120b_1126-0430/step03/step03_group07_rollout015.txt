def _optimize_sequence(seq0, deadline):
    n = len(seq0)
    eps=1e-15
    log_a = np.log(seq0 + eps)
    beta1=0.9; beta2=0.999; lr=0.05; maybe use smaller?
    m = np.zeros_like(seq0); v = np.zeros_like(seq0)
    best_seq = None; best_score = float('inf')
    t=0
    mu0 = 0.5; mu_max = 200.0
    # determine schedule: mu = mu0 + (mu_max - mu0) * min(1.0, t / 5000)
    # Hard-coded 5000 iterations for μ to reach max.

    while time.time() < deadline:
        # compute a vector
        a = np.exp(log_a)
        sum_a = a.sum()
        if sum_a == 0.0:

        else:
            a /= sum_a  # ensure sum=1

        # a = (a + a[::-1]) / 2

        fft_len = 1 << ((2 * n - 1).bit_length())
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        # compute smooth softmax weights
        mu = mu0 + (mu_max - mu0) * min(1.0, t / 2000)  # schedule
        # Use stable exp: subtract max
        max_conv = conv.max()

        exp_term = np.exp(mu * (conv - max_conv))
        p = exp_term / exp_term.sum()  # shape length 2n-1

        # compute gradient: g_i = sum_k p_k a_{k-i}

        t += 1

"""
Optimiser for the step‑function metric

    2·n·max(conv(a,a)) / (sum(a))²

The algorithm works directly on the final length (default 5000) and
uses a smooth approximation of the max‑operator.  For a given vector
`a` we compute the self‑convolution `c = a*a`.  The softmax
``p_k ∝ exp(μ·c_k)`` approximates the active‑set weighting of the
max‑operator; the gradient of the smoothed objective is
``g_i = Σ_k p_k·a_{k-i}``.  This gradient is obtained efficiently by a
single FFT‑based convolution.  An Adam optimiser is applied to the
log‑variables (exponentiated‑gradient / mirror descent on the simplex).
The temperature μ is annealed from a small value (very smooth) to a
large one (close to the true max), allowing the optimiser to first move
freely and later focus on the true objective.

Multiple seeds (arcsine, beta‑distortions, random Dirichlet, uniform and
the optional `height_sequence_1`) are tried with a time‑sliced budget;
the best result is refined for the remaining time and returned as a
plain Python list of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast evaluation (identical to the supplied evaluate_sequence but without
# the security checks).  The result is a float.
# ----------------------------------------------------------------------
def _fast_evaluation(seq: np.ndarray) -> float:
    """Metric 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    n = seq.size
    if n == 0:
        return float("inf")
    # next power of two for the convolution length
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------
def _beta_sequence(alpha: float, beta: float, length: int) -> np.ndarray:
    """Beta(α,β) density sampled at mid‑points, normalised to sum = 1."""
    xs = (np.arange(length, dtype=float) + 0.5) / length
    a = np.power(xs, alpha - 1.0, out=np.empty_like(xs))
    b = np.power(1.0 - xs, beta - 1.0, out=np.empty_like(xs))
    seq = a * b
    seq[np.isinf(seq) | np.isnan(seq)] = 0.0
    seq = np.maximum(seq, 0.0)
    if seq.sum() == 0.0:
        seq = np.full(length, 1.0 / length)
    else:
        seq /= seq.sum()
    return seq


def _random_dirichlet(length: int, concentration: float = 0.5) -> np.ndarray:
    """Draw a Dirichlet point with a single concentration parameter."""
    seq = np.random.gamma(concentration, 1.0, size=length)
    if seq.sum() == 0.0:
        seq = np.full(length, 1.0 / length)
    else:
        seq /= seq.sum()
    return seq


# ----------------------------------------------------------------------
# Adam optimiser with a smooth max (soft‑max) surrogate.
# ----------------------------------------------------------------------
def _adam_optimize(seq0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Refine ``seq0`` (non‑negative, sum = 1) using Adam on the log‑variables.
    Returns the best vector found before ``deadline``.
    """
    n = seq0.size
    eps = 1e-15
    # log‑variables (exponentiated‑gradient)
    log_a = np.log(seq0 + eps)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    base_lr = 0.25      # base step size in log‑space
    m = np.zeros_like(seq0)
    v = np.zeros_like(seq0)
    t = 0

    # FFT lengths (fixed for the whole optimisation)
    fft_len_self = 1 << ((2 * n - 1).bit_length())      # for self‑convolution
    conv_len_grad = 3 * n - 2                           # length of p*a_rev convolution
    fft_len_grad = 1 << ((conv_len_grad - 1).bit_length())

    best_seq = None
    best_score = float("inf")

    # temperature schedule for the soft‑max
    mu_start = 0.4
    mu_end = 150.0
    mu_ramp_iters = 8000   # after this many iterations μ≈mu_end

    while time.time() < deadline:
        # ------- obtain current vector and normalise -------
        a = np.exp(log_a)
        sum_a = a.sum()
        if sum_a == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= sum_a

        # optional symmetrisation (helps some seeds)
        a = 0.5 * (a + a[::-1])

        # ------- self‑convolution -------
        fa = np.fft.rfft(a, fft_len_self)
        conv = np.fft.irfft(fa * fa, fft_len_self)[: 2 * n - 1]

        # ------- soft‑max weights -------
        mu = mu_start + (mu_end - mu_start) * min(1.0, t / mu_ramp_iters)
        max_c = conv.max()
        # stable soft‑max
        exp_term = np.exp(mu * (conv - max_c))
        p = exp_term / exp_term.sum()        # length 2n‑1, non‑negative, sums to 1

        # ------- gradient of the smooth objective -------
        a_rev = a[::-1]
        fp = np.fft.rfft(p, fft_len_grad)
        fa_rev = np.fft.rfft(a_rev, fft_len_grad)
        grad_full = np.fft.irfft(fp * fa_rev, fft_len_grad)[:conv_len_grad]
        # slice corresponds to indices n‑1 … n‑1+n‑1
        g = grad_full[n - 1: n - 1 + n]

        # ------- Adam update on log‑variables (descent) -------
        t += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # learning‑rate decay (optional)
        lr = base_lr / (1.0 + 1e-5 * t)

        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        log_a -= step                       # descent step

        # keep log‑values in a reasonable range
        np.clip(log_a, -30.0, 30.0, out=log_a)

        # ------- keep the best candidate (real metric) -------
        score = _fast_evaluation(a)
        if score < best_score - 1e-12:
            best_score = score
            best_seq = a.copy()
            if t % 500 == 0:
                print(f"[opt] iter {t:6d}  best {best_score:.6f}")

        # ------- occasional jitter to escape plateaus -------
        if np.random.rand() < 0.001:
            jitter = np.random.randn(n) * 1e-3
            a = np.maximum(a + jitter, 0.0)
            if a.sum() == 0.0:
                a = np.full(n, 1.0 / n)
            a /= a.sum()
            log_a = np.log(a + eps)

    # if nothing was improved, return the last normalised vector
    if best_seq is None:
        a = np.exp(log_a)
        a /= a.sum()
        best_seq = a
    return best_seq


# ----------------------------------------------------------------------
# Main entry point required by the harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Produce a sequence of non‑negative heights with a low evaluation score.

    The routine:
      * builds a small collection of diverse seeds (arcsine, Beta, Dirichlet,
        uniform and optionally the provided ``height_sequence_1``),
      * runs a time‑sliced Adam optimiser with a soft‑max surrogate on each
        seed,
      * keeps the best vector and performs a final refinement with the
        remaining budget,
    finally returning the vector as a plain Python list.
    """
    np.random.seed(seed)
    start = time.time()
    # keep a small safety margin (<budget) for the harness
    deadline = start + budget_s - 5.0

    # final length – the original challenge used 5000 and the metric scales
    # linearly with n, so we stay with that size.
    n = 5000

    # ------------------------------------------------------------------
    # Build a list of seed vectors (all are normalised to sum = 1)
    # ------------------------------------------------------------------
    seeds = []

    # classic arcsine (Beta(0.5,0.5))
    seeds.append(("arcsine", _beta_sequence(0.5, 0.5, n)))

    # a few symmetric beta variants
    for a in [0.30, 0.40, 0.55, 0.70, 0.90, 1.20]:
        seeds.append((f"beta_sym_{a:.2f}", _beta_sequence(a, a, n)))

    # asymmetric beta examples
    for a, b in [(0.3, 0.8), (0.8, 0.3), (0.5, 0.2), (0.2, 0.5)]:
        seeds.append((f"beta_{a:.2f}_{b:.2f}", _beta_sequence(a, b, n)))

    # random Dirichlet draws (concentration < 1 gives spikier shapes)
    for i, conc in enumerate([0.3, 0.5, 0.7]):
        seeds.append((f"rand_dir_{i}", _random_dirichlet(n, concentration=conc)))

    # uniform distribution – a useful baseline
    seeds.append(("uniform", np.full(n, 1.0 / n)))

    # optional external seed that may already exist in the namespace
    if 'height_sequence_1' in globals():
        try:
            hs = np.asarray(height_sequence_1, dtype=float)
            if hs.ndim == 1 and hs.size > 0:
                hs = np.maximum(hs, 0.0)
                if hs.sum() == 0.0:
                    hs = np.full_like(hs, 1.0 / hs.size)
                else:
                    hs /= hs.sum()
                # resample / interpolate to desired length if needed
                if hs.size != n:
                    x_old = np.arange(hs.size, dtype=float)
                    x_new = np.linspace(0, hs.size - 1, n)
                    hs = np.interp(x_new, x_old, hs)
                hs = np.maximum(hs, 0.0)
                hs /= hs.sum()
                seeds.append(("external", hs))
        except Exception:
            pass  # ignore malformed external seed

    # ------------------------------------------------------------------
    # Time‑sliced optimisation of each seed
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    total_seeds = len(seeds)

    while seeds and time.time() < deadline:
        time_left = deadline - time.time()
        per_seed = max(5.0, time_left / (len(seeds) + 1))

        name, seq0 = seeds.pop(0)

        # ensure a proper probability vector
        seq0 = np.maximum(seq0, 0.0)
        if seq0.sum() == 0.0:
            seq0 = np.full(n, 1.0 / n)
        else:
            seq0 = seq0 / seq0.sum()

        seed_deadline = time.time() + per_seed
        refined = _adam_optimize(seq0, seed_deadline)

        score = _fast_evaluation(refined)
        print(f"[seed {name}]  score {score:.6f}")

        if score < best_score:
            best_score = score
            best_seq = refined.copy()
            print(f"*** new global best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final refinement on the overall best vector with any remaining time
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    if time.time() < deadline:
        print("[final] refining globally best sequence...")
        final_seq = _adam_optimize(best_seq, deadline)
        final_score = _fast_evaluation(final_seq)
        print(f"[final] best score {final_score:.6f}")
        best_seq = final_seq
        best_score = final_score

    print(f"[summary] final metric {best_score:.6f} with length {best_seq.size}")

    # return plain Python list (the harness will call evaluate_sequence on it)
    return best_seq.tolist()
