def _optimize_smooth(a0, max_iters, deadline):
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)
    n = a.size
    no_improve = 0
    patience = max_iters // 20
    for t in range(max_iters):
        if time.time() >= deadline: break
        # schedule
        alpha = alpha_start * (alpha_end/alpha_start)**(t / max_iters)
        eta = eta0 / (1 + t/5000)
        # compute conv and gradient
        conv = _conv_fft(a)  # length 2n-1
        max_val = conv.max()
        # softmax weights
        shifted = conv - max_val
        exp_vals = np.exp(alpha * shifted)
        Z = exp_vals.sum()
        p = exp_vals / Z
        # gradient via convolution with a_rev
        a_rev = a[::-1]
        size = 1 << ((3*n - 2).bit_length())
        P = np.fft.fft(p, n=size)
        AR = np.fft.fft(a_rev, n=size)
        conv_full = np.fft.ifft(P * AR).real
        g = 2.0 * conv_full[n-1:n-1+n]
        # optional: clip
        # update
        log_a = np.log(a + 1e-15) - eta * g
        # shift
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()
        # evaluate actual objective
        cur_score = _objective(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            print(f"[opt] iter {t:5d} new best O={best_score:.6f}")
        else:
            no_improve += 1
            if no_improve >= patience:
                # jitter
                jitter = np.random.randn(n) * 1e-4
                a = best_a + jitter
                a = np.clip(a, 0, None)
                a = _project_to_simplex(a)
                no_improve = 0
        # maybe also occasional jitter on g? not needed
    return best_a, best_score

"""
Propose a step‑function minimizing  O = 2·n·max(conv(a,a))  (with Σa = 1).

The algorithm uses a smoothed‑max (log‑sum‑exp) surrogate and an
exponentiated‑gradient descent on the simplex.  The smoothing parameter
α is annealed from a modest start to a larger final value, turning the
smooth surrogate gradually into the true max‑objective.  Multiple
initialisations (previous best, 1/√‑law, a U‑shaped “arcsine” law and a
random point) are tried; the best sequence found before the time budget
expires is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray, size: int = None) -> np.ndarray:
    """
    Linear convolution a*a using a power‑of‑two FFT.
    Returns a vector of length 2·len(a)‑1.
    """
    n = a.size
    if size is None:
        size = 1 << ((2 * n - 2).bit_length())      # next power of two ≥ 2n‑1
    A = np.fft.fft(a, n=size)
    conv = np.fft.ifft(A * A).real[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact bound 2·n·max(conv(a,a)) for a normalised to Σa = 1."""
    n = a.size
    conv = _conv_fft(a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
def _smooth_max_and_gradient(a: np.ndarray,
                             alpha: float,
                             size_conv: int,
                             size_grad: int,
                             grad_start: int) -> tuple[float, np.ndarray]:
    """
    Compute max(conv) and the gradient of the smoothed max
    (log‑sum‑exp with temperature 1/α) w.r.t. a.

    Returns
    -------
    max_val : float
        The (unsmoothed) maximum convolution value.
    grad : np.ndarray, shape (n,)
        Gradient of the smoothed max w.r.t. a.
    """
    n = a.size

    # ----- conv = a * a -------------------------------------------------
    conv = _conv_fft(a, size=size_conv)
    max_val = conv.max()

    # ----- soft‑max weights --------------------------------------------
    shifted = conv - max_val                     # max shift for stability
    exp_vals = np.exp(alpha * shifted)
    Z = exp_vals.sum()
    p = exp_vals / Z                             # p_k = soft‑max weight

    # ----- gradient = 2 * (p * a_rev) convolution --------------------
    a_rev = a[::-1]
    P = np.fft.fft(p, n=size_grad)
    AR = np.fft.fft(a_rev, n=size_grad)
    conv_full = np.fft.ifft(P * AR).real
    grad = 2.0 * conv_full[grad_start:grad_start + n]

    return max_val, grad


# ----------------------------------------------------------------------
def _optimize_smooth(a0: np.ndarray,
                     max_iters: int,
                     eta0: float,
                     alpha_start: float,
                     alpha_end: float,
                     deadline: float) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the smoothed max‑objective.

    Parameters
    ----------
    a0 : np.ndarray
        Initial distribution (already normalised to sum 1).
    max_iters : int
        Upper bound on iterations (the loop also stops when the deadline is hit).
    eta0 : float
        Initial learning‑rate for the multiplicative update.
    alpha_start, alpha_end : float
        Annealing schedule for the smoothing parameter (geometric).
    deadline : float
        Unix time at which the optimisation must stop.

    Returns
    -------
    best_a, best_score
        The best sequence seen (normalised) and its exact objective value.
    """
    a = a0.copy()
    n = a.size
    best_a = a.copy()
    best_score = _objective(a)

    # pre‑compute FFT sizes
    size_conv = 1 << ((2 * n - 2).bit_length())          # ≥ 2n‑1
    size_grad = 1 << ((3 * n - 2).bit_length())          # ≥ 3n‑2 (p length + a_rev)
    grad_start = n - 1                                    # where to slice the full conv

    # early‑stop handling
    patience = max_iters // 20
    no_improve = 0

    for t in range(max_iters):
        if time.time() >= deadline:
            break

        # ---- schedule --------------------------------------------------
        frac = t / max_iters
        alpha = alpha_start * (alpha_end / alpha_start) ** frac   # geometric increase
        eta = eta0 / (1.0 + t / 5000.0)                           # slowly decaying

        # ---- smoothed max & gradient ----------------------------------
        max_val, grad = _smooth_max_and_gradient(a, alpha,
                                                 size_conv, size_grad,
                                                 grad_start)

        # ---- multiplicative (exponentiated) update --------------------
        log_a = np.log(a + 1e-15) - eta * grad
        # keep numbers in a moderate range
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()                                          # project onto simplex

        # ---- exact objective for bookkeeping -------------------------
        cur_score = 2.0 * n * max_val
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            if t % 100 == 0:
                print(f"[opt] iter {t:6d}  new best O={best_score:.6f}")
        else:
            no_improve += 1
            if no_improve >= patience:
                # small random jitter to escape a flat region
                jitter = np.random.randn(n) * 1e-4
                a = best_a + jitter
                a = np.clip(a, 0.0, None)
                a = _project_to_simplex(a)
                no_improve = 0

    return best_a, best_score

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the provided
    ``evaluate_sequence`` bound.  Returns the sequence as a plain list of
    floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Choose a working length.  If a previous construction exists we keep
    # its length, otherwise we pick a power‑of‑two size that works well with FFT.
    prev = globals().get('height_sequence_1')
    if prev is not None:
        a_prev = np.asarray(prev, dtype=float)
        n = a_prev.size
        a_prev = a_prev / a_prev.sum()
    else:
        # 8192 gives a good trade‑off between flexibility and speed
        n = 8192
        a_prev = None

    # ------------------------------------------------------------------
    # Build a small pool of diverse starting points.
    starts: list[np.ndarray] = []

    if a_prev is not None:
        starts.append(a_prev)

    # 1/√ law (theoretically solid)
    idx = np.arange(1, n + 1, dtype=float)
    a_sqrt = 1.0 / np.sqrt(idx)
    a_sqrt /= a_sqrt.sum()
    starts.append(a_sqrt)

    # U‑shaped “arcsine” law: a_i ∝ 1 / √(i·(n‑i+1))
    i = np.arange(1, n + 1, dtype=float)
    a_arc = 1.0 / np.sqrt(i * (n - i + 1))
    a_arc /= a_arc.sum()
    starts.append(a_arc)

    # random point – adds extra stochasticity
    a_rand = np.random.rand(n)
    a_rand = _project_to_simplex(a_rand)
    starts.append(a_rand)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optimisation parameters
    max_iters_per_start = 150_000          # large – loop will stop early if time runs out
    eta0 = 0.03                             # base learning rate for the multiplicative update
    alpha_start = 0.5
    alpha_end = 30.0

    for si, a_start in enumerate(starts):
        if time.time() >= deadline:
            break
        init_score = _objective(a_start)
        print(f"[init {si}] start O={init_score:.6f}")

        a_opt, score_opt = _optimize_smooth(a_start,
                                            max_iters_per_start,
                                            eta0,
                                            alpha_start,
                                            alpha_end,
                                            deadline)

        print(f"[result {si}] optimized O={score_opt:.6f}")

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()

    # ------------------------------------------------------------------
    # Fallback: if nothing succeeded (e.g. time ran out), use the 1/√ shape.
    if best_seq is None:
        best_seq = a_sqrt.copy()
        best_score = _objective(best_seq)

    # Final exact evaluation using the supplied function.
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:               # pragma: no cover
        print("[final] evaluate_sequence call failed:", e)
        final_score = None

    return best_seq.tolist()
