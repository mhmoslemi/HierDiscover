# ----------------------------------------------------------------------
# Simplex projection – Euclidean projection onto {x ≥ 0, Σx = 1}
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all non‑positive
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # Numerical safety – enforce exact sum = 1
    w /= w.sum() if w.sum() > 0 else 1.0
    return w


# ----------------------------------------------------------------------
# FFT‑based full self‑convolution (length 2·n‑1)
def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Return a*a (full convolution) using an FFT of length fft_len."""
    fa = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(fa * fa, n=fft_len)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Exact objective for a simplex‑normalised vector (sum = 1)
def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact O(a) = 2·n·max(conv) for Σa = 1."""
    conv = _conv_fft(a, fft_len)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Surrogate gradient – log‑sum‑exp approximation of max(conv)
def _lse_grad(a: np.ndarray, conv: np.ndarray, alpha: float, fft_len: int) -> np.ndarray:
    """
    ∇_a  ( (2·n/α)·log Σ_k exp(α·conv_k) )
    Formula: 2·corr(w, a)  where w_k ∝ exp(α·conv_k)
    """
    # stable soft‑max weights
    scaled = alpha * conv
    max_s = np.max(scaled)
    exp_scaled = np.exp(scaled - max_s)
    w = exp_scaled / exp_scaled.sum()

    # correlation (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fw = np.fft.rfft(w, n=fft_len)
    fa_rev = np.fft.rfft(a_rev, n=fft_len)
    corr_rev = np.fft.irfft(fw * fa_rev, n=fft_len)
    n = a.size
    grad = 2.0 * corr_rev[n - 1 : n - 1 + n]   # central n entries
    return grad


# ----------------------------------------------------------------------
# Sub‑gradient of the maximal convolution entry (exact)
def _max_conv_subgrad(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. a."""
    n = a.size
    k = conv.argmax()                     # a single maximiser is enough
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    i_idx = np.arange(i_start, i_end + 1, dtype=int)
    j_idx = k - i_idx
    g = np.zeros_like(a)
    # each pair (i,j) contributes symmetrically
    g[i_idx] += a[j_idx]
    g[j_idx] += a[i_idx]
    # for odd size there may be a double count of the centre, correct by halving
    if (i_idx == j_idx).any():
        g[i_idx[i_idx == j_idx]] /= 2.0
    return g

        m = np.zeros_like(a)
        v = np.zeros_like(a)

        # ------------------- Stage A – Adam on log‑sum‑exp -----------------
        for it in range(max_iter_stage1):
            if time.time() >= deadline:
                break

            conv = _conv_fft(a, fft_len)
            cur_obj = _objective(a, n, fft_len)
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                print(f"[cid{cid}] stage‑A iter={it:,}  O={best_obj:.6f}")

            # linear annealing of α
            frac = it / max_iter_stage1
            alpha = alpha_start + (alpha_end - alpha_start) * frac

            # surrogate gradient
            grad = _lse_grad(a, conv, alpha, fft_len)

            # Adam update (bias‑corrected)
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** (it + 1))
            v_hat = v / (1.0 - beta2 ** (it + 1))
            step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

            a = _project_to_simplex(a - step)

        # ------------------- Stage B – exact sub‑gradient -----------------
        for it in range(max_iter_stage2):
            if time.time() >= deadline:
                break

            conv = _conv_fft(a, fft_len)
            cur_obj = _objective(a, n, fft_len)
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                print(f"[cid{cid}] stage‑B iter={it:,}  O={best_obj:.6f}")

            # sub‑gradient of the current maximum entry
            g = _max_conv_subgrad(a, conv)

            # linearly decaying step size
            eta = 0.08 * (1.0 - it / max_iter_stage2)
            a = a - eta * g
            a = _project_to_simplex(a)

            # occasional tiny jitter to escape plateaus
            if it % 2000 == 0 and it > 0:
                jitter = np.random.randn(n) * 2e-4
                a = _project_to_simplex(a + jitter)

        # ------------------------------------------------------------------
        # Deterministic hill‑climbing (no temperature)
        no_improve = 0
        max_hc_iters = 250_000
        for it in range(max_hc_iters):
            if time.time() >= deadline:
                break

            conv = _conv_fft(a, fft_len)
            cur_obj = _objective(a, n, fft_len)
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                no_improve = 0
                print(f"[cid{cid}] hillclimb iter={it:,}  O={best_obj:.6f}")
            else:
                no_improve += 1

            # stop early if we are stuck
            if no_improve > 5000:
                break

            # Identify the worst convolution index
            k = conv.argmax()
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_idx = np.arange(i_start, i_end + 1, dtype=int)
            # pick the index inside the active window with the largest a_i
            i = i_idx[a[i_idx].argmax()]

            # pick a destination index with minimal mass (outside the active set)
            mask = np.ones(n, dtype=bool)
            mask[i_idx] = False
            if not mask.any():
                # fall back to whole vector
                mask[:] = True
                mask[i] = False
            j = np.argmin(a[mask])
            j = np.flatnonzero(mask)[j]

            if a[i] <= 1e-12:
                continue

            delta = min(a[i] * 0.10, 0.05)          # move up to 10 % of source mass
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            a_new = _project_to_simplex(a_new)

            new_obj = _objective(a_new, n, fft_len)
            if new_obj < cur_obj:
                a = a_new
                no_improve = 0  # reset stagnation counter

        # end of candidate optimisation
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Final sanity check with the supplied evaluation function
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence failed:", exc)

    return best_a.tolist()
