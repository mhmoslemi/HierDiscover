# ----------------------------------------------------------------------
# Fast evaluation (identical to the supplied ``evaluate_sequence`` but
# without the security checks).  Returns ``float('inf')`` only for empty
# input – never for a correctly normalised vector.
# ----------------------------------------------------------------------
def _fast_evaluation(seq: np.ndarray) -> float:
    """Metric 2·n·max(conv(a,a)) / (sum(a))² (sum is assumed =1)."""
    n = seq.size
    if n == 0:
        return float("inf")
    # length for linear convolution via FFT
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    max_b = conv.max()
    # sum is forced to 1 in the optimiser, but keep the original formula
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean projection onto {x≥0, Σx=1})
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    # find the rho where u_i - (cssv_i / i) > 0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical safeguard – fall back to uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w

def _beta_sequence(alpha: float, beta: float, length: int) -> np.ndarray:
    """Beta(α,β) density sampled on the centre of each sub‑interval."""
    xs = (np.arange(length, dtype=float) + 0.5) / length
    a = np.power(xs, alpha - 1.0, out=np.empty_like(xs))
    b = np.power(1.0 - xs, beta - 1.0, out=np.empty_like(xs))
    seq = a * b
    seq[np.isinf(seq)] = 0.0
    seq = np.maximum(seq, 0.0)
    if seq.sum() == 0.0:
        seq = np.full(length, 1.0 / length)
    else:
        seq /= seq.sum()
    return seq


def _sin_power_sequence(p: float, length: int) -> np.ndarray:
    """sin(π·x)^p on (0,1) sampled at the centre of each sub‑interval."""
    xs = (np.arange(length, dtype=float) + 0.5) / length
    vals = np.sin(np.pi * xs) ** p
    vals = np.maximum(vals, 0.0)
    if vals.sum() == 0.0:
        vals = np.full(length, 1.0 / length)
    else:
        vals /= vals.sum()
    return vals


def _random_dirichlet(length: int, concentration: float = 0.5) -> np.ndarray:
    """Draw a Dirichlet point with identical concentration."""
    seq = np.random.gamma(concentration, 1.0, size=length)
    if seq.sum() == 0.0:
        seq = np.full(length, 1.0 / length)
    else:
        seq /= seq.sum()
    return seq


def _exp_decay_sequence(r: float, length: int) -> np.ndarray:
    """Exponential decay a_i ∝ r^i , 0<r<1."""
    xs = np.arange(length, dtype=float)
    seq = r ** xs
    seq = np.maximum(seq, 0.0)
    if seq.sum() == 0.0:
        seq = np.full(length, 1.0 / length)
    else:

    return seq


# ----------------------------------------------------------------------
# Smooth projected gradient descent on the simplex
# ----------------------------------------------------------------------
def _smooth_descent(
    seq: np.ndarray,
    deadline: float,
    *,
    initial_lr: float = 0.5,
    final_lr: float = 0.001,
    initial_beta: float = 0.5,
    final_beta: float = 5.0,
    momentum: float = 0.9,
    verbose_interval: int = 1000,
) -> np.ndarray:
    """
    Minimise the soft‑max of the self‑convolution.
    Parameters are schedules that interpolate linearly between the
    given start and end values as the optimisation proceeds.
    Returns the best vector (with sum = 1) found before ``deadline``.
    """
    n = seq.size
    a = np.copy(seq)                 # guaranteed to sum to 1
    a = _project_simplex(a)          # just in case

    # FFT length for linear convolutions of length n and 2n‑1
    fft_len = 1 << ((3 * n - 2).bit_length())

    # momentum buffer for the gradient
    m_buf = np.zeros_like(a)

    best_seq = a.copy()
    best_score = _fast_evaluation(a)

    start = time.time()
    it = 0
    while time.time() < deadline:
        # fraction of the allotted time that has elapsed
        elapsed = time.time() - start
        ratio = elapsed / (deadline - start)

        # linear schedules
        lr = initial_lr * (1.0 - ratio) + final_lr * ratio
        beta = initial_beta * (1.0 - ratio) + final_beta * ratio

        # ---------- self‑convolution ----------
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        # ---------- soft‑max weights ----------
        # shift by the maximum to keep the exponentials ≤ 1
        shift = conv.max()
        exp_arg = beta * (conv - shift)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---------- gradient (2·a * w_rev) ----------
        w_rev = w[::-1]
        fw = np.fft.rfft(w_rev, fft_len)
        grad_full = np.fft.irfft(fa * fw, fft_len)[: 3 * n - 2]
        # slice the centre part and reverse as derived in the analysis
        segment = grad_full[n - 1 : 2 * n - 1]          # length n
        grad = 2.0 * segment[::-1]                     # shape (n,)

        # ---------- momentum ----------
        m_buf = momentum * m_buf + (1.0 - momentum) * grad

        # ---------- gradient step + projection ----------
        a_next = a - lr * m_buf
        a = _project_simplex(a_next)

        # ---------- keep best ----------
        score = _fast_evaluation(a)
        if score < best_score - 1e-12:
            best_score = score
            best_seq = a.copy()
            if it % verbose_interval == 0:
                print(f"[smooth] iter {it:6d}  best {best_score:.6f}")

        # occasional tiny random jitter to help escape flat plateaux
        if np.random.rand() < 0.001:
            j = np.random.randint(n)
            a[j] = max(0.0, a[j] + np.random.randn() * 1e-4)
            a = _project_simplex(a)

        it += 1

    return best_seq

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:

    Produce a step‑function (list of non‑negative floats) with a low
    evaluation metric.  The routine follows the three‑stage strategy
    described in the module docstring.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety cushion for the harness
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # 1.  Small‑scale exploratory phase
    # --------------------------------------------------------------
    n_small = 2000
    seeds = []

    # classic arcsine (Beta(0.5,0.5))
    seeds.append(("arcsine", _beta_sequence(0.5, 0.5, n_small)))

    # a denser sweep of symmetric beta shapes
    for a_val in [0.2, 0.25, 0.3, 0.35, 0.4, 0.45, 0.5,
                  0.55, 0.6, 0.65, 0.7, 0.8, 1.0]:
        seeds.append((f"beta_sym_{a_val:.2f}",
                      _beta_sequence(a_val, a_val, n_small)))

    # asymmetric beta mixtures
    for a_val, b_val in [(0.2, 0.8), (0.8, 0.2), (0.3, 0.9), (0.9, 0.3)]:
        seeds.append((f"beta_{a_val:.2f}_{b_val:.2f}",
                      _beta_sequence(a_val, b_val, n_small)))

    # sine‑power shapes – include a few more negative exponents
    for p in [-0.8, -0.6, -0.4, -0.2, 0.0, 0.2, 0.4]:
        seeds.append((f"sin_pow_{p:.1f}", _sin_power_sequence(p, n_small)))

    # exponential decay profiles
    for r in [0.6, 0.7, 0.8, 0.85]:
        seeds.append((f"exp_decay_{r:.2f}", _exp_decay_sequence(r, n_small)))

    # a few random Dirichlet draws (concentration < 1 creates spikes)
    for i in range(6):
        seeds.append((f"rand_dir_{i}", _random_dirichlet(n_small, concentration=0.4)))

    # --------------------------------------------------------------
    # 2.  Refine each seed for a short, equal share of the remaining time
    # --------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    total_seeds = len(seeds)

    while seeds and time.time() < deadline:
        # allocate a fair slice (minimum 4 s per seed)
        time_left = deadline - time.time()
        per_seed = max(4.0, time_left / (len(seeds) + 1))

        name, seq0 = seeds.pop(0)

        # ensure a proper probability vector
        seq0 = np.maximum(seq0, 0.0)
        seq0 /= seq0.sum() if seq0.sum() > 0 else 1.0

        # short optimisation on the small size
        seed_deadline = time.time() + per_seed
        refined = _smooth_descent(seq0, seed_deadline,
                                  initial_lr=0.6, final_lr=0.002,
                                  initial_beta=0.4, final_beta=4.0,
                                  momentum=0.85,
                                  verbose_interval=2000)

        score = _fast_evaluation(refined)
        print(f"[seed {name}]  score {score:.6f}")

        if score < best_score:
            best_score = score
            best_seq = refined.copy()
            print(f"*** new global best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3.  Upscale the best seed to the final length and perform a long
    #     final refinement.
    # --------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never happen)
        best_seq = np.full(n_small, 1.0 / n_small)

    n_big = 5000
    # linear interpolation from the small grid to the large one
    x_small = np.arange(n_small, dtype=float)
    x_big = np.linspace(0, n_small - 1, n_big)
    up_seq = np.interp(x_big, x_small, best_seq)

    up_seq = np.maximum(up_seq, 0.0)
    up_seq /= up_seq.sum() if up_seq.sum() > 0 else 1.0

    if time.time() < deadline:
        print("[final] starting long refinement on full size...")
        final_seq = _smooth_descent(up_seq, deadline,
                                    initial_lr=0.4, final_lr=0.0005,
                                    initial_beta=0.3, final_beta=6.0,
                                    momentum=0.9,
                                    verbose_interval=5000)
        final_score = _fast_evaluation(final_seq)
        print(f"[final] best score {final_score:.6f}")
        best_seq = final_seq
        best_score = final_score

    # --------------------------------------------------------------
    # 4.  Return the best vector as a plain Python list
    # --------------------------------------------------------------
    print(f"[summary] final metric {best_score:.6f} with length {best_seq.size}")
    return best_seq.tolist()
