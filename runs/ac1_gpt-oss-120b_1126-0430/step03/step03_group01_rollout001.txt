def _optimize_fixed_n(a0, n, deadline):
    # a0 is numpy array shape (n,) sum to 1.
    eps = 1e-16
    z = np.log(a0 + eps)  # logits
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2=0.999
    eps_adam = 1e-8
    base_alpha = 0.2
    best_a = a0.copy()
    best_score = 2.0 * n * _conv_max_fft(a0, _fft_len_for_n(n))[0]

    it = 0
    last_improve = 0
    max_iters_no_improve = 100000
    sigma_min = 0.5
    sigma_max = 30.0
    sigma_inc_iter = 5000  # increase sigma gradually

    while time.time() < deadline:
        it += 1
        a = _softmax(z)
        # Compute conv
        fft_len = _fft_len_for_n(n)
        conv = np.fft.irfft(np.fft.rfft(a, fft_len) * np.fft.rfft(a, fft_len), fft_len)[:2*n-1]
        maxb = conv.max()
        cheap = 2.0 * n * maxb
        # Keep best
        if cheap < best_score:
            best_score = cheap
            best_a = a.copy()
            last_improve = it

        # Compute smoothing weight
        sigma = min(sigma_max, sigma_min + (sigma_max - sigma_min) * (it / sigma_inc_iter))
        # compute weights
        # conv is length 2n-1
        exp_vals = np.exp(sigma * (conv - maxb))
        w = exp_vals / exp_vals.sum()
        # gradient g = 2 * correlation of a with w
        # compute via FFT
        Lcorr = 1 << ((3*n - 2).bit_length())
        Fa = np.fft.rfft(a, Lcorr)
        Fw = np.fft.rfft(w, Lcorr)
        # correlation a ⋆ w = ifft(Fa * conj(Fw))
        corr = np.fft.irfft(Fa * np.conj(Fw), Lcorr)
        g = 2.0 * corr[:n]   # shape n
        # Mirror gradient
        mu = np.dot(g, a)
        grad_z = a * (g - mu)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * (grad_z ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = - m_hat / (np.sqrt(v_hat) + eps_adam)
        # learning rate schedule
        alpha = base_alpha / (math.sqrt(it) + 1.0)
        # line search
        accepted = False
        trial_alpha = alpha
        for _ in range(4):
            z_new = z + trial_alpha * step
            z_new = np.clip(z_new, -50.0, 50.0)
            a_new = _softmax(z_new)
            # compute max conv for new
            conv_new = np.fft.irfft(np.fft.rfft(a_new, fft_len) * np.fft.rfft(a_new, fft_len), fft_len)[:2*n-1]
            cheap_new = 2.0 * n * conv_new.max()
            if cheap_new <= cheap:
                z = z_new
                accepted = True
                break
            trial_alpha *= 0.5
        if not accepted:
            z += alpha * step
            z = np.clip(z, -50.0, 50.0)

        # random resets or perturb
        if it % 10000 == 0:
            # random noise
            noise = 0.01 * np.random.randn(n)
            z += noise
            m *= 0.5
            v *= 0.5

        # early stop if no improvement for many iterations
        if it - last_improve > max_iters_no_improve:
            break

    return best_a, best_score

"""
Hybrid Adam optimizer for the step‑function problem.

* Uses a probability‑simplex parametrisation (soft‑max of logits `z`).
* Instead of the raw `max` of the self‑convolution we minimise a
  smoothed version:
      w_k ∝ exp(σ·(c_k‑max(c))) ,   Σ w_k = 1
  where `c = a*a` is the full convolution.  The gradient of the
  smoothed objective is a correlation `g = 2·(a ⋆ w)`, which is
  evaluated with a single real FFT (O(N log N) per iteration).
* An Adam update is performed in log‑it space with a simple back‑track
  line‑search so that the cheap proxy `2·n·max(c)` never increases.
* The algorithm runs on several support sizes (up to 20 000).  For each
  size it performs a short Adam optimisation, optionally restarts from
  a perturbed copy of the current best, and keeps the overall best
  candidate (both cheap proxy and exact `evaluate_sequence` score).
* Random perturbations and a gradual increase of the smoothing factor
  `σ` help to escape plateaus.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max that returns a probability vector (sum = 1)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Returns logits that map to `a` after a soft‑max."""
    return np.log(a + eps)


def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two ≥ 2·n‑1 (suitable for a*a)."""
    return 1 << ((2 * n - 1).bit_length())


def _fft_len_for_corr(n: int) -> int:
    """Smallest power‑of‑two ≥ 3·n‑2 (suitable for a ⋆ w)."""
    return 1 << ((3 * n - 2).bit_length())


def _conv_max_fft(a: np.ndarray, fft_len: int) -> tuple[float, int]:
    """
    Full self‑convolution `a*a` (real FFT) and returns (max value, argmax).
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.size - 1]
    idx = int(conv.argmax())
    return float(conv.max()), idx


def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution `a*a` (real FFT) – returns the whole vector."""
    n = a.shape[0]
    fft_len = _fft_len_for_n(n)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def _corr_a_w(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Correlation g_i = Σ_k w_k * a_{k‑i} for i = 0 … n‑1.
    Implemented via FFT: g = 2·(a ⋆ w)  (the factor 2 is applied by the caller).
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    Fa = np.fft.rfft(a, L)
    Fw = np.fft.rfft(w, L)
    # correlation = irfft(Fa * conj(Fw))
    corr = np.fft.irfft(Fa * np.conj(Fw), L)
    return corr[:n]  # first n entries correspond to lags 0 … n‑1


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    s = new_a.sum()
    if s == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= s
    return new_a


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """Normalised power‑law a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a

# Core optimiser for a fixed support size
# ----------------------------------------------------------------------


def _optimize_fixed_n(a0: np.ndarray, n: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the simplex for a given `n`.

    Returns the best probability vector (sum = 1) found before `deadline`
    together with its cheap proxy score 2·n·max(a*a).
    """
    eps = 1e-16
    # logits
    z = _logits_from_distribution(a0, eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    base_alpha = 0.2

    # smoothing schedule
    sigma_min = 0.5
    sigma_max = 30.0
    sigma_ramp_iters = 5000

    # bookkeeping
    best_a = a0.copy()
    conv0 = _conv_full_fft(a0)
    best_score = 2.0 * n * conv0.max()
    last_improve_it = 0
    it = 0
    max_no_improve = 150_000  # early‑stop patience

    while time.time() < deadline:
        it += 1
        a = _softmax(z)

        # cheap convolution & proxy
        conv = _conv_full_fft(a)
        maxb = conv.max()
        cheap = 2.0 * n * maxb

        if cheap < best_score:
            best_score = cheap
            best_a = a.copy()
            last_improve_it = it

        # ----- smoothed gradient -----
        sigma = sigma_min + (sigma_max - sigma_min) * min(1.0, it / sigma_ramp_iters)
        # weights w_k ∝ exp(σ (conv_k - maxb))
        w_raw = np.exp(sigma * (conv - maxb))
        w_sum = w_raw.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = w_raw / w_sum

        # correlation gives ∂ Σ w_k conv_k / ∂ a
        g = 2.0 * _corr_a_w(a, w)          # shape (n,)

        # Mirror‑descent gradient in log‑it space
        mu = np.dot(g, a)
        grad_z = a * (g - mu)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step_dir = - m_hat / (np.sqrt(v_hat) + eps_adam)

        # learning‑rate schedule (inverse‑sqrt decay)
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)

        # ----- back‑tracking line‑search -----
        accepted = False
        trial_alpha = alpha_t
        for _ in range(4):
            z_new = z + trial_alpha * step_dir
            z_new = np.clip(z_new, -50.0, 50.0)
            a_new = _softmax(z_new)
            conv_new = _conv_full_fft(a_new)
            cheap_new = 2.0 * n * conv_new.max()
            if cheap_new <= cheap:               # descent
                z = z_new
                accepted = True
                break
            trial_alpha *= 0.5

        if not accepted:
            # modest step so optimisation keeps moving
            z += alpha_t * step_dir
            z = np.clip(z, -50.0, 50.0)

        # ----- random perturbations to escape plateaus -----
        if it % 5000 == 0 and np.random.rand() < 0.25:
            perturb = 0.01 * np.random.randn(n)
            z += perturb
            m *= 0.5
            v *= 0.5

        # early stop if stagnation
        if it - last_improve_it > max_no_improve:
            break

    return best_a, best_score
