```
def _smooth_gradient(a, conv, alpha, L):
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / np.sum(exp_scaled)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)  # length L
    n = a.shape[0]
    grad = 2.0 * corr_rev[n-1:n-1+n]
    return grad
```

```
def _subgradient_max_conv(a, conv):
    n = a.size
    max_val = conv.max()
    ks = np.where(conv == max_val)[0]
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k-(n-1))
        i_end = min(n-1, k)
        i_idx = np.arange(i_start, i_end+1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    g /= len(ks)
    return g
```

The program searches for a non‑negative step‑function `a = (a₀,…,aₙ₋₁)` that minimises  

```
O(a) = 2·n·max(conv(a,a)) / ( Σ a_i )² .
```

Because the objective is scale‑invariant we normalise the vector to the probability
simplex (`Σ a_i = 1`).  
The algorithm consists of four phases:

1. **Smart initialisation** – a small grid of Beta‑shaped distributions
   (`a_i ∝ x_i^{α‑1}(1‑x_i)^{β‑1}`) is examined; the best candidate becomes the
   starting point.

2. **Phase 1 – Adam on a smooth log‑sum‑exp surrogate** – we minimise  

   ```
   f_α(a) = (2·n/α)·log Σ_k exp(α·conv_k)
   ```

   with Adam and a linearly increasing temperature `α`.  This surrogate
   approximates the true `max` while staying smooth.

3. **Phase 2 – Deterministic sub‑gradient descent** – we directly follow the
   sub‑gradient of `max(conv)` (the most‑active convolution index) with an
   adaptively decreasing step size.  Tiny Gaussian jitter is added periodically
   to avoid stagnation.

4. **Phase 3 – Simulated‑annealing hill‑climbing** – random mass‑transfer moves
   (`δ` moved from one index to another) are accepted with the
   Metropolis rule.  The temperature decays exponentially and the local state is
   occasionally re‑initialised with the current global best.

All phases respect the 1000 s budget; they stop early if the deadline is near.
The best vector found is finally checked with the supplied `evaluate_sequence`
function and returned as a plain Python list.

The implementation is completely self‑contained, uses only `numpy` and the

```python
"""
Propose a non‑negative step‑function sequence that (approximately) minimises

    O(a) = 2·n·max(conv(a,a)) / ( Σ a_i )²

The algorithm:
  1. Initialise with a small grid of Beta‑shaped distributions and keep the
     best candidate (fast cheap sweep).
  2. Adam optimisation on the smooth log‑sum‑exp surrogate, with α annealed
     from 5 to 20000.
  3. Deterministic sub‑gradient descent on the exact max‑convolution entry.
  4. Simulated‑annealing hill‑climbing that repeatedly transfers a random
     fraction of mass between two indices.
Throughout the search the vector is projected onto the probability simplex.
All phases respect a total time budget of 1000 s.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the simplex {x ≥ 0, Σx = 1}
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:                     # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full (linear) self‑convolution
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return a*a (full linear convolution) using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Exact objective for a simplex‑normalised vector
def _objective(a: np.ndarray, n: int, L: int) -> float:
    """2·n·max(conv(a,a)) – a is assumed to sum to 1."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Sub‑gradient of the maximal convolution entry (used in Phase 2)
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Return the sub‑gradient of max(conv) w.r.t. a."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)          # indices achieving the maximum
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]                # each pair contributes symmetrically
    g /= len(ks)                                   # average over all maximisers
    return g


# ----------------------------------------------------------------------
# Gradient of the smooth log‑sum‑exp surrogate
def _smooth_gradient(a: np.ndarray, conv: np.ndarray,
                     alpha: float, L: int) -> np.ndarray:
    """
    Compute ∇_a ( (2·n/α)·log Σ_k exp(α·conv_k) ).
    Result = 2·corr(w, a) where w_k ∝ exp(α·conv_k).
    """
    # soft‑max weights (stable formulation)
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / exp_scaled.sum()

    # correlation w ⋆ a   (g_i = 2·Σ_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)   # length ≥ 3·n‑2
    n = a.shape[0]
    grad = 2.0 * corr_rev[n - 1 : n - 1 + n]          # central n entries
    return grad

        best_a = None
```
