def _softmax_weights(conv, beta):
    # conv shape (2n-1)
    # compute stable softmax of beta * conv
    scaled = beta * conv
    maxval = np.max(scaled)
    exp_shifted = np.exp(scaled - maxval)
    w = exp_shifted / np.sum(exp_shifted)
    return w

    # conv array length 2*n -1

#!/usr/bin/env python
"""
search for a non‑negative step‑function that minimises

    2·n·max_k (a*a)[k] / (∑ a_i)^2

The algorithm consists of three phases:

1.  Choose a working length `n` (≥ 5000) and initialise the vector with a
    normalised power‑law a_i ∝ (i+1)^‑p.  The exponent `p` is scanned on a
    fine grid using a cheap FFT‑based proxy for the objective.

2.  Optimise the vector on the probability simplex by **smooth‑max**
    optimisation.  The nonsmooth term `max(conv)` is replaced by a
    temperature‑controlled soft‑max:

        f_β(a) = (1/β)·log(∑_k exp(β·(a*a)[k]))

    The gradient of `f_β` can be written as a convolution and is obtained
    efficiently with two FFTs per iteration.  An exponentiated‑gradient
    (multiplicative) update keeps the iterate non‑negative and automatically
    normalises it.  The temperature `β` is increased slowly (annealing) so
    that the optimisation starts with a smooth landscape and ends by focussing
    on the true maximum.

3.  The best iterate (according to the cheap proxy `2·n·max(conv)`) is
    evaluated with the provided `evaluate_sequence`.  The corresponding
    sequence is returned.

The whole routine respects the given time budget and prints progress
information.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (sum = 1, v ≥ 0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all entries non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float) -> np.ndarray:
    """Return a normalised power‑law vector a_i ∝ (i+1)^‑p (∑a_i = 1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = i ** (-p)
    seq_sum = seq.sum()
    if seq_sum == 0:
        return np.full(n, 1.0 / n)
    seq /= seq_sum
    return seq


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Stable soft‑max over the convolution values.
    Returns a probability vector `w` of the same length as `conv`.
    """
    scaled = beta * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


def _conv_full_rfft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute the full linear convolution a*a (length 2·len(a)‑1) using real FFT.
    """
    Fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(Fa * Fa, fft_len)
    return conv[: 2 * a.size - 1]


def _gradient_via_softmax(a: np.ndarray, conv: np.ndarray,
                         beta: float, fft_len: int) -> np.ndarray:
    """
    Gradient of the smooth max (soft‑max) w.r.t. `a`.
    Implements   ∇f_β(a) = 2·(w * a_rev)[n‑1 : n‑1+n]
    where w is the soft‑max weight vector.
    """
    n = a.size
    # soft‑max weights over the convolution entries
    w = _softmax_weights(conv, beta)

    # real FFTs (both a and w are real)
    Fa = np.fft.rfft(a, fft_len)
    Fw = np.fft.rfft(w, fft_len)

    # convolution w * a_rev ; the FFT of a_rev is conj(Fa)
    conv_wr = np.fft.irfft(Fw * np.conj(Fa), fft_len)
    # slice the linear part that corresponds to the required gradient
    grad = 2.0 * conv_wr[n - 1: n - 1 + n]   # length n
    return grad

# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    the provided ``evaluate_sequence`` objective.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds; the function must return before the budget
        expires.

    Returns
    -------
    list[float]
        The best sequence found (the list is already normalised – the
        evaluation function will re‑normalise it anyway).
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the final exact evaluation
    deadline = start_time + budget_s - 2.0

    # --------------------------------------------------------------
    # 0. Determine the working length `n` and initialise `a`.
    # --------------------------------------------------------------
    try:
        prev = height_sequence_1               # type: ignore[name-defined]
        prev_arr = np.asarray(prev, dtype=np.float64)
        if prev_arr.size == 0 or prev_arr.sum() == 0:
            prev_arr = None
    except Exception:
        prev_arr = None

    # Choose a length – keep at least 5000 elements to stay in the
    # asymptotic regime while not being too expensive for the final exact
    # evaluation.  If a previous construction exists, reuse its length if it
    # is already large enough.
    if prev_arr is not None:
        n = max(prev_arr.size, 5000)
    else:
        n = 15000                               # default length

    # If the previous construction has a different length we linearly
    # interpolate it to the new size.
    if prev_arr is not None and prev_arr.size != n:
        x_old = np.linspace(0.0, 1.0, prev_arr.size)
        x_new = np.linspace(0.0, 1.0, n)
        a = np.interp(x_new, x_old, prev_arr)
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum == 0:
            a = _power_law_sequence(n, 0.60)
        else:
            a /= a_sum
    else:
        # No usable previous vector – start from a power‑law.
        a = _power_law_sequence(n, 0.60)

    # --------------------------------------------------------------
    # 1. Scan a fine grid of power‑law exponents – cheap proxy.
    # --------------------------------------------------------------
    p_grid = np.linspace(0.55, 0.68, 81)   # dense interval around known optimum
    best_a = a.copy()
    best_score = float("inf")
    best_p = None

    # FFT length for the cheap scan – we only need the convolution of a with itself.
    fft_len_scan = 1 << ((2 * n - 1).bit_length())

    for p in p_grid:
        if time.time() >= deadline:
            break
        a_candidate = _power_law_sequence(n, p)
        conv = _conv_full_rfft(a_candidate, fft_len_scan)
        cheap_score = 2.0 * n * np.max(conv)   # = exact objective for sum=1
        if cheap_score < best_score:
            best_score = cheap_score
            best_a = a_candidate.copy()
            best_p = p

    # Consider the (possibly interpolated) previous construction as a candidate.
    if prev_arr is not None:
        conv = _conv_full_rfft(a, fft_len_scan)
        cheap_score = 2.0 * n * np.max(conv)
        if cheap_score < best_score:
            best_score = cheap_score
            best_a = a.copy()
            best_p = "prev"

    if best_a is None:   # extremely unlikely, but keep a fallback
        best_a = _power_law_sequence(n, 0.60)
        best_p = 0.60
        conv = _conv_full_rfft(best_a, fft_len_scan)
        best_score = 2.0 * n * np.max(conv)

    print(f"Initial power‑law scan finished: best p = {best_p}, cheap score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2. Smooth‑max optimisation with exponentiated‑gradient updates.
    # --------------------------------------------------------------
    a = best_a.copy()                     # current iterate (always Σa = 1)
    best_iter_a = a.copy()
    best_iter_score = best_score

    # FFT length for the optimisation loop – must accommodate a*a
    # (length 2n‑1) and its convolution with a_rev (length 3n‑2).
    fft_len = 1 << ((3 * n - 2).bit_length())

    # Hyper‑parameters
    eta0 = 0.5                # initial step‑size factor for EG
    beta = 0.5                # initial temperature for soft‑max
    beta_inc = 1.001          # multiplicative increase per iteration
    beta_max = 80.0           # cap – large enough to approximate the true max
    stagnation_limit = 20000  # iterations without improvement before a perturbation
    last_improve = 0
    iter_count = 0

    while time.time() < deadline:
        iter_count += 1

        # ----- compute convolution a*a (full) -----
        Fa = np.fft.rfft(a, fft_len)
        conv_full = np.fft.irfft(Fa * Fa, fft_len)[: 2 * n - 1]

        # cheap proxy (exact for Σa = 1)
        cheap_score = 2.0 * n * np.max(conv_full)

        # keep the best vector seen so far
        if cheap_score < best_iter_score:
            best_iter_score = cheap_score
            best_iter_a = a.copy()
            last_improve = iter_count
            print(f"[Iter {iter_count:07d}] NEW BEST cheap score = {cheap_score:.6f}")

        # ----- compute gradient of smooth max -----
        grad = _gradient_via_softmax(a, conv_full, beta, fft_len)

        # ----- exponentiated‑gradient update (multiplicative) -----
        eta = eta0 / math.sqrt(iter_count)
        exp_factor = np.exp(-eta * grad)               # always positive
        a = a * exp_factor
        sum_a = a.sum()
        if sum_a == 0 or np.isnan(sum_a):
            # fallback to uniform if something went wrong
            a = np.full(n, 1.0 / n)
        else:
            a = a / sum_a

        # ----- anneal the temperature -----
        if beta < beta_max:
            beta = min(beta_max, beta * beta_inc)

        # ----- occasional random perturbation to escape plateaus -----
        if iter_count - last_improve > stagnation_limit:
            perturb = 0.01 * np.random.rand(n)
            a = a + perturb
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum == 0:
                a = np.full(n, 1.0 / n)
            else:
                a = a / a_sum
            last_improve = iter_count
            print(f"[Iter {iter_count:07d}] Restarted after stagnation (perturb).")

        # ----- keep the loop cheap – break if we are very close to deadline -----
        if time.time() > deadline - 5.0:
            # one last cheap evaluation before exiting
            pass

    # --------------------------------------------------------------
    # 3. Final exact evaluation using the provided ``evaluate_sequence``.
    # --------------------------------------------------------------
    final_seq = best_iter_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:      # defensive – should never happen
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # --------------------------------------------------------------
    # 4. Return the best sequence found.
    # --------------------------------------------------------------
    return final_seq
