"""
Propose a step‑function (non‑negative sequence) that minimises the bound

    O = 2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm combines three ingredients:

1. **Initialisation** – start from the previously discovered best sequence
   (global variable ``height_sequence_1``) if it exists, otherwise draw a
   random point on the probability simplex of length ``n≈4600``.

2. **Exact incremental convolution** – the self‑convolution is stored once.
   A transfer of a tiny amount ε from index i to index j can be applied in
   O(n) time by updating the two overlapping slices of the convolution and the
   three singleton positions (2·i, 2·j and i+j).  The update is exact, therefore
   the convolution never needs to be recomputed from scratch except for
   occasional re‑normalisations.

3. **Simulated‑annealing hill‑climbing** – repeatedly propose a random mass
   transfer, accept it if it improves the objective or with a temperature‑
   dependent probability otherwise.  The temperature and the maximal transfer
   size decay linearly with the elapsed time.  Every ~10 000 iterations the
   convolution is recomputed from scratch to avoid drift, and after a long
   stagnation period the walk is re‑initialised from the current best solution
   with a tiny random perturbation.

The method respects the 1000 s budget, prints occasional progress messages,
and finally returns the best sequence found as a plain list of floats.
"""

import time
import numpy as np

# -------------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:                      # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Small utility returning the next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (fast, O(n log n))."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv


# -------------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` bound.  Returns the best sequence as a plain list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # -----------------------------------------------------------------
    # 1) Initial sequence
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=float)
        total = a.sum()
        if total == 0.0:
            a = np.random.rand(a.size)
            total = a.sum()
        a = a / total                     # normalise to sum‑1
        n = a.size
    else:
        n = 4600
        a = np.random.rand(n)
        a = _project_to_simplex(a)

    # 2) Full convolution and current objective
    conv = _conv_self_fft(a)               # length 2n‑1
    cur_score = 2.0 * n * conv.max()       # sum(a)=1 → denominator =1
    best_seq = a.copy()
    best_score = cur_score
    print(f"[init] n={n}, O={best_score:.6f}")

    # -----------------------------------------------------------------
    # 3) Simulated‑annealing parameters
    start_move = 0.5          # start with up to 50 % of the source mass
    end_move   = 0.01         # end with very fine moves
    T_start    = max(1e-3, cur_score * 0.08)   # temperature ≈ 8 % of current O
    T_end      = 1e-7

    iter_cnt = 0
    last_improve_iter = 0

    # -----------------------------------------------------------------
    # 4) Main search loop
    while time.time() < deadline:
        iter_cnt += 1

        # --- linear schedules (based on elapsed fraction) ----------------
        elapsed = time.time() + budget_s - deadline - (budget_s - 0.5)
        fraction = min(elapsed / budget_s, 1.0)
        move_factor = start_move + (end_move - start_move) * fraction
        temperature = T_start + (T_end - T_start) * fraction

        # --- propose a random transfer -----------------------------------
        i = np.random.randint(n)
        if a[i] <= 1e-15:                     # nothing to move from here
            continue

        # pick a different target index
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        eps_max = a[i] * move_factor
        if eps_max <= 0.0:
            continue
        eps = np.random.rand() * eps_max          # amount moved from i → j

        di = -eps
        dj =  eps

        # --- apply incremental convolution update (in‑place) ------------
        #   use the *old* a for the shift contributions
        conv[i:i + n] += 2.0 * di * a
        conv[j:j + n] += 2.0 * dj * a
        conv[2 * i]   += di * di
        conv[2 * j]   += dj * dj
        conv[i + j]   += 2.0 * di * dj

        new_max = conv.max()
        new_score = 2.0 * n * new_max
        delta = new_score - cur_score

        # --- Metropolis acceptance ---------------------------------------
        accept = False
        if delta <= 0.0:
            accept = True
        else:
            # probability exp(-Δ/T)
            if temperature > 0.0:
                prob = np.exp(-delta / temperature)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            # keep the update, modify a
            a[i] -= eps
            a[j] += eps

            # occasional renormalisation to combat floating‑point drift
            if abs(a.sum() - 1.0) > 1e-12:
                a = a / a.sum()
                conv = _conv_self_fft(a)
                cur_score = 2.0 * n * conv.max()
            else:
                cur_score = new_score

            # update best solution
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_seq = a.copy()
                last_improve_iter = iter_cnt
                print(f"[iter {iter_cnt:07d}] new best O={best_score:.6f}")

        else:
            # revert the convolution (undo the additive updates)
            conv[i:i + n] -= 2.0 * di * a
            conv[j:j + n] -= 2.0 * dj * a
            conv[2 * i]   -= di * di
            conv[2 * j]   -= dj * dj
            conv[i + j]   -= 2.0 * di * dj
            # a unchanged

        # --- housekeeping -------------------------------------------------
        # every 10 000 steps recompute conv from scratch (removes accumulated error)
        if iter_cnt % 10000 == 0:
            conv = _conv_self_fft(a)
            cur_score = 2.0 * n * conv.max()

        # if we have been stuck for a while, restart from the best with a tiny jitter
        if iter_cnt - last_improve_iter > 200000:
            a = best_seq.copy()
            jitter = np.random.randn(n) * 0.001
            a = np.maximum(a + jitter, 0.0)
            a = a / a.sum()
            conv = _conv_self_fft(a)
            cur_score = 2.0 * n * conv.max()
            last_improve_iter = iter_cnt
            print(f"[restart] iter {iter_cnt}, re‑initialised from best.")

    # 5) Final verification with the supplied ``evaluate_sequence`` routine
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:                 # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_seq.tolist()
