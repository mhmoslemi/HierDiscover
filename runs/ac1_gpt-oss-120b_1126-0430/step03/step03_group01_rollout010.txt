    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 1.0

        # Ensure nonnegative and sum to 1

        s = init_seq.sum()
        if s == 0:

        else:
            init_seq = init_seq / s
        n = init_seq.shape[0]
        print(f"Using provided sequence length {n}")
    except Exception:
        n = 20000
        # power-law start
        p0 = 0.65
        idx = np.arange(1, n+1)
        a0 = idx ** (-p0)
        a0 = a0 / a0.sum()
        init_seq = a0
        print(f"Initialized power law length {n}")

    # Set up parameters
    eps = 1e-16
    z = np.log(init_seq + eps)  # logits

    # Adam parameters
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    base_alpha = 0.5  # base learning rate (may need bigger)

    # Temperature schedule
    beta_temp = 1.0
    beta_max = 500.0
    beta_inc = 1.001  # multiplicative increase per iteration

    # FFT length for conv (2n-1)
    conv_len = 2*n - 1
    L_conv = 1 << ((conv_len).bit_length())  # next power of two >= conv_len
    # For cross correlation length is n+conv_len-1 = 3n-2
    cross_len = n + conv_len - 1
    L_cross = 1 << ((cross_len).bit_length())

    best_seq = init_seq.copy()
    conv = np.fft.irfft(np.fft.rfft(best_seq, L_conv) * np.fft.rfft(best_seq, L_conv), L_conv)[:conv_len]
    best_score = 2.0 * n * conv.max()
    best_iter = 0

    it = 0
    while time.time() < deadline and it < 2000000:
        it += 1

        # compute a from logits
        a = np.exp(z - np.max(z))
        a = a / a.sum()

        # compute conv
        A = np.fft.rfft(a, L_conv)
        conv = np.fft.irfft(A * A, L_conv)[:conv_len]

        maxb = conv.max()
        cheap_score = 2.0 * n * maxb
        if cheap_score < best_score:
            best_score = cheap_score
            best_seq = a.copy()
            best_iter = it
            if (it % 1000) == 0:
                print(f"[Iter {it}] New best cheap {cheap_score:.6f}")

        # compute softmax weighting over conv

        # stable version

        c = np.max(temp)

        P = np.fft.rfft(p, L_cross)

        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning rate schedule
        alpha = base_alpha / (math.sqrt(it) + 1.0)

        step = - alpha * m_hat / (np.sqrt(v_hat) + eps_adam)
        # update logits

        # clip to avoid overflow
        z = np.clip(z, -50.0, 50.0)

        # Update temperature
        if beta_temp < beta_max:
            beta_temp = min(beta_max, beta_temp * beta_inc)

        # occasional random perturb (to avoid stagnation)
        if it % 5000 == 0 and np.random.rand() < 0.3:
            perturb = 0.01 * np.random.randn(n)
            z += perturb
            # damp momentum a bit
            m *= 0.8
            v *= 0.8

    # final evaluation using evaluate_sequence
    final_seq = best_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:
        final_score = float("inf")
        print(f"Error in final evaluation: {exc}")

    return final_seq

#!/usr/bin/env python
"""
Minimise the self‑convolution peak of a non‑negative step‑function.

The algorithm works on the probability simplex by parametrising the
vector `a` with logits `z` (soft‑max).  The objective `2·n·max(a*a)`
is approximated by a smooth version

    fβ(a) = 2·n·(1/β)·log Σ_k exp(β·conv_k),

where `conv = a*a` is the self‑convolution.  The gradient of `fβ` w.r.t.
`a` is obtained analytically (a cross‑correlation) and transformed to
log‑it space.  An Adam optimiser with an annealing temperature `β`
(gradually increased) performs the optimisation.  The best vector found
with the cheap proxy (`2·n·max(conv)`) is finally evaluated with the
provided `evaluate_sequence`.

The routine respects the given time budget (≈1000 s) and returns a list
of non‑negative heights.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x."""
    return 1 << (x - 1).bit_length()


def _softmax_logits(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping logits to a probability vector."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _interpolate_to_length(a: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_len = a.shape[0]
    if old_len == new_len:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    b = np.interp(new_x, old_x, a)
    b = np.maximum(b, 0.0)
    s = b.sum()
    if s == 0.0:
        b[:] = 1.0 / new_len
    else:
        b /= s
    return b

    sequence, using a temperature‑annealed soft‑max approximation of the

    """

    deadline = start_time + budget_s - 1.0  # tiny safety margin

    # ---------------------------------------------------------------
    # 0. Choose a working support size (capped at 20 000)
    # ---------------------------------------------------------------

        init_seq = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        init_seq = np.maximum(init_seq, 0.0)
        s = init_seq.sum()
        if s == 0.0:

        else:
            init_seq = init_seq / s
        # interpolate / truncate to MAX_N entries
        if init_seq.shape[0] != MAX_N:
            init_seq = _interpolate_to_length(init_seq, MAX_N)
        n = MAX_N
        print(f"Initial sequence taken from global variable, length = {n}")

        # fallback – power‑law start
        n = MAX_N
        p = 0.63
        idx = np.arange(1, n + 1, dtype=np.float64)
        init_seq = idx ** (-p)
        init_seq = init_seq / init_seq.sum()
        print(f"Power‑law initialise, length = {n}")

    # ---------------------------------------------------------------
    # 1. Log‑it parametrisation + Adam state.
    # ---------------------------------------------------------------
    eps = 1e-16
    z = np.log(init_seq + eps)          # logits (real numbers)

    m = np.zeros_like(z)               # first moment
    v = np.zeros_like(z)               # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    base_alpha = 0.4                   # base learning rate

    # ---------------------------------------------------------------
    # 2. Temperature schedule for the smooth max.
    # ---------------------------------------------------------------
    beta_temp = 1.0                     # current temperature
    beta_max = 500.0                    # maximum temperature (still safe)
    beta_mul = 1.001                    # multiplicative increase per iteration

    # ---------------------------------------------------------------
    # 3. Pre‑compute FFT length (power of two large enough for
    #    both self‑convolution and cross‑correlation).
    # ---------------------------------------------------------------
    conv_len = 2 * n - 1
    L = _next_pow2(3 * n - 2)           # ≥ n + conv_len - 1

    # ---------------------------------------------------------------
    # 4. Initialise best‑so‑far tracking (cheap proxy).
    # ---------------------------------------------------------------
    A = np.fft.rfft(init_seq, L)
    conv = np.fft.irfft(A * A, L)[:conv_len]
    best_score = 2.0 * n * conv.max()
    best_seq = init_seq.copy()
    print(f"Initial cheap score = {best_score:.6f}")

    it = 0
    last_improve = 0
    report_interval = 1000

    # ---------------------------------------------------------------
    # 5. Main optimisation loop.
    # ---------------------------------------------------------------
    while time.time() < deadline and it < 2_000_000:
        it += 1

        # probability vector from logits
        a = _softmax_logits(z)

        # FFT of the current vector (shared for conv & correlation)
        A = np.fft.rfft(a, L)

        # ---- self‑convolution (cheap proxy) ---------------------------------
        conv = np.fft.irfft(A * A, L)[:conv_len]
        maxb = conv.max()
        cheap_score = 2.0 * n * maxb

        if cheap_score < best_score:
            best_score = cheap_score
            best_seq = a.copy()
            last_improve = it
            if it % report_interval == 0:
                print(f"[Iter {it:07d}] NEW BEST cheap = {cheap_score:.6f}")

            # early stop if we already beat the target bound
            if cheap_score <= 1.503:
                print("Target cheap bound reached – exiting early.")
                break

        # ---- smooth max via temperature‑annealed soft‑max -------------------
        temp = beta_temp * conv
        c = np.max(temp)
        exp_temp = np.exp(temp - c)          # stable soft‑max
        p_weights = exp_temp / exp_temp.sum()   # length = conv_len

        # ---- gradient w.r.t. a (cross‑correlation a ⋆ p) --------------------
        P = np.fft.rfft(p_weights, L)
        # linear cross‑correlation via conj(A) * P
        corr = np.fft.irfft(np.conj(A) * P, L)  # length L
        # needed part: first n entries correspond to (a ⋆ p)[i] for i=0..n‑1
        grad_a = 4.0 * n * corr[:n]

        # ---- map gradient to log‑it space (mirror descent on simplex) ------
        mu = np.dot(grad_a, a)
        grad_z = a * (grad_a - mu)

        # ---- Adam updates --------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning‑rate schedule (inverse‑sqrt decay)
        alpha = base_alpha / (math.sqrt(it) + 1.0)

        step = -alpha * m_hat / (np.sqrt(v_hat) + eps_adam)
        # optional clipping of the step to keep logits bounded
        step = np.clip(step, -0.5, 0.5)

        z += step
        z = np.clip(z, -50.0, 50.0)   # avoid overflow in the soft‑max

        # ---- temperature annealing -----------------------------------------
        if beta_temp < beta_max:
            beta_temp = min(beta_max, beta_temp * beta_mul)

        # ---- occasional random perturbations to escape plateaus ------------
        if it % 4000 == 0 and np.random.rand() < 0.25:
            z += 0.02 * np.random.randn(n)
            m *= 0.5
            v *= 0.5

        # ---- early termination if stagnation observed ----------------------
        if it - last_improve > 200_000 and beta_temp >= beta_max:
            print(f"No improvement for {it - last_improve} iterations – stopping.")
            break

    # ---------------------------------------------------------------
    # 6. Final exact evaluation (uses the provided evaluate_sequence)
    # ---------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:   # defensive – should never happen
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # ---------------------------------------------------------------
    # 7. Return the best sequence found.
    # ---------------------------------------------------------------
    return final_seq
