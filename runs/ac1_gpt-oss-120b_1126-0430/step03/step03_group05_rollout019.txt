#!/usr/bin/env python
"""
Propose a step‑function (list of non‑negative heights) that minimises

    2·n·max_k (a * a)[k] / (Σ a_i)²

The algorithm consists of three phases:

1. **Exploratory initialisation** – a dense scan over simple families
   (power‑law, exponential, uniform and the previous best construction)
   on a moderate length (≈ 8000‑12000).  The best candidate (according to the
   cheap proxy 2·n·maxconv, i.e. the exact objective when Σa = 1) is kept.

2. **Smooth‑max gradient descent** – the non‑smooth objective
   `max(conv)` is replaced by a smooth approximation
   `F(a)= (1/λ)·log Σ_k exp(λ·conv_k)`.  Its gradient is a weighted sum of
   the individual convolution derivatives and can be obtained via a single
   FFT‑based correlation.  Adam (adaptive moment estimation) together with a
   projection onto the probability simplex keeps the iterate feasible.
   The smoothing parameter λ is increased linearly (annealing) so that the
   optimisation first moves in a smooth landscape and later focuses on the
   true maximum.

3. **Final evaluation** – the best iterate found during the search is
   evaluated with the provided ``evaluate_sequence`` and returned.

The whole routine respects a user‑supplied time budget (default 1000 s)
and never exceeds it.
"""

# ----------------------------------------------------------------------
# Imports and helper functions (must be top‑level – no closures)
# ----------------------------------------------------------------------
import time
import math
import numpy as np

def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                              # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^‑p, normalised so Σ a_i = 1."""
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = i ** (-p)
    seq /= seq.sum()
    return seq


def _exponential_sequence(n: int, decay: float) -> np.ndarray:
    """a_i ∝ exp(-decay·i), normalised so Σ a_i = 1."""
    i = np.arange(n, dtype=np.float64)
    seq = np.exp(-decay * i)
    seq /= seq.sum()
    return seq


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·n‑1) using real FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.size - 1]
    return conv


def _softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Soft‑max weighting of convolution values (numerically stable)."""
    maxc = conv.max()
    # subtract max to avoid overflow, then exponentiate
    w = np.exp(lam * (conv - maxc))
    s = w.sum()
    if s == 0.0:
        # extreme λ → all under‑flow; fall back to uniform weights
        w = np.full_like(conv, 1.0 / conv.size)
    else:
        w /= s
    return w


def _gradient_from_weights(a: np.ndarray, w: np.ndarray, fft_len_corr: int) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate:
        g = 2·corr(w, a)   (corr = linear correlation w ⋆ a)
    The correlation is computed by FFT (circular → linear because of zero‑padding).
    """
    # FFT of the two sequences (real input → rfft)
    fw = np.fft.rfft(w, fft_len_corr)
    fa = np.fft.rfft(a, fft_len_corr)

    # linear correlation w ⋆ a  =  ifft( conj(Fw) * Fa )
    corr_full = np.fft.irfft(np.conj(fw) * fa, fft_len_corr)

    n = a.shape[0]
    conv_len = 2 * n - 1                     # length of a*a
    centre = conv_len - 1                     # index of lag 0 in the linear correlation
    # g_i = 2 * corr[centre - i]   for i = 0 … n‑1
    g = 2.0 * corr_full[centre - np.arange(n)]
    return g


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The function runs for at most ``budget_s`` seconds and returns the best
    sequence found (as a plain Python list of floats).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0               # safety margin

    # --------------------------------------------------------------
    # 0. Determine a working length.
    # --------------------------------------------------------------
    try:
        prev = height_sequence_1                 # type: ignore[name-defined]
        prev_arr = np.asarray(prev, dtype=np.float64)
        if prev_arr.size == 0 or prev_arr.sum() == 0:
            prev_arr = None
    except Exception:
        prev_arr = None

    # Use at least 5000 entries, but not more than ~15000 to keep FFT cheap.
    n = max(5000, prev_arr.size if prev_arr is not None else 0, 10000)
    n = min(n, 15000)                     # hard cap for safety

    print(f"[Init] Working length n = {n}")

    # --------------------------------------------------------------
    # 1. Scan a few simple families (power‑law, exponential, uniform).
    # --------------------------------------------------------------
    best_a = None
    best_score = float("inf")

    # helper to evaluate the cheap objective (Σa = 1 ⇒ score = 2·n·maxconv)
    def cheap_score(vec):
        conv = _conv_full_fft(vec, fft_len_conv)
        return 2.0 * n * conv.max(), conv

    # FFT lengths (pre‑computed for the whole run)
    conv_len = 2 * n - 1
    fft_len_conv = 1 << ((conv_len - 1).bit_length())
    corr_len_needed = conv_len + n - 1               # = 3·n‑2
    fft_len_corr = 1 << ((corr_len_needed - 1).bit_length())

    # -------- power‑law grid --------
    p_grid = np.linspace(0.55, 0.70, 81)   # dense around the known optimum
    for p in p_grid:
        if time.time() >= deadline:
            break
        a = _power_law_sequence(n, p)
        sc, _ = cheap_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"  power‑law p={p: .4f} → cheap = {sc:.6f}")

    # -------- exponential grid --------
    decay_grid = np.linspace(0.0002, 0.0035, 71)
    for d in decay_grid:
        if time.time() >= deadline:
            break
        a = _exponential_sequence(n, d)
        sc, _ = cheap_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"  exponential decay={d: .6f} → cheap = {sc:.6f}")

    # -------- uniform (baseline) --------
    a = np.full(n, 1.0 / n, dtype=np.float64)
    sc, _ = cheap_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
        print(f"  uniform → cheap = {sc:.6f}")

    # -------- previous construction, if any --------
    if prev_arr is not None:
        # truncate / pad to length n, then renormalise to sum 1
        if prev_arr.size < n:
            pad = np.zeros(n - prev_arr.size, dtype=np.float64)
            arr = np.concatenate([prev_arr, pad])
        else:
            arr = prev_arr[:n].copy()
        if arr.sum() > 0:
            arr /= arr.sum()
            sc, _ = cheap_score(arr)
            if sc < best_score:
                best_score = sc
                best_a = arr.copy()
                print(f"  previous construction → cheap = {sc:.6f}")

    # Fallback (should never be needed)
    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_score, _ = cheap_score(best_a)
        print("[Fallback] using uniform")

    print(f"[Init] Initial cheap score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2. Refine with a smooth‑max Adam optimiser.
    # --------------------------------------------------------------
    a = best_a.copy()                     # current iterate, always Σa = 1
    m = np.zeros_like(a)                  # first‑moment accumulator (Adam)
    v = np.zeros_like(a)                  # second‑moment accumulator (Adam)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_iter_a = a.copy()
    best_iter_score = best_score
    last_improve = 0

    # lambda schedule (smooth‑max annealing)
    lam_start = 10.0
    lam_end = 800.0
    lam_ramp_iters = 120_000               # over how many iterations we increase λ

    iter_count = 0
    while time.time() < deadline and iter_count < 800_000:
        iter_count += 1

        # ----- convolution (full) -----
        fa = np.fft.rfft(a, fft_len_conv)
        conv = np.fft.irfft(fa * fa, fft_len_conv)[:conv_len]
        maxb = conv.max()
        cheap = 2.0 * n * maxb

        # keep the best cheap score seen so far
        if cheap < best_iter_score - 1e-9:
            best_iter_score = cheap
            best_iter_a = a.copy()
            last_improve = iter_count
            print(f"[Iter {iter_count:07d}] NEW BEST cheap = {cheap:.6f}")

        # ----- update smoothing parameter λ -----
        lam = lam_start + (lam_end - lam_start) * min(1.0, iter_count / lam_ramp_iters)

        # ----- soft‑max weights over the convolution entries -----
        w = np.exp(lam * (conv - maxb))       # maxb is subtracted to avoid overflow
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smooth surrogate -----
        g = _gradient_from_weights(a, w, fft_len_corr)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** iter_count)
        v_hat = v / (1.0 - beta2 ** iter_count)

        # step size decays like 1/sqrt(t) – works well for this scale‑invariant problem
        eta = 0.6 / math.sqrt(iter_count)

        a_candidate = a - eta * m_hat / (np.sqrt(v_hat) + eps)

        # ----- projection onto simplex (ensures feasibility) -----
        a = _simplex_projection(a_candidate)

        # ----- occasional random re‑initialisation on prolonged stagnation -----
        if iter_count - last_improve > 100_000:
            # small random perturbation of the current best
            perturb = 0.02 * np.random.rand(n)
            a = _simplex_projection(best_iter_a + perturb)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = iter_count
            print(f"[Iter {iter_count:07d}] Restarted after stagnation (perturb).")

        # occasional cheap check near the end of the budget
        if iter_count % 5_000 == 0 and time.time() > deadline - 30:
            print(f"[Iter {iter_count:07d}] intermediate cheap = {cheap:.6f}")

    # --------------------------------------------------------------
    # 3. Final exact evaluation (using the supplied ``evaluate_sequence``)
    # --------------------------------------------------------------
    final_seq = best_iter_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[Result] Final exact score = {final_score:.6f}")
    except Exception as exc:
        final_score = float("inf")
        print(f"[Result] Evaluation error: {exc}")

    # --------------------------------------------------------------
    # 4. Return the best sequence found.
    # --------------------------------------------------------------
    return final_seq
