def propose_candidate(seed=..., budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.3

    # initialize f: maybe random, or from height_sequence_1.
    if global variable height_sequence_1 exists and length <= 2000, start with that normalized else uniform.
    else:
        n = 2000
        f = np.random.rand(n)
        f = f / f.sum()
    # ensure positivity nonzero?

    # precompute L for FFT (next power of two >= 2*n -1 *2 maybe).
    L = 1 << ((3*n - 2).bit_length())

    best_f = f.copy()
    best_obj = _objective_simplex(f, n, L)
    print("init O", best_obj)

    # iteration counters
    it = 0
    # step size schedule (t from 0.5 initial to maybe 0.01)
    while time.time() < deadline:
        # compute conv_f_f using FFT
        conv_f = _conv_fft(f, L)
        M = conv_f.max()
        # build A matrix for LP
        # Use sparse matrix
        rows = []
        cols = []
        data = []
        for k in range(2*n-1):
            # j range
            j_start = max(0, k - (n-1))
            j_end = min(n-1, k)
            # i = k - j
            i_vals = k - np.arange(j_start, j_end+1)
            rows.extend([k] * (j_end - j_start + 1))
            cols.extend(np.arange(j_start, j_end+1))
            data.extend(f[i_vals])
        A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))

        # solve LP: maximize sum(b) -> minimize -sum(b)
        c = -np.ones(n)
        b_ub = M * np.ones(2*n-1)
        bounds = [(0, None)] * n
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            # fallback, break loop
            print("LP failed", res.message)
            break
        g = res.x
        sum_g = g.sum()
        if sum_g == 0:
            break

        # scale to sum 1
        g = g / sum_g

        # combine
        t = 0.5   # maybe adapt based on iteration count
        f = (1 - t) * f + t * g
        # ensure sum = 1
        f = f / f.sum()
        # compute objective
        obj = _objective_simplex(f, n, L)
        if obj < best_obj:
            best_obj = obj
            best_f = f.copy()
            print(f"[{time.time():.1f}] iter {it} new best O={best_obj:.6f}")

        # maybe runtime limit
        it += 1
        # optional early termination if improvement small
        if obj - best_obj < 1e-6 and it > 10:
            # maybe break after stable? Not needed.
            pass

    # final evaluation via evaluate_sequence
    final_score = evaluate_sequence(best_f.tolist())
    print("[final] evaluate_sequence reports O=", final_score)

    return best_f.tolist()

def build_conv_matrix(f):
    n = len(f)
    rows = []
    cols = []
    data = []
    for i in range(n):
        # f_i multiplied by each b_j where k = i + j

def propose_candidate(seed=123, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.2

    # Determine initial n
    if 'height_sequence_1' in globals():
        base = np.array(height_sequence_1, dtype=float)
        # truncate to length <=2000 for speed, else uniform
        n0 = base.size
        if n0 > 2000:
            # take first 2000 entries truncated? Or compress by resampling?
            # safer to use uniform.
            n = 2000
            a = np.full(n, 1.0/n)
        else:
            n = n0
            a = base / base.sum()
    else:
        n = 2000
        a = np.full(n, 1.0/n)

    # Ensure positivity
    a = np.maximum(a, 1e-12)
    a = a / a.sum()

    # Precomputations
    L = 1 << ((3 * n - 2).bit_length())
    # Precompute row/col pattern for convolution constraints
    i = np.arange(n, dtype=np.int32)[:, None]
    j = np.arange(n, dtype=np.int32)[None, :]
    row_idx = (i + j).ravel()
    col_idx = np.tile(np.arange(n, dtype=np.int32), n)

    best_seq = a.copy()
    best_obj = _objective_simplex(best_seq, n, L)
    print(f"[init] n={n} O={best_obj:.6f}")

    # Phase A: LP iterative improvement (use ~30% of budget)
    phase_a_deadline = start + budget_s * 0.3
    it_lp = 0
    while time.time() < phase_a_deadline:
        conv = _conv_fft(a, L)
        M = conv.max()
        # Build constraint matrix with current a
        data = np.repeat(a, n)   # length n*n
        A = scipy.sparse.csr_matrix((data, (row_idx, col_idx)), shape=(2*n-1, n))
        # Solve LP
        c = -np.ones(n)   # maximize sum(g)
        b_ub = M * np.ones(2*n-1)
        bounds = [(0, None)] * n
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            print("[lp] linprog failed:", res.message)
            break
        g = res.x
        s = g.sum()
        if s == 0:
            break
        g = g / s
        # mixing weight
        t = 0.5
        a = (1 - t) * a + t * g
        # renormalize (should already sum to 1)
        a = a / a.sum()
        # evaluate

        if cur_obj < best_obj:
            best_obj = cur_obj
            best_seq = a.copy()
            print(f"[lp] iter {it_lp} new best O={best_obj:.6f}")
        it_lp += 1

    # Phase B: stochastic local refinement (use remaining budget)
    print(f"[local] start, best O={best_obj:.6f}")
    # reuse same functions from previous code for mass transfer and subgradient
    subgrad_interval = 5

                    cur_obj = new_obj

                    if cur_obj < best_obj:

The evaluation is invariant under scaling, therefore we keep the vector
normalised to sum 1 and minimise 2·n·max(conv).  The algorithm consists of

1. **LP‑driven smoothing** – for the current distribution f we solve
   a linear programme that maximises the ℓ1‑mass of a candidate g
   under the convolution bound  f ★ g ≤ max(f★f).  The new iterate is a
   convex combination (1‑t)·f + t·g (t≈0.5).  This is the method described
   in the reference note and guarantees a non‑increasing objective.

2. **Stochastic local refinement** – starting from the best vector obtained
   above, we apply simulated‑annealing mass‑transfer moves together with
   occasional exact sub‑gradient steps on max(conv).  The temperature
   decays slowly and the step size of the sub‑gradient is annealed.

Both phases respect the 1000 s time budget, print progress, and finally
return the best sequence (as a plain Python list of floats).

import time
import math
import numpy as np
import scipy.optimize
import scipy.sparse

# FFT‑based full self‑convolution (length 2·n‑1)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return a*a (full convolution) using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Objective for a simplex‑normalised vector (sum(a)=1)
def _objective_simplex(a: np.ndarray, n: int, L: int) -> float:
    """2·n·max(conv(a,a)) (a is assumed to sum to 1)."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Sub‑gradient of the maximal convolution entry (as used in the reference code)
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Return the sub‑gradient of max(conv) w.r.t. a."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)  # indices where the max is attained
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g
