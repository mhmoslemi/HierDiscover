"""
Propose a step‑function sequence that minimises

    O = 2·n·max(conv(a,a)) / (sum(a)²)

The optimizer consists of two phases.

Phase 1 – *global exploration*:
    • Several problem sizes (n) are tried.
    • For each size several deterministic starting shapes are generated
      (random, edge‑heavy, centred Gaussian).
    • A projected Adam optimiser (simplex projection after every step) is
      run for a few‑ten‑thousand iterations.  The gradient is the exact
      sub‑gradient of max(conv) obtained from the active indices of the
      convolution.  The best sequence found over all sizes is kept.

Phase 2 – *fine‑tuning*:
    • Starting from the best sequence of phase 1 (or the provided
      ``height_sequence_1``) a longer Adam run with a much smaller
      learning‑rate schedule is performed.  This phase consumes the
      remaining time budget and aims at shaving the last bits off the
      bound.

All heavy work (full self‑convolution and the required correlation)
is performed by FFTs.  The algorithm respects the 1000 s budget,
prints progress, and finally returns the best sequence (list of floats).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:                     # all non‑positive → uniform vector
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Return the smallest power of two greater than or equal to *x*."""
    return 1 << (x - 1).bit_length()


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (real‑valued) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv


def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation required for the gradient.
    Returns  g_i = Σ_t w[i+t]·a[t]   for i = 0,…,n‑1.
    Implemented with FFTs (linear convolution).
    """
    n = a.shape[0]
    m = w.shape[0]               # = 2n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) (assumes sum(a)=1)."""
    n = a.size
    conv = _conv_self_fft(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Initialisation functions (all return a normalised (sum=1) ndarray)
# ----------------------------------------------------------------------
def _init_random(n: int) -> np.ndarray:
    x = np.random.rand(n)
    return _project_to_simplex(x)


def _init_edge_heavy(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    seq = 1.0 / (i + 1.0) + 1.0 / (n - i)
    seq = np.maximum(seq, 0.0)
    return seq / seq.sum()


def _init_center_gauss(n: int, sigma_ratio: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    sigma = sigma_ratio * n
    seq = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return seq / seq.sum()


def _init_two_peaks(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    sigma = 0.05 * n
    p1 = np.exp(-0.5 * ((i - 0.2 * n) / sigma) ** 2)
    p2 = np.exp(-0.5 * ((i - 0.8 * n) / sigma) ** 2)
    seq = p1 + p2
    return seq / seq.sum()


def _resample_sequence(seq: np.ndarray, target_len: int) -> np.ndarray:
    """Linear interpolation of an existing sequence to a new length."""
    if seq.size == target_len:
        return seq.copy()
    x_old = np.linspace(0, 1, seq.size)
    x_new = np.linspace(0, 1, target_len)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    return new_seq / new_seq.sum()


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the bound
    used by ``evaluate_sequence``.  Returns the best sequence (list of
    floats) found within the supplied time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5          # keep a safety margin

    # --------------------------------------------------------------
    # Global bookkeeping
    # --------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # Phase 1 – coarse global search over several lengths & starts
    # --------------------------------------------------------------
    # problem sizes to explore (feel free to add/remove)
    lengths = [800, 1600, 3200, 4600, 6400]
    # if a previous construction is available, add its length as well
    if "height_sequence_1" in globals():
        lengths.append(len(height_sequence_1))

    init_fns = [_init_random, _init_edge_heavy, _init_center_gauss, _init_two_peaks]

    # allocate roughly the first 20 % of the budget to this phase
    phase1_deadline = time.time() + 0.20 * budget_s

    for n in lengths:
        if time.time() >= phase1_deadline or time.time() >= deadline:
            break

        for init_fn in init_fns:
            if time.time() >= phase1_deadline or time.time() >= deadline:
                break

            # -----------------------------------------------------------------
            # Initialise a candidate of length n
            # -----------------------------------------------------------------
            if "height_sequence_1" in globals() and init_fn is _init_random:
                # use the known construction as a more informed start
                a = _resample_sequence(np.asarray(height_sequence_1, dtype=float), n)
                a = _project_to_simplex(a)
            else:
                a = init_fn(n)

            # Adam moments
            m = np.zeros_like(a)
            v = np.zeros_like(a)
            beta1, beta2, eps = 0.9, 0.999, 1e-12

            # iteration budget for the current (n, init) pair
            max_iters = 22000
            for it in range(1, max_iters + 1):
                if time.time() >= phase1_deadline or time.time() >= deadline:
                    break

                # -------------------------------------------------------------
                # Objective and best‑so‑far update
                # -------------------------------------------------------------
                conv = _conv_self_fft(a)
                M = conv.max()
                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-12:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(f"[phase1] new best O={best_score:.6f}  n={n}  init={init_fn.__name__}  it={it}")

                # -------------------------------------------------------------
                # Sub‑gradient of max(conv) – exact (active set = max index(es))
                # -------------------------------------------------------------
                # use a tiny tolerance to capture multiple active indices
                tol = max(1e-12, M * 1e-12)
                active = np.where(conv >= M - tol)[0]
                w = np.zeros_like(conv)
                w[active] = 1.0 / max(1, active.size)          # equal weighting

                # gradient = 2 * correlation(w, a)
                g = 2.0 * _correlation_fft(w, a)

                # -------------------------------------------------------------
                # Adam update + simplex projection
                # -------------------------------------------------------------
                m = beta1 * m + (1.0 - beta1) * g
                v = beta2 * v + (1.0 - beta2) * (g * g)

                # Learning‑rate schedule: start 0.25 → 0.02
                alpha = 0.25 * (1.0 - it / max_iters) + 0.02

                a = a - alpha * m / (np.sqrt(v) + eps)
                a = _project_to_simplex(a)

                # occasional jitter to escape flat plateaus
                if it % 4000 == 0:
                    jitter = np.random.randn(n) * 0.001
                    a += jitter
                    a = _project_to_simplex(a)

    # --------------------------------------------------------------
    # Phase 2 – long fine‑tuning of the best sequence found so far
    # --------------------------------------------------------------
    if best_seq is None:
        # fallback: a fresh random start of moderate size
        n = 4600
        a = _project_to_simplex(np.random.rand(n))
    else:
        a = best_seq.copy()
        n = a.size

    # Adam moments for the fine‑tuning stage
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-12

    # fine‑tuning runs until the overall deadline is reached
    it = 0
    while time.time() < deadline:
        it += 1

        conv = _conv_self_fft(a)
        M = conv.max()
        cur_score = 2.0 * n * M
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[phase2] new best O={best_score:.6f}  n={n}  it={it}")

        # exact sub‑gradient (active set)
        tol = max(1e-12, M * 1e-12)
        active = np.where(conv >= M - tol)[0]
        w = np.zeros_like(conv)
        w[active] = 1.0 / max(1, active.size)
        g = 2.0 * _correlation_fft(w, a)

        # Adam update – a slower schedule than in phase 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        # Learning‑rate slowly decays from 0.08 to 0.001
        alpha = 0.08 * (1.0 - it / 300000) + 0.001

        a = a - alpha * m / (np.sqrt(v) + eps)
        a = _project_to_simplex(a)

        # tiny jitter every few thousand steps
        if it % 5000 == 0:
            jitter = np.random.randn(n) * 5e-4
            a += jitter
            a = _project_to_simplex(a)

    # ------------------------------------------------------------------
    # Final evaluation with the provided security wrapper
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_seq.tolist()
