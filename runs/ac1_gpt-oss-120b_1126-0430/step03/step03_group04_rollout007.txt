"""
Propose a step‑function sequence that (approximately) minimizes

    O = 2·n·max(conv(a,a)) / (sum(a)**2)

with a_i ≥ 0.  The algorithm is a fast projected Adam optimiser
on a smooth approximation of the *max* (soft‑max with annealing β).
When β is large the soft‑max becomes a hard indicator of the maximal
convolution entries, i.e. a sub‑gradient of the true objective.
A simplex projection enforces the non‑negative unit‑sum constraint.
The schedule for β and the learning‑rate η is exponential, and the
method periodically injects a tiny random jitter and performs a
restart from the current best if no improvement has been seen for a
long time.  The routine respects the 1000 s budget and returns the
best sequence found as a plain list of floats.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:                     # all non‑positive -> uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute g_i = Σ_t w[i+t]·a[t] for i=0,…,n‑1
    (a correlation of w with a, without reversing a).  Implemented via FFT.
    """
    n = a.shape[0]
    m = w.shape[0]               # = 2n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.size
    conv = _conv_self_fft(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    Returns the best sequence found as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ---------------------------------------------------------------
    # Choose a length.  Empirically ~5 k gives a good trade‑off.
    n = kwargs.get("length", 5200)

    # Initialise from the global best if it exists, otherwise random.
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=float)
        a = a / a.sum()
        if a.size != n:
            # Re‑sample to the required length by simple interpolation.
            x_src = np.linspace(0, 1, a.size)
            x_tgt = np.linspace(0, 1, n)
            a = np.interp(x_tgt, x_src, a)
        a = a.astype(float)
    else:
        a = np.random.dirichlet(np.ones(n))

    # Keep the globally best solution.
    best_seq = a.copy()
    best_score = _objective(best_seq)
    print(f"[init] n={n}, objective={best_score:.6f}")

    # ---------------------------------------------------------------
    # Adam parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_iter = 0

    # Schedules
    beta_start, beta_end = 2.0, 5e4          # soft‑max temperature
    lr_start, lr_end = 0.5, 0.01            # learning rate (η)

    # Counters for restart logic
    no_improve = 0
    RESTART_AFTER = 60000          # iterations without improvement
    JITTER_EVERY = 3500

    while time.time() < deadline:
        t_iter += 1

        # -----------------------------------------------------------
        # Full self‑convolution and current objective
        conv = _conv_self_fft(a)
        M = conv.max()
        cur_score = 2.0 * n * M
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            no_improve = 0
            print(f"[iter {t_iter:07d}] new best O={best_score:.6f}")
        else:
            no_improve += 1

        # -----------------------------------------------------------
        # Fraction of the (estimated) total work
        # We use a generous estimate of 300 k iterations as the full run.
        frac = min(t_iter / 300_000.0, 1.0)

        # Exponential schedules for β and η
        beta = beta_start * (beta_end / beta_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac

        # -----------------------------------------------------------
        # Build the weight vector w for the smooth max.
        if beta < 1e4:                     # use soft‑max while β is moderate
            shifted = conv - M                      # ≤ 0
            exp_vals = np.exp(beta * shifted)
            w = exp_vals / exp_vals.sum()
        else:                                 # hard sub‑gradient (indicator)
            w = np.zeros_like(conv)
            # In case of ties, distribute uniformly over them
            max_mask = conv == M
            w[max_mask] = 1.0 / max_mask.sum()

        # -----------------------------------------------------------
        # Gradient of the smooth max – a correlation
        g = 2.0 * _correlation_fft(w, a)

        # -----------------------------------------------------------
        # Adam update on the simplex (projected)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = _project_to_simplex(a_new)

        # -----------------------------------------------------------
        # Periodic jitter to escape flat regions
        if t_iter % JITTER_EVERY == 0:
            jitter = np.random.randn(n) * 0.001
            a_new += jitter
            a_new = _project_to_simplex(a_new)

        # -----------------------------------------------------------
        # Restart if we have been stuck for a while
        if no_improve >= RESTART_AFTER:
            a_new = best_seq.copy()
            a_new += np.random.randn(n) * 0.002
            a_new = _project_to_simplex(a_new)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[restart] iter {t_iter}, re‑initialised from best.")

        a = a_new

    # ------------------------------------------------------------------
    # Final evaluation with the supplied ``evaluate_sequence`` routine.
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_seq.tolist()
