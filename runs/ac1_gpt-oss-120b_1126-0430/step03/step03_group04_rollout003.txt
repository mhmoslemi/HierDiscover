"""
Propose a step‑function sequence that minimises the bound
    O = 2·n·max(conv(a,a))   (with sum(a)=1, a_i ≥ 0)

Algorithm
---------
* Initialise the candidate from the globally‑available best construction
  (``height_sequence_1``) or from a random simplex point.
* Optimise the *soft‑max* approximation of the max‑convolution
  using a projected **Adam** optimiser on the probability simplex.
  The soft‑max temperature (β) grows exponentially from a small value
  to a large one, turning the smooth objective into the true max.
* The gradient of the smooth objective is obtained analytically:
        ∇  Σ_k w_k (a∗a)_k  =  2·corr(w, a)
  where w are the soft‑max weights and ``corr`` is the (non‑reversed)
  correlation, both evaluated by FFTs.
* After each Adam step we project back onto the simplex (non‑negative,
  unit‑sum) using the classic O(n) algorithm.
* Periodically a tiny Gaussian jitter is added to escape plateaus and,
  every few × 10⁴ iterations, the current best solution is re‑initialised
  with a small perturbation (re‑start).
* The loop runs until the allotted time budget (≈ 1000 s) is exhausted.
  The best sequence encountered (according to the exact objective
  ``2·n·max(conv)``) is finally evaluated with ``evaluate_sequence`` and
  returned.

The method improves on the earlier plain projected gradient descent by
using Adam’s adaptive moments, a more aggressive β‑schedule and a
few robustness tricks; it typically pushes the bound below the
previous 1.522263 value within the time limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the simplex {x ≥ 0, sum(x)=1}
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of a vector onto the probability simplex."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation g_i = Σ_t w[i+t]·a[t] for i = 0,…,n‑1.
    Implemented via FFT.
    """
    n = a.shape[0]
    m = w.shape[0]                # = 2n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) under ∑a = 1."""
    n = a.size
    conv = _conv_self_fft(a)
    return float(2.0 * n * conv.max())

    v = np.zeros_like(a)

    best_score = _objective(best_seq)
    print(f"[init] n={n}, objective={best_score:.6f}")

    # Hyper‑parameters
    max_iters_est = 250_000          # rough estimate for schedule
    beta1 = 0.9                      # Adam exponential decay for the 1st moment
    beta2 = 0.999                    # Adam exponential decay for the 2nd moment
    eps = 1e-8                       # Adam epsilon
    eta_start = 0.45                 # initial learning‑rate
    eta_end = 0.02                   # final learning‑rate
    beta_start = 1.0                 # start of soft‑max temperature

        # ---------------------------------------------------------
        # 3) soft‑max weights for the smoothed max
        shifted = conv - M                     # ≤ 0, max entry becomes 0
        # avoid overflow/underflow: the largest exponent is 0
        exp_vals = np.exp(beta * shifted)
        w = exp_vals / exp_vals.sum()          # length 2n‑1, sums to 1

        # ---------------------------------------------------------
        # 4) gradient of the smooth objective (2·correlation)
        g = 2.0 * _correlation_fft(w, a)

        # ---------------------------------------------------------
        # 5) Adam update (on the simplex, then re‑project)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = eta * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = _project_to_simplex(a_new)

        # ---------------------------------------------------------
        # 6) occasional jitter to break plateaus
        if it % 4000 == 0:
            jitter = np.random.randn(n) * 0.001
            a_new += jitter
            a_new = _project_to_simplex(a_new)

        # ---------------------------------------------------------
        # 7) simple restart strategy if stuck for a long while
        if it % 30000 == 0:
            if it - last_improve_iter > 25000:
                a_new = best_seq.copy()
                a_new += np.random.randn(n) * 0.002
                a_new = _project_to_simplex(a_new)
                print(f"[restart] iter {it}, re‑initialised from best (stagnation).")
            else:
                # gentle restart even without stagnation to explore new basins
                a_new = best_seq.copy()
                a_new += np.random.randn(n) * 0.001
                a_new = _project_to_simplex(a_new)
                print(f"[restart] iter {it}, exploration restart.")

        a = a_new

    # ------------------------------------------------------------------
    # Final evaluation with the provided ``evaluate_sequence`` routine.
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_seq.tolist()
