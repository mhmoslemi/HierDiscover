def _mass_transfer_move(a, g, max_frac=0.2):
    # choose source i with probability proportional to g (higher gradient)
    prob_source = g / g.sum() if g.sum()>0 else np.ones_like(g) / len(g)
    i = np.random.choice(len(a), p=prob_source)
    # choose target j with probability proportional to (1/g) or low g
    inv_g = np.max(g) - g
    if inv_g.sum() == 0:
        prob_target = np.ones_like(g) / len(g)
    else:
        prob_target = inv_g / inv_g.sum()
    j = np.random.choice(len(a), p=prob_target)
    if i==j: return a
    # compute delta up to max_frac * a[i]
    max_transfer = a[i] * max_frac
    delta = np.random.rand() * max_transfer
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    a_new = _project_to_simplex(a_new)
    return a_new

* **Stage 1 – Smoothed Adam**  
  Optimises the log‑sum‑exp surrogate  
  fα(a) = (2·n/α)·log Σ_k e^{α·conv_k}  
  with Adam on the Euclidean simplex (projection after each step).  The
  smoothing parameter α is linearly annealed from 5 to 20000, allowing the
  optimisation to focus gradually on the true maximum.

* **Stage 2 – Exact sub‑gradient descent with line‑search**  
  Uses the true sub‑gradient of ``max(conv)``.  A back‑tracking line search
  chooses the largest step that yields improvement, followed by projection
  onto the probability simplex.

* **Stage 3 – Gradient‑guided stochastic refinement**  
  Repeatedly transfers a small fraction of probability mass from indices
  with large sub‑gradient (i.e. those that increase the peak) to indices
  with small sub‑gradient.  Moves are accepted if they improve the exact
  objective, otherwise with a Metropolis probability that slowly cools.
  Interleaved sub‑gradient steps keep the search directed.

The best sequence found (as a plain Python list) is returned.  All
operations respect the constraints required by ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        # all non‑positive – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution via FFT (full length = 2·n‑1)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution a*a using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Exact objective for a vector normalised to sum‑1
def _objective_simplex(a: np.ndarray, n: int, L: int) -> float:
    """Return O = 2·n·max(conv) assuming sum(a)=1."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Gradient of the log‑sum‑exp (soft‑max) surrogate
def _smooth_gradient(a: np.ndarray, conv: np.ndarray, alpha: float, L: int) -> np.ndarray:
    """
    ∇_a ( (2·n/α)·log Σ_k e^{α·conv_k} ).
    This reduces to 2·corr(w, a) where w_k ∝ e^{α·conv_k}.
    """
    # soft‑max on the convolution entries
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)          # stability
    w = exp_scaled / exp_scaled.sum()

    # correlation w ⋆ a   (g_i = 2·Σ_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)    # length ≥ 3·n‑2
    n_len = a.size
    grad = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]
    return grad


# ----------------------------------------------------------------------
# Sub‑gradient of the exact max‑convolution entry
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a.
    For each maximising index k we add 2·a_{k‑i} to the component i.
    """
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)   # possibly several maximisers
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        # each pair (i,j) contributes 2*a[j] to gradient at i
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size > 0:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
# Exact line‑search step for a given sub‑gradient
def _line_search_step(a: np.ndarray, g: np.ndarray,
                     cur_obj: float, n: int, L: int,
                     eta_init: float = 0.2, min_eta: float = 1e-5) -> tuple:
    """
    Perform back‑tracking line‑search on the sub‑gradient direction.
    Returns (a_new, new_obj, conv_new, eta_used).  If no improvement is
    found, returns the original a.
    """
    eta = eta_init
    while eta >= min_eta:
        a_candidate = a - eta * g
        a_candidate = _project_to_simplex(a_candidate)
        conv = _conv_fft(a_candidate, L)
        new_obj = 2.0 * n * conv.max()
        if new_obj < cur_obj:
            return a_candidate, new_obj, conv, eta
        eta *= 0.5
    return a, cur_obj, None, 0.0


# ----------------------------------------------------------------------
# Gradient‑guided stochastic mass‑transfer move
def _mass_transfer_move(a: np.ndarray, g: np.ndarray,
                       max_frac: float = 0.18) -> np.ndarray:
    """
    Transfer a fraction of mass from a high‑gradient index to a low‑gradient
    one.  The result is projected back to the simplex.
    """
    n = a.size
    # source probabilities proportional to gradient magnitude
    src_tot = g.sum()
    if src_tot == 0:
        src_prob = np.full(n, 1.0 / n)
    else:
        src_prob = g / src_tot

    i = np.random.choice(n, p=src_prob)

    # target probabilities favour low gradient entries
    inv = np.max(g) - g
    inv_tot = inv.sum()
    if inv_tot == 0:
        tgt_prob = np.full(n, 1.0 / n)
    else:
        tgt_prob = inv / inv_tot

    j = np.random.choice(n, p=tgt_prob)

    if i == j or a[i] < 1e-15:
        return a  # nothing to do

    delta = a[i] * np.random.rand() * max_frac
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    a_new = _project_to_simplex(a_new)
    return a_new


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that minimizes
    the evaluation function ``evaluate_sequence``.  Returns the best sequence
    as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 0.2

    # ------------------------------------------------------------------
    # Choose problem size and initialise a
    if 'height_sequence_1' in globals():
        base = np.array(height_sequence_1, dtype=float)
        n = base.size
        a = base / base.sum()
        print(f"[init] using supplied sequence, n={n}")
    else:
        # Larger n tends to give a better bound (still cheap enough for FFT)
        n = 8000
        a = np.full(n, 1.0 / n)   # uniform start on the simplex
        print(f"[init] no supplied sequence – using uniform, n={n}")

    # FFT length – power of two covering correlation length
    L = 1 << ((3 * n - 2).bit_length())

    # ------------------------------------------------------------------
    # Stage 1 – Adam on the smoothed log‑sum‑exp surrogate
    best_seq = a.copy()
    best_score = _objective_simplex(best_seq, n, L)
    print(f"[stage‑1] start O={best_score:.6f}")

    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    lr0 = 0.6                     # initial learning rate (Euclidean)
    alpha_start = 5.0
    alpha_end = 20000.0
    max_iter_phase1 = 250_000     # upper bound, actual stop by time

    phase1_deadline = start_time + budget_s * 0.35   # ~35 % of budget

    it = 0
    while time.time() < phase1_deadline and it < max_iter_phase1:
        # Full convolution and current objective
        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[stage‑1] iter {it:,} new best O={best_score:.6f}")

        # Linear schedule for α
        frac = it / max_iter_phase1
        alpha = alpha_start + (alpha_end - alpha_start) * frac

        # Smooth gradient
        grad = _smooth_gradient(a, conv, alpha, L)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        lr = lr0 / math.sqrt(it + 1)   # simple 1/√t decay

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape plateaus
        if np.random.rand() < 0.001:
            a_new += np.random.randn(n) * 4e-4

        # projection onto simplex
        a_new = _project_to_simplex(a_new)

        # prune the tiniest tail every few thousand steps
        if it % 5000 == 0 and it > 0:
            thresh = np.percentile(a_new, 0.5)   # drop lowest 0.5 %
            a_new[a_new < thresh] = 0.0
            a_new = _project_to_simplex(a_new)

        a = a_new
        it += 1

    # ------------------------------------------------------------------
    # Stage 2 – exact sub‑gradient descent with back‑tracking line search
    print(f"[stage‑2] start, best O={best_score:.6f}")
    a = best_seq.copy()
    conv = _conv_fft(a, L)
    cur_obj = 2.0 * n * conv.max()
    phase2_deadline = start_time + budget_s * 0.65   # up to ~65 % total

    it2 = 0
    while time.time() < phase2_deadline and it2 < 150_000:
        g = _subgradient_max_conv(a, conv)

        # line‑search step
        a_new, new_obj, conv_new, eta_used = _line_search_step(
            a, g, cur_obj, n, L,
            eta_init=0.15, min_eta=1e-7
        )

        if new_obj < cur_obj:
            a = a_new
            conv = conv_new
            cur_obj = new_obj
            if cur_obj < best_score:
                best_score = cur_obj
                best_seq = a.copy()
                print(f"[stage‑2] iter {it2:,} new best O={best_score:.6f} (η={eta_used:.2e})")
        else:
            # no improvement – fall back to a tiny Adam‑style jitter
            jitter = np.random.randn(n) * 2e-4
            a = _project_to_simplex(a + jitter)
            conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max()

        it2 += 1

    # ------------------------------------------------------------------
    # Stage 3 – gradient‑guided stochastic refinement
    print(f"[stage‑3] start, best O={best_score:.6f}")
    temperature0 = 0.04
    temperature = temperature0
    max_iter_local = 120_000
    it3 = 0
    while time.time() < deadline and it3 < max_iter_local:
        # compute sub‑gradient for the current state
        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        g = _subgradient_max_conv(a, conv)

        # mass‑transfer move guided by the gradient
        a_candidate = _mass_transfer_move(a, g, max_frac=0.15)
        conv_cand = _conv_fft(a_candidate, L)
        cand_obj = 2.0 * n * conv_cand.max()

        # Metropolis acceptance
        if cand_obj < cur_obj or np.random.rand() < math.exp((cur_obj - cand_obj) / max(temperature, 1e-12)):
            a = a_candidate
            cur_obj = cand_obj
            conv = conv_cand
            if cur_obj < best_score:
                best_score = cur_obj
                best_seq = a.copy()
                tnow = time.time() - start_time
                print(f"[{tnow:6.1f}s] new global best O={best_score:.6f} (iter={it3})")

        # every few iterations perform a deterministic sub‑gradient step
        if it3 % 9 == 0:
            eta = 0.07 * (1.0 - it3 / max_iter_local)   # linear cooling
            a_step = a - eta * g
            a_step = _project_to_simplex(a_step)
            conv_step = _conv_fft(a_step, L)
            step_obj = 2.0 * n * conv_step.max()
            if step_obj < best_score:
                best_score = step_obj
                best_seq = a_step.copy()
                print(f"[{time.time() - start_time:6.1f}s] sub‑grad improvement O={best_score:.6f}")
            if step_obj < cur_obj:
                a = a_step
                conv = conv_step
                cur_obj = step_obj

        # temperature decay
        temperature *= 0.99995
        it3 += 1

    # ------------------------------------------------------------------
    # Final sanity check with the provided evaluator
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:
        print("[final] evaluate_sequence failed:", e)

    return best_seq.tolist()
