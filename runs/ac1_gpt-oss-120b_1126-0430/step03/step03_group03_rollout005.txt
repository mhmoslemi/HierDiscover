Implementation details for _smooth_gradient:

```
def _smooth_gradient(a, conv, alpha, L):
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / exp_scaled.sum()
    # Compute correlation w ⋆ a: g_i = 2* Σ_j w_{i+j} a_j
    # This is convolution of w with a (same length?), but correlation is reversed a.
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n = a.size
    # central n entries correspond to correlation
    grad = 2.0 * corr_rev[n-1 : n-1 + n]
    return grad
```

"""
Propose a non‑negative step‑function sequence that (approximately) minimises

    O(a) = 2·n·max(conv(a,a))   with   Σ a_i = 1

Algorithm (≈1000 s budget)
------------------------------------
1. Initialise from a previously found sequence (`height_sequence_1`) if it exists,
   otherwise from a uniform distribution.  The vector is projected onto the
   probability simplex.

2. Phase A – multi‑restart Adam on a smooth log‑sum‑exp surrogate.
   * For each restart we perturb the current best vector, run a short Adam
     optimisation of f_α(a) = (2·n/α)·log∑_k exp(α·conv_k) and linearly anneal
     α from 10 to 5000.  After each inner iteration the vector is projected onto
     the simplex.  The best vector over all restarts is kept.

3. Phase B – deterministic sub‑gradient descent with back‑tracking line‑search.
   * The exact sub‑gradient of max(conv) is computed.
   * An adaptive step size (initially 0.5) is reduced by a factor 0.5 until a
     decrease in the true objective O is observed, then the step size is
     slightly increased for the next iteration.  Every 1000 iterations a tiny
     Gaussian jitter is added to escape plateaus.

4. Phase C – guided mass‑transfer refinement.
   * Let K be the set of convolution indices attaining the maximum.
   * For a randomly chosen shift k∈K we compute the contribution of each
     coefficient to that shift.  The index with the largest contribution is
     decreased while the index with the smallest contribution is increased,
     keeping the sum equal to 1.  The move is accepted only if it improves O.
   * Occasionally a completely random transfer is attempted (Metropolis
     acceptance) to provide global exploration.

5. The best vector found is finally evaluated with the supplied
   `evaluate_sequence` function and returned as a plain Python list.

All heavy operations (convolution, gradient) use FFTs, and every update is
projected onto the simplex to guarantee feasibility.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        # all non‑positive – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full self‑convolution (length 2·n‑1)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return a*a (full convolution) using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Exact objective for a simplex‑normalised vector
def _objective(a: np.ndarray, n: int, L: int) -> float:
    """2·n·max(conv(a,a)) – a is assumed to sum to 1."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Sub‑gradient of the maximal convolution entry (used in Phase B/C)
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Return the sub‑gradient of max(conv) w.r.t. a."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)          # indices achieving the maximum
    g = np.zeros_like(a)
    # average contributions over all maximising shifts
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        # each pair (i,j) contributes a_j to the derivative w.r.t a_i (and vice‑versa)
        g[i_idx] += 2.0 * a[j_idx]
    g /= len(ks)
    return g


# ----------------------------------------------------------------------
# Gradient of the smooth log‑sum‑exp surrogate (Phase A)
def _smooth_gradient(a: np.ndarray, conv: np.ndarray, alpha: float, L: int) -> np.ndarray:
    """
    Compute ∇_a ( (2·n/α)·log∑_k e^{α·conv_k} ).
    Result = 2·corr(w, a) where w_k ∝ e^{α·conv_k}.
    """
    # soft‑max weights (stable)
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / exp_scaled.sum()

    # correlation w ⋆ a   (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)   # length ≥ 3·n‑2
    n = a.size
    grad = 2.0 * corr_rev[n - 1 : n - 1 + n]          # central n entries
    return grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence that minimises the evaluation function.
    Returns the sequence as a list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5        # safety margin for final steps

    # ------------------------------------------------------------------
    # Initialise a (use previous best if present)
    if 'height_sequence_1' in globals():
        base = np.array(height_sequence_1, dtype=float)
        n = base.size
        a = base.copy()
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-12:
            a = np.full(n, 1.0 / n)
        else:
            a = a / a.sum()
        print(f"[init] Using provided sequence, n={n}")
    else:
        n = 4096
        a = np.full(n, 1.0 / n)
        print(f"[init] No prior sequence – start uniform, n={n}")

    # FFT length – must accommodate both convolution (2n‑1) and correlation (3n‑2)
    L = 1 << ((3 * n - 2).bit_length())   # next power of two

    # ------------------------------------------------------------------
    # Evaluate the initial solution
    conv = _conv_fft(a, L)
    best_a = a.copy()
    best_obj = _objective(best_a, n, L)
    print(f"[init] O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Phase A – multi‑restart Adam on the smooth surrogate
    # ----------------------------------------------------
    phase_a_deadline = start_time + budget_s * 0.30   # ~30 % of the budget
    max_restarts = 12
    restart = 0
    while time.time() < phase_a_deadline and restart < max_restarts:
        restart += 1
        # start from current best with a small random perturbation
        a_cur = best_a + 0.02 * np.random.randn(n)
        a_cur = np.maximum(a_cur, 0.0)
        a_cur = a_cur / a_cur.sum()
        # Adam state
        m = np.zeros_like(a_cur)
        v = np.zeros_like(a_cur)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        lr = 0.08
        inner_iters = 5000
        alpha_start, alpha_end = 10.0, 5000.0
        for it in range(inner_iters):
            if time.time() >= phase_a_deadline - 5.0:
                break
            # full convolution and surrogate gradient
            conv_cur = _conv_fft(a_cur, L)
            # actual objective for monitoring
            cur_obj = 2.0 * n * conv_cur.max()
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a_cur.copy()
                print(f"[phase‑A] restart={restart}, iter={it}, O={best_obj:.6f}")

            # linear annealing of α
            frac = it / inner_iters
            alpha = alpha_start + (alpha_end - alpha_start) * frac

            # smooth gradient
            grad = _smooth_gradient(a_cur, conv_cur, alpha, L)

            # Adam update (bias‑corrected)
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** (it + 1))
            v_hat = v / (1.0 - beta2 ** (it + 1))
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            # projected update
            a_next = a_cur - step
            a_cur = _project_to_simplex(a_next)

        # end of one restart
    # ------------------------------------------------------------------
    # Phase B – deterministic sub‑gradient descent with back‑tracking line‑search
    # ------------------------------------------------------------------
    phase_b_deadline = start_time + budget_s * 0.65   # up to ~65 % of total
    a_cur = best_a.copy()
    step_size = 0.5
    subgrad_iter = 0
    while time.time() < phase_b_deadline:
        if subgrad_iter % 1000 == 0 and subgrad_iter > 0:
            # tiny jitter to possibly escape tiny plateaus
            jitter = np.random.randn(n) * 3e-4
            a_cur = _project_to_simplex(a_cur + jitter)

        conv_cur = _conv_fft(a_cur, L)
        cur_obj = 2.0 * n * conv_cur.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a_cur.copy()
            print(f"[subgrad] iter={subgrad_iter}, O={best_obj:.6f}")

        # sub‑gradient of the maximal entry
        g = _subgradient_max_conv(a_cur, conv_cur)

        # back‑tracking line‑search
        eta = step_size
        while True:
            a_candidate = _project_to_simplex(a_cur - eta * g)
            conv_cand = _conv_fft(a_candidate, L)
            cand_obj = 2.0 * n * conv_cand.max()
            if cand_obj < cur_obj or eta < 1e-7:
                break
            eta *= 0.5

        # accept the move
        a_cur = a_candidate
        # adapt step size for the next iteration (slightly larger if we succeeded)
        step_size = max(eta * 1.2, 1e-7)

        subgrad_iter += 1
        if time.time() >= phase_b_deadline - 5.0:
            break

    # ------------------------------------------------------------------
    # Phase C – guided mass‑transfer refinement (local search)
    # ------------------------------------------------------------------
    phase_c_deadline = deadline
    a_cur = best_a.copy()
    local_obj = _objective(a_cur, n, L)
    move_iter = 0
    while time.time() < phase_c_deadline:
        conv_cur = _conv_fft(a_cur, L)
        max_val = conv_cur.max()
        ks = np.flatnonzero(conv_cur == max_val)
        # pick a random maximising shift (to add stochasticity)
        k = ks[np.random.randint(len(ks))]

        # contribution of each index to this shift
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        contrib = np.zeros(n)
        contrib[i_idx] = a_cur[j_idx]      # each i contributes a_j to the shift
        # we use the sub‑gradient magnitude (2*contrib) only for ordering
        i_high = i_idx[np.argmax(contrib[i_idx])]
        i_low = i_idx[np.argmin(contrib[i_idx])]

        # propose a deterministic transfer
        delta = min(a_cur[i_high] * 0.12, 5e-4)  # small fraction of the source mass
        if delta > 0:
            a_new = a_cur.copy()
            a_new[i_high] -= delta
            a_new[i_low] += delta
            a_new = _project_to_simplex(a_new)
            conv_new = _conv_fft(a_new, L)
            new_obj = 2.0 * n * conv_new.max()
        else:
            new_obj = local_obj
            a_new = a_cur

        # also occasionally try a random transfer (Metropolis style)
        if np.random.rand() < 0.03:
            i_rand, j_rand = np.random.randint(n, size=2)
            if i_rand != j_rand and a_cur[i_rand] > 0:
                delta_rand = a_cur[i_rand] * np.random.rand() * 0.15
                a_rand = a_cur.copy()
                a_rand[i_rand] -= delta_rand
                a_rand[j_rand] += delta_rand
                a_rand = _project_to_simplex(a_rand)
                conv_rand = _conv_fft(a_rand, L)
                obj_rand = 2.0 * n * conv_rand.max()
                # accept if improvement or with Metropolis probability
                if obj_rand < local_obj or np.random.rand() < math.exp((local_obj - obj_rand) / max(0.01, (move_iter+1)*1e-4)):
                    a_new = a_rand
                    new_obj = obj_rand

        # acceptance
        if new_obj < local_obj:
            a_cur = a_new
            local_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_a = a_cur.copy()
                print(f"[mass‑transfer] iter={move_iter}, O={best_obj:.6f}")

        move_iter += 1
        if time.time() >= phase_c_deadline - 1.0:
            break

    # ------------------------------------------------------------------
    # Final sanity check with the supplied evaluation function
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:
        print("[final] evaluate_sequence failed:", e)

    return best_a.tolist()
