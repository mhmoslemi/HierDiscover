def gradient_logsumexp(a, alpha):

    n = len(a)
    # compute conv = a*a (full)
    fft_len_conv = _fft_len_for_n(2 * n - 1)  # maybe same.
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[:2*n-1]

    # compute weights
    max_conv = np.max(conv)
    # to avoid overflow: subtract max
    w = np.exp(alpha * (conv - max_conv))
    w_sum = np.sum(w)
    if w_sum == 0:
        w_norm = np.full_like(w, 1.0 / w.size)
    else:
        w_norm = w / w_sum

    # compute gradient via correlation of w_norm and a
    # convolution of w_norm with a_rev.
    # Determine fft size
    fft_len_corr = 1 << ((len(w_norm) + n - 1).bit_length())
    fa_w = np.fft.rfft(w_norm, fft_len_corr)
    fa_rev = np.fft.rfft(a[::-1], fft_len_corr)
    corr_full = np.fft.irfft(fa_w * fa_rev, fft_len_corr)[:len(w_norm)+n-1]
    grad = 2.0 * corr_full[n-1:n-1+n]  # length n
    return conv, grad

#!/usr/bin/env python
"""
Propose a candidate step‑function (non‑negative heights) that minimizes
the provided ``evaluate_sequence``.  The optimisation works on the
probability simplex (∑a = 1) and uses a smooth approximation of the
max‑convolution objective (log‑sum‑exp) together with Adam on the log‑it
parameters.  A simple schedule raises the temperature (α) so the
optimiser first sees a smooth landscape and later concentrates on the
true maximum.  The support size is enlarged progressively (up to 20 000)
by linear interpolation of the best distribution found so far.
The final sequence is rescaled to the required ℓ1‑norm
√(2 n) before being handed to ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector that sums to 1)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it representation that yields ``a`` after ``softmax``."""
    return np.log(a + eps)


def _fft_len_for_n(m: int) -> int:
    """Smallest power‑of‑two ≥ ``m`` (efficient size for real FFT)."""
    return 1 << ((m - 1).bit_length())


def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """
    Full (linear) self‑convolution a*a using a real FFT.
    Returns a length ``2·len(a)‑1`` array.
    """
    n = a.shape[0]
    fft_len = _fft_len_for_n(2 * n - 1)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def _logsumexp_gradient(a: np.ndarray, alpha: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Compute the full self‑convolution ``c = a*a`` and the gradient of the
    smoothed objective ``F(a)= (1/α)·log Σ exp(α·c_k)``.
    Returns the convolution vector ``c`` and the gradient ∂F/∂a (size len(a)).
    """
    n = a.shape[0]

    # ----- convolution -----
    c = _conv_full_fft(a)                     # length 2n-1

    # ----- soft‑max weights for the convolution entries -----
    c_max = np.max(c)
    # shift to avoid overflow (c - c_max ≤ 0)
    w = np.exp(alpha * (c - c_max))
    w_sum = np.sum(w)
    if w_sum == 0.0:
        w_norm = np.full_like(w, 1.0 / w.size)
    else:
        w_norm = w / w_sum

    # ----- gradient:  g_i = 2·Σ_k w_k·a_{k‑i}  (cross‑correlation) -----
    # Correlation can be computed as convolution of w with the reversed a.
    rev_a = a[::-1]
    len_w = w_norm.shape[0]                     # 2n‑1
    fft_len = _fft_len_for_n(len_w + n - 1)     # size for correlation
    fw = np.fft.rfft(w_norm, fft_len)
    fa_rev = np.fft.rfft(rev_a, fft_len)
    corr_full = np.fft.irfft(fw * fa_rev, fft_len)[: len_w + n - 1]
    # extract the n needed entries (centered)
    grad = 2.0 * corr_full[n - 1 : n - 1 + n]   # size n
    return c, grad


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    total = new_a.sum()
    if total == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= total
    return new_a


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """Normalised power‑law a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _random_simplex(n: int) -> np.ndarray:
    """Draw a random point on the n‑dimensional probability simplex."""
    x = np.random.rand(n)
    s = x.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return x / s


# ----------------------------------------------------------------------
# Optimisation routine (Adam on the simplex, smooth max)
# ----------------------------------------------------------------------


def _optimize(
    a_start: np.ndarray,
    deadline: float,
    max_iters: int = 1_000_000,
    max_alpha: float = 40.0,
    start_alpha: float = 2.0,
    alpha_ramp_iters: int = 30_000,
    base_lr: float = 0.6,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps_adam: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the probability simplex.
    Returns the best probability vector (sum=1) and the associated cheap score
    (which equals the final evaluation after the ℓ1‑scaling).
    """
    n = a_start.shape[0]
    a = a_start.copy()
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0.0:
        a[:] = 1.0 / n
    else:
        a /= a_sum

    # logits (log‑it) representation
    z = _logits_from_distribution(a)

    # Adam state
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    # bookkeeping
    best_a = a.copy()
    # initial cheap score
    c0 = _conv_full_fft(a)
    best_score = 2.0 * n * np.max(c0)

    it = 0
    last_improve = 0

    while time.time() < deadline and it < max_iters:
        it += 1

        # ----- forward pass -----
        a = _softmax(z)                      # stay on the simplex
        alpha = start_alpha + (max_alpha - start_alpha) * min(1.0, it / alpha_ramp_iters)
        conv, grad_a = _logsumexp_gradient(a, alpha)

        cheap_score = 2.0 * n * np.max(conv)

        # ----- record best -----
        if cheap_score < best_score:
            best_score = cheap_score
            best_a = a.copy()
            last_improve = it
            if it % 2000 == 0:
                print(f"[Iter {it:07d}] new best cheap = {cheap_score:.6f}")

        # ----- gradient in log‑it space (mirror descent on simplex) -----
        mu = np.dot(grad_a, a)
        grad_z = a * (grad_a - mu)

        # ----- Adam updates (bias‑corrected) -----
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning‑rate schedule (inverse‑sqrt decay)
        lr_t = base_lr / (math.sqrt(it) + 1.0)

        # ----- parameter update -----
        z -= lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)
        # keep exponentials in a safe range
        z = np.clip(z, -50.0, 50.0)

        # ----- occasional random perturbation (helps escape plateaus) -----
        if it % 5000 == 0 and np.random.rand() < 0.2:
            z += 0.015 * np.random.randn(n)
            m *= 0.5
            v *= 0.5

        # ----- early stopping (if stagnation for a long time) -----
        if it - last_improve > 250_000 and time.time() < deadline - 30:
            # no improvement for many iterations – keep moving with a smaller step
            base_lr *= 0.7
            last_improve = it  # reset counter to avoid immediate repeat

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) beats the target bound.
    The routine respects the given time budget (≈1000 s) and uses an Adam
    optimiser on a smoothed max‑convolution objective.  The support size is
    progressively increased up to 20 000 entries.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0  # small safety margin

    # ---------------------------------------------------------------
    # 0. Determine initial support size and initial distribution.
    # ---------------------------------------------------------------
    try:
        # If a previous construction is present, start from it.
        init_global = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        n0 = max(2000, init_global.shape[0])
        a0 = np.maximum(init_global, 0.0)
        s = a0.sum()
        if s == 0.0:
            a0 = np.full_like(a0, 1.0 / a0.shape[0])
        else:
            a0 /= s
        use_global = True
        print(f"Starting from global variable, length = {n0}")
    except Exception:
        init_global = None
        use_global = False
        n0 = 5000
        a0 = None

    # If no global start, scan a few power‑law exponents and keep the best cheap score.
    if not use_global:
        p_grid = np.linspace(0.55, 0.70, 81)
        best_score = float("inf")
        best_a = None
        for p in p_grid:
            if time.time() >= deadline:
                break
            cand = _power_law_distribution(n0, p)
            conv = _conv_full_fft(cand)
            cheap = 2.0 * n0 * np.max(conv)
            if cheap < best_score:
                best_score = cheap
                best_a = cand
                best_p = p
        # fall back to a random simplex if everything failed
        if best_a is None:
            best_a = _random_simplex(n0)
            best_score = 2.0 * n0 * np.max(_conv_full_fft(best_a))
            best_p = None
        a0 = best_a
        print(f"Power‑law init: best p = {best_p}, cheap score = {best_score:.6f}")

    # ---------------------------------------------------------------
    # 1. Main optimisation loop with progressive support enlargement.
    # ---------------------------------------------------------------
    max_support = 20000                     # hard limit for support size
    current_n = n0
    current_a = a0.copy()
    best_overall_a = None
    best_overall_score = float("inf")

    while time.time() < deadline:
        # allocate a slice of the remaining time for the current support size
        time_left = deadline - time.time()
        phase_deadline = time.time() + min(time_left, 180.0)  # 3 min per phase max

        # optimise at the current support size
        best_a_phase, best_score_phase = _optimize(
            current_a, phase_deadline, max_iters=800_000, max_alpha=40.0, start_alpha=2.0
        )

        if best_score_phase < best_overall_score:
            best_overall_score = best_score_phase
            best_overall_a = best_a_phase.copy()
            print(
                f"[Phase] length={current_n} improved overall cheap = {best_overall_score:.6f}"
            )

        # decide whether to enlarge the support
        if current_n >= max_support:
            # nothing more to do
            break
        # ensure we have enough time for an enlargement step (interpolation + a few iterations)
        if time.time() + 30.0 > deadline:
            break

        new_n = min(max_support, int(current_n * 1.5) + 1)
        print(f"[Phase] expanding support: {current_n} → {new_n}")
        # interpolate the best distribution obtained so far to the larger size
        current_a = _interpolate_distribution(best_a_phase, new_n)
        current_n = new_n
        # reset the optimisation state for the new problem size
        # (the next loop iteration will call _optimize again)

    # ---------------------------------------------------------------
    # 2. Final scaling to the required ℓ1‑norm √(2 n) and exact evaluation.
    # ---------------------------------------------------------------
    if best_overall_a is None:
        # fallback – should never happen
        best_overall_a = _random_simplex(current_n)

    final_n = best_overall_a.shape[0]
    target_sum = math.sqrt(2.0 * final_n)
    final_seq = (best_overall_a * target_sum).tolist()

    # ---------------------------------------------------------------
    # 3. Log the exact score (the platform will also do this).
    # ---------------------------------------------------------------
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:  # defensive – should never happen
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # ---------------------------------------------------------------
    # 4. Return the constructed sequence.
    # ---------------------------------------------------------------
    return final_seq
