def propose_candidate(seed=42, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.3

    # Determine sequence length
    if 'height_sequence_1' in globals():
        base = np.array(height_sequence_1, dtype=float)
        n = base.size
        a = base / np.sum(base)  # simplex normalised
        print("[init] using supplied sequence, n=", n)
    else:
        n = 8000  # larger than baseline
        a = np.full(n, 1.0/n)
        print("[init] no supplied sequence – using uniform, n=", n)

    # compute L as power of two > 3n-2
    L = 1 << ((3*n - 2).bit_length())

    # Phase A: param search over beta (geometric shape)
    best_score = float('inf')
    best_seq = None
    # define beta range adaptively based on n
    beta_min = 0.0
    beta_max = 0.02 if n <= 6000 else 0.015
    steps = 300
    for s in range(steps):
        beta = beta_min + (beta_max - beta_min) * s / (steps - 1)
        # compute candidate
        exponents = beta * np.arange(n)
        # subtract max for stability
        exponents -= np.max(exponents)
        a_candidate = np.exp(exponents)
        a_candidate_sum = a_candidate.sum()
        a_candidate /= a_candidate_sum
        # compute objective using FFT
        conv = _conv_fft(a_candidate, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = a_candidate.copy()
            print(f"[phase-A] beta {beta:.6f} new best O={best_score:.6f}")

    # Phase A may return best_seq; else fallback to a.

    if best_seq is None:
        best_seq = a.copy()
        best_score = _objective_simplex(best_seq, n, L)

    # Now Phase B: Adam on surrogate
    a = best_seq.copy()
    best_seq = a.copy()
    best_score = _objective_simplex(a, n, L)
    print(f"[phase-B] start O={best_score:.6f}")

    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    lr0 = 0.8
    alpha_start = 5.0
    alpha_end = 8000.0

    # allocate portion of budget
    phaseB_deadline = start_time + budget_s * 0.6
    iter_B = 0
    max_iter_B = 400_000
    while time.time() < phaseB_deadline and iter_B < max_iter_B:
        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[phase-B] iter {iter_B} new best O={best_score:.6f}")

        # schedule alpha
        frac = iter_B / max_iter_B
        alpha = alpha_start + (alpha_end - alpha_start) * frac
        # smooth gradient
        grad = _smooth_gradient(a, conv, alpha, L)
        # Adam updates
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** (iter_B + 1))
        v_hat = v / (1 - beta2 ** (iter_B + 1))
        lr = lr0 / math.sqrt(iter_B + 1)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        # jitter
        if np.random.rand() < 0.001:
            a_new += np.random.randn(n) * 1e-4
        # projection
        a_new = _project_to_simplex(a_new)

            a = _project_to_simplex(a)

            i, j = np.random.randint(n, size=2)

                delta = a[i] * np.random.rand() * 0.15
                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta
                a_new = _project_to_simplex(a_new)
                conv_new = _conv_fft(a_new, L)
                new_obj = 2.0 * n * conv_new.max()

                if new_obj < cur_obj or np.random.rand() < math.exp((cur_obj - new_obj) / max(T, 1e-12)):

                        best_score = cur_obj
                        best_seq = a.copy()
                        print(f"[{time.time():.1f}] New global best O={best_score:.6f} (restart={max_restarts}, iter={it_loc})")
                else:
                    no_improve += 1
            else:
                no_improve += 1

            # subgradient step
            if it_loc % subgrad_interval == 0:

                g = _subgradient_max_conv(a, conv)
                eta = 0.06 * (1.0 - it_loc / max_iter_local)

                a = _project_to_simplex(a)
                conv = _conv_fft(a, L)
                cur_obj = 2.0 * n * conv.max()
                if cur_obj < best_score:
                    best_score = cur_obj
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] Sub‑grad improved O={best_score:.6f}")

            T *= 0.99995
            it_loc += 1

            if no_improve >= no_improve_limit:
                break

        max_restarts += 1
        if time.time() >= deadline:
            break

    # final evaluation
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:
        print(f"[final] evaluate_sequence failed: {e}")

    return best_seq.tolist()

"""
Search for a non‑negative step‑function sequence that minimizes the
evaluation objective

    O(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm consists of three stages:

1. **Exponential‑shape sweep** – a cheap 1‑D search over
   a_i ∝ exp(β·i) (β ≥ 0).  The best β provides a strong
   starting point that already beats the uniform baseline.

2. **Adam on a smooth surrogate** – minimise the log‑sum‑exp
   approximation of the max‑convolution using Adam with Euclidean
   projection onto the probability simplex.

3. **Stochastic local refinement** – random mass‑transfer moves
   with a Metropolis acceptance rule together with periodic
   exact sub‑gradient steps on the true max‑convolution.

All stages respect a global time budget (default 1000 s).  The
best sequence found is returned as a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper 1 – Euclidean projection onto the probability simplex
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto {x >= 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:                      # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper 2 – Full convolution via FFT (length 2·n‑1)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return the full convolution a*a using an FFT of size L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Helper 3 – Exact objective for a simplex‑normalised vector
def _objective_simplex(a: np.ndarray, n: int, L: int) -> float:
    """2·n·max(conv(a,a)) (a is assumed to sum to 1)."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Helper 4 – Sub‑gradient of the maximal convolution entry
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. a."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
# Helper 5 – Gradient of the log‑sum‑exp surrogate
def _smooth_gradient(a: np.ndarray, conv: np.ndarray, alpha: float, L: int) -> np.ndarray:
    """
    Gradient of   (2·n/α)·log Σ_k exp(α·conv_k) .
    Uses the identity ∇ = 2·corr(w, a) where w_k ∝ exp(α·conv_k).
    """
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)          # stability
    w = exp_scaled / exp_scaled.sum()

    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n_len = a.size
    grad = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]
    return grad

# Helper 6 – Fast evaluation of the exponential‑shape candidate
def _eval_exponential(beta: float, n: int, L: int) -> tuple[float, np.ndarray]:
    """Return (objective, normalised vector) for a_i ∝ exp(beta·i)."""
    # safe exponentiation
    exponents = beta * np.arange(n, dtype=np.float64)
    exponents -= np.max(exponents)                 # avoid overflow
    a = np.exp(exponents)
    a /= a.sum()
    obj = _objective_simplex(a, n, L)
    return obj, a

                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta
                a_new = _project_to_simplex(a_new)

                conv_new = _conv_fft(a_new, L)
                new_obj = 2.0 * n * conv_new.max()

                if (new_obj < cur_obj) or (np.random.rand() <
                        math.exp((cur_obj - new_obj) / max(T, 1e-12))):
                    a = a_new
                    conv = conv_new
                    cur_obj = new_obj
                    no_improve = 0
                    if cur_obj < best_score:
                        best_score = cur_obj
                        best_seq = a.copy()
                        print(f"[{time.time():.1f}] New global best O={best_score:.6f} "
                              f"(restart={max_restarts}, iter={it_loc})")
                else:
                    no_improve += 1
            else:
                no_improve += 1

            # ---- Exact sub‑gradient step on the current maximal entry ----------
            if it_loc % subgrad_interval == 0:
                g = _subgradient_max_conv(a, conv)
                eta = 0.07 * (1.0 - it_loc / max_iter_local)    # linear decay
                a = a - eta * g
                a = _project_to_simplex(a)
                conv = _conv_fft(a, L)
                cur_obj = 2.0 * n * conv.max()
                if cur_obj < best_score:
                    best_score = cur_obj
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] sub‑grad improved O={best_score:.6f}")

            # ---- Temperature decay ------------------------------------------------
            T *= 0.99995
            it_loc += 1

            # ---- Restart if stuck -------------------------------------------------
            if no_improve >= no_improve_limit:
                break

        max_restarts += 1
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Final polishing – small exponentiation / rotation tweaks
    # ------------------------------------------------------------------
    print("[final] polishing tweaks")
    for power in np.linspace(0.5, 1.5, 21):
        cand = np.power(best_seq, power)
        cand /= cand.sum()
        sc = _objective_simplex(cand, n, L)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[final] power {power:.3f} improved O={best_score:.6f}")

    # random circular shifts
    for _ in range(15):
        shift = np.random.randint(n)
        cand = np.roll(best_seq, shift)
        sc = _objective_simplex(cand, n, L)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[final] shift {shift} improved O={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check with the provided evaluator
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:
        print("[final] evaluate_sequence failed:", e)

    return best_seq.tolist()
