        O(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

    vector z (a_i = exp(z_i)/∑_j exp(z_j)).  This automatically enforces

    α is annealed from 5 to 5 000.  The gradient of f_α w.r.t. a is
    2·corr(w,a) with w_k ∝ e^{α·conv_k}; the chain‑rule gives the gradient
    w.r.t. z.  An Adam optimiser (bias‑corrected moments) is applied.
  • Occasionally a tiny random jitter is added to z and extremely small

  • Starts from the best candidate of Phase 1 (or a jittered version /

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex (kept for occasional use)
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Soft‑max parametrisation
def _softmax(z: np.ndarray) -> np.ndarray:
    """Return soft‑max of z (numerically stable)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# FFT‑based full self‑convolution (length 2·n‑1)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full convolution a*a using an FFT of length L (real‑FFT)."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Gradient of the log‑sum‑exp (smooth) surrogate wrt a
def _smooth_gradient(a: np.ndarray, conv: np.ndarray, alpha: float, L: int) -> np.ndarray:
    """∇_a ( (2·n/α)·log∑_k e^{α·conv_k} )."""
    # soft‑max weights w_k ∝ e^{α·conv_k}
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / exp_scaled.sum()

    # correlation w ⋆ a  (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n_len = a.size
    grad = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]
    return grad


# ----------------------------------------------------------------------
# Full gradient of the surrogate wrt the unconstrained vector z
def _grad_surrogate_z(z: np.ndarray, alpha: float, L: int):
    """Return (grad_z, a, conv) for the smooth surrogate."""
    a = _softmax(z)
    conv = _conv_fft(a, L)
    grad_a = _smooth_gradient(a, conv, alpha, L)
    dot = np.dot(a, grad_a)
    grad_z = a * (grad_a - dot)
    return grad_z, a, conv


# ----------------------------------------------------------------------
# Exact objective for a simplex‑normalised vector (sum(a)=1)
def _objective_simplex(a: np.ndarray, n: int, L: int) -> float:
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Sub‑gradient of the maximal convolution entry (as used in reference)
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) wrt a."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g

                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] Sub‑grad improved O={best_score:.6f}")

            # ----- temperature decay -----
            T *= 0.9999
            it_loc += 1

            # ----- stagnation check -----
            if no_improve >= no_improve_limit:
                break

        restart_count += 1
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Final sanity check via the provided evaluator
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence failed:", exc)

    # Return the best sequence as a plain list
    return best_seq.tolist()
