def solve_lp_for_f(f, M):
    n = f.shape[0]
    # Build constraints matrix A: each diagonal offset i has f[i]
    offsets = np.arange(n, dtype=np.int32)
    A = scipy.sparse.diags(f, offsets, shape=(2*n-1, n), format='csc')
    b_ub = np.full(2*n-1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)  # maximize sum

    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                                 method='highs')
    if not res.success:
        # fallback: return zero vector
        return np.zeros_like(f)
    return res.x

"""
Propose a step‐function sequence that minimises the evaluation metric.

Strategy
--------
1. Initialise a large Catalan distribution (a_i ∝ C(2i,i)/4^i, normalised to
   Σa_i = 1).  This distribution is known to be close to optimal for the
   continuous version of the problem.

2. Iteratively improve it by solving a *linear programme*:
      maximise   Σ b_j
      subject to b_j ≥ 0  and   (a * b)[k] ≤ M  for all k,
   where M = max_k (a * a)[k].  The constraints are linear in b, because the
   convolution (a * b) is a linear operator.  After the LP we obtain a vector
   `b` with L1‑norm larger than 1; we scale it back to a probability vector
   `g = b / Σb`.  The mixture `a ← (1‑t) a + t g` (followed by a simplex
   projection) provably does not increase M and usually reduces it.

3. After a few LP‑updates we switch to a cheap projected sub‑gradient /
   mirror‑descent optimiser (the same idea as the baseline code) to polish
   the solution when the LP stops making progress.

Both phases run as long as the time budget permits.  The code works with
`numpy`, `scipy.optimize` and `scipy.sparse`; if SciPy is unavailable the
fallback is the pure sub‑gradient optimiser.

The returned list consists of plain Python floats (the LP and the
optimiser keep the vector on the probability simplex, so the evaluation
function never returns ``inf``).
"""

import math
import numpy as np

# Helper: full linear convolution using a real FFT (fast, O(n log n))
# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray) -> np.ndarray:
    """Return the linear convolution a*a (full length 2·len(a)-1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Projection onto the probability simplex Σx = 1, x ≥ 0
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = 1}."""
    n = v.size
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)          # 1‑based indices
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # Numerical safety – return the uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]                # last true index
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Catalan distribution, normalised to Σ = 1
# ----------------------------------------------------------------------
def _catalan_normalised(n: int) -> np.ndarray:
    """First n Catalan‑type coefficients, normalised to sum 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)   # recurrence for C(2i,i)/4^i
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Objective used throughout: 2·n·max(a*a)   (identical to evaluate_sequence
# when Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    return float(2.0 * n * np.max(_fft_convolve(a)))


# ----------------------------------------------------------------------
# Linear‑programming step: given a (∑a = 1) and M = max(a*a), find a
# vector b maximising Σb under (a * b) ≤ M.
# ----------------------------------------------------------------------
def _solve_lp_for_a(a: np.ndarray, M: float):
    """Return the optimal b for the LP described in the docstring.
    If the LP fails, a zero vector is returned."""
    try:
        import scipy.optimize
        import scipy.sparse
    except Exception:                     # SciPy not available
        return np.zeros_like(a)

    n = a.shape[0]
    # Build the Toeplitz‑like constraint matrix: each diagonal offset i
    # contains the constant a[i].
    offsets = np.arange(n, dtype=np.int32)          # 0 … n‑1
    A = scipy.sparse.diags(a, offsets, shape=(2 * n - 1, n), format='csc')
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # maximise Σb  <=> minimise –Σb
    c = -np.ones(n, dtype=np.float64)

    bounds = [(0.0, None)] * n

    # The HiGHS solver (default) handles large sparse LPs efficiently.
    res = scipy.optimize.linprog(
        c,
        A_ub=A,
        b_ub=b_ub,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )
    if not res.success:
        # In the unlikely event of failure fall back to the zero vector.
        return np.zeros_like(a)

    return res.x


# ----------------------------------------------------------------------
# Sub‑gradient / mirror‑descent optimiser (the baseline method)
# ----------------------------------------------------------------------
def _subgradient_optimize(a0: np.ndarray, deadline: float,
                          max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Fine‑tune a probability vector using projected sub‑gradient descent."""
    a = a0.copy()
    n = a.shape[0]

    best = a.copy()
    best_val = _objective(best)

    # Heavy‑ball momentum parameters
    beta = 0.9
    momentum = np.zeros_like(a)

    t = 0
    while time.time() < deadline and t < max_iters:
        conv = _fft_convolve(a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active quadratic form (g = 2·J_k·a)
        grad = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        grad[idx] = 2.0 * a[k - idx]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # diminishing step size
        lr = 0.5 / math.sqrt(t + 1)

        a = a - lr * momentum
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best = a.copy()
            # occasional progress report
            if t % 10_000 == 0:
                print(f"[subgrad] iter {t:06d}  obj {best_val:.9f}")

        t += 1

    return best, best_val


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Return a list of non‑negative heights (Python floats) that minimises
    ``evaluate_sequence`` within the allotted time budget.
    """
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # 1) Problem size.  n≈2000 gives a good trade‑off between LP size and
    #    quality of the discrete approximation.
    # ------------------------------------------------------------------
    n = 2000

    # ------------------------------------------------------------------
    # 2) Initial candidate – Catalan distribution (near‑optimal analytically)
    # ------------------------------------------------------------------
    a = _catalan_normalised(n)

    # ------------------------------------------------------------------
    # 3) Time management
    # ------------------------------------------------------------------
    # Leave a small safety margin for the harness.
    deadline = time.time() + budget_s - 2.0

    best_a = a.copy()
    best_obj = _objective(best_a)
    print(f"[init] objective ≈ {best_obj:.9f}")

    # ------------------------------------------------------------------
    # 4) LP‑based refinement loop
    # ------------------------------------------------------------------
    lp_iter = 0
    while time.time() < deadline - 5.0 and lp_iter < 40:
        conv = _fft_convolve(a)
        M = conv.max()

        b = _solve_lp_for_a(a, M)
        sum_b = b.sum()
        if sum_b <= 1e-12:
            # Something went wrong – abort the LP phase.
            break

        # Scale b to a probability vector (if sum_b > 0)
        if sum_b > 1.0:
            g = b / sum_b          # scaling down – keeps feasibility
        else:
            g = b                  # sum_b ≤ 1, we keep it as is; renorm later

        # Attempt a *direct* replacement with g (if it improves the objective)
        g_norm = g / g.sum() if g.sum() > 0 else g
        obj_g = _objective(g_norm)
        if obj_g + 1e-12 < best_obj:
            a = g_norm
            best_a = a.copy()
            best_obj = obj_g
            print(f"[LP {lp_iter:02d}] direct replace improved obj {best_obj:.9f}")
        else:
            # Otherwise perform a modest convex blend and re‑project onto the simplex.
            t = 0.5 / math.sqrt(lp_iter + 1)   # decreasing step size
            a = (1.0 - t) * a + t * g_norm
            a = _project_simplex(a)
            cur_obj = _objective(a)
            if cur_obj + 1e-12 < best_obj:
                best_a = a.copy()
                best_obj = cur_obj
                print(f"[LP {lp_iter:02d}] blended improved obj {best_obj:.9f}")

        lp_iter += 1

    # ------------------------------------------------------------------
    # 5) Fine‑tuning with projected sub‑gradient descent (fallback if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        a_opt, obj_opt = _subgradient_optimize(best_a, deadline)
        if obj_opt < best_obj:
            best_a = a_opt
            best_obj = obj_opt
            print(f"[final subgrad] improved obj {best_obj:.9f}")

    # ------------------------------------------------------------------
    # 6) Final sanity print (the harness will also evaluate the sequence)
    # ------------------------------------------------------------------
    print(f"[final] best objective ≈ {best_obj:.9f} with n={n}")

    # Ensure we return plain Python floats.
    return best_a.tolist()
