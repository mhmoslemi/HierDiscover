"""
Propose a step‑function that minimises the bound

    O = 2·n·max(conv(a,a))      (with Σa = 1, a_i ≥ 0)

The algorithm is a *smoothed* exponentiated‑gradient descent.
It replaces the non‑smooth max‑operator by a temperature‑controlled
log‑sum‑exp (soft‑max) and follows the gradient of the smooth surrogate.
A decreasing temperature (σ) lets the method gradually focus on the true
maximum while the learning‑rate η is decayed as well.  The update is
performed in log‑space (multiplicative mirror descent) and the vector is
renormalised, which automatically enforces the simplex constraints.
If progress stalls a small random jitter is added to escape flat regions.
The whole optimisation runs until the allocated time budget (≈1000 s)
is exhausted.  Finally the candidate sequence is handed to the supplied
``evaluate_sequence`` function for the exact score.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small utility returning the next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Compute the unsmoothed bound 2·n·max(conv(a,a)) for a
    already normalised (Σa = 1).  Uses a small FFT for speed.
    """
    n = a.size
    size = _next_pow2(2 * n - 1)
    A = np.fft.fft(a, n=size)
    conv = np.fft.ifft(A * A).real[: 2 * n - 1]
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
def _smooth_optimize(a0: np.ndarray,
                     max_iters: int,
                     eta0: float,
                     eta_decay: float,
                     sigma0: float,
                     sigma_decay: float,
                     sigma_min: float,
                     deadline: float) -> tuple[np.ndarray, float]:
    """
    Smoothed exponentiated‑gradient descent.

    a0            – starting point (already normalised)
    max_iters     – hard iteration limit (the loop also respects the deadline)
    eta0, eta_decay
                  – initial learning rate and geometric decay factor
    sigma0, sigma_decay, sigma_min
                  – temperature schedule for the soft‑max
    deadline      – Unix timestamp when the algorithm must stop
    Returns the best vector found and its (unsmoothed) objective value.
    """
    a = a0.copy()
    n = a.size
    # FFT size sufficient for both the convolution a*a (length 2n-1) and the
    # correlation a ⋆ w (length 3n-2)
    L = _next_pow2(3 * n - 2)

    best_a = a.copy()
    best_score = _objective(a)

    no_improve = 0
    patience = max(10, max_iters // 20)       # when to inject jitter

    eta = eta0
    sigma = sigma0

    for t in range(max_iters):
        if time.time() >= deadline:
            break

        # ---------- convolution a*a ----------
        A = np.fft.fft(a, n=L)
        conv_full = np.fft.ifft(A * A).real[: 2 * n - 1]

        cur_score = 2.0 * n * conv_full.max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            print(f"[smooth] iter {t:6d} new best O={best_score:.6f} "
                  f"sigma={sigma:.6f} eta={eta:.6f}")
        else:
            no_improve += 1
            if no_improve >= patience:
                # ─── small jitter to escape a flat region ───
                jitter = np.random.randn(n) * 1e-4
                a = best_a + jitter
                a = np.clip(a, 0.0, None)
                a /= a.sum()
                # reset schedules a little to give the jitter a chance
                sigma = sigma0
                eta = eta0
                no_improve = 0
                continue

        # ---------- soft‑max weights ----------
        maxc = conv_full.max()
        # stable soft‑max: subtract the maximum before exponentiation
        exp_vals = np.exp((conv_full - maxc) / sigma)
        w = exp_vals / exp_vals.sum()                     # w_k ≥ 0, Σw = 1

        # ---------- correlation a ⋆ w ----------
        # correlation r_i = Σ_j a_j w_{i+j}  (i = 0..n‑1)
        w_rev = w[::-1]
        W_rev = np.fft.fft(w_rev, n=L)
        corr_full = np.fft.ifft(A * W_rev).real
        # indices (2n‑2) … (n‑1) give the needed n values
        idx = (2 * n - 2) - np.arange(n)
        corr = corr_full[idx]                # shape (n,)

        # ---------- gradient of the smoothed objective ----------
        # O_σ = 2·n·σ·log Σ exp(conv/σ)
        # ∇O_σ = 4·n·(a ⋆ w)            (see derivation in analysis)
        grad = 4.0 * n * corr

        # ---------- multiplicative (exponentiated) update ----------
        log_a = np.log(a + 1e-15)            # avoid log(0)
        log_a -= eta * grad
        # keep numbers moderate
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()

        # ---------- schedule update ----------
        sigma = max(sigma_min, sigma * sigma_decay)
        eta *= eta_decay

    return best_a, best_score
