"""
Propose a non‑negative step‑function sequence that minimizes  

    O(a) = 2·n·max(conv(a,a)) / (sum(a)^2)  

The algorithm proceeds in three stages:

1️⃣ **Smart initialisation** – generate a few parametric families
   (uniform, cosine, sine‑squared, symmetric Beta(α,α) for many α)
   and keep the one with the smallest surrogate objective.

2️⃣ **Phase 1 – Adam on a smooth log‑sum‑exp surrogate** –  
   minimise the differentiable surrogate  
   fα(a) = (2·n/α)·log∑k e^{α·conv_k}  
   using Adam with an exponentially increasing smoothing parameter α.
   After each step we project onto the probability simplex.

3️⃣ **Phase 2 – Stochastic refinement** –  
   repeated random mass‑transfer moves (Metropolis acceptance) combined
   with exact sub‑gradient steps on the current maximal convolution entry.
   The temperature is slowly annealed and occasional restarts with jitter
   keep the search diverse.

All heavy work (convolutions, gradients) is performed with FFTs.
The function respects the global time‑budget (default 1000 s) and returns
the best sequence found as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        # all non‑positive – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full self‑convolution (length 2·n‑1)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return a*a (full convolution) using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Exact objective for a simplex‑normalised vector
def _objective_simplex(a: np.ndarray, n: int, L: int) -> float:
    """2·n·max(conv(a,a)) (a is assumed to sum to 1)."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Sub‑gradient of the maximal convolution entry (as used in the reference code)
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Return the sub‑gradient of max(conv) w.r.t. a."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)          # possibly several maximisers
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size > 0:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
# Gradient of the log‑sum‑exp (soft‑max) surrogate of the max
def _smooth_gradient(a: np.ndarray, conv: np.ndarray, alpha: float, L: int) -> np.ndarray:
    """
    Compute ∇_a ( (2·n/α)·log∑_k e^{α·conv_k} ).
    The formula reduces to 2·corr(w, a) where w_k ∝ e^{α·conv_k}.
    """
    # soft‑max weights
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)          # numerical stability
    w = exp_scaled / exp_scaled.sum()

    # correlation w ⋆ a   (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)   # length ≥ 3·n‑2
    n_len = a.size
    grad = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]
    return grad


# ----------------------------------------------------------------------
# Parametric family: symmetric Beta(α,α) density discretised on a grid
def _beta_distribution(alpha: float, n: int) -> np.ndarray:
    """
    Return a discrete probability vector of length n proportional to
    x^{α-1} (1-x)^{α-1} on (0,1), where x = (i+0.5)/n.
    """
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    # work in log‑space to avoid overflow when α<1
    log_pdf = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    # shift for numerical stability
    log_pdf -= np.max(log_pdf)
    pdf = np.exp(log_pdf)
    pdf_sum = pdf.sum()
    if pdf_sum == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= pdf_sum
    return pdf

    Returns a plain list of non‑negative floats.

    L = 1 << ((3 * n - 2).bit_length())

    # cosine‑shaped bump (zero at ends, peak in middle)

    obj_sin = _objective_simplex(a_sin, n, L)

    # --------------------------------------------------------------
    # 2️⃣ Phase 1 – Adam optimiser on the smooth surrogate
    a = best_seq.copy()
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    alpha_start = 5.0
    alpha_end = 20000.0

    while time.time() < deadline:
