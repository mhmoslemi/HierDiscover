def smooth_gradient(a, conv, alpha, L):
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / exp_scaled.sum()
    # correlation
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n = a.size
    grad = 2.0 * corr_rev[n-1 : n-1 + n]
    return grad
```

"""
Optimisation of a non‑negative step‑function sequence for the evaluation function

    O(a) = 2·n·max(conv(a,a)) / (∑ a_i)² ,

where conv is the full linear convolution.  The objective is scale‑invariant,
so we constrain the sequence to lie on the probability simplex (∑ a_i = 1).

The algorithm works in three stages:

1. **Coarse optimisation (small size).**
   The problem is first solved for a modest length (≈2048) using
   Adam on a smooth log‑sum‑exp surrogate of the max‑convolution.
   The surrogate gradually tightens (α from ≈1 to ≈20000) so that the
   gradient eventually points at the true max‑convolution entry.
   After each Adam step the vector is projected onto the simplex, which
   automatically creates sparsity.

2. **Upscaling and fine‑tuning (large size).**
   The best coarse solution is interpolated to a larger target size
   (default 8192, or the length of a provided `height_sequence_1` if larger).
   The same Adam optimiser is run again, but for far fewer iterations,
   now on the full‑size problem.

3. **Deterministic sub‑gradient refinement.**
   Starting from the best vector obtained so far, a classic sub‑gradient
   method on the exact max‑convolution is applied with a linearly decaying
   stepsize.  This removes the remaining bias introduced by the smooth
   surrogate.

The whole routine respects a 1000 s wall‑clock budget and prints periodic
progress reports.  The final sequence is returned as a plain Python list
and is also evaluated with the supplied `evaluate_sequence` for sanity.

No file or network I/O is performed; only NumPy and the standard library
are used.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        # all non‑positive – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full linear convolution (length 2·n‑1)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return a * a (full linear convolution) using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Smooth surrogate gradient (log‑sum‑exp approximation of max)
def _smooth_gradient(a: np.ndarray, conv: np.ndarray, alpha: float, L: int) -> np.ndarray:
    """
    Compute ∇_a ( (2·n/α)·log ∑_k exp(α·conv_k) ).
    Result = 2·corr(w, a) where w_k ∝ exp(α·conv_k).
    """
    # soft‑max weights (stable)
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / exp_scaled.sum()          # length = 2·n‑1

    # correlation w ⋆ a  (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)   # length ≥ 3·n‑2
    n = a.size
    grad = 2.0 * corr_rev[n - 1 : n - 1 + n]          # central n entries
    return grad


# ----------------------------------------------------------------------
# Exact sub‑gradient of the maximal convolution entry
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Return a sub‑gradient of max(conv) w.r.t. a."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)          # indices achieving the maximum
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]                # each ordered pair contributes symmetrically
    g /= len(ks)                                   # average if several maximisers
    return g


# ----------------------------------------------------------------------
# Exact objective (max‑convolution) for a simplex‑normalised vector
def _objective_simplex(a: np.ndarray, L: int) -> float:
    """Return O = 2·n·max(conv(a,a)) (a already sums to 1)."""
    n = a.size
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Simple linear interpolation up‑sampling
def _upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate a probability vector to a larger length."""
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    up = np.interp(x_new, x_old, seq)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s == 0.0:
        up[:] = 1.0 / new_n
    else:
        up /= s
    return up


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function sequence minimizing the provided
    evaluation function.  Returns the best sequence found as a list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # tiny safety margin

    # ------------------------------------------------------------------
    # 1) Choose a target size and initialise the coarse problem
    if 'height_sequence_1' in globals():
        prior = np.array(height_sequence_1, dtype=float)
        prior = np.maximum(prior, 0.0)
        prior_sum = prior.sum()
        if prior_sum < 1e-12:
            prior = np.full_like(prior, 1.0 / prior.size)
        else:
            prior /= prior_sum
        target_n = max(8192, prior.size)       # ensure a decently large problem
        print(f"[init] Using provided sequence, target_n={target_n}")
    else:
        prior = None
        target_n = 8192
        print(f"[init] No prior sequence – target_n={target_n}")

    # Coarse problem size (powers of two are convenient)
    coarse_n = 2048
    if coarse_n > target_n:
        coarse_n = target_n // 2
        coarse_n = max(512, coarse_n)

    # Initialise coarse vector (interpolate prior if available)
    if prior is not None:
        a_coarse = _upsample_sequence(prior, coarse_n)
    else:
        a_coarse = np.full(coarse_n, 1.0 / coarse_n)

    # FFT length for coarse problem (must handle correlation of length 3·n‑2)
    def _fft_len(n):
        return 1 << ((3 * n - 2).bit_length())   # next power of two
    L_coarse = _fft_len(coarse_n)

    # ------------------------------------------------------------------
    # 2) Phase‑1: Adam on smooth surrogate (coarse size)
    max_iter_phase1 = 250_000
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.5
    # α schedule: start small, finish large (helps the optimiser see a smooth landscape first)
    alpha_start, alpha_end = 1.0, 20000.0

    m = np.zeros_like(a_coarse)
    v = np.zeros_like(a_coarse)

    best_a = a_coarse.copy()
    best_obj = _objective_simplex(best_a, L_coarse)

    phase1_deadline = start_time + budget_s * 0.35   # allocate ~35 % of time

    for it in range(max_iter_phase1):
        if time.time() >= phase1_deadline - 5.0:
            break

        # Full convolution and exact objective (tiny overhead, useful for monitoring)
        conv = _conv_fft(a_coarse, L_coarse)
        cur_obj = 2.0 * coarse_n * conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a_coarse.copy()
            print(f"[phase‑1] it={it:06d}, O={best_obj:.6f}")

        # Anneal α linearly (could use any schedule – linear works fine)
        frac = it / max_iter_phase1
        alpha = alpha_start + (alpha_end - alpha_start) * frac

        # Smooth gradient of the surrogate loss
        grad = _smooth_gradient(a_coarse, conv, alpha, L_coarse)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        lr = base_lr * (1.0 - frac)                # linear decay of learning rate
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # Project back onto the simplex (ensures non‑negativity and Σ=1)
        a_candidate = a_coarse - step
        a_coarse = _project_to_simplex(a_candidate)

        # Small jitter to avoid stagnation
        if it % 5000 == 0 and it > 0:
            jitter = np.random.randn(coarse_n) * 5e-4
            a_coarse = _project_to_simplex(a_coarse + jitter)

    # ------------------------------------------------------------------
    # 3) Upsample to target size and fine‑tune
    a_fine = _upsample_sequence(best_a, target_n)
    L_fine = _fft_len(target_n)

    # Re‑initialise Adam state for the fine phase
    m = np.zeros_like(a_fine)
    v = np.zeros_like(a_fine)

    best_fine = a_fine.copy()
    best_obj_fine = _objective_simplex(best_fine, L_fine)

    # Allocate the next chunk of the budget (≈40 %)
    phase2_deadline = start_time + budget_s * 0.75

    max_iter_fine = 120_000
    base_lr_fine = 0.3
    alpha_start_fine, alpha_end_fine = 5.0, 20000.0

    for it in range(max_iter_fine):
        if time.time() >= phase2_deadline - 5.0:
            break

        conv = _conv_fft(a_fine, L_fine)
        cur_obj = 2.0 * target_n * conv.max()
        if cur_obj < best_obj_fine:
            best_obj_fine = cur_obj
            best_fine = a_fine.copy()
            print(f"[fine‑tune] it={it:06d}, O={best_obj_fine:.6f}")

        frac = it / max_iter_fine
        alpha = alpha_start_fine + (alpha_end_fine - alpha_start_fine) * frac

        grad = _smooth_gradient(a_fine, conv, alpha, L_fine)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        lr = base_lr_fine * (1.0 - frac)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a_fine - step
        a_fine = _project_to_simplex(a_candidate)

        # Jitter occasionally
        if it % 3000 == 0 and it > 0:
            a_fine = _project_to_simplex(a_fine + np.random.randn(target_n) * 3e-4)

    # ------------------------------------------------------------------
    # 4) Deterministic sub‑gradient refinement on the full‑size problem
    a_sub = best_fine.copy()
    best_sub = a_sub.copy()
    best_obj_sub = best_obj_fine
    max_iter_sub = 150_000

    sub_deadline = deadline - 5.0   # leave a couple of seconds for final sanity check

    for it in range(max_iter_sub):
        if time.time() >= sub_deadline:
            break

        conv = _conv_fft(a_sub, L_fine)
        cur_obj = 2.0 * target_n * conv.max()
        if cur_obj < best_obj_sub:
            best_obj_sub = cur_obj
            best_sub = a_sub.copy()
            print(f"[subgrad] it={it:06d}, O={best_obj_sub:.6f}")

        # Sub‑gradient of the true max‑convolution entry
        g = _subgradient_max_conv(a_sub, conv)

        # Linear decay of the stepsize – the constant 0.12 works well for the
        # range of problems we consider.
        eta = 0.12 * (1.0 - it / max_iter_sub)
        a_sub = a_sub - eta * g
        a_sub = _project_to_simplex(a_sub)

        # Tiny Gaussian perturbation every 2000 steps to escape possible plateaus
        if it % 2000 == 0 and it > 0:
            a_sub = _project_to_simplex(a_sub + np.random.randn(target_n) * 2e-4)

    # ------------------------------------------------------------------
    # 5) Final sanity check with the supplied evaluation function
    final_seq = best_sub.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:
        print("[final] evaluate_sequence failed:", e)

    return final_seq
