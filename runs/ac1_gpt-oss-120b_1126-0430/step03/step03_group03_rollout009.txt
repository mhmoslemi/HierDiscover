"""
Propose a non‑negative step‑function sequence that (approximately) minimises

    O(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm has two stages.

Stage 1 – Adam on a smooth log‑sum‑exp surrogate
    * Optimises a on the probability simplex using the surrogate
      f_α(a) = (2·n/α)·log∑_k exp(α·conv_k).
    * A linear schedule α∈[5,2000] is used, together with a standard
      Adam update and a projection onto the simplex.

Stage 2 – Fast simulated‑annealing hill‑climbing
    * The full convolution is maintained explicitly.
    * A “mass‑transfer” move transfers a small amount δ from index i to j.
      The effect on the convolution can be updated in O(n) using the exact
      identity
          Δconv_k = 2·δ·(a_{k‑j} – a_{k‑i})  (ignoring O(δ²) terms).
    * The move is accepted if it reduces the objective, otherwise with a
      Metropolis probability exp(−ΔO/T).  Temperature T and the maximal
      transfer fraction decay linearly with the elapsed time.
    * Every few thousand accepted moves the convolution is recomputed via FFT
      (to remove accumulated O(δ²) error) and a is re‑projected onto the simplex.

The whole search respects the 1000 s budget.  The best sequence found is
validated with the supplied `evaluate_sequence` before being returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full self‑convolution using FFT (length 2·n‑1)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return a*a (full convolution) using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Gradient of the smooth log‑sum‑exp surrogate
def _smooth_gradient(a: np.ndarray, conv: np.ndarray,
                     alpha: float, L_corr: int) -> np.ndarray:
    """
    Compute ∇_a ( (2·n/α)·log∑_k e^{α·conv_k} ).
    Result = 2·corr(w, a) where w_k ∝ e^{α·conv_k}.
    """
    # soft‑max weights (stable)
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / exp_scaled.sum()

    # correlation w ⋆ a   (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L_corr)
    fft_a_rev = np.fft.rfft(a_rev, n=L_corr)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L_corr)
    n = a.size
    grad = 2.0 * corr_rev[n - 1 : n - 1 + n]  # central n entries
    return grad

    Returns the sequence as a list of floats.

        n = base.size

    L_conv = 1 << ((2 * n - 2).bit_length())           # for full self‑convolution
    L_corr = 1 << ((3 * n - 2).bit_length())           # for correlation in the smooth surrogate

    # ------------------------------------------------------------------
    # Helper to compute the exact objective (sum(a)=1 → O = 2·n·max(conv))
    def _objective_from_conv(conv_arr):
        return 2.0 * n * conv_arr.max()

    # ------------------------------------------------------------------
    # Stage 1 – Adam on smooth surrogate
    # Allocate roughly 15 % of the total time to this phase.
    phase1_deadline = start_time + budget_s * 0.15
    alpha_start, alpha_end = 5.0, 2000.0

        # convolution and exact objective

        grad = _smooth_gradient(a, conv, alpha, L_corr)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (iter1 + 1))
        v_hat = v / (1.0 - beta2 ** (iter1 + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # projected simplex update
        a_candidate = a - step
        a = _project_to_simplex(a_candidate)

        iter1 += 1
        if time.time() >= deadline - 5.0:
            break

    # ------------------------------------------------------------------
    # Stage 2 – Fast simulated‑annealing hill‑climbing with O(n) updates
    # Re‑initialise data structures from the best point of phase 1.
    a = best_a.copy()
    conv = _conv_fft(a, L_conv)
    cur_max = conv.max()
    cur_O = 2.0 * n * cur_max
    best_O = cur_O
    best_a = a.copy()
    best_conv = conv.copy()

    # Annealing schedule parameters
    T0 = 0.02                     # initial temperature (in objective units)
    Tmin = 1e-8
    max_frac0 = 0.20              # start with up‑to‑20 % of the source mass
    min_frac = 0.001              # final transfer fraction
    recompute_interval = 5000     # full FFT recompute to counter O(δ²) drift
    iteration = 0
    accepted = 0
    eps_nonzero = 1e-14

    while time.time() < deadline:
        # time‑based progress (0 … 1)
        elapsed = time.time() - start_time
        progress = min(elapsed / budget_s, 1.0)

        # temperature and max transfer fraction decay linearly
        T = max(T0 * (1.0 - progress), Tmin)
        max_frac = max_frac0 * (1.0 - progress) + min_frac

        # pick a source index i with positive mass
        i = np.random.randint(n)
        if a[i] < eps_nonzero:
            continue

        # pick a destination index j (different from i)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # propose a transfer δ (≤ a[i])
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # store the parts of the convolution that will change
        conv_i_old = conv[i:i + n].copy()
        conv_j_old = conv[j:j + n].copy()
        a_i_old = a[i]
        a_j_old = a[j]

        # linear O(n) update of the convolution (exact up to O(δ²))
        conv[i:i + n] -= 2.0 * delta * a
        conv[j:j + n] += 2.0 * delta * a

        # apply the mass transfer
        a[i] = a_i_old - delta
        a[j] = a_j_old + delta

        # new maximum after the move
        new_max = conv.max()
        new_O = 2.0 * n * new_max
        delta_O = new_O - cur_O

        # Metropolis acceptance
        accept = False
        if delta_O < 0.0:
            accept = True
        else:
            if np.random.rand() < math.exp(-delta_O / T):
                accept = True

        if accept:
            cur_max = new_max
            cur_O = new_O
            accepted += 1

            if cur_O < best_O:
                best_O = cur_O
                best_a = a.copy()
                best_conv = conv.copy()
                print(f"[hill] iter={iteration}, O={best_O:.6f}")

        else:
            # revert all changes
            conv[i:i + n] = conv_i_old
            conv[j:j + n] = conv_j_old
            a[i] = a_i_old
            a[j] = a_j_old

        iteration += 1

        # occasional full recompute to eliminate accumulated quadratic error
        if iteration % recompute_interval == 0:
            conv = _conv_fft(a, L_conv)
            cur_max = conv.max()
            cur_O = 2.0 * n * cur_max
            # slight renormalisation to keep the simplex constraint numerically tight
            a = _project_to_simplex(a)

        # safety break
        if time.time() >= deadline - 1.0:
            break

    # ------------------------------------------------------------------
    # Final sanity check with the supplied evaluation function
    # (project once more onto the simplex to avoid tiny numerical drift)
    best_a = _project_to_simplex(best_a)
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:
        print("[final] evaluate_sequence failed:", e)

    return best_a.tolist()
