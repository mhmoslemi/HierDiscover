import time

import numpy as np

def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto simplex {x | sum(x)=1, x >=0}."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        # all non-positive -> uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _power_law_sequence(n: int, p: float) -> np.ndarray:
    """Return normalized power-law a_i ∝ (i+1)^-p with sum 1."""
    i = np.arange(1, n+1, dtype=np.float64)
    seq = i ** (-p)
    seq /= seq.sum()
    return seq

def _conv_full(a: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution a*a using FFT of length L (>=2*len(a)-1). Returns full length."""
    fa = np.fft.fft(a, L)
    conv = np.fft.ifft(fa * fa).real
    return conv

def _grad_softmax(a: np.ndarray, conv: np.ndarray, alpha: float, L: int) -> np.ndarray:
    """
    Compute gradient of softmax-weighted max-convolution objective.
    conv is full convolution a*a (length 2n-1).
    Returns gradient vector of length n.
    """
    n = a.shape[0]
    # softmax weights over conv entries
    maxc = conv.max()
    shift = conv - maxc                # shift for numerical stability (max becomes 0)
    # exponentiate
    exp_term = np.exp(alpha * shift)
    w = exp_term / exp_term.sum()      # length 2n-1, sum 1
    # reverse both a and w for cross-correlation via convolution
    a_rev = a[::-1]
    w_rev = w[::-1]
    # compute convolution a_rev * w_rev
    fa_rev = np.fft.fft(a_rev, L)
    fw_rev = np.fft.fft(w_rev, L)
    grad_full = np.fft.ifft(fa_rev * fw_rev).real
    # extract entries corresponding to indices n-1 .. n-1 + n -1
    g = 2.0 * grad_full[n-1 : n-1 + n]
    return g

#!/usr/bin/env python
"""
Search for a non‑negative step‑function (list of heights) that minimises the
provided ``evaluate_sequence`` function.

Method
------
* Initialise a probability vector a (sum = 1) using a power‑law scan.
* Optimise a with Adam on a *smooth* surrogate of the max‑convolution:
      f(a) = (1/α)·log Σ_k exp(α·(a*a)[k]).
  The gradient of f is a soft‑max weighted sum of the individual
  convolution gradients, which can be computed by a single FFT–based
  convolution.
* Use a linear learning‑rate schedule and slowly increase the soft‑max
  temperature α (annealing) so that the optimisation first moves
  globally and later focuses on the true max.
* After each update project back onto the probability simplex
  (non‑negative, Σa = 1).
* Keep the best vector seen (according to the cheap proxy
  ``2·n·max(a*a)``) and finally evaluate it exactly with
  ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – must be top level (no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all entries ≤ 0 → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float) -> np.ndarray:
    """Return a normalised power‑law vector a_i ∝ (i+1)^‑p (Σa_i = 1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = i ** (-p)
    seq /= seq.sum()
    return seq


def _conv_full(a: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution a*a using FFT length L (L ≥ 2·len(a)‑1)."""
    fa = np.fft.fft(a, L)
    conv = np.fft.ifft(fa * fa).real
    return conv


def _grad_softmax(a: np.ndarray, conv: np.ndarray,
                  alpha: float, L: int) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate of max(a*a).

    Parameters
    ----------
    a      – current probability vector (length n)
    conv   – full linear convolution a*a (length 2·n‑1)
    alpha  – soft‑max temperature
    L      – FFT length ≥ 3·n‑2 (for the gradient convolution)

    Returns
    -------
    Gradient w.r.t. a (length n).
    """
    n = a.shape[0]

    # soft‑max weights over the convolution entries
    maxc = conv.max()
    shift = conv - maxc                    # numerical stability
    exp_term = np.exp(alpha * shift)
    w = exp_term / exp_term.sum()          # length 2·n‑1, Σw = 1

    # convolution of reversed a with w gives the required cross‑correlation
    a_rev = a[::-1]                        # length n
    fa_rev = np.fft.fft(a_rev, L)
    fw = np.fft.fft(w, L)
    grad_full = np.fft.ifft(fa_rev * fw).real

    # extract the slice that corresponds to indices 0 … n‑1
    g = 2.0 * grad_full[n - 1: n - 1 + n]
    return g
