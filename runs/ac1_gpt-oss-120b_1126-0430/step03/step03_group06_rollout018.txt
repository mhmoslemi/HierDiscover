def propose_candidate(seed=42, budget_s=1000):

    n = 20000 # maybe adjust after time passes?

    # Random search for power-law shape
    for i in range(2000):   # limited by time
        alpha = np.random.uniform(0.55, 0.95)

        # optionally use exponential interpolation between best shapes

        if time.time() > deadline:
            break

    # Run mirror descent starting from best_a
    best_a, best_val = _optimize_mirror(best_a, deadline, max_iters=200000, verbose=False)

    # Additional random restarts if time remains
    while time.time() < deadline - 2:
        a_rand = np.random.rand(n)
        a_rand /= a_rand.sum()
        a_opt, val = _optimize_mirror(a_rand, deadline, max_iters=50000, verbose=False)

            best_val, best_a = val, a_opt.copy()

"""
Proposes a step‑function (non‑negative list) that minimises
``evaluate_sequence``.  

The method is a hybrid search:

1.  A fast random exploration of a two‑parameter family
    a_i ∝ (i + offset)^(-α) (α∈[0.55,0.95], offset>0) – this family
    contains the optimal continuous “Catalan” shape (α=0.5) and many
    candidates that improve on it.

2.  The best candidate is refined with two variants of projected
    (sub‑)gradient descent on the exact objective
        O(a)=2·n·max(a*a)    (the sum of a is fixed to 1).
    • Euclidean projection onto the simplex (heavy‑ball momentum).
    • KL‑mirror descent (multiplicative updates) – often yields a
      lower plateau.

3.  If time permits a few cheap random restarts are launched and the
    best result found overall is returned.

The whole search respects the 1000 s budget, stays within the simplex,
and works with any length `n` (default 20 000).  The returned list can be
directly fed to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear convolution via a real FFT (full mode length = 2·n‑1)
# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, size: int) -> np.ndarray:
    """Return full linear convolution a*a using a real FFT of given size."""
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[: 2 * a.shape[0] - 1]


# ----------------------------------------------------------------------
# Euclidean projection of a vector onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto {x ≥ 0, Σx = 1} (the Euclidean version)."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Catalan distribution (discrete analogue of the 1/√x law)
# ----------------------------------------------------------------------
def _catalan_normalised(n: int) -> np.ndarray:
    """First n Catalan‑type coefficients, normalised to Σ = 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Objective used throughout the optimisation (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """2·n·max(a*a) for a already normalised to Σa = 1."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    conv = _fft_convolve(a, size)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Power‑law family generator  a_i ∝ (i+offset)^{-alpha}
# ----------------------------------------------------------------------
def _powerlaw(alpha: float, offset: float, n: int) -> np.ndarray:
    """Return a normalised power‑law sequence of length n."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-alpha)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


# ----------------------------------------------------------------------
# Gradient of the active quadratic term (the index where conv is maximal)
# ----------------------------------------------------------------------
def _active_gradient(a: np.ndarray) -> tuple[np.ndarray, int]:
    """Return (gradient, k) where k is an argmax of a*a."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    conv = _fft_convolve(a, size)
    k = int(np.argmax(conv))

    # gradient = 2 * J_k * a  (J_k is the anti‑diagonal selector)
    grad = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    idx = np.arange(i_start, i_end + 1, dtype=int)
    grad[idx] = 2.0 * a[k - idx]
    return grad, k


# ----------------------------------------------------------------------
# Euclidean projected sub‑gradient/momentum descent
# ----------------------------------------------------------------------
def _optimize_euclidean(a0: np.ndarray,
                        deadline: float,
                        max_iters: int = 300_000,
                        verbose: bool = False) -> tuple[np.ndarray, float]:
    """Heavy‑ball projected sub‑gradient optimiser."""
    a = a0.copy()
    n = a.shape[0]

    best = a.copy()
    best_val = _objective(best)

    beta = 0.9
    momentum = np.zeros_like(a)

    for t in range(max_iters):
        if time.time() >= deadline:
            break

        grad, _ = _active_gradient(a)

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # step size  ∝ 1/√t
        lr = 0.5 / np.sqrt(t + 1)

        # additive update + simplex projection
        a = a - lr * momentum
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best = a.copy()
            if verbose:
                print(f"[euclid improve] iter {t:06d}  obj {best_val:.9f}")

        # tiny random perturbation every few thousand steps
        if (t % 5_000) == 0 and t > 0:
            pert = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + pert, 0.0)
            a = _project_simplex(a)

    return best, best_val


# ----------------------------------------------------------------------
# KL‑mirror descent (multiplicative updates) optimiser
# ----------------------------------------------------------------------
def _optimize_mirror(a0: np.ndarray,
                     deadline: float,
                     max_iters: int = 300_000,
                     verbose: bool = False) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on the simplex."""
    a = a0.copy()
    n = a.shape[0]

    best = a.copy()
    best_val = _objective(best)

    for t in range(max_iters):
        if time.time() >= deadline:
            break

        grad, _ = _active_gradient(a)

        # learning rate schedule  ∝ 1/√t
        lr = 0.5 / np.sqrt(t + 1)

        # multiplicative update – guarantees non‑negativity
        a = a * np.exp(-lr * grad)

        # renormalise to the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best = a.copy()
            if verbose:
                print(f"[mirror improve] iter {t:06d}  obj {best_val:.9f}")

        # occasional random jitter to escape plateaus
        if (t % 5_000) == 0 and t > 0:
            pert = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + pert, 0.0)
            a /= a.sum()

    return best, best_val


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a sequence of non‑negative heights that minimises
    ``evaluate_sequence`` under the given time budget.
    """
    np.random.seed(seed)
    # leave a small safety margin for the surrounding harness
    deadline = time.time() + budget_s - 2.0

    # ------------------------------------------------------------------
    # 1) Choose a problem size (large enough for a good bound,
    #    still cheap enough for thousands of FFTs)
    # ------------------------------------------------------------------
    n = 20_000

    # ------------------------------------------------------------------
    # 2) Baseline candidates
    # ------------------------------------------------------------------
    best_seq = _catalan_normalised(n)
    best_val = _objective(best_seq)
    print(f"[init] Catalan baseline obj {best_val:.9f}")

    # ------------------------------------------------------------------
    # 3) Random exploration of the two‑parameter power‑law family
    # ------------------------------------------------------------------
    while time.time() < deadline - 20.0:   # reserve time for refinements
        # log‑uniform offset, uniform alpha
        alpha = np.random.uniform(0.55, 0.95)
        log_off = np.random.uniform(np.log(0.1), np.log(20.0))
        offset = np.exp(log_off)

        a = _powerlaw(alpha, offset, n)
        val = _objective(a)
        if val < best_val:
            best_val, best_seq = val, a.copy()
            print(f"[powerlaw] α={alpha:.4f} off={offset:.3f}  obj={val:.9f}")

        # limit the number of pure random trials to keep within budget
        if time.time() > deadline - 30.0:
            break

    # ------------------------------------------------------------------
    # 4) Refine the current best with Euclidean projected sub‑gradient
    # ------------------------------------------------------------------
    seq_euc, val_euc = _optimize_euclidean(best_seq, deadline,
                                            max_iters=250_000,
                                            verbose=False)
    if val_euc < best_val:
        best_val, best_seq = val_euc, seq_euc.copy()
        print(f"[euc refine] improved to {best_val:.9f}")

    # ------------------------------------------------------------------
    # 5) Refine the (possibly new) best with KL‑mirror descent
    # ------------------------------------------------------------------
    seq_mir, val_mir = _optimize_mirror(best_seq, deadline,
                                        max_iters=250_000,
                                        verbose=False)
    if val_mir < best_val:
        best_val, best_seq = val_mir, seq_mir.copy()
        print(f"[mirror refine] improved to {best_val:.9f}")

    # ------------------------------------------------------------------
    # 6) If any time is left, launch a few cheap random restarts
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        a0 = np.random.rand(n)
        a0 /= a0.sum()
        seq_tmp, val_tmp = _optimize_euclidean(a0, deadline,
                                               max_iters=80_000,
                                               verbose=False)
        if val_tmp < best_val:
            best_val, best_seq = val_tmp, seq_tmp.copy()
            print(f"[restart] new best obj {best_val:.9f}")

    # ------------------------------------------------------------------
    # 7) Final sanity print (the external harness will invoke
    #    ``evaluate_sequence`` on the returned list)
    # ------------------------------------------------------------------
    final_obj = _objective(best_seq)
    print(f"[final] best objective ≈ {final_obj:.9f}  n={n}")

    return best_seq.tolist()
