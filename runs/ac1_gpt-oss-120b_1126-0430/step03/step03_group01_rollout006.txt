#!/usr/bin/env python
"""
A smooth‐gradient Adam optimiser on the probability simplex.

The objective to be minimised is
    E(a) = 2·n·max_k (a*a)[k]   (a ≥ 0, ∑a = 1)

Because the exact objective is nondifferentiable we replace the max by a
soft‑max (or a power‑law) weight over all convolution entries and compute the
gradient of the weighted sum.  The weighted gradient is
    g_i = 2·∑_k w_k a_{k‑i} .
The weights w_k are defined from the current convolution c_k = (a*a)[k] as
    w_k ∝ c_k^β   (β → ∞ recovers the true sub‑gradient).

The distribution a is parametrised by logits `z` (soft‑max) so the simplex
constraint ∑a = 1 is satisfied automatically.  Adam is used in the log‑it
space and a simple schedule for the annealing exponent β and the step‑size
is employed.  The algorithm runs for the whole time budget (≈ 1000 s) and
keeps the best (lowest cheap) score; the final candidate is evaluated with
the provided ``evaluate_sequence`` before being returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max, returns a probability vector that sums to 1."""
    m = np.max(z)
    e = np.exp(z - m)
    s = e.sum()
    if s == 0.0:
        # fallback to uniform
        return np.full_like(z, 1.0 / z.size)
    return e / s


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it representation that yields `a` after soft‑max."""
    return np.log(a + eps)


def _fft_len(min_len: int) -> int:
    """Smallest power‑of‑two ≥ min_len (efficient length for real FFT)."""
    return 1 << ((min_len - 1).bit_length())


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """Normalised power‑law a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _v_shape_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric V‑shaped distribution:
       a_i ∝ (i+1)^{-p} + (n-i)^{-p}   (i = 0 … n‑1)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    left = idx ** (-p)
    right = (n - idx + 1) ** (-p)
    a = left + right
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    tot = new_a.sum()
    if tot == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= tot
    return new_a


# ----------------------------------------------------------------------
# Core optimisation primitives
# ----------------------------------------------------------------------


def _self_convolution_fft(a: np.ndarray, Nfft: int) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT of length Nfft.
    Returns the first 2·len(a)‑1 entries (the genuine convolution)."""
    Fa = np.fft.rfft(a, Nfft)
    conv_full = np.fft.irfft(Fa * Fa, Nfft)
    return np.maximum(conv_full[: 2 * a.size - 1], 0.0)  # clip tiny negatives


def _weighted_gradient(a: np.ndarray,
                       Fa: np.ndarray,
                       conv: np.ndarray,
                       beta: float,
                       Nfft: int) -> np.ndarray:
    """
    Compute ∇_a  ( ∑_k w_k (a*a)[k] )   with  w_k ∝ conv_k^β .
    Returns a vector of length len(a).
    """
    # build the weights
    w = conv ** beta
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(conv, 1.0 / conv.size)
    else:
        w /= w_sum

    # convolution of a with reversed weights
    w_rev = w[::-1]
    Fw = np.fft.rfft(w_rev, Nfft)
    conv2_full = np.fft.irfft(Fa * Fw, Nfft)

    n = a.size
    idx = (2 * n - 2) - np.arange(n)           # positions of interest
    grad_a = 2.0 * conv2_full[idx]
    return grad_a


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) beats the target bound.
    The routine respects the given time budget (≈1000 s) and uses an Adam optimiser
    on the simplex with a smooth weighted surrogate of the max‑convolution.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # 0. Choose a working support size (the maximum we will ever use)
    # ------------------------------------------------------------------
    N_MAX = 20000
    n = N_MAX

    # ------------------------------------------------------------------
    # 1. Initial distribution (use global if present, otherwise try a few heuristics)
    # ------------------------------------------------------------------
    init_from_global = False
    try:
        init_seq = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if init_seq.ndim == 1 and init_seq.size >= 5:
            a = init_seq.copy()
            a = np.maximum(a, 0.0)
            s = a.sum()
            a = a / s if s > 0 else np.full_like(a, 1.0 / a.size)
            # possibly truncate / pad to the working size
            if a.size != n:
                a = _interpolate_distribution(a, n)
            init_from_global = True
            print(f"Initial sequence taken from global variable, length = {a.size}")
    except Exception:
        init_from_global = False

    if not init_from_global:
        # try a small grid of power‑law exponents; keep the best cheap score
        best_a = None
        best_score = float("inf")
        fft_len = _fft_len(3 * n)                     # enough for all convolutions
        for p in np.linspace(0.55, 0.70, 81):
            if time.time() >= deadline:
                break
            cand = _power_law_distribution(n, p)
            conv = _self_convolution_fft(cand, fft_len)
            cheap = 2.0 * n * conv.max()
            if cheap < best_score:
                best_score = cheap
                best_a = cand
                best_p = p
        # also test a V‑shaped family
        for p in np.linspace(0.55, 0.80, 51):
            if time.time() >= deadline:
                break
            cand = _v_shape_distribution(n, p)
            conv = _self_convolution_fft(cand, fft_len)
            cheap = 2.0 * n * conv.max()
            if cheap < best_score:
                best_score = cheap
                best_a = cand
                best_p = p
        if best_a is None:   # very unlikely
            best_a = np.full(n, 1.0 / n)
        a = best_a
        print(f"Initialisation finished, best power‑law/V‑shape cheap = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Log‑it parametrisation + Adam state
    # ------------------------------------------------------------------
    eps = 1e-16
    z = _logits_from_distribution(a, eps)

    m = np.zeros_like(z)      # first‑moment estimate
    v = np.zeros_like(z)      # second‑moment estimate
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    base_alpha = 0.5          # relatively aggressive start

    # ------------------------------------------------------------------
    # 3. Main optimisation loop
    # ------------------------------------------------------------------
    Nfft = _fft_len(3 * n)    # same FFT length for all transforms
    best_seq = a.copy()
    best_cheap = float("inf")
    it = 0
    report_every = 10000
    beta_max = 30.0           # final exponent for the weight w_k ∝ conv_k^β
    max_iters_est = 300000    # rough upper bound used for scheduling

    while time.time() < deadline:
        it += 1

        # ----- probability vector from current logits -----
        a = _softmax(z)

        # ----- self‑convolution (cheap proxy) -----
        Fa = np.fft.rfft(a, Nfft)
        conv_full = np.fft.irfft(Fa * Fa, Nfft)
        conv = np.maximum(conv_full[: 2 * n - 1], 0.0)
        max_conv = conv.max()
        cheap_score = 2.0 * n * max_conv

        # ----- keep the best ever observed (according to cheap proxy) -----
        if cheap_score < best_cheap:
            best_cheap = cheap_score
            best_seq = a.copy()
            if it % report_every == 0:
                print(f"[Iter {it:08d}] NEW BEST cheap = {cheap_score:.6f}")

        # ----- annealing schedule for β (smoothness of max) -----
        progress = min(1.0, it / max_iters_est)
        beta = 1.0 + (beta_max - 1.0) * (progress ** 1.5)   # a slightly convex schedule

        # ----- weighted gradient w.r.t. a -----
        grad_a = _weighted_gradient(a, Fa, conv, beta, Nfft)

        # ----- mirror‑descent gradient in log‑it space (KL geometry) -----
        mu = np.dot(grad_a, a)
        grad_z = a * (grad_a - mu)

        # ----- Adam moments (bias‑corrected) -----
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # ----- step size (inverse‑sqrt decay) -----
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)

        # ----- update logits -----
        step_direction = - m_hat / (np.sqrt(v_hat) + eps_adam)
        z = z + alpha_t * step_direction
        # keep logits in a numerically safe window
        z = np.clip(z, -50.0, 50.0)

        # ----- occasional random perturbations to escape plateaux -----
        if it % 5000 == 0 and np.random.rand() < 0.25:
            perturb = 0.02 * np.random.randn(n)
            z += perturb
            m *= 0.5
            v *= 0.5

        # ----- early exit if we are out of time -----
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # 4. Final exact evaluation (uses the supplied evaluate_sequence)
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact evaluation: score = {final_score:.6f}")
    except Exception as exc:   # defensive – should never happen
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # ------------------------------------------------------------------
    # 5. Return the best sequence found.
    # ------------------------------------------------------------------
    return final_seq
