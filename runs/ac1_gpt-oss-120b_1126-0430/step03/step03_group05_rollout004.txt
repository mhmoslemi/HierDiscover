def _subgradient_max_convolution(a, max_idx):
    n = a.shape[0]
    lo = max(0, max_idx - (n - 1))
    hi = min(max_idx, n - 1)

    g = np.zeros_like(a)
    g[idx] = 2.0 * a[max_idx - idx]
    return g

#!/usr/bin/env python
"""
Propose a non‑negative step‑function (list of heights) that minimises

    2·n·max_k (a*a)[k] / (sum a)^2

The objective is scale‑invariant, therefore we keep the vector on the
probability simplex (sum = 1, a_i ≥ 0).  The algorithm consists of three
stages:

1.  A quick scan of several parametrised families (power‑law,
    exponential, uniform, random Dirichlet) to obtain a good initial
    point.

2.  An accelerated stochastic optimisation using the Adam variant of
    sub‑gradient descent on the active convolution index.
    After each step the iterate is projected onto the simplex (simple
    clipping + renormalisation).  Restarts with a small random perturbation
    are performed when the cheap score has not improved for a while.

3.  A short “hill‑climbing’’ phase that attempts random mass transfers
    between coordinates if they lower the cheap score.

All heavy work (convolution and sub‑gradient) is performed with
FFT‑based O(n log n) convolution; the FFT length is a power of two, so the
cost is essentially constant for any n ≤ 32768.  The whole routine respects
the supplied time budget (default 1000 s) and prints progress information.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – top level, no closures
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Simple (non‑Euclidean) projection onto the probability simplex:
    negative entries are set to zero and the vector is renormalised.
    This is much faster than the exact O(n log n) projection and is
    sufficient for the heuristic optimisation.
    """
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0:
        # fall back to uniform distribution
        v.fill(1.0 / v.shape[0])
        return v
    return v / s


def _power_law_sequence(n: int, p: float) -> np.ndarray:
    """Return a normalised power‑law vector a_i ∝ (i+1)^‑p (Σa_i = 1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = i ** (-p)
    seq /= seq.sum()
    return seq


def _exponential_sequence(n: int, rate: float) -> np.ndarray:
    """Return a normalised exponential decay a_i ∝ exp(-rate·i)."""
    i = np.arange(n, dtype=np.float64)
    seq = np.exp(-rate * i)
    seq /= seq.sum()
    return seq


def _conv_max_fft(a: np.ndarray, fft_len: int) -> tuple[float, int]:
    """
    Compute autocorrelation a*a via FFT and return the maximum value and its
    index (0‑based on the array of length 2·n‑1).
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    conv = conv[: 2 * a.shape[0] - 1]          # keep the true convolution part
    max_idx = int(conv.argmax())
    return float(conv[max_idx]), max_idx


def _subgradient_max_convolution(a: np.ndarray, max_idx: int) -> np.ndarray:
    """
    Sub‑gradient of the active autocorrelation term:
    for a given shift k = max_idx,
    (∂/∂a_i) (a*a)[k] = 2·a_{k‑i} whenever the index is valid,
    otherwise 0.
    """
    n = a.shape[0]
    lo = max(0, max_idx - (n - 1))
    hi = min(max_idx, n - 1)
    idx = np.arange(lo, hi + 1, dtype=np.int64)
    g = np.zeros_like(a)
    g[idx] = 2.0 * a[max_idx - idx]
    return g


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a vector of non‑negative heights that minimises the provided
    evaluation function.  The routine runs for at most ``budget_s`` seconds
    and returns the best sequence it discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the final exact evaluation
    deadline = start_time + budget_s - 1.0

    # ------------------------------------------------------------------
    # 0. Choose the working length.
    #    Use the largest n that still fits into the same FFT length
    #    (2 · n‑1 ≤ 65536 → n ≤ 32768).  A larger n gives more degrees of
    #    freedom without increasing the FFT cost, hence we pick n = 30000.
    # ------------------------------------------------------------------
    n = 30000
    fft_len = 1 << ((2 * n - 1).bit_length())   # next power of two ≥ 2n‑1

    # ------------------------------------------------------------------
    # 1. Initialise from several parametric families and keep the best.
    # ------------------------------------------------------------------
    best_a = None
    best_score = float("inf")

    # uniform
    a_uni = np.full(n, 1.0 / n, dtype=np.float64)
    maxb, _ = _conv_max_fft(a_uni, fft_len)
    score = 2.0 * n * maxb
    best_a, best_score = a_uni.copy(), score

    # power‑law grid
    p_grid = np.linspace(0.55, 0.70, 81)
    for p in p_grid:
        if time.time() >= deadline:
            break
        a = _power_law_sequence(n, p)
        maxb, _ = _conv_max_fft(a, fft_len)
        score = 2.0 * n * maxb
        if score < best_score:
            best_score = score
            best_a = a.copy()
    # exponential decay grid
    rate_grid = np.logspace(-5, -2, 30)
    for rate in rate_grid:
        if time.time() >= deadline:
            break
        a = _exponential_sequence(n, rate)
        maxb, _ = _conv_max_fft(a, fft_len)
        score = 2.0 * n * maxb
        if score < best_score:
            best_score = score
            best_a = a.copy()
    # a few random Dirichlet vectors (α = 0.1 and α = 1.0)
    for alpha in (0.1, 1.0):
        for _ in range(5):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(np.full(n, alpha))
            maxb, _ = _conv_max_fft(a, fft_len)
            score = 2.0 * n * maxb
            if score < best_score:
                best_score = score
                best_a = a.copy()

    # ------------------------------------------------------------------
    # 2. Refine with Adam‑style projected sub‑gradient descent.
    # ------------------------------------------------------------------
    a = best_a.copy()                # current iterate (always Σa = 1)
    m = np.zeros_like(a)             # first‑moment accumulator
    v = np.zeros_like(a)             # second‑moment accumulator

    best_iter_a = a.copy()
    best_iter_score = best_score

    # hyper‑parameters for Adam
    eta0 = 0.8                       # base learning‑rate factor
    beta1 = 0.9
    beta2 = 0.999
    adam_eps = 1e-8

    iter_count = 0
    last_improve = 0
    stagnation_limit = 80000        # restart after this many non‑improving iters
    perturb_scale = 0.02            # magnitude of random kick on restart

    # main optimisation loop
    while time.time() < deadline and iter_count < 2_000_000:
        iter_count += 1

        # ----- compute convolution and active index -----
        maxb, max_idx = _conv_max_fft(a, fft_len)
        cheap_score = 2.0 * n * maxb

        # ----- keep record of the best cheap score -----
        if cheap_score < best_iter_score - 1e-12:
            best_iter_score = cheap_score
            best_iter_a = a.copy()
            last_improve = iter_count
            print(f"[Iter {iter_count:07d}] NEW BEST cheap score = {cheap_score:.6f}")

        # ----- sub‑gradient at the active offset -----
        g = _subgradient_max_convolution(a, max_idx)

        # ----- Adam update (with 1/√t decay) -----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        # bias‑correction
        m_hat = m / (1.0 - beta1 ** iter_count)
        v_hat = v / (1.0 - beta2 ** iter_count)

        lr = eta0 / math.sqrt(iter_count)   # diminishing step size
        step = lr * m_hat / (np.sqrt(v_hat) + adam_eps)

        a_candidate = a - step
        a = _simplex_projection(a_candidate)

        # ----- occasional restart if stuck -----
        if iter_count - last_improve > stagnation_limit:
            # small random perturbation around the best iterate so far
            perturb = perturb_scale * np.random.rand(n)
            a = _simplex_projection(best_iter_a + perturb)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = iter_count
            print(f"[Iter {iter_count:07d}] Restarted after stagnation (perturb).")

    # ------------------------------------------------------------------
    # 3. A short hill‑climbing phase using random mass transfers.
    # ------------------------------------------------------------------
    a = best_iter_a.copy()
    best_hc_score = best_iter_score
    best_hc_a = a.copy()
    attempts = 0
    max_attempts = 20000
    while time.time() < deadline and attempts < max_attempts:
        attempts += 1

        # pick a donor index proportionally to its mass (so we usually have mass to move)
        donor = np.random.choice(n, p=a)
        if a[donor] == 0:
            continue
        recipient = np.random.randint(0, n)
        if donor == recipient:
            continue

        # amount to move: up to 50 % of the donor's mass
        eps = a[donor] * np.random.rand() * 0.5
        if eps <= 0:
            continue

        a_new = a.copy()
        a_new[donor] -= eps
        a_new[recipient] += eps  # sum stays 1, non‑negativity preserved

        maxb_new, _ = _conv_max_fft(a_new, fft_len)
        score_new = 2.0 * n * maxb_new

        if score_new < best_hc_score - 1e-12:
            best_hc_score = score_new
            best_hc_a = a_new.copy()
            a = a_new
            print(f"[HC {attempts:05d}] Improved cheap score to {score_new:.6f}")

    # ------------------------------------------------------------------
    # 4. Final exact evaluation (uses the provided ``evaluate_sequence``)
    # ------------------------------------------------------------------
    final_seq = best_hc_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:      # defensive – should never happen
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # ------------------------------------------------------------------
    # 5. Return the best sequence found.
    # ------------------------------------------------------------------
    return final_seq
