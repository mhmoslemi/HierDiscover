"""
Propose a non‑negative step‑function (a probability vector) that minimizes

        O(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm proceeds in three stages:

1. **Fast parametric sweep** – try a few simple families (exponential,
   power‑law) to obtain a decent starting point.
2. **Adam on a smooth log‑sum‑exp surrogate** – minimise a smooth
   approximation of the max‑convolution while always projecting onto the
   probability simplex.
3. **Stochastic refinement** – mass‑transfer Metropolis steps combined
   with projected sub‑gradient updates (the exact sub‑gradient of the
   maximal convolution entry).  This phase explores the neighbourhood of
   the best candidate for the remaining time.

All helper functions are defined at top‑level so that the grader can
import them without any closure issues.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# 1.  Simplex projection (Euclidean)
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto the simplex {x>=0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]                      # descending sort
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:                         # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# 2.  Convolution via FFT (full linear convolution, length 2·n‑1)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full convolution a*a using an FFT of length L (L must be ≥ 2·len(a)-1)."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# 3.  Exact objective for a simplex‑normalised vector
def _objective(a: np.ndarray, n: int, L: int) -> float:
    """Return O(a) = 2·n·max(conv(a,a)) (assumes sum(a)=1)."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# 4.  Sub‑gradient of the maximal convolution entry
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a.
    g_i = sum_{j+i=k*} 2·a_j   over all maximising shifts k*.
    """
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)      # all maximisers
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size > 0:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
# 5.  Gradient of the smooth log‑sum‑exp surrogate
def _smooth_gradient(a: np.ndarray, conv: np.ndarray, alpha: float, L: int) -> np.ndarray:
    """
    Gradient of  (2·n/α)·log Σ_k exp(α·conv_k)   w.r.t. a.
    This reduces to 2·corr(w, a)  where w_k ∝ exp(α·conv_k).
    """
    # soft‑max weights on the convolution entries
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)          # for numerical stability
    w = exp_scaled / exp_scaled.sum()

    # correlation w ⋆ a  (g_i = 2·Σ_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)   # length ≥ 3·n‑2
    n_len = a.size
    grad = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]
    return grad

    # Power‑law family a_i ∝ (i+1)^{-p}
    for p in np.linspace(0.0, 5.0, 61):
        seq = 1.0 / (np.arange(n, dtype=float) + 1) ** p
        seq /= seq.sum()
        sc = _objective(seq, n, L)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[sweep power] p={p:.3f} → O={sc:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – Adam on the smooth surrogate (log‑sum‑exp)
    print(f"[phase‑1] Adam start O={best_score:.6f}")

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    m = np.zeros_like(best_seq)
    v = np.zeros_like(best_seq)

    # learning‑rate schedule (base value will be decayed)
    lr0 = 0.6
    # smoothing schedule: α grows from a small to a large value
    alpha_start = 5.0
    alpha_end = 2500.0
    max_iter_phase1 = 250_000                     # generous upper bound

    # allocate ~45 % of the total budget to this phase
    phase1_deadline = start_time + budget_s * 0.45
    a = best_seq.copy()
    it = 0
    while time.time() < phase1_deadline and it < max_iter_phase1:
        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[phase‑1] it {it:06d} new best O={best_score:.6f}")

        # anneal α linearly in the iteration budget
        frac = it / max_iter_phase1
        alpha = alpha_start + (alpha_end - alpha_start) * frac

        # smooth gradient of the surrogate
        grad = _smooth_gradient(a, conv, alpha, L)

        # Adam bias‑corrected update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        lr = lr0 / math.sqrt(it + 1)               # 1/√t decay

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional tiny jitter to escape flat plateaus
        if np.random.rand() < 0.001:
            a_new += np.random.randn(n) * 1e-4

        # projection onto simplex
        a_new = _project_to_simplex(a_new)

        # prune extremely small entries every few thousand iterations
        if it % 4000 == 0 and it > 0:
            thresh = np.percentile(a_new, 0.5)       # drop the lowest 0.5 %
            a_new[a_new < thresh] = 0.0
            a_new = _project_to_simplex(a_new)

        a = a_new
        it += 1

    # ------------------------------------------------------------------
    # Stage 2 – stochastic refinement (mass‑transfer + sub‑gradient steps)
    print(f"[phase‑2] entering stochastic refinement, best O={best_score:.6f}")

    # parameters for the refinement loop
    max_local_iters = 150_000
    temp_start = 0.04
    subgrad_every = 9
    restart_counter = 0

    while time.time() < deadline:
        # ----- restart (either jittered best or fresh) -----
        if np.random.rand() < 0.55:
            a = best_seq.copy()
            a += np.random.randn(n) * 0.0015
            a = np.clip(a, 0, None)
            a = _project_to_simplex(a)
        else:
            a = np.random.rand(n)
            a = _project_to_simplex(a)

        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        it_loc = 0
        no_improve = 0
        T = temp_start

        while time.time() < deadline and it_loc < max_local_iters:
            # ---- random mass‑transfer move ----
            i, j = np.random.randint(n, size=2)
            if i != j and a[i] > 1e-14:
                # move a fraction (up to 15 %) of the source mass
                delta = a[i] * np.random.rand() * 0.15
                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta
                a_new = _project_to_simplex(a_new)

                conv_new = _conv_fft(a_new, L)
                new_obj = 2.0 * n * conv_new.max()

                # Metropolis acceptance
                if new_obj < cur_obj or np.random.rand() < math.exp((cur_obj - new_obj) / max(T, 1e-12)):
                    a = a_new
                    conv = conv_new
                    cur_obj = new_obj
                    no_improve = 0
                    if cur_obj < best_score:
                        best_score = cur_obj
                        best_seq = a.copy()
                        print(f"[{time.time():.1f}] new global best O={best_score:.6f} (restart={restart_counter}, it={it_loc})")
                else:
                    no_improve += 1
            else:
                no_improve += 1

            # ---- projected sub‑gradient step on the current maximal entry ----
            if it_loc % subgrad_every == 0:
                g = _subgradient_max_conv(a, conv)
                eta = 0.07 * (1.0 - it_loc / max_local_iters)   # linearly decaying step
                a = a - eta * g
                a = _project_to_simplex(a)
                conv = _conv_fft(a, L)
                cur_obj = 2.0 * n * conv.max()
                if cur_obj < best_score:
                    best_score = cur_obj
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] sub‑grad improved O={best_score:.6f}")

            # temperature decay
            T *= 0.9999
            it_loc += 1

            # restart if stuck for a while
            if no_improve >= 6000:
                break

        restart_counter += 1
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Final sanity check with the provided evaluator
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:
        print("[final] evaluate_sequence failed:", e)

    # Return the best sequence as a plain list
    return best_seq.tolist()
