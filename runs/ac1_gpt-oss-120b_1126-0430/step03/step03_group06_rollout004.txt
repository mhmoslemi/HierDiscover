def _fft_convolve_full(u: np.ndarray, v: np.ndarray, size: int) -> np.ndarray:
    """
    Full convolution of real vectors u and v using a real FFT.
    The returned length is len(u)+len(v)-1.

# the Euclidean projection would require sorting but is slower.
# ----------------------------------------------------------------------
def _simplex_projection(a: np.ndarray) -> np.ndarray:
    a = np.maximum(a, 0.0)
    s = a.sum()
    # safeguard against degenerate zero vector
    if s == 0.0:
        return np.full_like(a, 1.0 / a.shape[0])
    return a / s


# ----------------------------------------------------------------------
# Catalan (Inverse‑square‑root) normalised distribution.
# ----------------------------------------------------------------------
def _catalan_normalised(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Exact objective used for reporting (and for the final answer).
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    conv = _fft_convolve_full(a, a, 1 << ((2 * n - 1).bit_length()))
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed objective.
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   max_iters: int = 1_000_000,
                   verbose: bool = True) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed objective.
    Returns the best vector discovered (sum = 1) and its true objective value.
    """
    a = a0.copy()
    n = a.shape[0]

    # Pre‑compute the FFT size that can handle both convolutions (a*a and
    # (soft‑max weights) * a).  The longest convolution length is 3n‑2.
    fft_size = 1 << ((3 * n - 2).bit_length())

    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.05   # base learning rate (scaled later by Adam’s moments)

    # Temperature schedule for the soft‑max smoothing
    alpha_start = 1.0
    alpha_final = 2000.0
    ramp_iters = 20000  # linear increase over these many steps

    m = np.zeros_like(a)  # 1st moment
    v = np.zeros_like(a)  # 2nd moment

    best = a.copy()
    best_val = _objective(best)

    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1

        # ----- 1. Convolution a * a  (full) ---------------------------------
        conv_full = _fft_convolve_full(a, a, fft_size)[: 2 * n - 1]
        conv_max = conv_full.max()
        cur_obj = 2.0 * n * conv_max

        if cur_obj < best_val:
            best_val = cur_obj
            best = a.copy()
            if verbose:
                print(f"[improve] iter {t:06d}  obj {best_val:.9f}")

        # ----- 2. Soft‑max weights -----------------------------------------
        # Linear temperature schedule
        alpha = alpha_start + (alpha_final - alpha_start) * min(1.0, t / ramp_iters)

        # shift by the max to keep exponentials well‑scaled
        exp_term = np.exp(alpha * (conv_full - conv_max))
        p = exp_term / exp_term.sum()            # p_k ≥ 0, Σ p_k = 1

        # ----- 3. Gradient of the smoothed objective -----------------------
        # Gradient = 4·n·(P·a) with Toeplitz matrix P built from p.
        p_rev = p[::-1]                          # reversed weight vector
        conv_rev = _fft_convolve_full(p_rev, a, fft_size)[: 3 * n - 2]
        # (P·a)_i = conv_rev[2n‑2‑i]   for i = 0 … n‑1
        pa = conv_rev[2 * n - 2 - np.arange(n)]
        grad = 4.0 * n * pa

        # ----- 4. Adam update -----------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a -= step

        # ----- 5. Feasibility projection (non‑negative, sum = 1) ------------
        a = _simplex_projection(a)

        # occasional tiny random jitter to escape flat regions
        if t % 10_000 == 0:
            jitter = np.random.normal(scale=0.003, size=n)
            a = _simplex_projection(a + jitter)

    return best, best_val


# ----------------------------------------------------------------------
# Main entry point expected by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Return a non‑negative height sequence that (approximately) minimises
    ``evaluate_sequence`` within the given time budget.
    """
    np.random.seed(seed)
    # keep a small safety margin for the harness
    deadline = time.time() + budget_s - 2.0

    # problem size – 20 000 provides a good trade‑off between quality
    # and per‑iteration runtime.
    n = 20_000

    # ---- 1) initial candidates (Catalan + a few random Dirichlet draws) ----
    initial_vectors = [_catalan_normalised(n)]
    for _ in range(3):
        rnd = np.random.rand(n)
        initial_vectors.append(rnd / rnd.sum())

    best_seq = None
    best_val = float('inf')

    # ---- 2) run Adam optimisation from each start while time permits -----
    for idx, a0 in enumerate(initial_vectors):
        if time.time() > deadline - 5.0:
            break
        print(f"[restart] start {idx} (sum={a0.sum():.3f})")
        a_opt, val = _adam_optimize(a0, deadline, max_iters=200_000, verbose=False)
        if val < best_val:
            best_val = val
            best_seq = a_opt.copy()
            print(f"[restart] new best obj {best_val:.9f}")

    # ---- 3) final sanity check -------------------------------------------
    final_obj = _objective(best_seq)
    print(f"[final] best objective ≈ {final_obj:.9f} with n={n}")

    return best_seq.tolist()
