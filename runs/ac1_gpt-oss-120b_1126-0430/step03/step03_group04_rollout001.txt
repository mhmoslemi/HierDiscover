"""
Propose a step‑function (non‑negative vector summing to 1) that
minimises  O = 2·n·max(conv(a,a)) .
The implementation uses a simple sub‑gradient descent on the
max‑convolution with Nesterov‑style momentum and a projection onto
the probability simplex.  Several random restarts (different lengths,
different initial shapes) are performed until the 1000 s budget is
exhausted.  The best sequence found is returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x ≥ 0, sum(x)=1}
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        # all non‑positive – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two greater than or equal to x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation of w with a (no reversal):
        (w ⋆ a)[i] = Σ_t w[i+t]·a[t] for i = 0,…,n‑1 .
    Implemented via FFT.
    """
    n = a.shape[0]
    m = w.shape[0]               # = 2·n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) (a sums to 1)."""
    n = a.shape[0]
    conv = _conv_self_fft(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _initial_vector(n: int, mode: str = "random") -> np.ndarray:
    """Generate a non‑negative vector of length n that sums to 1."""
    if mode == "gaussian":
        x = np.arange(n)
        sigma = n / 6.0
        center = (n - 1) / 2.0
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    elif mode == "tent":
        half = n // 2
        a = np.concatenate([np.arange(1, half + 1), np.arange(half, 0, -1)])
        if a.shape[0] < n:
            a = np.append(a, 0.0)
    elif mode == "uniform":
        a = np.ones(n)
    else:   # "random"
        a = np.random.rand(n)
    a = a.astype(float)
    a = _project_to_simplex(a)
    return a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that minimises the bound.
    Returns the best sequence as a list of floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Global best across all restarts
    best_seq = None
    best_score = float("inf")

    # If a previous construction is available, seed from it
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=float)
        if a0.size == 0:
            a0 = None
        else:
            a0 = a0 / a0.sum()
    else:
        a0 = None

    restart_counter = 0
    while time.time() < deadline:
        restart_counter += 1

        # --------------------------------------------------------------
        # Choose a length (a few thousand works well) and an initial mode
        n = np.random.randint(3000, 6000)   # length between 3k and 6k
        if a0 is not None and a0.shape[0] == n:
            a = a0.copy()
        else:
            mode = np.random.choice(["random", "gaussian", "tent", "uniform"])
            a = _initial_vector(n, mode=mode)

        # --------------------------------------------------------------
        # Sub‑gradient descent parameters for this run
        max_iters_run = 120_000                     # generous upper bound
        mu = 0.9                                     # momentum coefficient
        vel = np.zeros_like(a)                      # momentum buffer
        eta0 = 0.8                                   # initial step size coefficient
        tol = 1e-12                                  # tolerance for active set

        # Keep local best for this restart (may be useful for early restarts)
        local_best_seq = a.copy()
        local_best_score = _objective(local_best_seq)

        for it in range(1, max_iters_run + 1):
            if time.time() >= deadline:
                break

            # ---------------------------------------------------------
            # 1) Compute full self‑convolution and the objective
            conv = _conv_self_fft(a)
            M = conv.max()
            cur_score = 2.0 * n * M

            # Update global best
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[restart {restart_counter}] new global best O={best_score:.6f} "
                      f"(n={n}, it={it})")

            # Update local best (used for occasional re‑initialisation)
            if cur_score < local_best_score - 1e-12:
                local_best_score = cur_score
                local_best_seq = a.copy()

            # ---------------------------------------------------------
            # 2) Determine the active set (indices attaining the max)
            active_mask = (conv >= M - tol).astype(float)
            if active_mask.sum() == 0:   # numerical safety
                active_mask[conv.argmax()] = 1.0

            w = active_mask / active_mask.sum()   # probability on active indices

            # ---------------------------------------------------------
            # 3) Sub‑gradient of the (non‑smooth) max‑convolution
            #    ∇ = 2 * correlation(w, a)
            g = 2.0 * _correlation_fft(w, a)

            # ---------------------------------------------------------
            # 4) Momentum step and projection onto simplex
            eta = eta0 / math.sqrt(it)          # diminishing step size
            vel = mu * vel - eta * g
            a_new = a + vel
            a_new = _project_to_simplex(a_new)

            # ---------------------------------------------------------
            # 5) Occasionally add a tiny random jitter to escape plateaus
            if it % 5000 == 0:
                jitter = np.random.randn(n) * 0.0005
                a_new += jitter
                a_new = _project_to_simplex(a_new)

            a = a_new

        # End of one restart – optionally re‑initialise from the best so far
        # (already done at the start of the next loop)

    # ------------------------------------------------------------------
    # Final evaluation with the provided evaluate_sequence (returns the
    # exact bound with the same normalisation as the competition).
    if best_seq is None:
        # Fallback: a simple uniform distribution
        best_seq = np.full(1000, 1.0 / 1000)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_seq.tolist()
