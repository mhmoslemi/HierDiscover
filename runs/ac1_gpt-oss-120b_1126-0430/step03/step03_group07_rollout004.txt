        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        # each seed maybe get min(120, remaining/remaining_seeds)

def _project_simplex(v):
    # returns projection onto the simplex (non-negative sum = 1)
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _gradient_smooth(a, alpha, L):
    # compute conv
    fa = np.fft.rfft(a, L)
    # but compute conv: use ifft of fa * fa
    conv = np.fft.irfft(fa * fa, L)[:2*n-1]
    # compute weights w
    maxc = conv.max()
    exp_arg = alpha * (conv - maxc)  # subtract max for stability
    w = np.exp(exp_arg)
    w = w / w.sum()
    # compute correlation g = 2 * corr(w, a)
    fw = np.fft.rfft(w, L)
    # correlation g = ifft(fw * conj(fa))
    corr = np.fft.irfft(fw * np.conj(fa), L)[:n]
    g = 2.0 * corr

def _smooth_gradient_descent(a0, deadline):
    a = np.copy(a0)
    n = a.size
    # FFT length
    L = 1 << ((2*n - 1).bit_length())
    # Adam's parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-15
    lr = 0.2 # base step
    m = np.zeros(n)
    v = np.zeros(n)
    t=0
    best_a = a.copy()
    best_score = fast_evaluation(a)
    # schedule
    alphas = [5,10,20,40,80,160]
    alpha_idx = 0
    alpha = alphas[alpha_idx]
    # per-alpha iterations limit maybe 3000
    iter_per_alpha = max(2000, int((deadline - time.time()) / (len(alphas) * 1))) # just placeholder
    # But we just increment alpha when t reaches certain thresholds: e.g., after 3000 iterations.
    max_iters = 1000000  # just large; stop by time.

    while time.time() < deadline and t < max_iters:
        # compute gradient and conv_max
        # step 1: FFT of a
        fa = np.fft.rfft(a, L)
        # compute convolution a*a
        conv = np.fft.irfft(fa * fa, L)[: 2*n - 1]
        maxc = conv.max()
        # compute weights w (softmax)
        exp_arg = alpha * (conv - maxc)  # prevents overflow; max exponent 0
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            # fallback to uniform weight
            w = np.full_like(w, 1.0/w.size)
        else:
            w /= w_sum

        # compute gradient = 2 * correlation(w, a)
        fw = np.fft.rfft(w, L)
        # correlation = ifft(fw * conj(fa))
        corr = np.fft.irfft(fw * np.conj(fa), L)[:n]
        g = 2.0 * corr

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = a - step
        # project onto simplex
        a = _project_simplex(a_new)

        # maybe apply small epsilon to avoid zero values causing log issues later
        # but not needed.

        # evaluate metric
        score = fast_evaluation(a)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            # optional print
            if t % 500 == 0:
                print(f"[GD] iter {t:6d} alpha {alpha:5.1f} best {best_score:.6f}")

            a[idx] = max(0.0, a[idx] + np.random.randn() * 1e-4)
            a = _project_simplex(a)

    return best_a

def _project_simplex(v):
    """Project vector v onto the probability simplex (non-negative, sum = 1)."""
    # Based on algorithm by Duchi et al.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        # all values are zero or negative, return uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
propose_candidate
-----------------
Search for a non‑negative step‑function (probability vector) that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

using a continuation‑style smooth‑max optimisation.
The algorithm:

1. Generate a diverse pool of seeds (Beta‑densities, sin‑powers,
   random Dirichlet draws, and optionally the previously best known
   sequence ``height_sequence_1``).

2. For each seed run an Adam‑style projected gradient descent on the
   *soft‑max* approximation of the max‑operator:

       S_α(a) = (1/α)·log Σ_k exp(α·(a*a)[k])

   Gradient of S_α can be written analytically as
   `g = 2·corr(w, a)` where `w` are the soft‑max weights.
   All convolutions and correlations are performed with FFTs (O(n log n)).

   The smoothness parameter α is increased geometrically (5 → 10 → … → 160)
   during the run (continuation), allowing large steps early and a
   tighter focus on the true max later.

3. Each Adam update is followed by a projection onto the probability
   simplex (non‑negative, sum = 1).

4. The best vector seen across all seeds and a final refinement on the
   remaining time budget are returned as a plain Python list.

The method explores many directions that the previous exponentiated‑gradient
approach could not reach and typically yields a lower bound (< 1.5) within the
1000 s budget.

"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast evaluation (identical to the supplied evaluate_sequence but without
# security checks).  The result is a float, not np.inf for valid input.
# ----------------------------------------------------------------------
def _fast_evaluation(seq: np.ndarray) -> float:
    """Return the metric 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    n = seq.size
    if n == 0:
        return float("inf")
    # FFT length: next power of two ≥ 2·n‑1
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex (sum = 1, v >= 0)."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # Find rho
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive → uniform simplex
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1  # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Seed generators

def _beta_sequence(alpha: float, beta: float, length: int) -> np.ndarray:
    """Beta(α,β) density sampled on (0,1) and normalised to sum = 1."""
    xs = (np.arange(length, dtype=float) + 0.5) / length
    # avoid zero powers that would produce infinities
    a = np.power(xs, alpha - 1.0, out=np.empty_like(xs))
    b = np.power(1.0 - xs, beta - 1.0, out=np.empty_like(xs))
    seq = a * b
    seq = np.nan_to_num(seq, nan=0.0, posinf=0.0, neginf=0.0)
    seq = np.maximum(seq, 0.0)
    s = seq.sum()
    if s == 0.0:
        seq = np.full(length, 1.0 / length)
    else:
        seq /= s
    return seq


def _sin_power_sequence(p: float, length: int) -> np.ndarray:
    """Sequence proportional to sin(π·x)^p (x in (0,1)), normalised."""
    xs = (np.arange(length, dtype=float) + 0.5) / length
    # protect against sin(x)≈0 when p<0
    eps = 1e-12
    s = np.maximum(np.sin(np.pi * xs), eps)
    vals = np.power(s, p, out=np.empty_like(s))
    vals = np.nan_to_num(vals, nan=0.0, posinf=0.0, neginf=0.0)
    vals = np.maximum(vals, 0.0)
    ssum = vals.sum()
    if ssum == 0.0:
        vals = np.full(length, 1.0 / length)
    else:
        vals /= ssum
    return vals


def _random_dirichlet(length: int, concentration: float = 0.5) -> np.ndarray:
    """Draw a random point from Dirichlet(concentration,…,concentration)."""
    seq = np.random.gamma(concentration, 1.0, size=length)
    s = seq.sum()
    if s == 0.0:
        seq = np.full(length, 1.0 / length)
    else:
        seq /= s
    return seq


# ----------------------------------------------------------------------
# Smoothed‑max gradient descent with Adam and simplex projection
# ----------------------------------------------------------------------
def _smooth_gradient_descent(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Refine a probability vector ``a0`` (sum = 1, a0≥0) using an Adam‑style
    projected gradient descent on the smoothed max‐objective.
    The continuation schedule for the smoothness parameter α is
    [5,10,20,40,80,160]; it is increased every ~2500 iterations.
    """
    a = a0.astype(float).copy()
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())          # FFT length (power of two)
    # Adam state
    beta1, beta2 = 0.9, 0.999
    lr = 0.12                                     # base learning rate
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # continuation schedule
    alphas = [5, 10, 20, 40, 80, 160]
    alpha_idx = 0
    alpha = alphas[alpha_idx]

    # bookkeeping
    best_a = a.copy()
    best_score = _fast_evaluation(a)

    # main loop
    while time.time() < deadline:
        # FFT of current a
        fa = np.fft.rfft(a, L)

        # self‑convolution a*a (real part only)
        conv = np.fft.irfft(fa * fa, L)[: 2 * n - 1]

        # soft‑max weights w_k ∝ exp(α·conv_k)
        maxc = conv.max()
        exp_arg = alpha * (conv - maxc)          # shift for numerical stability
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient g = 2 * corr(w, a)
        fw = np.fft.rfft(w, L)
        corr = np.fft.irfft(fw * np.conj(fa), L)[:n]
        g = 2.0 * corr

        # Adam update with bias correction
        t += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = a - step

        # project onto simplex
        a = _project_simplex(a_new)

        # occasional tiny jitter to escape plateaus
        if np.random.rand() < 0.0005:
            j = np.random.randint(n)
            a[j] = max(0.0, a[j] + np.random.randn() * 1e-4)
            a = _project_simplex(a)

        # evaluate true metric (no smoothing)
        cur_score = _fast_evaluation(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            if t % 500 == 0:
                print(f"[GD] iter {t:6d} α={alpha:5.1f} best {best_score:.6f}")

        # increase α according to schedule
        if t % 2500 == 0 and alpha_idx + 1 < len(alphas):
            alpha_idx += 1
            alpha = alphas[alpha_idx]
            print(f"[GD] increased smoothing α → {alpha}")

    return best_a

        seeds.append((f"sin_pow_{p:.1f}", _sin_power_sequence(p, n)))

    # random Dirichlet draws (concentration < 1 ⇒ spikier)
    for i in range(3):
        seeds.append((f"rand_dir_{i}", _random_dirichlet(n, concentration=0.4)))

    # ------------------------------------------------------------------
    # 2. Run a time‑sliced refinement on each seed, keep the best
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    total_seeds = len(seeds)

    while seeds and time.time() < deadline:
        # allocate a fair slice of remaining time to the current seed
        time_left = deadline - time.time()
        per_seed = min(200.0, max(10.0, time_left / (len(seeds) + 1)))
        name, seq0 = seeds.pop(0)

        # ensure proper simplex normalisation
        seq0 = np.maximum(seq0, 0.0)
        s0 = seq0.sum()
        if s0 > 0.0:
            seq0 /= s0
        else:
            seq0 = np.full(n, 1.0 / n)

        refined = _smooth_gradient_descent(seq0, time.time() + per_seed)
        score = _fast_evaluation(refined)
        print(f"[seed {name}] score {score:.6f}")

        if score < best_score:
            best_score = score
            best_seq = refined.copy()
            print(f"*** new global best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Final refinement on the remaining budget (if any)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)  # fallback – uniform distribution

    if time.time() < deadline:
        print("[final] performing last refinement on the best candidate")
        final_seq = _smooth_gradient_descent(best_seq, deadline)
        final_score = _fast_evaluation(final_seq)
        print(f"[final] best score {final_score:.6f}")
        if final_score < best_score:
            best_seq = final_seq
            best_score = final_score

    # ------------------------------------------------------------------
    # 4. Return as plain Python list
    # ------------------------------------------------------------------
    print(f"[summary] final metric {best_score:.6f} with length {best_seq.size}")
    return best_seq.tolist()
