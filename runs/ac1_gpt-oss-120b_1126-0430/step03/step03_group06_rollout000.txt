"""
Propose a step‑function sequence that (approximately) minimises
``evaluate_sequence``.  

The algorithm builds on three ideas:

1. **Catalan start** – the analytically motivated Catalan distribution is a
   strong baseline.
2. **Exponentiated‑gradient / Adam update** – we treat the problem as optimisation
   on the probability simplex and perform mirror‑descent in the log‑domain.
   The gradient is the sub‑gradient of the active maximal convolution entry.
3. **Multi‑scale search** – a few quick runs on several problem sizes
   (4000, 8000, 16000, 20000) are performed, the best size is then refined for
   the remaining time.  Random restarts with Dirichlet starts are added if time
   permits.

The method respects the 1000 s budget, prints progress and finally returns the
best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based full linear convolution (real input)
# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Objective for a normalized sequence (sum == 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """2·n·max(a*a) when Σa = 1."""
    n = a.shape[0]
    return float(2.0 * n * np.max(_fft_convolve(a)))


# ----------------------------------------------------------------------
# Catalan coefficients, normalised to Σa = 1
# ----------------------------------------------------------------------
def _catalan_normalised(n: int) -> np.ndarray:
    """First n Catalan‑type coefficients C_i/4^i, normalised to a simplex."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1) / (2.0 * i)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Adam‑style exponentiated‑gradient optimiser on the simplex
# ----------------------------------------------------------------------
def _optimize_adam_eg(a0: np.ndarray,
                      deadline: float,
                      max_iters: int = 300_000,
                      verbose: bool = True):
    """
    Mirror‑descent on the simplex using Adam on the log‑variables.
    Returns the best vector encountered and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]

    # work in log‑space (mirror map = KL‑divergence)
    log_a = np.log(np.maximum(a, 1e-30))

    # Adam accumulators
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0

    best = a.copy()
    best_val = _objective(best)

    t = 0
    while time.time() < deadline and t < max_iters:
        # ---------- evaluate ----------
        conv = _fft_convolve(a)
        k = int(np.argmax(conv))
        max_c = conv[k]
        cur_val = 2.0 * n * max_c
        if cur_val < best_val:
            best_val = cur_val
            best = a.copy()
            if verbose:
                print(f"[improve] iter {t:06d}  obj {best_val:.9f}")

        # ---------- sub‑gradient of the active quadratic ----------
        grad = np.zeros_like(a)
        i0 = max(0, k - (n - 1))
        i1 = min(k, n - 1)
        idx = np.arange(i0, i1 + 1, dtype=int)
        # contribution: 2 * a_{k-i}
        grad[idx] = 2.0 * a[k - idx]

        # ---------- Adam update on log‑variables ----------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        beta1_pow *= beta1
        beta2_pow *= beta2

        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        # step size schedule (decreasing like 1/√t)
        lr = 0.1 / np.sqrt(t + 1.0)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # log‑space descent
        log_a -= step

        # keep numbers in a numerically safe range
        log_a -= np.max(log_a)          # shift – avoids overflow in exp
        a = np.exp(log_a)

        # enforce simplex constraints
        a_sum = a.sum()
        if a_sum == 0.0:                # safety fallback
            a = np.full_like(a, 1.0 / n)
        else:
            a /= a_sum
        # re‑initialise log_a from the projected point
        log_a = np.log(np.maximum(a, 1e-30))

        # occasional jitter to escape plateaus
        if t % 10_000 == 0 and t > 0:
            log_a += np.random.normal(scale=0.001, size=n)

        t += 1

    return best, best_val


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a sequence of non‑negative heights (a step function) that
    minimises ``evaluate_sequence`` under the given time budget.

    The routine:
        1. tries a few problem sizes with short Adam‑EG runs,
        2. refines the best candidate for the remaining time,
        3. performs random restarts if time permits,
        4. returns the best sequence found.
    """
    np.random.seed(seed)

    # leave a small margin for the harness / final printing
    deadline = time.time() + budget_s - 2.0

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # (a)  Small quick explorations on several lengths
    # ------------------------------------------------------------------
    candidate_ns = [4_000, 8_000, 16_000, 20_000]
    for n in candidate_ns:
        if time.time() >= deadline:
            break
        # start from Catalan distribution (a strong analytical baseline)
        a0 = _catalan_normalised(n)

        # give each size a modest time slice (≈10 s)
        sub_deadline = min(deadline, time.time() + 12.0)
        seq, val = _optimize_adam_eg(a0, sub_deadline,
                                     max_iters=150_000,
                                     verbose=False)
        if val < best_val:
            best_val, best_seq = val, seq
            print(f"[candidate] n={n}  new best obj {best_val:.9f}")

    # ------------------------------------------------------------------
    # (b)  If a good candidate exists, run a long refinement on it
    # ------------------------------------------------------------------
    if best_seq is not None:
        n = len(best_seq)
        # remaining time is used for a deeper optimisation on the best size
        seq, val = _optimize_adam_eg(best_seq, deadline,
                                     max_iters=800_000,
                                     verbose=True)
        if val < best_val:
            best_val, best_seq = val, seq
            print(f"[refine] n={n}  improved obj {best_val:.9f}")

    # ------------------------------------------------------------------
    # (c)  Random restarts with Dirichlet starts (if time still left)
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        # pick a size (reuse a previously successful size for speed)
        n = np.random.choice(candidate_ns)
        a_rand = np.random.rand(n)
        a_rand /= a_rand.sum()
        seq, val = _optimize_adam_eg(a_rand, deadline,
                                     max_iters=70_000,
                                     verbose=False)
        if val < best_val:
            best_val, best_seq = val, seq
            print(f"[restart] n={n}  new best obj {best_val:.9f}")

    # ------------------------------------------------------------------
    # (d)  Final safety fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.array([1.0], dtype=np.float64)

    # ------------------------------------------------------------------
    #  Convert to plain Python list, enforce the bounds used in
    #  ``evaluate_sequence`` (0 ≤ x ≤ 1000).
    # ------------------------------------------------------------------
    best_seq = [float(min(1000.0, max(0.0, x))) for x in best_seq]

    # ------------------------------------------------------------------
    #  Report final value (the harness will evaluate it independently)
    # ------------------------------------------------------------------
    print(f"[final] best objective ≈ {best_val:.9f} with n={len(best_seq)}")
    return best_seq
