"""
Propose a non‑negative step function that minimises the bound

    O = 2·n·max(conv(a,a)) / (sum(a)^2)

The bound is scale‑invariant, so we keep the vector on the unit simplex.
The optimizer works in two phases:

1. A fast phase on a modest length (n≈4000) using a smooth approximation
   of the max‑operator.  The gradient of the smooth max is obtained via a
   single FFT‑based convolution (O(n log n)).  An exponentiated‑gradient
   update (mirrored SGD) keeps the variables non‑negative and normalised.

2. The best vector from phase‑1 is interpolated to a larger length
   (n≈12000–20000) and further refined with the same scheme; longer
   vectors give a lower theoretical bound because the convolution can be
   spread over more positions.

Both phases use a cooling schedule for the temperature of the smooth
max and for the learning‑rate.  When the objective does not improve for
a while a small random jitter is added to escape possible plateaus.

The final sequence is returned as a plain list of floats; it is then
evaluated with the supplied ``evaluate_sequence`` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray, size: int = None) -> np.ndarray:
    """Linear convolution a * a using a power‑of‑two FFT."""
    n = a.size
    if size is None:
        size = 1 << ((2 * n - 2).bit_length())
    A = np.fft.fft(a, n=size)
    conv = np.fft.ifft(A * A).real[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def _objective(a: np.ndarray, conv_cache: np.ndarray | None = None) -> float:
    """Bound 2·n·max(conv(a,a)) for a already normalised (Σa = 1)."""
    n = a.size
    if conv_cache is None:
        conv = _conv_fft(a)
    else:
        conv = conv_cache
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
def _gradient_from_weights(w: np.ndarray, a_rev: np.ndarray,
                           size: int) -> np.ndarray:
    """
    Compute ∇_a  Σ_k w_k (a*a)[k] efficiently.
    With w = soft‑weights over the convolution entries, the gradient is
    2 * (w ⋆ a_rev) shifted by (n‑1).
    """
    # FFT of both arguments
    W = np.fft.fft(w, n=size)
    A = np.fft.fft(a_rev, n=size)
    rev_conv = np.fft.ifft(W * A).real
    n = a_rev.size
    # slice corresponds to indices n‑1 … n‑1+n‑1 (inclusive)
    grad = 2.0 * rev_conv[n - 1: n - 1 + n]
    return grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises the
    provided bound.  The algorithm runs two optimisation phases with
    increasing vector length and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # small safety margin

    # ------------------------------------------------------------------
    # Phase 1: modest length, quick exploration
    n1 = 4000
    idx = np.arange(1, n1 + 1, dtype=float)
    a = 1.0 / np.sqrt(idx)                     # 1/√i law
    a /= a.sum()                               # simplex

    best_a = a.copy()
    best_obj = _objective(a)

    # FFT size for conv(a,a)
    size_conv1 = 1 << ((2 * n1 - 2).bit_length())
    # FFT size for w * a_rev (needs length 3n‑2)
    size_grad1 = 1 << ((3 * n1 - 2).bit_length())

    # hyper‑parameters for phase 1
    max_iters1 = 30000
    eta0_1 = 0.45
    tau_factor_1 = 0.02
    min_tau = 1e-12
    patience = 3000
    stagnation = 0

    for it in range(max_iters1):
        if time.time() >= deadline:
            break

        # ---- convolution a*a -------------------------------------------------
        conv = _conv_fft(a, size=size_conv1)
        max_val = conv.max()
        cur_obj = 2.0 * n1 * max_val
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            stagnation = 0
            print(f"[phase1 {it:5d}] new best O={best_obj:.6f}")
        else:
            stagnation += 1

        # ---- smooth max weights ------------------------------------------------
        tau = max(tau_factor_1 * max_val, min_tau) * (1.0 - it / max_iters1)
        # use a numerically stable softmax (shift by max)
        w = np.exp((conv - max_val) / tau)
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient ---------------------------------------------------------
        a_rev = a[::-1]
        grad = _gradient_from_weights(w, a_rev, size_grad1)

        # ---- exponentiated‑gradient step --------------------------------------
        eta = eta0_1 * (1.0 - it / max_iters1)
        log_a = np.log(a + 1e-15) - eta * grad
        # keep numbers in a reasonable range
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()

        # ---- occasional jitter to escape plateaus -----------------------------
        if stagnation >= patience:
            jitter = np.random.randn(n1) * 1e-4
            a = best_a + jitter
            a = np.clip(a, 0.0, None)
            a = _project_to_simplex(a)
            stagnation = 0

    # ------------------------------------------------------------------
    # Phase 2: enlarge the vector, continue optimisation
    # Only start phase 2 if we still have a reasonable amount of time.
    remaining = deadline - time.time()
    if remaining > 120.0:   # need at least ~2 minutes for phase 2
        # target length – a power of two is convenient for FFTs
        n2 = 16384  # 2^14  ->  size for FFT will be 2^15 or 2^16, still cheap
        # interpolate the best vector from phase 1 to the new length
        x_old = np.linspace(0.0, 1.0, best_a.size)
        x_new = np.linspace(0.0, 1.0, n2)
        a = np.interp(x_new, x_old, best_a)
        a = _project_to_simplex(a)

        best_a2 = a.copy()
        best_obj2 = _objective(a)

        size_conv2 = 1 << ((2 * n2 - 2).bit_length())
        size_grad2 = 1 << ((3 * n2 - 2).bit_length())

        max_iters2 = 80000
        eta0_2 = 0.30
        tau_factor_2 = 0.015
        stagnation = 0

        for it in range(max_iters2):
            if time.time() >= deadline:
                break

            conv = _conv_fft(a, size=size_conv2)
            max_val = conv.max()
            cur_obj = 2.0 * n2 * max_val
            if cur_obj < best_obj2 - 1e-12:
                best_obj2 = cur_obj
                best_a2 = a.copy()
                stagnation = 0
                print(f"[phase2 {it:5d}] new best O={best_obj2:.6f}")
            else:
                stagnation += 1

            tau = max(tau_factor_2 * max_val, min_tau) * (1.0 - it / max_iters2)
            w = np.exp((conv - max_val) / tau)
            w_sum = w.sum()
            if w_sum == 0 or not np.isfinite(w_sum):
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            a_rev = a[::-1]
            grad = _gradient_from_weights(w, a_rev, size_grad2)

            eta = eta0_2 * (1.0 - it / max_iters2)
            log_a = np.log(a + 1e-15) - eta * grad
            log_a -= np.max(log_a)
            a = np.exp(log_a)
            a /= a.sum()

            if stagnation >= patience:
                jitter = np.random.randn(n2) * 1e-5
                a = best_a2 + jitter
                a = np.clip(a, 0.0, None)
                a = _project_to_simplex(a)
                stagnation = 0

        # keep the best from phase 2 if it beats phase 1
        if best_obj2 < best_obj:
            best_obj = best_obj2
            best_a = best_a2.copy()

    # ------------------------------------------------------------------
    # Final exact evaluation (the provided function clamps negatives etc.)
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] internal best O={best_obj:.6f}, evaluate_sequence O={final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence failed:", exc)
        final_score = None

    return best_a.tolist()
