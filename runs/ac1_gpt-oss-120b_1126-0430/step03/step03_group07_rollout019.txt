def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import time, numpy as np
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0 # safety margin

    n_small = 2000 # for seed generation
    n_big = 5000   # final length

    # generate seeds
    seeds = []

    # Beta symmetric
    alphas = [0.05,0.1,0.15,0.2,0.25,0.3,0.35,0.4,0.45,0.5,0.6,0.8,1.0]
    for a in alphas:
        seeds.append((f"beta_sym_{a:.2f}", _beta_sequence(a, a, n_small)))

    # Beta asymmetric
    asym_pairs = [(0.05, 0.9), (0.1,0.9), (0.2,0.8), (0.3,0.7), (0.4,0.6), (0.6,0.4), (0.7,0.3), (0.8,0.2), (0.9,0.1), (0.9,0.05)]
    for a,b in asym_pairs:
        seeds.append((f"beta_{a:.2f}_{b:.2f}", _beta_sequence(a,b,n_small)))

    # sin power
    p_vals = [-0.9,-0.8,-0.7,-0.6,-0.5,-0.4,-0.3,-0.2,-0.1,0.0,0.1,0.2,0.3]
    for p in p_vals:
        seeds.append((f"sin_pow_{p:.1f}", _sin_power_sequence(p, n_small)))

    # power-law end shapes
    exps = [-0.9,-0.7,-0.5,-0.3,-0.1,0.0,0.2,0.4,0.6]
    for pl in exps:
        for pr in exps:
            # include only a modest number of combos
            if (pl,pr) in [(0.0,0.0), (0.2,0.2), (0.4,0.4)]:
                continue
            seeds.append((f"pow_{pl:.1f}_{pr:.1f}", _powerlaw_sequence(pl, pr, n_small)))

    # window shapes
    seeds.append(("hann", _window_hann(n_small)))
    seeds.append(("bartlett", _window_bartlett(n_small)))
    seeds.append(("blackman", _window_blackman(n_small)))

    # Random Dirichlet draws
    for i in range(6):
        seeds.append((f"rand_dir_{i}", _random_dirichlet(n_small, concentration=0.4)))
    for i in range(4):
        seeds.append((f"rand_dir_high_{i}", _random_dirichlet(n_small, concentration=0.8)))

    # Evaluate each seed's score
    scored_seeds = []
    for name, seq in seeds:
        seq = np.maximum(seq, 0.0)
        if seq.sum() == 0.0:
            seq = np.full_like(seq, 1.0/seq.size)
        else:
            seq = seq/seq.sum()
        score = _fast_evaluation(seq)
        scored_seeds.append((score, name, seq))
        print(f"[seed {name}] raw score {score:.6f}")

    # Sort and keep top candidates
    scored_seeds.sort(key=lambda x: x[0])
    top_k = min(15, len(scored_seeds))
    top_candidates = scored_seeds[:top_k]
    print(f"[selection] keeping top {top_k} seeds")

    best_seq = None
    best_score = float("inf")

    # refine each top candidate using Adam
    # allocate time per candidate (equal share of remaining time)
    while top_candidates and time.time() < deadline:
        time_left = deadline - time.time()
        per_candidate = max(5.0, time_left / (len(top_candidates) + 1))
        score, name, seq0 = top_candidates.pop(0)
        seed_deadline = time.time() + per_candidate
        refined = _adam_refine(seq0, seed_deadline)
        refined_score = _fast_evaluation(refined)
        print(f"[refined {name}] score {refined_score:.6f}")
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined
            print(f"*** new global best {best_score:.6f}")

    # If still None, fallback uniform
    if best_seq is None:
        best_seq = np.full(n_small, 1.0 / n_small)

    # upscale to final length
    x_small = np.arange(n_small, dtype=float)
    x_big = np.linspace(0, n_small - 1, n_big)
    up_seq = np.interp(x_big, x_small, best_seq)
    up_seq = np.maximum(up_seq, 0.0)
    up_seq /= up_seq.sum()

    # final refinement
    if time.time() < deadline:
        print("[final] refining full-size sequence")
        final_seq = _adam_refine(up_seq, deadline)
        final_score = _fast_evaluation(final_seq)
        print(f"[final] final score {final_score:.6f}")
        if final_score < best_score:
            best_seq = final_seq
            best_score = final_score

    print(f"[summary] metric {best_score:.6f} length {best_seq.size}")
    return best_seq.tolist()

"""
propose_candidate
-----------------
A multi‑stage optimizer for the “self‑convolution peak” metric.

Stage 1 – seed generation
    • Symmetric and asymmetric Beta(α,β) densities with a wide range of
      parameters (including α<0.5 where the distribution is more edge‑heavy).
    • Power‑law shapes a_i ∝ (i+½)^pL·(n‑i‑½)^pR, both symmetric and skewed.
    • Classical windows (Hann, Bartlett, Blackman) and sin‑power profiles.
    • Random Dirichlet draws with several concentrations.

All seeds are normalised to sum = 1 and scored with a fast FFT‑based
evaluation of the metric.  The best ~15 candidates are kept.

Stage 2 – local refinement
    An Adam‑style exponentiated‑gradient descent on the simplex is applied
    to each top seed (log‑space updates, per‑coordinate moments, active‑set
    sub‑gradient of the max‑operator).  Time is split evenly among the
    candidates.

Stage 3 – final upscale & polishing
    The best candidate is linearly interpolated to the target length
    (n = 5000) and refined again with the remaining budget.

The returned list contains non‑negative floats; the harness will call
`evaluate_sequence` on it.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast metric (identical to evaluate_sequence but without security checks)
# ----------------------------------------------------------------------
def _fast_evaluation(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) / (sum(seq)²)  –  n = len(seq)."""
    n = seq.size
    if n == 0:
        return float("inf")
    # FFT length: next power of two ≥ 2·n‑1
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Seed factories
# ----------------------------------------------------------------------
def _beta_sequence(alpha: float, beta: float, length: int) -> np.ndarray:
    """Beta(α,β) density sampled at the mid‑points of the interval."""
    xs = (np.arange(length, dtype=float) + 0.5) / length
    a = np.power(xs, alpha - 1.0, out=np.empty_like(xs))
    b = np.power(1.0 - xs, beta - 1.0, out=np.empty_like(xs))
    seq = a * b
    seq[np.isinf(seq)] = 0.0
    seq = np.maximum(seq, 0.0)
    if seq.sum() == 0.0:
        seq = np.full(length, 1.0 / length)
    else:
        seq /= seq.sum()
    return seq


def _random_dirichlet(length: int, concentration: float = 0.5) -> np.ndarray:
    """Draw a point from Dirichlet(concentration,…,concentration)."""
    seq = np.random.gamma(concentration, 1.0, size=length)
    if seq.sum() == 0.0:
        seq = np.full(length, 1.0 / length)
    else:
        seq /= seq.sum()
    return seq


def _sin_power_sequence(p: float, length: int) -> np.ndarray:
    """a_i ∝ sin(π·(i+0.5)/length)^p  (p may be negative)."""
    xs = (np.arange(length, dtype=float) + 0.5) / length
    vals = np.sin(np.pi * xs) ** p
    vals = np.maximum(vals, 0.0)
    if vals.sum() == 0.0:
        vals = np.full(length, 1.0 / length)
    else:
        vals /= vals.sum()
    return vals


def _powerlaw_sequence(p_left: float, p_right: float, length: int) -> np.ndarray:
    """
    a_i ∝ (i+½)^p_left · (n‑i‑½)^p_right .
    Allows independent power‑law behaviour at the left and right edges.
    """
    i = np.arange(length, dtype=float)
    left = np.power(i + 0.5, p_left, out=np.empty_like(i))
    right = np.power(length - i - 0.5, p_right, out=np.empty_like(i))
    seq = left * right
    seq = np.maximum(seq, 0.0)
    if seq.sum() == 0.0:
        seq = np.full(length, 1.0 / length)
    else:
        seq /= seq.sum()
    return seq


def _window_hann(length: int) -> np.ndarray:
    """Classic Hann (raised‑cosine) window."""
    i = np.arange(length, dtype=float)
    seq = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (length - 1)))
    seq = np.maximum(seq, 0.0)
    seq /= seq.sum()
    return seq


def _window_bartlett(length: int) -> np.ndarray:
    """Triangular (Bartlett) window."""
    i = np.arange(length, dtype=float)
    seq = 1.0 - np.abs((i - (length - 1) / 2.0) / ((length - 1) / 2.0))
    seq = np.maximum(seq, 0.0)
    seq /= seq.sum()
    return seq


def _window_blackman(length: int) -> np.ndarray:
    """Blackman window – more aggressive tapering."""
    i = np.arange(length, dtype=float)
    a0 = 0.42
    a1 = 0.5
    a2 = 0.08
    seq = (a0
           - a1 * np.cos(2.0 * np.pi * i / (length - 1))
           + a2 * np.cos(4.0 * np.pi * i / (length - 1)))
    seq = np.maximum(seq, 0.0)
    seq /= seq.sum()
    return seq


# ----------------------------------------------------------------------
# Adam‑style exponentiated‑gradient optimiser on the simplex
# ----------------------------------------------------------------------
def _adam_refine(seq: np.ndarray, deadline: float) -> np.ndarray:
    """
    Multiplicative Adam updates on log‑coefficients.
    Stops when ``time.time()`` exceeds ``deadline`` and returns the
    best vector seen during the run.
    """
    n = seq.size
    eps = 1e-15
    log_seq = np.log(seq + eps)

    beta1 = 0.9
    beta2 = 0.999
    lr = 0.12
    m = np.zeros_like(seq)
    v = np.zeros_like(seq)
    t = 0

    # FFT length (next power of two)
    fft_len = 1 << ((2 * n - 1).bit_length())

    best_seq = None
    best_score = float("inf")
    it = 0

    while time.time() < deadline:
        # current vector on the simplex
        seq = np.exp(log_seq)
        sum_seq = seq.sum()
        if sum_seq == 0.0:
            seq = np.full(n, 1.0 / n)
            log_seq = np.log(seq)
        else:
            seq /= sum_seq

        # convolution & active set
        fa = np.fft.rfft(seq, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        M = conv.max()
        tol = 1e-12
        active = np.where(conv >= M - tol)[0]

        # sub‑gradient of the max‑operator
        g = np.zeros_like(seq)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            i_range = np.arange(lo, hi + 1, dtype=int)
            j_range = k - i_range
            g[i_range] += seq[j_range]

        # Adam moment updates
        t += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # log‑space multiplicative step
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        log_seq -= step
        np.clip(log_seq, -30.0, 30.0, out=log_seq)

        # keep the best encountered vector
        score = _fast_evaluation(seq)
        if score < best_score - 1e-12:
            best_score = score
            best_seq = seq.copy()
            if it % 500 == 0:
                print(f"[Adam] iter {it:6d}  best {best_score:.6f}")

        # occasional jitter to escape plateaus
        if np.random.rand() < 0.001:
            j = np.random.randint(n)
            seq[j] = max(0.0, seq[j] + np.random.randn() * 1e-3)
            seq /= seq.sum()
            log_seq = np.log(seq + eps)

        it += 1

    # fallback – return the last normalised point if nothing improved
    if best_seq is None:
        seq = np.exp(log_seq)
        seq /= seq.sum()
        best_seq = seq
    return best_seq


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Produce a step‑function (list of non‑negative floats) with a low
    self‑convolution peak.  The routine respects the time budget,
    explores a diverse family of parametric shapes, refines the most
    promising candidates with Adam, and finally returns the best vector
    found (length ≈ 5000).
    """
    np.random.seed(seed)
    start = time.time()
    # leave a small safety margin
    deadline = start + budget_s - 5.0

    # ------------------------------------------------------------------
    # 1.  Exploration length (fast evaluation)
    # ------------------------------------------------------------------
    n_small = 2000
    n_big = 5000

    # ------------------------------------------------------------------
    # 2.  Build a large pool of parametric seeds
    # ------------------------------------------------------------------
    seeds = []

    # symmetric beta – wide range of α
    alphas = [0.05, 0.10, 0.15, 0.20, 0.25, 0.30,
              0.35, 0.40, 0.45, 0.50, 0.60, 0.80, 1.00]
    for a in alphas:
        seeds.append((f"beta_sym_{a:.2f}", _beta_sequence(a, a, n_small)))

    # asymmetric beta pairs (edge‑heavy vs centre‑heavy)
    asym_pairs = [(0.05, 0.90), (0.10, 0.90), (0.20, 0.80),
                  (0.30, 0.70), (0.40, 0.60), (0.60, 0.40),
                  (0.70, 0.30), (0.80, 0.20), (0.90, 0.10), (0.90, 0.05)]
    for a, b in asym_pairs:
        seeds.append((f"beta_{a:.2f}_{b:.2f}", _beta_sequence(a, b, n_small)))

    # sin‑power family (including negative exponents)
    p_vals = [-0.9, -0.8, -0.7, -0.6, -0.5, -0.4,
              -0.3, -0.2, -0.1, 0.0, 0.1, 0.2, 0.3]
    for p in p_vals:
        seeds.append((f"sin_pow_{p:.1f}", _sin_power_sequence(p, n_small)))

    # power‑law end shapes (both symmetric and skewed)
    exps = [-0.9, -0.7, -0.5, -0.3, -0.1, 0.0, 0.2, 0.4]
    for pl in exps:
        for pr in exps:
            # skip the trivial uniform case and a few duplicates
            if pl == pr == 0.0:
                continue
            name = f"pow_{pl:.1f}_{pr:.1f}"
            seeds.append((name, _powerlaw_sequence(pl, pr, n_small)))

    # classic window shapes
    seeds.append(("hann", _window_hann(n_small)))
    seeds.append(("bartlett", _window_bartlett(n_small)))
    seeds.append(("blackman", _window_blackman(n_small)))

    # random Dirichlet draws – low & medium concentration
    for i in range(5):
        seeds.append((f"rand_dir_{i}", _random_dirichlet(n_small, concentration=0.4)))
    for i in range(3):
        seeds.append((f"rand_dir_high_{i}", _random_dirichlet(n_small, concentration=0.8)))

    # ------------------------------------------------------------------
    # 3.  Score all seeds quickly and keep the best few
    # ------------------------------------------------------------------
    scored = []
    for name, seq in seeds:
        seq = np.maximum(seq, 0.0)
        if seq.sum() == 0.0:
            seq = np.full_like(seq, 1.0 / seq.size)
        else:
            seq = seq / seq.sum()
        score = _fast_evaluation(seq)
        scored.append((score, name, seq))
        print(f"[seed {name}] raw score {score:.6f}")

    scored.sort(key=lambda x: x[0])
    top_k = min(15, len(scored))
    top_candidates = scored[:top_k]
    print(f"[selection] retaining top {top_k} seeds for refinement")

    # ------------------------------------------------------------------
    # 4.  Adam refinement of the shortlisted seeds (time‑sliced)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    while top_candidates and time.time() < deadline:
        # allocate a fair slice of the remaining budget
        time_left = deadline - time.time()
        per_seed = max(5.0, time_left / (len(top_candidates) + 1))
        score, name, seq0 = top_candidates.pop(0)

        seed_deadline = time.time() + per_seed
        refined = _adam_refine(seq0, seed_deadline)
        refined_score = _fast_evaluation(refined)
        print(f"[refined {name}] score {refined_score:.6f}")

        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined
            print(f"*** new global best {best_score:.6f}")

    # fallback – uniform if everything failed
    if best_seq is None:
        best_seq = np.full(n_small, 1.0 / n_small)

    # ------------------------------------------------------------------
    # 5.  Upscale to the final length and run a final polish
    # ------------------------------------------------------------------
    x_small = np.arange(n_small, dtype=float)
    x_big = np.linspace(0, n_small - 1, n_big)
    up_seq = np.interp(x_big, x_small, best_seq)
    up_seq = np.maximum(up_seq, 0.0)
    up_seq /= up_seq.sum()

    if time.time() < deadline:
        print("[final] starting full‑size refinement")
        final_seq = _adam_refine(up_seq, deadline)
        final_score = _fast_evaluation(final_seq)
        print(f"[final] final metric {final_score:.6f}")
        if final_score < best_score:
            best_seq = final_seq
            best_score = final_score

    print(f"[summary] best metric {best_score:.6f} (length {best_seq.size})")
    return best_seq.tolist()
