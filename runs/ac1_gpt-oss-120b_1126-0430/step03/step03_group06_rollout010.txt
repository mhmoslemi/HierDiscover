def _fft_convolve_full(a, b):
    size = 1 << ((len(a) + len(b) - 1).bit_length())
    Fa = np.fft.rfft(a, size)
    Fb = np.fft.rfft(b, size)
    conv = np.fft.irfft(Fa * Fb, size)
    return conv[:len(a)+len(b)-1]

def _smooth_max_and_grad(a, tau):
    conv = _fft_convolve(a)
    max_c = conv.max()
    # compute softmax weights
    # shift conv to avoid large exponent
    shifted = conv - max_c
    w = np.exp(shifted / tau)
    sum_w = w.sum()
    p = w / sum_w
    smooth_max = max_c + tau * np.log(sum_w)

    d = conv_rev[start : start + len(a)]
    grad = 4.0 * len(a) * d
    obj = 2 * len(a) * smooth_max

def _full_convolve(x, y):
    n = len(x)

    conv_self = _fft_convolve_self(a)

    min_tau = max_c * 1e-5 if max_c>0 else 1e-12
    decay = 0.9999

"""
Propose a step‑function that minimises
    O(a) = 2·n·max(a*a)   (a_i ≥ 0, Σ a_i = 1)

The optimizer works in three stages:

1.  A smoothed‑max gradient descent (soft‑max of the self‑convolution)
    with Adam‑style adaptive steps, projected onto the probability simplex.
    The temperature τ starts relatively large and is slowly annealed,
    giving a well‑behaved gradient that eventually concentrates on the
    active constraints.

2.  If time permits the solution is up‑sampled to a larger problem size
    (more degrees of freedom) and refined again – the larger discretisation
    can better approximate the continuous optimum.

3.  Cheap random restarts are performed while budget remains.

The code is completely self‑contained (only numpy and time are used) and
conforms to the competition interface – it returns a plain Python list of
non‑negative heights.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of a vector onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Catalan (1/√(1‑z)) coefficients, normalised to unit L¹‑norm
# ----------------------------------------------------------------------
def _catalan_normalised(n: int) -> np.ndarray:
    """First n Catalan‑type terms a_i = C(2i,i)/4**i, normalised to sum = 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# FFT‑based full linear convolution (real inputs)
# ----------------------------------------------------------------------
def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n).bit_length())                 # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)
    return conv[:n]


def _self_convolution(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Upsample a vector to a larger length by linear interpolation
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    old_n = a.shape[0]
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
# Smoothed‑max (log‑sum‑exp) objective and its gradient
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(a: np.ndarray, tau: float) -> tuple[float, np.ndarray]:
    """
    Returns (objective, gradient) where the objective is
        2·n·smooth_max( a*a )
    and the gradient is the exact gradient of that smoothed objective.
    """
    n = a.shape[0]

    # self‑convolution and its (exact) maximum
    conv = _self_convolution(a)                # length 2n‑1
    max_c = conv.max()

    # soft‑max weights over the convolution entries
    shifted = conv - max_c                    # ≤ 0
    # avoid division by zero – tau is always > 0
    w = np.exp(shifted / tau)                # may underflow to 0 for far‑away entries
    sum_w = w.sum()
    if sum_w == 0.0:                         # numerical safeguard
        p = np.zeros_like(w)
        p[conv.argmax()] = 1.0
        smooth_max = max_c
    else:
        p = w / sum_w
        smooth_max = max_c + tau * np.log(sum_w)

    # gradient: d = Σ_k p_k J_k a   ;  grad = 4·n·d
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, p)       # length 3n‑2
    start = n - 1
    d = conv_rev[start:start + n]            # slice of length n
    grad = 4.0 * n * d

    obj = 2.0 * n * smooth_max
    return float(obj), grad


# ----------------------------------------------------------------------
# Main optimiser – smoothed max + Adam + simplex projection
# ----------------------------------------------------------------------
def _optimize_smoothing(a0: np.ndarray,
                        deadline: float,
                        max_iters: int = 300_000,
                        verbose: bool = False) -> tuple[np.ndarray, float]:
    """
    Refine a starting vector a0 (already Σa = 1) using a smoothed‑max
    objective.  Returns the best vector encountered and its exact objective
    value (2·n·max(a*a)).
    """
    a = a0.copy()
    n = a.shape[0]

    # initial exact objective (used for best‑track)
    conv0 = _self_convolution(a)
    best_val = 2.0 * n * conv0.max()
    best_a = a.copy()

    # Adam hyper‑parameters
    lr = 0.12
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    b1_pow = 1.0
    b2_pow = 1.0

    # temperature schedule
    max_c0 = conv0.max()
    tau = max_c0 * 0.2 if max_c0 > 0 else 1e-8
    min_tau = max_c0 * 1e-5 if max_c0 > 0 else 1e-12
    decay = 0.9995

    it = 0
    while time.time() < deadline and it < max_iters:
        # ---- smoothed objective & gradient ---------------------------------
        obj, grad = _smooth_obj_and_grad(a, tau)

        # ---- Adam update ----------------------------------------------------
        b1_pow *= beta1
        b2_pow *= beta2
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - b1_pow)
        v_hat = v / (1.0 - b2_pow)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---- projection onto the simplex ------------------------------------
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        # ---- evaluate exact objective periodically ---------------------------
        if (it & 1023) == 0:            # every 1024 iterations
            conv = _self_convolution(a)
            cur_val = 2.0 * n * conv.max()
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                if verbose:
                    print(f"[improve] iter {it:06d}  obj {best_val:.9f}")

        # ---- temperature annealing -------------------------------------------
        tau = max(min_tau, tau * decay)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine performs a smoothed‑max gradient
    descent on a Catalan start, optionally refines an up‑sampled version,
    and finishes with a few random restarts.
    """
    np.random.seed(seed)
    # leave a couple of seconds for the harness
    deadline = time.time() + budget_s - 2.0

    # ------------------------------------------------------------------
    # Phase 1 – optimise at a moderate size (n = 20000)
    # ------------------------------------------------------------------
    n1 = 20_000
    a_start = _catalan_normalised(n1)

    best_seq, best_val = _optimize_smoothing(a_start, deadline,
                                             max_iters=250_000,
                                             verbose=True)

    # ------------------------------------------------------------------
    # Phase 2 – if time permits, up‑sample and continue optimisation
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 40.0:  # need enough room for a second run
        n2 = 50_000
        a_up = _upsample(best_seq, n2)
        seq2, val2 = _optimize_smoothing(a_up, deadline,
                                         max_iters=150_000,
                                         verbose=False)
        if val2 < best_val:
            best_seq, best_val = seq2, val2
            print(f"[up‑sample] improvement to {best_val:.9f}")

    # ------------------------------------------------------------------
    # Phase 3 – cheap random restarts while time remains
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        cur_n = best_seq.shape[0]
        a_rand = np.random.rand(cur_n)
        a_rand /= a_rand.sum()
        seq_r, val_r = _optimize_smoothing(a_rand, deadline,
                                           max_iters=80_000,
                                           verbose=False)
        if val_r < best_val:
            best_seq, best_val = seq_r, val_r
            print(f"[restart] new best obj {best_val:.9f}")

    # Final sanity print (the evaluation harness will call evaluate_sequence)
    print(f"[final] best objective ≈ {best_val:.9f} with n={best_seq.shape[0]}")
    return best_seq.tolist()
