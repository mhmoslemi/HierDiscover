def _optimize_gradient(a0, deadline, max_iters=200_000, verbose=True):
    n = a0.shape[0]
    a = a0.copy()
    size = 1 << ((3*n-2).bit_length())
    best = a.copy()
    best_val = _objective(a)  # compute conv max using _objective (fast conv)
    # momentum or Adam
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.05  # initial learning rate maybe
    lambda_ = 20000.0  # smoothing parameter

    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        # compute FFT once
        fa = np.fft.rfft(a, size)
        # convolution
        conv_full = np.fft.irfft(fa*fa, size)
        conv = conv_full[: 2*n - 1]

        # evaluate objective
        cur_obj = 2.0 * n * np.max(conv)
        if cur_obj < best_val:
            best_val = cur_obj
            best = a.copy()
            if verbose:
                print(f"[improve] iter {t} obj {best_val:.9f}")

        # softmax weights for max approximation
        conv_shift = conv - np.max(conv)  # shift for numeric stability
        w = np.exp(lambda_ * conv_shift)
        w_sum = w.sum()
        w /= w_sum

        # correlation to get gradient wrt a
        rev_w = w[::-1]
        fw_rev = np.fft.rfft(rev_w, size)
        corr_full = np.fft.irfft(fa * fw_rev, size)
        start = 2*n - 2
        grad_a = 4.0 * n * corr_full[start: start + n]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_a
        v = beta2 * v + (1 - beta2) * (grad_a ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        lr = lr0 / np.sqrt(t)  # diminishing
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        # enforce non-negativity via projection onto simplex
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        # occasional random perturbation
        if t % 5000 == 0:
            pert = np.random.normal(scale=0.001, size=n)
            a = np.maximum(a + pert, 0.0)
            a = _project_simplex(a)

    return best, best_val

"""
Propose a step‑function (non‑negative sequence) that approximately minimises

    O(a) = 2·n·max_k (a * a)[k]   with   Σ a = 1 , a_i ≥ 0.

The optimisation is performed on the probability simplex using a
soft‑max (exponential) parametrisation a_i = exp(y_i)/∑_j exp(y_j).
A smooth approximation of the max (log‑sum‑exp with a growing
temperature) provides a dense gradient, which is then turned into a
gradient w.r.t. the logits y via the Jacobian of the soft‑max.
The gradient of the smoothed objective is computed efficiently with
FFT‑based convolutions (auto‑convolution of a and the correlation of a
with the soft‑max weights).  Adam is used to adapt the step size.
The algorithm works with a large number of coefficients (≈50 000)
and runs for as long as the allotted time budget permits.  Several
random restarts (Catalan, Dirichlet and Beta‑shaped starts) are added
to avoid local minima.

The returned list is directly usable by `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: next power of two
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
# FFT based full linear convolution a*a (length 2·n‑1)
# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Projection onto the simplex (not used in the main optimiser,
# kept for possible fall‑backs)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Catalan distribution (first n Catalan numbers, normalised to sum 1)
# ----------------------------------------------------------------------
def _catalan_normalised(n: int) -> np.ndarray:
    """Catalan coefficients a_i = C(2i,i)/4**i, normalised to Σa = 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Beta(α,α)‑like distribution, normalised to Σa = 1
# ----------------------------------------------------------------------
def _beta_normalised(n: int, alpha: float = 0.7) -> np.ndarray:
    """Density ∝ x^{α‑1}(1‑x)^{α‑1} on {0,…,n‑1}, normalised."""
    i = np.arange(n) + 0.5          # avoid the exact 0/1 endpoints
    x = i / n
    a = np.power(x * (1.0 - x), alpha - 1.0)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Objective (exact value returned by evaluate_sequence for Σa=1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """2·n·max(a*a) for a already normalised to sum 1."""
    n = a.shape[0]
    return float(2.0 * n * np.max(_fft_convolve(a)))


# ----------------------------------------------------------------------
# Core optimiser – smooth‑max gradient + Adam on log‑probabilities
# ----------------------------------------------------------------------
def _optimize_softmax(a0: np.ndarray,
                      deadline: float,
                      max_iters: int = 200_000,
                      verbose: bool = True) -> tuple[np.ndarray, float]:
    """
    Optimises a probability vector a (Σa=1, a_i≥0) by maximising a smooth
    approximation of the max‑convolution term and using Adam on the
    log‑odds y (a = softmax(y)).  Returns the best vector found and its
    exact objective value.
    """
    n = a0.shape[0]
    size_fft = _next_pow2(3 * n - 2)          # enough for correlation a↔w

    # initialise logits y so that softmax(y) == a0
    y = np.log(a0 + 1e-30)

    # Adam parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(y)
    v = np.zeros_like(y)

    # schedule for the smooth‑max temperature λ
    lam_start, lam_end = 2_000.0, 20_000.0

    # keep the best exact solution encountered
    best_a = np.exp(y - np.max(y))
    best_a /= best_a.sum()
    best_val = _objective(best_a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ---------- current distribution ----------
        # soft‑max (stable)
        max_y = np.max(y)
        exp_y = np.exp(y - max_y)
        a = exp_y / exp_y.sum()

        # ---------- convolution a*a ----------
        fa = np.fft.rfft(a, size_fft)
        conv_full = np.fft.irfft(fa * fa, size_fft)
        conv = conv_full[: 2 * n - 1]                # length 2n‑1

        # exact objective (for bookkeeping)
        cur_obj = 2.0 * n * np.max(conv)
        if cur_obj < best_val:
            best_val = cur_obj
            best_a = a.copy()
            if verbose:
                print(f"[improve] iter {t:06d}  obj {best_val:.9f}")

        # ---------- smooth‑max weights ----------
        lam = lam_start + (lam_end - lam_start) * (t / max_iters)
        conv_max = np.max(conv)
        # shift to avoid overflow/underflow
        shifted = conv - conv_max
        w_raw = np.exp(lam * shifted)
        w = w_raw / w_raw.sum()                         # shape (2n‑1,)

        # ---------- gradient w.r.t. a ----------
        # correlation of w (reversed) with a gives Σ_i w_i a_{i‑j}
        rev_w = w[::-1]
        fw_rev = np.fft.rfft(rev_w, size_fft)
        corr_full = np.fft.irfft(fa * fw_rev, size_fft)
        start = 2 * n - 2
        grad_a = 4.0 * n * corr_full[start: start + n]   # length n

        # ---------- gradient w.r.t. logits ----------
        avg = np.dot(a, grad_a)
        grad_y = a * (grad_a - avg)

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = 0.01 * m_hat / (np.sqrt(v_hat) + eps)   # lr≈0.01
        y -= step

        # occasional tiny random perturbation to escape plateaus
        if t % 5_000 == 0:
            y += np.random.normal(scale=0.001, size=n)

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Return a sequence of non‑negative heights that (within the time
    budget) tries to minimise `evaluate_sequence`.  The routine uses
    a large‑scale Adam optimiser on a smooth approximation of the
    objective, starting from a Catalan distribution and performing a
    few random restarts (Dirichlet and Beta‑shaped starts).
    """
    np.random.seed(seed)

    # Leave a tiny safety margin for the harness
    deadline = time.time() + budget_s - 2.0

    # --------------------------------------------------------------
    # 1) problem size – 50 000 gives a good trade‑off between quality
    #    and speed (FFT size ≈ 2²⁰, still very fast).
    # --------------------------------------------------------------
    n = 50_000

    # --------------------------------------------------------------
    # 2) initial candidate – Catalan distribution (good analytical start)
    # --------------------------------------------------------------
    a0 = _catalan_normalised(n)

    best_seq, best_val = _optimize_softmax(
        a0, deadline, max_iters=150_000, verbose=True
    )

    # --------------------------------------------------------------
    # 3) random restarts while time permits
    # --------------------------------------------------------------
    while time.time() < deadline - 5.0:
        # choose a random starting shape
        if np.random.rand() < 0.5:
            # Beta‑like start with a random alpha ∈ (0.55, 0.95)
            alpha = np.random.uniform(0.55, 0.95)
            a_start = _beta_normalised(n, alpha)
        else:
            # plain Dirichlet (uniform on the simplex)
            a_start = np.random.rand(n)
            a_start /= a_start.sum()

        a_opt, val = _optimize_softmax(
            a_start, deadline, max_iters=60_000, verbose=False
        )
        if val < best_val:
            best_val = val
            best_seq = a_opt.copy()
            print(f"[restart] new best obj {best_val:.9f}")

    # --------------------------------------------------------------
    # 4) final sanity check (the framework will call evaluate_sequence)
    # --------------------------------------------------------------
    final_obj = _objective(best_seq)
    print(f"[final] best objective ≈ {final_obj:.9f} with n={n}")

    # return as plain Python list
    return best_seq.tolist()
