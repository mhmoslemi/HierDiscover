"""
Height‑sequence optimiser.

The optimiser works in three stages:

1. **Coarse parametric search** – a family a_i ∝ (i+δ)^{-p} (p∈(0,1), δ≥0)
   is evaluated on a fixed length (default 2048).  The best (p,δ) is kept.

2. **Exponentiated‑gradient refinement** – starting from the best parametric
   vector we run a mirror‑descent (EG) loop on the true objective
   O=2·n·max(conv(a,a)) while staying on the probability simplex.
   An adaptive learning‑rate and a small momentum term are used.

3. **Linear‑programming step (optional)** – given the current a,
   we solve the LP
       maximise Σ b_j
       s.t.   a * b ≤ M   (point‑wise),   b_j ≥ 0
   where M = max(conv(a,a)).  The optimiser g = b / Σb is mixed with the
   current a (convex combination).  This is exactly the “LP‑step’’ described
   in the statement and often yields a further reduction.

All steps are time‑capped (default 1000 s).  The best sequence found is
returned as a plain list of floats.
"""

import time
import numpy as np

# optional LP support
try:
    from scipy.optimize import linprog
    import scipy.sparse as sp
    _LP_AVAILABLE = True
except Exception:               # pragma: no cover
    _LP_AVAILABLE = False


# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a power‑of‑two FFT (real part only)."""
    n = a.size
    size = 1 << ((2 * n - 2).bit_length())
    A = np.fft.fft(a, n=size)
    conv = np.fft.ifft(A * A).real[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return the bound 2·n·max(conv(a,a)) for a already normalised (Σa=1)."""
    n = a.size
    conv = _conv_fft(a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv(a,a)) w.r.t. a."""
    n = a.size
    conv = _conv_fft(a)
    max_val = conv.max()
    ks = np.where(np.abs(conv - max_val) < 1e-12)[0]
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]          # derivative of a_i*a_j (both orders)
    if ks.size:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
def _exp_update(a: np.ndarray, g: np.ndarray, eta: float) -> np.ndarray:
    """Exponentiated‑gradient update on the simplex."""
    log_a = np.log(a + 1e-15) - eta * g
    log_a -= np.max(log_a)               # keep numbers moderate
    a_new = np.exp(log_a)
    a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
def _exp_gradient_descent(a0: np.ndarray,
                         max_iters: int,
                         eta0: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent optimiser (EG) with a linearly decaying learning rate.
    Returns the best vector seen and its objective value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)

    n = a.size
    no_improve = 0
    patience = max_iters // 30

    for t in range(max_iters):
        if time.time() >= deadline:
            break

        g = _subgradient_max_conv(a)
        eta = eta0 * (1.0 - t / max_iters)

        a = _exp_update(a, g, eta)

        cur = _objective(a)
        if cur < best_score - 1e-12:
            best_score = cur
            best_a = a.copy()
            no_improve = 0
            print(f"[EG] iter {t:5d}  new best O={best_score:.6f}")
        else:
            no_improve += 1
            if no_improve >= patience:
                # small jitter to escape plateaus
                jitter = np.random.randn(n) * 1e-5
                a = best_a + jitter
                a = np.clip(a, 0.0, None)
                a = _project_to_simplex(a)
                no_improve = 0
    return best_a, best_score


# ----------------------------------------------------------------------
def _build_lp_matrix(a: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2n‑1)×n matrix that encodes the constraints
        (a * b)[k] = Σ_i a_i b_{k-i} ≤ M .
    The matrix entries are a_i placed on the diagonal band.
    """
    n = a.shape[0]
    rows = []
    cols = []
    data = []
    # iterate over i (fixed a_i) and fill its contribution to all shifts
    for i in range(n):
        a_i = a[i]
        # for each j, k=i+j
        row_idx = i + np.arange(n)           # shape (n,)
        col_idx = np.arange(n)               # (j)
        rows.append(row_idx)
        cols.append(col_idx)
        data.append(np.full(n, a_i))
    rows = np.concatenate(rows)
    cols = np.concatenate(cols)
    data = np.concatenate(data)
    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A


# ----------------------------------------------------------------------
def _lp_step(a: np.ndarray,
             M: float,
             deadline: float,
             mixing: float = 0.5) -> tuple[np.ndarray, bool]:
    """
    One LP‑based improvement step.
    Returns the improved vector (or the original) and a flag indicating success.
    """
    if not _LP_AVAILABLE:
        return a, False
    n = a.shape[0]

    # build constraints a * b ≤ M  (point‑wise)
    A = _build_lp_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    c_obj = -np.ones(n)                     # maximise Σ b_j

    # Use HiGHS (fast) – set a generous iteration limit but respect deadline
    lp_deadline = min(deadline, time.time() + 15.0)
    options = {"presolve": True, "time_limit": max(0.0, lp_deadline - time.time())}

    res = linprog(c=c_obj,
                  A_ub=A,
                  b_ub=b_ub,
                  bounds=[(0, None)] * n,
                  method="highs",
                  options=options)

    if res.status != 0 or not res.success:
        return a, False

    b_opt = res.x
    sum_b = b_opt.sum()
    if sum_b <= 0:
        return a, False

    # normalise to a probability distribution and mix
    g = b_opt / sum_b
    a_new = (1.0 - mixing) * a + mixing * g
    a_new = _project_to_simplex(a_new)      # re‑project for numerical safety
    return a_new, True


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The function respects the 1000 s time limit.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Choose a working length
    if 'height_sequence_1' in globals():
        a_prev = np.asarray(height_sequence_1, dtype=float)
        n = a_prev.size
        a_prev = a_prev / a_prev.sum()
    else:
        n = 2048                       # power‑of‑two friendly size
        a_prev = None

    # ------------------------------------------------------------------
    # Helper to evaluate a candidate quickly (assumes sum==1)
    def quick_obj(v):
        return _objective(v)

    # ------------------------------------------------------------------
    # Phase 1 – coarse parametric search
    best_a = None
    best_score = float("inf")
    # parameter grids
    p_grid = np.linspace(0.45, 0.70, 31)            # 0.45 … 0.70
    shift_grid = np.linspace(0, n * 0.15, 21)       # up to ~15% of n

    for p in p_grid:
        if time.time() >= deadline:
            break
        for delta in shift_grid:
            idx = np.arange(1, n + 1, dtype=float) + delta
            # power‑law shape
            a = idx ** (-p)
            a /= a.sum()
            score = quick_obj(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                print(f"[grid] p={p:.3f} δ={delta:.1f}  O={best_score:.6f}")

    # also consider the previous best if available
    if a_prev is not None:
        score_prev = quick_obj(a_prev)
        if score_prev < best_score:
            best_score = score_prev
            best_a = a_prev.copy()
            print(f"[init] using previous sequence  O={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient refinement
    if best_a is None:
        # fallback – simple 1/√ law
        idx = np.arange(1, n + 1, dtype=float)
        best_a = (1.0 / np.sqrt(idx))
        best_a /= best_a.sum()
        best_score = quick_obj(best_a)

    # EG parameters (tuned empirically)
    max_iters = 25000
    eta0 = 0.55

    a_opt, score_opt = _exp_gradient_descent(best_a,
                                             max_iters=max_iters,
                                             eta0=eta0,
                                             deadline=deadline)
    if score_opt < best_score:
        best_score = score_opt
        best_a = a_opt.copy()
        print(f"[EG] improved to O={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – optional LP polishing (run while time permits)
    while time.time() < deadline:
        # current bound
        M = _conv_fft(best_a).max()
        # try an LP step with a modest mixing factor
        best_a, succeeded = _lp_step(best_a, M, deadline, mixing=0.3)
        if not succeeded:
            break
        new_score = quick_obj(best_a)
        if new_score < best_score - 1e-12:
            best_score = new_score
            print(f"[LP] new best O={best_score:.6f}")
        else:
            # no improvement -> stop LP loop
            break

    # ------------------------------------------------------------------
    # Final exact evaluation by the supplied function
    seq = best_a.tolist()
    try:
        final_val = evaluate_sequence(seq)
        print(f"[final] evaluate_sequence reports O={final_val:.6f}")
    except Exception as e:  # pragma: no cover
        print("[final] evaluate_sequence failed:", e)
        final_val = None

    # Return the plain list (the environment will use it)
    return seq
