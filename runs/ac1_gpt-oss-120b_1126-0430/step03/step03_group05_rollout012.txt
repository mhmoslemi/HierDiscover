#!/usr/bin/env python
"""
Propose a step‑function (list of non‑negative heights) that minimizes
``evaluate_sequence``.

Algorithm
---------
1. Choose a working length n (≥ 8000, or reuse the previous construction).
2. Search a two–parameter family of vectors
        a_i ∝ (i+1)^(-p) · q^i ,   q = exp(-α/n) .
   The parameters p∈[0,0.8] and α≈1.0–3.5 are explored on a coarse grid.
   For every candidate we compute an inexpensive “cheap score”
        S = 2·n·max(a*a)      (the denominator Σa_i is 1, so this equals the
   true objective up to the fixed normalisation).
   The best candidate from the grid is kept.
3. Refine the best candidate with a projected Adam optimisation that
   directly minimises the exact objective  f(a)=max_k (a*a)[k] .
   * The full convolution is obtained by FFT (O(n log n)).
   * The sub‑gradient of the active peak(s) is
        g_i = Σ_{k active} 2·a_{k-i}   (zero where the index is out of range).
   * Adam updates are performed and the iterate is projected onto the
     probability simplex (Σa_i=1, a_i≥0).
   * If no improvement is seen for many iterations a small random
     perturbation of the current best vector is applied (escape from plateaus).
4. After the time budget the best vector found is evaluated with the
   provided ``evaluate_sequence`` and returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top level – no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (Σv=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all entries non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Exact linear convolution a*a via FFT (real part, length 2·len(a)-1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa).real[: 2 * a.size - 1]
    return conv


def _subgradient_active(a: np.ndarray, conv: np.ndarray,
                        thresh: float) -> np.ndarray:
    """
    Sub‑gradient of f(a)=max(conv) at the set of active indices
    where conv >= thresh.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    active = np.where(conv >= thresh)[0]
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        # a[k-idx] is always inside [0,n‑1] due to the range choice above
        g[idx] += 2.0 * a[k - idx]
    return g


def _parametric_sequence(n: int, p: float, alpha: float) -> np.ndarray:
    """
    Return a normalised vector
        a_i ∝ (i+1)^(-p) · q^i,   q = exp(-α/n),
    with Σ a_i = 1.
    """
    i = np.arange(n, dtype=np.float64)
    lam = alpha / n
    q = math.exp(-lam)
    # raw = (i+1)^(-p) * q**i
    if p == 0.0:
        raw = q ** i
    else:
        raw = (i + 1) ** (-p) * (q ** i)
    s = raw.sum()
    if s == 0.0:
        # fallback to uniform (should never happen)
        return np.full(n, 1.0 / n, dtype=np.float64)
    return raw / s


def _cheap_score(a: np.ndarray, fft_len: int) -> float:
    """2·n·max(a*a) for a normalised (Σa=1) vector."""
    n = a.shape[0]
    conv = _full_convolution(a, fft_len)
    maxb = conv.max()
    return 2.0 * n * maxb

# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising
    ``evaluate_sequence`` under the given time budget.
    Returns the best sequence found as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # --------------------------------------------------------------
    # 0. Determine working length
    # --------------------------------------------------------------
    try:
        prev = height_sequence_1               # may be defined by the platform
        n = max(len(prev), 8000)               # ensure a reasonable size
        prev_arr = np.asarray(prev, dtype=np.float64)
        if prev_arr.sum() == 0.0:
            prev_arr = None
        else:
            # normalise to sum = 1 for the optimisation
            prev_arr = prev_arr / prev_arr.sum()
    except Exception:
        prev_arr = None
        n = 15000                               # default working size

    # Cap n to keep FFTs cheap but still in the asymptotic regime
    if n > 20000:
        n = 20000

    fft_len = 1 << ((2 * n - 1).bit_length())   # next power of two ≥ 2n‑1

    # --------------------------------------------------------------
    # 1. Coarse grid search over the (p,α) family
    # --------------------------------------------------------------
    best_a = None
    best_score = float("inf")
    best_params = None

    # grids – a bit coarse, adjustable if time permits
    p_grid = np.linspace(0.0, 0.8, 21)          # 0,0.04,…,0.8
    alpha_grid = np.linspace(0.8, 3.5, 27)      # 0.8,0.92,…,3.5

    print(f"[Init] Working length n = {n}, FFT length = {fft_len}")

    for p in p_grid:
        for alpha in alpha_grid:
            if time.time() >= deadline:
                break
            a = _parametric_sequence(n, p, alpha)
            score = _cheap_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                best_params = (p, alpha)
                print(f"[Grid] New best – p={p:.3f}, α={alpha:.3f}, cheap={score:.6f}")

    # also evaluate the previous construction if available
    if prev_arr is not None and prev_arr.size == n:
        score = _cheap_score(prev_arr, fft_len)
        if score < best_score:
            best_score = score
            best_a = prev_arr.copy()
            best_params = ("prev", None)
            print(f"[Prev] Previous sequence beats grid: cheap={score:.6f}")

    # fallback (should never happen)
    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _cheap_score(best_a, fft_len)
        best_params = ("uniform", None)
        print("[Fallback] Using uniform distribution")

    print(f"[Init] Best after grid – params={best_params}, cheap={best_score:.6f}")

    # --------------------------------------------------------------
    # 2. Refine with projected Adam on the exact max‑convolution objective
    # --------------------------------------------------------------
    a = best_a.copy()                     # current iterate (Σa = 1)
    m = np.zeros_like(a)                  # first moment
    v_adam = np.zeros_like(a)             # second moment
    beta1 = 0.90
    beta2 = 0.999
    eps_adam = 1e-8
    base_lr = 0.6                         # base learning rate
    iter_count = 0
    last_improve = 0
    best_iter_a = a.copy()
    best_iter_score = best_score

    # restart parameters
    stagnation_limit = 60000               # iterations without improvement
    report_interval = 500                  # how often we print cheap score

    while time.time() < deadline and iter_count < 1_500_000:
        iter_count += 1

        # full convolution and maximum
        conv = _full_convolution(a, fft_len)
        maxb = conv.max()
        cheap = 2.0 * n * maxb

        # sub‑gradient from all near‑max indices
        rel_tol = 1e-8
        thresh = maxb - max(rel_tol * maxb, 1e-12)
        g = _subgradient_active(a, conv, thresh)

        # Adam update (descending direction)
        lr = base_lr / math.sqrt(iter_count)   # simple decay
        m = beta1 * m + (1.0 - beta1) * g
        v_adam = beta2 * v_adam + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** iter_count)
        v_hat = v_adam / (1.0 - beta2 ** iter_count)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = _simplex_projection(a_new)

        a = a_new

        # Track best cheap score
        if (iter_count % report_interval == 0) or (time.time() > deadline - 5):
            if cheap < best_iter_score:
                best_iter_score = cheap
                best_iter_a = a.copy()
                last_improve = iter_count
                print(f"[Iter {iter_count:07d}] NEW BEST cheap = {cheap:.6f}")

        # Stagnation handling – random perturbation of the best so far
        if iter_count - last_improve > stagnation_limit:
            pert = 0.02 * np.random.rand(n)
            a = _simplex_projection(best_iter_a + pert)
            m.fill(0.0)
            v_adam.fill(0.0)
            last_improve = iter_count
            print(f"[Iter {iter_count:07d}] Restart after stagnation (perturb).")

    # --------------------------------------------------------------
    # 3. Final exact evaluation with the provided function
    # --------------------------------------------------------------
    final_seq = best_iter_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[Result] Final exact score = {final_score:.6f}")
    except Exception as exc:   # defensive – should never happen
        final_score = float("inf")
        print(f"[Error] Evaluation failed: {exc}")

    # Store the best sequence globally for possible reuse in a later run
    global height_sequence_1
    height_sequence_1 = final_seq   # noqa: PLW0603

    return final_seq
