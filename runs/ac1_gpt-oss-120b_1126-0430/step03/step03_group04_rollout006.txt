def propose_candidate(seed=123, budget_s=1000):
    import time, numpy as np
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    # choose initial length
    base_n = 2000
    a = np.random.rand(base_n)
    a = _project_to_simplex(a)

    # ensure symmetry
    a = (a + a[::-1]) / 2
    a = a / a.sum()

    # log domain
    epsilon = 1e-12
    loga = np.log(a + epsilon)

    best_seq = a.copy()
    best_score = _objective(best_seq)

    # hyperparameters
    max_iter = 2000000  # high but break by time
    t = 0
    mu = 0.9
    v = np.zeros_like(a)

    # schedule parameters
    eta_start = 0.5
    eta_end = 0.02
    beta_start = 0.5
    beta_end = 2000.0

    # counters for stagnation
    last_improve_iter = 0

    while time.time() < deadline and t < max_iter:
        t += 1

        a = np.exp(loga - np.max(loga))
        a = a / a.sum()

        # evaluate
        conv = _conv_self_fft(a)
        M = conv.max()
        cur_score = 2.0 * a.size * M

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            last_improve_iter = t
            if t % 1000 == 0:
                print(f"[iter {t}] new best O={best_score:.6f}")

        # schedule beta and eta
        fraction = min(t / 500000.0, 1.0)  # fudge
        beta = beta_start + (beta_end - beta_start) * fraction
        eta = eta_start + (eta_end - eta_start) * fraction  # linear decay

        # compute softmax weights
        shifted = conv - M
        exp_vals = np.exp(beta * shifted)   # max weight is 1
        w = exp_vals / exp_vals.sum()

        # gradient
        g = 2.0 * _correlation_fft(w, a)

        # momentum update
        v = mu * v + (1 - mu) * g

        # log update (mirror descent)
        loga = loga - eta * v

        # enforce symmetry
        if t % 5 == 0:   # enforce occasionally to reduce cost
            # reconstruct a for symmetry enforcement
            a_temp = np.exp(loga - np.max(loga))
            a_temp = a_temp / a_temp.sum()
            a_sym = (a_temp + a_temp[::-1]) / 2
            # keep same loga by taking log of symmetric
            loga = np.log(a_sym + epsilon)

        # occasional jitter
        if t % 10000 == 0:
            jitter = np.random.randn(a.size) * 1e-3
            loga += jitter

        # restart if stagnation
        if t - last_improve_iter > 50000:
            # reinitialize around best
            a0 = best_seq.copy()
            # add small noise
            a0 = a0 + np.random.randn(a0.size) * 5e-3
            a0 = _project_to_simplex(a0)
            a0 = (a0 + a0[::-1]) / 2
            loga = np.log(a0 + epsilon)
            v = np.zeros_like(a)
            last_improve_iter = t
            print(f"[restart] iter {t}, reset from best")

    # final best evaluation using evaluate_sequence
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence O={final_score:.6f}")
    except Exception as e:
        print("[final] evaluate_sequence failed:", e)

    return best_seq.tolist()

non‑smooth `max`.  The main novelties over the reference implementation are

* **Momentum / Adam‑like adaptation** – a velocity term (`v`) accumulated from
  past gradients accelerates convergence.
* **Mirror‑descent on the simplex** – the parameters are stored in log‑space

* **Enforced symmetry** – after every few iterations the vector is symmetrised

The optimiser works on the probability simplex using mirror‑descent
(exponentiated‑gradient) with momentum.  A soft‑max approximation of the
non‑smooth max is annealed (parameter β) during the run.  Symmetry of the
candidate is enforced periodically, and a simple stagnation‑restart heuristic
keeps the search from getting stuck.  The implementation is self‑contained,
uses only NumPy and the provided ``evaluate_sequence`` routine and finishes
well within the 1000 s budget.
"""

import time

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x ≥ 0, sum(x)=1}
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:               # all entries non‑positive
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute g_i = Σ_t w[i+t]·a[t]   (i = 0,…,n‑1)
    i.e. correlation of w with a (no reversal), via FFT.
    """
    n = a.shape[0]
    m = w.shape[0]                 # = 2n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) (assumes sum(a)=1)."""
    n = a.size
    conv = _conv_self_fft(a)
    return 2.0 * n * conv.max()
