def propose_candidate(seed=123, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    n = 4600  # default
    # maybe allow adjust to 5000 if time.
    a = np.random.rand(n)
    a = _project_to_simplex(a)

    best = a.copy()
    best_score = _objective(best) # objective without evaluate_sequence (maybe use evaluate later)
    print(f"[init] n={n}, initial objective {best_score}")

    # Preallocate solver objects across iterations? cvxpy cannot easily update constraints.
    # We'll repeatedly solve with changed A (since A depends on f). Could rebuild.

    t = 0.2 # step size for convex combination
    # loop:
    iter_count = 0
    while time.time() < deadline:
        # compute conv_f_f and M
        conv = _conv_self_fft(a)
        M = conv.max()
        # Build constraint matrix A using f = a
        # Use scipy.sparse.diags
        from scipy import sparse
        diag_vals = [np.full(n, a[i]) for i in range(n)]
        A = sparse.diags(diag_vals, offsets=range(n), shape=(2*n-1, n), format='csc')
        # Solve LP
        g_var = cp.Variable(n, nonneg=True)
        constraints = [A @ g_var <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
        # solve with ECOS
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000)
        if g_var.value is None:
            print("[LP] solver failed, break")
            break
        g = np.array(g_var.value).reshape(-1)
        # combine new candidate
        a_new = (1 - t) * a + t * g
        a_new = _project_to_simplex(a_new)
        a = a_new
        # evaluate objective
        cur_obj = _objective(a)
        if cur_obj < best_score - 1e-12:
            best_score = cur_obj
            best = a.copy()
            print(f"[iter {iter_count}] new best O={best_score:.6f}")
        iter_count += 1
        # adapt t schedule (decrease)
        if iter_count % 20 == 0:
            t = max(0.01, t * 0.95) # dec
        # random jitter occasionally
        if iter_count % 200 == 0:
            jitter = np.random.randn(n) * 0.0005
            a += jitter
            a = _project_to_simplex(a)

    # final evaluation via evaluate_sequence
    try:
        final_val = evaluate_sequence(best.tolist())
        print("[final] evaluate_sequence O=", final_val)
    except Exception as e:
        print("[final] evaluate_sequence error", e)

    return best.tolist()

        beta = beta_start * (beta_end / beta_start) ** fraction  # exponential
        eta = eta_start * (eta_end / eta_start) ** fraction   # exponential decay

        # compute softmax weights
        shifted = conv - M
        # Use stable exp
        exp_vals = np.exp(beta * shifted)
        w = exp_vals / exp_vals.sum()

        # gradient
        grad = 2.0 * _correlation_fft(w, a)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** iter_cnt)
        v_hat = v / (1 - beta2 ** iter_cnt)
        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_to_simplex(a_new)

        # small jitter
        if iter_cnt % 5000 == 0:
            jitter = np.random.randn(n) * 0.0002
            a_new += jitter
            a_new = _project_to_simplex(a_new)

        # restart if stuck
        if no_improve_counter >= 20000:
            # reset momentum
            m[:] = 0.0
            v[:] = 0.0
            a_new = best_seq.copy()
            a_new += np.random.randn(n) * 0.001
            a_new = _project_to_simplex(a_new)
            no_improve_counter = 0
            print(f"[restart] iter {iter_cnt}, re‑initialized from best.")

        a = a_new

    # final evaluation
    try:
        final = evaluate_sequence(best_seq.tolist())
        print("[final] evaluate_sequence O=", final)
    except Exception as e:
        print("[final] error", e)

    return best_seq.tolist()

"""
Propose a step‑function that minimizes the bound
    O = 2·n·max(conv(a,a))      (sum(a)=1, a_i ≥ 0)

The algorithm is a *projected Adam* optimiser on the soft‑max
approximation of the max‑convolution.  For a current candidate
`a` we

1. compute the full self‑convolution `conv = a*a` (FFT, O(n log n));
2. build a soft‑max weight vector  
   w_k ∝ exp(β·(conv_k − max(conv)))   – β grows during the run;
3. obtain the gradient of the smooth max by correlating `w`
   with `a` (again via FFT);
4. apply an Adam step and project the result back onto the
   probability simplex {x≥0, Σx=1}.

The step‑size η and the smoothing β follow exponential schedules.
Every few thousand iterations a tiny random jitter is added and,
if no improvement is seen for a long stretch, the search is
re‑initialised from the best solution found so far.  The routine
runs for at most the supplied time budget and finally returns the
best sequence (as a plain list) it discovered.
"""

import time
import math
import numpy as np
from scipy import sparse


# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x ≥ 0, Σx = 1}
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex by the algorithm of
    Wang & Carreira‑Perpiñán (2013)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:               # everything non‑positive → uniform vector
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation g_i = Σ_t w[i+t]·a[t]   for i = 0,…,n‑1.
    Implemented via FFT (linear convolution of w with reversed a,
    but without the reversal the formula above holds).
    """
    n = a.shape[0]
    m = w.shape[0]                 # = 2n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) with Σa = 1."""
    n = a.size
    conv = _conv_self_fft(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the bound.
    Returns the best sequence as a plain list of floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # Choose length and starting point.
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=float)
        a = a / a.sum()
        n = a.size
    else:
        n = 4600                     # a length that is known to work well
        a = np.random.rand(n)
        a = _project_to_simplex(a)

    # Keep the best solution seen so far.
    best_seq = a.copy()
    best_score = _objective(best_seq)
    print(f"[init] n={n}, objective={best_score:.6f}")

    # --------------------------------------------------------------
    # Adam state
    m = np.zeros_like(a)            # first moment
    v = np.zeros_like(a)            # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # Hyper‑parameter schedules (exponential)
    beta_start = 0.5                # start of soft‑max smoothing
    beta_end = 8000.0               # final sharpness
    eta_start = 0.2                 # initial learning‑rate
    eta_end = 0.005                 # final learning‑rate

    # Rough estimate of total iterations (will be cut by the time budget)
    max_iters_est = 300_000

    iter_cnt = 0
    no_improve = 0

    while time.time() < deadline and iter_cnt < 500_000:
        iter_cnt += 1

        # ----------------------------------------------------------
        # 1) current objective and best‑so‑far update
        conv = _conv_self_fft(a)
        M = conv.max()
        cur_score = 2.0 * n * M
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            no_improve = 0
            print(f"[iter {iter_cnt:06d}] new best O={best_score:.6f}")
        else:
            no_improve += 1

        # ----------------------------------------------------------
        # 2) schedules for β (soft‑max) and η (step size)
        frac = min(iter_cnt / max_iters_est, 1.0)
        beta = beta_start * (beta_end / beta_start) ** frac   # exponential growth
        eta = eta_start * (eta_end / eta_start) ** frac      # exponential decay

        # ----------------------------------------------------------
        # 3) soft‑max weights over the convolution entries
        shifted = conv - M                                 # ≤ 0, max is at zero
        # use shift to avoid overflow; all numbers ≤ 0
        exp_vals = np.exp(beta * shifted)
        w = exp_vals / exp_vals.sum()                     # length 2n‑1, sums to 1

        # ----------------------------------------------------------
        # 4) gradient of the smooth max (2· correlation)
        g = 2.0 * _correlation_fft(w, a)                  # shape (n,)

        # ----------------------------------------------------------
        # 5) Adam update (projected)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)

        m_hat = m / (1.0 - beta1 ** iter_cnt)
        v_hat = v / (1.0 - beta2 ** iter_cnt)

        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_to_simplex(a_new)

        # ----------------------------------------------------------
        # 6) occasional jitter to escape plateaus
        if iter_cnt % 5000 == 0:
            jitter = np.random.randn(n) * 0.0002
            a_new += jitter
            a_new = _project_to_simplex(a_new)

        # ----------------------------------------------------------
        # 7) restart if we have been stuck for a while
        if no_improve >= 20000:
            # Reset Adam moments and re‑initialize from the current best
            m.fill(0.0)
            v.fill(0.0)
            a_new = best_seq.copy()
            a_new += np.random.randn(n) * 0.001
            a_new = _project_to_simplex(a_new)
            no_improve = 0
            print(f"[restart] iter {iter_cnt}, re‑initialised from best.")

        a = a_new

    # ------------------------------------------------------------------
    # Final evaluation with the official routine (for reporting only)
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_val:.6f}")
    except Exception as exc:                                 # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    # Return the best sequence found (as plain Python list)
    return best_seq.tolist()
