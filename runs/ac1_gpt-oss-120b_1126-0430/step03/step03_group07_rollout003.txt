# Fast evaluation (identical to the provided evaluate_sequence but

def _fast_evaluation(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    n = seq.size
    if n == 0:
        return float("inf")
    # FFT length: next power of two ≥ 2·n‑1
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))

def _beta_sequence(alpha: float, beta: float, length: int) -> np.ndarray:
    """
    Sample the Beta(α,β) density on a regular grid (mid‑point rule) and
    normalise to sum‑one.  Works for α,β > 0 (including values < 1 which
    give U‑shaped distributions).
    """
    xs = (np.arange(length, dtype=float) + 0.5) / length
    # unnormalised density
    a = np.power(xs, alpha - 1.0, out=np.empty_like(xs))
    b = np.power(1.0 - xs, beta - 1.0, out=np.empty_like(xs))
    seq = a * b
    seq[np.isinf(seq) | np.isnan(seq)] = 0.0
    seq = np.maximum(seq, 0.0)
    if seq.sum() == 0.0:
        seq = np.full(length, 1.0 / length)
    else:
        seq /= seq.sum()
    return seq


def _sin_power_sequence(p: float, length: int) -> np.ndarray:
    """
    a_i ∝ sin(π·(i+0.5)/length)^p  (p may be negative, zero or positive).
    """
    xs = (np.arange(length, dtype=float) + 0.5) / length
    vals = np.sin(np.pi * xs) ** p
    vals = np.maximum(vals, 0.0)
    if vals.sum() == 0.0:
        vals = np.full(length, 1.0 / length)
    else:
        vals /= vals.sum()
    return vals


def _random_dirichlet(length: int, concentration: float = 0.5) -> np.ndarray:

    return seq

def _smooth_adam_refine(seq: np.ndarray, deadline: float) -> np.ndarray:

    Refine a probability vector ``seq`` (non‑negative, sum = 1) by
    minimising a temperature‑controlled soft‑max of the self‑convolution.
    The optimisation runs until ``time.time()`` reaches ``deadline`` and
    returns the best vector seen.
    """

    eps = 1e-15
    # initialise log‑variables (guarantees positivity)
    log_seq = np.log(seq + eps)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999

    v = np.zeros_like(seq)

    best_seq = None
    best_score = float("inf")
    it = 0

    while time.time() < deadline:
        # ----- obtain current probability vector -----
        a = np.exp(log_seq)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        # ----- self‑convolution (FFT) -----
        fa = np.fft.rfft(a, conv_fft_len)
        conv = np.fft.irfft(fa * fa, conv_fft_len)[:conv_len]
        max_conv = conv.max()

        # start with a relatively warm temperature and cool it gradually
        T_factor = 0.5                       # initial factor of max_conv

        best_score = final_score

    # ------------------------------------------------------------------
    # 5.  Return the best candidate as a plain Python list.
    # ------------------------------------------------------------------
    print(f"[summary] final metric {best_score:.6f} with length {best_seq.size}")
    # Clip to the limits enforced by the original evaluate_sequence
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return best_seq.tolist()
