"""
Search for a non‑negative step‑function that minimizes

    O(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The bound is scale‑invariant, so the algorithm works on the probability
simplex {a_i≥0, Σa_i=1}.  The core optimizer is an Adam‑style projected
gradient method applied to a smooth approximation of the max‑convolution.
The smooth max is obtained with a soft‑max (log‑sum‑exp) that is annealed
from a low to a high inverse temperature β.  Several diverse initial
vectors (random, triangular, sine, exponential and the previous best if
available) are tried, each for a short run.  The best result is then
refined with a long run.  The routine respects a 1000 s time budget,
prints progress, and finally returns the best sequence as a plain list
of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x ≥ 0, Σx = 1}
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        # all entries non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a * a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute the correlation g_i = Σ_t w[i+t]·a[t] for i = 0,…,n‑1
    (a linear correlation without reversing a).  Implemented via FFT.
    """
    n = a.shape[0]
    m = w.shape[0]                     # = 2·n‑1
    L = _next_pow2(m + n - 1)          # length for linear convolution
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) (a is assumed to sum to 1)."""
    n = a.size
    conv = _conv_self_fft(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    max_iters: int,
    deadline: float,
    best_obj_initial: float | None = None,
) -> tuple[np.ndarray, float]:
    """
    Adam‑style projected gradient descent on the smooth max‑convolution.
    Returns the best vector encountered and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # schedules
    beta_start, beta_end = 2.0, 4000.0          # temperature for soft‑max
    lr_start, lr_end = 0.15, 0.001             # learning‑rate decay

    best_seq = a.copy()
    best_obj = _objective(a) if best_obj_initial is None else best_obj_initial

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # --------------------------------------------------------------
        # convolution and objective
        conv = _conv_self_fft(a)
        M = conv.max()
        cur_obj = 2.0 * n * M
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_seq = a.copy()
            print(f"[adam] iter {t:06d}  new best O={best_obj:.6f}")

        # --------------------------------------------------------------
        # annealing / learning‑rate schedules (linear in iteration fraction)
        fraction = t / max_iters
        beta = beta_start + (beta_end - beta_start) * fraction
        lr = lr_start + (lr_end - lr_start) * fraction

        # --------------------------------------------------------------
        # smooth max weights (soft‑max)
        shifted = conv - M                       # ≤ 0, max entry becomes 0
        exp_vals = np.exp(beta * shifted)       # under‑flow to 0 for far‑away entries
        w = exp_vals / exp_vals.sum()           # length 2·n‑1, sums to 1

        # --------------------------------------------------------------
        # gradient of the smoothed max (chain rule)
        g = 2.0 * _correlation_fft(w, a)        # shape (n,)

        # --------------------------------------------------------------
        # Adam update (gradient descent direction)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = _project_to_simplex(a)

        # --------------------------------------------------------------
        # occasional jitter to shake out of plateaus
        if t % 5000 == 0:
            a += np.random.randn(n) * 0.001
            a = _project_to_simplex(a)

    return best_seq, best_obj

def propose_candidate(budget_s: int = 1000) -> list[float]:
    """
    Main entry point.  Returns a list of non‑negative heights that (hopefully)
    yields a bound lower than the previous best 1.522263.
    The routine respects the supplied time budget.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.3    # small safety margin

    # ------------------------------------------------------------------
    # Choose a problem size.  The previous best used ~4600 entries;
    # this size balances expressive power and iteration speed.
    n = 4600
    if 'height_sequence_1' in globals():
        a_initial = np.asarray(height_sequence_1, dtype=float)
        a_initial = a_initial / a_initial.sum()
        n = a_initial.size
    else:
        a_initial = None

    # ------------------------------------------------------------------
    # Assemble a collection of diverse starting vectors.
    init_vectors = []

    # 1) the previous construction if available
    if a_initial is not None:
        init_vectors.append(a_initial.copy())

    # 2) random Dirichlet‑style start
    init_vectors.append(_project_to_simplex(np.random.rand(n)))

    # 3) symmetric triangular shape
    tri = np.minimum(np.arange(1, n + 1), np.arange(n, 0, -1)).astype(float)
    init_vectors.append(_project_to_simplex(tri))

    # 4) half‑sine shape (positive and smooth)
    sine = np.sin(np.pi * (np.arange(1, n + 1) / (n + 1))).astype(float)
    init_vectors.append(_project_to_simplex(sine))

    # 5) exponential decay
    exp_decay = np.exp(-np.arange(n) / (n / 10.0)).astype(float)
    init_vectors.append(_project_to_simplex(exp_decay))

    # ------------------------------------------------------------------
    # Phase 1 – short explorations from each seed
    best_seq = None
    best_obj = float('inf')
    short_iters = 20000

    for idx, vec in enumerate(init_vectors):
        if time.time() > deadline:
            break
        print(f"[phase1] start seed {idx}, n={n}")
        cand_seq, cand_obj = _adam_optimize(vec, short_iters, deadline, None)
        if cand_obj < best_obj:
            best_obj = cand_obj
            best_seq = cand_seq.copy()
            print(f"[phase1] new global best O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – long refinement from the best candidate so far
    if best_seq is None:
        # fallback: generate a fresh random start
        best_seq = _project_to_simplex(np.random.rand(n))
        best_obj = _objective(best_seq)

    remaining_iters = 200000
    print("[phase2] long refinement started")
    best_seq, best_obj = _adam_optimize(best_seq, remaining_iters, deadline, best_obj)

    # ------------------------------------------------------------------
    # Final evaluation with the provided security‑checked routine
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:                         # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_seq.tolist()
