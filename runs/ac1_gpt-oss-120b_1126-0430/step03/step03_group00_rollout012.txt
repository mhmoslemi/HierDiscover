"""
Propose a non‑negative step‑function sequence that (approximately) minimises

    O(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The routine uses two phases:

1️⃣  **Smoothed log‑sum‑exp optimisation (Adam)**  
    We minimise the surrogate  
        fα(a) = (2·n/α)·log∑k exp(α·convk(a)) + λ·∑i a_i·log a_i   (entropy term)  
    The gradient of the soft‑max part is 2·corr(w,a) with
        w_k ∝ exp(α·convk).  
    Adam with bias‑correction updates the parameters, after each step we
    project onto the probability simplex (∑a_i=1, a_i≥0).  The smoothing
    parameter α is annealed from a small value up to a large one (default 5000)
    so that the optimisation gradually focuses on the true maximum.
    The entropy coefficient λ (default 0.02) favours a spread‑out shape
    and helps to escape flat plateaus.

2️⃣  **Stochastic local refinement**  
    Starting from the best candidate of phase 1 we perform many cheap
    “mass‑transfer’’ moves (move a random fraction of probability mass from one
    index to another).  Moves are accepted if they improve the exact objective,
    otherwise with a Metropolis probability that slowly cools down.
    Every few iterations we take a projected sub‑gradient step for the exact
    max‑convolution entry.  A few restarts (jittered copies of the current best
    or fresh random points) improve exploration.

Both phases respect the overall 1000 s budget.  The function returns the
best sequence found as a plain Python list of floats.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------
# Simplex projection (Euclidean)
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
# Full linear convolution via FFT (length 2·n‑1)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return the full a⨂a convolution (length 2·n‑1) using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# --------------------------------------------------------------
# Objective for a vector that is already normalised (∑a=1)
def _objective_simplex(a: np.ndarray, n: int, L: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for sum(a)=1."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# --------------------------------------------------------------
# Sub‑gradient of max(conv) (used in phase 2)
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. a (averaged over all maximisers)."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)          # indices achieving the max
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


# --------------------------------------------------------------
# Gradient of the smooth surrogate (log‑sum‑exp) plus entropy term
def _smooth_gradient_plus_entropy(
    a: np.ndarray,
    conv: np.ndarray,
    alpha: float,
    L: int,
    lam: float,
) -> np.ndarray:
    """
    Returns ∇_a ( (2·n/α)·log∑_k e^{α·conv_k} + λ·∑ a_i·log a_i ).
    The first part reduces to 2·corr(w,a) with w_k ∝ e^{α·conv_k};
    the second part contributes λ·(log a_i + 1).
    """
    # ---- soft‑max weights ----------
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)          # stability
    w = exp_scaled / exp_scaled.sum()                 # shape (2n‑1,)

    # ---- correlation w ⋆ a (g_i = 2·∑_j w_{i+j} a_j) ----
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n_len = a.size
    grad_corr = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]

    # ---- entropy part ----------
    eps = 1e-20
    grad_ent = lam * (np.log(a + eps) + 1.0)

    return grad_corr + grad_ent


# --------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that minimises the
    provided evaluation function.  Returns the best sequence found (plain list).
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin so we never exceed the budget
    deadline = start_time + budget_s - 0.3

    # --------------------------------------------------------------
    # Problem size – we use a length that is a power of two plus a small margin;
    # a larger n gives more freedom without blowing up the FFT cost.
    if "height_sequence_1" in globals():
        base = np.array(height_sequence_1, dtype=float)
        n = base.size
        a = base / base.sum()
        print(f"[init] start from supplied sequence, n={n}")
    else:
        n = 5000                      # a little larger than the previous 4600
        a = np.full(n, 1.0 / n)       # uniform start
        print(f"[init] start from uniform, n={n}")

    # FFT length – next power of two larger than the longest vector we need
    L = 1 << ((3 * n - 2).bit_length())

    # --------------------------------------------------------------
    # Phase‑1 : Adam on the smooth surrogate (log‑sum‑exp + entropy)
    best_seq = a.copy()
    best_score = _objective_simplex(best_seq, n, L)
    print(f"[phase‑1] start O={best_score:.6f}")

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    m = np.zeros_like(a)       # first moment estimate
    v = np.zeros_like(a)       # second moment estimate
    lr0 = 0.6                   # base learning rate
    alpha_start = 5.0
    alpha_end = 8000.0          # larger final temperature
    lam = 0.02                  # entropy weight (tiny, but helpful)

    max_iter_phase1 = 400_000   # generous upper bound; we exit on time
    it = 0
    phase1_deadline = start_time + budget_s * 0.55   # approx 55 % of budget

    while time.time() < phase1_deadline and it < max_iter_phase1:
        # ----- exact objective (for bookkeeping) -----
        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[phase‑1] iter {it:,} new best O={best_score:.6f}")

        # ----- anneal smoothing parameter α -----
        frac = it / max_iter_phase1
        alpha = alpha_start + (alpha_end - alpha_start) * frac

        # ----- smooth gradient (soft‑max + entropy) -----
        grad = _smooth_gradient_plus_entropy(a, conv, alpha, L, lam)

        # ----- Adam update (bias‑corrected) -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        lr = lr0 / math.sqrt(it + 1)          # simple 1/√t decay

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional tiny jitter to escape plateaus
        if np.random.rand() < 0.001:
            a_new += np.random.randn(n) * 1e-4

        # project onto the simplex (guarantees feasibility)
        a_new = _project_to_simplex(a_new)

        # prune extremely tiny entries every few thousand iterations
        if it % 5000 == 0 and it > 0:
            thresh = np.percentile(a_new, 0.5)   # drop the lowest half‑percent
            a_new[a_new < thresh] = 0.0
            a_new = _project_to_simplex(a_new)

        a = a_new
        it += 1

    # --------------------------------------------------------------
    # Phase‑2 : Stochastic local refinement (mass‑transfer + sub‑grad)
    print(f"[phase‑2] entering local refinement, best O={best_score:.6f}")

    max_restarts = 0
    subgrad_interval = 5        # sub‑gradient step every 5 iterations
    temperature0 = 0.04
    max_iter_local = 150_000
    no_improve_limit = 3_000

    while time.time() < deadline:
        # ----- initialise a restart (jittered best or fresh) -----
        if np.random.rand() < 0.6:
            a = best_seq.copy()
            a += np.random.randn(n) * 0.0015   # small Gaussian jitter
            a = np.clip(a, 0, None)
            a = _project_to_simplex(a)
        else:
            a = np.random.rand(n)
            a = _project_to_simplex(a)

        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        it_loc = 0
        no_improve = 0
        T = temperature0

        while time.time() < deadline and it_loc < max_iter_local:
            # ------------- random mass‑transfer move -------------
            i, j = np.random.randint(n, size=2)
            if i != j and a[i] > 1e-14:
                delta = a[i] * np.random.rand() * 0.15   # up to 15 % of source mass
                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta
                a_new = _project_to_simplex(a_new)

                conv_new = _conv_fft(a_new, L)
                new_obj = 2.0 * n * conv_new.max()

                # Metropolis acceptance
                if new_obj < cur_obj or np.random.rand() < math.exp((cur_obj - new_obj) / max(T, 1e-14)):
                    a, conv, cur_obj = a_new, conv_new, new_obj
                    no_improve = 0
                    if cur_obj < best_score:
                        best_score = cur_obj
                        best_seq = a.copy()
                        print(f"[{time.time():.1f}] new global best O={best_score:.6f} (restart={max_restarts}, iter={it_loc})")
                else:
                    no_improve += 1
            else:
                no_improve += 1

            # ------------- sub‑gradient step on the current max entry -------------
            if it_loc % subgrad_interval == 0:
                g = _subgradient_max_conv(a, conv)
                # step size slowly decays to zero
                eta = 0.07 * (1.0 - it_loc / max_iter_local)
                a = a - eta * g
                a = _project_to_simplex(a)
                conv = _conv_fft(a, L)
                cur_obj = 2.0 * n * conv.max()
                if cur_obj < best_score:
                    best_score = cur_obj
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] sub‑grad improved O={best_score:.6f}")

            # ------------- (optional) tiny smoothing move -------------
            if it_loc % 4000 == 0 and it_loc > 0:
                # blend a little bit with a 5‑point moving average
                kernel = np.array([0.125, 0.25, 0.25, 0.25, 0.125])
                a_smooth = np.convolve(a, kernel, mode="same")
                a = _project_to_simplex(0.85 * a + 0.15 * a_smooth)

            # temperature decay
            T *= 0.9998
            it_loc += 1

            # restart if stuck for a while
            if no_improve >= no_improve_limit:
                break

        max_restarts += 1
        if time.time() >= deadline:
            break

    # --------------------------------------------------------------
    # Final sanity check using the provided evaluator
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:
        print("[final] evaluate_sequence failed:", e)

    # Return the best sequence discovered (as a plain Python list)
    return best_seq.tolist()
