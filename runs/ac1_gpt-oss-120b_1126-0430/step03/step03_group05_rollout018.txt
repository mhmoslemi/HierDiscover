def _conv_max_fft(a, fft_len):
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa).real[:2*a.size - 1]

"""
Propose a step function sequence minimizing evaluate_sequence.

Algorithm:
1. Choose a sequence length (default 15000 or inherited from previous run).
2. Initialise the vector using the *discrete arcsine* distribution
   a_i ∝ 1/√((i+α)*(n‑α‑i)), which is the known extremiser of the
   max‑convolution problem (its self‑convolution is asymptotically flat).
   Several α values are tried and the best cheap score (2·n·max(conv))
   is kept.
3. Refine the candidate with an Adam‑style projected stochastic
   gradient descent.  The gradient is the sub‑gradient of the
   active convolution entry (g_i = 2·a_{k*‑i}).  After each step the
   iterate is projected onto the probability simplex (Σa=1, a≥0).
4. Periodically store the best iterate seen (according to the cheap
   score) and, after a long stagnation, restart from the best vector
   with a small random perturbation.
5. When the time budget expires evaluate the best vector with the
   provided ``evaluate_sequence`` and return it.
"""

import time
import math
import numpy as np

def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (sum = 1, non‑negative)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # all entries non‑positive – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_max_fft(a: np.ndarray, fft_len: int) -> tuple[float, int]:
    """Return (maximum convolution value, index) of a*a (linear convolution)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa).real[: 2 * a.size - 1]
    max_idx = int(conv.argmax())
    return float(conv[max_idx]), max_idx


def _subgradient_max_convolution(a: np.ndarray, max_idx: int) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)[k] at the active index ``max_idx``.
    For the active shift k we have (∂/∂a_i) (a*a)[k] = 2·a_{k‑i}
    wherever the index is valid.
    """
    n = a.shape[0]
    lo = max(0, max_idx - (n - 1))
    hi = min(max_idx, n - 1)
    idx = np.arange(lo, hi + 1, dtype=np.int64)
    g = np.zeros_like(a)
    g[idx] = 2.0 * a[max_idx - idx]
    return g


def _arcsine_sequence(n: int, alpha: float = 0.5) -> np.ndarray:
    """
    Discrete approximation of the arcsine distribution on {0,…,n‑1}.
    a_i ∝ 1 / sqrt( (i+α)·(n‑α‑i) ), then normalised to sum to 1.
    """
    i = np.arange(n, dtype=np.float64)
    # shift by alpha to avoid division by zero at the boundaries
    denom = (i + alpha) * (n - alpha - i)
    # Enforce positivity (the expression is >0 for 0<alpha<1)
    a = 1.0 / np.sqrt(denom)
    a_sum = a.sum()
    if a_sum == 0.0:
        # fall back to uniform (should never happen)
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a /= a_sum
    return a

            m.fill(0.0)
            v.fill(0.0)
            last_improve = iter_count
            print(f"[Iter {iter_count:07d}] Restarted after stagnation (perturb).")

    # ------------------------------------------------------------------

    except Exception as exc:
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # ------------------------------------------------------------------
    # 4. Return.
    # ------------------------------------------------------------------
    return final_seq

    # ----- previous construction (if any) --------------------------------
    if prev_arr is not None and prev_arr.size == n:
        maxb, _ = _conv_max_fft(prev_arr, fft_len)
        cheap = 2.0 * n * maxb
        if cheap < best_score:
            best_score = cheap
            best_a = prev_arr.copy()
            best_desc = "previous"
            print(f"[Init] previous cheap = {cheap:.6f}")

    # ----- final safety fallback -----------------------------------------
    if best_a is None:
        best_a = _arcsine_sequence(n, 0.5)
        maxb, _ = _conv_max_fft(best_a, fft_len)
        best_score = 2.0 * n * maxb
        best_desc = "fallback arcsine"
        print(f"[Init] fallback cheap = {best_score:.6f}")

    print(f"Initialisation finished ({best_desc}): cheap = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Refine the best start with projected Adam.
    # ------------------------------------------------------------------
    a = best_a.copy()                # always Σa = 1
    m = np.zeros_like(a)             # first‑moment accumulator
    v = np.zeros_like(a)             # second‑moment accumulator
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.05                        # base learning‑rate (scaled as 1/√t)

    best_iter_a = a.copy()
    best_iter_score = best_score
    last_improve = 0
    stagnation_limit = 150_000        # iterations without improvement → perturb
    perturb_scale = 0.02

    iter_count = 0
    while time.time() < deadline and iter_count < 2_000_000:
        iter_count += 1

        # ----- convolution and active index -----
        maxb, max_idx = _conv_max_fft(a, fft_len)

        # ----- sub‑gradient at the active shift -----
        g = _subgradient_max_convolution(a, max_idx)

        # ----- Adam update (projected onto simplex) -----
        t = iter_count
        lr = lr0 / math.sqrt(t)

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        # bias‑correction (only while the power terms are not numerically zero)
        if t < 1000:
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
        else:
            m_hat = m
            v_hat = v

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_candidate = a - step
        a = _simplex_projection(a_candidate)

        # ----- bookkeeping – cheap score every few iterations -----
        if (iter_count % 400 == 0) or (time.time() > deadline - 4):
            cheap = 2.0 * n * maxb               # exact because Σa = 1
            if cheap < best_iter_score:
                best_iter_score = cheap
                best_iter_a = a.copy()
                last_improve = iter_count
                print(f"[Iter {iter_count:07d}] NEW BEST cheap = {cheap:.6f}")

        # ----- restart after stagnation -----
        if iter_count - last_improve > stagnation_limit:
            perturb = perturb_scale * np.random.rand(n)
            a = _simplex_projection(best_iter_a + perturb)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = iter_count
            print(f"[Iter {iter_count:07d}] Restarted (perturb).")

    # ------------------------------------------------------------------
    # 3. Final exact evaluation (uses the provided ``evaluate_sequence``)
    # ------------------------------------------------------------------
    final_seq = best_iter_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:      # defensive – should never happen
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # ------------------------------------------------------------------
    # 4. Return the best sequence found.
    # ------------------------------------------------------------------
    return final_seq
```
