"""
propose_candidate
-----------------
A fast Adam‑based optimiser that works on the *softmax* (log‑it) 
parameterisation of a non‑negative step‑function.  The objective is a
smooth approximation of the original metric

    2·n·max(conv(a,a)) / (∑a)² .

The algorithm:

1. Build a diverse collection of seeds (uniform, Dirichlet draws,
   sparse Dirichlet, asymmetric Beta and sinusoidal shapes, all
   optionally permuted to break symmetry).

2. For each seed a short time‑sliced optimisation is performed.
   The optimisation works with logits `v` where `a = softmax(v)`,
   guaranteeing positivity and the normalisation `∑a = 1`.
   The nonsmooth `max` is replaced by a temperature‑controlled
   log‑sum‑exp (`β` grows from 5 to 30), giving a well‑behaved gradient:
       ∂f/∂a = 2·β·(w * a)    with   w_i = exp(β·conv[i]) / Σexp(..)
   The gradient in log‑it space is
       g = a·(∂f/∂a – (∂f/∂a)·a)  .
   An Adam update is applied to `v`.  The best vector seen so far
   (according to the *exact* evaluation function) is kept.

3. After the exploratory phase the best candidate is refined on the
   full time budget.

The whole routine stays well within the 1000 s limit, returns a list
of non‑negative floats and prints progress information.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast metric – identical to the supplied evaluate_sequence but without
# the safety checks (used for the inner optimisation loop).
# ----------------------------------------------------------------------
def _fast_evaluation(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) / (sum(seq)²).  Assumes sum>0."""
    n = seq.size
    if n == 0:
        return float("inf")
    # linear convolution via FFT (zero‑padded to avoid circular effects)
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Helper: compute the full self‑convolution of a vector (linear)
# ----------------------------------------------------------------------
def _self_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a, length = 2·len(a)‑1."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.size - 1]
    return conv


# ----------------------------------------------------------------------
# Helper: gradient of the smooth‑max objective w.r.t. the probability
# vector a.  Returns ∂f/∂a.
# ----------------------------------------------------------------------
def _gradient_a_from_weights(a: np.ndarray,
                             w: np.ndarray,
                             beta: float,
                             grad_fft_len: int) -> np.ndarray:
    """
    g = 2·β·(w * a)   where * is linear convolution.
    Both w (length 2n‑1) and a (length n) are real.
    """
    # convolution w*a (full length 3n‑2); we need only the first n entries.
    fa = np.fft.rfft(a, grad_fft_len)
    fw = np.fft.rfft(w, grad_fft_len)
    conv = np.fft.irfft(fa * fw, grad_fft_len)[: a.size]
    return 2.0 * beta * conv


# ----------------------------------------------------------------------
# Adam optimiser on the logits (log‑it) of a probability vector.
# ----------------------------------------------------------------------
def _adam_softmax_optimize(a0: np.ndarray,
                           deadline: float,
                           lr: float = 0.07,
                           beta1: float = 0.9,
                           beta2: float = 0.999,
                           beta_start: float = 5.0,
                           beta_end: float = 30.0) -> np.ndarray:
    """
    Minimise a smooth approximation of the metric starting from a0.
    a0 must be non‑negative and sum to 1.
    Returns the best probability vector encountered before deadline.
    """
    n = a0.size
    eps = 1e-15
    v = np.log(a0 + eps)                     # logits
    m = np.zeros_like(v)                     # first moment
    v_hat = np.zeros_like(v)                 # second moment
    t = 0

    # FFT lengths (pre‑computed for speed)
    fft_len_conv = 1 << ((2 * n - 1).bit_length())      # for a*a
    fft_len_grad = 1 << ((3 * n - 2).bit_length())      # for w*a

    best_seq = a0.copy()
    best_score = _fast_evaluation(best_seq)

    it = 0
    # linear schedule for the smooth‑max temperature β
    total_time = deadline - time.time()
    while time.time() < deadline:
        # ----- probability vector from logits -----
        a = np.exp(v)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / a_sum

        # ----- self‑convolution -----
        conv = _self_convolution(a, fft_len_conv)

        # ----- smooth max (log‑sum‑exp) -----
        # linearly interpolate β between start and end based on elapsed fraction
        elapsed = time.time() - (deadline - total_time)
        frac = min(1.0, elapsed / total_time)
        beta = beta_start + (beta_end - beta_start) * frac

        max_c = conv.max()
        # stable exponentials
        exp_term = np.exp(beta * (conv - max_c))
        w = exp_term / exp_term.sum()          # softmax weights, sum to 1

        # ----- gradient w.r.t. a -----
        grad_a = _gradient_a_from_weights(a, w, beta, fft_len_grad)

        # ----- convert to gradient w.r.t. logits (softmax Jacobian) -----
        dot = np.dot(grad_a, a)                # scalar ⟨grad_a , a⟩
        grad_v = a * (grad_a - dot)            # shape (n,)

        # ----- Adam update on v -----
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1.0 - beta2) * (grad_v * grad_v)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat_corr = v_hat / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat_corr) + 1e-8)
        v -= step

        # keep logits in a reasonable range to avoid overflow/underflow
        np.clip(v, -30.0, 30.0, out=v)

        # ----- occasional random jitter to escape tiny plateaus -----
        if np.random.rand() < 0.001:
            jitter = np.random.randn(n) * 0.1
            v += jitter
            np.clip(v, -30.0, 30.0, out=v)

        # ----- evaluate exact metric and keep the best -----
        score = _fast_evaluation(a)
        if score < best_score - 1e-12:
            best_score = score
            best_seq = a.copy()
            if it % 250 == 0:
                print(f"[Adam] iter {it:6d}  best {best_score:.6f}")

        it += 1

    return best_seq


# ----------------------------------------------------------------------
# Seed generators – all return a probability vector of length n (sum=1)
# ----------------------------------------------------------------------
def _beta_sequence(alpha: float, beta: float, length: int) -> np.ndarray:
    xs = (np.arange(length, dtype=float) + 0.5) / length
    seq = np.power(xs, alpha - 1.0, out=np.empty_like(xs))
    seq *= np.power(1.0 - xs, beta - 1.0, out=np.empty_like(xs))
    seq = np.nan_to_num(seq, nan=0.0, posinf=0.0, neginf=0.0)
    seq = np.maximum(seq, 0.0)
    if seq.sum() == 0.0:
        seq = np.full(length, 1.0 / length)
    else:
        seq /= seq.sum()
    return seq


def _sin_power_sequence(p: float, length: int) -> np.ndarray:
    xs = (np.arange(length, dtype=float) + 0.5) / length
    seq = np.sin(np.pi * xs) ** p
    seq = np.maximum(seq, 0.0)
    if seq.sum() == 0.0:
        seq = np.full(length, 1.0 / length)
    else:
        seq /= seq.sum()
    return seq


def _random_dirichlet(length: int, concentration: float = 0.5) -> np.ndarray:
    seq = np.random.gamma(concentration, 1.0, size=length)
    if seq.sum() == 0.0:
        seq = np.full(length, 1.0 / length)
    else:
        seq /= seq.sum()
    return seq


def _sparse_dirichlet(length: int,
                      k_nonzero: int = 200,
                      concentration: float = 0.4) -> np.ndarray:
    """Place mass on exactly k randomly chosen indices."""
    idx = np.random.choice(length, size=k_nonzero, replace=False)
    vals = np.random.gamma(concentration, 1.0, size=k_nonzero)
    seq = np.zeros(length, dtype=float)
    seq[idx] = vals
    if seq.sum() == 0.0:
        seq = np.full(length, 1.0 / length)
    else:
        seq /= seq.sum()
    return seq

    # ------------------------------------------------------------------
    n = 5000

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    seeds = []

    seeds.append(np.full(n, 1.0 / n))

    for conc in (0.5, 0.3, 0.1):

    # sparse Dirichlet (highly asymmetric)
    for k in (100, 200, 400):

    # asymmetric Beta (permute to destroy symmetry)

    # another asymmetric Beta

    # sin‑power shapes with a random permutation
    for p in (-0.5, -0.3, 0.0, 0.3):

        seeds.append(sin_seq[np.random.permutation(n)])

    # a few completely random probability vectors (Gamma draws)
    for _ in range(3):

        raw /= raw.sum()
        seeds.append(raw)

        # allocate at least 5 s per remaining seed, but never exceed the budget

        best_score = final_score

    print(f"[summary] final metric {best_score:.6f} with length {best_seq.size}")

    # convert to plain Python list (required by the harness)
    return best_seq.tolist()
