"""
Propose a non‑negative step‑function (probability vector) that minimises

        O(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

The denominator is handled by keeping the vector on the probability simplex
(sum = 1) throughout the optimisation.  The algorithm consists of three
stages:

1. **Smooth‑surrogate Adam** – minimise a log‑sum‑exp approximation of the
   max‑convolution using Adam and Euclidean projection onto the simplex.
   The smoothing parameter α is annealed from 5 to 5000.

2. **Deterministic sub‑gradient descent** – repeatedly take projected
   sub‑gradient steps for the true max‑convolution (the exact non‑smooth
   objective).  The step size decays linearly.

3. **Stochastic local search** – mass‑transfer moves together with a simple
   simulated‑annealing acceptance rule.  Occasionally a “gradient‑guided”
   transfer (from the highest‑gradient index to the lowest‑gradient index)
   is performed.  Restarts, jitter and occasional support‑pruning keep the
   search diverse while staying well inside the 1000 s budget.

The best vector found is returned as a plain Python list.  All intermediate
vectors are kept on the simplex, therefore the final call to
`evaluate_sequence` is guaranteed to be valid.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Projection onto the probability simplex {x >= 0, sum(x)=1}
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:                     # all non‑positive
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full convolution a*a using FFT (returns length 2·n‑1 array)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return a * a (full convolution) using FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Exact objective for a simplex‑normalised vector (sum = 1)
def _objective_simplex(a: np.ndarray, n: int, L: int) -> float:
    """2·n·max(conv(a,a))  (assumes sum(a)=1)."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Sub‑gradient for the max‑convolution entry (exact, not smoothed)
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. a (vector of same length as a)."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)          # all maximisers
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size > 0:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
# Gradient of the log‑sum‑exp surrogate (smooth approximation of max)
def _smooth_gradient(a: np.ndarray, conv: np.ndarray, alpha: float, L: int) -> np.ndarray:
    """
    ∇_a ( (2·n/α)·log Σ_k exp(α·conv_k) )
    = 2·corr(w, a) with w_k ∝ exp(α·conv_k).
    The correlation is evaluated via FFT for speed.
    """
    # soft‑max weights
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)      # stability
    w = exp_scaled / exp_scaled.sum()

    # correlation w ⋆ a   (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n_len = a.size
    grad = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]
    return grad


# ----------------------------------------------------------------------
# Main entry point ----------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that minimises the given evaluation function.
    Returns the best sequence as a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin so we do not overrun the limit
    deadline = start_time + budget_s - 0.3

    # ------------------------------------------------------------------
    # Choose problem size and initial point
    if 'height_sequence_1' in globals():
        base = np.array(height_sequence_1, dtype=float)
        n = base.size
        a = base / base.sum()                      # simplex normalisation
        print(f"[init] using supplied sequence, n={n}")
    else:
        n = 4600
        a = np.full(n, 1.0 / n)                    # uniform start
        print(f"[init] no supplied sequence – using uniform, n={n}")

    # FFT length – next power of two larger than the longest needed convolution/correlation
    L = 1 << ((3 * n - 2).bit_length())

    # ------------------------------------------------------------------
    # Phase 1 – Adam on a smoothed surrogate (≈ 55 % of budget)
    phase1_deadline = start_time + budget_s * 0.55
    best_seq = a.copy()
    best_score = _objective_simplex(best_seq, n, L)
    print(f"[phase‑1] start O={best_score:.6f}")

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    lr0 = 0.7            # a slightly larger base learning rate
    alpha_start = 5.0
    alpha_end = 5000.0
    max_iter_phase1 = 250_000

    it = 0
    while time.time() < phase1_deadline and it < max_iter_phase1:
        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[phase‑1] iter {it} new best O={best_score:.6f}")

        # linear schedule for α
        frac = it / max_iter_phase1
        alpha = alpha_start + (alpha_end - alpha_start) * frac

        # gradient of smooth surrogate
        grad = _smooth_gradient(a, conv, alpha, L)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        # learning‑rate decay ~ 1/√t
        lr = lr0 / math.sqrt(it + 1)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape plateaus
        if np.random.rand() < 0.001:
            a_new += np.random.randn(n) * 1e-4

        # projection onto simplex
        a_new = _project_to_simplex(a_new)

        # light support pruning every few thousand iterations
        if it % 4000 == 0 and it > 0:
            # zero out the lowest 1 % of entries (by value) to keep support compact
            thresh = np.percentile(a_new, 1)
            a_new[a_new < thresh] = 0.0
            a_new = _project_to_simplex(a_new)

        a = a_new
        it += 1

    # ------------------------------------------------------------------
    # Phase 2 – deterministic sub‑gradient descent (≈ 15 % of budget)
    print(f"[phase‑2] deterministic sub‑gradient start, best O={best_score:.6f}")
    subgrad_iters = 30_000
    subgrad_deadline = start_time + budget_s * 0.70
    it = 0
    while time.time() < subgrad_deadline and it < subgrad_iters:
        conv = _conv_fft(a, L)
        g = _subgradient_max_conv(a, conv)
        # step size linearly decaying from 0.12 to 0
        eta = 0.12 * (1.0 - it / subgrad_iters)
        a = a - eta * g
        a = _project_to_simplex(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[phase‑2] iter {it} new best O={best_score:.6f}")
        it += 1

    # ------------------------------------------------------------------
    # Phase 3 – stochastic hill‑climbing + guided mass transfers
    print(f"[phase‑3] stochastic refinement, best O={best_score:.6f}")

    max_restarts = 0
    subgrad_interval = 9          # how often we take a deterministic sub‑grad step
    temperature0 = 0.04
    max_iter_local = 150_000
    no_improve_limit = 6_000

    while time.time() < deadline:
        # ------- initialise a restart -------
        if np.random.rand() < 0.5:
            # jittered version of the current global best
            a = best_seq.copy()
            a += np.random.randn(n) * 0.0015
            a = np.clip(a, 0, None)
            a = _project_to_simplex(a)
        else:
            # fresh random point
            a = np.random.rand(n)
            a = _project_to_simplex(a)

        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        it_loc = 0
        no_improve = 0
        T = temperature0

        while time.time() < deadline and it_loc < max_iter_local:
            # ---- random mass‑transfer move (basic Metropolis) ----
            i, j = np.random.randint(n, size=2)
            if i != j and a[i] > 1e-14:
                delta = a[i] * np.random.rand() * 0.12    # up to 12 % of source mass
                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta
                a_new = _project_to_simplex(a_new)

                conv_new = _conv_fft(a_new, L)
                new_obj = 2.0 * n * conv_new.max()

                if new_obj < cur_obj or np.random.rand() < math.exp((cur_obj - new_obj) / max(T, 1e-12)):
                    a = a_new
                    conv = conv_new
                    cur_obj = new_obj
                    no_improve = 0
                    if cur_obj < best_score:
                        best_score = cur_obj
                        best_seq = a.copy()
                        print(f"[{time.time():.1f}] New global best O={best_score:.6f} (restart={max_restarts}, iter={it_loc})")
                else:
                    no_improve += 1
            else:
                no_improve += 1

            # ---- gradient‑guided transfer (every few iterations) ----
            if it_loc % 13 == 0:
                # compute sub‑gradient for current max entry
                g = _subgradient_max_conv(a, conv)
                # pick an index with big positive gradient (contributes a lot)
                # and one with small gradient (contributes little)
                idx_high = np.argmax(g)
                idx_low = np.argmin(g)
                if a[idx_high] > 1e-12 and idx_high != idx_low:
                    # move a modest fraction from high‑gradient to low‑gradient
                    delta = a[idx_high] * np.random.rand() * 0.08
                    a_tmp = a.copy()
                    a_tmp[idx_high] -= delta
                    a_tmp[idx_low] += delta
                    a_tmp = _project_to_simplex(a_tmp)
                    conv_tmp = _conv_fft(a_tmp, L)
                    obj_tmp = 2.0 * n * conv_tmp.max()
                    if obj_tmp < cur_obj:
                        a = a_tmp
                        conv = conv_tmp
                        cur_obj = obj_tmp
                        no_improve = 0
                        if cur_obj < best_score:
                            best_score = cur_obj
                            best_seq = a.copy()
                            print(f"[{time.time():.1f}] Guided transfer improved O={best_score:.6f}")

            # ---- deterministic sub‑gradient step (periodic) ----
            if it_loc % subgrad_interval == 0:
                g = _subgradient_max_conv(a, conv)
                eta = 0.07 * (1.0 - it_loc / max_iter_local)   # decaying step
                a = a - eta * g
                a = _project_to_simplex(a)
                conv = _conv_fft(a, L)
                cur_obj = 2.0 * n * conv.max()
                if cur_obj < best_score:
                    best_score = cur_obj
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] Sub‑grad deterministic improved O={best_score:.6f}")

            # temperature decay
            T *= 0.99995
            it_loc += 1

            # restart if stagnating
            if no_improve >= no_improve_limit:
                break

        max_restarts += 1
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Final sanity check with the provided evaluator
    try:

        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence failed:", exc)

    # Return the best found sequence as a plain list of floats
    return best_seq.tolist()
