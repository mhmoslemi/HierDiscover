def grad_softmax(a, temp):
    n = len(a)
    conv = fft_convolve(a)  # length 2n-1
    max_val = np.max(conv)
    # compute weights
    # to avoid overflow, subtract max_val
    exps = np.exp((conv - max_val) / temp)
    w = exps / np.sum(exps)
    # compute gradient: g = 2 * (reverse(a) * w) conv at offset
    a_rev = a[::-1]
    size = 1 << ((len(a_rev) + len(w) - 1).bit_length())
    # use fft conv
    FA = np.fft.rfft(a_rev, size)
    FW = np.fft.rfft(w, size)
    G_full = np.fft.irfft(FA * FW, size)
    # extract
    g = 2 * G_full[n-1: n-1 + n]
    return g, max_val  # return gradient and current max (for evaluation)
```

def _mirror_descent_optimize(a0, deadline, max_iters):
    a = a0.copy()
    log_a = np.log(a + 1e-12)
    # We'll maintain a as exp(log_a)
    best = a.copy()
    best_val = _objective(best)
    t=0
    while time.time()<deadline and t<max_iters:
        conv = _fft_convolve(a)
        max_val = conv.max()
        # gradient of max w.r.t a (subgrad)
        k = int(np.argmax(conv))
        grad = np.zeros_like(a)
        start = max(0, k - (n - 1))
        end = min(n-1, k)
        idx = np.arange(start, end+1)
        grad[idx] = 2.0 * a[k - idx]
        # incorporate step
        eta = 0.1 / np.sqrt(t + 1)
        log_a -= eta * grad
        # re-exponentiate
        a = np.exp(log_a)
        a = np.maximum(a, 1e-12)
        a = a / a.sum()
        # evaluate
        cur_val = 2.0*n*conv.max()
        if cur_val < best_val:
            best_val = cur_val
            best = a.copy()
        t+=1
    return best, best_val
```

"""
Propose a low‑value step‑function for `evaluate_sequence`.

The search consists of three stages
----------------------------------
1. **Coarse parametric sweep** –  a family  a_i ∝ (i+1)^(-α)
   (α ∈ [0.3, 0.9]) is examined on the simplex.  The best α gives
   a good starting point that already beats the plain Catalan vector.

2. **Projected sub‑gradient descent** –  starting from the best α‑vector,
   we run a heavy‑weight projected sub‑gradient optimizer (the same
   scheme as the reference code but with a larger step‑size and
   longer horizon).  Momentum (heavy‑ball) and occasional Gaussian
   nudges help to escape plateaus.

3. **Adam‑style mirror descent** –  the best point from stage 2
   is refined with an Adam optimiser that works directly on the
   simplex (additive updates + Euclidean projection).  This
   converges faster on the non‑smooth “max‑convolution’’ objective.

All updates keep the vector non‑negative and re‑normalise it to a
probability simplex (∑a_i = 1).  The value returned by
`evaluate_sequence` is invariant to a global scaling, therefore the
simplex normalisation is sufficient.

The routine stops a couple of seconds before the supplied time budget
to guarantee a result is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based full linear convolution (real input, output length 2*n‑1)
# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple families for quick grid‑search initialisation
# ----------------------------------------------------------------------
def _power_sequence(alpha: float, n: int) -> np.ndarray:
    """a_i ∝ (i+1)^(-α), normalised to sum 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-alpha)
    a /= a.sum()
    return a


def _catalan_normalised(n: int) -> np.ndarray:
    """Catalan‑type start used in the reference implementation."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Objective (exactly what `evaluate_sequence` returns for Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    return float(2.0 * n * np.max(_fft_convolve(a)))


# ----------------------------------------------------------------------
# Stage‑2: heavy‑ball projected sub‑gradient descent
# ----------------------------------------------------------------------
def _subgrad_optimize(a0: np.ndarray,
                      deadline: float,
                      max_iters: int = 300_000,
                      verbose: bool = False) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]

    best = a.copy()
    best_val = _objective(best)

    beta = 0.95               # stronger momentum than the reference code
    momentum = np.zeros_like(a)

    t = 0
    while time.time() < deadline and t < max_iters:
        conv = _fft_convolve(a)
        k = int(np.argmax(conv))

        cur_val = 2.0 * n * conv[k]
        if cur_val < best_val:
            best_val = cur_val
            best = a.copy()
            if verbose:
                print(f"[subgrad] iter {t:06d}  obj {best_val:.9f}")

        # sub‑gradient of the active quadratic form
        grad = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        grad[idx] = 2.0 * a[k - idx]

        # heavy‑ball update
        momentum = beta * momentum + (1.0 - beta) * grad

        # diminishing step‑size (1/√t schedule, a bit larger prefactor)
        lr = 0.8 / np.sqrt(t + 1)

        a = a - lr * momentum
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        # occasional random jitter to shake off plateaus
        if t % 7_000 == 0 and t > 0:
            pert = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + pert, 0.0)
            a = _project_simplex(a)

        t += 1

    return best, best_val


# ----------------------------------------------------------------------
# Stage‑3: Adam‑style mirror descent on the same (non‑smooth) objective
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   max_iters: int = 300_000,
                   verbose: bool = False) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]

    best = a.copy()
    best_val = _objective(best)

    m = np.zeros_like(a)        # first moment
    v = np.zeros_like(a)        # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.02

    t = 0
    while time.time() < deadline and t < max_iters:
        conv = _fft_convolve(a)
        k = int(np.argmax(conv))

        cur_val = 2.0 * n * conv[k]
        if cur_val < best_val:
            best_val = cur_val
            best = a.copy()
            if verbose:
                print(f"[adam] iter {t:06d}  obj {best_val:.9f}")

        # sub‑gradient of the active index
        grad = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        grad[idx] = 2.0 * a[k - idx]

        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = base_lr / (np.sqrt(v_hat) + eps)

        a = a - step * m_hat
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        # light random perturbations
        if t % 12_000 == 0:
            pert = np.random.normal(scale=0.008, size=n)
            a = np.maximum(a + pert, 0.0)
            a = _project_simplex(a)

    return best, best_val
