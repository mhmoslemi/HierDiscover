"""
Propose a step‑function (non‑negative list of heights) that minimises

    O = 2·n·max(conv(a,a)) / (sum(a)²)

The search consists of three stages:

1. **Parametric sweep** – try simple power‑law shapes
       a_i ∝ (i+β)^(-α)        (i = 1..n)
   on a grid for α and β and keep the best candidate.

2. **Local random refinement** – perturb the best (α,β) a few thousand
   times and accept any improvement.  An optional exponential tail
   factor exp(-γ·i/n) is also tried.

3. **Mirror‑descent (exponentiated‑gradient) polishing** – start from the
   best shape and run a projected exponentiated‑gradient descent on the
   true objective (sub‑gradient of the max‑convolution).  The loop is
   time‑capped, uses a decaying learning rate and occasional jitter to
   escape plateaus.

The whole routine respects the 1000 s budget, prints progress and finally
returns the best sequence as a plain ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a power‑of‑two FFT."""
    n = a.size
    size = 1 << ((2 * n - 2).bit_length())        # next power of two ≥ 2n‑1
    A = np.fft.fft(a, n=size)
    conv = np.fft.ifft(A * A).real[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def _objective_norm1(a: np.ndarray) -> float:
    """Objective 2·n·max(conv) assuming Σa = 1."""
    n = a.size
    conv = _conv_fft(a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv(a,a)) w.r.t. a (on the simplex)."""
    n = a.size
    conv = _conv_fft(a)
    max_val = conv.max()
    # indices achieving (almost) the maximum
    ks = np.where(np.abs(conv - max_val) < 1e-14)[0]
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        # derivative a_i*a_j appears twice (i↔j)
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
def _exp_update(a: np.ndarray, g: np.ndarray, eta: float) -> np.ndarray:
    """
    Exponentiated‑gradient update on the simplex.
    Works in log‑space for stability.
    """
    log_a = np.log(a + 1e-15)          # avoid log(0)
    log_a -= eta * g
    log_a -= np.max(log_a)            # shift to keep numbers moderate
    a_new = np.exp(log_a)
    a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
def _optimize(a0: np.ndarray,
              max_iters: int,
              eta0: float,
              deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) optimisation.
    Returns the best vector seen and its objective value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective_norm1(a)

    n = a.size
    no_improve = 0
    patience = max_iters // 20  # early‑stop after a while without improvement

    for t in range(max_iters):
        if time.time() >= deadline:
            break

        g = _subgradient_max_conv(a)

        # linearly decaying learning rate
        eta = eta0 * (1.0 - t / max_iters)

        a = _exp_update(a, g, eta)

        cur_score = _objective_norm1(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            print(f"[opt] iter {t:6d}  new best O={best_score:.6f}")
        else:
            no_improve += 1
            if no_improve >= patience:
                # small jitter to hop out of a flat region
                jitter = np.random.randn(n) * 1e-4
                a = best_a + jitter
                a = np.clip(a, 0.0, None)
                a = _project_to_simplex(a)
                no_improve = 0

    return best_a, best_score


# ----------------------------------------------------------------------
def _shape_power(alpha: float, beta: float, n: int) -> np.ndarray:
    """
    Produce a monotone decreasing sequence a_i ∝ (i+β)^(-α),
    i = 1..n, then normalise to sum‑1.
    """
    i = np.arange(1, n + 1, dtype=float)
    a = (i + beta) ** (-alpha)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


# ----------------------------------------------------------------------
def _shape_power_exp(alpha: float, beta: float, gamma: float, n: int) -> np.ndarray:
    """
    Power‑law with an exponential tail:
        a_i ∝ (i+β)^(-α) * exp(-γ·(i‑1)/n)
    """
    i = np.arange(1, n + 1, dtype=float)
    a = (i + beta) ** (-alpha) * np.exp(-gamma * (i - 1) / n)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


# ----------------------------------------------------------------------
