"""
Propose a non‑negative step‑function sequence that (approximately) minimizes

    O(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm has three stages:

1. **Smooth surrogate optimisation** – Adam on a log‑sum‑exp surrogate of the
   max‑convolution (the same idea as the reference code) but with a faster
   schedule, less console output and a modest learning‑rate decay.  The iterate
   is always projected onto the probability simplex and short‑range pruning
   keeps the support compact.

2. **Linear‑programming refinement** – With the current point `a` we compute
   `M = max(conv(a,a))`.  We then solve the LP

        maximise   Σ b_j
        subject to  A·b ≤ M·1 ,   b ≥ 0

   where `A` is the (2n‑1)×s convolution matrix of `a` restricted to a
   (small) support `S` of the largest coefficients of `a`.  The LP yields a
   vector `b` that carries as much mass as possible without violating the
   convolution bound.  After normalising `b` to the simplex we move a
   fraction `τ` towards it (`a ← (1‑τ)a + τ·b`).  This step flattens the
   convolution profile and typically yields a lower objective.

3. **Stochastic local search** – A cheap hill‑climbing loop that performs
   random mass‑transfer moves and occasional sub‑gradient steps on the
   currently active maximal convolution index.  It finishes the run with a
   final polishing pass.

All three phases respect the global time budget (default 1000 s).  The best
sequence encountered is returned as a plain Python list.

# Helper: Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        # all non‑positive – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full self‑convolution (length 2·n‑1)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return a*a (full convolution) using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Exact objective for a simplex‑normalised vector (sum = 1)
def _objective_simplex(a: np.ndarray, n: int, L: int) -> float:
    """2·n·max(conv(a,a)) (a is assumed to sum to 1)."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Sub‑gradient of the maximal convolution entry (as used in the reference code)
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Return the sub‑gradient of max(conv) w.r.t. a."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
# Gradient of the log‑sum‑exp (soft‑max) surrogate of the max
def _smooth_gradient(a: np.ndarray, conv: np.ndarray, alpha: float, L: int) -> np.ndarray:
    """
    Compute ∇_a ( (2·n/α)·log∑_k e^{α·conv_k} ).
    The formula reduces to 2·corr(w, a) where w_k ∝ e^{α·conv_k}.
    """
    # soft‑max weights
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)          # numerical stability
    w = exp_scaled / exp_scaled.sum()

    # correlation w ⋆ a   (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)   # length ≥ 3·n‑2
    n_len = a.shape[0]
    grad = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]
    return grad


# ----------------------------------------------------------------------
# Build the sparse convolution matrix A for a given support
def _build_constraint_matrix(a: np.ndarray, support_idx: np.ndarray) -> scipy.sparse.csr_matrix:
    """
    Return a sparse matrix A (shape (2n-1, |support|)) such that
    A @ b_sub = conv(a, b) for a full‑size vector b that is zero outside `support_idx`.
    """
    n = a.shape[0]
    s = support_idx.size
    # Allocate flat arrays for data, rows, cols
    data = np.empty(s * n, dtype=np.float64)
    rows = np.empty(s * n, dtype=np.int32)
    cols = np.empty(s * n, dtype=np.int32)

    pos = 0
    for col_pos, j in enumerate(support_idx):
        rows[pos:pos + n] = np.arange(j, j + n, dtype=np.int32)
        cols[pos:pos + n] = col_pos
        data[pos:pos + n] = a
        pos += n

    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, s))
    return A


# ----------------------------------------------------------------------
# Solve the LP  maximise Σ b_j  subject to A·b ≤ M·1 , b ≥ 0
def _solve_lp_direction(a: np.ndarray, M: float, support_idx: np.ndarray):
    """
    Return a full‑size vector b (non‑negative) solving the LP described above.
    If the LP fails, None is returned.
    """
    A = _build_constraint_matrix(a, support_idx)
    b_ub = np.full(2 * a.shape[0] - 1, M, dtype=np.float64)
    c = -np.ones(support_idx.size, dtype=np.float64)   # minimise -Σb  ⇔ maximise Σb
    # Bounds: all variables ≥ 0, no upper bound
    bounds = [(0, None)] * support_idx.size

    # Use HiGHS (fast and robust)
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                                 method='highs', options={'presolve': True})
    if not res.success:
        return None
    b_sub = res.x
    b_full = np.zeros_like(a)
    b_full[support_idx] = b_sub
    return b_full


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function sequence that minimises the given
    evaluation function.  Returns the sequence as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.3        # safety margin

    # ------------------------------------------------------------------
    # Initialise a (simplex‑normalised)
    if 'height_sequence_1' in globals():
        a = np.array(height_sequence_1, dtype=float)
        a = a / a.sum()
        n = a.size
        print(f"[init] using supplied sequence, n={n}")
    else:
        n = 4600
        a = np.full(n, 1.0 / n)
        print(f"[init] uniform start, n={n}")

    # FFT length – next power of two larger than the longest convolution/correlation
    L = 1 << ((3 * n - 2).bit_length())

    # ------------------------------------------------------------------
    # Phase 1 – Adam on smooth surrogate (≈45 % of total budget)
    phase1_deadline = start_time + budget_s * 0.45
    best_seq = a.copy()
    best_score = _objective_simplex(best_seq, n, L)
    print(f"[phase‑1] start O={best_score:.6f}")

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    lr0 = 0.7                     # slightly larger base LR
    alpha_start = 5.0
    alpha_end = 4000.0
    max_iter_phase1 = 250_000    # upper bound; loop stops early by time

    it = 0
    while time.time() < phase1_deadline and it < max_iter_phase1:
        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[phase‑1] iter {it} new best O={best_score:.6f}")

        # linear annealing of α
        frac = it / max_iter_phase1
        alpha = alpha_start + (alpha_end - alpha_start) * frac

        # smooth gradient
        grad = _smooth_gradient(a, conv, alpha, L)

        # Adam bias‑corrected update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        # step size decays ∝ 1/√t
        lr = lr0 / math.sqrt(it + 1)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional tiny jitter
        if np.random.rand() < 0.001:
            a_new += np.random.randn(n) * 1e-4

        # project back onto simplex
        a_new = _project_to_simplex(a_new)

        # prune extremely tiny entries every 5000 iter
        if it % 5000 == 0 and it > 0:
            thresh = np.percentile(a_new, 1)      # drop lowest 1 %
            a_new[a_new < thresh] = 0.0
            a_new = _project_to_simplex(a_new)

        a = a_new
        it += 1

    # ------------------------------------------------------------------
    # Phase 2 – Linear‑programming refinement (≈40 % of budget)
    phase2_deadline = start_time + budget_s * 0.85
    print(f"[phase‑2] entering LP refinement, best O={best_score:.6f}")

    # mixing coefficient schedule (larger early, smaller later)
    mix_start = 0.35
    mix_end = 0.07

    lp_iter = 0
    while time.time() < phase2_deadline:
        # current conv & objective
        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[LP] iter {lp_iter} new global best O={best_score:.6f}")

        # decide a support (keep the bulk of mass, drop negligible tails)
        # keep indices that together hold at least 99 % of the probability mass
        sorted_idx = np.argsort(-a)
        cum = np.cumsum(a[sorted_idx])
        keep_mask = cum <= 0.99
        support_idx = sorted_idx[keep_mask]
        # guarantee a minimal support size
        if support_idx.size < max(20, int(0.01 * n)):
            support_idx = sorted_idx[:max(20, int(0.01 * n))]

        M = conv.max()

        # solve LP on this reduced support
        b_full = _solve_lp_direction(a, M, support_idx)
        if b_full is None:
            # fallback: small random perturbation
            a += np.random.randn(n) * 0.001
            a = np.clip(a, 0, None)
            a = _project_to_simplex(a)
            continue

        # mixing coefficient (linearly decreasing)
        tau = mix_start + (mix_end - mix_start) * (lp_iter / 2000.0)
        tau = max(min(tau, 0.5), 0.02)   # keep it in a safe interval

        # normalise b to the simplex and take a convex combination
        sum_b = b_full.sum()
        if sum_b > 0:
            g = b_full / sum_b
            a = (1.0 - tau) * a + tau * g
            a = _project_to_simplex(a)      # numerical safety

        lp_iter += 1

    # ------------------------------------------------------------------
    # Phase 3 – Stochastic local refinement (remaining time)
    print(f"[phase‑3] final stochastic polishing, best O={best_score:.6f}")

    max_iter_local = 200_000
    no_improve_limit = 4_000
    it_loc = 0
    no_improve = 0
    T0 = 0.04                     # initial temperature for Metropolis acceptance

    while time.time() < deadline and it_loc < max_iter_local:
        # ----- random mass‑transfer move -----
        i, j = np.random.randint(n, size=2)
        if i != j and a[i] > 1e-14:
            delta = a[i] * np.random.rand() * 0.13   # up to 13 % of source mass
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            a_new = _project_to_simplex(a_new)

            conv_new = _conv_fft(a_new, L)
            new_obj = 2.0 * n * conv_new.max()

            # Metropolis acceptance
            if new_obj < cur_obj or np.random.rand() < math.exp((cur_obj - new_obj) / max(T0 * (1 - it_loc / max_iter_local), 1e-12)):
                a = a_new
                conv = conv_new
                cur_obj = new_obj
                no_improve = 0
                if cur_obj < best_score:
                    best_score = cur_obj
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] New best O={best_score:.6f} (local iter={it_loc})")
            else:
                no_improve += 1
        else:
            no_improve += 1

        # ----- occasional sub‑gradient step on the current maximal entry -----
        if it_loc % 9 == 0:
            gsg = _subgradient_max_conv(a, conv)
            eta = 0.09 * (1.0 - it_loc / max_iter_local)
            a = a - eta * gsg
            a = _project_to_simplex(a)
            conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_score:
                best_score = cur_obj
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Sub‑grad improved O={best_score:.6f}")

        # temperature decay
        T0 *= 0.99995
        it_loc += 1

        if no_improve >= no_improve_limit:
            # restart from the best found so far with slight jitter
            a = best_seq.copy()
            a += np.random.randn(n) * 0.001
            a = np.clip(a, 0, None)
            a = _project_to_simplex(a)
            conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max()
            no_improve = 0

    # ------------------------------------------------------------------
    # Final sanity check via the provided evaluator
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence failed:", exc)

    return best_seq.tolist()
