"""
Propose a step‑function (non‑negative sequence) that minimises

    O = 2·n·max(conv(a,a))   with   sum(a)=1

The algorithm works in two phases:

1. **Coarse parametric search** – a few simple families (uniform,
   exponential decay, power‑law) are evaluated on a range of parameters.
   The best sequence from this stage becomes the starter for phase 2.

2. **Adam‑optimised projected gradient descent** – the objective
   `max(conv(a,a))` is smoothed by a soft‑max with an annealing temperature
   β.  The gradient w.r.t. the log‑softmax parameters is computed via FFT‑based
   correlation, an Adam update is applied, and the resulting vector is
   projected back onto the probability simplex.  Occasional jitter and
   cheap restarts improve robustness.

The routine respects the given time budget, prints progress, and finally
returns the best sequence found (as a plain ``list[float]``).  It uses only
NumPy and the provided ``evaluate_sequence`` function.
"""

import time
import math
import numpy as np

# -------------------------------------------------------------
# Helper: projection onto the simplex {x ≥ 0, sum(x)=1}
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of a vector onto the probability simplex."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        # all non‑positive -> uniform vector
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# -------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# -------------------------------------------------------------
def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv


# -------------------------------------------------------------
def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation g_i = Σ_t w[i+t]·a[t]   for i=0,…,n‑1
    (no reversal of a).  Implemented via FFT.
    """
    n = a.shape[0]
    m = w.shape[0]                     # = 2n‑1
    L = _next_pow2(m + n - 1)          # length for linear convolution
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


# -------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective O = 2·n·max(conv(a,a))  (sum(a)=1)."""
    n = a.size
    conv = _conv_self_fft(a)
    return 2.0 * n * conv.max()


# -------------------------------------------------------------
def _make_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i), normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    a = np.exp(-lam * i)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


# -------------------------------------------------------------
def _make_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p), normalised to sum = 1."""
    i = np.arange(1, n + 1, dtype=float)
    a = i ** (-p)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


# -------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the bound.
    Returns the best sequence as a plain list of floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # ---------------------------------------------------------
    # Choose a length – the original code used ~4600, which is a
    # good trade‑off between expressivity and speed.
    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=float)
        if a0.size == 0:
            n = 4600
        else:
            a0 = a0 / a0.sum()
            n = a0.size
    else:
        n = 4600
        a0 = np.full(n, 1.0 / n)   # fallback (uniform)

    # ---------------------------------------------------------
    # -----------------------------------------------------------------
    # Phase 1 – coarse parametric search
    best_seq = a0.copy()
    best_score = _objective(best_seq)

    print(f"[phase‑1] Initial length n={n}, starting objective={best_score:.6f}")

    # uniform (already covered by a0) – evaluate once more for safety
    # exponential family
    lam_candidates = np.logspace(-5, -1, num=30)   # 1e‑5 … 0.1
    for lam in lam_candidates:
        a = _make_exponential(n, lam)
        sc = _objective(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[phase‑1] exp λ={lam:.5g} → O={sc:.6f}")

    # power‑law family
    p_candidates = np.linspace(0.1, 3.0, num=30)
    for p in p_candidates:
        a = _make_powerlaw(n, p)
        sc = _objective(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[phase‑1] power p={p:.3f} → O={sc:.6f}")

    # random start (a few seeds)
    for _ in range(5):
        a = np.random.rand(n)
        a = _project_to_simplex(a)
        sc = _objective(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[phase‑1] random start → O={sc:.6f}")

    # ---------------------------------------------------------
    # Phase 2 – Adam‑optimised projected gradient descent on a soft‑max
    #          approximation of the max‑convolution.
    # ---------------------------------------------------------

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    eta0 = 0.45                     # base learning rate
    lr_decay = 5e-5                 # simple decay per iteration
    beta_start = 4.0
    beta_end = 4000.0
    max_iters_est = 200_000        # used for beta schedule

    # initialise from the best sequence of phase‑1
    a = best_seq.copy()
    # log‑softmax parameters (z) – ensures positivity and sum‑1 after softmax
    eps = 1e-12
    z = np.log(a + eps)

    m = np.zeros_like(z)    # first moment
    v = np.zeros_like(z)    # second moment
    t = 0

    iter_cnt = 0
    last_improvement = 0

    while time.time() < deadline and iter_cnt < 500_000:
        iter_cnt += 1
        t += 1

        # -----------------------------------------------------------------
        # 1) current convolution and objective
        conv = _conv_self_fft(a)
        M = conv.max()
        cur_score = 2.0 * n * M
        if cur_score + 1e-12 < best_score:
            best_score = cur_score
            best_seq = a.copy()
            last_improvement = iter_cnt
            print(f"[iter {iter_cnt:06d}] new best O={best_score:.6f}")

        # -----------------------------------------------------------------
        # 2) annealing schedule for the soft‑max temperature β
        fraction = min(iter_cnt / max_iters_est, 1.0)
        beta = beta_start + (beta_end - beta_start) * fraction

        # -----------------------------------------------------------------
        # 3) compute soft‑max weights for the max‑convolution
        if beta > 2000:                     # switch to exact sub‑gradient when very sharp
            mask = (conv == M)
            w = mask.astype(float) / mask.sum()
        else:
            shifted = conv - M               # ≤ 0
            exp_vals = np.exp(beta * shifted)
            w = exp_vals / exp_vals.sum()

        # -----------------------------------------------------------------
        # 4) gradient of the smooth max w.r.t. a
        grad_a = 2.0 * _correlation_fft(w, a)

        # -----------------------------------------------------------------
        # 5) convert gradient to the log‑softmax variables (z)
        mu = np.dot(grad_a, a)          # ⟨grad_a, a⟩
        grad_z = a * (grad_a - mu)

        # -----------------------------------------------------------------
        # 6) Adam update on z
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * (grad_z ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        eta = eta0 / (1.0 + iter_cnt * lr_decay)
        step = eta * m_hat / (np.sqrt(v_hat) + eps_adam)
        z = z - step

        # -----------------------------------------------------------------
        # 7) map back to the simplex (soft‑max)
        max_z = np.max(z)                 # for numerical stability
        a = np.exp(z - max_z)
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / a_sum

        # -----------------------------------------------------------------
        # 8) occasional jitter to escape plateaus
        if iter_cnt % 4000 == 0:
            jitter = np.random.randn(n) * 1e-4
            z += jitter

        # -----------------------------------------------------------------
        # 9) cheap restart if stagnation lasts too long
        if iter_cnt - last_improvement > 30_000:
            # re‑initialise from the current best with a small perturbation
            a = best_seq.copy()
            jitter = np.random.randn(n) * 5e-4
            a = a + jitter
            a = _project_to_simplex(a)
            z = np.log(a + eps)
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            last_improvement = iter_cnt
            print(f"[restart] iter {iter_cnt}, re‑initialised from best.")

    # -----------------------------------------------------------------
    # Final evaluation with the official routine.
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_seq.tolist()
