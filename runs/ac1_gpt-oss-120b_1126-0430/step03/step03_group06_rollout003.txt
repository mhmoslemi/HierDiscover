"""
Propose a near‑optimal step‑function sequence.

Algorithm
---------

1. **Parameter search** – on a moderate size `n≈20 000` we generate many
   candidate shapes (power laws, shifted power laws, beta‑type,
   sine‑type and the Catalan distribution).  Each candidate is normalised
   to sum‑to‑one and its objective
   `O(a)=2·n·max(a*a)` is evaluated using an FFT‑based convolution.
   The best shape found so far is kept.

2. **Scale‑up** – if time permits we double the length (linear interpolation
   of the best shape) and continue optimisation on the larger problem.

3. **Projected sub‑gradient descent** – starting from the best candidate we
   iteratively minimise `O(a)`.  In each iteration we
   * locate the index `k` where the convolution attains its maximum,
   * compute the sub‑gradient `g = 2·J_k·a` (only the active Toeplitz block
     of the convolution contributes),
   * apply heavy‑ball momentum and a diminishing step‑size,
   * project back onto the probability simplex (non‑negative, sum = 1).

   Random perturbations are injected every few thousand steps to avoid
   plateaus.  The loop runs until the allocated time expires.

4. **Optional random restarts** – if any time remains we launch a few cheap
   random restarts of the sub‑gradient optimiser.

The returned list is a plain Python list of non‑negative numbers.  The
objective value (as returned by `evaluate_sequence`) is printed for
reference; the evaluation harness will call `evaluate_sequence` on the
output.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear convolution (full mode)
# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2*len(a)-1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[: 2 * n - 1]

# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x ≥ 0, Σx = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Objective (identical to evaluate_sequence for Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    return float(2.0 * n * np.max(_fft_convolve(a)))

# ----------------------------------------------------------------------
# Analytic families of candidate sequences (all returned normalised)
# ----------------------------------------------------------------------
def _catalan_normalised(n: int) -> np.ndarray:
    """Catalan‐type coefficients, normalised to unit L1 norm."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    a /= a.sum()
    return a


def _power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p)
    return a / a.sum()


def _shifted_power(n: int, p: float, shift: float) -> np.ndarray:
    """a_i ∝ (i+shift)^{-p}, shift ≥ 0."""
    i = np.arange(n, dtype=np.float64) + shift
    a = i ** (-p)
    a[a < 0] = 0.0
    return a / a.sum()


def _beta_family(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^α·(n‑i)^β   (α,β may be negative)."""
    i = np.arange(n, dtype=np.float64) + 1.0
    x = i / n
    a = (x ** max(alpha, 0)) * ((1 - x) ** max(beta, 0))
    # for negative exponents we set a small epsilon to avoid division by zero
    eps = 1e-12
    if alpha < 0:
        a *= (x + eps) ** alpha
    if beta < 0:
        a *= ((1 - x) + eps) ** beta
    a = np.maximum(a, 0.0)
    if a.sum() == 0:
        a = np.ones_like(a)
    return a / a.sum()


def _sine_family(n: int, phase: float) -> np.ndarray:
    """a_i ∝ sin(π·(i+0.5)/n + phase), clipped to non‑negative."""
    i = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * i / n + phase)
    a = np.maximum(a, 0.0)
    if a.sum() == 0:
        a = np.ones_like(a)
    return a / a.sum()


# ----------------------------------------------------------------------
# Projected sub‑gradient optimiser (heavy‑ball momentum)
# ----------------------------------------------------------------------
def _optimize(a0: np.ndarray,
              deadline: float,
              max_iters: int = 300_000,
              verbose: bool = True) -> tuple[np.ndarray, float]:
    """
    Refine a simplex‑normalised vector a0 by sub‑gradient descent on
    `2·n·max(a*a)`.  Returns the best vector seen and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]

    best = a.copy()
    best_val = _objective(best)

    beta = 0.9                     # heavy‑ball momentum coefficient
    momentum = np.zeros_like(a)

    t = 0
    while time.time() < deadline and t < max_iters:
        conv = _fft_convolve(a)
        k = int(np.argmax(conv))          # active convolution index
        max_c = conv[k]

        cur_val = 2.0 * n * max_c
        if cur_val < best_val:
            best_val = cur_val
            best = a.copy()
            if verbose:
                print(f"[improve] iter {t:06d}  obj {best_val:.9f}")

        # sub‑gradient for the active index (2·J_k·a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # diminishing step‑size (1/√t schedule)
        lr = 0.6 / np.sqrt(t + 1)

        # gradient step + projection onto simplex
        a = a - lr * momentum
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        # occasional random jitter to escape flat regions
        if t % 7_000 == 0 and t > 0:
            pert = np.random.normal(scale=0.015, size=n)
            a = np.maximum(a + pert, 0.0)
            a = _project_simplex(a)

        t += 1

    return best, best_val

# Main entry point required by the competition harness

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a sequence of non‑negative heights that minimises
    ``evaluate_sequence`` under the given time budget.
    """
    np.random.seed(seed)
    # leave a small safety margin for the harness
    deadline = time.time() + budget_s - 2.0

    # --------------------------------------------------------------
    # 1) Parameter search on a moderate size (fast evaluations)
    # --------------------------------------------------------------
    n_small = 20_000
    best_seq = _catalan_normalised(n_small)
    best_val = _objective(best_seq)
    print(f"[init] Catalan objective {best_val:.9f}")

    # Allocate roughly half of the time to the random‑parameter sweep
    search_deadline = deadline - 30.0   # keep 30 s for refinement / restarts

    while time.time() < search_deadline:
        choice = np.random.rand()
        if choice < 0.15:                               # pure power law
            p = np.random.uniform(0.6, 2.4)
            a = _power_law(n_small, p)

        elif choice < 0.30:                             # shifted power law
            p = np.random.uniform(0.6, 2.4)
            shift = np.random.uniform(0.0, 5.0)
            a = _shifted_power(n_small, p, shift)

        elif choice < 0.45:                             # beta‑type family
            alpha = np.random.uniform(-0.8, 4.5)
            beta = np.random.uniform(-0.8, 4.5)
            a = _beta_family(n_small, alpha, beta)

        elif choice < 0.60:                             # sine family
            phase = np.random.uniform(0.0, np.pi / 2)
            a = _sine_family(n_small, phase)

        elif choice < 0.75:                             # Catalan (again, but with random scaling)
            a = _catalan_normalised(n_small)

        else:                                           # small random Dirichlet start
            a = np.random.rand(n_small)
            a /= a.sum()

        val = _objective(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[search] new best obj {best_val:.9f}")

    # --------------------------------------------------------------
    # 2) If time permits, upscale and run projected sub‑gradient descent
    # --------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 90.0:
        # upscale factor – 2 works well without blowing up memory
        upscale = 2
        n_big = n_small * upscale
        # linear interpolation of the current best shape
        x_old = np.linspace(0.0, 1.0, n_small)
        x_new = np.linspace(0.0, 1.0, n_big)
        a_big = np.interp(x_new, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        # run the optimiser on the larger problem
        best_seq, best_val = _optimize(a_big, deadline,
                                       max_iters=250_000,
                                       verbose=True)

        n_small = n_big
    else:
        # not enough time to enlarge – just polish the current best
        best_seq, best_val = _optimize(best_seq, deadline,
                                       max_iters=200_000,
                                       verbose=False)

    # --------------------------------------------------------------
    # 3) Optional cheap random restarts while time remains
    # --------------------------------------------------------------
    while time.time() < deadline - 5.0:
        a_rand = np.random.rand(n_small)
        a_rand /= a_rand.sum()
        a_opt, val = _optimize(a_rand, deadline,
                               max_iters=60_000,
                               verbose=False)
        if val < best_val:
            best_val = val
            best_seq = a_opt.copy()
            print(f"[restart] new best obj {best_val:.9f}")

    # --------------------------------------------------------------
    # 4) Final report
    # --------------------------------------------------------------
    final_obj = _objective(best_seq)
    print(f"[final] best objective ≈ {final_obj:.9f} with n={len(best_seq)}")
    # Convert to plain Python list as required by the harness
    return best_seq.tolist()
