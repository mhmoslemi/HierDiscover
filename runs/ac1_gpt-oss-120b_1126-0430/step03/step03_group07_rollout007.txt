author:  ChatGPT (optimization via parametric mixtures + DE + Adam)
date:    2026‑11‑26

summary
-------
The metric to minimise is  

    2·n·max(conv(a,a)) / (∑a)² .

It is scale–invariant, therefore we fix ∑a = 1 and search for a
probability vector a∈ℝⁿ₊ that makes the self‑convolution as flat as
possible (the average of the convolution is 1/(2n‑1), so the optimum
would be a constant convolution giving a metric of 1).  A smooth
distribution that concentrates mass at the ends reduces the centre of
the convolution, while a uniform part reduces the spikes at the ends.
We therefore search a *mixture* of three elementary shapes:

    • Beta(α₁,β₁)   – a flexible “U‑shape” (heavy at the ends)
    • Beta(α₂,β₂)   – a second, possibly more central, shape
    • Uniform        – a flat background

The mixture coefficients are kept on the simplex by a soft‑max on three
log‑weights.  The whole 7‑dimensional parameter vector is optimised with
a differential‑evolution global optimiser (SciPy).  For every candidate
the metric is evaluated extremely fast by an FFT‑based convolution
(`_fast_evaluation`).  The best mixture found is finally polished with
an Adam‑style multiplicative gradient descent on the simplex
(`_adam_refine`).

The whole routine respects the 1000‑second budget, prints a few
progress messages and returns a list of non‑negative floats whose sum
is 1.  The returned sequence has length 5000 (the length used in the

"""

# ----------------------------------------------------------------------
# Fast evaluation – identical to the supplied `evaluate_sequence`
# (but without the security checks).  Returns the metric.
# ----------------------------------------------------------------------
def _fast_evaluation(seq: np.ndarray) -> float:
    """Return   2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    n = seq.size
    if n == 0:
        return float("inf")
    # length of FFT (next power of two ≥ 2·n‑1)
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))

# ----------------------------------------------------------------------
def _construct_sequence(params: np.ndarray) -> np.ndarray:
    v0, v1, v2, a1, b1, a2, b2 = params

    # soft‑max to obtain mixture weights that sum to one
    w_raw = np.exp([v0, v1, v2])
    w = w_raw / w_raw.sum()
    w0, w1, w2 = w

    # Beta‑pdf 1  (computed in log‑space for stability)
    with np.errstate(divide="ignore", invalid="ignore"):
        log_pdf1 = (a1 - 1.0) * _LOG_X + (b1 - 1.0) * _LOG_ONE_MINUS_X
    pdf1 = np.exp(log_pdf1)
    pdf1[np.isnan(pdf1) | np.isinf(pdf1)] = 0.0

    # Beta‑pdf 2
    with np.errstate(divide="ignore", invalid="ignore"):
        log_pdf2 = (a2 - 1.0) * _LOG_X + (b2 - 1.0) * _LOG_ONE_MINUS_X
    pdf2 = np.exp(log_pdf2)
    pdf2[np.isnan(pdf2) | np.isinf(pdf2)] = 0.0

    # Uniform component
    uniform = np.ones_like(_X)

    # linear mixture
    seq = w0 * pdf1 + w1 * pdf2 + w2 * uniform

    # enforce exact symmetry (the objective is symmetric)
    seq = 0.5 * (seq + seq[::-1])

    # normalise to sum = 1 (the metric is scale‑invariant)
    total = seq.sum()
    if total == 0.0:
        seq = np.full_like(seq, 1.0 / seq.size)
    else:
        seq = seq / total
    return seq


# ----------------------------------------------------------------------
# Objective function for the global optimiser (returns the metric).
# ----------------------------------------------------------------------
def _objective(params: np.ndarray) -> float:
    seq = _construct_sequence(params)
    return _fast_evaluation(seq)


# ----------------------------------------------------------------------
# Adam‑style exponentiated‑gradient optimiser on the simplex.
# (identical to the one used in the previous state‑of‑the‑art code).
# ----------------------------------------------------------------------
def _adam_refine(seq: np.ndarray, deadline: float) -> np.ndarray:
    """
    Refine a probability vector `seq` (sum = 1, non‑negative) by
    projected Adam updates in log‑space until `deadline`.
    Returns the best vector seen.
    """
    n = seq.size
    eps = 1e-15
    log_seq = np.log(seq + eps)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    lr = 0.12
    m = np.zeros_like(seq)
    v = np.zeros_like(seq)
    t = 0

    # FFT length (fixed for the whole run)
    fft_len = 1 << ((2 * n - 1).bit_length())

    best_seq = None
    best_score = float("inf")
    it = 0

    while time.time() < deadline:
        # current probability vector
        seq = np.exp(log_seq)
        sum_seq = seq.sum()
        if sum_seq == 0.0:
            seq = np.full(n, 1.0 / n)
            log_seq = np.log(seq)
        else:
            seq /= sum_seq

        # convolution and active set (indices attaining the maximum)
        fa = np.fft.rfft(seq, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        M = conv.max()
        tol = 1e-12
        active = np.where(conv >= M - tol)[0]

        # sub‑gradient of the max‑operator
        g = np.zeros_like(seq)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            i_range = np.arange(lo, hi + 1, dtype=int)
            j_range = k - i_range
            g[i_range] += seq[j_range]

        # Adam moments
        t += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # multiplicative (log‑space) update
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        log_seq -= step

        # keep log‑values in a reasonable range
        np.clip(log_seq, -30.0, 30.0, out=log_seq)

        # evaluate current point
        score = _fast_evaluation(seq)
        if score < best_score - 1e-12:
            best_score = score
            best_seq = seq.copy()
            if it % 500 == 0:
                print(f"[Adam] iter {it:6d}  best {best_score:.6f}")

        # tiny random jitter to escape flat regions
        if np.random.rand() < 0.001:
            j = np.random.randint(n)
            seq[j] = max(0.0, seq[j] + np.random.randn() * 1e-3)
            seq /= seq.sum()
            log_seq = np.log(seq + eps)

        it += 1

    # fallback (should never happen)
    if best_seq is None:
        seq = np.exp(log_seq)
        seq /= seq.sum()
        best_seq = seq
    return best_seq


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Produce a step‑function (list of non‑negative floats) whose metric is
    as low as possible within the allotted time.
    The method:
      1. Global optimisation (differential evolution) over a 7‑parameter
         mixture of two Beta PDFs and a uniform background.
      2. A final Adam‑style local refinement on the full‑size vector.
    The returned list sums to 1 and has length 5000.
    """
    np.random.seed(seed)
    start = time.time()
    # keep a small safety margin
    deadline = start + budget_s - 5.0

    # ------------------------------------------------------------------
    # 1. Global optimisation with differential evolution
    # ------------------------------------------------------------------
    #   params = [v0, v1, v2, a1, b1, a2, b2]
    bounds = [
        (-5.0, 5.0),  # log‑weight v0
        (-5.0, 5.0),  # log‑weight v1
        (-5.0, 5.0),  # log‑weight v2
        (0.01, 5.0),  # α₁
        (0.01, 5.0),  # β₁
        (0.01, 5.0),  # α₂
        (0.01, 5.0),  # β₂
    ]

    # The DE run is very cheap (a few thousand evaluations).  We limit
    # the number of generations to stay well inside the time budget.
    print("[DE] start global search")
    result = differential_evolution(
        _objective,
        bounds,
        maxiter=30,
        popsize=15,
        seed=seed,
        polish=False,
        updating="deferred",
    )
    print(f"[DE] best metric {result.fun:.6f}")

    # ------------------------------------------------------------------
    # 2. Build the best sequence from the DE result
    # ------------------------------------------------------------------
    best_seq = _construct_sequence(result.x)

    # ------------------------------------------------------------------
    # 3. Local Adam refinement (uses the remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[Adam] start local refinement")
        best_seq = _adam_refine(best_seq, deadline)

    # ------------------------------------------------------------------
    # 4. Final report and return
    # ------------------------------------------------------------------
    final_metric = _fast_evaluation(best_seq)
    print(f"[final] metric {final_metric:.6f}   length {best_seq.size}")

    # Return a plain Python list as required by the harness
    return best_seq.tolist()
