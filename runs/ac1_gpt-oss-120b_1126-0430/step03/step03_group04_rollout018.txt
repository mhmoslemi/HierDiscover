"""
Optimisation of a non‑negative step function for the bound

    O = 2·n·max(conv(a,a))   (with sum(a)=1)

The method is a **mirror‑descent / exponentiated‑gradient** optimiser on the
probability simplex.  At each iteration we

1. compute the full self‑convolution of the current candidate,
2. build a weight vector over the convolution entries:
   * early in the run a smooth soft‑max (log‑sum‑exp) is used,
   * later the weight is concentrated on the (few) maximal entries
     (hard‑max / active‑set).
3. obtain the gradient of the (smoothed) max‑convolution via a correlation
   (FFT based, O(n log n)).
4. update the logarithmic variables  u_i = log a_i  by a decreasing
   learning rate η and then renormalise (i.e. a_i = exp(u_i) / Σ_j exp(u_j)).
5. occasional random jitter and a cheap restart from the best‑so‑far
   solution keep the search from stagnating.

The routine respects a 1000 s time budget, prints progress, evaluates the
final solution with the provided ``evaluate_sequence`` and returns the best
list of coefficients found.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv


def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute the linear correlation  g_i = Σ_t w[i+t]·a[t]   for i=0,…,n‑1.
    Implemented via FFT.  ``w`` has length 2·n‑1.
    """
    n = a.shape[0]
    m = w.shape[0]                # = 2·n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` bound.  Returns the best sequence as a plain
    ``list`` of ``float``.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # -----------------------------------------------------------------
    # 1) Choose length and initialise the log‑probabilities.
    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=float)
        a0 = a0 / a0.sum()
        n = a0.size
        # start from the known solution, but add a tiny jitter
        eps = 1e-12
        u = np.log(a0 + eps)
    else:
        # a moderately large length – larger than the previous 4600
        n = 8000
        u = np.full(n, -np.log(n))               # uniform → log(1/n)

    best_u = u.copy()
    best_score = np.inf
    best_iter = 0

    # -----------------------------------------------------------------
    # 2) Schedules
    beta_start = 4.0
    beta_end   = 20000.0            # much stronger annealing than before
    eta_start  = 0.5                # initial step‑size in the log‑domain
    eta_end    = 0.03               # final step‑size
    max_iters_est = 300_000        # rough estimate – the loop stops earlier
    jitter_interval = 5000
    jitter_scale = 1e-3
    restart_interval = 30_000

    it = 0
    while time.time() < deadline and it < 700_000:
        it += 1

        # -------------------------------------------------------------
        # (a) build the probability vector and its convolution
        #   use log‑normalisation to avoid overflow
        a = np.exp(u - u.max())
        a_sum = a.sum()
        if a_sum == 0.0:                     # numerical safety
            a = np.full_like(a, 1.0 / n)
            a_sum = 1.0
        a = a / a_sum

        conv = _conv_self_fft(a)
        M = conv.max()
        cur_score = 2.0 * n * M               # because sum(a)=1

        # -------------------------------------------------------------
        # (b) update the best solution
        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_u = u.copy()
            best_iter = it
            print(f"[iter {it:07d}] new best O={best_score:.6f}")

        # -------------------------------------------------------------
        # (c) schedule parameters
        fraction = min(it / max_iters_est, 1.0)
        beta = beta_start + (beta_end - beta_start) * fraction
        # exponential decay of η (more aggressive early, gentle later)
        eta = eta_start * (eta_end / eta_start) ** fraction

        # -------------------------------------------------------------
        # (d) build the weight vector over convolution entries
        #     use soft‑max early, hard‑max later
        if fraction < 0.4:                     # soft‑max phase
            shifted = conv - M                 # ≤ 0, max is 0
            # subtraction of M already guarantees the exponent ≤ 0,
            # so no overflow occurs even for large beta.
            w = np.exp(beta * shifted)
            w /= w.sum()
        else:                                 # active‑set (hard‑max) phase
            # tolerance relative to the current maximum – tiny to capture
            # only the truly binding constraints.
            tol = max(1e-12, M * 1e-8)
            active = np.where(conv >= M - tol)[0]
            if active.size == 0:               # fallback – should not happen
                w = np.zeros_like(conv)
                w[conv.argmax()] = 1.0
            else:
                w = np.zeros_like(conv)
                w[active] = 1.0 / active.size

        # -------------------------------------------------------------
        # (e) gradient of the (smoothed) max‑convolution
        g = 2.0 * _correlation_fft(w, a)       # shape (n,)

        # -------------------------------------------------------------
        # (f) mirror‑descent / exponentiated‑gradient step on the log‑variables
        u = u - eta * g

        # -------------------------------------------------------------
        # (g) occasional random jitter to escape flat plateaus
        if it % jitter_interval == 0:
            u += np.random.randn(n) * jitter_scale

        # -------------------------------------------------------------
        # (h) cheap restart if we have been unproductive for a while
        if it % restart_interval == 0:
            # Re‑initialise from the best solution with a small perturbation.
            u = best_u.copy()
            u += np.random.randn(n) * (jitter_scale * 0.5)
            print(f"[restart] iter {it}, re‑initialised from best (O={best_score:.6f})")

    # -----------------------------------------------------------------
    # 3) Final evaluation with the provided evaluator
    best_seq = np.exp(best_u - best_u.max())
    best_seq = best_seq / best_seq.sum()
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f} "
              f"(best iteration {best_iter})")
    except Exception as exc:                                   # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_seq.tolist()
