#!/usr/bin/env python
"""
Propose a step–function (non‑negative heights) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm follows a *smoothed* mirror‑descent on the probability simplex:

* The vector `a` is represented by logits `z`; `a = softmax(z)` guarantees
  non‑negativity and Σa = 1, so the exact evaluation reduces to
  the cheap proxy `2·n·max(conv)`.
* At each iteration we form the self‑convolution `c = a*a` (via a real FFT)
  and a *soft‑max* weighting `w_k ∝ exp(β·c_k)`.  For small `β` the
  objective behaves like an average of the convolution entries, for large
  `β` it approaches the true maximum.  By slowly annealing `β` we obtain a
  smooth landscape early on and a precise refinement later.
* The gradient of the smoothed objective w.r.t. `a` is
      ∇ₐ f = 2·n· Σ_k w_k ∇ₐ c_k .
  Using the identity `∇ₐ c_k[i] = 2·a_{k‑i}` (with out‑of‑range terms zero)
  this becomes a weighted cross‑correlation, which can be evaluated with a
  single extra FFT.
* Adam is applied in log‑it space (a mirror‑descent step on the simplex).
* A short back‑tracking line‑search guarantees monotone decrease of the
  cheap proxy.  Random perturbations and occasional restarts help escape
  plateaus.
* The search starts from the best of a small family of analytic shapes
  (power‑law, symmetric Beta, sine) and finally evaluates the best
  candidate with the supplied `evaluate_sequence`.

The routine respects a time budget (≈ 1000 s) and returns the best
sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max; returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it representation that yields `a` after soft‑max."""
    return np.log(a + eps)


def _fft_len_for_n(k: int) -> int:
    """Smallest power‑of‑two ≥ k (efficient size for real FFT)."""
    return 1 << ((k - 1).bit_length())


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """Normalised a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    a /= a.sum()
    return a


def _beta_symmetric_distribution(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) on {0,…,n‑1}: a_i ∝ (i+1)^{α‑1}·(n‑i)^{α‑1}."""
    i = np.arange(n, dtype=np.float64) + 1.0          # 1 … n
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (alpha - 1.0))
    a /= a.sum()
    return a


def _sine_distribution(n: int) -> np.ndarray:
    """A half‑period sine shape a_i ∝ sin(π·(i+0.5)/n)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * i / n)
    a /= a.sum()
    return a


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    s = new_a.sum()
    if s == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= s
    return new_a
