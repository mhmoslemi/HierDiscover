def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (sum=1, non‑negative)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all entries non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
```

"""
Propose a step‑function (non‑negative heights) that minimizes

    2·n·max_k (a*a)[k] / (Σ a_i)²

using a smooth‑max (log‑sum‑exp) surrogate and Adam optimisation in the
simplex.  The algorithm works with any length (default ≈20 000) and stays
well within the 1000 s budget by using FFTs for all convolutions.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (must be top level – no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project **v** onto the probability simplex (Σ = 1, v ≥ 0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all entries non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_seq(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)  (normalized to Σ=1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = i ** (-p)
    seq /= seq.sum()
    return seq


def _exp_seq(n: int, q: float) -> np.ndarray:
    """a_i ∝ q**i  (0<q<1)  normalized to Σ=1."""
    i = np.arange(n, dtype=np.float64)
    seq = q ** i
    seq /= seq.sum()
    return seq


def _beta_seq(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric Beta‐like shape a_i ∝ (i+1)^{α-1} (n-i)^{β-1} , Σ=1."""
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    seq = np.maximum(seq, 0.0)
    seq /= seq.sum()
    return seq


def _conv_linear(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa).real[: 2 * a.size - 1]
    return conv


def _smooth_max_weights(conv: np.ndarray, tau: float) -> np.ndarray:
    """
    Return soft‑max weights w_k ≈ 1_{k = argmax conv}
    using temperature τ.
    """
    # shift for numerical stability (max subtraction)
    c_max = conv.max()
    exp_terms = np.exp((conv - c_max) / tau)
    w = exp_terms / exp_terms.sum()
    return w


def _gradient_smooth_max(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of τ·log Σ exp(conv/τ) w.r.t. a.
    It is 2·(w ⋆ a) where ⋆ denotes linear cross‑correlation.
    Returns a vector of length len(a).
    """
    n = a.size
    # linear cross‑correlation (w ⋆ a)[i] = Σ_j w[i+j]·a[j]
    L = len(w) + n - 1
    fft_len = 1 << ((L - 1).bit_length())
    fa = np.fft.fft(a, fft_len)
    fw = np.fft.fft(w, fft_len)
    # cross‑correlation via conj(FFT(a))
    corr = np.real(np.fft.ifft(fw * np.conj(fa)))[:n]
    return 2.0 * corr

# Main optimisation routine required by the platform

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence``.  The routine respects the given time budget.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0               # safety margin

    # --------------------------------------------------------------
    # 1. Choose working length.
    # --------------------------------------------------------------
    # reuse a previous construction if it exists, otherwise start with
    # a moderate size (≈20 000).  Larger n gives a better asymptotic bound.
    try:
        prev = height_sequence_1                      # type: ignore[name-defined]
        n = max(len(prev), 8000)                     # never smaller than 8 k
        prev_arr = np.asarray(prev, dtype=np.float64)
        if prev_arr.sum() == 0:
            prev_arr = None
        else:
            prev_arr = prev_arr / prev_arr.sum()
    except Exception:
        prev_arr = None
        n = 20000

    # FFT length for the a*a convolution (needs ≥ 2n‑1).
    fft_len_conv = 1 << ((2 * n - 1).bit_length())

    # --------------------------------------------------------------
    # 2. Scan a few parametric families (power‑law, exponential,
    #    symmetric Beta) – cheap evaluation with the FFT.
    # --------------------------------------------------------------
    best_a = None
    best_score = float("inf")          # cheap score = 2·n·max(conv)
    # power‑law grid
    for p in np.linspace(0.55, 0.70, 41):
        if time.time() >= deadline:
            break
        a = _power_law_seq(n, p)
        conv = _conv_linear(a, fft_len_conv)
        cheap = 2.0 * n * conv.max()
        if cheap < best_score:
            best_score, best_a = cheap, a.copy()

    # exponential grid
    for q in np.linspace(0.90, 0.999, 31):
        if time.time() >= deadline:
            break
        a = _exp_seq(n, q)
        conv = _conv_linear(a, fft_len_conv)
        cheap = 2.0 * n * conv.max()
        if cheap < best_score:
            best_score, best_a = cheap, a.copy()

    # symmetric beta grid (α=β)
    for a_param in np.linspace(0.3, 2.0, 35):
        if time.time() >= deadline:
            break
        a = _beta_seq(n, a_param, a_param)
        conv = _conv_linear(a, fft_len_conv)
        cheap = 2.0 * n * conv.max()
        if cheap < best_score:
            best_score, best_a = cheap, a.copy()

    # consider the previous construction if it matches the current length
    if prev_arr is not None and prev_arr.size == n:
        conv = _conv_linear(prev_arr, fft_len_conv)
        cheap = 2.0 * n * conv.max()
        if cheap < best_score:
            best_score, best_a = cheap, prev_arr.copy()

    # Fallback – uniform (should never happen)
    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_score = 2.0 * n * _conv_linear(best_a, fft_len_conv).max()

    print(f"Initial scan finished: n={n}, best cheap score={best_score:.6f}")

    # --------------------------------------------------------------
    # 3. Refine with Adam on the smooth‑max surrogate.
    # --------------------------------------------------------------
    a = best_a.copy()                 # current iterate (always Σ=1)
    m = np.zeros_like(a)              # 1st moment
    v = np.zeros_like(a)              # 2nd moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    lr0 = 0.5                         # initial learning rate
    tau_start = 0.5
    tau_min = 1e-3

    best_iter_a = a.copy()
    best_iter_score = best_score
    last_improve = 0
    iter_cnt = 0
    max_iters = 1_200_000

    # Pre‑compute FFT length for gradient step (w length = 2n‑1)
    fft_len_grad = 1 << ((3 * n - 2).bit_length())

    while time.time() < deadline and iter_cnt < max_iters:
        iter_cnt += 1

        # ------- convolution and smooth‑max weights -------
        conv = _conv_linear(a, fft_len_conv)          # length 2n‑1
        cheap = 2.0 * n * conv.max()
        tau = max(tau_min, tau_start / math.sqrt(iter_cnt))
        w = _smooth_max_weights(conv, tau)            # soft‑max over the 2n‑1 coeffs

        # ------- gradient of the surrogate -------
        grad = _gradient_smooth_max(a, w)             # length n

        # ------- Adam update -------
        lr = lr0 / math.sqrt(iter_cnt)                # simple decay
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** iter_cnt)
        v_hat = v / (1.0 - beta2 ** iter_cnt)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a - step
        a_candidate = _simplex_projection(a_candidate)

        a = a_candidate

        # ------- keep the best cheap score -------
        if (iter_cnt % 300 == 0) or (time.time() > deadline - 5):
            if cheap < best_iter_score:
                best_iter_score = cheap
                best_iter_a = a.copy()
                last_improve = iter_cnt
                print(f"[Iter {iter_cnt:07d}] NEW BEST cheap = {cheap:.6f}")

        # ------- occasional random perturbation to escape plateaus -------
        if iter_cnt - last_improve > 120_000:
            perturb = 0.02 * np.random.rand(n)
            a = _simplex_projection(best_iter_a + perturb)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = iter_cnt
            print(f"[Iter {iter_cnt:07d}] Restarted after stagnation (perturb).")

    # --------------------------------------------------------------
    # 4. Final exact evaluation (uses the provided evaluate_sequence)
    # --------------------------------------------------------------
    final_seq = best_iter_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:                     # defensive – should never happen
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # --------------------------------------------------------------
    # 5. Return the best sequence found.
    # --------------------------------------------------------------
    return final_seq
