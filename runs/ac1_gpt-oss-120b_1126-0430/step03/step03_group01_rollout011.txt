def adam_optimize(a0, max_iters, deadline):
    n = len(a0)
    # Initialize logits via log softmax
    eps = 1e-16
    z = logits_from_distribution(a0, eps)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1=0.9; beta2=0.999; eps_adam=1e-8
    best_a = a0.copy()
    best_score = cheap_score(a0)
    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        a = softmax(z)
        conv = conv_full(a) # full conv
        maxb = conv.max()
        cheap = 2.0 * n * maxb
        if cheap < best_score:
            best_score = cheap
            best_a = a.copy()
        # compute subgradient
        tol = 1e-12
        active_mask = np.abs(conv - maxb) <= tol
        active_idxs = np.where(active_mask)[0]
        # Use average subgradient:
        g = np.zeros_like(a)
        for k in active_idxs:
            lo = max(0, k - (n-1))
            hi = min(k, n-1)
            i = np.arange(lo, hi+1)
            g[i] += 2.0 * a[k - i]
        g /= len(active_idxs)
        mu = np.dot(g, a)
        grad_z = a * (g - mu)
        # Adam update
        m = beta1 * m + (1-beta1) * grad_z
        v = beta2 * v + (1-beta2) * (grad_z**2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = - (0.2 / (np.sqrt(it)+1.0)) * m_hat / (np.sqrt(v_hat) + eps_adam)
        # Might also clip step magnitude?
        z_new = z + step
        # clip logits to avoid overflow
        z = np.clip(z_new, -50, 50)
        # Optionally add occasional random perturbations
        if it % 1000 == 0 and np.random.rand() < 0.2:
            z += 0.01*np.random.randn(n)
            m *= 0.5
            v *= 0.5
    return best_a, best_score
```

def local_refine(a, deadline, max_iters=5000):
    n = a.shape[0]
    best_a = a.copy()
    best_score = cheap_score(a)
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        # pick two indices at random
        i, j = np.random.randint(n, size=2)
        if i==j: continue
        # choose delta proportion of a[i]
        δ = 0.01 * a[i] * (np.random.rand() - 0.5) # random +/- up to 1%?
        if a[i] - δ < 0:
            continue
        a_new = best_a.copy()
        a_new[i] -= δ
        a_new[j] += δ
        a_new = np.maximum(a_new, 0.0)
        a_new_sum = a_new.sum()
        if a_new_sum == 0:
            continue
        a_new /= a_new_sum
        score = cheap_score(a_new)

#!/usr/bin/env python
"""
A multi‑start mirror‑descent optimiser for the step‑function bound.

The algorithm
--------------
* 1️⃣  Generate many candidate probability vectors of different lengths
      using several analytic families (power law, exponential, beta,
      mixtures).  The vectors are normalised to sum to 1, so the
      evaluation metric 2·n·max(a*a) does not depend on the scale.
* 2️⃣  Keep the best few candidates for each length and improve them with
      Adam on the simplex (log‑softmax parametrisation).  The sub‑gradient
      of the active convolution entries is computed via an FFT; all
      active (maximal) indices are used jointly.
* 3️⃣  A tiny amount of random noise is injected periodically to escape
      plateaus.  The learning‑rate follows an inverse‑sqrt schedule.
* 4️⃣  After the time budget is exhausted the best candidate is finally
      evaluated with the supplied ``evaluate_sequence`` and returned.

The optimiser runs entirely with cheap FFT‑based evaluations, so thousands
of candidates can be tried inside the ~1000 s budget.  It improves on the
previous Adam‑only approach by (i) exploring many initial shapes and
lengths, (ii) using all active constraints for the sub‑gradient, and (iii)
performing aggressive restarts and noise injection.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max; returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it representation that yields ``a`` after ``softmax``."""
    return np.log(a + eps)


def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two ≥ 2·n‑1 (efficient size for a real FFT)."""
    return 1 << ((2 * n - 1).bit_length())


def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """
    Full self‑convolution a*a using a power‑of‑two FFT.
    Returns an array of length 2·n‑1.
    """
    n = a.shape[0]
    fft_len = _fft_len_for_n(n)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def _max_conv_and_active(a: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Return (max_value, bool_mask_of_active_indices) of the self‑convolution
    ``a*a``.  ``active`` is a mask of all indices attaining the maximum
    (within a tiny tolerance).
    """
    conv = _conv_full_fft(a)
    max_val = float(conv.max())
    # tolerance accounts for floating‑point noise
    tol = max_val * 1e-12 + 1e-14
    active = np.abs(conv - max_val) <= tol
    return max_val, active


def _subgradient_multi(a: np.ndarray, active_mask: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of ``max_k (a*a)[k]`` using *all* active convolution indices.
    For a single active offset k the gradient is
        g_i = 2·a_{k‑i}  for i in the overlap range.
    The result is the average over the active set.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    # positions of active convolution entries
    active_idxs = np.nonzero(active_mask)[0]
    if active_idxs.size == 0:
        return g
    for k in active_idxs:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i = np.arange(lo, hi + 1, dtype=np.int64)
        g[i] += 2.0 * a[k - i]
    g /= float(active_idxs.size)
    return g


def _cheap_score(a: np.ndarray) -> float:
    """
    Cheap proxy for ``evaluate_sequence``: 2·n·max(a*a) with ∑a = 1.
    The input may be un‑normalised; it is normalised internally.
    """
    s = a.sum()
    if s == 0.0:
        return float("inf")
    a = a / s
    n = a.shape[0]
    conv = _conv_full_fft(a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Families of analytic distributions (all are normalised to sum 1)
# ----------------------------------------------------------------------


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _exponential_distribution(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ x_i^{alpha-1} (1-x_i)^{beta-1}, x_i = i/(n-1)."""
    if n == 1:
        return np.array([1.0])
    x = np.linspace(0.0, 1.0, n)
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    # Replace NaNs caused by 0^{neg} with 0
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _mixture_distribution(n: int, w: float, base: np.ndarray, mix: np.ndarray) -> np.ndarray:
    """
    Convex combination w·mix + (1‑w)·base, then renormalised.
    Both ``base`` and ``mix`` must already be normalised.
    """
    a = (1.0 - w) * base + w * mix
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Optimisation core (Adam on the simplex)
# ----------------------------------------------------------------------


def _adam_optimize(
    a0: np.ndarray,
    max_iters: int,
    deadline: float,
    verbose: bool = False,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent/Adam on the probability simplex.
    Returns the best vector found (normalised) and its cheap score.
    """
    n = a0.shape[0]
    eps = 1e-16
    z = _logits_from_distribution(a0, eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    base_alpha = 0.3

    best_a = a0.copy()
    best_score = _cheap_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        a = _softmax(z)

        # cheap convolution, max value and active mask
        maxb, active_mask = _max_conv_and_active(a)

        cheap = 2.0 * n * maxb
        if cheap < best_score:
            best_score = cheap
            best_a = a.copy()
            if verbose:
                print(f"[Iter {it:07d}] new best cheap = {cheap:.6f}")

        # sub‑gradient (average over all active indices)
        g_a = _subgradient_multi(a, active_mask)

        # gradient w.r.t. logits (mirror descent on the simplex)
        mu = np.dot(g_a, a)
        grad_z = a * (g_a - mu)

        # Adam moment updates (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning‑rate schedule (inverse‑sqrt decay)
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)

        # step in log‑it space
        step = -alpha_t * m_hat / (np.sqrt(v_hat) + eps_adam)

        # simple back‑track: accept if we do not increase the cheap score
        accepted = False
        trial_step = step.copy()
        for _ in range(3):
            new_z = z + trial_step
            new_z = np.clip(new_z, -50.0, 50.0)
            new_a = _softmax(new_z)
            new_maxb, _ = _max_conv_and_active(new_a)
            new_score = 2.0 * n * new_maxb
            if new_score <= cheap:
                z = new_z
                accepted = True
                break
            trial_step *= 0.5
        if not accepted:
            # take the original step even if it does not improve;
            # this helps escape shallow plateaus
            z = np.clip(z + step, -50.0, 50.0)

        # occasional random perturbations (helps exploration)
        if it % 2500 == 0 and np.random.rand() < 0.3:
            delta = 0.02 * np.random.randn(n)
            z += delta
            m *= 0.5
            v *= 0.5

    return best_a, best_score
