"""
Propose a non‑negative step‑function (a probability vector)
that minimises the bound used by ``evaluate_sequence``.
The algorithm is a *projected Adam* optimiser on the exact
max‑convolution (no log‑sum‑exp smoothing).  The gradient of
the objective  O = 2·n·max(conv(a,a)) (with ∑a=1) is a sub‑gradient
obtained from the indices where the convolution attains its
maximum.  After each Adam update the vector is projected onto the
simplex {x≥0, ∑x=1}.  A few heuristics are added:

* start from a random simplex point (or from the global
  ``height_sequence_1`` if available);
* use an increasing problem size (default n≈6000, which gives a
  good trade‑off between resolution and speed);
* adapt the learning rate slowly; occasional Gaussian jitter
  helps escape plateaus;
* if no improvement is seen for many iterations we restart
  from the current best solution with a small random perturbation.

All heavy work (convolution and correlation) is performed with
FFT‑based O(n log n) routines, allowing many thousands of iterations
within the 1000 s budget.  The best sequence found is returned
as a plain list of floats and its exact score is printed via
``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x ≥ 0, sum(x)=1}
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:               # all entries are <=0
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute the (non‑reversed) correlation
        g_i = Σ_t w[i+t] * a[t]   for i=0,…,n‑1
    using FFT.  ``w`` has length 2n‑1, ``a`` has length n.
    """
    n = a.shape[0]
    m = w.shape[0]                     # should be 2n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective used for internal comparison: 2·n·max(conv)."""
    n = a.size
    conv = _conv_self_fft(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that minimises the bound in
    ``evaluate_sequence``.  Returns the best sequence (list of floats)
    found within the time budget.
    """
    np.random.seed(seed)
    # small safety margin so we do not overrun the allotted time
    deadline = time.time() + budget_s - 0.5

    # --------------------------------------------------------------
    # Initialise the vector a (sum = 1).  If a previous solution exists,
    # reuse it; otherwise start from a random point of length ~6000.
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=float)
        a = a / a.sum()
        n = a.shape[0]
    else:
        n = 6000                     # problem size – a good compromise
        a = np.random.rand(n)
        a = _project_to_simplex(a)

    # Adam state
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    t_iter = 0

    # Track the best solution seen so far
    best_seq = a.copy()
    best_score = _objective(best_seq)
    print(f"[init] n={n}, objective={best_score:.8f}")

    # Hyper‑parameters
    lr_initial = 0.5              # starting learning rate
    lr_min = 0.03                 # lower bound for the learning rate
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # Restart control
    last_improve_iter = 0
    RESTART_AFTER = 25000         # iterations without improvement

    # --------------------------------------------------------------
    while time.time() < deadline:
        t_iter += 1

        # ---- current objective -------------------------------------------------
        conv = _conv_self_fft(a)
        M = conv.max()
        cur_score = 2.0 * n * M

        # ---- keep the best ----------------------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            last_improve_iter = t_iter
            print(f"[iter {t_iter:06d}] new best O={best_score:.6f}")

        # ---- sub‑gradient of max(conv) ----------------------------------------
        # weight vector w: uniform over all indices attaining the max
        max_mask = (conv == M)
        w = max_mask.astype(float)
        nz = w.sum()
        if nz > 0:
            w /= nz
        else:                       # should never happen
            w = np.ones_like(conv) / conv.size

        # gradient of (2·n·max(conv)) w.r.t. a
        g = 2.0 * _correlation_fft(w, a)

        # ---- Adam update -------------------------------------------------------
        lr = max(lr_min, lr_initial * (0.98 ** (t_iter // 2000)))
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = _project_to_simplex(a_new)

        # ---- occasional jitter to escape plateaus -----------------------------
        if t_iter % 3000 == 0:
            jitter = np.random.randn(n) * 0.001
            a_new = _project_to_simplex(a_new + jitter)

        a = a_new

        # ---- restart if we are stuck -------------------------------------------
        if t_iter - last_improve_iter > RESTART_AFTER:
            # re‑initialise close to the best solution
            a = best_seq + np.random.randn(n) * 0.002
            a = _project_to_simplex(a)
            m.fill(0.0)
            v.fill(0.0)
            last_improve_iter = t_iter
            print(f"[restart] iter {t_iter}, re‑initialised from best.")

    # ----------------------------------------------------------------------
    # Final evaluation with the provided ``evaluate_sequence`` routine.
    # The routine is scale‑invariant, so we simply feed the best simplex
    # vector found.
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_seq.tolist()
