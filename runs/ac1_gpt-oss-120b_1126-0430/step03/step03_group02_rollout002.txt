"""
Propose a non‑negative step‑function (list of heights) that minimises

    O = 2·n·max(conv(a,a))   (scale‑invariant, sum(a) can be enforced =1)

The algorithm combines three ideas:

1. **Smooth approximation of the max** – use a soft‑max (log‑sum‑exp)
   with an increasing temperature β to obtain a differentiable surrogate.

2. **Exponentiated‑gradient (mirror descent)** – update in log‑space,
   which automatically respects non‑negativity and the simplex constraint.

3. **Multiple diverse starts** – 1/√ law, a few power‑law variants,
   a random start and, if available, the previous best construction.

All heavy operations (convolutions, gradients) are performed with FFTs,
so each iteration costs only O(n log n).  The routine stops after the
budget (default 1000 s) and returns the best sequence found.
"""

import time
import numpy as np

# ------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
def _next_pow_two(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


# ------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using FFT (full length 2·n‑1)."""
    n = a.size
    size = _next_pow_two(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return conv


# ------------------------------------------------------------
def _conv_fft_general(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT (full length)."""
    n = x.size
    m = y.size
    size = _next_pow_two(n + m - 1)
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)[: n + m - 1]
    return conv


# ------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Objective 2·n·max(conv(a,a)) for a already normalised (Σa=1)."""
    n = a.size
    conv = _conv_fft(a)
    return 2.0 * n * np.max(conv)


# ------------------------------------------------------------
def _smooth_max_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Return soft‑max weights w_k = exp(beta·conv_k)/∑exp(beta·conv)
    (numerically stable by subtracting the maximum).
    """
    max_val = conv.max()
    # shift to avoid overflow/underflow
    shifted = beta * (conv - max_val)
    # clipping protects against overflow for very large beta
    # (np.exp of very negative numbers will under‑flow to 0, which is fine)
    exp_vals = np.exp(shifted)
    sum_exp = exp_vals.sum()
    if sum_exp == 0.0 or not np.isfinite(sum_exp):
        # fallback to plain hard max (single one‑hot)
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w
    return exp_vals / sum_exp


# ------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the gradient of the smoothed max w.r.t. a:
        g_i = 2 * Σ_j a_j * w_{i+j}
    using a single FFT‑based convolution.
    """
    n = a.size
    a_rev = a[::-1]                     # reversed a
    conv2 = _conv_fft_general(a_rev, w)  # length = 3·n‑2
    # slice the relevant n entries (indices n‑1 … 2·n‑2)
    g = 2.0 * conv2[n - 1 : n - 1 + n]
    return g


# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the evaluation
    bound.  Returns the sequence as a plain list of floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Choose a length.  If a previous construction exists we keep it,
    # otherwise use a power‑of‑two size that works well with FFT.
    if "height_sequence_1" in globals():
        a_prev = np.asarray(height_sequence_1, dtype=float)
        n = a_prev.size
        a_prev = a_prev / a_prev.sum()
    else:
        a_prev = None
        n = 8000                     # default length (power‑of‑two friendly)

    # ------------------------------------------------------------------
    # Build a small diverse pool of starting points.
    starts = []

    if a_prev is not None:
        starts.append(a_prev)

    # 1/√ law (theoretically good)
    idx = np.arange(1, n + 1, dtype=float)
    a_sqrt = 1.0 / np.sqrt(idx)
    a_sqrt /= a_sqrt.sum()
    starts.append(a_sqrt)

    # a few other power‑law exponents
    for p in (0.4, 0.6):
        a_power = 1.0 / np.power(idx + 0.5, p)
        a_power /= a_power.sum()
        starts.append(a_power)

    # random start – adds diversity
    a_rand = np.random.rand(n)
    a_rand = _project_to_simplex(a_rand)
    starts.append(a_rand)

    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    # Hyper‑parameters (tuned empirically)
    iters_per_start = 20000           # enough to converge for each start
    eta0 = 0.6                        # initial learning rate for EG
    beta0 = 1.0                       # start smoothing parameter
    beta_factor = 1.02                # slow increase of β each iteration
    beta_max = 2000.0                 # cap to avoid numerical issues

    for s_idx, a_start in enumerate(starts):
        if time.time() >= deadline:
            break
        a = a_start.copy()
        # evaluate the initial point (objective assumes Σa=1)
        cur_obj = _objective(a)
        print(f"[start {s_idx}] initial O={cur_obj:.6f}")

        beta = beta0
        no_improve = 0
        patience = max(10, iters_per_start // 30)   # for occasional jitter

        for t in range(iters_per_start):
            if time.time() >= deadline:
                break

            # ---------- convolution and current objective ----------
            conv = _conv_fft(a)                     # length 2·n‑1
            max_val = conv.max()
            cur_obj = 2.0 * n * max_val

            # update global best if improved
            if cur_obj < best_score - 1e-12:
                best_score = cur_obj
                best_seq = a.copy()
                print(f"[global] iter {t:5d} start {s_idx} new best O={best_score:.6f}")

            # ---------- smoothed max weighting ----------
            w = _smooth_max_weights(conv, beta)

            # ---------- gradient computation ----------
            g = _gradient_from_weights(a, w)

            # ---------- learning‑rate schedule ----------
            eta = eta0 * (1.0 - t / iters_per_start)   # linear decay

            # ---------- exponentiated‑gradient update ----------
            log_a = np.log(a + 1e-15) - eta * g
            log_a -= np.max(log_a)                     # stabilise exponentials
            a = np.exp(log_a)
            a /= a.sum()                               # re‑project to simplex

            # ---------- schedule for β ----------
            beta = min(beta * beta_factor, beta_max)

            # ---------- occasional jitter to escape plateaus ----------
            if cur_obj >= best_score - 1e-12:
                no_improve += 1
                if no_improve >= patience:
                    jitter = np.random.randn(n) * 1e-4
                    a = best_seq + jitter
                    a = np.clip(a, 0.0, None)
                    a = _project_to_simplex(a)
                    no_improve = 0
            else:
                no_improve = 0

    # ------------------------------------------------------------------
    # Fallback in the unlikely event that no start produced a candidate.
    if best_seq is None:
        best_seq = a_sqrt.copy()
        best_score = _objective(best_seq)

    # Final exact evaluation (scale‑invariant, sum is forced to 1)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:       # pragma: no cover
        print("[final] evaluate_sequence call failed:", e)
        final_score = None

    return best_seq.tolist()
