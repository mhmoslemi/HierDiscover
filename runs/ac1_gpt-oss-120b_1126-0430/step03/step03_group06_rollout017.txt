"""
Propose a sequence of non‑negative heights that minimizes ``evaluate_sequence``.
The algorithm works on a simplex with a fixed ℓ₁‑norm S = √(2·n) (the
normalisation used by the reference implementation).  The objective to be
minimised is

    F(a) = max_k (a * a)[k]                      (a has ℓ₁‑norm S)

because 2·n·max/(S²) = max when S = √(2·n).  To obtain a better solution than
the simple sub‑gradient method we use a *smooth* surrogate for the max
(the log‑sum‑exp) and perform Adam‑style mirror descent on the simplex.
A softmax weight vector w_k = softmax_λ((a*a)_k) gives a differentiable
approximation of the active set; the gradient of the surrogate is

    ∇F_λ(a)_i = 2· Σ_k w_k a_{k‑i}
               = 2· (w ⊛ a_rev)[i]  (⊛ = linear convolution,
                                      a_rev = a reversed).

Both the self‑convolution and the gradient are computed by a real FFT,
so each iteration costs only O(n log n).  λ is started small (≈0.5) so that
many entries of the convolution influence the update, and is slowly increased
towards a large value so that the surrogate approaches the true max.
The iterate is projected onto the simplex {x≥0, Σx=S} after each update.
Periodic random perturbations help escape plateaus.  Several cheap random
restarts are performed if time permits.

The result is returned as a plain Python list; ``evaluate_sequence`` will
automatically apply the necessary clipping and scaling.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based *linear* convolution (real inputs)
# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of real vectors a and b using a real FFT."""
    n = a.shape[0]
    m = b.shape[0]
    size = 1 << ((n + m - 1).bit_length())        # next power of two
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    conv = np.fft.irfft(fa * fb, size)
    return conv[: n + m - 1]                     # length = n+m‑1


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex of non‑negative vectors with L1‑norm s."""
    n = v.size
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the uniform vector
        return np.full_like(v, s / n)
    rho = ind[cond][-1]          # last index where condition holds
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Catalan‑type coefficients, normalized to ℓ₁‑norm S = √(2n)
# ----------------------------------------------------------------------
def _catalan_normalised(n: int) -> np.ndarray:
    """First n terms a_i = C(2i,i)/4**i, scaled so that Σa = √(2n)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1) / (2.0 * i)   # (2i‑1)/(2i)
    # sum of raw Catalan terms ≈ 2·√(n/π)
    S_target = np.sqrt(2.0 * n)
    a = a / a.sum() * S_target
    return a


# ----------------------------------------------------------------------
# Max‑convolution (the exact value returned by ``evaluate_sequence`` when
# the ℓ₁‑norm is √(2·n)).
# ----------------------------------------------------------------------
def _max_convolution(a: np.ndarray) -> float:
    """Return max_k (a*a)[k] for a vector a."""
    conv = _fft_convolve(a, a)
    return float(conv.max())


# ----------------------------------------------------------------------
# Core optimiser – Adam on a smoothed max (log‑sum‑exp)
# ----------------------------------------------------------------------
def _optimize_smooth(
    a0: np.ndarray,
    deadline: float,
    max_iters: int = 100_000,
    verbose: bool = True,
) -> tuple[np.ndarray, float]:
    """
    Adam‑style optimisation of the smoothed max‑convolution.
    Returns the best vector found and its exact max‑convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # smoothing parameter λ (log‑sum‑exp temperature)
    lam = 0.5               # start very soft
    lam_inc = 1.001         # multiplicative increase every few steps

    best = a.copy()
    best_val = _max_convolution(best)

    t = 0
    while time.time() < deadline and t < max_iters:
        # 1) self‑convolution (full length 2n‑1)
        conv = _fft_convolve(a, a)
        max_conv = float(conv.max())

        # keep the best ever seen
        if max_conv < best_val:
            best_val = max_conv
            best = a.copy()
            if verbose:
                print(f"[improve] iter {t:06d}  max_conv {best_val:.9f}  λ {lam:.3f}")

        # 2) softmax weights for the smoothed max
        #    w_k = exp(λ·(conv_k‑max)) / Σ exp(...)
        shifted = conv - max_conv                # max shift → stability
        exp_term = np.exp(lam * shifted)
        w = exp_term / exp_term.sum()            # shape (2n‑1,)

        # 3) gradient of the smoothed objective:
        #    grad_i = 2· Σ_k w_k a_{k‑i}
        #    This is a convolution of w with the reversed a.
        grad_full = _fft_convolve(w, a[::-1])    # length 3n‑2
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]  # central n entries

        # 4) Adam update (additive)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (t + 1))
        v_hat = v / (1.0 - beta2 ** (t + 1))

        # learning‑rate schedule (inverse‑sqrt decay)
        lr = 0.2 / np.sqrt(t + 1)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # 5) enforce feasibility (non‑negative, ℓ₁‑norm = √(2n))
        a = np.maximum(a, 0.0)
        a = _project_onto_simplex(a, S)

        # occasional perturbation to jump out of plateaus
        if t % 4000 == 0 and t > 0:
            perturb = np.random.normal(scale=0.01 * S, size=n)
            a = np.maximum(a + perturb, 0.0)
            a = _project_onto_simplex(a, S)

        # slowly tighten the smoothing (increase λ)
        if t % 1000 == 0 and t > 0:
            lam = min(lam * lam_inc, 30.0)   # never let λ explode

        t += 1

    return best, best_val


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Return a list of non‑negative heights (as Python floats) that minimises
    ``evaluate_sequence`` within the given time budget.
    """
    np.random.seed(seed)
    # leave a couple of seconds for the wrapper that will call us
    deadline = time.time() + budget_s - 2.0

    # problem size – 20 000 is a good trade‑off between quality and speed
    n = 20_000
    S = np.sqrt(2.0 * n)

    # ---------- 1) start from the well‑known Catalan distribution ----------
    a0 = _catalan_normalised(n)

    # ---------- 2) main optimisation loop (smooth max + Adam) -------------
    best_seq, best_val = _optimize_smooth(a0, deadline, max_iters=150_000, verbose=True)

    # ---------- 3) cheap random restarts if time is left -----------------
    while time.time() < deadline - 5.0:
        # Dirichlet‑type random start on the simplex (α = 0.5 gives a bit of
        # front‑loading, which is often useful)
        rnd = np.random.gamma(shape=0.5, scale=1.0, size=n)
        a_rand = rnd / rnd.sum() * S
        a_opt, val = _optimize_smooth(a_rand, deadline, max_iters=30_000, verbose=False)
        if val < best_val:
            best_val = val
            best_seq = a_opt.copy()
            print(f"[restart] new best max_conv {best_val:.9f}")

    # final sanity check (the harness will also call evaluate_sequence)
    final_max = _max_convolution(best_seq)
    print(f"[final] best max convolution ≈ {final_max:.9f} (objective = {final_max:.9f}) n={n}")

    # return as a plain Python list
    return best_seq.tolist()
