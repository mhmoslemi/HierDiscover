"""
Propose a step‑function with a low evaluation value.

Algorithm
---------
1.  Work on the probability simplex: a = softmax(z).  This guarantees
    non‑negative entries that sum to 1, which makes the objective
    scale‑invariant.
2.  At each iteration we minimise a *smooth* approximation of
    max_k (a*a)[k] :

           L(a) = (1/β)·log Σ_k exp(β·(a*a)[k])

    The gradient of L is a convex combination of the gradients of all
    convolution entries, not only the single active one.  This gives a
    more informative direction than a sub‑gradient of the true max.
3.  The gradient of L with respect to a can be written as a convolution:

        ∇_a L = 2·(reverse(a) * w)[n-1:n-1+n]

    where w are the normalised soft‑max weights
    w_k ∝ exp(β·((a*a)[k]‑max_conv)).  The convolution is evaluated
    with a real FFT (O(n log n)).
4.  Mirror‑descent on the logits (Adam on the KL‑geometry) is used to
    update z.  An inexpensive back‑tracking line‑search guarantees a
    non‑increasing cheap objective (2·n·max_conv).
5.  The support size is gradually enlarged (up to `n_max`) by
    linear interpolation of the best distribution found so far.
6.  The best distribution (according to the cheap proxy) is finally
    evaluated with the provided `evaluate_sequence` and returned.

The method differs from the previous approach by employing a smooth
max and an exact convolution‑based gradient, which typically reduces
the peak of the self‑convolution more efficiently.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (must be top‑level – no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it representation that yields `a` after soft‑max."""
    return np.log(a + eps)


def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two ≥ 2·n‑1 (efficient size for a real FFT)."""
    return 1 << ((2 * n - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real arrays using a real FFT."""
    n = x.size + y.size - 1
    fft_len = 1 << ((n - 1).bit_length())
    fx = np.fft.rfft(x, fft_len)
    fy = np.fft.rfft(y, fft_len)
    conv = np.fft.irfft(fx * fy, fft_len)[:n]
    return conv


def _conv_max_fft(a: np.ndarray, fft_len: int) -> tuple[float, int]:
    """
    Return (max value, argmax) of the self‑convolution a*a
    using a pre‑computed FFT length.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.size - 1]
    idx = int(conv.argmax())
    return float(conv.max()), idx


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    total = new_a.sum()
    if total == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= total
    return new_a


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """Normalised power‑law a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a

# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) beats the target bound.
    The routine respects the given time budget (≈1000 s) and uses a smooth‑max
    gradient descent on the probability simplex with FFT‑based convolution.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # safety margin

    # ---------------------------------------------------------------
    # 0. Choose a working support size.
    # ---------------------------------------------------------------
    try:
        # If a previous construction is available, start from it.
        init_global = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        n = max(2000, init_global.shape[0])
        use_global = True
        print(f"Warm start from global variable, length = {n}")
    except Exception:
        init_global = None
        n = 15000                               # start size – can be enlarged later
        use_global = False
        print(f"No global start – using random initialisation, length = {n}")

    n_max = 40000                               # absolute ceiling on the support size

    # ---------------------------------------------------------------
    # 1. Initial distribution on the simplex.
    # ---------------------------------------------------------------
    if use_global and init_global is not None:
        a = np.maximum(init_global, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / a.shape[0])
        else:
            a /= s
        print("Initial distribution taken from global variable.")
    else:
        # Mix a random start with a power‑law seed – both are cheap to evaluate.
        rand = np.random.rand(n)
        rand /= rand.sum()
        plaw = _power_law_distribution(n, 0.62)
        a = 0.5 * rand + 0.5 * plaw
        a = a / a.sum()
        print("Initial distribution = mixture of uniform random and power law.")

    # ---------------------------------------------------------------
    # 2. Log‑it parametrisation + Adam state.
    # ---------------------------------------------------------------
    eps = 1e-16
    z = _logits_from_distribution(a, eps)

    m = np.zeros_like(z)          # 1st moment
    v = np.zeros_like(z)          # 2nd moment
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    base_alpha = 0.25             # base step‑size (will be scaled)

    # ---------------------------------------------------------------
    # 3. Main optimisation loop.
    # ---------------------------------------------------------------
    best_seq = a.copy()
    # cheap proxy = 2·n·max_conv (since sum(a)=1)
    fft_len = _fft_len_for_n(n)
    max_conv, _ = _conv_max_fft(a, fft_len)
    best_score = 2.0 * n * max_conv
    it = 0
    last_improve_it = 0
    report_every = 2000

    while time.time() < deadline and it < 1_000_000:
        it += 1

        # ----- probability vector from logits -----
        a = _softmax(z)

        # ----- cheap convolution -----
        fft_len = _fft_len_for_n(n)
        conv = _conv_fft(a, a)                # length 2n‑1
        max_conv = conv.max()
        cheap_score = 2.0 * n * max_conv

        # ----- keep the best ever found -----
        if cheap_score < best_score:
            best_score = cheap_score
            best_seq = a.copy()
            last_improve_it = it
            if it % report_every == 0:
                print(f"[Iter {it:07d}] NEW BEST cheap = {cheap_score:.6f}")

        # ----- smooth‑max weights -----
        # scale β so that β·max_conv ≈ cur_beta_target (≈15‑30) → smooth but
        # increasingly sharp as optimisation proceeds.
        cur_beta_target = 15.0 + it * 5e-5   # slowly grow, stays modest
        beta = cur_beta_target / (max_conv + 1e-20)

        # w_k ∝ exp(β·(conv_k – max_conv))  ≤ 1, max at 0
        w_raw = np.exp(beta * (conv - max_conv))
        w = w_raw / w_raw.sum()            # normalised, length 2n‑1

        # ----- gradient of smooth max w.r.t. a -----
        a_rev = a[::-1]
        # convolution a_rev * w (full length = n + (2n‑1) – 1 = 3n‑2)
        grad_full = _conv_fft(a_rev, w)    # length 3n‑2
        # extract the slice that corresponds to ∂L/∂a_i
        g_a = 2.0 * grad_full[n - 1 : n - 1 + n]

        # ----- mirror‑descent (Adam) on logits -----
        mu = np.dot(g_a, a)                 # <∇L , a>
        grad_z = a * (g_a - mu)             # ∇_z L  (KL‑geometry)

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

        it_f = float(it)
        m_hat = m / (1.0 - beta1 ** it_f)
        v_hat = v / (1.0 - beta2 ** it_f)

        step_dir = - m_hat / (np.sqrt(v_hat) + eps_adam)

        # learning‑rate schedule: inverse‑sqrt decay
        alpha_t = base_alpha / (math.sqrt(it_f) + 1.0)

        # ----- back‑tracking line search (max 3 halvings) -----
        accepted = False
        trial_alpha = alpha_t
        for _ in range(3):
            z_new = z + trial_alpha * step_dir
            z_new = np.clip(z_new, -50.0, 50.0)      # keep exp stable
            a_new = _softmax(z_new)
            conv_new = _conv_fft(a_new, a_new)
            new_score = 2.0 * n * conv_new.max()
            if new_score <= cheap_score:
                z = z_new
                accepted = True
                break
            trial_alpha *= 0.5

        if not accepted:
            # take a modest step anyway to keep moving
            z = z + alpha_t * step_dir
            z = np.clip(z, -50.0, 50.0)

        # ----- random perturbations (helps escape plateaus) -----
        if it % 5000 == 0 and np.random.rand() < 0.2:
            perturb = 0.02 * np.random.randn(n)
            z += perturb
            m *= 0.5
            v *= 0.5

        # ----- progressive support enlargement -----
        if (it % 20000 == 0) and (n < n_max) and (time.time() + 30 < deadline):
            new_n = min(n_max, n + 5000)
            print(f"[Iter {it:07d}] Expanding support: {n} → {new_n}")
            a = _interpolate_distribution(best_seq, new_n)
            z = _logits_from_distribution(a, eps)
            m = np.zeros_like(z)
            v = np.zeros_like(z)
            n = new_n
            # reset counters for the enlarged problem
            it = 0
            last_improve_it = 0
            fft_len = _fft_len_for_n(n)
            max_conv, _ = _conv_max_fft(a, fft_len)
            best_score = 2.0 * n * max_conv
            best_seq = a.copy()
            continue  # next iteration works with the larger size

        # ----- early termination if stagnant (after max size reached) -----
        if n >= n_max and it - last_improve_it > 250_000:
            print(f"No improvement for {it - last_improve_it} iters – stopping early.")
            break

    # ---------------------------------------------------------------
    # 4. Final exact evaluation (uses the provided evaluate_sequence)
    # ---------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact evaluation score = {final_score:.6f}")
    except Exception as exc:   # defensive – should never happen
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # ---------------------------------------------------------------
    # 5. Return the best sequence found.
    # ---------------------------------------------------------------
    return final_seq
