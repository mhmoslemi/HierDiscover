def _smooth_gradient(a, beta):
    conv = _conv_fft(a)
    # compute softmax weights
    max_val = conv.max()
    exp_vals = np.exp(beta * (conv - max_val))
    sum_exp = exp_vals.sum()
    w = exp_vals / sum_exp
    # compute gradient
    w_rev = w[::-1]
    # conv of a and w_rev
    sz = 1 << ((len(a)+len(w_rev)-2).bit_length())
    A = np.fft.fft(a, sz)
    W = np.fft.fft(w_rev, sz)
    c = np.fft.ifft(A * W).real
    L = len(w)  # 2n-1
    # gradient for indices i=0..n-1: g_i = 2 * c[L - 1 - i]
    g = 2.0 * c[(L - 1) - np.arange(len(a))]
    return g

Potential numeric errors: due to round-off may exceed small negativity; ensure g is non-negative.

def _exp_update_smooth(a, g, eta):
    log_a = np.log(a + 1e-15) - eta * g
    # ensure numeric stability
    log_a -= np.max(log_a)
    a_new = np.exp(log_a)
    a_new /= a_new.sum()

    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)
    n = a.size

    for t in range(max_iters):

        beta = beta0 + (beta_max - beta0) * (t / max_iters)
        g = _smooth_gradient(a, beta)

"""
Propose a low‑upper‑bound step function.

The bound under evaluation is
    O(a) = 2·n·max(conv(a,a)) / (sum(a)^2)      (n = len(a))

O is scale‑invariant, so we work on the simplex Σa=1.
The algorithm combines three ideas:

1.  A cheap parametric search over power‑law shapes a_i ∝ (i+1+c)^(-α)
    (α∈(0.4,0.9), c≥0) to obtain good starting points.
2.  A smoothed‑gradient descent on the *soft* max of the convolution.
    The soft‑max uses a temperature β that is increased during the run,
    turning the smooth objective into the true max.
    The gradient is obtained analytically via a second convolution, all
    performed with FFTs (O(n log n)).
3.  A classic exponentiated‑gradient sub‑gradient loop (as in the
    reference code) is used as a fall‑back and for extra jitter.

Both loops run on the probability simplex by multiplicative updates
followed by renormalisation.  The best vector seen (according to the
exact bound computed by ``evaluate_sequence``) is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a power‑of‑two FFT."""
    n = a.size
    size = 1 << ((2 * n - 2).bit_length())  # next power of two ≥2n‑1
    A = np.fft.fft(a, n=size)
    conv = np.fft.ifft(A * A).real[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact bound 2·n·max(conv(a,a)) for a already normalized (Σa=1)."""
    n = a.size
    conv = _conv_fft(a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smoothed max M_β(a) = (1/β)·log Σ exp(β·conv_k) .
    Returns ∇M_β (the direction of *increase*; we move opposite).
    """
    n = a.size
    conv = _conv_fft(a)

    # soft‑max weights
    max_val = conv.max()
    exp_vals = np.exp(beta * (conv - max_val))   # prevent overflow
    w = exp_vals / exp_vals.sum()                # Σ w = 1

    # gradient = 2·(a * w_rev) taken at appropriate indices
    # w_rev is w reversed; we need c = a ⊛ w_rev (linear convolution)
    w_rev = w[::-1]
    size = 1 << ((n + w_rev.size - 2).bit_length())
    A = np.fft.fft(a, n=size)
    B = np.fft.fft(w_rev, n=size)
    c = np.fft.ifft(A * B).real

    L = w.size                      # = 2·n‑1
    idx = L - 1 - np.arange(n)      # positions that give Σ a_j w_{i+j}
    g = 2.0 * c[idx]
    return g


# ----------------------------------------------------------------------
def _exp_update(a: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:
    """
    Multiplicative (exponentiated‑gradient) update on the simplex.
    The gradient points towards *increase* of the objective; we step
    opposite to minimise.
    """
    log_a = np.log(a + 1e-15) - eta * grad
    log_a -= np.max(log_a)                 # keep numbers moderate
    a_new = np.exp(log_a)
    a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) for use with plain EG descent.
    """
    n = a.size
    conv = _conv_fft(a)
    max_val = conv.max()
    ks = np.where(np.abs(conv - max_val) < 1e-14)[0]
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size > 0:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
def _smooth_optimize(a0: np.ndarray,
                     max_iters: int,
                     eta0: float,
                     beta0: float,
                     beta_max: float,
                     deadline: float) -> tuple[np.ndarray, float]:
    """
    Smoothed‑gradient descent with a temperature schedule.
    Returns the best vector found and its exact objective value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)

    n = a.size
    patience = max_iters // 20
    no_improve = 0

    for t in range(max_iters):
        if time.time() >= deadline:
            break

        beta = beta0 + (beta_max - beta0) * (t / max_iters)
        g = _smooth_gradient(a, beta)
        eta = eta0 * (1.0 - t / max_iters)

        a = _exp_update(a, g, eta)

        cur = _objective(a)
        if cur < best_score - 1e-12:
            best_score = cur
            best_a = a.copy()
            no_improve = 0
            print(f"[smooth_opt] iter {t:5d}  best O={best_score:.6f}")
        else:
            no_improve += 1
            if no_improve >= patience:
                jitter = np.random.randn(n) * 1e-4
                a = best_a + jitter
                a = np.clip(a, 0.0, None)
                a = _project_to_simplex(a)
                no_improve = 0

    return best_a, best_score


# ----------------------------------------------------------------------
def _plain_optimize(a0: np.ndarray,
                    max_iters: int,
                    eta0: float,
                    deadline: float) -> tuple[np.ndarray, float]:
    """
    Classic exponentiated‑gradient descent on the true max‑operator.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)

    n = a.size
    patience = max_iters // 20
    no_improve = 0

    for t in range(max_iters):
        if time.time() >= deadline:
            break

        eta = eta0 * (1.0 - t / max_iters)
        a = _exp_update(a, g, eta)

        cur = _objective(a)
        if cur < best_score - 1e-12:
            best_score = cur
            best_a = a.copy()
            no_improve = 0

        else:

                jitter = np.random.randn(n) * 1e-4
                a = best_a + jitter
                a = np.clip(a, 0.0, None)
                a = _project_to_simplex(a)
                no_improve = 0

    return best_a, best_score


# ----------------------------------------------------------------------
def _powerlaw_shape(alpha: float, c: float, n: int) -> np.ndarray:
    """Generate a_i ∝ (i+1+c)^{-α} and normalise to sum 1."""
    idx = np.arange(1, n + 1, dtype=float)
    a = (idx + c) ** (-alpha)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the bound.
    Returns the sequence as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # Choose length n (prefer a power of two for FFT speed)
    if 'height_sequence_1' in globals():
        prev = np.asarray(height_sequence_1, dtype=float)
        n = prev.size
        # we will start from the *normalised* version of the previous best
        a_prev = prev / prev.sum()
    else:
        n = 4096                     # default length (power‑of‑two)
        a_prev = None

    print(f"[info] Using length n={n}")

    # --------------------------------------------------------------
    # Build a small pool of promising starts
    starts = []

    if a_prev is not None:
        starts.append(a_prev)

    # classic 1/√ law
    a_sqrt = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=float))
    a_sqrt /= a_sqrt.sum()
    starts.append(a_sqrt)

    # random simplex point
    a_rand = np.random.rand(n)
    a_rand = _project_to_simplex(a_rand)
    starts.append(a_rand)

    # --------------------------------------------------------------
    # Parametric power‑law search (cheap coarse scan)
    best_param_score = float("inf")
    best_param_vec = None

    param_trials = 500
    for _ in range(param_trials):
        if time.time() >= deadline:
            break
        alpha = np.random.uniform(0.45, 0.85)
        c = np.random.exponential(scale=5.0)   # bias towards small offsets
        a_candidate = _powerlaw_shape(alpha, c, n)
        score = _objective(a_candidate)
        if score < best_param_score:
            best_param_score = score
            best_param_vec = a_candidate.copy()
            print(f"[param] new best O={best_param_score:.6f} (α={alpha:.4f},c={c:.4f})")
    if best_param_vec is not None:
        starts.append(best_param_vec)

    # --------------------------------------------------------------
    # Optimisation from each start
    best_seq = None
    best_score = float("inf")

    # parameters for the optimisers
    smooth_iters = 12000
    plain_iters = 12000
    eta0 = 0.45
    beta0 = 5.0
    beta_max = 120.0  # final temperature (more aggressive sharpening)

    for idx, a_start in enumerate(starts):
        if time.time() >= deadline:
            break

        init_score = _objective(a_start)
        print(f"[start {idx}] initial O={init_score:.6f}")

        # --- first a smoothed run -------------------------------------------------
        a_opt, score_opt = _smooth_optimize(a_start.copy(),
                                            max_iters=smooth_iters,
                                            eta0=eta0,
                                            beta0=beta0,
                                            beta_max=beta_max,
                                            deadline=deadline)
        print(f"[smooth result {idx}] O={score_opt:.6f}")

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()

        # --- then a plain EG run (helps escape possible smoothing artefacts) -------
        if time.time() >= deadline:
            break
        a_opt2, score_opt2 = _plain_optimize(best_seq.copy(),
                                             max_iters=plain_iters,
                                             eta0=eta0 * 0.6,
                                             deadline=deadline)
        print(f"[plain result {idx}] O={score_opt2:.6f}")

        if score_opt2 < best_score:
            best_score = score_opt2
            best_seq = a_opt2.copy()

    # --------------------------------------------------------------
    # Fallback: if nothing was found, use a simple 1/√ shape
    if best_seq is None:
        print("[fallback] using 1/√ law")
        best_seq = a_sqrt.copy()
        best_score = _objective(best_seq)

    # --------------------------------------------------------------
    # Final exact evaluation with the supplied function (just for reporting)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence failed:", exc)
        final_score = None

    # Store the best sequence globally for possible reuse by later runs
    global height_sequence_1
    height_sequence_1 = best_seq.tolist()

    return best_seq.tolist()
