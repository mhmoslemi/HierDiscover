"""
Propose a non‑negative step‑function that minimizes the bound

    O = 2·n·max(conv(a,a)) / (sum(a)**2)

The implementation uses a multi‑restart Adam optimiser with a soft‑max
approximation of the max‑convolution.  Each iterate is projected onto the
probability simplex (non‑negative, sum = 1) by a cheap scaling step, the
vector is kept symmetric (a[i]=a[n‑1‑i]) and occasional random jitter is
added to escape plateaus.  Restarts are performed from the current best
solution as well as from random perturbations and from interpolated longer
versions of the best solution, allowing the search to explore many
different lengths.  The algorithm respects a 1000 s time budget, prints
progress, and finally returns the best sequence found (as a plain list of
floats) which can be fed to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Fast projection onto the simplex {x ≥ 0, sum(x)=1} by
    clipping negatives and renormalising.  This is not the Euclidean
    projection but is sufficient and O(n) for our optimisation loop.
    """
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 0.0:
        v /= s
    else:
        v.fill(1.0 / v.size)
    return v


# ----------------------------------------------------------------------
# Convolution / correlation via FFT (real‑valued)
# ----------------------------------------------------------------------
def _conv_and_fft(a: np.ndarray):
    """
    Compute the full self‑convolution a*a together with the rFFT of a.
    Returns (conv, af, L) where conv is the linear convolution of length
    2·n‑1, af is the rFFT of a with length L, and L is the FFT length used.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv, af, L


def _correlation_fft(w: np.ndarray, af: np.ndarray, L: int, n: int) -> np.ndarray:
    """
    Compute the correlation g_i = Σ_t w[i+t]·a[t] for i=0,…,n‑1
    using the previously computed FFT of a (af) and a fresh FFT of w.
    """
    wf = np.fft.rfft(w, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


# ----------------------------------------------------------------------
# Exact objective (used for bookkeeping)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact bound O = 2·n·max(conv(a,a)) / (sum(a)**2)."""
    n = a.size
    conv, _, _ = _conv_and_fft(a)
    return 2.0 * n * conv.max() / (a.sum() ** 2)

        # Choose a new length and initialise a candidate.
        # --------------------------------------------------------------
        if restart_counter % 3 == 0 and best_seq.size < 12000:
            # Occasionally try a longer sequence by interpolating the best.
            factor = 1.3
            new_n = max(2, int(best_seq.size * factor))
            x_old = np.linspace(0.0, 1.0, best_seq.size)
            x_new = np.linspace(0.0, 1.0, new_n)
            a = np.interp(x_new, x_old, best_seq)
            a = a + 0.001 * np.random.randn(new_n)   # tiny perturbation
        else:
            n = best_seq.size
            a = np.random.rand(n)

        a = _project_simplex(a)

        # Adam state
        m = np.zeros_like(a)
        v = np.zeros_like(a)

        # --------------------------------------------------------------
        # Inner optimisation loop
        # --------------------------------------------------------------
        for it in range(1, max_iters_per_restart + 1):
            if time.time() >= deadline:
                break

            # ---- 1) convolution and current objective
            conv, af, L = _conv_and_fft(a)
            M = conv.max()
            cur_O = 2.0 * a.size * M / (a.sum() ** 2)   # sum(a) stays =1
            if cur_O < best_score - 1e-12:
                best_score = cur_O
                best_seq = a.copy()
                print(f"[best] iter {it:05d} restart {restart_counter} O={best_score:.6f}")

            # ---- 2) annealing schedule for soft‑max and learning rate
            frac = it / max_iters_per_restart
            beta = beta_start + (beta_end - beta_start) * frac
            lr = lr_start + (lr_end - lr_start) * (frac - 1.0)   # linear decay

            # ---- 3) soft‑max weights approximating the max entry
            #    (conv - M) ≤ 0, so exponentials are ≤ 1.
            shifted = conv - M
            exp_vals = np.exp(beta * shifted)          # safe: all ≤ 1
            w = exp_vals / exp_vals.sum()              # length 2n‑1, sums to 1

            # ---- 4) gradient of 2·max(conv) (soft version)
            g = 2.0 * _correlation_fft(w, af, L, a.size)

            # ---- 5) Adam update
            m = beta1 * m + (1.0 - beta1) * g
            v = beta2 * v + (1.0 - beta2) * (g * g)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            a = a - step

            # ---- 6) enforce symmetry (helps the problem structure)
            a = (a + a[::-1]) * 0.5

            # ---- 7) project back onto the simplex
            a = _project_simplex(a)

            # ---- 8) occasional jitter to leave flat regions
            if it % 3000 == 0:
                a += 0.001 * np.random.randn(a.size)
                a = _project_simplex(a)

        # If we are very close to the deadline, break early.
        if time.time() + 0.2 >= deadline:
            break

    # ------------------------------------------------------------------
    # Final evaluation using the provided evaluate_sequence routine.
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_val:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_seq.tolist()
