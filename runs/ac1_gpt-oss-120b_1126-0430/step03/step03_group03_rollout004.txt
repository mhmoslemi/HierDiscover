"""
Propose a non‑negative step‑function (probability vector) that minimises
    O(a) = 2·n·max(conv(a,a))   with   Σ a_i = 1 .
Algorithm
---------
1. Initialise a with a *U‑shaped* (arcsine) distribution – mass is concentrated
   near the ends of the interval, which is known to give a low self‑convolution
   peak.

2. Compute the full self‑convolution a*a once by FFT (O(n log n)).  From then on
   every elementary “mass‑transfer” move is updated **incrementally** in O(n)
   time using the exact formula
        (a+Δ)*(a+Δ) = a*a + a*Δ + Δ*a + Δ*Δ,
   where Δ contains only two non‑zero components (source – δ, destination + δ).

3. Repeatedly:
   * locate the index k₀ where conv is maximal;
   * the sub‑gradient of this entry is
        g_i = 2·a_{k₀‑i}   for i that participate in the peak;
   * move a small random fraction of mass from the index with the largest
     gradient (the worst contributor to the peak) to a uniformly chosen
     destination;
   * the move is accepted unconditionally – the update is always performed.
     Because the source is chosen according to the gradient, the move is
     biased towards reducing the peak.
   * keep the globally best vector seen so far.

4. To avoid stagnation a tiny Gaussian jitter is added every few thousand
   iterations; after jitter the convolution is recomputed from scratch.

5. The loop runs until the allotted time (≈1000 s) is exhausted.  The best
   sequence found is finally checked with the supplied ``evaluate_sequence``
   and returned as a plain Python list.

The incremental update makes it possible to perform millions of cheap moves,
exploring the huge search space far more thoroughly than the previous
FFT‑based Adam/annealing pipeline.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full self‑convolution a*a using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    # keep only the first 2·n‑1 entries
    return conv_full[: 2 * a.size - 1]


def _init_arcsine(n: int) -> np.ndarray:
    """U‑shaped (arcsine) start: a_i ∝ 1/√(x_i·(1‑x_i)), x_i∈(0,1)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-12
    a = 1.0 / np.sqrt(np.clip(x * (1.0 - x), eps, None))
    a /= a.sum()
    return a


def _incremental_update(conv: np.ndarray, a: np.ndarray,
                        src: int, dst: int, delta: float) -> None:
    """
    Apply the exact update of the full convolution when mass δ is moved
    from index ``src`` to ``dst``.
    The array ``conv`` (length 2·n‑1) and the vector ``a`` (length n)
    are modified in place.
    """
    n = a.size
    # cross‑terms: conv[i:i+n] ← conv[i:i+n] ± 2·δ·a
    conv[src:src + n] -= 2.0 * delta * a
    conv[dst:dst + n] += 2.0 * delta * a
    # quadratic self‑terms
    conv[2 * src] += delta * delta
    conv[2 * dst] += delta * delta
    conv[src + dst] -= 2.0 * delta * delta


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that (approximately) minimises the evaluation
    function.  Returns the sequence as a list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin for the final sanity check
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Problem size – a few‑thousand entries give a good trade‑off.
    n = 8000                       # feel free to adjust if memory permits
    a = _init_arcsine(n)           # U‑shaped start
    L = 1 << ((2 * n - 1).bit_length())   # FFT length ≥ 2·n‑1

    # initial full convolution (exact)
    conv = _conv_fft(a, L)
    cur_max = conv.max()
    best_max = cur_max
    best_a = a.copy()
    print(f"[init] n={n}, O={2 * n * cur_max:.6f}")

    # ------------------------------------------------------------------
    # Main stochastic mass‑transfer loop (high‑frequency cheap moves)
    max_iters = 6_000_000          # generous upper bound – the loop stops earlier
    iter_cnt = 0
    # schedule for the maximal transferred fraction (large → fine)
    frac_start = 0.40
    frac_end = 1e-4
    jitter_every = 5000            # recompute convolution from scratch occasionally

    while time.time() < deadline and iter_cnt < max_iters:
        # 1) locate the peak of the current convolution
        k0 = int(conv.argmax())     # index of the maximal entry
        cur_max = conv[k0]

        # 2) indices that participate in the peak (i + j = k0)
        i_start = max(0, k0 - (n - 1))
        i_end = min(n - 1, k0)
        # slice of a that participates (order i = i_start … i_end)
        a_slice = a[i_start : i_end + 1]
        # gradient g_i = 2·a_{k0‑i} = 2·reversed a_slice
        g_slice = 2.0 * a_slice[::-1]

        # 3) pick a source index with the largest gradient (worst contributor)
        src_rel = int(g_slice.argmax())
        src = i_start + src_rel
        src_val = a[src]
        if src_val < 1e-15:                      # nothing to move – skip
            iter_cnt += 1
            continue

        # 4) pick a destination uniformly (different from src)
        dst = np.random.randint(n)
        while dst == src:
            dst = np.random.randint(n)

        # 5) determine the transferred mass (fraction of src‐mass)
        frac = frac_start * (1.0 - iter_cnt / max_iters) + frac_end
        delta = src_val * np.random.rand() * frac
        if delta <= 0.0:
            iter_cnt += 1
            continue

        # 6) apply the exact incremental convolution update
        _incremental_update(conv, a, src, dst, delta)

        # 7) update the probability vector (mass is conserved)
        a[src] -= delta
        a[dst] += delta

        # 8) record the best vector encountered so far
        new_max = conv.max()
        if new_max < best_max:
            best_max = new_max
            best_a = a.copy()
            print(f"[iter {iter_cnt:08d}] Improved O={2 * n * best_max:.6f}")

        # 9) occasional jitter + full recompute to shake the system
        if (iter_cnt % jitter_every) == 0 and iter_cnt > 0:
            # tiny Gaussian noise, then re‑normalise onto the simplex
            jitter = np.random.randn(n) * 5e-5
            a += jitter
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s == 0.0:
                a = np.full(n, 1.0 / n)
                s = 1.0
            a /= s
            # recompute convolution from scratch (still cheap thanks to FFT)
            conv = _conv_fft(a, L)

        iter_cnt += 1

        # ------------------------------------------------------------------
        # early exit if we are within a second of the deadline
        if time.time() >= deadline - 1.0:
            break

    # ------------------------------------------------------------------
    # Final sanity check with the supplied evaluation function
    final_seq = best_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence failed:", exc)

    return final_seq
