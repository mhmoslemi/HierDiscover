def run_opt(n, time_limit):
    start = time.time()
    # initialize
    if n == len(height_sequence_1) and 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=float)
        a = np.clip(a, 0, None)
        a = a / a.sum()
    else:
        a = np.random.rand(n).astype(float)
        a = a / a.sum()
    # convert to logits for Adam
    x = np.log(a + 1e-12)  # avoid -inf
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    # schedules
    beta_start = 0.1
    beta_end = 5000.0
    lr_start = 0.2
    lr_end = 0.001
    max_iters = 200000
    iter_cnt = 0
    best_a = a.copy()
    best_obj = _objective(a)
    while time.time() - start < time_limit and iter_cnt < max_iters:
        iter_cnt += 1
        # compute a from x
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()
        # convolution
        conv = _conv_self_fft(a)
        M = conv.max()
        obj = 2.0 * n * M
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            # optionally print
            print(f"[n={n} iter={iter_cnt}] new best O={best_obj:.6f}")
        # schedule parameters
        t = min(iter_cnt / max_iters, 1.0)
        beta = beta_start * (beta_end / beta_start) ** t
        lr = lr_start * (lr_end / lr_start) ** t
        # compute softmax weights
        if beta > 500:   # approximate as one-hot
            kmax = np.argmax(conv)
            w = np.zeros_like(conv)
            w[kmax] = 1.0
        else:
            shifted = conv - M
            exp_vals = np.exp(beta * shifted)
            w = exp_vals / exp_vals.sum()
        # gradient wrt a
        g_a = 2.0 * _correlation_fft(w, a)  # shape n
        # transform to gradient wrt x
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)
        # Adam update
        m = beta1 * m + (1 - beta1) * g_x
        v = beta2 * v + (1 - beta2) * (g_x ** 2)
        m_hat = m / (1 - beta1 ** iter_cnt)
        v_hat = v / (1 - beta2 ** iter_cnt)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # random jitter occasionally
        if iter_cnt % 5000 == 0:
            jitter = np.random.randn(n) * 0.01
            x += jitter
        # optional restart if stagnation
        if iter_cnt % 50000 == 0:
            # reinitialize from best with noise
            x = np.log(best_a + 1e-12)
            x += np.random.randn(n) * 0.005

    # compress near-zero entries
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.shape[0]:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
    return best_a.tolist(), best_obj
```

"""
Improved search for a non‑negative step‑function that minimises

    O(a) = 2·n·max(conv(a,a)),     sum(a)=1 , a_i ≥ 0

The algorithm is built around an Adam optimiser on the log‑it (softmax)
parameterisation of the simplex.  A smooth “soft‑max’’ approximation of
the max is used (controlled by an annealing parameter β that grows
exponentially).  The gradient of the smooth objective is obtained via
FFT‑based convolution / correlation, exactly as in the reference code,
but Adam’s adaptive moments give a far more stable descent.  We run a
few independent optimisation runs for different support sizes,
compress negligible entries after each run and keep the overall best
sequence.  A light‑weight local‑search (random mass transfers) is
performed at the end to escape tiny plateaus.

The routine respects the 1000 s budget, prints intermediate best
values and finally returns the best sequence (as a plain list of floats)
found so far.
"""

import time
import numpy as np

# -------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# -------------------------------------------------------------
def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (real‐valued) using FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv


# -------------------------------------------------------------
def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute g_i = Σ_t w[i+t]·a[t]  for i=0,…,n‑1  (correlation of w with a).
    Implemented via FFT.
    """
    n = a.shape[0]
    m = w.shape[0]                     # = 2n‑1
    L = _next_pow2(m + n - 1)          # length for linear convolution
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


# -------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) (assumes sum(a)=1)."""
    n = a.size
    conv = _conv_self_fft(a)
    return 2.0 * n * conv.max()


# -------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


# -------------------------------------------------------------
def _run_opt(
        n: int,
        time_limit: float,
        init_a: np.ndarray | None = None,
        seed: int = 0,
) -> tuple[list[float], float]:
    """
    Optimise a vector of length ``n`` for ``time_limit`` seconds.
    Returns the best sequence (as a plain list) and its exact objective.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # ----- initialise -------------------------------------------------
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = np.random.rand(n)
        a = a / a.sum()
    else:
        a = rng.random(n)
        a = a / a.sum()

    # work in log‑it space to keep the simplex constraint automatically
    x = np.log(a + 1e-16)          # +eps avoids log(0)

    m = np.zeros_like(x)          # Adam first moment
    v = np.zeros_like(x)          # Adam second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # ----- schedules --------------------------------------------------
    beta_start = 0.5               # smoothing start
    beta_end   = 8000.0            # smoothing end (very sharp)
    lr_start   = 0.15              # initial learning rate on logits
    lr_end     = 0.001             # final learning rate
    max_iters  = 300_000

    best_a = a.copy()
    best_obj = _objective(best_a)

    it = 0
    while time.time() - start < time_limit and it < max_iters:
        it += 1

        # ----- probabilities from logits --------------------------------
        a = _softmax(x)

        # ----- convolution & exact objective ----------------------------
        conv = _conv_self_fft(a)
        M = conv.max()
        obj = 2.0 * n * M
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[n={n:5d} it={it:06d}] new best O={best_obj:.6f}")

        # ----- schedule parameters --------------------------------------
        t = min(it / max_iters, 1.0)
        # exponential growth for β
        beta = beta_start * np.exp(np.log(beta_end / beta_start) * t)
        # exponential decay for learning rate
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t)

        # ----- soft‑max weights for the smoothed max --------------------
        if beta > 4000.0:                  # very sharp → use true arg‑max
            kmax = int(np.argmax(conv))
            w = np.zeros_like(conv)
            w[kmax] = 1.0
        else:
            shifted = conv - M                # ≤ 0
            w = np.exp(beta * shifted)
            w = w / w.sum()                  # length 2n‑1, sums to 1

        # ----- gradient w.r.t. a (smooth max) ---------------------------
        g_a = 2.0 * _correlation_fft(w, a)   # shape (n,)

        # ----- transform gradient to log‑it space (natural gradient) ---
        dot = np.dot(a, g_a)                # scalar = ⟨a, g_a⟩
        g_x = a * (g_a - dot)               # shape (n,)

        # ----- Adam update ---------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- occasional jitter to escape plateaus --------------------
        if it % 6000 == 0:
            jitter = rng.normal(scale=0.02, size=n)
            x += jitter

        # ----- restart from current best if stuck for long -------------
        if it % 70_000 == 0:
            # re‑initialise logits from the best solution + tiny noise
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.005, size=n)
            m.fill(0.0)
            v.fill(0.0)
            print(f"[n={n:5d}] restart at iter {it}")

    # ----- compress negligible entries ---------------------------------
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        n_new = best_a.size
        best_obj = _objective(best_a)
        print(f"[compress] reduced length {n} → {n_new}, O={best_obj:.6f}")

    # ----- very short local search (random pair swaps) ----------------
    remaining = time_limit - (time.time() - start)
    if remaining > 0.5:
        # simple greedy mass‑transfer moves
        local_start = time.time()
        a_loc = best_a.copy()
        obj_loc = best_obj
        n_loc = a_loc.size
        while time.time() - local_start < remaining * 0.3:
            i, j = rng.choice(n_loc, size=2, replace=False)
            # transfer a small random fraction of mass from i to j
            delta = a_loc[i] * rng.random() * 0.02   # ≤2 % of a_i
            if delta <= 0.0:
                continue
            a_new = a_loc.copy()
            a_new[i] -= delta
            a_new[j] += delta
            # keep positivity (already guaranteed by construction)
            conv_new = _conv_self_fft(a_new)
            obj_new = 2.0 * n_loc * conv_new.max()
            if obj_new < obj_loc:
                a_loc, obj_loc = a_new, obj_new
                # keep the globally best if we improved
                if obj_new < best_obj:
                    best_a, best_obj = a_new, obj_new
        print(f"[local] final local O={best_obj:.6f}")

    return best_a.tolist(), best_obj

def propose_candidate(
        seed: int = 42,
        budget_s: int = 1000,
        **_ignore,
) -> list[float]:
    """
    Main entry point used by the evaluation harness.
    Runs several optimisation attempts (different support sizes) within the
    allotted time budget and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # -----------------------------------------------------------------
    # Determine a set of candidate lengths.
    lengths = [500, 1000, 2000, 4000]          # a few useful sizes
    if 'height_sequence_1' in globals():
        seq0 = np.asarray(height_sequence_1, dtype=float).copy()
        seq0 = np.clip(seq0, 0.0, None)
        if seq0.sum() < 0.01:
            seq0 = np.ones_like(seq0)
        seq0 = seq0 / seq0.sum()
        n0 = seq0.size
        # ensure we try the length from the previous state as well
        if n0 not in lengths:
            lengths.insert(0, n0)

    best_seq = None
    best_obj = float('inf')

    # -----------------------------------------------------------------
    # Give the longest run (≈80 % of remaining time) to the largest n.
    for idx, n in enumerate(lengths):
        remaining = deadline - time.time()
        if remaining <= 1.0:
            break

        # allocate time: first run gets most, later runs share the rest
        if idx == 0:
            run_time = remaining * 0.75
        else:
            run_time = (remaining - 0.5) / (len(lengths) - idx)

        init = None
        if idx == 0 and 'height_sequence_1' in globals():
            init = seq0

        seq, val = _run_opt(
            n=n,
            time_limit=run_time,
            init_a=init,
            seed=seed + idx * 13,
        )
        if val < best_obj:
            best_obj = val
            best_seq = seq
            print(f"[global] new overall best O={best_obj:.6f} (n={n})")

    # -----------------------------------------------------------------
    # Final evaluation with the provided evaluate_sequence routine.
    try:
        final_score = evaluate_sequence(best_seq)
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_seq
