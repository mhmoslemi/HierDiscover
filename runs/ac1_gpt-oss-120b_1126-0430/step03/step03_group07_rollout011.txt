def _adam_softmax_opt(seq, deadline):
    n = seq.size
    # ensure nonnegative and sum to 1
    seq = np.maximum(seq,0.0)
    s = seq.sum()
    if s == 0.0:
        seq = np.full(n, 1.0/n)
    else:
        seq = seq / s
    # adam parameters
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    lr = 0.2  # maybe bigger than earlier?
    # compute FFT lengths
    L_conv = 1 << ((2*n - 1).bit_length())
    L_corr = 1 << ((3*n - 2).bit_length())
    # pre allocate arrays? Not needed.
    m = np.zeros_like(seq)
    v = np.zeros_like(seq)
    t = 0
    start = time.time()
    total_time = max(deadline - start, 1e-6)
    best_seq = seq.copy()
    best_score = _fast_evaluation(seq)
    # show start
    print(f"[adam_softmax] start n={n}, total_time={total_time:.1f}s")
    while time.time() < deadline:
        # compute convolution via FFT
        Fa = np.fft.rfft(seq, L_conv)
        conv = np.fft.irfft(Fa * Fa, L_conv)[:2*n-1]
        # compute softmax weights
        max_conv = conv.max()
        # linear scaling (beta schedule)
        elapsed = time.time() - start
        fraction = elapsed / total_time
        beta_min = 0.5
        beta_max = 2000.0
        beta = beta_min * (beta_max / beta_min) ** fraction
        # subtract max for numerical stability
        exp_vals = np.exp(beta * (conv - max_conv))
        w = exp_vals / exp_vals.sum()
        # compute gradient via cross-correlation
        Fw = np.fft.rfft(w, L_corr)
        Fseq_rev = np.fft.rfft(seq[::-1], L_corr)
        grad_full = np.fft.irfft(Fw * Fseq_rev, L_corr)
        grad = 2.0 * grad_full[n-1:n-1+n]
        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + epsilon)
        seq = seq - step
        # enforce nonnegativity and renormalize to sum 1 (simple scaling)
        seq = np.maximum(seq, 0.0)
        s = seq.sum()
        if s == 0.0:
            seq = np.full(n, 1.0/n)
        else:
            seq = seq / s
        # occasional random jitter to escape plateaus
        if np.random.rand() < 0.001:
            jitter_idx = np.random.randint(n)
            seq[jitter_idx] += np.random.randn() * 1e-3
            seq = np.maximum(seq, 0.0)
            seq = seq / seq.sum()

"""

summary
-------
The metric to be minimised is

    2·n·max(conv(a,a)) / (sum(a)²) ,

scale‑invariant, so we fix ∑a = 1 and minimise the peak of the
self‑convolution.  The optimiser works in three stages:

1. **Diverse deterministic seeds** – symmetric / asymmetric Beta
   densities, sin‑power shapes and a few random Dirichlet draws
   (all normalised to sum = 1).  If a previous best vector
   ``height_sequence_1`` is available it is also added.

2. **Adam‑style “soft‑max” gradient descent** on the simplex.
   Instead of the non‑smooth max we use a temperature‑controlled
   soft‑max:

        w_k = exp(β·conv_k) / Σ_j exp(β·conv_j) ,

   with β increasing from 0.5 to 2000 during the optimisation
   (deterministic annealing).  The gradient of the soft‑max‑averaged
   convolution is

        ∂/∂a_i  Σ_k w_k·(a*a)_k = 2·(w ⋆ a)[i] ,

   i.e. a cross‑correlation that can be evaluated by a single FFT.
   Adam’s adaptive moments give stable per‑coordinate steps.  After
   each update the vector is projected onto the probability simplex
   (non‑negative, sum = 1).

3. **Final refinement** – the best seed receives a long optimisation
   pass; if time permits a tiny random‑perturbation local search is

# ----------------------------------------------------------------------
# Fast evaluation (identical to the supplied evaluate_sequence but without
# the security checks).  Used inside the optimiser.
# ----------------------------------------------------------------------
def _fast_evaluation(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) / (sum(seq)²) for a non‑empty vector."""
    n = seq.size
    if n == 0:
        return float("inf")
    # FFT length – next power of two ≥ 2·n‑1
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))

def _beta_sequence(alpha: float, beta: float, length: int) -> np.ndarray:
    """Beta(α,β) density sampled on the mid‑points of a regular grid."""
    xs = (np.arange(length, dtype=float) + 0.5) / length
    a = np.power(xs, alpha - 1.0, out=np.empty_like(xs))
    b = np.power(1.0 - xs, beta - 1.0, out=np.empty_like(xs))
    seq = a * b
    seq[np.isinf(seq)] = 0.0
    seq = np.maximum(seq, 0.0)
    total = seq.sum()
    if total == 0.0:
        seq = np.full(length, 1.0 / length)
    else:
        seq /= total
    return seq


def _sin_power_sequence(p: float, length: int) -> np.ndarray:
    """a_i ∝ sin(π·(i+0.5)/length)^p (p may be negative)."""
    xs = (np.arange(length, dtype=float) + 0.5) / length
    vals = np.sin(np.pi * xs) ** p
    vals = np.maximum(vals, 0.0)
    total = vals.sum()
    if total == 0.0:
        vals = np.full(length, 1.0 / length)
    else:
        vals /= total
    return vals


def _random_dirichlet(length: int, concentration: float = 0.5) -> np.ndarray:
    """Sample from Dirichlet(concentration,…,concentration)."""
    seq = np.random.gamma(concentration, 1.0, size=length)
    total = seq.sum()
    if total == 0.0:
        seq = np.full(length, 1.0 / length)
    else:
        seq /= total
    return seq

# Simple projection onto the probability simplex (non‑negative, sum = 1)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find the last index where the condition holds
    cond = u > cssv / ind
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Adam optimiser that uses a soft‑max approximation of the max‑convolution.
# ----------------------------------------------------------------------
def _adam_softmax_opt(seq: np.ndarray, deadline: float) -> np.ndarray:
    """
    Optimise a probability vector (sum = 1, non‑negative) w.r.t.
    the peak of its self‑convolution using an Adam update on the
    soft‑max surrogate of the max.
    """
    n = seq.size
    seq = np.maximum(seq, 0.0)
    total = seq.sum()
    if total == 0.0:
        seq = np.full(n, 1.0 / n)
    else:
        seq /= total

    # Adam hyper‑parameters
    adam_beta1 = 0.9
    adam_beta2 = 0.999
    adam_lr = 0.18
    eps = 1e-8

    # FFT lengths (fixed for the whole optimisation)
    L_conv = 1 << ((2 * n - 1).bit_length())          # for a*a
    L_corr = 1 << ((3 * n - 2).bit_length())          # for w ⋆ a

    m = np.zeros_like(seq)
    v_adam = np.zeros_like(seq)

    t = 0
    start = time.time()
    total_time = max(deadline - start, 1e-6)

    best_seq = seq.copy()
    best_score = _fast_evaluation(seq)

    while time.time() < deadline:
        # ----- convolution a*a (real FFT) -----
        Fa = np.fft.rfft(seq, L_conv)
        conv = np.fft.irfft(Fa * Fa, L_conv)[: 2 * n - 1]

        # ----- soft‑max weights (annealed β) -----
        max_conv = conv.max()
        elapsed = time.time() - start
        frac = elapsed / total_time
        beta_min, beta_max = 0.5, 2000.0
        beta = beta_min * (beta_max / beta_min) ** frac
        # stability trick: subtract max before exponentiation
        exp_vals = np.exp(beta * (conv - max_conv))
        w = exp_vals / exp_vals.sum()                     # shape (2n‑1,)

        # ----- gradient = 2·(w ⋆ a) -----
        Fw = np.fft.rfft(w, L_corr)
        Fseq_rev = np.fft.rfft(seq[::-1], L_corr)
        grad_full = np.fft.irfft(Fw * Fseq_rev, L_corr)
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # ----- Adam moment updates -----
        t += 1
        m = adam_beta1 * m + (1.0 - adam_beta1) * grad
        v_adam = adam_beta2 * v_adam + (1.0 - adam_beta2) * (grad * grad)

        m_hat = m / (1.0 - adam_beta1 ** t)
        v_hat = v_adam / (1.0 - adam_beta2 ** t)

        # ----- multiplicative‑style step (in log‑space we subtract) -----
        step = adam_lr * m_hat / (np.sqrt(v_hat) + eps)
        seq = seq - step

        # ----- enforce feasibility -----
        seq = np.maximum(seq, 0.0)
        s = seq.sum()
        if s == 0.0:
            seq = np.full(n, 1.0 / n)
        else:
            seq /= s

            seq[j] = max(0.0, seq[j] + np.random.randn() * 1e-3)
            seq /= seq.sum()

            cur_score = _fast_evaluation(seq)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = seq.copy()
                print(f"[adam_softmax] iter {t:6d}  best {best_score:.6f}")

    if final_score < best_score:
        best_seq = seq.copy()
        best_score = final_score
        print(f"[adam_softmax] final best {best_score:.6f}")
