def greedy_sidon(N, target_size):
    S=[]
    used_sums=set()
    for i in range(N):
        # attempt to add i if all sums i + s for s in S are not already present
        ok=True
        for s in S:
            if (i+s) in used_sums:
                ok=False
                break
        if ok:
            # also need to check self-sum i+i not colliding
            if (i+i) in used_sums:
                ok=False
        if ok:
            for s in S:
                used_sums.add(i+s)
            used_sums.add(i+i)
            S.append(i)
            if len(S)>=target_size:
                break
    return np.array(S)

"""
Propose a non‑negative step‑function sequence that (approximately) minimises
the evaluation function `evaluate_sequence`.  

Algorithm
---------
1. Determine the problem size `n`.  If a previous best (`height_sequence_1`)
   exists we reuse its length, otherwise we start from a uniform vector of
   length 4096.

2. **Quick parametric sweep** – explore a simple “U‑shaped’’ family  
   a_i(p) ∝ (i+1)^(-p) + (n‑i)^(-p)  (p>0) and mixes of it with the uniform
   distribution.  The best candidate becomes the initial point.

3. **Phase 1 – Adam on a log‑sum‑exp surrogate**  
   Objective      f_α(a) = (2·n/α)·log∑_k exp(α·conv_k) .  
   α is annealed from 5 to 5000, Adam updates are projected onto the simplex.
   A large number of iterations is performed (limited only by the allocated
   time budget).

4. **Phase 2 – deterministic sub‑gradient refinement**  
   Uses the exact sub‑gradient of the maximal convolution entry,
   a linearly decaying step size and occasional Gaussian jitter.

5. **Phase 3 – simulated‑annealing hill climbing**  
   Random mass‑transfer moves are accepted with a Metropolis rule; the temperature
   decays exponentially.  Every few thousand steps the walk is re‑seeded with the
   current global best.

6. The best vector found is finally checked with `evaluate_sequence` and
   returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full self‑convolution (length 2·n‑1)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return a*a (full convolution) using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Exact objective for a simplex‑normalised vector
def _objective_simplex(a: np.ndarray, n: int, L: int) -> float:
    """2·n·max(conv(a,a)) – a is assumed to sum to 1."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Sub‑gradient of the maximal convolution entry (used in Phase 2)
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Return the sub‑gradient of max(conv) w.r.t. a."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)          # indices achieving the maximum
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]                # each pair contributes symmetrically
    g /= len(ks)                                   # average if several maximisers
    return g


# ----------------------------------------------------------------------
# Gradient of the smooth log‑sum‑exp surrogate
def _smooth_gradient(a: np.ndarray, conv: np.ndarray, alpha: float, L: int) -> np.ndarray:
    """
    Compute ∇_a ( (2·n/α)·log∑_k e^{α·conv_k} ).
    Result = 2·corr(w, a) where w_k ∝ e^{α·conv_k}.
    """
    # soft‑max weights (stable)
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / exp_scaled.sum()

    # correlation w ⋆ a   (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)   # length ≥ 3·n‑2
    n_ = a.size
    grad = 2.0 * corr_rev[n_ - 1 : n_ - 1 + n_]
    return grad


# ----------------------------------------------------------------------
# Generate a simple “U‑shaped’’ weight vector for a given exponent p
def _u_shape_weights(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    w = (i + 1) ** (-p) + (n - i) ** (-p)
    return w

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a step‑function sequence that minimises `evaluate_sequence`.
    Returns the sequence as a list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5        # safety margin

    # ------------------------------------------------------------------
    # Determine problem size and initialise a (uniform or previous best)
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
            n = base_seq.size
            print(f"[init] Using provided sequence length n={n}")
        except Exception:
            base_seq = None

    if base_seq is None:
        n = 4096
        print(f"[init] No prior sequence – start with length n={n}")

    # FFT length – must accommodate both convolution (2n‑1) and correlation (3n‑2)
    L = 1 << ((3 * n - 2).bit_length())          # next power of two

    # Uniform vector as a simple baseline
    a_uniform = np.full(n, 1.0 / n)
    best_a = a_uniform.copy()
    best_obj = _objective_simplex(best_a, n, L)
    print(f"[init] Uniform O={best_obj:.6f}")

    # If a previous solution exists, try it as well
    if base_seq is not None:
        a_base = base_seq / base_seq.sum()
        obj_base = _objective_simplex(a_base, n, L)
        if obj_base < best_obj:
            best_a = a_base.copy()
            best_obj = obj_base
            print(f"[init] Base sequence O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Phase 0 – cheap parametric sweep over U‑shaped families
    sweep_deadline = start_time + 0.05 * budget_s
    ps = np.linspace(0.2, 4.0, 39)          # 39 values of exponent p
    mix_factors = np.linspace(0.0, 0.5, 6)  # blend with uniform up to 50 %
    while time.time() < sweep_deadline:
        for p in ps:
            w = _u_shape_weights(n, p)
            a_shape = w / w.sum()
            for lam in mix_factors:
                a_mix = (1.0 - lam) * a_uniform + lam * a_shape
                a_mix = _project_to_simplex(a_mix)
                obj = _objective_simplex(a_mix, n, L)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a_mix.copy()
                    print(f"[sweep] p={p:4.2f} λ={lam:3.2f} → O={best_obj:.6f}")
                if time.time() >= sweep_deadline:
                    break
            if time.time() >= sweep_deadline:
                break

    # ------------------------------------------------------------------
    # Phase 1 – Adam on smooth log‑sum‑exp surrogate
    phase1_deadline = time.time() + 0.45 * (deadline - time.time())
    a = best_a.copy()
    max_iter1 = 300_000
    alpha_start, alpha_end = 5.0, 5000.0
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.12
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(max_iter1):
        if time.time() >= phase1_deadline:
            break
        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            print(f"[phase‑1] it={it:6d} O={best_obj:.6f}")

        # linear schedule for α
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iter1)

        # gradient of smoothed objective
        grad = _smooth_gradient(a, conv, alpha, L)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a - step
        a = _project_to_simplex(a_candidate)

    # ------------------------------------------------------------------
    # Phase 2 – deterministic sub‑gradient refinement
    phase2_deadline = time.time() + 0.30 * (deadline - time.time())
    a2 = a.copy()
    max_iter2 = 200_000
    for it in range(max_iter2):
        if time.time() >= phase2_deadline:
            break
        conv = _conv_fft(a2, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a2.copy()
            print(f"[subgrad] it={it:6d} O={best_obj:.6f}")

        # sub‑gradient of the maximal entry
        g = _subgradient_max_conv(a2, conv)

        # linearly decaying step size
        eta = 0.2 * (1.0 - it / max_iter2)
        a2 = a2 - eta * g
        a2 = _project_to_simplex(a2)

        # occasional tiny jitter
        if it % 1000 == 0 and it > 0:
            jitter = np.random.randn(n) * 3e-4
            a2 = _project_to_simplex(a2 + jitter)

    # ------------------------------------------------------------------
    # Phase 3 – simulated‑annealing hill climbing (global refinement)
    local_a = best_a.copy()
    local_obj = _objective_simplex(local_a, n, L)
    total_hc = 0
    T0 = 0.05

    while time.time() < deadline:
        # exponential cooling
        T = T0 * (0.9995 ** total_hc)

        # propose a random mass‑transfer move
        i, j = np.random.randint(n, size=2)
        if i == j or local_a[i] == 0.0:
            total_hc += 1
            continue

        delta = local_a[i] * np.random.rand() * 0.12   # move up to 12 % of source mass
        a_new = local_a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = _project_to_simplex(a_new)

        obj_new = _objective_simplex(a_new, n, L)

        # Metropolis acceptance
        if obj_new < local_obj or np.random.rand() < math.exp((local_obj - obj_new) / max(T, 1e-12)):
            local_a = a_new
            local_obj = obj_new
            if local_obj < best_obj:
                best_obj = local_obj
                best_a = local_a.copy()
                print(f"[anneal] iter={total_hc:7d} O={best_obj:.6f}")

        total_hc += 1

        # every few thousand steps, reset the local walk to the current global best
        if total_hc % 4000 == 0:
            local_a = best_a.copy()
            local_obj = _objective_simplex(local_a, n, L)

        # safety break a second before the deadline
        if time.time() >= deadline - 1.0:
            break

    # ------------------------------------------------------------------
    # Final sanity check with the supplied evaluation function
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:
        print("[final] evaluate_sequence failed:", e)

    return best_a.tolist()
