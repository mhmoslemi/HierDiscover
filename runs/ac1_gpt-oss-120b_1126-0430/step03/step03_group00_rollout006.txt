"""
Propose a step‑function sequence that (approximately) minimises

    O(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm consists of four stages:

1. **Adam on a log‑sum‑exp smooth surrogate** – the smoothed objective
   (2·n/α)·log∑ exp(α·conv) is differentiable; its gradient is a
   correlation that can be evaluated with FFTs.  An exponential schedule
   for α (from 5 to 5000) and a decreasing learning‑rate are used.
   After each Adam step the vector is projected onto the probability
   simplex.

2. **Deterministic projected sub‑gradient descent** – the exact objective
   contains a max; its sub‑gradient is computed analytically and a
   back‑tracking line search guarantees monotone improvement.

3. **Stochastic mass‑transfer with simulated annealing** – a classic
   Metropolis search that moves a random fraction of mass between two
   coordinates, accepting downhill moves or occasionally uphill moves.
   Periodically a sub‑gradient step is taken to accelerate convergence.

4. **Final polishing** – a short run of pure sub‑gradient descent to
   clean up any remaining slack.

All phases respect the global time budget (default 1000 s).  The best
sequence found is returned as a plain Python list of floats.  The code
uses only NumPy (FFT based convolution) and pure‑Python arithmetic, so
it runs comfortably within the 2‑CPU limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        # all entries non‑positive – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full self‑convolution (length 2·n‑1)
def conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return a*a (full convolution) using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    prod = fft_a * fft_a
    conv_full = np.fft.irfft(prod, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Objective for a simplex‑normalised vector (sum = 1)
def objective_simplex(a: np.ndarray, n: int, L: int) -> float:
    """2·n·max(conv(a,a))."""
    conv = conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Gradient of the log‑sum‑exp (soft‑max) surrogate of the max
def smooth_gradient(a: np.ndarray, conv: np.ndarray, alpha: float, L: int) -> np.ndarray:
    """
    Compute ∇_a ( (2·n/α)·log∑_k e^{α·conv_k} ).
    The gradient reduces to 2·corr(w, a) where w_k ∝ e^{α·conv_k}.
    """
    # soft‑max weights
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)          # numeric stability
    w = exp_scaled / exp_scaled.sum()

    # correlation w ⋆ a   (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n_len = a.size
    grad = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]
    return grad


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) w.r.t. a
def subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Return the sub‑gradient of max(conv) (the max‑convolution entry)."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)      # all maximising positions
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        # each pair (i,j) contributes a_j (and a_i when i!=j)
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size > 0:
        g /= ks.size
    return g

    evaluate_sequence().  Returns the best sequence (list of floats).

        print(f"[init] no supplied sequence – using uniform, n={n}")

    n = a.size
    # FFT length – next power of two > 3·n‑2 (covers both convolution and correlation)
    L = 1 << ((3 * n - 2).bit_length())

    conv = conv_fft(a, L)
    best_score = objective_simplex(a, n, L)
    best_seq = a.copy()
    print(f"[init] O={best_score:.6f}")

    # Phase 1 – Adam on the smooth log‑sum‑exp surrogate
    phase1_deadline = start_time + budget_s * 0.55

    beta1 = 0.9

    # Phase 2 – deterministic projected sub‑gradient descent (line search)
    phase2_deadline = start_time + budget_s * 0.85
    max_iter_sub = 200_000
    it_sub = 0

    while time.time() < phase2_deadline and it_sub < max_iter_sub:
        conv = conv_fft(a, L)
        cur_obj = objective_simplex(a, n, L)
        g = subgradient_max_conv(a, conv)

        # decreasing step size schedule
        eta = 0.5 * (1.0 - it_sub / max_iter_sub)
        if eta < 1e-5:
            eta = 1e-5

        improved = False
        # simple back‑tracking (max 10 attempts)
        for _ in range(10):
            a_candidate = a - eta * g
            a_candidate = project_to_simplex(a_candidate)
            new_obj = objective_simplex(a_candidate, n, L)
            if new_obj < cur_obj:
                a = a_candidate
                cur_obj = new_obj
                improved = True
                break
            eta *= 0.5

        if improved and cur_obj < best_score:
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[subgrad] iter {it_sub} improved O={best_score:.6f}")

        # occasional random jitter to shake the system
        if not improved and np.random.rand() < 0.02:
            a += np.random.randn(n) * 5e-4
            a = np.clip(a, 0.0, None)
            a = project_to_simplex(a)

        it_sub += 1

    # ------------------------------------------------------------------
    # Phase 3 – stochastic mass‑transfer moves with simulated annealing
    # Remaining time is devoted to this global search.
    while time.time() < deadline:
        # ---- initialise a restart ----
        if np.random.rand() < 0.5:
            a = best_seq.copy()
            a += np.random.randn(n) * 0.002
            a = np.clip(a, 0.0, None)
            a = project_to_simplex(a)
        else:
            a = np.random.rand(n)
            a = project_to_simplex(a)

        conv = conv_fft(a, L)
        cur_obj = objective_simplex(a, n, L)

        # inner loop for this restart
        it_loc = 0
        no_improve = 0
        T = 0.03                               # simulated‑annealing temperature
        max_iter_loc = 150_000
        while time.time() < deadline and it_loc < max_iter_loc:
            # ---- random mass‑transfer move ----
            i, j = np.random.randint(n, size=2)
            if i != j and a[i] > 1e-14:
                delta = a[i] * np.random.rand() * 0.12
                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta
                a_new = project_to_simplex(a_new)

                conv_new = conv_fft(a_new, L)
                new_obj = objective_simplex(a_new, n, L)

                # Metropolis acceptance
                if new_obj < cur_obj or np.random.rand() < math.exp((cur_obj - new_obj) / max(T, 1e-12)):
                    a = a_new
                    conv = conv_new
                    cur_obj = new_obj
                    no_improve = 0
                    if cur_obj < best_score:
                        best_score = cur_obj
                        best_seq = a.copy()
                        print(f"[{time.time():.1f}] New global best O={best_score:.6f}")
                else:
                    no_improve += 1
            else:
                no_improve += 1

            # ---- occasional sub‑gradient refinement ----
            if it_loc % 9 == 0:
                g = subgradient_max_conv(a, conv)
                eta = 0.07 * (1.0 - it_loc / max_iter_loc)
                a = a - eta * g
                a = project_to_simplex(a)
                conv = conv_fft(a, L)
                cur_obj = objective_simplex(a, n, L)
                if cur_obj < best_score:
                    best_score = cur_obj
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] Sub‑grad O={best_score:.6f}")

            # temperature decay
            T *= 0.99995
            it_loc += 1

            # restart if stuck for a while
            if no_improve >= 4000:
                break

    # ------------------------------------------------------------------
    # Phase 4 – final polishing with a short sub‑gradient run
    a = best_seq.copy()
    polish_start = time.time()
    max_polish = 5000
    it_polish = 0
    while time.time() < deadline and it_polish < max_polish:
        conv = conv_fft(a, L)
        cur_obj = objective_simplex(a, n, L)
        g = subgradient_max_conv(a, conv)
        eta = 0.02 * (1.0 - it_polish / max_polish)
        a = a - eta * g
        a = project_to_simplex(a)
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[polish] iter {it_polish} O={best_score:.6f}")
        it_polish += 1

    # ------------------------------------------------------------------
    # Final sanity check with the provided evaluator
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence O={final_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence failed:", exc)

    # Return the best found sequence as a plain list
    return best_seq.tolist()
