def smooth_max_and_grad(a, beta):
    n = a.shape[0]
    # compute convolution a*a via FFT
    size = 1 << ((2*n-2).bit_length())
    A_fft = np.fft.rfft(a, n=size)  # real FFT for speed
    conv = np.fft.irfft(A_fft * A_fft, n=size)[:2*n-1]

    # smoothing
    max_val = conv.max()
    # shift for stability:
    shifted = conv - max_val
    exp_beta_shift = np.exp(beta * shifted)
    Z = exp_beta_shift.sum()
    smooth = max_val + (1.0/beta) * np.log(Z)  # smoothed max
    w = exp_beta_shift / Z

def _adam_optimize(a0, max_iters, lr, beta_start, beta_end, deadline):
    n = a0.shape[0]
    size_conv = 1 << ((2 * n - 2).bit_length())
    size_corr = 1 << ((3 * n - 2).bit_length())
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)  # non-smooth
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9; beta2=0.999; eps=1e-8
    for t in range(1, max_iters+1):
        if time.time() >= deadline: break
        beta = beta_start + (beta_end - beta_start) * (t / max_iters)
        # compute smooth objective and gradient
        # compute conv via fft for conv_max
        A_fft = np.fft.rfft(a, n=size_conv)
        conv = np.fft.irfft(A_fft * A_fft, n=size_conv)[:2*n-1]
        conv_max = conv.max()
        # smoothing
        max_val = conv_max
        shifted = conv - max_val
        exp_beta = np.exp(beta * shifted)
        Z = exp_beta.sum()
        # w
        w = exp_beta / Z
        # correlation
        A_fft_corr = np.fft.rfft(a, n=size_corr)
        W_fft_corr = np.fft.rfft(w, n=size_corr)
        corr_full = np.fft.irfft(A_fft_corr * np.conj(W_fft_corr), n=size_corr)
        offset = (2*n-1) - 1
        corr_seg = corr_full[offset: offset + n]
        grad = 4.0 * n * corr_seg
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        # projection onto simplex
        a = _project_to_simplex(a)
        # compute non-smooth objective for best tracking
        cur_score = 2.0 * n * conv_max
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[adam] iter {t:5d} new best O={best_score:.6f}")
        # possibly early stop? Not needed
    return best_a, best_score

"""
Improved search for a non‑negative step‑function that minimizes

        O = 2·n·max(conv(a, a))      with Σ a = 1.

The algorithm combines three ideas:

1. **Exponentiated‑gradient warm‑start** – a fast EG loop (the same
   method that was used previously) drives the initial vector into a
   reasonable region.

2. **Smoothed objective with Adam** – we replace the nondifferentiable
   max by a log‑sum‑exp (soft‑max) with a temperature β.
   The gradient of the smoothed objective can be written analytically
   using a single cross‑correlation (computable with FFTs).  An
   Adam‑style adaptive step‑size is then applied, while the iterate is
   projected onto the probability simplex after each step.

3. **Final refinement with the original sub‑gradient** – a short EG
   pass finishes the optimisation.

Multiple starting points are tried (previous best, the theoretical
1/√‑law and a random simplex point).  The best sequence found before the
time limit (≈1000 s) is returned.

All helper functions are defined at module level – there are no
closures or lambdas – and the code only relies on NumPy (which is
available in the execution environment.
"""

def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Non‑smooth objective 2·n·max(conv(a, a)) for a simplex‑vector a."""
    n = a.size
    size = 1 << ((2 * n - 2).bit_length())          # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv(a,a)) on the simplex (used for EG)."""
    n = a.size
    size = 1 << ((2 * n - 2).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    max_val = conv.max()
    ks = np.where(np.abs(conv - max_val) < 1e-14)[0]
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
def _exp_update(a: np.ndarray, g: np.ndarray, eta: float) -> np.ndarray:
    """Exponentiated‑gradient step (mirrored descent on the simplex)."""
    log_a = np.log(a + 1e-15) - eta * g
    log_a -= np.max(log_a)                # keep numbers moderate
    a_new = np.exp(log_a)
    a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
def _optimize(a0: np.ndarray, max_iters: int, eta0: float,
              deadline: float) -> tuple[np.ndarray, float]:
    """Simple EG optimiser – used as a quick warm‑start."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)
    n = a.size
    patience = max_iters // 20
    no_improve = 0

    for t in range(max_iters):
        if time.time() >= deadline:
            break
        g = _subgradient_max_conv(a)
        eta = eta0 * (1.0 - t / max_iters)
        a = _exp_update(a, g, eta)
        cur = _objective(a)
        if cur < best_score - 1e-12:
            best_score = cur
            best_a = a.copy()
            no_improve = 0
            print(f"[eg] iter {t:5d}  new best O={best_score:.6f}")
        else:
            no_improve += 1
            if no_improve >= patience:
                # tiny jitter to try escaping a flat region
                jitter = np.random.randn(n) * 1e-4
                a = best_a + jitter
                a = np.clip(a, 0.0, None)
                a = _project_to_simplex(a)
                no_improve = 0
    return best_a, best_score


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, max_iters: int,
                   lr: float, beta_start: float, beta_end: float,
                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the *smoothed* version of the objective.
    The smooth max is  (1/β)·log Σ exp(β·conv) .
    The gradient of 2·n·smooth_max can be written as
          4·n·(a ⋆ w)   with   w_i ∝ exp(β·conv_i) .
    Both the convolution and the cross‑correlation are obtained by FFTs.
    """
    a = a0.copy()
    n = a.size

    # FFT sizes (powers of two) – pre‑computed once per run
    size_conv = 1 << ((2 * n - 2).bit_length())          # ≥ 2·n‑1
    size_corr = 1 << ((3 * n - 2).bit_length())          # ≥ n + (2·n‑1) – 1

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = a.copy()
    best_score = _objective(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- temperature schedule -----
        beta = beta_start + (beta_end - beta_start) * (t / max_iters)

        # ----- convolution a*a -----
        A_fft = np.fft.rfft(a, n=size_conv)
        conv = np.fft.irfft(A_fft * A_fft, n=size_conv)[: 2 * n - 1]

        # ----- soft‑max weights w -----
        max_val = conv.max()
        shifted = conv - max_val                     # for numerical stability
        exp_beta = np.exp(beta * shifted)            # all ≤ 1
        Z = exp_beta.sum()
        w = exp_beta / Z                              # size = 2·n‑1

        # ----- gradient of the smoothed objective -----
        A_fft_corr = np.fft.rfft(a, n=size_corr)
        W_fft_corr = np.fft.rfft(w, n=size_corr)
        corr_full = np.fft.irfft(A_fft_corr * np.conj(W_fft_corr), n=size_corr)
        # first n entries give (a ⋆ w)[i] = Σ_j a_j w_{i+j}
        corr_seg = corr_full[:n]
        grad = 4.0 * n * corr_seg                     # ∇_a (2·n·smooth_max)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step
        a = _project_to_simplex(a)

        # ----- track the true (non‑smooth) objective -----
        cur_score = 2.0 * n * max_val
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[adam] iter {t:5d}  new best O={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine combines an EG warm‑start,
    a smoothed‑objective Adam phase and a final EG refinement.
    The best sequence found before the deadline is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Determine the working length.
    if "height_sequence_1" in globals():
        a_prev = np.asarray(height_sequence_1, dtype=float)
        n = a_prev.size
        a_prev = a_prev / a_prev.sum()
    else:
        n = 4000                 # a power‑of‑two‑friendly default
        a_prev = None

    # --------------------------------------------------------------
    # Build a small pool of starting points.
    starts = []

    if a_prev is not None:
        starts.append(a_prev)

    # 1/√ law (theoretically sound)
    idx = np.arange(1, n + 1, dtype=float)
    a_sqrt = 1.0 / np.sqrt(idx)
    a_sqrt /= a_sqrt.sum()
    starts.append(a_sqrt)

    # pure random point on the simplex
    a_rand = np.random.rand(n)
    a_rand = _project_to_simplex(a_rand)
    starts.append(a_rand)

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # Optimisation parameters (chosen to fit comfortably into the 1000 s budget)
    eg_iters_warm = 3000
    eg_eta0_warm = 0.35

    adam_iters = 25000
    adam_lr = 0.12
    beta_start = 5.0
    beta_end = 80.0

    eg_iters_final = 4000
    eg_eta0_final = 0.15

    # --------------------------------------------------------------
    for i, a_start in enumerate(starts):
        if time.time() >= deadline:
            break
        print(f"[run {i}] start length={n}  O={_objective(a_start):.6f}")

        # -- warm‑up with EG (fast reduction of the raw max) --
        a_mid, score_mid = _optimize(a_start, eg_iters_warm, eg_eta0_warm, deadline)
        print(f"[run {i}] after EG warm‑up O={score_mid:.6f}")

        # -- main Adam optimisation on the smoothed objective --
        a_adam, score_adam = _adam_optimize(
            a_mid,
            max_iters=adam_iters,
            lr=adam_lr,
            beta_start=beta_start,
            beta_end=beta_end,
            deadline=deadline,
        )
        print(f"[run {i}] after Adam O={score_adam:.6f}")

        # -- final polishing with EG on the true max --
        a_final, score_final = _optimize(a_adam, eg_iters_final, eg_eta0_final, deadline)
        print(f"[run {i}] after final EG O={score_final:.6f}")

        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()

    # --------------------------------------------------------------
    # Fallback – should never happen but guarantees a return value.
    if best_seq is None:
        best_seq = a_sqrt.copy()
        best_score = _objective(best_seq)

    # --------------------------------------------------------------
    # Exact evaluation with the supplied function
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:  # pragma: no cover
        print("[final] evaluate_sequence call failed:", e)
        final_score = None

    return best_seq.tolist()
