```
def _interpolate_and_normalise(seq, n_new):
    old_len = len(seq)
    if old_len == n_new:
        a = seq.copy()
    else:
        x_old = np.linspace(0, 1, old_len)
        x_new = np.linspace(0, 1, n_new)
        a = np.interp(x_new, x_old, seq)
    # Ensure nonnegative
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full_like(a, 1.0/len(a))
    else:
        a /= a_sum
    return a
```

def _adam_optimize(a0, deadline, max_iters=500_000, verbose=False):
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_val = _objective(best)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.1  # base learning rate
    t = 0
    while time.time() < deadline and t < max_iters:
        conv = _fft_convolve(a)
        max_c = conv.max()
        cur_val = 2.0 * n * max_c
        if cur_val < best_val:
            best_val = cur_val
            best = a.copy()
            if verbose:
                print(f"[improve] iter {t:06d}  obj {best_val:.9f}")
        # subgradient
        k = int(np.argmax(conv))
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** (t + 1))
        v_hat = v / (1 - beta2 ** (t + 1))
        lr = lr0 / np.sqrt(t + 1)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)
        # occasional perturb

            perturb = np.random.normal(scale=0.005, size=n)
            a = np.maximum(a + perturb, 0.0)
            a = _project_simplex(a)
        t += 1
    return best, best_val

"""
Propose an improved non‑negative step‑function sequence.

Algorithm
---------

1. Initialise a probability vector on the simplex using the Catalan
   distribution (a_i = C(2i,i)/4**i, normalised to sum = 1).  The Catalan
   law is a known near‑optimal analytic start.

2. Optimise the vector with a projected Adam optimiser.
   The objective to minimise is       
        O(a) = 2·n·max_k (a*a)[k] ,  Σa = 1 .
   For a given vector the active index `k* = argmax (a*a)` yields the exact
   sub‑gradient `∇ = 2·J_k*·a`.  Adam’s adaptive step‑sizes and bias‑corrected
   moments speed up convergence compared with the heavy‑ball method used
   previously.

3. The optimiser runs as long as the time budget permits, keeping the best
   vector observed.

4. After the first phase (n = 20 000) the best vector is linearly
   interpolated to a larger size (n = 40 000) – a larger dimension gives a
   better asymptotic constant.  The Adam optimiser is invoked again on this
   up‑sampled vector.

5. If time remains, a few cheap random restarts (Dirichlet draws) are
   performed; any improvement replaces the current best.

6. The final best vector (already on the simplex) is returned as a plain
   Python list.  All numbers are non‑negative and the sum is exactly 1,
   so `evaluate_sequence` will use the exact objective value.

The code respects the 1000 s budget, uses only NumPy and the standard
library, and is fully self‑contained.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based full linear convolution (real FFT)
# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x ≥ 0, Σx = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Catalan coefficients, normalised to unit L¹‑norm
# ----------------------------------------------------------------------
def _catalan_normalised(n: int) -> np.ndarray:
    """First n Catalan‑type coefficients, normalised to sum = 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Exact objective (the value returned by evaluate_sequence for Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    return float(2.0 * n * np.max(_fft_convolve(a)))


# ----------------------------------------------------------------------
# Linear interpolation from one length to another, with re‑normalisation
# ----------------------------------------------------------------------
def _interpolate_and_normalise(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resample seq to length new_n by linear interpolation and renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        a = seq.copy()
    else:
        x_old = np.linspace(0.0, 1.0, old_n)
        x_new = np.linspace(0.0, 1.0, new_n)
        a = np.interp(x_new, x_old, seq)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / new_n
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Adam optimiser on the exact max‑convolution objective
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   max_iters: int = 600_000,
                   verbose: bool = False) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser for  O(a)=2·n·max(a*a)  on the simplex.
    Returns the best vector seen and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]

    best = a.copy()
    best_val = _objective(best)

    m = np.zeros_like(a)          # 1st moment
    v = np.zeros_like(a)          # 2nd moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.12                    # base learning rate
    beta1_pow = beta1
    beta2_pow = beta2
    t = 0

    while time.time() < deadline and t < max_iters:
        # convolution and active index
        conv = _fft_convolve(a)
        max_c = conv.max()
        cur_val = 2.0 * n * max_c
        if cur_val < best_val:
            best_val = cur_val
            best = a.copy()
            if verbose:
                print(f"[improve] iter {t:06d}  obj {best_val:.9f}")

        # sub‑gradient of the active quadratic form
        k = int(np.argmax(conv))
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # Adam bias‑corrected moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # bias‑correction
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        # diminishing step size
        lr = lr0 / np.sqrt(t + 1)

        # Adam update
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # keep in the feasible set
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        # occasional stochastic kick to escape plateaus
        if t % 5_000 == 0 and t > 0:
            scale = 0.1 * a.mean()          # small relative perturbation
            a = a + np.random.normal(scale=scale, size=n)
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)

        # update bias‑correction factors
        beta1_pow *= beta1
        beta2_pow *= beta2
        t += 1

    return best, best_val


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Return a non‑negative height sequence (Python list) that minimises
    ``evaluate_sequence`` within the allotted time budget.
    """
    np.random.seed(seed)
    # leave a safety margin for the harness
    deadline = time.time() + budget_s - 2.0

    # ------------------------------------------------------------------
    # Phase 1 – moderate size optimisation (n = 20 000)
    # ------------------------------------------------------------------
    n_small = 20_000
    a0_small = _catalan_normalised(n_small)

    # If a previously discovered sequence exists, use it as a warm‑start
    if 'height_sequence_1' in globals():
        seq = np.asarray(height_sequence_1, dtype=np.float64)
        if seq.size > 0:
            if seq.size != n_small:
                a0_small = _interpolate_and_normalise(seq, n_small)
            else:
                a0_small = np.maximum(seq, 0.0)
                s = a0_small.sum()
                if s == 0.0:
                    a0_small[:] = 1.0 / n_small
                else:
                    a0_small /= s

    best_small, best_val_small = _adam_optimize(
        a0_small,
        deadline,
        max_iters=800_000,
        verbose=True,
    )

    # ------------------------------------------------------------------
    # Phase 2 – up‑sample to a larger dimension (n = 40 000)
    # ------------------------------------------------------------------
    if time.time() < deadline - 10.0:
        n_large = 40_000
        a0_large = _interpolate_and_normalise(best_small, n_large)

        best_large, best_val_large = _adam_optimize(
            a0_large,
            deadline,
            max_iters=500_000,
            verbose=True,
        )

        # keep the better of the two phases
        if best_val_large < best_val_small:
            best_seq = best_large
            best_val = best_val_large
        else:
            best_seq = best_small
            best_val = best_val_small
    else:
        best_seq = best_small
        best_val = best_val_small

    # ------------------------------------------------------------------
    # Optional cheap random restarts (if time permits)
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        n_rand = 20_000
        a_rand = np.random.rand(n_rand)
        a_rand /= a_rand.sum()
        a_opt, val = _adam_optimize(
            a_rand,
            deadline,
            max_iters=80_000,
            verbose=False,
        )
        # upscale to the current best size for a fair comparison
        a_opt_up = _interpolate_and_normalise(a_opt, best_seq.shape[0])
        val_up = _objective(a_opt_up)
        if val_up < best_val:
            best_val = val_up
            best_seq = a_opt_up.copy()
            print(f"[restart] new best obj {best_val:.9f}")

    # ------------------------------------------------------------------
    # Final sanity report
    # ------------------------------------------------------------------
    final_obj = _objective(best_seq)
    print(f"[final] best objective ≈ {final_obj:.9f} with n={best_seq.shape[0]}")

    return best_seq.tolist()
