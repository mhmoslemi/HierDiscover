```python
"""
Hybrid LP + projected subgradient optimizer for the max self‑convolution
objective.  The outer loop repeatedly solves a linear program that, for a
fixed f, finds the most “massive’’ non‑negative vector b such that the
convolution f*b stays below the current worst value M.  After scaling b to
the required ℓ₁‑norm and mixing it with f we obtain a new f with a
strictly smaller maximal convolution.  When the time budget is almost
exhausted the algorithm switches to the fast FFT‑based projected
sub‑gradient refinement used in earlier submissions to polish the result.
"""

import time
import numpy as np
from scipy.optimize import linprog
import scipy.sparse as sp

def _catalan_normalised(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i-1] * (2*i-1) / (2*i)
    a /= a.sum()
    return a

def _fft_convolve(a, b):
    n = a.shape[0] + b.shape[0] - 1
    size = 1 << ((n).bit_length())
    Fa = np.fft.rfft(a, size)
    Fb = np.fft.rfft(b, size)
    conv = np.fft.irfft(Fa*Fb, size)
    return conv[:n]

def _max_selfconv(f):
    # compute max of f*f
    conv = _fft_convolve(f, f)
    return conv.max()

def _conv_matrix(f):
    """Return sparse matrix A of shape (2*n-1, n) such that A @ b = f*b (full convolution)."""
    n = f.shape[0]
    # each diagonal offset i contains constant f[i] repeated n times
    diags = [np.full(n, f[i]) for i in range(n)]
    offsets = np.arange(n)   # 0,1,...,n-1
    A = sp.diags(diags, offsets, shape=(2*n-1, n), format='csr')
    return A

def _lp_step(f, M):
    """Solve the LP max Σ b  s.t.  b>=0, f*b <= M."""
    n = f.shape[0]
    A = _conv_matrix(f)  # (2n-1) × n sparse
    # Upper bound for each convolution entry
    ub = np.full(2*n-1, M, dtype=np.float64)
    # Objective: maximize sum(b) = - minimize -sum(b)
    c = -np.ones(n, dtype=np.float64)
    res = linprog(c, A_ub=A, b_ub=ub, bounds=(0, None), method='highs')
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")
    return res.x

def _projected_subgradient(a, deadline, max_iters=200_000, verbose=False):
    """A cheap refinement based on the sub‑gradient code from the previous
    best solution.  The vector a must have ℓ₁‑norm 1; the function returns a
    vector with the same norm."""
    n = a.shape[0]
    a = a.copy()
    best = a.copy()
    best_val = 2.0*n*_fft_convolve(a, a).max()
    beta = 0.9
    momentum = np.zeros_like(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        conv = _fft_convolve(a, a)
        k = int(np.argmax(conv))
        max_c = conv[k]
        cur_val = 2.0*n*max_c
        if cur_val < best_val:
            best_val = cur_val
            best = a.copy()
            if verbose:
                print(f"[subgrad improve] iter {t} obj {best_val:.9f}")
        # subgradient of active term
        grad = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        grad[idx] = 2.0 * a[k - idx]
        # momentum
        momentum = beta * momentum + (1.0 - beta) * grad
        lr = 0.5 / np.sqrt(t + 1)
        a = a - lr * momentum
        a = np.maximum(a, 0.0)
        # project onto simplex Σ a = 1
        # Euclidean projection onto simplex
        u = np.sort(a)[::-1]
        cssv = np.cumsum(u) - 1.0
        rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = cssv[rho] / (rho + 1)
        a = np.maximum(a - theta, 0.0)
        if t % 5_000 == 0 and t > 0:
            pert = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + pert, 0.0)
            # project again
            u = np.sort(a)[::-1]
            cssv = np.cumsum(u) - 1.0
            rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
            if rho.size == 0:
                theta = 0.0
            else:
                rho = rho[-1]
                theta = cssv[rho] / (rho + 1)
            a = np.maximum(a - theta, 0.0)
        t += 1
    return best

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """Return a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence``.  The routine uses a small LP based
    descent followed by a short projected sub‑gradient polishing phase.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # keep a safety margin

    # --- problem size -------------------------------------------------
    n = 2000                      # moderate size, balances quality & speed
    target_sum = np.sqrt(2.0 * n) # we keep the ℓ₁‑norm equal to √(2n)

    # --- initialise with a Catalan distribution ----------------------
    f = _catalan_normalised(n) * target_sum

    best_f = f.copy()
    best_obj = _max_selfconv(best_f)   # with the chosen normalisation M = objective
    print(f"[init] n={n}, initial M={best_obj:.9f}")

    iteration = 0
    # dynamic mixing parameter: start relatively aggressive, then decay
    t_mix = 0.5

    while time.time() < deadline - 5.0:
        # current bound
        M = _max_selfconv(f)

        if M < best_obj:
            best_obj = M
            best_f = f.copy()
            print(f"[lp improve] iter {iteration}  M {M:.9f}")

        # Solve LP to find the most massive vector b respecting the bound M
        try:
            b = _lp_step(f, M)
        except RuntimeError as e:
            print(f"[lp] {e}")
            break

        sum_b = b.sum()
        if sum_b <= 0:
            # no progress possible
            break

        # Scale b to the required ℓ₁‑norm
        g = b * (target_sum / sum_b)

        # Mix f and g
        f = (1.0 - t_mix) * f + t_mix * g

        # slowly decay the mixing weight to improve stability
        t_mix = max(0.05, t_mix * 0.97)

        iteration += 1
        # optional early stop if improvement stalls
        if iteration % 5 == 0:
            # recompute M and see relative change
            new_M = _max_selfconv(f)
            if abs(new_M - M) < 1e-6:
                break

    # ------------------------------------------------------------------
    # Polishing phase: switch to the fast sub‑gradient optimizer on the
    # simplex (sum = 1) for a few thousand cheap iterations.
    # ------------------------------------------------------------------
    # Re‑normalise to sum = 1 for the sub‑gradient method
    a_norm = best_f / best_f.sum()
    # give the optimiser at most the remaining time
    remaining = deadline - time.time()
    if remaining > 5.0:
        sub_deadline = time.time() + remaining - 1.0
        polished = _projected_subgradient(a_norm, sub_deadline,
                                          max_iters=200_000, verbose=False)
        # re‑scale back to the original ℓ₁‑norm
        polished = polished * target_sum
        # keep whichever is better
        polished_M = _max_selfconv(polished)
        if polished_M < best_obj:
            best_obj = polished_M
            best_f = polished
            print(f"[polish] improved M to {best_obj:.9f}")

    # Final sanity check before returning
    final_sum = best_f.sum()
    if final_sum < 0.01:
        # fallback to a trivial uniform distribution
        best_f = np.full(n, target_sum / n)

    # Clip to the limits accepted by evaluate_sequence
    best_f = np.clip(best_f, 0.0, 1000.0)

    print(f"[final] best M ≈ {best_obj:.9f} with n={n}")

    return best_f.tolist()
```

The objective to minimise is  

\[
O(a)=\frac{2\,n\;\max (a*a)}{(\sum_i a_i)^2},
\qquad a_i\ge 0 .
\]

If the vector is kept on the ℓ₁‑sphere  

\[
\sum_i a_i = \sqrt{2n},
\]

the denominator cancels and the objective is simply  

\[
M=\max (a*a).
\]

The algorithm works in two phases  

1. **LP‑descent** – for a fixed vector `f` we solve the linear programme  

   \[
   \max\; \sum_j b_j \qquad
   \text{s.t. } b_j\ge0,\;(f*b)_k\le M\;\;\forall k,
   \]

   where `M = max(f*f)`.  
   The optimal `b` is the most massive vector that does **not** increase the
   worst convolution entry.  After scaling `b` to the required ℓ₁‑norm we either
   replace `f` (if the new max‑convolution is smaller) or mix the two vectors.

2. **Projected sub‑gradient polishing** – when only a few seconds remain the
   fast FFT‑based sub‑gradient optimiser (the one used in the previous state‑of‑the‑art) is run for a few‑hundred thousand cheap iterations.  This removes the last “plateau” left by the LP phase.

Both phases respect the time budget (≈ 1000 s).  The LP phase is performed on a
moderate size (`n≈2000`) – large enough to approach the asymptotic limit but
still cheap to build and solve the linear programme.  The final vector is clipped
to the range accepted by `evaluate_sequence` and returned as a plain Python
list.

```python
"""
Hybrid optimisation for the max‑self‑convolution objective.

* Phase 1 – LP descent:
  Repeatedly solve a linear programme that maximises the ℓ₁‑mass of a vector
  `b` while keeping the convolution `f*b` below the current worst value `M`.
  After scaling `b` to the required ℓ₁‑norm we either replace `f` or mix it
  with `b`.  This strictly decreases the maximal convolution entry.

* Phase 2 – Projected sub‑gradient polishing:
  When the time budget is almost exhausted a fast FFT‑based projected sub‑gradient
  method (the one from the previous best solution) is run for a few thousand
  cheap iterations to fine‑tune the result.

The returned list contains non‑negative heights and is directly usable by
`evaluate_sequence`.
"""

import time
import numpy as np
from scipy.optimize import linprog
import scipy.sparse as sp


# ----------------------------------------------------------------------
# Helper – Catalan (1/√(1‑z)) normalised to unit ℓ₁‑norm
# ----------------------------------------------------------------------
def _catalan_normalised(n: int) -> np.ndarray:
    """First n terms a_i = C(2i,i)/4**i, normalised so that Σ a_i = 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)   # recurrence for central binomial coeff.
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# FFT based full convolution (linear, not circular)
# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of a and b using a real FFT."""
    n = a.shape[0] + b.shape[0] - 1
    size = 1 << ((n).bit_length())           # next power of two
    Fa = np.fft.rfft(a, size)
    Fb = np.fft.rfft(b, size)
    conv = np.fft.irfft(Fa * Fb, size)
    return conv[:n]


# ----------------------------------------------------------------------
# Maximal entry of the self‑convolution
# ----------------------------------------------------------------------
def _max_selfconv(v: np.ndarray) -> float:
    """Return max_k (v*v)[k] for a real vector v."""
    return _fft_convolve(v, v).max()


# ----------------------------------------------------------------------
# Build the (2n‑1) × n convolution matrix A such that A @ b = f*b
# ----------------------------------------------------------------------
def _conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    """Sparse matrix representing the linear map b ↦ f*b (full convolution)."""
    n = f.shape[0]
    # each diagonal offset i (0 ≤ i < n) contains the constant f[i] repeated n times
    diags = [np.full(n, f[i], dtype=np.float64) for i in range(n)]
    offsets = np.arange(n)                # 0,1,…,n‑1
    A = sp.diags(diags, offsets, shape=(2 * n - 1, n), format="csr")
    return A


# ----------------------------------------------------------------------
# One LP step: maximise Σ b_j under the current bound M
# ----------------------------------------------------------------------
def _lp_step(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve   max Σ b_j
    s.t.    b_j ≥ 0,
            (f * b)_k ≤ M   for all k.

    Returns the optimal vector b (still on the ℓ₁‑scale of f, i.e. Σb ≤ √(2n)).
    """
    n = f.shape[0]
    A = _conv_matrix(f)                     # (2n‑1) × n
    ub = np.full(2 * n - 1, M, dtype=np.float64)

    # maximise Σb -> minimise -Σb
    c = -np.ones(n, dtype=np.float64)

    res = linprog(c, A_ub=A, b_ub=ub, bounds=(0, None), method="highs")
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")
    return res.x


# ----------------------------------------------------------------------
# Project a vector onto the probability simplex {x≥0, Σx=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex Σx = 1, x ≥ 0."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical safeguard – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast projected sub‑gradient optimiser (the previous best method)
# ----------------------------------------------------------------------
def _subgradient_refine(a: np.ndarray, deadline: float,
                        max_iters: int = 200_000,
                        verbose: bool = False) -> np.ndarray:
    """
    Refine a probability vector `a` (Σa = 1) with projected sub‑gradient descent.
    Returns the best vector encountered before the deadline.
    """
    n = a.shape[0]
    a = a.copy()
    best = a.copy()
    best_val = 2.0 * n * _max_selfconv(a)

    beta = 0.9
    momentum = np.zeros_like(a)

    t = 0
    while time.time() < deadline and t < max_iters:
        conv = _fft_convolve(a, a)
        k = int(np.argmax(conv))
        max_c = conv[k]
        cur_val = 2.0 * n * max_c
        if cur_val < best_val:
            best_val = cur_val
            best = a.copy()
            if verbose:
                print(f"[subgrad improve] iter {t:06d} obj {best_val:.9f}")

        # sub‑gradient of the active quadratic form
        grad = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end   = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        grad[idx] = 2.0 * a[k - idx]

        # heavy‑ball momentum
        momentum = beta * momentum + (1.0 - beta) * grad
        lr = 0.5 / np.sqrt(t + 1)

        a = a - lr * momentum
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        # occasional random jitter to escape plateaus
        if t % 5_000 == 0 and t > 0:
            pert = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + pert, 0.0)
            a = _project_simplex(a)

        t += 1

    return best


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Return a list of non‑negative heights that (empirically) minimises
    ``evaluate_sequence`` within the allotted time budget.

    The routine first runs an LP‑based descent on a moderate size problem
    (n≈2000) and, if time permits, finishes with a short projected
    sub‑gradient polishing phase.
    """
    np.random.seed(seed)
    # keep a small safety margin for the surrounding harness
    deadline = time.time() + budget_s - 2.0

    # --------------------------------------------------------------
    # 1) problem size – a compromise between quality and LP cost
    # --------------------------------------------------------------
    n = 2000
    target_sum = np.sqrt(2.0 * n)          # ℓ₁‑norm we will keep fixed

    # --------------------------------------------------------------
    # 2) initialise with the Catalan distribution (known good start)
    # --------------------------------------------------------------
    f = _catalan_normalised(n) * target_sum

    best_f = f.copy()
    best_obj = _max_selfconv(best_f)       # objective when Σf = √(2n)
    print(f"[init] n={n}, initial M={best_obj:.9f}")

    # --------------------------------------------------------------
    # 3) LP descent loop
    # --------------------------------------------------------------
    iteration = 0
    mix_weight = 0.5                       # start with an aggressive mix

    while time.time() < deadline - 5.0:
        M = _max_selfconv(f)

        # record improvement
        if M < best_obj:
            best_obj = M
            best_f = f.copy()
            print(f"[lp improve] iter {iteration:04d}  M {M:.9f}")

        # solve the LP for the current bound M
        try:
            b = _lp_step(f, M)
        except RuntimeError as e:
            print(f"[lp] {e}")
            break

        sum_b = b.sum()
        if sum_b <= 0.0:
            # nothing to improve any more
            break

        # scale to the required ℓ₁‑norm
        g = b * (target_sum / sum_b)

        # try a full replacement – it is safe only if it really improves M
        M_g = _max_selfconv(g)
        if M_g < M:
            f = g
            # after a successful replacement we can be a little bolder
            mix_weight = min(0.9, mix_weight * 1.05)
        else:
            # otherwise perform a convex combination
            f = (1.0 - mix_weight) * f + mix_weight * g
            # slowly decay the mixing weight to stabilise the sequence
            mix_weight = max(0.05, mix_weight * 0.97)

        iteration += 1

        # stop early if the progress becomes negligible
        if iteration % 5 == 0:
            new_M = _max_selfconv(f)
            if abs(new_M - M) < 1e-8:
                break

    # --------------------------------------------------------------
    # 4) Polishing phase – cheap projected sub‑gradient optimisation
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        # rescale to the probability simplex required by the sub‑gradient code
        a_norm = best_f / best_f.sum()          # now Σa = 1
        sub_deadline = time.time() + remaining - 1.0
        polished = _subgradient_refine(a_norm, sub_deadline,
                                       max_iters=200_000,
                                       verbose=False)
        # bring back to the ℓ₁‑norm √(2n)
        polished = polished * target_sum
        polished_M = _max_selfconv(polished)
        if polished_M < best_obj:
            best_obj = polished_M
            best_f = polished
            print(f"[polish] improved M to {best_obj:.9f}")

    # --------------------------------------------------------------
    # 5) Final sanity checks and return
    # --------------------------------------------------------------
    # ensure a tiny positive sum (evaluate_sequence rejects <0.01)
    if best_f.sum() < 0.01:
        best_f = np.full(n, target_sum / n)

    # clip to the limits accepted by evaluate_sequence (0 ≤ x ≤ 1000)
    best_f = np.clip(best_f, 0.0, 1000.0)

    print(f"[final] best M ≈ {best_obj:.9f} with n={n}")
    return best_f.tolist()
```
