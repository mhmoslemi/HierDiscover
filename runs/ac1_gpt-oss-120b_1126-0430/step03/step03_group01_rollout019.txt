#!/usr/bin/env python
"""
Search for a step‑function (non‑negative sequence) that minimizes

    score = 2·n·max_{k} (a∗a)[k] / (sum a)^2

where ``a`` is the sequence and ``*`` denotes the linear convolution.
The algorithm consists of three phases:

1. **Coarse grid search** over a two‑parameter family
   a_i ∝ (i+1+shift)^{-p}.  This family already gets close to the
   theoretical optimum (≈π/2) and often beats the previous best
   cheap score (≈1.52).

2. **Fine‑grained optimisation** starting from the best grid point.
   An Adam optimiser works on the log‑softmax parametrisation of the
   vector (so the simplex constraint ∑a = 1 is automatic).  The gradient
   is a (sub‑)gradient of the *max* convolution; to obtain a more
   informative direction we sum the gradients of the K largest
   convolution entries (K≈3).  A short back‑tracking line‑search and
   occasional random perturbations improve robustness.

3. **Exact evaluation** of the best candidate found with the provided
   ``evaluate_sequence`` function.

The routine respects a time budget (≈1000 s) and prints progress messages.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max; returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it representation that yields `a` after soft‑max."""
    return np.log(a + eps)


def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two ≥ 2·n‑1 (efficient size for real FFT)."""
    return 1 << ((2 * n - 1).bit_length())


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a using a real FFT of length ``fft_len``.
    Returns the full result of length 2·len(a)‑1.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.size - 1]
    return conv


def _subgradient_multi_shifts(a: np.ndarray, shifts: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the sum of convolution entries of the given shifts.
    For each active offset k we have (∂/∂a_i) (a*a)[k] = 2·a_{k‑i}
    whenever the index lies inside the support.
    The result is the sum of these contributions over all ``shifts``.
    """
    n = a.shape[0]
    g = np.zeros_like(a)

    for k in shifts:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i = np.arange(lo, hi + 1, dtype=np.int64)
        g[i] += 2.0 * a[k - i]
    return g


def _shifted_power_law_distribution(n: int, p: float, shift: float) -> np.ndarray:
    """
    Normalised distribution a_i ∝ (i+1+shift)^{-p}.
    ``shift`` may be non‑integer; it is added before the exponent.
    """
    idx = np.arange(1, n + 1, dtype=np.float64) + shift
    a = idx ** (-p)
    total = a.sum()
    if total == 0.0:
        a[:] = 1.0 / n
    else:
        a /= total
    return a


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Normalised Beta‑density discretised on n points:
        f(x) ∝ x^{alpha‑1} (1‑x)^{beta‑1},  x∈[0,1].
    Uses the mid‑point rule for stability.
    """
    x = (np.arange(n) + 0.5) / n
    # avoid zero to the power of a negative number
    a = np.power(x + 1e-16, alpha - 1.0) * np.power(1.0 - x + 1e-16, beta - 1.0)
    total = a.sum()
    if total == 0.0:
        a[:] = 1.0 / n
    else:
        a /= total
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) beats the target bound.
    The routine respects the given time budget (≈1000 s) and uses a
    combination of grid search and Adam optimisation on the probability simplex.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # small safety margin

    # ---------------------------------------------------------------
    # 0. Choose a working support size.
    # ---------------------------------------------------------------
    # Use the globally supplied sequence if it exists – it is often a good
    # starting point for the Adam phase.
    try:
        init_global = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        n = max(2000, init_global.shape[0])
        use_global = True
        print(f"[Info] Using global sequence as warm‑start, length = {n}")
    except Exception:
        init_global = None
        n = 20000
        use_global = False
        print(f"[Info] No global sequence found, using length = {n}")

    fft_len = _fft_len_for_n(n)

    # ---------------------------------------------------------------
    # 1. Coarse grid search for a good initial distribution.
    # ---------------------------------------------------------------
    best_a = None
    best_score = float("inf")
    best_descr = None

    # If a global vector is available we still run the grid search, hoping
    # to improve upon it.
    p_grid = np.linspace(0.45, 0.70, 27)               # 0.45 … 0.70
    shift_grid = np.arange(0, 301, 30)               # 0 … 300 in steps of 30
    beta_grid = np.linspace(0.3, 3.0, 14)            # for Beta family (optional)

    # first try the shifted‑power‑law family
    for p in p_grid:
        for sh in shift_grid:
            if time.time() >= deadline:
                break
            cand = _shifted_power_law_distribution(n, p, sh)
            conv = _conv_fft(cand, fft_len)
            maxb = conv.max()
            cheap_score = 2.0 * n * maxb
            if cheap_score < best_score:
                best_score = cheap_score
                best_a = cand.copy()
                best_descr = f"pow(p={p:.3f}, shift={sh})"
    # optional second family: Beta distributions (asymmetric)
    for alpha in beta_grid:
        for beta in beta_grid:
            if time.time() >= deadline:
                break
            cand = _beta_distribution(n, alpha, beta)
            conv = _conv_fft(cand, fft_len)
            maxb = conv.max()
            cheap_score = 2.0 * n * maxb
            if cheap_score < best_score:
                best_score = cheap_score
                best_a = cand.copy()
                best_descr = f"beta(a={alpha:.2f},b={beta:.2f})"

    # fall back to a uniform vector if everything failed (should never happen)
    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_score = 2.0 * n * _conv_fft(best_a, fft_len).max()
        best_descr = "uniform"

    print(f"[Init] Best start: {best_descr}, cheap score = {best_score:.6f}")

    # ---------------------------------------------------------------
    # 2. Adam optimiser on the simplex (log‑it parametrisation).
    # ---------------------------------------------------------------
    eps = 1e-16
    z = _logits_from_distribution(best_a, eps)

    m = np.zeros_like(z)      # first‑moment estimate
    v = np.zeros_like(z)      # second‑moment estimate
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    base_alpha = 0.25         # a slightly larger base learning‑rate

    # track the best ever seen
    best_seq = best_a.copy()
    best_score = 2.0 * n * _conv_fft(best_seq, fft_len).max()
    it = 0
    last_improve = 0
    report_every = 2000
    K = 3                     # number of top convolution entries used for the sub‑gradient

    while time.time() < deadline and it < 2_000_000:
        it += 1

        # probability vector from logits
        a = _softmax(z)

        # full convolution (needed for both max and multi‑shift gradient)
        conv = _conv_fft(a, fft_len)
        maxb = float(conv.max())
        cheap_score = 2.0 * n * maxb

        # keep the best ever seen (cheap proxy)
        if cheap_score < best_score:
            best_score = cheap_score
            best_seq = a.copy()
            last_improve = it
            if it % report_every == 0:
                print(f"[Iter {it:07d}] NEW BEST cheap = {cheap_score:.6f}")

        # choose the K largest convolution entries (including the absolute max)
        if K == 1:
            active_idx = np.array([int(conv.argmax())], dtype=np.int64)
        else:
            # argpartition is O(N) and cheap for our modest K
            active_idx = np.argpartition(conv, -K)[-K:]

        # sub‑gradient of the sum of the selected entries
        g_a = _subgradient_multi_shifts(a, active_idx)

        # gradient in logits space (mirror descent with KL divergence)
        mu = np.dot(g_a, a)
        grad_z = a * (g_a - mu)

        # Adam moment updates (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # direction (negative Adam step)
        step_dir = - m_hat / (np.sqrt(v_hat) + eps_adam)

        # basic step‑size schedule: inverse‑sqrt decay
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)

        # -----------------------------------------------------------
        # Simple back‑tracking line search (up to 3 halvings)
        # -----------------------------------------------------------
        accepted = False
        trial_alpha = alpha_t
        for _ in range(3):
            z_new = z + trial_alpha * step_dir
            z_new = np.clip(z_new, -50.0, 50.0)       # keep exp safe
            a_new = _softmax(z_new)
            maxb_new = _conv_fft(a_new, fft_len).max()
            new_score = 2.0 * n * maxb_new
            if new_score <= cheap_score:
                z = z_new
                accepted = True
                break
            trial_alpha *= 0.5

        if not accepted:
            # take a modest step anyway (helps escaping plateaus)
            z += alpha_t * step_dir
            z = np.clip(z, -50.0, 50.0)

        # -----------------------------------------------------------
        # Random perturbations (helps escape flat regions)
        # -----------------------------------------------------------
        if it % 5000 == 0 and np.random.rand() < 0.25:
            z += 0.02 * np.random.randn(n)
            m *= 0.5
            v *= 0.5

        # -----------------------------------------------------------
        # Early stop on stagnation
        # -----------------------------------------------------------
        if it - last_improve > 250_000:
            print(f"[Iter {it:07d}] No improvement for {it - last_improve} iters – stopping.")
            break

    # ---------------------------------------------------------------
    # 3. Final exact evaluation (uses the provided evaluate_sequence)
    # ---------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[Result] Exact score = {final_score:.6f}")
    except Exception as exc:   # defensive – should never happen
        final_score = float("inf")
        print(f"[Error] Evaluation failed: {exc}")

    # ---------------------------------------------------------------
    # 4. Return the best sequence found.
    # ---------------------------------------------------------------
    return final_seq
