"""
Propose a non‑negative step‑function sequence that (approximately) minimises

    O(a) = 2·n·max(conv(a,a))   with   Σ a_i = 1  (i.e. a lies on the probability simplex).

The algorithm works in several stages:

1.  Initialise from the previously‑found best sequence (``height_sequence_1``) if
    present, otherwise start with a uniform distribution.
2.  OPTIONAL smoothing: a few low‑pass “blur” passes (kernel [0.25,0.5,0.25])
    to get rid of tiny spikes that may hinder optimisation.
3.  Adam on a smooth log‑sum‑exp surrogate of the max‑convolution.  The surrogate
    becomes tighter as the annealing parameter α grows, allowing a fast descent
    towards a low‑max‑convolution region.
4.  Deterministic sub‑gradient descent with an adaptive step size (Armijo‑style
    back‑tracking).  The sub‑gradient of the true max‑convolution entry is
    computed efficiently via FFT‑based correlation.
5.  Simulated‑annealing hill climbing.  Moves are *directed*: source indices
    are chosen with probability proportional to their sub‑gradient (high
    contribution to the current maximum) and target indices preferentially
    have low sub‑gradient.  The temperature decays exponentially; occasional
    Gaussian jitter helps to escape plateaus.
6.  Final tiny sub‑gradient polishing.

All operations stay on the simplex (projection onto the probability simplex)
and the total running time respects the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
#  Simplex projection ----------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        # all non‑positive – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
#  FFT‑based full self‑convolution ----------------------------------------
def conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return the linear convolution a*a (full, length 2·n‑1) using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
#  Exact objective for a simplex‑normalised vector ------------------------
def objective(a: np.ndarray, n: int, L: int) -> float:
    """2·n·max(conv(a,a)) – a is assumed to sum to 1."""
    conv = conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
#  Sub‑gradient of the maximal convolution entry (FFT version) ----------
def subgradient_max(a: np.ndarray, conv: np.ndarray, L: int) -> np.ndarray:
    """
    Efficiently compute the (averaged) sub‑gradient of max(conv) w.r.t. a.
    Returns a vector of length n.
    """
    n = a.size
    mask = np.zeros(2 * n - 1, dtype=float)
    max_val = conv.max()
    # binary mask of active constraints (could be several)
    mask[conv == max_val] = 1.0
    # average over all active positions
    mask /= mask.sum()

    # correlation mask ★ a   (central n entries)
    a_rev = a[::-1]
    fft_mask = np.fft.rfft(mask, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_mask * fft_a_rev, n=L)
    g = 2.0 * corr_rev[n - 1 : n - 1 + n]  # central slice
    return g


# ----------------------------------------------------------------------
#  Gradient of the smooth log‑sum‑exp surrogate ---------------------------
def smooth_gradient(a: np.ndarray, conv: np.ndarray, alpha: float, L: int) -> np.ndarray:
    """
    Compute ∇_a ( (2·n/α)·log∑_k e^{α·conv_k} ).
    Result = 2·corr(w, a) where w_k ∝ e^{α·conv_k}.
    """
    # soft‑max weights
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / exp_scaled.sum()

    # correlation w ★ a   (central n entries)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n = a.size
    grad = 2.0 * corr_rev[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
#  Small low‑pass blur (helps to eliminate microscopic spikes) ----------
def blur(a: np.ndarray, steps: int = 10) -> np.ndarray:
    """Apply a few passes of a 3‑tap smoothing kernel and renormalise."""
    kernel = np.array([0.25, 0.5, 0.25], dtype=float)
    for _ in range(steps):
        a = np.convolve(a, kernel, mode="same")
        a = np.maximum(a, 0.0)
        a /= a.sum()
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):

    Search for a step‑function sequence that minimises the evaluation function.
    Returns the sequence as a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the final sanity check
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Initialise ---------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.array(height_sequence_1, dtype=float)
        a = np.maximum(a, 0.0)
        a /= a.sum()
        print(f"[init] Using provided sequence, n={a.size}")
    else:
        n = 4096
        a = np.full(n, 1.0 / n, dtype=float)
        print(f"[init] No prior sequence – start uniform, n={a.size}")

    n = a.size
    # FFT length – must handle convolution (2n‑1) and correlation (3n‑2)
    L = 1 << ((3 * n - 2).bit_length())  # next power of two

    a = blur(a, steps=15)

    best_a = a.copy()
    best_obj = objective(best_a, n, L)
    print(f"[init] O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Helper for time checks
    def time_left() -> float:
        return deadline - time.time()

    # ------------------------------------------------------------------
    # Phase 1 – Adam on smooth log‑sum‑exp surrogate ----------------------
    alpha_start, alpha_end = 5.0, 5000.0
    max_iter_phase1 = 250_000
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.15
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    phase1_deadline = start_time + budget_s * 0.35  # roughly 35% of budget
    iter1 = 0
    while time.time() < phase1_deadline and iter1 < max_iter_phase1:
        conv = conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            print(f"[phase‑1] iter={iter1}, O={best_obj:.6f}")

        # linear α‑annealing
        frac = iter1 / max_iter_phase1
        alpha = alpha_start + (alpha_end - alpha_start) * frac

        # smooth gradient
        grad = smooth_gradient(a, conv, alpha, L)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (iter1 + 1))
        v_hat = v / (1.0 - beta2 ** (iter1 + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a - step
        a = project_to_simplex(a_candidate)

        iter1 += 1
        if time.time() >= deadline - 4.0:
            break

    # ------------------------------------------------------------------

    max_iter_sub = 180_000
    eta0 = 0.25           # initial step size
    eta = eta0
    min_eta = 1e-5
    iter2 = 0
    while time.time() < deadline and iter2 < max_iter_sub:

        cur_obj = 2.0 * n * conv.max()

            best_a = a.copy()
            print(f"[subgrad] iter={iter2}, O={best_obj:.6f}")

        for _ in range(5):

            a_new = project_to_simplex(a_new)
            conv_new = conv_fft(a_new, L)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < cur_obj:
                a = a_new
                eta = min(eta * 1.1, eta0)   # a tiny increase after success
                improved = True
                break
            else:
                eta *= 0.5
                if eta < min_eta:
                    break
        if not improved:
            # if we failed to improve, keep the current a but shrink η strongly
            eta = max(eta * 0.3, min_eta)

        # occasional small Gaussian jitter to avoid being stuck
        if iter2 % 2000 == 0 and iter2 > 0:
            jitter = np.random.randn(n) * 4e-4
            a = project_to_simplex(a + jitter)

        iter2 += 1
        # linear decay of a baseline step‑size (acts as a fallback)
        eta0 = eta0 * (1.0 - iter2 / max_iter_sub)
        if time.time() >= deadline - 4.0:
            break

    # ------------------------------------------------------------------
    # Phase 3 – directed simulated‑annealing hill climbing -----------------
    total_hc = 0
    temperature0 = 0.06
    # we keep a local copy that we modify during the climb; reset periodically

    local_conv = conv_fft(local_a, L)
    local_obj = 2.0 * n * local_conv.max()

    # Pre‑compute a static sub‑gradient for move‑direction bias (updated every 2000 iters)
    bias_g = None
    bias_update = 0

        T = temperature0 * (0.9995 ** total_hc)

        # Update bias every few thousand steps
        if total_hc - bias_update >= 2000:
            conv_cur = conv_fft(local_a, L)
            bias_g = subgradient_max(local_a, conv_cur, L)
            bias_update = total_hc

        # Sample source index preferentially where gradient is large
        if bias_g is not None and np.random.rand() < 0.7:
            prob_src = bias_g / bias_g.sum()
            i = np.searchsorted(np.cumsum(prob_src), np.random.rand())
        else:
            i = np.random.randint(n)

        # Ensure we have some mass to move
        if local_a[i] <= 1e-12:
            total_hc += 1
            continue

        # Sample target index where gradient is small (preferentially)
        if bias_g is not None and np.random.rand() < 0.6:
            inv = bias_g.max() - bias_g + 1e-9
            prob_tgt = inv / inv.sum()

        a_new[i] -= delta
        a_new[j] += delta
        a_new = project_to_simplex(a_new)

        obj_new = 2.0 * n * conv_new.max()

            local_a, local_obj = a_new, obj_new
            # global best update
            if obj_new < best_obj:
                best_obj, best_a = obj_new, local_a.copy()
                print(f"[anneal] iter={total_hc}, O={best_obj:.6f}")

        total_hc += 1

        if total_hc % 4000 == 0:

        a_candidate = project_to_simplex(a_candidate)
        obj_cand = objective(a_candidate, n, L)
        if obj_cand < best_obj:
            best_a, best_obj = a_candidate, obj_cand

    # Final sanity check with the provided evaluation function
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:
        print("[final] evaluate_sequence failed:", e)

    return best_a.tolist()
