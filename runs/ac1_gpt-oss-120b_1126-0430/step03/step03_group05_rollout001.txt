    best_local_a = a.copy()
    best_local_score = np.inf
    lam = lambda0
    a_current = a.copy()
    for iter in range(max_iter):
        if time.time() >= deadline: break
        # compute conv
        conv = _full_convolution_fft(a_current, a_current)
        max_conv = conv.max()
        cheap_score = 2.0 * n * max_conv
        if cheap_score < best_local_score:
            best_local_score = cheap_score
            best_local_a = a_current.copy()

        # compute gradient
        a_rev = a_current[::-1]
        grad_full = _full_convolution_fft(a_rev, w)

        # EG update
        a_current = a_current * np.exp(-eta * grad)
        sum_a = a_current.sum()
        if sum_a == 0:
            a_current = np.full(n, 1.0/n)
        else:
            a_current /= sum_a

#!/usr/bin/env python
"""
Improved search for a non‑negative step function that minimises

    2·n·max_k (a*a)[k] / ( Σ a_i )²

The algorithm is a fast projected *exponentiated‑gradient* descent on the
simplex, using a smooth approximation of the max‑operator (soft‑max).  At
each iteration we

1. compute the full autocorrelation a*a by FFT,
2. build a soft‑max weight vector w (λ controls the smoothness),
3. obtain the gradient of the smoothed objective via another FFT,
4. update a_i ← a_i·exp(‑η·grad_i) and renormalise onto the probability
   simplex,
5. slowly increase λ so the approximation converges to the true max.

Multiple random restarts with different lengths (≈5 k–15 k) and
initialisations (uniform, power‑law, Dirichlet) are performed until the
time budget (≈1000 s) is exhausted.  The best vector (according to the
cheap proxy 2·n·max_conv, i.e. the exact value for Σa=1) is finally
evaluated with the provided ``evaluate_sequence`` function and returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the probability simplex:
        Σ v_i = 1 , v_i ≥ 0.
    Implementation of the algorithm from:
    Wang & Carreira‑Perpiñán, “Projection onto the probability simplex”.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all entries non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Return the *full* linear convolution of two real vectors using FFT.
    The length of the result is len(x)+len(y)-1.
    """
    n = len(x) + len(y) - 1
    L = 1 << ((n - 1).bit_length())          # next power of two ≥ n
    fx = np.fft.rfft(x, L)
    fy = np.fft.rfft(y, L)
    conv = np.fft.irfft(fx * fy, L)[:n]
    return conv


def _conv_and_max(a: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Convolve a with itself, returning both the full array and its maximum.
    """
    conv = _full_convolution_fft(a, a)
    return conv, float(conv.max())


def _softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Given a convolution array, compute a soft‑max weight vector.
    The weights sum to 1 and concentrate on the largest entries as λ grows.
    """
    m = conv.max()
    # subtract max to keep exponentials ≤ 1 → no overflow
    exp_term = np.exp(lam * (conv - m))
    w = exp_term / exp_term.sum()
    return w


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute ∇_a  Σ_k w_k (a*a)[k]  = 2·(rev(a) * w) over the appropriate slice.
    Returns a vector of the same length as a.
    """
    a_rev = a[::-1]
    full = _full_convolution_fft(a_rev, w)           # length = 3·n‑2
    n = a.shape[0]
    # slice that corresponds to indices 0 … n‑1 in the original ordering
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


def _power_law_sequence(n: int, p: float) -> np.ndarray:
    """Return a normalised power‑law vector a_i ∝ (i+1)^‑p (sum = 1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = i ** (-p)
    seq /= seq.sum()
    return seq


def _random_dirichlet(n: int, alpha: float = 1.0) -> np.ndarray:
    """Sample a random point on the simplex using a Dirichlet distribution."""
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()


# ----------------------------------------------------------------------
# Core optimisation routine for a single run
# ----------------------------------------------------------------------


def _optimize_one_run(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 20000,
    eta: float = 0.6,
    lam0: float = 8.0,
    lam_factor: float = 1.001,
    perturb_interval: int = 5000,
) -> tuple[np.ndarray, float]:
    """
    Perform an exponentiated‑gradient descent on the simplex.
    Returns the best vector (as a NumPy array) seen during the run and its
    cheap score 2·n·max(a*a).  The run stops early if the overall deadline is
    reached.
    """
    a = a0.copy()
    best_a = a.copy()
    # cheap score for the current best
    conv, mval = _conv_and_max(a)
    best_score = 2.0 * n * mval

    lam = lam0
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # --- 1. full convolution and cheap objective -----------------
        conv, max_conv = _conv_and_max(a)
        cheap_score = 2.0 * n * max_conv
        if cheap_score < best_score:
            best_score = cheap_score
            best_a = a.copy()

        # --- 2. soft‑max weights ------------------------------------
        w = _softmax_weights(conv, lam)

        # --- 3. gradient of the smoothed max -----------------------
        grad = _gradient_from_weights(a, w)

        # --- 4. exponentiated‑gradient update + projection ----------
        # a_i ← a_i·exp(‑η·grad_i)  (gradient points towards larger loss)
        a = a * np.exp(-eta * grad)
        # re‑normalise onto the simplex (keeps Σa = 1, a_i ≥ 0)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / a_sum

        # --- 5. gradually sharpen the soft‑max ---------------------
        lam = min(lam * lam_factor, 200.0)

        # --- 6. occasional random perturbation to escape plateaus ---
        if perturb_interval and (it % perturb_interval == 0):
            pert = 0.02 * np.random.rand(n)
            a = _simplex_projection(a + pert)

    return best_a, best_score
