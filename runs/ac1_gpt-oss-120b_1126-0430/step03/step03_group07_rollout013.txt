def compute_gradient(a, gamma):
    # a shape (n,)
    n = a.shape[0]
    # compute conv = a*a via FFT
    m = 1 << ((2*n-1).bit_length())
    fa = np.fft.rfft(a, m)
    conv = np.fft.irfft(fa*fa, m)[:2*n-1]   # length L
    # compute softmax weights
    # to avoid overflow, subtract max conv * gamma: compute w = exp(gamma*(conv - max(conv))) then normalize.
    maxc = conv.max()
    exp_vals = np.exp(gamma * (conv - maxc))
    w = exp_vals / exp_vals.sum()
    # compute cross-correlation g = 2 * (corr(a, w))
    # compute convolution of reversed a with w
    a_rev = a[::-1]
    L = 2*n-1
    conv_len = n + L -1
    m2 = 1 << ((conv_len).bit_length())
    fa_rev = np.fft.rfft(a_rev, m2)
    fw = np.fft.rfft(w, m2)
    corr_full = np.fft.irfft(fa_rev*fw, m2)[:conv_len]  # length conv_len
    # g_a = 2 * corr_full[n-1:n-1+n]
    g_a = 2.0 * corr_full[n-1:n-1+n]

"""
Propose a low‑score step‑function.

Algorithm
---------
* Generate a diverse collection of symmetric seeds (Beta, sin‑power,
  random Dirichlet, and the previously best known construction if it
  exists).
* For each seed run a fast Adam‑style optimiser that works on the
  log‑softmax representation of the vector (guarantees non‑negativity
  and ∑a=1).  The objective is a smooth approximation of
      2·n·max(conv(a,a))
  using a soft‑max (temperature γ) of the convolution.  γ is increased
  linearly during optimisation so the early stage follows a smooth
  landscape and the final stage targets the true max.
* Keep the best vector seen across all seeds.
* Upsample the best vector to the final length (≈5000) and run a
  longer final optimisation phase with a higher temperature schedule.
* Return the best vector as a plain Python list.

All heavy work uses FFT‑based convolution and works entirely with
NumPy.  The optimiser respects symmetry (a[i]=a[-i‑1]) which is known
to be optimal for the continuous version of the problem.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast metric (identical to the provided evaluate_sequence but without
# the security checks).  Returns the value of 2·n·max(conv(a,a)) /
# (sum(a)**2).  The input is assumed to be a 1‑D NumPy array.
# ----------------------------------------------------------------------
def _fast_evaluation(a: np.ndarray) -> float:
    n = a.shape[0]
    if n == 0:
        return float("inf")
    # convolution length = 2·n‑1
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len).bit_length())
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[:conv_len]  # linear convolution
    max_val = conv.max()
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_val / (s * s))


# ----------------------------------------------------------------------
# Seed generators (all produce a probability vector, i.e. sum = 1)
# ----------------------------------------------------------------------
def _beta_sequence(alpha: float, beta: float, length: int) -> np.ndarray:
    xs = (np.arange(length, dtype=float) + 0.5) / length
    seq = np.power(xs, alpha - 1.0, out=np.empty_like(xs))
    seq *= np.power(1.0 - xs, beta - 1.0, out=np.empty_like(xs))
    seq[np.isinf(seq)] = 0.0
    seq = np.maximum(seq, 0.0)
    if seq.sum() == 0.0:
        seq = np.full(length, 1.0 / length)
    else:
        seq /= seq.sum()
    return seq


def _sin_power_sequence(p: float, length: int) -> np.ndarray:
    xs = (np.arange(length, dtype=float) + 0.5) / length
    vals = np.sin(np.pi * xs) ** p
    vals = np.maximum(vals, 0.0)
    if vals.sum() == 0.0:
        vals = np.full(length, 1.0 / length)
    else:
        vals /= vals.sum()
    return vals


def _random_dirichlet(length: int, concentration: float = 0.5) -> np.ndarray:
    seq = np.random.gamma(concentration, 1.0, size=length)
    if seq.sum() == 0.0:
        seq = np.full(length, 1.0 / length)
    else:
        seq /= seq.sum()
    return seq


# ----------------------------------------------------------------------
# Helper: cross‑correlation a ⋆ w (linear, length = n)
#   g_i = Σ_j a_j w_{i+j}
# This is implemented via FFT convolution of reversed a with w.
# ----------------------------------------------------------------------
def _cross_corr(a: np.ndarray, w: np.ndarray, fft_len_cross: int) -> np.ndarray:
    """Return 2·Σ_j a_j w_{i+j} for i=0,…,n‑1 (the factor 2 is omitted)."""
    n = a.shape[0]
    # reversed a
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_cross)
    fw = np.fft.rfft(w, fft_len_cross)
    full = np.fft.irfft(fa_rev * fw, fft_len_cross)
    # the full linear convolution length = n + (2n‑1) – 1 = 3n‑2
    # slice the centre part that corresponds to indices i = 0 … n‑1
    start = n - 1
    return full[start:start + n]


# ----------------------------------------------------------------------
# Adam optimiser on the log‑softmax parameters with a temperature schedule.
# ----------------------------------------------------------------------
def _adam_optimize(
    init_seq: np.ndarray,
    deadline: float,
    start_gamma: float = 5.0,
    end_gamma: float = 200.0,
    base_lr: float = 0.18,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> np.ndarray:
    """
    Refine `init_seq` (non‑negative, sum≈1) using Adam on the log‑softmax
    representation.  Returns the best vector (as NumPy array) seen before
    the deadline.
    """
    n = init_seq.shape[0]
    # --- pre‑compute FFT lengths -------------------------------------------------
    conv_len = 2 * n - 1
    fft_len_self = 1 << ((conv_len).bit_length())
    cross_len = n + conv_len - 1          # = 3·n‑2
    fft_len_cross = 1 << ((cross_len).bit_length())

    # --- initialise log‑variables (softmax) ------------------------------------
    # tiny epsilon avoids log(0); later we will use a numerically stable softmax
    log_x = np.log(init_seq + 1e-15)

    # Adam state
    m = np.zeros_like(log_x)
    v = np.zeros_like(log_x)
    t = 0

    best_seq = None
    best_score = float("inf")
    start_time = time.time()

    while time.time() < deadline:
        # ---- softmax to obtain a (guaranteed sum = 1) -------------------------
        max_log = np.max(log_x)
        a = np.exp(log_x - max_log)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= a_sum

        # ---- enforce symmetry (a[i] == a[-i‑1]) --------------------------------
        a = (a + a[::-1]) * 0.5
        a_sum = a.sum()
        a /= a_sum   # re‑normalise after symmetrisation

        # ---- compute self‑convolution (linear) ---------------------------------
        fa = np.fft.rfft(a, fft_len_self)
        conv = np.fft.irfft(fa * fa, fft_len_self)[:conv_len]

        # ---- soft‑max weighting of the convolution -----------------------------
        max_c = conv.max()
        # stabilised exponentials
        exp_vals = np.exp(start_gamma + (end_gamma - start_gamma) *
                          ((time.time() - start_time) / (deadline - start_time)) *
                          (conv - max_c))
        w = exp_vals / exp_vals.sum()   # shape (2·n‑1,)

        # ---- gradient of the smoothed objective w.r.t. a -----------------------
        # g_a = 2 * Σ_j a_j w_{i+j}
        g_a = 2.0 * _cross_corr(a, w, fft_len_cross)

        # ---- convert to gradient on the log‑softmax parameters ------------------
        mu = np.dot(a, g_a)                # scalar = Σ a_i g_i
        grad_x = a * (g_a - mu)            # shape (n,)

        # ---- Adam moments -------------------------------------------------------
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # ---- adaptive step (log‑space update) ---------------------------------
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)
        log_x -= step

        # ---- keep log‑variables symmetric ---------------------------------------
        log_x = (log_x + log_x[::-1]) * 0.5

        # ---- evaluate current candidate -----------------------------------------
        cur_score = _fast_evaluation(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()

        # ---- occasional tiny jitter to escape flat regions ----------------------
        if np.random.rand() < 0.001:
            idx = np.random.randint(n)
            perturb = np.random.randn() * 1e-3
            a[idx] = max(0.0, a[idx] + perturb)
            a /= a.sum()
            log_x = np.log(a + 1e-15)

    # fallback – if nothing improved return the last normalised vector
    if best_seq is None:
        a = np.exp(log_x - np.max(log_x))
        a /= a.sum()
        best_seq = a
    return best_seq


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Produce a step‑function (list of non‑negative floats) with a low metric.
    The routine respects the global time budget (default 1000 s) and returns
    the best vector found.
    """
    np.random.seed(seed)
    start = time.time()
    # keep a small safety margin for the harness
    deadline = start + budget_s - 2.0

    # ------------------------------------------------------------------
    # 1.  Choose a modest working length for the exploratory phase.
    # ------------------------------------------------------------------
    n_small = 1800               # fast enough for many seeds, still detailed

    # ------------------------------------------------------------------
    # 2.  Build a list of seed generators (all symmetric, normalised).
    # ------------------------------------------------------------------
    seeds = []

    # classic arcsine (Beta(0.5,0.5))
    seeds.append(("arcsine", _beta_sequence(0.5, 0.5, n_small)))

    # a few symmetric beta families around α≈0.5
    for a in [0.30, 0.35, 0.40, 0.45, 0.55, 0.60, 0.70, 0.90, 1.00]:
        seeds.append((f"beta_sym_{a:.2f}", _beta_sequence(a, a, n_small)))

    # asymmetric Beta shapes (mirrored to stay symmetric overall)
    for a, b in [(0.25, 0.75), (0.75, 0.25), (0.20, 0.80), (0.80, 0.20)]:
        # generate a half‑vector and mirror it
        half = _beta_sequence(a, b, n_small // 2 + 1)
        full = np.concatenate((half, half[-2::-1]))
        full /= full.sum()
        seeds.append((f"beta_asym_{a:.2f}_{b:.2f}", full))

    # sin‑power shapes (p negative → heavy edges, p positive → flat centre)
    for p in [-0.7, -0.5, -0.3, -0.1, 0.0, 0.2, 0.4]:
        seeds.append((f"sin_pow_{p:.1f}", _sin_power_sequence(p, n_small)))

    # random Dirichlet draws (lower concentration → spikier)
    for i in range(5):
        seeds.append((f"dirich_{i}", _random_dirichlet(n_small, concentration=0.35)))

    # incorporate the previously best known construction if present
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            prev_arr = np.array(previous, dtype=float)
            if prev_arr.size > 0:
                # interpolate / truncate to n_small
                if prev_arr.size != n_small:
                    x_old = np.arange(prev_arr.size, dtype=float)
                    x_new = np.linspace(0, prev_arr.size - 1, n_small)
                    prev_arr = np.interp(x_new, x_old, prev_arr)
                prev_arr = np.maximum(prev_arr, 0.0)
                if prev_arr.sum() == 0.0:
                    prev_arr = np.full(n_small, 1.0 / n_small)
                else:
                    prev_arr /= prev_arr.sum()
                seeds.append(("previous_best", prev_arr))
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 3.  Quick evaluation of all seeds (to order them) and prepare list.
    # ------------------------------------------------------------------
    seed_infos = []
    for name, arr in seeds:
        a = np.maximum(arr, 0.0)
        if a.sum() == 0.0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a /= a.sum()
        score = _fast_evaluation(a)
        seed_infos.append((score, name, a))

    # sort by the raw metric – the better seeds are refined first
    seed_infos.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # 4.  Run a time‑sliced Adam optimisation on each seed.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    while seed_infos and time.time() < deadline:
        remaining = deadline - time.time()
        # give each remaining seed at least a few seconds
        per_seed = max(6.0, remaining / (len(seed_infos) + 1))

        score0, name, seq0 = seed_infos.pop(0)
        # refine this seed
        seed_deadline = time.time() + per_seed
        refined = _adam_optimize(
            seq0,
            seed_deadline,
            start_gamma=5.0,
            end_gamma=120.0,
            base_lr=0.18,
        )
        refined_score = _fast_evaluation(refined)
        print(f"[seed {name}]  refined score {refined_score:.6f}")

        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined.copy()
            print(f"*** new global best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Upscale the best vector to the final length and perform final refinement.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n_small, 1.0 / n_small)

    n_big = 5000
    x_small = np.arange(best_seq.size, dtype=float)
    x_big = np.linspace(0, best_seq.size - 1, n_big)
    up_seq = np.interp(x_big, x_small, best_seq)

    up_seq = np.maximum(up_seq, 0.0)
    if up_seq.sum() == 0.0:
        up_seq = np.full(n_big, 1.0 / n_big)
    else:
        up_seq /= up_seq.sum()

    if time.time() < deadline:
        print("[final] starting long‑run refinement on full size...")
        final_seq = _adam_optimize(
            up_seq,
            deadline,
            start_gamma=20.0,
            end_gamma=300.0,
            base_lr=0.15,
        )
        final_score = _fast_evaluation(final_seq)
        print(f"[final] best score {final_score:.6f}")
        best_seq = final_seq
        best_score = final_score

    # ------------------------------------------------------------------
    # 6.  Return the best sequence as a plain Python list.
    # ------------------------------------------------------------------
    print(f"[summary] final metric {best_score:.6f} with length {best_seq.size}")
    return best_seq.tolist()
