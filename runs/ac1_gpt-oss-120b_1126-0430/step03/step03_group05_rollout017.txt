def _subgradient_topk(a, conv, K=3):
    # conv is whole conv array
    # Get indices of top K values (including ties)
    top_idx = np.argpartition(conv, -K)[-K:]
    # compute weighted average of subgradients (equal weights)
    g = np.zeros_like(a)
    n = a.size
    for idx in top_idx:
        lo = max(0, idx - (n - 1))
        hi = min(idx, n - 1)
        indices = np.arange(lo, hi + 1, dtype=np.int64)
        # gradient contribution: 2*a[idx - indices]
        g[indices] += 2 * a[idx - indices]
    # average
    g /= K
    return g

"""
Propose a non‑negative step function (list of heights) that minimises

        2·n·max_k (a*a)[k] / (Σ a_i)²

The implementation combines several cheap heuristics:

*  a short scan of power‑law shapes   a_i ∝ (i+1)^‑p
*  many random Dirichlet draws with a wide range of
   concentration parameters (to explore sparse and dense shapes)
*  a projected heavy‑ball sub‑gradient descent that uses the
   gradients of the *K* largest convolution entries (instead of only the
   single active shift).  The update is occasionally performed with an
   exponentiated‑gradient step which respects the simplex constraint
   without an explicit projection.
*  automatic restarts when progress stalls and a final exact call to
   ``evaluate_sequence`` to obtain the true score.

The routine respects the given time budget (default 1000 s) and always
returns a feasible sequence (non‑negative, sum > 0).  The code is fully
self‑contained – only ``numpy`` and the already imported ``evaluate_sequence``
are required.
"""

import time
import math

# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float) -> np.ndarray:
    """Return a normalised power‑law a_i ∝ (i+1)^‑p (Σa_i = 1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = i ** (-p)
    seq /= seq.sum()
    return seq


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using FFT (returns length 2·n‑1 array)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa).real[: 2 * a.size - 1]
    return conv


def _conv_max_fft(a: np.ndarray, fft_len: int) -> tuple[float, int]:
    """Maximum value and its index of the autocorrelation a*a."""
    conv = _conv_full_fft(a, fft_len)
    idx = int(conv.argmax())
    return float(conv[idx]), idx


def _topk_subgradient(a: np.ndarray, conv: np.ndarray, k: int = 3) -> np.ndarray:
    """
    Sub‑gradient built from the k largest convolution entries.
    For each selected shift s we add the gradient 2·a_{s‑i}.
    The result is the average of the k contributions.
    """
    n = a.shape[0]
    # indices of the k largest values (ties are broken arbitrarily)
    top_idx = np.argpartition(conv, -k)[-k:]
    g = np.zeros_like(a)
    for idx in top_idx:
        lo = max(0, idx - (n - 1))
        hi = min(idx, n - 1)
        rng = np.arange(lo, hi + 1, dtype=np.int64)
        g[rng] += 2.0 * a[idx - rng]
    g /= float(k)
    return g

            last_improve = iter_cnt
            print(f"[Iter {iter_cnt:07d}] Restart after stagnation (perturb).")

    # --------------------------------------------------------------
    # 3. Final exact evaluation (uses the provided ``evaluate_sequence``)
    # --------------------------------------------------------------
    final_seq = best_iter_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:      # defensive – should never happen
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # --------------------------------------------------------------
    # 4. Return the best sequence found.
    # --------------------------------------------------------------
    return final_seq
