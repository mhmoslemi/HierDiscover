def _smooth_gradient(a, conv, alpha, L):
    # compute softmax weights w of conv
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / exp_scaled.sum()
    # correlation of w with a: g_i = 2* sum_j w_{i+j} a_j, i from 0..n-1
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n = a.shape[0]
    grad = 2.0 * corr_rev[n - 1 : n - 1 + n]
    return grad
```

def _subgradient_max_conv(a, conv):
    n = a.shape[0]
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    g /= len(ks)
    return g

"""
A lightweight multi‑phase optimiser for the step‑function problem.

* Phase 1 – Adam on the smooth log‑sum‑exp surrogate of max‑convolution.
  The parameters are tuned for a longer run (larger α, larger learning rate)
  and the optimisation is performed on the probability simplex via a
  Euclidean projection.

* Phase 2 – Adaptive sub‑gradient descent on the true max‑convolution.
  A simple back‑tracking line‑search adapts the step size and guarantees
  a decrease whenever possible.  Occasional Gaussian jitter helps escape
  plateaus.

* Phase 3 – Simulated‑annealing hill climbing that repeatedly transfers a
  random fraction of mass between two indices.  The Metropolis criterion
  and a periodic reset to the current global best help explore the search
  space.

All helper functions are defined at module level; no closures or lambdas are
used.  The routine respects the global 1000 s budget and returns the best
sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex."""
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        # all non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full self‑convolution (length 2·n‑1)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return a*a (full convolution) using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Exact objective for a simplex‑normalised vector
def _objective_simplex(a: np.ndarray, n: int, L: int) -> float:
    """2·n·max(conv(a,a)) – a is assumed to sum to 1."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Sub‑gradient of the maximal convolution entry (used in Phase 2)
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Return the sub‑gradient of max(conv) w.r.t. a."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)          # indices achieving the maximum
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    g /= len(ks)                                   # average if several maximisers
    return g


# ----------------------------------------------------------------------
# Gradient of the smooth log‑sum‑exp surrogate
def _smooth_gradient(a: np.ndarray, conv: np.ndarray, alpha: float, L: int) -> np.ndarray:
    """
    Compute ∇_a ( (2·n/α)·log∑_k e^{α·conv_k} ).
    Result = 2·corr(w, a) where w_k ∝ e^{α·conv_k}.
    """
    # soft‑max weights (stable)
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / exp_scaled.sum()

    # correlation w ⋆ a   (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)   # length ≥ 3·n‑2
    n = a.size
    grad = 2.0 * corr_rev[n - 1 : n - 1 + n]          # central n entries
    return grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function sequence that minimises `evaluate_sequence`.
    Returns the best sequence found (list of non‑negative floats).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # small safety margin

    # --------------------------------------------------------------
    # Problem size and initial vector
    if 'height_sequence_1' in globals():
        base = np.array(height_sequence_1, dtype=float)
        n = base.size
        a = base / base.sum()
        print(f"[init] Using provided sequence, n={n}")
    else:
        n = 5000                                     # a comfortable default
        a = np.full(n, 1.0 / n)
        print(f"[init] No prior sequence – start uniform, n={n}")

    # FFT length – must accommodate both convolution (2n‑1) and correlation (3n‑2)
    L = 1 << ((3 * n - 2).bit_length())            # next power of two

    # --------------------------------------------------------------
    # Initial evaluation
    best_a = a.copy()
    best_obj = _objective_simplex(best_a, n, L)
    print(f"[init] O={best_obj:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – Adam on the smooth surrogate
    alpha_start, alpha_end = 5.0, 5000.0
    max_iter_phase1 = 200_000
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.2                 # a slightly larger base learning rate
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    iter1 = 0

    phase1_deadline = start_time + budget_s * 0.40   # allocate ~40 % of the budget

    while time.time() < phase1_deadline and iter1 < max_iter_phase1:
        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            print(f"[phase‑1] iter={iter1}, O={best_obj:.6f}")

        # linear annealing of α
        frac = iter1 / max_iter_phase1
        alpha = alpha_start + (alpha_end - alpha_start) * frac

        # gradient of the smooth surrogate
        grad = _smooth_gradient(a, conv, alpha, L)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (iter1 + 1))
        v_hat = v / (1.0 - beta2 ** (iter1 + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # projected update onto the simplex
        a_candidate = a - step
        a = _project_to_simplex(a_candidate)

        iter1 += 1

        # early exit if we are about to exceed total budget
        if time.time() >= deadline - 5.0:
            break

    # --------------------------------------------------------------
    # Phase 2 – Adaptive sub‑gradient descent on the true objective
    max_iter_subgrad = 150_000
    iter2 = 0
    step_size = 0.2                                 # initial step length

    phase2_deadline = start_time + budget_s * 0.70   # ~30 % more time

    while time.time() < phase2_deadline and iter2 < max_iter_subgrad:
        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            print(f"[subgrad] iter={iter2}, O={best_obj:.6f}")

        # sub‑gradient of the maximal entry
        g = _subgradient_max_conv(a, conv)

        # back‑tracking line‑search (max 5 attempts)
        eta = step_size
        for _ in range(5):
            a_candidate = a - eta * g
            a_candidate = _project_to_simplex(a_candidate)
            conv_cand = _conv_fft(a_candidate, L)
            obj_cand = 2.0 * n * conv_cand.max()
            if obj_cand < cur_obj:
                a = a_candidate
                step_size = min(step_size * 1.1, 0.5)   # modest increase on success
                break
            else:
                eta *= 0.5
        else:
            # no improvement – shrink step size
            step_size *= 0.5

        # occasional jitter to avoid stagnation
        if iter2 % 2000 == 0 and iter2 > 0:
            jitter = np.random.randn(n) * 1e-4
            a = _project_to_simplex(a + jitter)

        iter2 += 1

        if time.time() >= deadline - 5.0:
            break

    # --------------------------------------------------------------
    # Phase 3 – Simulated‑annealing hill climbing (global refinement)
    temperature0 = 0.02
    total_hc_iters = 0
    local_a = best_a.copy()
    local_conv = _conv_fft(local_a, L)
    local_obj = 2.0 * n * local_conv.max()

    while time.time() < deadline:
        # exponential cooling
        T = temperature0 * (0.9995 ** total_hc_iters)

        # propose a random mass‑transfer move
        i, j = np.random.randint(n, size=2)
        if i == j or local_a[i] == 0.0:
            total_hc_iters += 1
            continue
        delta = local_a[i] * np.random.rand() * 0.2   # move up to 20 % of source mass
        a_new = local_a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = _project_to_simplex(a_new)

        conv_new = _conv_fft(a_new, L)
        obj_new = 2.0 * n * conv_new.max()

        # Metropolis acceptance
        if obj_new < local_obj or np.random.rand() < math.exp((local_obj - obj_new) / max(T, 1e-12)):
            local_a = a_new
            local_obj = obj_new
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = local_a.copy()
                print(f"[anneal] iter={total_hc_iters}, O={best_obj:.6f}")

        total_hc_iters += 1

        # every few thousand steps, re‑seed the local state with the current global best
        if total_hc_iters % 5000 == 0:
            local_a = best_a.copy()
            local_obj = 2.0 * n * _conv_fft(local_a, L).max()

        # safety break if we are within a second of the deadline
        if time.time() >= deadline - 1.0:
            break

    # --------------------------------------------------------------
    # Final sanity check with the supplied evaluation function
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:
        print("[final] evaluate_sequence failed:", e)

    return best_a.tolist()
