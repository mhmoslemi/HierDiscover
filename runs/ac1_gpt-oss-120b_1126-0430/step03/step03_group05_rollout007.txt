#!/usr/bin/env python
"""
Propose a non‑negative step‑function that minimises
    2·n·max_k (a*a)[k] / (Σ a_i)²

Algorithm
---------
* Choose a moderate length (≈5 000–8 000).  Longer sequences do not
  improve the asymptotic constant but cost more time.
* Initialise with a *shifted* power‑law
        a_i ∝ (i + shift + 1)^{-p},
  where both the exponent p and the shift are swept on a grid.
* Keep the best normalised candidate (Σ a = 1) according to the cheap
  score 2·n·max(a*a) (the denominator is 1 after normalisation).
* Refine the best candidate with an accelerated projected sub‑gradient
  method that uses a *smoothed* sub‑gradient:
      – at each iteration compute the full autocorrelation by FFT,
      – take the K largest convolution entries (K≈5) and
        average their sub‑gradients (the sub‑gradient for a shift k is
        g_i = 2·a_{k‑i} on the overlapping indices),
      – apply a heavy‑ball momentum update and project onto the probability
        simplex (Σ a = 1, a_i ≥ 0).
* When progress stalls, perturb the current best vector slightly and
  restart the momentum.
* After the time budget a final exact evaluation using the provided
  ``evaluate_sequence`` is performed and the best sequence is returned.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the probability simplex Σx = 1, x ≥ 0.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # all entries non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _shifted_power_law_sequence(n: int, p: float, shift: float) -> np.ndarray:
    """
    Return a normalised shifted power‑law vector
        a_i ∝ (i + shift + 1)^{-p}   (Σ a = 1, a_i ≥ 0).
    """
    i = np.arange(n, dtype=np.float64)
    seq = (i + shift + 1.0) ** (-p)
    seq_sum = seq.sum()
    if seq_sum == 0.0:
        # fallback to uniform if something went wrong
        return np.full(n, 1.0 / n, dtype=np.float64)
    return seq / seq_sum


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full linear convolution a*a performed via FFT.
    Returns a real ndarray of length 2*len(a)-1.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa).real
    # keep only the first 2*len(a)-1 entries (the rest are padding)
    return conv[: 2 * a.size - 1]


def _conv_max_fft(a: np.ndarray, fft_len: int) -> tuple[float, int]:
    """
    Compute max value of a*a and the corresponding index using FFT.
    """
    conv = _conv_full_fft(a, fft_len)
    idx = int(conv.argmax())
    return float(conv[idx]), idx


def _multi_subgradient(a: np.ndarray, topk_idx: np.ndarray) -> np.ndarray:
    """
    Compute the average sub‑gradient over a set of active convolution
    indices (the indices in ``topk_idx``).  For each active shift k,
    the sub‑gradient entry i (if i lies in the overlapping range) is
    2·a_{k‑i}.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in topk_idx:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        # reflected part a[k-idx] is always in‑range by construction
        g[idx] += 2.0 * a[k - idx]
    # average over the selected indices
    if topk_idx.size > 0:
        g /= float(topk_idx.size)
    return g
