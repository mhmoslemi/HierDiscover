"""
propose_candidate
-----------------
Search for a low‑score non‑negative step‑function using a cheap
parameter sweep + a simplex Exponentiated‑Gradient (mirror‑descent)
refinement.

The algorithm works in three phases:

1. **Seed generation (small size).**  
   A large collection of simple parametric shapes is built on a short
   vector (n≈800).  The families are:
   * symmetric Beta(α,α) for many α∈[0.05,0.6];
   * asymmetric Beta(α,β) mirrored to enforce symmetry;
   * sin⁽ᵖ⁾(πx) for p∈[−2,0.5];
   * random Dirichlet draws (concentration<1);
   * convex mixtures of two symmetric Betas.
   Every candidate is normalised to have L¹‑norm 1.

2. **Fast evaluation & short EG refinement.**  
   After a quick evaluation on the short length the best ~30 candidates
   are kept.  Each of them is refined with a few‑hundred exponentiated‑gradient
   steps (multiplicative update on the simplex) while staying inside the
   given time budget.

3. **Upscaling & final refinement (full size).**  
   The best refined short vector is interpolated to the required length
   (n=5000).  A longer EG run (thousands of iterations) is performed on
   the full size until the overall deadline is reached.  The best vector
   encountered is returned as a plain Python list.

All operations stay in the feasible region (non‑negative, sum>0) and the
search uses only the lightweight `_fast_evaluation` routine; the final
sequence will be scored by the provided `evaluate_sequence` function.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast evaluation (identical to the supplied evaluate_sequence but without
# the security checks).  The result is a float, not np.inf for valid input.
# ----------------------------------------------------------------------
def _fast_evaluation(seq: np.ndarray) -> float:
    """Metric 2·n·max(conv(seq,seq)) / (sum(seq)²).  Assumes seq is non‑negative."""
    n = seq.size
    if n == 0:
        return float("inf")
    # next power of two for FFT (≥ 2·n‑1)
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Simple shape generators (all return a *probability* vector, i.e. sum=1)
# ----------------------------------------------------------------------
def _beta_sequence(alpha: float, beta: float, length: int) -> np.ndarray:
    """Beta(α,β) density sampled on the mid‑points of `length` bins."""
    xs = (np.arange(length, dtype=float) + 0.5) / length
    # unnormalised density
    a = np.power(xs, alpha - 1.0, out=np.empty_like(xs))
    b = np.power(1.0 - xs, beta - 1.0, out=np.empty_like(xs))
    seq = a * b
    seq[np.isinf(seq)] = 0.0
    seq = np.maximum(seq, 0.0)
    if seq.sum() == 0.0:
        seq = np.full(length, 1.0 / length)
    else:
        seq /= seq.sum()
    return seq


def _sin_power_sequence(p: float, length: int) -> np.ndarray:
    """Sequence a_i ∝ sin(π·(i+0.5)/length)ᵖ  (p may be negative)."""
    xs = (np.arange(length, dtype=float) + 0.5) / length
    vals = np.sin(np.pi * xs) ** p
    vals = np.maximum(vals, 0.0)
    if vals.sum() == 0.0:
        vals = np.full(length, 1.0 / length)
    else:
        vals /= vals.sum()
    return vals


def _random_dirichlet(length: int, concentration: float = 0.5) -> np.ndarray:
    """Draw a Dirichlet(concentration,…,concentration) point."""
    seq = np.random.gamma(concentration, 1.0, size=length)
    if seq.sum() == 0.0:
        seq = np.full(length, 1.0 / length)
    else:
        seq /= seq.sum()
    return seq


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) on the simplex.
# ----------------------------------------------------------------------
def _eg_refine(seq: np.ndarray, deadline: float,
               eta0: float = 0.6, decay: float = 1e-4,
               tol_active: float = 1e-12,
               jitter_prob: float = 0.001) -> np.ndarray:
    """
    Refine a probability vector `seq` (sum = 1) using multiplicative updates.

    The update rule is
        a_i ← a_i * exp(-η_t * g_i)
    where g_i is a sub‑gradient of the max‑convolution term.
    After each step the vector is re‑normalised.

    Parameters
    ----------
    seq : np.ndarray
        Initial probability vector (non‑negative, sum = 1).
    deadline : float
        Absolute time (seconds since epoch) after which the routine stops.
    eta0, decay : float
        Learning‑rate schedule η_t = η0 / (1 + decay * t).
    tol_active : float
        Tolerance for treating an index as “active” (i.e. achieving the max).
    jitter_prob : float
        Probability of adding a tiny random perturbation to escape plateaus.

    Returns
    -------
    np.ndarray
        The best vector seen during the run (still normalised).
    """
    n = seq.size
    # ensure proper normalisation
    seq = np.maximum(seq, 0.0)
    if seq.sum() == 0.0:
        seq = np.full(n, 1.0 / n)
    else:
        seq /= seq.sum()

    best_seq = seq.copy()
    best_score = _fast_evaluation(seq)

    # pre‑compute FFT length (power of two ≥ 2·n‑1)
    fft_len = 1 << ((2 * n - 1).bit_length())

    t = 0
    while time.time() < deadline:
        # ----- convolution & active set (max indices) -----
        fa = np.fft.rfft(seq, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        M = conv.max()
        active = np.where(conv >= M - tol_active)[0]

        # ----- sub‑gradient of the max‑convolution term -----
        g = np.zeros_like(seq)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            i_range = np.arange(lo, hi + 1, dtype=int)
            j_range = k - i_range
            g[i_range] += seq[j_range]

        # ----- learning‑rate schedule -----
        eta = eta0 / (1.0 + decay * t)

        # ----- multiplicative update (log‑space) -----
        # To keep the numbers stable we clip the exponent.
        expo = -eta * g
        np.clip(expo, -30.0, 30.0, out=expo)
        seq = seq * np.exp(expo)

        # ----- projection back onto the simplex -----
        sum_seq = seq.sum()
        if sum_seq == 0.0:
            seq = np.full(n, 1.0 / n)
        else:
            seq /= sum_seq

        # ----- optional jitter -----
        if np.random.rand() < jitter_prob:
            j = np.random.randint(n)
            seq[j] = max(0.0, seq[j] + np.random.randn() * 1e-4)
            seq /= seq.sum()

        # ----- keep the best -----
        score = _fast_evaluation(seq)
        if score < best_score - 1e-12:
            best_score = score
            best_seq = seq.copy()

        t += 1

    return best_seq


# ----------------------------------------------------------------------
# Main entry point required by the harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Produce a step‑function (list of non‑negative floats) with a low
    `evaluate_sequence` score.

    The routine uses a quick parametric sweep on a short vector,
    short EG refinements for each promising seed and a final long
    EG refinement on the full length (≈5000).  It respects the given
    time budget (default 1000 s).
    """
    np.random.seed(seed)
    start = time.time()
    # Leave a small safety margin for the harness.
    deadline = start + budget_s - 5.0

    # ------------------------------------------------------------------
    # 1.  Choose a modest working length for the exploratory phase.
    # ------------------------------------------------------------------
    n_small = 800          # fast enough for many evaluations
    # ------------------------------------------------------------------
    # 2.  Build a diverse set of seed vectors (all normalised).
    # ------------------------------------------------------------------
    seeds = []

    # symmetric Beta shapes (α from 0.05 to 0.60)
    for a in [0.05, 0.07, 0.10, 0.13, 0.15, 0.18,
              0.20, 0.25, 0.30, 0.35, 0.40, 0.45, 0.50, 0.55, 0.60]:
        seeds.append(_beta_sequence(a, a, n_small))

    # asymmetric Beta (mirrored to enforce symmetry)
    for a, b in [(0.1, 0.9), (0.15, 0.85), (0.2, 0.8),
                 (0.25, 0.75), (0.3, 0.7), (0.35, 0.65)]:
        raw = _beta_sequence(a, b, n_small)
        sym = (raw + raw[::-1]) / 2.0
        seeds.append(sym)

    # sin‑power family
    for p in [-2.0, -1.5, -1.0, -0.75, -0.5, -0.25, 0.0, 0.25, 0.5]:
        seeds.append(_sin_power_sequence(p, n_small))

    # random Dirichlet draws (concentration < 1 → spikier)
    for _ in range(8):
        seeds.append(_random_dirichlet(n_small, concentration=0.35))

    # mixtures of two symmetric Betas
    betas = [0.10, 0.20, 0.30, 0.40, 0.50, 0.60]
    for a1 in betas:
        for a2 in betas:
            if a1 >= a2:
                continue
            seq1 = _beta_sequence(a1, a1, n_small)
            seq2 = _beta_sequence(a2, a2, n_small)
            for w in [0.2, 0.5, 0.8]:
                mix = (1.0 - w) * seq1 + w * seq2
                mix /= mix.sum()
                seeds.append(mix)

    # ------------------------------------------------------------------
    # 3.  Quick evaluation of all seeds, keep the top few.
    # ------------------------------------------------------------------
    # Evaluate on the short length (fast).  Lower value ≈ lower final metric.
    scores = [_fast_evaluation(s) for s in seeds]
    # sort indices by score
    sorted_idx = np.argsort(scores)
    # keep the best 30 candidates (adjust if time is very low)
    keep = min(30, len(seeds))
    top_indices = sorted_idx[:keep]

    top_seeds = [seeds[i].copy() for i in top_indices]

    # ------------------------------------------------------------------
    # 4.  Short EG refinement for each promising seed.
    # ------------------------------------------------------------------
    best_small = None
    best_small_score = float("inf")

    # allocate roughly equal time to each top seed
    while top_seeds and time.time() < deadline:
        time_left = deadline - time.time()
        per_seed = max(2.0, time_left / (len(top_seeds) + 1))
        seq0 = top_seeds.pop(0)

        # run a short refinement (a few hundred iterations)
        seed_deadline = time.time() + per_seed
        refined = _eg_refine(seq0, seed_deadline,
                             eta0=0.8, decay=2e-4,
                             jitter_prob=0.002)
        score = _fast_evaluation(refined)

        print(f"[refine small] score {score:.6f}")

        if score < best_small_score:
            best_small_score = score
            best_small = refined.copy()
            print(f"  >>> new global small best {best_small_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Upscale the best small vector to the final length.
    # ------------------------------------------------------------------
    if best_small is None:
        # fallback – uniform distribution
        best_small = np.full(n_small, 1.0 / n_small)

    n_big = 5000
    x_small = np.arange(n_small, dtype=float)
    x_big = np.linspace(0, n_small - 1, n_big)
    upscaled = np.interp(x_big, x_small, best_small)

    upscaled = np.maximum(upscaled, 0.0)
    if upscaled.sum() == 0.0:
        upscaled = np.full(n_big, 1.0 / n_big)
    else:
        upscaled /= upscaled.sum()

    # ------------------------------------------------------------------
    # 6.  Final, longer EG refinement on the full size.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[final] starting long refinement on full size...")
        final_seq = _eg_refine(upscaled, deadline,
                               eta0=0.5, decay=1e-4,
                               jitter_prob=0.001)
    else:
        final_seq = upscaled

    final_score = _fast_evaluation(final_seq)
    print(f"[summary] final metric (fast) {final_score:.6f} with length {final_seq.size}")

    # ------------------------------------------------------------------
    # 7.  Return plain Python list (the harness will later call
    #     evaluate_sequence on it).
    # ------------------------------------------------------------------
    return final_seq.tolist()
