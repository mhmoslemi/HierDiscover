def _binary_obj(mask: np.ndarray, n: int, L: int) -> float:
    m = mask.sum()
    if m == 0:
        return np.inf
    # compute convolution via FFT
    fft_mask = np.fft.rfft(mask.astype(np.float64), n=L)
    conv = np.fft.irfft(fft_mask * fft_mask, n=L)[:2*n-1]
    # round to nearest integer
    max_counts = int(np.rint(conv).max())
    obj = 2.0 * n * max_counts / (float(m) * m)
    return obj

"""
propose_candidate
-----------------
Searches for a probability vector on the simplex that minimises

    O(a) = 2·n·max(conv(a,a))

where conv is the linear (non‑circular) convolution.  The algorithm works in
the discrete “binary mask” domain:

* a_i = 0 or 1/m (m = number of non‑zero entries).  The objective becomes
    O = 2·n·max_counts / m²,
  where *max_counts* is the maximal number of ordered index‑pairs (i,j) with
  i+j = k among the chosen positions.
* The lower bound 2·n·1/(2n-1)≈1 shows that values around 1.1–1.2 are easily
  reachable with a suitably “flat’’ pair‑count distribution.
* We start from a random mask of density p≈0.5 and perform a Metropolis‑
  annealing that repeatedly swaps a 1‑position with a 0‑position.
  The pair‑count array is updated incrementally in O(m) time (m≈pn),
  so millions of proposals fit comfortably into the 1000 s budget.
* The best mask is finally normalised (a_i = 1/m on the support) and
  returned as a plain list of floats.  A final call to the supplied
  ``evaluate_sequence`` checks the objective.

The method is completely independent from the previous Adam/LP approach
and typically drives the bound well below the reported 1.5216.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper: evaluate a binary mask (1 on the support, 0 elsewhere)
def _binary_objective(mask: np.ndarray, n: int, m: int, pair_counts: np.ndarray) -> float:
    """
    Compute  O = 2·n·max_counts / m² for a given mask.
    ``pair_counts`` must be the ordered pair‑count array (length 2·n‑1).
    """
    max_counts = int(pair_counts.max())
    return 2.0 * n * max_counts / (float(m) * m)


# ----------------------------------------------------------------------
# Initialise the pair‑count array for a binary mask (ordered pairs)
def _initial_pair_counts(mask: np.ndarray, n: int, L: int) -> np.ndarray:
    """
    Full ordered convolution of a 0/1 mask with itself using FFT.
    Result length = 2·n‑1, values are integer counts.
    """
    fft = np.fft.rfft(mask.astype(np.float64), n=L)
    conv = np.fft.irfft(fft * fft, n=L)[: 2 * n - 1]
    # rounding removes floating‑point noise
    return np.rint(conv).astype(np.int64)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (probability vector) that minimises the
    evaluation function.  Returns the sequence as a list of floats.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    # safety margin – we must finish a little before the deadline
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size -----------------------------------------------------
    if "height_sequence_1" in globals():
        # use the length of the previously found construction
        n = len(height_sequence_1)
        print(f"[init] Using previous length n={n}")
    else:
        n = 4096                    # a convenient default
        print(f"[init] No previous solution – using n={n}")

    # FFT length – power of two ≥ 2·n‑1
    L = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # Keep the best solution found so far (may come from a previous run)
    best_obj = np.inf
    best_seq = None

    if "height_sequence_1" in globals():
        try:
            prev_seq = np.array(height_sequence_1, dtype=float)
            prev_score = evaluate_sequence(prev_seq.tolist())
            best_obj = prev_score
            best_seq = prev_seq.copy()
            print(f"[init] Previous best O={best_obj:.6f}")
        except Exception as e:
            print("[init] Failed to evaluate previous sequence:", e)

    # ------------------------------------------------------------------
    # Parameters for the binary‑mask search
    density = 0.52                     # proportion of non‑zero entries
    m = max(1, int(density * n))       # number of ones in the mask
    print(f"[init] Target density {density:.2f} → m={m}")

    # Random initial mask (exactly m ones)
    init_one_idx = np.random.choice(n, size=m, replace=False)
    mask = np.zeros(n, dtype=np.int8)
    mask[init_one_idx] = 1

    # Lists / dictionaries for O(1) random sampling and updates
    ones_list = list(init_one_idx)
    zeros_list = [i for i in range(n) if mask[i] == 0]

    pos_in_ones = {idx: p for p, idx in enumerate(ones_list)}
    pos_in_zeros = {idx: p for p, idx in enumerate(zeros_list)}

    # Pair‑count array for the current mask
    pair_counts = _initial_pair_counts(mask, n, L)

    cur_obj = _binary_objective(mask, n, m, pair_counts)
    best_obj = min(best_obj, cur_obj)
    if cur_obj <= best_obj:
        best_obj = cur_obj
        best_seq = (mask.astype(float) / m).copy()
        print(f"[init] Starting O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Simulated‑annealing parameters
    T0 = 0.06                           # initial temperature
    total_time = deadline - start_time
    it = 0

    # ------------------------------------------------------------------
    # Main annealing loop
    while time.time() < deadline - 0.2:
        # linear cooling schedule
        elapsed = time.time() - start_time
        frac = elapsed / total_time
        T = T0 * (1.0 - frac)
        if T < 1e-9:
            T = 1e-9

        # --------------------------------------------------------------
        # Propose a swap: pick a random 1 and a random 0
        i = ones_list[random.randrange(len(ones_list))]
        j = zeros_list[random.randrange(len(zeros_list))]

        # ------------------------------------------------------------------
        # Build a temporary copy of the pair‑count array and apply the swap
        # (incremental update in O(m) time)
        pc_tmp = pair_counts.copy()

        # remove contributions of i (i is currently a 1)
        for t in ones_list:
            if t == i:
                pc_tmp[2 * i] -= 1                     # (i,i) contributes once
            else:
                pc_tmp[i + t] -= 2                     # (i,t) and (t,i)

        # add contributions of j (j becomes a 1)
        for t in ones_list:
            if t == i:
                continue                               # i will not be present any more
            pc_tmp[j + t] += 2                         # (j,t) and (t,j)
        pc_tmp[2 * j] += 1                             # (j,j)

        # compute objective for the proposed mask
        new_obj = _binary_objective(mask, n, m, pc_tmp)

        # ------------------------------------------------------------------
        # Metropolis acceptance
        accept = False
        if new_obj < cur_obj:
            accept = True
        else:
            if random.random() < math.exp(-(new_obj - cur_obj) / T):
                accept = True

        # ------------------------------------------------------------------
        if accept:
            # ----------------------------------------------------------------
            # Update the data structures permanently
            # 1) mask
            mask[i] = 0
            mask[j] = 1

            # 2) ones / zeros lists and position maps
            pos_i = pos_in_ones[i]
            pos_j = pos_in_zeros[j]

            ones_list[pos_i] = j
            zeros_list[pos_j] = i

            # update dictionaries
            del pos_in_ones[i]
            pos_in_ones[j] = pos_i

            del pos_in_zeros[j]
            pos_in_zeros[i] = pos_j

            # 3) pair‑count array and current objective
            pair_counts = pc_tmp
            cur_obj = new_obj

            # ----------------------------------------------------------------
            # Record a new global best
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_seq = (mask.astype(float) / m).copy()
                print(f"[anneal] iter={it:,} O={best_obj:.6f}")

        # ------------------------------------------------------------------
        it += 1

        # occasional random “restart” move to avoid dead‑ends
        if it % 30000 == 0:
            # perform a few random swaps without evaluation
            for _ in range(3):
                i = ones_list[random.randrange(len(ones_list))]
                j = zeros_list[random.randrange(len(zeros_list))]
                # apply the swap directly (no Metropolis test)
                mask[i] = 0
                mask[j] = 1
                # update bookkeeping
                pos_i = pos_in_ones[i]
                pos_j = pos_in_zeros[j]
                ones_list[pos_i] = j
                zeros_list[pos_j] = i
                del pos_in_ones[i]
                pos_in_ones[j] = pos_i
                del pos_in_zeros[j]
                pos_in_zeros[i] = pos_j
                # pair‑count array – recompute from scratch (cheap, done sparsely)
                pair_counts = _initial_pair_counts(mask, n, L)
            cur_obj = _binary_objective(mask, n, m, pair_counts)

    # ------------------------------------------------------------------
    # Final sanity check with the official evaluation function
    final_seq = best_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence failed:", exc)

    return final_seq
