"""
Propose a non‑negative step function (list of floats) that minimizes

        O = 2·n·max(conv(a,a)) / (sum(a)²)

The bound is scale‑invariant, so we restrict the search to the simplex
Σ a_i = 1.  The algorithm consists of three phases:

1. **Parametric seeding** – a coarse grid search over two simple families
   (power‑law a_i ∝ (i+β)^{-p} and exponential a_i ∝ exp(-c·i)) to obtain a
   good initial shape.

2. **Refinement by simulated annealing** – a Metropolis walk on the simplex.
   In each step a random mass δ is moved from one index to another,
   the new objective is evaluated with an FFT based convolution and the move
   is accepted w.r.t. a temperature schedule.  The walk adapts the step size
   and temperature linearly so that early moves are large and later moves
   become fine‑grained.

3. **Optional up‑sampling** – if time permits the best solution found for a
   moderate length (≈3000) is interpolated to a larger length (≈6000‑8000)
   and refined again by simulated annealing.

All helper functions are top‑level (no closures) and the total run time is
capped by the supplied budget (default 1000 s).  The final sequence is passed
to ``evaluate_sequence`` for the exact score before being returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow_two(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a power‑of‑two FFT (real‑valued output)."""
    n = a.size
    size = _next_pow_two(2 * n - 1)
    A = np.fft.fft(a, n=size)
    conv = np.fft.ifft(A * A).real[: 2 * n - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """
    Objective 2·n·max(conv(a,a)) for a simplex‑normalised vector (sum(a)=1).
    """
    n = a.size
    conv = _conv_fft(a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
def _power_law(n: int, p: float, beta: float) -> np.ndarray:
    """Generate a_i ∝ (i+β)^{-p}, i=1..n, then normalise to sum 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (idx + beta) ** (-p)
    a /= a.sum()
    return a


def _exponential(n: int, c: float) -> np.ndarray:
    """Generate a_i ∝ exp(-c·i), i=0..n‑1, then normalise."""
    idx = np.arange(n, dtype=np.float64)
    a = np.exp(-c * idx)
    a /= a.sum()
    return a


def _grid_search_initial(n: int) -> np.ndarray:
    """Return the best vector from a quick grid over the two families."""
    best_a = None
    best_val = np.inf

    # ---- power‑law grid -------------------------------------------------
    p_vals = np.linspace(0.40, 0.80, 41)          # step 0.01
    beta_vals = np.linspace(0.0, 5.0, 51)        # step 0.1
    for p in p_vals:
        for beta in beta_vals:
            a = _power_law(n, p, beta)
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_a = a

    # ---- exponential grid ---------------------------------------------
    c_vals = np.logspace(-4, 0, 40)  # from 1e‑4 to 1
    for c in c_vals:
        a = _exponential(n, c)
        val = _objective(a)
        if val < best_val:
            best_val = val
            best_a = a

    print(f"[grid] best initial O={best_val:.6f} for n={n}")
    return best_a


# ----------------------------------------------------------------------
def _simulated_annealing(
    a_start: np.ndarray,
    max_iters: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Simple SA on the simplex.
    Moves: transfer a random fraction of the source coordinate to a destination.
    """
    a = a_start.copy()
    n = a.size
    cur_val = _objective(a)

    best_a = a.copy()
    best_val = cur_val

    # temperature schedule (linear cooling) and step‑size schedule
    T0 = 0.12
    T_min = 1e-5
    step0 = 0.5
    step_min = 1e-4

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # linear schedules
        t_frac = it / max_iters
        T = max(T0 * (1 - t_frac), T_min)
        step = max(step0 * (1 - t_frac), step_min)

        # propose a mass transfer
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        # always move from the larger to the smaller entry (helps flatten)
        if a[i] > a[j]:
            src, dst = i, j
        else:
            src, dst = j, i
        max_move = a[src]
        if max_move <= 0.0:
            continue

        delta = max_move * step * np.random.rand()
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        # small negative correction due to rounding
        if a_new[src] < 0.0:
            a_new[src] = 0.0

        new_val = _objective(a_new)

        # Metropolis acceptance
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # probability exp(-(Δ)/T)
            prob = np.exp(-(new_val - cur_val) / max(T, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[SA] iter {it:7d} new best O={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
def _interpolate_to_length(a: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a simplex‑vector to a new length."""
    old_len = a.shape[0]
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    a_new = np.interp(x_new, x_old, a)
    a_new /= a_new.sum()
    return a_new

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the bound.
    Returns the sequence as a plain list of floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # small safety margin

    # ------------------------------------------------------------------
    # 1. Choose a base length (moderate size – fast for many iterations)
    n_base = 3000
    if 'height_sequence_1' in globals():
        # use the previous construction as a warm start
        a_start = np.asarray(height_sequence_1, dtype=float)
        a_start = a_start / a_start.sum()
        n_base = a_start.size
        print(f"[init] using previous sequence of length {n_base}")
    else:
        a_start = _grid_search_initial(n_base)

    # ------------------------------------------------------------------
    # 2. Refine the base sequence with simulated annealing
    # allocate about 60 % of the total budget to the base refinement
    time_left = deadline - time.time()
    iters_base = int(6e5)                     # large enough for a few seconds
    a_refined, val_refined = _simulated_annealing(a_start, iters_base, deadline)

    # ------------------------------------------------------------------
    # 3. Upsample and refine again if time permits (≈30 % of remaining budget)
    # The up‑sampled vector has more degrees of freedom and can lower O further.
    if time.time() < deadline:
        n_big = 6000
        a_big = _interpolate_to_length(a_refined, n_big)
        # give a modest number of SA iterations for the larger problem
        iters_big = int(3e5)

        # pick the better of the two refined solutions

            best_seq = a_big_ref
            best_val = val_big_ref
            print(f"[final] up‑sampled improvement: O={best_val:.6f}")
        else:
            best_seq = a_refined
            best_val = val_refined
    else:
        best_seq = a_refined
        best_val = val_refined

    # ------------------------------------------------------------------
    # 4. Final exact evaluation (the harness will also evaluate)
    try:

    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence failed:", exc)

    return best_seq.tolist()
