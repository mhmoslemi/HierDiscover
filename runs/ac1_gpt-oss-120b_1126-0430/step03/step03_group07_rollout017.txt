def _softmax_adam_refine(seq, deadline):
    n = len(seq)
    eps=1e-15
    log_seq = np.log(seq + eps)
    # Adam parameters
    beta1=0.9; beta2=0.999; lr=0.02 (maybe smaller)
    m=np.zeros_like(seq); v=np.zeros_like(seq)
    t=0
    best_seq= None; best_score=inf

    # precompute FFT lengths
    L_conv = 1 << ((2 * n - 1).bit_length())
    L_grad = 1 << ((3 * n - 2).bit_length())
    while time.time()<deadline:
        # compute a
        a = np.exp(log_seq)
        a_sum = a.sum()
        if a_sum <=0:
            a = np.full(n, 1.0/n)
            a_sum = 1.0
        a /= a_sum

        # conv via FFT
        Fa = np.fft.rfft(a, L_conv)
        conv = np.fft.irfft(Fa * Fa, L_conv)[:2*n-1]

        # compute weighted softmax
        alpha = 10.0  # could increase gradually via schedule: alpha = 10 + 40* t/(max_iters)
        # tune
        max_c = conv.max()
        shifted = conv - max_c
        w = np.exp(alpha * shifted)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0/w.size)
        else:
            w /= w_sum

        # gradient via cross-correlation
        w_rev = w[::-1]   # length 2n-1
        Fa2 = np.fft.rfft(a, L_grad)
        Fwr = np.fft.rfft(w_rev, L_grad)
        corr = np.fft.irfft(Fa2 * Fwr, L_grad)[:3*n-2]
        # gradient extraction
        idx_start = (2*n-2) - (n-1)   # = n-1?
        # we need r[(2n-2) - k] for k=0..n-1
        g = 2.0 * corr[(2*n-2) - np.arange(n)]

        # Adam moments
        t += 1
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)

        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # update log parameters

        # clip log for numerical stability
        np.clip(log_seq, -50.0, 50.0, out=log_seq)

        # evaluate

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            # optionally print progress

    # after loop
    if best_seq is None:
        a = np.exp(log_seq)
        a /= a.sum()
        best_seq = a
    return best_seq

"""
propose_candidate
-----------------
A 2‑stage optimiser for the step‑function metric

    2·n·max(conv(a,a)) / (sum(a))²

The metric is scale‑invariant; we therefore work on the simplex
∑a = 1, a ≥ 0.

Stage 1  – diverse seeds (beta, sin‑power, exponential, logistic,
Dirichlet, and the provided `height_sequence_1`).  Each seed is refined
by a *soft‑max Adam* routine that minimises a smooth approximation
of the max‑operator:
    L(a) = log‑sum‑exp(α·conv(a,a)).
The gradient of L is obtained efficiently by a single FFT‑based
cross‑correlation.  The parameters are kept in log‑space (exponentiated
gradient) and re‑normalised after each step.

Stage 2  – the best small seed is linearly interpolated to the final
length (≈5000) and refined again with the same optimiser for the
remaining time budget.

The optimiser works entirely with NumPy/FFT, evaluates the exact
metric only when reporting progress and therefore stays well within
the 1000 s limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast metric (identical to evaluate_sequence but without the safety checks)
# ----------------------------------------------------------------------
def _fast_metric(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) / (sum(seq)²).  seq is assumed non‑negative."""
    n = seq.size
    if n == 0:
        return float("inf")
    # FFT length for the self‑convolution
    L = 1 << ((2 * n - 1).bit_length())
    Fa = np.fft.rfft(seq, L)
    conv = np.fft.irfft(Fa * Fa, L)[: 2 * n - 1]
    max_b = conv.max()
    s = seq.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------
def _beta_sequence(alpha: float, beta: float, length: int) -> np.ndarray:
    xs = (np.arange(length, dtype=float) + 0.5) / length
    a = np.power(xs, alpha - 1.0, out=np.empty_like(xs))
    b = np.power(1.0 - xs, beta - 1.0, out=np.empty_like(xs))
    seq = a * b
    seq[np.isinf(seq)] = 0.0
    seq = np.maximum(seq, 0.0)
    if seq.sum() == 0.0:
        seq = np.full(length, 1.0 / length)
    else:
        seq /= seq.sum()
    return seq


def _sin_power_sequence(p: float, length: int) -> np.ndarray:
    xs = (np.arange(length, dtype=float) + 0.5) / length
    vals = np.sin(np.pi * xs) ** p
    vals = np.maximum(vals, 0.0)
    if vals.sum() == 0.0:
        vals = np.full(length, 1.0 / length)
    else:
        vals /= vals.sum()
    return vals


def _exponential_sequence(rate: float, length: int) -> np.ndarray:
    xs = np.arange(length, dtype=float)
    seq = np.exp(-rate * xs)
    if seq.sum() == 0.0:
        seq = np.full(length, 1.0 / length)
    else:
        seq /= seq.sum()
    return seq


def _logistic_sequence(mid: float, scale: float, length: int) -> np.ndarray:
    xs = np.arange(length, dtype=float)
    seq = 1.0 / (1.0 + np.exp((xs - mid) / scale))
    seq = np.maximum(seq, 0.0)
    if seq.sum() == 0.0:
        seq = np.full(length, 1.0 / length)
    else:
        seq /= seq.sum()
    return seq


def _random_dirichlet(length: int, concentration: float = 0.5) -> np.ndarray:
    seq = np.random.gamma(concentration, 1.0, size=length)
    if seq.sum() == 0.0:
        seq = np.full(length, 1.0 / length)
    else:
        seq /= seq.sum()
    return seq


# ----------------------------------------------------------------------
# Soft‑max Adam optimiser on the simplex (log‑variables)
# ----------------------------------------------------------------------
def _softmax_adam_refine(seq: np.ndarray, deadline: float) -> np.ndarray:
    """
    Refine ``seq`` (non‑negative, sum ≈ 1) using an Adam‑style
    multiplicative update on log‑coefficients.  The loss is a soft‑max
    approximation of max(conv(seq,seq)).
    """
    n = seq.size
    eps = 1e-15
    log_a = np.log(seq + eps)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    lr = 0.02
    m = np.zeros_like(seq)
    v = np.zeros_like(seq)

    # FFT lengths (next powers of two)
    L_conv = 1 << ((2 * n - 1).bit_length())
    L_grad = 1 << ((3 * n - 2).bit_length())

    best_seq = None
    best_score = float("inf")
    t = 0

    while time.time() < deadline:
        # ----- current point on the simplex -----
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
            s = 1.0
        a /= s

        # ----- self‑convolution (FFT) -----
        Fa = np.fft.rfft(a, L_conv)
        conv = np.fft.irfft(Fa * Fa, L_conv)[: 2 * n - 1]
        max_b = conv.max()
        score = 2.0 * n * max_b          # sum(a)=1 ⇒ metric = 2·n·max_b
        if score < best_score - 1e-12:
            best_score = score
            best_seq = a.copy()
            # occasional progress report
            if t % 2000 == 0:
                print(f"[soft‑max Adam] iter {t:6d}  best {best_score:.6f}")

        # ----- soft‑max weights (α controls smoothness) -----
        # α grows slowly with the iteration count
        alpha = 5.0 + 0.005 * t
        if alpha > 30.0:
            alpha = 30.0
        shifted = conv - max_b                # stabilise the exponentials
        w = np.exp(alpha * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the soft‑max loss -----
        # g_k = 2 * Σ_j a_j w_{j+k}
        w_rev = w[::-1]                       # reverse for cross‑correlation
        Fa2 = np.fft.rfft(a, L_grad)
        Fwr = np.fft.rfft(w_rev, L_grad)
        corr = np.fft.irfft(Fa2 * Fwr, L_grad)[: 3 * n - 2]
        g = 2.0 * corr[(2 * n - 2) - np.arange(n)]

        # ----- Adam moment updates -----
        t += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # ----- multiplicative update in log‑space -----
        log_a -= step
        np.clip(log_a, -50.0, 50.0, out=log_a)

    if best_seq is None:
        a = np.exp(log_a)
        a /= a.sum()
        best_seq = a
    return best_seq


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Produce a low‑score step‑function.

    1. generate a wide collection of seeds (Beta, sin‑power, exponential,
       logistic, Dirichlet, and the supplied ``height_sequence_1``);
    2. run a time‑sliced soft‑max Adam refinement on each seed;
    3. keep the best, upscale to length 5000 and perform a final
       refinement with the remaining budget.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # 1.  Working length for the exploratory phase
    # ------------------------------------------------------------------
    n_small = 2500

    # ------------------------------------------------------------------
    # 2.  Build the seed list (name, array of length n_small)
    # ------------------------------------------------------------------
    seeds = []

    # classic arcsine (Beta(0.5,0.5))
    seeds.append(("arcsine", _beta_sequence(0.5, 0.5, n_small)))

    # symmetric beta family
    for a in [0.30, 0.35, 0.40, 0.45, 0.55, 0.60, 0.70, 0.80, 1.0]:
        seeds.append((f"beta_sym_{a:.2f}", _beta_sequence(a, a, n_small)))

    # asymmetric beta shapes
    for a, b in [(0.25, 0.75), (0.75, 0.25), (0.20, 0.80), (0.80, 0.20),
                 (0.15, 0.85), (0.85, 0.15)]:
        seeds.append((f"beta_{a:.2f}_{b:.2f}", _beta_sequence(a, b, n_small)))

    # sin‑power shapes
    for p in [-0.6, -0.5, -0.4, -0.3, -0.2, -0.1, 0.0, 0.1, 0.2]:
        seeds.append((f"sin_pow_{p:.1f}", _sin_power_sequence(p, n_small)))

    # exponential decays (various rates)
    for rate in [0.0005, 0.001, 0.002, 0.005, 0.010]:
        seeds.append((f"exp_{rate:.4f}", _exponential_sequence(rate, n_small)))

    # logistic curves (different centres / scales)
    for mid, scale in [(int(0.2 * n_small), 30), (int(0.4 * n_small), 30),
                       (int(0.6 * n_small), 30), (int(0.8 * n_small), 30),
                       (int(0.5 * n_small), 50)]:
        seeds.append((f"logistic_m{mid}_s{scale}", _logistic_sequence(mid, scale, n_small)))

    # random Dirichlet draws (different concentrations)
    for conc in [0.2, 0.4, 0.7]:
        for i in range(3):
            seeds.append((f"dir_{conc:.1f}_{i}", _random_dirichlet(n_small, concentration=conc)))

    # include the previously found best construction if it exists
    try:
        import __main__ as _main  # the harness places globals here
        if hasattr(_main, "height_sequence_1"):
            hs = np.array(_main.height_sequence_1, dtype=float)
            # interpolate / truncate to n_small
            x_src = np.linspace(0, len(hs) - 1, len(hs))
            x_tgt = np.linspace(0, len(hs) - 1, n_small)
            hs_small = np.interp(x_tgt, x_src, hs)
            hs_small = np.maximum(hs_small, 0.0)
            if hs_small.sum() == 0.0:
                hs_small = np.full(n_small, 1.0 / n_small)
            else:
                hs_small /= hs_small.sum()
            seeds.append(("prev_best", hs_small))
    except Exception:
        pass  # harmless if the variable is not present

    # ------------------------------------------------------------------
    # 3.  Time‑sliced refinement of each seed
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    total_seeds = len(seeds)

    while seeds and time.time() < deadline:
        time_left = deadline - time.time()
        per_seed = max(10.0, time_left / (len(seeds) + 1))

        name, seed_vec = seeds.pop(0)

        # guarantee a proper probability vector
        seed_vec = np.maximum(seed_vec, 0.0)
        if seed_vec.sum() == 0.0:
            seed_vec = np.full(n_small, 1.0 / n_small)
        else:
            seed_vec /= seed_vec.sum()

        refined = _softmax_adam_refine(seed_vec, time.time() + per_seed)
        score = _fast_metric(refined)

        print(f"[seed {name}]  score {score:.6f}")

        if score < best_score:
            best_score = score
            best_seq = refined.copy()
            print(f"*** new global best = {best_score:.6f}")

        # guard against exhausting the budget prematurely
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # 4.  Upscale the best vector to the final length and final pass
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n_small, 1.0 / n_small)

    n_big = 5000
    # linear interpolation to the target size
    x_small = np.arange(n_small, dtype=float)
    x_big = np.linspace(0, n_small - 1, n_big)
    up_seq = np.interp(x_big, x_small, best_seq)

    up_seq = np.maximum(up_seq, 0.0)
    if up_seq.sum() == 0.0:
        up_seq = np.full(n_big, 1.0 / n_big)
    else:
        up_seq /= up_seq.sum()

    if time.time() < deadline:
        print("[final] starting full‑size refinement...")
        final_seq = _softmax_adam_refine(up_seq, deadline)
        final_score = _fast_metric(final_seq)
        print(f"[final] best score {final_score:.6f}")
        best_seq = final_seq
        best_score = final_score

    # ------------------------------------------------------------------
    # 5.  Return the result as a plain Python list
    # ------------------------------------------------------------------
    print(f"[summary] final metric {best_score:.6f} with length {best_seq.size}")
    return best_seq.tolist()
