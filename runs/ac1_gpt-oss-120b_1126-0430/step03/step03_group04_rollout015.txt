# Helper: projection onto the simplex {x >= 0, sum(x)=1}
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:                     # all non‑positive → uniform vector
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute   g_i = Σ_t w[i+t]·a[t]   for i = 0,…,n‑1
    (correlation of w with a, no reversal of a).  Implemented via FFT.
    """
    n = a.shape[0]
    m = w.shape[0]               # = 2n‑1
    L = _next_pow2(m + n - 1)    # length for linear convolution
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective O = 2·n·max(conv(a,a)) / (sum(a)²)."""
    n = a.size
    conv = _conv_self_fft(a)
    return 2.0 * n * conv.max() / (a.sum() ** 2)


# ----------------------------------------------------------------------
def _init_mixture(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Initialise a mixture of a uniform component and a decreasing
    geometric component.  The mixture weight ``u`` and the geometric
    ratio ``r`` are chosen heuristically.
    """
    # mixture weight (how much uniform we keep)
    u = 0.12
    # geometric ratio – a value a little below 1 gives a soft decay
    r = 0.95

    # uniform part
    uniform = np.full(n, 1.0 / n)

    # geometric part (decreasing)
    geom = np.arange(n, dtype=float)
    geom = r ** geom
    geom = geom * (1.0 - r) / (1.0 - r ** n)   # normalise to sum 1

    mix = u * uniform + (1.0 - u) * geom
    mix = mix / mix.sum()
    return mix


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` bound.  The method combines a sub‑gradient
    descent on the true maximum of the convolution with back‑tracking,
    exponential‑gradient updates on the simplex and occasional random
    restarts.  The best sequence found within the time budget is
    returned as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Choose a set of lengths to explore.  Larger n gives a lower
    # theoretical bound but costs more per iteration.
    lengths = [2600, 4600, 6600]   # three lengths fitting comfortably in the budget

    best_overall_seq = None
    best_overall_score = np.inf

    for n in lengths:
        # --------------------------------------------------------------
        # Initial candidate – use the global best if available, otherwise
        # a mixture of uniform + geometric.
        if 'height_sequence_1' in globals():
            a0 = np.asarray(height_sequence_1, dtype=float)
            # if the stored sequence has a different length, resample it
            if a0.size != n:
                idx = np.linspace(0, a0.size - 1, n).astype(int)
                a0 = a0[idx]
        else:
            a0 = _init_mixture(n, rng)

        a = a0.copy()
        a = a / a.sum()          # ensure exact simplex
        # store the best for this length
        best_seq = a.copy()
        best_score = _objective(best_seq)

        # ADAM‑style accumulators for momentum (optional – can be switched off)
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1, beta2, eps = 0.9, 0.999, 1e-8

        # parameters for the sub‑gradient method
        eta0 = 0.45                # initial step size (will be adapted)
        top_k = 5                  # number of entries used to form the sub‑gradient
        no_improve_cnt = 0
        max_no_improve = 15000    # after this many steps we restart

        it = 0
        while time.time() < deadline and it < 300_000:
            it += 1

            # -----------------------------------------------------------
            # 1) current objective and best‑so‑far update
            conv = _conv_self_fft(a)
            M = conv.max()
            cur_score = 2.0 * n * M / (a.sum() ** 2)
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_seq = a.copy()
                no_improve_cnt = 0
                print(f"[n={n:05d} it={it:06d}] new best O={best_score:.6f}")
            else:
                no_improve_cnt += 1

            # -----------------------------------------------------------
            # 2) build a sub‑gradient: average over the top‑k convolution entries
            #    (this approximates the true sub‑gradient of the max)
            if top_k >= conv.size:
                w = np.ones_like(conv) / conv.size
            else:
                top_idx = np.argpartition(conv, -top_k)[-top_k:]
                w = np.zeros_like(conv)
                w[top_idx] = 1.0 / top_k
            # gradient of the (average of) top‑k entries
            g = 2.0 * _correlation_fft(w, a)

            # -----------------------------------------------------------
            # 3) Adam momentum update (on the logits) – we stay on the simplex
            #    by using an exponential‑gradient step after the momentum term.
            m = beta1 * m + (1.0 - beta1) * g
            v = beta2 * v + (1.0 - beta2) * (g ** 2)
            it_float = float(it)
            m_hat = m / (1.0 - beta1 ** it_float)
            v_hat = v / (1.0 - beta2 ** it_float)

            # propose an exponential‑gradient update
            step = eta0 / (np.sqrt(v_hat) + eps)
            a_candidate = a * np.exp(-step * m_hat)   # element‑wise
            a_candidate = a_candidate / a_candidate.sum()

            # -----------------------------------------------------------
            # 4) back‑tracking line search on the *exact* objective
            #    (using the true max, not the soft‑max)
            eta_bt = 1.0
            while eta_bt > 1e-4:
                a_new = a * np.exp(-eta_bt * step * m_hat)
                a_new = a_new / a_new.sum()
                conv_new = _conv_self_fft(a_new)
                O_new = 2.0 * n * conv_new.max() / (a_new.sum() ** 2)

                if O_new <= cur_score:
                    a = a_new
                    break
                eta_bt *= 0.5
            # if line‑search failed we keep the current a (no update)

            # -----------------------------------------------------------
            # 5) occasional random jitter to escape flat regions
            if it % 3000 == 0:
                jitter = rng.normal(scale=0.001, size=n)
                a = a + jitter
                a = _project_to_simplex(a)

            # -----------------------------------------------------------
            # 6) restart if we have been stuck for a long while
            if no_improve_cnt >= max_no_improve:
                a = best_seq.copy()
                # inject small noise before the restart
                a += rng.normal(scale=0.002, size=n)
                a = _project_to_simplex(a)
                no_improve_cnt = 0
                print(f"[n={n}] restart after {it} iterations – re‑initialising from best.")

        # ------------------------------------------------------------------
        # Keep the best sequence across all lengths
        if best_score < best_overall_score:
            best_overall_score = best_score
            best_overall_seq = best_seq.copy()
            print(f"[global] new overall best using n={n}: O={best_overall_score:.6f}")

    # ----------------------------------------------------------------------
    # Final validation via the provided evaluate_sequence function
    try:
        final_val = evaluate_sequence(best_overall_seq.tolist())
        print(f"[final] evaluate_sequence returns O={final_val:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence failed:", exc)

    return best_overall_seq.tolist()
