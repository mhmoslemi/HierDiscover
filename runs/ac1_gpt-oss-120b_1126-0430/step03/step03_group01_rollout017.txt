def _smooth_max_and_weights(conv, beta):
    max_val = conv.max()
    # Compute softmax weights for smoothing
    # To avoid overflow, subtract max_val
    exp_beta = np.exp(beta * (conv - max_val))
    sum_exp = exp_beta.sum()
    w = exp_beta / sum_exp
    # soft max value approximated as (log sum exp)/beta + max_val
    smooth_max = max_val + np.log(sum_exp) / beta
    return smooth_max, w

#!/usr/bin/env python
"""
Propose a step‑function (non‑negative sequence) that minimises

    2·n·max_k (a*a)[k] / (sum a)^2 ,

where ``a`` is the sequence.  The objective is scale‑invariant, so we work
with probability vectors (∑a=1).  The algorithm consists of three phases:

1. **Parameter sweep** – generate many simple monotone families
   (power‑law, exponential, and a product of both) on a modest size
   ``n_small`` (≈5 000).  The cheapest candidate (by the cheap FFT‑based
   bound) is kept.

2. **Refinement on the small support** – optimise the chosen candidate
   with Adam on the probability simplex.  The non‑smooth “max’’ is
   approximated by a soft‑max with a temperature β that is increased
   gradually (continuation method).  Gradients are obtained via the
   convolution theorem (two FFTs per iteration).

3. **Upscaling and long‑run optimisation** – the best small‑support
   distribution is interpolated to the target size ``n_large`` (≈20 000)
   and further optimised with the same Adam‑soft‑max scheme until the
   time budget (≈1000 s) is exhausted.

The final best sequence is evaluated with the provided ``evaluate_sequence``
and returned.  Print statements report progress and intermediate bounds.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top‑level functions only
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max; returns a probability vector (sum = 1)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it representation that yields `a` after soft‑max."""
    return np.log(a + eps)


def _fft_len_for_n(n: int) -> int:
    """Smallest power of two ≥ 2·n‑1 (efficient length for real FFT)."""
    return 1 << ((2 * n - 1).bit_length())


def _conv_max_fft(a: np.ndarray, fft_len: int) -> tuple[float, int]:
    """Return (max value, arg‑max) of the auto‑convolution a*a."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.size - 1]
    idx = int(conv.argmax())
    return float(conv.max()), idx


def _conv_vals_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full auto‑convolution a*a (real) of length 2·n‑1 using an FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.size - 1]
    return conv


def _smooth_max_and_weights(conv: np.ndarray, beta: float) -> tuple[float, np.ndarray]:
    """
    Soft‑max approximation of ``max(conv)``.
    Returns (smooth_max, weights) where ``weights`` is the soft‑max
    distribution over the convolution entries (sum = 1).
    """
    max_val = conv.max()
    # subtract the max to avoid overflow
    shifted = conv - max_val
    exp_beta = np.exp(beta * shifted)
    sum_exp = exp_beta.sum()
    weights = exp_beta / sum_exp
    smooth_max = max_val + np.log(sum_exp) / beta
    return float(smooth_max), weights


def _gradient_from_distribution(a: np.ndarray,
                                w: np.ndarray,
                                fft_len: int) -> tuple[np.ndarray, np.ndarray]:
    """
    Given a probability vector ``a`` (len n) and a weight vector ``w``
    (len 2n‑1) from the smooth max, compute:

    * ``conv_self`` – the auto‑convolution a*a  (len 2n‑1)
    * ``grad_a``   – gradient of the smooth max w.r.t. ``a`` (len n)

    Both are obtained by two FFTs (FFT of ``a`` and of ``w``) and a
    single inverse FFT.
    """
    n = a.shape[0]
    # FFT of a (real → complex)
    fa = np.fft.rfft(a, fft_len)

    # auto‑convolution a*a
    conv_self = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

    # FFT of the weight vector w
    fw = np.fft.rfft(w, fft_len)

    # convolution of a with the time‑reversed w (cross‑correlation)
    conv_rev = np.fft.irfft(fa * np.conj(fw), fft_len)

    # Linear part of a * w_rev has length n + (2n‑1) - 1 = 3n‑2
    conv_rev_lin = conv_rev[: 3 * n - 2]

    # Index that picks the appropriate element for each i:
    #   grad_i = 2 * sum_k w_k a_{k-i}
    idx_base = 2 * n - 2
    grad_a = 2.0 * conv_rev_lin[idx_base - np.arange(n)]

    return conv_self, grad_a


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    total = new_a.sum()
    if total == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= total
    return new_a


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """Normalised a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _exponential_distribution(n: int, lam: float) -> np.ndarray:
    """Normalised a_i ∝ exp(-lam·i).  ``lam`` may be zero (uniform)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _combined_distribution(n: int, p: float, lam: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}·exp(-lam·i)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-lam * i)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a

            maxb, _ = _conv_max_fft(cand, fft_len)

    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    base_alpha = 0.20

    # continuation (temperature) schedule for the smooth max
    beta_smooth = 2.0          # start very smooth
    beta_max = 200.0
    beta_inc_iter = 2000
    beta_factor = 1.4

    best_seq_small = a.copy()
    best_score_small_exact = best_score_small
    it_small = 0
    report_every = 1000

    while time.time() < deadline and it_small < 40000:
        it_small += 1

        a = _softmax(z)

        # compute convex‑smooth objective
        conv_self, g_a = _gradient_from_distribution(a, None, fft_len)  # placeholder, will be overwritten
        # Actually need weights; we compute them based on current convolution:
        conv_self = _conv_vals_fft(a, fft_len)
        smooth_max, w = _smooth_max_and_weights(conv_self, beta_smooth)

        # recompute gradient using the weights w
        _, g_a = _gradient_from_distribution(a, w, fft_len)

        # move to log‑it space (mirror descent w.r.t. KL)
        mu = np.dot(g_a, a)
        grad_z = a * (g_a - mu)

        # Adam moment updates (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it_small)
        v_hat = v / (1.0 - beta2 ** it_small)

        step_dir = - m_hat / (np.sqrt(v_hat) + eps_adam)

        # basic step size schedule: inverse‑sqrt decay
        alpha_t = base_alpha / (math.sqrt(it_small) + 1.0)

        # simple back‑tracking (max 3 halvings)
        accepted = False
        trial_alpha = alpha_t
        for _ in range(3):
            z_new = z + trial_alpha * step_dir
            z_new = np.clip(z_new, -50.0, 50.0)
            a_new = _softmax(z_new)
            conv_new = _conv_vals_fft(a_new, fft_len)
            smooth_new, _ = _smooth_max_and_weights(conv_new, beta_smooth)
            cheap_new = 2.0 * n_small * smooth_new
            if cheap_new <= smooth_max * 2.0 * n_small:
                z = z_new
                accepted = True
                break
            trial_alpha *= 0.5
        if not accepted:
            # still move a little
            z = z + alpha_t * step_dir
            z = np.clip(z, -50.0, 50.0)

        # occasional random perturbation to escape plateaux
        if it_small % 5000 == 0 and np.random.rand() < 0.2:
            perturb = 0.02 * np.random.randn(n_small)
            z += perturb
            m *= 0.5
            v *= 0.5

        # increase smoothness (β) periodically
        if it_small % beta_inc_iter == 0 and beta_smooth < beta_max:
            beta_smooth = min(beta_max, beta_smooth * beta_factor)

        # exact max (for best‑track) every few hundred iterations
        if it_small % report_every == 0:
            maxb, _ = _conv_max_fft(a, fft_len)
            cheap_exact = 2.0 * n_small * maxb
            if cheap_exact < best_score_small_exact:
                best_score_small_exact = cheap_exact
                best_seq_small = a.copy()
                print(f"[Iter {it_small:06d}] NEW BEST small cheap={cheap_exact:.6f}")

    # ---------------------------------------------------------------
    # 3. Upscale to the large support and continue optimisation
    # ---------------------------------------------------------------
    n = n_large
    a = _interpolate_distribution(best_seq_small, n)
    fft_len = _fft_len_for_n(n)

    z = _logits_from_distribution(a, eps)
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    # reset continuation parameters for the larger problem
    beta_smooth = 2.0
    beta_max = 200.0
    beta_inc_iter = 5000
    beta_factor = 1.35

    # best candidate on the large support
    maxb, _ = _conv_max_fft(a, fft_len)
    best_seq = a.copy()
    best_score = 2.0 * n * maxb

    it = 0
    while time.time() < deadline:
        it += 1
        a = _softmax(z)

        conv_self = _conv_vals_fft(a, fft_len)
        smooth_max, w = _smooth_max_and_weights(conv_self, beta_smooth)

        # gradient for the smoothed objective
        _, g_a = _gradient_from_distribution(a, w, fft_len)

        mu = np.dot(g_a, a)
        grad_z = a * (g_a - mu)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step_dir = - m_hat / (np.sqrt(v_hat) + eps_adam)
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)

        # back‑tracking line search (up to three halvings)
        accepted = False
        trial_alpha = alpha_t
        for _ in range(3):
            z_new = z + trial_alpha * step_dir
            z_new = np.clip(z_new, -50.0, 50.0)
            a_new = _softmax(z_new)
            conv_new = _conv_vals_fft(a_new, fft_len)
            smooth_new, _ = _smooth_max_and_weights(conv_new, beta_smooth)
            cheap_new = 2.0 * n * smooth_new
            if cheap_new <= smooth_max * 2.0 * n:
                z = z_new
                accepted = True
                break
            trial_alpha *= 0.5
        if not accepted:
            z = z + alpha_t * step_dir
            z = np.clip(z, -50.0, 50.0)

        # increase β from time to time (makes the approximation sharper)
        if it % beta_inc_iter == 0 and beta_smooth < beta_max:
            beta_smooth = min(beta_max, beta_smooth * beta_factor)

        # random perturbations occasionally
        if it % 5000 == 0 and np.random.rand() < 0.15:
            perturb = 0.01 * np.random.randn(n)
            z += perturb
            m *= 0.5
            v *= 0.5

        # exact evaluation for best‑track (every ~1000 iters)
        if it % 1000 == 0:
            maxb, _ = _conv_max_fft(a, fft_len)
            cheap_exact = 2.0 * n * maxb
            if cheap_exact < best_score:
                best_score = cheap_exact
                best_seq = a.copy()
                print(f"[Iter {it:06d}] NEW BEST cheap={cheap_exact:.6f}")

    # ---------------------------------------------------------------
    # 4. Final (exact) evaluation and return
    # ---------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"FINAL exact score = {final_score:.6f}")
    except Exception as exc:   # defensive – should never happen
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    return final_seq
