#!/usr/bin/env python
"""
Propose a step‑function sequence that minimises
    2·n·max(conv(a,a)) / (sum(a)^2)

The optimal continuous solution is the arcsine law:
    f(x) ∝ 1/√(x(1‑x))   on (0,1)

Its self‑convolution is exactly uniform, giving the theoretical
lower bound 2·n·max≈2·n·(1/(2n‑1)) ≈ 1.
We discretise this law, normalise to a probability vector and
search a tiny neighbourhood of the exponent (p) in
    a_i ∝ ((i+1)(n‑i))^{‑p}
by cheap FFT‑based evaluation.  The best candidate is returned.
All work respects the given time budget (≈1000 s) and uses only
NumPy – no external solvers are required.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# --------------------------------------------------------------

def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two ≥ 2·n‑1 (efficient size for real FFT)."""
    return 1 << ((2 * n - 1).bit_length())


def _conv_max_fft(a: np.ndarray, fft_len: int) -> float:
    """
    Return the maximum value of the self‑convolution a*a using an FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.size - 1]
    return float(conv.max())


def _arcsine_like_distribution(n: int, p: float = 0.5) -> np.ndarray:
    """
    Return a normalised distribution on {0,…,n‑1} with
        a_i ∝ ((i+1)*(n-i))^{-p}
    For p=0.5 this is the discrete analogue of the arcsine law.
    """
    i = np.arange(n, dtype=np.float64)
    # ((i+1)*(n-i)) is always positive for 0≤i<n
    a = ( (i + 1) * (n - i) ) ** (-p)
    total = a.sum()
    if total == 0.0:               # safety – should never happen
        a[:] = 1.0 / n
    else:
        a /= total
    return a


def _cheap_score(a: np.ndarray) -> float:
    """
    Compute the cheap proxy of evaluate_sequence:
        2·n·max(conv(a,a))   when sum(a)=1.
    """
    n = a.shape[0]
    maxb = _conv_max_fft(a, _fft_len_for_n(n))
    return 2.0 * n * maxb


# --------------------------------------------------------------
# Main entry point required by the platform
# --------------------------------------------------------------

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Generate a step‑function using a discretised arcsine distribution.
    A small one‑dimensional sweep over the exponent p is performed
    (p≈0.5 gives the continuous optimal).  The best sequence found
    within the time budget is evaluated with ``evaluate_sequence`` and
    returned.
    """
    np.random.seed(seed)
    start = time.time()
    # keep a small safety margin so that we finish before hard timeout
    deadline = start + budget_s - 2.0

    # ---------------------------------------------------------
    # 0. Choose a few candidate lengths.
    # ---------------------------------------------------------
    # The previous best used 20000 entries – we stay around that range.
    # Larger n gives a tighter theoretical bound but the exact
    # evaluation becomes slower (quadratic).  20000 is a good compromise.
    candidate_lengths = [2000, 5000, 10000, 20000]

    # ---------------------------------------------------------
    # 1. Scan exponent p in a modest grid and keep the best cheap score.
    # ---------------------------------------------------------
    best_a = None
    best_score = float("inf")
    best_n = None
    best_p = None

    # grid around the theoretical p=0.5; a slightly wider range is cheap
    p_grid = np.linspace(0.35, 0.65, 31)

    for n in candidate_lengths:
        if time.time() >= deadline:
            break
        fft_len = _fft_len_for_n(n)

        for p in p_grid:
            if time.time() >= deadline:
                break
            a = _arcsine_like_distribution(n, p)
            # cheap evaluation – sum(a)=1 by construction
            score = _cheap_score(a)

            if score < best_score:
                best_score = score
                best_a = a
                best_n = n
                best_p = p
                print(f"[{time.time()-start:6.1f}s] New best: n={n}, p={p:.4f}, cheap={score:.6f}")

    # If for some reason nothing was generated (e.g. time ran out),
    # fall back to a simple uniform distribution.
    if best_a is None:
        best_n = 2000
        best_a = np.full(best_n, 1.0 / best_n)
        best_score = _cheap_score(best_a)
        print("Fallback to uniform distribution.")

    # ---------------------------------------------------------
    # 2. Optional local refinement (few iterations of Adam).
    #    This step is cheap because we keep the same length.
    # ---------------------------------------------------------
    # As the arcsine shape is already close to optimal,
    # a light‑weight Adam run can smooth out any discretisation artefacts.
    # We stop early if the budget is exhausted.
    try:
        # Adam parameters
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-8
        base_alpha = 0.3

        # logits parametrisation (soft‑max ensures sum=1 automatically)
        def _softmax(z):
            z_max = np.max(z)
            e = np.exp(z - z_max)
            return e / e.sum()

        # initialise logits from current distribution
        z = np.log(best_a + 1e-16)

        m = np.zeros_like(z)
        v = np.zeros_like(z)

        it = 0
        last_improve = 0
        best_local_a = best_a.copy()
        best_local_score = best_score

        while time.time() < deadline and it < 30_000:
            it += 1

            a = _softmax(z)                     # probability vector, sum≈1
            maxb, max_idx = None, None          # we need max and its index
            # compute convolution and its maximum
            fft_len = _fft_len_for_n(best_n)
            fa = np.fft.rfft(a, fft_len)
            conv = np.fft.irfft(fa * fa, fft_len)[: 2 * best_n - 1]
            maxb = float(conv.max())
            max_idx = int(conv.argmax())
            cheap = 2.0 * best_n * maxb

            # keep global best seen during refinement
            if cheap < best_local_score:
                best_local_score = cheap
                best_local_a = a.copy()
                last_improve = it

            # sub‑gradient of the active convolution entry
            lo = max(0, max_idx - (best_n - 1))
            hi = min(max_idx, best_n - 1)
            g = np.zeros_like(a)
            i_vals = np.arange(lo, hi + 1, dtype=np.int64)
            g[i_vals] = 2.0 * a[max_idx - i_vals]

            # gradient in logit space (mirror descent with KL)
            mu = np.dot(g, a)
            grad_z = a * (g - mu)

            # Adam updates (bias‑corrected)
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            step_dir = - m_hat / (np.sqrt(v_hat) + eps_adam)

            # learning‑rate schedule (inverse‑sqrt decay)
            alpha = base_alpha / (math.sqrt(it) + 1.0)

            # simple back‑tracking line‑search (max 3 halvings)
            accepted = False
            trial_alpha = alpha
            for _ in range(3):
                z_new = z + trial_alpha * step_dir
                z_new = np.clip(z_new, -50.0, 50.0)   # avoid overflow
                a_new = _softmax(z_new)
                fa_new = np.fft.rfft(a_new, fft_len)
                conv_new = np.fft.irfft(fa_new * fa_new, fft_len)[: 2 * best_n - 1]
                cheap_new = 2.0 * best_n * float(conv_new.max())
                if cheap_new <= cheap:
                    z = z_new
                    accepted = True
                    break
                trial_alpha *= 0.5

            if not accepted:
                # take a small step anyway – keeps the dynamics moving
                z += alpha * step_dir
                z = np.clip(z, -50.0, 50.0)

            # occasional random perturbation to escape plateaus
            if it % 5000 == 0 and np.random.rand() < 0.2:
                z += 0.02 * np.random.randn(best_n)
                m *= 0.5
                v *= 0.5

            # early stop if no improvement for a long time
            if it - last_improve > 20000:
                break

        # after refinement keep the best seen
        best_a = best_local_a
        best_score = best_local_score
        print(f"Refinement finished after {it} iters, best cheap = {best_score:.6f}")

    except Exception as exc:
        # If anything goes wrong (e.g. numerical issues), keep the original.
        print(f"Refinement step failed: {exc}")

    # ---------------------------------------------------------
    # 3. Final exact evaluation (uses the provided evaluate_sequence)
    # ---------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}  (n={best_n}, p={best_p:.4f})")
    except Exception as exc:
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # ---------------------------------------------------------
    # 4. Return the best sequence found.
    # ---------------------------------------------------------
    return final_seq
