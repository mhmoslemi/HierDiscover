"""
Propose a step-function sequence minimizing 2·n·max(conv(a,a)) / (sum(a)²).

Algorithm overview:
  * Generate a rich pool of candidate seed distributions (Beta, sin‑power,
    random Dirichlet, and the previously‑found height_sequence_1 if available).
  * Evaluate each seed quickly (FFT‑based convolution) and keep the few
    best‑scoring ones.

    max‑operator is approximated by a temperature‑controlled soft‑max
    (log‑sum‑exp) of the self‑convolution.  The gradient of this smooth
    surrogate is computed analytically via a correlation, and the update
    is performed in log‑space (exponentiated‑gradient) to stay on the
    probability simplex.
  * The temperature (soft‑max β) is increased from a modest value to a
    large value during the run, so the optimisation first follows a
    smooth landscape and finally concentrates on the true peak.
  * After the smooth phase a short vanilla sub‑gradient Adam refinement
    (active‑set gradient) is performed to fine‑tune the result.
  * The final vector is returned as a plain Python list; all entries are
    non‑negative and sum to 1, which is scale‑invariant for the metric.
"""

def _fast_metric(seq: np.ndarray) -> float:
    """Metric 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    n = seq.size
    if n == 0:
        return float("inf")
    # next power of two for linear convolution
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))

def _conv_self(seq: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·n‑1)."""
    n = seq.size
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    return conv

def _correlation(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Cross‑correlation r[l]=∑_i a[i]·b[i+l] for l≥0.
    Returns the first len(a) values (l=0..len(a)-1)."""
    n = a.size
    L = b.size
    m = 1 << ((n + L - 1).bit_length())
    Fa = np.fft.rfft(a, m)
    Fb = np.fft.rfft(b, m)
    corr = np.fft.irfft(np.conj(Fa) * Fb, m)
    return corr[:n]

def _beta_seq(alpha: float, beta: float, length: int) -> np.ndarray:
    xs = (np.arange(length, dtype=float) + 0.5) / length
    # avoid division by zero when alpha<1 or beta<1 near endpoints
    with np.errstate(divide='ignore', invalid='ignore'):
        a = np.power(xs, alpha - 1.0)
        b = np.power(1.0 - xs, beta - 1.0)
    seq = a * b
    seq[~np.isfinite(seq)] = 0.0
    seq = np.maximum(seq, 0.0)
    tot = seq.sum()
    if tot == 0.0:
        seq = np.full(length, 1.0 / length)
    else:
        seq /= tot
    return seq

def _sin_power_seq(p: float, length: int) -> np.ndarray:
    xs = (np.arange(length, dtype=float) + 0.5) / length
    vals = np.sin(np.pi * xs) ** p
    vals = np.maximum(vals, 0.0)
    tot = vals.sum()
    if tot == 0.0:
        vals = np.full(length, 1.0 / length)
    else:
        vals /= tot
    return vals

def _rand_dirichlet_seq(length: int, concentration: float) -> np.ndarray:
    seq = np.random.gamma(concentration, 1.0, size=length)
    tot = seq.sum()
    if tot == 0.0:
        seq = np.full(length, 1.0 / length)
    else:
        seq /= tot
    return seq

    m = 1 << ((2 * n - 1).bit_length())          # power‑of‑two FFT length
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _conv_self(seq: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·n‑1)."""
    n = seq.size
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    return conv


def _correlation(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Cross‑correlation r[l] = Σ_i a[i]·b[i+l] for l ≥ 0.
    Returns the first len(a) values (l = 0 … len(a)-1).
    """
    n = a.size
    L = b.size
    m = 1 << ((n + L - 1).bit_length())
    Fa = np.fft.rfft(a, m)
    Fb = np.fft.rfft(b, m)
    corr = np.fft.irfft(np.conj(Fa) * Fb, m)
    return corr[:n]


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------
def _beta_seq(alpha: float, beta: float, length: int) -> np.ndarray:
    xs = (np.arange(length, dtype=float) + 0.5) / length
    with np.errstate(divide="ignore", invalid="ignore"):
        a = np.power(xs, alpha - 1.0)
        b = np.power(1.0 - xs, beta - 1.0)
    seq = a * b
    seq[~np.isfinite(seq)] = 0.0
    seq = np.maximum(seq, 0.0)
    s = seq.sum()
    if s == 0.0:
        seq = np.full(length, 1.0 / length)
    else:
        seq /= s
    return seq


def _sin_power_seq(p: float, length: int) -> np.ndarray:
    xs = (np.arange(length, dtype=float) + 0.5) / length
    vals = np.sin(np.pi * xs) ** p
    vals = np.maximum(vals, 0.0)
    s = vals.sum()
    if s == 0.0:
        vals = np.full(length, 1.0 / length)
    else:
        vals /= s
    return vals


def _rand_dirichlet_seq(length: int, concentration: float) -> np.ndarray:
    seq = np.random.gamma(concentration, 1.0, size=length)
    s = seq.sum()
    if s == 0.0:
        seq = np.full(length, 1.0 / length)
    else:
        seq /= s
    return seq


# ----------------------------------------------------------------------
# Adam optimiser on the *smooth* surrogate (log‑sum‑exp of the convolution)
# ----------------------------------------------------------------------
def _adam_smooth_refine(
    seq: np.ndarray,
    deadline: float,
    beta_start: float = 0.5,
    beta_end: float = 200.0,
    lr0: float = 0.05,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-15,
) -> np.ndarray:
    """
    Refine ``seq`` (non‑negative, sum = 1) using Adam on the
    soft‑max surrogate of ``max(conv)``.
    """
    n = seq.size
    log_a = np.log(np.maximum(seq, eps))

    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    t = 0

    best_seq = seq.copy()
    best_score = _fast_metric(seq)

    start = time.time()
    while time.time() < deadline:
        # current probability vector
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        # self‑convolution
        conv = _conv_self(a)                     # length 2·n‑1

        # temperature schedule (geometric increase)
        frac = (time.time() - start) / (deadline - start)
        beta = beta_start * (beta_end / beta_start) ** frac

        # soft‑max weights
        shift = conv.max()
        exp_vals = np.exp(beta * (conv - shift))
        w = exp_vals / exp_vals.sum()            # length 2·n‑1, sums to 1

        # gradient w.r.t. a (smooth objective)
        grad_a = 2.0 * _correlation(a, w)        # size n

        # log‑space gradient (exponentiated‑gradient)
        grad_log = a * grad_a

        # Adam moments
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log * grad_log)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # step size (decreases with iteration count)
        lr = lr0 / math.sqrt(t)
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # update in log‑space and clip
        log_a -= step
        np.clip(log_a, -30.0, 30.0, out=log_a)

        # occasional jitter to shake out of plateaus
        if np.random.rand() < 0.001:
            j = np.random.randint(n)
            delta = np.random.randn() * 1e-3
            a[j] = max(0.0, a[j] + delta)
            a /= a.sum()
            log_a = np.log(np.maximum(a, eps))

        # evaluate current point
        a_eval = np.exp(log_a)
        a_eval_sum = a_eval.sum()
        if a_eval_sum == 0.0:
            a_eval = np.full(n, 1.0 / n)
        else:
            a_eval /= a_eval_sum

        score = _fast_metric(a_eval)
        if score < best_score - 1e-12:
            best_score = score
            best_seq = a_eval.copy()
            print(f"[smooth Adam] iter {t:6d}  β {beta:7.3f}  best {best_score:.6f}")

    return best_seq


# ----------------------------------------------------------------------
# Sub‑gradient Adam (active‑set version) – retained for final polishing
# ----------------------------------------------------------------------
def _adam_subgrad_refine(seq: np.ndarray, deadline: float) -> np.ndarray:
    n = seq.size
    eps = 1e-15
    log_a = np.log(seq + eps)

    beta1 = 0.9
    beta2 = 0.999
    lr = 0.12
    m = np.zeros_like(seq)
    v = np.zeros_like(seq)
    t = 0

    best_seq = None
    best_score = float("inf")
    it = 0
    fft_len = 1 << ((2 * n - 1).bit_length())

    while time.time() < deadline:
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
            log_a = np.log(a)
        else:
            a /= s

        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        M = conv.max()
        tol = 1e-12
        active = np.where(conv >= M - tol)[0]

        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            i_range = np.arange(lo, hi + 1, dtype=int)
            j_range = k - i_range
            g[i_range] += a[j_range]

        grad_log = a * g

        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log * grad_log)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        log_a -= step
        np.clip(log_a, -30.0, 30.0, out=log_a)

        a = np.exp(log_a)
        a /= a.sum()
        score = _fast_metric(a)
        if score < best_score - 1e-12:
            best_score = score
            best_seq = a.copy()
            if it % 500 == 0:
                print(f"[subgrad Adam] iter {it:6d}  best {best_score:.6f}")

        if np.random.rand() < 0.001:
            j = np.random.randint(n)
            a[j] = max(0.0, a[j] + np.random.randn() * 1e-3)
            a /= a.sum()
            log_a = np.log(np.maximum(a, eps))

        it += 1

    if best_seq is None:
        a = np.exp(log_a)
        a /= a.sum()
        best_seq = a
    return best_seq


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Return a non‑negative step‑function (list of floats) of length ≈5000
    that attempts to minimise the supplied metric.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    n = 5000

    # --------------------------------------------------------------
    # 1. Build a pool of diverse candidate seeds
    # --------------------------------------------------------------
    seeds = []

    # classic arcsine (Beta(0.5,0.5))
    seeds.append(("arcsine", _beta_seq(0.5, 0.5, n)))

    # symmetric Betas covering a wide range
    for a in [0.1, 0.15, 0.2, 0.25, 0.3, 0.35, 0.4, 0.45,
              0.55, 0.6, 0.65, 0.7, 0.8, 0.9, 1.0]:
        seeds.append((f"beta_sym_{a:.2f}", _beta_seq(a, a, n)))

    # a few asymmetric Betas
    for a, b in [(0.2, 0.8), (0.8, 0.2), (0.3, 0.7), (0.7, 0.3), (0.4, 0.9), (0.9, 0.4)]:
        seeds.append((f"beta_{a:.2f}_{b:.2f}", _beta_seq(a, b, n)))

    # sin‑power shapes
    for p in [-1.0, -0.8, -0.6, -0.4, -0.2, 0.0, 0.2, 0.4]:
        seeds.append((f"sin_pow_{p:.1f}", _sin_power_seq(p, n)))

    # random Dirichlet draws with various concentrations
    for conc in [0.05, 0.1, 0.2, 0.4, 0.6]:
        for i in range(2):
            seeds.append((f"dir_{conc:.2f}_{i}", _rand_dirichlet_seq(n, conc)))

    # optionally use the previously best known construction
    try:
        prev = np.asarray(height_sequence_1, dtype=float)
        if prev.size != n:
            # linear interpolation to the required length
            x_old = np.arange(prev.size, dtype=float)
            x_new = np.linspace(0, prev.size - 1, n)
            prev = np.interp(x_new, x_old, prev)
        prev = np.maximum(prev, 0.0)
        s = prev.sum()
        if s == 0.0:
            prev = np.full(n, 1.0 / n)
        else:
            prev /= s
        seeds.append(("previous_best", prev))
    except NameError:
        pass

    # --------------------------------------------------------------
    # 2. Quick evaluation of all seeds; keep the top few
    # --------------------------------------------------------------
    scored_seeds = []
    for name, seq in seeds:
        seq = np.maximum(seq, 0.0)
        total = seq.sum()
        if total == 0.0:
            seq = np.full(n, 1.0 / n)
        else:
            seq /= total
        score = _fast_metric(seq)
        scored_seeds.append((score, name, seq))
        print(f"[seed] {name:20s}  score {score:.6f}")

    scored_seeds.sort(key=lambda x: x[0])
    top_k = 5
    top_candidates = scored_seeds[:top_k]

    # --------------------------------------------------------------
    # 3. Allocate remaining budget to intensive optimisation
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining <= 0.0:
        # fallback – return the best initial seed
        best_seq = top_candidates[0][2]
        print("[timeout] returning best initial seed")
        return best_seq.tolist()

    # 80 % of the remaining time for smooth Adam, 20 % for sub‑gradient fine‑tune
    smooth_time = remaining * 0.80
    subgrad_time = remaining * 0.20

    # --------------------------------------------------------------
    # 3a. Smooth Adam on the best seed
    # --------------------------------------------------------------
    init_seq = top_candidates[0][2]
    smooth_deadline = time.time() + smooth_time
    best_seq = _adam_smooth_refine(init_seq, smooth_deadline,
                                   beta_start=0.5, beta_end=200.0,
                                   lr0=0.05)

    # --------------------------------------------------------------
    # 3b. Short sub‑gradient Adam polishing
    # --------------------------------------------------------------
    subgrad_deadline = time.time() + subgrad_time
    best_seq = _adam_subgrad_refine(best_seq, subgrad_deadline)

    # --------------------------------------------------------------
    # 4. Final metric and return
    # --------------------------------------------------------------
    final_score = _fast_metric(best_seq)
    print(f"[final] metric {final_score:.6f} (length {best_seq.size})")
    return best_seq.tolist()
