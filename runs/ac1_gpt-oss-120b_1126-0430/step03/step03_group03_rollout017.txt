def propose_candidate(seed=123, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    best_obj = float('inf')
    best_a = None

    # try list of n's
    ns = [4096, 8192, 16384]
    for n in ns:
        if time.time() >= deadline:
            break
        L = 1 << ((3 * n - 2).bit_length())
        avg = 1.0 / (2 * n - 1)

        # init a uniform
        a = np.full(n, 1.0 / n)
        # optionally add small random noise
        a += np.random.rand(n) * 1e-3
        a = _project_to_simplex(a)

        # flatten gradient descent
        max_iters_fd = 20000
        eta0 = 1.0  # step size, might need scaling based on gradient magnitude
        beta = 0.9
        vel = np.zeros_like(a)

        for it in range(max_iters_fd):
            if time.time() >= deadline - 2.0:
                break
            conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                print(f"[n={n}] iter={it}, O={best_obj:.6f}")

            # compute error to constant avg
            err = conv - avg

            # gradient via cross-correlation
            fft_a = np.fft.rfft(a, n=L)
            fft_err = np.fft.rfft(err, n=L)
            grad_full = np.fft.irfft(np.conj(fft_a) * fft_err, n=L)
            grad = 4.0 * grad_full[:n]

            # momentum
            vel = beta * vel + grad
            eta = eta0 * (1.0 - it / max_iters_fd)
            a_candidate = a - eta * vel
            a = _project_to_simplex(a_candidate)

        # after flattening, maybe refinement via Adam on LSE (optional) but skip for brevity

    # final annealing (optional)

    if best_a is None:
        raise RuntimeError("No solution found")
    # final sanity evaluation
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print("[final] evaluate_sequence reports O=", final_score)
    except Exception as e:
        print("[final] evaluate_sequence failed:", e)

    return best_a.tolist()

"""
Propose a non‑negative step‑function (probability vector) that minimises
the evaluation function:

    O(a) = 2·n·max(conv(a,a))   with   Σ a_i = 1 .

The key idea is to *flatten* the self‑convolution: we minimise the
L2‑distance between the convolution and the constant vector
c = 1/(2n‑1)·1 (the average value of the convolution).  This gives the
gradient

    ∇ L = 4·corr(a, conv(a,a)‑c)

which can be computed efficiently with one forward FFT of `a` and one
FFT of the error term.  After each gradient step the vector is projected
onto the probability simplex.  The method quickly drives mass from the
centre of the sequence to the ends, thereby lowering the maximal
convolution entry far below the uniform value (O≈2).

The algorithm proceeds as follows

* try a few problem sizes (default 4096, 8192, 16384);
* for each size initialise with a uniform distribution (and a tiny
  random jitter);
* run many iterations of the flattening gradient descent (with simple
  momentum and a linearly decaying step size);
* keep the best vector seen over all sizes;
* finish with a short simulated‑annealing hill‑climb to escape any tiny
  plateaus;
* return the best sequence as a plain Python list.

All helper functions are top‑level; no closures or lambdas are used.
Printing is used for progress monitoring.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:            # all non‑positive – return uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear convolution a*a using an FFT of length L (real‑valued input).
    Returns the full convolution of length 2·len(a)‑1.
    """
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int, L: int) -> float:
    """Exact objective for a simplex‑normalised vector: 2·n·max(conv)."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _flatten_gradient(a: np.ndarray, conv: np.ndarray, avg: float, L: int) -> np.ndarray:
    """
    Gradient of the squared error ½·∥conv‑avg∥² with respect to `a`.
    Formula: 4·corr(a, conv‑avg).  Computed via FFT.
    """
    err = conv - avg                     # length 2·n‑1
    fft_a = np.fft.rfft(a, n=L)
    fft_err = np.fft.rfft(err, n=L)
    # cross‑correlation a ⋆ err   (real‑valued)
    corr_full = np.fft.irfft(np.conj(fft_a) * fft_err, n=L)
    grad = 4.0 * corr_full[: a.size]      # first n entries are the needed lags
    return grad

        print(f"\n=== Starting optimisation for n={n} ===")
        L = 1 << ((3 * n - 2).bit_length())      # power‑of‑two ≥ 3·n‑2
        avg = 1.0 / (2 * n - 1)                  # target constant convolution

        a = np.full(n, 1.0 / n)                  # uniform

        max_iters_flat = 30_000                     # enough for convergence
