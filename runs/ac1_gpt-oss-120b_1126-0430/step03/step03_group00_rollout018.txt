Improved search for step‑function sequence minimizing O(a).
Uses an extended Adam stage on a log‑sum‑exp surrogate with a higher
smoothness parameter and a long deterministic sub‑gradient refinement
that transfers probability mass from indices that contribute most to
the current maximal convolution entry towards indices that contribute
least.  The routine respects the 1000 s budget and works on the simplex
by Euclidean projection.
"""

import time, math, numpy as np

def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n+1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        # all non‑positive – return uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]

def _smooth_gradient(a: np.ndarray, conv: np.ndarray, alpha: float, L: int) -> np.ndarray:
    # w_k ∝ exp(alpha * conv_k)
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / exp_scaled.sum()
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n = a.size
    grad = 2.0 * corr_rev[n-1 : n-1 + n]
    return grad

def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g

    best_seq = a.copy()

    print(f"[phase‑1] start O={best_score:.6f}")

    # -------------------------------------------------------------------------

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    m = np.zeros_like(a)
    v = np.zeros_like(a)

        m = beta1 * m + (1.0 - beta1) * grad

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        lr = lr0 / math.sqrt(it + 1)

        # pick a destination index with small gradient (least responsible)
        i_dst = np.argmin(g)

        delta = min(transfer_frac * a[i_src], a[i_src] - min_transfer)
        if delta > 0:
            a[i_src] -= delta
            a[i_dst] += delta

        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:
        print("[final] evaluate_sequence failed:", e)

    return best_seq.tolist()

"""
Improved search for a non‑negative step‑function sequence that minimizes

        O(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm consists of two stages:

* **Stage 1 – Adam on a log‑sum‑exp surrogate**  
  The smooth surrogate fα(a) = (2·n/α)·log∑k exp(α·convk) approximates the
  true objective.  Adam is run for many iterations while α is increased
  linearly up to a very large value (20000) so that the surrogate becomes
  close to the exact max.  After each step the vector is projected onto the
  probability simplex.

* **Stage 2 – Deterministic sub‑gradient refinement**  
  The exact sub‑gradient of max(conv) is inexpensive to compute.
  At every iteration we move a small fraction of probability mass from the
  index with the largest sub‑gradient (i.e. the index that contributes most
  to the current maximal convolution entry) to the index with the smallest
  sub‑gradient, then re‑project onto the simplex.  This “mass‑transfer”
  step directly reduces the peak of the convolution.  Tiny random jitters
  are added occasionally to escape plateaus.

Both stages respect the 1000 s time limit; the best sequence found is
returned as a plain list of floats.  The code is deliberately self‑contained
and uses only NumPy (no external LP or optimisation libraries).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Project a vector onto the probability simplex {x >= 0, sum(x)=1}
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:                     # all entries non‑positive
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear convolution via FFT (full length = 2·n‑1)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return a*a (full linear convolution) using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Gradient of the smooth log‑sum‑exp surrogate
def _smooth_gradient(a: np.ndarray, conv: np.ndarray, alpha: float, L: int) -> np.ndarray:
    """
    ∇_a ( (2·n/α)·log∑_k e^{α·conv_k} )
    = 2·corr(w, a)  where w_k ∝ e^{α·conv_k}.
    """
    # soft‑max weights (stable)
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / exp_scaled.sum()

    # correlation w ⋆ a   (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n = a.size
    grad = 2.0 * corr_rev[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) w.r.t. a (as used in the reference implementation)
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Return the sub‑gradient of max(conv) w.r.t. a.
    For each k attaining the maximum we add the contribution
    ∂/∂a_i (∑_{i+j=k} a_i a_j) = 2·a_j .
    The result is averaged over all maximising k.
    """
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)      # possibly several maximisers
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g
