#!/usr/bin/env python
"""
Propose a non‑negative step‑function (list of heights) that minimises
the evaluation function defined in ``evaluate_sequence``.

The algorithm consists of three stages:

1. **Initialisation** – scan a fine grid of power‑law exponents and the
   previously known construction (if any).  The best candidate is kept
   normalised to sum = 1.

2. **Refinement** – accelerated projected (heavy‑ball) sub‑gradient descent.
   The sub‑gradient is a convex combination of the usual active‑index
   sub‑gradient and a *soft‑max* smoothed gradient (temperature β
   increasing during optimisation, mixing weight α increasing).  This
   gives a smoother descent direction and helps the iterates escape
   plateaus.

   The iteration uses FFTs for cheap O(n log n) convolutions and projects
   onto the probability simplex after each update.  Random perturbations
   with restart are applied when no improvement is seen for a long time.

3. **LP‑boost (optional)** – after the gradient phase a small number of
   linear‑programming steps are performed.  For the current distribution
   `f` we compute `M = max_k (f*f)[k]` and solve

       maximise   Σ b_i
       subject to  b_i ≥ 0,
                   (f * b)[k] ≤ M   for all k,

   which is a linear program (cvxpy).  The solution `b` is renormalised
   to a probability vector `g` and blended with `f` (step size `t`).
   A few such updates drastically move mass to indices where `f` is small,
   lowering the autocorrelation peak.

All steps respect the supplied time budget (default 1000 s).  The best
vector found is finally evaluated with the original ``evaluate_sequence``
and returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (sum = 1, v >= 0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                         # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float) -> np.ndarray:
    """Return a normalised power‑law vector a_i ∝ (i+1)^‑p (Σa_i = 1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = i ** (-p)
    seq /= seq.sum()
    return seq


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full linear convolution of a with itself using FFT.
    Returns the first (2*len(a)-1) entries of the circular convolution.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa).real
    return conv[: 2 * a.size - 1]


def _conv_max_fft(a: np.ndarray, fft_len: int) -> tuple[float, int]:
    """Maximum value and its index of the autocorrelation a*a (FFT based)."""
    conv = _conv_full_fft(a, fft_len)
    idx = int(conv.argmax())
    return float(conv[idx]), idx


def _subgradient_max_convolution(a: np.ndarray, max_idx: int) -> np.ndarray:
    """
    Sub‑gradient of the active autocorrelation entry (the one attaining the
    maximum).  For the active shift k we have
        (∂/∂a_i) (a*a)[k] = 2·a_{k‑i}   (zero outside the support).
    """
    n = a.shape[0]
    lo = max(0, max_idx - (n - 1))
    hi = min(max_idx, n - 1)
    idx = np.arange(lo, hi + 1, dtype=np.int64)
    g = np.zeros_like(a)
    g[idx] = 2.0 * a[max_idx - idx]
    return g


# ----------------------------------------------------------------------
# Optional LP boost (uses cvxpy)
# ----------------------------------------------------------------------
def _lp_boost(a: np.ndarray, max_iters: int = 5, t0: float = 0.15,
              solver: str = "ECOS") -> np.ndarray:
    """
    Perform a few linear‑programming updates.

    For the current distribution `a` (Σa=1) compute M = max_k (a*a)[k].
    Solve   max Σ b_i   s.t.  b ≥ 0, (a*b) ≤ M.
    The solution `b` is renormalised to a probability vector `g` and
    blended with `a` using a decreasing step size.
    """
    try:
        import cvxpy as cp
    except Exception:
        # cvxpy not available – simply skip the LP boost.
        return a

    n = a.shape[0]
    a_param = cp.Parameter(n, value=a, nonneg=True)
    # Pre‑allocate the variable once; we will reuse the problem object.
    b = cp.Variable(n, nonneg=True)
    conv_fb = cp.conv(b, a_param)            # length 2n‑1, linear in b
    # No need to keep the whole constraint list as a Python list; a single
    # vector inequality is accepted by cvxpy.
    prob = cp.Problem(cp.Maximize(cp.sum(b)), [conv_fb <= 0.0])  # placeholder

    a_cur = a.copy()
    for it in range(max_iters):
        # recompute the bound M for the current a_cur
        fft_len = 1 << ((2 * n - 1).bit_length())
        maxb, _ = _conv_max_fft(a_cur, fft_len)
        M = maxb * (1.0 + 1e-12)               # tiny slack for numerics

        # Update the parameter and the bound constraint
        a_param.value = a_cur
        # Replace the old constraint with the new bound
        prob.constraints[0] = (conv_fb <= M)

        # Solve the LP
        try:
            prob.solve(solver=solver, warm_start=True,
                       max_iters=4000, verbose=False)
        except Exception:
            break
        if b.value is None:
            break

        b_val = np.maximum(b.value, 0.0)
        sum_b = b_val.sum()
        if sum_b <= 0.0:
            break

        g = b_val / sum_b                     # renormalise to Σg = 1
        # decreasing blend factor
        t = t0 / (it + 1.0)
        a_cur = (1.0 - t) * a_cur + t * g
        a_cur = _simplex_projection(a_cur)

    return a_cur

    v = np.zeros_like(a)                  # momentum buffer

    best_iter_a = a.copy()
    best_iter_score = best_score

    # Hyper‑parameters
    eta0 = 0.7                             # initial step‑size factor
    mu = 0.88                              # momentum coefficient
    stagnation_patience = 80_000           # iterations without improvement before restart
    last_improve_iter = 0

    iter_count = 0
    while time.time() < deadline and iter_count < 1_300_000:
        iter_count += 1

        # ----- convolution and active index -----
        conv_full = _conv_full_fft(a, fft_len)       # length 2n‑1
        maxb = conv_full.max()
        max_idx = int(conv_full.argmax())
        cheap_score = 2.0 * n * maxb

        # ----- build the gradient -----
        # 1) classic sub‑gradient at the active shift
        g_active = _subgradient_max_convolution(a, max_idx)

        # 2) soft‑max smoothed gradient
        # temperature schedule: start small, increase to ~50
        beta = min(50.0, 1.0 + iter_count * (49.0 / 200_000.0))
        # mixing weight: grow from 0 to 1 over the same horizon
        alpha = min(1.0, iter_count / 200_000.0)

        # soft‑max weights (numerically stable)
        w = np.exp(beta * (conv_full - maxb))
        w_sum = w.sum()
        if w_sum == 0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # convolution of w_norm with reversed a to obtain the smooth gradient
        a_rev = a[::-1]
        fw = np.fft.fft(w_norm, grad_fft_len)
        fa_rev = np.fft.fft(a_rev, grad_fft_len)
        conv_rev = np.fft.ifft(fw * fa_rev).real
        # slice the relevant part: indices n‑1 … n‑1+n‑1
        g_soft = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # blended gradient
        g = (1.0 - alpha) * g_active + alpha * g_soft

        # ----- heavy‑ball momentum update + projection onto simplex -----
        eta = eta0 / math.sqrt(iter_count)
        v = mu * v - eta * g
        a_candidate = a + v
        a_candidate = _simplex_projection(a_candidate)

        a = a_candidate

        # ----- keep track of the best cheap score -----
        if cheap_score < best_iter_score:
            best_iter_score = cheap_score
            best_iter_a = a.copy()
            last_improve_iter = iter_count
            print(f"[Iter {iter_count:07d}] New best cheap score = {cheap_score:.6f}")

        # ----- occasional restart on stagnation -----
        if iter_count - last_improve_iter > stagnation_patience:
            perturb = 0.02 * np.random.rand(n)
            a = _simplex_projection(best_iter_a + perturb)
            v = np.zeros_like(a)               # reset momentum
            last_improve_iter = iter_count
            print(f"[Iter {iter_count:07d}] Restarted after stagnation (perturb).")

        # ----- optional LP boost after a certain amount of progress -----
        # perform at most a few LP updates near the end of the budget
        if iter_count % 250_000 == 0 and time.time() < deadline - 30:
            # a quick LP step with a modest blend factor
            a_lp = _lp_boost(a, max_iters=4, t0=0.12)
            # evaluate the LP‑updated vector
            conv_full_lp = _conv_full_fft(a_lp, fft_len)
            cheap_score_lp = 2.0 * n * conv_full_lp.max()
            if cheap_score_lp < best_iter_score:
                best_iter_score = cheap_score_lp
                best_iter_a = a_lp.copy()
                a = a_lp.copy()
                print(f"[Iter {iter_count:07d}] LP‑boost improved score to {cheap_score_lp:.6f}")

    # ------------------------------------------------------------------
    # 3. Final exact evaluation (uses the provided ``evaluate_sequence``)
    # ------------------------------------------------------------------
    final_seq = best_iter_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # ------------------------------------------------------------------
    # 4. Return the best sequence found.
    # ------------------------------------------------------------------
    return final_seq
