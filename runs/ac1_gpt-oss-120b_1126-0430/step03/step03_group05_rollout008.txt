            a = a + perturb
            a = a / a.sum()
            last_improve = iter_count
            print(f"[Smooth {iter_count}] Perturb due to stagnation.")

    # 3. Heavy-ball subgradient phase (similar to previous code)
    a = best_iter_a.copy()  # start from best found after smooth phase
    v = np.zeros_like(a)
    best_hb_a = a.copy()
    best_hb_score = best_iter_score
    last_improve_hb = 0
    eta0_hb = 0.6
    mu = 0.85

            best_hb_a = a.copy()
            last_improve_hb = iter_hb

            last_improve_hb = iter_hb

    # final best candidate

    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:

        print(f"Error during final evaluation: {exc}")

#!/usr/bin/env python
"""
Propose a step‑function that minimises
    2·n·max_k (a*a)[k] / (sum a)^2
under non‑negativity and sum‑to‑one constraints.

The algorithm consists of three stages:

1.  **Broad initial search** – scan simple parametric families
    (power‑law, truncated geometric) on the chosen length `n`
    and keep the best cheap score `2·n·max(conv)`.

2.  **Smooth gradient descent** – minimise a smoothed version of the
    objective.  The convolution values are turned into a soft‑max
    weight vector `w` (temperature `α`).  The gradient of the weighted
    sum `∑ w_k (a*a)[k]` can be obtained by one extra FFT.  An
    exponentiated‑gradient (mirror‑descent w.r.t. KL) step is performed
    and the temperature `α` is increased gradually (continuation).

3.  **Accelerated sub‑gradient (heavy‑ball) refinement** – after the
    smooth phase the problem is close to the true max‑norm.  A
    heavy‑ball update with projection onto the simplex is applied,
    exactly as in the reference implementation, with occasional random
    perturbations to escape plateaus.

The whole routine respects a user supplied time budget (default 1000 s)
and returns the best sequence found (as a plain Python list).  The
sequence is automatically normalised, so the evaluation function is
scale‑invariant.
"""

import time
import math
import numpy as np

# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (sum=1, non‑negative)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # all entries non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _subgradient_max_convolution(a: np.ndarray, max_idx: int) -> np.ndarray:
    """Sub‑gradient of max_k (a*a)[k] at the active offset ``max_idx``."""
    n = a.shape[0]
    lo = max(0, max_idx - (n - 1))
    hi = min(max_idx, n - 1)
    idx = np.arange(lo, hi + 1, dtype=np.int64)
    g = np.zeros_like(a)
    g[idx] = 2.0 * a[max_idx - idx]
    return g


def _power_law_sequence(n: int, p: float) -> np.ndarray:
    """Return a normalised power‑law vector a_i ∝ (i+1)^‑p (Σa_i = 1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = i ** (-p)
    seq /= seq.sum()
    return seq


def _truncated_geometric(n: int, r: float) -> np.ndarray:
    """Return a normalised truncated geometric vector a_i ∝ r**i (0<r<1)."""
    i = np.arange(n, dtype=np.float64)
    seq = (1 - r) * (r ** i)
    seq_sum = seq.sum()
    if seq_sum == 0:
        seq = np.full(n, 1.0 / n)
    else:
        seq /= seq_sum
    return seq


def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution a*a via real FFT (full length 2·len(a)-1)."""
    Fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Fa * Fa, n=L)
    return conv[: 2 * a.shape[0] - 1]


def _conv_weight_rev_fft(w: np.ndarray, a_rev: np.ndarray, L: int) -> np.ndarray:
    """
    Linear convolution w * a_rev via real FFT.
    Returns the full linear convolution of length len(w)+len(a_rev)-1.
    """
    Fw = np.fft.rfft(w, n=L)
    Fa_rev = np.fft.rfft(a_rev, n=L)
    conv = np.fft.ifft(Fw * Fa_rev).real
    # linear part occupies the first len(w)+len(a_rev)-1 entries
    return conv[: w.shape[0] + a_rev.shape[0] - 1]
