"""

Approach
--------
* Choose a relatively large problem size (default 20000) – larger n
  improves the asymptotic bound.
* Initialise the vector with a family of simple analytic shapes
  (power‑law and exponential decay).  Each candidate is normalised
  to sum‑one and evaluated cheaply by an FFT‑based autocorrelation;
  the best seed is kept.
* Refine the seed using *exponentiated gradient descent* on the
  probability simplex (mirror descent w.r.t. KL).  For the active
  shift `k*` where the autocorrelation attains its maximum we use the
  exact sub‑gradient `g_i = 2·a_{k*-i}` (zero outside the overlap).  
  A momentum term (heavy‑ball) accelerates convergence.
* When no improvement has been observed for a long stretch we
  inject a small random perturbation and reset the momentum –
  this helps escape plateaus.
* The cheap score `2·n·max_b` (exact when the sum is one) is monitored
  and the best vector seen so far is stored.  After the time budget a
  final exact evaluation using the provided ``evaluate_sequence`` is
  performed and the best sequence is returned.
"""

import time
import math
import numpy as np

def _power_law_sequence(n: int, p: float) -> np.ndarray:
    """Return a normalised power‑law a_i ∝ (i+1)^‑p with Σ a_i = 1."""
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = i ** (-p)
    seq /= seq.sum()
    return seq

def _exponential_sequence(n: int, r: float) -> np.ndarray:
    """Return a normalised geometric decay a_i ∝ r**i with Σ a_i = 1."""
    i = np.arange(n, dtype=np.float64)
    seq = r ** i
    seq /= seq.sum()
    return seq

def _conv_max_fft(a: np.ndarray, fft_len: int) -> tuple[float, int]:
    """Compute the full autocorrelation a*a via FFT and return its max value and index."""
    # Real FFT for speed
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len).real
    # Keep only the first 2*len(a)-1 entries (full convolution)
    conv = conv[: 2 * a.size - 1]
    max_idx = int(conv.argmax())
    return float(conv[max_idx]), max_idx

def _subgradient_max_convolution(a: np.ndarray, max_idx: int) -> np.ndarray:
    """
    Sub‑gradient of the active autocorrelation entry (a*a)[max_idx].
    For each i in the overlap we have ∂/∂a_i = 2·a_{max_idx‑i}.
    """
    n = a.shape[0]
    lo = max(0, max_idx - (n - 1))
    hi = min(max_idx, n - 1)
    idx = np.arange(lo, hi + 1, dtype=np.int64)
    g = np.zeros_like(a)
    g[idx] = 2.0 * a[max_idx - idx]
    return g

    # ------------------------------------------------------------------

    best_score = float("inf")
