        v_hat = v / (1 - beta2 ** it)

            z += perturb

            v *= 0.5

            z = np.log(best_seq + eps) + 0.1 * np.random.randn(n)
            m = np.zeros_like(z)
            v = np.zeros_like(z)
            last_improve_iter = it

        # break if time low

    try:

        print(f"Final exact score = {final_score:.6f}")
    except Exception as e:
        final_score = float('inf')
        print(f"Evaluation error: {e}")

    return final_seq

#!/usr/bin/env python
"""
Optimisation of a non‑negative step function via a smoothed
max‑convolution objective.

* The vector a lives on the probability simplex (∑a = 1) – this
  makes the cheap proxy 2·n·max(a*a) identical to the exact
  evaluation returned by ``evaluate_sequence``.
* A temperature‑controlled soft‑max over the self‑convolution
  provides a smooth approximation of the max; the exact sub‑gradient
  of the smooth objective is obtained analytically and expressed
  via a single FFT‑based convolution.
* Adam on the logit (soft‑max) parameters drives the optimisation;
  the schedule slowly reduces the temperature so the algorithm
  finishes on the true max‑objective.
* Occasional random perturbations and a simple restart mechanism
  avoid stagnation.  The code respects a 1000 s time budget
  (≈2 s safety margin) and prints occasional progress reports.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max; returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it representation that yields `a` after soft‑max."""
    return np.log(a + eps)


def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """Normalised power‑law a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    a /= a.sum()
    return a


def _cheap_score(a: np.ndarray, fft_len_self: int) -> float:
    """2·n·max(self‑convolution) for a probability vector a."""
    fa = np.fft.rfft(a, fft_len_self)
    conv = np.fft.irfft(fa * fa, fft_len_self)[: 2 * a.size - 1]
    return float(2 * a.size * conv.max())


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) yields a low
    value of ``evaluate_sequence``.  The routine runs for at most
    ``budget_s`` seconds (≈1000 s) and prints progress information.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # safety margin

    # ---------------------------------------------------------------
    # 0. Choose a working support size.
    # ---------------------------------------------------------------
    try:
        # If a global candidate exists, start from it.
        init_seq = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        n = max(5_000, min(20_000, init_seq.shape[0]))
        a = np.maximum(init_seq, 0.0)
        s = a.sum()
        if s == 0.0:
            a[:] = 1.0 / n
        else:
            a /= s
        print(f"[Init] Using global sequence, length = {n}")
    except Exception:
        # No global sequence – start from a power‑law distribution.
        n = 20_000
        best_score = float("inf")
        best_a = None
        fft_len_self = _next_pow2(2 * n - 1)
        # coarse grid of exponents
        for p in np.linspace(0.55, 0.70, 81):
            if time.time() >= deadline:
                break
            cand = _power_law_distribution(n, p)
            score = _cheap_score(cand, fft_len_self)
            if score < best_score:
                best_score = score
                best_a = cand
        if best_a is None:
            best_a = np.full(n, 1.0 / n)
            best_score = _cheap_score(best_a, fft_len_self)
        a = best_a
        print(f"[Init] Power‑law start, cheap score = {best_score:.6f}")

    # 1. Log‑it parametrisation + Adam state.
    # ---------------------------------------------------------------
    eps = 1e-16
    z = _logits_from_distribution(a, eps)

    v = np.zeros_like(z)        # second‑moment estimate

        mu = np.dot(g, a)                       # ⟨g,a⟩
        grad_z = a * (g - mu)                   # ∇_z  (KL‑projection)

        # ---- Adam moments (bias‑corrected) ----
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # ---- learning‑rate schedule (inverse‑sqrt decay) ----
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)

        # ---- Adam step in logit space ----
        step = - m_hat / (np.sqrt(v_hat) + eps_adam)
        z_new = z + alpha_t * step
        # clip logits to avoid overflow in the subsequent exp()
        z_new = np.clip(z_new, -50.0, 50.0)

        # simple back‑tracking line search (up to three halvings)
        accepted = False
        trial_alpha = alpha_t
        for _ in range(3):
            z_trial = z + trial_alpha * step
            z_trial = np.clip(z_trial, -50.0, 50.0)
            a_trial = np.exp(z_trial - np.max(z_trial))
            a_trial /= a_trial.sum()

            ft = np.fft.rfft(a_trial, fft_len_self)
            conv_trial = np.fft.irfft(ft * ft, fft_len_self)[: 2 * n - 1]
            cheap_trial = 2.0 * n * conv_trial.max()
            if cheap_trial <= cheap:
                z = z_trial
                accepted = True
                break
            trial_alpha *= 0.5

        if not accepted:
            # no descent found – still move a bit to keep the dynamics alive
            z = z_new

        # ---- temperature annealing ----
        T = max(T * T_decay, T_min)

        # ---- occasional random perturbations (helps escape plateaus) ----
        if it % 10_000 == 0 and np.random.rand() < 0.25:
            perturb = 0.015 * np.random.randn(n)
            z += perturb
            m *= 0.5
            v *= 0.5
            print(f"[Iter {it:07d}] Random perturbation applied, T={T:.2e}")

        if it - last_improve_iter > stagnation_limit:
            # re‑initialise logits around the best sequence found so far
            z = _logits_from_distribution(best_seq, eps) + 0.05 * np.random.randn(n)
            m.fill(0.0)
            v.fill(0.0)
            last_improve_iter = it
            print(f"[Iter {it:07d}] Restart due to stagnation, best cheap = {best_score:.6f}")

        print(f"Error during final evaluation: {exc}")

    # ---------------------------------------------------------------
    return final_seq
