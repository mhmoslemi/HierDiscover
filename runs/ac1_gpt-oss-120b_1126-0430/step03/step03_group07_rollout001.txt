# ----------------------------------------------------------------------
# Fast metric (identical to the supplied evaluate_sequence but without
# the defensive checks).  Used only inside the optimiser.
# ----------------------------------------------------------------------
def _fast_metric(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) / (sum(a)²)  for a (non‑negative, not all zero)."""
    n = a.size
    if n == 0:
        return float("inf")
    # FFT length = next power of two ≥ 2·n‑1
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    max_b = conv.max()
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
# Seed generators (all produce vectors that sum to 1)
# ----------------------------------------------------------------------
def _beta_seq(alpha: float, beta: float, length: int) -> np.ndarray:
    xs = (np.arange(length, dtype=float) + 0.5) / length
    a = np.power(xs, alpha - 1.0, out=np.empty_like(xs))
    b = np.power(1.0 - xs, beta - 1.0, out=np.empty_like(xs))
    seq = a * b
    seq[np.isinf(seq)] = 0.0
    seq = np.maximum(seq, 0.0)
    s = seq.sum()
    if s == 0.0:
        seq[:] = 1.0 / length
    else:
        seq /= s
    return seq


def _sin_power_seq(p: float, length: int) -> np.ndarray:
    xs = (np.arange(length, dtype=float) + 0.5) / length
    vals = np.sin(np.pi * xs) ** p
    vals = np.maximum(vals, 0.0)
    s = vals.sum()
    if s == 0.0:
        vals[:] = 1.0 / length
    else:
        vals /= s
    return vals

    s = seq.sum()
    if s == 0.0:
        seq[:] = 1.0 / length
    else:

    return seq

# Adam optimiser that works on the *soft‑max* approximation of the
# convolution peak.  The update is performed on log‑variables, guaranteeing
# positivity; after each step we renormalise to stay on the simplex.
# ----------------------------------------------------------------------
def _adam_softmax(
    init_seq: np.ndarray,
    deadline: float,
    beta_start: float = 5.0,
    beta_end: float = 400.0,
) -> np.ndarray:
    """
    Refine ``init_seq`` (∑=1, ≥0) using Adam on log‑variables.
    ``beta`` (soft‑max temperature) is increased linearly from
    ``beta_start`` to ``beta_end`` over the allotted time.
    Returns the best sequence (according to the exact metric) seen.
    """
    n = init_seq.size
    eps = 1e-15
    log_a = np.log(init_seq + eps)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    lr = 0.08

    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    t = 0

    best_seq = None
    best_score = float("inf")
    start = time.time()

    # pre‑compute FFT lengths (they stay constant during the run)
    fft_len_conv = _next_pow2(2 * n - 1)               # for a*a
    fft_len_grad = _next_pow2(3 * n - 2)               # for gradient via correlation

    while time.time() < deadline:
        # ------- obtain current probability vector --------------------
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a[:] = 1.0 / n
            s = 1.0
        a /= s

        # ------- self‑convolution (for the soft‑max) ----------------
        fa = np.fft.rfft(a, fft_len_conv)
        conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

        # ------- soft‑max weights ------------------------------------
        elapsed = time.time() - start
        total = deadline - start
        # linear annealing of β
        beta = beta_start + (beta_end - beta_start) * min(1.0, elapsed / total)

        # stable soft‑max
        max_c = conv.max()
        w = np.exp(beta * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w[:] = 1.0 / w.size
            w_sum = 1.0
        w /= w_sum

        # ------- gradient of the soft‑max peak -----------------------
        # g = 2 * corr(w, a)  (valid mode, length n)
        # Compute via FFT to keep O(n log n)
        w_rev = w[::-1]
        af = np.fft.rfft(a, fft_len_grad)
        wf = np.fft.rfft(w_rev, fft_len_grad)
        corr = np.fft.irfft(af * wf, fft_len_grad)[: 3 * n - 2]
        offset = 2 * n - 2
        g = 2.0 * corr[offset - np.arange(n)]

        # ------- Adam moment updates --------------------------------
        t += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # ------- multiplicative (log‑space) Adam step ---------------
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        log_a -= step

        # keep log‑variables in a reasonable range (prevents overflow)
        np.clip(log_a, -40.0, 40.0, out=log_a)

        # ------- occasional random jitter to escape flat regions -----
        if np.random.rand() < 0.001:
            j = np.random.randint(n)
            a[j] = max(0.0, a[j] + np.random.randn() * 1e-3)
            a /= a.sum()
            log_a = np.log(a + eps)

        # ------- evaluate exact metric and store the best -----------
        score = _fast_metric(a)
        if score < best_score - 1e-12:
            best_score = score
            best_seq = a.copy()
            # Log progress (sparse to avoid clutter)

    # If no improvement happened (unlikely) return the last normalised a
    if best_seq is None:
        a = np.exp(log_a)
        a /= a.sum()
        best_seq = a
    return best_seq
