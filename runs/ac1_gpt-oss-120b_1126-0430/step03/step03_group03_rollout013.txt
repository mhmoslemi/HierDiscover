"""
Propose a step‑function (non‑negative sequence) that minimizes

    O(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

subject to a_i ≥ 0.
The implementation follows three stages:

1. **Adam on a smooth log‑sum‑exp surrogate** – the surrogate
   f_α(a) = (2·n/α)·log Σ_k exp(α·conv_k) approximates the true
   objective for large α.  α is increased exponentially from 5 to a large
   value (≈1e4) while a is kept on the probability simplex by Euclidean
   projection.

2. **Deterministic sub‑gradient refinements** – once the surrogate is
   close to the true max, we switch to the exact sub‑gradient of the
   maximal convolution entry.  A linearly decaying step‑size together
   with tiny Gaussian jitter keeps the iterates moving.

3. **Simulated‑annealing hill climbing** – a final global search that
   repeatedly transfers a random fraction of mass from one coordinate to
   another.  Moves are accepted according to a Metropolis rule with an
   exponentially cooling temperature.  The best vector seen during the
   whole run is kept.

The algorithm respects the 1000 s budget, splits the time roughly
(35 % / 35 % / 30 %), and works with any initial sequence (the previously
found one is used when available).  The final sequence is returned as a
plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:                     # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full self‑convolution via FFT (returns length 2·n‑1 array)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Compute a*a (full convolution) using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Exact objective for a simplex‑normalised vector
def _objective(a: np.ndarray, n: int, L: int) -> float:
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Sub‑gradient of the maximal convolution entry
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. a (averaged over all maximisers)."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]          # each pair contributes symmetrically
    if ks.size > 0:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
# Gradient of the smooth log‑sum‑exp surrogate
def _smooth_gradient(a: np.ndarray, conv: np.ndarray,
                     alpha: float, L: int) -> np.ndarray:
    """
    ∇_a  ( (2·n/α)·log Σ_k exp(α·conv_k) )
    = 2 * correlation( w , a )
    where w_k ∝ exp(α·conv_k).
    """
    # soft‑max (stable)
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / exp_scaled.sum()

    # correlation w ⋆ a   (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)   # length ≥ 3·n‑2
    n = a.size
    grad = 2.0 * corr_rev[n - 1 : n - 1 + n]          # central n entries
    return grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence that minimises the evaluation
    function.  Returns the sequence as a list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # a little safety margin

    # ------------------------------------------------------------------
    # Problem size and initial vector
    if 'height_sequence_1' in globals():
        base = np.array(height_sequence_1, dtype=float)
        n = base.size
        a = base / base.sum()
        print(f"[init] Using provided sequence, n={n}")
    else:
        n = 4096                                 # modest default size
        # start from a sorted Dirichlet – this tends to reduce the
        # self‑overlap of a and its reversal.
        a = np.random.dirichlet(np.full(n, 0.5))
        a = np.sort(a)[::-1]                    # descending order
        a = _project_to_simplex(a)
        print(f"[init] No prior sequence – start random, n={n}")

    # FFT length – enough for convolution (2n‑1) and correlation (3n‑2)
    L = 1 << ((3 * n - 2).bit_length())          # next power of two

    # ------------------------------------------------------------------
    # Keep the globally best solution
    conv = _conv_fft(a, L)
    best_a = a.copy()
    best_obj = _objective(best_a, n, L)
    print(f"[init] O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam on the smooth surrogate (≈35 % of total time)
    phase1_end = start_time + budget_s * 0.35
    # schedule parameters
    alpha_start, alpha_end = 5.0, 10000.0
    max_iter1 = 300_000
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.2
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(max_iter1):
        now = time.time()
        if now >= phase1_end or now >= deadline - 1.0:
            break

        # full convolution and current objective
        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            print(f"[phase‑1] it={it}, O={best_obj:.6f}")

        # exponential growth of α (more aggressive than linear)
        frac = it / max_iter1
        alpha = alpha_start * (alpha_end / alpha_start) ** frac

        # smooth gradient and Adam update
        grad = _smooth_gradient(a, conv, alpha, L)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        # learning rate decays linearly to keep steps small near the end
        lr = lr0 * (1.0 - frac)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a - step
        a = _project_to_simplex(a_candidate)

    # ------------------------------------------------------------------
    # Phase 2 – deterministic sub‑gradient refinement (≈35 % of time)
    phase2_end = start_time + budget_s * 0.70
    max_iter2 = 300_000
    for it in range(max_iter2):
        now = time.time()
        if now >= phase2_end or now >= deadline - 1.0:
            break

        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            print(f"[subgrad] it={it}, O={best_obj:.6f}")

        # sub‑gradient of the (possibly multiple) maximal entries
        g = _subgradient_max_conv(a, conv)

        # linearly decaying step size (starts larger, ends tiny)
        eta = 0.2 * (1.0 - it / max_iter2)
        a = a - eta * g
        a = _project_to_simplex(a)

        # occasional jitter to escape plateaus
        if it % 1000 == 0 and it > 0:
            jitter = np.random.randn(n) * 3e-4
            a = _project_to_simplex(a + jitter)

    # ------------------------------------------------------------------
    # Phase 3 – simulated‑annealing hill climbing (≈30 % of time)
    # Initialise local copy from the current global best
    local_a = best_a.copy()
    local_obj = _objective(local_a, n, L)

    # temperature parameters
    T0 = 0.08          # start a bit hotter than before
    decay = 0.998      # exponential cooling factor
    total_iters = 0

    while time.time() < deadline:
        T = T0 * (decay ** total_iters)

        # propose a random mass‑transfer move
        i, j = np.random.randint(n, size=2)
        if i == j or local_a[i] == 0.0:
            total_iters += 1
            continue
        # transfer a random fraction (up to 20 %) of the source mass
        delta = local_a[i] * np.random.rand() * 0.20
        a_new = local_a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = _project_to_simplex(a_new)

        obj_new = _objective(a_new, n, L)

        # Metropolis acceptance
        if obj_new < local_obj or np.random.rand() < math.exp((local_obj - obj_new) / max(T, 1e-12)):
            local_a = a_new
            local_obj = obj_new
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = local_a.copy()
                print(f"[anneal] it={total_iters}, O={best_obj:.6f}")

        total_iters += 1

        # every few thousand steps, reset the local walk to the global best
        if total_iters % 5000 == 0:
            local_a = best_a.copy()
            local_obj = _objective(local_a, n, L)

        # safety break
        if time.time() >= deadline - 0.5:
            break

    # ------------------------------------------------------------------
    # Final sanity check with the supplied evaluation function
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:
        print("[final] evaluate_sequence failed:", e)

    return best_a.tolist()
