def _optimize_smooth(a0, deadline, max_iters=200000, verbose=True):

   a = a0.copy()
   n = a.shape[0]
   size = 1 << ((3 * n - 2).bit_length())
   eps_factor = 0.2
   min_eps_factor = 0.001
   decay_eps = 0.9995

   beta = 0.9
   momentum = np.zeros_like(a)

   lr0 = 0.04

   best_val = float('inf')
   best_seq = None

   t = 0
   while time.time() < deadline and t < max_iters:
       grad, max_c, _, _ = _smooth_obj_and_grad(a, size, eps_factor)
       cur_val = 2.0 * n * max_c
       if cur_val < best_val:
           best_val = cur_val
           best_seq = a.copy()
           if verbose:
               print(f"[improve] iter {t:06d} obj {best_val:.9f}")
       momentum = beta * momentum + (1.0 - beta) * grad
       lr = lr0 / np.sqrt(t + 1)
       a = a - lr * momentum
       # enforce non-negativity (project)
       a = np.maximum(a, 0.0)
       a = _project_simplex(a)
       # update epsilon factor
       eps_factor = max(eps_factor * decay_eps, min_eps_factor)

       # optionally random perturbation
       if t % 10000 == 0 and t > 0:
           pert = np.random.normal(scale=0.005, size=n)
           a = np.maximum(a + pert, 0.0)
           a = _project_simplex(a)

       t += 1

   return best_seq, best_val

"""...
Optimization of step sequence to minimize evaluate_sequence objective using smoothed
max‑convolution gradient descent (log‑sum‑exp smoothing) with simplex projection.
"""

import time
import numpy as np

# Helper: projection onto simplex
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Catalan distribution normalised
def _catalan_normalised(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    a /= a.sum()
    return a

# Full linear convolution via real FFT (used for final objective)
def _fft_convolve(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[: 2 * n - 1]

# Smoothed gradient of 2 n·max(a*a)
def _smooth_grad(a: np.ndarray, size: int, eps_factor: float) -> tuple[np.ndarray, float]:
    """Return (gradient, max_convolution) of the smoothed objective."""
    n = a.shape[0]
    # FFT of a
    fa = np.fft.rfft(a, size)
    # linear convolution a*a (full length 2n-1)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    max_c = conv.max()
    # smoothing parameter, never smaller than 1e-12
    epsilon = max(eps_factor * max_c, 1e-12)
    # soft‑max weights (log‑sum‑exp trick)
    w = np.exp((conv - max_c) / epsilon)          # length 2n‑1, max weight = 1
    sum_w = w.sum()
    # correlation a ⋆ w (full length 3n‑2)
    w_rev = w[::-1]
    fw_rev = np.fft.rfft(w_rev, size)
    conv_rev = np.fft.irfft(fa * fw_rev, size)[: n + (2 * n - 1) - 1]   # length 3n‑2
    M = 2 * n - 1
    # correlation for each index j = 0…n‑1
    corr = conv_rev[M - 1 - np.arange(n)]
    # gradient of the smoothed max
    grad = (4.0 * n / sum_w) * corr
    return grad, max_c

def _optimize_smooth(a0: np.ndarray,
                     deadline: float,
                     max_iters: int = 250_000,
                     verbose: bool = True) -> tuple[np.ndarray, float]:
    """Projected (heavy‑ball) descent on the smoothed max‑convolution objective."""
    a = a0.copy()
    n = a.shape[0]
    # FFT size that can accommodate the correlation (3n‑2)
    size = 1 << ((3 * n - 2).bit_length())

    # Hyper‑parameters
    beta = 0.9                         # momentum coefficient
    momentum = np.zeros_like(a)
    lr0 = 0.04                         # initial learning‑rate scale
    eps_factor = 0.3                   # start smoothing at 30 % of the current max
    min_eps_factor = 0.001
    decay_eps = 0.9994                 # geometric decay per iteration

    best = a.copy()
    # initial objective
    _, max_c0 = _smooth_grad(a, size, eps_factor)  # we don't really need the grad here
    best_val = 2.0 * n * max_c0

    t = 0
    while time.time() < deadline and t < max_iters:
        # gradient of the smoothed objective
        grad, max_c = _smooth_grad(a, size, eps_factor)

        cur_val = 2.0 * n * max_c
        if cur_val < best_val:
            best_val = cur_val
            best = a.copy()
            if verbose:

        # diminishing learning rate (1/√t schedule)
        lr = lr0 / np.sqrt(t + 1)

        # additive step + projection onto the simplex
        a = a - lr * momentum
        a = np.maximum(a, 0.0)

            a = np.maximum(a + pert, 0.0)
            a = _project_simplex(a)

        t += 1

"""Optimisation of a non‑negative step sequence.

We minimise the value returned by ``evaluate_sequence`` which, up to a harmless
scale, is the objective

    O(a) = 2·n·max_k (a * a)[k]      with a_i ≥ 0 ,  Σ a_i = 1 .

The programme starts from a Catalan‑type distribution (known to be close to
optimal) and refines it with a smoothed version of the objective (log‑sum‑exp
approximation of the max).  The gradient of the smooth surrogate is computed
exactly by means of FFT‑based convolution, after which a heavy‑ball (momentum)
step followed by Euclidean projection onto the probability simplex is taken.
The smoothing parameter is gradually reduced, allowing the method to first
explore a smooth landscape and later focus on the true non‑smooth objective.
A few cheap random restarts are performed if time permits.

The whole routine respects the 1000 s budget and returns a plain Python list
of heights ready for ``evaluate_sequence``.
"""

# Projection onto the probability simplex Σx=1, x≥0
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
def _smooth_grad(a: np.ndarray, size: int, eps_factor: float) -> tuple[np.ndarray, float]:
    """Return (gradient, current maximal convolution) for the smoothed objective.

    eps_factor scales the smoothing parameter ε = eps_factor * max(conv).
    """

    fa = np.fft.rfft(a, size)

    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
