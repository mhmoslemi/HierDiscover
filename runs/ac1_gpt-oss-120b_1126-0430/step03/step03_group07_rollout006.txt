def _conv_full_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    n = a.size + b.size - 1
    m = 1 << ((n-1).bit_length())
    fa = np.fft.rfft(a, m)
    fb = np.fft.rfft(b, m)
    conv = np.fft.irfft(fa * fb, m)[:n]
    return conv
```

def _adam_optimize(seq, deadline, lr=0.12, beta1=0.9, beta2=0.999):
    n = seq.size
    eps = 1e-15
    log_a = np.log(seq + eps)
    m = np.zeros_like(seq)
    v = np.zeros_like(seq)
    t = 0
    start = time.time()
    total = deadline - start
    lam_start = 2.0
    lam_end = 30.0
    lam_rate = (lam_end - lam_start) / max(total, 1e-9)
    best_seq = seq.copy()
    best_score = float('inf')
    # precompute FFT length as next_pow2(3*n - 2)
    fft_len = 1 << ((3*n - 2).bit_length())
    while time.time() < deadline:
        # recompute a
        a = np.exp(log_a)
        sum_a = a.sum()
        if sum_a == 0:
            a = np.full(n, 1.0/n)
            log_a = np.log(a + eps)
            sum_a = 1.0
        a /= sum_a
        # compute conv: a*a
        fa = np.fft.rfft(a, fft_len)
        conv_full = np.fft.irfft(fa * fa, fft_len)[:2*n-1]
        M = conv_full.max()
        # compute smoothed weights
        lam = lam_start + lam_rate * (time.time() - start)
        if lam > lam_end:
            lam = lam_end
        # use softmax over (conv - M) scaled by lam
        # (conv - M) <=0, so max weight = 1
        w = np.exp(lam * (conv_full - M))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # compute gradient: conv(w, a_rev)
        a_rev = a[::-1]
        fw = np.fft.rfft(w, fft_len)
        farev = np.fft.rfft(a_rev, fft_len)
        grad_full = np.fft.irfft(fw * farev, fft_len)[:3*n-2]
        g = grad_full[n-1:n-1+n]
        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        log_a -= step
        # clip
        np.clip(log_a, -30.0, 30.0, out=log_a)
        # evaluate metric
        metric = 2 * n * M
        if metric < best_score:
            best_score = metric
            best_seq = a.copy()
            # print occasionally
            if t % 1000 == 0:
                elapsed = time.time() - start
                print(f"[Adam] iter {t:6d}  metric {best_score:.6f}  lam {lam:.2f}  time {elapsed:.1f}s")
        # random jitter occasionally to escape
        if np.random.rand() < 0.001:
            # add small noise to random coordinate
            idx = np.random.randint(n)
            # small change
            a[idx] = max(0.0, a[idx] + np.random.randn() * 1e-3)
            a /= a.sum()
            log_a = np.log(a + eps)
    return best_seq

def _next_pow2(x: int) -> int:

def _conv_full_fft(a: np.ndarray, b: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full (linear) convolution of real vectors a and b using a fixed FFT length.
    Returns a real array of length len(a)+len(b)-1.
    """
    fa = np.fft.rfft(a, fft_len)
    fb = np.fft.rfft(b, fft_len)
    conv = np.fft.irfft(fa * fb, fft_len)[: a.size + b.size - 1]
    return conv


def _metric(seq: np.ndarray) -> float:
    """
    Metric for a *normalised* sequence (∑seq == 1):
        metric = 2·n·max(conv(seq,seq))
    """
    n = seq.size
    conv = _conv_full_fft(seq, seq, _next_pow2(3 * n - 2))[: 2 * n - 1]
    return 2.0 * n * conv.max()

def _beta_sequence(alpha: float, beta: float, length: int) -> np.ndarray:
    """Sample the (unnormalised) Beta(α,β) density on a regular grid, then normalise."""
    xs = (np.arange(length, dtype=float) + 0.5) / length
    a = np.power(xs, alpha - 1.0, out=np.empty_like(xs))
    b = np.power(1.0 - xs, beta - 1.0, out=np.empty_like(xs))
    seq = a * b
    seq[np.isinf(seq)] = 0.0
    seq = np.maximum(seq, 0.0)
    if seq.sum() == 0.0:
        seq = np.full(length, 1.0 / length)
    else:
        seq /= seq.sum()
    return seq


def _sin_power_sequence(p: float, length: int) -> np.ndarray:
    """a_i ∝ sin(π·(i+0.5)/length)^p, with p possibly negative."""
    xs = (np.arange(length, dtype=float) + 0.5) / length
    vals = np.sin(np.pi * xs) ** p
    vals = np.maximum(vals, 0.0)
    if vals.sum() == 0.0:
        vals = np.full(length, 1.0 / length)
    else:
        vals /= vals.sum()
    return vals


def _random_dirichlet(length: int, concentration: float = 0.5) -> np.ndarray:
    """Draw a random Dirichlet point with identical concentration parameters."""
    seq = np.random.gamma(concentration, 1.0, size=length)
    if seq.sum() == 0.0:
        seq = np.full(length, 1.0 / length)
    else:
        seq /= seq.sum()
    return seq

    n = initial_seq.size
    eps = 1e-15                     # to avoid log(0)
    log_a = np.log(initial_seq + eps)

    m = np.zeros_like(initial_seq)
    v = np.zeros_like(initial_seq)
    t = 0

    # FFT length big enough for all convolutions (a*a and w*a_rev)
    fft_len = _next_pow2(3 * n - 2)

    # time‑dependent temperature schedule
    start_time = time.time()
    total_time = max(deadline - start_time, 1e-9)
    lam_rate = (lam_end - lam_start) / total_time

    best_seq = None
    best_score = float('inf')
    it_print = 0

    while time.time() < deadline:
        # ---- obtain current probability vector, normalise ----
        a = np.exp(log_a)
        sum_a = a.sum()
        if sum_a == 0.0:
            a = np.full(n, 1.0 / n)
            log_a = np.log(a + eps)
            sum_a = 1.0
        a /= sum_a

        # ---- convolution a * a (full length 2n‑1) ----
        fa = np.fft.rfft(a, fft_len)
        conv_full = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        M = conv_full.max()

        # ---- temperature (soft‑max) ----
        lam = lam_start + lam_rate * (time.time() - start_time)
        lam = min(lam, lam_end)

        # weights proportional to exp(λ·(conv – M))  (max weight = 1)
        w = np.exp(lam * (conv_full - M))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of the smooth max:  g = (w * a_rev) convolution ----
        a_rev = a[::-1]
        fw = np.fft.rfft(w, fft_len)
        farev = np.fft.rfft(a_rev, fft_len)
        grad_full = np.fft.irfft(fw * farev, fft_len)[: 3 * n - 2]
        # the required slice corresponds to indices n‑1 … n‑1+n‑1
        g = grad_full[n - 1: n - 1 + n]

        # ---- Adam moment updates ----
        t += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # ---- exponentiated (log‑space) update ----
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        log_a -= step
        np.clip(log_a, -30.0, 30.0, out=log_a)

        # ---- keep the best seen so far ----
        score = 2.0 * n * M          # metric (since sum=1)
        if score < best_score - 1e-12:
            best_score = score
            best_seq = a.copy()
            # occasional progress output
            if it_print % 500 == 0:
                elapsed = time.time() - start_time
                print(f"[Adam] iter {t:6d}  metric {best_score:.6f}  λ {lam:.2f}  time {elapsed:.1f}s")
            it_print += 1

        # ---- tiny random jitter to avoid dead‑ends ----
        if np.random.rand() < 0.001:
            idx = np.random.randint(n)
            a[idx] = max(0.0, a[idx] + np.random.randn() * 1e-3)
            a /= a.sum()
            log_a = np.log(a + eps)

    # return the best vector discovered
    return best_seq if best_seq is not None else np.exp(log_a) / np.exp(log_a).sum()


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Produce a step‑function (list of non‑negative floats) that minimises the
    evaluation metric.  The routine runs for up to ``budget_s`` seconds
    (a safety margin of a few seconds is left for the harness).  It builds
    several analytic seeds, picks the most promising one and refines it
    with a smooth‑max Adam optimiser.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0           # safety margin

    # ------------------------------------------------------------------
    # 1.  Choose final length.
    # ------------------------------------------------------------------
    n = 5000

    # ------------------------------------------------------------------
    # 2.  Generate a diverse set of seed vectors (already normalised).
    # ------------------------------------------------------------------
    seeds = []

    # classic arcsine (Beta(0.5,0.5))
    seeds.append(_beta_sequence(0.5, 0.5, n))

    # symmetric Beta with several α‑values
    for a in [0.35, 0.40, 0.45, 0.55, 0.60, 0.65, 0.80, 1.00]:
        seeds.append(_beta_sequence(a, a, n))

    # a handful of asymmetric Beta shapes
    for a, b in [(0.3, 0.7), (0.7, 0.3), (0.4, 0.9), (0.9, 0.4)]:
        seeds.append(_beta_sequence(a, b, n))

    # sin‑power shapes (p negative → heavier edges)
    for p in [-0.6, -0.5, -0.4, -0.3, -0.2, 0.0, 0.2]:
        seeds.append(_sin_power_sequence(p, n))

    # a few random Dirichlet draws (concentration < 1 gives spikier vectors)
    for _ in range(5):
        seeds.append(_random_dirichlet(n, concentration=0.4))

    # optional: incorporate a previously found construction if it exists
    try:
        hs = np.array(height_sequence_1, dtype=float)
        if hs.ndim == 1:
            if hs.size == n:
                seeds.append(hs / hs.sum())
            else:
                # linear interpolation to required length
                x_src = np.arange(hs.size, dtype=float)
                x_dst = np.linspace(0, hs.size - 1, n)
                hs_res = np.interp(x_dst, x_src, hs)
                seeds.append(hs_res / hs_res.sum())
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Quick evaluation of all seeds; keep the best.
    # ------------------------------------------------------------------
    best_seed = None
    best_metric = float('inf')
    for idx, s in enumerate(seeds):
        s = np.maximum(s, 0.0)
        if s.sum() == 0.0:
            s = np.full_like(s, 1.0 / s.size)
        else:
            s = s / s.sum()
        cur_metric = _metric(s)
        print(f"[seed {idx:02d}] metric {cur_metric:.6f}")
        if cur_metric < best_metric:
            best_metric = cur_metric
            best_seed = s.copy()
    print(f"[summary] best initial metric {best_metric:.6f}")

    # ------------------------------------------------------------------
    # 4.  Refine the chosen seed with the Adam optimiser until time runs out.
    # ------------------------------------------------------------------
    if best_seed is None:
        best_seed = np.full(n, 1.0 / n)

    refined = _adam_optimize(best_seed, deadline)

    final_metric = _metric(refined)
    print(f"[final] metric {final_metric:.6f}")

    # ------------------------------------------------------------------
    # 5.  Return a plain Python list (the harness will call evaluate_sequence).
    # ------------------------------------------------------------------
    return refined.tolist()
