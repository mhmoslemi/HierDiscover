def softmax(x):
    e = np.exp(x - np.max(x))
    return e / e.sum()

def compute_conv(a):
    # Use FFT with size = power of two >= 2n-1
    n = a.shape[0]
    size = 1 << ((2*n-2).bit_length())
    A = np.fft.fft(a, n=size)
    conv = np.fft.ifft(A * A).real[:2*n-1]
    return conv

def smoothed_max(conv, beta):
    # Use log-sum-exp with scaling
    c = conv * beta
    maxc = np.max(c)
    # Avoid overflow
    w = np.exp(c - maxc)
    sumw = np.sum(w)
    m = (np.log(sumw) + maxc) / beta
    # Also return softmax weights w_norm = w / sumw
    w_norm = w / sumw
    return m, w_norm

```
import time, numpy as np

def softmax(y):
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()

def compute_conv_fft(a, size):
    A = np.fft.fft(a, n=size)
    conv = np.fft.ifft(A * A).real
    # conv length = size, but actual linear conv is first (2n-1) entries
    return conv

def linear_conv(a):
    # compute full convolution length 2n-1 using FFT size pow2>=2n-1
    n = a.shape[0]
    size = 1 << ((2*n-2).bit_length())
    A = np.fft.fft(a, n=size)
    conv = np.fft.ifft(A * A).real[:2*n-1]
    return conv

def cross_corr(a, w):
    # cross correlation g_j = sum_i w_i a_{i-j}
    n = a.shape[0]
    len_w = w.shape[0]  # 2n-1
    size = 1 << ((n + len_w - 1).bit_length())
    A = np.fft.fft(a, n=size)
    W_rev = np.fft.fft(w[::-1], n=size)
    c = np.fft.ifft(A * W_rev).real
    offset = len_w - 1
    g = c[offset - np.arange(n)]
    return g

def smoothed_max_and_weights(conv, beta):
    # compute smooth max and weight distribution
    c = conv * beta
    maxc = np.max(c)
    # subtract maxc for numerical stability
    exp_c = np.exp(c - maxc)
    sum_exp = np.sum(exp_c)
    smooth_max = (np.log(sum_exp) + maxc) / beta
    weights = exp_c / sum_exp  # sum to 1
    return smooth_max, weights

def objective_exact(conv, n):
    return 2.0 * n * np.max(conv)

def gradient_a(a, weights):
    # weights length = 2n-1
    n = a.shape[0]
    # cross-correlation
    g = cross_corr(a, weights)
    grad = 4.0 * n * g
    return grad

def adam_update(y, grad, m, v, t, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * (grad * grad)
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)
    y_new = y - lr * m_hat / (np.sqrt(v_hat) + eps)
    return y_new, m, v

def optimize_from_start(y0, n, deadline, max_iters=200000, beta_start=0.5, beta_end=200.0):
    y = y0.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    best_a = softmax(y)
    best_obj = objective_exact(linear_conv(best_a), n)
    t_iter = 0
    while time.time() < deadline and t_iter < max_iters:
        # compute a
        a = softmax(y)
        # conv
        conv = linear_conv(a)  # length 2n-1
        # beta schedule
        # linear schedule:
        beta = beta_start + (beta_end - beta_start) * (t_iter / max_iters)
        # smoothed max and weights
        smooth_max, w = smoothed_max_and_weights(conv, beta)
        # gradient wrt a
        grad_a = gradient_a(a, w)
        # gradient wrt y
        dot = np.dot(a, grad_a)
        grad_y = a * (grad_a - dot)
        # adam step
        lr = 0.02  # could adapt
        # increment iteration count
        t = t_iter + 1
        y, m, v = adam_update(y, grad_y, m, v, t, lr=lr)
        # optionally evaluate exact objective periodically
        if (t_iter % 50) == 0:
            obj = objective_exact(conv, n)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
                # print progress
                print(f"[opt] iter {t_iter:6d} best O={best_obj:.6f}, beta={beta:.2f}")
        t_iter += 1
    return best_a, best_obj

    e = np.exp(y - m)

def _linear_convolution(a):
    n = a.shape[0]
    size = 1 << ((2*n-2).bit_length())
    A = np.fft.fft(a, n=size)
    conv = np.fft.ifft(A * A).real[:2*n-1]
    return conv

def _cross_corr(a, w, size):
    # cross correlation of a with w: g_j = sum_i w_i a_{i-j}
    n = a.shape[0]
    len_w = w.shape[0]
    # compute using FFT
    A = np.fft.fft(a, n=size)
    W_rev = np.fft.fft(w[::-1], n=size)
    c = np.fft.ifft(A * W_rev).real
    offset = len_w - 1
    g = c[offset - np.arange(n)]
    return g

def _smooth_max_and_weights(conv, beta):
    # Smoothed max via log-sum-exp
    c = conv * beta
    maxc = np.max(c)
    exp_c = np.exp(c - maxc)
    sum_exp = np.sum(exp_c)
    m = (np.log(sum_exp) + maxc) / beta
    w = exp_c / sum_exp

    n = a.shape[0]
    g = _cross_corr(a, w, size_cross)
    grad = 4.0 * n * g
    return grad

    v_hat = v / (1 - beta2**t)
    y = y - lr * m_hat / (np.sqrt(v_hat) + eps)
    return y, m, v

"""
Search for a non‑negative step‑function that minimises

    O = 2·n·max(conv(a,a))    (with Σa = 1)

The bound is scale‑invariant, therefore we optimise over the simplex.
We parametrise a = softmax(y) (y∈ℝⁿ) which automatically satisfies a≥0,
Σa=1.  The non‑smooth max is replaced by a smooth log‑sum‑exp
approximation with a temperature β that is slowly increased.
We minimise the smoothed objective with Adam (a first‑order method with
momentum) while projecting back onto the simplex via the softmax.
A few sensible starts (previous best, the 1/√‑law, a random point) are
optimised in turn; the best sequence found is returned and stored in the
global variable ``height_sequence_1`` for later runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


# ----------------------------------------------------------------------
def _linear_conv(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a power‑of‑two FFT (full length)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 2).bit_length())          # next power of two ≥ 2n‑1
    A = np.fft.fft(a, n=size)
    conv = np.fft.ifft(A * A).real[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def _cross_corr(a: np.ndarray, w: np.ndarray, size: int) -> np.ndarray:
    """
    Compute g_j = Σ_i w_i·a_{i‑j} (cross‑correlation) via FFT.
    ``size`` must be a power of two ≥ n + len(w) – 1.
    """
    n = a.shape[0]
    len_w = w.shape[0]                     # = 2n‑1
    A = np.fft.fft(a, n=size)
    W_rev = np.fft.fft(w[::-1], n=size)    # w reversed
    c = np.fft.ifft(A * W_rev).real
    offset = len_w - 1
    return c[offset - np.arange(n)]


# ----------------------------------------------------------------------
def _smooth_max_and_weights(conv: np.ndarray, beta: float):
    """
    Smoothed maximum via log‑sum‑exp and the corresponding soft weights.
    Returns (M, w) where M≈max(conv) and Σw = 1.
    """
    c = conv * beta
    m = np.max(c)
    exp_c = np.exp(c - m)                 # stabilise
    sum_exp = np.sum(exp_c)
    M = (np.log(sum_exp) + m) / beta
    w = exp_c / sum_exp
    return M, w


# ----------------------------------------------------------------------
def _objective_exact(conv: np.ndarray, n: int) -> float:
    """Exact bound 2·n·max(conv)."""
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
def _grad_a(a: np.ndarray, w: np.ndarray, size_cross: int) -> np.ndarray:
    """
    Gradient of the smoothed objective w.r.t. a.
    grad_a = 4·n· Σ_i w_i·a_{i‑j}  (cross‑correlation).
    """
    n = a.shape[0]
    g = _cross_corr(a, w, size_cross)
    return 4.0 * n * g


# ----------------------------------------------------------------------
def _adam_step(y: np.ndarray, grad: np.ndarray,
               m: np.ndarray, v: np.ndarray,
               t: int, lr: float = 0.02,
               beta1: float = 0.9, beta2: float = 0.999,
               eps: float = 1e-8):
    """One Adam update on the unconstrained variable y."""
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad * grad)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    y = y - lr * m_hat / (np.sqrt(v_hat) + eps)
    return y, m, v


# ----------------------------------------------------------------------
def _optimize_from_y(y0: np.ndarray, n: int,
                     deadline: float,
                     max_iters: int = 200_000,
                     beta_start: float = 0.5,
                     beta_end: float = 200.0,

    Adam optimisation on the smoothed objective.
    Returns the best a (already normalised) and its exact bound.
    """

    size_cross = 1 << ((3 * n - 2).bit_length())
