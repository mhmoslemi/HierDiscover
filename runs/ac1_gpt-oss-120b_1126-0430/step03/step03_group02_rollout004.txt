    beta = 0.9
    v = np.zeros_like(a)
    eta0 = 0.2

"""
Improved search for a non‑negative step‑function that minimises

    O = 2·n·max(conv(a,a)) / (sum(a))²

The bound is scale‑invariant, so we normalise Σa = 1 and minimise
    F(a) = 2·n·max(conv(a,a)).

Algorithm
---------

1. **Power‑law sweep** – try a_i ∝ (i+1)^{-p} for several exponents p,
   keep the best as a start.

2. **Sub‑gradient EG (mirror descent)** – an exponentiated‑gradient
   descent on the true sub‑gradient of the max‑operator, with momentum
   and a decaying learning rate.

3. **Soft‑max‑smoothed EG** – replace the non‑smooth max by a
   log‑sum‑exp (ε‑softmax).  The gradient of this smooth surrogate is
   computed via an FFT‑based correlation, allowing a second phase of
   mirror descent that can escape plateaus.

4. **Hill‑climbing local search** – random pairwise mass transfers
   (keeping the simplex constraint).  Only improving moves are kept;
   this final refinement often yields a small but consistent gain.

All phases respect the simplex by normalising after each update.
The code is time‑capped (default 1000 s).  The best sequence found is
returned as a plain list of floats and finally evaluated with the
provided ``evaluate_sequence`` function.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a power‑of‑two FFT."""
    n = a.size
    size = 1 << ((2 * n - 2).bit_length())
    A = np.fft.fft(a, n=size)
    conv = np.fft.ifft(A * A).real[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return the bound 2·n·max(conv(a,a)) for a normalised (Σa=1)."""
    n = a.size
    conv = _conv_fft(a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) w.r.t. a on the simplex.
    It is the average of the gradients of all active (max) indices.
    """
    n = a.size
    conv = _conv_fft(a)
    max_val = conv.max()
    ks = np.where(np.abs(conv - max_val) < 1e-14)[0]
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        # each pair (i,j) appears twice in the derivative of a_i*a_j
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size > 0:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
def _exp_update(a: np.ndarray, g: np.ndarray, eta: float) -> np.ndarray:
    """
    Exponentiated‑gradient step with entropy mirror map.
    a – current distribution (already normalised)
    g – gradient (or sub‑gradient)
    eta – step size
    """
    log_a = np.log(a + 1e-15)          # avoid log(0)
    log_a -= eta * g
    log_a -= np.max(log_a)            # numerical stabilisation
    a_new = np.exp(log_a)
    a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, eps: float) -> np.ndarray:
    """
    Gradient of the ε‑softmax surrogate
        f_ε(a) = ε·log Σ_k exp(conv_k/ε)
    w.r.t. a.  This equals the weighted average of the individual
    convolution gradients, where the weights are the softmax of conv/ε.
    """
    n = a.size
    conv = _conv_fft(a)                       # length 2n‑1
    # Stabilised softmax
    max_c = conv.max()
    scaled = (conv - max_c) / eps
    w = np.exp(scaled)
    w_sum = w.sum()

    # Correlation w ⋆ a (cross‑correlation, mode='valid')
    size = 1 << ((len(w) + n - 1).bit_length())
    W = np.fft.fft(w, n=size)
    A_rev = np.fft.fft(a[::-1], n=size)
    corr = np.fft.ifft(W * A_rev).real
    # Desired slice corresponds to lags 0 … n‑1
    grad = 2.0 * corr[n - 1: n - 1 + n] / w_sum
    return grad


# ----------------------------------------------------------------------
def _phase_subgrad(a: np.ndarray,
                   best_a: np.ndarray,
                   best_score: float,
                   max_iters: int,
                   deadline: float) -> Tuple[np.ndarray, np.ndarray, float]:
    """
    Mirror‑descent on the true sub‑gradient of max(conv).
    Uses momentum (β=0.9) and a linearly decaying learning rate.
    """
    n = a.size
    v = np.zeros_like(a)                # momentum buffer
    beta = 0.9
    eta0 = 0.55

    for t in range(max_iters):
        if time.time() >= deadline:
            break
        g = _subgradient_max_conv(a)
        v = beta * v + (1.0 - beta) * g
        eta = eta0 * (1.0 - t / max_iters)
        a = _exp_update(a, v, eta)

        cur = _objective(a)
        if cur + 1e-12 < best_score:
            best_score = cur
            best_a = a.copy()
            print(f"[subgrad] iter {t:5d}  best O={best_score:.6f}")

    return a, best_a, best_score


# ----------------------------------------------------------------------
def _phase_softmax(a: np.ndarray,
                   best_a: np.ndarray,
                   best_score: float,
                   max_iters: int,
                   deadline: float) -> Tuple[np.ndarray, np.ndarray, float]:
    """
    Mirror‑descent on a smooth log‑sum‑exp surrogate of max(conv).
    The smoothing parameter ε is geometrically decreased.
    """
    n = a.size
    v = np.zeros_like(a)
    beta = 0.9
    eta0 = 0.25
    eps_start = 0.08
    eps_end = 0.001

    for t in range(max_iters):
        if time.time() >= deadline:
            break
        # geometric decay of ε
        eps = eps_start * ((eps_end / eps_start) ** (t / max_iters))
        # gradient of the smooth objective (include factor 2·n)
        g = 2.0 * n * _softmax_gradient(a, eps)
        v = beta * v + (1.0 - beta) * g
        eta = eta0 * (1.0 - t / max_iters)
        a = _exp_update(a, v, eta)

        cur = _objective(a)
        if cur + 1e-12 < best_score:
            best_score = cur
            best_a = a.copy()
            print(f"[softmax] iter {t:5d}  best O={best_score:.6f}")

    return a, best_a, best_score


# ----------------------------------------------------------------------
def _phase_hill_climb(a: np.ndarray,
                      best_a: np.ndarray,
                      best_score: float,
                      max_iters: int,
                      deadline: float) -> Tuple[np.ndarray, np.ndarray, float]:
    """
    Simple stochastic hill‑climbing: move a random fraction of mass from
    a source index to a target index, accepting only improving moves.
    """
    n = a.size
    cur_a = a.copy()
    cur_score = best_score

    for t in range(max_iters):
        if time.time() >= deadline:
            break

        # pick a source index proportionally to its mass
        src = np.random.choice(n, p=cur_a / cur_a.sum())
        if cur_a[src] < 1e-12:
            continue
        # pick a different target
        tgt = np.random.randint(n - 1)
        if tgt >= src:
            tgt += 1

        # transfer a random fraction (up to 30 %) of the source mass
        delta = cur_a[src] * np.random.rand() * 0.30
        a_new = cur_a.copy()
        a_new[src] -= delta
        a_new[tgt] += delta

        new_score = _objective(a_new)
        if new_score < cur_score - 1e-12:        # strictly better
            cur_a, cur_score = a_new, new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = cur_a.copy()
                print(f"[hillclimb] iter {t:5d}  new best O={best_score:.6f}")

    return cur_a, best_a, best_score

def _optimize_full(a0: np.ndarray, deadline: float) -> Tuple[np.ndarray, float]:
    """
    Run the three optimisation phases in sequence, respecting the global
    deadline.  Returns the best sequence found and its raw objective value.
    """
    # Phase lengths are allocated heuristically; they shrink if time runs out.
    remaining = deadline - time.time()
    # Rough budget split (seconds): 0.45 – subgrad, 0.35 – softmax, 0.20 – hill‑climb
    sub_iters = max_iters_sub = int(15000 * (remaining / 950.0))
    soft_iters = max_iters_soft = int(12000 * (remaining / 950.0))

    a, best_a, best_score = _phase_hill_climb(

    return best_a, best_score
