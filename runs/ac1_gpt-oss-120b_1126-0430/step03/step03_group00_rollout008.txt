"""
Propose a non‑negative step‑function (probability vector) that minimizes

    O(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm consists of three stages:

1. **Smooth surrogate optimisation** – minimise a soft‑max of the
   convolution using Adam on the log‑sum‑exp surrogate.
   The surrogate is (2·n/α)·log∑_k exp(α·conv_k).
   α is annealed from a small to a large value, turning the surrogate
   gradually into the true max.  After each Adam step the vector is
   projected onto the probability simplex.

2. **Local refinement** – stochastic “mass‑transfer’’ moves combined with
   a Metropolis acceptance rule, and occasional exact sub‑gradient steps
   on the current maximal convolution entry.  This phase performs a
   simulated‑annealing style hill‑climb.

3. **Final sanity check** – the best candidate found is evaluated with the
   provided `evaluate_sequence` function and returned as a plain list.

The implementation works for any length; a relatively large size
(`n≈15000`) is used because the theoretical lower bound of the objective
approaches 1 as n grows.  All operations respect the 1000 s time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:                         # all non‑positive
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution a*a using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
def _objective_simplex(a: np.ndarray, n: int, L: int) -> float:
    """Exact objective 2·n·max(conv) for a vector a that already sums to 1."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. a (averaged over all maximising shifts)."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, conv: np.ndarray,
                     alpha: float, L: int) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp surrogate

        (2·n/α)·log Σ_k exp(α·conv_k)

    The formula reduces to 2·corr(w,a) where w_k ∝ exp(α·conv_k).
    """
    # soft‑max weights
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)          # numerical stability
    w = exp_scaled / exp_scaled.sum()

    # correlation w ⋆ a   (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)   # length >= 3·n‑2
    n_len = a.size
    grad = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]
    return grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence that minimises the evaluation
    function. Returns the best sequence as a list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin
    deadline = start_time + budget_s - 0.3

    # --------------------------------------------------------------
    # Problem size
    # A larger n gives a lower theoretical bound; we use n≈15000.
    n = 15000

    # --------------------------------------------------------------
    # Initial point (simplex‑normalised)
    if 'height_sequence_1' in globals():
        base = np.array(height_sequence_1, dtype=float)
        if base.size < n:
            a = np.pad(base, (0, n - base.size))
        else:
            a = base[:n].copy()
        a = np.maximum(a, 0.0)
    else:
        a = np.full(n, 1.0 / n)      # uniform start
    a = a / a.sum()                  # ensure simplex feasibility

    # FFT length – next power of two > 3·n‑2 (covers both convolution & correlation)
    L = 1 << ((3 * n - 2).bit_length())

    # --------------------------------------------------------------
    # Phase 1 – Adam on the smooth surrogate
    best_seq = a.copy()
    best_score = _objective_simplex(best_seq, n, L)
    print(f"[phase‑1] start O={best_score:.6f}")

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    lr0 = 0.8
    alpha_start = 5.0
    alpha_end = 20000.0
    max_iter_phase1 = 200_000                # generous upper bound

    # allocate roughly 55 % of the total budget to phase 1
    phase1_deadline = start_time + budget_s * 0.55
    it = 0
    while time.time() < phase1_deadline and it < max_iter_phase1:
        conv = _conv_fft(a, L)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[phase‑1] iter {it} new best O={best_score:.6f}")

        # linear schedule for the smoothing parameter
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iter_phase1)

        # smooth gradient of the log‑sum‑exp surrogate
        grad = _smooth_gradient(a, conv, alpha, L)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        # learning‑rate decays like 1/√t
        lr = lr0 / math.sqrt(it + 1)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # tiny jitter to escape flat regions
        if np.random.rand() < 0.001:
            a_new += np.random.randn(n) * 1e-4

        # projection back onto the simplex
        a_new = _project_to_simplex(a_new)

        # occasional pruning of the tiniest entries (keep support reasonably compact)
        if it % 8000 == 0 and it > 0:
            thresh = np.percentile(a_new, 0.5)   # drop ~0.5 % smallest weights
            a_new[a_new < thresh] = 0.0
            a_new = _project_to_simplex(a_new)

        a = a_new
        it += 1

    # --------------------------------------------------------------
    # Phase 2 – stochastic hill‑climbing + sub‑gradient refinements
    print(f"[phase‑2] entering local refinement, best O={best_score:.6f}")

    max_restarts = 0
    subgrad_interval = 5
    temperature0 = 0.04
    max_iter_local = 150_000
    no_improve_limit = 4000

    while time.time() < deadline:
        # ----------------------------------------------------------
        # initialise a restart – jittered best or fresh random start
        if np.random.rand() < 0.5:
            a = best_seq.copy()
            a += np.random.randn(n) * 0.001
            a = np.clip(a, 0.0, None)
            a = _project_to_simplex(a)
        else:
            a = np.random.rand(n)
            a = _project_to_simplex(a)

        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        it_loc = 0
        no_improve = 0
        T = temperature0

        while time.time() < deadline and it_loc < max_iter_local:
            # -------- random mass‑transfer move ----------
            i, j = np.random.randint(n, size=2)
            if i != j and a[i] > 1e-14:
                delta = a[i] * np.random.rand() * 0.10   # up to 10 % of source mass
                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta
                a_new = _project_to_simplex(a_new)

                conv_new = _conv_fft(a_new, L)
                new_obj = 2.0 * n * conv_new.max()

                # Metropolis acceptance
                if new_obj < cur_obj or np.random.rand() < math.exp(
                        (cur_obj - new_obj) / max(T, 1e-12)):
                    a = a_new
                    conv = conv_new
                    cur_obj = new_obj
                    no_improve = 0
                    if cur_obj < best_score:
                        best_score = cur_obj
                        best_seq = a.copy()
                        print(f"[{time.time():.1f}] New global best O={best_score:.6f}"
                              f" (restart={max_restarts}, iter={it_loc})")
                else:
                    no_improve += 1
            else:
                no_improve += 1

            # -------- sub‑gradient step on the current maximal entry ----------
            if it_loc % subgrad_interval == 0:
                g = _subgradient_max_conv(a, conv)
                eta = 0.06 * (1.0 - it_loc / max_iter_local)   # decreasing step size
                a = a - eta * g
                a = _project_to_simplex(a)
                conv = _conv_fft(a, L)
                cur_obj = 2.0 * n * conv.max()
                if cur_obj < best_score:
                    best_score = cur_obj
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] Sub‑grad improved O={best_score:.6f}")

            # temperature decay (simulated‑annealing style)
            T *= 0.9995
            it_loc += 1

            # restart if we have been stuck for a while
            if no_improve >= no_improve_limit:
                break

        max_restarts += 1
        if time.time() >= deadline:
            break

    # --------------------------------------------------------------
    # Final sanity check with the provided evaluator
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:
        print("[final] evaluate_sequence failed:", e)

    return best_seq.tolist()
