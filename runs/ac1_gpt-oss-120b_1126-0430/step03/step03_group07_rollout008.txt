def propose_candidate(seed=42, budget_s=1000, **kwargs):

        seq = np.maximum(init_seq, 0.0)

        M = conv.max()

        # if time exhausted break
        if time.time() >= deadline:

    if best_f is None:
        best_f = np.full(n_small, total_sum_target / n_small)
        best_M = np.fft.irfft(np.fft.rfft(best_f, 1 << ((2*n_small-1).bit_length()))**2, 1 << ((2*n_small-1).bit_length()))[:2*n_small-1].max()

def _fw_optimize(seq_init, target_sum, deadline, step=0.2):
    f = seq_init.copy()
    best_f = f.copy()
    best_M =  float('inf')
    it = 0
    while time.time() < deadline:
        # compute conv and M
        n = len(f)
        # Use FFT:
        mlen = 1 << ((2*n - 1).bit_length())
        fa = np.fft.rfft(f, mlen)
        conv = np.fft.irfft(fa * fa, mlen)[:2*n-1]
        M = conv.max()
        if M < best_M:
            best_M = M
            best_f = f.copy()
        # optional early break if improvement small? Not necessary.
        # Build A_ub matrix (2n-1, n) as described
        A = np.empty((2*n - 1, n), dtype=np.float64)
        # using broadcasting may be slower; use loop
        for i in range(n):
            A[i:i+n, i] = f[i]
        # Solve LP maximize sum b subject to A @ b <= M, b >=0
        c = -np.ones(n)  # minimizing -sum b = maximizing sum b
        # use linprog with bounds (0, None)
        res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
        if not res.success:
            # fallback: break
            print("[FW] LP failed")
            break
        b = res.x
        sum_b = b.sum()
        if sum_b <= 0:
            break
        # scale to target sum
        scale = target_sum / sum_b
        g = b * scale
        # update f with convex combination
        # step may be adaptive: reduce when no improvement
        f = (1 - step) * f + step * g
        # optional normalization to exact target sum (should be)
        f_sum = f.sum()
        if f_sum != target_sum:
            f = f * (target_sum / f_sum)
        it += 1
        # adjust step based on improvement
        # maybe gradually reduce step
        if it % 20 == 0:
            step = max(step * 0.9, 0.01)
    return best_f

"""
Propose a step‐function with a low evaluation score.

Algorithm
---------
1. **Small‑scale Frank–Wolfe optimisation**
   * Work with a modest size `n_small` (≈1500) to keep the linear program
     tractable.
   * Initialise the vector with a few diverse seeds (uniform, symmetric
     beta, asymmetric beta, Dirichlet draws).  All seeds are scaled to the
     target ℓ₁‑norm `√(2·n_small)`.  The Frank–Wolfe iteration proceeds as
     described in the challenge statement:
        – compute `M = max(conv(f,f))`;
        – solve   max ∑bᵢ  s.t. conv(f,b) ≤ M , b ≥ 0   (a linear program);
        – rescale `b` to the same ℓ₁‑norm,
        – update `f ← (1‑t)·f + t·b` with a decreasing step size `t`.
   The LP is solved with `scipy.optimize.linprog` (HiGHS).  The best `f`
   (smallest `M`) seen during the time slice is kept.

2. **Upscaling**
   The best small vector is interpolated to the final length `n_big = 5000`
   and renormalised to sum = 1.

3. **Fine‑grained Adam‑style refinement**
   A multiplicative (log‑space) Adam update is applied to the full‑size
   vector, using the exact sub‑gradient of the max‑convolution (the same
   idea as in the reference code).  This final stage exploits the remaining
   time budget.

The whole routine respects the 1000‑second total budget and returns a list
of non‑negative floats suitable for `evaluate_sequence`.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Fast evaluation (identical to the supplied evaluate_sequence but without
# the security checks).  Used only for internal monitoring.
# ----------------------------------------------------------------------
def _fast_evaluation(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) / (sum(seq)²).  The metric is scale‑invariant."""
    n = seq.size
    if n == 0:
        return float("inf")
    m = 1 << ((2 * n - 1).bit_length())      # next power of two
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    M = conv.max()
    s = seq.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * M / (s * s))


# ----------------------------------------------------------------------
# Seed generators (same as in the previous solution)
# ----------------------------------------------------------------------
def _beta_sequence(alpha: float, beta: float, length: int) -> np.ndarray:
    xs = (np.arange(length, dtype=float) + 0.5) / length
    a = np.power(xs, alpha - 1.0, out=np.empty_like(xs))
    b = np.power(1.0 - xs, beta - 1.0, out=np.empty_like(xs))
    seq = a * b
    seq[np.isinf(seq)] = 0.0
    seq = np.maximum(seq, 0.0)
    if seq.sum() == 0.0:
        seq = np.full(length, 1.0 / length)
    else:
        seq /= seq.sum()
    return seq


def _random_dirichlet(length: int, concentration: float = 0.5) -> np.ndarray:
    seq = np.random.gamma(concentration, 1.0, size=length)
    if seq.sum() == 0.0:
        seq = np.full(length, 1.0 / length)
    else:
        seq /= seq.sum()
    return seq


# ----------------------------------------------------------------------
# Frank‑Wolfe optimisation on the simplex (LP‑based)
# ----------------------------------------------------------------------
def _fw_optimize(f0: np.ndarray, target_sum: float, deadline: float,
                 step0: float = 0.25, step_min: float = 0.02) -> np.ndarray:
    """
    Perform a Frank–Wolfe style optimisation:
        maximise   ∑bᵢ     subject to   (f * b) ≤ M   and  b ≥ 0,
    where M = max(conv(f,f)).  The solution `b` is rescaled to the same ℓ₁‑norm
    as `f` and mixed back using a decreasing convex‑combination step size.
    Returns the best vector (smallest max convolution) encountered.
    """
    f = f0.copy()
    n = f.size
    best_f = f.copy()
    best_M = _fast_evaluation(f)                # same as max(conv)/norm² for sum=target_sum
    step = step0
    it = 0
    # pre‑allocate RHS vector for the LP
    rhs = None

    while time.time() < deadline:
        # ----- current max convolution -----
        mfft = 1 << ((2 * n - 1).bit_length())
        fa = np.fft.rfft(f, mfft)
        conv = np.fft.irfft(fa * fa, mfft)[: 2 * n - 1]
        M = conv.max()
        if M < best_M:
            best_M = M
            best_f = f.copy()
            print(f"[FW] iter {it:5d}  best M = {best_M:.8f}")

        # ----- construct the convolution matrix A (A @ b ≤ M) -----
        # A has shape (2n‑1) × n, column i is f[i] placed on rows i … i+n‑1
        A = np.empty((2 * n - 1, n), dtype=np.float64)
        for i in range(n):
            A[i:i + n, i] = f[i]
        rhs = np.full(2 * n - 1, M, dtype=np.float64)

        # ----- linear program: maximise sum(b)  <=> minimise -sum(b) -----
        c = -np.ones(n, dtype=np.float64)          # minimise cᵀb = -∑bᵢ
        bounds = [(0.0, None)] * n
        res = linprog(c, A_ub=A, b_ub=rhs, bounds=bounds,
                      method="highs", options={"presolve": True})

        if not res.success:
            # If the LP fails we fall back to a tiny random perturbation.
            print("[FW] LP failed – applying small jitter")
            jitter = np.random.rand(n) * 1e-6
            f = np.maximum(f + jitter, 0.0)
            f_sum = f.sum()
            if f_sum > 0:
                f = f * (target_sum / f_sum)
            continue

        b = res.x
        sb = b.sum()
        if sb <= 0:
            break

        # ----- rescale b to the target ℓ₁‑norm -----
        scale = target_sum / sb
        g = b * scale

        # ----- convex combination update (log‑space is not needed here) -----
        f = (1.0 - step) * f + step * g
        # re‑normalise exactly (numeric drift)
        f_sum = f.sum()
        if f_sum > 0:
            f = f * (target_sum / f_sum)

        # ----- step‑size schedule -----
        it += 1
        if it % 10 == 0:
            step = max(step * 0.93, step_min)

    return best_f


# ----------------------------------------------------------------------
# Adam‑style multiplicative refinement on the simplex (same idea as before)
# ----------------------------------------------------------------------
def _adam_refine(seq: np.ndarray, deadline: float) -> np.ndarray:
    n = seq.size
    eps = 1e-15
    log_seq = np.log(seq + eps)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    lr = 0.15
    m = np.zeros_like(seq)
    v = np.zeros_like(seq)
    t = 0

    best_seq = None
    best_score = float("inf")
    it = 0
    fft_len = 1 << ((2 * n - 1).bit_length())

    while time.time() < deadline:
        # ----- recover probability vector -----
        a = np.exp(log_seq)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
            log_seq = np.log(a)
        else:
            a /= s

        # ----- convolution and active set -----
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        M = conv.max()
        tol = 1e-12
        active = np.where(conv >= M - tol)[0]

        # ----- sub‑gradient of the max‑operator -----
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            i_range = np.arange(lo, hi + 1, dtype=int)
            j_range = k - i_range
            g[i_range] += a[j_range]

        # ----- Adam moments -----
        t += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # ----- multiplicative (log‑space) update -----
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        log_seq -= step
        np.clip(log_seq, -30.0, 30.0, out=log_seq)

        # ----- keep the best -----
        score = _fast_evaluation(a)
        if score < best_score - 1e-12:
            best_score = score
            best_seq = a.copy()
            if it % 500 == 0:
                print(f"[Adam] iter {it:6d}  best score {best_score:.6f}")

        # occasional jitter to escape stagnation
        if np.random.rand() < 0.001:
            j = np.random.randint(n)
            a[j] = max(0.0, a[j] + np.random.randn() * 1e-3)
            a /= a.sum()
            log_seq = np.log(a + eps)

        it += 1

    if best_seq is None:
        a = np.exp(log_seq)
        a /= a.sum()
        best_seq = a
    return best_seq


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Produce a step‑function whose metric 2·n·max(conv)/sum² is as small as
    possible within the allotted time budget.
    """
    np.random.seed(seed)
    start = time.time()
    # leave a small safety margin for the harness
    deadline = start + budget_s - 5.0

    # --------------------------------------------------------------
    # 1. Small‑scale Frank–Wolfe stage
    # --------------------------------------------------------------
    n_small = 1500
    target_sum_small = math.sqrt(2.0 * n_small)

    # a few diverse initial vectors (all are scaled to the target ℓ₁‑norm)
    init_seeds = []

    # uniform
    init_seeds.append(np.full(n_small, target_sum_small / n_small))

    # symmetric beta families (including the classic arcsine)
    for a in [0.5, 0.35, 0.25, 0.20, 0.15, 0.10]:
        beta_vec = _beta_sequence(a, a, n_small) * target_sum_small
        init_seeds.append(beta_vec)

    # a handful of asymmetric shapes
    for a, b in [(0.3, 0.7), (0.7, 0.3), (0.2, 0.8), (0.8, 0.2)]:
        beta_vec = _beta_sequence(a, b, n_small) * target_sum_small
        init_seeds.append(beta_vec)

    # random Dirichlet draws (concentration < 1 → spikier)
    for _ in range(3):
        d = _random_dirichlet(n_small, concentration=0.4) * target_sum_small
        init_seeds.append(d)

    # time budget for the FW phase (≈70 % of total)
    fw_deadline = start + 0.70 * budget_s
    best_small = None
    best_M_small = float("inf")

    for idx, seed_vec in enumerate(init_seeds):
        if time.time() >= fw_deadline:
            break
        # Normalise exactly to the required ℓ₁‑norm
        s = seed_vec.sum()
        if s == 0.0:
            seed_vec = np.full(n_small, target_sum_small / n_small)
        else:
            seed_vec = seed_vec * (target_sum_small / s)

        # allocate roughly equal remaining time for this seed
        remaining = fw_deadline - time.time()
        per_seed = max(5.0, remaining / (len(init_seeds) - idx))
        seed_deadline = time.time() + per_seed

        refined = _fw_optimize(seed_vec, target_sum_small, seed_deadline,
                               step0=0.25, step_min=0.02)

        # evaluate the obtained vector (metric = max‑convolution because sum is √(2 n))
        m_val = _fast_evaluation(refined)
        print(f"[seed #{idx}]  metric {m_val:.6f}")

        if m_val < best_M_small:
            best_M_small = m_val
            best_small = refined.copy()
            print(f"*** new global best (small) = {best_M_small:.6f}")

    if best_small is None:
        best_small = np.full(n_small, target_sum_small / n_small)

    # --------------------------------------------------------------
    # 2. Upscale to the final length
    # --------------------------------------------------------------
    n_big = 5000
    x_small = np.arange(n_small, dtype=float)
    x_big = np.linspace(0, n_small - 1, n_big)
    up_seq = np.interp(x_big, x_small, best_small)

    # enforce non‑negativity and renormalise to sum = 1 (any scale works)
    up_seq = np.maximum(up_seq, 0.0)
    if up_seq.sum() == 0.0:
        up_seq = np.full(n_big, 1.0 / n_big)
    else:
        up_seq /= up_seq.sum()

    # --------------------------------------------------------------
    # 3. Final Adam‑style refinement on the full size
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("[final] starting Adam refinement on full size...")
        final_seq = _adam_refine(up_seq, deadline)
    else:
        final_seq = up_seq

    # final clean‑up
    final_seq = np.maximum(final_seq, 0.0)
    total = final_seq.sum()
    if total == 0.0:
        final_seq = np.full_like(final_seq, 1.0 / final_seq.size)
    else:
        final_seq /= total

    final_metric = _fast_evaluation(final_seq)
    print(f"[summary] final metric {final_metric:.6f} with length {final_seq.size}")

    # return a plain Python list as required by the harness
    return final_seq.tolist()
