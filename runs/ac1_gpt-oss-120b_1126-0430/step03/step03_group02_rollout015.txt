def _smoothmax_gradient(a, tau):
    conv = _conv_fft(a)  # length 2n-1
    max_conv = conv.max()
    # compute softmax with temperature tau
    # subtract max for stability
    scaled = (conv - max_conv) / tau
    exp_scaled = np.exp(scaled)
    weights = exp_scaled / exp_scaled.sum()
    # compute gradient g = 2 * cross-correlation of a with weights
    # Need cross-correlation: g_j = sum_{i where i-j in range} weights[i] * a[i-j]
    # Equivalent to convolution of a and weights_rev
    n = a.size
    L = weights.size
    size = 1 << ((n + L - 2).bit_length())
    A = np.fft.fft(a, n=size)
    W_rev = np.fft.fft(weights[::-1], n=size)
    conv2 = np.fft.ifft(A * W_rev).real
    # g = 2 * conv2[n-1 : n-1+n]
    g = 2.0 * conv2[n-1 : n-1 + n]
    return max_conv, weights, g

def _optimize_smoothmax(a0, deadline, eta=0.7, tau=0.05, epsilon=1e-12, patience=2000):
    a = a0.copy()
    n = a.size
    best_a = a.copy()
    best_obj = _objective(a)
    # accum for AdaGrad
    accum = np.zeros_like(a)
    # maybe have step count indexing
    it = 0
    no_improve = 0

    # compute FFT size based on n (max needed for conv+grad)
    size = 1 << ((3 * n - 2).bit_length())
    # pre-allocate maybe?

    while time.time() < deadline:
        # compute FFT of a once
        A = np.fft.fft(a, n=size)
        # conv = a*a
        conv_full = np.fft.ifft(A * A).real
        conv = conv_full[: 2 * n - 1]

        # compute softmax weights
        max_conv = conv.max()
        # temperature maybe schedule: tau decaying
        # we could compute tau_now = max(0.01, tau * (1 - it / 500000))
        tau_now = max(0.01, tau * (1.0 - it / 200000.0))
        # compute scaled
        scaled = (conv - max_conv) / tau_now
        # for stability, subtract max of scaled
        # but max of scaled is 0 (since conv-max=0 at max), so safe.
        exp_scaled = np.exp(scaled)
        w = exp_scaled / exp_scaled.sum()

        # gradient via cross-correlation
        # compute FFT of reversed w
        B = np.fft.fft(w[::-1], n=size)
        grad_full = np.fft.ifft(A * B).real
        # extract g entries
        idx = (2 * n - 2) - np.arange(n)  # descending
        g = 2.0 * grad_full[idx]

        # update accum
        accum += g * g

        # compute step
        step = eta * g / (np.sqrt(accum) + epsilon)

        # exponentiated update (log domain)
        log_a = np.log(a + epsilon) - step
        # shift to avoid overflow
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a_sum = a.sum()
        if a_sum == 0:
            # fallback to uniform
            a = np.full_like(a, 1.0 / n)
        else:
            a /= a_sum

        # evaluate occasional

                    a = a / a.sum()
                    accum = np.zeros_like(a)
                    no_improve = 0

        it += 1

    return best_a, best_obj

assistantfinal**Solution Overview**

We minimise  

\[
O(a)=\frac{2\,n\,\max_k (a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\qquad a_i\ge 0,\; \sum_i a_i>0 .
\]

Because the bound is scale‑invariant we may restrict ourselves to the simplex  
\(\{a\ge 0,\;\sum a_i = 1\}\).  
`O(a)` is a convex, non‑smooth function (a maximum of quadratic forms).  
We smooth the maximum with a temperature‑parameterised soft‑max

\[
\tilde O_\tau(a)=\tau\log\Bigl(\sum_k \exp\bigl((a*a)_k/\tau\bigr)\Bigr),
\]

whose gradient is

\[
\nabla \tilde O_\tau(a)_j = 2\sum_k
\underbrace{\frac{\exp((a*a)_k/\tau)}{\sum_\ell \exp((a*a)_\ell/\tau)}}_{w_k}
a_{k-j},
\]

i.e. a weighted cross–correlation of the current vector `a` with the soft‑max
weights `w`.  The gradient can be evaluated with two FFTs per iteration.

We perform **mirrored (exponentiated) gradient descent** on the simplex:

```
log_a ← log(a)  –  η · (∇Ō / √(acc + ε))
a      ← exp(log_a) / sum(exp(log_a))
```

`acc` stores the accumulated squared gradients (AdaGrad‑style) so that each
coordinate gets an adaptive step‑size.  The temperature τ is slowly reduced
during the run, starting with a smooth objective and ending with an
approximation of the true maximum.

The optimiser is run from several sensible starting points (previous best,
√‑law, power‑law p‑= 0.6, random).  For each length we keep the best vector
found, finally returning the globally best sequence.  All heavy work is
performed with FFTs, so even for `n≈8000` the inner loop runs comfortably
within the 1000 s budget.

The returned list is fed to the supplied `evaluate_sequence` to obtain the
exact bound.

```python
"""
Propose a step‑function that minimises the bound
    O = 2·n·max(conv(a,a)) / (sum(a)**2)

The method:
  1. Work on the probability simplex (Σa = 1, a ≥ 0).
  2. Smooth the max‑operator with a soft‑max (temperature τ) and
     obtain the exact gradient via two FFTs per iteration.
  3. Perform mirrored (exponentiated) gradient descent with AdaGrad‑style
     per‑coordinate step size and a slowly decreasing temperature.
  4. Run the optimiser from several diverse starts and a few different
     lengths; keep the best solution.
  5. Return the best sequence (list of floats) – the caller will invoke
     evaluate_sequence() for the final score.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small helper – smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a * a using a power‑of‑two FFT.
    Returns the full (2·n‑1) convolution.
    """
    n = a.size
    size = _next_pow2(2 * n - 1)
    A = np.fft.fft(a, n=size)
    conv_full = np.fft.ifft(A * A).real
    return conv_full[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact bound 2·n·max(conv(a,a)) for a already normalised (Σa=1)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    A = np.fft.fft(a, n=size)
    conv = np.fft.ifft(A * A).real[: 2 * n - 1]
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
def _smoothmax_gradient(a: np.ndarray, tau: float, size: int):
    """
    Returns
        conv      – the full convolution a*a (length 2·n‑1)
        w         – soft‑max weights (sum to 1)
        g         – gradient of the smoothed max w.r.t. a
    All operations use the provided FFT size (≥ 3·n‑2).
    """
    n = a.size
    # FFT of a (once)
    A = np.fft.fft(a, n=size)

    # ----- convolution a*a (needed for the soft‑max) -----
    conv_full = np.fft.ifft(A * A).real
    conv = conv_full[: 2 * n - 1]

    # ----- soft‑max over convolution entries -----
    max_conv = conv.max()
    # (conv‑max)/τ ≤ 0, numerically stable
    scaled = (conv - max_conv) / tau
    exp_scaled = np.exp(scaled)          # max element becomes 1
    w = exp_scaled / exp_scaled.sum()    # length 2·n‑1, sums to 1

    # ----- gradient: 2 * (a ⋆ w)  (cross‑correlation) -----
    # cross‑correlation a ⋆ w = a * reverse(w)
    B = np.fft.fft(w[::-1], n=size)
    grad_full = np.fft.ifft(A * B).real
    # indices giving Σ_i a_i w_{i+j}  for j = 0..n‑1
    idx = (2 * n - 2) - np.arange(n)
    g = 2.0 * grad_full[idx]

    return conv, w, g


# ----------------------------------------------------------------------
def _optimize_smoothmax(a0: np.ndarray, deadline: float,
                        eta: float = 0.7,
                        tau0: float = 0.05,
                        tau_min: float = 0.01,
                        eps: float = 1e-12,
                        patience: int = 2000) -> tuple[np.ndarray, float]:
    """
    Mirrored Adam/AdaGrad on the soft‑max smoothed objective.
    Returns the best vector encountered and its exact (non‑smoothed) bound.
    """
    a = a0.copy()
    n = a.size
    best_a = a.copy()
    best_val = _objective(a)

    # AdaGrad accumulator
    acc = np.zeros_like(a)

    # FFT size that can hold both the convolution and the cross‑correlation
    fft_size = _next_pow2(3 * n - 2)

    it = 0
    no_improve = 0

    while time.time() < deadline:
        # linearly decay temperature to tau_min
        tau = max(tau_min, tau0 * (1.0 - it / 200_000.0))

        conv, w, g = _smoothmax_gradient(a, tau, fft_size)

        # AdaGrad update on the log‑domain (mirrored gradient descent)
        acc += g * g
        step = eta * g / (np.sqrt(acc) + eps)

        log_a = np.log(a + eps) - step
        # shift to avoid overflow/underflow
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a_sum = a.sum()
        if a_sum == 0.0:                     # extreme – fall back to uniform
            a = np.full_like(a, 1.0 / n)
        else:
            a /= a_sum

        # occasional exact evaluation
        if it % 100 == 0:
            cur = _objective(a)
            if cur < best_val - 1e-12:
                best_val = cur
                best_a = a.copy()
                no_improve = 0
                print(f"[opt] iter {it:6d}  new best O={best_val:.6f}")
            else:
                no_improve += 1
                if no_improve >= patience:
                    # escape flat regions by a small random jitter
                    jitter = np.random.randn(n) * 1e-4
                    a = best_a + jitter
                    a = np.clip(a, 0.0, None)
                    a = a / a.sum()
                    acc.fill(0.0)
                    no_improve = 0

        it += 1

    return best_a, best_val

    def _prepare_start(base, n):

            a = base[:n]
        else:

        a = np.maximum(a, 0.0)
        if a.sum() == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= a.sum()
        return a

    # ------------------------------------------------------------------
    # Choose a few lengths – larger n offers more freedom but costs a bit
    # more per iteration.  We split the budget roughly equally.
    lengths = [4000, 8000]
    best_seq = None
    best_score = float("inf")

    for n in lengths:
        if time.time() >= deadline:
            break

        # Build start points

        a_rand = np.random.rand(n)
        a_rand /= a_rand.sum()
```
