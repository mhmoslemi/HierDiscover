def propose_candidate(seed=123, budget_s=1000):
    import time, numpy as np
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0

    # parameters
    small_n = 5000
    big_n = 40000

    # define helper: compute objective for sum=1 vec
    def objective(a):
        # a is np array sum=1
        n = a.shape[0]
        # use FFT to compute conv
        size = 1 << ((2*n - 1).bit_length())
        fa = np.fft.rfft(a, size)
        conv = np.fft.irfft(fa * fa, size)[:2*n-1]
        return float(2.0 * n * np.max(conv))

    # Candidate types:
    # type: 'catalan', 'power', 'exp'
    best_obj = np.inf
    best_type = None
    best_params = None

    # search power law
    alphas = np.arange(1.01, 2.01, 0.05)
    for alpha in alphas:
        a = np.arange(1, small_n+1, dtype=np.float64) ** (-alpha)
        a /= a.sum()
        val = objective(a)
        if val < best_obj:
            best_obj = val
            best_type = 'power'
            best_params = alpha

    # search exponential
    rs = np.arange(0.5, 0.99, 0.02)
    for r in rs:
        a = (1 - r) * (r ** np.arange(small_n, dtype=np.float64))
        # sum already =1 for infinite series; truncated sum <1, normalize
        a /= a.sum()
        val = objective(a)
        if val < best_obj:
            best_obj = val
            best_type = 'exp'
            best_params = r

    # test catalan
    a_catalan = np.empty(small_n, dtype=np.float64)
    a_catalan[0] = 1.0

    a_catalan /= a_catalan.sum()
    val = objective(a_catalan)
    if val < best_obj:
        best_obj = val
        best_type = 'catalan'
        best_params = None

    print(f"[scan] best type {best_type} param {best_params} obj {best_obj:.9f}")

    # Build large initial vector based on best type
    n = big_n
    if best_type == 'power':

"""
Propose a low‑objective step‑function via a multi‑stage search.

1.  Scan a few simple parametric families (Catalan, power‑law,
    geometric) on a modest size (n≈5 000) to pick a promising shape.
2.  Upscale the best shape to a larger problem size (n≈50 000)
    and refine it with an Adam‑style projected optimizer that works
    directly on the simplex (sum = 1, a_i≥0).  The gradient is the
    exact sub‑gradient of the active max‑convolution entry.
3.  If time permits, perform a few cheap random restarts and a final
    hill‑climbing phase that swaps small masses between coordinates.
The whole routine respects the 1000 s budget and returns the best
vector found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear convolution via real FFT (full “linear” mode)
# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2*len(a)-1) using real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {x≥0, Σx=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective (exactly the value returned by evaluate_sequence for Σa=1)
# ----------------------------------------------------------------------
def _objective_normed(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) for a of L1‑norm 1."""
    n = a.shape[0]
    conv = _fft_convolve(a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------
def _catalan_normalised(n: int) -> np.ndarray:
    """Catalan coefficients a_i = C(2i,i)/4**i, normalised to sum = 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    a /= a.sum()
    return a


def _power_law_normalised(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-alpha}, normalised to sum = 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-alpha)
    a /= a.sum()
    return a


def _geometric_normalised(n: int, r: float) -> np.ndarray:
    """Geometric distribution a_i = (1‑r)·r**i (truncated & renormalised)."""
    # infinite‑series sum would be 1, but we truncate for finite n
    a = (1.0 - r) * (r ** np.arange(n, dtype=np.float64))
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Adam‑style projected optimizer on the simplex
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   max_iters: int = 500_000,
                   verbose: bool = True) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimisation of 2·n·max(a*a) with sum(a)=1.
    Returns the best vector discovered and its objective.
    """
    a = a0.copy()
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best = a.copy()
    best_val = _objective_normed(best)

    t = 0
    while time.time() < deadline and t < max_iters:
        # convolution and max index
        fa = np.fft.rfft(a, size)
        conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
        max_c = conv.max()
        # active set (indices whose value is within a tiny tolerance)
        tol = 1e-12
        active = np.where(conv >= max_c - tol)[0]

        # compute sub‑gradient (average over the active set)
        grad = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx = np.arange(i_start, i_end + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]
        grad /= max(1, len(active))

        # Adam parameter update
        t += 1
        lr = 0.5 / np.sqrt(t)            # simple diminishing schedule
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Project back onto the simplex
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        cur_val = 2.0 * n * max_c
        if cur_val < best_val:
            best_val = cur_val
            best = a.copy()
            if verbose:
                print(f"[improve] iter {t:07d}  obj {best_val:.9f}")

        # occasional tiny random kick to escape plateaus
        if t % 5_000 == 0:
            pert = np.random.normal(scale=0.005, size=n)
            a = np.maximum(a + pert, 0.0)
            a = _project_simplex(a)

    return best, best_val


# ----------------------------------------------------------------------
# Simple hill‑climbing refinement (swap a small amount of mass)
# ----------------------------------------------------------------------
def _hill_climb(a: np.ndarray,
                deadline: float,
                steps: int = 20_000) -> tuple[np.ndarray, float]:
    best = a.copy()
    best_val = _objective_normed(best)
    n = a.shape[0]

    for step in range(steps):
        if time.time() >= deadline:
            break
        # pick source and destination indices
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # move a tiny fraction of the source mass to the destination
        delta = 0.01 * best[i]            # 1 % of source mass
        if delta <= 0.0:
            continue
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        # ensure feasibility (non‑negative + sum = 1)
        if cand[i] < 0.0:
            continue
        cand = _project_simplex(cand)      # tiny correction for numerical drift
        val = _objective_normed(cand)
        if val < best_val:
            best_val = val
            best = cand
    return best, best_val

# ----------------------------------------------------------------------

    """

    # 1) Quick scan on a modest size to decide which family looks best.
    # ------------------------------------------------------------------
    small_n = 5_000

    best_type = None
    best_param = None

    cand = _catalan_normalised(small_n)

    # Power‑law exponents
    for alpha in np.arange(1.05, 2.05, 0.05):
        cand = _power_law_normalised(small_n, alpha)
        val = _objective_normed(cand)
        if val < best_obj:
            best_obj, best_type, best_param = val, "power", alpha

    # Geometric ratios
    for r in np.arange(0.55, 0.96, 0.03):
        cand = _geometric_normalised(small_n, r)
        val = _objective_normed(cand)
        if val < best_obj:
            best_obj, best_type, best_param = val, "geom", r

    print(f"[scan] best family {best_type} param {best_param} obj {best_obj:.9f}")

    # ------------------------------------------------------------------
    # 2) Upscale to a larger problem size for the final optimisation.
    # ------------------------------------------------------------------
    big_n = 50_000
    if best_type == "catalan":
        a0 = _catalan_normalised(big_n)
    elif best_type == "power":
        a0 = _power_law_normalised(big_n, best_param)   # type: ignore[arg-type]
    else:  # geometric
        a0 = _geometric_normalised(big_n, best_param)   # type: ignore[arg-type]

    # ------------------------------------------------------------------
    # 3) Main Adam optimisation (as many iterations as time permits).
    # ------------------------------------------------------------------
    best_seq, best_val = _adam_optimize(a0, deadline, max_iters=400_000, verbose=True)

    # ------------------------------------------------------------------
    # 4) Optional random restarts if the budget still allows.
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        a_rand = np.random.rand(big_n)
        a_rand /= a_rand.sum()
        cand_seq, cand_val = _adam_optimize(a_rand, deadline, max_iters=80_000, verbose=False)
        if cand_val < best_val:
            best_val, best_seq = cand_val, cand_seq.copy()
            print(f"[restart] new best obj {best_val:.9f}")

    # ------------------------------------------------------------------
    # 5) Final local hill‑climbing refinement.
    # ------------------------------------------------------------------
    best_seq, best_val = _hill_climb(best_seq, deadline, steps=30_000)

    print(f"[final] best objective ≈ {best_val:.9f} with n={big_n}")
    return best_seq.tolist()
