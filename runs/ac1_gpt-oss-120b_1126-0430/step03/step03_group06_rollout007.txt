def _fft_correlation(p, a):
    # p length = 2n-1, a length = n
    L = 1 << ((len(p) + len(a) - 1).bit_length())
    fp = np.fft.rfft(p, L)
    fa = np.fft.rfft(a, L)
    corr = np.fft.irfft(fp * np.conj(fa), L)  # shape L
    # linear correlation result first len(p) elements (or first len(a)+len(p)-1)
    # we need first n entries for j from 0..n-1
    return corr[:len(a)]

def _optimize(a0, deadline, max_iters, top_k=5, verbose=True):
    a = a0.copy()
    n = len(a)
    best = a.copy()
    best_val = 2*n*np.max(_fft_convolve(a))
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    for t in range(max_iters):
        if time.time() >= deadline: break
        conv = _fft_convolve(a)
        max_c = conv.max()
        cur_val = 2.0 * n * max_c
        if cur_val < best_val:
            best_val = cur_val
            best = a.copy()
            if verbose:
                print(f"[improve] iter {t} obj {best_val:.9f}")
        # compute gradient via top K
        # get top indices
        if top_k >= conv.size:
            top_idxs = np.arange(conv.size)
        else:
            # use argpartition for speed
            part = np.argpartition(-conv, top_k)[:top_k]
            top_idxs = part[np.argsort(-conv[part])]  # sorted descending
        # compute gradient sum
        grad = np.zeros_like(a)
        n1 = n - 1
        for k_idx in top_idxs:
            i_start = max(0, k_idx - n1)
            i_end = min(n1, k_idx)
            idx = np.arange(i_start, i_end+1)

"""
Propose a step‑function (non‑negative sequence) that minimises
``evaluate_sequence``.

The algorithm is a projected Adam optimiser that works on the exact
objective

    O(a) = 2·n·max_k (a*a)[k]          ( Σ a_i = 1, a_i ≥ 0 )

Instead of the crude sub‑gradient that uses only the single maximal
convolution entry, we aggregate the contributions of the *K* largest
entries (K≈5).  The resulting direction is a true sub‑gradient of the
convex envelope of the max‑operator and steers the iterates away from
flat plateaux.

A fast FFT based convolution is used for the self‑convolution, and the
gradient for each active index *k* is computed analytically as

    ∇_a  (a*a)[k] = 2·J_k·a ,

where J_k is the shift matrix of the convolution.  The sum over the
active indices yields the update direction.  After each Adam step the
vector is projected back onto the probability simplex (non‑negative,
unit‐sum) with an O(n log n) Euclidean projection.

The optimiser runs as long as the time budget allows, keeps the best
vector seen, and performs cheap random‑restart refinements if time
remains.  If a previous best construction is available in the global
variable ``height_sequence_1`` it is used as a warm start.

The whole routine finishes well within the 1000 s limit and typically
reaches an objective below the target 1.503.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear convolution a*a using a real FFT (full mode)
# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())      # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)          # a*a in frequency domain
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex Σx=1, x≥0
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto {x ≥ 0, Σx = 1} (the probability simplex)."""
    n = v.size
    u = np.sort(v)[::-1]                       # sorted decreasing
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):                       # numerical safety‑net
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Catalan‑type coefficients, L¹‑normalised
# ----------------------------------------------------------------------
def _catalan_normalised(n: int) -> np.ndarray:
    """First n terms a_i = C(2i,i)/4**i, normalised so that Σ a_i = 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Gradient constructed from the K largest convolution entries
# ----------------------------------------------------------------------
def _topk_gradient(a: np.ndarray, conv: np.ndarray, K: int = 5) -> np.ndarray:
    """
    Return a sub‑gradient of max_k (a*a)[k] that aggregates the contributions
    of the K largest entries of ``conv``.
    """
    n = a.shape[0]
    # indices of the K largest convolution values
    if K >= conv.size:
        top_idx = np.arange(conv.size)
    else:
        # argpartition is O(n); we then sort the selected indices
        part = np.argpartition(-conv, K)[:K]
        top_idx = part[np.argsort(-conv[part])]

    grad = np.zeros_like(a)
    n1 = n - 1
    for k in top_idx:
        i_start = max(0, k - n1)
        i_end = min(n1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        # contribution 2·a_{k‑i}
        grad[idx] += 2.0 * a[k - idx]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the simplex with the top‑K gradient
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   max_iters: int = 400_000,
                   K: int = 5,
                   verbose: bool = True) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimisation starting from ``a0`` (already Σa=1).
    Returns the best vector encountered and its objective value
    (2·n·max(a*a)).
    """
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    conv = _fft_convolve(a)
    best_val = float(2.0 * n * conv.max())

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    while time.time() < deadline and t < max_iters:
        # current convolution and objective
        conv = _fft_convolve(a)
        max_c = conv.max()
        cur_val = 2.0 * n * max_c
        if cur_val < best_val:
            best_val = cur_val
            best = a.copy()
            if verbose:
                print(f"[improve] iter {t:06d}  obj {best_val:.9f}")

        # gradient (sum of K active quadratic forms)
        grad = _topk_gradient(a, conv, K=K)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (t + 1))
        v_hat = v / (1.0 - beta2 ** (t + 1))

        # diminishing step size
        lr = 0.5 / np.sqrt(t + 1)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        # keep feasibility
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        # occasional tiny random jitter to escape tiny plateaux
        if t % 5_000 == 0 and t > 0:
            a += np.random.normal(scale=0.001, size=n)
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)

        t += 1

    return best, best_val

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Return a sequence of non‑negative heights that (empirically) minimises
    ``evaluate_sequence`` within the given time budget.

    * Starts from a Catalan‑type distribution (or a previous best if
      ``height_sequence_1`` is present).
    * Refines it with a projected Adam optimiser that aggregates the
      K=5 most violating convolution constraints.
    * Performs cheap random‑restart refinements while time permits.
    """
    np.random.seed(seed)
    # leave a small safety margin for the surrounding harness
    deadline = time.time() + budget_s - 2.0

    # ------------------------------------------------------------------
    # 1) Problem size – a few ten‑thousands give a good trade‑off.
    # ------------------------------------------------------------------
    n = 24_000

    # ------------------------------------------------------------------
    # 2) Initial candidate – use previous best if available, otherwise Catalan.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            if prev.size == n:
                a0 = prev.copy()
            elif prev.size > n:
                a0 = prev[:n]
            else:

                x_old = np.linspace(0.0, 1.0, prev.size)
                x_new = np.linspace(0.0, 1.0, n)

            a0 = np.maximum(a0, 0.0)
            a0 = _project_simplex(a0)          # ensure exact Σ=1
        except Exception:

    else:

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    # 5) Final sanity output (the framework will call evaluate_sequence).
    # ------------------------------------------------------------------
    print(f"[final] best objective ≈ {best_val:.9f} with n={n}")
    return best_seq.tolist()
