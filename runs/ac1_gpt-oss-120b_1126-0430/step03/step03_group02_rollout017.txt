"""
Propose a non‑negative step function that minimises the bound

    O = 2·n·max(conv(a,a)) / (sum(a)²)

as evaluated by the supplied ``evaluate_sequence``.  

The method combines three ideas:

1. **Parametric sweeps** – generate simple power‑law families
   a_i ∝ (i+1)^(-α) (including the classical 1/√i law) on several
   support sizes and keep the best one.

2. **Exponential‑gradient refinement** – starting from the best
   candidates, run a mirror‑descent / exponentiated‑gradient loop
   that follows a sub‑gradient of the max‑convolution while projecting
   onto the probability simplex.

3. **Tail pruning** – after each refinement we cut off the trailing
   part of the vector that carries only a tiny amount of total mass,
   renormalise and re‑evaluate.  Since the objective contains the factor
   ``2·n``, shortening the support can considerably lower the bound.

All helpers are top‑level (no closures) and the whole search is
time‑capped (default 1000 s).  Progress and the final exact score are
printed.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all entries are equal
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a power‑of‑two FFT."""
    n = a.size
    size = 1 << ((2 * n - 2).bit_length())
    A = np.fft.fft(a, n=size)
    conv = np.fft.ifft(A * A).real[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Compute the surrogate objective 2·n·max(conv(a,a)) for a vector
    already normalised to Σa = 1.
    """
    n = a.size
    conv = _conv_fft(a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) w.r.t. a (on the simplex).
    The sub‑gradient is formed from *all* indices where the convolution
    attains its maximum (with a tiny tolerance).
    """
    n = a.size
    conv = _conv_fft(a)
    max_val = conv.max()
    ks = np.where(np.abs(conv - max_val) < 1e-12)[0]
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
def _exp_update(a: np.ndarray, g: np.ndarray, eta: float) -> np.ndarray:
    """
    Exponentiated‑gradient step respecting the simplex.
    Works in log‑space for stability.
    """
    log_a = np.log(a + 1e-15) - eta * g
    log_a -= np.max(log_a)                # avoid overflow
    a_new = np.exp(log_a)
    a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
def _optimize(a0: np.ndarray, max_iters: int, eta0: float,
              deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple exponentiated‑gradient descent on the sub‑gradient of the
    max‑convolution.  Returns the best vector found and its objective.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)

    n = a.size
    patience = max_iters // 20 + 1
    no_improve = 0

    for t in range(max_iters):
        if time.time() >= deadline:
            break

        g = _subgradient_max_conv(a)
        eta = eta0 * (1.0 - t / max_iters)    # linear decay
        a = _exp_update(a, g, eta)

        cur_score = _objective(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= patience:       # escape flat regions
                jitter = np.random.randn(n) * 1e-4
                a = best_a + jitter
                a = np.clip(a, 0.0, None)
                a = _project_to_simplex(a)
                no_improve = 0

    return best_a, best_score


# ----------------------------------------------------------------------
def _prune_tail(a: np.ndarray, mass: float = 0.999,
                min_len: int = 30) -> np.ndarray:
    """
    Drop the (right) tail of the vector that carries at most (1‑mass) of the
    total probability mass, then renormalise.
    The function assumes a is roughly monotone decreasing, which is true
    for all shapes we generate.
    """
    csum = np.cumsum(a)
    idx = np.searchsorted(csum, mass, side='right')
    if idx < min_len:
        idx = min_len - 1
    idx = min(idx, a.size - 1)
    a_trunc = a[: idx + 1]
    a_trunc = a_trunc / a_trunc.sum()
    return a_trunc

    deadline = time.time() + budget_s - 1.0   # safety margin

    best_seq = None
