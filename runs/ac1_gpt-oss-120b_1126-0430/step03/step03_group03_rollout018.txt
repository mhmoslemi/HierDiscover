```python
"""
Optimization of a non‑negative step‑function sequence to minimise
evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a)^2).

The algorithm combines:
  1. Adam on a smooth log‑sum‑exp surrogate (progressively sharpening).
  2. Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
  3. A small linear programme (cvxpy) that maximises the L1‑norm of a
     partner vector while keeping its convolution with the current iterate
     below the current worst value; the partner is blended into the
     iterate.
  4. Simulated‑annealing hill‑climbing with random mass transfers.
All phases respect the 1000 s budget; the best vector found is returned.
"""

import time
import math
import numpy as np
import cvxpy as cp

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full (non‑circular) convolution a*a using FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
def _objective_simplex(a: np.ndarray, n: int, L: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) assuming sum(a)=1."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. a (average over all maximizing indices)."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        # each unordered pair contributes twice, but the sum over i already accounts for both
        g[i_idx] += 2.0 * a[j_idx]
    g /= len(ks)
    return g


# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, conv: np.ndarray, alpha: float, L: int) -> np.ndarray:
    """
    Gradient of the smooth surrogate
        (2·n/α)·log Σ_k exp(α·conv_k)
    """
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / exp_scaled.sum()                     # soft‑max weights

    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n = a.size
    grad = 2.0 * corr_rev[n - 1 : n - 1 + n]               # central n entries
    return grad


# ----------------------------------------------------------------------
def _exp_update(a: np.ndarray, g: np.ndarray, eta: float) -> np.ndarray:
    """Mirror‑descent / exponentiated‑gradient step on the simplex."""
    a_new = a * np.exp(-eta * g)
    # renormalise – the product is already non‑negative
    a_new_sum = a_new.sum()
    if a_new_sum == 0:
        # fallback to uniform if something went wrong
        return np.full_like(a, 1.0 / a.size)
    return a_new / a_new_sum


# ----------------------------------------------------------------------
def _lp_step(f: np.ndarray, M: float, max_iters: int = 2000) -> np.ndarray:
    """
    Solve the LP
        maximise   sum(g)
        s.t.       conv(f,g) <= M   (element‑wise)
                 g >= 0
    Returns a feasible g normalised to have unit sum.
    """
    n = f.shape[0]
    f_param = cp.Parameter(n, nonneg=True)
    f_param.value = f
    g = cp.Variable(n, nonneg=True)

    conv_fg = cp.conv(f_param, g)          # length 2·n‑1
    constraints = [conv_fg <= M]

    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=max_iters, verbose=False)
    except Exception:               # fall back to a generic solver if ECOS fails
        prob.solve(solver=cp.SCS, max_iters=max_iters, verbose=False)

    if g.value is None:
        # Infeasible or solver failure – return the input unchanged
        return f.copy()
    g_vec = np.maximum(g.value, 0.0)        # numerical safety
    total = g_vec.sum()
    if total == 0:
        return np.full_like(f, 1.0 / n)
    return g_vec / total                    # renormalise to the simplex


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Main entry point used by the evaluator.
    Returns a list of non‑negative floats that (approximately) minimises
    evaluate_sequence().
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # leave a safety margin

    # --------------------------------------------------------------
    # Choose problem size; reuse a previous solution if available.
    if 'height_sequence_1' in globals():
        base = np.array(height_sequence_1, dtype=float)
        n = base.size
        a = base / base.sum()
        print(f"[init] Using supplied sequence, n={n}")
    else:
        n = 5000                     # a good compromise between quality and speed
        a = np.full(n, 1.0 / n)
        print(f"[init] No prior solution – start uniform, n={n}")

    # FFT length sufficient for both convolution and correlation steps
    L = 1 << ((3 * n - 2).bit_length())   # next power of two

    # --------------------------------------------------------------
    # Keep track of the best vector seen so far
    conv = _conv_fft(a, L)
    best_a = a.copy()
    best_obj = _objective_simplex(best_a, n, L)
    print(f"[init] O={best_obj:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – Adam on a smooth log‑sum‑exp surrogate
    phase1_deadline = start_time + budget_s * 0.30
    alpha_start, alpha_end = 5.0, 20000.0
    max_iter1 = 200_000
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.2
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    iter1 = 0
    while time.time() < phase1_deadline and iter1 < max_iter1:
        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:

            print(f"[phase‑1] iter={iter1}, O={best_obj:.6f}")

        if iter2 % 2000 == 0 and iter2 > 0:

            jitter = np.random.randn(n) * 5e-4
            a = _project_to_simplex(a + jitter)

"""
Optimization of a non‑negative step‑function sequence to minimise

    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm consists of four phases:

1. Adam optimisation on a smooth log‑sum‑exp surrogate of the max‑convolution.
2. Mirror‑descent (exponentiated‑gradient) using the exact max‑convolution
   sub‑gradient.
3. A few aggressive linear‑programming refinements (cvxpy) that maximise the
   L1‑norm of a partner vector while keeping its convolution with the current
   iterate below the current worst value. The partner is blended into the
   iterate.
4. Simulated‑annealing hill‑climbing with random mass‑transfer moves.

All phases respect the 1000 s time budget; the best vector encountered is
returned as a plain Python list.
"""

import time
import math
import numpy as np
import cvxpy as cp

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full (non‑circular) convolution a*a using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
def _objective_simplex(a: np.ndarray, n: int, L: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) assuming sum(a)=1."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. a (averaged over all maximisers)."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    g /= len(ks)
    return g


# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, conv: np.ndarray, alpha: float, L: int) -> np.ndarray:
    """
    Gradient of the smooth surrogate
        (2·n/α)·log Σ_k exp(α·conv_k)
    """
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / exp_scaled.sum()                     # soft‑max weights

    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n = a.size
    grad = 2.0 * corr_rev[n - 1 : n - 1 + n]               # central n entries
    return grad


# ----------------------------------------------------------------------
def _exp_update(a: np.ndarray, g: np.ndarray, eta: float) -> np.ndarray:
    """Mirror‑descent / exponentiated‑gradient step on the simplex."""
    a_new = a * np.exp(-eta * g)
    total = a_new.sum()
    if total == 0.0:
        return np.full_like(a, 1.0 / a.size)
    return a_new / total


# ----------------------------------------------------------------------
def _lp_step(f: np.ndarray, M: float, max_iters: int = 2000) -> np.ndarray:
    """
    Solve the linear programme
        maximise   sum(g)
        s.t.       conv(f, g) <= M   (element‑wise)
                   g >= 0
    Returns a feasible g normalized to the probability simplex.
    """
    n = f.shape[0]
    f_param = cp.Parameter(n, nonneg=True)
    f_param.value = f
    g = cp.Variable(n, nonneg=True)

    # convolution in cvxpy (length 2·n‑1)
    conv_fg = cp.conv(f_param, g)
    constraints = [conv_fg <= M]

    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=max_iters, verbose=False)
    except Exception:
        prob.solve(solver=cp.SCS, max_iters=max_iters, verbose=False)

    if g.value is None:
        return f.copy()                     # fallback – keep the current vector

    g_vec = np.maximum(g.value, 0.0)
    total = g_vec.sum()
    if total == 0.0:
        return np.full_like(f, 1.0 / n)
    return g_vec / total
