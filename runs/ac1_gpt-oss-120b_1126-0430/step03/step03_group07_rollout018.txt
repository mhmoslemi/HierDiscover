"""
optimize_step_function.py
-------------------------

A fast, large‑scale optimizer for the step‑function metric

    2·n·max(conv(a,a)) / (sum(a))²

The algorithm works in three phases:

1. **Seed selection** – a handful of analytic probability vectors
   (Beta‑distributions, sin‑power shapes, random Dirichlet draws) and,
   if present, the previously discovered ``height_sequence_1`` are
   evaluated with a tiny FFT‑based evaluator.  The best seed becomes the
   starting point.

2. **Smoothed‑max gradient descent** – the non‑smooth max is replaced
   by a log‑sum‑exp (soft‑max) with a temperature β that is slowly
   increased (continuation).  The gradient of the smooth objective is
   computed analytically via a single FFT convolution, giving an
   O(n log n) per‑iteration cost.  An Adam‑style mirror‑descent update
   (multiplicative in the original variables) keeps the iterates on the
   simplex and adapts per‑coordinate step sizes.

3. **Final polish** – after the main loop the best vector is returned.
   All entries are non‑negative and automatically satisfy the 0 ≤ aᵢ ≤ 1000
   guards of ``evaluate_sequence``.

The whole optimisation runs for the full user‑supplied budget (default
1000 s) and prints periodic progress information.

"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fast_evaluation(seq: np.ndarray) -> float:
    """
    Fast replica of the supplied ``evaluate_sequence`` (without security checks).
    Returns the metric value for a non‑negative vector `seq`.
    """
    n = seq.size
    if n == 0:
        return float("inf")
    # linear convolution via FFT
    m = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    conv = np.maximum(conv, 0.0)
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Seed factories
# ----------------------------------------------------------------------
def _beta_sequence(alpha: float, beta: float, length: int) -> np.ndarray:
    """Beta(α,β) density sampled uniformly on (0,1) and normalised."""
    xs = (np.arange(length, dtype=float) + 0.5) / length
    seq = np.power(xs, alpha - 1.0, out=np.empty_like(xs))
    seq *= np.power(1.0 - xs, beta - 1.0, out=np.empty_like(xs))
    seq[np.isinf(seq)] = 0.0
    seq = np.maximum(seq, 0.0)
    tot = seq.sum()
    if tot == 0.0:
        seq.fill(1.0 / length)
    else:
        seq /= tot
    return seq


def _sin_power_sequence(p: float, length: int) -> np.ndarray:
    """a_i ∝ sin(π·(i+0.5)/length)^p, normalised."""
    xs = (np.arange(length, dtype=float) + 0.5) / length
    vals = np.sin(np.pi * xs) ** p
    vals = np.maximum(vals, 0.0)
    tot = vals.sum()
    if tot == 0.0:
        vals.fill(1.0 / length)
    else:
        vals /= tot
    return vals


def _random_dirichlet(length: int, concentration: float = 0.5) -> np.ndarray:
    """Draw a Dirichlet point with identical concentration parameters."""
    seq = np.random.gamma(concentration, 1.0, size=length)
    tot = seq.sum()
    if tot == 0.0:
        seq.fill(1.0 / length)
    else:
        seq /= tot
    return seq


# ----------------------------------------------------------------------
# Core optimisation routine (smoothed‑max Adam on the simplex)
# ----------------------------------------------------------------------
def _smooth_max_adam(
    init_seq: np.ndarray,
    deadline: float,
    *,
    beta1: float = 0.9,
    beta2: float = 0.999,
    base_lr: float = 0.12,
    beta_start: float = 1.0,
    beta_factor: float = 1.02,
    beta_max: float = 500.0,
) -> np.ndarray:
    """
    Refine ``init_seq`` (non‑negative, sum = 1) with Adam on the
    log‑variables.  The objective is the soft‑max of the convolution,
    i.e.

        F(a) = (1/β)·log Σ_k exp(β·(a*a)[k])

    whose gradient can be written as

        ∂F/∂a_i = 2·Σ_k w_k·a_{k‑i},

    where w_k is the soft‑max weight at shift k.
    """
    n = init_seq.size
    eps = 1e-15

    # log‑variables (mirror‑descent on the simplex)
    log_a = np.log(init_seq + eps)

    m = np.zeros_like(init_seq)      # first Adam moment
    v = np.zeros_like(init_seq)      # second Adam moment
    t = 0

    # FFT lengths (chosen once for efficiency)
    fft_len_self = _next_pow2(2 * n - 1)      # for a*a
    fft_len_grad = _next_pow2(3 * n - 2)      # for a * w_rev
    idx_for_grad = 2 * n - 2 - np.arange(n)   # positions extracting the gradient

    best_seq = None
    best_score = float("inf")
    beta = beta_start

    while time.time() < deadline:
        # ------------------------------------------------------------------
        # (a) build the probability vector from log‑variables and normalise
        # ------------------------------------------------------------------
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        # ------------------------------------------------------------------
        # (b) self‑convolution a*a  (linear, not circular)
        # ------------------------------------------------------------------
        Fa = np.fft.rfft(a, fft_len_self)
        conv_self = np.fft.irfft(Fa * Fa, fft_len_self)[: 2 * n - 1]
        conv_self = np.maximum(conv_self, 0.0)    # numerical clean‑up

        # ------------------------------------------------------------------
        # (c) soft‑max weights over all shifts
        # ------------------------------------------------------------------
        beta = min(beta_max, beta * beta_factor)   # slowly heat up
        c_max = conv_self.max()
        w = np.exp(beta * (conv_self - c_max))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ------------------------------------------------------------------
        # (d) gradient of the smooth objective:
        #     g_i = 2 * Σ_r a_r * w_{i+r}
        # ------------------------------------------------------------------
        w_rev = w[::-1]
        Fa_big = np.fft.rfft(a, fft_len_grad)
        Fw_big = np.fft.rfft(w_rev, fft_len_grad)
        conv_rev = np.fft.irfft(Fa_big * Fw_big, fft_len_grad)[: 3 * n - 2]
        g = 2.0 * conv_rev[idx_for_grad]

        # ------------------------------------------------------------------
        # (e) Adam update in log‑space (mirror descent)
        # ------------------------------------------------------------------
        t += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = base_lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        log_a -= step

        # keep log‑variables in a reasonable numeric range
        np.clip(log_a, -30.0, 30.0, out=log_a)

        # ------------------------------------------------------------------
        # (f) evaluate – keep the best seen vector
        # ------------------------------------------------------------------
        a = np.exp(log_a)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= a_sum

        score = _fast_evaluation(a)
        if score < best_score - 1e-12:
            best_score = score
            best_seq = a.copy()
            if t % 500 == 0:
                print(f"[adam] iter {t:6d}  best {best_score:.6f}")

        # occasional tiny random jitter to escape flat regions
        if np.random.rand() < 0.001:
            j = np.random.randint(n)
            a[j] = max(0.0, a[j] + np.random.randn() * 1e-3)
            a /= a.sum()
            log_a = np.log(a + eps)

    # ----------------------------------------------------------------------
    # Return the best vector discovered (fallback to the last one if needed)
    # ----------------------------------------------------------------------
    if best_seq is None:
        a = np.exp(log_a)
        a /= a.sum()
        best_seq = a
    return best_seq


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Produce a step‑function with a low metric value.

    The routine builds several analytic seeds, picks the best,
    and then runs a long Adam‑type optimisation on the full length
    (≈ 5000) using a smoothed max objective.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # 1. Desired length – the challenge uses 5000
    # ------------------------------------------------------------------
    n = 5000

    # ------------------------------------------------------------------
    # 2. Assemble a collection of promising seeds
    # ------------------------------------------------------------------
    seeds = []

    # (a) reuse the previously discovered construction if available
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=float)
            if cand.ndim == 1:
                if cand.size != n:
                    # interpolate/resample to the required length
                    x_old = np.linspace(0, 1, cand.size)
                    x_new = np.linspace(0, 1, n)
                    cand = np.interp(x_new, x_old, cand)
                cand = np.maximum(cand, 0.0)
                tot = cand.sum()
                if tot == 0.0:
                    cand = np.full(n, 1.0 / n)
                else:
                    cand /= tot
                seeds.append(("prev_best", cand))
        except Exception:
            pass

    # (b) analytic families
    seeds.append(("arcsine", _beta_sequence(0.5, 0.5, n)))
    for a in [0.35, 0.45, 0.55, 0.65, 0.80, 1.0]:
        seeds.append((f"beta_sym_{a:.2f}", _beta_sequence(a, a, n)))
    for a, b in [(0.3, 0.7), (0.7, 0.3), (0.4, 0.9), (0.9, 0.4)]:
        seeds.append((f"beta_{a:.2f}_{b:.2f}", _beta_sequence(a, b, n)))
    for p in [-0.6, -0.5, -0.4, -0.3, -0.2, 0.0, 0.2]:
        seeds.append((f"sin_pow_{p:.1f}", _sin_power_sequence(p, n)))
    for _ in range(3):
        seeds.append(("rand_dir", _random_dirichlet(n, concentration=0.4)))

    # ------------------------------------------------------------------
    # 3. Pick the best seed using the fast evaluator
    # ------------------------------------------------------------------
    best_seed_seq = None
    best_seed_score = float("inf")
    for name, seq in seeds:
        seq = np.maximum(seq, 0.0)
        total = seq.sum()
        if total == 0.0:
            seq = np.full(n, 1.0 / n)
        else:
            seq /= total
        score = _fast_evaluation(seq)
        print(f"[seed] {name:15s}  score {score:.6f}")
        if score < best_seed_score:
            best_seed_score = score
            best_seed_seq = seq.copy()

    # ------------------------------------------------------------------
    # 4. Main optimisation – smoothed‑max Adam on the simplex
    # ------------------------------------------------------------------
    if best_seed_seq is None:  # extremely unlikely
        best_seed_seq = np.full(n, 1.0 / n)

    print("[optim] starting long‑run Adam refinement...")
    refined = _smooth_max_adam(best_seed_seq, deadline)

    final_score = _fast_evaluation(refined)
    print(f"[summary] final metric {final_score:.6f} with length {refined.size}")

    return refined.tolist()
