import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max; returns a probability vector (sum = 1)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it representation that yields `a` after soft‑max."""
    return np.log(a + eps)


def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two ≥ 2·n‑1 (efficient size for real FFT)."""
    return 1 << ((2 * n - 1).bit_length())


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using a real FFT of length `fft_len`."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.size - 1]
    return conv


def _conv_max_fft(a: np.ndarray, fft_len: int) -> tuple[float, int]:
    """Return (max_value, argmax) of the self‑convolution a*a."""
    conv = _conv_full_fft(a, fft_len)
    idx = int(conv.argmax())
    return float(conv.max()), idx


def _subgradient_max_conv(a: np.ndarray, idx: int) -> np.ndarray:
    """Sub‑gradient of a*a at a single active index `idx`."""
    n = a.shape[0]
    g = np.zeros_like(a)
    lo = max(0, idx - (n - 1))
    hi = min(idx, n - 1)
    i = np.arange(lo, hi + 1, dtype=np.int64)
    g[i] = 2.0 * a[idx - i]
    return g


def _subgradient_multi_conv(a: np.ndarray, idxs: np.ndarray) -> np.ndarray:
    """Average sub‑gradient over a set of active indices."""
    n = a.shape[0]
    g = np.zeros_like(a)
    for idx in idxs:
        lo = max(0, idx - (n - 1))
        hi = min(idx, n - 1)
        i = np.arange(lo, hi + 1, dtype=np.int64)
        g[i] += 2.0 * a[idx - i]
    if len(idxs) > 0:
        g /= float(len(idxs))
    return g


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    total = new_a.sum()
    if total == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= total
    return new_a


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}, normalised."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _exponential_distribution(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i), normalised."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _sin_power_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ sin(π·(i+0.5)/n)^p, normalised."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n) ** p
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _arcsine_distribution(n: int) -> np.ndarray:
    """
    a_i ∝ 1 / sqrt((i+0.5)*(n-i-0.5)), i.e. the discrete arcsine law.
    """
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights (a step function) that aims
    to minimise the provided evaluation function.  The routine runs for
    at most `budget_s` seconds and uses a two‑stage search:
    (1) broad parametric scan, (2) local Adam optimisation with an
    active‑set sub‑gradient.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0     # safety margin

    # ------------------------------------------------------------------
    # 1. Global parametric scan – keep the best K candidates.
    # ------------------------------------------------------------------
    K = 5                                 # size of the elite set
    elite = []                            # list of tuples (score, seq)

    # support sizes we will explore
    n_list = [5000, 8000, 12000, 20000]

    print("[Phase 1] Global parametric scan")
    for n in n_list:
        if time.time() >= deadline:
            break
        fft_len = _fft_len_for_n(n)

        # ----- uniform -------------------------------------------------
        a_uni = np.full(n, 1.0 / n, dtype=np.float64)
        maxb, _ = _conv_max_fft(a_uni, fft_len)
        score = 2.0 * n * maxb
        elite.append((score, a_uni.copy()))
        print(f" n={n:5d} | uniform    | cheap_score={score:.6f}")

        # ----- power‑law grid -----------------------------------------
        for p in np.linspace(0.5, 3.0, 16):
            if time.time() >= deadline:
                break
            a = _power_law_distribution(n, p)
            maxb, _ = _conv_max_fft(a, fft_len)
            score = 2.0 * n * maxb
            elite.append((score, a.copy()))
        # ----- exponential grid ----------------------------------------
        for lam in np.linspace(0.0, 0.0015, 16):
            if time.time() >= deadline:
                break
            a = _exponential_distribution(n, lam)
            maxb, _ = _conv_max_fft(a, fft_len)
            score = 2.0 * n * maxb
            elite.append((score, a.copy()))
        # ----- sine‑power ---------------------------------------------
        for p in (0.5, 1.0, 2.0, 3.0):
            if time.time() >= deadline:
                break
            a = _sin_power_distribution(n, p)
            maxb, _ = _conv_max_fft(a, fft_len)
            score = 2.0 * n * maxb
            elite.append((score, a.copy()))
        # ----- arcsine -------------------------------------------------
        a = _arcsine_distribution(n)
        maxb, _ = _conv_max_fft(a, fft_len)
        score = 2.0 * n * maxb
        elite.append((score, a.copy()))

    # Keep the K best candidates (lowest cheap score)
    elite.sort(key=lambda x: x[0])
    elite = elite[:K]
    print(f"[Phase 1] Elite size = {len(elite)} (targets for refinement)")

    # ------------------------------------------------------------------
    # 2. Local Adam optimisation from each elite start.
    # ------------------------------------------------------------------
    best_score = float("inf")
    best_seq = None

    # optimisation hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    base_alpha = 0.25      # a bit larger than the previous 0.2

    # tolerance for the active set (percent of the maximum)
    active_tol = 0.005     # 0.5 % below the peak

    for idx, (init_score, a0) in enumerate(elite):
        if time.time() >= deadline:
            break
        n = a0.shape[0]
        fft_len = _fft_len_for_n(n)

        print(f"[Phase 2] Refinement {idx + 1}/{len(elite)} – n={n}")

        # initialise logits and Adam moments
        z = _logits_from_distribution(a0)
        m = np.zeros_like(z)
        v = np.zeros_like(z)

        it = 0
        last_improve = 0
        local_best_seq = a0.copy()
        local_best_score = init_score

        # main optimisation loop
        while time.time() < deadline and it < 500_000:
            it += 1

            # probability vector from logits
            a = _softmax(z)

            # full convolution
            conv = _conv_full_fft(a, fft_len)
            maxb = float(conv.max())
            cheap_score = 2.0 * n * maxb

            # keep local best
            if cheap_score < local_best_score:
                local_best_score = cheap_score
                local_best_seq = a.copy()
                last_improve = it

            # active set (indices within tolerance of the maximum)
            threshold = maxb * (1.0 - active_tol)
            active_idxs = np.where(conv >= threshold)[0]

            # sub‑gradient (average over the active set)
            g_a = _subgradient_multi_conv(a, active_idxs)

            # gradient in log‑it space (mirror descent)
            mu = np.dot(g_a, a)
            grad_z = a * (g_a - mu)

            # Adam moment updates (bias‑corrected)
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            # direction (negative Adam step)
            step_dir = - m_hat / (np.sqrt(v_hat) + eps_adam)

            # basic step‑size schedule: inverse‑sqrt decay
            alpha_t = base_alpha / (math.sqrt(it) + 1.0)

            # -------------------------------------------------------------
            # back‑tracking line search (up to 4 halvings)
            # -------------------------------------------------------------
            accepted = False
            trial_alpha = alpha_t
            for _ in range(4):
                z_new = z + trial_alpha * step_dir
                z_new = np.clip(z_new, -50.0, 50.0)
                a_new = _softmax(z_new)
                conv_new = _conv_full_fft(a_new, fft_len)
                new_score = 2.0 * n * float(conv_new.max())
                if new_score <= cheap_score:
                    z = z_new
                    accepted = True
                    break
                trial_alpha *= 0.5

            if not accepted:
                # modest step even if no improvement
                z = z + alpha_t * step_dir
                z = np.clip(z, -50.0, 50.0)

            # occasional random perturbations
            if it % 4000 == 0 and np.random.rand() < 0.15:
                perturb = 0.02 * np.random.randn(n)
                z += perturb
                m *= 0.5
                v *= 0.5

            # early stop if no improvement for a long time
            if it - last_improve > 120_000:
                print(f"  No improvement for {it - last_improve} iters – stopping this run.")
                break

        # after the run, compare the local best with the global best
        if local_best_score < best_score:
            best_score = local_best_score
            best_seq = local_best_seq.copy()
            print(f"[Phase 2] New global cheap best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Final exact evaluation
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – very simple uniform distribution
        best_seq = np.full(5000, 1.0 / 5000, dtype=np.float64)

    final_seq = best_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[Result] Final exact score = {final_score:.6f}")
    except Exception as exc:   # defensive – should never happen
        final_score = float("inf")
        print(f"[Result] Error during final evaluation: {exc}")

    return final_seq
