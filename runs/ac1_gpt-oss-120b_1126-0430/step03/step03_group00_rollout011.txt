    simple analytic families (Beta‑densities, power‑laws, the known
    “arcsine” shape, the previous best sequence, uniform).  The best
    candidate from this cheap scan seeds the long optimisation.

2️⃣  **Smooth surrogate optimisation** – Adam on the log‑sum‑exp
    approximation of the max‑convolution (`f_α(a) = (2·n/α)·log Σ e^{α·conv}`).
    The smoothing parameter α is linearly annealed (5 → 15000), the
    iterate is projected onto the probability simplex after every step.

3️⃣  **Refinement on the exact objective** – a mixture of
    *deterministic projected gradient descent* (sub‑gradient of the
    exact max‑convolution with a back‑tracking line‑search) and
    *biased stochastic mass‑transfer* moves that preferentially move
    probability mass from indices with large sub‑gradient values to
    indices with small ones.  The best vector seen so far is always
    stored.

All phases respect the global time budget (default 1000 s).  The
function finally returns the best sequence as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex {x ≥ 0, sum(x)=1}
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex."""
    n = v.shape[0]
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        # all non‑positive – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full self‑convolution (length 2·n‑1)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return a*a (full convolution) using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Exact objective for a simplex‑normalised vector
def _objective_simplex(a: np.ndarray, n: int, L: int) -> float:
    """Return 2·n·max(conv(a,a)) (a is assumed to sum to 1)."""
    conv = _conv_fft(a, L)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Sub‑gradient of the maximal convolution entry (as used in the reference code)
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Return the sub‑gradient of max(conv) w.r.t. a."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)          # possibly several maximisers
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
# Gradient of the log‑sum‑exp (soft‑max) surrogate of the max
def _smooth_gradient(a: np.ndarray, conv: np.ndarray, alpha: float, L: int) -> np.ndarray:
    """
    Compute ∇_a ( (2·n/α)·log∑_k e^{α·conv_k} ).
    The formula reduces to 2·corr(w, a) where w_k ∝ e^{α·conv_k}.
    """
    # soft‑max weights (stable)
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / exp_scaled.sum()

    # correlation w ⋆ a   (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)   # length ≥ 3·n‑2
    n_len = a.size
    grad = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]
    return grad


# ----------------------------------------------------------------------
# Parametrised families (Beta, power‑law, arcsine) --------------------
def _beta_family(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density discretised on n points and normalised."""
    x = (np.arange(n) + 0.5) / n                       # mid‑point rule
    # avoid NaNs for α<1 at the boundaries by clipping
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    pdf = (x ** (alpha - 1.0)) * ((1.0 - x) ** (alpha - 1.0))
    a = pdf / pdf.sum()
    return a


def _powerlaw_family(n: int, p: float) -> np.ndarray:
    """
    Symmetric power‑law: a_i ∝ (min(i,n‑1‑i)+0.5)^p .
    p>0 concentrates mass near the edges, p<0 near the centre.
    """
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 0.5
    a = d ** p
    a = a / a.sum()
    return a


def _arcsine_family(n: int) -> np.ndarray:
    """Discrete version of the arcsine density on [0,1]."""
    # arcsine is Beta(0.5,0.5)
    return _beta_family(n, 0.5)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    O(a) = 2·n·max(conv(a,a)) / (sum(a)^2).  The function returns the best
    sequence found before the allocated time expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.3   # tiny safety margin

    # ------------------------------------------------------------------
    # Choose problem size -------------------------------------------------
    if 'height_sequence_1' in globals():
        base = np.array(height_sequence_1, dtype=float)
        n = base.size
        a0 = base / base.sum()                     # simplex normalisation
        print(f"[init] Using supplied sequence of length {n}")
    else:
        # a moderate size that keeps FFT cheap but gives enough freedom
        n = 4600
        a0 = np.full(n, 1.0 / n)
        print(f"[init] No supplied sequence – using uniform of length {n}")

    # FFT length – next power of two sufficient for all our transforms
    L = 1 << ((3 * n - 2).bit_length())

    # ------------------------------------------------------------------
    # Stage 0 – cheap parametric scan (≈ 10 % of time budget) ----------
    best_seq = a0.copy()
    best_score = _objective_simplex(best_seq, n, L)
    print(f"[scan‑0] start O={best_score:.6f}")

    # grid of parameters
    beta_alphas = [0.2, 0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0, 5.0]
    power_ps = [-2.0, -1.5, -1.0, -0.5, 0.0, 0.5, 1.0, 1.5, 2.0]

    scan_deadline = start_time + budget_s * 0.1
    # Beta family
    for α in beta_alphas:
        if time.time() >= scan_deadline:
            break
        a = _beta_family(n, α)
        cur = _objective_simplex(a, n, L)
        if cur < best_score:
            best_score = cur
            best_seq = a.copy()
            print(f"[scan‑beta] α={α:.2f} new best O={best_score:.6f}")

    # Power‑law family
    for p in power_ps:
        if time.time() >= scan_deadline:
            break
        a = _powerlaw_family(n, p)
        cur = _objective_simplex(a, n, L)
        if cur < best_score:
            best_score = cur
            best_seq = a.copy()
            print(f"[scan‑pow]  p={p:.2f} new best O={best_score:.6f}")

    # Explicit arcsine (α=0.5) – already covered but keep for safety
    a = _arcsine_family(n)
    cur = _objective_simplex(a, n, L)
    if cur < best_score:
        best_score = cur
        best_seq = a.copy()
        print(f"[scan‑arc] new best O={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – Adam on the smooth log‑sum‑exp surrogate ---------------
    phase1_deadline = start_time + budget_s * 0.45
    a = best_seq.copy()
    m = np.zeros_like(a)          # 1st moment
    v = np.zeros_like(a)          # 2nd moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    lr0 = 0.6                     # a slightly larger base LR
    alpha_start = 5.0
    alpha_end = 15000.0           # steer the surrogate very close to max
    it = 0
    max_iter_phase1 = 500_000

    print(f"[phase‑1] start O={best_score:.6f}")

    while time.time() < phase1_deadline and it < max_iter_phase1:
        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_score - 1e-12:
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[phase‑1] iter {it} new best O={best_score:.6f}")

        # linearly anneal α
        frac = it / max_iter_phase1
        alpha = alpha_start + (alpha_end - alpha_start) * frac

        grad = _smooth_gradient(a, conv, alpha, L)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        lr = lr0 / math.sqrt(it + 1.0)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional tiny jitter
        if np.random.rand() < 0.001:
            a_new += np.random.randn(n) * 1e-4

        # projection onto simplex
        a_new = _project_to_simplex(a_new)

        # optional pruning of minute entries every 8000 iter
        if it % 8000 == 0 and it > 0:
            thresh = np.percentile(a_new, 0.5)   # drop the lowest 0.5 %
            a_new[a_new < thresh] = 0.0
            a_new = _project_to_simplex(a_new)

        a = a_new
        it += 1

    # ------------------------------------------------------------------
    # Stage 2 – Stochastic local refinement with gradient‑biased moves ---
    phase2_deadline = start_time + budget_s * 0.85
    max_restarts = 0
    subgrad_interval = 6
    temperature0 = 0.04
    max_iter_local = 180_000
    no_improve_limit = 4_000

    print(f"[phase‑2] entering refinement, best O={best_score:.6f}")

    while time.time() < phase2_deadline:
        # ----- initialise a restart (jittered best or fresh) ----------
        if np.random.rand() < 0.6:
            a = best_seq.copy()
            a += np.random.randn(n) * 0.0015
            a = np.clip(a, 0, None)
            a = _project_to_simplex(a)
        else:
            a = np.random.rand(n)
            a = _project_to_simplex(a)

        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        it_loc = 0
        no_improve = 0
        T = temperature0

        # pre‑compute gradient once to bias the random moves
        while time.time() < phase2_deadline and it_loc < max_iter_local:
            # ----- biased mass‑transfer move -------------------------
            # compute sub‑gradient for the current maximal entry
            g = _subgradient_max_conv(a, conv)

            # source probabilities proportional to gradient magnitude
            src_prob = g / (g.sum() + 1e-32)
            i = np.random.choice(n, p=src_prob)

            # target probabilities inversely proportional to gradient
            inv = 1.0 / (g + 1e-12)
            tgt_prob = inv / inv.sum()
            j = np.random.choice(n, p=tgt_prob)

            if i != j and a[i] > 1e-14:
                delta = a[i] * np.random.rand() * 0.10   # up to 10 %
                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta
                a_new = _project_to_simplex(a_new)

                conv_new = _conv_fft(a_new, L)
                new_obj = 2.0 * n * conv_new.max()

                # Metropolis acceptance
                if new_obj < cur_obj or np.random.rand() < math.exp((cur_obj - new_obj) / max(T, 1e-12)):
                    a = a_new
                    conv = conv_new
                    cur_obj = new_obj
                    no_improve = 0
                    if cur_obj < best_score - 1e-12:
                        best_score = cur_obj
                        best_seq = a.copy()
                        print(f"[{time.time():.1f}] New global best O={best_score:.6f} (restart={max_restarts}, iter={it_loc})")
                else:
                    no_improve += 1
            else:
                no_improve += 1

            # ----- sub‑gradient step on the exact max (every few iters) -----
            if it_loc % subgrad_interval == 0:
                # direction opposite to the sub‑gradient, mean‑adjusted to keep sum = 0
                direction = -(g - np.mean(g))
                # back‑tracking line‑search
                step = 0.07
                for _ in range(12):
                    a_tmp = _project_to_simplex(a + step * direction)
                    conv_tmp = _conv_fft(a_tmp, L)
                    obj_tmp = 2.0 * n * conv_tmp.max()
                    if obj_tmp <= cur_obj:
                        a = a_tmp
                        conv = conv_tmp
                        cur_obj = obj_tmp
                        if cur_obj < best_score - 1e-12:
                            best_score = cur_obj
                            best_seq = a.copy()
                            print(f"[{time.time():.1f}] Sub‑grad improvement O={best_score:.6f}")
                        break
                    step *= 0.5

            # temperature decay
            T *= 0.9997
            it_loc += 1

            # restart if stuck for a while
            if no_improve >= no_improve_limit:
                break

        max_restarts += 1

    # ------------------------------------------------------------------
    # Stage 3 – Deterministic projected gradient descent with line‑search
    phase3_deadline = deadline
    a = best_seq.copy()
    conv = _conv_fft(a, L)
    cur_obj = 2.0 * n * conv.max()
    it_det = 0

    print(f"[phase‑3] deterministic refinement, start O={best_score:.6f}")

    while time.time() < phase3_deadline:
        g = _subgradient_max_conv(a, conv)
        # direction that conserves the simplex sum (mean‑zero)
        direction = -(g - np.mean(g))

        # try a decreasing series of step sizes
        step = 0.05
        improved = False
        for _ in range(15):
            a_new = _project_to_simplex(a + step * direction)
            conv_new = _conv_fft(a_new, L)
            new_obj = 2.0 * n * conv_new.max()
            if new_obj < cur_obj - 1e-12:
                a = a_new
                conv = conv_new
                cur_obj = new_obj
                improved = True
                if cur_obj < best_score - 1e-12:
                    best_score = cur_obj
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] Deterministic step → O={best_score:.6f}")
                break
            step *= 0.5
        if not improved:
            # if no progress, inject a tiny random jitter to escape flat region
            a += np.random.randn(n) * 5e-5
            a = _project_to_simplex(a)
            conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max()
        it_det += 1
        if it_det % 10_000 == 0:
            # occasional sanity check and early exit if stagnating
            if time.time() > phase3_deadline - 5:
                break

    # ------------------------------------------------------------------
    # Final sanity check with the provided evaluator
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:
        print("[final] evaluate_sequence failed:", e)

    # Return the best sequence found (plain list)
    return best_seq.tolist()
