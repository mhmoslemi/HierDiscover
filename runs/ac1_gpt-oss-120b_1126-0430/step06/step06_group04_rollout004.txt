"""
Advanced step‑function optimiser (v3).

This version builds on the previous optimiser but adds several
enhancements aimed at pushing the objective below 1.503:

*   Multiple independent restarts (≈6) with diverse analytical
    initial shapes, including the previously discovered best
    construction (``height_sequence_1`` if it exists).

*   Adam optimisation on the log‑it simplex with a smoother
    temperature schedule and occasional jitter/restart.

*   A dedicated projected‑gradient (PG) phase that
    – uses a soft‑max weight on the autocorrelation (β grows
      from 5 to ~1e6) so that several near‑maximal entries are
      considered together,
    – performs a simple back‑tracking line‑search on the step
      size (up to 5 tries) to guarantee a decrease of the exact
      objective,
    – injects occasional global noise to escape shallow basins.

*   After the PG phase a short simulated‑annealing / PG hybrid
    (the same routine as the earlier “Stage 3”) is run with the
    remaining budget.

All heavy lifting (convolutions, correlations, projections) is
implemented with real FFTs for speed.  The optimiser respects the
global time budget (default 1000 s) and always returns a non‑negative
list that ``evaluate_sequence`` can consume.
"""

import time
import math

# Low‑level utilities

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·len(a)‑1) using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return conv_full[: 2 * n - 1]


def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation of a with the weight vector w (length 2·n−1):
        g[i] = Σ_t w[i+t]·a[t]   for i=0,…,n−1
    Implemented via FFT.
    """
    n = a.shape[0]
    m = w.shape[0]                     # expected 2·n−1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Stable soft‑max."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of vector v onto the probability simplex
    {x ≥ 0, Σ x = 1}.
    Implements the O(n log n) algorithm of Duchi et al.
    """
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback to uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) assuming Σ a = 1.
    Returns a Python float.
    """
    n = a.size
    conv = _conv_self_fft(a)
    return float(2.0 * n * conv.max())

def _shape_candidates(n: int, rng: np.random.Generator):
    """
    Produce a modest set of analytically‑motivated initial sequences.
    Each candidate is a NumPy array normalised to sum‑to‑1.
    """
    cand = []

    # Uniform
    cand.append(np.full(n, 1.0 / n))

    # Exponential decays (forward and reversed)
    r_vals = np.linspace(0.90, 0.9997, 15)
    idx = np.arange(n, dtype=float)
    for r in r_vals:
        a = r ** idx
        a = a / a.sum()
        cand.append(a)
        cand.append(a[::-1])

    # Gaussian windows centred in the middle
    centre = (n - 1) / 2.0
    sigma_vals = np.linspace(0.10 * n, 0.40 * n, 10)
    for sigma in sigma_vals:
        a = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
        a = a / a.sum()
        cand.append(a)

    # Sparse Dirichlet‑like (few non‑zero entries)
    k = max(2, int(0.03 * n))
    for _ in range(4):
        a = np.zeros(n, dtype=float)
        pos = rng.choice(n, size=k, replace=False)
        vals = rng.random(k)
        a[pos] = vals
        a = a / a.sum()
        cand.append(a)

    # Pure random Dirichlet (α=1)
    a = rng.random(n)
    a = a / a.sum()
    cand.append(a)

    return cand


def _global_candidates(n: int):

    If a global variable ``height_sequence_1`` exists, turn it into a

    """
    Adam optimisation on the log‑it parametrisation of the simplex.
    Returns the best sequence (as ndarray, sum=1) and its exact objective.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    n = a0.size
    a = a0.copy()
    # logits
    x = np.log(a + 1e-16)
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # Temperature (smooth‑max) schedule
    beta_start, beta_end = 0.5, 1e6
    # Learning‑rate schedule
    lr_start, lr_end = 0.25, 0.0003

    best_a = a.copy()
    best_obj = _objective(best_a)

    it = 0
    while time.time() - start < time_limit:
        it += 1

        # stay on simplex
        a = _softmax(x)

        conv = _conv_self_fft(a)
        M = conv.max()
        obj = float(2.0 * n * M)

        if obj < best_obj - 1e-14:
            best_obj, best_a = obj, a.copy()
            print(f"[Adam n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        # schedule (elapsed fraction)
        t = min((time.time() - start) / time_limit, 1.0)
        beta = beta_start * np.exp(np.log(beta_end / beta_start) * t)
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t)

        # smooth‑max weights
        if beta > 5e5:                     # essentially hard max
            w = np.zeros_like(conv)
            w[int(np.argmax(conv))] = 1.0
        else:
            shifted = conv - M               # ≤ 0
            w = np.exp(beta * shifted)
            s = w.sum()
            if s == 0.0:
                w = np.ones_like(conv) / conv.size
            else:
                w /= s

        # gradient wrt a (smooth max)
        g_a = 2.0 * _correlation_fft(w, a)

        # natural gradient in log‑it space
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to shake out of plateaus
        if it % 3000 == 0:
            jitter = rng.normal(scale=0.015, size=n)
            x += jitter

        # periodic restart from current best
        if it % 60000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.006, size=n)
            m.fill(0.0)
            v.fill(0.0)

    return best_a, best_obj


# ----------------------------------------------------------------------
# Projected‑gradient refinement (exact max with line‑search)
# ----------------------------------------------------------------------


def _pg_refine(a0: np.ndarray,
               time_limit: float,
               seed: int = 0) -> tuple[np.ndarray, float]:
    """
    Refine a sequence using projected gradient descent on the exact
    max‑objective, employing a soft‑max weight that becomes sharper
    over time and a back‑tracking line‑search.
    Returns the best sequence found and its exact objective.
    """
    start = time.time()
    rng = np.random.default_rng(seed)
    a = a0.copy()
    n = a.size

    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())
    best_a = a.copy()

    # soft‑max temperature – start modest, grow toward hard max
    beta = 5.0
    beta_factor = 1.02
    max_beta = 1e6

    it = 0
    while time.time() - start < time_limit:
        it += 1

        conv = _conv_self_fft(a)
        M = conv.max()
        k_max = int(np.argmax(conv))

        # soft‑max over all entries (sharpening)
        shifted = conv - M                     # ≤0, max at 0
        w = np.exp(beta * shifted)
        s = w.sum()
        if s == 0.0:
            w = np.zeros_like(conv)
            w[k_max] = 1.0
        else:
            w /= s

        # exact gradient of smoothed max
        g = 2.0 * _correlation_fft(w, a)

        # back‑tracking line‑search (max 5 probes)
        step = 1.0
        improved = False
        for _ in range(5):
            a_candidate = _project_simplex(a - step * g)
            conv_cand = _conv_self_fft(a_candidate)
            obj_cand = float(2.0 * n * conv_cand.max())
            if obj_cand < best_obj - 1e-14:
                # accept
                a, conv, best_obj, best_a = a_candidate, conv_cand, obj_cand, a_candidate.copy()
                improved = True
                break
            step *= 0.5

        if not improved:
            # if no improvement, still project a tiny step to keep moving
            a = _project_simplex(a - 0.1 * step * g)

        # update best if needed
        if best_obj < _objective(best_a):
            best_a, best_obj = a.copy(), best_obj

        # temperature schedule
        beta = min(beta * beta_factor, max_beta)

        # occasional global jitter to escape flat regions
        if it % 800 == 0:
            noise = rng.normal(scale=0.008, size=n)
            a = _project_simplex(a + noise)

        # simple termination criterion
        if step < 1e-8:
            break

    return best_a, best_obj


# ----------------------------------------------------------------------
# Targeted simulated‑annealing + projected‑gradient (final polish)
# ----------------------------------------------------------------------


def _run_sa(init_seq: list[float],
            time_limit: float,
            seed: int = 0) -> tuple[list[float], float]:
    """
    Fast SA / PG hybrid used as a final polishing step.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    a = np.asarray(init_seq, dtype=float)
    if a.sum() < 1e-12:
        a = np.ones_like(a) / a.size
    else:
        a = a / a.sum()
    n = a.shape[0]

    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())
    best_seq = a.copy()

    # SA schedule
    T = best_obj * 0.2 + 1e-8
    T_min = best_obj * 1e-6 + 1e-12
    decay = 0.997

    it = 0
    while time.time() - start < time_limit * 0.65:
        it += 1
        k_max = int(np.argmax(conv))

        # contributions a[i] * a[k_max-i]
        i_vals = np.arange(n)
        j_vals = k_max - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i_vals[mask]] * a[j_vals[mask]]
        total_contrib = contrib.sum()

        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = rng.choice(n, p=probs)
        else:
            i = rng.integers(n)

        # pick a distinct destination uniformly
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # move a modest fraction of mass
        delta = a[i] * rng.random() * 0.07
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.clip(a_new, 0.0, None)
        a_new /= a_new.sum()

        conv_new = _conv_self_fft(a_new)
        obj_new = float(2.0 * n * conv_new.max())

        if obj_new < best_obj - 1e-14:
            a, conv, best_obj, best_seq = a_new, conv_new, obj_new, a_new.copy()
            print(f"[SA it={it}] new best O={best_obj:.6f}")
        else:
            # Metropolis acceptance
            if rng.random() < math.exp(-(obj_new - best_obj) / T):
                a, conv = a_new, conv_new

        if it % 2500 == 0:
            T = max(T * decay, T_min)

        if it % 8000 == 0:
            noise = rng.normal(scale=0.008, size=n)
            a = np.clip(a + noise, 0.0, None)
            a /= a.sum()
            conv = _conv_self_fft(a)

    # final projected‑gradient descent (exact max)
    remaining = time_limit - (time.time() - start)
    if remaining > 0.3:
        a = best_seq.copy()
        n = a.size
        it_g = 0
        eta0 = 0.25
        while time.time() - start < time_limit:
            it_g += 1
            conv = _conv_self_fft(a)
            k_max = int(np.argmax(conv))
            w = np.zeros_like(conv)
            w[k_max] = 1.0
            g = 2.0 * _correlation_fft(w, a)
            eta = eta0 * (0.995 ** it_g)
            a_new = _project_simplex(a - eta * g)
            obj_new = float(2.0 * n * _conv_self_fft(a_new).max())
            if obj_new < best_obj - 1e-14:
                a, best_obj, best_seq = a_new, obj_new, a_new.copy()
                print(f"[PG final it={it_g}] improved O={best_obj:.6f}")
            else:
                eta0 *= 0.99
                if eta0 < 1e-6:
                    break

    return best_seq.tolist(), best_obj


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **_ignore) -> list[float]:
    """
    Entry‑point used by the evaluation harness.
    Performs several independent optimisation runs (Adam + PG) on a
    fixed support size (n=2000), keeps the best result and finishes
    with a short SA/PG polishing stage.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5      # safety margin

    # ------------------------------------------------------------------
    # Problem size – 2000 gives a good trade‑off between flexibility
    # and per‑iteration cost.
    n = 2000

    # Assemble a pool of diverse initial sequences
    init_pool = _shape_candidates(n, rng)
    init_pool.extend(_global_candidates(n))

    best_seq = None
    best_obj = float("inf")

    # Determine how many independent restarts we can afford.
    # Reserve a few seconds for the final SA/PG polish.
    remaining = deadline - time.time()
    if remaining <= 10.0:
        max_restarts = 1
    else:
        # Each restart will try to consume roughly equal time.
        max_restarts = min(6, int(remaining // 20))
    if max_restarts < 1:
        max_restarts = 1

    per_restart = (deadline - time.time()) * 0.9 / max_restarts

    print(f"[info] n={n}, restarts={max_restarts}, per_restart≈{per_restart:.1f}s")

    for r in range(max_restarts):
        if time.time() >= deadline - 5.0:
            break

        # pick a random seed‑initialiser
        a0 = rng.choice(init_pool)
        # jitter a little to get variability
        a0 = a0 + rng.normal(scale=0.004, size=n)
        a0 = np.clip(a0, 0.0, None)
        s = a0.sum()
        if s <= 0:
            a0 = np.full(n, 1.0 / n)
        else:
            a0 = a0 / s

        # split the allocated time roughly 60 % Adam / 40 % PG
        t_now = time.time()
        time_left = deadline - t_now
        if time_left <= 0:
            break
        adam_time = min(per_restart * 0.6, time_left * 0.6)
        pg_time   = min(per_restart * 0.4, time_left * 0.4)

        # ---- Adam phase ----
        seq_adam, obj_adam = _adam_optimize(a0, adam_time, seed + r * 12345)

        # ---- Projected‑gradient refinement ----
        seq_pg, obj_pg = _pg_refine(seq_adam, pg_time, seed + r * 67890)

        if obj_pg < best_obj:
            best_obj, best_seq = obj_pg, np.asarray(seq_pg, dtype=float)
            print(f"[global restart {r}] new overall best O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Final polishing with SA + PG if time permits
    remaining = deadline - time.time()
    if best_seq is not None and remaining > 6.0:
        seq_ref, val_ref = _run_sa(best_seq.tolist(),
                                   time_limit=remaining,
                                   seed=seed + 999999)
        if val_ref < best_obj:
            best_obj, best_seq = val_ref, np.asarray(seq_ref, dtype=float)
            print(f"[final polish] after SA+PG new best O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Safety fallback
    if best_seq is None or best_seq.size == 0:
        best_seq = np.full(200, 1.0 / 200.0)

    # ------------------------------------------------------------------
    # Final sanity check (diagnostic only)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    # Ensure the sequence respects evaluator caps and sum bound
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    if best_seq.sum() < 0.01:
        best_seq = np.ones_like(best_seq) / best_seq.size

    return best_seq.tolist()
