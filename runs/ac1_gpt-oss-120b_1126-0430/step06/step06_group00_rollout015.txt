def _conv_max_and_grad(a, eps_factor):
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[:2*n-1]
    max_conv = float(conv.max())
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0 or np.isnan(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]
    grad_a = 4.0 * n * corr
    return max_conv, grad_a
```

   autocorrelation, obtained by a soft‑max with a decaying temperature.
   This yields an analytical gradient (via FFTs) that is cheap to compute.
   The Adam step size is scheduled as `α / (sqrt(t)+1)` and the
   temperature is multiplied by a factor < 1 each iteration, ending at a
   very small value so that the gradient essentially targets the true
   maximal convolution entry.

   The loop runs until the time budget (≈ 1000 s) is exhausted, keeping
   the best vector ever seen.

4. A short random‑perturbation phase (pairwise mass transfers) is run
   at the very end to squeeze out any remaining improvement.

5. The function returns the best sequence as a plain Python list.
   A final call to `evaluate_sequence` is made only for logging – the
   platform will independently evaluate the returned list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Top‑level helpers (no closures, no lambdas)
# ----------------------------------------------------------------------


def arcsine_cdf(x: np.ndarray) -> np.ndarray:
    """Arcsine CDF  F(x) = (2/π)·arcsin(√x)  on [0,1]."""
    return (2.0 / np.pi) * np.arcsin(np.sqrt(np.clip(x, 0.0, 1.0)))


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max; always returns a probability vector."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _fft_len_for_corr(n: int) -> int:
    """
    Smallest power‑of‑two ≥ 3·n‑2.
    This length is sufficient for both the linear self‑convolution
    (length 2·n‑1) and the correlation (length 3·n‑2) needed for the
    gradient computation.
    """
    return 1 << ((3 * n - 2).bit_length())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Smoothed maximum of the self‑convolution `a*a` together with its
    gradient w.r.t. `a`.

    Parameters
    ----------
    a : np.ndarray
        Probability vector (non‑negative, sum = 1).
    eps_factor : float
        Multiplicative factor that controls the smoothing constant:
        eps = max(eps_factor * max(conv), 1e‑12).

    Returns
    -------
    max_conv : float
        Maximum entry of the linear self‑convolution.
    grad_a : np.ndarray
        Gradient of 2·n·max_conv w.r.t. `a` (already multiplied by 4·n).
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    # FFT of the current vector
    a_fft = np.fft.rfft(a, L)

    # Linear self‑convolution (length 2·n‑1)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # Soft‑max weights over all convolution entries
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv          # shift for numeric stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or np.isnan(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # Cross‑correlation a ⋆ w (length 3·n‑2, we need the first n entries)
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # Gradient (the factor 4·n is 2·n times the derivative of the soft‑max)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution `a*a` (FFT based)."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())

# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that minimises the evaluation metric.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a safety margin for final logging / cleanup
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # 0. Choose the length of the sequence
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
        print(f"[Info] Reusing external length: n = {n}")
    except Exception:
        n = 20000
        print(f"[Info] No external length – using default n = {n}")

    # ------------------------------------------------------------------
    # 1. Build a set of initial candidates
    # ------------------------------------------------------------------
    candidates = []

    # a) exact arcsine distribution (area discretisation)
    idx = np.arange(n, dtype=np.float64)
    x0 = idx / n
    x1 = (idx + 1) / n
    a_arc = arcsine_cdf(x1) - arcsine_cdf(x0)
    a_arc /= a_arc.sum()
    candidates.append(a_arc)

    # b) regularised arcsine shapes  a_i ∝ 1/√((i+δ)(n‑1‑i+δ))
    for delta in (0.05, 0.2, 0.5, 1.0, 2.0):
        i = np.arange(n, dtype=np.float64)
        a = 1.0 / np.sqrt((i + delta) * (n - 1 - i + delta))
        a = np.maximum(a, 0.0)
        a /= a.sum()
        candidates.append(a)

    # c) uniform distribution
    candidates.append(np.full(n, 1.0 / n))

    # d) a couple of random Dirichlet samples (α=0.5 gives sparse vectors)
    for _ in range(2):
        a = np.random.gamma(0.5, 1.0, size=n)
        a /= a.sum()
        candidates.append(a)

    # ------------------------------------------------------------------
    # 2. Quick evaluation of all starters (hard metric, no smoothing)
    # ------------------------------------------------------------------
    best_a = None
    best_score = float("inf")
    for a in candidates:
        if time.time() >= deadline:
            break
        # metric for a probability vector (sum = 1) is 2·n·max(a*a)
        score = 2.0 * n * _conv_max(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    if best_a is None:                     # fallback – should never happen
        best_a = np.full(n, 1.0 / n)
        best_score = 2.0 * n * _conv_max(best_a)

    print(f"[Info] Initial best metric = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Adam optimisation on the simplex (logits = log a)
    # ------------------------------------------------------------------
    # initialise logits (softmax will recover a)
    z = np.log(best_a + 1e-16)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    # step‑size and smoothing schedules
    base_alpha = 0.7          # initial magnitude for the Adam step
    eps_factor = 0.5          # start temperature for the soft‑max of the convolution
    eps_factor_min = 5e-5     # final temperature (very close to the true max)
    eps_decay = 0.9996        # exponential decay per iteration

    it = 0
    max_iters = 5_000_000

    # helper for occasional progress reports
    def _log(it_, sc_):
        elapsed = time.time() - start_time
        print(f"[Iter {it_:7d}] best_score = {sc_:.6f}   elapsed = {elapsed:.1f}s")

    while time.time() < deadline and it < max_iters:
        it += 1

        # current probability vector (always sums to 1)
        a_cur = _softmax(z)

        # smoothed max‑convolution and its gradient
        max_conv, grad_a = _conv_max_and_grad(a_cur, eps_factor)

        cheap_score = 2.0 * n * max_conv

        # keep the best vector seen so far
        if cheap_score < best_score:
            best_score = cheap_score
            best_a = a_cur.copy()
            if it % 5000 == 0 or cheap_score < 1.05:
                _log(it, best_score)

        # natural‑gradient step on the logits (mirror descent on the simplex)
        mu = np.dot(grad_a, a_cur)                # 〈∇, a〉
        grad_z = a_cur * (grad_a - mu)            # ∇ₖJ

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # inverse‑sqrt decay of the learning rate
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step = - alpha_t * m_hat / (np.sqrt(v_hat) + eps_adam)

        z = z + step
        # keep logits numerically safe
        np.clip(z, -80.0, 80.0, out=z)

        # temperature schedule
        eps_factor = max(eps_factor_min, eps_factor * eps_decay)

        # occasional jitter to escape shallow plateaus
        if it % 20000 == 0 and np.random.rand() < 0.2:
            z += 0.01 * np.random.randn(n)

        # early exit if we are already extremely close to the continuous optimum
        if best_score < 1.01 and it > 5000:
            print("[Info] Near‑optimal score reached – terminating early.")
            break

    # ------------------------------------------------------------------
    # 4. Tiny random local perturbations (mass transfers) – final polish
    # ------------------------------------------------------------------
    while time.time() < deadline:
        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        if i == j:
            continue
        # move a tiny fraction of mass from i to j
        eps = best_a[i] * np.random.uniform(0.0, 0.005)
        if eps <= 0.0:
            continue
        a_new = best_a.copy()
        a_new[i] -= eps
        a_new[j] += eps
        # evaluate the new candidate (hard metric)
        new_score = 2.0 * n * _conv_max(a_new)
        if new_score < best_score:
            best_score = new_score
            best_a = a_new
            elapsed = time.time() - start_time
            print(f"[Perturb] improved score = {best_score:.6f}   elapsed = {elapsed:.1f}s")

    # ------------------------------------------------------------------
    # 5. Final logging (the platform will evaluate the sequence anyway)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[Info] Final exact score = {final_score:.6f}")
    except Exception as exc:
        print(f"[Warning] Evaluation error (ignored): {exc}")

    # return a plain Python list as required by the platform
    return best_a.tolist()
```

This implementation respects all platform restrictions, performs a thorough
search (initialisation + long Adam refinement + final random polishing) within
the 1000‑second budget, and is designed to push the evaluation metric well
below the 1.503 target.
