def project_simplex(v):
    """ Projects a real vector v onto the probability simplex (sum=1, v_i >= 0)."""
    # https://arxiv.org/abs/1309.1541 algorithm O(n log n).
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a non‑negative step‑function (a probability vector) that minimises
the metric 2·n·max(a*a)/(sum(a)^2).

Algorithm
---------
1. Initialise the vector from the exact discretisation of the arcsine
   distribution (the known near‑optimal continuous shape).  If the platform
   already supplied a sequence `height_sequence_1`, we reuse it after normalising
   to sum = 1.

2. Parameterise the vector as a soft‑max of unrestricted logits `z`.  This
   automatically enforces non‑negativity and the simplex constraint,
   avoiding an expensive projection step.

3. Perform a long “mirror‑descent” optimisation on `z` using the Adam
   optimiser.  The gradient is the exact sub‑gradient of the objective:
   for the current maximum entry of the self‑convolution (index `k*`),

        ∂/∂a_j  max(a*a) = 2·a_{k*‑j}   (if the index is valid, else 0).

   The natural‑gradient on the simplex is
        ∇_z J = a ∘ (∇_a J − ⟨∇_a J, a⟩),

   where `a = softmax(z)`.  Only a single FFT (to obtain the convolution
   and its maximising index) is required per iteration, making the method
   fast enough to run several‑hundred thousand steps inside the 1000 s budget.

4. Keep the best candidate seen (lowest cheap objective
   `2·n·max_conv`) and return it.  A final exact evaluation via
   `evaluate_sequence` is performed only for logging.

The implementation is completely self‑contained, uses only NumPy and the
standard library, and respects the platform’s time limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / np.sum(e)


def _fft_len_for_corr(n: int) -> int:
    """
    Smallest power‑of‑two ≥ 3·n‑2.
    Used for linear self‑convolution via FFT.
    """
    return 1 << ((3 * n - 2).bit_length())


def _conv_max_and_index(a: np.ndarray, n: int, L: int):
    """
    Returns the maximum entry of the linear self‑convolution `a*a`,
    together with its index (0 ≤ idx < 2·n‑1).
    """
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    idx = int(np.argmax(conv))
    max_val = float(conv[idx])
    return max_val, idx


def _init_arcsine(n: int) -> np.ndarray:
    """
    Exact area‑discretisation of the arcsine distribution on [0,1].
    Returns a probability vector (sum = 1).
    """
    idx = np.arange(n, dtype=np.float64)
    x0 = idx / n
    x1 = (idx + 1) / n

    def arcsine_cdf(x):
        # Guard against numerical slip at the boundaries.
        return (2.0 / np.pi) * np.arcsin(np.sqrt(np.clip(x, 0.0, 1.0)))

    a = arcsine_cdf(x1) - arcsine_cdf(x0)
    s = a.sum()
    if s <= 0.0 or not np.isfinite(s):
        raise RuntimeError("Failed to initialise a valid arcsine distribution.")
    a /= s
    return a


def _initialise_from_global(n: int) -> np.ndarray | None:
    """
    If the global variable `height_sequence_1` exists, use it (after
    normalising to sum = 1).  Returns ``None`` if the variable is missing or
    unusable.
    """
    try:
        seq = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        seq = np.maximum(seq, 0.0)                     # drop accidental negatives
        s = seq.sum()
        if s > 0.0 and np.isfinite(s):
            seq /= s
            return seq
    except Exception:
        pass
    return None

    # ------------------------------------------------------------------

        # ---------- Exact sub‑gradient of the max‑convolution ----------
        # Gradient w.r.t. a (size n).
        grad_a = np.zeros_like(a)

        # Valid j‑range for the active index `idx_max`.
        j_min = max(0, idx_max - (n - 1))
        j_max = min(n - 1, idx_max)
        if j_min <= j_max:
            idxs = np.arange(j_min, j_max + 1, dtype=np.int64)
            # grad_a[j] = 4·n·a[idx_max‑j]  (the factor 4·n comes from 2·n·max)
            grad_a[idxs] = 4.0 * n * a[idx_max - idxs]

        # Natural‑gradient on the simplex (mirrored descent).
        mu = np.dot(grad_a, a)               # ⟨∇,a⟩
        grad_z = a * (grad_a - mu)           # ∇_z J

        # Adam update (bias‑corrected).
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # Inverse‑sqrt decay of the step size.
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step = -alpha_t * m_hat / (np.sqrt(v_hat) + eps_adam)

        # Logits update and clipping for numerical safety.
        z = z + step
        z = np.clip(z, -60.0, 60.0)

        # Small random jitter every few thousand iterations to escape plateaus.
        if it % 5000 == 0 and np.random.rand() < 0.1:
            z += 0.01 * np.random.randn(n)

        # Early exit if we have already crossed the target and a reasonable
        # number of iterations have elapsed.
        if best_score < 1.503 and it > 5000:
            print("[Info] Target threshold reached – stopping early.")
            break

    # ------------------------------------------------------------------
    # 3. Return the best vector found (as a plain Python list).
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()

    # Optional final exact evaluation (only for logging; the platform will
    # evaluate the returned sequence anyway).
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[Info] Final exact score = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print(f"[Warning] Evaluation error (ignored): {exc}")

    return final_seq
