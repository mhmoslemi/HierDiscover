#!/usr/bin/env python
"""
Hybrid optimiser for the max‑convolution metric.

* Initialise from the exact discrete arcsine law (the known good construction).
* Perform a long Adam‑style mirror‑descent on the simplex, using a
  smoothed soft‑max approximation of the infinity‑norm of the self‑convolution.
  The schedule for the smoothing constant and the learning rate is deliberately
  aggressive so that the optimiser can keep making progress for many minutes.
* After the gradient phase, run a few iterations of the classic alternating‑LP
  refinement (the “linear‑programming boost” from the state‑of‑the‑art paper):
  for the current distribution `f` solve
        maximise   Σ g_j
        subject to 0 ≤ g_j ,   (f * g)_k ≤ M_f   for all k,
  where `M_f = max (f*f)`.  Rescale `g` to unit sum, mix it with `f`,
  and accept the mixture if it strictly reduces `M`.  This step can
  dramatically lower the peak of the self‑convolution.
* Keep the best sequence seen during the whole run and return it.

The routine works for any length `n`; if a global ``height_sequence_1`` is
present its length is used, otherwise a moderate default (2000) is chosen so
that the linear programmes stay cheap.  All work is stopped well before the
budget expires, and the final list is printed together with the exact
evaluation score.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
from scipy.linalg import toeplitz

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max → probability vector."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Logits that map to a via the soft‑max (adds a tiny epsilon for safety)."""
    return np.log(a + eps)


def _fft_len_for_conv(n: int) -> int:
    """Smallest power‑of‑two ≥ 2·n‑1, sufficient for linear convolution."""
    return 1 << ((2 * n - 1).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a (no gradient)."""
    n = a.shape[0]
    L = _fft_len_for_conv(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Smoothed maximum of a*a together with its gradient w.r.t. *a*.

    Parameters
    ----------
    a : np.ndarray
        Probability vector (non‑negative, sum = 1).
    eps_factor : float
        Multiplicative factor for the smoothing constant:
        eps = max(max_conv * eps_factor, 1e-12).

    Returns
    -------
    max_conv : float
        Maximum entry of the self‑convolution.
    grad_a : np.ndarray
        Gradient of 4·n·smooth_max w.r.t. a (already multiplied by 4·n).
    """
    n = a.shape[0]
    L = _fft_len_for_conv(n)

    a_fft = np.fft.rfft(a, L)

    # linear self‑convolution (length 2·n‑1)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # smoothed soft‑max weights
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv               # improve stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum

    # cross‑correlation a ⋆ w (linear, length n)
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # gradient (already multiplied by 4·n)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    `evaluate_sequence`.  The routine follows the two‑phase hybrid
    strategy described in the module docstring.
    """
    np.random.seed(seed)
    start = time.time()
    # leave a small safety margin for final logging / return
    deadline = start + budget_s - 2.0

    # --------------------------------------------------------------
    # 0. Choose length ------------------------------------------------
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
        print(f"Using external length n = {n}")
    except Exception:
        # 2000 is a comfortable size for the LP stage while still giving
        # a good bound for the metric.
        n = 2000
        print(f"No external length – using default n = {n}")

    # --------------------------------------------------------------
    # 1. Exact arcsine discretisation (probability vector)
    # --------------------------------------------------------------
    i = np.arange(n, dtype=np.float64)
    x0 = i / n
    x1 = (i + 1) / n
    sqrt0 = np.sqrt(np.clip(x0, 0.0, 1.0))
    sqrt1 = np.sqrt(np.clip(x1, 0.0, 1.0))
    F0 = (2.0 / np.pi) * np.arcsin(sqrt0)
    F1 = (2.0 / np.pi) * np.arcsin(sqrt1)
    a = F1 - F0                     # bin probabilities, sum ≈ 1
    a_sum = a.sum()
    if not np.isfinite(a_sum) or a_sum <= 0.0:
        raise RuntimeError("Failed to generate arcsine bins.")
    a /= a_sum                      # enforce exact normalisation

    best_a = a.copy()
    best_score = 2.0 * n * _conv_max(best_a)
    print(f"Initial score (2·n·max_conv) = {best_score:.8f}")

    # --------------------------------------------------------------
    # 2. Long Adam optimisation on the simplex
    # --------------------------------------------------------------
    # Parameter choices are tuned for a long, steady descent.
    z = _logits_from_distribution(best_a)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_alpha = 1.0               # larger base step size than the baseline
    eps_factor = 0.5               # start with a fairly smooth approximation
    eps_factor_min = 5e-5
    eps_factor_decay = 0.9997
    it = 0
    max_iters = 200_000

    while time.time() < deadline and it < max_iters:

        max_conv, grad_a = _conv_max_and_grad(a_cur, eps_factor)
        cheap_score = 2.0 * n * max_conv

        if cheap_score < best_score:
            best_score = cheap_score
            best_a = a_cur.copy()
            if it % 5000 == 0:
                print(f"Iter {it:6d}: improved score = {best_score:.8f}")

        # --- Adam step on the logits (mirror descent) ---
        mu = np.dot(grad_a, a_cur)                # ⟨grad, a⟩
        grad_z = a_cur * (grad_a - mu)            # keep us on the simplex manifold

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

        # bias‑corrected moments
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step size schedule (inverse‑sqrt)
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step = -alpha_t * m_hat / (np.sqrt(v_hat) + eps_adam)
        z = z + step
        # keep the logits bounded to avoid overflow in the exponent
        z = np.clip(z, -60.0, 60.0)

        # decay the smoothing constant
        eps_factor = max(eps_factor_min, eps_factor * eps_factor_decay)

        # occasional tiny random jitter to shake the system out of plateaus
        if it % 10000 == 0 and np.random.rand() < 0.1:
            z += 0.005 * np.random.randn(n)

        # stop early if we already beat the target
        if cheap_score < 1.503:
            print(f"Target crossed at iteration {it}, score {cheap_score:.8f}")
            break

    # --------------------------------------------------------------
    # 3. Alternating linear‑programming boost
    # --------------------------------------------------------------
    a = best_a.copy()
    M = _conv_max(a)          # current worst convolution entry
    mix_t = 0.5               # mixing weight for the LP‑derived candidate
    lp_iter = 0

    while time.time() < deadline - 2.0:
        lp_iter += 1
        # Build the (2n‑1)×n convolution matrix T(f) such that T(f) @ g = f*g
        # Toeplitz structure: first column = [f, 0, …, 0], first row = [f0, 0,…,0]
        col = np.concatenate([a, np.zeros(len(a) - 1)])
        row = np.concatenate([a[:1], np.zeros(len(a) - 1)])
        A = toeplitz(col, row)               # shape (2n‑1, n)

        c = -np.ones(len(a))                  # maximise Σ g_j  → minimise –Σ g_j
        b_ub = np.full(A.shape[0], M)
        bounds = [(0.0, None)] * len(a)

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})

        if not res.success:
            print(f"LP #{lp_iter}: solver failed ({res.message}); aborting LP phase.")
            break

        g_raw = res.x
        if g_raw.sum() == 0.0:
            print(f"LP #{lp_iter}: zero solution – aborting.")
            break

        # Normalise to unit sum (the metric is scale‑invariant)
        g = g_raw / g_raw.sum()

        # Mix the new candidate with the current distribution
        a_new = (1.0 - mix_t) * a + mix_t * g
        M_new = _conv_max(a_new)
        score_new = 2.0 * n * M_new

        print(f"LP #{lp_iter}: mixed score = {score_new:.8f}")

        # Keep the best overall
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            print(f"  *** New best after LP #{lp_iter}: {best_score:.8f}")

        # Accept the mixture only if it strictly improves M (guarantees descent)
        if M_new < M:
            a = a_new
            M = M_new
        else:
            print(f"LP #{lp_iter}: no improvement in M, stopping LP iterations.")
            break

        # After each LP step run a short burst of Adam updates
        # (helps to polish the freshly mixed point)
        z = _logits_from_distribution(a)
        m = np.zeros_like(z)
        v = np.zeros_like(z)
        inner_max = 3000
        inner_it = 0
        while time.time() < deadline - 1.0 and inner_it < inner_max:
            inner_it += 1
            a_cur = _softmax(z)
            max_conv, grad_a = _conv_max_and_grad(a_cur, eps_factor)

            cheap_score = 2.0 * n * max_conv
            if cheap_score < best_score:
                best_score = cheap_score
                best_a = a_cur.copy()
                if inner_it % 1000 == 0:
                    print(f"    (inner) iter {inner_it:4d}: new best {best_score:.8f}")

            mu = np.dot(grad_a, a_cur)
            grad_z = a_cur * (grad_a - mu)

            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

            m_hat = m / (1.0 - beta1 ** inner_it)
            v_hat = v / (1.0 - beta2 ** inner_it)

            alpha_t = base_alpha / (math.sqrt(inner_it) + 1.0)
            step = -alpha_t * m_hat / (np.sqrt(v_hat) + eps_adam)
            z = z + step
            z = np.clip(z, -60.0, 60.0)

            # shrink eps_factor a little more during the fine‑tuning bursts
            eps_factor = max(eps_factor_min, eps_factor * 0.9995)

    # --------------------------------------------------------------
    # 4. Return the best sequence found
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        exact_score = evaluate_sequence(final_seq)
        print(f"Exact evaluation (final): {exact_score:.8f}")
    except Exception as exc:
        print(f"Exact evaluation failed: {exc}")

    return final_seq
