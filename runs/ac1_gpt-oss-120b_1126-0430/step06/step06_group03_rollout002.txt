"""
Hybrid optimizer for the self‑convolution bound.

The routine works in four stages:

1. **Initialisation** – use a provided sequence (if any) or a uniform
   distribution of length ≈25 000.  The vector is normalised to sum = 1.

2. **Exploratory sampling** – a short random Dirichlet sweep followed by
   a dense parametric sweep over several one‑ and two‑parameter families
   (power‑law, offset power‑law, double‑power‑law, stretched‑exponential,
   exponential, logistic, symmetric product‑law, cosine‑power, beta‑law).
   The cheap score `2·n·max(a*a)` is used to keep the best candidate.

3. **Mirror‑descent on the simplex** – starting from the best candidate,
   an Adam optimiser works on the log‑softmax (“logits”) variables.
   A smooth‑max surrogate (softmax of the convolution) supplies
   differentiable gradients.  The learning‑rate and the smoothing
   parameter are decayed over time; a back‑tracking line‑search guarantees
   monotone improvement of the cheap score.  If progress stalls the
   optimiser is re‑initialised from the best point seen so far.

4. **Fine‑grained random walk** – the final phase performs cheap mass‑transfer
   moves between random positions.  Only moves that strictly improve the
   cheap score are accepted; occasional larger jumps help escape plateaus.

The function returns the best sequence found before the time budget
expires.  The sequence is a plain list of non‑negative floats; its sum
is ≈ 1, satisfying the checks performed by `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    s = e.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / z.size)
    return e / s


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Logit that reproduces `a` after a soft‑max."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """Smallest power‑of‑two ≥ 3·n‑2, enough for the convolution and its gradient."""
    return 1 << ((3 * n - 2).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Smooth‑max of the self‑convolution and its gradient w.r.t. the
    probability vector `a`.

    Returns (max_conv, grad_a) where

        grad_a = ∇_a [ 2·n·smooth_max(a*a) ].
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv                     # ≤ 0 for stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # gradient of 2·n·smooth_max (the factor 4·n comes from the derivative of a*a)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a


# ----------------------------------------------------------------------
# Parametric families of probability vectors (length n, sum = 1)
# ----------------------------------------------------------------------


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    a /= s if s > 0 else 1.0
    return a


def _power_law_offset_distribution(n: int, p: float, offset: float) -> np.ndarray:
    """a_i ∝ (i+offset)^{-p}."""
    idx = np.arange(offset, offset + n, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    a /= s if s > 0 else 1.0
    return a


def _double_power_law_distribution(n: int, p: float, q: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}·(n-i)^{-q}."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) ** (-p)) * ((n - i) ** (-q))
    s = a.sum()
    a /= s if s > 0 else 1.0
    return a


def _stretched_exp_distribution(n: int, tau_factor: float, gamma: float) -> np.ndarray:
    """
    a_i ∝ exp( - (i/(tau_factor·n))^{gamma} ).
    """
    tau = max(1e-12, tau_factor * n)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(- (i / tau) ** gamma)
    s = a.sum()
    a /= s if s > 0 else 1.0
    return a


def _exp_decay_distribution(n: int, lam: float) -> np.ndarray:
    """a_i ∝ e^{-lam·i}+e^{-lam·(n‑1‑i)}."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    s = a.sum()
    a /= s if s > 0 else 1.0
    return a


def _logistic_distribution(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ 1/(1+e^{α·(i‑mid)})."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = 1.0 / (1.0 + np.exp(alpha * (i - mid)))
    s = a.sum()
    a /= s if s > 0 else 1.0
    return a


def _sym_product_power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ ((i+1)*(n‑i))^{-p}  (symmetric)."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) * (n - i)) ** (-p)
    s = a.sum()
    a /= s if s > 0 else 1.0
    return a


def _cosine_power_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ (1‑cos(π·i/(n‑1)))^{p}."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - np.cos(np.pi * i / (n - 1))) ** p
    s = a.sum()
    a /= s if s > 0 else 1.0
    return a


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ (x)^{α}·(1‑x)^{β} with x = i/(n‑1)."""
    i = np.arange(n, dtype=np.float64)
    x = i / (n - 1)
    # add a tiny epsilon to avoid 0^{exp} when α or β < 0
    eps = 1e-12
    a = (x + eps) ** alpha * ((1.0 - x) + eps) ** beta
    s = a.sum()
    a /= s if s > 0 else 1.0
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Returns a list of non‑negative heights that (hopefully) gives a low
    `evaluate_sequence` score.  The routine respects the supplied time
    budget (~1000 s) and uses a hybrid parametric sweep + Adam optimiser
    on the simplex, followed by a cheap random walk.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # margin for final evaluation

    # --------------------------------------------------------------
    # 0. Problem size & initial distribution
    # --------------------------------------------------------------
    try:
        init_seq = np.array(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        n = max(20000, init_seq.shape[0])
        a0 = np.maximum(init_seq, 0.0)
        s = a0.sum()
        if s > 0.0:
            a0 /= s
        else:
            a0 = np.full_like(a0, 1.0 / a0.shape[0])
        print(f"[Init] Using provided sequence, length = {n}")
    except Exception:
        n = 25000
        a0 = np.full(n, 1.0 / n)
        print(f"[Init] No provided sequence – start from uniform, length = {n}")

    best_seq = a0.copy()
    best_score = 2.0 * n * _conv_max(best_seq)

    # --------------------------------------------------------------
    # 1. Random Dirichlet sampling (quick diversifier)
    # --------------------------------------------------------------
    dirichlet_trials = 80
    for _ in range(dirichlet_trials):
        if time.time() >= deadline:
            break
        # Dirichlet with concentration 0.5 → spikier vectors
        gamma = np.random.gamma(0.5, 1.0, size=n)
        cand = gamma / gamma.sum()
        score = 2.0 * n * _conv_max(cand)
        if score < best_score:
            best_score = score
            best_seq = cand.copy()

    # --------------------------------------------------------------
    # 2. Coarse parametric sweep (extended ranges)
    # --------------------------------------------------------------
    families = [
        ("power_law", _power_law_distribution,
         [dict(p=p) for p in np.linspace(0.5, 2.0, 16)]),

        ("power_law_offset", _power_law_offset_distribution,
         [dict(p=p, offset=off)
          for p in np.linspace(0.5, 2.0, 11)
          for off in np.linspace(0.1, 2.0, 11)]),

        ("double_power_law", _double_power_law_distribution,
         [dict(p=p, q=q)
          for p in np.linspace(0.5, 2.0, 9)
          for q in np.linspace(0.5, 2.0, 9)]),

        ("stretched_exp", _stretched_exp_distribution,
         [dict(tau_factor=t, gamma=g)
          for t in np.linspace(0.2, 0.9, 8)
          for g in np.linspace(0.5, 4.0, 8)]),

        ("exp_decay", _exp_decay_distribution,
         [dict(lam=lam) for lam in np.logspace(-8, -2, 30)]),

        ("logistic", _logistic_distribution,
         [dict(alpha=alpha) for alpha in np.linspace(0.5, 10.0, 20)]),

        ("sym_product_power_law", _sym_product_power_law_distribution,
         [dict(p=p) for p in np.linspace(0.5, 2.0, 16)]),

        ("cosine_power", _cosine_power_distribution,
         [dict(p=p) for p in np.linspace(0.5, 5.0, 10)]),

        ("beta", _beta_distribution,
         [dict(alpha=a, beta=b)
          for a in np.linspace(0.5, 5.0, 8)
          for b in np.linspace(0.5, 5.0, 8)]),
    ]

    print("[Sweep] Starting parametric search")
    for name, generator, param_dicts in families:
        for params in param_dicts:
            if time.time() >= deadline:
                break
            try:
                cand = generator(n, **params)
            except Exception:
                continue
            score = 2.0 * n * _conv_max(cand)
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
        if time.time() >= deadline:
            break

    print(f"[Sweep] Best cheap score after sweep = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3. Mirror‑descent (Adam) on the simplex
    # --------------------------------------------------------------
    logits = _logits_from_distribution(best_seq) + 0.03 * np.random.randn(n)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    base_alpha = 0.45                     # larger initial step size
    eps_factor = 0.20                     # start with more smoothing
    eps_factor_min = 3e-5
    eps_decay = 0.9985                    # faster decay → sharper gradients earlier

    it = 0
    last_improve = 0
    best_seq_sofar = best_seq.copy()
    best_score_sofar = best_score

    while time.time() < deadline and it < 1_500_000:
        it += 1
        a = _softmax(logits)                     # probability vector (sum≈1)

        max_conv, grad_a = _conv_max_and_grad(a, eps_factor)
        cheap_score = 2.0 * n * max_conv

        # record best ever seen
        if cheap_score < best_score_sofar - 1e-12:
            best_score_sofar = cheap_score
            best_seq_sofar = a.copy()
            last_improve = it

        # mirror‑descent gradient w.r.t. logits (natural gradient on simplex)
        mu = np.dot(grad_a, a)
        grad_z = a * (grad_a - mu)

        # Adam updates (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step‑size schedule (inverse‑sqrt decay, slightly more aggressive)
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step = - m_hat / (np.sqrt(v_hat) + eps_adam)

        # back‑tracking line‑search (up to 3 halvings)
        accepted = False
        trial_alpha = alpha_t
        for _ in range(3):
            new_logits = logits + trial_alpha * step
            new_logits = np.clip(new_logits, -60.0, 60.0)
            a_new = _softmax(new_logits)
            new_score = 2.0 * n * _conv_max(a_new)
            if new_score <= cheap_score:
                logits = new_logits
                accepted = True
                break
            trial_alpha *= 0.5
        if not accepted:
            logits = logits + alpha_t * step
            logits = np.clip(logits, -60.0, 60.0)

        # occasional stochastic jitter to escape flat regions
        if it % 3500 == 0 and np.random.rand() < 0.12:
            logits += 0.03 * np.random.randn(n)
            m *= 0.55
            v *= 0.55

        # decay the smoothing constant
        eps_factor = max(eps_factor_min, eps_factor * eps_decay)

        # restart if no improvement for a long stretch
        if it - last_improve > 300_000:
            print(f"[Iter {it}] Stagnation – restarting from best-so‑far")
            logits = _logits_from_distribution(best_seq_sofar) + 0.04 * np.random.randn(n)
            m.fill(0.0)
            v.fill(0.0)
            eps_factor = 0.20
            last_improve = it

    best_seq = best_seq_sofar
    best_score = best_score_sofar
    print(f"[Adam] Finished after {it} iterations, best cheap score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Final cheap random walk (coordinate mass transfers)
    # --------------------------------------------------------------
    a = best_seq.copy()
    cheap_score = best_score

    while time.time() < deadline:
        # pick two distinct indices
        i = np.random.randint(0, n)
        j = np.random.randint(0, n - 1)
        if j >= i:
            j += 1
        if a[i] <= 0.0:
            continue
        # transfer a random fraction (0–1 %) of the mass from i to j
        delta = np.random.rand() * 0.01 * a[i]
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = 2.0 * n * _conv_max(a_new)
        if new_score < cheap_score:
            a = a_new
            cheap_score = new_score
            best_seq = a.copy()
            best_score = cheap_score
            # sparse progress report
            if np.random.rand() < 0.002:
                print(f"[Walk] Improved cheap = {cheap_score:.8f}")

        # every few hundred steps try a bigger shuffle (swap two masses)
        if np.random.rand() < 0.001:
            p, q = np.random.randint(0, n, size=2)
            if p != q:
                a_swap = a.copy()
                a_swap[p], a_swap[q] = a_swap[q], a_swap[p]
                new_score2 = 2.0 * n * _conv_max(a_swap)
                if new_score2 < cheap_score:
                    a = a_swap
                    cheap_score = new_score2
                    best_seq = a.copy()
                    best_score = cheap_score

    # --------------------------------------------------------------
    # 5. Final exact evaluation
    # --------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        exact_score = evaluate_sequence(final_seq)   # provided by the platform
        print(f"Final exact score = {exact_score:.6f}")
    except Exception as exc:          # safety net – should never happen
        exact_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    return final_seq
