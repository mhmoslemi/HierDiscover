def proj_simplex(v):
    # returns projection onto simplex {x>=0, sum x =1}
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u > (cssv - 1) / np.arange(1, len(v)+1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Hybrid optimizer for the self‑convolution bound.

The algorithm works as follows:

1. **Initialization** – use the “arcsine” (Beta(½,½)) distribution,
   which is a theoretically good starting point for flattening the
   self‑convolution.  If the platform supplies a sequence `height_sequence_1`,
   it is normalised and used as a fallback.

2. **Coarse parametric sweep** – a very cheap grid‑search over a few
   families (power‑law, offset power‑law, double power‑law, stretched‑exp,
   logistic).  The sweep is limited in depth so that it finishes quickly
   but may still discover a better starting point than the arcsine law.

3. **Exact mirror‑descent on the simplex** – we maintain a soft‑max
   parametrisation `a = softmax(logits)`.  In each iteration we evaluate
   the exact (non‑smoothed) sub‑gradient of the objective
   `2·n·max(a*a)`.  The sub‑gradient is cheap to obtain because the
   maximising index `k*` yields a simple shift of `a`.  An Adam optimiser
   with back‑tracking line‑search guarantees monotone improvement of the
   cheap score.  The loop runs until the time budget is exhausted.

4. **Final random walk** – any remaining time is spent on cheap coordinate
   moves (transfer a tiny mass between two entries) that are accepted only
   if they improve the cheap score.

5. **Exact evaluation** – the best sequence found is handed to the
   provided `evaluate_sequence` function for the final score.

The whole routine respects the 1000 s time budget, prints progress, and
returns a list of non‑negative heights.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable softmax."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    s = e.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / z.size)
    return e / s


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it that reproduces `a` after a softmax."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """Smallest power‑of‑two ≥ 3·n‑2 (enough for all needed convolutions)."""
    return 1 << ((3 * n - 2).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_subgrad(a: np.ndarray):
    """
    Returns (max_conv, grad_a) where

        grad_a = ∇_a [ max_k (a*a)_k ]

    The gradient is a sub‑gradient obtained from the index `k` that attains
    the maximum (if several indices tie, one of them is chosen arbitrarily).
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())
    k = int(conv.argmax())            # an index achieving the maximum

    # sub‑gradient of max_k (a*a)_k at the chosen k
    i = np.arange(n)
    j = k - i
    mask = (j >= 0) & (j < n)
    grad = np.zeros_like(a)
    # derivative of sum_i a_i a_{k-i} w.r.t. a_i is 2*a_{k-i}
    grad[mask] = 2.0 * a[j[mask]]
    return max_conv, grad


# ----------------------------------------------------------------------
# Parametric families (all return a normalised probability vector)
# ----------------------------------------------------------------------


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}   (normalised)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _power_law_offset_distribution(n: int, p: float, offset: float) -> np.ndarray:
    """a_i ∝ (i+offset)^{-p}   (normalised)."""
    idx = np.arange(offset, offset + n, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _double_power_law_distribution(n: int, p: float, q: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}·(n-i)^{-q}   (normalised)."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) ** (-p)) * ((n - i) ** (-q))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _stretched_exp_distribution(n: int, tau_factor: float, gamma: float) -> np.ndarray:
    """a_i ∝ exp( - (i/(tau_factor·n))^{gamma} )   (normalised)."""
    tau = max(1e-12, tau_factor * n)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(- (i / tau) ** gamma)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _exp_decay_distribution(n: int, lam: float) -> np.ndarray:
    """a_i ∝ e^{-lam·i}+e^{-lam·(n‑1‑i)}   (normalised)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _logistic_distribution(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ 1/(1+e^{α·(i‑mid)})   (normalised)."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = 1.0 / (1.0 + np.exp(alpha * (i - mid)))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discrete Beta(α,β) on {0,…,n‑1} using CDF differences.
    a_i = F((i+1)/n) - F(i/n)  where F is the regularised beta CDF.
    """
    # use the regularised incomplete beta function from scipy if available,
    # otherwise fall back to the simple power‑law form.
    try:
        from scipy.special import betainc
        x = np.arange(n + 1, dtype=np.float64) / n
        cdf = betainc(alpha, beta, x)  # regularised incomplete beta
        a = np.diff(cdf)
    except Exception:
        # fallback: treat as (i+0.5)^{α-1} (1-i/n)^{β-1}
        i = np.arange(n, dtype=np.float64) + 0.5
        t = i / n
        a = (t ** (alpha - 1)) * ((1.0 - t) ** (beta - 1))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) improves the
    `evaluate_sequence` score.  The routine respects the supplied time budget
    (~1000 s) and uses a hybrid parametric sweep + Adam optimiser on the
    probability simplex, followed by a cheap random walk.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0   # a small safety margin

    # --------------------------------------------------------------
    # 0. Problem size and initial distribution
    # --------------------------------------------------------------
    try:
        init_seq = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        n = max(20000, init_seq.shape[0])
        a0 = np.maximum(init_seq, 0.0)
        total = a0.sum()
        if total == 0.0:
            a0 = np.full_like(a0, 1.0 / a0.shape[0])
        else:
            a0 /= total
        print(f"[Init] Using provided sequence, length = {n}")
    except Exception:
        n = 25000
        # arcsine (Beta(0.5,0.5)) – a strong baseline
        a0 = _beta_distribution(n, 0.5, 0.5)
        print(f"[Init] No provided sequence – start from beta(0.5,0.5), length = {n}")

    best_seq = a0.copy()
    best_score = 2.0 * n * _conv_max(best_seq)
    print(f"[Init] Starting cheap score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 1. Coarse parametric sweep (very cheap)
    # --------------------------------------------------------------
    families = [
        ("power_law", _power_law_distribution,
         [dict(p=p) for p in np.linspace(0.20, 2.00, 37)]),
        ("power_law_offset", _power_law_offset_distribution,
         [dict(p=p, offset=off)
          for p in np.linspace(0.20, 2.00, 19)
          for off in np.linspace(0.01, 5.0, 15)]),
        ("double_power_law", _double_power_law_distribution,
         [dict(p=p, q=q)
          for p in np.linspace(0.20, 2.00, 19)
          for q in np.linspace(0.20, 2.00, 19)]),
        ("stretched_exp", _stretched_exp_distribution,
         [dict(tau_factor=t, gamma=g)
          for t in np.linspace(0.05, 0.95, 13)
          for g in np.linspace(0.4, 3.0, 13)]),
        ("exp_decay", _exp_decay_distribution,
         [dict(lam=lam) for lam in np.logspace(-7, -2, 30)]),
        ("logistic", _logistic_distribution,
         [dict(alpha=alpha) for alpha in np.linspace(0.01, 8.0, 30)]),
        ("beta_5_5", lambda nn, **_: _beta_distribution(nn, 5.0, 5.0), [{}]),
        ("beta_1_3", lambda nn, **_: _beta_distribution(nn, 1.0, 3.0), [{}]),
        ("beta_3_1", lambda nn, **_: _beta_distribution(nn, 3.0, 1.0), [{}]),
    ]

    print("[Sweep] Starting coarse parametric search")
    for name, gen, param_grid in families:
        for params in param_grid:
            if time.time() >= deadline:
                break
            try:
                cand = gen(n, **params)
            except Exception:
                continue
            score = 2.0 * n * _conv_max(cand)
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                print(f"[Sweep] New best {name} {params} -> {score:.6f}")
        if time.time() >= deadline:
            break

    print(f"[Sweep] End of sweep, best cheap score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2. Exact mirror‑descent on the simplex (Adam + line‑search)
    # --------------------------------------------------------------
    # Initialise logits from the best vector found so far
    logits = _logits_from_distribution(best_seq) + 0.02 * np.random.randn(n)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    base_alpha = 0.45                     # base learning‑rate for logits
    report_every = 50000
    it = 0
    last_improve = 0
    best_seq_sofar = best_seq.copy()
    best_score_sofar = best_score

    while time.time() < deadline and it < 1_200_000:
        it += 1

        a = _softmax(logits)               # probability vector, sum≈1
        max_conv, grad_a = _conv_max_and_subgrad(a)
        cheap_score = 2.0 * n * max_conv

        # keep the best ever seen
        if cheap_score < best_score_sofar - 1e-15:
            best_score_sofar = cheap_score
            best_seq_sofar = a.copy()
            last_improve = it
            if it % report_every == 0:
                print(f"[Iter {it:07d}] NEW BEST cheap = {cheap_score:.8f}")

        # sub‑gradient w.r.t. logits (mirror‑descent)
        mu = np.dot(grad_a, a)                 # = 2*max_conv
        grad_z = a * (grad_a - mu)             # ∇_z f = a ⊙ (∇_a f – μ)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step size schedule (inverse‑sqrt decay)
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step = - m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- back‑tracking line‑search (max 5 halvings) ----
        accepted = False
        trial_alpha = alpha_t
        for _ in range(5):
            new_logits = logits + trial_alpha * step
            # keep logits in a numerically safe range
            new_logits = np.clip(new_logits, -50.0, 50.0)
            a_new = _softmax(new_logits)
            new_score = 2.0 * n * _conv_max(a_new)
            if new_score <= cheap_score:
                logits = new_logits
                accepted = True
                break
            trial_alpha *= 0.5
        if not accepted:
            # take a small step even if it does not improve (to escape flats)
            logits = logits + alpha_t * step
            logits = np.clip(logits, -50.0, 50.0)

        # occasional random jitter to get out of plateaus
        if it % 4000 == 0 and np.random.rand() < 0.12:
            logits += 0.02 * np.random.randn(n)
            m *= 0.6
            v *= 0.6

        # restart if stuck for a long time
        if it - last_improve > 300_000:
            print(f"[Iter {it}] Stagnation – restarting from best so far")
            logits = _logits_from_distribution(best_seq_sofar) + 0.03 * np.random.randn(n)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = it

    best_seq = best_seq_sofar
    best_score = best_score_sofar
    print(f"[Adam] Finished after {it} iterations, best cheap score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3. Final cheap random walk (mass transfer moves)
    # --------------------------------------------------------------
    a = best_seq.copy()
    cheap_score = best_score

    while time.time() < deadline:
        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        if i == j:
            continue
        delta = 0.001 * a[i]          # move 0.1 % of mass from i to j
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = 2.0 * n * _conv_max(a_new)
        if new_score < cheap_score:
            a = a_new
            cheap_score = new_score
            best_seq = a.copy()
            best_score = cheap_score
            # sporadic progress reports
            if np.random.rand() < 0.001:
                print(f"[Walk] Improved cheap = {cheap_score:.8f}")

    # --------------------------------------------------------------
    # 4. Final exact evaluation
    # --------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        exact_score = evaluate_sequence(final_seq)   # provided by the platform
        print(f"Final exact score = {exact_score:.6f}")
    except Exception as exc:
        exact_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    return final_seq
