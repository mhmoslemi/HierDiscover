import time

def _simplex_projection(v):
    """Project v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _next_pow2(x):
    return 1 << ((x - 1).bit_length())

def _conv_self_fft(a, fft_len):
    """Full self‑convolution a*a using real FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[:2 * a.size - 1]

def _conv_cross_fft(x, y, fft_len):
    """Full convolution x*y using real FFT."""
    fx = np.fft.rfft(x, fft_len)
    fy = np.fft.rfft(y, fft_len)
    conv = np.fft.irfft(fx * fy, fft_len)

        # 2. Very cheap permutation / swap local search (optional)
        # --------------------------------------------------------------

        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

Propose a non‑negative step function (list of heights) that minimises
``evaluate_sequence``.

The algorithm consists of three phases:

1. **Broad parametric sweep** – a variety of analytic families
   (uniform, exponential, power‑law, Gaussian, cosine, triangular,
   Dirichlet draws, …) are instantiated for several lengths
   (5000, 8000, 12000, 20000).  The best candidate of each length is kept.
2. **Smoothed exponential‑gradient descent** – the objective
   `max_k (a*a)_k` is approximated by a log‑sum‑exp smoothing.
   A mirror‑descent (exponentiated‑gradient) update on the simplex,
   together with an annealing schedule for the smoothing parameter β,

3. **Lightweight permutation local search** – random swaps are tried

The routine respects a user‑supplied time budget (default 1000 s) and

def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Small utility: next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.size - 1]


def _conv_cross_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution x*y using a real FFT."""
    fx = np.fft.rfft(x, fft_len)
    fy = np.fft.rfft(y, fft_len)
    conv = np.fft.irfft(fx * fy, fft_len)
    return conv[: len(x) + len(y) - 1]


def _cheap_score(a: np.ndarray, fft_len_self: int) -> float:
    """Return 2·n·max(conv) assuming sum(a)=1."""
    conv = _conv_self_fft(a, fft_len_self)
    return 2.0 * a.size * conv.max()
