def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    s = e.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / z.size)
    return e / s

    """Returns log‑it vector that reproduces `a` after a soft‑max."""
    return np.log(a + eps)

    """Smallest power‑of‑two ≥ 3·n‑2 (enough for linear self‑convolution)."""
    return 1 << ((3 * n - 2).bit_length())

def _conv_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1)."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return conv

    """Maximum entry of the linear self‑convolution a*a."""
    conv = _conv_full(a)
    return float(conv.max())


def _conv_max_and_exact_grad(a: np.ndarray):
    """
    Returns (max_conv, grad_a) where

        grad_a = ∇_a [ 2·n·max(a*a) ]

    The gradient is the exact sub‑gradient obtained from the indices where
    the convolution attains its maximum.
    """

    conv = _conv_full(a)
    max_val = float(conv.max())

    # tolerance for detecting all maximisers
    tol = 1e-12 * max(1.0, max_val)
    max_idxs = np.where(np.abs(conv - max_val) <= tol)[0]

    grad = np.zeros_like(a)
    for k in max_idxs:
        if k < n:
            # contribution from a[0..k] reversed
            grad[: k + 1] += a[k::-1]
        else:

            grad[start:] += a[n - 1 : start - 1 : -1]

        grad /= float(len(max_idxs))

    grad_a = 4.0 * n * grad
    return max_val, grad_a

def _power_law_distribution(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    a /= s if s != 0.0 else 1.0
    return a


def _power_law_offset_distribution(n: int, p: float, offset: float) -> np.ndarray:
    idx = np.arange(offset, offset + n, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    a /= s if s != 0.0 else 1.0
    return a


def _double_power_law_distribution(n: int, p: float, q: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) ** (-p)) * ((n - i) ** (-q))
    s = a.sum()
    a /= s if s != 0.0 else 1.0
    return a


def _stretched_exp_distribution(n: int, tau_factor: float, gamma: float) -> np.ndarray:
    tau = max(1e-12, tau_factor * n)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(- (i / tau) ** gamma)
    s = a.sum()
    a /= s if s != 0.0 else 1.0
    return a


def _exp_decay_distribution(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    s = a.sum()
    a /= s if s != 0.0 else 1.0
    return a


def _logistic_distribution(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = 1.0 / (1.0 + np.exp(alpha * (i - mid)))
    s = a.sum()
    a /= s if s != 0.0 else 1.0
    return a


def _sym_product_power_law_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) * (n - i)) ** (-p)
    s = a.sum()
    a /= s if s != 0.0 else 1.0
    return a

                print(f"[Walk] Improved cheap = {cheap_score:.8f}")

    # ------------------------------------------------------------------
    # 5. Exact evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        exact_score = evaluate_sequence(final_seq)   # provided by the platform
        print(f"Final exact score = {exact_score:.6f}")
    except Exception as exc:
        exact_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    return final_seq
