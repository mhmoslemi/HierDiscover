def propose_candidate(seed=42, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2

    # Determine n: use height_sequence_1 length if available else default 20000 but cap maybe 30000 not too big.

    # Use multiple restarts: e.g., restarts = 4

    best_score = float('inf')
    best_seq = None

    # Helper to initialize distribution:

    def init_arcsine(n):
        idx = np.arange(n, dtype=np.float64)
        x0 = idx / n
        x1 = (idx+1)/n
        F = lambda x: (2/np.pi)*np.arcsin(np.sqrt(np.clip(x,0,1)))
        a = F(x1)-F(x0)
        a_sum = a.sum()
        a = a / a_sum
        return a

    def init_beta(n, alpha):
        # Beta distribution with parameters alpha, alpha -> symmetric
        # pdf f(x) = x^{alpha-1} * (1-x)^{alpha-1} / B(alpha,alpha)
        # approximated via differences of CDF using regularized Beta CDF.
        from scipy.stats import beta
        # Use vector of size n: compute CDF at x1, x0.
        x0 = np.arange(n, dtype=np.float64)/n
        x1 = (np.arange(n)+1)/n
        cdf0 = beta.cdf(x0, alpha, alpha)
        cdf1 = beta.cdf(x1, alpha, alpha)
        a = cdf1 - cdf0
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full(n, 1.0/n)
        else:
            a /= a_sum
        return a

    def init_random(n):
        a = np.random.rand(n)  # uniform random positive
        a = a / a.sum()
        return a

    # optimization routine
    def run_opt(initial_a, time_limit):
        # return best vector and its score within time limit
        n = initial_a.shape[0]
        # Convert to logits
        z = np.log(initial_a + 1e-16)
        m = np.zeros_like(z)
        v = np.zeros_like(z)
        beta1, beta2 = 0.9, 0.999
        eps_adam = 1e-8

        base_alpha = 1.0 # start bigger
        eps_factor = 0.2
        eps_factor_min = 1e-6
        eps_decay = 0.999

        best_local_a = initial_a.copy()
        best_local_score = 2.0 * n * _conv_max(initial_a)  # cheap

        start_local = time.time()
        it = 0
        max_iters = 500_000

        while time.time() < start_local + time_limit and it < max_iters:
            it += 1
            a = _softmax(z)

            max_conv, grad_a = _conv_max_and_grad(a, eps_factor)

            score = 2.0 * n * max_conv

            if score < best_local_score:
                best_local_score = score
                best_local_a = a.copy()
                # Log occasional improvements
                if it % 5000 == 0:
                    elapsed = time.time() - start
                    print(f"[Restart] iter {it:6d}, best_score {best_local_score:.6f}, elapsed {elapsed:.1f}s")

            # Mirror descent gradient in logits
            mu = np.dot(grad_a, a)
            grad_z = a * (grad_a - mu)

            # Adam update
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * (grad_z ** 2)
            m_hat = m / (1 - beta1 ** it)
            v_hat = v / (1 - beta2 ** it)

            # Step size schedule
            alpha_t = base_alpha / (math.sqrt(it) + 1.0)
            step = - alpha_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            z = z + step

            # Clip to prevent overflow
            z = np.clip(z, -80.0, 80.0)

            eps_factor = max(eps_factor_min, eps_factor * eps_decay)

            # Occasionally jitter
            if it % 10000 == 0 and np.random.rand() < 0.2:
                z += 0.005 * np.random.randn(n)

        return best_local_a, best_local_score

    # Determine n
    try:
        n = len(height_sequence_1)
    except NameError:
        n = 20000

    print(f"[Info] Using length n={n}")

    # Time allocations for restarts
    total_time = deadline - time.time()
    # For simplicity allocate equal time for each restart (maybe 4)
    restarts = 4
    time_per = total_time / restarts

    for r in range(restarts):
        # Choose init based on round
        if r == 0:
            init_a = init_arcsine(n)
            init_name = "arcsine"
        elif r == 1:
            init_a = init_beta(n, 0.35) # more peaked
            init_name = "beta0.35"
        elif r == 2:
            init_a = init_beta(n, 0.45)
            init_name = "beta0.45"
        else:
            init_a = init_random(n)
            init_name = "random"

        print(f"[Restart {r}] init {init_name}")
        # Run optimization with allocated time
        remaining_time = deadline - time.time()
        run_time = min(time_per, remaining_time * 0.9)  # keep buffer

"""
Multi‑restart Adam optimisation on the simplex for the max‑convolution
objective.

The algorithm:
  1. Choose a length `n` (reuse the length of a previously supplied
     `height_sequence_1` when possible, otherwise default to 20 000).
  2. Initialise several candidates:
        – exact arcsine discretisation (the current state‑of‑the‑art);
        – a Beta‑type shape with a stronger singularity at the borders;
        – a simple random start.
  3. For each start run a projected Adam optimiser on the probability
     simplex.  The optimiser works on logits `z = log a` (mirror descent);
     the objective is a smoothed max of the self‑convolution obtained via
     a soft‑max with a decreasing temperature (`eps_factor`).  The gradient
     is computed analytically by FFT (linear‐time per iteration).
  4. Keep the best sequence seen across all restarts.
  5. If time permits, perform a final fine‑tuning run with a very
     small temperature to sharpen the max‑approximation.
  6. Zero out any entries smaller than 1e‑12, re‑normalise and return the
     list.  The platform will assess the returned sequence with
     `evaluate_sequence`.

The method stays within the 1000 s budget, uses only NumPy (no file or
network I/O) and leaves the scale‑invariance to the evaluator (the sum
is normalised to 1 throughout optimisation).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    s = e.sum()
    if s == 0.0 or np.isnan(s):
        # fallback to uniform distribution – should never happen
        return np.full_like(z, 1.0 / z.size)
    return e / s


def _fft_len_for_corr(n: int) -> int:
    """
    Smallest power‑of‑two ≥ 3·n‑2.
    Required length for linear convolution (2·n‑1) and the correlation
    (3·n‑2) used in the gradient computation.
    """
    return 1 << ((3 * n - 2).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a (via FFT)."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Smoothed max of a*a and its gradient w.r.t. a.

    Parameters
    ----------
    a : np.ndarray
        Probability vector (sum = 1, non‑negative).
    eps_factor : float
        Multiplicative factor for the smoothing constant:
        eps = max(eps_factor * max(conv), 1e‑12).

    Returns
    -------
    max_conv : float
        Exact maximum of the linear self‑convolution.
    grad_a : np.ndarray
        Gradient of 2·n·smooth_max w.r.t. a (already multiplied by 4·n).
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    a_fft = np.fft.rfft(a, L)

    # linear self‑convolution (length 2·n‑1)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # soft‑max weights over the convolution entries
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv          # for numerical stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or np.isnan(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # cross‑correlation a ⋆ w (length 3·n‑2, we need first n entries)
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # gradient (the factor 4·n comes from the derivative of 2·n·smooth_max)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a


# ----------------------------------------------------------------------
# Initialisers for different starting shapes
# ----------------------------------------------------------------------


def _init_arcsine(n: int) -> np.ndarray:
    """Exact discretisation of the arcsine CDF (optimal continuous case)."""
    idx = np.arange(n, dtype=np.float64)
    x0 = idx / n
    x1 = (idx + 1) / n

    def arcsine_cdf(x):
        return (2.0 / np.pi) * np.arcsin(np.sqrt(np.clip(x, 0.0, 1.0)))

    a = arcsine_cdf(x1) - arcsine_cdf(x0)
    a_sum = a.sum()
    if a_sum == 0.0 or np.isnan(a_sum):
        raise RuntimeError("Arcsine initialisation failed.")
    return a / a_sum


def _init_beta_power(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta‑type density proportional to
        x^{alpha-1} (1-x)^{alpha-1}
    approximated by a Riemann sum over bin mid‑points.
    """
    mids = (np.arange(n, dtype=np.float64) + 0.5) / n
    # avoid exact 0 or 1 which would give infinities
    eps = 1e-30
    mids = np.clip(mids, eps, 1 - eps)

    w = (mids ** (alpha - 1.0)) * ((1.0 - mids) ** (alpha - 1.0))
    w_sum = w.sum()
    if w_sum == 0.0 or np.isnan(w_sum):
        # fall back to uniform
        return np.full(n, 1.0 / n, dtype=np.float64)
    return w / w_sum


def _init_random(n: int) -> np.ndarray:
    """Uniform‑random Dirichlet(1) start."""
    a = np.random.rand(n)
    a_sum = a.sum()
    return a / a_sum


# ----------------------------------------------------------------------
# Optimisation routine (Adam on the simplex)
# ----------------------------------------------------------------------


def _run_adam_opt(
    init_a: np.ndarray,
    time_budget: float,
    base_alpha: float = 0.8,
    eps_factor_start: float = 0.25,
    eps_factor_min: float = 1e-6,
    eps_decay: float = 0.9993,
    max_iters: int = 500_000,
):
    """
    Perform Adam optimisation on the simplex starting from `init_a`.
    Returns the best vector (probability distribution) found and its
    cheap objective value 2·n·max(conv).
    """
    n = init_a.shape[0]
    # logits for the soft‑max parametrisation
    z = np.log(init_a + 1e-16)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    eps_factor = eps_factor_start

    best_a = init_a.copy()
    best_score = 2.0 * n * _conv_max(init_a)

    start = time.time()
    it = 0

    while it < max_iters and (time.time() - start) < time_budget:
        it += 1

        a = _softmax(z)

        max_conv, grad_a = _conv_max_and_grad(a, eps_factor)

        score = 2.0 * n * max_conv

        if score < best_score:
            best_score = score
            best_a = a.copy()

        # Mirror‑descent gradient on logits (natural gradient on simplex)
        mu = np.dot(grad_a, a)               # 〈∇, a〉
        grad_z = a * (grad_a - mu)           # ∇_z J

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # Inverse‑sqrt decay of the step size
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step = - alpha_t * m_hat / (np.sqrt(v_hat) + eps_adam)

        z = z + step
        # Clamp logits to keep the exponentials numerically safe
        z = np.clip(z, -80.0, 80.0)

        # Reduce smoothing (temperature) gradually
        eps_factor = max(eps_factor_min, eps_factor * eps_decay)

        # Occasionally add a tiny jitter to escape plateaus
        if it % 10000 == 0 and np.random.rand() < 0.15:
            z += 0.005 * np.random.randn(n)

    return best_a, best_score

    fine‑tuning run with a very low temperature is executed if time permits.
    """
    np.random.seed(seed)
    start_time = time.time()

    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # 0. Choose the length of the sequence
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]

    # ------------------------------------------------------------------
    # 2. Run a few Adam restarts, each with a share of the remaining budget
    # ------------------------------------------------------------------
    best_overall_a = None
    best_overall_score = float("inf")

    total_time_left = max(0.0, deadline - time.time())
    # Allocate roughly equal time to each restart, keep a small reserve.
    n_restarts = len(initialisers)
    time_per_restart = max(0.0, total_time_left * 0.85 / n_restarts)

    for name, maker in initialisers:
        print(f"[Restart] initialiser = {name}")

        a0 = maker()
        # Run optimisation for the allocated budget

        if score_opt < best_overall_score:

    # ------------------------------------------------------------------
    # 3. Optional fine‑tuning pass (sharper temperature, smaller steps)
    # ------------------------------------------------------------------
    remaining = max(0.0, deadline - time.time())

        a_opt, score_opt = _run_adam_opt(

            eps_factor_min=1e-7,
            eps_decay=0.9999,

        print(f"[Fine‑tune] refined cheap score = {score_opt:.6f}")
        if score_opt < best_overall_score:
            best_overall_score = score_opt
            best_overall_a = a_opt.copy()

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    s = best_overall_a.sum()
    if s == 0.0 or np.isnan(s):
        # Very unlikely – fall back to uniform
        best_overall_a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        best_overall_a = best_overall_a / s

    final_seq = best_overall_a.tolist()

    # ------------------------------------------------------------------
    # 5. Log the final exact evaluation (optional – the platform will do it)
    # ------------------------------------------------------------------
    try:
        final_eval = evaluate_sequence(final_seq)
        print(f"[Result] Final exact score = {final_eval:.6f}")
    except Exception as exc:  # pragma: no cover
        print(f"[Warning] evaluate_sequence failed: {exc}")

    return final_seq
