"""
Hybrid optimizer for the self‑convolution bound.

The function `propose_candidate` searches a probability vector `a`
(non‑negative, sum ≈ 1) that minimizes

        2·n·max_k (a*a)[k]

where `n = len(a)`.  The evaluation used by the platform is
`evaluate_sequence`, which returns exactly this value when the sum of
the returned list is 1.

The algorithm consists of four stages:

1. **Broad parametric sweep** – a few extra families (cosine‑power,
   sparse random patterns) are added to the original sweep.
2. **Adam on a smooth max** – the same mirror‑descent as before but with
   a more aggressive learning‑rate schedule and a faster decay of the
   smoothing constant.
3. **Coordinate polishing** – cheap random moves with an adaptive step
   size are performed until the time budget expires.
4. **Exact evaluation** – the best sequence found is returned.

The code respects the 1000 s time budget, never writes files, and uses
only NumPy and the already‑imported `evaluate_sequence`.  All helper
functions are top‑level and contain no closures.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (stateless)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max, returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    s = e.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / z.size)
    return e / s


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it that reproduces `a` after a soft‑max (used for mirror‑descent)."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """Smallest power‑of‑two ≥ 3·n‑2 (enough for all needed convolutions)."""
    return 1 << ((3 * n - 2).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a (FFT based)."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Returns (max_conv, grad_a) where

        grad_a = ∇_a [ 2·n·smooth_max(a*a) ]

    `eps_factor` multiplies the smoothing constant ε = eps_factor·max_conv.
    The gradient is obtained via a soft‑max weighting of the convolution.
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv                     # ≤ 0 for stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # Gradient of 2·n·smooth_max (the factor 4·n comes from the derivative of a*a)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a

# Parametric families of distributions

def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}   (normalised)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _power_law_offset_distribution(n: int, p: float, offset: float) -> np.ndarray:
    """a_i ∝ (i+offset)^{-p}   (normalised)."""
    idx = np.arange(offset, offset + n, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _double_power_law_distribution(n: int, p: float, q: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}·(n-i)^{-q}   (normalised)."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) ** (-p)) * ((n - i) ** (-q))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _stretched_exp_distribution(n: int, tau_factor: float, gamma: float) -> np.ndarray:
    """
    a_i ∝ exp( - (i/(tau_factor·n))^{gamma} )   (normalised).
    `tau_factor` ∈ (0,1] stretches the decay length.
    """
    tau = max(1e-12, tau_factor * n)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(- (i / tau) ** gamma)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _exp_decay_distribution(n: int, lam: float) -> np.ndarray:
    """a_i ∝ e^{-lam·i}+e^{-lam·(n‑1‑i)}   (normalised)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _logistic_distribution(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ 1/(1+e^{α·(i‑mid)})   (normalised)."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = 1.0 / (1.0 + np.exp(alpha * (i - mid)))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _sym_product_power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ ((i+1)*(n‑i))^{-p}   (symmetric, normalised)."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) * (n - i)) ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _cosine_power_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ (½·(1+cos(π·i/(n‑1))))^{p}  (peaks at the ends)."""
    i = np.arange(n, dtype=np.float64)
    base = (np.cos(np.pi * i / (n - 1)) + 1.0) * 0.5
    a = base ** p
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _sparse_uniform_distribution(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """k uniformly‑random positions receive equal mass ≈1/k."""
    a = np.zeros(n, dtype=np.float64)
    if k <= 0:
        a[:] = 1.0 / n
        return a
    idx = rng.choice(n, size=min(k, n), replace=False)
    a[idx] = 1.0 / len(idx)
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) improves the
    `evaluate_sequence` score.  The routine respects the supplied time
    budget (~1000 s) and uses a hybrid grid‑search + Adam optimiser on
    the probability simplex, followed by a cheap random walk.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    # small safety margin for the final exact evaluation
    deadline = start + budget_s - 1.0

    # ------------------------------------------------------------------
    # 0. Determine problem size and a fallback initial distribution
    # ------------------------------------------------------------------
    try:
        init_seq = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        n = max(20000, init_seq.shape[0])
        a0 = np.maximum(init_seq, 0.0)
        s = a0.sum()
        if s == 0.0:
            a0 = np.full_like(a0, 1.0 / a0.shape[0])
        else:
            a0 /= s
        print(f"[Init] Using supplied sequence, length = {n}")
    except Exception:
        n = 25000
        a0 = np.full(n, 1.0 / n)
        print(f"[Init] No provided sequence – start from uniform, length = {n}")

    best_seq = a0.copy()
    best_score = 2.0 * n * _conv_max(best_seq)

    # ------------------------------------------------------------------
    # 1. Coarse parametric sweep (including new families)
    # ------------------------------------------------------------------
    families = [
        ("power_law", _power_law_distribution,
         [dict(p=p) for p in np.linspace(0.30, 1.00, 71)]),
        ("power_law_offset", _power_law_offset_distribution,
         [dict(p=p, offset=off)
          for p in np.linspace(0.30, 0.80, 26)
          for off in np.linspace(0.5, 10.0, 21)]),
        ("double_power_law", _double_power_law_distribution,
         [dict(p=p, q=q)
          for p in np.linspace(0.30, 0.90, 13)
          for q in np.linspace(0.30, 0.90, 13)]),
        ("stretched_exp", _stretched_exp_distribution,
         [dict(tau_factor=t, gamma=g)
          for t in np.linspace(0.05, 0.95, 19)
          for g in np.linspace(0.5, 3.0, 19)]),
        ("exp_decay", _exp_decay_distribution,
         [dict(lam=lam) for lam in np.logspace(-7, -2, 35)]),
        ("logistic", _logistic_distribution,
         [dict(alpha=alpha) for alpha in np.linspace(0.01, 10.0, 40)]),
        ("sym_product_power_law", _sym_product_power_law_distribution,
         [dict(p=p) for p in np.linspace(0.30, 1.50, 61)]),
        ("cosine_power", _cosine_power_distribution,
         [dict(p=p) for p in np.linspace(0.5, 5.0, 46)]),
    ]

    print("[Sweep] Starting parametric search")
    for name, generator, param_dicts in families:
        for params in param_dicts:
            if time.time() >= deadline:
                break
            try:
                cand = generator(n, **params)
            except Exception:
                continue
            score = 2.0 * n * _conv_max(cand)
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
        if time.time() >= deadline:
            break

    # Sparse random patterns (few non‑zeros) – these can be surprisingly good.
    sparsity_levels = [int(np.sqrt(n)), int(2 * np.sqrt(n)), n // 20, n // 10]
    print("[Sweep] Sparse random candidates")
    for k in sparsity_levels:
        if time.time() >= deadline:
            break
        cand = _sparse_uniform_distribution(n, k, rng)
        score = 2.0 * n * _conv_max(cand)
        if score < best_score:
            best_score = score
            best_seq = cand.copy()

    print(f"[Sweep] Best cheap score after sweep = {best_score:.8f}")

    # ------------------------------------------------------------------
    # 2. Mirror‑descent (Adam) on the simplex, smoother max
    # ------------------------------------------------------------------
    logits = _logits_from_distribution(best_seq) + 0.03 * rng.standard_normal(n)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    base_alpha = 0.45          # larger initial steps
    eps_factor = 0.20          # start with fairly smooth max
    eps_factor_min = 5e-6
    eps_decay = 0.9995

    report_every = 50000
    it = 0
    last_improve = 0
    best_seq_sofar = best_seq.copy()
    best_score_sofar = best_score

    # Main optimisation loop
    while time.time() < deadline and it < 2_000_000:
        it += 1

        a = _softmax(logits)               # probability vector, sum ≈ 1
        max_conv, grad_a = _conv_max_and_grad(a, eps_factor)
        cheap_score = 2.0 * n * max_conv

        # Track the best seen so far
        if cheap_score < best_score_sofar - 1e-12:
            best_score_sofar = cheap_score
            best_seq_sofar = a.copy()
            last_improve = it
            if it % report_every == 0:
                print(f"[Iter {it:07d}] NEW BEST cheap = {cheap_score:.8f}")

        # Mirror‑descent gradient w.r.t. logits (KL‑geometry)
        mu = np.dot(grad_a, a)
        grad_z = a * (grad_a - mu)

        # Adam updates (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # Step‑size schedule (inverse‑sqrt decay)
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step = - m_hat / (np.sqrt(v_hat) + eps_adam)

        # Simple back‑tracking line‑search (≤ 4 halvings)
        accepted = False
        trial_alpha = alpha_t
        for _ in range(4):
            new_logits = logits + trial_alpha * step
            new_logits = np.clip(new_logits, -60.0, 60.0)
            a_new = _softmax(new_logits)
            new_score = 2.0 * n * _conv_max(a_new)
            if new_score <= cheap_score:
                logits = new_logits
                accepted = True
                break
            trial_alpha *= 0.5
        if not accepted:
            # fall back to a modest step even if not improving
            logits += alpha_t * step
            logits = np.clip(logits, -60.0, 60.0)

        # Random jitter every few thousand iterations to escape plateaus
        if it % 3000 == 0 and rng.random() < 0.12:
            logits += 0.015 * rng.standard_normal(n)
            m *= 0.7
            v *= 0.7

        # Decay smoothing constant
        eps_factor = max(eps_factor_min, eps_factor * eps_decay)

        # Restart if we have not improved for a long while
        if it - last_improve > 300_000:
            print(f"[Iter {it}] Stagnation – restarting from best so far")
            logits = _logits_from_distribution(best_seq_sofar) + 0.04 * rng.standard_normal(n)
            m.fill(0.0)
            v.fill(0.0)
            eps_factor = 0.20
            last_improve = it

    best_seq = best_seq_sofar
    best_score = best_score_sofar
    print(f"[Adam] Finished after {it} iterations, best cheap score = {best_score:.8f}")

    # ------------------------------------------------------------------
    # 3. Final cheap random walk (coordinate transfers)
    # ------------------------------------------------------------------
    a = best_seq.copy()
    cheap_score = best_score

    while time.time() < deadline:
        # Transfer a small amount of mass between two random indices
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        delta = 0.005 * a[i]  # 0.5 % of the source mass
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = 2.0 * n * _conv_max(a_new)
        if new_score < cheap_score:
            a = a_new
            cheap_score = new_score
            best_seq = a.copy()
            best_score = cheap_score
            # occasional progress report
            if rng.random() < 0.001:
                print(f"[Walk] Improved cheap = {cheap_score:.8f}")

        # Occasionally perform a three‑way shuffle (move a chunk of mass)
        if rng.random() < 0.0005:
            idxs = rng.choice(n, size=3, replace=False)
            i, j, k = idxs
            amt = 0.003 * a[i]
            if amt > 0:
                a[i] -= amt
                a[j] += amt / 2
                a[k] += amt / 2
                cheap_score = 2.0 * n * _conv_max(a)
                if cheap_score < best_score:
                    best_seq = a.copy()
                    best_score = cheap_score

    # ------------------------------------------------------------------
    # 4. Final exact evaluation with evaluate_sequence()
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        exact_score = evaluate_sequence(final_seq)  # provided by the platform
        print(f"Final exact score = {exact_score:.8f}")
    except Exception as exc:
        exact_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    return final_seq
