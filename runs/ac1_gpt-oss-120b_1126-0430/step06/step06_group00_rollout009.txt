"""
Optimising a non‑negative step‑function for the metric

    2·n·max(a*a) / (sum(a)^2)

The code works in two stages:

1️⃣  **Variance‑reduction stage** – minimise the L2‑norm of the
    self‑convolution (∑(a∗a)²).  This produces a distribution whose
    autocorrelation is already very flat.
2️⃣  **Smoothed‑max stage** – a soft‑max approximation of the true
    ∞‑norm is optimised with Adam on the probability simplex.
    A small weight of the variance‑gradient is kept to stabilise the
    search.

Both stages use the natural‑gradient / mirror‑descent trick
(gradient on logits = a·(∇_aJ – 〈∇_aJ,a〉)) so the iterate always
stays on the simplex.  The implementation works for any length *n*;
if a previous solution ``height_sequence_1`` is present its length is
re‑used, otherwise a default of 20 000 is taken.

The whole optimisation runs until the supplied time‑budget (≈ 1000 s)
expires and returns the best sequence seen.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max, returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _fft_len_for_corr(n: int) -> int:
    """
    Smallest power of two ≥ 3·n‑2.
    Sufficient for linear convolution (length 2·n‑1) and correlation
    (length 3·n‑2) needed in the gradient computation.
    """
    return 1 << ((3 * n - 2).bit_length())


def _arcsine_init(n: int) -> np.ndarray:
    """
    Initialise a probability vector from the exact discretisation of
    the arcsine distribution on [0, 1] :

        F(x) = (2/π)·arcsin(√x)

    The vector entries are the increments of the CDF, i.e.
    a_i = F((i+1)/n) – F(i/n).
    """
    idx = np.arange(n, dtype=np.float64)
    x0 = idx / n
    x1 = (idx + 1) / n

    # CDF of the arcsine law, safely clipped at the boundaries.
    def arcsine_cdf(x):
        return (2.0 / np.pi) * np.arcsin(np.sqrt(np.clip(x, 0.0, 1.0)))

    a = arcsine_cdf(x1) - arcsine_cdf(x0)
    s = a.sum()
    if s == 0.0 or np.isnan(s):
        raise RuntimeError("Failed to initialise a valid arcsine vector.")
    a /= s
    return a


def _grad_sum_sq(a: np.ndarray, a_fft: np.ndarray, L: int) -> np.ndarray:
    """
    Gradient of  Σ (a*a)²  w.r.t. a.

    Using the identity
        ∇_a Σ (a*a)² = 4 * corr(a , a*a)
    and the FFT representation, the correlation can be obtained as

        irfft( a_fft * |a_fft|² , L )[:n]

    Parameters
    ----------
    a      – current probability vector (length n)
    a_fft  – rFFT of ``a`` with length ``L``
    L      – FFT length (power of two ≥ 3·n‑2)

    Returns
    -------
    grad   – numpy array of shape (n,) containing the gradient.
    """
    # a_fft * |a_fft|² = a_fft * (a_fft * conj(a_fft))
    grad = 4.0 * np.fft.irfft(a_fft * (np.abs(a_fft) ** 2), L)[: a.shape[0]]
    return grad


def _grad_smoothed_max(a: np.ndarray,
                       a_fft: np.ndarray,
                       conv: np.ndarray,
                       eps_factor: float,
                       n: int,
                       L: int) -> (float, np.ndarray):
    """
    Smoothed max of the self‑convolution and its gradient w.r.t. ``a``.
    The smoothing is a soft‑max with temperature ``eps = eps_factor *
    max(conv)``.

    Returns
    -------
    max_conv : float
        The exact maximum of the linear self‑convolution.
    grad_a   : np.ndarray
        Gradient of the *unsclaed* objective 2·n·max(conv) w.r.t. a.
    """
    max_conv = float(conv.max())
    eps = max(max_conv * eps_factor, 1e-12)

    # soft‑max weights over the 2·n‑1 convolution entries
    shifted = conv - max_conv
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or np.isnan(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # correlation a ⋆ w (first n entries)
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # gradient of 2·n·max(conv) (the factor 4·n comes from the derivative)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a

# Main entry point (required by the platform)
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Produce a list of non‑negative heights that (hopefully) achieves an
    evaluation score below the target 1.503.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin for final logging / cleanup
    deadline = start_time + budget_s - 2.0

    # --------------------------------------------------------------
    # 0. Choose the length of the sequence
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)  # type: ignore[name-defined]
        print(f"[Info] Reusing external length: n = {n}")
    except Exception:
        n = 20000
        print(f"[Info] No external length – using default n = {n}")

    L = _fft_len_for_corr(n)          # FFT length used throughout

    # --------------------------------------------------------------
    # 1. Initialise from the exact arcsine CDF (probability vector)
    # --------------------------------------------------------------
    a = _arcsine_init(n)

    # --------------------------------------------------------------
    # 2. Optimisation variables (log‑softmax parametrisation)
    # --------------------------------------------------------------
    # logits such that a = softmax(z)  → always a ≥ 0 and Σa = 1
    z = np.log(a + 1e-16)

    # Adam state
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    # Phase‑split parameters
    phase1_fraction = 0.55                      # proportion of time for variance stage
    phase1_alpha = 1.0                          # base step size during variance stage
    phase2_alpha = 0.6                          # base step size during smoothed‑max stage
    eps_factor = 0.5                            # start temperature for smoothed‑max
    eps_factor_min = 1e-4
    eps_decay = 0.9995

    # Keep the best candidate seen so far
    best_a = a.copy()
    best_score = 2.0 * n * np.max(np.convolve(a, a))
    it = 0
    max_iters = 2_000_000                         # generous upper bound, real stop is time‑driven

    # --------------------------------------------------------------
    # 3. Main optimisation loop
    # --------------------------------------------------------------
    while time.time() < deadline and it < max_iters:
        it += 1

        # ---- current probability vector (softmax of logits) ----
        a_cur = _softmax(z)

        # ---- FFT of the current vector (re‑used for both gradients) ----
        a_fft = np.fft.rfft(a_cur, L)

        # ---- linear self‑convolution (needed for both cheap score and max‑grad) ----
        conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]

        # ---- cheap evaluation of the true metric (no clipping) ----
        cheap_score = 2.0 * n * float(conv.max())
        if cheap_score < best_score:
            best_score = cheap_score
            best_a = a_cur.copy()
            # occasional progress report
            if it % 2000 == 0 or cheap_score < 1.02:
                elapsed = time.time() - start_time
                print(f"[Iter {it:6d}] best_score = {best_score:.6f}  elapsed = {elapsed:.1f}s")

        # ----------------------------------------------------------
        # Decide which phase we are in (based on elapsed time)
        # ----------------------------------------------------------
        now = time.time()
        phase1 = now < start_time + (deadline - start_time) * phase1_fraction

        if phase1:
            # ----------- Phase 1 : minimise Σ(conv)² ----------------
            grad_a = _grad_sum_sq(a_cur, a_fft, L)
            alpha_t = phase1_alpha / (math.sqrt(it) + 1.0)
        else:
            # ----------- Phase 2 : minimise smoothed max -------------
            max_conv, grad_max = _grad_smoothed_max(
                a_cur, a_fft, conv, eps_factor, n, L
            )
            # add a small weight of the variance gradient to keep the shape
            # well‑behaved far from the current max.
            grad_var = _grad_sum_sq(a_cur, a_fft, L)
            grad_a = grad_max + 0.15 * grad_var

            # decay the temperature for the soft‑max
            eps_factor = max(eps_factor_min, eps_factor * eps_decay)

            alpha_t = phase2_alpha / (math.sqrt(it) + 1.0)

        # ----------- Natural‑gradient step on the logits -------------
        mu = np.dot(grad_a, a_cur)                     # 〈∇, a〉
        grad_z = a_cur * (grad_a - mu)                # mirror‑descent gradient

        # Adam bias‑corrected updates
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = - alpha_t * m_hat / (np.sqrt(v_hat) + eps_adam)
        z = z + step

        # keep logits in a numerically safe window
        z = np.clip(z, -80.0, 80.0)

        # occasional random jitter to escape tiny plateaus
        if it % 10000 == 0 and np.random.rand() < 0.08:
            z += 0.01 * np.random.randn(n)

        # ----------------------------------------------------------------
        # (optional) early exit when we are almost out of time
        # ----------------------------------------------------------------
        if time.time() >= deadline:
            break

    # --------------------------------------------------------------
    # 4. Return the best sequence (plain Python list)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()

    # optional final exact evaluation – useful for debugging / logging
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[Info] Final exact score = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover – platform will handle errors
        print(f"[Warning] Evaluation error (ignored): {exc}")

    return final_seq
