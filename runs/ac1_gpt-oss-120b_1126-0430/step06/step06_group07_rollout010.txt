def _run_adam(a0, n, fft_len, deadline, max_iters):
    a = a0.copy()
    best_a = a.copy()
    best_val = cheap_score(a, n, fft_len)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.5  # step size
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        conv = _conv_full_fft(a, fft_len)
        maxb = conv.max()
        cheap = 2.0 * n * maxb
        if cheap < best_val - 1e-12:
            best_val = cheap
            best_a = a.copy()
        kmax = int(conv.argmax())
        # subgradient
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        rng = np.arange(lo, hi+1, dtype=np.int64)
        j = kmax - rng
        g = np.zeros_like(a)
        g[rng] = 2.0 * a[j]
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)

    return best_a, best_val

#!/usr/bin/env python
"""
Search for a non‑negative step function that minimises ``evaluate_sequence``.

Algorithm
---------
1. Build a diverse pool of initial candidates (uniform, previous solution,
   several exponential / power‑law families and random Dirichlet draws).
2. For each candidate run a projected Adam optimiser on the true
   ``max(conv)`` objective.  The exact sub‑gradient of the maximal
   convolution entry is used (including the diagonal 2·a_i term).  The
   optimiser works on the simplex by clipping negatives and renormalising.
   Multiple restarts are performed until the time budget is exhausted.
3. Starting from the best Adam iterate, a simulated‑annealing style
   random‑swap local search on the ordering of the entries is performed
   (the values themselves are fixed).  This often reduces the maximal
   convolution a little further.
4. The final sequence is evaluated with the provided ``evaluate_sequence``
   and returned.

The routine respects a user‑supplied time budget (default 1000 s) and
always returns a feasible sequence.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using a power‑of‑two FFT (real FFT)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.size - 1]


def _cheap_score(a: np.ndarray, n: int, fft_len: int) -> float:
    """Fast proxy for evaluate_sequence when sum(a)=1: 2·n·max(conv)."""
    conv = _conv_full_fft(a, fft_len)
    return 2.0 * n * conv.max()


def _maxconv_subgradient(a: np.ndarray, kmax: int, n: int) -> np.ndarray:
    """
    Exact sub‑gradient of the maximal autocorrelation entry.
    For the maximal index kmax we have
        ∂/∂a_i sum_j a_j a_{kmax-j} = 2·a_{kmax-i},
    with the diagonal case (i=j) also receiving the factor 2.
    """
    lo = max(0, kmax - (n - 1))
    hi = min(kmax, n - 1)
    rng = np.arange(lo, hi + 1, dtype=np.int64)
    j = kmax - rng
    g = np.zeros_like(a)
    g[rng] = 2.0 * a[j]          # includes the diagonal term automatically
    return g


def _run_adam(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimisation of  max(conv)  on the simplex.
    Returns the best vector found and its cheap (2·n·max(conv)) score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _cheap_score(a, n, fft_len)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    lr = 0.2                     # base learning rate
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1

        # convolution and cheap proxy
        conv = _conv_full_fft(a, fft_len)
        maxb = conv.max()
        cheap = 2.0 * n * maxb

        # keep the best seen so far
        if cheap < best_val - 1e-12:
            best_val = cheap
            best_a = a.copy()
            # occasional logging
            if t % 50000 == 0:
                print(f"[Adam] iter {t:07d} new best cheap = {cheap:.6f}")

        # sub‑gradient of the maximal entry
        kmax = int(conv.argmax())
        g = _maxconv_subgradient(a, kmax, n)

        # Adam update (on the simplex)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # projection onto the non‑negative simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s <= 0.0 or not np.isfinite(s):
            a = best_a.copy()
            m.fill(0.0)
            v.fill(0.0)
            continue
        a /= s

    return best_a, best_val


def _swap_local_search(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style random‑swap search on the ordering of the
    entries of ``a0``.  The values themselves stay fixed.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _cheap_score(a, n, fft_len)

    temperature = 1.0
    cooling = 0.99997

    while time.time() < deadline:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        a[i], a[j] = a[j], a[i]

        conv = _conv_full_fft(a, fft_len)
        cheap = 2.0 * n * conv.max()

        accept = False
        if cheap < best_val - 1e-12:
            accept = True
        else:
            delta = cheap - best_val
            prob = math.exp(-delta / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            if cheap < best_val - 1e-12:
                best_val = cheap
                best_a = a.copy()
                print(f"[Swap] NEW BEST cheap = {cheap:.6f}")
        else:
            # revert swap
            a[i], a[j] = a[j], a[i]

        temperature = max(temperature * cooling, 1e-6)

    return best_a, best_val


# ----------------------------------------------------------------------
# Core search routine required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` within the allotted time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the final exact evaluation
    deadline = start_time + budget_s - 5.0
    eps = 1e-30

    # --------------------------------------------------------------
    # Choose a working length.
    # --------------------------------------------------------------
    try:
        prev = height_sequence_1          # may be defined globally by caller
        n = max(len(prev), 5000)
        prev_arr = np.asarray(prev, dtype=np.float64)
        prev_arr = np.maximum(prev_arr, 0.0)
        if prev_arr.sum() > 0.0:
            prev_arr = prev_arr / prev_arr.sum()
        else:
            prev_arr = None
    except Exception:
        prev_arr = None
        n = 5000

    # keep n reasonable for the budget
    n = min(max(n, 5000), 20000)

    # FFT length – next power of two ≥ 2·n‑1 (also sufficient for 3n‑2)
    fft_len = 1 << ((2 * n - 1).bit_length())

    # --------------------------------------------------------------
    # Build a pool of diverse initial candidates.
    # --------------------------------------------------------------
    seeds = []

    # uniform baseline
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    seeds.append(uniform)

    # previous solution, if any
    if prev_arr is not None and prev_arr.size == n:
        seeds.append(prev_arr)

    # exponential (left‑decaying)
    for lam in np.linspace(0.0, 10.0, 9):
        a = np.exp(-lam * np.arange(n, dtype=np.float64))
        a /= a.sum()
        seeds.append(a)

    # symmetric exponential (U‑shaped)
    for lam in np.linspace(0.0, 10.0, 9):
        idx = np.arange(n, dtype=np.float64)
        a = np.exp(-lam * np.minimum(idx, n - 1 - idx))
        a /= a.sum()
        seeds.append(a)

    # power‑law (left)
    for p in np.linspace(0.05, 3.0, 7):
        a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
        a /= a.sum()
        seeds.append(a)

    # symmetric power‑law (U‑shaped)
    for p in np.linspace(0.05, 3.0, 7):
        i = np.arange(n, dtype=np.float64)
        a = np.minimum(i + 1, n - i) ** (-p)
        a /= a.sum()
        seeds.append(a)

    # random Dirichlet seeds with various concentrations
    for conc in [0.001, 0.01, 0.1, 0.5, 1.0, 2.0, 5.0]:
        for _ in range(2):
            g = np.random.gamma(conc, 1.0, size=n)
            a = g / g.sum()
            seeds.append(a)

    # shuffle the pool to avoid systematic ordering
    np.random.shuffle(seeds)

    # --------------------------------------------------------------
    # 0. Quick cheap evaluation of the pool, keep the best.
    # --------------------------------------------------------------
    best_a = None
    best_val = float("inf")
    for a in seeds:
        val = _cheap_score(a, n, fft_len)
        if val < best_val:
            best_val = val
            best_a = a.copy()
    print(f"[Init] best cheap from pool = {best_val:.6f}")

    # --------------------------------------------------------------
    # 1. Multiple Adam restarts (projected gradient) on the simplex.
    # --------------------------------------------------------------
    # We reuse the pool – each vector becomes a start point for Adam.
    for idx, a0 in enumerate(seeds):
        if time.time() >= deadline:
            break
        # allocate a generous iteration budget; the inner loop also checks time.
        a_opt, val_opt = _run_adam(a0, n, fft_len, deadline, max_iters=200_000)
        if val_opt < best_val - 1e-12:
            best_val = val_opt
            best_a = a_opt.copy()
            print(f"[Adam] seed {idx:02d} improved global cheap = {best_val:.6f}")

    # --------------------------------------------------------------
    # 2. Permutation / swap local search (simulated‑annealing style)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _swap_local_search(best_a, n, fft_len, deadline)
        print(f"[Swap] final cheap after swaps = {best_val:.6f}")

    # --------------------------------------------------------------
    # 3. Final exact evaluation and return.
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    return final_seq
