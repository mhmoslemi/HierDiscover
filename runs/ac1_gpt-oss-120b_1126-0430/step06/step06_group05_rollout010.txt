"""
Hybrid optimisation for the self‑convolution L∞‑objective.

The routine consists of three stages:

1. **Smooth‑max Adam** – minimise a differentiable surrogate
   τ·log ∑ exp(conv(a,a)/τ) where τ is annealed from a relatively
   large value to a small one.  The gradient of the surrogate can be
   written as a convolution and is obtained via two FFTs per iteration.
2. **Hard‑max Adam** – switch to the exact sub‑gradient of the maximal
   convolution entry (the code from the reference runs).  This finetunes
   the solution found by stage 1.
3. **Targeted simulated‑annealing** – a cheap local‑search that moves a
   small fraction of mass between two indices.  It removes tiny entries
   and usually yields a small final improvement.

All phases respect the simplex constraint by working in log‑it (exponentiated
gradient) space and projecting onto the simplex via a soft‑max.

The whole search stops after the supplied time budget (≈1000 s) and returns
the best sequence found.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Basic convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full self‑convolution a*a using a real FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return conv_full[: 2 * n - 1]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (maps ℝⁿ → simplex)."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


# ----------------------------------------------------------------------
# Gradient of weighted convolution (used for the smooth‑max surrogate)
# ----------------------------------------------------------------------


def _grad_weighted_conv(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute ∇_a  Σ_k w_k · (a*a)_k  where w ≥ 0, Σw = 1.
    The exact formula is   g_i = 2·∑_j a_j·w_{i+j} .
    Implemented efficiently via an FFT‑based convolution.
    """
    n = a.shape[0]
    # reverse w for convolution
    w_rev = w[::-1]
    L = _next_pow2(n + w_rev.shape[0] - 1)   # = next pow2 of 3n‑2
    af = np.fft.rfft(a, n=L)
    wf = np.fft.rfft(w_rev, n=L)
    conv = np.fft.irfft(af * wf, n=L)        # length = n + (2n‑1)‑1 = 3n‑2
    # slice [n‑1 : 2n‑1) and reverse to obtain g_i/2
    slice_start = n - 1
    slice_end = slice_start + n
    g = 2.0 * conv[slice_start:slice_end][::-1]
    return g


# ----------------------------------------------------------------------
# Hard‑max Adam optimiser (identical to the reference implementation)
# ----------------------------------------------------------------------


def _run_opt_hardmax(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
) -> Tuple[List[float], float]:
    """Adam optimiser using the exact sub‑gradient of the maximal entry."""
    start = time.time()
    rng = np.random.default_rng(seed)

    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
    else:
        a = rng.random(n)

    a = np.clip(a, 0.0, None)
    s = a.sum()
    a = a / s if s > 1e-12 else np.ones(n) / n

    # work in log‑it space
    x = np.log(a + 1e-16)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    lr_start, lr_end = 0.5, 0.001
    best_a = a.copy()
    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())

    it = 0
    while time.time() - start < time_limit:
        it += 1
        a = _softmax(x)

        conv = _conv_self_fft(a)
        M = conv.max()
        obj = float(2.0 * n * M)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[hardmax n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        k_max = int(np.argmax(conv))
        # exact sub‑gradient of the maximal entry
        g_a = np.zeros_like(a)
        idx = np.arange(n)
        j = k_max - idx
        mask = (j >= 0) & (j < n)
        if mask.any():
            g_a[mask] = 2.0 * a[j[mask]]

        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)

        elapsed = time.time() - start
        t = min(elapsed / time_limit, 1.0)
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t)

        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter and restart
        if it % 5000 == 0:
            x += rng.normal(scale=0.01, size=n)
        if it % 80000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.005, size=n)
            m.fill(0.0)
            v.fill(0.0)

    # ------------------------------------------------------------------
    # prune negligible entries
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = float(2.0 * best_a.size * _conv_self_fft(best_a).max())
        print(f"[hardmax compress] reduced length → {best_a.size}, O={best_obj:.6f}")

    return best_a.tolist(), best_obj

# Smooth‑max Adam optimiser
# ----------------------------------------------------------------------


def _run_opt_smoothmax(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
    tau_start: float = 0.2,
    tau_end: float = 0.005,
) -> Tuple[List[float], float]:
    """
    Adam optimiser that minimises the smooth surrogate
        τ·log ∑_k exp(conv(a,a)_k/τ)
    with an exponentially decaying temperature τ.
    Returns the best sequence found and its hard‑max objective.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
    else:
        a = rng.random(n)

    a = np.clip(a, 0.0, None)
    s = a.sum()
    a = a / s if s > 1e-12 else np.ones(n) / n

    x = np.log(a + 1e-16)            # logits
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    lr_start, lr_end = 0.5, 0.001
    best_a = a.copy()
    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())

    it = 0
    while time.time() - start < time_limit:
        it += 1
        a = _softmax(x)

        # self‑convolution
        conv = _conv_self_fft(a)
        M = conv.max()
        obj = float(2.0 * n * M)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[smoothmax n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        # temperature schedule (exponential decay)
        elapsed = time.time() - start
        frac = min(elapsed / time_limit, 1.0)
        tau = tau_start * np.exp(np.log(tau_end / tau_start) * frac)

        # soft‑max weights over convolution entries
        conv_shift = conv - conv.max()            # stabilise exponent
        # safeguard τ against zero (should not happen)
        safe_tau = max(tau, 1e-12)
        exp_vals = np.exp(conv_shift / safe_tau)
        w = exp_vals / exp_vals.sum()            # shape (2n‑1,)

        # gradient of the smooth surrogate
        g_a = _grad_weighted_conv(a, w)

        # mirror‑gradient (log‑it space)
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)

        # learning‑rate schedule (same as hard‑max)
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * frac)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to avoid plateaus
        if it % 4000 == 0:
            x += rng.normal(scale=0.01, size=n)

        # occasional full restart from current best
        if it % 75000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.005, size=n)
            m.fill(0.0)
            v.fill(0.0)

    # ------------------------------------------------------------------
    # prune negligible entries (same as hard‑max)
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = float(2.0 * best_a.size * _conv_self_fft(best_a).max())
        print(f"[smoothmax compress] reduced length → {best_a.size}, O={best_obj:.6f}")

    return best_a.tolist(), best_obj

def _run_sa(
    init_seq: List[float],
    time_limit: float,
    seed: int = 0,
) -> Tuple[List[float], float]:
    """Refine a sequence by small mass‑transfer moves with a Metropolis schedule."""
    start = time.time()
    rng = np.random.default_rng(seed)
    a = np.asarray(init_seq, dtype=float)
    if a.sum() < 1e-12:
        a = np.ones_like(a) / a.size
    else:
        a = a / a.sum()
    n = a.shape[0]

    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())
    best_seq = a.copy()

    T = best_obj * 0.1 + 1e-6
    T_min = best_obj * 1e-6 + 1e-9
    decay = 0.995
    it = 0

    while time.time() - start < time_limit:
        it += 1
        k_max = int(np.argmax(conv))

        i_vals = np.arange(n)
        j_vals = k_max - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i_vals[mask]] * a[j_vals[mask]]
        total_contrib = contrib.sum()

        if total_contrib > 0:
            probs = contrib / total_contrib
            i = rng.choice(n, p=probs)
        else:
            i = rng.integers(n)

        # pick a distinct destination uniformly
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.05  # up to 5 % of a[i]
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.clip(a_new, 0.0, None)
        a_new /= a_new.sum()

        conv_new = _conv_self_fft(a_new)
        obj_new = float(2.0 * n * conv_new.max())

        if obj_new < best_obj - 1e-12:
            a, conv, best_obj, best_seq = a_new, conv_new, obj_new, a_new.copy()
            print(f"[SA it={it}] new best O={best_obj:.6f}")
        else:
            prob = np.exp(-(obj_new - best_obj) / T) if T > 0 else 0.0
            if rng.random() < prob:
                a, conv = a_new, conv_new

        if it % 2000 == 0:
            T = max(T * decay, T_min)

        if it % 10000 == 0:
            noise = rng.normal(scale=0.01, size=n)
            a = np.clip(a + noise, 0.0, None)
            a /= a.sum()
            conv = _conv_self_fft(a)

    # final compression of tiny entries
    thresh = 1e-12
    mask = best_seq > thresh
    if mask.sum() < best_seq.size:
        comp = best_seq[mask]
        comp /= comp.sum()
        comp_obj = float(2.0 * comp.size * _conv_self_fft(comp).max())
        if comp_obj < best_obj:
            best_seq, best_obj = comp, comp_obj
            print(f"[SA compress] reduced length → {comp.size}, O={best_obj:.6f}")

    return best_seq.tolist(), best_obj

        x_new = np.linspace(0.0, 1.0, target_n, endpoint=False)
        init_a = np.interp(x_new, x_old, init_seq)
        init_a = np.clip(init_a, 0.0, None)
        init_a = init_a / init_a.sum()
    else:
        init_a = init_seq

    # ------------------------------------------------------------------
    # Allocate time for the three stages
    # ------------------------------------------------------------------
    now = time.time()
    time_left = deadline - now
    t_smooth = time_left * 0.45          # 45 % for smooth‑max Adam
    t_hard   = time_left * 0.40          # 40 % for hard‑max Adam
    # remaining time will be used for SA (≈15 %)

    # ------------------------------------------------------------------
    # Stage 1 – smooth‑max Adam
    # ------------------------------------------------------------------
    seq_smooth, obj_smooth = _run_opt_smoothmax(
        n=target_n,
        time_limit=t_smooth,
        init_a=init_a,
        seed=seed + 111,
    )
    print(f"[stage‑1] smooth‑max finished O={obj_smooth:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – hard‑max Adam (fine‑tune)
    # ------------------------------------------------------------------
    seq_hard, obj_hard = _run_opt_hardmax(
        n=target_n,
        time_limit=t_hard,
        init_a=np.asarray(seq_smooth),
        seed=seed + 222,
    )
    print(f"[stage‑2] hard‑max finished O={obj_hard:.6f}")

    best_seq = np.asarray(seq_hard, dtype=float)
    best_obj = obj_hard

    # ------------------------------------------------------------------
    # Stage 3 – simulated‑annealing refinement (if time left)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        seq_ref, obj_ref = _run_sa(best_seq.tolist(), time_limit=remaining, seed=seed + 999)
        if obj_ref < best_obj:
            best_seq = np.asarray(seq_ref, dtype=float)
            best_obj = obj_ref
            print(f"[stage‑3] SA improvement O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check via the provided evaluator
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_seq.tolist()
