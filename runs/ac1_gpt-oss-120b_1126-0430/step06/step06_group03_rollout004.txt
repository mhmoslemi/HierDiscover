"""
Hybrid optimisation for the self‑convolution bound – version 2.

Main ideas
----------

* **Exact sub‑gradient** – instead of the soft‑max smoothing the previous
  implementation used, we compute a true sub‑gradient of the max‑convolution
  (the gradient of `max_k (a*a)[k]`).  For a probability vector `a` the
  sub‑gradient can be written in closed form as a sum of two shifted copies
  of the reversed vector; this can be evaluated in O(|S|·n) where `S` is the
  (typically tiny) set of indices where the convolution attains its maximum.

* **Exponentiated‑gradient (Adam) on the simplex** – we keep the convenient
  soft‑max parametrisation (`logits → a`) and apply Adam to the exact
  sub‑gradient.  The update is the same as before, but with a *perfect*
  gradient, which yields a noticeably lower cheap score.

* **Broader parametric sweep** – a few extra families (asymmetric double
  exponential) are tested before the descent starts, providing a better
  initial point.

* **Final cheap random walk** – after the Adam phase a fast coordinate‑wise
  mass‑transfer walk is performed, again using the exact `max(a*a)` for
  acceptance.

The routine respects the 1000 s budget, prints progress and finally calls
`evaluate_sequence` on the best candidate found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max, returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    s = e.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / z.size)
    return e / s


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it that reproduces `a` after a soft‑max."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """Smallest power‑of‑two ≥ 3·n‑2 (enough for all needed convolutions)."""
    return 1 << ((3 * n - 2).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_subgrad(a: np.ndarray):
    """
    Compute both the maximum entry of a*a and a sub‑gradient of the
    max‑convolution with respect to `a`.

    Returns
    -------
    max_val : float
        max_k (a*a)[k]
    grad    : np.ndarray, shape (n,)
        Sub‑gradient of `max_k (a*a)[k]`; satisfies
        `max_val + <grad, d>` is an upper bound on the value after a
        feasible perturbation `d` on the simplex.
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]

    max_val = float(conv.max())
    # indices where the maximum is attained (allow tiny numerical slack)
    eps = max(1e-14 * max_val, 1e-15)
    max_idx = np.where(conv >= max_val - eps)[0]

    # sub‑gradient: 2 * sum_{k in max_idx} a shifted by (n‑1‑k)
    a_rev = a[::-1]
    grad = np.zeros_like(a)

    for k in max_idx:
        offset = n - 1 - k               # could be negative
        if offset >= 0:
            # shift right by `offset`
            grad[: n - offset] += a_rev[offset:]
        else:
            o = -offset
            grad[o:] += a_rev[: n - o]

    grad *= 2.0
    return max_val, grad

# Parametric families of distributions
# ----------------------------------------------------------------------


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}   (normalised)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _power_law_offset_distribution(n: int, p: float, offset: float) -> np.ndarray:
    """a_i ∝ (i+offset)^{-p}   (normalised)."""
    idx = np.arange(offset, offset + n, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _double_power_law_distribution(n: int, p: float, q: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}·(n-i)^{-q}   (normalised)."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) ** (-p)) * ((n - i) ** (-q))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _stretched_exp_distribution(n: int, tau_factor: float, gamma: float) -> np.ndarray:
    """
    a_i ∝ exp( - (i/(tau_factor·n))^{gamma} )   (normalised).
    `tau_factor` ∈ (0,1] stretches the decay length.
    """
    tau = max(1e-12, tau_factor * n)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(- (i / tau) ** gamma)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _exp_decay_distribution(n: int, lam: float) -> np.ndarray:
    """a_i ∝ e^{-lam·i}+e^{-lam·(n‑1‑i)}   (normalised)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _asym_exp_distribution(n: int, lam_left: float, lam_right: float) -> np.ndarray:
    """
    Asymmetric double‑exponential:
      a_i ∝ e^{-lam_left·i} + e^{-lam_right·(n‑1‑i)}.
    Allows the two tails to decay at different rates.
    """
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam_left * i) + np.exp(-lam_right * (n - 1 - i))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _logistic_distribution(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ 1/(1+e^{α·(i‑mid)})   (normalised)."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = 1.0 / (1.0 + np.exp(alpha * (i - mid)))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _sym_product_power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ ((i+1)*(n‑i))^{-p}   (symmetric, normalised)."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) * (n - i)) ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) improves the
    `evaluate_sequence` score.  The routine respects a time budget (≈1000 s)
    and uses:
      1. an expanded parametric sweep,
      2. Adam on the simplex with the *exact* sub‑gradient of the max‑convolution,
      3. a cheap random‑walk fine‑tuner,
    before finally calling ``evaluate_sequence`` on the best candidate.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0   # small safety margin

    # ------------------------------------------------------------------
    # 0. Problem size & fallback initial distribution
    # ------------------------------------------------------------------
    try:
        init_seq = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        n = max(20000, init_seq.shape[0])
        a0 = np.maximum(init_seq, 0.0)
        s = a0.sum()
        if s == 0.0:
            a0 = np.full_like(a0, 1.0 / a0.shape[0])
        else:
            a0 /= s
        print(f"[Init] Using provided sequence, length = {n}")
    except Exception:
        n = 25000
        a0 = np.full(n, 1.0 / n)
        print(f"[Init] No provided sequence – start from uniform, length = {n}")

    best_seq = a0.copy()
    best_score = 2.0 * n * _conv_max(best_seq)
    print(f"[Init] Initial cheap score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1. Coarse parametric sweep (expanded)
    # ------------------------------------------------------------------
    families = [
        ("power_law", _power_law_distribution,
         [dict(p=p) for p in np.linspace(0.30, 0.90, 31)]),
        ("power_law_offset", _power_law_offset_distribution,
         [dict(p=p, offset=off)
          for p in np.linspace(0.30, 0.90, 11)
          for off in np.linspace(0.1, 4.0, 11)]),
        ("double_power_law", _double_power_law_distribution,
         [dict(p=p, q=q)
          for p in np.linspace(0.30, 0.80, 13)
          for q in np.linspace(0.30, 0.80, 13)]),
        ("stretched_exp", _stretched_exp_distribution,
         [dict(tau_factor=t, gamma=g)
          for t in np.linspace(0.05, 0.95, 13)
          for g in np.linspace(0.4, 2.5, 13)]),
        ("exp_decay", _exp_decay_distribution,
         [dict(lam=lam) for lam in np.logspace(-7, -1, 35)]),
        ("asym_exp", _asym_exp_distribution,
         [dict(lam_left=lam1, lam_right=lam2)
          for lam1 in np.logspace(-6, -2, 12)
          for lam2 in np.logspace(-6, -2, 12)]),
        ("logistic", _logistic_distribution,
         [dict(alpha=alpha) for alpha in np.linspace(0.01, 8.0, 30)]),
        ("sym_product_power_law", _sym_product_power_law_distribution,
         [dict(p=p) for p in np.linspace(0.30, 1.30, 31)]),
    ]

    print("[Sweep] Starting expanded parametric search")
    for name, generator, param_dicts in families:
        for params in param_dicts:
            if time.time() >= deadline:
                break
            try:
                cand = generator(n, **params)
            except Exception:
                continue
            cheap = 2.0 * n * _conv_max(cand)
            if cheap < best_score:
                best_score = cheap
                best_seq = cand.copy()
                # occasional log
                if np.random.rand() < 0.03:
                    print(f"[Sweep] New best {name} {params} -> {cheap:.6f}")
        if time.time() >= deadline:
            break
    print(f"[Sweep] Best cheap score after sweep = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Mirror‑descent (Adam) on the simplex with exact sub‑gradient
    # ------------------------------------------------------------------
    # Initialise logits from the best distribution (small jitter added)
    logits = _logits_from_distribution(best_seq) + 0.01 * np.random.randn(n)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    base_alpha = 0.35          # base learning‑rate for logits
    report_every = 20000
    it = 0
    last_improve = 0
    best_seq_sofar = best_seq.copy()
    best_score_sofar = best_score

    while time.time() < deadline and it < 1_200_000:
        it += 1

        a = _softmax(logits)                     # probability vector, sum≈1
        max_conv, grad_a = _conv_max_and_subgrad(a)
        cheap_score = 2.0 * n * max_conv

        # keep the best ever seen
        if cheap_score < best_score_sofar - 1e-13:
            best_score_sofar = cheap_score
            best_seq_sofar = a.copy()
            last_improve = it
            if it % report_every == 0:
                print(f"[Iter {it:07d}] NEW BEST cheap = {cheap_score:.8f}")

        # Mirror‑descent gradient w.r.t. logits (natural gradient on simplex)
        mu = np.dot(grad_a, a)
        grad_z = a * (grad_a - mu)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step‑size schedule (inverse‑sqrt decay)
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step = - m_hat / (np.sqrt(v_hat) + eps_adam)

        # back‑tracking line‑search (up to 4 halvings)
        accepted = False
        trial_alpha = alpha_t
        for _ in range(4):
            new_logits = logits + trial_alpha * step
            new_logits = np.clip(new_logits, -50.0, 50.0)
            a_new = _softmax(new_logits)
            new_score = 2.0 * n * _conv_max(a_new)
            if new_score <= cheap_score:
                logits = new_logits
                accepted = True
                break
            trial_alpha *= 0.5
        if not accepted:
            # still move, but with the original step
            logits = logits + alpha_t * step
            logits = np.clip(logits, -50.0, 50.0)

        # occasional random jitter to escape plateaus
        if it % 5000 == 0 and np.random.rand() < 0.12:
            logits += 0.02 * np.random.randn(n)
            m *= 0.6
            v *= 0.6

        # restart if stuck for a long time
        if it - last_improve > 300_000:
            print(f"[Iter {it}] Stagnation – restarting from best so far")
            logits = _logits_from_distribution(best_seq_sofar) + 0.025 * np.random.randn(n)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = it

        if it % 50000 == 0:
            elapsed = time.time() - start
            print(f"[Iter {it}] cheap = {cheap_score:.6f}  elapsed = {elapsed:.1f}s")

    # store the best seen after the Adam phase
    best_seq = best_seq_sofar
    best_score = best_score_sofar
    print(f"[Adam] Finished after {it} iterations, best cheap = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Final cheap random walk (coordinate mass transfers)
    # ------------------------------------------------------------------
    a = best_seq.copy()
    cheap_score = best_score

    while time.time() < deadline:
        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        if i == j:
            continue
        # transfer a small fraction of the mass at i to j
        delta = 0.001 * a[i]
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = 2.0 * n * _conv_max(a_new)
        if new_score < cheap_score:
            a = a_new
            cheap_score = new_score
            best_seq = a.copy()
            best_score = cheap_score
            if np.random.rand() < 0.002:
                print(f"[Walk] improved cheap = {cheap_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Final exact evaluation with evaluate_sequence()
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        exact_score = evaluate_sequence(final_seq)  # provided by the platform
        print(f"Final exact score = {exact_score:.6f}")
    except Exception as exc:
        exact_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    return final_seq
