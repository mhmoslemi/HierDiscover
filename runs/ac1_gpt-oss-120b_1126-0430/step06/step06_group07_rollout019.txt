def multiactive_gradient_descent(a, max_iters, deadline):
    best_a = a.copy()
    best_val = cheap_score(a)
    last_improve = 0
    for iter in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        conv = _conv_full_fft(a, fft_len)
        maxb = conv.max()
        cheap = 2.0 * n * maxb
        if cheap < best_val - 1e-12:
            best_val = cheap; best_a = a.copy(); last_improve = iter
            print(f"Iter {iter}: new best cheap = {cheap:.6f}")
        # compute active indices: top K
        K = 3
        if K >= conv.size:
            active = np.arange(conv.size)
        else:
            # use argpartition
            idx_part = np.argpartition(-conv, K-1)[:K]
            active = idx_part
        # compute gradient
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            i = np.arange(lo, hi + 1)
            j = k - i
            g[i] += 2.0 * a[j]
        # exponentiated gradient update
        eta = eta0 / math.sqrt(iter)
        log_a = np.log(a + eps)
        log_a -= eta * g
        a = np.exp(log_a)
        # renorm
        sum_a = a.sum()
        if sum_a <= 0 or not np.isfinite(sum_a):
            a = best_a.copy()
            continue
        a /= sum_a

        # restart if no improvement for many iterations
        if iter - last_improve > stagnation_limit:
            perturb = np.random.rand(n) * 0.01
            a = _simplex_projection(best_a + perturb)
            last_improve = iter
            print(f"Iter {iter}: Restart after stagnation.")
    return best_a, best_val

#!/usr/bin/env python
"""
Propose a non‑negative step function (list of heights) that minimises
``evaluate_sequence``.

Algorithm
---------

* **1. Warm‑start.**  Sweep a few parametric families (exponential,
  symmetric exponential, power law, β‑distribution, Gaussian, triangular,
  ramp) and keep the best cheap score (2·n·max(conv) with Σa=1).

* **2. Multi‑active mirror descent.**  Repeatedly perform exponentiated‑
  gradient updates on the simplex.  Instead of using only the single
  maximal convolution entry we take the *K* largest entries (K≈3–5) and
  sum their exact sub‑gradients.  This gives a stronger descent direction
  and avoids getting stuck at a local saddle.

* **3. Random restarts.**  Whenever progress stalls we perturb the best
  iterate and restart the mirror‑descent; we also occasionally launch a
  fresh Dirichlet seed to explore different basins.

* **4. Local refinement.**  Two cheap neighbourhood searches are applied
  while time permits:
      – random swaps of two positions (permuting the multiset of heights);
      – random mass transfers between two positions (changing the values
        while preserving the ℓ₁‑norm).

* **5. Final evaluation.**  The best candidate found is evaluated with the
  provided ``evaluate_sequence`` and returned.

All heavy work uses FFT‑based convolution (O(n log n)).  The routine
respects the supplied time budget (default 1000 s) and always returns a
feasible sequence.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using a power‑of‑two FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa).real[: 2 * a.size - 1]
    return conv


def _conv_max_fft(a: np.ndarray, fft_len: int) -> tuple[float, int]:
    """Maximum value of the convolution a*a and its index."""
    conv = _conv_full_fft(a, fft_len)
    idx = int(conv.argmax())
    return float(conv[idx]), idx


# ----------------------------------------------------------------------
# Core search routine required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that
    minimises ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the final exact evaluation
    deadline = start_time + budget_s - 5.0
    eps = 1e-30

    # --------------------------------------------------------------
    # Determine a working length.
    # --------------------------------------------------------------
    try:
        prev = height_sequence_1               # may be defined globally
        n = max(len(prev), 5000)
        prev_arr = np.asarray(prev, dtype=np.float64)
        if prev_arr.sum() > 0.0:
            prev_arr = prev_arr / prev_arr.sum()
        else:
            prev_arr = None
    except Exception:
        prev_arr = None
        n = 5000

    n = int(min(max(n, 5000), 20000))           # keep it reasonable
    fft_len = 1 << ((2 * n - 1).bit_length())   # next power‑of‑two ≥ 2·n‑1

    # --------------------------------------------------------------
    # Cheap score (assumes Σa_i = 1): 2·n·max(conv)
    # --------------------------------------------------------------
    def cheap_score(vec: np.ndarray) -> float:
        maxb, _ = _conv_max_fft(vec, fft_len)
        return 2.0 * n * maxb

    # --------------------------------------------------------------
    # 0. Initialise with cheap families
    # --------------------------------------------------------------
    best_a = None
    best_val = float("inf")

    # uniform baseline
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    val = cheap_score(uniform)
    best_a, best_val = uniform, val

    # previous construction, if any and same length
    if prev_arr is not None and prev_arr.size == n:
        val = cheap_score(prev_arr)
        if val < best_val:
            best_a, best_val = prev_arr.copy(), val

    # ----- exponential (left) -------------------------------------------------
    for lam in np.linspace(0.0, 10.0, 41):
        if time.time() >= deadline:
            break
        a = np.exp(-lam * np.arange(n, dtype=np.float64))
        a /= a.sum()
        val = cheap_score(a)
        if val < best_val:
            best_a, best_val = a.copy(), val

    # ----- symmetric exponential (U‑shaped) ----------------------------------
    for lam in np.linspace(0.0, 10.0, 41):
        if time.time() >= deadline:
            break
        idx = np.arange(n, dtype=np.float64)
        a = np.exp(-lam * np.minimum(idx, n - 1 - idx))
        a /= a.sum()
        val = cheap_score(a)
        if val < best_val:
            best_a, best_val = a.copy(), val

    # ----- power‑law (left) ----------------------------------------------------
    for p in np.linspace(0.01, 3.0, 61):
        if time.time() >= deadline:
            break
        a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
        a /= a.sum()
        val = cheap_score(a)
        if val < best_val:
            best_a, best_val = a.copy(), val

    # ----- symmetric power‑law (U‑shaped) ------------------------------------
    for p in np.linspace(0.01, 3.0, 61):
        if time.time() >= deadline:
            break
        i = np.arange(n, dtype=np.float64)
        a = np.minimum(i + 1, n - i) ** (-p)
        a /= a.sum()
        val = cheap_score(a)
        if val < best_val:
            best_a, best_val = a.copy(), val

    # ----- beta families (U‑shaped when α,β<1) -------------------------------
    x = (np.arange(1, n + 1, dtype=np.float64) / n)
    alphas = np.concatenate((
        np.linspace(0.01, 0.2, 5),
        np.linspace(0.2, 1.0, 9),
        np.linspace(1.0, 5.0, 5)
    ))
    betas = alphas.copy()
    for alpha in alphas:
        for beta in betas:
            if time.time() >= deadline:
                break
            a = (x ** alpha) * ((1.0 - x) ** beta)
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = cheap_score(a)
            if val < best_val:
                best_a, best_val = a.copy(), val

    # ----- Gaussian (centered) ------------------------------------------------
    centres = [n // 2]
    sigmas = np.linspace(0.2 * n, 0.5 * n, 7)
    for sigma in sigmas:
        if time.time() >= deadline:
            break
        a = np.exp(-0.5 * ((np.arange(n) - centres[0]) / sigma) ** 2)
        a /= a.sum()
        val = cheap_score(a)
        if val < best_val:
            best_a, best_val = a.copy(), val

    # ----- Triangular (peak at centre) ---------------------------------------
    a = 1.0 - np.abs(np.arange(n) - (n - 1) / 2) / ((n - 1) / 2)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    val = cheap_score(a)
    if val < best_val:
        best_a, best_val = a.copy(), val

    # ----- Linear ramp (decreasing) -------------------------------------------
    a = (np.arange(n, 0, -1, dtype=np.float64))
    a /= a.sum()
    val = cheap_score(a)
    if val < best_val:
        best_a, best_val = a.copy(), val

    # ----- random Dirichlet seeds ---------------------------------------------
    for conc in [0.001, 0.01, 0.1, 0.5, 1.0, 2.0, 5.0]:
        if time.time() >= deadline:
            break
        g = np.random.gamma(conc, 1.0, size=n)
        a = g / g.sum()
        val = cheap_score(a)
        if val < best_val:
            best_a, best_val = a.copy(), val

    print(f"[Init] best cheap = {best_val:.6f}")

    # --------------------------------------------------------------
    # 1. Multi‑active exponentiated‑gradient mirror descent
    # --------------------------------------------------------------
    a = best_a.copy()
    best_iter_a = a.copy()
    best_iter_val = best_val
    last_improve = 0
    iter_cnt = 0
    eta0 = 0.7
    stagnation_limit = 250_000
    max_iters = 2_000_000

    while time.time() < deadline and iter_cnt < max_iters:
        iter_cnt += 1

        conv = _conv_full_fft(a, fft_len)
        maxb = conv.max()
        cheap = 2.0 * n * maxb

        if cheap < best_iter_val - 1e-12:
            best_iter_val = cheap
            best_iter_a = a.copy()
            last_improve = iter_cnt
            print(f"[Iter {iter_cnt:07d}] NEW BEST cheap = {cheap:.6f}")

        # ----- multi‑active sub‑gradient (top‑K largest convolution entries) -----
        K = 4                                            # number of active shifts
        if K >= conv.size:
            active = np.arange(conv.size)
        else:
            # get indices of the K largest values
            inds = np.argpartition(-conv, K - 1)[:K]
            active = inds

        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            i = np.arange(lo, hi + 1, dtype=np.int64)
            j = k - i
            # exact sub‑gradient: for each i add 2*a[j]
            g[i] += 2.0 * a[j]

        eta = eta0 / math.sqrt(iter_cnt)
        log_a = np.log(a + eps)
        log_a -= eta * g
        a = np.exp(log_a)
        sum_a = a.sum()
        if sum_a <= 0.0 or not np.isfinite(sum_a):
            a = best_iter_a.copy()
            continue
        a /= sum_a

        # occasional restart if we have stalled for a while
        if iter_cnt - last_improve > stagnation_limit:
            # perturb current best a a little and project back onto the simplex
            perturb = np.random.rand(n) * 0.01
            a = _simplex_projection(best_iter_a + perturb)
            last_improve = iter_cnt
            print(f"[Iter {iter_cnt:07d}] Restart after stagnation.")

    # --------------------------------------------------------------
    # 2. Permutation‑local search (swap) and height‑transfer search
    # --------------------------------------------------------------
    a = best_iter_a.copy()
    best_swap_a = a.copy()
    best_swap_val = best_iter_val
    temperature = 1.0
    cooling = 0.99997

    while time.time() < deadline:
        # ---- random swap of two positions ---------------------------------
        i, j = np.random.randint(0, n, size=2)
        if i != j:
            a[i], a[j] = a[j], a[i]

            conv = _conv_full_fft(a, fft_len)
            cheap = 2.0 * n * conv.max()
            accept = False
            if cheap < best_swap_val - 1e-12:
                accept = True
            else:
                delta = cheap - best_swap_val
                prob = math.exp(-delta / max(temperature, 1e-12))
                if np.random.rand() < prob:
                    accept = True

            if accept:
                if cheap < best_swap_val - 1e-12:
                    best_swap_val = cheap
                    best_swap_a = a.copy()
                    print(f"[Swap] NEW BEST cheap = {cheap:.6f}")
            else:
                # revert the swap
                a[i], a[j] = a[j], a[i]

        # ---- random mass transfer between two positions --------------------
        i, j = np.random.randint(0, n, size=2)
        if i != j:
            # choose a transfer amount δ ∈ [‑a_i, a_j]
            delta = (np.random.rand() - 0.5) * 2.0 * min(a[i], a[j]) * 0.1
            # ensure non‑negativity
            delta = max(-a[i], min(delta, a[j]))
            a[i] -= delta
            a[j] += delta

            conv = _conv_full_fft(a, fft_len)
            cheap = 2.0 * n * conv.max()
            accept = False
            if cheap < best_swap_val - 1e-12:
                accept = True
            else:
                delta_val = cheap - best_swap_val
                prob = math.exp(-delta_val / max(temperature, 1e-12))
                if np.random.rand() < prob:
                    accept = True

            if accept:
                if cheap < best_swap_val - 1e-12:
                    best_swap_val = cheap
                    best_swap_a = a.copy()
                    print(f"[Transfer] NEW BEST cheap = {cheap:.6f}")
            else:
                # revert the transfer
                a[i] += delta
                a[j] -= delta

        temperature = max(temperature * cooling, 1e-6)

    # --------------------------------------------------------------
    # 3. Final exact evaluation
    # --------------------------------------------------------------
    final_seq = best_swap_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:  # defensive – should never happen
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    return final_seq
