"""
Improved non‑negative step‑function optimiser.

Algorithm
---------

1. **Shape pre‑search** – generate many analytic candidates
   (uniform, exponential, Gaussian, power‑law, sparse Dirichlet).
   Pick the best by the exact objective.

2. **Adam on the simplex** – optimise the soft‑max (log‑it) parametrisation
   of the probability simplex.  The smooth‑max of the autocorrelation
   is controlled by an exponential temperature schedule (β grows from 0.5
   to 1e7).  Adam hyper‑parameters are tuned for a faster decay of the
   learning rate.

3. **Deterministic projected‑gradient descent** – after Adam a long
   projected‑gradient phase runs on the *exact* max‑convolution (hard
   max, no smoothing).  A back‑tracking line‑search guarantees descent
   and the step‑size is adapted automatically.  Periodic jitter and
   occasional random restarts avoid stagnation.

The routine tries several support lengths (2000 … 8000) and allocates the
remaining time budget dynamically.  The best sequence found is returned
as a plain Python list ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return conv_full[: 2 * n - 1]


def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation of ``a`` with weight vector ``w``.
    Implements g[i] = Σ_t w[i+t]·a[t]  (real FFT).
    """
    n = a.shape[0]
    m = w.shape[0]                     # = 2n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Stable soft‑max."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) assuming Σ a = 1.
    """
    n = a.size
    conv = _conv_self_fft(a)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σ x = 1}.
    """
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Stage 1 – candidate shapes
# ----------------------------------------------------------------------


def _shape_candidates(n: int, rng: np.random.Generator):
    """Create a diverse pool of candidate sequences (sum to 1)."""
    cand = []

    # Uniform
    cand.append(np.full(n, 1.0 / n))

    # Exponential decays (forward & reversed)
    r_vals = np.linspace(0.90, 0.9997, 20)
    idx = np.arange(n, dtype=float)
    for r in r_vals:
        a = r ** idx
        cand.append(a / a.sum())
        cand.append(a[::-1] / a.sum())

    # Gaussian windows
    centre = (n - 1) / 2.0
    sigma_vals = np.linspace(0.08 * n, 0.35 * n, 12)
    for sigma in sigma_vals:
        a = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
        cand.append(a / a.sum())

    # Power‑law (inverse) – concentrates mass near left side
    p_vals = np.linspace(0.5, 2.5, 10)
    for p in p_vals:
        a = 1.0 / (idx + 1) ** p
        cand.append(a / a.sum())

    # Sparse Dirichlet‑like – few non‑zero entries
    k = max(2, int(0.02 * n))
    for _ in range(6):
        a = np.zeros(n, dtype=float)
        pos = rng.choice(n, size=k, replace=False)
        vals = rng.random(k)
        a[pos] = vals
        cand.append(a / a.sum())

    # Pure Dirichlet (α=1)
    a = rng.random(n)
    cand.append(a / a.sum())

    # Alpha‑skewed Dirichlet (α<1 → sparser)
    for alpha in [0.3, 0.5, 0.7]:
        a = rng.gamma(alpha, 1.0, size=n)
        cand.append(a / a.sum())

    return cand


def _best_initial_sequence(n: int, rng: np.random.Generator, time_limit: float):
    """Select the best candidate from the analytical pool."""
    start = time.time()
    best_a = None
    best_obj = float("inf")
    for a in _shape_candidates(n, rng):
        if time.time() - start > time_limit:
            break
        obj = _objective(a)
        if obj < best_obj:
            best_obj, best_a = obj, a
    if best_a is None:
        a = rng.random(n)
        best_a = a / a.sum()
        best_obj = _objective(best_a)
    return best_a, best_obj

def _run_adam(
    n: int,
    time_limit: float,
    init_a: np.ndarray,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the log‑it simplex.
    Returns the best sequence (numpy array) and its exact objective.
    """
    start = time.time()
    a = np.asarray(init_a, dtype=float)
    if a.size != n:
        a = rng.random(n)
    a = a / a.sum()

    # log‑it parameters
    x = np.log(a + 1e-16)
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # schedules (time‑based)
    beta_start, beta_end = 0.5, 1e7
    lr_start, lr_end = 0.30, 0.0002

    best_a = a.copy()
    best_obj = _objective(best_a)

    it = 0
    while time.time() - start < time_limit:
        it += 1
        a = _softmax(x)

        conv = _conv_self_fft(a)
        M = conv.max()
        obj = float(2.0 * n * M)

        if obj < best_obj - 1e-14:
            best_obj, best_a = obj, a.copy()
            print(f"[n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        t = min((time.time() - start) / time_limit, 1.0)
        beta = beta_start * np.exp(np.log(beta_end / beta_start) * t)
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t)

        if beta > 5e5:          # hard max
            w = np.zeros_like(conv)
            w[int(np.argmax(conv))] = 1.0
        else:
            shifted = conv - M
            w = np.exp(beta * shifted)
            s = w.sum()
            if s == 0.0:
                w = np.ones_like(conv) / conv.size
            else:
                w /= s

        g_a = 2.0 * _correlation_fft(w, a)
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)

        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if it % 2500 == 0:
            jitter = rng.normal(scale=0.012, size=n)
            x += jitter

        if it % 50000 == 0:
            # restart from best with small noise
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.008, size=n)
            m.fill(0.0)
            v.fill(0.0)

    # prune negligible entries (helps later stages)
    thresh = 1e-13
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = _objective(best_a)
        print(f"[compress] new length {best_a.size}, O={best_obj:.6f}")

    return best_a, best_obj


# ----------------------------------------------------------------------
# Stage 3 – Deterministic Projected Gradient Descent (exact max)
# ----------------------------------------------------------------------


def _run_pg_exact(
    init_seq: np.ndarray,
    time_limit: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Projected‑gradient descent on the exact max‑convolution.
    Uses a back‑tracking line‑search and occasional random restarts.
    Returns the best sequence found and its exact objective.
    """
    start = time.time()
    a = init_seq.copy()
    n = a.size
    best_a = a.copy()
    best_obj = _objective(best_a)

    # initial step size
    eta = 0.5
    it = 0

    while time.time() - start < time_limit:
        it += 1
        conv = _conv_self_fft(a)
        k = int(np.argmax(conv))
        # gradient of the hard max
        w = np.zeros_like(conv)
        w[k] = 1.0
        grad = 2.0 * _correlation_fft(w, a)

        # back‑tracking line search
        improved = False
        eta_try = eta
        for _ in range(7):
            a_new = a - eta_try * grad
            a_new = _project_simplex(a_new)
            obj_new = _objective(a_new)
            if obj_new < best_obj - 1e-14:
                best_a, best_obj = a_new, obj_new
                a = a_new
                improved = True
                # slightly increase step size for next iteration
                eta = min(eta_try * 1.1, 1.0)
                break
            eta_try *= 0.5
        if not improved:
            # reduce step size aggressively
            eta = max(eta * 0.5, 1e-8)
            if eta < 1e-8:
                # random restart if stuck
                a = rng.dirichlet(np.ones(n))
                a = a / a.sum()
                eta = 0.5
                continue

        # occasional jitter to shake out shallow basins
        if it % 800 == 0:
            jitter = rng.normal(scale=0.004, size=n)
            a = _project_simplex(a + jitter)

    return best_a, best_obj


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **_ignore) -> list[float]:
    """
    Optimisation driver – explores several support lengths,
    runs Adam followed by a long exact projected‑gradient phase,
    and returns the best non‑negative step‑function found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.4  # safety margin

    # support lengths to explore (larger lengths give a better theoretical bound)
    lengths = [2000, 4000, 6000, 8000]

    best_seq = None
    best_obj = float("inf")

    for idx, n in enumerate(lengths):
        now = time.time()
        remaining = deadline - now
        if remaining < 5.0:
            break

        # simple equal split of the remaining budget among the lengths still to try
        lengths_left = len(lengths) - idx
        per_len_budget = remaining / lengths_left

        # ------------------------------------------------------------------
        # Stage 1 – shape pre‑search (quick)
        shape_budget = min(4.0, per_len_budget * 0.07)
        init_a, init_obj = _best_initial_sequence(n, rng, shape_budget)
        print(f"[init n={n}] shape‑obj={init_obj:.6f}")

        # ------------------------------------------------------------------
        # Stage 2 – Adam optimisation
        adam_budget = per_len_budget * 0.55
        seq_adam, obj_adam = _run_adam(
            n=n,
            time_limit=adam_budget,
            init_a=init_a,
            rng=rng,
        )
        if obj_adam < best_obj:
            best_obj, best_seq = obj_adam, seq_adam.copy()
            print(f"[global] new overall best O={best_obj:.6f} (n={n})")

        # ------------------------------------------------------------------
        # Stage 3 – exact projected‑gradient descent (major portion of left time)
        pg_budget = per_len_budget * 0.35
        seq_pg, obj_pg = _run_pg_exact(
            init_seq=seq_adam,
            time_limit=pg_budget,
            rng=rng,
        )
        if obj_pg < best_obj:
            best_obj, best_seq = obj_pg, seq_pg.copy()
            print(f"[global] new overall best after PG O={best_obj:.6f} (n={n})")

        if time.time() >= deadline - 0.5:
            break

    # ----------------------------------------------------------------------
    # Final sanity check (diagnostic only)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    # Clamp to evaluator limits and avoid too‑small sums
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    if best_seq.sum() < 0.01:
        best_seq = np.ones_like(best_seq) / best_seq.size

    return best_seq.tolist()
