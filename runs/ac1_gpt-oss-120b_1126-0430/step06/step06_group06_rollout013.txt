"""
Propose a non‑negative step function (list of heights) that minimises
    2·n·max_k (a*a)[k] / (Σ a_i)²

Algorithm
---------
1. Choose a relatively large support size (default 20 000).  Larger n
   gives a lower theoretical bound (≈1) because the average of the
   self‑convolution is 1/(2n‑1).  The memory and time cost are still
   modest (FFT length ≤ 2²⁶).
2. Scan several low‑dimensional parametric families (uniform, power‑law,
   exponential, mixed, beta, Dirichlet, simple ramps).  The cheapest
   candidate is kept.
3. Refine the best candidate with a *projected sub‑gradient descent* on
   the true non‑smooth objective.  The sub‑gradient of
   max_k (a*a)[k] is obtained from the active convolution indices.
   After each step the vector is projected onto the probability simplex
   (non‑negative, sum = 1).
4. Perform a short (1+1)‑Evolution Strategy hill‑climb on the cheap
   objective to escape possible plateaus.  Mutations are multiplicative
   in log‑space and the mutation strength is adapted with the
   1/5‑success rule.
5. Return the best sequence found; it is guaranteed to be feasible
   (all entries ≥0, sum > 0.01).  The routine respects the 1000 s time
   budget.

The code contains no closures or lambda expressions and only uses
standard scientific libraries (numpy, math, time).  It can be run as
is – the platform will provide ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (L1‑normalisation to sum = 1)
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities – FFT based (real input)
# ----------------------------------------------------------------------
def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution (a*a) using a real FFT, length 2·n‑1."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.size - 1]


def _conv_max_fft(a: np.ndarray, fft_len: int) -> float:
    """Maximum coefficient of a*a."""
    conv = _conv_full_fft(a, fft_len)
    return float(conv.max())


# ----------------------------------------------------------------------
# Cheap objective (assumes Σa_i = 1): 2·n·max(conv)
# ----------------------------------------------------------------------
def _cheap_score(a: np.ndarray, n: int, fft_len: int) -> float:
    maxb = _conv_max_fft(a, fft_len)
    return 2.0 * n * maxb


# ----------------------------------------------------------------------
# Parametric families (all return a normalised probability vector)
# ----------------------------------------------------------------------
def _uniform_seq(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _power_law_seq(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = i ** (-p)
    seq /= seq.sum()
    return seq


def _exponential_seq(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    seq = np.exp(-lam * i)
    seq /= seq.sum()
    return seq


def _power_law_exp_seq(n: int, p: float, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    seq = (i + 1.0) ** (-p) * np.exp(-lam * i)
    seq /= seq.sum()
    return seq


def _stretched_exponential_seq(n: int, lam: float, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    seq = np.exp(-lam * (i ** gamma))
    seq /= seq.sum()
    return seq


def _beta_seq(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(1, n + 1, dtype=np.float64) / n)
    seq = (x ** alpha) * ((1.0 - x) ** beta)
    s = seq.sum()
    if s == 0.0:
        return None
    seq /= s
    return seq


def _dirichlet_seq(n: int, conc: float) -> np.ndarray:
    g = np.random.gamma(conc, 1.0, size=n)
    seq = g / g.sum()
    return seq


def _linear_ramp_seq(n: int, increasing: bool = True) -> np.ndarray:
    """Simple arithmetic progression, normalised."""
    if increasing:
        i = np.arange(1, n + 1, dtype=np.float64)
    else:
        i = np.arange(n, 0, -1, dtype=np.float64)
    seq = i
    seq /= seq.sum()
    return seq
