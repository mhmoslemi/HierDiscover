#!/usr/bin/env python
"""
Propose a non‑negative step function minimizing

        2·n·max_k (a*a)[k] / (Σ a_i)²

using a smooth soft‑max approximation of the max and an Adam‐type
gradient descent on the simplex (exponentiated‑gradient after each
Adam step).  The algorithm works entirely in the probability simplex
(Σa_i=1) so the final objective equals 2·n·max(conv).  It starts from a
few simple parametric families (including the previous best if
available), runs a fast FFT‑based gradient loop with adaptive learning
rate, and finishes with a lightweight (1+1)‑ES hill‑climb.  All
operations respect the 1000 s budget and keep the sequence non‑negative
and normalised.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (mirrored gradient step target set)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        # all non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT based convolution utilities
# ----------------------------------------------------------------------


def _conv_full_fft(a: np.ndarray, b: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full linear convolution of real vectors a and b using a real FFT.
    Returns the first len(a)+len(b)-1 entries of the linear convolution.
    """
    fa = np.fft.rfft(a, fft_len)
    fb = np.fft.rfft(b, fft_len)
    conv = np.fft.irfft(fa * fb, fft_len).real
    return conv[: a.shape[0] + b.shape[0] - 1]


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Convenience wrapper for a*a."""
    return _conv_full_fft(a, a, fft_len)


# ----------------------------------------------------------------------
# Parametric families (return a normalised probability vector)
# ----------------------------------------------------------------------


def _uniform_seq(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exponential_seq(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    seq = np.exp(-lam * i)
    seq /= seq.sum()
    return seq


def _power_law_seq(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = i ** (-p)
    seq /= seq.sum()
    return seq


def _dirichlet_seq(n: int, conc: float) -> np.ndarray:
    g = np.random.gamma(conc, 1.0, size=n)
    seq = g / g.sum()
    return seq


# ----------------------------------------------------------------------
# Soft‑max (log‑sum‑exp) weight computation
# ----------------------------------------------------------------------


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Return w_k = exp(alpha*conv_k) / Σexp(alpha*conv) in a numerically stable way."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    return exp_scaled / exp_scaled.sum()


# ----------------------------------------------------------------------
# Gradient of the soft‑max approximation
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, w: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute ∇_a  ( 1/α log Σ exp(α·conv) )
    where conv = a*a and w is the soft‑max weight vector on conv.
    The gradient equals term1 + term2:
        term1_i = Σ_j a_j w_{i+j}
        term2_i = Σ_j a_j w_{i-j}
    Both terms are obtained by two FFT‑based convolutions.
    """
    n = a.shape[0]
    L = w.shape[0]                     # L = 2·n‑1
    # a * w
    conv_aw = _conv_full_fft(a, w, fft_len)
    term2 = conv_aw[:n]                # Σ_j a_j w_{i-j}
    # a * w_rev
    w_rev = w[::-1]
    conv_rev = _conv_full_fft(a, w_rev, fft_len)
    # term1_i = conv_rev[L‑1‑i]
    idx = L - 1 - np.arange(n)
    term1 = conv_rev[idx]
    return term1 + term2

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that
    minimises ``evaluate_sequence``.  The routine adapts its schedule to
    the available time budget and always returns a feasible (Σ>0.01)
    sequence.
    """
    np.random.seed(seed)
    start_time = time.time()
    # give a small safety margin for the final exact evaluation
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # Choose a working length n (5 k ≤ n ≤ 25 k)
    # --------------------------------------------------------------
    try:
        prev = globals()["height_sequence_1"]
        n = max(len(prev), 5000)
        prev_arr = np.asarray(prev, dtype=np.float64)
        if prev_arr.sum() > 0.0:
            prev_arr = prev_arr / prev_arr.sum()
        else:
            prev_arr = None
    except Exception:
        prev_arr = None
        n = 12000                         # comfortable default
    n = int(min(max(n, 5000), 25000))

    # FFT length that can handle both a*a and a*w convolutions
    fft_len = 1 << ((3 * n - 2).bit_length())   # ≥ 3·n‑2 (covers a*w)
    L = 2 * n - 1                                 # length of a*a

    # ------------------------------------------------------------------
    # Cheap score (Σa_i = 1) : 2·n·max(conv)
    # ------------------------------------------------------------------
    def cheap_score(vec: np.ndarray) -> float:
        conv = _conv_self_fft(vec, fft_len)
        return 2.0 * n * conv.max()

    # ------------------------------------------------------------------
    # Phase 0 – pick a decent starting point from a few parametric families
    # ------------------------------------------------------------------
    best_a = _uniform_seq(n)
    best_val = cheap_score(best_a)

    # incorporate the previous construction if length matches
    if prev_arr is not None and prev_arr.size == n:
        val = cheap_score(prev_arr)
        if val < best_val:
            best_a, best_val = prev_arr.copy(), val

    # a few deterministic families
    for lam in (0.0, 0.5, 1.0, 2.0, 3.0):
        cand = _exponential_seq(n, lam)
        val = cheap_score(cand)
        if val < best_val:
            best_a, best_val = cand, val

    for p in (0.2, 0.5, 1.0, 1.5):
        cand = _power_law_seq(n, p)
        val = cheap_score(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # a few random Dirichlet seeds
    for conc in (0.01, 0.05, 0.1, 0.5, 1.0):
        cand = _dirichlet_seq(n, conc)
        val = cheap_score(cand)
        if val < best_val:
            best_a, best_val = cand, val

    print(f"[Init] length={n}, best cheap score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam optimisation on the smooth soft‑max surrogate
    # ------------------------------------------------------------------
    a = best_a.copy()
    best_iter_a = a.copy()
    best_iter_val = best_val

    # Adam hyper‑parameters
    lr = 0.12
    beta1, beta2 = 0.9, 0.999
    eps_opt = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # soft‑max temperature (larger => closer to true max)
    alpha = 80.0

    it = 0
    last_improve = 0

    while time.time() < deadline - 5.0:
        it += 1

        # full convolution a*a
        conv = _conv_self_fft(a, fft_len)          # length L = 2n‑1
        max_conv = conv.max()
        cheap = 2.0 * n * max_conv

        # keep the best seen so far
        if cheap < best_iter_val - 1e-12:
            best_iter_val = cheap
            best_iter_a = a.copy()
            last_improve = it
            print(f"[Iter {it:07d}] NEW BEST cheap = {cheap:.6f}")

        # compute soft‑max weights on the convolution
        w = _softmax_weights(conv, alpha)

        # gradient of the smooth objective
        g = _gradient_softmax(a, w, fft_len)

        # Adam update (moving opposite to the gradient)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps_opt)
        a = a - step

        # project back onto the simplex
        a = _simplex_projection(a)

        # occasional random restart if stagnating for a long time
        if it - last_improve > 180_000:
            perturb = 0.025 * np.random.rand(n)
            a = _simplex_projection(best_iter_a + perturb)
            m[:] = 0.0
            v[:] = 0.0
            last_improve = it
            print(f"[Iter {it:07d}] Restart after stagnation.")

        # slowly increase the temperature to sharpen the focus on the true max
        if it % 30_000 == 0:
            alpha = min(alpha * 1.25, 300.0)

    # ------------------------------------------------------------------
    # Phase 2 – (1+1)‑ES hill‑climb on the true cheap objective
    # ------------------------------------------------------------------
    sigma = 0.13                     # log‑space mutation strength
    fail_cnt = 0
    it_es = 0
    while time.time() < deadline - 2.0:
        it_es += 1
        cand = best_iter_a * np.exp(sigma * np.random.randn(n))
        cand = _simplex_projection(cand)
        val = cheap_score(cand)
        if val < best_iter_val:
            best_iter_val = val
            best_iter_a = cand
            sigma = max(sigma * 0.85, 1e-4)
            fail_cnt = 0
            print(f"[ES {it_es:07d}] improved cheap = {val:.6f}  sigma={sigma:.5f}")
        else:
            fail_cnt += 1
            if fail_cnt > 300:
                sigma = min(sigma * 1.40, 0.6)
                fail_cnt = 0

        # occasional fresh restart from a Dirichlet when the hill‑climb stalls
        if it_es % 120_000 == 0:
            a_restart = _dirichlet_seq(n, np.random.uniform(0.001, 2.0))
            best_iter_a = _simplex_projection(best_iter_a + 0.02 * a_restart)

    # ------------------------------------------------------------------
    # Phase 3 – final exact evaluation
    # ------------------------------------------------------------------
    final_seq = best_iter_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    return final_seq
