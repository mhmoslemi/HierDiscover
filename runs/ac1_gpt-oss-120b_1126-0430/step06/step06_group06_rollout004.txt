#!/usr/bin/env python
"""
Propose a non‑negative step function (list of heights) that minimises

        2·n·max_k (a*a)[k] / (Σ a_i)²

Algorithm
---------
1. Choose a working length n (5 k – 20 k).  If a previous solution
   ``height_sequence_1`` exists we reuse its length, otherwise we start
   with n = 8000.
2. Exhaustively scan a few parametric families (uniform, power‑law,
   exponential, stretched‑exponential, beta, Dirichlet).  The best
   candidate (with normalised sum = 1) becomes the start point.
3. Smooth optimisation:
   * Parameterise a = softmax(b) (b∈ℝⁿ, a_i≥0, Σa_i=1 automatically).
   * Use a smoothed version of the max‑norm:
         f_β(a)= 2·n·(1/β)·log Σ_k exp(β·conv_k)   with conv = a*a.
     The gradient ∂f/∂a = 4·n·(w * a) where w is the soft‑max
     weights w_k ∝ exp(β·conv_k).
   * Optimise b with Adam (exponential‑moving‑average of the gradient)
     and a continuation schedule for β (β grows from 5 to ≈200).
   * Keep the best vector seen during the run.
4. Final refinement – a cheap (1+1) evolution‑strategy on the true
   cheap score 2·n·max(conv) to polish the result.
5. Return the best vector; the final exact score is printed for
   reference.

All heavy work is done with real FFTs; the routine respects the 1000 s
budget and always returns a feasible sequence (non‑negative, sum>0.01).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level – no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using real FFT (length 2·n‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len).real[: 2 * a.size - 1]
    return conv


def _conv_max_fft(a: np.ndarray, fft_len: int) -> float:
    """Maximum coefficient of a*a (sum of a is assumed 1)."""
    conv = _conv_full_fft(a, fft_len)
    return float(conv.max())


def _cheap_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Cheap objective = 2·n·max(conv) (sum(a) = 1)."""
    return 2.0 * n * _conv_max_fft(a, fft_len)


# ----------------------------------------------------------------------
# Parametric families (all return a normalised probability vector)
# ----------------------------------------------------------------------


def _uniform_seq(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _power_law_seq(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = i ** (-p)
    return seq / seq.sum()


def _exponential_seq(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    seq = np.exp(-lam * i)
    return seq / seq.sum()


def _power_law_exp_seq(n: int, p: float, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    seq = (i + 1.0) ** (-p) * np.exp(-lam * i)
    return seq / seq.sum()


def _stretched_exponential_seq(n: int, lam: float, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    seq = np.exp(-lam * (i ** gamma))
    return seq / seq.sum()


def _beta_seq(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(1, n + 1, dtype=np.float64) / n)
    seq = (x ** alpha) * ((1.0 - x) ** beta)
    s = seq.sum()
    if s == 0.0:
        return None
    return seq / s


def _dirichlet_seq(n: int, conc: float) -> np.ndarray:
    g = np.random.gamma(conc, 1.0, size=n)
    return g / g.sum()

    minimises ``evaluate_sequence``.  The routine adapts its schedule
