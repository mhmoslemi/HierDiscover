def gradient_descent(a):
    best_a = a.copy()
    best_score = cheap_score(a)
    max_iters = 1000000
    eta0 = 0.5
    tau_start = 1.0
    tau_end = 30.0
    tau_ramp_iters = 20000  # iterations to increase tau
    stagnation_counter = 0
    last_best_iter = 0
    for it in range(1, max_iters+1):
        if time.time() >= deadline: break
        # update tau
        tau = tau_start + (tau_end - tau_start) * min(1.0, it / tau_ramp_iters)
        # FFT of a
        a_fft = np.fft.fft(a, fft_len)
        conv = np.fft.ifft(a_fft * a_fft).real[:2*n-1]
        max_conv = conv.max()
        # compute cheap
        cheap = 2.0 * n * max_conv
        if cheap < best_score - 1e-12:
            best_score = cheap
            best_a = a.copy()
            last_best_iter = it
            stagnation_counter = 0
            print(f"[GD {it}] new best cheap {cheap:.6f}")
        else:
            stagnation_counter +=1

        # compute weight for smoothed max
        # to avoid overflow:
        conv_shifted = conv - max_conv   # max_conv ensures max_shift = 0 for max entry
        w_unnorm = np.exp(tau * conv_shifted)
        w_sum = w_unnorm.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            # fallback: use exact subgradient with max index only
            grad = np.zeros_like(a)
            kmax = int(conv.argmax())
            lo = max(0, kmax - (n - 1))
            hi = min(kmax, n - 1)
            rng = np.arange(lo, hi+1)
            j = kmax - rng
            # diag contributions
            eq = rng == j
            grad[rng[~eq]] = 2.0 * a[j[~eq]]
            grad[rng[eq]] = a[rng[eq]]
        else:
            w = w_unnorm / w_sum
            # compute gradient via convolution w * rev(a)
            a_rev = a[::-1]
            w_fft = np.fft.fft(w, fft_len)
            a_rev_fft = np.fft.fft(a_rev, fft_len)
            grad_full = np.fft.ifft(w_fft * a_rev_fft).real
            # slice grad
            grad = 2.0 * grad_full[n-1 : n-1+n]   # length n

        # update
        eta = eta0 / math.sqrt(it)
        # multiplicative update with exponentiated gradient
        log_a = np.log(a + 1e-30)
        log_a -= eta * grad
        a = np.exp(log_a)
        a_sum = a.sum()
        if not np.isfinite(a_sum) or a_sum == 0.0:
            # revert to best known
            a = best_a.copy()
            continue
        a /= a_sum

        # occasional restart if stagnation > some threshold
        if stagnation_counter > 200000:
            # add small random noise and project onto simplex
            perturb = np.random.rand(n) * 0.001
            a = _simplex_projection(a + perturb)
            stagnation_counter = 0
            print(f"[GD {it}] restart due to stagnation")
    return best_a, best_score

def permutation_local_search(a, best_score, deadline):
    a_curr = a.copy()
    best_a = a.copy()
    best_val = best_score
    temperature = 1.0
    cooling = 0.99995
    n = len(a)
    # precompute FFT length for cheap score
    fft_len = 1 << ((2*n - 1).bit_length())
    while time.time() < deadline:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        a_curr[i], a_curr[j] = a_curr[j], a_curr[i]
        # compute cheap quickly: need conv max
        conv = np.fft.ifft(np.fft.fft(a_curr, fft_len) * np.fft.fft(a_curr, fft_len)).real[:2*n-1]
        cheap = 2.0 * n * conv.max()
        if cheap < best_val - 1e-12:
            best_val = cheap
            best_a = a_curr.copy()
            temperature = max(temperature*0.9, 1e-6)  # accept big improvement and cool
        else:
            # annealed acceptance
            delta = cheap - best_val
            prob = math.exp(-delta / max(temperature, 1e-12))
            if np.random.rand() < prob:
                # accept worse swap
                pass
            else:
                # revert
                a_curr[i], a_curr[j] = a_curr[j], a_curr[i]
        temperature = max(temperature * cooling, 1e-8)
    return best_a, best_val

        else:

        # schedule tau
        tau = tau_start + (tau_end - tau_start) * min(1.0, iter_cnt / tau_ramp)

        # smoothed weight
        max_conv = conv.max()

            a = _simplex_projection(best_iter_a + perturb)

    a = best_iter_a.copy()
    best_swap_a = a.copy()

                best_swap_val = cheap
                best_swap_a = a.copy()
                print(f"[Swap] NEW BEST cheap = {cheap:.6f}")
        else:
            a[i], a[j] = a[j], a[i]  # revert
        temperature = max(temperature * cooling, 1e-8)

    return best_swap_a, best_swap_val

"""Propose a non‑negative step function that minimizes `evaluate_sequence`.

The algorithm works in three stages:

1. **Broad parametric sweep** – evaluates many analytic families
   (exponential, symmetric “U”, power‑law, cosine‑U, Gaussian‑mixture,
   beta, Dirichlet) and keeps the best vector on the simplex.
2. **Smooth max‑convolution descent** – minimises a smooth approximation
   of the maximal convolution using an exponentiated‑gradient / projection
   scheme.  The temperature τ is annealed from a low to a high value,
   turning the smooth objective into the true max.
3. **Permutation‑swap local search** – a simulated‑annealing style random
   swap of indices refines the ordering of the entries.

The routine respects the user‑supplied time budget and always returns a
feasible sequence.  It makes use only of NumPy (no external optimisation
packages)."""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_idx = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho_idx.size == 0:
        # all entries are non‑positive – give the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho_idx[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_max_fft(a: np.ndarray, fft_len: int) -> float:
    """Return the maximal value of the full linear convolution a*a."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa).real[: 2 * a.size - 1]
    return float(conv.max())


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using an FFT of length `fft_len`."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa).real[: 2 * a.size - 1]
    return conv


def _cheap_score(a: np.ndarray, n: int, fft_len: int) -> float:
    """Cheap surrogate: 2·n·max(conv(a,a)).  Assumes sum(a)=1."""
    maxb = _conv_max_fft(a, fft_len)
    return 2.0 * n * maxb


# ----------------------------------------------------------------------
# Core optimisation routine for a single length `n`
# ----------------------------------------------------------------------


def _optimize_for_n(n: int, prev_arr: np.ndarray, deadline: float):
    """
    Perform the three‑stage search for a fixed length `n`.
    Returns the best vector (as a NumPy array) and its cheap score.
    """
    # FFT length for the cheap score (convolution of length 2·n‑1)
    fft_len = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 0.  Broad parametric sweep – keep the best vector on the simplex.
    # ------------------------------------------------------------------
    best_a = None
    best_val = float("inf")

    # Uniform baseline
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    val = _cheap_score(uniform, n, fft_len)
    best_a, best_val = uniform, val

    # Previous construction if it has the same length
    if prev_arr is not None and prev_arr.size == n:
        val = _cheap_score(prev_arr, n, fft_len)
        if val < best_val:
            best_a, best_val = prev_arr.copy(), val

    # ----- Exponential (left) ----------------------------------------
    for lam in np.linspace(0.0, 20.0, 41):
        if time.time() >= deadline:
            break
        a = np.exp(-lam * np.arange(n, dtype=np.float64))
        a /= a.sum()
        val = _cheap_score(a, n, fft_len)
        if val < best_val:
            best_a, best_val = a.copy(), val

    # ----- Symmetric exponential (U‑shaped) -------------------------
    for lam in np.linspace(0.0, 20.0, 41):
        if time.time() >= deadline:
            break
        idx = np.arange(n, dtype=np.float64)
        a = np.exp(-lam * np.minimum(idx, n - 1 - idx))
        a /= a.sum()
        val = _cheap_score(a, n, fft_len)
        if val < best_val:
            best_a, best_val = a.copy(), val

    # ----- Power‑law (left) ------------------------------------------
    for p in np.linspace(0.01, 5.0, 51):
        if time.time() >= deadline:
            break
        a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
        a /= a.sum()
        val = _cheap_score(a, n, fft_len)
        if val < best_val:
            best_a, best_val = a.copy(), val

    # ----- Symmetric power‑law (U‑shaped) ----------------------------
    for p in np.linspace(0.01, 5.0, 51):
        if time.time() >= deadline:
            break
        i = np.arange(n, dtype=np.float64)
        a = np.minimum(i + 1, n - i) ** (-p)
        a /= a.sum()
        val = _cheap_score(a, n, fft_len)
        if val < best_val:
            best_a, best_val = a.copy(), val

    # ----- Cosine‑U shape (high at the ends, low in the centre) -----
    if time.time() < deadline:
        i = np.arange(n, dtype=np.float64)
        a = np.cos(np.pi * i / (n - 1)) ** 2
        a /= a.sum()
        val = _cheap_score(a, n, fft_len)
        if val < best_val:
            best_a, best_val = a.copy(), val

    # ----- Gaussian mixture at the two ends -------------------------
    for sigma_frac in (0.02, 0.05, 0.1):
        if time.time() >= deadline:
            break
        sigma = sigma_frac * n
        x = np.arange(n, dtype=np.float64)
        a = np.exp(- (x ** 2) / (2.0 * sigma ** 2)) + np.exp(- ((x - (n - 1)) ** 2) / (2.0 * sigma ** 2))
        a /= a.sum()
        val = _cheap_score(a, n, fft_len)
        if val < best_val:

    alphas = np.concatenate((
        np.logspace(-4, -2, 5),
        np.linspace(0.01, 0.5, 5),
        np.linspace(0.5, 2.0, 5),
        np.linspace(2.0, 5.0, 5)

    betas = alphas.copy()
    for alpha in alphas:
        for beta in betas:
            if time.time() >= deadline:
                break
            a = (x ** alpha) * ((1.0 - x) ** beta)
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _cheap_score(a, n, fft_len)
            if val < best_val:
                best_a, best_val = a.copy(), val

    for conc in (0.0005, 0.001, 0.005, 0.01, 0.05, 0.1, 0.5, 1.0, 2.0, 5.0):
        if time.time() >= deadline:
            break
        g = np.random.gamma(conc, 1.0, size=n)
        a = g / g.sum()
        val = _cheap_score(a, n, fft_len)
        if val < best_val:
            best_a, best_val = a.copy(), val

    print(f"[Init n={n}] best cheap = {best_val:.6f}")

    # 1.  Smooth‑max gradient descent on the simplex.
    # --------------------------------------------------------------
    a = best_a.copy()
    best_iter_a = a.copy()
    best_iter_val = best_val
    iter_cnt = 0
    stagnation = 0
    eta0 = 0.7                # base step‑size for the additive update
    tau_start = 0.5           # start with a very smooth objective
    tau_end = 30.0            # finish close to the true max
    tau_ramp = 50000          # iterations over which τ is increased
    max_iters = 2_000_000
    # FFT length large enough for the gradient convolution (≈3·n)
    grad_fft_len = _next_pow2(3 * n)

        # Full convolution a*a (used for cheap score and for the weight)
        a_fft = np.fft.fft(a, grad_fft_len)
        conv = np.fft.ifft(a_fft * a_fft).real[: 2 * n - 1]
        maxb = conv.max()
        cheap = 2.0 * n * maxb

            best_iter_val = cheap
            best_iter_a = a.copy()
            stagnation = 0
            print(f"[GD {iter_cnt:07d}] NEW BEST cheap = {cheap:.6f}")
        else:
            stagnation += 1

        tau = tau_start + (tau_end - tau_start) * min(1.0, iter_cnt / tau_ramp)

        w_sum = w_unnorm.sum()

            grad = np.zeros_like(a)
            kmax = int(conv.argmax())
            lo = max(0, kmax - (n - 1))
            hi = min(kmax, n - 1)
            rng = np.arange(lo, hi + 1, dtype=np.int64)
            j = kmax - rng
            eq = (rng == j)
            grad[rng[~eq]] = 2.0 * a[j[~eq]]
            grad[rng[eq]] = a[rng[eq]]
        else:
            w = w_unnorm / w_sum
            # Gradient = 2 * (w ⋆ rev(a)) sliced appropriately
            a_rev = a[::-1]
            w_fft = np.fft.fft(w, grad_fft_len)
            a_rev_fft = np.fft.fft(a_rev, grad_fft_len)
            grad_full = np.fft.ifft(w_fft * a_rev_fft).real
            grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        eta = eta0 / math.sqrt(iter_cnt)
        a_new = a - eta * grad
        a = _simplex_projection(a_new)

        if stagnation > 200_000:

            a = _simplex_projection(best_iter_a + perturb)
            stagnation = 0
            print(f"[GD {iter_cnt:07d}] Restart after stagnation.")

    # 2.  Permutation / swap local search (simulated‑annealing style)
    # --------------------------------------------------------------
    a = best_iter_a.copy()
    best_swap_a = a.copy()
    best_swap_val = best_iter_val
    temperature = 1.0
    cooling = 0.99997

    # pre‑compute FFT length for the cheap evaluation inside the loop
    conv_fft_len = fft_len

    while time.time() < deadline:
        i, j = np.random.randint(0, n, size=2)
        if i == j:

        cheap = 2.0 * n * conv.max()

        accept = False
        if cheap < best_swap_val - 1e-12:
            accept = True
        else:
            delta = cheap - best_swap_val
            prob = math.exp(-delta / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            if cheap < best_swap_val - 1e-12:
                best_swap_val = cheap
                best_swap_a = a.copy()
                print(f"[Swap] NEW BEST cheap = {cheap:.6f}")
        else:
            # revert the swap
            a[i], a[j] = a[j], a[i]

        temperature = max(temperature * cooling, 1e-8)

    return best_swap_a, best_swap_val

The algorithm works in three stages:

1. **Broad parametric sweep** – evaluates many analytic families  
   (exponential, symmetric “U”, power‑law, cosine‑U, Gaussian‑mixture,
   beta, Dirichlet) and keeps the best vector on the simplex.
2. **Smooth‑max gradient descent** – minimises a smooth approximation of
   the maximal convolution using an exponentiated‑gradient / projection
   scheme.  The temperature τ is annealed from low to high, turning the
   smooth objective into the true max.
3. **Permutation‑swap local search** – a simulated‑annealing style random
   swap of indices refines the ordering of the entries.

The routine respects the user‑supplied time budget and always returns a
feasible sequence.  It uses only NumPy (no external optimisation packages)."""

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_idx = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho_idx.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho_idx[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_max_fft(a: np.ndarray, fft_len: int) -> float:
    """Maximum entry of the full linear convolution a*a."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa).real[: 2 * a.size - 1]
    return float(conv.max())


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a (used during gradient descent)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa).real[: 2 * a.size - 1]
    return conv


def _cheap_score(a: np.ndarray, n: int, fft_len: int) -> float:
    """Cheap surrogate: 2·n·max(conv(a,a)).  Assumes sum(a)=1."""
    maxb = _conv_max_fft(a, fft_len)
    return 2.0 * n * maxb
