#!/usr/bin/env python
"""
Propose a step‑function (non‑negative heights) that minimizes the
evaluation metric used in the challenge.

The algorithm:
1. Build the exact arcsine bin distribution (probability masses) on a
   uniform grid of length `n`.  This distribution is known to give a
   very low max‑convolution value.
2. Parameterise the sequence by soft‑max logits `z` (so that the
   heights are always non‑negative and sum to 1).  This turns the
   constrained problem into an unconstrained one.
3. Optimise `z` with a continuation of L‑BFGS‑B runs.  In each
   continuation step we minimise a *smoothed* version of the objective
   `2·n·max(a*a)`.  The smoothing (soft‑max) makes the gradient
   well‑behaved; the smoothing factor `eps_factor` is decreased
   geometrically so that the optimisation gradually focuses on the
   true maximum.
4. After each L‑BFGS phase we evaluate the true (non‑smoothed)
   metric using `evaluate_sequence`.  The best sequence seen so far is
   kept.
5. If the time budget permits a final short Adam refinement is run
   (similar to the reference code) to squeeze a few extra digits.
6. The best sequence (as a plain Python list) is returned.

The implementation respects the 1000 s budget, uses only NumPy, SciPy
(if available) and stays within the required API.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max, returns a probability vector."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Logits that map to `a` via soft‑max."""
    return np.log(a + eps)


def _fft_len_for_conv(n: int) -> int:
    """Smallest power‑of‑two ≥ 2·n‑1 (length required for linear convolution)."""
    return 1 << ((2 * n - 1).bit_length())


def _conv_and_smooth_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Compute the (smoothed) maximum of the self‑convolution `a*a`
    together with the gradient of `2·n·smooth_max` w.r.t. `a`.

    Parameters
    ----------
    a : np.ndarray
        Probability vector (non‑negative, sums to 1).
    eps_factor : float
        Relative smoothing factor: eps = eps_factor * max(conv).

    Returns
    -------
    max_conv : float
        Maximum entry of the raw (unsmoothed) self‑convolution.
    smooth_max : float
        Smoothed version of the maximum:
        max_conv + eps·log∑exp((conv‑max_conv)/eps).
    grad_a : np.ndarray
        Gradient of `2·n·smooth_max` w.r.t. `a`.
    """
    n = a.shape[0]
    L = _fft_len_for_conv(n)

    # FFT of a
    a_fft = np.fft.rfft(a, L)

    # linear self‑convolution (length 2·n‑1)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]

    max_conv = float(conv.max())

    # smoothed soft‑max weights
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv          # improve stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / w.size)
        w_sum = 1.0
    else:
        w = w / w_sum

    # smoothed maximum value
    smooth_max = max_conv + eps * math.log(w_sum)

    # cross‑correlation a ⋆ w (length n)
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # gradient of 2·n·smooth_max  (the factor 2*n already included)
    grad_a = 2.0 * n * corr * 2.0   # = 4·n·corr, matches the original implementation
    return max_conv, smooth_max, grad_a


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    the evaluation metric.  The routine builds an arcsine bin vector,
    then runs a continuation of L‑BFGS‑B optimisations with decreasing
    smoothing, followed by an optional short Adam fine‑tuning.
    """
    np.random.seed(seed)
    start = time.time()
    # keep a small safety margin for final logging / return
    deadline = start + budget_s - 2.0

    # ------------------------------------------------------------------
    # 0. Choose length `n`.  Use the length of the externally supplied
    #    sequence if present, otherwise pick a size depending on the time
    #    budget (larger n gives a finer discretisation).
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
        print(f"Using external length n = {n}")
    except Exception:
        # increase n for a longer budget; cap it to keep FFT size modest
        if budget_s >= 800:
            n = 40000
        elif budget_s >= 500:
            n = 30000
        else:
            n = 20000
        print(f"No external sequence – using length n = {n}")

    # ------------------------------------------------------------------
    # 1. Exact discretisation of the arcsine distribution (probability
    #    masses of the bins).  The vector sums to 1.
    # ------------------------------------------------------------------
    i = np.arange(n, dtype=np.float64)
    x0 = i / n
    x1 = (i + 1) / n
    sqrt0 = np.sqrt(np.clip(x0, 0.0, 1.0))
    sqrt1 = np.sqrt(np.clip(x1, 0.0, 1.0))
    F0 = (2.0 / np.pi) * np.arcsin(sqrt0)
    F1 = (2.0 / np.pi) * np.arcsin(sqrt1)
    a = F1 - F0                     # probability mass per bin
    a_sum = a.sum()
    if not np.isfinite(a_sum) or a_sum <= 0.0:
        raise RuntimeError("Failed to generate arcsine bins.")
    a /= a_sum                      # enforce exact normalisation

    # ------------------------------------------------------------------
    # 2. Helper to compute the *raw* metric (no smoothing) for monitoring.
    # ------------------------------------------------------------------
    def _raw_score(vec):
        """2·n·max(a*a) for a probability vector (sum≈1)."""
        nloc = vec.shape[0]
        Lloc = _fft_len_for_conv(nloc)
        fft = np.fft.rfft(vec, Lloc)
        conv = np.fft.irfft(fft * fft, Lloc)[: 2 * nloc - 1]
        return 2.0 * nloc * float(conv.max())

    best_a = a.copy()
    best_raw = _raw_score(best_a)
    print(f"Initial raw score (2·n·max_conv) = {best_raw:.8f}")

    # ------------------------------------------------------------------
    # 3. Parameterise by logits and run a continuation of L‑BFGS‑B.
    # ------------------------------------------------------------------
    try:
        from scipy.optimize import minimize
        scipy_available = True
    except Exception:
        scipy_available = False
        print("SciPy not available – will fall back to Adam optimisation.")

    # Initialise logits from the arcsine distribution
    z = _logits_from_distribution(best_a)

    # List of smoothing factors (large → smooth, small → close to true max)
    eps_schedule = [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]

    # Keep track of the best *exact* evaluation (using evaluate_sequence)
    try:
        best_exact = float(evaluate_sequence(best_a.tolist()))
    except Exception:
        best_exact = float('inf')
    print(f"Exact evaluation of initial sequence = {best_exact:.8f}")

    # --------------------------------------------------------------
    # Helper for a single L‑BFGS phase
    # --------------------------------------------------------------
    def _lbfgs_phase(z_start, eps_factor, maxfun):
        """Run L‑BFGS‑B for a given smoothing factor, return final logits."""
        def obj_and_grad(z_vec):
            a_vec = _softmax(z_vec)
            max_conv, smooth_max, grad_full = _conv_and_smooth_max_and_grad(
                a_vec, eps_factor
            )
            # Objective = 2·n·smooth_max
            obj = 2.0 * n * smooth_max
            # Gradient w.r.t. a is 0.5 * grad_full (since grad_full = 4·n·corr)
            grad_a = 0.5 * grad_full
            # Chain rule through soft‑max (mirror‑descent gradient)
            mu = np.dot(grad_a, a_vec)
            grad_z = a_vec * (grad_a - mu)
            return obj, grad_z.astype(np.float64)

        # Use L‑BFGS‑B; we do not set box bounds because logits are unconstrained.
        res = minimize(
            fun=obj_and_grad,
            x0=z_start,
            method='L-BFGS-B',
            jac=True,
            options={'maxfun': maxfun, 'maxiter': maxfun, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        return res.x, res.fun

    # --------------------------------------------------------------
    # Main continuation loop
    # --------------------------------------------------------------
    if scipy_available:
        for eps_factor in eps_schedule:
            if time.time() >= deadline:
                print("Time budget exhausted before next L‑BFGS phase.")
                break
            remaining = deadline - time.time()
            # Rough estimate: a single L‑BFGS evaluation costs ~3 FFTs.
            # Assume we can afford at most 200 evaluations per second.
            maxfun = max(20, int(min(3000, remaining * 150)))
            print(f"Starting L‑BFGS phase with eps_factor={eps_factor:.6f}, maxfun={maxfun}")

            z, _ = _lbfgs_phase(z, eps_factor, maxfun)

            # Evaluate the raw (unsmoothed) metric for the new candidate
            a_candidate = _softmax(z)
            raw_score = _raw_score(a_candidate)

            if raw_score < best_raw:
                best_raw = raw_score
                best_a = a_candidate.copy()
                print(f"  New best raw score = {best_raw:.8f}")

            # Exact evaluation (expensive, do it only when time permits)
            try:
                exact_score = float(evaluate_sequence(a_candidate.tolist()))
                if exact_score < best_exact:
                    best_exact = exact_score
                    best_a = a_candidate.copy()
                    print(f"  New best exact score = {best_exact:.8f}")
            except Exception as exc:
                print(f"  Exact evaluation failed: {exc}")

            # Add a tiny random jitter before the next phase to escape tiny plateaus
            jitter_scale = 0.02 / (eps_factor + 1e-12)
            z += jitter_scale * np.random.randn(n)
            z = np.clip(z, -60.0, 60.0)

    # ------------------------------------------------------------------
    # 4. If time permits, perform a short Adam refinement (the reference
    #    implementation).  This often gives a few additional digits.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("Starting optional Adam refinement.")
        # Adam hyper‑parameters – deliberately conservative
        m = np.zeros_like(z)
        v = np.zeros_like(z)
        beta1, beta2 = 0.9, 0.999
        eps_adam = 1e-8
        base_alpha = 0.6
        eps_factor = 0.12
        eps_factor_min = 5e-4
        eps_decay = 0.9993

        # The Adam loop runs until the time budget is exhausted
        it = 0
        while time.time() < deadline:
            it += 1
            a_cur = _softmax(z)

            max_conv, smooth_max, grad_full = _conv_and_smooth_max_and_grad(
                a_cur, eps_factor
            )
            raw_score = 2.0 * n * max_conv
            if raw_score < best_raw:
                best_raw = raw_score
                best_a = a_cur.copy()
                print(f"Adam iter {it:6d}: improved raw score = {best_raw:.8f}")

            # Gradient w.r.t. a (objective = 2·n·smooth_max)
            grad_a = 0.5 * grad_full
            mu = np.dot(grad_a, a_cur)
            grad_z = a_cur * (grad_a - mu)

            # Adam bias‑corrected update
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            alpha_t = base_alpha / (math.sqrt(it) + 1.0)
            step = -alpha_t * m_hat / (np.sqrt(v_hat) + eps_adam)
            z = z + step
            z = np.clip(z, -60.0, 60.0)

            eps_factor = max(eps_factor_min, eps_factor * eps_decay)

            if it % 5000 == 0 and np.random.rand() < 0.07:
                z += 0.004 * np.random.randn(n)

            # Early exit if we have essentially converged
            if it % 20000 == 0 and best_raw < 1.02:
                print(f"Early stop Adam at iteration {it}, best_raw={best_raw:.8f}")
                break

    # ------------------------------------------------------------------
    # 5. Final exact evaluation and return the best sequence.
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact evaluation score = {final_score:.8f}")
    except Exception as exc:
        print(f"Final exact evaluation failed: {exc}")

    return final_seq
