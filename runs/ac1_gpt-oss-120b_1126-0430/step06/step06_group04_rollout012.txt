def evaluate_sequence(sequence):
    # after cleaning negative values, cap values to [0,1000], etc.
    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)
    if sum_a < 0.01:
        return np.inf
    return float(2 * n * max_b / (sum_a**2))

    import numpy as np
    from scipy.optimize import minimize
    rng = np.random.default_rng(seed)
    a0 = np.asarray(init_seq, dtype=float)
    if a0.sum() < 1e-12:
        a0 = np.ones_like(a0)/a0.size
    else:
        a0 = a0 / a0.sum()
    n = a0.size
    # logits
    eps=1e-12
    x0 = np.log(a0 + eps)
    # define objective and gradient
    def fun(x):
        a = _softmax(x)  # sum to 1
        conv = _conv_self_fft(a)
        M = conv.max()
        obj = float(2.0 * n * M)
        return obj

    def grad(x):
        a = _softmax(x)
        conv = _conv_self_fft(a)
        k_max = int(np.argmax(conv))
        w = np.zeros_like(conv)
        w[k_max] = 1.0
        g_a = 2.0 * _correlation_fft(w, a)
        # natural gradient in logit space
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)
        return g_x

    # define callback to enforce time limit
    max_iter = 10000  # large
    start = time.time()
    def callback(xk):
        # L-BFGS will call this after each iteration
        if time.time() - start > time_limit:
            raise StopIteration

    options = {'maxiter': max_iter, 'disp': False}
    try:
        res = minimize(fun, x0, method='L-BFGS-B', jac=grad, callback=callback, options=options)
        if not res.success:
            # still return result
            pass
    except StopIteration:
        # time out, use current x
        x_res = x
    else:
        x_res = res.x

    a_opt = _softmax(x_res)
    obj_opt = _objective(a_opt)  # exact objective
    return a_opt.tolist(), obj_opt

"""
Non‑negative step‑function optimiser (v3).

Key ideas
---------
* Enlarge the family of analytical “shape” candidates (exponential,
  stretched‑exponential, power‑law, mixed‑exponential, sparse Dirichlet,
  etc.) and use a short exhaustive scan to pick the best starting point.
* Run Adam on the log‑it simplex with a more aggressive temperature /
  learning‑rate schedule and occasional large jitter.
* Refine with a more exploratory simulated‑annealing (higher start temperature,
  slower decay) that moves mass preferentially away from the peak‑contributing
  indices.
* Finish with a limited L‑BFGS‑B optimisation on the logit variables (exact
  max‑gradient) when SciPy is available – this gives a fast second‑order
  push toward a better local optimum.
* All stages respect the global time budget (default 1000 s) and fallback
  gracefully if any library is missing.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Global state used by the L‑BFGS callback
# ----------------------------------------------------------------------
_lbfgs_deadline = None
_lbfgs_last_x = None


def _lbfgs_callback(xk):
    """Callback for SciPy L‑BFGS‑B that stops after the global time deadline."""
    global _lbfgs_deadline, _lbfgs_last_x
    _lbfgs_last_x = xk
    if time.time() > _lbfgs_deadline:
        raise StopIteration


# ----------------------------------------------------------------------
# Low‑level utilities (FFT‑based convolution / correlation)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return conv_full[: 2 * n - 1]


def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation of a with weight vector w:
        g[i] = Σ_t w[i+t]·a[t]   for i=0,…,n‑1
    Implemented via FFT (real‑valued).
    """
    n = a.shape[0]
    m = w.shape[0]                     # = 2n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Stable soft‑max."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of vector v onto the probability simplex
    {x ≥ 0, Σ x = 1}.
    O(n log n) algorithm from Duchi et al.
    """
    if v.ndim != 1:
        raise ValueError("v must be 1‑D")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) assuming Σ a = 1.
    Returns a Python float.
    """
    n = a.size
    conv = _conv_self_fft(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Shape‑candidate generation (extended family)
# ----------------------------------------------------------------------


def _shape_candidates_extended(n: int, rng: np.random.Generator):
    """
    Produce a list of candidate initial sequences for support length n.
    Each candidate is a NumPy array normalized to sum‑to‑1.
    """
    cand = []

    idx = np.arange(n, dtype=float)
    half = (n - 1) / 2.0

    # Uniform
    cand.append(np.full(n, 1.0 / n))

    # Exponential decay (forward & reversed)
    for r in np.linspace(0.90, 0.9999, 12):
        a = r ** idx
        a = a / a.sum()
        cand.append(a)
        cand.append(a[::-1])

    # Stretched exponential
    for p in [0.5, 0.75, 1.0, 1.25, 1.5, 2.0]:
        for c in np.linspace(0.5, 10.0, 7):
            a = np.exp(-c * (idx / (n - 1)) ** p)
            if a.sum() == 0:
                continue
            a = a / a.sum()
            cand.append(a)
            cand.append(a[::-1])

    # Power‑law
    for p in np.linspace(0.5, 3.0, 6):
        a = 1.0 / ( (idx + 1) ** p )
        a = a / a.sum()
        cand.append(a)
        cand.append(a[::-1])

    # Mixed exponential (bimodal)
    for _ in range(5):
        r1 = rng.uniform(0.90, 0.999)
        r2 = rng.uniform(0.90, 0.999)
        alpha = rng.random()
        a = alpha * (r1 ** idx) + (1 - alpha) * (r2 ** (n - 1 - idx))
        a = a / a.sum()
        cand.append(a)

    # Sparse Dirichlet‑like (few non‑zero entries)
    k = max(2, int(0.02 * n))
    for _ in range(6):
        a = np.zeros(n, dtype=float)
        pos = rng.choice(n, size=k, replace=False)
        vals = rng.random(k)
        a[pos] = vals
        a = a / a.sum()
        cand.append(a)

    # Random Dirichlet (concentration <1 for sparsity)
    a = rng.gamma(0.3, 1.0, size=n)
    a = a / a.sum()
    cand.append(a)

    # Pure random Dirichlet (α=1)
    a = rng.random(n)
    a = a / a.sum()
    cand.append(a)

    # Gaussian‑like windows
    sigma_vals = np.linspace(0.10 * n, 0.40 * n, 6)
    for sigma in sigma_vals:
        a = np.exp(-0.5 * ((idx - half) / sigma) ** 2)
        a = a / a.sum()
        cand.append(a)

    return cand


def _best_initial_sequence(n: int, rng: np.random.Generator, time_limit: float):
    """
    Scan a modest set of analytical shapes and return the one with the
    smallest exact objective.  Abort early if time runs out.
    """
    start = time.time()
    best_a = None
    best_obj = float("inf")
    for a in _shape_candidates_extended(n, rng):
        if time.time() - start > time_limit:
            break
        obj = _objective(a)
        if obj < best_obj:
            best_obj, best_a = obj, a
    if best_a is None:
        # Fallback to a random Dirichlet
        a = rng.random(n)
        best_a = a / a.sum()
        best_obj = _objective(best_a)
    return best_a, best_obj


# ----------------------------------------------------------------------
# Stage 2 – Adam optimiser on the simplex (log‑it parametrisation)
# ----------------------------------------------------------------------


def _run_opt(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
) -> tuple[list[float], float]:
    """
    Adam optimisation on the log‑it simplex.
    Returns the best sequence (as list) and its exact objective.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # Initialise ---------------------------------------------------------
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
        a = a / a.sum()
    else:
        a = rng.random(n)
        a = a / a.sum()

    # logits (log‑it parametrisation)
    x = np.log(a + 1e-16)
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # Annealing schedule parameters (more aggressive)
    beta_start = 0.5
    beta_end = 1e7                # sharper soft‑max
    lr_start = 0.30               # slightly larger initial LR
    lr_end = 0.0002

    best_a = a.copy()
    best_obj = _objective(best_a)

    it = 0
    while time.time() - start < time_limit:
        it += 1

        # Stay on the simplex
        a = _softmax(x)

        conv = _conv_self_fft(a)
        M = conv.max()
        obj = float(2.0 * n * M)

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        # -----------------------------------------------------------------
        # schedule (time‑based)
        t = min((time.time() - start) / time_limit, 1.0)
        beta = beta_start * np.exp(np.log(beta_end / beta_start) * t)
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t)

        # smooth‑max weight vector
        if beta > 5e5:  # essentially a hard max
            k_max = int(np.argmax(conv))
            w = np.zeros_like(conv)
            w[k_max] = 1.0
        else:
            shifted = conv - M               # ≤ 0
            w = np.exp(beta * shifted)
            s = w.sum()
            if s == 0.0:
                w = np.ones_like(conv) / conv.size
            else:
                w /= s

        # gradient w.r.t. a (exact for the smooth max)
        g_a = 2.0 * _correlation_fft(w, a)

        # natural gradient in log‑it space
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape plateaus
        if it % 2500 == 0:
            jitter = rng.normal(scale=0.018, size=n)
            x += jitter

        # periodic restart from the current best (helps to escape shallow basins)
        if it % 50000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.008, size=n)
            m.fill(0.0)
            v.fill(0.0)

    # -----------------------------------------------------------------
    # prune near‑zero entries (helps the final objective)
    thresh = 1e-13
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = _objective(best_a)
        print(f"[compress] length → {best_a.size}, O={best_obj:.6f}")

    # -----------------------------------------------------------------
    # cheap local refinement (mass‑transfer moves)
    remaining = time_limit - (time.time() - start)
    if remaining > 0.5:
        a_loc = best_a.copy()
        obj_loc = best_obj
        n_loc = a_loc.size
        while time.time() - start < time_limit * 0.3:
            i, j = rng.choice(n_loc, size=2, replace=False)
            delta = a_loc[i] * rng.random() * 0.02
            if delta <= 0.0:
                continue
            a_new = a_loc.copy()
            a_new[i] -= delta
            a_new[j] += delta
            conv_new = _conv_self_fft(a_new)
            obj_new = float(2.0 * n_loc * conv_new.max())
            if obj_new < obj_loc:
                a_loc, obj_loc = a_new, obj_new
                if obj_new < best_obj:
                    best_a, best_obj = a_new, obj_new
        print(f"[local] post‑Adam O={best_obj:.6f}")

    return best_a.tolist(), best_obj


# ----------------------------------------------------------------------
# Stage 3 – targeted simulated‑annealing + projected gradient
# ----------------------------------------------------------------------


def _run_sa(
    init_seq: list[float],
    time_limit: float,
    seed: int = 0,
) -> tuple[list[float], float]:
    """
    Targeted simulated‑annealing refinement followed by a few
    projected‑gradient descent steps (exact max‑gradient).
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    a = np.asarray(init_seq, dtype=float)
    if a.sum() < 1e-12:
        a = np.ones_like(a) / a.size
    else:
        a = a / a.sum()
    n = a.shape[0]

    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())
    best_seq = a.copy()

    # SA schedule – more aggressive exploration
    T = best_obj * 0.5 + 1e-9          # higher start temperature
    T_min = best_obj * 1e-7 + 1e-12
    decay = 0.9995                    # slower cooling

    it = 0
    while time.time() - start < time_limit * 0.65:
        it += 1
        k_max = int(np.argmax(conv))

        # contribution of each index to the current peak
        i_vals = np.arange(n)
        j_vals = k_max - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i_vals[mask]] * a[j_vals[mask]]
        total_contrib = contrib.sum()

        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = rng.choice(n, p=probs)
        else:
            i = rng.integers(n)

        # pick destination far from i with a bias towards large separation
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        # occasionally encourage long jumps
        if rng.random() < 0.15:
            # choose a far index deliberately
            far = rng.choice(np.where(np.abs(np.arange(n) - i) > n // 4)[0])
            j = far

        delta = a[i] * rng.random() * 0.07   # up to 7 % of a[i]
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.clip(a_new, 0.0, None)
        a_new /= a_new.sum()

        conv_new = _conv_self_fft(a_new)
        obj_new = float(2.0 * n * conv_new.max())

        if obj_new < best_obj - 1e-14:
            a, conv, best_obj, best_seq = a_new, conv_new, obj_new, a_new.copy()
            print(f"[SA it={it}] new best O={best_obj:.6f}")
        else:
            # Metropolis acceptance
            if rng.random() < math.exp(-(obj_new - best_obj) / T):
                a, conv = a_new, conv_new

        if it % 3000 == 0:
            T = max(T * decay, T_min)

        if it % 9000 == 0:
            # global jitter (larger amplitude)
            noise = rng.normal(scale=0.012, size=n)
            a = np.clip(a + noise, 0.0, None)
            a /= a.sum()
            conv = _conv_self_fft(a)

    # -----------------------------------------------------------------
    # Projected‑gradient descent with the exact max‑gradient (more iterations)
    remaining = time_limit - (time.time() - start)
    if remaining > 0.35:
        a = best_seq.copy()
        n = a.size
        it_g = 0
        eta0 = 0.22
        while time.time() - start < time_limit:
            it_g += 1
            conv = _conv_self_fft(a)
            k_max = int(np.argmax(conv))
            w = np.zeros_like(conv)
            w[k_max] = 1.0
            g = 2.0 * _correlation_fft(w, a)    # exact max‑gradient
            eta = eta0 * (0.996 ** it_g)
            a_new = a - eta * g
            a_new = _project_simplex(a_new)
            obj_new = float(2.0 * n * _conv_self_fft(a_new).max())
            if obj_new < best_obj - 1e-14:
                a, best_obj, best_seq = a_new, obj_new, a_new.copy()
                print(f"[PG it={it_g}] improved O={best_obj:.6f}")
            else:
                eta0 *= 0.99
                if eta0 < 1e-6:
                    break

    return best_seq.tolist(), best_obj


# ----------------------------------------------------------------------
# Stage 4 – optional L‑BFGS‑B refinement (requires SciPy)
# ----------------------------------------------------------------------


def _run_lbfgs(
    init_seq: list[float],
    time_limit: float,
    seed: int = 0,
) -> tuple[list[float], float]:
    """
    L‑BFGS‑B refinement on the log‑it variables, using the exact max‑gradient.
    Falls back to the input if SciPy is unavailable.
    """
    global _lbfgs_deadline, _lbfgs_last_x
    try:
        from scipy.optimize import minimize
    except Exception:  # pragma: no cover
        # SciPy not present – skip this stage
        a = np.asarray(init_seq, dtype=float)
        a = a / a.sum()
        return a.tolist(), _objective(a)

    rng = np.random.default_rng(seed)
    a0 = np.asarray(init_seq, dtype=float)
    if a0.sum() < 1e-12:
        a0 = np.ones_like(a0) / a0.size
    else:
        a0 = a0 / a0.sum()
    n = a0.size
    eps = 1e-16
    x0 = np.log(a0 + eps)

    def fun(x):
        a = _softmax(x)
        conv = _conv_self_fft(a)
        M = conv.max()
        return float(2.0 * n * M)

    def grad(x):
        a = _softmax(x)
        conv = _conv_self_fft(a)
        k_max = int(np.argmax(conv))
        w = np.zeros_like(conv)
        w[k_max] = 1.0
        g_a = 2.0 * _correlation_fft(w, a)
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)
        return g_x

    # Set the global deadline for the callback
    _lbfgs_deadline = time.time() + time_limit
    _lbfgs_last_x = x0.copy()

    max_iter = max(10, int(time_limit * 25))  # rough budget of iterations
    try:
        res = minimize(fun, x0, method="L-BFGS-B", jac=grad,
                       callback=_lbfgs_callback,
                       options={"maxiter": max_iter, "disp": False})
        x_opt = res.x
    except StopIteration:
        # Timed out – use the last stored iterate
        x_opt = _lbfgs_last_x
    except Exception:  # pragma: no cover
        # Any other failure – fall back
        x_opt = x0

    a_opt = _softmax(x_opt)
    obj_opt = _objective(a_opt)
    return a_opt.tolist(), obj_opt


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **_ignore,
) -> list[float]:
    """
    Main optimisation routine.
    Searches over several support sizes, uses an extended analytical
    shape pre‑search, performs Adam optimisation, and finalises with a
    more exploratory simulated‑annealing + projected‑gradient refinement,
    followed by an optional L‑BFGS‑B push.
    Returns a plain list of non‑negative floats suitable for
    ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.6  # small safety margin

    # --------------------------------------------------------------
    # Choose support lengths – a mix of moderate and large sizes.
    lengths = [1200, 2400, 4800, 9600, 12000]

    best_seq = None
    best_obj = float("inf")

    # --------------------------------------------------------------
    # Stage‑1 + Stage‑2 for each length
    for idx, n in enumerate(lengths):
        now = time.time()
        remaining = deadline - now
        if remaining <= 4.0:
            break

        # allocate time: first length gets a bit more, later ones share the rest
        if idx == 0:
            run_time = remaining * 0.45
        else:
            run_time = (remaining * 0.35) / (len(lengths) - idx)

        # ----- Shape pre‑search (more generous) -----
        shape_time = min(6.0, run_time * 0.15)
        init_a, init_obj = _best_initial_sequence(n, rng, shape_time)
        print(f"[init n={n}] shape‑obj={init_obj:.6f}")

        # ----- Adam optimisation -----
        seq, val = _run_opt(
            n=n,
            time_limit=run_time,
            init_a=init_a,
            seed=seed + idx * 12345,
        )
        if val < best_obj:
            best_obj, best_seq = val, np.asarray(seq, dtype=float)
            print(f"[global] new overall best O={best_obj:.6f} (n={n})")

        if time.time() >= deadline - 0.5:
            break

    # --------------------------------------------------------------
    # Stage‑3 – targeted SA + PG (larger share of remaining budget)
    remaining = deadline - time.time()
    if best_seq is not None and remaining > 10.0:
        seq_sa, val_sa = _run_sa(
            best_seq.tolist(),
            time_limit=remaining * 0.60,
            seed=seed + 777777,
        )
        if val_sa < best_obj:
            best_obj, best_seq = val_sa, np.asarray(seq_sa, dtype=float)
            print(f"[refine] SA+PG improved O={best_obj:.6f}")

        # Stage‑4 – optional L‑BFGS‑B refinement
        remaining = deadline - time.time()
        if remaining > 3.0:
            seq_lb, val_lb = _run_lbfgs(
                best_seq.tolist(),
                time_limit=remaining,
                seed=seed + 888888,
            )
            if val_lb < best_obj:
                best_obj, best_seq = val_lb, np.asarray(seq_lb, dtype=float)
                print(f"[refine] L‑BFGS‑B improved O={best_obj:.6f}")

    # --------------------------------------------------------------
    # Fallback – should never be needed
    if best_seq is None or best_seq.size == 0:
        best_seq = np.ones(200, dtype=float) / 200.0

    # --------------------------------------------------------------
    # Final sanity check (diagnostic)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    # Enforce evaluator caps
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    if best_seq.sum() < 0.01:
        best_seq = np.ones_like(best_seq) / best_seq.size

    return best_seq.tolist()
