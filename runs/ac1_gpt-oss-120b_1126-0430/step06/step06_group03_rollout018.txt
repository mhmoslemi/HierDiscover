"""
Hybrid optimizer for the self‑convolution bound.

The routine works in three phases:

1. **Very‑wide parametric sweep** – a handful of analytical families
   (power‑law, offset power‑law, double‑power‑law, stretched‑exponential,
   exponential‑decay, logistic, symmetric product‑power‑law and a few
   custom shapes) are sampled on generous parameter ranges.  The cheap
   bound 2·n·max(a*a) (where the sum of the vector is 1) is used to keep
   the best candidate.

2. **Mirror‑descent (Adam) on the simplex** – starting from the best
   candidate a soft‑max parametrisation a = softmax(z) is optimised.
   The “smooth‑max’’ of the autocorrelation is obtained by a soft‑max
   with a temperature ε that is slowly cooled.  Adam updates are applied
   to the logits; occasional restarts and jitter help to escape plateaus.

3. **Final cheap local random walk** – tiny mass transfers between two
   coordinates are accepted only if they improve the exact cheap bound.
   This phase finishes the search before the time budget expires.

The returned list is a probability vector (non‑negative, sum≈1).  The
provided `evaluate_sequence` rescales it if necessary and returns the
final exact score.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities (no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max, returns a probability vector (sums to 1)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    s = e.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / z.size)
    return e / s


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Logits that reproduce `a` after a soft‑max."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """Smallest power‑of‑two ≥ 3·n‑2 (sufficient for full linear self‑convolution)."""
    return 1 << ((3 * n - 2).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Returns (max_conv, grad_a) for the smoothed objective

        f(a) = 2·n·smooth_max(a*a)

    where the smoothing constant is ε = eps_factor·max_conv.
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    eps = max(max_conv * eps_factor, 1e-12)

    # soft‑max over the convolution values (shifted for numerical stability)
    shifted = conv - max_conv                     # ≤ 0
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # derivative of 2·n·smooth_max – the factor 4·n comes from d(a*a)/da = 2·a (convolution)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a

def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}  (normalised)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _power_law_offset_distribution(n: int, p: float, offset: float) -> np.ndarray:
    """a_i ∝ (i+offset)^{-p}  (normalised)."""
    idx = np.arange(offset, offset + n, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _double_power_law_distribution(n: int, p: float, q: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}·(n-i)^{-q}  (normalised)."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) ** (-p)) * ((n - i) ** (-q))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _stretched_exp_distribution(n: int, tau_factor: float, gamma: float) -> np.ndarray:
    """a_i ∝ exp( - (i/(tau_factor·n))^{gamma} )  (normalised)."""
    tau = max(1e-12, tau_factor * n)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(- (i / tau) ** gamma)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _exp_decay_distribution(n: int, lam: float) -> np.ndarray:
    """a_i ∝ e^{-lam·i}+e^{-lam·(n‑1‑i)}  (normalised)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _logistic_distribution(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ 1/(1+e^{α·(i‑mid)})  (normalised)."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = 1.0 / (1.0 + np.exp(alpha * (i - mid)))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _sym_product_power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ ((i+1)*(n‑i))^{-p}  (symmetric, normalised)."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) * (n - i)) ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _two_sided_geometric_distribution(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·min(i, n‑1‑i))  (normalised, peaks at both ends)."""
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-lam * d)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _tent_distribution(n: int) -> np.ndarray:
    """Linear “tent’’ shape, rising to the centre and falling symmetrically."""
    i = np.arange(n, dtype=np.float64)
    half = n // 2
    a = np.where(i <= half, i + 1, n - i)
    a /= a.sum()
    return a

            # occasional jitter to escape flat regions
            if it % 3000 == 0 and rng.random() < 0.12:
                logits += 0.02 * rng.standard_normal(n)
                m *= 0.6
                v *= 0.6

            # cooling of the smooth‑max temperature
            eps_factor = max(eps_factor_min, eps_factor * eps_decay)

            # restart if we have not improved for a long stretch
            if it - last_improve > 200_000:
                # fall back to the best point seen so far in this run
                if best_local_seq is not None:
                    logits = _logits_from_distribution(best_local_seq) + 0.03 * rng.standard_normal(n)
                else:
                    logits = _logits_from_distribution(best_seq) + 0.03 * rng.standard_normal(n)
                m.fill(0.0)
                v.fill(0.0)
                eps_factor = 0.20
                last_improve = it

        # Return the best point found in this run
        return best_local_seq, best_local_score

    # Run the three Adam starts (or fewer if time runs out)
    for idx, start_vec in enumerate(starts):
        if time.time() >= adam_deadline:
            break
        print(f"[Adam] Run {idx + 1} starting")
        # convert start vector (which might be raw logits) to a proper logits array
        if start_vec.ndim == 1 and start_vec.shape[0] == n:
            start_logits = start_vec.copy()
        else:
            start_logits = _logits_from_distribution(best_seq)  # fallback
        seq, sc = _run_adam(start_logits)
        if seq is not None and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[Adam] Run {idx + 1} produced new global best cheap = {sc:.8f}")

    print(f"[Adam] Finished, best cheap = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Final cheap random walk (tiny mass swaps)
    # ------------------------------------------------------------------
    a = best_seq.copy()
    cheap_score = best_score

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        delta = 0.001 * a[i]          # move 0.1 % of the mass at i
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # non‑negativity is guaranteed by construction
        new_score = 2.0 * n * _conv_max(a_new)
        if new_score < cheap_score:
            a = a_new
            cheap_score = new_score
            best_seq = a.copy()
            best_score = cheap_score
            # very sparse progress report
            if rng.random() < 0.001:
                print(f"[Walk] Improved cheap = {cheap_score:.8f}")

    # ------------------------------------------------------------------
    # 5. Final exact evaluation via the provided function.
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        exact_score = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[Result] Final exact score = {exact_score:.6f}")
    except Exception as exc:
        exact_score = float("inf")
        print("[Result] Evaluation error:", exc)

    return final_seq
