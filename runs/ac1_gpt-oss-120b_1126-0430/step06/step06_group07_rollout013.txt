#!/usr/bin/env python
"""
Propose a non‑negative step function (list of heights) that minimises
``evaluate_sequence``.  The algorithm proceeds in three phases:

1. **Broad parametric sweep** – generates many candidate families
   (exponential, symmetric exponential, power‑law, symmetric power‑law,
   beta, Dirichlet) and keeps the best according to the cheap proxy
   ``cheap_score`` (which equals the exact evaluation when the sum of
   heights is normalised to 1).

2. **Exponentiated‑gradient mirror descent** – starting from the best
   candidate, repeatedly reduces the maximal entry of the self‑convolution
   using a true sub‑gradient (the gradient of the active bin is
   ``2*a[kmax‑i]``).  The step‑size is η₀/√t and occasional restarts
   keep the iterates from stagnating.

3. **Hybrid simulated‑annealing local search** – a combination of
   *swap* (permutation), *mass‑transfer* (move a fraction of weight from a
   high‑contributing index to a low‑contributing index) and *global
   perturb* (add small Gaussian noise and re‑project onto the simplex) is
   performed until the time budget is exhausted.  The best vector found
   is finally evaluated with ``evaluate_sequence`` and returned.

All helper functions are defined at module level; no closures or lambdas
are used.  The routine respects a user‑supplied time budget and always
returns a feasible sequence.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_max_fft(a: np.ndarray, fft_len: int) -> tuple[float, int]:
    """Maximum value of the full convolution a*a and its index."""
    # real FFT is faster for real inputs
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.size - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx


def _cheap_score(a: np.ndarray, n: int, fft_len: int) -> float:
    """Score 2·n·max(conv) for a vector with sum(a)=1."""
    maxb, _ = _conv_max_fft(a, fft_len)
    return 2.0 * n * maxb


# ----------------------------------------------------------------------
# Core search routine required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that
    minimises ``evaluate_sequence``.  The routine adapts its schedule to
    the available time budget and always returns a feasible sequence.
    """
    np.random.seed(seed)
    start_time = time.time()
    # reserve a few seconds for the final exact evaluation
    deadline = start_time + budget_s - 5.0
    eps = 1e-30

    # --------------------------------------------------------------
    # Choose a working length.
    # --------------------------------------------------------------
    try:
        prev = height_sequence_1               # may be defined globally
        prev_arr = np.asarray(prev, dtype=np.float64)
        if prev_arr.sum() > 0.0:
            prev_arr = prev_arr / prev_arr.sum()
        else:
            prev_arr = None
    except Exception:
        prev_arr = None

    # Start with at least 5000 entries, allow a modest increase if a
    # previous solution exists – a longer vector gives more flexibility.
    n = 5000
    if prev_arr is not None:
        n = max(5000, min(15000, int(len(prev_arr) * 1.2)))
    # keep n in a reasonable interval for the time budget
    n = min(max(n, 5000), 15000)

    # FFT length – next power of two ≥ 2·n‑1
    fft_len = 1 << ((2 * n - 1).bit_length())

    # --------------------------------------------------------------
    # 0. Initialise with cheap families
    # --------------------------------------------------------------
    best_a = None
    best_val = float("inf")

    # uniform baseline

    val = _cheap_score(uniform, n, fft_len)
    best_a, best_val = uniform, val

    if prev_arr is not None and prev_arr.size == n:
        val = _cheap_score(prev_arr, n, fft_len)
        if val < best_val:

    # ----- exponential (left) ------------------------------------
    for lam in np.linspace(0.0, 8.0, 41):
        if time.time() >= deadline:
            break
        a = np.exp(-lam * np.arange(n, dtype=np.float64))
        a /= a.sum()
        val = _cheap_score(a, n, fft_len)
        if val < best_val:
            best_a, best_val = a.copy(), val

    # ----- symmetric exponential (U‑shaped) -----------------------
    for lam in np.linspace(0.0, 8.0, 41):
        if time.time() >= deadline:

        a = np.exp(-lam * np.minimum(idx, n - 1 - idx))
        a /= a.sum()
        val = _cheap_score(a, n, fft_len)

    for p in np.linspace(0.01, 2.5, 41):

        a /= a.sum()
        val = _cheap_score(a, n, fft_len)
        if val < best_val:

        if time.time() >= deadline:

        i = np.arange(n, dtype=np.float64)
        a = np.minimum(i + 1, n - i) ** (-p)

        val = _cheap_score(a, n, fft_len)

        # full convolution via FFT (real variant)
        fa = np.fft.rfft(a, fft_len)
        conv_full = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        maxb = conv_full.max()
        cheap = 2.0 * n * maxb

        if cheap < best_iter_val - 1e-12:
            best_iter_val = cheap
            best_iter_a = a.copy()
            last_improve = iter_cnt
            print(f"[Iter {iter_cnt:07d}] NEW BEST cheap = {cheap:.6f}")

        # index of the maximal bin
        kmax = int(conv_full.argmax())

        # sub‑gradient of the active bin (correctly 2*a[kmax‑i] everywhere)
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        i_range = np.arange(lo, hi + 1, dtype=np.int64)
        j_range = kmax - i_range
        g = np.zeros_like(a)
        g[i_range] = 2.0 * a[j_range]

        eta = eta0 / math.sqrt(iter_cnt)
        log_a = np.log(a + eps)
        log_a -= eta * g
        a = np.exp(log_a)
        sum_a = a.sum()
