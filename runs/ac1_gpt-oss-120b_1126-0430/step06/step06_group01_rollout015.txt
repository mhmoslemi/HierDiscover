#!/usr/bin/env python
"""
Adaptive Adam optimisation of a non‑negative step function.

The metric to minimise is
    2·n·max(conv(a,a)) / (sum a)²,
which is scale‑invariant, so we enforce sum(a)=1 via a soft‑max
parameterisation.  We start from the exact arcsine‑bin construction
(which has a provably flat continuous convolution) and run a long
Adam optimisation with a progressively tightening smooth‑max
approximation.  Several random restarts (including noisy restarts)
are performed within the time budget; the best sequence found is
returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all pure top‑level utilities
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max, returns a probability vector."""
    zmax = np.max(z)
    e = np.exp(z - zmax)
    s = e.sum()
    if s == 0.0 or not np.isfinite(s):
        return np.full_like(z, 1.0 / z.size)
    return e / s


def _logits_from_distribution(p: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Logits that map to the probability vector `p` via soft‑max."""
    return np.log(p + eps)


def _fft_len_for_conv(n: int) -> int:
    """Smallest power‑of‑two ≥ 2·n‑1 (sufficient for linear convolution)."""
    return 1 << ((2 * n - 1).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a (no gradient)."""
    n = a.shape[0]
    L = _fft_len_for_conv(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Smoothed maximum of a*a together with its gradient.

    Parameters
    ----------
    a : np.ndarray
        Probability vector (non‑negative, sum = 1).
    eps_factor : float
        Multiplicative factor for the smoothing constant:
        eps = eps_factor * max(conv).

    Returns
    -------
    max_conv : float
        Exact maximum entry of the linear self‑convolution.
    grad_a : np.ndarray
        Gradient of 4·n·smooth_max w.r.t. a (already multiplied by 4·n).
    """
    n = a.shape[0]
    L = _fft_len_for_conv(n)

    # a_fft is reused for both forward and backward passes.
    a_fft = np.fft.rfft(a, L)

    # linear self‑convolution (length 2·n‑1)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # Smoothed soft‑max weights over the (2·n‑1) positions.
    eps = max(max_conv * eps_factor, 1e-15)
    shifted = conv - max_conv
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum

    # Correlation a ⋆ w (length n) – this yields ∂smooth_max/∂a.
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # Gradient (already multiplied by 4·n, see derivation).
    grad_a = 4.0 * n * corr
    return max_conv, grad_a


def _adam_optimize(
    z0: np.ndarray,
    n: int,
    deadline: float,
    *,
    base_alpha: float = 0.9,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps_adam: float = 1e-8,
    eps_factor_start: float = 0.25,
    eps_factor_min: float = 1e-6,
    eps_decay: float = 0.998,
    jitter_prob: float = 0.15,
    jitter_scale: float = 0.02,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the simplex (soft‑max re‑parametrisation).

    Returns the best probability vector and the associated cheap score
    (2·n·max_conv).  The loop runs until the supplied ``deadline``.
    """
    z = z0.copy()
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    best_a = None
    best_score = np.inf
    it = 0
    eps_factor = eps_factor_start
    last_improve = 0

    # Main optimisation loop
    while time.time() < deadline:
        it += 1

        a_cur = _softmax(z)                     # distribution (sum=1)
        max_conv, grad_a = _conv_max_and_grad(a_cur, eps_factor)
        cheap_score = 2.0 * n * max_conv

        if cheap_score < best_score:
            best_score = cheap_score
            best_a = a_cur.copy()
            last_improve = it
            if it % 500 == 0:
                print(f"Iter {it:6d} – new best score = {best_score:.8f}")

        # Mirror‑descent gradient on the logits
        mu = np.dot(grad_a, a_cur)
        grad_z = a_cur * (grad_a - mu)

        # Adam bias‑corrected moments
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # Step‑size schedule (inverse‑sqrt scaling)
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step = -alpha_t * m_hat / (np.sqrt(v_hat) + eps_adam)
        z = z + step
        # Keep logits bounded for numerical safety
        z = np.clip(z, -70.0, 70.0)

        # Reduce the smoothing constant gradually
        eps_factor = max(eps_factor_min, eps_factor * eps_decay)

        # Occasional random jitter to escape plateaus
        if (it % 2000 == 0) and (np.random.rand() < jitter_prob):
            z += jitter_scale * np.random.randn(n)

        # If we have stagnated for a long time, lower the step‑size a bit.
        if it - last_improve > 20000:
            base_alpha *= 0.7
            last_improve = it

    if best_a is None:
        # Should never happen, but fall back to the current point.
        best_a = _softmax(z)
        best_score = 2.0 * n * _conv_max(best_a)
    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    the evaluation metric.  The routine builds the exact arcsine
    discretisation, then performs a long Adam optimisation on the
    simplex, using several random restarts if time permits.  The
    best sequence found before the time budget expires is returned.
    """
    np.random.seed(seed)
    start = time.time()
    # leave a small safety margin for final logging / return
    deadline = start + budget_s - 2.0

    # ------------------------------------------------------------------
    # 0. Choose length of the vector.
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
        print(f"Using external length n = {n}")
    except Exception:
        n = 20000
        print(f"No external sequence – using default length n = {n}")

    # ------------------------------------------------------------------
    # 1. Exact discretisation of the arcsine distribution (bin masses).
    # ------------------------------------------------------------------
    i = np.arange(n, dtype=np.float64)
    x0 = i / n
    x1 = (i + 1) / n

    # CDF of the arcsine law: F(x) = (2/π)·arcsin(√x)
    sqrt0 = np.sqrt(np.clip(x0, 0.0, 1.0))
    sqrt1 = np.sqrt(np.clip(x1, 0.0, 1.0))
    F0 = (2.0 / np.pi) * np.arcsin(sqrt0)
    F1 = (2.0 / np.pi) * np.arcsin(sqrt1)

    a = F1 - F0                     # probability mass of each bin
    a_sum = a.sum()
    if not np.isfinite(a_sum) or a_sum <= 0.0:
        raise RuntimeError("Failed to generate arcsine bins.")
    a /= a_sum                      # enforce exact normalisation (sum = 1)

    # ------------------------------------------------------------------
    # 2. Record the initial score.
    # ------------------------------------------------------------------
    init_max = _conv_max(a)
    best_score = 2.0 * n * init_max
    best_a = a.copy()
    print(f"Initial score (2·n·max_conv) = {best_score:.8f}")

    # ------------------------------------------------------------------
    # 3. Run optimisation – possibly several restarts.
    # ------------------------------------------------------------------
    # Settings for the optimiser (tuned for the 1000 s budget).
    base_alpha = 0.9
    eps_factor_start = 0.25
    eps_decay = 0.998
    eps_min = 1e-6
    jitter_prob = 0.15
    jitter_scale = 0.02

    # Number of restarts – we start from the arcsine vector and then
    # try a few noisy restarts if time permits.
    restart = 0
    while time.time() < deadline:
        remaining = deadline - time.time()
        # allocate at most 30 % of the remaining budget to a single run
        run_deadline = time.time() + min(remaining * 0.3, 200.0)

        if restart == 0:
            # First run – initialise from the best known sequence.
            z0 = _logits_from_distribution(best_a)
        else:
            # Subsequent runs – random perturbations around the current best.
            perturb = np.random.randn(n) * 0.4
            z0 = _logits_from_distribution(best_a) + perturb

        # Run Adam optimisation for as long as we have time.
        a_candidate, score_candidate = _adam_optimize(
            z0,
            n,
            run_deadline,
            base_alpha=base_alpha,
            eps_factor_start=eps_factor_start,
            eps_factor_min=eps_min,
            eps_decay=eps_decay,
            jitter_prob=jitter_prob,
            jitter_scale=jitter_scale,
        )

        if score_candidate < best_score:
            best_score = score_candidate
            best_a = a_candidate.copy()
            print(f"Restart {restart}: improved global best = {best_score:.8f}")

        restart += 1
        # After a few restarts we keep only the best and stop if time is low.
        if restart >= 5:
            # we have already tried enough restarts; give the remaining time
            # to the best‑found trajectory in the next outer loop iteration.
            break

    # ------------------------------------------------------------------
    # 4. Final logging and return.
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        exact_score = evaluate_sequence(final_seq)
        print(f"Exact evaluation (after optimisation): {exact_score:.8f}")
    except Exception as exc:  # pragma: no cover – safety net
        print(f"Exact evaluation failed: {exc}")

    return final_seq
