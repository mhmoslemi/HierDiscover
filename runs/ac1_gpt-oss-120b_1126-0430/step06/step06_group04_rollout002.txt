"""
Non‑negative step‑function optimiser (v3).

Goal:
    minimise 2·n·max(conv(a,a))  with  a_i ≥ 0  and  Σ a_i = 1.
    (the evaluator divides by Σ a_i², therefore the objective is
    scaling‑invariant – we may keep the simplex normalisation.)

Strategy
---------
1. **Richer analytic pre‑search** – a much larger catalogue of
   handcrafted shapes is generated for every support length.
   In addition to the uniform/exponential/Gaussian families used
   previously we add:
        • Beta‑type densities   a_i ∝ (i+1)^{p‑1}(n‑i)^{q‑1}
        • Power‑law edge shapes a_i ∝ (i+½)^{-α}(n‑i‑½)^{-α}
        • Sine‑powers           a_i ∝ sin^p(π(i+½)/n)
        • Sparse Dirichlet / pure Dirichlet
   The best candidate (exact objective computed via FFT) seeds the
   optimisation.

2. **Long Adam on the simplex** – we run Adam on the log‑it
   parametrisation of the probability simplex.  The smooth‑max
   weight w is obtained from a temperature β that grows
   exponentially from 0.5 to 1e7, i.e. we transition from a smooth
   approximation to the true max.  Learning‑rate and momentum
   schedules are tuned for the larger support sizes.  Restarts and
   jitter are employed to leave shallow basins.

3. **Intensive final refinement** – a two‑phase local search:
      * a targeted simulated‑annealing that relocates mass from
        indices contributing heavily to the current maximal
        autocorrelation entry;
      * a prolonged projected‑gradient descent with the exact
        max‑gradient (step‑size decays until no further progress).
   Both phases run until the allocated time budget is exhausted.

The optimiser explores several support lengths (2000 – 12000)
and returns the sequence with the smallest exact objective found
within the global budget (default 1000 s).  All numbers are clipped
to the evaluator’s range [0,1000] and the sum is forced to be at
least 0.01 to avoid the “sum too small” guard.
"""

import time

# ----------------------------------------------------------------------
# Low‑level FFT utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a⋆a using a real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return conv_full[: 2 * n - 1]


def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation of a with weight vector w:
        (w * a)[i] = Σ_t w[i+t]·a[t]   (i = 0,…,n‑1)
    Implemented via FFT (real‑valued).
    """
    n = a.shape[0]
    m = w.shape[0]                     # = 2n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0, Σ x = 1}.  O(n log n) algorithm of Duchi et al.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) under the assumption Σ a = 1.
    Returns a Python float.
    """
    n = a.size
    conv = _conv_self_fft(a)
    return float(2.0 * n * conv.max())

def _beta_shapes(n: int, rng: np.random.Generator):
    """Beta‑type densities (two parameters p,q)."""
    cand = []
    pp = [0.3, 0.4, 0.5, 0.7, 1.0, 2.0]
    for p in pp:
        for q in pp:
            idx = np.arange(n, dtype=float)
            a = np.power(idx + 1.0, p - 1.0) * np.power(n - idx, q - 1.0)
            a = np.clip(a, 0.0, None)
            if a.sum() == 0.0:
                continue
            a /= a.sum()
            cand.append(a)
    return cand


def _powerlaw_edge_shapes(n: int):
    """Shapes concentrating mass near the ends: a_i ∝ (i+½)^-α (n-i-½)^-α."""
    cand = []
    alphas = [0.2, 0.3, 0.4, 0.5, 0.7]
    idx = np.arange(n, dtype=float)
    for a in alphas:
        a_vec = np.power(idx + 0.5, -a) * np.power(n - idx - 0.5, -a)
        a_vec = np.clip(a_vec, 0.0, None)
        a_vec /= a_vec.sum()
        cand.append(a_vec)
    return cand


def _sine_power_shapes(n: int):
    """Shapes a_i ∝ sin^p(π(i+½)/n)."""
    cand = []
    exps = [2, 3, 4, 6]
    idx = np.arange(n, dtype=float)
    theta = np.pi * (idx + 0.5) / n
    base = np.sin(theta)
    for p in exps:
        a = np.power(base, p)
        a = np.clip(a, 0.0, None)
        a /= a.sum()
        cand.append(a)
    return cand


def _shape_candidates(n: int, rng: np.random.Generator):
    """
    Assemble a diverse collection of analytically‑motivated
    candidates for support length n.  Every candidate is a 1‑D
    NumPy array normalised to sum‑to‑1.
    """
    cand = []

    # ---- simple baselines -------------------------------------------------
    cand.append(np.full(n, 1.0 / n))                # uniform

    # exponentials (forward & reversed)
    r_vals = np.linspace(0.90, 0.9997, 15)
    idx = np.arange(n, dtype=float)
    for r in r_vals:
        a = r ** idx
        a /= a.sum()
        cand.append(a)
        cand.append(a[::-1])

    # Gaussian windows
    centre = (n - 1) / 2.0
    sigma_vals = np.linspace(0.10 * n, 0.40 * n, 10)
    for sigma in sigma_vals:
        a = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
        a /= a.sum()
        cand.append(a)

    # ---- richer families -------------------------------------------------
    cand.extend(_beta_shapes(n, rng))
    cand.extend(_powerlaw_edge_shapes(n))
    cand.extend(_sine_power_shapes(n))

    # Sparse Dirichlet‑like (few non‑zero entries)
    k = max(2, int(0.03 * n))
    for _ in range(4):
        a = np.zeros(n, dtype=float)
        pos = rng.choice(n, size=k, replace=False)
        vals = rng.random(k)
        a[pos] = vals
        a /= a.sum()
        cand.append(a)

    # Pure Dirichlet (α=1)
    a = rng.random(n)
    a /= a.sum()
    cand.append(a)

    return cand


def _best_initial_sequence(n: int, rng: np.random.Generator, time_limit: float):
    """
    Scan the analytic candidate set for the smallest exact objective.
    The scan aborts early if the time budget expires.
    """
    start = time.time()
    best_a, best_val = None, float("inf")
    for a in _shape_candidates(n, rng):
        if time.time() - start > time_limit:
            break
        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a
    # fallback to a random Dirichlet if nothing was evaluated
    if best_a is None:
        a = rng.random(n)
        best_a = a / a.sum()
        best_val = _objective(best_a)
    return best_a, best_val

def _run_adam(
    n: int,
    time_limit: float,
    init_a: np.ndarray,
    seed: int = 0,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the soft‑max parametrisation.
    Returns the best sequence (as a NumPy array) and its exact
    objective 2·n·max(conv).
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    a = init_a.astype(float)
    a = np.clip(a, 0.0, None)
    if a.sum() == 0.0:
        a = np.ones(n, dtype=float) / n
    else:
        a /= a.sum()

    # log‑it parameters
    x = np.log(a + 1e-16)
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # annealing of the soft‑max temperature (β) and learning‑rate (lr)
    beta_start, beta_end = 0.5, 1e7
    lr_start, lr_end = 0.3, 0.0002

    best_a = a.copy()
    best_obj = _objective(best_a)

    it = 0
    while time.time() - start < time_limit:
        it += 1

        # map logits to simplex
        a = _softmax(x)

        conv = _conv_self_fft(a)
        max_val = conv.max()
        obj = float(2.0 * n * max_val)

        # record improvement
        if obj < best_obj - 1e-15:
            best_obj, best_a = obj, a.copy()
            print(f"[n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        # schedule (linear in time)
        t = min((time.time() - start) / time_limit, 1.0)
        beta = beta_start * np.exp(np.log(beta_end / beta_start) * t)
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t)

        # smooth‑max weights
        if beta > 5e5:  # essentially a hard max
            w = np.zeros_like(conv)
            w[int(np.argmax(conv))] = 1.0
        else:
            shifted = conv - max_val
            w = np.exp(beta * shifted)
            s = w.sum()
            if s == 0.0:
                w = np.ones_like(conv) / conv.size
            else:
                w /= s

        # gradient of smooth max w.r.t a
        g_a = 2.0 * _correlation_fft(w, a)

        # natural gradient in log‑it space
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape plateaus
        if it % 2500 == 0:
            jitter = rng.normal(scale=0.01, size=n)
            x += jitter

        # periodic restart from current best
        if it % 50000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.007, size=n)
            m.fill(0.0)
            v.fill(0.0)

    # -----------------------------------------------------------------
    # Remove near‑zero entries (helps the final refinement)
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a /= best_a.sum()
        best_obj = _objective(best_a)
        print(f"[compress] length → {best_a.size}, O={best_obj:.6f}")

    return best_a, best_obj


# ----------------------------------------------------------------------
# Targeted simulated annealing + projected gradient (stage 3)
# ----------------------------------------------------------------------


def _run_local_refinement(
    init_seq: np.ndarray,
    time_limit: float,
    seed: int = 0,
) -> tuple[np.ndarray, float]:
    """
    Hybrid local search: a fast targeted SA followed by a long
    projected‑gradient descent using the exact max‑gradient.
    Returns the best sequence and its exact objective.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    a = np.copy(init_seq)
    a = np.clip(a, 0.0, None)
    if a.sum() == 0.0:
        a = np.ones_like(a) / a.size
    else:
        a /= a.sum()
    n = a.size

    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())
    best_seq = a.copy()

    # -------------------- part 1: targeted SA -----------------------
    T = best_obj * 0.25 + 1e-9
    T_min = best_obj * 1e-6 + 1e-12
    decay = 0.996

    it = 0
    while time.time() - start < time_limit * 0.55:
        it += 1
        k_max = int(np.argmax(conv))

        # contribution of each index to the maximal autocorrelation term
        i_idx = np.arange(n)
        j_idx = k_max - i_idx
        mask = (j_idx >= 0) & (j_idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i_idx[mask]] * a[j_idx[mask]]
        total = contrib.sum()

        if total > 0.0:
            probs = contrib / total
            i = rng.choice(n, p=probs)
        else:
            i = rng.integers(n)

        # pick a destination different from i uniformly
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # move a modest fraction of mass from i to j
        delta = a[i] * rng.random() * 0.07  # up to 7 % of a[i]
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.clip(a_new, 0.0, None)
        a_new /= a_new.sum()

        conv_new = _conv_self_fft(a_new)
        obj_new = float(2.0 * n * conv_new.max())

        if obj_new < best_obj - 1e-15:
            a, conv, best_obj, best_seq = a_new, conv_new, obj_new, a_new.copy()
            print(f"[SA it={it}] new best O={best_obj:.6f}")
        else:
            # Metropolis acceptance
            if rng.random() < math.exp(-(obj_new - best_obj) / T):
                a, conv = a_new, conv_new

        if it % 3000 == 0:
            T = max(T * decay, T_min)

        if it % 7000 == 0:
            # global jitter
            noise = rng.normal(scale=0.009, size=n)
            a = np.clip(a + noise, 0.0, None)
            a /= a.sum()
            conv = _conv_self_fft(a)

    # -------------------- part 2: projected gradient -----------------
    eta0 = 0.25
    it_g = 0
    while time.time() - start < time_limit:
        it_g += 1
        conv = _conv_self_fft(a)
        k_max = int(np.argmax(conv))
        w = np.zeros_like(conv)
        w[k_max] = 1.0
        grad = 2.0 * _correlation_fft(w, a)   # exact max‑gradient

        eta = eta0 * (0.996 ** it_g)
        a_new = _project_simplex(a - eta * grad)
        obj_new = float(2.0 * n * _conv_self_fft(a_new).max())

        if obj_new < best_obj - 1e-15:
            a, best_obj, best_seq = a_new, obj_new, a_new.copy()
            print(f"[PG it={it_g}] improved O={best_obj:.6f}")
        else:
            # adapt step size if stuck
            eta0 *= 0.98
            if eta0 < 1e-7:
                break

    return best_seq, best_obj

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **_ignore,
) -> list[float]:
    """
    Search for a step‑function that minimises the evaluator's score.
    Returns a plain list of non‑negative floats.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # support lengths to explore – larger n gives more degrees of freedom
    lengths = [2000, 4000, 8000, 12000]

    best_seq = None
    best_obj = float("inf")

    for idx, n in enumerate(lengths):
        now = time.time()
        remaining = deadline - now
        if remaining <= 2.0:
            break

        # allocate more budget to the larger sizes
        if idx == 0:
            run_time = remaining * 0.45
        elif idx == len(lengths) - 1:
            run_time = remaining * 0.3
        else:
            run_time = remaining * 0.25 / (len(lengths) - idx - 1)

        # ----- stage 1 : rich analytic pre‑search -----------------
        shape_time = min(4.0, run_time * 0.08)
        init_a, init_obj = _best_initial_sequence(n, rng, shape_time)
        print(f"[init n={n}] shape‑obj={init_obj:.6f}")

        # ----- stage 2 : Adam optimisation -------------------------
        seq_opt, obj_opt = _run_adam(
            n=n,
            time_limit=run_time * 0.65,
            init_a=init_a,
            seed=seed + idx * 99991,
        )
        if obj_opt < best_obj:
            best_obj, best_seq = obj_opt, np.asarray(seq_opt, dtype=float)
            print(f"[global] new overall best O={best_obj:.6f} (n={n})")

        # ----- stage 3 : SA + projected gradient -------------------
        remaining = deadline - time.time()
        if remaining > 5.0 and best_seq is not None:
            seq_ref, obj_ref = _run_local_refinement(
                best_seq,
                time_limit=remaining,
                seed=seed + 111111 * (idx + 1),
            )
            if obj_ref < best_obj:
                best_obj, best_seq = obj_ref, np.asarray(seq_ref, dtype=float)
                print(f"[final] after SA+PG new best O={best_obj:.6f}")

        if time.time() >= deadline - 0.5:
            break

    # -----------------------------------------------------------------
    # Fallback in the unlikely event that nothing was found
    if best_seq is None or best_seq.size == 0:
        best_seq = np.ones(2000, dtype=float) / 2000.0
        best_obj = _objective(best_seq)

    # final sanity check (for diagnostics)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:          # pragma: no cover
        print("[final] evaluate_sequence failed:", exc)

    # Apply the evaluator's caps and guard against an almost‑zero sum
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    if best_seq.sum() < 0.01:
        best_seq = np.ones_like(best_seq) / best_seq.size

    return best_seq.tolist()

This implementation extends the analytic pre‑search with Beta, power‑law‑edge, and sine‑power families, allocates the bulk of the time budget to a longer Adam optimisation on the simplex, and follows up with a more exhaustive local refinement (targeted simulated‑annealing + projected gradient).  By exploring several support lengths (up to 12000) and richer initial shapes, it is expected to push the final objective below the previous 1.513709 threshold and, with the extra optimisation budget, to achieve the target ≈ 1.503.
