"""
Algorithm:
- Choose a large support size (default 80000) and initialize a 
  probability vector by discretising the exact arcsine distribution,
  which is the continuous minimiser of the max‑convolution problem.
- If the initial objective is already below the target (1.503) we return
  it directly.
- Otherwise we refine the vector on the simplex using Adam on the
  logits.  The gradient is the exact sub‑gradient of the objective
  (the hard max of the linear self‑convolution) computed by FFTs.
- The smoothing factor is gradually reduced to zero, so the optimiser
  first sees a smooth approximation and then a true sub‑gradient.
- The best candidate seen during the run is returned.
"""

import time
import math
import numpy as np

def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable softmax (sums to 1)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()

def _fft_len_for_corr(n: int) -> int:
    """Smallest power of two ≥ 3·n‑2, sufficient for linear convolution."""
    return 1 << ((3 * n - 2).bit_length())

def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Smoothed max of a*a and its gradient w.r.t. a.
    eps = max(eps_factor * max(conv), 1e‑12).
    Returns (max_conv, grad_a) where grad_a = 4·n·corr.
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    a_fft = np.fft.rfft(a, L)

    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or np.isnan(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    grad_a = 4.0 * n * corr
    return max_conv, grad_a

def _conv_max_and_hard_grad(a: np.ndarray):
    """
    Exact gradient of 2·n·max(a*a) using a hard (one‑hot) max.
    Returns (max_conv, grad_a) where grad_a = 4·n·corr with w a one‑hot.
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    a_fft = np.fft.rfft(a, L)

    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())
    m_idx = int(np.argmax(conv))

    # one‑hot weight vector of length 2·n‑1
    w = np.zeros(2 * n - 1, dtype=np.float64)
    w[m_idx] = 1.0

"""
Algorithm Overview
------------------
1. Choose a sufficiently large support size (default 100 000).  A larger
   discretisation of the arcsine distribution reduces the discretisation
   error of the continuous optimal solution, bringing the objective
   arbitrarily close to its theoretical limit (~1.5).

2. Initialise the probability vector by exact area–discretisation of the
   arcsine law:
       a_i = F((i+1)/n) - F(i/n),  where
       F(x) = (2/π)·arcsin(√x), x∈[0,1].

3. If the initial metric (computed via ``evaluate_sequence``) already

4. Otherwise we refine the vector on the simplex using Adam on the
   logits (soft‑max parametrisation).  The gradient is the exact
   sub‑gradient of 2·n·max(a∗a) computed by FFTs.  In the early phase a
   smoothed max (soft‑max) is used for stability; the smoothing factor
   decays to zero, after which a hard‑max (one‑hot) gradient is used.
   The best candidate seen during the run is kept.

5. The final sequence (as a plain Python list) is returned.  A final
   exact evaluation is printed for logging, but the platform will evaluate

def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable softmax; result sums to one."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _fft_len_for_corr(n: int) -> int:
    """Smallest power‑of‑two ≥ 3·n‑2 (sufficient for convolution/correlation)."""
    return 1 << ((3 * n - 2).bit_length())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Smoothed max of a∗a and its gradient w.r.t. a.

    Parameters
    ----------
    a : np.ndarray
        Probability vector (sum = 1, non‑negative).
    eps_factor : float
        Multiplicative factor for the smoothing constant:
        eps = max(eps_factor * max(conv), 1e‑12).

    Returns
    -------
    max_conv : float
        Maximum entry of the linear self‑convolution.
    grad_a : np.ndarray
        Gradient of 2·n·max_conv w.r.t. a (already multiplied by 4·n).
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    a_fft = np.fft.rfft(a, L)

    # linear self‑convolution
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or np.isnan(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    grad_a = 4.0 * n * corr
    return max_conv, grad_a


def _conv_max_and_hard_grad(a: np.ndarray):
    """
    Exact gradient of 2·n·max(a∗a) using a hard (one‑hot) max.

    Returns
    -------
    max_conv : float
        Maximum entry of the linear self‑convolution.
    grad_a : np.ndarray
        Gradient of 2·n·max_conv w.r.t. a (already multiplied by 4·n).
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    a_fft = np.fft.rfft(a, L)

    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())
    m_idx = int(np.argmax(conv))

    # one‑hot weight vector for the max entry
    w = np.zeros(2 * n - 1, dtype=np.float64)
    w[m_idx] = 1.0

    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    grad_a = 4.0 * n * corr
    return max_conv, grad_a
