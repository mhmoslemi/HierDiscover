"""
Propose a non‑negative step‑function that minimises
    2·n·max(a*a) / (sum(a)^2)

The algorithm
---------------
1. Choose a support length `n`.  Re‑use the length of the external
   variable `height_sequence_1` if it exists, otherwise fall back to
   `n = 20000`.

2. Generate a family of symmetric beta‑type probability vectors
   `a_α` (including the arcsine case α=0.5) by area‑discretising the
   Beta(α,α) distribution:
        a_α[i] = I_{(i+1)/n}(α,α) – I_{i/n}(α,α)
   where `I_x` is the regularised incomplete beta function.
   The best of those candidates (in terms of the cheap metric
   2·n·max(conv)) is selected as the starting point.

3. Refine the chosen vector with many Adam steps on the simplex.
   The optimisation target is a smoothed version of `max(a*a)`
   (soft‑max over the autocorrelation).  The gradient is obtained
   analytically using FFTs, and the parameters (step size, temperature)
   are slowly annealed.  The loop runs until the time budget (≈ 1000 s)
   is exhausted or the target bound 1.503 is reached.

4. Return the best sequence seen during the optimisation as a plain
   Python list.  A final call to `evaluate_sequence` is performed only
   for logging – the platform will evaluate the returned list itself.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level helpers (no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _fft_len_for_corr(n: int) -> int:
    """Smallest power‑of‑two ≥ 3·n‑2 (enough for linear conv + corr)."""
    return 1 << ((3 * n - 2).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a (via FFT)."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Smoothed max of a*a and gradient w.r.t. a.

    Parameters
    ----------
    a : np.ndarray
        Probability vector (sum = 1, non‑negative).
    eps_factor : float
        Multiplicative factor for the smoothing constant:
        eps = max(eps_factor * max(conv), 1e‑12).

    Returns
    -------
    max_conv : float
        Maximum entry of the linear self‑convolution.
    grad_a : np.ndarray
        Gradient of 2·n·smooth_max w.r.t. a (already multiplied by 4·n).
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    a_fft = np.fft.rfft(a, L)

    # linear self‑convolution (length 2·n‑1)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # soft‑max weights over the convolution entries
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv          # for numerical stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or np.isnan(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # cross‑correlation a ⋆ w (length 3·n‑2, we need first n entries)
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # gradient (the factor 4·n comes from the derivative of 2·n·max)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a

    # ------------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)
    x0 = idx / n
    x1 = (idx + 1) / n

    # ------------------------------------------------------------------
    # 2. Build a few symmetric beta‑type initialisations
    # ------------------------------------------------------------------
    # The regularised incomplete beta is provided by scipy.special.betainc.
    # If scipy is unavailable we simply fall back to the arcsine distribution.
    use_beta = False
    try:
        from scipy.special import betainc  # noqa: F401
        use_beta = True
    except Exception:
        print("[Warning] scipy not available – will use only the arcsine init.")

    # List of α‑values to try (α = 0.5 corresponds to the arcsine law)
    alphas = [0.10, 0.15, 0.20, 0.25, 0.30, 0.35, 0.40, 0.45, 0.50]
    best_score = np.inf
    best_a = None

    # Helper to evaluate the cheap bound (2·n·max(conv)) without calling
    # evaluate_sequence (which does extra clamping and a full convolution).
    def cheap_score_of(vec):
        return 2.0 * n * _conv_max(vec)

    for α in alphas:
        if use_beta:
            # area discretisation of Beta(α,α)
            a = betainc(α, α, x1) - betainc(α, α, x0)
        else:
            # fall back to the exact arcsine CDF (α = 0.5)
            def arcsine_cdf(x):
                return (2.0 / np.pi) * np.arcsin(np.sqrt(np.clip(x, 0.0, 1.0)))
            a = arcsine_cdf(x1) - arcsine_cdf(x0)

        # Numerical safety – enforce non‑negativity and renormalise
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0 or np.isnan(s):
            continue
        a /= s

        score = cheap_score_of(a)
        print(f"[Init α={α:.2f}] cheap_score = {score:.6f}")

        if score < best_score:
            best_score = score
            best_a = a.copy()

    # Sanity – if something went wrong fall back to the arcsine init
    if best_a is None:
        print("[Warning] No valid initialisation – falling back to arcsine.")
        def arcsine_cdf(x):
            return (2.0 / np.pi) * np.arcsin(np.sqrt(np.clip(x, 0.0, 1.0)))
        best_a = arcsine_cdf(x1) - arcsine_cdf(x0)
        best_a = np.maximum(best_a, 0.0)
        best_a /= best_a.sum()
        best_score = cheap_score_of(best_a)

    # ------------------------------------------------------------------
    # 3. Adam refinement on the simplex (log‑its parametrisation)
    # ------------------------------------------------------------------
    # We work with logits `z` such that a = softmax(z).
    z = np.log(best_a + 1e-16)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    # Adam hyper‑parameters
    base_alpha = 0.7          # initial step magnitude for logits
    eps_factor = 0.15         # temperature for the soft‑max over the convolution
    eps_factor_min = 5e-5
    eps_decay = 0.9992

    it = 0
    max_iters = 1_000_000
    log_interval = 2000

    best_seq = best_a.copy()
    best_score = cheap_score_of(best_seq)

    # Helper for intermittent logging
    def _log(iters, score):
        elapsed = time.time() - start_time
        print(f"[Iter {iters:6d}] best_score = {score:.6f}  elapsed = {elapsed:.1f}s")

    while time.time() < deadline and it < max_iters:
        it += 1

        # Current probability vector (always sums to 1)
        a_cur = _softmax(z)

        # Smoothed max‑convolution and its gradient
        max_conv, grad_a = _conv_max_and_grad(a_cur, eps_factor)

        cheap = 2.0 * n * max_conv

        # Keep the best candidate so far
        if cheap < best_score:
            best_score = cheap
            best_seq = a_cur.copy()
            if it % log_interval == 0 or cheap < 1.02:
                _log(it, best_score)

            # Early exit if we have already passed the target bound
            if best_score < 1.503:
                print("[Info] Target bound reached – stopping early.")
                break

        # Natural‑gradient step on the simplex (mirror‑descent on logits)
        mu = np.dot(grad_a, a_cur)          # 〈∇,a〉
        grad_z = a_cur * (grad_a - mu)      # ∇_z J

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # Inverse‑sqrt decay of the learning‑rate
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step = - alpha_t * m_hat / (np.sqrt(v_hat) + eps_adam)

        z = z + step
        # Clip logits to avoid numerical overflow / underflow
        z = np.clip(z, -80.0, 80.0)

        # Anneal the temperature for the soft‑max over the convolution
        eps_factor = max(eps_factor_min, eps_factor * eps_decay)

        # Occasionally inject a tiny random jitter to escape flat plateaus
        if it % 5000 == 0 and np.random.rand() < 0.1:
            z += 0.01 * np.random.randn(n)

    # ------------------------------------------------------------------
    # 4. Return the best sequence (plain Python list)
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()

    # Optional exact evaluation for logging purposes only.
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[Info] Final exact score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print(f"[Warning] Evaluation error (ignored): {exc}")

    return final_seq
