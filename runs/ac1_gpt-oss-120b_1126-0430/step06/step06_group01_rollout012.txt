def propose_candidate(seed=42, budget_s=1000):

    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0

    # Determine n (external or default)
    n = determine_n()

    # Helper: compute arcsine distribution a0
    a0 = arcsine_bins(n)

    # Prepare best
    best_a = a0.copy()
    best_score = 2.0*n*_conv_max(best_a)  # initial

    # Function to run Adam optimization from given start z
    def run_adam(z_start, max_iters, eps_factor_start, eps_factor_min, eps_decay):
        nonlocal best_a, best_score, deadline
        z = z_start.copy()
        m = np.zeros_like(z)
        v = np.zeros_like(z)
        beta1 = 0.9; beta2 = 0.999
        eps_adam = 1e-8
        base_alpha = 0.8

        eps_factor = eps_factor_start
        it = 0
        while time.time() < deadline and it < max_iters:
            it += 1
            a_cur = _softmax(z)
            max_conv, grad_a = _conv_max_and_grad(a_cur, eps_factor)
            score = 2.0 * n * max_conv
            if score < best_score:
                best_score = score
                best_a = a_cur.copy()
                # maybe print improvement
            
            mu = np.dot(grad_a, a_cur)
            grad_z = a_cur * (grad_a - mu)

            # Adam update
            m = beta1*m + (1-beta1)*grad_z
            v = beta2*v + (1-beta2)*(grad_z**2)
            m_hat = m / (1-beta1**(it))
            v_hat = v / (1-beta2**(it))
            alpha_t = base_alpha / (math.sqrt(it) + 1.0)
            step = - alpha_t * m_hat / (np.sqrt(v_hat) + eps_adam)
            z = z + step
            z = np.clip(z, -60.0, 60.0)
            # decay epsilon
            eps_factor = max(eps_factor_min, eps_factor * eps_decay)

            # jitter occasionally
            if it % 5000 == 0 and np.random.rand() < 0.07:
                z += 0.004 * np.random.randn(n)

        return

    # Perform a few short restarts with random initializations
    num_restarts = 5
    for r in range(num_restarts):
        # jitter arcsine logits: log(a0) + sigma*rand
        sigma = 0.5
        z0 = _logits_from_distribution(a0) + sigma * np.random.randn(n)
        max_iters_start = 3000
        eps_start = 0.12
        eps_min = 5e-4
        eps_decay = 0.9995
        run_adam(z0, max_iters_start, eps_start, eps_min, eps_decay)

    # Fine-tune the best solution for longer
    # Convert best_a to logits.
    z_best = _logits_from_distribution(best_a)
    remaining_iters = 50000
    run_adam(z_best, remaining_iters, eps_factor_start=5e-3, eps_factor_min=5e-5, eps_decay=0.9998)

    final_seq = best_a.tolist()

        print(f"Exact evaluation: {exact_score:.8f}")
    except Exception as e:
        print("Exact evaluation failed:", e)

    return final_seq

#!/usr/bin/env python
"""
Mirror‑descent / Adam optimisation of a step‑function on the simplex.

The algorithm:
1. Initialise a probability vector with the exact discrete arcsine law
   (the continuous arcsine density has a perfectly flat self‑convolution).
2. Perform a number of short Adam runs from slightly perturbed initial
   points – this helps to escape tiny local minima and gives several
   candidate solutions.
3. Take the best candidate and run a long, low‑temperature Adam
   optimisation where the smoothing constant for the “soft‑max” of the
   convolution is gradually reduced.  This fine‑tunes the discrete
   distribution and drives the objective
        2·n·max(a*a) / (sum a)²
   (scale‑invariant) down as far as the time budget permits.
4. Return the best sequence found (a plain Python list of non‑negative
   floats).  An exact evaluation with ``evaluate_sequence`` is printed for
   reference.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
# Basic numerical helpers (no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Logits that map to a via the soft‑max."""
    return np.log(a + eps)


def _fft_len_for_conv(n: int) -> int:
    """Smallest power‑of‑two ≥ 2·n‑1 (sufficient for linear self‑convolution)."""
    return 1 << ((2 * n - 1).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a (no gradient)."""
    n = a.shape[0]
    L = _fft_len_for_conv(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Compute the (unsmoothed) maximum of a*a together with a smoothed
    gradient of 2·n·max(a*a).

    Parameters
    ----------
    a : np.ndarray
        Probability vector (non‑negative, sum = 1).
    eps_factor : float
        Multiplicative factor for the smoothing constant:
        eps = eps_factor * max(conv).

    Returns
    -------
    max_conv : float
        Maximum entry of the self‑convolution.
    grad_a : np.ndarray
        Gradient of 4·n·smooth_max with respect to a (already multiplied
        by 4·n).  This serves as an approximate gradient of the original
        objective 2·n·max_conv.
    """
    n = a.shape[0]
    L = _fft_len_for_conv(n)

    a_fft = np.fft.rfft(a, L)

    # linear self‑convolution (length 2·n‑1)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # smoothed soft‑max weights
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv      # improve stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum

    # cross‑correlation a ⋆ w (linear, length 3·n‑2)
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # gradient (4·n multiplied)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a


def _arcsine_bins(n: int) -> np.ndarray:
    """
    Exact discretisation of the arcsine law on a uniform grid.
    Returns a probability vector of length n (sum = 1).
    """
    i = np.arange(n, dtype=np.float64)
    x0 = i / n
    x1 = (i + 1) / n

    sqrt0 = np.sqrt(np.clip(x0, 0.0, 1.0))
    sqrt1 = np.sqrt(np.clip(x1, 0.0, 1.0))
    F0 = (2.0 / np.pi) * np.arcsin(sqrt0)
    F1 = (2.0 / np.pi) * np.arcsin(sqrt1)

    a = F1 - F0
    a_sum = a.sum()
    if not np.isfinite(a_sum) or a_sum <= 0.0:
        raise RuntimeError("Failed to generate arcsine bins.")
    a /= a_sum
    return a


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (mirror‑descent formulation)
# ----------------------------------------------------------------------


def run_adam(
    z_start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int,
    eps_factor_start: float,
    eps_factor_min: float,
    eps_decay: float,
    base_alpha: float = 0.8,
    verbose: bool = False,
):
    """
    Mirror‑descent (Adam) optimisation of the objective
    ``2·n·max(a*a)`` with a simplex constraint enforced via the soft‑max
    parametrisation.

    Parameters
    ----------
    z_start : np.ndarray
        Starting logits.
    n : int
        Length of the sequence.
    deadline : float
        Unix timestamp after which the optimisation must stop.
    max_iters : int
        Upper bound on the number of iterations (actual count may be lower
        because of the time limit).
    eps_factor_start, eps_factor_min, eps_decay : float
        Parameters controlling the smoothing constant for the soft‑max of
        the convolution (see ``_conv_max_and_grad``).
    base_alpha : float
        Base step‑size for the Adam schedule.
    verbose : bool
        If true, prints progress messages every 1000 iterations.

    Returns
    -------
    best_a : np.ndarray
        Probability vector that achieved the lowest objective value during
        the run.
    best_score : float
        Corresponding value of ``2·n·max(a*a)``.
    """
    z = z_start.copy()
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    best_a = None
    best_score = np.inf
    it = 0
    eps_factor = eps_factor_start

    while time.time() < deadline and it < max_iters:
        it += 1

        a = _softmax(z)                     # probability vector, sum=1
        max_conv, grad_a = _conv_max_and_grad(a, eps_factor)

        score = 2.0 * n * max_conv           # cheap objective (scale‑invariant)

        if score < best_score:
            best_score = score
            best_a = a.copy()
            if verbose and (it % 1000 == 0):
                print(f"Iter {it:6d}: improved score = {best_score:.8f}")

        # Mirror‑descent gradient on the logits (soft‑max parametrisation)
        mu = np.dot(grad_a, a)
        grad_z = a * (grad_a - mu)

        # Adam bias‑corrected update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # Inverse‑sqrt step‑size schedule
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step = -alpha_t * m_hat / (np.sqrt(v_hat) + eps_adam)
        z = z + step

        # Keep logits bounded (prevent overflow in exp)
        z = np.clip(z, -60.0, 60.0)

        # Gradually tighten the smoothing constant
        eps_factor = max(eps_factor_min, eps_factor * eps_decay)

        # Tiny random jitter every few thousand steps (helps escape plateaus)
        if it % 5000 == 0 and np.random.rand() < 0.07:
            z += 0.004 * np.random.randn(n)

    # Fallback in the unlikely case that no improvement was recorded
    if best_a is None:
        a = _softmax(z)
        best_a = a
        best_score = 2.0 * n * _conv_max(best_a)

    return best_a, best_score
