def _project_onto_simplex(v):
    """Project vector v onto probability simplex (nonnegative, sum=1)."""
    # reference: Duchi et al 2008.
    # v is array.
    n = v.shape[0]
    u = np.sort(v)[::-1]  # descending sort
    cssv = np.cumsum(u) - 1
    # find rho = max {j | u_j > cssv_j / (j+1)}
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _conv_max_and_grad_exact(a):
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[:2*n - 1]
    max_conv = float(conv.max())
    # compute mask: entries within tolerance of max
    tol = max(1e-12, max_conv * 1e-12) # relative?
    mask = (conv >= max_conv - tol).astype(np.float64)
    # do FFT of mask
    mask_fft = np.fft.rfft(mask, L)
    corr = np.fft.irfft(np.conj(a_fft) * mask_fft, L)[:n]
    grad = 4.0 * n * corr
    return max_conv, grad

"""
Propose a step‑function (non‑negative sequence) that minimises
the metric 2·n·max(a*a)/(sum(a)^2).

Algorithm
---------
1. Choose the support length ``n``.  Re‑use the length of the externally
   provided ``height_sequence_1`` if it exists, otherwise fall back to
   ``n = 20000``.

2. Build a small library of candidate distributions:
   * the exact arcsine distribution (the known continuous optimum);
   * the uniform distribution;
   * a family of symmetric Beta(α,α) densities (α sampled on a fine grid);
   * a few simple convex mixtures of arcsine and uniform.
   Every candidate is normalised to have sum = 1.

3. Evaluate each candidate by a *cheap* objective
      cheap = 2·n·max(conv(a,a))
   (the full convolution is obtained with a single FFT).  The best
   candidate from this grid search is taken as the starting point.

4. Refine the starter with a projected‑gradient/Adam optimiser that
   works **directly on the true max‑convolution** (no soft‑max
   smoothing).  The gradient of ``max(conv)`` is a sub‑gradient:
      ∇max = 2·(a ⋆ 𝟙_{max})   (cross‑correlation with the indicator of
      the maximal convolution entry).  Using the FFT we obtain this
   gradient in O(n log n) time.

   The parameters are kept on the probability simplex via a
   soft‑max (mirrored) parametrisation; Adam’s adaptive step size and
   a 1/√t decay give stable progress.

5. Keep the best sequence seen during the optimisation, log progress,
   and stop as soon as the target score 1.503 is reached or when the
   time budget (≈ 1000 s) is exhausted.

The function returns the best sequence as a plain Python list; the
platform will evaluate it with ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from scipy import special  # for the Beta CDF

# ----------------------------------------------------------------------
# Helper functions – must be top‑level (no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max; returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _fft_len_for_corr(n: int) -> int:
    """
    Smallest power‑of‑two ≥ 3·n‑2.
    Sufficient for the linear self‑convolution (length 2·n‑1) and the
    correlation (length 3·n‑2) needed for the gradient.
    """
    return 1 << ((3 * n - 2).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a (via FFT)."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad_exact(a: np.ndarray):
    """
    Exact max of a*a and gradient w.r.t. a (sub‑gradient of the max).

    Returns
    -------
    max_conv : float
        Maximum entry of the linear self‑convolution.
    grad_a : np.ndarray
        Gradient of   J(a)=2·n·max_conv   w.r.t. a (already multiplied by 1).
        The quantity needed for mirror‑descent is ``grad_a``.
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    # FFT of a
    a_fft = np.fft.rfft(a, L)

    # Linear self‑convolution (length 2·n‑1)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # Build mask of indices attaining the maximum (within a tiny tolerance)
    tol = max(1e-14, max_conv * 1e-12)
    mask = (conv >= max_conv - tol).astype(np.float64)

    # FFT of the mask
    mask_fft = np.fft.rfft(mask, L)

    # Cross‑correlation a ⋆ mask (first n entries)
    corr = np.fft.irfft(np.conj(a_fft) * mask_fft, L)[:n]

    # Gradient of 2·n·max_conv w.r.t. a: 4·n·corr (derived in the analysis)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ_i x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that minimises the evaluation metric.
    The optimiser runs for up to ``budget_s`` seconds and returns the best
    sequence it discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    # Keep a safety margin for final logging / cleanup.
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # 0. Choose the length of the sequence
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
        print(f"[Info] Reusing external length: n = {n}")
    except Exception:
        n = 20000
        print(f"[Info] No external length – using default n = {n}")

    # ------------------------------------------------------------------
    # 1. Build a small library of candidate distributions
    # ------------------------------------------------------------------
    candidates = []

    # ---- arcsine distribution (exact area discretisation) ----
    idx = np.arange(n, dtype=np.float64)
    x0 = idx / n
    x1 = (idx + 1) / n

    def arcsine_cdf(x):
        # (2/π)·arcsin(√x)  ; clip to avoid rounding errors at the boundaries.
        return (2.0 / np.pi) * np.arcsin(np.sqrt(np.clip(x, 0.0, 1.0)))

    a_arc = arcsine_cdf(x1) - arcsine_cdf(x0)
    a_arc /= a_arc.sum()
    candidates.append(a_arc)

    # ---- uniform distribution ----
    a_uni = np.full(n, 1.0 / n, dtype=np.float64)
    candidates.append(a_uni)

    # ---- symmetric Beta(α,α) family (α ∈ [0.12, 2.0]) ----
    # The CDF of Beta(α,α) is the regularised incomplete beta function.
    alphas = np.concatenate(
        [
            np.linspace(0.12, 0.48, 13),   # dense region around the known optimum
            np.linspace(0.5, 2.0, 10),
        ]
    )
    xx = np.arange(n + 1, dtype=np.float64) / n   # points 0,…,1
    for a in alphas:
        # Using scipy.special.betainc for the CDF
        cdf = special.betainc(a, a, xx)
        pmf = cdf[1:] - cdf[:-1]
        s = pmf.sum()
        if s == 0.0 or np.isnan(s):
            continue
        pmf /= s
        candidates.append(pmf)

    # ---- simple mixtures of arcsine and uniform (λ ∈ [0, 0.3]) ----
    for lam in np.linspace(0.0, 0.30, 7):
        mix = (1.0 - lam) * a_arc + lam * a_uni
        mix /= mix.sum()
        candidates.append(mix)

    # ------------------------------------------------------------------
    # 2. Grid‑search the candidates (cheap objective)
    # ------------------------------------------------------------------
    best_a = None
    best_score = np.inf

    for idx_c, a in enumerate(candidates):
        if time.time() >= deadline:
            break
        max_conv = _conv_max(a)
        cheap = 2.0 * n * max_conv
        if cheap < best_score:
            best_score = cheap
            best_a = a.copy()
            print(f"[Grid] Candidate {idx_c:02d}: new best cheap = {best_score:.6f}")

    if best_a is None:
        # Fallback – should never happen.
        best_a = a_uni.copy()
        best_score = 2.0 * n * _conv_max(best_a)

    print(f"[Info] Initial best score after grid search: {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Refine with Adam on the simplex (exact max‑gradient)
    # ------------------------------------------------------------------
    # We optimise the *logits* z such that a = softmax(z).  This keeps the
    # iterate on the probability simplex automatically.
    z = np.log(best_a + 1e-16)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_alpha = 0.8          # initial step‑size for the logits
    it = 0
    max_iters = 500_000

    # Helper for logging
    log_interval = 5000

    while time.time() < deadline and it < max_iters:
        it += 1

        # Current probability vector (always sums to 1)
        a_cur = _softmax(z)

        # Exact max‑convolution and its sub‑gradient
        max_conv, grad_a = _conv_max_and_grad_exact(a_cur)

        cheap = 2.0 * n * max_conv

        # Record the best candidate seen so far
        if cheap < best_score:
            best_score = cheap
            best_a = a_cur.copy()
            if it % log_interval == 0 or cheap < 1.51:
                elapsed = time.time() - start_time
                print(f"[Iter {it:6d}] best_score = {best_score:.6f}  elapsed = {elapsed:.1f}s")

        # Natural (mirror‑descent) gradient on the logits
        mu = np.dot(grad_a, a_cur)           # 〈∇J, a〉
        grad_z = a_cur * (grad_a - mu)       # ∇_z J

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # Inverse‑sqrt decay of the learning‑rate
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step = -alpha_t * m_hat / (np.sqrt(v_hat) + eps_adam)

        z = z + step
        # Clamp logits to keep the exponentials numerically safe
        z = np.clip(z, -80.0, 80.0)

        # Occasional jitter to escape tiny plateaus
        if it % 10000 == 0 and np.random.rand() < 0.2:
            z += 0.01 * np.random.randn(n)

        # Early exit if target already achieved
        if best_score < 1.503 and it > 5000:
            print("[Info] Target score reached – terminating early.")
            break

    # ------------------------------------------------------------------
    # 4. Return the best sequence (plain Python list)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()

    # Optional exact evaluation – purely for logging.
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[Info] Final exact score = {final_score:.6f}")
    except Exception as exc:
        print(f"[Warning] Evaluation error (ignored): {exc}")

    return final_seq
