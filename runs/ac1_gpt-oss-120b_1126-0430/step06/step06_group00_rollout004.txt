    start_time = time.time()
    deadline = start_time + budget_s - 2.0

    try:

        # choose larger n

        else:

        n = 65536
        print(f"[Info] Using default n = {n}")

    # arcsine init
    idx = np.arange(n, dtype=np.float64)
    x0 = idx / n
    x1 = (idx + 1) / n
    def arcsine_cdf(x):
        return (2.0 / np.pi) * np.arcsin(np.sqrt(np.clip(x, 0.0, 1.0)))

    a = arcsine_cdf(x1) - arcsine_cdf(x0)
    a_sum = a.sum()
    a /= a_sum

    # compute initial best
    best_a = a.copy()
    best_score = 2.0 * n * _conv_max(best_a)

    # parameterization via logits
    z = np.log(best_a + 1e-16)
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta2 = 0.999
    eps_adam = 1e-8
    base_alpha = 0.6

    it = 0
    max_iters = 500_000

    # periodic logging
    last_log = start_time

    while time.time() < deadline and it < max_iters:
        it += 1

        # get probabilities
        a_cur = _softmax(z)

        # compute convolution and max
        # use our own conv function
        conv = _linear_conv(a_cur) # we need to implement _linear_conv returning full conv array

        max_conv = float(conv.max())
        cheap_score = 2.0 * n * max_conv

        # keep best
        if cheap_score < best_score:
            best_score = cheap_score
            best_a = a_cur.copy()

        # compute mask of max entries (with tolerance)
        tol = max_conv * 1e-12 + 1e-16
        mask = (conv >= max_conv - tol).astype(np.float64)

        # gradient of max (convolution of a with mask)
        # compute correlation a ⋆ mask
        # we can compute via FFT similarly to _conv_max_and_grad but with mask variable
        # We want corr_i = sum_{j} a_j * mask_{i+j}
        # Use fft of mask

"""
Candidate generator for the step‑function minimisation task.

Algorithm
---------
1. Initialisation  
   • If a global variable `height_sequence_1` is present, it is taken as a
     starting point (normalised to sum = 1).  
   • Otherwise the exact discretisation of the arcsine distribution
     (the known continuous optimum) is used:
       a_i = F((i+1)/n) – F(i/n) ,   F(x) = (2/π)·arcsin(√x).

2. Optimisation on the simplex  
   • The vector is re‑parameterised by logits `z` (soft‑max) so that
     `a = softmax(z)` is always non‑negative and sums to 1.
   • At each iteration the linear self‑convolution `c = a∗a` is computed
     with a single FFT (length chosen as the next power‑of‑two ≥ 3·n‑2).
   • The exact sub‑gradient of the objective  
        J(a) = 2·n·max(c)  
     is  
        ∇_a J = 4·n·(a ⋆ m) , m_k = 1 if c_k = max(c) else 0,
     where “⋆” is the (linear) cross‑correlation.  Both the
     convolution and the correlation are obtained via FFTs.
   • Natural‑gradient updates of the logits are performed with Adam
     (bias‑corrected first and second moments).  The step size decays
     like 1/√t and logits are kept in a safe range.
   • Small random jitter is added every few thousand steps to escape
     plateaus.

3. The best sequence seen during the run (lowest cheap score
   `2·n·max(c)`) is kept and finally returned as a plain Python list.
   The platform will evaluate the returned sequence with
   `evaluate_sequence`.

The whole optimisation respects the 1000 s budget; it stops early
once the time limit is reached.  Larger `n` reduces discretisation
error, while the Adam‑driven natural gradient quickly drives the
objective toward the theoretical optimum ≈ 1.5.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (must be defined at top‑level)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max, returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / np.sum(e)


def _fft_len_for_corr(n: int) -> int:
    """
    Smallest power‑of‑two ≥ 3·n‑2.
    Sufficient for linear convolution (length 2·n‑1) and cross‑correlation
    (length 3·n‑2) used in the gradient computation.
    """
    return 1 << ((3 * n - 2).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a∗a (via FFT)."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def arcsine_cdf(x: np.ndarray) -> np.ndarray:
    """
    Cumulative distribution function of the arcsine law on [0,1]:
        F(x) = (2/π)·arcsin(√x) .
    The input is clipped to [0,1] to avoid NaNs.
    """
    return (2.0 / np.pi) * np.arcsin(np.sqrt(np.clip(x, 0.0, 1.0)))


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise the
    evaluation metric defined in ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for any final logging/cleanup
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # 0. Choose the length n and initialise the sequence
    # ------------------------------------------------------------------
    DEFAULT_N = 50000                     # a compromise between resolution and speed
    use_external = False
    try:
        # ``height_sequence_1`` may be provided by the platform
        ext = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if ext.ndim == 1 and ext.size > 0:
            # sanitise the external data
            ext = np.clip(ext, 0.0, 1000.0)
            s = ext.sum()
            if s >= 0.01:
                a = ext / s
                n = ext.size
                use_external = True
                print(f"[Info] Initialising from provided sequence, n = {n}")
    except Exception:
        pass

    if not use_external:
        n = DEFAULT_N
        # exact arcsine discretisation on [0,1]
        idx = np.arange(n, dtype=np.float64)
        x0 = idx / n
        x1 = (idx + 1) / n
        a = arcsine_cdf(x1) - arcsine_cdf(x0)
        s = a.sum()
        if s == 0.0 or np.isnan(s):
            raise RuntimeError("Failed to initialise a valid arcsine distribution.")
        a /= s
        print(f"[Info] Using arcsine initialisation, n = {n}")

    # ------------------------------------------------------------------
    # 1. Prepare optimisation objects
    # ------------------------------------------------------------------
    # best candidate seen so far (always normalised to sum = 1)
    best_a = a.copy()
    best_score = 2.0 * n * _conv_max(best_a)

    # logits for the exponential‑map (soft‑max) parametrisation
    z = np.log(best_a + 1e-16)
    m = np.zeros_like(z)          # 1st‑moment estimate for Adam
    v = np.zeros_like(z)          # 2nd‑moment estimate for Adam
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_alpha = 0.6              # base step‑size for logits

    # FFT length (constant throughout the run)
    L = _fft_len_for_corr(n)

    it = 0
    max_iters = 300_000           # a hard cap; the loop also stops at the time deadline

    # ------------------------------------------------------------------
    # 2. Main optimisation loop (natural‑gradient Adam on the simplex)
    # ------------------------------------------------------------------
    while time.time() < deadline and it < max_iters:
        it += 1

        # -------------------- probability vector -----------------------
        a_cur = _softmax(z)                     # shape (n,); sum(a_cur) = 1

        # -------------------- self‑convolution -------------------------
        a_fft = np.fft.rfft(a_cur, L)
        conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]   # length 2n‑1, real

        max_conv = float(conv.max())
        cheap_score = 2.0 * n * max_conv

        # -------------------- keep the best so far --------------------
        if cheap_score < best_score:
            best_score = cheap_score
            best_a = a_cur.copy()
            if it % 2000 == 0 or cheap_score < 1.02:
                elapsed = time.time() - start_time
                print(f"[Iter {it:6d}] best_score = {best_score:.6f}  elapsed = {elapsed:.1f}s")

        # -------------------- sub‑gradient of max(conv) ----------------
        # mask of entries attaining (or very close to) the maximum
        tol = max_conv * 1e-12 + 1e-15
        mask = (conv >= max_conv - tol).astype(np.float64)   # shape (2n‑1,)

        # correlation a ⋆ mask  (first n entries only)
        mask_fft = np.fft.rfft(mask, L)
        corr = np.fft.irfft(np.conj(a_fft) * mask_fft, L)[:n]   # real, length n

        # gradient of the objective  J = 2·n·max(conv)
        grad_a = 4.0 * n * corr

        # -------------------- natural gradient on logits -------------
        mu = np.dot(grad_a, a_cur)               # scalar 〈∇,a〉
        grad_z = a_cur * (grad_a - mu)           # shape (n,)

        # -------------------- Adam update ------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step = -alpha_t * m_hat / (np.sqrt(v_hat) + eps_adam)

        z = z + step
        # keep logits in a numerically safe range
        z = np.clip(z, -80.0, 80.0)

        # -------------------- occasional jitter ------------------------
        if it % 10000 == 0 and np.random.rand() < 0.07:
            z += 0.02 * np.random.randn(n)

        # -------------------- periodic logging ------------------------
        if it % 5000 == 0:
            elapsed = time.time() - start_time
            print(f"[Iter {it:6d}] best_score = {best_score:.6f}  elapsed = {elapsed:.1f}s")

        # optional very early exit if we have already well below the target
        if best_score < 1.505 and it > 5000:
            # we keep running a bit longer to see if we can push further,
            # but the loop will stop automatically when the budget expires.
            pass

    # ------------------------------------------------------------------
    # 3. Return the best sequence (as a plain Python list)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()

    # Optional: log the exact evaluation (the platform will also evaluate)
    try:
        final_score = evaluate_sequence(final_seq)   # noqa: F821 – imported by the platform
        print(f"[Info] Final exact evaluation score = {final_score:.6f}")
    except Exception as exc:
        print(f"[Warning] evaluate_sequence failed: {exc}")

    return final_seq
