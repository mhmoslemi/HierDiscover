def _project_onto_simplex(v):
    # v may be any vector; returns projection onto simplex.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

#!/usr/bin/env python
"""
Improved arcsine‑like construction with exact max‑gradient and a
parameter‑search over Beta‑distribution exponents.

Algorithm
---------

1. Choose the length `n` (either the length of the supplied
   ``height_sequence_1`` or a default of 20000).

2. Scan a one‑parameter family of symmetric Beta densities
   f_p(x) ∝ [x(1‑x)]^{-p}  (p∈[0.45,0.55]).
   For each p the exact bin probabilities are
       a_i = I_{(i+1)/n}(1-p,1-p) – I_{i/n}(1-p,1-p)
   where I_x is the regularised incomplete beta function.
   The best p gives a good initial step‑function.

3. Refine the best candidate with Adam on the simplex.
   The gradient of the objective
       J(a) = 2·n·max(a*a)   (a already normalised to sum=1)
   is computed **exactly** (no smoothing):
       * evaluate the linear self‑convolution a*a via FFT,
       * build a weight vector w that is uniform on the arg‑max indices,
       * the gradient of max(a*a) w.r.t. a is `2 * (a ⋆ w)`,
         where `⋆` is linear cross‑correlation.
         In the logit (soft‑max) parametrisation the gradient w.r.t.
         the logits is `a * (grad_a – μ)`.
   Adam’s adaptive step size, a √t decay and occasional random jitter
   keep the optimisation stable.

4. The best sequence found before the time budget expires is returned.
   A final call to ``evaluate_sequence`` prints the exact score.

The whole routine runs under the 1000 s budget, uses only NumPy and
SciPy, and respects the non‑negativity / normalisation constraints."""

import time
import math
import numpy as np
from scipy.special import betainc

# ----------------------------------------------------------------------
# FFT utilities (linear convolution length 2·n‑1)
# ----------------------------------------------------------------------


def _fft_len_for_conv(n: int) -> int:
    """Smallest power‑of‑two ≥ 2·n‑1, sufficient for a linear self‑convolution."""
    return 1 << ((2 * n - 1).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a."""
    n = a.shape[0]
    L = _fft_len_for_conv(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_hard_grad(a: np.ndarray):
    """
    Exact max of the self‑convolution and its (sub)gradient.

    Returns
    -------
    max_conv : float
        max_i (a*a)[i].
    grad_a : np.ndarray
        Gradient of 2·n·max_conv w.r.t. a (already multiplied by 2·n).
    """
    n = a.shape[0]
    L = _fft_len_for_conv(n)

    # FFT of a
    a_fft = np.fft.rfft(a, L)

    # Linear self‑convolution (length 2·n‑1)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # Build a weight vector w that is uniform on the (near) arg‑max set.
    # Use a tiny tolerance to catch several indices that are numerically equal.
    tol = max(max_conv * 1e-12, 1e-15)
    mask = conv >= max_conv - tol
    cnt = int(mask.sum())
    if cnt == 0:                     # safety fallback – should never happen
        idx = int(np.argmax(conv))
        w = np.zeros_like(conv)
        w[idx] = 1.0
    else:
        w = np.zeros_like(conv)
        w[mask] = 1.0 / cnt

    # Cross‑correlation a ⋆ w (linear, length n)
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # Gradient of 2·n·max_conv (the optimisation objective)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a


# ----------------------------------------------------------------------
# Family of symmetric Beta‑bin distributions (parameter p)
# ----------------------------------------------------------------------


def _beta_bin_distribution(n: int, p: float) -> np.ndarray:
    """
    Return the exact bin probabilities of the density
        f_p(x) ∝ [x (1‑x)]^{-p},   0 ≤ x ≤ 1,
    i.e. a_i = I_{(i+1)/n}(1‑p,1‑p) – I_{i/n}(1‑p,1‑p).

    The result is already normalised to sum = 1.
    """
    i = np.arange(n, dtype=np.float64)
    x0 = i / n
    x1 = (i + 1) / n
    a = betainc(1.0 - p, 1.0 - p, x1) - betainc(1.0 - p, 1.0 - p, x0)
    a_sum = a.sum()
    if not np.isfinite(a_sum) or a_sum <= 0.0:
        raise RuntimeError(f"Failed to generate Beta‑bins for p={p}")
    a /= a_sum
    return a

    # --------------------------------------------------------------

    best_score = float("inf")

    # grid of p values around the arcsine exponent (p=0.5)
    p_grid = np.linspace(0.45, 0.55, 81)

    for p in p_grid:
        if time.time() >= deadline:
            break
        a = _beta_bin_distribution(n, p)
        max_conv = _conv_max(a)
        score = 2.0 * n * max_conv
        if score < best_score:
            best_score = score
            best_a = a.copy()
        # Occasionally log progress
        if int(p * 1000) % 100 == 0:
            print(f"p={p:.3f} → score={score:.8f}")

    # compare with the supplied external sequence (if any)
    if external is not None:
        max_conv_ext = _conv_max(external)
        score_ext = 2.0 * n * max_conv_ext
        if score_ext < best_score:
            best_score = score_ext
            best_a = external.copy()
            print(f"External sequence gave better start: score={score_ext:.8f}")

    # fallback safety
    if best_a is None:
        raise RuntimeError("Failed to initialise a starting vector.")

    print(f"Initial best score after p‑search: {best_score:.8f}")

    # --------------------------------------------------------------
    # 2. Refine with Adam on the simplex (exact max‑gradient).
    # --------------------------------------------------------------
    # Initialise logits for the soft‑max parametrisation (guarantees sum = 1)
    z = np.log(best_a + 1e-16)
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_alpha = 0.8            # a bit larger than in the reference code
    it = 0
    max_iters = 200_000

    while time.time() < deadline and it < max_iters:
        it += 1

        # Current point on the simplex
        a_cur = np.exp(z - np.max(z))
        a_cur = a_cur / a_cur.sum()

        # Exact max‑gradient
        max_conv, grad_a = _conv_max_and_hard_grad(a_cur)
        cur_score = 2.0 * n * max_conv

        if cur_score < best_score:
            best_score = cur_score
            best_a = a_cur.copy()
            if it % 1000 == 0:
                print(f"Iter {it:6d}: improved score = {best_score:.8f}")

        # Gradient w.r.t. logits (mirrored‑gradient)
        mu = np.dot(grad_a, a_cur)
        grad_z = a_cur * (grad_a - mu)

        # Adam bias‑corrected update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # Step size decays as O(1/√t)
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step = - alpha_t * m_hat / (np.sqrt(v_hat) + eps_adam)

        z = z + step
        # Keep logits in a safe range for the exponential
        z = np.clip(z, -60.0, 60.0)

        # Small random jitter every few thousand steps to escape plateaus
        if it % 5000 == 0 and np.random.rand() < 0.07:
            z += 0.004 * np.random.randn(n)

        # Early exit if the improvement is already excellent
        if best_score < 1.02:
            print(f"Early stop – metric already below 1.02 at iter {it}")
            break

    # --------------------------------------------------------------
    # 3. Return the best sequence found.
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        exact_score = evaluate_sequence(final_seq)
        print(f"Exact evaluation (final): {exact_score:.8f}")
    except Exception as exc:
        print(f"Exact evaluation failed: {exc}")

    return final_seq
