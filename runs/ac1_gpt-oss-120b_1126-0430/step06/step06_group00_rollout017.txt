def beta_sequence(alpha, n):
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    # compute pdf values
    # Compute in log-space for stability: log_u = (alpha-1)*(log(x) + log(1-x)).
    # Then u = exp(log_u) to avoid overflow.
    log_u = (alpha - 1) * (np.log(x) + np.log(1.0 - x))
    u = np.exp(log_u)
    # clip nan for x=0/1 boundaries? x in (0,1) so okay.
    total = u.sum()
    seq = u / total
    return seq

"""
Propose a step‑function (non‑negative sequence) that minimises

        2·n·max(a*a) / (sum(a)^2)

The metric is scale‑invariant, so we keep sum(a)=1.
The algorithm consists of three stages:

1. **Parametric sweep** – the family of symmetric Beta(α,α) densities
   (mid‑point discretisation) contains the classic arcsine case (α=0.5).
   By scanning α∈[0.5, 2] on the target length we quickly locate a
   promising shape with a lower max‑convolution than the arcsine start.

2. **Adam refinement on the simplex** – the chosen distribution is turned
   into logits and optimised with Adam using the smooth‑max of the
   autocorrelation as objective.  The gradient is obtained analytically
   via FFTs (see `_conv_max_and_grad`).  The optimisation runs until the
   time budget is exhausted or no further progress is seen.

3. **Light random‑perturbation hill‑climbing** – when time permits we add
   tiny Gaussian noise to the best vector, re‑project onto the simplex
   and keep improvements.  This helps escape possible plateaus.

All helper functions are top‑level, there are no closures or file I/O,
and the final sequence is returned as a plain Python list.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT helpers, soft‑max, gradient of smoothed max)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max; returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _fft_len_for_corr(n: int) -> int:
    """
    Smallest power‑of‑two ≥ 3·n‑2.
    Sufficient for linear convolution (length 2·n‑1) and correlation
    (length 3·n‑2) used in the gradient computation.
    """
    return 1 << ((3 * n - 2).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a (via FFT)."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Smoothed max of a*a and gradient w.r.t. a.

    Parameters
    ----------
    a : np.ndarray
        Probability vector (sum = 1, non‑negative).
    eps_factor : float
        Multiplicative factor for the smoothing constant:
        eps = max(eps_factor * max(conv), 1e‑12).

    Returns
    -------
    max_conv : float
        Maximum entry of the linear self‑convolution.
    grad_a : np.ndarray
        Gradient of 2·n·smooth_max w.r.t. a (already multiplied by 4·n).
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    a_fft = np.fft.rfft(a, L)

    # linear self‑convolution (length 2·n‑1)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # soft‑max weights over the convolution entries
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv          # for numerical stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or np.isnan(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # cross‑correlation a ⋆ w (length 3·n‑2, we need first n entries)
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # gradient (the factor 4·n comes from the derivative of 2·n·max)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a


# ----------------------------------------------------------------------
# Construction of a Beta(α,α) discretised distribution
# ----------------------------------------------------------------------


def _beta_distribution(alpha: float, n: int) -> np.ndarray:
    """
    Return a length‑`n` non‑negative vector approximating the symmetric
    Beta(α,α) density (mid‑point Riemann sum, then normalised to sum = 1).
    """
    # mid‑points of the bins
    idx = np.arange(n, dtype=np.float64)
    x = (idx + 0.5) / n

    # log‑space evaluation for stability
    # pdf ∝ x^{α‑1}·(1‑x)^{α‑1}
    log_pdf = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    pdf = np.exp(log_pdf)

    # normalise to total mass 1
    total = pdf.sum()
    if total == 0.0 or np.isnan(total):
        raise RuntimeError(f"Invalid Beta PDF for α={alpha}")
    return pdf / total


# ----------------------------------------------------------------------
# Scoring utility (exact objective, no clamping)
# ----------------------------------------------------------------------


def _objective(a: np.ndarray) -> float:
    """Exact evaluation (2·n·max_conv) for a probability vector a."""
    n = a.shape[0]
    max_conv = _conv_max(a)
    return 2.0 * n * max_conv
