#!/usr/bin/env python
"""
Search for a non‑negative step function that minimises ``evaluate_sequence``.

Algorithm (≈ 1000 s budget)
----------------------------
1. **Length choice & initial interpolation** – if a previous best
   ``height_sequence_1`` exists it is interpolated to a longer length
   (up to 15000) to give the search more degrees of freedom.
2. **Broad parametric sweep** – many analytic families (exponential,
   symmetric exponential, power‑law, symmetric power‑law, beta,
   Gaussian, cosine‑tent and a few Dirichlet draws) are tried and the
   best cheap score (2·n·max(conv)) is kept.
3. **Smoothed mirror descent** – the true max‑convolution is replaced by a
   log‑sum‑exp smoothing (parameter ``γ``).  The gradient of the smooth
   objective is obtained by a single FFT‑based correlation, and an
   exponentiated‑gradient (mirror descent) update is performed on the
   simplex.  ``γ`` is increased slowly so that the algorithm first moves
   globally and later focuses on the true maximum.
4. **Permutation / simulated‑annealing local search** – starting from the
   best iterate, random swaps of entries are attempted and accepted
   either if they improve the cheap score or with an annealed probability.
5. **Exact evaluation** – the best vector found is converted to a list
   and passed to ``evaluate_sequence`` for the final score.

All helper functions are defined at module level (no closures or lambdas)
and the routine respects the supplied time budget.  It always returns a
feasible (non‑negative) sequence.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using a power‑of‑two FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa).real[: 2 * a.size - 1]
    return conv


def _cheap_score(vec: np.ndarray, n: int, fft_len: int) -> float:
    """Cheap surrogate of evaluate_sequence for sum‑normalised vectors."""
    conv = _conv_full_fft(vec, fft_len)
    return float(2.0 * n * conv.max())


def _smooth_weights(conv: np.ndarray, gamma: float) -> np.ndarray:
    """Log‑sum‑exp smoothing of the max – returns a probability vector."""
    max_c = conv.max()
    exp_term = np.exp(gamma * (conv - max_c))
    w = exp_term / exp_term.sum()
    return w


def _gradient_smooth(a: np.ndarray, w: np.ndarray,
                     n: int, corr_fft_len: int) -> np.ndarray:
    """
    Gradient of the smoothed max‑convolution objective.
    Implements  g_i = 2*sum_j w_{i+j} a_j – w_{2i} a_i .
    The sum is performed by an FFT‑based linear convolution with w reversed.
    """
    # reverse w for the convolution a * w_rev
    w_rev = w[::-1]
    # FFT of both vectors (size chosen once globally)
    fa = np.fft.fft(a, corr_fft_len)
    fw_rev = np.fft.fft(w_rev, corr_fft_len)
    conv_rev = np.fft.ifft(fa * fw_rev).real[: n + len(w_rev) - 1]
    # indices picking the needed entries (see derivation in analysis)
    idx = len(w) - 1 - np.arange(n)
    S = conv_rev[idx]                     # S_i = sum_j w_{i+j} a_j
    diag = w[2 * np.arange(n)] * a
    grad = 2.0 * S - diag
    return grad


def _exp_gradient_step(a: np.ndarray, grad: np.ndarray,
                       eta: float, eps: float = 1e-30) -> np.ndarray:
    """Exponentiated‑gradient (mirror‑descent) update on the simplex."""
    log_a = np.log(a + eps) - eta * grad
    a_new = np.exp(log_a)
    a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
# Core search routine required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that
    minimises ``evaluate_sequence``.  The routine adapts its schedule
    to the available time budget and always returns a feasible sequence.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the final exact evaluation
    deadline = start_time + budget_s - 5.0
    eps = 1e-30

    # --------------------------------------------------------------
    # 0. Choose a working length and possibly initialise from the
    #    previously found global ``height_sequence_1``.
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        prev_arr = np.asarray(globals()['height_sequence_1'], dtype=np.float64)
        prev_len = prev_arr.size
        # aim for a longer vector (up to 15000) to increase flexibility
        n = min(max(prev_len * 2, 5000), 15000)
        # linear interpolation to the new length
        if prev_len != n:
            x_old = np.linspace(0.0, 1.0, prev_len)
            x_new = np.linspace(0.0, 1.0, n)
            prev_arr = np.interp(x_new, x_old, prev_arr)
        # normalise to sum = 1
        tot = prev_arr.sum()
        if tot > 0.0:
            prev_arr /= tot
        else:
            prev_arr = None
    else:
        prev_arr = None
        n = 10000                     # reasonable default length

    # Ensure minimum size
    n = max(5000, n)

    # FFT lengths (next powers of two)
    fft_len = 1 << ((2 * n - 1).bit_length())          # for a*a
    corr_fft_len = 1 << ((3 * n - 2).bit_length())     # for correlation a * w_rev

    # --------------------------------------------------------------
    # 1. Broad parametric sweep (cheap score = 2·n·max(conv))
    # --------------------------------------------------------------
    best_a = None
    best_val = float("inf")

    # uniform baseline
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    val = _cheap_score(uniform, n, fft_len)
    best_a, best_val = uniform, val

    # previous construction (if any)
    if prev_arr is not None and prev_arr.size == n:
        val = _cheap_score(prev_arr, n, fft_len)
        if val < best_val:
            best_a, best_val = prev_arr.copy(), val

    # ----- exponential (left) ------------------------------------
    for lam in np.linspace(0.0, 10.0, 31):
        if time.time() >= deadline:
            break
        a = np.exp(-lam * np.arange(n, dtype=np.float64))
        a /= a.sum()
        val = _cheap_score(a, n, fft_len)
        if val < best_val:
            best_a, best_val = a.copy(), val

    # ----- symmetric exponential (U‑shaped) -----------------------
    for lam in np.linspace(0.0, 10.0, 31):
        if time.time() >= deadline:
            break
        i = np.arange(n, dtype=np.float64)
        a = np.exp(-lam * np.minimum(i, n - 1 - i))
        a /= a.sum()
        val = _cheap_score(a, n, fft_len)
        if val < best_val:
            best_a, best_val = a.copy(), val

    # ----- power‑law (left) --------------------------------------
    for p in np.linspace(0.01, 3.0, 41):
        if time.time() >= deadline:
            break
        a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
        a /= a.sum()
        val = _cheap_score(a, n, fft_len)
        if val < best_val:
            best_a, best_val = a.copy(), val

    # ----- symmetric power‑law (U‑shaped) ------------------------
    for p in np.linspace(0.01, 3.0, 41):
        if time.time() >= deadline:
            break
        i = np.arange(n, dtype=np.float64)
        a = np.minimum(i + 1, n - i) ** (-p)
        a /= a.sum()
        val = _cheap_score(a, n, fft_len)
        if val < best_val:
            best_a, best_val = a.copy(), val

    # ----- beta families (U‑shaped when α,β<1) --------------------
    x = np.arange(1, n + 1, dtype=np.float64) / n
    alphas = np.concatenate((
        np.linspace(0.01, 0.2, 5),
        np.linspace(0.2, 1.0, 9),
        np.linspace(1.0, 5.0, 5)
    ))
    betas = alphas.copy()
    for alpha in alphas:
        for beta in betas:
            if time.time() >= deadline:
                break
            a = (x ** alpha) * ((1.0 - x) ** beta)
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _cheap_score(a, n, fft_len)
            if val < best_val:
                best_a, best_val = a.copy(), val

    # ----- Gaussian families (centered) -------------------------
    for c in np.linspace(0.5, 12.0, 31):
        if time.time() >= deadline:
            break
        i = np.arange(n, dtype=np.float64)
        centre = (n - 1) / 2.0
        a = np.exp(-c * ((i - centre) / (n / 2.0)) ** 2)
        a_sum = a.sum()
        if a_sum == 0.0:
            continue
        a /= a_sum
        val = _cheap_score(a, n, fft_len)
        if val < best_val:
            best_a, best_val = a.copy(), val

    # ----- cosine‑tent (smooth U‑shape) -------------------------
    for power in [1.0, 2.0, 3.0]:
        if time.time() >= deadline:
            break
        i = np.arange(n, dtype=np.float64)
        a = (1.0 - np.cos(np.pi * i / (n - 1))) ** power
        a_sum = a.sum()
        if a_sum == 0.0:
            continue
        a /= a_sum
        val = _cheap_score(a, n, fft_len)
        if val < best_val:
            best_a, best_val = a.copy(), val

    # ----- random Dirichlet seeds --------------------------------
    for conc in [0.001, 0.01, 0.1, 0.5, 1.0, 2.0, 5.0]:
        if time.time() >= deadline:
            break
        g = np.random.gamma(conc, 1.0, size=n)
        a = g / g.sum()
        val = _cheap_score(a, n, fft_len)
        if val < best_val:
            best_a, best_val = a.copy(), val

    print(f"[Init] best cheap = {best_val:.6f}")

    # --------------------------------------------------------------
    # 2. Smoothed mirror descent (exponentiated gradient)
    # --------------------------------------------------------------
    a = best_a.copy()
    best_iter_a = a.copy()
    best_iter_val = best_val
    last_improve = 0
    iter_cnt = 0
    max_iters = 300_000
    eta0 = 0.4
    gamma = 5.0                     # start with a fairly smooth approximation

    while time.time() < deadline and iter_cnt < max_iters:
        iter_cnt += 1

        # full convolution and cheap score (sum is 1)
        conv = _conv_full_fft(a, fft_len)
        cheap = 2.0 * n * conv.max()

        if cheap < best_iter_val - 1e-12:
            best_iter_val = cheap
            best_iter_a = a.copy()
            last_improve = iter_cnt
            print(f"[Iter {iter_cnt:07d}] NEW BEST cheap = {cheap:.6f}")

        # smooth weights
        w = _smooth_weights(conv, gamma)

        # gradient of the smoothed objective
        grad = _gradient_smooth(a, w, n, corr_fft_len)

        # exponentiated‑gradient update
        eta = eta0 / math.sqrt(iter_cnt)
        a = _exp_gradient_step(a, grad, eta, eps)

        # adapt smoothing parameter slowly (focus on true max later)
        if iter_cnt % 20_000 == 0 and gamma < 60.0:
            gamma *= 1.2

        # occasional restart if progress stalls
        if iter_cnt - last_improve > 80_000:
            noise = np.random.rand(n) * 0.02
            a = _simplex_projection(best_iter_a + noise)
            last_improve = iter_cnt
            print(f"[Iter {iter_cnt:07d}] Restart after stagnation.")

    # --------------------------------------------------------------
    # 3. Permutation / swap local search (simulated‑annealing style)
    # --------------------------------------------------------------
    a = best_iter_a.copy()
    best_swap_a = a.copy()
    best_swap_val = best_iter_val
    temperature = 1.0
    cooling = 0.99995

    while time.time() < deadline:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # propose a swap
        a[i], a[j] = a[j], a[i]

        conv = _conv_full_fft(a, fft_len)
        cheap = 2.0 * n * conv.max()
        accept = False
        if cheap < best_swap_val - 1e-12:
            accept = True
        else:
            delta = cheap - best_swap_val
            prob = math.exp(-delta / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            if cheap < best_swap_val - 1e-12:
                best_swap_val = cheap
                best_swap_a = a.copy()
                print(f"[Swap] NEW BEST cheap = {cheap:.6f}")
        else:
            # revert swap
            a[i], a[j] = a[j], a[i]

        temperature = max(temperature * cooling, 1e-6)

    # --------------------------------------------------------------
    # 4. Final exact evaluation
    # --------------------------------------------------------------
    final_seq = best_swap_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:          # defensive – should never happen
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    return final_seq
