#!/usr/bin/env python
"""
Non‑negative step‑function optimiser (improved).

The optimiser works in three stages:

1️⃣  **Parametric warm‑start** – a fine golden‑section search over the
    symmetric product‑power‑law family  a_i ∝ ((i+1)*(n‑i))^{‑p}
    yields a distribution whose self‑convolution is already close to flat.
    The search is performed on a modest support (≈ 5 000 points) so that it
    costs only a few seconds.

2️⃣  **Mirror‑descent (Adam) refinement** – the distribution is written as a
    soft‑max of logits.  A smoothed‑max gradient of `max(a*a)` is obtained
    via the exponential weights trick and fed to Adam.  The smoothing
    constant ε = eps_factor·max_conv is gradually shrunk to the true
    sub‑gradient.  The optimiser runs until the time budget is exhausted,
    occasionally jittering the logits to escape plateaux and up‑sampling
    the support (first to ~8 000, then to ~16 000 points) to gain extra
    degrees of freedom.

3️⃣  **Local random‑swap polishing** – after the Adam phase a cheap
    random‑swap hill‑climber makes tiny mass transfers between two
    coordinates; a move is kept only if it lowers the exact objective.
    Because the support is modest this stage can afford many trials.

The best sequence seen throughout the run is finally evaluated with the
provided `evaluate_sequence` routine and returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures or lambdas
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max, returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    s = e.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / z.size)
    return e / s


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it that reproduces `a` after a soft‑max."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """Smallest power‑of‑two ≥ 3·n‑2 (enough for all needed convolutions)."""
    return 1 << ((3 * n - 2).bit_length())


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    s = new_a.sum()
    if s == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= s
    return new_a


def _sym_product_power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ ((i+1)*(n‑i))^{‑p}, symmetric, normalised."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) * (n - i)) ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{‑p}, normalised."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _exp_decay_distribution(n: int, lam: float) -> np.ndarray:
    """a_i ∝ e^{‑lam·i}+e^{‑lam·(n‑1‑i)}, symmetric exponential."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _logistic_distribution(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ 1/(1+e^{α·(i‑mid)}), normalised; left‑heavy."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = 1.0 / (1.0 + np.exp(alpha * (i - mid)))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Returns (max_conv, grad_a) where

        grad_a = ∇_a [ 2·n·smooth_max(a*a) ]

    `eps_factor` multiplies the smoothing constant ε = eps_factor·max_conv.
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    # FFT of a (zero‑padded)
    a_fft = np.fft.rfft(a, L)

    # Linear self‑convolution a*a
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # Smoothed soft‑max weights
    eps = max(max_conv * eps_factor, 1e-15)
    shifted = conv - max_conv                                 # ≤ 0
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # Correlation a ⋆ w
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # Gradient (already multiplied by 4·n)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a


def _sym_product_power_law_score(p: float, n: int) -> float:
    """Cheap score = 2·n·max(a*a) for a given p."""
    a = _sym_product_power_law_distribution(n, p)
    return 2.0 * n * _conv_max(a)


def _optimize_sym_product_power_law(n: int, lo: float = 0.35, hi: float = 0.80,
                                   iters: int = 30):
    """
    Golden‑section search for the exponent p that minimises the cheap score.
    Returns a tuple (best_a, best_score, best_p).
    """
    phi = (1 + 5 ** 0.5) / 2
    invphi = 1 / phi
    invphi2 = invphi * invphi

    a = lo
    b = hi
    c = b - invphi * (b - a)
    d = a + invphi * (b - a)
    fc = _sym_product_power_law_score(c, n)
    fd = _sym_product_power_law_score(d, n)

    for _ in range(iters):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - invphi * (b - a)
            fc = _sym_product_power_law_score(c, n)
        else:
            a = c
            c = d
            fc = fd
            d = a + invphi * (b - a)
            fd = _sym_product_power_law_score(d, n)

    best_p = (a + b) / 2.0
    best_a = _sym_product_power_law_distribution(n, best_p)
    best_score = 2.0 * n * _conv_max(best_a)
    return best_a, best_score, best_p

        if n < max_n and (it % 250_000 == 0) and (time.time() + 30.0 < deadline):
            # Target a larger support (≈ 1.6× the current size, capped at max_n)
            new_n = min(max_n, int(n * 1.6))
            print(f"[Iter {it}] Upsampling support: {n} → {new_n}")
            best_seq = _interpolate_distribution(best_seq, new_n)
            z = _logits_from_distribution(best_seq) + 0.007 * np.random.randn(new_n)
            m = np.zeros_like(z)
            v = np.zeros_like(z)
            n = new_n
            # recompute best_score on the new grid
            best_score = 2.0 * n * _conv_max(best_seq)
            # reset counters after a structural change
            it = 0
            last_improve_iter = 0
            eps_factor = 0.20
            continue   # skip the rest of the loop and start fresh

    # --------------------------------------------------------------
    # 3. Local random‑swap polishing (cheap hill‑climber)
    # --------------------------------------------------------------
    # The current best_seq lives on the latest support size.
    a = best_seq.copy()
    current_score = 2.0 * n * _conv_max(a)

    swap_trials = 0
    max_swap_trials = 12_000
    max_step = 0.02               # initial mass‑transfer magnitude
    decay_per_trial = 0.9995

    while time.time() < deadline and swap_trials < max_swap_trials:
        swap_trials += 1
        # pick two distinct indices
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # draw a small amount of mass from i to j
        delta = max_step * a[i] * np.random.rand()
        if delta <= 0.0:
            continue
        a_trial = a.copy()
        a_trial[i] -= delta
        a_trial[j] += delta
        # ensure non‑negativity (since delta ≤ a[i] it holds)
        trial_score = 2.0 * n * _conv_max(a_trial)
        if trial_score < current_score - 1e-14:
            a = a_trial
            current_score = trial_score
            if trial_score < best_score:
                best_score = trial_score
                best_seq = a.copy()
        # shrink the step size gradually
        max_step *= decay_per_trial

    # --------------------------------------------------------------
    # 4. Final exact evaluation with the provided `evaluate_sequence`
    # --------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:          # pragma: no cover
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # --------------------------------------------------------------
    # 5. Return the best sequence discovered.
    # --------------------------------------------------------------
    return final_seq
