#!/usr/bin/env python
"""
Subgradient mirror‑descent optimiser for the step‑function problem.

The optimiser works on a simplex (soft‑max parametrisation) and
uses a *exact* sub‑gradient of the non‑smooth objective
`2·n·max(a*a)` (the max of the linear self‑convolution).  Only a
single FFT is required per iteration; the gradient of the active
convolution entry is obtained by a simple slice‑wise shift, which
makes the inner loop much faster than the smoothed‑max version.
The algorithm proceeds in three stages, gradually increasing the
support size (2000 → 8000 → 20000).  At each stage a few
hand‑crafted families (power‑law, symmetric product‑law,
exponential, logistic) are scanned to obtain a good initial
distribution, then an Adam‑style mirror‑descent loop is run for
as many iterations as the remaining time permits.  Periodic jitter
restarts and up‑sampling allow the optimiser to escape plateaux.
The best sequence found is finally evaluated with the provided
`evaluate_sequence` routine and returned.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top‑level, no closures or lambdas
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max, returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    s = e.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / z.size)
    return e / s


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it that reproduces `a` after a soft‑max."""
    # a may contain zeros – add eps to avoid log(0)
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """Smallest power‑of‑two ≥ 3·n‑2 (large enough for all convolutions)."""
    return 1 << ((3 * n - 2).bit_length())


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    s = new_a.sum()
    if s == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= s
    return new_a


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{‑p}, normalised."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _sym_product_power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ ((i+1)*(n‑i))^{‑p}, symmetric, normalised."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) * (n - i)) ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _exp_decay_distribution(n: int, lam: float) -> np.ndarray:
    """a_i ∝ e^{‑lam·i}+e^{‑lam·(n‑1‑i)}, symmetric exponential."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _logistic_distribution(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ 1/(1+e^{α·(i‑mid)}), normalised; heavy on the left side."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = 1.0 / (1.0 + np.exp(alpha * (i - mid)))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _conv_sequence(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a power‑of‑two FFT."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return conv


def _max_conv_and_subgrad(a: np.ndarray):
    """
    Returns (max_conv, subgrad_a) where

        subgrad_a = ∇_a [ 2·n·max(a*a) ]

    The sub‑gradient is computed from the exact active indices of
    the convolution maximum (no smoothing).  Only one FFT is needed.
    """
    n = a.shape[0]
    conv = _conv_sequence(a)
    max_conv = float(conv.max())

    # indices where the maximum is attained (within a tiny tolerance)
    # Numerical tolerance is set to 0.0 because we compare exact max.
    # If floating‑point noise creates a few equal values we still treat
    # them as active.
    active_idx = np.where(conv >= max_conv - 1e-15)[0]

    # Build the sum of the shifted vectors a_{s‑j}.
    shift_sum = np.zeros_like(a)
    for s in active_idx:
        if s < n:
            # j runs from 0 … s, a_{s‑j} = a[ s‑j ] = a[0…s] reversed
            shift_sum[: s + 1] += a[: s + 1][::-1]
        else:
            # s ≥ n.  Valid j are s‑(n‑1) … n‑1
            lo = s - (n - 1)
            # a[lo…n‑1] reversed aligns with j = lo…n‑1
            shift_sum[lo:] += a[lo:][::-1]

    # Average over the active set (any convex combination works)
    if active_idx.size == 0:
        # Should never happen – fall back to uniform gradient.
        subgrad = np.zeros_like(a)
    else:
        factor = 4.0 * n / float(active_idx.size)
        subgrad = factor * shift_sum

    return max_conv, subgrad

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) improves the
    `evaluate_sequence` score.  The routine respects the given time budget.
    """
    np.random.seed(seed)
    start = time.time()
    # keep a small safety margin for the final evaluation call
    deadline = start + budget_s - 2.0

    # ------------------------------------------------------------------
    # 0.  Choose a series of support sizes (coarse → fine)
    # ------------------------------------------------------------------
    stages = [
        # (support size, number of cheap iterations allocated)
        (2000, 250_000),
        (8000, 300_000),
        (20000, None),  # run until time runs out
    ]

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Helper to evaluate cheap score (no final clip) given a probability vector
    # ------------------------------------------------------------------
    def cheap_score_of(a):
        return 2.0 * a.shape[0] * float(_conv_sequence(a).max())

    # ------------------------------------------------------------------
    # 1.  Loop over stages, up‑sampling when a stage finishes
    # ------------------------------------------------------------------
    for stage_id, (n, max_iter_stage) in enumerate(stages, start=1):
        if time.time() >= deadline:
            break

        # Initialise distribution for this stage
        if best_seq is not None:
            # Upsample previous best to current support
            a0 = _interpolate_distribution(np.array(best_seq, dtype=np.float64), n)
        else:
            # No previous result – scan a few parametric families quickly
            print(f"[Stage {stage_id}] Scanning parametric families for n={n}.")
            a0 = None
            local_best = float("inf")
            # Power‑law
            for p in np.linspace(0.5, 1.5, 13):
                if time.time() >= deadline:
                    break
                cand = _power_law_distribution(n, p)
                sc = cheap_score_of(cand)

#!/usr/bin/env python
"""
Subgradient mirror‑descent optimiser for the step‑function problem.

The optimiser works on a simplex (soft‑max parametrisation) and
uses a *exact* sub‑gradient of the non‑smooth objective
`2·n·max(a*a)` (the max of the linear self‑convolution).  Only a
single FFT is required per iteration; the gradient of the active
convolution entry is obtained by a simple slice‑wise shift, which
makes the inner loop much faster than the smoothed‑max version.
The algorithm proceeds in three stages, gradually increasing the
support size (2000 → 8000 → 20000).  At each stage a few
hand‑crafted families (power‑law, symmetric product‑law,
exponential, logistic) are scanned to obtain a good initial
distribution, then an Adam‑style mirror‑descent loop is run for
as many iterations as the remaining time permits.  Periodic jitter
restarts and up‑sampling allow the optimiser to escape plateaux.
The best sequence found is finally evaluated with the provided
`evaluate_sequence` routine and returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top‑level, no closures or lambdas
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max, returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    s = e.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / z.size)
    return e / s


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it that reproduces `a` after a soft‑max (avoids log(0))."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """Smallest power‑of‑two ≥ 3·n‑2 (large enough for all needed convolutions)."""
    return 1 << ((3 * n - 2).bit_length())


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    s = new_a.sum()
    if s == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= s
    return new_a


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{‑p}, normalised."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _sym_product_power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ ((i+1)*(n‑i))^{‑p}, symmetric, normalised."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) * (n - i)) ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _exp_decay_distribution(n: int, lam: float) -> np.ndarray:
    """a_i ∝ e^{‑lam·i}+e^{‑lam·(n‑1‑i)}, symmetric exponential."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _logistic_distribution(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ 1/(1+e^{α·(i‑mid)}), normalised; heavy on the left side."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = 1.0 / (1.0 + np.exp(alpha * (i - mid)))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _conv_sequence(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a power‑of‑two FFT."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return conv


def _max_conv_and_subgrad(a: np.ndarray):
    """
    Returns (max_conv, subgrad_a) where

        subgrad_a = ∇_a [ 2·n·max(a*a) ]

    The sub‑gradient is computed from the exact active indices of
    the convolution maximum (no smoothing).  Only one FFT is needed.
    """
    n = a.shape[0]
    conv = _conv_sequence(a)
    max_conv = float(conv.max())

    # indices where the maximum is attained (tiny tolerance to guard rounding)
    active_idx = np.where(conv >= max_conv - 1e-15)[0]

    # sum of the shifted vectors a_{s‑j}
    shift_sum = np.zeros_like(a)
    for s in active_idx:
        if s < n:
            # j = 0 … s  → a_{s‑j} = a[0…s] reversed
            shift_sum[: s + 1] += a[: s + 1][::-1]
        else:
            # s ≥ n:   j = s‑(n‑1) … n‑1
            lo = s - (n - 1)
            shift_sum[lo:] += a[lo:][::-1]

    if active_idx.size == 0:
        subgrad = np.zeros_like(a)
    else:
        factor = 4.0 * n / float(active_idx.size)
        subgrad = factor * shift_sum

    return max_conv, subgrad
