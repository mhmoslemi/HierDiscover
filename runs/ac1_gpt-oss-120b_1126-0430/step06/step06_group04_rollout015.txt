def _adam_opt(a_init, time_limit, seed):
    rng = np.random.default_rng(seed)
    n = len(a_init)
    x = np.log(a_init + 1e-16)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    best_a = None; best_obj = inf

    start = time.time()
    it = 0
    while time.time() - start < time_limit:
        it += 1
        a = _softmax(x)
        conv = _conv_self_fft(a)
        M = conv.max()
        obj = float(2.0 * n * M)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        # schedule: use linear schedule for beta from 0.5 to 1e7 over time.
        t = (time.time() - start) / time_limit
        beta = 0.5 * np.exp(np.log(1e7/0.5) * t)
        lr = 0.3 * np.exp(np.log(0.001/0.3) * t)
        # smooth max weight
        shifted = conv - M
        if beta > 1e6:
            w = np.zeros_like(conv); w[np.argmax(conv)] = 1.0
        else:
            w = np.exp(beta * shifted)
            s = w.sum()
            w = w / s if s > 0 else np.full_like(conv, 1.0/conv.size)
        # gradient w.r.t a
        g_a = 2.0 * _correlation_fft(w, a)
        # natural gradient in logit space
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)
        # Adam step
        m = beta1 * m + (1 - beta1) * g_x
        v = beta2 * v + (1 - beta2) * (g_x**2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # jitter occasional
        if it % 2000 == 0:
            jitter = rng.normal(scale=0.01, size=n)
            x += jitter
    return best_a, best_obj

def _random_hill_climb(seq, time_limit):
    a = seq.copy()
    n = len(a)
    best_a = a.copy()
    best_obj = float(2.0*n * _conv_self_fft(a).max())
    start = time.time()
    rng = np.random.default_rng()
    while time.time() - start < time_limit:
        # pick i proportional to a_i * a_{k_max-i}
        conv = _conv_self_fft(a)
        k = int(np.argmax(conv))
        # compute contributions
        i_idx = np.arange(n)
        j_idx = k - i_idx
        mask = (j_idx >= 0) & (j_idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i_idx[mask]] * a[j_idx[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            probs = contrib / total_contrib
            i = rng.choice(n, p=probs)
        else:
            i = rng.integers(0, n)
        # choose j uniform from other indices
        j = rng.integers(0, n-1)
        if j >= i:
            j += 1
        # move fraction
        max_move = a[i] * 0.1  # up to 10%
        delta = rng.random() * max_move
        if delta <= 0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = a_new / a_new.sum()
        conv_new = _conv_self_fft(a_new)
        obj_new = float(2.0*n * conv_new.max())
        if obj_new < best_obj:
            a, best_a, best_obj = a_new, a_new.copy(), obj_new
    return best_a, best_obj

3. **Targeted refinement** – a fast simulated‑annealing / projected‑
   gradient descent hybrid (identical to the SA+PG stage of the previous
   version) is run on the remaining time budget, followed by a very
   cheap greedy “mass‑transfer” local search that repeatedly moves a
   fraction of the mass from indices contributing most to the current
   maximal convolution entry to low‑mass indices.

The final sequence is clipped to ``[0, 1000]`` (the evaluator’s caps) and
returned as a plain ``list[float]`` suitable for ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return conv_full[: 2 * n - 1]


def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation of a with weight vector w:
        g[i] = Σ_t w[i+t]·a[t]   for i=0,…,n‑1
    Implemented via FFT (real‑valued).
    """
    n = a.shape[0]
    m = w.shape[0]                     # = 2n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Stable soft‑max."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}."""
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) assuming Σ a = 1."""
    n = a.size
    conv = _conv_self_fft(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Stage 2 – Adam on the simplex (enhanced)
# ----------------------------------------------------------------------


def _adam_opt(
    n: int,
    init_a: np.ndarray,
    time_limit: float,
    seed: int = 0,
) -> Tuple[List[float], float]:
    """
    Adam optimiser on the log‑it simplex with an aggressive
    temperature schedule (β up to 1e8).  Returns the best sequence
    (as a Python list) and its exact objective.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    a = init_a.copy()
    a = np.clip(a, 0.0, None)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()

    # log‑it parametrisation
    x = np.log(a + 1e-16)
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # schedule parameters (tuned for a deeper search)
    beta_start = 0.5
    beta_end = 1e8          # sharper final max
    lr_start = 0.35
    lr_end = 0.00015

    best_a = a.copy()
    best_obj = _objective(best_a)

    it = 0
    while time.time() - start < time_limit:
        it += 1

        # current point on the simplex
        a = _softmax(x)

        conv = _conv_self_fft(a)
        M = conv.max()
        obj = float(2.0 * n * M)

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[Adam n={n} it={it:07d}] new best O={best_obj:.6f}")

        # --- schedule (time‑based) ---
        t = min((time.time() - start) / time_limit, 1.0)
        beta = beta_start * np.exp(np.log(beta_end / beta_start) * t)
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t)

        # soft‑max weights for the smooth max
        if beta > 5e6:
            # essentially the hard max
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            shifted = conv - M          # ≤ 0
            w = np.exp(beta * shifted)
            s = w.sum()
            if s == 0.0:
                w = np.full_like(conv, 1.0 / conv.size)
            else:
                w = w / s

        # gradient of (smooth) max w.r.t. a
        g_a = 2.0 * _correlation_fft(w, a)

        # natural gradient in log‑it space (mirrored gradient)
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape plateaus
        if it % 2000 == 0:
            jitter = rng.normal(scale=0.012, size=n)
            x += jitter

        # periodic restart from the current best (helps escape shallow basins)
        if it % 50000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.008, size=n)
            m.fill(0.0)
            v.fill(0.0)

    # -----------------------------------------------------------------
    # prune near‑zero entries (helps the final objective)
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = _objective(best_a)
        print(f"[Adam compress] length → {best_a.size}, O={best_obj:.6f}")

    return best_a.tolist(), best_obj


# ----------------------------------------------------------------------
# Stage 3 – Simulated annealing + projected gradient (as before)
# ----------------------------------------------------------------------


def _run_sa(
    init_seq: List[float],
    time_limit: float,
    seed: int = 0,
) -> Tuple[List[float], float]:
    """
    Targeted simulated‑annealing refinement followed by projected‑gradient
    descent (exact max‑gradient).  Returns the best sequence found.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    a = np.asarray(init_seq, dtype=float)
    if a.sum() < 1e-12:
        a = np.ones_like(a) / a.size
    else:
        a = a / a.sum()
    n = a.shape[0]

    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())
    best_seq = a.copy()

    # SA schedule
    T = best_obj * 0.2 + 1e-8
    T_min = best_obj * 1e-6 + 1e-12
    decay = 0.996

    it = 0
    while time.time() - start < time_limit * 0.65:
        it += 1
        k_max = int(np.argmax(conv))

        # contributions a[i] * a[k_max-i]
        i_vals = np.arange(n)
        j_vals = k_max - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i_vals[mask]] * a[j_vals[mask]]
        total_contrib = contrib.sum()

        # pick source index biased by contribution
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = rng.choice(n, p=probs)
        else:
            i = rng.integers(n)

        # pick destination uniformly (different from i)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # move a modest fraction of mass
        delta = a[i] * rng.random() * 0.07  # up to 7 % of a[i]
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.clip(a_new, 0.0, None)
        a_new /= a_new.sum()

        conv_new = _conv_self_fft(a_new)
        obj_new = float(2.0 * n * conv_new.max())

        if obj_new < best_obj - 1e-14:
            a, conv, best_obj, best_seq = a_new, conv_new, obj_new, a_new.copy()
            print(f"[SA it={it}] new best O={best_obj:.6f}")
        else:
            # Metropolis acceptance
            if rng.random() < math.exp(-(obj_new - best_obj) / T):
                a, conv = a_new, conv_new

        if it % 2500 == 0:
            T = max(T * decay, T_min)

        if it % 8000 == 0:
            # occasional global jitter
            noise = rng.normal(scale=0.008, size=n)
            a = np.clip(a + noise, 0.0, None)
            a /= a.sum()
            conv = _conv_self_fft(a)

    # -----------------------------------------------------------------
    # Projected‑gradient descent with the exact max‑gradient
    remaining = time_limit - (time.time() - start)
    if remaining > 0.3:
        a = best_seq.copy()
        n = a.size
        it_g = 0
        eta0 = 0.25
        while time.time() - start < time_limit:
            it_g += 1
            conv = _conv_self_fft(a)
            k_max = int(np.argmax(conv))
            w = np.zeros_like(conv)
            w[k_max] = 1.0
            g = 2.0 * _correlation_fft(w, a)    # exact gradient of max(conv)
            eta = eta0 * (0.995 ** it_g)
            a_new = a - eta * g
            a_new = _project_simplex(a_new)
            obj_new = float(2.0 * n * _conv_self_fft(a_new).max())
            if obj_new < best_obj - 1e-14:
                a, best_obj, best_seq = a_new, obj_new, a_new.copy()
                print(f"[PG it={it_g}] improved O={best_obj:.6f}")
            else:
                eta0 *= 0.99
                if eta0 < 1e-6:
                    break

    return best_seq.tolist(), best_obj


# ----------------------------------------------------------------------
# Stage 3b – cheap greedy mass‑transfer refinement
# ----------------------------------------------------------------------


def _greedy_refine(
    init_seq: List[float],
    time_limit: float,
    seed: int = 0,
) -> Tuple[List[float], float]:
    """
    Very fast local search: repeatedly move a small fraction of mass from the
    index most responsible for the current maximal convolution entry to a
    low‑mass index.  Accept only improvements.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    a = np.asarray(init_seq, dtype=float)
    a = a / a.sum()
    n = a.size

    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())
    best_a = a.copy()

    while time.time() - start < time_limit:
        conv = _conv_self_fft(a)
        k = int(np.argmax(conv))

        i_idx = np.arange(n)
        j_idx = k - i_idx
        mask = (j_idx >= 0) & (j_idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i_idx[mask]] * a[j_idx[mask]]
        if contrib.max() == 0.0:
            break

        i = int(np.argmax(contrib))

        # choose a destination among the indices with the smallest current mass
        low = np.where(a <= np.median(a))[0]
        if low.size == 0:
            break
        j = int(rng.choice(low))
        if j == i:
            continue

        delta = a[i] * 0.05   # move 5 % of the source mass
        if delta < 1e-12:
            break

        a[i] -= delta
        a[j] += delta
        a /= a.sum()

        obj = float(2.0 * n * _conv_self_fft(a).max())
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[greedy] new best O={best_obj:.6f}")

    return best_a.tolist(), best_obj


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **_ignore,
) -> List[float]:
    """
    Optimise a non‑negative step‑function under the given time budget.
    The routine focusses on a single support length (the length of the
    provided ``height_sequence_1`` or 2000 by default) and runs a deep
    Adam optimiser followed by simulated‑annealing + projected‑gradient
    refinement and a final cheap greedy sweep.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # 1️⃣ initialise – use previous best if available, otherwise a simple
    # exponential decay shape.
    try:
        init_seq = list(height_sequence_1)          # type: ignore[name-defined]
        print("[init] using supplied height_sequence_1")
    except Exception:
        n = 2000
        idx = np.arange(n, dtype=float)
        r = 0.985
        a = r ** idx
        a = a / a.sum()
        init_seq = a.tolist()
        print("[init] generated exponential start (n=2000)")

    a0 = np.asarray(init_seq, dtype=float)
    a0 = np.clip(a0, 0.0, None)
    if a0.sum() == 0.0:
        a0 = np.full_like(a0, 1.0 / a0.size)
    else:
        a0 = a0 / a0.sum()
    n = a0.size

    best_seq = None
    best_obj = float("inf")

    # ------------------------------------------------------------------
    # 2️⃣ Adam optimisation – allocate the bulk of the budget
    remaining = deadline - time.time()
    if remaining > 5.0:
        adam_time = remaining * 0.70
        print(f"[phase1] Adam optimisation for {adam_time:.1f}s (n={n})")
        seq_adam, obj_adam = _adam_opt(
            n=n,
            init_a=a0,
            time_limit=adam_time,
            seed=seed + 1001,
        )
        best_seq = np.asarray(seq_adam, dtype=float)
        best_obj = obj_adam
        print(f"[phase1] best O after Adam = {best_obj:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Simulated annealing + projected gradient
    remaining = deadline - time.time()
    if remaining > 4.0 and best_seq is not None:
        sa_time = remaining * 0.85
        print(f"[phase2] SA + PG for {sa_time:.1f}s")
        seq_sa, obj_sa = _run_sa(best_seq.tolist(), time_limit=sa_time, seed=seed + 2002)
        if obj_sa < best_obj:
            best_seq = np.asarray(seq_sa, dtype=float)
            best_obj = obj_sa
            print(f"[phase2] new overall best O = {best_obj:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣ Final cheap greedy refinement
    remaining = deadline - time.time()
    if remaining > 2.0 and best_seq is not None:
        print(f"[phase3] greedy refinement for {remaining:.1f}s")
        seq_greedy, obj_greedy = _greedy_refine(best_seq.tolist(),
                                                time_limit=remaining,
                                                seed=seed + 3003)
        if obj_greedy < best_obj:
            best_seq = np.asarray(seq_greedy, dtype=float)
            best_obj = obj_greedy
            print(f"[phase3] improved O = {best_obj:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣ final sanity checks and return
    if best_seq is None or best_seq.size == 0:
        best_seq = np.ones(200, dtype=float) / 200.0

    # Clamp to evaluator limits and avoid too‑small total mass
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    if best_seq.sum() < 0.01:
        best_seq = np.ones_like(best_seq) / best_seq.size

    # Final diagnostic (does not affect the returned value)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_seq.tolist()
