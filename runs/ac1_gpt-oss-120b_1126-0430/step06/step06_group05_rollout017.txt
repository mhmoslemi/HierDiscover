def _grad_smooth_max(a: np.ndarray, conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Compute gradient of the smoothed max (softmax) of convolution entries
    with respect to a.
    Returns vector of shape (n,) representing d (softmax(conv))/d a.
    """
    # compute softmax weights
    # Use log-sum-exp trick
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    w = np.exp(scaled - max_scaled)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(conv, 1.0 / conv.size)
    else:
        w /= w_sum
    # reverse weights for convolution
    w_rev = w[::-1]
    n = a.shape[0]
    L = _next_pow2(n + w_rev.shape[0] - 1)
    af = np.fft.rfft(a, n=L)
    wf = np.fft.rfft(w_rev, n=L)
    conv_w = np.fft.irfft(af * wf, n=L)

"""
Non‑negative step‑function optimiser (smooth‑max Adam + hard‑max fine‑tune).

The optimiser works in three stages within the given time budget:

1. **Smooth‑max Adam** – minimise a smooth approximation of the
   maximal self‑convolution entry.
   The gradient of the soft‑max of the convolution is obtained by a
   weighted convolution (computed via a single FFT), which gives a
   smoother landscape than the hard‑max sub‑gradient and helps escape
   plateaux.

2. **Hard‑max Adam** – start from the best smooth‑max solution and
   switch to the exact hard‑max sub‑gradient to polish the peak.

3. **Simulated‑annealing** – a brief local search that moves a small
   amount of mass between indices, accepting occasional uphill moves.

All three phases keep the sequence on the probability simplex
(`a = softmax(x)`) so the evaluation metric simplifies to
`O = 2·n·max(conv(a,a))`.  The code respects the overall time limit,
prunes negligible entries and finally returns a list of non‑negative
floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Basic FFT utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return conv_full[: 2 * n - 1]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (maps ℝⁿ → simplex)."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


# ----------------------------------------------------------------------
# Gradient helpers
# ----------------------------------------------------------------------


def _grad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """Exact gradient of a single convolution entry (index k) w.r.t. a."""
    n = a.shape[0]
    idx = np.arange(n)
    j = k - idx
    mask = (j >= 0) & (j < n)
    g = np.zeros_like(a)
    if mask.any():
        g[mask] = 2.0 * a[j[mask]]
    return g


def _grad_smooth_max(a: np.ndarray, conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the soft‑max (temperature 1/α) of the convolution entries.
    Returns ∂/∂a  (softmax_α(conv)) .
    """
    # soft‑max weights over the convolution entries
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    w = np.exp(scaled - max_scaled)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(conv, 1.0 / conv.size)
    else:
        w /= w_sum

    # weighted convolution: g = 2 * (a * w_rev)  (cross‑correlation)
    w_rev = w[::-1]
    n = a.shape[0]
    L = _next_pow2(n + w_rev.shape[0] - 1)          # length >= 3·n‑2
    af = np.fft.rfft(a, n=L)
    wf = np.fft.rfft(w_rev, n=L)
    conv_w = np.fft.irfft(af * wf, n=L)            # full convolution a * w_rev
    idx = 2 * n - 2 - np.arange(n)                 # pick the required slice
    g = 2.0 * conv_w[idx]
    return g

    s = a.sum()

        M = conv.max()

            print(f"[hard n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        k_max = int(np.argmax(conv))

def _run_sa(
    init_seq: List[float],
    time_limit: float,
    seed: int = 0,
) -> Tuple[List[float], float]:
    """Refine a sequence by small mass‑transfer moves with a Metropolis schedule."""
    start = time.time()
    rng = np.random.default_rng(seed)
    a = np.asarray(init_seq, dtype=float)
    if a.sum() < 1e-12:
        a = np.ones_like(a) / a.size
    else:
        a = a / a.sum()
    n = a.shape[0]

    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())
    best_seq = a.copy()

    # temperature schedule
    T = best_obj * 0.1 + 1e-6
    T_min = best_obj * 1e-6 + 1e-9
    decay = 0.995
    it = 0

    while time.time() - start < time_limit:
        it += 1
        k_max = int(np.argmax(conv))

        # contribution of each index to the maximal entry
        i_vals = np.arange(n)
        j_vals = k_max - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i_vals[mask]] * a[j_vals[mask]]
        total_contrib = contrib.sum()

        if total_contrib > 0:
            probs = contrib / total_contrib
            i = rng.choice(n, p=probs)
        else:
            i = rng.integers(n)

        # distinct destination uniformly
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # move a modest fraction of mass
        delta = a[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.clip(a_new, 0.0, None)
        a_new /= a_new.sum()

        conv_new = _conv_self_fft(a_new)
        obj_new = float(2.0 * n * conv_new.max())

        # Metropolis acceptance
        if obj_new < best_obj - 1e-12:
            a, conv, best_obj, best_seq = a_new, conv_new, obj_new, a_new.copy()
            print(f"[SA it={it}] new best O={best_obj:.6f}")
        else:
            prob = np.exp(-(obj_new - best_obj) / T) if T > 0 else 0.0
            if rng.random() < prob:
                a, conv = a_new, conv_new

        if it % 2000 == 0:
            T = max(T * decay, T_min)
        if it % 10000 == 0:
            noise = rng.normal(scale=0.01, size=n)
            a = np.clip(a + noise, 0.0, None)
            a /= a.sum()
            conv = _conv_self_fft(a)

    # final compression
    thresh = 1e-12
    mask = best_seq > thresh
    if mask.sum() < best_seq.size:
        comp = best_seq[mask]
        comp /= comp.sum()
        comp_obj = float(2.0 * comp.size * _conv_self_fft(comp).max())
        if comp_obj < best_obj:
            best_seq, best_obj = comp, comp_obj
            print(f"[SA compress] reduced length → {comp.size}, O={best_obj:.6f}")

    return best_seq.tolist(), best_obj

    if init_seq is not None and init_seq.size != target_n:
        x_old = np.linspace(0.0, 1.0, init_seq.size, endpoint=False)
        x_new = np.linspace(0.0, 1.0, target_n, endpoint=False)
        init_a = np.interp(x_new, x_old, init_seq)
        init_a = np.clip(init_a, 0.0, None)
        init_a = init_a / init_a.sum()
    else:
        init_a = init_seq

    # ------------------------------------------------------------------
    # Split the remaining budget.
    remaining = deadline - time.time()
    smooth_frac = 0.55
    hardmax_frac = 0.35
    smooth_budget = remaining * smooth_frac
    hardmax_budget = remaining * hardmax_frac
    sa_budget = remaining - smooth_budget - hardmax_budget

    # ------------------------------------------------------------------
    # Phase 1 – smooth‑max Adam optimisation.
    seq_s, obj_s = _run_opt_smoothmax(
        n=target_n,
        time_limit=smooth_budget,
        init_a=init_a,
        seed=seed + 111,
    )
    best_seq = np.asarray(seq_s, dtype=float)
    best_obj = obj_s
    print(f"[global] after smooth O={best_obj:.6f} (n={target_n})")

    # ------------------------------------------------------------------
    # Phase 2 – hard‑max Adam refinement.
    remaining = deadline - time.time()
    if remaining > 5.0:
        hardmax_budget = remaining  # use whatever is left before SA
    seq_h, obj_h = _run_opt_hardmax(
        n=target_n,
        time_limit=hardmax_budget,
        init_a=best_seq.tolist(),
        seed=seed + 222,
    )
    if obj_h < best_obj:
        best_obj, best_seq = obj_h, np.asarray(seq_h, dtype=float)
        print(f"[refine] hard‑max new best O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated‑annealing local search (if time permits).
    remaining = deadline - time.time()
    if remaining > 5.0:
        seq_sa, obj_sa = _run_sa(best_seq.tolist(), time_limit=remaining, seed=seed + 333)
        if obj_sa < best_obj:
            best_obj, best_seq = obj_sa, np.asarray(seq_sa, dtype=float)
            print(f"[final] SA new best O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check – call the supplied evaluator.
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_seq.tolist()
