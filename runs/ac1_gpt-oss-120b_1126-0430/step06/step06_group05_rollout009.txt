"""
Non‑negative step‑function optimiser (annealed‑p‑norm Adam + SA).

The optimiser minimises the objective
        O(f) = 2·n·max( f * f )
using a smooth surrogate based on the p‑norm of the self‑convolution.
The p‑norm is annealed from p=2 (very smooth) up to p≈40 (close to the
hard‑max) while Adam updates are performed in log‑it (exponentiated‑gradient)
space to respect the simplex constraint.
After the main phase (≈80 % of the time budget) the result is
compressed (tiny entries are dropped) and a short simulated‑annealing
refinement is applied.  The whole routine respects the 1000 s wall‑clock
limit and returns a list of non‑negative floats ready for
``evaluate_sequence``.
"""

import time
import numpy as np
from typing import List, Tuple


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full (linear) self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return conv_full[: 2 * n - 1]


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of a and b using a real FFT."""
    n = a.shape[0]
    m = b.shape[0]
    L = _next_pow2(n + m - 1)
    af = np.fft.rfft(a, n=L)
    bf = np.fft.rfft(b, n=L)
    conv_full = np.fft.irfft(af * bf, n=L)
    return conv_full[: n + m - 1]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


# ----------------------------------------------------------------------
# Annealed p‑norm Adam optimiser (log‑it space)
# ----------------------------------------------------------------------


def _run_opt_pnorm(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
    p_start: float = 2.0,
    p_end: float = 40.0,
) -> Tuple[List[float], float]:
    """
    Adam optimiser that minimises the p‑norm of the self‑convolution as a
    smooth surrogate for the hard‑max.  Returns the best sequence (list) and
    its hard‑max objective (float).
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # ----- initial point on the simplex -----
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
    else:
        a = rng.random(n)

    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n, dtype=float) / n
    else:
        a = a / s

    # work in log‑it (mirrored) space
    x = np.log(a + 1e-16)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # learning‑rate schedule (exponential decay)
    lr_start = 0.6
    lr_end = 0.001

    best_a = a.copy()
    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())
    it = 0

    while time.time() - start < time_limit:
        it += 1

        # current simplex point
        a = _softmax(x)

        # self‑convolution and hard‑max objective (for tracking)
        conv = _conv_self_fft(a)
        M = conv.max()
        obj = float(2.0 * n * M)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[p‑norm n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        # ----- annealed p‑norm computation -----
        elapsed = time.time() - start
        progress = min(elapsed / time_limit, 1.0)
        p = p_start + (p_end - p_start) * progress

        # avoid exact zeros (they would kill the gradient for large p)
        conv_pos = np.maximum(conv, 1e-30)
        p_weights = p * conv_pos ** (p - 1)          # shape (2n‑1,)

        # gradient of the p‑norm surrogate w.r.t. a:
        #   g = 2 * (p_weights * a) convolved   -> take first n entries
        grad = 2.0 * _conv_fft(p_weights, a)[:n]

        # natural gradient in log‑it coordinates (mirrored / exponentiated)
        dot = np.dot(a, grad)
        g_x = a * (grad - dot)

        # time‑dependent learning‑rate (exponential decay)
        t = progress
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # Adam step
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape flat regions
        if it % 3000 == 0:
            jitter = rng.normal(scale=0.008, size=n)
            x += jitter

        # full restart from the best‑so‑far (helps when stuck)
        if it % 60000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.004, size=n)
            m.fill(0.0)
            v.fill(0.0)

    # ----- final compression of negligible entries -----
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = float(2.0 * best_a.size * _conv_self_fft(best_a).max())
        print(f"[compress] reduced length → {best_a.size}, O={best_obj:.6f}")

    return best_a.tolist(), best_obj

def _run_sa(
    init_seq: List[float],
    time_limit: float,
    seed: int = 0,
) -> Tuple[List[float], float]:
    """Refine a sequence by small mass‑transfer moves with a Metropolis schedule."""
    start = time.time()
    rng = np.random.default_rng(seed)
    a = np.asarray(init_seq, dtype=float)
    if a.sum() < 1e-12:
        a = np.ones_like(a) / a.size
    else:
        a = a / a.sum()
    n = a.shape[0]

    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())
    best_seq = a.copy()

    # temperature schedule
    T = best_obj * 0.1 + 1e-6
    T_min = best_obj * 1e-6 + 1e-9
    decay = 0.995
    it = 0

    while time.time() - start < time_limit:
        it += 1
        k_max = int(np.argmax(conv))

        # contribution of each index to the maximal entry
        i_vals = np.arange(n)
        j_vals = k_max - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i_vals[mask]] * a[j_vals[mask]]
        total_contrib = contrib.sum()

        if total_contrib > 0:
            probs = contrib / total_contrib
            i = rng.choice(n, p=probs)
        else:
            i = rng.integers(n)

        # choose a distinct destination uniformly
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # transfer a modest fraction of mass from i to j
        delta = a[i] * rng.random() * 0.05  # up to 5 % of a[i]
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.clip(a_new, 0.0, None)
        a_new /= a_new.sum()

        conv_new = _conv_self_fft(a_new)
        obj_new = float(2.0 * n * conv_new.max())

        # Metropolis acceptance
        if obj_new < best_obj - 1e-12:
            a, conv, best_obj, best_seq = a_new, conv_new, obj_new, a_new.copy()
            print(f"[SA it={it}] new best O={best_obj:.6f}")
        else:
            prob = np.exp(-(obj_new - best_obj) / T) if T > 0 else 0.0
            if rng.random() < prob:
                a, conv = a_new, conv_new

        # temperature decay
        if it % 2000 == 0:
            T = max(T * decay, T_min)

        # occasional global jitter
        if it % 10000 == 0:
            noise = rng.normal(scale=0.01, size=n)
            a = np.clip(a + noise, 0.0, None)
            a /= a.sum()
            conv = _conv_self_fft(a)

    # final compression of tiny entries
    thresh = 1e-12
    mask = best_seq > thresh
    if mask.sum() < best_seq.size:
        comp = best_seq[mask]
        comp /= comp.sum()
        comp_obj = float(2.0 * comp.size * _conv_self_fft(comp).max())
        if comp_obj < best_obj:
            best_seq, best_obj = comp, comp_obj
            print(f"[SA compress] reduced length → {comp.size}, O={best_obj:.6f}")

    return best_seq.tolist(), best_obj

            time_limit=sub_budget,
            init_a=best_seq,
            seed=seed + 777,
            p_start=4.0,
            p_end=60.0,
        )
        if obj2 < best_obj:
            best_seq = np.asarray(seq2, dtype=float)
            best_obj = obj2
            print(f"[re‑opt] after second‑pass new best O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated‑annealing refinement (if time permits).
    remaining = deadline - time.time()
    if remaining > 5.0:
        seq_ref, obj_ref = _run_sa(best_seq.tolist(), time_limit=remaining, seed=seed + 999)
        if obj_ref < best_obj:
            best_obj, best_seq = obj_ref, np.asarray(seq_ref, dtype=float)
            print(f"[final] after SA new best O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check – call the supplied evaluator.
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_seq.tolist()
