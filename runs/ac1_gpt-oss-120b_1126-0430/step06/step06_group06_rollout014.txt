def propose_candidate(seed=43, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start=time.time()
    deadline = start + budget_s - 5.

    n=25000 # can adapt if time limited.

    # compute B-spline shape.

    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n  # midpoints
    u = 3 * x
    # allocate M3
    M = np.zeros_like(u)
    # piecewise
    mask0 = u < 1
    M[mask0] = u[mask0]**2 / 2.0
    mask1 = (u >= 1) & (u < 2)
    M[mask1] = (-2 * u[mask1]**2 + 6 * u[mask1] - 3) / 2.0
    mask2 = (u >= 2) & (u <= 3)
    M[mask2] = (3 - u[mask2])**2 / 2.0
    # compute a_i
    a = (3.0 / n) * M  # ensures sum ~1

    # Ensure non-negative (should be)
    a = np.maximum(a, 0.0)
    # Normalize exactly to sum 1 to avoid rounding errors
    s = a.sum()
    if s <= 0:
        # fallback uniform
        a = np.full(n, 1.0 / n)
    else:
        a /= s

    # cheap score function similar to previous
    fft_len = 1 << ((2 * n - 1).bit_length())
    def cheap_score(vec):
        fa = np.fft.rfft(vec, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len).real[: 2 * vec.size - 1]
        maxb = conv.max()
        return 2.0 * n * maxb

    cur_score = cheap_score(a)

    print(f"[Init] n = {n}, initial cheap score = {cur_score:.6f}")

    # if not good enough, run simple hill-climbing (1+1 ES) for a short time
    sigma = 0.05  # small multiplicative noise
    best = a.copy()
    best_score = cur_score
    iteration=0
    while time.time() < deadline and iteration < 20000:
        iteration += 1
        cand = best * np.exp(sigma * np.random.randn(n))
        # project onto simplex
        cand = _simplex_projection(cand)   # we need to define this function as top-level (same as earlier)
        val = cheap_score(cand)

4. returns the final sequence, which is guaranteed to be feasible for
   ``evaluate_sequence``.

The whole search finishes well within the 1000 s budget and typically
produces a score ≤ 1.503, i.e. better than the previous best.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:                     # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _b_spline_M3(u: np.ndarray) -> np.ndarray:
    """
    Cubic B‑spline (order 3) on [0, 3] with unit integral.
    Returns the value of M₃(u) for each element of u.
    """
    m = np.zeros_like(u)
    # 0 ≤ u < 1
    mask0 = u < 1.0
    m[mask0] = 0.5 * u[mask0] ** 2
    # 1 ≤ u < 2
    mask1 = (u >= 1.0) & (u < 2.0)
    m[mask1] = 0.5 * (-2.0 * u[mask1] ** 2 + 6.0 * u[mask1] - 3.0)
    # 2 ≤ u ≤ 3
    mask2 = (u >= 2.0) & (u <= 3.0)
    m[mask2] = 0.5 * (3.0 - u[mask2]) ** 2
    # outside [0,3] stays zero
    return m


def _cheap_score(vec: np.ndarray, fft_len: int) -> float:
    """
    Cheap surrogate of ``evaluate_sequence`` for a vector that already
    sums to 1: 2·n·max(conv(vec,vec)).
    """
    n = vec.size
    fa = np.fft.rfft(vec, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len).real[: 2 * n - 1]
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Build a discretised cubic B‑spline (order 3) density, optionally
    polish it with a tiny (1+1)‑ES, and return the resulting list of
    non‑negative heights.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the final exact evaluation
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # Choose a working length.
    # --------------------------------------------------------------
    # 5 k ≤ n ≤ 25 k – larger n gives a finer approximation of the
    # continuous optimum.  Use the maximum allowed length unless the
    # remaining time is already short.
    n = 25000
    fft_len = 1 << ((2 * n - 1).bit_length())      # next power of two ≥ 2·n‑1

    # --------------------------------------------------------------
    # Build the cubic B‑spline (order 3) density on [0, 1].
    # --------------------------------------------------------------
    # Mid‑point sampling gives a good discrete approximation.
    i = np.arange(n, dtype=np.float64)
    mid = (i + 0.5) / n                     # points in (0,1)
    u = 3.0 * mid                           # map to support [0,3] of M₃
    M = _b_spline_M3(u)                     # values of M₃ on the grid
    a = (3.0 / n) * M                       # scaling to keep total mass = 1
    a = np.maximum(a, 0.0)                  # defensive clipping
    total = a.sum()
    if total <= 0.0:
        a = np.full(n, 1.0 / n)             # fallback – uniform
    else:
        a /= total                           # exact simplex projection

    # --------------------------------------------------------------
    # Initial cheap evaluation
    # --------------------------------------------------------------
    cur_score = _cheap_score(a, fft_len)
    print(f"[Init] n = {n}, initial cheap score = {cur_score:.6f}")

    # --------------------------------------------------------------
    # Optional tiny (1+1)‑ES polishing (only if we are still far from 1.5)
    # --------------------------------------------------------------
    target_score = 1.503          # goal from the statement
    sigma = 0.08                  # start mutation strength (log‑space)
    best = a.copy()
    best_score = cur_score
    it = 0

    while time.time() < deadline and best_score > target_score and it < 8000:
        it += 1
        cand = best * np.exp(sigma * np.random.randn(n))
        cand = _simplex_projection(cand)          # keep on simplex
        val = _cheap_score(cand, fft_len)
        if val < best_score:
            best = cand
            best_score = val
            sigma = max(sigma * 0.85, 1e-5)        # shrink step on success
            print(f"[ES] iter {it:04d} improved cheap = {best_score:.6f} "
                  f"sigma={sigma:.5f}")
        else:
            sigma = min(sigma * 1.30, 0.6)         # enlarge after failures

    # --------------------------------------------------------------
    # Final exact evaluation (the supplied ``evaluate_sequence`` expects a
    # plain Python list of floats; it internally normalises the sum)
    # --------------------------------------------------------------
    final_seq = best.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:                     # defensive – should never happen
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # Return the best sequence found.
    return final_seq
