def adam_refine(a_init, deadline, start_time):
    import numpy as np, math, time, random
    n = a_init.shape[0]
    # logits
    z = np.log(a_init + 1e-16)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_alpha = 0.4
    eps_factor = 0.2
    eps_factor_min = 5e-5
    eps_decay = 0.9999
    best_a = a_init.copy()
    best_score = 2.0 * n * _conv_max(best_a)
    it = 0
    max_iters = 10_000_000
    while time.time() < deadline and it < max_iters:
        it += 1
        a_cur = _softmax(z)
        max_conv, grad_a = _conv_max_and_grad(a_cur, eps_factor)
        cur_score = 2.0 * n * max_conv
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_cur.copy()
            if it % 2000 == 0:
                elapsed = time.time() - start_time
                print(f"[Adam {it:6d}] best_score = {best_score:.6f}  elapsed = {elapsed:.1f}s")
        # mirror descent gradient
        mu = np.dot(grad_a, a_cur)
        grad_z = a_cur * (grad_a - mu)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * (grad_z ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step = -alpha_t * m_hat / (np.sqrt(v_hat) + eps_adam)
        z = z + step
        z = np.clip(z, -60.0, 60.0)
        eps_factor = max(eps_factor_min, eps_factor * eps_decay)
        # optional jitter
        if it % 5000 == 0 and random.random() < 0.05:
            z += 0.01 * np.random.randn(n)
    return best_a, best_score

"""
Step‑function optimizer.

The algorithm has three stages:

1. **Initialisation** – a probability vector is built from the exact
   discretisation of the arcsine distribution (the continuous optimum
   for the max‑convolution problem).

2. **Hill‑climbing** – we repeatedly move a small amount of probability
   mass from the index with the largest sub‑gradient of the current
   max‑convolution to the index with the smallest sub‑gradient.
   This directly reduces the term `max(a*a)` while keeping the sum of
   the sequence equal to 1.

3. **Adam polishing** – a short mirror‑descent/Adam optimisation on the
   simplex refines the result using a smoothed version of the max‑convolution
   (soft‑max).  The soft‑max smoothing constant is gradually reduced.

All helper functions are defined at top level, no closures or lambdas are
used.  The routine respects the 1000 s time budget, adapts step sizes,
and finally returns the best non‑negative sequence it found.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Low‑level FFT utilities
# ----------------------------------------------------------------------


def _fft_len_for_conv(n: int) -> int:
    """Smallest power‑of‑two ≥ 2·n‑1 (sufficient for linear self‑convolution)."""
    return 1 << ((2 * n - 1).bit_length())


def _fft_len_for_corr(n: int) -> int:
    """Smallest power‑of‑two ≥ 3·n‑2 (used for the gradient of the smooth max)."""
    return 1 << ((3 * n - 2).bit_length())


def conv_self_fft(a: np.ndarray, L: int = None) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    if L is None:
        L = _fft_len_for_conv(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Soft‑max utilities (mirror‑descent on the simplex)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max producing a probability vector."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Smoothed max of a*a and gradient w.r.t. a (already multiplied by 4·n).

    Returns
    -------
    max_conv : float
        Maximum entry of the linear self‑convolution.
    grad_a : np.ndarray
        Gradient of 2·n·smooth_max (including the factor 4·n).
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    a_fft = np.fft.rfft(a, L)

    # linear self‑convolution (length 2·n‑1)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # soft‑max weights over the convolution entries
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or np.isnan(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # cross‑correlation a ⋆ w (length 3·n‑2, we need first n entries)
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # gradient (the factor 4·n comes from d (2·n·max) / d a)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a


def _conv_max(a: np.ndarray) -> float:
    """Plain (non‑smoothed) maximum entry of a*a."""
    n = a.shape[0]
    L = _fft_len_for_conv(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Problem‑specific helpers
# ----------------------------------------------------------------------


def initialize_arcsine(n: int) -> np.ndarray:
    """Exact discretisation of the arcsine CDF on n equal bins (sum = 1)."""
    idx = np.arange(n, dtype=np.float64)
    x0 = idx / n
    x1 = (idx + 1) / n

    def arcsine_cdf(x):
        return (2.0 / np.pi) * np.arcsin(np.sqrt(np.clip(x, 0.0, 1.0)))

    a = arcsine_cdf(x1) - arcsine_cdf(x0)
    a_sum = a.sum()
    if a_sum == 0.0 or np.isnan(a_sum):
        raise RuntimeError("Failed to initialise a valid arcsine vector.")
    return a / a_sum


def gradient_of_max(a: np.ndarray, max_idx: int) -> np.ndarray:
    """
    Sub‑gradient of the entry (a*a)[max_idx] with respect to a.
    For each i:  ∂/∂a_i (a*a)[max_idx] = a[max_idx-i] + a[i-max_idx]
    (terms out of range are taken as zero).
    """
    n = a.shape[0]
    g = np.zeros_like(a)

    # i where max_idx - i ≥ 0  → i ≤ max_idx
    i1 = np.arange(max_idx + 1)
    g[i1] += a[max_idx - i1]

    # i where i - max_idx ≥ 0  → i ≥ max_idx
    i2 = np.arange(max_idx, n)
    g[i2] += a[i2 - max_idx]

    return g


# ----------------------------------------------------------------------
# Hill‑climbing stage
# ----------------------------------------------------------------------


def hill_climb_optimize(a: np.ndarray,
                        deadline: float,
                        start_time: float,
                        log_interval: int = 2000,
                        no_improve_limit: int = 50000) -> (np.ndarray, float):
    """
    Simple coordinate‑wise hill‑climbing:
    move a small amount of mass from the index with the largest
    sub‑gradient of the current max‑convolution to the index with the
    smallest sub‑gradient, keeping the total sum equal to 1.
    """
    n = a.shape[0]
    L = _fft_len_for_conv(n)

    conv = conv_self_fft(a, L)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()

    step_factor = 0.30          # initial fraction of source mass to move
    min_step_factor = 0.001
    decay = 0.9995

    it = 0
    no_improve = 0

    while time.time() < deadline:
        it += 1

        conv = conv_self_fft(a, L)
        max_idx = int(conv.argmax())

        # compute sub‑gradient of the active max entry
        g = gradient_of_max(a, max_idx)
        src = int(g.argmax())
        dst = int(g.argmin())

        if src == dst:
            break

        delta = step_factor * a[src]
        if delta < 1e-14:
            break
        delta = min(delta, a[src])

        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        conv_new = conv_self_fft(a_new, L)
        new_score = 2.0 * n * conv_new.max()

        if new_score < best_score - 1e-12:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            if it % log_interval == 0:
                elapsed = time.time() - start_time
                print(f"[Hill {it:6d}] improved score = {best_score:.6f} "
                      f"elapsed = {elapsed:.1f}s")
        else:
            no_improve += 1
            # occasional random jitter to escape tiny plateaus
            if it % 5000 == 0 and random.random() < 0.1:
                perturb = (np.random.rand(n) - 0.5) * 1e-5
                a = a + perturb
                a = np.maximum(a, 0.0)
                a_sum = a.sum()
                if a_sum > 0:
                    a = a / a_sum

        # decay the step size
        step_factor = max(min_step_factor, step_factor * decay)

        if no_improve > no_improve_limit:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Adam refinement stage (mirror‑descent on the simplex)
# ----------------------------------------------------------------------


def adam_refine(a_init: np.ndarray,
                deadline: float,
                start_time: float) -> (np.ndarray, float):
    """
    Mirror‑descent / Adam optimisation on the simplex using a
    smoothed max‑convolution.  The routine runs until the time budget
    expires and returns the best vector seen.
    """
    n = a_init.shape[0]

    # logits for the probability simplex
    z = np.log(a_init + 1e-16)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    base_alpha = 0.6          # initial step size for logits
    eps_factor = 0.12         # start smoothing relative to max
    eps_factor_min = 5e-5
    eps_decay = 0.9999

    best_a = a_init.copy()
    best_score = 2.0 * n * _conv_max(best_a)

    it = 0
    max_iters = 5_000_000

    while time.time() < deadline and it < max_iters:
        it += 1

        a_cur = _softmax(z)
        max_conv, grad_a = _conv_max_and_grad(a_cur, eps_factor)
        cur_score = 2.0 * n * max_conv

        if cur_score < best_score:
            best_score = cur_score
            best_a = a_cur.copy()
            if it % 2000 == 0:
                elapsed = time.time() - start_time
                print(f"[Adam {it:6d}] best_score = {best_score:.6f} "
                      f"elapsed = {elapsed:.1f}s")

        # natural‑gradient step on the simplex (mirror descent)
        mu = np.dot(grad_a, a_cur)          # 〈∇, a〉
        grad_z = a_cur * (grad_a - mu)      # ∇_z J

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step = -alpha_t * m_hat / (np.sqrt(v_hat) + eps_adam)

        z = z + step
        z = np.clip(z, -60.0, 60.0)        # keep logits numerically safe

        # gradual reduction of the smoothing constant
        eps_factor = max(eps_factor_min, eps_factor * eps_decay)

        # tiny random jitter now and then
        if it % 5000 == 0 and random.random() < 0.05:
            z += 0.01 * np.random.randn(n)

    return best_a, best_score
