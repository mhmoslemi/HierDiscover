"""
Step‑function optimiser.

The optimiser works in three stages for each chosen support length:

1. **Shape pre‑search** – a handful of analytically motivated shapes
   (uniform, exponential, Gaussian, sparse Dirichlet, random) are
   generated and the best (lowest exact objective) is kept as a starting
   point.

2. **Exponentiated‑gradient descent on the simplex** – the objective
   2·n·max(conv(a,a)) is approximated by a temperature‑controlled
   soft‑max of the autocorrelation.  The gradient of the smooth max is
   obtained via a single FFT‑based correlation, and a multiplicative
   (exponentiated) update (a_i ← a_i·exp(−η·g_i)) keeps the iterates
   strictly positive.  The temperature β grows from ≈0.5 to 1e7 while
   the step‑size η shrinks, so the optimisation gradually focuses on the
   true L∞‑norm.

3. **Targeted refinement** – a fast simulated‑annealing phase that
   moves mass away from the indices that contribute most to the current
   maximal autocorrelation entry, followed by a few exact projected‑gradient
   steps (gradient of the hard max).

The routine tries several support sizes (8000, 6000, 4000, 2000) and keeps
the overall best sequence found within the total time budget (default
1000 s).  All numbers are non‑negative and the final sequence is clipped
to the limits imposed by ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level FFT utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full self‑convolution a*a using a real FFT (length 2·len(a)‑1).
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return conv_full[: 2 * n - 1]


def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation of a with weight vector w (length 2·n‑1):
        g[i] = Σ_t w[i+t]·a[t]    for i = 0,…,n‑1
    Implemented via FFT (real‑valued).  Returns vector of length n.
    """
    n = a.shape[0]
    m = w.shape[0]                     # = 2n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of vector v onto the probability simplex
    {x ≥ 0, Σ x = 1}.  Uses the O(n log n) algorithm of Duchi et al.
    """
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero or negative – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) assuming Σ a = 1.
    Returns a Python float.
    """
    n = a.size
    conv = _conv_self_fft(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Shape pre‑search
# ----------------------------------------------------------------------


def _shape_candidates(n: int, rng: np.random.Generator):
    """
    Produce a list of candidate initial sequences for support length n.
    Each candidate is a NumPy array normalised to sum‑to‑1.
    """
    cand = []

    # Uniform
    cand.append(np.full(n, 1.0 / n))

    # Exponential decay (forward and reversed)
    r_vals = np.linspace(0.90, 0.9997, 15)
    idx = np.arange(n, dtype=float)
    for r in r_vals:
        a = r ** idx
        a = a / a.sum()
        cand.append(a)
        cand.append(a[::-1])

    # Gaussian windows centred in the middle
    centre = (n - 1) / 2.0
    sigma_vals = np.linspace(0.10 * n, 0.40 * n, 10)
    for sigma in sigma_vals:
        a = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
        a = a / a.sum()
        cand.append(a)

    # Sparse Dirichlet‑like (few non‑zero entries)
    k = max(2, int(0.03 * n))
    for _ in range(4):
        a = np.zeros(n, dtype=float)
        pos = rng.choice(n, size=k, replace=False)
        vals = rng.random(k)
        a[pos] = vals
        a = a / a.sum()
        cand.append(a)

    # Pure random Dirichlet (α=1)
    a = rng.random(n)
    a = a / a.sum()
    cand.append(a)

    return cand


def _best_initial_sequence(n: int, rng: np.random.Generator, time_limit: float):
    """
    Scan a modest set of analytical shapes and return the one with the
    smallest exact objective.  The scan aborts if the time budget expires.
    """
    start = time.time()
    best_a = None
    best_obj = float("inf")
    for a in _shape_candidates(n, rng):
        if time.time() - start > time_limit:
            break
        obj = _objective(a)
        if obj < best_obj:
            best_obj, best_a = obj, a
    # Fallback to a random Dirichlet if nothing evaluated
    if best_a is None:
        a = rng.random(n)
        a = a / a.sum()
        best_a = a
        best_obj = _objective(a)
    return best_a, best_obj


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (Stage 2)
# ----------------------------------------------------------------------


def _run_eg_opt(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
) -> tuple[list[float], float]:
    """
    Exponentiated‑gradient descent on the simplex using a temperature‑
    controlled soft‑max of the autocorrelation.  Returns the best sequence
    (as a list) discovered and its exact objective.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # Initialise ---------------------------------------------------------
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
    else:
        a = rng.random(n)

    a = np.clip(a, 1e-16, None)      # avoid exact zeros
    a = a / a.sum()

    # Annealing / step‑size schedules
    beta_start = 0.5                  # smoothness at start
    beta_end = 1e7                    # practically a hard max
    eta_start = 0.5                   # relatively large initial step
    eta_end = 1e-5                    # fine‑grained final step

    best_a = a.copy()
    best_obj = _objective(best_a)

    it = 0
    while time.time() - start < time_limit:
        it += 1
        conv = _conv_self_fft(a)                # full autocorrelation
        M = conv.max()
        # exact objective (hard max) – keep track of the best seen
        obj_exact = float(2.0 * n * M)
        if obj_exact < best_obj - 1e-14:
            best_obj = obj_exact
            best_a = a.copy()
            print(f"[EG n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        # Schedule factors (based on elapsed fraction)
        t_frac = (time.time() - start) / time_limit
        beta = beta_start * (beta_end / beta_start) ** t_frac
        eta = eta_start * (eta_end / eta_start) ** t_frac

        # Smoothed‑max weights
        if beta > 1e6:                         # treat as hard max
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            shifted = conv - M                 # ≤0
            # stable exponentiation
            w_exp = np.exp(beta * shifted)
            w_sum = w_exp.sum()
            if w_sum == 0.0:
                w = np.full_like(conv, 1.0 / conv.size)
            else:
                w = w_exp / w_sum

        # Gradient of the (smooth) max w.r.t. a
        g = 2.0 * _correlation_fft(w, a)       # shape (n,)

        # Exponentiated‑gradient multiplicative update
        a = a * np.exp(-eta * g)

        # Normalise back onto the probability simplex
        s = a.sum()
        if s == 0.0 or np.isnan(s):
            a = rng.random(n)
            a = a / a.sum()
        else:
            a = a / s

        # Occasional jitter to shake out of plateaus
        if it % 500 == 0:
            a = np.clip(a + rng.normal(scale=0.001, size=n), 0.0, None)
            a = a / a.sum()

    return best_a.tolist(), best_obj


# ----------------------------------------------------------------------
# Targeted simulated‑annealing + projected‑gradient refinement (Stage 3)
# ----------------------------------------------------------------------


def _run_sa(
    init_seq: list[float],
    time_limit: float,
    seed: int = 0,
) -> tuple[list[float], float]:
    """
    Fast simulated‑annealing phase that preferentially moves mass away
    from indices contributing to the current maximal autocorrelation entry,
    followed by a short projected‑gradient descent with the exact max‑gradient.
    Returns the best sequence and its exact objective.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    a = np.asarray(init_seq, dtype=float)
    if a.sum() < 1e-12:
        a = np.ones_like(a) / a.size
    else:
        a = a / a.sum()
    n = a.shape[0]

    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())
    best_seq = a.copy()

    # SA schedule
    T = best_obj * 0.2 + 1e-8
    T_min = best_obj * 1e-6 + 1e-12
    decay = 0.997

    it = 0
    while time.time() - start < time_limit * 0.65:
        it += 1
        k_max = int(np.argmax(conv))

        # contribution of each i to the maximal entry
        i_vals = np.arange(n)
        j_vals = k_max - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i_vals[mask]] * a[j_vals[mask]]
        total_contrib = contrib.sum()

        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = rng.choice(n, p=probs)
        else:
            i = rng.integers(n)

        # pick a distinct destination uniformly
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.07   # up to 7 % of a[i]
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.clip(a_new, 0.0, None)
        a_new /= a_new.sum()

        conv_new = _conv_self_fft(a_new)
        obj_new = float(2.0 * n * conv_new.max())

        if obj_new < best_obj - 1e-14:
            a, conv, best_obj, best_seq = a_new, conv_new, obj_new, a_new.copy()
            print(f"[SA it={it}] new best O={best_obj:.6f}")
        else:
            # Metropolis acceptance
            if rng.random() < math.exp(-(obj_new - best_obj) / T):
                a, conv = a_new, conv_new

        if it % 2500 == 0:
            T = max(T * decay, T_min)

        if it % 8000 == 0:
            # occasional global jitter
            noise = rng.normal(scale=0.008, size=n)
            a = np.clip(a + noise, 0.0, None)
            a /= a.sum()
            conv = _conv_self_fft(a)

    # -----------------------------------------------------------------
    # Projected‑gradient descent with the exact max‑gradient
    remaining = time_limit - (time.time() - start)
    if remaining > 0.3:
        a = best_seq.copy()
        n = a.size
        it_g = 0
        eta0 = 0.25
        while time.time() - start < time_limit:
            it_g += 1
            conv = _conv_self_fft(a)
            k_max = int(np.argmax(conv))
            w = np.zeros_like(conv)
            w[k_max] = 1.0
            g = 2.0 * _correlation_fft(w, a)     # exact max‑gradient
            eta = eta0 * (0.995 ** it_g)
            a_new = a - eta * g
            a_new = _project_simplex(a_new)
            obj_new = float(2.0 * n * _conv_self_fft(a_new).max())
            if obj_new < best_obj - 1e-14:
                a, best_obj, best_seq = a_new, obj_new, a_new.copy()
                print(f"[PG it={it_g}] improved O={best_obj:.6f}")
            else:
                # shrink step size aggressively if not improving
                eta0 *= 0.99
                if eta0 < 1e-6:
                    break

    return best_seq.tolist(), best_obj


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **_ignore,
) -> list[float]:
    """
    Main optimisation routine.
    Tries several support lengths, builds an analytically motivated start,
    runs exponentiated‑gradient descent, then refines with simulated‑annealing
    and projected‑gradient steps.  Returns a list of non‑negative floats
    ready for ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # Try decreasing lengths (larger lengths first give more flexibility)
    lengths = [8000, 6000, 4000, 2000]

    best_seq = None
    best_obj = float("inf")

    for idx, n in enumerate(lengths):
        now = time.time()
        remaining = deadline - now
        if remaining <= 2.0:
            break

        # Allocate time for this length.
        # First (largest) length gets the bulk of the budget.
        if idx == 0:
            run_time = min(remaining * 0.70, remaining - (len(lengths) - idx - 1) * 2.0)
        else:
            run_time = min(remaining * 0.25, remaining - (len(lengths) - idx - 1) * 2.0)

        # ------------------------------------------------------------------
        # Stage 1 – shape pre‑search (quick)
        shape_time = min(3.0, run_time * 0.07)
        init_a, init_obj = _best_initial_sequence(n, rng, shape_time)
        print(f"[init n={n}] shape‑obj={init_obj:.6f}")

        # ------------------------------------------------------------------
        # Stage 2 – exponentiated‑gradient optimisation
        eg_time = run_time * 0.60
        seq, val = _run_eg_opt(
            n=n,
            time_limit=eg_time,
            init_a=init_a,
            seed=seed + idx * 12345,
        )
        if val < best_obj:
            best_obj, best_seq = val, np.asarray(seq, dtype=float)
            print(f"[global] new overall best O={best_obj:.6f} (n={n})")

        # ------------------------------------------------------------------
        # Stage 3 – targeted simulated annealing + projected gradient
        remain_after_eg = run_time - eg_time
        if remain_after_eg > 4.0:
            seq_ref, val_ref = _run_sa(
                seq,
                time_limit=remain_after_eg,
                seed=seed + idx * 12345 + 999,
            )
            if val_ref < best_obj:
                best_obj, best_seq = val_ref, np.asarray(seq_ref, dtype=float)
                print(f"[final] after SA+PG O={best_obj:.6f} (n={n})")

        if time.time() >= deadline - 0.5:
            break

    # ----------------------------------------------------------------------
    # Fallback (should never happen)
    if best_seq is None or best_seq.size == 0:
        best_seq = np.ones(200, dtype=float) / 200.0

    # ----------------------------------------------------------------------
    # Final sanity checks and clipping to evaluator limits
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    if best_seq.sum() < 0.01:
        best_seq = np.ones_like(best_seq) / best_seq.size

    # Optional diagnostic evaluation (does not affect the returned value)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_seq.tolist()
```

This implementation replaces the Adam optimiser used previously with an exponentiated‑gradient scheme that works directly on the simplex, adds a more aggressive temperature schedule, and keeps the effective simulated‑annealing + projected‑gradient refinement. It explores several support lengths while respecting the 1000 s budget, and returns the best sequence found.
