def _run_opt_softmax_adam(n, time_limit, init_a=None, seed=0):
    start = time.time()
    rng = np.random.default_rng(seed)
    # init a (simplex)
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
    else:
        a = rng.random(n)
    a = np.clip(a, 0, None)
    a_sum = a.sum()
    if a_sum < 1e-12:
        a = np.ones(n)/n
    else:
        a = a / a_sum

    x = np.log(a + 1e-16)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    best_a = a.copy()
    conv = _conv_self_fft(a)
    best_obj = float(2 * n * conv.max())
    it = 0

    # schedule for alpha
    alpha_start = 0.5
    alpha_end = 30.0

    while time.time() - start < time_limit:
        it += 1
        a = _softmax(x)   # current
        conv = _conv_self_fft(a)
        # compute smooth max with alpha based on progress
        elapsed = time.time() - start
        t = min(elapsed / time_limit, 1.0)
        alpha = alpha_start + (alpha_end - alpha_start) * (t ** 2)  # quadratic increase
        # compute softmax weights
        max_c = conv.max()
        exp_alpha = np.exp(alpha * (conv - max_c))
        w = exp_alpha / exp_alpha.sum()
        # compute gradient of smoothed max
        # weight contributions from all entries but we approximate using topK
        topK = min(10, w.size)  # maybe top 10
        # get indices of largest w
        idx = np.argpartition(-w, topK-1)[:topK]  # unsorted topK
        # compute gradient contributions
        g_a = np.zeros_like(a)
        for k in idx:
            # shift grad = 2*a_{k-i}
            # we can vectorize: g_a += w[k] * 2*a shifted
            # shift: produce vector of length n where at position i we have a_{k-i} if valid
            # vectorized: create j = k - np.arange(n); mask = (j>=0) & (j<n); g[k where mask] += w[k] * 2*a[j[mask]]
            j = k - np.arange(n)
            mask = (j >= 0) & (j < n)
            g_a[mask] += w[k] * 2.0 * a[j[mask]]
        # gradient of full objective: multiply by 2*n (same scaling for all directions, may skip)
        # Map to logit space
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)

        # learning rate schedule
        lr_start = 0.6
        lr_end = 0.001
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t)

        # Adam moments
        m = beta1 * m + (1 - beta1) * g_x
        v = beta2 * v + (1 - beta2) * (g_x ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # track best obj with real max conv
        obj = float(2 * n * conv.max())
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[softAdam n={n} it={it} O={best_obj:.6f} α={alpha:.2f}]")
        # occasional jitter & restart
        if it % 3000 == 0:
            jitter = rng.normal(scale=0.005, size=n)
            x += jitter
        if it % 50000 == 0:
            x = np.log(best_a + 1e-16) + rng.normal(scale=0.002, size=n)
            m.fill(0.)
            v.fill(0.)

    # compress negligible entries
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = float(2 * best_a.size * _conv_self_fft(best_a).max())
        print(f"[compress] length {best_a.size}, O={best_obj:.6f}")

    return best_a.tolist(), best_obj

"""
Non‑negative step‑function optimiser (annealed‑soft‑max Adam + refinement).

The algorithm differs from the previous hard‑max Adam in two ways:

1.  It minimises a *smooth* approximation of the maximal convolution entry:
        sm_α(c) = (1/α)·log∑_k exp(α·c_k) .
    The smoothing parameter α is increased gradually (deterministic annealing)
    so the optimiser first sees a smooth landscape and later focuses on the true
    hard maximum.

2.  The gradient of the smooth objective is the weighted sum of the gradients
    of all convolution entries (weights = soft‑max over the entries).  This
    provides a richer descent direction than using only the single maximal
    entry.

The optimisation is performed on the simplex by means of an exponentiated‑
gradient (log‑it) parametrisation together with Adam.  After the main phase,
a short simulated‑annealing refinement (the same routine as used previously)
polishes the result.  The routine respects the time‑budget and returns a list
of non‑negative floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return conv_full[: 2 * n - 1]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


# ----------------------------------------------------------------------
# Annealed‑soft‑max Adam optimiser
# ----------------------------------------------------------------------


def _run_opt_softmax_adam(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
) -> Tuple[List[float], float]:
    """
    Adam optimiser that minimises a *soft‑max* approximation of the
    maximal self‑convolution entry.  The smoothing parameter α is raised
    from ``alpha_start`` to ``alpha_end`` during the run.
    Returns the best sequence (as a list) and its hard‑max objective
    value 2·n·max(conv).
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # ----- initialise a on the simplex ---------------------------------
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
    else:
        a = rng.random(n)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n, dtype=float) / n
    else:
        a = a / s

    # log‑it (exponentiated‑gradient) parametrisation
    x = np.log(a + 1e-16)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # learning‑rate schedule (exponential decay)
    lr_start = 0.6
    lr_end = 0.001

    # annealing schedule for the soft‑max temperature
    alpha_start = 0.5          # very smooth
    alpha_end = 30.0           # close to hard max

    best_a = a.copy()
    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())

    it = 0
    topK = 10                 # number of convolution entries used in the gradient

    while time.time() - start < time_limit:
        it += 1

        # current point on the simplex
        a = _softmax(x)

        # convolution of a with itself
        conv = _conv_self_fft(a)

        # progress‑dependent parameters
        elapsed = time.time() - start
        t = min(elapsed / time_limit, 1.0)                # in [0,1]
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t)
        alpha = alpha_start + (alpha_end - alpha_start) * (t ** 2)

        # soft‑max weights over the convolution entries
        max_c = conv.max()
        exp_alpha = np.exp(alpha * (conv - max_c))
        w = exp_alpha / exp_alpha.sum()

        # gradient of the *smooth* max w.r.t. a:
        #   g_a = 2 * Σ_k w_k * shift_k(a)
        # we approximate the sum by the top‑K weights (most influence)
        idx = np.argpartition(-w, min(topK, w.size) - 1)[:topK]
        g_a = np.zeros_like(a)
        for k in idx:
            wk = w[k]
            # shift contribution: 2 * a_{k-i} (if index in range)
            j = k - np.arange(n)
            mask = (j >= 0) & (j < n)
            if mask.any():
                g_a[mask] += wk * 2.0 * a[j[mask]]

        # map gradient to log‑it space (mirror / natural gradient)
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)

        bias_correction1 = 1.0 - beta1 ** it
        bias_correction2 = 1.0 - beta2 ** it
        m_hat = m / bias_correction1
        v_hat = v / bias_correction2

        # Adam step in log‑it space
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # keep track of the *hard* objective (the one used by evaluate_sequence)
        obj = float(2.0 * n * conv.max())
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[softAdam n={n:5d} it={it:07d} O={best_obj:.6f} α={alpha:.2f}]")

        # occasional jitter to escape plateaus
        if it % 4000 == 0:
            jitter = rng.normal(scale=0.005, size=n)
            x += jitter

        # occasional full restart from the current best
        if it % 90000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.002, size=n)
            m.fill(0.0)
            v.fill(0.0)

    # ------------------------------------------------------------------
    # prune negligible entries (they increase n without helping)
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = float(2.0 * best_a.size * _conv_self_fft(best_a).max())
        print(f"[compress] reduced length → {best_a.size}, O={best_obj:.6f}")

    return best_a.tolist(), best_obj


# ----------------------------------------------------------------------
# Targeted simulated‑annealing refinement (unchanged)
# ----------------------------------------------------------------------


def _run_sa(
    init_seq: List[float],
    time_limit: float,
    seed: int = 0,
) -> Tuple[List[float], float]:
    """Refine a sequence by small mass‑transfer moves with a Metropolis schedule."""
    start = time.time()
    rng = np.random.default_rng(seed)
    a = np.asarray(init_seq, dtype=float)
    if a.sum() < 1e-12:
        a = np.ones_like(a) / a.size
    else:
        a = a / a.sum()
    n = a.shape[0]

    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())
    best_seq = a.copy()

    # temperature schedule
    T = best_obj * 0.1 + 1e-6
    T_min = best_obj * 1e-6 + 1e-9
    decay = 0.995
    it = 0

    while time.time() - start < time_limit:
        it += 1
        k_max = int(np.argmax(conv))

        # contribution of each index to the maximal entry
        i_vals = np.arange(n)
        j_vals = k_max - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i_vals[mask]] * a[j_vals[mask]]
        total_contrib = contrib.sum()

        if total_contrib > 0:
            probs = contrib / total_contrib
            i = rng.choice(n, p=probs)
        else:
            i = rng.integers(n)

        # choose a distinct destination uniformly
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # transfer a modest fraction of mass from i to j
        delta = a[i] * rng.random() * 0.05  # up to 5 % of a[i]
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.clip(a_new, 0.0, None)
        a_new /= a_new.sum()

        conv_new = _conv_self_fft(a_new)
        obj_new = float(2.0 * n * conv_new.max())

        # Metropolis acceptance
        if obj_new < best_obj - 1e-12:
            a, conv, best_obj, best_seq = a_new, conv_new, obj_new, a_new.copy()
            print(f"[SA it={it}] new best O={best_obj:.6f}")
        else:
            prob = np.exp(-(obj_new - best_obj) / T) if T > 0 else 0.0
            if rng.random() < prob:
                a, conv = a_new, conv_new

        # temperature decay
        if it % 2000 == 0:
            T = max(T * decay, T_min)

        # occasional global jitter
        if it % 10000 == 0:
            noise = rng.normal(scale=0.01, size=n)
            a = np.clip(a + noise, 0.0, None)
            a /= a.sum()
            conv = _conv_self_fft(a)

    # final compression of tiny entries
    thresh = 1e-12
    mask = best_seq > thresh
    if mask.sum() < best_seq.size:
        comp = best_seq[mask]
        comp /= comp.sum()
        comp_obj = float(2.0 * comp.size * _conv_self_fft(comp).max())
        if comp_obj < best_obj:
            best_seq, best_obj = comp, comp_obj
            print(f"[SA compress] reduced length → {comp.size}, O={best_obj:.6f}")

    return best_seq.tolist(), best_obj

def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **_ignore,
) -> List[float]:
    """
    Search for a non‑negative step‑function minimising
    ``evaluate_sequence``.  The routine spends most of the available
    budget on an annealed‑soft‑max Adam optimisation (large support,
    ~15‑20 k entries) and finishes with a short simulated‑annealing
    refinement.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # initialise from a previously discovered construction (if any)
    init_seq = None
    if 'height_sequence_1' in globals():
        seq0 = np.asarray(height_sequence_1, dtype=float)
        seq0 = np.clip(seq0, 0.0, None)
        if seq0.sum() < 0.01:
            seq0 = np.ones_like(seq0)
        seq0 = seq0 / seq0.sum()
        init_seq = seq0

    # ------------------------------------------------------------------
    # Choose a target length based on remaining time.
    remaining = deadline - time.time()
    # we aim for a fairly long support (≈18 k) if we have enough time
    if remaining > 60.0:
        target_n = 18000
    elif remaining > 30.0:
        target_n = 12000
    else:
        target_n = 8000

    # upsample any initial sequence to the target length
    if init_seq is not None and init_seq.size != target_n:
        x_old = np.linspace(0.0, 1.0, init_seq.size, endpoint=False)
        x_new = np.linspace(0.0, 1.0, target_n, endpoint=False)
        init_a = np.interp(x_new, x_old, init_seq)
        init_a = np.clip(init_a, 0.0, None)
        init_a = init_a / init_a.sum()
    else:
        init_a = init_seq

    # ------------------------------------------------------------------
    # Main optimisation phase (annealed‑soft‑max Adam)
    remaining = deadline - time.time()
    main_budget = remaining * 0.80
    seq, obj = _run_opt_softmax_adam(

    )

    best_obj = obj
    print(f"[global] after soft‑Adam best O={best_obj:.6f} (n={target_n})")

    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_seq.tolist()
