def _run_opt_smooth(n, time_limit, init_a, seed):
    start = time.time()
    rng = np.random.default_rng(seed)

    # init a on simplex
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            # interpolate to n using linear interpolation
            old = np.linspace(0, 1, a.size, endpoint=False)
            new = np.linspace(0, 1, n, endpoint=False)
            a = np.interp(new, old, a)
    else:
        a = rng.random(n)
    a = np.clip(a, 0, None)
    if a.sum() < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / a.sum()
    # logits
    x = np.log(a + 1e-16)

    # Adam
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    # schedule for beta (smoothness)
    beta_start = 0.5
    beta_end = 200.0

    # schedule for LR: maybe from 0.5 to 0.001 (like earlier)
    lr_start = 0.5
    lr_end = 0.001

    best_a = a.copy()
    best_obj = float('inf')
    # Actually compute objective via evaluate_sequence (hard max)
    conv = _conv_self_fft(best_a)
    best_obj = float(2.0 * n * conv.max())

    # main loop
    it = 0
    while time.time() - start < time_limit:
        it += 1

        # compute current a
        a = _softmax(x) # sum=1

        # compute conv
        conv = _conv_self_fft(a)
        # compute hard objective
        obj_hard = float(2.0 * n * conv.max())
        if obj_hard < best_obj - 1e-12:
            best_obj = obj_hard
            best_a = a.copy()
            print(f"[smooth n={n} it={it} obj={best_obj:.6f}]")

        # compute smoothed max with beta schedule
        elapsed = time.time() - start
        t_frac = min(elapsed / time_limit, 1.0)
        beta = beta_start * (beta_end / beta_start) ** t_frac

        # stable log-sum-exp for conv
        conv_max = conv.max()
        exp_vals = np.exp(beta * (conv - conv_max))
        w = exp_vals / exp_vals.sum()

        # compute gradient g w.r.t a
        # Compute convolution w * a_rev
        a_rev = a[::-1]
        L = _next_pow2(len(w) + len(a_rev) - 1)
        wf = np.fft.rfft(w, n=L)
        af = np.fft.rfft(a_rev, n=L)
        conv_rev = np.fft.irfft(wf * af, n=L)
        start_idx = n - 1
        g = 2.0 * conv_rev[start_idx:start_idx + n]

        # compute gradient in logit x space: g_x = a * (g - np.dot(a,g))
        dot = np.dot(a, g)
        g_x = a * (g - dot)

        # learning rate schedule
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t_frac)

        # Adam moments update
        m = beta1 * m + (1 - beta1) * g_x
        v = beta2 * v + (1 - beta2) * (g_x ** 2)

        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        # step
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # jitter occasionally
        if it % 2000 == 0:
            jitter = rng.normal(scale=0.01, size=n)
            x += jitter

        # restart occasionally
        if it % 50000 == 0:
            # restart from best
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.005, size=n)
            m.fill(0)
            v.fill(0)

    # compress negligible entries
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        # recompute best_obj with new length
        conv = _conv_self_fft(best_a)
        best_obj = float(2.0 * best_a.size * conv.max())
        print(f"[compress] new length {best_a.size}, O={best_obj:.6f}")

    return best_a.tolist(), best_obj

"""
Step‑function optimizer with a continuation (smooth‑max) scheme.

The algorithm works directly on the simplex (via a softmax of logits) and
optimises a smoothed version of the maximum self‑convolution entry.
The smoothing parameter β is increased from a very soft value to a large
one, turning the problem gradually from a smooth objective to the true
hard max.  Adam updates are performed in log‑it space, occasional jitter
and restarts help escape plateaus, and a short simulated‑annealing
refinement is applied at the end.  The routine respects the overall
time budget (default 1000 s) and returns a non‑negative sequence that
minimises ``evaluate_sequence``.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return conv_full[: 2 * n - 1]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces a probability vector)."""
    mx = np.max(x)
    e = np.exp(x - mx)
    s = e.sum()
    if s == 0.0:
        # fallback to uniform
        return np.full_like(x, 1.0 / x.size)
    return e / s


# ----------------------------------------------------------------------
# Smoothed‑max Adam optimiser
# ----------------------------------------------------------------------


def _run_opt_smooth(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
) -> Tuple[List[float], float]:
    """
    Optimise a non‑negative simplex vector `a` (length n) to minimise
    2·n·max(conv(a,a)) using a smooth‑max (log‑sum‑exp) surrogate.
    Returns the best sequence (as a list) and the corresponding hard
    objective value.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # ---- initialise a on the simplex -------------------------------------------------
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
    else:
        a = rng.random(n)

    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n, dtype=float) / n
    else:
        a = a / s

    # logits for exponentiated (mirror) updates
    x = np.log(a + 1e-16)
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # learning‑rate schedule (exponential decay)
    lr_start, lr_end = 0.5, 0.001

    # smooth‑max schedule
    beta_start, beta_end = 0.5, 5000.0

    best_a = a.copy()
    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())

    it = 0
    while time.time() - start < time_limit:
        it += 1

        # current point on the simplex
        a = _softmax(x)

        # self‑convolution (hard objective)
        conv = _conv_self_fft(a)
        conv = np.maximum(conv, 0.0)                     # guard against tiny negative artefacts
        obj_hard = float(2.0 * n * conv.max())
        if obj_hard < best_obj - 1e-12:
            best_obj = obj_hard
            best_a = a.copy()
            print(f"[smooth n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        # ---- smooth‑max surrogate ----------------------------------------------------
        elapsed = time.time() - start
        t_frac = min(elapsed / time_limit, 1.0)

        # smoothing parameter β
        beta = beta_start * (beta_end / beta_start) ** t_frac

        # stable log‑sum‑exp weights
        mx = conv.max()
        exp_vals = np.exp(beta * (conv - mx))
        w = exp_vals / exp_vals.sum()          # shape (2n‑1,)

        # gradient of the smooth‑max w.r.t. a:
        #   g_j = 2 * Σ_i w_i * a_{i‑j}
        a_rev = a[::-1]
        L = _next_pow2(len(w) + len(a_rev) - 1)
        wf = np.fft.rfft(w, n=L)
        af = np.fft.rfft(a_rev, n=L)
        conv_rev = np.fft.irfft(wf * af, n=L)
        start_idx = n - 1
        g = 2.0 * conv_rev[start_idx:start_idx + n]   # shape (n,)

        # natural gradient in log‑it space (mirror / exponentiated)
        dot = np.dot(a, g)
        g_x = a * (g - dot)

        # learning‑rate (exponential decay)
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t_frac)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to shake out of flat regions
        if it % 3000 == 0:
            jitter = rng.normal(scale=0.01, size=n)
            x += jitter

        # occasional full restart from the best found so far
        if it % 60000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.005, size=n)
            m.fill(0.0)
            v.fill(0.0)

    # ------------------------------------------------------------------
    # prune negligible entries (they increase length without helping)
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        conv = _conv_self_fft(best_a)
        best_obj = float(2.0 * best_a.size * conv.max())
        print(f"[compress] reduced length → {best_a.size}, O={best_obj:.6f}")

    return best_a.tolist(), best_obj


# ----------------------------------------------------------------------
# Targeted simulated‑annealing refinement (unchanged)
# ----------------------------------------------------------------------


def _run_sa(
    init_seq: List[float],
    time_limit: float,
    seed: int = 0,
) -> Tuple[List[float], float]:
    """Refine a sequence by small mass‑transfer moves with a Metropolis schedule."""
    start = time.time()
    rng = np.random.default_rng(seed)
    a = np.asarray(init_seq, dtype=float)
    if a.sum() < 1e-12:
        a = np.ones_like(a) / a.size
    else:
        a = a / a.sum()
    n = a.shape[0]

    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())
    best_seq = a.copy()

    # temperature schedule
    T = best_obj * 0.1 + 1e-6
    T_min = best_obj * 1e-6 + 1e-9
    decay = 0.995
    it = 0

    while time.time() - start < time_limit:
        it += 1
        k_max = int(np.argmax(conv))

        # contribution of each index to the maximal entry
        i_vals = np.arange(n)
        j_vals = k_max - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i_vals[mask]] * a[j_vals[mask]]
        total_contrib = contrib.sum()

        if total_contrib > 0:
            probs = contrib / total_contrib
            i = rng.choice(n, p=probs)
        else:
            i = rng.integers(n)

        # choose a distinct destination uniformly
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # transfer a modest fraction of mass from i to j
        delta = a[i] * rng.random() * 0.05  # up to 5 % of a[i]
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.clip(a_new, 0.0, None)
        a_new /= a_new.sum()

        conv_new = _conv_self_fft(a_new)
        obj_new = float(2.0 * n * conv_new.max())

        # Metropolis acceptance
        if obj_new < best_obj - 1e-12:
            a, conv, best_obj, best_seq = a_new, conv_new, obj_new, a_new.copy()
            print(f"[SA it={it}] new best O={best_obj:.6f}")
        else:
            prob = np.exp(-(obj_new - best_obj) / T) if T > 0 else 0.0
            if rng.random() < prob:
                a, conv = a_new, conv_new

        # temperature decay
        if it % 2000 == 0:
            T = max(T * decay, T_min)

        # occasional global jitter
        if it % 10000 == 0:
            noise = rng.normal(scale=0.01, size=n)
            a = np.clip(a + noise, 0.0, None)
            a /= a.sum()
            conv = _conv_self_fft(a)

    # final compression of tiny entries
    thresh = 1e-12
    mask = best_seq > thresh
    if mask.sum() < best_seq.size:
        comp = best_seq[mask]
        comp /= comp.sum()
        comp_obj = float(2.0 * comp.size * _conv_self_fft(comp).max())
        if comp_obj < best_obj:
            best_seq, best_obj = comp, comp_obj
            print(f"[SA compress] reduced length → {comp.size}, O={best_obj:.6f}")

    return best_seq.tolist(), best_obj

    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
