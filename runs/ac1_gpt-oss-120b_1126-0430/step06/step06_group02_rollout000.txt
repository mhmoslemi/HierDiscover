"""
Hybrid L‑BFGS / Adam optimiser for the step‑function norm problem.

The algorithm builds a probability vector `a` (non‑negative, sum = 1) on a
support of size *n*.  The evaluation metric is
    score = 2·n·max(a*a)   (max = maximum entry of the linear self‑convolution)
which is scale‑invariant, so we may work on the simplex.
A softmax parametrisation `a = softmax(z)` is used; gradient information
for the smooth max (log‑sum‑exp) of the convolution is obtained by FFT‑based
correlation.  Optimisation proceeds in several stages:

1.  Initialise from the global sequence if provided, otherwise from a few
    hand‑crafted families and a handful of random Dirichlet draws.
2.  Run a schedule of L‑BFGS‑B sub‑optimisations with decreasing smoothing
    factors `eps_factor`.  After each stage the best distribution seen so far
    is remembered.
3.  Occasionally up‑sample the support (≈ 1.5×) and continue the schedule –
    larger supports allow a finer discretisation of the optimal continuous
    shape and usually lower the peak of the convolution.
4.  A final short Adam fine‑tuning (mirror‑descent on the logits) is performed
    to polish the solution.
5.  The best candidate is evaluated with the supplied `evaluate_sequence`
    function and returned.

The whole routine respects the time budget (`budget_s`, default 1000 s) and
will stop early if the deadline is reached.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top‑level, no closures
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable softmax returning a probability vector (sum = 1)."""
    # Clamp to avoid overflow in exp()
    z = np.clip(z, -50.0, 50.0)
    z_max = np.max(z)
    e = np.exp(z - z_max)
    s = e.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / z.size)
    return e / s


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it that reproduces `a` after a softmax."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """Smallest power‑of‑two ≥ 3·n‑2 (enough for linear self‑convolution)."""
    return 1 << ((3 * n - 2).bit_length())


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    s = new_a.sum()
    if s == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= s
    return new_a


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{‑p}, normalised."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _sym_product_power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ ((i+1)*(n‑i))^{‑p}, symmetric, normalised."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) * (n - i)) ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _exp_decay_distribution(n: int, lam: float) -> np.ndarray:
    """a_i ∝ e^{‑lam·i}+e^{‑lam·(n‑1‑i)}, symmetric exponential."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _logistic_distribution(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ 1/(1+e^{α·(i‑mid)}), normalised; heavy on the left side."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = 1.0 / (1.0 + np.exp(alpha * (i - mid)))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _random_dirichlet_distribution(n: int, conc: float) -> np.ndarray:
    """Sample a Dirichlet(alpha=conc) distribution and return a probability vector."""
    if conc <= 0.0:
        conc = 1.0
    x = np.random.gamma(conc, 1.0, size=n)
    s = x.sum()
    if s == 0.0:
        x[:] = 1.0 / n
    else:
        x /= s
    return x


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Returns (max_conv, grad_a) where

        grad_a = ∇_a [ 2·n·smooth_max(a*a) ]

    `eps_factor` multiplies the smoothing constant ε = eps_factor·max_conv.
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    # FFT of a (zero‑padded)
    a_fft = np.fft.rfft(a, L)

    # Linear self‑convolution a*a
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # Smoothed soft‑max weights
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv                # ≤ 0 for numerical stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # Correlation a ⋆ w
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # Gradient (already multiplied by 4·n)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a


def _objective_and_grad(z: np.ndarray, eps_factor: float):
    """
    Return (value, gradient) for the smooth objective

        f(z) = 2·n·smooth_max(softmax(z) * softmax(z))

    Gradient is taken w.r.t. the logits `z` (mirror‑descent on the simplex).
    """
    a = _softmax(z)
    max_conv, grad_a = _conv_max_and_grad(a, eps_factor)
    n = a.shape[0]
    value = 2.0 * n * max_conv
    mu = np.dot(grad_a, a)               # <grad_a, a>
    grad_z = a * (grad_a - mu)           # Jacobian of softmax
    return value, grad_z

                mu = np.dot(grad_a, a_cur)
                grad_z = a_cur * (grad_a - mu)

                # Adam moments
                m = beta1 * m + (1.0 - beta1) * grad_z
                v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
                it = inner + 1
                m_hat = m / (1.0 - beta1 ** it)
                v_hat = v / (1.0 - beta2 ** it)

                # Step size (inverse‑sqrt decay)
                alpha_t = base_alpha / (math.sqrt(it) + 1.0)
                step = - m_hat / (np.sqrt(v_hat) + eps_adam)

                # Simple back‑tracking
                trial_alpha = alpha_t
                accepted = False
                for _ in range(4):
                    z_trial = np.clip(z + trial_alpha * step, -50.0, 50.0)
                    a_trial = _softmax(z_trial)
                    new_score = _cheap_score(a_trial)
                    cur_score = _cheap_score(a_cur)
                    if new_score <= cur_score:
                        z = z_trial
                        accepted = True
                        break
                    trial_alpha *= 0.5
                if not accepted:
                    z = np.clip(z + alpha_t * step, -50.0, 50.0)

                # occasional jitter to avoid plateaux
                if (inner % 1000 == 0) and (np.random.rand() < 0.1):
                    z += 0.02 * np.random.randn(n)

            z_new = z.copy()

        # ----------------------------------------------------------
        # 2b. Evaluate the candidate produced by this stage.
        # ----------------------------------------------------------
        a_new = _softmax(z_new)
        cur_score = _cheap_score(a_new)

        # Keep the global best ever seen.
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a_new.copy()
            last_improve_iter = total_iters
            print(f"[Stage {stage_counter:03d}] NEW BEST cheap = {cur_score:.6f}")

        # Prepare for the next stage.
        z = z_new
        eps_factor = max(eps_factor_min, eps_factor * eps_decay)
        stage_counter += 1
        total_iters += max_iter_per_stage

        # ----------------------------------------------------------
        # 2c. Periodic jitter restart if stagnation observed.
        # ----------------------------------------------------------
        if total_iters - last_improve_iter > 800000:
            print(f"[Stage {stage_counter}] Stagnation – jitter restart.")
            if best_seq is not None:
                z = _logits_from_distribution(best_seq) + 0.03 * np.random.randn(n)
            else:
                z = np.log(np.full(n, 1.0 / n)) + 0.03 * np.random.randn(n)
            last_improve_iter = total_iters

        # ----------------------------------------------------------
        # 2d. Upsample when we have spare time and the support is still modest.
        # ----------------------------------------------------------
        if n < max_n and (time.time() + 30.0) < deadline:
            # Decide new size (≈ 1.6× but not exceeding max_n)
            new_n = min(max_n, int(n * 1.6))
            print(f"[Stage {stage_counter}] Upsampling support: {n} → {new_n}")
            best_seq = _interpolate_distribution(best_seq, new_n)
            # Reset optimisation variables for the larger problem.
            z = _logits_from_distribution(best_seq) + 0.005 * np.random.randn(new_n)
            n = new_n
            # Reset smoothing schedule to allow a fresh descent on the finer grid.
            eps_factor = eps_factor_start
            # Reset Adam moments if we later fall back.
            # Continue with the next iteration (no 'continue' needed).

        # ----------------------------------------------------------
        # End‑of‑loop housekeeping – ensure we don't overrun the deadline.
        # ----------------------------------------------------------
        if time.time() >= deadline:
            break

    # --------------------------------------------------------------
    # 3. Final exact evaluation with the provided `evaluate_sequence`
    # --------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # --------------------------------------------------------------
    # 4. Return the best sequence discovered.
    # --------------------------------------------------------------
    return final_seq
