"""
Propose a step‑function (list of non‑negative heights) that minimises
evaluate_sequence().  The method consists of three phases:

1.  **Extensive parametric sweep** – a variety of analytic families
    (power‑law, offset power‑law, double power‑law, stretched‑exponential,
    exponential, logistic, symmetric product‑power‑law, Laplacian,
    Gaussian, Beta‑asymmetric and cosine‑bell) are sampled on a fine grid.
    The cheap bound 2·n·max(a*a) is used to keep the sweep cheap.

2.  **Local refinement** – the best few families are refined by a
    simple simulated‑annealing / hill‑climbing on their parameters.
    The refinement uses the exact cheap bound and respects the time budget.

3.  **Full‑vector optimisation** – the best distribution found so far
    is further improved by a mirror‑descent (Adam) optimiser on the
    probability simplex, using a smooth max‑convolution gradient.
    A final cheap random walk finishes the search.

All steps respect the 1000 s time budget and keep the sequence non‑negative,
normalised to sum≈1 (the evaluator rescales anyway).  The returned list is
ready for evaluate_sequence().

def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max, returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    s = e.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / z.size)
    return e / s


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it that reproduces `a` after a soft‑max."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """Smallest power‑of‑two ≥ 3·n‑2 (enough for all needed convolutions)."""
    return 1 << ((3 * n - 2).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Returns (max_conv, grad_a) where

        grad_a = ∇_a [ 2·n·smooth_max(a*a) ]

    `eps_factor` multiplies the smoothing constant ε = eps_factor·max_conv.
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv                     # ≤ 0 for stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # Gradient of 2·n·smooth_max (the factor 4·n comes from the derivative of a*a)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a


def _cheap_score(a: np.ndarray) -> float:
    """2·n·max(a*a) – the exact value of evaluate_sequence when sum(a)=1."""
    n = a.shape[0]
    return 2.0 * n * _conv_max(a)

def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}   (normalised)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _power_law_offset_distribution(n: int, p: float, offset: float) -> np.ndarray:
    """a_i ∝ (i+offset)^{-p}   (normalised)."""
    idx = np.arange(offset, offset + n, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _double_power_law_distribution(n: int, p: float, q: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}·(n-i)^{-q}   (normalised)."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) ** (-p)) * ((n - i) ** (-q))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _stretched_exp_distribution(n: int, tau_factor: float, gamma: float) -> np.ndarray:
    """
    a_i ∝ exp( - (i/(tau_factor·n))^{gamma} )   (normalised).
    `tau_factor` ∈ (0,1] stretches the decay length.
    """
    tau = max(1e-12, tau_factor * n)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(- (i / tau) ** gamma)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _exp_decay_distribution(n: int, lam: float) -> np.ndarray:
    """a_i ∝ e^{-lam·i}+e^{-lam·(n‑1‑i)}   (normalised)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _logistic_distribution(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ 1/(1+e^{α·(i‑mid)})   (normalised)."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = 1.0 / (1.0 + np.exp(alpha * (i - mid)))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _sym_product_power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ ((i+1)*(n‑i))^{-p}   (symmetric, normalised)."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) * (n - i)) ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _laplacian_distribution(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·|i‑mid|)   (normalised)."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = np.exp(-lam * np.abs(i - mid))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _gaussian_distribution(n: int, sigma_frac: float) -> np.ndarray:
    """a_i ∝ exp(-½·((i‑mid)/(σ))²)   (normalised)."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    sigma = max(1e-12, sigma_frac * n)
    a = np.exp(-0.5 * ((i - mid) / sigma) ** 2)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _beta_asym_distribution(n: int, a_param: float, b_param: float) -> np.ndarray:
    """
    a_i ∝ x^{a-1}(1‑x)^{b-1}   with x = (i+0.5)/n, normalised.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, a_param - 1.0) * np.power(1.0 - x, b_param - 1.0)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _cosine_bell_distribution(n: int, width_frac: float) -> np.ndarray:
    """Cosine‑bell of given half‑width as a fraction of the total length."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    half_width = width_frac * (n - 1) / 2.0
    d = np.abs(i - mid)
    mask = d <= half_width
    a = np.zeros_like(i)
    a[mask] = 0.5 * (1.0 + np.cos(np.pi * d[mask] / half_width))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a

def _refine_parameters(name, generator, init_params, bounds, n, budget, best_score):
    """
    Simple stochastic hill‑climbing / simulated annealing on the parameters
    of a parametric family.  `bounds` is a dict {param: (low, high)}.
    Returns (best_seq, best_score, best_params).
    """
    # time allocation: at most `budget` seconds for this family
    start = time.time()
    deadline = start + budget
    # start from the supplied init_params
    cur_params = dict(init_params)
    cur_seq = generator(n, **cur_params)
    cur_score = _cheap_score(cur_seq)
    best_seq = cur_seq.copy()
    best_par = cur_params.copy()
    best_local_score = cur_score

    # step sizes: 10 % of the range initially
    step = {k: 0.10 * (high - low) for k, (low, high) in bounds.items()}

    # annealing schedule
    temperature = 1.0

    while time.time() < deadline:
        # propose a new point
        cand_params = {}
        for k, (low, high) in bounds.items():
            delta = np.random.randn() * step[k]
            cand = cur_params[k] + delta
            # clip to bounds
            cand = max(low, min(high, cand))
            cand_params[k] = cand

        try:
            cand_seq = generator(n, **cand_params)
        except Exception:
            continue

        cand_score = _cheap_score(cand_seq)

        # accept if better, otherwise maybe accept with prob.
        improve = cand_score < cur_score
        if improve or (np.random.rand() < math.exp(-(cand_score - cur_score) / max(temperature, 1e-12))):
            cur_params = cand_params
            cur_seq = cand_seq
            cur_score = cand_score
            if cand_score < best_local_score:
                best_local_score = cand_score
                best_seq = cand_seq.copy()
                best_par = cand_params.copy()

        # adapt step sizes
        for k in step:
            if cand_score < cur_score:
                step[k] *= 1.02
            else:
                step[k] *= 0.99
            # keep steps reasonable
            low, high = bounds[k]
            step[k] = max(step[k], (high - low) * 1e-4)

        # temperature decay
        temperature *= 0.9995

        # early exit if we have already out‑performed the global best
        if best_local_score < best_score - 1e-8:
            break

    return best_seq, best_local_score, best_par

    # --------------------------------------------------------------
    # Initialise logits from the current best
    logits = _logits_from_distribution(best_seq) + 0.02 * np.random.randn(n)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.90, 0.999
    eps_adam = 1e-8

    base_alpha = 0.30
    eps_factor = 0.08
    eps_factor_min = 5e-5
    eps_decay = 0.9994

    it = 0
    last_improve = 0
    best_seq_sofar = best_seq.copy()
    best_score_sofar = best_score

    # Allocate about 25 % of the remaining budget for Adam
    adam_deadline = time.time() + (deadline - time.time()) * 0.25
    while time.time() < adam_deadline and it < 1_200_000:
        it += 1
        a = _softmax(logits)
        max_conv, grad_a = _conv_max_and_grad(a, eps_factor)
        cheap_score = 2.0 * n * max_conv

        # keep the best ever seen
        if cheap_score < best_score_sofar - 1e-12:
            best_score_sofar = cheap_score
            best_seq_sofar = a.copy()
            last_improve = it
            if it % 30000 == 0:
                print(f"[Adam {it:07d}] NEW BEST cheap = {cheap_score:.8f}")

        # mirror‑descent on logits
        mu = np.dot(grad_a, a)
        grad_z = a * (grad_a - mu)

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step = - m_hat / (np.sqrt(v_hat) + eps_adam)

        # back‑tracking line‑search
        accepted = False
        trial_alpha = alpha_t
        for _ in range(4):
            new_logits = logits + trial_alpha * step
            new_logits = np.clip(new_logits, -50.0, 50.0)
            a_new = _softmax(new_logits)
            new_score = 2.0 * n * _conv_max(a_new)
            if new_score <= cheap_score:
                logits = new_logits
                accepted = True
                break
            trial_alpha *= 0.5
        if not accepted:
            logits += alpha_t * step
            logits = np.clip(logits, -50.0, 50.0)

        # occasional jitter to get out of plateaus
        if it % 4000 == 0 and np.random.rand() < 0.15:
            logits += 0.02 * np.random.randn(n)
            m *= 0.6
            v *= 0.6

        # decay smoothing parameter
        eps_factor = max(eps_factor_min, eps_factor * eps_decay)

        # restart if we have been stuck for a long time
        if it - last_improve > 250_000:
            print(f"[Adam {it}] Stagnation – restarting from best so far")
            logits = _logits_from_distribution(best_seq_sofar) + 0.03 * np.random.randn(n)
            m.fill(0.0)
            v.fill(0.0)
            eps_factor = 0.08
            last_improve = it

    best_seq = best_seq_sofar
    best_score = best_score_sofar
    print(f"[Adam] Finished after {it} iters, best cheap score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Final cheap random walk (coordinate moves)
    # --------------------------------------------------------------
    a = best_seq.copy()
    cheap_score = best_score
    while time.time() < deadline:
        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        if i == j:
            continue
        delta = 0.001 * a[i]                # move 0.1 % of mass at i
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = 2.0 * n * _conv_max(a_new)
        if new_score < cheap_score:
            a = a_new
            cheap_score = new_score
            best_seq = a.copy()
            best_score = cheap_score
            # sparse progress reports
            if np.random.rand() < 0.001:
                print(f"[Walk] Improved cheap = {cheap_score:.8f}")

    # --------------------------------------------------------------
    # 5. Final exact evaluation
    # --------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        exact_score = evaluate_sequence(final_seq)   # provided by the platform
        print(f"Final exact score = {exact_score:.6f}")
    except Exception as exc:  # pragma: no cover
        exact_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    return final_seq
