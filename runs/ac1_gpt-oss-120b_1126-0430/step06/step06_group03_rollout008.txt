def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum x = 1}."""
    # from Duchi et al. 2008
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

import time

# ----------------------------------------------------------------------
# Utility functions (no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max, returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    s = e.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / z.size)
    return e / s


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it that reproduces `a` after a soft‑max."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """Smallest power‑of‑two ≥ 3·n‑2 (enough for all needed convolutions)."""
    return 1 << ((3 * n - 2).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Returns (max_conv, grad_a) where

        grad_a = ∇_a [ 2·n·smooth_max(a*a) ]

    `eps_factor` multiplies the smoothing constant ε = eps_factor·max_conv.
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv                     # ≤ 0 for stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # Gradient of 2·n·smooth_max (the factor 4·n comes from the derivative of a*a)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a


def _conv_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution (length 2·n‑1)."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
        {x | x_i ≥ 0, Σ x_i = 1}
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}   (normalised)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _power_law_offset_distribution(n: int, p: float, offset: float) -> np.ndarray:
    """a_i ∝ (i+offset)^{-p}   (normalised)."""
    idx = np.arange(offset, offset + n, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _double_power_law_distribution(n: int, p: float, q: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}·(n-i)^{-q}   (normalised)."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) ** (-p)) * ((n - i) ** (-q))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _stretched_exp_distribution(n: int, tau_factor: float, gamma: float) -> np.ndarray:
    """
    a_i ∝ exp( - (i/(tau_factor·n))^{gamma} )   (normalised).
    `tau_factor` ∈ (0,1] stretches the decay length.
    """
    tau = max(1e-12, tau_factor * n)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(- (i / tau) ** gamma)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _exp_decay_distribution(n: int, lam: float) -> np.ndarray:
    """a_i ∝ e^{-lam·i}+e^{-lam·(n‑1‑i)}   (normalised)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _logistic_distribution(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ 1/(1+e^{α·(i‑mid)})   (normalised)."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = 1.0 / (1.0 + np.exp(alpha * (i - mid)))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _sym_product_power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ ((i+1)*(n‑i))^{-p}   (symmetric, normalised)."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) * (n - i)) ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _two_level_distribution(n: int, alpha: float, r: float) -> np.ndarray:
    """
    Simple two‑level step distribution:
        first ⌈α·n⌉ entries = high value,
        remaining entries = low value,
    with ratio high/low = r.
    The vector is normalised to sum = 1.
    """
    m = max(1, int(alpha * n))
    low_len = n - m
    if low_len < 0:
        m = n
        low_len = 0
    denom = m * r + low_len
    low_val = 1.0 / denom
    high_val = r * low_val
    a = np.full(n, low_val, dtype=np.float64)
    a[:m] = high_val
    return a

        ("exp_decay", _exp_decay_distribution,
         lambda: dict(lam=np.random.uniform(1e-6, 0.5))),
        ("logistic", _logistic_distribution,
         lambda: dict(alpha=np.random.uniform(0.1, 20.0))),
        ("sym_product_power_law", _sym_product_power_law_distribution,
         lambda: dict(p=np.random.uniform(0.1, 1.5))),
        ("two_level", _two_level_distribution,
         lambda: dict(alpha=np.random.uniform(0.01, 0.9),
                      r=10 ** np.random.uniform(0.0, 2.5)))  # r ∈ [1, ≈ 316]
    ]

    max_samples_per_family = 250
    print("[Sweep] Starting broad random sweep")
    for name, generator, sampler in families:
        for _ in range(max_samples_per_family):
            if time.time() >= deadline:
                break
            params = sampler()
            try:

            except Exception:

            cand = np.maximum(cand, 0.0)
            score = 2.0 * n * _conv_max(cand)
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                print(f"[Sweep] New best {name} {params} -> {score:.6f}")
        if time.time() >= deadline:
            break

    print(f"[Sweep] Best cheap score after sweep = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Mirror‑descent (Adam) on the simplex
    # ------------------------------------------------------------------
    logits = _logits_from_distribution(best_seq) + 0.02 * np.random.randn(n)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    base_alpha = 0.35
    eps_factor = 0.20                      # start with a relatively smooth max
    eps_factor_min = 5e-5
    eps_decay = 0.9995

    it = 0
    best_seq_sofar = best_seq.copy()
    best_score_sofar = best_score
    last_improve = 0
    report_every = 20000

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

        v_hat = v / (1.0 - beta2 ** it)

    best_score = best_score_sofar
    print(f"[Adam] Finished after {it} iterations, best cheap = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Exact sub‑gradient descent on the max‑shift
    # ------------------------------------------------------------------
    a = best_seq.copy()
    cheap_score = best_score
    it_sub = 0
    last_improve_sub = 0

    while time.time() < deadline and it_sub < 30_000:
        it_sub += 1
        conv_full = _conv_full(a)
        max_conv = float(conv_full.max())
        cheap_curr = 2.0 * n * max_conv

            cheap_score = cheap_curr
            best_seq = a.copy()
            best_score = cheap_score

    # 4. Final cheap random walk (coordinate moves)

            cheap_score = new_score

        print(f"Final exact score = {exact_score:.6f}")
