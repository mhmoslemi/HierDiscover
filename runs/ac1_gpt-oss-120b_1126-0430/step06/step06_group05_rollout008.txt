"""
Annealed‑Adam optimiser for the max‑self‑convolution problem.

The optimiser works in two stages:
1.  A “soft‑max” Adam loop where the objective is the smooth
    approximation  τ·log∑exp(conv/τ).  The temperature τ is exponentially
    decreased from 1e‑3 to 1e‑9, allowing the optimiser to first move in a
    smooth landscape and then focus on the true hard‑max.
2.  A short hard‑max Adam polishing step (identical to the reference
    implementation) followed by a light simulated‑annealing refinement.

All updates are performed in log‑it (exponentiated‑gradient) space,
so the simplex constraint (non‑negative entries summing to one) is
preserved automatically.  The gradient of the smooth max is obtained
by a weighted sum of the per‑index gradients, which can be evaluated
with a single extra FFT per iteration (cross‑correlation of the weight
vector with the current sequence).

The routine respects the global time budget (default 1000 s) and
returns the best sequence found.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Basic utilities – FFT‑based convolution helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (no padding beyond needed)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return conv_full[: 2 * n - 1]


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of a and b via real FFT."""
    n = a.shape[0]
    m = b.shape[0]
    L = _next_pow2(n + m - 1)
    af = np.fft.rfft(a, n=L)
    bf = np.fft.rfft(b, n=L)
    conv_full = np.fft.irfft(af * bf, n=L)
    return conv_full[: n + m - 1]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces a probability vector)."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


# ----------------------------------------------------------------------
# Annealed Adam optimiser (smooth‑max version)
# ----------------------------------------------------------------------


def _run_annealed_adam(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
) -> Tuple[List[float], float]:
    """
    Adam optimiser that minimises a smoothed version of the maximal
    self‑convolution.  Returns the best sequence (as a list) and its
    objective value 2·n·max(conv) (i.e. the raw bound before the
    evaluate_sequence scaling).
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # ---------- initialise -------------------------------------------------
    if init_a is not None:
        a = np.asarray(init_a, dtype=float).copy()
        if a.size != n:
            a = rng.random(n)
    else:
        a = rng.random(n)

    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n, dtype=float) / n
    else:
        a = a / s

    # work in log‑it space
    x = np.log(a + 1e-16)

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # learning‑rate schedule (exponential decay)
    lr_start = 0.5
    lr_end = 0.001

    # temperature schedule for the smooth‑max
    tau_start = 1e-3
    tau_end = 1e-9

    best_a = a.copy()
    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())
    it = 0

    # pre‑compute FFT length for the gradient convolution (a * w_rev)
    L_grad = _next_pow2(3 * n - 2)

    while time.time() - start < time_limit:
        it += 1

        # current point on the simplex
        a = _softmax(x)

        # self‑convolution and current bound
        conv = _conv_self_fft(a)
        M = conv.max()
        obj = float(2.0 * n * M)

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[anneal n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        # ---------- temperature ----------
        elapsed = time.time() - start
        t_frac = min(elapsed / time_limit, 1.0)
        tau = tau_start * (tau_end / tau_start) ** t_frac
        if tau < 1e-12:
            tau = 1e-12

        # ---------- smooth‑max weights ----------
        max_conv = M
        diff = conv - max_conv            # ≤ 0
        # stable exponentiation
        w_exp = np.exp(diff / tau)
        w = w_exp / w_exp.sum()           # shape (2n‑1,)

        # ---------- gradient of the smooth‑max ----------
        # g_a = 2 * Σ_i w_i * ∂conv_i/∂a  = 2 * cross‑correlation(w, a)
        w_rev = w[::-1]                   # reverse for cross‑correlation
        # FFT based cross‑correlation via convolution a * w_rev
        Af = np.fft.rfft(a, n=L_grad)
        Wf = np.fft.rfft(w_rev, n=L_grad)
        conv_rev = np.fft.irfft(Af * Wf, n=L_grad)
        # needed slice: indices (2n‑2)‑k for k=0‥n‑1 → slice [n‑1 : 2n‑1] reversed
        slice_part = conv_rev[n - 1 : 2 * n - 1]
        g_a = 2.0 * slice_part[::-1]      # shape (n,)

        # ---------- mirror (natural) gradient in log‑it space ----------
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)

        # ---------- Adam step ----------
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t_frac)

        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to shake out of plateaus
        if it % 4000 == 0:
            x += rng.normal(scale=0.001, size=n)

        # periodic restart from the current best (helps escape)
        if it % 60000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.005, size=n)
            m.fill(0.0)
            v.fill(0.0)

    # ------------------------------------------------------------------
    # prune negligible entries (they only increase n)
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = float(2.0 * best_a.size * _conv_self_fft(best_a).max())
        print(f"[anneal compress] reduced length → {best_a.size}, O={best_obj:.6f}")

    return best_a.tolist(), best_obj

def _run_opt_hardmax(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
) -> Tuple[List[float], float]:
    """
    Adam optimiser that always uses the exact max‑convolution gradient.
    Returns the best sequence (as a list) and its objective value
    2·n·max(conv(a,a)).
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
    else:
        a = rng.random(n)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n, dtype=float) / n
    else:
        a = a / s

    x = np.log(a + 1e-16)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    lr_start = 0.5
    lr_end = 0.001

    best_a = a.copy()
    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())

    it = 0
    while time.time() - start < time_limit:
        it += 1

        a = _softmax(x)

        conv = _conv_self_fft(a)
        M = conv.max()
        obj = float(2.0 * n * M)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[hardmax n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        k_max = int(np.argmax(conv))
        # exact gradient for the maximal entry
        nlen = a.shape[0]
        idx = np.arange(nlen)
        j = k_max - idx
        mask = (j >= 0) & (j < nlen)
        g_a = np.zeros_like(a)
        if mask.any():
            g_a[mask] = 2.0 * a[j[mask]]

        # natural gradient in log‑it space
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)

        elapsed = time.time() - start
        t = min(elapsed / time_limit, 1.0)
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t)

        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if it % 5000 == 0:
            x += rng.normal(scale=0.01, size=n)

        if it % 80000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.005, size=n)
            m.fill(0.0)
            v.fill(0.0)

    # compress negligible entries
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = float(2.0 * best_a.size * _conv_self_fft(best_a).max())
        print(f"[hardmax compress] reduced length → {best_a.size}, O={best_obj:.6f}")

    return best_a.tolist(), best_obj

def _run_sa(
    init_seq: List[float],
    time_limit: float,
    seed: int = 0,
) -> Tuple[List[float], float]:
    """Refine a sequence by small mass‑transfer moves with a Metropolis schedule."""
    start = time.time()
    rng = np.random.default_rng(seed)
    a = np.asarray(init_seq, dtype=float)
    if a.sum() < 1e-12:
        a = np.ones_like(a) / a.size
    else:
        a = a / a.sum()
    n = a.shape[0]

    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())
    best_seq = a.copy()

    # temperature schedule
    T = best_obj * 0.1 + 1e-6
    T_min = best_obj * 1e-6 + 1e-9
    decay = 0.995
    it = 0

    while time.time() - start < time_limit:
        it += 1
        k_max = int(np.argmax(conv))

        i_vals = np.arange(n)
        j_vals = k_max - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i_vals[mask]] * a[j_vals[mask]]
        total_contrib = contrib.sum()

        if total_contrib > 0:
            probs = contrib / total_contrib
            i = rng.choice(n, p=probs)
        else:
            i = rng.integers(n)

        # distinct destination
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.clip(a_new, 0.0, None)
        a_new /= a_new.sum()

        conv_new = _conv_self_fft(a_new)
        obj_new = float(2.0 * n * conv_new.max())

        if obj_new < best_obj - 1e-12:
            a, conv, best_obj, best_seq = a_new, conv_new, obj_new, a_new.copy()
            print(f"[SA it={it}] new best O={best_obj:.6f}")
        else:
            prob = np.exp(-(obj_new - best_obj) / T) if T > 0 else 0.0
            if rng.random() < prob:
                a, conv = a_new, conv_new

        if it % 2000 == 0:
            T = max(T * decay, T_min)

        if it % 10000 == 0:
            noise = rng.normal(scale=0.01, size=n)
            a = np.clip(a + noise, 0.0, None)
            a /= a.sum()
            conv = _conv_self_fft(a)

    # final compression of tiny entries
    thresh = 1e-12
    mask = best_seq > thresh
    if mask.sum() < best_seq.size:
        comp = best_seq[mask]
        comp /= comp.sum()
        comp_obj = float(2.0 * comp.size * _conv_self_fft(comp).max())
        if comp_obj < best_obj:
            best_seq, best_obj = comp, comp_obj
            print(f"[SA compress] reduced length → {comp.size}, O={best_obj:.6f}")

    return best_seq.tolist(), best_obj

def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **_ignore,
) -> List[float]:
    """
    Find a non‑negative step‑function minimizing the evaluation metric.
    The routine first performs an annealed‑Adam optimisation, then a
    short hard‑max Adam polishing phase and finally a light simulated‑annealing
    refinement.  The best sequence found before the deadline is returned.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # initialise from a known construction if available
    init_seq = None
    if 'height_sequence_1' in globals():
        seq0 = np.asarray(height_sequence_1, dtype=float)
        seq0 = np.clip(seq0, 0.0, None)
        if seq0.sum() < 0.01:
            seq0 = np.ones_like(seq0)
        seq0 = seq0 / seq0.sum()
        init_seq = seq0

    # ------------------------------------------------------------------
    # target support size – keep it comparable to the reference (≈12k)
    target_n = 12000
    remaining = deadline - time.time()
    anneal_budget = remaining * 0.55
    hardmax_budget = remaining * 0.30
    sa_budget = remaining - (anneal_budget + hardmax_budget)

    # up‑sample initial sequence if needed
    if init_seq is not None and init_seq.size != target_n:
        x_old = np.linspace(0.0, 1.0, init_seq.size, endpoint=False)
        x_new = np.linspace(0.0, 1.0, target_n, endpoint=False)
        init_a = np.interp(x_new, x_old, init_seq)
        init_a = np.clip(init_a, 0.0, None)
        init_a = init_a / init_a.sum()
    else:
        init_a = init_seq

    # ------------------------------------------------------------------
    # (1) annealed Adam optimisation
    seq, obj = _run_annealed_adam(
        n=target_n,
        time_limit=anneal_budget,
        init_a=init_a,
        seed=seed + 111,
    )
    best_seq = np.asarray(seq, dtype=float)
    best_obj = obj
    print(f"[global] after annealed Adam O={best_obj:.6f} (n={target_n})")

    # ------------------------------------------------------------------
    # (2) short hard‑max Adam polishing
    remaining = deadline - time.time()
    if remaining > 5.0:
        seq2, obj2 = _run_opt_hardmax(
            n=best_seq.size,
            time_limit=min(hardmax_budget, remaining * 0.9),
            init_a=best_seq,
            seed=seed + 222,
        )
        if obj2 < best_obj:
            best_obj, best_seq = obj2, np.asarray(seq2, dtype=float)
            print(f"[global] after hard‑max polish O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # (3) simulated‑annealing fine‑tuning
    remaining = deadline - time.time()
    if remaining > 5.0:
        seq3, obj3 = _run_sa(best_seq.tolist(), time_limit=remaining, seed=seed + 333)
        if obj3 < best_obj:
            best_obj, best_seq = obj3, np.asarray(seq3, dtype=float)
            print(f"[global] after SA final O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check – evaluate via the supplied evaluator.
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_seq.tolist()
