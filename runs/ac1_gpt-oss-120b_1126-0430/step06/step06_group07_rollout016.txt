    α = r * β
    a = np.full(n, β, dtype=np.float64)
    a[:L+1] = α
    a[-(L+1):] = α
    return a

"""
Propose a non‑negative step function (list of heights) that minimises
``evaluate_sequence``.  The algorithm proceeds in four stages:

1. **Length selection** – re‑use a previous solution if available,
   otherwise fall back to a length of 5000 (adjustable up to 20 000).

2. **Broad parametric sweep** – evaluates a collection of parametric families
   (symmetric exponential, symmetric power‑law, a simple two‑level “U” shape,
   and random Dirichlet draws).  The symmetric families are refined with a
   cheap golden‑section line search to obtain a near‑optimal continuous
   parameter.

3. **Projected sub‑gradient descent** – starting from the best candidate,
   performs many additive sub‑gradient steps (the true gradient of the
   active convolution entry) with a diminishing step‑size and projection onto
   the probability simplex.  Restarts are triggered after long stagnation.

4. **Continuous simulated‑annealing local search** – repeatedly transfers a
   small amount of mass between two random positions, accepting worsening
   moves with an annealed probability.  The best vector found is finally
   evaluated with ``evaluate_sequence`` and returned.

All helper utilities are defined at module level; no closures or lambdas are
used.  The routine respects a user‑supplied time budget (default 1000 s) and
always returns a feasible sequence.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using a power‑of‑two FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa).real[: 2 * a.size - 1]
    return conv


def _conv_max_fft(a: np.ndarray, fft_len: int) -> tuple[float, int]:
    """Maximum value of the autocorrelation a*a and its index."""
    conv = _conv_full_fft(a, fft_len)
    idx = int(conv.argmax())
    return float(conv[idx]), idx


def _compute_subgradient(a: np.ndarray, kmax: int) -> np.ndarray:
    """Exact sub‑gradient of the maximal convolution entry w.r.t. a."""
    n = a.shape[0]
    lo = max(0, kmax - (n - 1))
    hi = min(kmax, n - 1)
    rng = np.arange(lo, hi + 1, dtype=np.int64)
    j = kmax - rng
    g = np.zeros_like(a)
    eq = (rng == j)
    # off‑diagonal contributions (appear twice in the sum)
    g[rng[~eq]] = 2.0 * a[j[~eq]]
    # diagonal contribution (i == j)
    g[rng[eq]] = a[rng[eq]]
    return g

        return a

    c = hi - invphi * (hi - lo)
    d = lo + invphi * (hi - lo)
    a_c = _sym_exp_vec(c)
    val_c = cheap_score(a_c)
    a_d = _sym_exp_vec(d)
    val_d = cheap_score(a_d)

    while (hi - lo) > tol and time.time() < deadline:
        if val_c < val_d:
            hi = d
            d = c
            val_d = val_c
            c = hi - invphi * (hi - lo)
            a_c = _sym_exp_vec(c)
            val_c = cheap_score(a_c)
        else:
            lo = c
            c = d
            val_c = val_d
            d = lo + invphi * (hi - lo)
            a_d = _sym_exp_vec(d)
            val_d = cheap_score(a_d)

    # pick the better endpoint
    if val_c < val_d:
        if val_c < best_val:
            best_a, best_val = a_c.copy(), val_c
            print(f"[Refine] sym‑exp λ={c:.4f} cheap={val_c:.6f}")
    else:
        if val_d < best_val:
            best_a, best_val = a_d.copy(), val_d
            print(f"[Refine] sym‑exp λ={d:.4f} cheap={val_d:.6f}")

    # ----- symmetric power‑law -----
    def _sym_pow_vec(p):
        i = np.arange(n, dtype=np.float64)
        a = np.minimum(i + 1, n - i) ** (-p)
        a /= a.sum()
        return a

    lo, hi = 0.01, 5.0
    c = hi - invphi * (hi - lo)
    d = lo + invphi * (hi - lo)
    a_c = _sym_pow_vec(c)
    val_c = cheap_score(a_c)
    a_d = _sym_pow_vec(d)
    val_d = cheap_score(a_d)

    while (hi - lo) > tol and time.time() < deadline:
        if val_c < val_d:
            hi = d
            d = c
            val_d = val_c
            c = hi - invphi * (hi - lo)
            a_c = _sym_pow_vec(c)
            val_c = cheap_score(a_c)
        else:
            lo = c
            c = d
            val_c = val_d
            d = lo + invphi * (hi - lo)
            a_d = _sym_pow_vec(d)
            val_d = cheap_score(a_d)

    if val_c < val_d:
        if val_c < best_val:
            best_a, best_val = a_c.copy(), val_c
            print(f"[Refine] sym‑pow p={c:.4f} cheap={val_c:.6f}")
    else:
        if val_d < best_val:
            best_a, best_val = a_d.copy(), val_d
            print(f"[Refine] sym‑pow p={d:.4f} cheap={val_d:.6f}")

    print(f"[Init] best cheap after families = {best_val:.6f}")

    # --------------------------------------------------------------
    # 2. Projected sub‑gradient descent (additive updates)
    # --------------------------------------------------------------
    a = best_a.copy()
    best_iter_a = a.copy()
    best_iter_val = best_val
    last_improve = 0
    iter_cnt = 0
    eta0 = 0.7
    max_iters = 2_000_000

    while time.time() < deadline and iter_cnt < max_iters:
        # full convolution and cheap score (sum is 1)
        conv = _conv_full_fft(a, fft_len)
        maxb = conv.max()
        cheap = 2.0 * n * maxb

        if cheap < best_iter_val - 1e-12:
            best_iter_val = cheap
            best_iter_a = a.copy()
            last_improve = iter_cnt
            print(f"[GD {iter_cnt:07d}] NEW BEST cheap = {cheap:.6f}")

        # exact sub‑gradient at the maximal entry
        kmax = int(conv.argmax())
        g = _compute_subgradient(a, kmax)

        iter_cnt += 1
        eta = eta0 / math.sqrt(iter_cnt + 1)

        # additive step with projection onto the simplex
        a = a - eta * g
        # clipping ensures non‑negativity before projection
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s <= 0.0 or not np.isfinite(s):
            a = best_iter_a.copy()
            continue
        a = a / s

        # occasional restart if no improvement for a long stretch
        if iter_cnt - last_improve > 200_000:
            pert = np.random.rand(n) * 0.01
            a = _simplex_projection(best_iter_a + pert)
            last_improve = iter_cnt
            print(f"[GD {iter_cnt:07d}] Restart after stagnation.")

    # --------------------------------------------------------------
    # 3. Continuous simulated‑annealing local search (mass swaps)
    # --------------------------------------------------------------
    a = best_iter_a.copy()
    best_swap_a = a.copy()
    best_swap_val = best_iter_val
    temperature = 1.0
    cooling = 0.99995

    while time.time() < deadline:
        # propose moving a random mass amount from i -> j
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        if a[i] <= 0.0:
            continue
        # move up to 10 % of the source mass
        amount = np.random.rand() * min(a[i], 0.1)
        a[i] -= amount
        a[j] += amount

        cheap = cheap_score(a)

        if cheap < best_swap_val - 1e-12:
            best_swap_val = cheap
            best_swap_a = a.copy()
            print(f"[SA] NEW BEST cheap = {cheap:.6f}")
        else:
            # Metropolis acceptance
            delta = cheap - best_swap_val
            prob = math.exp(-delta / max(temperature, 1e-12))
            if np.random.rand() < prob:
                # accept
                pass
            else:
                # revert
                a[i] += amount
                a[j] -= amount

        temperature = max(temperature * cooling, 1e-6)

    # --------------------------------------------------------------
    # 4. Final exact evaluation
    # --------------------------------------------------------------
    final_seq = best_swap_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:          # defensive – should never happen
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    return final_seq
