"""
non‑negative step‑function optimiser (dual‑temperature Adam + SA).

The optimiser works in three stages within the allotted time budget:

1. **Dual‑temperature Adam** –  a mirrored‑gradient Adam on the
   probability simplex, where the objective is a *soft‑max* of the
   self‑convolution values.
   The temperature (soft‑max “τ”) is exponentially cooled, so the
   optimiser first sees a smooth landscape and gradually focuses on
   the true max‑convolution.  Gradients are obtained analytically via
   FFT‑based convolutions (O(n log n) per iteration).

2. **Hard‑max Adam** – a short refinement that uses the exact
   sub‑gradient of the maximal convolution entry (the method from the
   reference implementation).  This helps to “snap’’ onto the final
   optimal support.

3. **Targeted simulated‑annealing** – a classic SA that moves a
   modest fraction of mass between indices, accepting uphill moves
   with a temperature schedule.  It finishes the search with a fine‑grained
   local optimisation.

All three stages share the same simplex‑projection via the log‑it
parameterisation (softmax of logits).  Tiny entries are pruned after
each stage to reduce the support size (and therefore the final
objective, which contains a factor `2·n`).  The best sequence seen
throughout the whole run is returned.

The routine respects the overall time budget (default 1000 s) and
produces a list of non‑negative floats that can be passed directly to
``evaluate_sequence``.
"""

import time
import numpy as np
from typing import List, Tuple, Optional

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution a*b using a real FFT (output length = len(a)+len(b)-1)."""
    n = a.shape[0]
    m = b.shape[0]
    L = _next_pow2(n + m - 1)
    Af = np.fft.rfft(a, n=L)
    Bf = np.fft.rfft(b, n=L)
    conv_full = np.fft.irfft(Af * Bf, n=L)
    return conv_full[: n + m - 1]


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (real FFT)."""
    return _conv_fft(a, a)


# ----------------------------------------------------------------------
# Soft‑max Adam optimiser (temperature‑annealed)
# ----------------------------------------------------------------------


def _run_opt_softmax_adam(
    n: int,
    time_limit: float,
    init_a: Optional[np.ndarray] = None,
    seed: int = 0,
) -> Tuple[np.ndarray, float]:
    """
    Mirrors Adam on the log‑it space using a soft‑max of the self‑convolution.
    Returns the best sequence (as a NumPy array) and its objective value.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # initialise a on the simplex
    if init_a is not None and init_a.size == n:
        a = np.array(init_a, dtype=float).copy()
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s < 1e-12:
            a = np.ones(n, dtype=float) / n
        else:
            a = a / s
    else:
        a = rng.random(n)
        a = a / a.sum()

    # logits for exponentiated‑gradient updates (mirror descent on simplex)
    x = np.log(a + 1e-16)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # learning‑rate schedule (exponential decay)
    lr_start, lr_end = 0.6, 0.002

    # temperature (soft‑max τ) schedule (exponential cooling)
    # τ will go from τ_start ≈ 0.5·max(conv) to τ_end ≈ 1e‑10
    # we compute τ dynamically from elapsed fraction of the phase.
    tau_start_factor = 0.5   # fraction of current max(conv) at the *beginning* of the phase
    tau_end = 1e-10

    best_a = a.copy()
    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())
    it = 0

    # main optimisation loop
    while time.time() - start < time_limit:
        it += 1

        # ---- simplex point ----
        a = np.exp(x - np.max(x))          # avoid overflow
        a = a / a.sum()

        # ---- convolution ----
        conv = _conv_self_fft(a)
        M = conv.max()
        obj = float(2.0 * n * M)

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[softmax Adam n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        # ---- temperature ----
        elapsed = time.time() - start
        phase_frac = min(elapsed / time_limit, 1.0)
        # start τ as a fraction of the *initial* max‑conv; keep it positive
        tau_start = max(tau_start_factor * M, 1e-12)
        tau = max(tau_end,
                  tau_start * np.exp(np.log(tau_end / tau_start) * phase_frac))

        # ---- soft‑max weights ----
        # w_i = exp(conv_i / τ) / Σ_j exp(conv_j / τ)
        # for stability shift by max(conv)
        shifted = (conv - conv.max()) / tau
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum == 0.0:          # numerical safeguard
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        # ---- gradient of the smooth max (in a‑space) ----
        # g_a = 2 * Σ_i w_i * a_{i‑k}
        w_rev = w[::-1]                      # reversed weight vector
        conv_weight = _conv_fft(a, w_rev)    # length = n + (2n‑1) – 1 = 3n‑2
        idx = (2 * n - 2) - np.arange(n)    # indices that give Σ_i w_i a_{i‑k}
        g_a = 2.0 * conv_weight[idx]

        # ---- mirror‑gradient (log‑it space) ----
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)

        # ---- learning‑rate (exponential decay) ----
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * phase_frac)

        # ---- Adam moments ----
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # ---- Adam step ----
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to shake out flat plateaus
        if it % 4000 == 0:
            jitter = rng.normal(scale=0.008, size=n)
            x += jitter

        # occasional full restart from the current best (helps escape)
        if it % 60000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.003, size=n)
            m.fill(0.0)
            v.fill(0.0)

    # ------------------------------------------------------------------
    # prune negligible entries (they increase n without helping)
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        conv = _conv_self_fft(best_a)
        best_obj = float(2.0 * best_a.size * conv.max())
        print(f"[softmax compress] reduced length → {best_a.size}, O={best_obj:.6f}")

    return best_a, best_obj


# ----------------------------------------------------------------------
# Hard‑max Adam optimiser (exact sub‑gradient of the maximal entry)
# ----------------------------------------------------------------------


def _run_opt_hardmax(
    n: int,
    time_limit: float,
    init_a: Optional[np.ndarray] = None,
    seed: int = 0,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser that uses the exact max‑convolution gradient.
    Returns the best sequence and its objective value.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # initialise a on the simplex
    if init_a is not None and init_a.size == n:
        a = np.array(init_a, dtype=float)
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s < 1e-12:
            a = np.ones(n, dtype=float) / n
        else:
            a = a / s
    else:
        a = rng.random(n)
        a = a / a.sum()

    # logits for exponentiated updates
    x = np.log(a + 1e-16)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # learning‑rate schedule
    lr_start, lr_end = 0.5, 0.001

    best_a = a.copy()
    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())
    it = 0

    while time.time() - start < time_limit:
        it += 1

        a = np.exp(x - np.max(x))
        a = a / a.sum()

        conv = _conv_self_fft(a)
        M = conv.max()
        obj = float(2.0 * n * M)

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[hardmax Adam n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        # index of the current maximal convolution entry
        k_max = int(np.argmax(conv))

        # exact gradient of that entry
        idx = np.arange(n)
        j = k_max - idx
        mask = (j >= 0) & (j < n)
        g_a = np.zeros_like(a)
        g_a[mask] = 2.0 * a[j[mask]]

        # mirror‑gradient in log‑it space
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)

        # learning‑rate (exponential decay)
        elapsed = time.time() - start
        phase_frac = min(elapsed / time_limit, 1.0)
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * phase_frac)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # jitter & restart
        if it % 5000 == 0:
            x += rng.normal(scale=0.01, size=n)

        if it % 80000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.005, size=n)
            m.fill(0.0)
            v.fill(0.0)

    # prune negligible entries
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        conv = _conv_self_fft(best_a)
        best_obj = float(2.0 * best_a.size * conv.max())
        print(f"[hardmax compress] reduced length → {best_a.size}, O={best_obj:.6f}")

    return best_a, best_obj


# ----------------------------------------------------------------------
# Targeted simulated‑annealing refinement (unchanged)
# ----------------------------------------------------------------------


def _run_sa(
    init_seq: List[float],
    time_limit: float,
    seed: int = 0,
) -> Tuple[np.ndarray, float]:
    """Refine a sequence by small mass‑transfer moves with a Metropolis schedule."""
    start = time.time()
    rng = np.random.default_rng(seed)
    a = np.asarray(init_seq, dtype=float)
    if a.sum() < 1e-12:
        a = np.ones_like(a) / a.size
    else:
        a = a / a.sum()
    n = a.shape[0]

    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())
    best_seq = a.copy()

    # temperature schedule
    T = best_obj * 0.1 + 1e-9
    T_min = best_obj * 1e-6 + 1e-12
    decay = 0.995
    it = 0

    while time.time() - start < time_limit:
        it += 1
        k_max = int(np.argmax(conv))

        # contribution of each index to the maximal entry
        i_vals = np.arange(n)
        j_vals = k_max - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i_vals[mask]] * a[j_vals[mask]]
        total_contrib = contrib.sum()

        if total_contrib > 0:
            probs = contrib / total_contrib
            i = rng.choice(n, p=probs)
        else:
            i = rng.integers(n)

        # random destination distinct from i
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # transfer a modest fraction of mass
        delta = a[i] * rng.random() * 0.06      # up to 6 % of a[i]
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.clip(a_new, 0.0, None)
        a_new /= a_new.sum()

        conv_new = _conv_self_fft(a_new)
        obj_new = float(2.0 * n * conv_new.max())

        # Metropolis acceptance
        if obj_new < best_obj - 1e-12:
            a, conv, best_obj, best_seq = a_new, conv_new, obj_new, a_new.copy()
            print(f"[SA it={it}] new best O={best_obj:.6f}")
        else:
            prob = np.exp(-(obj_new - best_obj) / T) if T > 0 else 0.0
            if rng.random() < prob:
                a, conv = a_new, conv_new

        # temperature decay
        if it % 2000 == 0:
            T = max(T * decay, T_min)

        # occasional global jitter
        if it % 10000 == 0:
            noise = rng.normal(scale=0.015, size=n)
            a = np.clip(a + noise, 0.0, None)
            a /= a.sum()
            conv = _conv_self_fft(a)

    # final compression of tiny entries
    thresh = 1e-12
    mask = best_seq > thresh
    if mask.sum() < best_seq.size:
        comp = best_seq[mask]
        comp /= comp.sum()
        comp_obj = float(2.0 * comp.size * _conv_self_fft(comp).max())
        if comp_obj < best_obj:
            best_seq, best_obj = comp, comp_obj
            print(f"[SA compress] reduced length → {comp.size}, O={best_obj:.6f}")

    return best_seq, best_obj

# Main entry point required by the competition harness

def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **_ignore,
) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The algorithm runs three stages (soft‑max Adam → hard‑max Adam → SA)
    within the given time budget and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1) obtain an initial sequence (use the globally stored one if present)
    # ------------------------------------------------------------------
    init_seq: Optional[np.ndarray] = None
    if "height_sequence_1" in globals():
        seq0 = np.asarray(globals()["height_sequence_1"], dtype=float)
        seq0 = np.clip(seq0, 0.0, None)
        if seq0.sum() < 0.01:
            seq0 = np.ones_like(seq0)
        seq0 = seq0 / seq0.sum()
        init_seq = seq0

    # ------------------------------------------------------------------
    # 2) Choose a target length (large enough to give flexibility)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining < 10.0:
        raise RuntimeError("Not enough time to run the optimiser.")
    # Allocate time: 60% soft‑max Adam, 20% hard‑max Adam, 20% SA
    softmax_budget = remaining * 0.60
    hardmax_budget = remaining * 0.20
    sa_budget = remaining - softmax_budget - hardmax_budget

    # Aim for a long support unless the remaining time is tiny
    target_n = 12000 if softmax_budget > 30.0 else 8000

    # Upsample an existing init_seq to the target length if needed
    if init_seq is not None and init_seq.size != target_n:
        x_old = np.linspace(0.0, 1.0, init_seq.size, endpoint=False)
        x_new = np.linspace(0.0, 1.0, target_n, endpoint=False)
        init_a = np.interp(x_new, x_old, init_seq)
        init_a = np.clip(init_a, 0.0, None)
        init_a = init_a / init_a.sum()
    else:
        init_a = init_seq

    # ------------------------------------------------------------------
    # 3) Soft‑max Adam (dual‑temperature) optimisation
    # ------------------------------------------------------------------
    seq_soft, obj_soft = _run_opt_softmax_adam(
        n=target_n,
        time_limit=softmax_budget,
        init_a=init_a,
        seed=seed + 111,
    )
    best_seq = seq_soft
    best_obj = obj_soft
    print(f"[global] after soft‑max Adam best O={best_obj:.6f} (n={target_n})")

    # ------------------------------------------------------------------
    # 4) Short hard‑max Adam refinement (if time permits)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0 and hardmax_budget > 2.0:
        seq_hard, obj_hard = _run_opt_hardmax(
            n=best_seq.size,
            time_limit=min(hardmax_budget, remaining),
            init_a=best_seq,
            seed=seed + 222,
        )
        if obj_hard < best_obj:
            best_seq, best_obj = seq_hard, obj_hard
            print(f"[global] after hard‑max Adam new best O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # 5) Simulated‑annealing refinement (if time permits)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        seq_sa, obj_sa = _run_sa(
            best_seq.tolist(),
            time_limit=remaining,
            seed=seed + 333,
        )
        if obj_sa < best_obj:
            best_seq, best_obj = seq_sa, obj_sa
            print(f"[global] after SA new best O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # 6) Final sanity check – call the supplied evaluator
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)
        final_score = float("inf")

    # Return the best sequence (as a plain Python list)
    return best_seq.tolist()
