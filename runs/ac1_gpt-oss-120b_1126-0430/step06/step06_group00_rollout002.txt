"""
Propose a step‑function (non‑negative sequence) that minimises the
evaluation metric 2·n·max(a*a)/(sum(a)^2).

Algorithm
---------
1. Choose the support size `n`.  If a sequence called `height_sequence_1`
   is already present we reuse its length, otherwise we fall back to
   `n = 20000`.

2. Initialise the vector from the exact discretisation of the arcsine
   distribution (the best known analytical starting point).

3. Run a **long‑running Adam optimisation on the probability simplex**
   using a smoothed‐max surrogate for the L∞‑norm of the self‑convolution.
   *   The smoothing constant (`eps_factor`) starts relatively large
       (≈ 0.3) and is decayed very slowly, so the optimiser first sees a
       broad, well‑behaved objective and only later focuses on the true
       max.
   *   The learning‑rate base (`base_alpha`) is increased and the
       decay is `1/√t`, providing fast early progress and fine‑grained
       updates later on.
   *   Random jitter is injected every few ten‑thousands of iterations
       and a simple restart is triggered if the best score has not
       improved for a long stretch.

4. When the time budget is almost exhausted a short **fine‑tuning**
   phase with an almost‑hard max (ε≈10⁻¹²) is performed to polish the
   solution.

5. The best sequence seen during the whole run is returned as a plain
   Python list.  The platform will call `evaluate_sequence` on it, but we
   also log the final exact score for debugging.

The whole optimisation respects the 1000 s budget, stays on the
probability simplex (all entries ≥ 0 and sum = 1) and works entirely with
NumPy/FFT – no external solvers are required.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _fft_len_for_corr(n: int) -> int:
    """
    Smallest power‑of‑two ≥ 3·n‑2.
    Sufficient for linear convolution (length 2·n‑1) and correlation
    (length 3·n‑2) used in the gradient computation.
    """
    return 1 << ((3 * n - 2).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a (via FFT)."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Smoothed max of a*a and gradient w.r.t. a.

    Parameters
    ----------
    a : np.ndarray
        Probability vector (sum = 1, non‑negative).
    eps_factor : float
        Multiplicative factor for the smoothing constant:
        eps = max(eps_factor * max(conv), 1e‑12).

    Returns
    -------
    max_conv : float
        Maximum entry of the linear self‑convolution.
    grad_a : np.ndarray
        Gradient of 2·n·smooth_max w.r.t. a (already multiplied by 4·n).
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    # FFT of a
    a_fft = np.fft.rfft(a, L)

    # Linear self‑convolution (length 2·n‑1)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # Soft‑max weights over the convolution entries
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv          # for numerical stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or np.isnan(w_sum):
        w.fill(1.0 / w.size)
    else:
        w /= w_sum

    # Cross‑correlation a ⋆ w (first n entries are needed)
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # Gradient (the factor 4·n comes from the derivative of 2·n·max)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a


def log_progress(start_time: float, it: int, score: float):
    """Print a short progress line."""
    elapsed = time.time() - start_time
    print(f"[Iter {it:6d}] best_score = {score:.6f}  elapsed = {elapsed:.1f}s")

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that minimises the evaluation metric.
    The optimisation runs for as many Adam steps as the time budget allows,
    using a slowly decaying smoothing factor and occasional random restarts.
    """
    np.random.seed(seed)
    start_time = time.time()
    # Keep a small safety margin for final logging / cleanup
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # 0. Choose the length of the sequence
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
        print(f"[Info] Reusing external length: n = {n}")
    except Exception:
        n = 20000
        print(f"[Info] No external length – using default n = {n}")

    # ------------------------------------------------------------------
    # 1. Initialise from the exact arcsine CDF (area discretisation)
    # ------------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)
    x0 = idx / n
    x1 = (idx + 1) / n

    def arcsine_cdf(x):
        # Clip to avoid sqrt of a tiny negative due to rounding
        return (2.0 / np.pi) * np.arcsin(np.sqrt(np.clip(x, 0.0, 1.0)))

    a = arcsine_cdf(x1) - arcsine_cdf(x0)
    a_sum = a.sum()
    if a_sum == 0.0 or np.isnan(a_sum):
        raise RuntimeError("Failed to initialise a valid distribution.")
    a /= a_sum                      # now sum(a) == 1

    # ------------------------------------------------------------------
    # 2. Adam optimisation on the simplex (logits = log a)
    # ------------------------------------------------------------------
    best_a = a.copy()
    best_score = 2.0 * n * _conv_max(best_a)

    # Parameterisation via logits (soft‑max) guarantees the simplex constraint
    z = np.log(best_a + 1e-16)
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    # Learning‑rate schedule (larger base, slower decay)
    base_alpha = 0.9          # initial step size for logits
    # Smoothing schedule for the soft‑max of the convolution
    eps_factor = 0.30          # start fairly smooth
    eps_factor_min = 5e-5      # we allow a very hard max at the end
    eps_decay = 0.9998         # decay extremely slowly

    # Stagnation detection
    last_improve_it = 0
    improve_tol = 1e-6

    it = 0
    max_iters = 2_500_000      # very large – actual stop is time‑driven

    # ------------------------------------------------------------------
    # 3. Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline and it < max_iters:
        it += 1

        # Current probability vector (always sums to 1)
        a_cur = _softmax(z)

        # Smoothed max‑convolution and its gradient
        max_conv, grad_a = _conv_max_and_grad(a_cur, eps_factor)

        cheap_score = 2.0 * n * max_conv

        # Keep the best candidate seen so far
        if cheap_score < best_score - improve_tol:
            best_score = cheap_score
            best_a = a_cur.copy()
            last_improve_it = it
            if it % 2000 == 0 or cheap_score < 1.02:
                log_progress(start_time, it, best_score)

        # Mirror‑descent gradient on logits (natural gradient on the simplex)
        mu = np.dot(grad_a, a_cur)          # 〈∇, a〉
        grad_z = a_cur * (grad_a - mu)      # ∇_z J

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # Inverse‑sqrt decay of the step size
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step = - alpha_t * m_hat / (np.sqrt(v_hat) + eps_adam)

        z = z + step
        # Keep logits within a numerically safe range
        z = np.clip(z, -70.0, 70.0)

        # Decay smoothing very slowly
        eps_factor = max(eps_factor_min, eps_factor * eps_decay)

        # Occasionally inject a tiny random jitter to escape flat regions
        if it % 10000 == 0 and np.random.rand() < 0.15:
            z += 0.01 * np.random.randn(n)

        # Simple restart if we have been stuck for a long time
        if it - last_improve_it > 100_000:
            # Re‑initialise logits around the best found point
            z = np.log(best_a + 1e-12) + 0.02 * np.random.randn(n)
            last_improve_it = it
            print(f"[Info] Restart at iteration {it}")

    # ------------------------------------------------------------------
    # 4. Fine‑tuning with an almost‑hard max (ε≈10⁻¹²)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # Reduce smoothing to the smallest possible value
        eps_factor = max(eps_factor_min, 1e-5)
        # Short, aggressive run
        fine_steps = 8_000
        for ft in range(fine_steps):
            if time.time() >= deadline:
                break
            a_cur = _softmax(z)
            max_conv, grad_a = _conv_max_and_grad(a_cur, eps_factor)
            cheap_score = 2.0 * n * max_conv
            if cheap_score < best_score - improve_tol:
                best_score = cheap_score
                best_a = a_cur.copy()
                if ft % 1000 == 0:
                    log_progress(start_time, it + ft, best_score)

            mu = np.dot(grad_a, a_cur)
            grad_z = a_cur * (grad_a - mu)

            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
            m_hat = m / (1.0 - beta1 ** (it + ft + 1))
            v_hat = v / (1.0 - beta2 ** (it + ft + 1))
            alpha_t = base_alpha * 0.4 / (math.sqrt(it + ft + 1) + 1.0)
            step = - alpha_t * m_hat / (np.sqrt(v_hat) + eps_adam)
            z = z + step
            z = np.clip(z, -70.0, 70.0)

    # ------------------------------------------------------------------
    # 5. Return the best sequence (plain Python list)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()

    # Optional final exact evaluation (only for logging; the platform will
    # evaluate the returned sequence anyway).
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[Info] Final exact score = {final_score:.6f}")
    except Exception as exc:
        print(f"[Warning] Evaluation error (ignored): {exc}")

    return final_seq
