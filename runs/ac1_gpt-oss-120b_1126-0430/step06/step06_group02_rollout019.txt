def _lp_step(f, M, t):
    n = len(f)
    # Build convolution matrix
    mat = np.zeros((2*n-1, n), dtype=np.float64)
    for i in range(n):
        mat[i:i+n, i] = f
    # Solve LP
    import cvxpy as cp
    g = cp.Variable(n)
    constraints = [mat @ g <= M, g >= 0]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    if g.value is None:
        # fallback: return f unchanged
        return f
    g_val = np.array(g.value, dtype=np.float64)
    sum_f = f.sum()
    sum_g = g_val.sum()
    if sum_g <= 0:
        return f
    g_scaled = g_val * (sum_f / sum_g)  # maintain sum
    # mix
    new_f = (1 - t) * f + t * g_scaled
    # ensure nonneg and normalising maybe?
    new_f = np.maximum(new_f, 0.0)
    s = new_f.sum()
    if s > 0:
        new_f /= s  # renorm to sum=1
    return new_f

#!/usr/bin/env python
"""
Non‑negative step‑function optimiser – hybrid LP + Adam + local search.

The optimiser works in three stages:

1. **Coarse search on a small support** (≈2000 points).  
   A handful of parametric families (exponential, power‑law,
   logistic, product‑power‑law) are scanned.  The best candidate
   is then refined by a single linear‑programming step (if cvxpy
   is available) that maximises the total mass while keeping the
   convolution bound unchanged.  This produces a good “starting
   shape”.

2. **Fine‑grained Adam optimisation** on a larger support
   (≈10 000–20 000 points).  The coefficients are encoded as
   soft‑max logits and the objective is the smooth approximation
   of `max (a*a)`.  Adaptive Adam moments, a decaying smoothing
   factor `eps` and occasional random jitter allow the algorithm
   to escape plateaux.  The support is up‑sampled once during the
   run to give the optimiser extra degrees of freedom.

3. **Local stochastic refinement** – random mass transfers
   between two positions are accepted only when they improve the
   exact (unsmoothed) objective.  This cheap hill‑climbing step
   often squeezes a few extra percent out of the score.

The whole routine respects the user‑provided time budget and
returns the best sequence found (a plain Python list of floats).
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures or lambdas
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max producing a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    s = e.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / z.size)
    return e / s


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it that reproduces `a` after a soft‑max."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """Smallest power‑of‑two ≥ 3·n‑2 (enough for all needed convolutions)."""
    return 1 << ((3 * n - 2).bit_length())


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    s = new_a.sum()
    if s == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= s
    return new_a


# ---- small parametric families ------------------------------------------------

def _exp_decay_distribution(n: int, lam: float) -> np.ndarray:
    """a_i ∝ e^{‑lam·i}  (purely monotone exponential, not symmetric)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{‑p}, normalised."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _logistic_distribution(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ 1/(1+e^{α·(i‑mid)}), left‑heavy."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = 1.0 / (1.0 + np.exp(alpha * (i - mid)))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _sym_product_power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ ((i+1)*(n‑i))^{‑p}, symmetric, normalised."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) * (n - i)) ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a (fast FFT)."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = rfft(a, L)
    conv = irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Returns (max_conv, grad_a) where

        grad_a = ∇_a [ 2·n·smooth_max(a*a) ]

    `eps_factor` multiplies the smoothing constant ε = eps_factor·max_conv.
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    a_fft = rfft(a, L)
    conv = irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # smooth soft‑max weights
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv            # ≤ 0 for numerical stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    w_fft = rfft(w, L)
    corr = irfft(np.conj(a_fft) * w_fft, L)[:n]

    grad_a = 4.0 * n * corr
    return max_conv, grad_a

def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the dense Toeplitz‑type matrix C (shape (2n‑1, n)) such that
    C @ g == np.convolve(f, g) for any vector g.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    C = np.zeros((rows, n), dtype=np.float64)
    # column j contains f shifted down by j positions
    for j in range(n):
        C[j:j + n, j] = f
    return C


def _lp_step(f: np.ndarray, M: float, mix: float = 0.5):
    """
    One LP improvement step (if cvxpy is available).

    For the current distribution `f` (sum = 1) and the current bound `M`,
    solve
        maximise    sum(g)
        subject to   conv(f, g) ≤ M  (element‑wise)
                     g ≥ 0
    Then mix the result with the old `f` using weight `mix` and renormalise.
    The function returns the mixed distribution.
    """
    try:
        import cvxpy as cp
    except Exception:          # pragma: no cover
        # cvxpy not available – skip the LP step.
        return f

    n = f.shape[0]
    C = _build_conv_matrix(f)                     # (2n‑1) × n matrix
    g = cp.Variable(n, nonneg=True)

    constraints = [C @ g <= M, g >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    # warm start and moderate iteration limit keep it fast
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, abstol=1e-9, reltol=1e-9)

    if g.value is None:
        # LP failed – keep the old distribution.
        return f

    g_val = np.maximum(0.0, np.array(g.value, dtype=np.float64))
    sum_f = f.sum()
    sum_g = g_val.sum()
    if sum_g == 0.0:
        return f

    # renormalise g to have the same total mass as f
    g_scaled = g_val * (sum_f / sum_g)

    # mix & re‑normalise (the mix weight is a hyper‑parameter)
    new_f = (1.0 - mix) * f + mix * g_scaled
    new_f = np.maximum(new_f, 0.0)
    s = new_f.sum()
    if s > 0.0:
        new_f /= s
    return new_f

    # ------------------------------------------------------------------
    # Adam optimisation loop (core)
    # ------------------------------------------------------------------
    while time.time() < deadline and it < 4_000_000:
        it += 1

        a = _softmax(z)
        max_conv, grad_a = _conv_max_and_grad(a, eps_factor)
        cheap_score = 2.0 * n * max_conv

        if cheap_score + 1e-12 < best_score:
            best_score = cheap_score
            best_seq = a.copy()
            last_improve = it
            if it % 50000 == 0:
                print(f"[Iter {it:07d}] NEW BEST cheap = {cheap_score:.6f}")

        # gradient wrt logits (mirror descent on the simplex)
        mu = np.dot(grad_a, a)
        grad_z = a * (grad_a - mu)

        # Adam moment updates (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step‑size schedule (inverse‑sqrt decay)
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step = - m_hat / (np.sqrt(v_hat) + eps_adam)

        # simple back‑tracking line‑search (up to 4 halvings)
        accepted = False
        trial_alpha = alpha_t
        for _ in range(4):
            z_new = z + trial_alpha * step
            z_new = np.clip(z_new, -50.0, 50.0)
            a_new = _softmax(z_new)
            new_score = 2.0 * n * _conv_max(a_new)
            if new_score <= cheap_score:
                z = z_new
                accepted = True
                break
            trial_alpha *= 0.5
        if not accepted:
            # forced move – helps to escape flat regions
            z += alpha_t * step
            z = np.clip(z, -50.0, 50.0)

        # occasional stochastic jitter to shake plateaux
        if (it % 3000 == 0) and (np.random.rand() < 0.18):
            z += 0.03 * np.random.randn(n)
            m *= 0.5
            v *= 0.5

        # smoothness decay
        eps_factor = max(eps_factor_min, eps_factor * eps_decay)

        # stagnation handling – restart from the current best
        if it - last_improve > 300_000:
            print(f"[Iter {it}] Stagnation – restarting with jitter.")
            z = _logits_from_distribution(best_seq) + 0.04 * np.random.randn(n)
            m.fill(0.0)
            v.fill(0.0)
            eps_factor = 0.3
            last_improve = it

        # one optional up‑sampling stage (once)
        if (it % 600_000 == 0) and (n < 18000) and (time.time() + 30.0 < deadline):
            new_n = min(18000, int(n * 1.4))
            print(f"[Iter {it}] Upsampling support: {n} → {new_n}")
            best_seq = _interpolate_distribution(best_seq, new_n)
            z = _logits_from_distribution(best_seq) + 0.005 * np.random.randn(new_n)
            m = np.zeros_like(z)
            v = np.zeros_like(z)
            n = new_n
            best_score = 2.0 * n * _conv_max(best_seq)
            # reset counters after a drastic change
            it = 0
            last_improve = 0
            eps_factor = 0.3

    # ------------------------------------------------------------------
    # 3. Local stochastic refinement (mass‑transfer hill‑climbing)
    # ------------------------------------------------------------------
    a = best_seq
    max_conv = _conv_max(a)
    best_score = 2.0 * n * max_conv
    print(f"[Refine] Starting local search, score = {best_score:.6f}")

    while time.time() < deadline:
        # propose a tiny mass transfer between two random positions
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        delta = (np.random.rand() - 0.5) * 0.002 * a.max()  # tiny signed amount
        if delta == 0.0:
            continue

        a_new = a.copy()
        a_new[i] = max(0.0, a_new[i] - delta)
        a_new[j] = max(0.0, a_new[j] + delta)

        # renormalise to keep the total mass fixed
        s = a_new.sum()
        if s == 0.0:
            continue
        a_new /= s

        new_score = 2.0 * n * _conv_max(a_new)
        if new_score + 1e-12 < best_score:
            a = a_new
            best_score = new_score
            best_seq = a
            # occasional logging
            if np.random.rand() < 0.05:
                print(f"[Refine] Improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Final exact evaluation with the provided evaluate_sequence()
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    return final_seq
