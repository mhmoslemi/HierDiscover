def proj_simplex(v):
    # returns projection onto simplex {x>=0, sum x = 1}
    # using algorithm from Wang & Carreira-Perpinan 2013
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]  # last true index
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

#!/usr/bin/env python
"""
Projected sub‑gradient optimiser for the step‑function bound.

The routine works on a probability vector (sum = 1, a_i ≥ 0).  The
objective 2 n·max(a*a) is equivalent to the given evaluation
function, because the scaling of the sequence does not change the
score.  The optimiser repeatedly

    * evaluates the self‑convolution by FFT,
    * extracts the index of the current maximal entry,
    * builds the sub‑gradient of max(a*a) w.r.t. a (which is simply a
      reversed copy of a shifted so that a_i contributes to the entry
      where the maximum occurs),
    * takes a step in the negative sub‑gradient direction,
    * projects back onto the simplex using the standard O(nlogn) projection,
    * backs off the step size with a tiny line‑search.

A few restarts (different parametric families, random Dirichlet draws)
provide a good initial point.  The support size is enlarged
gradually (≈ 5 k → 20 k) by linear interpolation, which gives the optimiser
extra degrees of freedom without blowing up the computational cost.
The best distribution seen is finally evaluated with the supplied
`evaluate_sequence` routine and returned.

The whole search respects the time budget (≈ 1000 s) and never returns
an empty list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures or lambdas
# ----------------------------------------------------------------------


def _fft_len_for_corr(n: int) -> int:
    """Smallest power‑of‑two ≥ 3·n‑2 (enough for self‑convolution)."""
    return 1 << ((3 * n - 2).bit_length())


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    s = new_a.sum()
    if s == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= s
    return new_a


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{‑p}, normalised."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _sym_product_power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ ((i+1)*(n‑i))^{‑p}, symmetric, normalised."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) * (n - i)) ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _exp_decay_distribution(n: int, lam: float) -> np.ndarray:
    """a_i ∝ e^{‑lam·i}+e^{‑lam·(n‑1‑i)}, symmetric exponential."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _logistic_distribution(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ 1/(1+e^{α·(i‑mid)}), normalised; heavier on the left."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = 1.0 / (1.0 + np.exp(alpha * (i - mid)))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _conv_full_and_max(a: np.ndarray):
    """Return (full self‑convolution a*a, its maximum entry)."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    af = np.fft.rfft(a, L)
    conv = np.fft.irfft(af * af, L)[: 2 * n - 1]
    return conv, float(conv.max())


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative step heights that (hopefully) minimizes
    `evaluate_sequence`.  The routine respects the supplied time budget.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0     # safety margin for the final eval

    # --------------------------------------------------------------
    # 0. Choose an initial support size and an initial distribution.
    # --------------------------------------------------------------
    n = 5000                     # start modest, will be up‑sampled later
    candidates = []

    # warm‑start from the global sequence if it exists
    global_seq = globals().get("height_sequence_1", None)
    if global_seq is not None:
        try:
            init_global = np.array(global_seq, dtype=np.float64).ravel()
            if init_global.ndim == 1 and init_global.size > 0:
                candidates.append(_interpolate_distribution(init_global, n))
                print("[Init] Warm‑started from provided global sequence.")
        except Exception:
            pass

    # hand‑crafted families
    for p in np.linspace(0.5, 2.0, 25):
        candidates.append(_power_law_distribution(n, p))
        candidates.append(_sym_product_power_law_distribution(n, p))
    for loglam in np.linspace(-6, -2, 20):
        lam = 10.0 ** loglam
        candidates.append(_exp_decay_distribution(n, lam))
    for alpha in np.linspace(0.1, 5.0, 30):
        candidates.append(_logistic_distribution(n, alpha))

    # a few random Dirichlet draws (uniform Dirichlet)
    for _ in range(8):
        a = np.random.rand(n)
        a /= a.sum()
        candidates.append(a)

    # include the uniform distribution as a fallback
    candidates.append(np.full(n, 1.0 / n))

    # pick the best among the candidates (cheap score = 2·n·max_conv)
    best_a = None
    best_score = float("inf")
    for cand in candidates:
        if time.time() >= deadline:
            break
        _, mx = _conv_full_and_max(cand)
        score = 2.0 * n * mx
        if score < best_score:
            best_score = score
            best_a = cand.copy()
    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_score = 2.0 * n * _conv_full_and_max(best_a)[1]
    print(f"[Init] Selected starting point with cheap score {best_score:.6f}")

    # --------------------------------------------------------------
    # 1. Projected sub‑gradient optimisation (mirror descent on simplex)
    # --------------------------------------------------------------
    a = best_a.copy()                 # current iterate
    best_seq = a.copy()               # best ever seen
    best_global_score = best_score
    last_improve_iter = 0
    it = 0
    step0 = 0.5                       # base step size for logits

    # schedule for smoothing factor used in the original code – not needed
    # here but kept for potential future extensions.
    while time.time() < deadline and it < 3500000:
        it += 1

        # ----- evaluate current point -----
        conv, max_c = _conv_full_and_max(a)
        k_max = int(conv.argmax())
        cheap_score = 2.0 * n * max_c

        # ----- keep best ever -----
        if cheap_score < best_global_score - 1e-12:
            best_global_score = cheap_score
            best_seq = a.copy()
            last_improve_iter = it
            if it % 10000 == 0:
                print(f"[Iter {it:07d}] NEW BEST cheap = {cheap_score:.6f}")

        # ----- sub‑gradient of max_conv at the active index -----
        # grad_i = 2 * a_{k_max - i}   (zero when out of range)
        grad = np.zeros_like(a)
        low = max(0, k_max - (n - 1))
        high = min(n - 1, k_max)
        idx = np.arange(low, high + 1, dtype=np.int64)
        grad[idx] = 2.0 * a[k_max - idx]

        # ----- step‑size (inverse‑square‑root decay) -----
        alpha = step0 / (math.sqrt(it) + 1.0)

        # ----- back‑tracking line search (up to 6 halvings) -----
        accepted = False
        trial_alpha = alpha
        for _ in range(6):
            a_trial = _proj_simplex(a - trial_alpha * grad)
            _, max_c_trial = _conv_full_and_max(a_trial)
            cheap_trial = 2.0 * n * max_c_trial
            if cheap_trial <= cheap_score:
                a = a_trial
                accepted = True
                break
            trial_alpha *= 0.5

        if not accepted:
            # no improvement – still take a modest step
            a = _proj_simplex(a - alpha * grad)

        # ----- occasional random jitter to shake plateaux -----
        if (it % 2500 == 0) and (np.random.rand() < 0.12):
            jitter = 0.02 * np.random.randn(n)
            a = _proj_simplex(a + jitter)

        # ----- gradual up‑sampling (once per ~150k iters) -----
        if (it % 150000 == 0) and (n < 20000) and (time.time() + 30.0 < deadline):
            new_n = min(20000, int(n * 1.5))
            print(f"[Iter {it}] Upsampling support: {n} → {new_n}")
            a = _interpolate_distribution(a, new_n)
            best_seq = _interpolate_distribution(best_seq, new_n)
            n = new_n
            # recompute best score for the new size
            _, mx = _conv_full_and_max(best_seq)
            best_global_score = 2.0 * n * mx
            # reset stagnation counter after a drastic change
            last_improve_iter = it
            continue   # restart loop with new dimension

        # ----- restart if stuck for a very long time -----
        if it - last_improve_iter > 250000:
            print(f"[Iter {it}] Stagnation – performing a random restart.")
            a = np.random.rand(n)
            a /= a.sum()
            last_improve_iter = it

    # --------------------------------------------------------------
    # 2. Final exact evaluation with the supplied `evaluate_sequence`
    # --------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # --------------------------------------------------------------
    # 3. Return the best sequence discovered.
    # --------------------------------------------------------------
    return final_seq
