"""
Improved step‑function optimiser.

This version uses a two‑phase Adam optimisation on the probability simplex.
Phase 1 employs a smooth surrogate of the maximal self‑convolution (a
soft‑max/temperature schedule) which provides a well‑behaved gradient and
helps escape plateaus.  Phase 2 refines the result with the exact hard‑max
sub‑gradient optimiser used previously.  Finally a brief simulated‑annealing
local search polishes the solution.  All convolutions/correlations are
computed with FFTs (O(n log n) per iteration).  The routine respects the
overall time budget and returns a list of non‑negative floats suitable for
``evaluate_sequence``.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Basic FFT utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full self‑convolution a*a (no reversal) using a real FFT.
    Returns the length‑(2·n‑1) array.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return conv_full[: 2 * n - 1]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (maps ℝⁿ → simplex)."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


def _corr_linear(w: np.ndarray, a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear correlation of w (length 2n‑1) with a (length n).
    Returns a length‑n vector g_i = Σ_j w[i+j]·a_j.
    Implemented by FFT convolution with the reversed a.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(rev_a, n=L)
    conv_full = np.fft.irfft(wf * af, n=L)
    start = n - 1
    return conv_full[start:start + n]


# ----------------------------------------------------------------------
# Phase 1 – soft‑max (temperature‑annealed) Adam optimiser
# ----------------------------------------------------------------------


def _run_opt_softmax_adam(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
) -> Tuple[List[float], float]:
    """
    Adam optimiser that minimises a smooth approximation of the
    maximal self‑convolution using a soft‑max (temperature) surrogate.
    Returns the best sequence (as a list) and its true objective
    2·n·max(conv(a,a)).
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # initialise a on the simplex
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
    else:
        a = rng.random(n)

    a = np.clip(a, 0.0, None)
    s = a.sum()
    a = a / s if s > 1e-12 else np.ones(n) / n

    # logits for exponentiated‑gradient mirror updates
    x = np.log(a + 1e-16)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # learning‑rate schedule (exponential decay)
    lr_start, lr_end = 0.6, 0.001

    # temperature (inverse‑temperature α) schedule
    alpha_start, alpha_end = 0.01, 3000.0

    # FFT length for the correlation step (w length = 2n‑1, a length = n)
    L_corr = _next_pow2(3 * n - 2)

    best_a = a.copy()
    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())

    it = 0
    while time.time() - start < time_limit:
        it += 1

        # Current point on simplex
        a = _softmax(x)

        # Self‑convolution and true objective
        conv = _conv_self_fft(a)
        M = conv.max()
        obj = float(2.0 * n * M)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[softmax n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        # Progress factor in [0,1]
        elapsed = time.time() - start
        t = min(elapsed / time_limit, 1.0)

        # Inverse‑temperature schedule (exponential)
        alpha = alpha_start * (alpha_end / alpha_start) ** t

        # Soft‑max weights over convolution entries
        scaled = alpha * conv
        max_scaled = np.max(scaled)
        w = np.exp(scaled - max_scaled)
        w /= w.sum()          # w is a probability vector over the 2n‑1 sums

        # Gradient = 2·α·corr(w, a)  (exact derivative of τ·log‑sum‑exp)
        corr = _corr_linear(w, a, L_corr)          # length n
        g = 2.0 * alpha * corr

        # Mirror‑gradient in log‑it space
        dot = np.dot(a, g)
        g_x = a * (g - dot)

        # Learning‑rate (exponential decay)
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # Adam step in log‑it space
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # Occasional jitter to leave flat regions
        if it % 6000 == 0:
            x += rng.normal(scale=0.015, size=n)

        # Full restart from the best point once in a while
        if it % 100000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.004, size=n)
            m.fill(0.0)
            v.fill(0.0)

    # ------------------------------------------------------------------
    # Prune negligible entries (they only increase n without help)
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = float(2.0 * best_a.size * _conv_self_fft(best_a).max())
        print(f"[softmax compress] reduced length → {best_a.size}, O={best_obj:.6f}")

    return best_a.tolist(), best_obj


# ----------------------------------------------------------------------
# Phase 2 – hard‑max Adam optimiser (identical to the reference version)
# ----------------------------------------------------------------------


def _run_opt_hardmax(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
) -> Tuple[List[float], float]:
    """
    Adam optimiser that always uses the exact max‑convolution sub‑gradient.
    Returns the best sequence (as a list) and its objective value
    2·n·max(conv(a,a)).
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
    else:
        a = rng.random(n)

    a = np.clip(a, 0.0, None)
    s = a.sum()
    a = a / s if s > 1e-12 else np.ones(n) / n

    x = np.log(a + 1e-16)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    lr_start, lr_end = 0.5, 0.001

    best_a = a.copy()
    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())

    it = 0
    while time.time() - start < time_limit:
        it += 1
        a = _softmax(x)

        conv = _conv_self_fft(a)
        M = conv.max()
        obj = float(2.0 * n * M)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[hardmax n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        k_max = int(np.argmax(conv))
        # exact sub‑gradient of the active convolution entry
        idx = np.arange(n)
        j = k_max - idx
        mask = (j >= 0) & (j < n)
        g_a = np.zeros_like(a)
        if mask.any():
            g_a[mask] = 2.0 * a[j[mask]]

        # mirror‑gradient in log‑it space
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)

        elapsed = time.time() - start
        t = min(elapsed / time_limit, 1.0)
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t)

        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if it % 5000 == 0:
            x += rng.normal(scale=0.01, size=n)

        if it % 80000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.005, size=n)
            m.fill(0.0)
            v.fill(0.0)

    # prune negligible entries
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = float(2.0 * best_a.size * _conv_self_fft(best_a).max())
        print(f"[hardmax compress] reduced length → {best_a.size}, O={best_obj:.6f}")

    return best_a.tolist(), best_obj


# ----------------------------------------------------------------------
# Simulated‑annealing polishing (unchanged from reference)
# ----------------------------------------------------------------------


def _run_sa(
    init_seq: List[float],
    time_limit: float,
    seed: int = 0,
) -> Tuple[List[float], float]:
    """Refine a sequence by small mass‑transfer moves with a Metropolis schedule."""
    start = time.time()
    rng = np.random.default_rng(seed)
    a = np.asarray(init_seq, dtype=float)
    if a.sum() < 1e-12:
        a = np.ones_like(a) / a.size
    else:
        a = a / a.sum()
    n = a.shape[0]

    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())
    best_seq = a.copy()

    T = best_obj * 0.1 + 1e-6
    T_min = best_obj * 1e-6 + 1e-9
    decay = 0.995
    it = 0

    while time.time() - start < time_limit:
        it += 1
        k_max = int(np.argmax(conv))

        i_vals = np.arange(n)
        j_vals = k_max - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i_vals[mask]] * a[j_vals[mask]]
        total_contrib = contrib.sum()

        if total_contrib > 0:
            probs = contrib / total_contrib
            i = rng.choice(n, p=probs)
        else:
            i = rng.integers(n)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.clip(a_new, 0.0, None)
        a_new /= a_new.sum()

        conv_new = _conv_self_fft(a_new)
        obj_new = float(2.0 * n * conv_new.max())

        if obj_new < best_obj - 1e-12:
            a, conv, best_obj, best_seq = a_new, conv_new, obj_new, a_new.copy()
            print(f"[SA it={it}] new best O={best_obj:.6f}")
        else:
            prob = np.exp(-(obj_new - best_obj) / T) if T > 0 else 0.0
            if rng.random() < prob:
                a, conv = a_new, conv_new

        if it % 2000 == 0:
            T = max(T * decay, T_min)

        if it % 10000 == 0:
            noise = rng.normal(scale=0.01, size=n)
            a = np.clip(a + noise, 0.0, None)
            a /= a.sum()
            conv = _conv_self_fft(a)

    thresh = 1e-12
    mask = best_seq > thresh
    if mask.sum() < best_seq.size:
        comp = best_seq[mask]
        comp /= comp.sum()
        comp_obj = float(2.0 * comp.size * _conv_self_fft(comp).max())
        if comp_obj < best_obj:
            best_seq, best_obj = comp, comp_obj
            print(f"[SA compress] reduced length → {comp.size}, O={best_obj:.6f}")

    return best_seq.tolist(), best_obj

def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **_ignore,
) -> List[float]:
    """
    Find a non‑negative step‑function minimizing the evaluation metric.
    The routine spends most of the budget on a smooth‑soft‑max Adam phase,
    then refines with a hard‑max Adam phase and finishes with a short
    simulated‑annealing polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Initialise from a previous construction (if present)
    # ------------------------------------------------------------------
    init_seq = None
    if "height_sequence_1" in globals():
        seq0 = np.asarray(height_sequence_1, dtype=float)
        seq0 = np.clip(seq0, 0.0, None)
        if seq0.sum() < 0.01:
            seq0 = np.ones_like(seq0)
        seq0 = seq0 / seq0.sum()
        init_seq = seq0

    # ------------------------------------------------------------------
    # Choose a target length based on the remaining budget.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    # If we still have a decent budget, go for a longer support.
    target_n = 15000 if remaining > 40.0 else 10000

    # Upsample any initial sequence to the target length.
    if init_seq is not None and init_seq.size != target_n:
        x_old = np.linspace(0.0, 1.0, init_seq.size, endpoint=False)
        x_new = np.linspace(0.0, 1.0, target_n, endpoint=False)
        init_a = np.interp(x_new, x_old, init_seq)
        init_a = np.clip(init_a, 0.0, None)
        init_a = init_a / init_a.sum()
    else:
        init_a = init_seq

    # ------------------------------------------------------------------
    # Phase 1 – smooth soft‑max Adam optimisation
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    phase1_budget = remaining * 0.65
    seq_soft, obj_soft = _run_opt_softmax_adam(
        n=target_n,
        time_limit=phase1_budget,
        init_a=init_a,
        seed=seed + 1001,
    )
    best_seq = np.asarray(seq_soft, dtype=float)
    best_obj = obj_soft
    print(f"[global] after soft‑max phase O={best_obj:.6f} (n={target_n})")

    # ------------------------------------------------------------------
    # Phase 2 – hard‑max Adam refinement (if time permits)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        phase2_budget = remaining * 0.70
        seq_hard, obj_hard = _run_opt_hardmax(
            n=best_seq.size,
            time_limit=phase2_budget,
            init_a=best_seq,
            seed=seed + 2022,
        )
        if obj_hard < best_obj:
            best_seq = np.asarray(seq_hard, dtype=float)
            best_obj = obj_hard
            print(f"[global] after hard‑max refinement O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated‑annealing polishing (if any time left)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        seq_ref, obj_ref = _run_sa(best_seq.tolist(), time_limit=remaining, seed=seed + 3033)
        if obj_ref < best_obj:
            best_seq = np.asarray(seq_ref, dtype=float)
            best_obj = obj_ref
            print(f"[final] after SA new best O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check – evaluate with the supplied scorer
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_seq.tolist()
