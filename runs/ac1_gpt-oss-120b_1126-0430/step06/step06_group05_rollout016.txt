"""
Non‑negative step‑function optimiser (soft‑max annealed Adam + hard‑max Adam).

The optimiser consists of three phases:

1.  **Soft‑max annealed Adam** – optimisation of a smooth approximation of
    the maximal self‑convolution.  The temperature is exponentially
    decreased, allowing the algorithm to first move in a “soft” landscape
    and then to focus on the true hard maximum.

2.  **Hard‑max Adam** – a short fine‑tuning phase that uses the exact
    sub‑gradient of the current maximal convolution entry (the same idea
    as the reference implementation).

3.  **Targeted simulated‑annealing** – the final polish that moves a few
    percent of the mass between indices with a Metropolis schedule.

All phases respect the simplex constraint (non‑negative entries,
normalised sum = 1) via the log‑it (exponentiated) parametrisation.
The routine splits the provided time budget (default 1000 s) among the
phases, tries a few random restarts and returns the best sequence found.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return conv_full[: 2 * n - 1]


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of a and b (both real) using FFT."""
    n, m = a.shape[0], b.shape[0]
    L = _next_pow2(n + m - 1)
    af = np.fft.rfft(a, n=L)
    bf = np.fft.rfft(b, n=L)
    conv_full = np.fft.irfft(af * bf, n=L)
    return conv_full[: n + m - 1]


# ----------------------------------------------------------------------
# Soft‑max annealed Adam optimiser
# ----------------------------------------------------------------------


def _run_opt_softmax_adam(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
) -> Tuple[List[float], float]:
    """
    Adam optimisation of a smooth (soft‑max) approximation of the maximal
    self‑convolution.  Temperature is exponentially annealed from an
    initial value proportional to the spread of the convolution down to a
    very small value (≈hard max).  Returns the best sequence seen and its
    objective value 2·n·max(conv).
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # initialise a on the simplex
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
    else:
        a = rng.random(n)

    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n, dtype=float) / n
    else:
        a = a / s

    # logits for exponentiated‑gradient updates (mirror descent)
    x = np.log(a + 1e-16)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # learning‑rate schedule (exponential decay)
    lr_start, lr_end = 0.8, 0.002

    best_a = a.copy()
    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())

    it = 0
    while time.time() - start < time_limit:
        it += 1

        # current point on the simplex
        a = np.exp(x)
        a /= a.sum()

        conv = _conv_self_fft(a)
        M = conv.max()
        obj = float(2.0 * n * M)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[soft‑max n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        # --- temperature (annealing) ------------------------------------
        # use a fraction of the current spread of the convolution as scale
        spread = conv.max() - conv.min()
        spread = max(spread, 1e-12)
        T0 = spread * 0.4          # start‑temperature (relative)
        T1 = spread * 0.001        # final temperature
        t_frac = (time.time() - start) / time_limit
        T = T0 * np.exp(np.log(T1 / T0) * t_frac) if T0 > 0 else 0.0

        # soft‑max weights over the convolution entries
        if T > 0.0:
            w = np.exp((conv - conv.max()) / T)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w /= w_sum
        else:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0

        # gradient of the smoothed objective (2·a * w‑convolution)
        # compute g = 2 * (a * w_rev) evaluated at appropriate indices
        w_rev = w[::-1]
        conv_aw = _conv_fft(a, w_rev)                     # length 3n‑2
        idx = 2 * n - 2 - np.arange(n)                    # map to g[i]
        g_a = 2.0 * conv_aw[idx]

        # natural gradient in log‑it space (mirror‑descent formulation)
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)

        # time‑based learning‑rate (exponential decay)
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t_frac)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # Adam step in the log‑it space
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape flat plateaus
        if it % 4000 == 0:
            jitter = rng.normal(scale=0.015, size=n)
            x += jitter

        # occasional full restart from the current best (helps escape)
        if it % 70000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.006, size=n)
            m.fill(0.0)
            v.fill(0.0)

    # ------------------------------------------------------------------
    # prune negligible entries (they increase n without helping)
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = float(2.0 * best_a.size * _conv_self_fft(best_a).max())
        print(f"[soft‑max compress] reduced length → {best_a.size}, O={best_obj:.6f}")

    return best_a.tolist(), best_obj

def _run_opt_hardmax(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
) -> Tuple[List[float], float]:
    """
    Adam optimiser that always uses the exact max‑convolution gradient.
    Returns the best sequence (as a list) and its objective value
    2·n·max(conv(a,a)).
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # initialise a on the simplex
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
    else:
        a = rng.random(n)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n, dtype=float) / n
    else:
        a = a / s

    x = np.log(a + 1e-16)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    lr_start = 0.5
    lr_end = 0.001

    best_a = a.copy()
    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())

    it = 0
    while time.time() - start < time_limit:
        it += 1
        a = np.exp(x)
        a /= a.sum()

        conv = _conv_self_fft(a)
        M = conv.max()
        obj = float(2.0 * n * M)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[hard‑max n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        k_max = int(np.argmax(conv))
        # exact sub‑gradient of that entry
        nlen = a.shape[0]
        idx = np.arange(nlen)
        j = k_max - idx
        mask = (j >= 0) & (j < nlen)
        g_a = np.zeros_like(a)
        if mask.any():
            g_a[mask] = 2.0 * a[j[mask]]

        # natural gradient in log‑it space
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)

        # time‑based learning‑rate (exponential)
        elapsed = time.time() - start
        t = min(elapsed / time_limit, 1.0)
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if it % 5000 == 0:
            jitter = rng.normal(scale=0.012, size=n)
            x += jitter

        if it % 80000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.005, size=n)
            m.fill(0.0)
            v.fill(0.0)

    # compress negligible entries
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = float(2.0 * best_a.size * _conv_self_fft(best_a).max())
        print(f"[hard‑max compress] reduced length → {best_a.size}, O={best_obj:.6f}")

    return best_a.tolist(), best_obj

    time_limit: float,

    """Refine a sequence by small mass‑transfer moves with a Metropolis schedule."""
    start = time.time()
    rng = np.random.default_rng(seed)
    a = np.asarray(init_seq, dtype=float)
    if a.sum() < 1e-12:
        a = np.ones_like(a) / a.size
    else:
        a = a / a.sum()
    n = a.shape[0]

    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())
    best_seq = a.copy()

    T = best_obj * 0.1 + 1e-8
    T_min = best_obj * 1e-7 + 1e-9
    decay = 0.996
    it = 0

    while time.time() - start < time_limit:
        it += 1
        k_max = int(np.argmax(conv))

        # contribution of each index to the maximal entry
        i_vals = np.arange(n)
        j_vals = k_max - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i_vals[mask]] * a[j_vals[mask]]
        total_contrib = contrib.sum()

        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = rng.choice(n, p=probs)
        else:
            i = rng.integers(n)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.07   # up to 7 % transfer
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.clip(a_new, 0.0, None)
        a_new /= a_new.sum()

        conv_new = _conv_self_fft(a_new)
        obj_new = float(2.0 * n * conv_new.max())

        if obj_new < best_obj - 1e-13:
            a, conv, best_obj, best_seq = a_new, conv_new, obj_new, a_new.copy()
            print(f"[SA it={it}] new best O={best_obj:.6f}")
        else:
            prob = np.exp(-(obj_new - best_obj) / T) if T > 0 else 0.0
            if rng.random() < prob:
                a, conv = a_new, conv_new

        if it % 2500 == 0:
            T = max(T * decay, T_min)

        if it % 12000 == 0:
            noise = rng.normal(scale=0.008, size=n)
            a = np.clip(a + noise, 0.0, None)
            a /= a.sum()
            conv = _conv_self_fft(a)

    # final compression of tiny entries
    thresh = 1e-12
    mask = best_seq > thresh
    if mask.sum() < best_seq.size:
        comp = best_seq[mask]
        comp /= comp.sum()
        comp_obj = float(2.0 * comp.size * _conv_self_fft(comp).max())
        if comp_obj < best_obj:
            best_seq, best_obj = comp, comp_obj
            print(f"[SA compress] reduced length → {comp.size}, O={best_obj:.6f}")

    return best_seq.tolist(), best_obj

def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **_ignore,
) -> List[float]:
    """
    Search for a non‑negative step‑function minimising the evaluation metric.
    The budget is split into three phases (soft‑max Adam → hard‑max Adam →
    simulated annealing).  Several random restarts are performed; the best
    sequence found is returned.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # leave a small safety margin

    # --------------------------------------------------------------
    # Initialise from a previously discovered construction if present.
    init_seq = None
    if 'height_sequence_1' in globals():
        seq0 = np.asarray(height_sequence_1, dtype=float)
        seq0 = np.clip(seq0, 0.0, None)
        if seq0.sum() < 0.01:
            seq0 = np.ones_like(seq0)
        seq0 = seq0 / seq0.sum()
        init_seq = seq0

    # --------------------------------------------------------------
    # Choose a target length.
    remaining = deadline - time.time()
    target_n = 12000 if remaining > 40.0 else 9000

    # Upsample any initial sequence to the target length.
    if init_seq is not None and init_seq.size != target_n:
        x_old = np.linspace(0.0, 1.0, init_seq.size, endpoint=False)
        x_new = np.linspace(0.0, 1.0, target_n, endpoint=False)
        init_a = np.interp(x_new, x_old, init_seq)
        init_a = np.clip(init_a, 0.0, None)
        init_a = init_a / init_a.sum()
    else:
        init_a = init_seq

    # --------------------------------------------------------------
    # Budget split
    now = time.time()
    remaining = deadline - now
    soft_budget = remaining * 0.55
    hard_budget = remaining * 0.30
    sa_budget   = remaining - soft_budget - hard_budget

    # --------------------------------------------------------------
    # Phase 1 – soft‑max annealed Adam (several short restarts)
    best_overall_seq = None
    best_overall_obj = float('inf')
    n_restarts = 3
    for r in range(n_restarts):
        # random perturbation of the initial point for each restart
        pert = rng.normal(scale=0.02, size=target_n)
        start_a = np.clip(init_a + pert, 0.0, None)
        start_a = start_a / start_a.sum()
        seq, obj = _run_opt_softmax_adam(
            n=target_n,
            time_limit=soft_budget / n_restarts,
            init_a=start_a,
            seed=seed + r * 101,
        )
        if obj < best_overall_obj:
            best_overall_obj, best_overall_seq = obj, np.asarray(seq, dtype=float)
            print(f"[restart {r}] soft‑max best O={best_overall_obj:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – hard‑max Adam fine‑tuning
    now = time.time()
    remaining = deadline - now
    if remaining > 5.0:
        seq, obj = _run_opt_hardmax(
            n=target_n,
            time_limit=min(hard_budget, remaining * 0.9),
            init_a=best_overall_seq,
            seed=seed + 777,
        )
        if obj < best_overall_obj:
            best_overall_obj, best_overall_seq = obj, np.asarray(seq, dtype=float)
            print(f"[hard‑max] refined best O={best_overall_obj:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – simulated‑annealing refinement
    now = time.time()
    remaining = deadline - now
    if remaining > 3.0:
        seq, obj = _run_sa(
            init_seq=best_overall_seq.tolist(),
            time_limit=remaining,
            seed=seed + 999,
        )
        if obj < best_overall_obj:
            best_overall_obj, best_overall_seq = obj, np.asarray(seq, dtype=float)
            print(f"[final SA] improved best O={best_overall_obj:.6f}")

    # --------------------------------------------------------------
    # Final sanity check – call the supplied evaluator.
    try:
        final_score = evaluate_sequence(best_overall_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_overall_seq.tolist()
