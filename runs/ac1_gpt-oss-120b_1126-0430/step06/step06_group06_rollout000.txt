#!/usr/bin/env python
"""
Propose a non‑negative step function that minimises

        2·n·max_k (a*a)[k] / (Σ a_i)²

Algorithm
---------
1. Scan a few parametric families (uniform, power‑law, exponential,
   stretched‑exponential, β‑shaped, Dirichlet) for several values of n.
   Keep the best cheap score (2·n·max(conv)) for each n.

2. Refine the best candidate of each n with a smooth approximation of the
   max‑operator:
        smooth_max_β(x) = (1/β)·log Σ exp(β·x_i)
   The optimisation variables are the *soft‑max* parameters y (a_i = softmax(y)).
   The objective is   f(y) = 2·n·smooth_max_β(conv(a(y))).
   Its gradient w.r.t. y is obtained analytically via FFT‑based convolutions.
   We run a limited number of L‑BFGS‑B restarts (different initial noises).

3. After L‑BFGS, perform a cheap (1+1)‑ES hill‑climb on the exact cheap
   objective (2·n·max(conv)).  This removes any remaining tiny irregularities.

4. Return the best vector (already normalised to sum = 1, thus feasible for
   ``evaluate_sequence``).  The whole routine respects the 1000 s budget.

The implementation avoids closures and lambda‑expressions, uses only top‑level
functions, and leans on NumPy / SciPy for fast FFT and optimisation.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using real FFT, length 2·n‑1."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    # cut to the exact linear‑convolution length
    return conv[: 2 * a.size - 1]


def _next_pow2(x: int) -> int:
    """Small helper returning the next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def cheap_score(vec: np.ndarray, n: int, fft_len: int) -> float:
    """Cheap objective assuming Σ a_i = 1:
       2·n·max(conv)."""
    conv = _conv_full_fft(vec, fft_len)
    return 2.0 * n * conv.max()


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Parametric families (return probability vectors)
# ----------------------------------------------------------------------


def _uniform_seq(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _power_law_seq(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = i ** (-p)
    seq /= seq.sum()
    return seq


def _exponential_seq(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    seq = np.exp(-lam * i)
    seq /= seq.sum()
    return seq


def _power_law_exp_seq(n: int, p: float, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    seq = (i + 1.0) ** (-p) * np.exp(-lam * i)
    seq /= seq.sum()
    return seq


def _stretched_exponential_seq(n: int, lam: float, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    seq = np.exp(-lam * (i ** gamma))
    seq /= seq.sum()
    return seq


def _beta_seq(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(1, n + 1, dtype=np.float64) / n)
    seq = (x ** alpha) * ((1.0 - x) ** beta)
    s = seq.sum()
    if s == 0.0:
        return None
    seq /= s
    return seq


def _dirichlet_seq(n: int, conc: float) -> np.ndarray:
    g = np.random.gamma(conc, 1.0, size=n)
    seq = g / g.sum()
    return seq


# ----------------------------------------------------------------------
# Smooth objective + gradient (used by L‑BFGS‑B)
# ----------------------------------------------------------------------


def _objective_and_grad(y: np.ndarray,
                        n: int,
                        fft_len_conv: int,
                        fft_len_grad: int,
                        beta: float) -> tuple[float, np.ndarray]:
    """
    Returns (f, grad) where
        f   = 2·n·smooth_max_β(conv(a(y)))
        grad = ∂f/∂y   (same shape as y)

    a = softmax(y) is the probability vector (Σ a = 1, a ≥ 0).
    """
    # ----- a = softmax(y) -----
    a = softmax(y)

    # ----- convolution a*a (full, length 2n‑1) -----
    Fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(Fa * Fa, fft_len_conv)[: (2 * n - 1)]

    # ----- smooth max -----
    scaled = beta * conv_full
    lse = logsumexp(scaled)          # log Σ exp(β·conv_i)
    smooth = lse / beta              # smooth max
    f = 2.0 * n * smooth

    # ----- weights w_i = exp(β·conv_i) / Σ exp(β·conv) -----
    w = np.exp(scaled - lse)         # length 2n‑1, already normalised

    # ----- gradient wrt a -----
    # g_a_i = 2 * Σ_k w_k a_{k-i}   (indices out of range give 0)
    # This is a convolution of a with the reversed weight vector.
    w_rev = w[::-1]                  # length 2n‑1
    Fa_grad = np.fft.rfft(a, fft_len_grad)
    Fw_rev = np.fft.rfft(w_rev, fft_len_grad)
    conv_aw = np.fft.irfft(Fa_grad * Fw_rev, fft_len_grad)
    conv_lin = conv_aw[: (3 * n - 2)]   # linear convolution length 3n‑2
    # index mapping: g_a[i] = 2 * conv_lin[ (2n‑2) – i ]
    idx = (2 * n - 2) - np.arange(n, dtype=int)
    g_a = 2.0 * conv_lin[idx]

    # ----- chain rule: ∂f/∂y = (∂f/∂a)·(∂a/∂y) -----
    # ∂a_i/∂y_j = a_i (δ_{ij} – a_j)
    dot = np.dot(g_a, a)             # scalar = Σ_j g_a_j a_j
    grad_y = a * (g_a - dot)         # elementwise

    return f, grad_y

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that
    minimises ``evaluate_sequence``.  The routine adapts its schedule to the
    available time budget and always returns a feasible sequence.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # small safety margin

    # ------------------------------------------------------------------
    # Working lengths (increasing order of effort).  Upper bound ≈ 2·n/(2n‑1)
    # -> large n gives theoretical lower bound close to 1.
    # ------------------------------------------------------------------
    lengths = [20000, 15000, 10000, 5000]   # all ≤ 25000 (limit of original code)

    global_best_seq = None
    global_best_val = float("inf")

    # ------------------------------------------------------------------
    # Loop over candidate lengths
    # ------------------------------------------------------------------
    for n in lengths:
        now = time.time()
        if now >= deadline:
            break

        print(f"[INFO] ----- Working with n = {n} -----")

        # FFT sizes needed later
        fft_len_conv = _next_pow2(2 * n - 1)      # for a*a
        fft_len_grad = _next_pow2(3 * n - 2)      # for a * w_rev

        # --------------------------------------------------------------
        # Phase 0 – cheap scan of parametric families
        # --------------------------------------------------------------
        candidates = []

        # uniform
        candidates.append(_uniform_seq(n))

        # power‑law
        for p in [0.3, 0.5, 0.7, 1.0, 1.3, 1.6]:
            candidates.append(_power_law_seq(n, p))

        # exponential
        for lam in [0.0, 0.001, 0.01, 0.05, 0.1, 0.2, 0.5]:
            candidates.append(_exponential_seq(n, lam))

        # mixed power‑law × exponential (a few random draws)
        for _ in range(6):
            p = np.random.uniform(0.2, 2.0)
            lam = np.random.exponential(scale=0.5)
            candidates.append(_power_law_exp_seq(n, p, lam))

        # stretched exponential
        for gamma in [0.5, 0.9, 1.0, 1.2, 1.5, 2.0]:
            lam = np.random.uniform(0.0, 3.0)
            candidates.append(_stretched_exponential_seq(n, lam, gamma))

        # beta‑shaped families
        xs = (np.arange(1, n + 1, dtype=np.float64) / n)
        for alpha in np.linspace(0.2, 5.0, 9):
            for beta in np.linspace(0.2, 5.0, 9):
                seq = (xs ** alpha) * ((1.0 - xs) ** beta)
                s = seq.sum()
                if s > 0:
                    candidates.append(seq / s)

        # Dirichlet seeds (wide concentration range)
        for conc in [0.001, 0.005, 0.01, 0.05, 0.1, 0.5, 1.0, 2.0, 5.0]:
            candidates.append(_dirichlet_seq(n, conc))

        # Previous construction, if it matches the current length
        try:
            prev = globals()["height_sequence_1"]
            if isinstance(prev, list) and len(prev) == n:
                prev_arr = np.asarray(prev, dtype=np.float64)
                if prev_arr.sum() > 0:
                    prev_arr = prev_arr / prev_arr.sum()
                    candidates.append(prev_arr)
        except Exception:
            pass

        # Find the best cheap candidate for this n
        best_local_vec = None
        best_local_val = float("inf")
        for cand in candidates:
            if time.time() >= deadline - 2.0:
                break
            if cand is None:
                continue
            val = cheap_score(cand, n, fft_len_conv)
            if val < best_local_val:
                best_local_val = val
                best_local_vec = cand.copy()
        print(f"[n={n}] best cheap after parametric scan = {best_local_val:.6f}")

        # --------------------------------------------------------------
        # Phase 1 – smooth optimisation (L‑BFGS‑B) on a soft‑max parametrisation
        # --------------------------------------------------------------
        beta_smooth = 12.0                # smoothing parameter (larger → closer to true max)
        restarts = 5                      # number of different initialisations
        best_vec = best_local_vec.copy()
        best_val = best_local_val

        for r in range(restarts):
            if time.time() >= deadline - 5.0:
                break

            if r == 0:
                a_start = best_local_vec
            else:
                # Slight random perturbation of the current best
                a_start = best_local_vec * np.exp(0.5 * np.random.randn(n))
                a_start = _simplex_projection(a_start)

            # safe start (avoid exact zeros)
            eps = 1e-16
            a_start = np.maximum(a_start, eps)

            # initialise soft‑max variables y = log(a)
            y0 = np.log(a_start)

            args = (n, fft_len_conv, fft_len_grad, beta_smooth)

            res = minimize(_objective_and_grad,
                           y0,
                           args=args,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 120, 'ftol': 1e-12, 'disp': False})

            y_opt = res.x
            a_opt = softmax(y_opt)
            val = cheap_score(a_opt, n, fft_len_conv)

            if val < best_val - 1e-12:
                best_val = val
                best_vec = a_opt.copy()
                best_local_vec = a_opt.copy()     # keep the improved version for later restarts
                print(f"[n={n}] L‑BFGS restart {r} -> new cheap = {val:.6f}")

        # --------------------------------------------------------------
        # Phase 2 – (1+1)‑ES hill‑climb on the exact cheap objective
        # --------------------------------------------------------------
        sigma = 0.12
        fail_streak = 0
        a_curr = best_vec.copy()
        curr_val = best_val

        while time.time() < deadline - 2.0:
            cand = a_curr * np.exp(sigma * np.random.randn(n))
            cand = _simplex_projection(cand)
            val = cheap_score(cand, n, fft_len_conv)

            if val < curr_val:
                curr_val = val
                a_curr = cand
                sigma = max(sigma * 0.85, 1e-5)
                fail_streak = 0
                print(f"[n={n}] ES improved cheap = {val:.6f} (σ={sigma:.5f})")
            else:
                fail_streak += 1
                if fail_streak > 250:
                    sigma = min(sigma * 1.30, 0.6)
                    fail_streak = 0

        # --------------------------------------------------------------
        # Record global best
        # --------------------------------------------------------------
        if curr_val < global_best_val:
            global_best_val = curr_val
            global_best_seq = a_curr.tolist()
            print(f"[GLOBAL] New overall cheap = {global_best_val:.6f} from n={n}")

    # ------------------------------------------------------------------
    # Final exact evaluation (for logging only; the returned list is already feasible)
    # ------------------------------------------------------------------
    if global_best_seq is None:
        # Fallback – a very simple uniform vector of length 5000
        n_fallback = 5000
        global_best_seq = _uniform_seq(n_fallback).tolist()
        global_best_val = cheap_score(np.asarray(global_best_seq), n_fallback,
                                      _next_pow2(2 * n_fallback - 1))
        print("[FALLBACK] Using uniform vector (unlikely to happen).")

    try:
        final_score = evaluate_sequence(global_best_seq)
        print(f"[RESULT] Exact evaluate_sequence = {final_score:.6f}")
    except Exception as exc:      # defensive – should never happen
        print(f"[ERROR] evaluate_sequence raised: {exc}")

    return global_best_seq
