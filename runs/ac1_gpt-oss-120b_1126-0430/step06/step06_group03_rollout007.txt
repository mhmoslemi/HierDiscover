"""
Hybrid optimizer for the self‑convolution bound.

The algorithm consists of four independent stages that fit in the
1000 s time budget:

1.  Quick parametric sweep – a handful of families (power‑law,
    double‑power‑law, stretched‑exponential, logistic, …) are sampled.
    The cheap objective `2·n·max(a*a)` is used to keep the best candidate.

2.  Smoothed mirror‑descent (Adam) – starting from the best candidate,
    Adam is applied to the log‑softmax parametrisation with a
    log‑sum‑exp smoothing of the convolution maximum.
    This stage quickly reaches a good local optimum.

3.  Exact sub‑gradient descent – once the smoothing constant is tiny,
    the true max‑convolution is used.  The sub‑gradient of the
    maximum ∇ₐ max(a*a) is obtained by a single FFT (no loops).
    An exponentiated‑gradient step on the simplex refines the solution
    to a tighter local minimum.

4.  Simulated‑annealing random walk – small random mass transfers
    between two positions are accepted if they improve the cheap score,
    or with a Metropolis probability that slowly cools down.
    This stage removes the remaining tiny ripples.

The best sequence found (a probability vector, Σa = 1) is returned.
`evaluate_sequence` rescales it if necessary and gives the exact score.
All numbers are non‑negative and stay far below the 1000 limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities (no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max, returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    s = e.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / z.size)
    return e / s


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it that reproduces `a` after a soft‑max."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """Smallest power‑of‑two ≥ 3·n‑2 (enough for full self‑convolution)."""
    return 1 << ((3 * n - 2).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Returns (max_conv, grad_a) where

        grad_a = ∇_a [ 2·n·smooth_max(a*a) ]

    `eps_factor` multiplies the smoothing constant ε = eps_factor·max_conv.
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv                     # ≤ 0 for stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # Gradient of 2·n·smooth_max (the factor 4·n comes from the derivative of a*a)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a


def _conv_max_and_subgrad(a: np.ndarray, tol: float = 1e-12):
    """
    Exact version: returns (max_conv, subgrad_a) where subgrad_a is any
    sub‑gradient of the true maximum of the self‑convolution.
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    a_fft = np.fft.rfft(a, L)
    conv_full = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv_full.max())

    mask = conv_full >= max_conv - tol
    cnt = mask.sum()
    if cnt == 0:
        # fallback – should never happen
        mask = np.zeros_like(conv_full, dtype=bool)
        mask[conv_full.argmax()] = True
        cnt = 1
    w = mask.astype(np.float64) / cnt          # weight 1/|argmax|
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # derivative of each (a_i * a_{k-i}) term contributes twice a_{k-i}
    subgrad = 2.0 * corr
    return max_conv, subgrad


# ----------------------------------------------------------------------
# Parametric families (probability vectors)
# ----------------------------------------------------------------------


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    a /= max(s, 1e-12)
    return a


def _power_law_offset_distribution(n: int, p: float, offset: float) -> np.ndarray:
    idx = np.arange(offset, offset + n, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    a /= max(s, 1e-12)
    return a


def _double_power_law_distribution(n: int, p: float, q: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) ** (-p)) * ((n - i) ** (-q))
    s = a.sum()
    a /= max(s, 1e-12)
    return a


def _stretched_exp_distribution(n: int, tau_factor: float, gamma: float) -> np.ndarray:
    tau = max(1e-12, tau_factor * n)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(- (i / tau) ** gamma)
    s = a.sum()
    a /= max(s, 1e-12)
    return a


def _exp_decay_distribution(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    s = a.sum()
    a /= max(s, 1e-12)
    return a


def _logistic_distribution(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = 1.0 / (1.0 + np.exp(alpha * (i - mid)))
    s = a.sum()
    a /= max(s, 1e-12)
    return a


def _sym_product_power_law_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) * (n - i)) ** (-p)
    s = a.sum()
    a /= max(s, 1e-12)
    return a


def _triangular_distribution(n: int) -> np.ndarray:
    """Deterministic symmetric triangle (1,2,…,⌈n/2⌉,…,2,1)."""
    i = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    a = np.where(i <= half, i + 1.0, n - i)
    s = a.sum()
    a /= max(s, 1e-12)
    return a


def _sinusoidal_distribution(n: int) -> np.ndarray:
    """Half‑sine shape: a_i ∝ sin(π·(i+0.5)/n)."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n)
    s = a.sum()
    a /= max(s, 1e-12)
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) improves the
    `evaluate_sequence` score.  The routine respects a time budget
    (~1000 s) and combines a parametric sweep, Adam on a smoothed
    objective, exact sub‑gradient descent, and a final simulated‑annealing
    random walk.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # a small safety margin

    # ------------------------------------------------------------------
    # 0. Initialise a probability vector (sum = 1)
    # ------------------------------------------------------------------
    try:
        init_seq = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        n = max(20000, init_seq.shape[0])
        a0 = np.maximum(init_seq, 0.0)
        s = a0.sum()
        if s > 0.0:
            a0 /= s
        else:
            a0 = np.full(n, 1.0 / n)
        print(f"[Init] Using provided sequence, length = {n}")
    except Exception:
        n = 25000
        a0 = np.full(n, 1.0 / n)
        print(f"[Init] No provided sequence – start from uniform, length = {n}")

    best_a = a0.copy()
    best_max = _conv_max(best_a)
    best_score = 2.0 * n * best_max
    print(f"[Init] Initial cheap score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1. Coarse parametric sweep (fast cheap score)
    # ------------------------------------------------------------------
    families = [
        ("power_law", _power_law_distribution,
         [dict(p=p) for p in np.linspace(0.40, 0.80, 41)]),
        ("power_law_offset", _power_law_offset_distribution,
         [dict(p=p, offset=off)
          for p in np.linspace(0.40, 0.80, 21)
          for off in np.linspace(0.5, 5.0, 21)]),
        ("double_power_law", _double_power_law_distribution,
         [dict(p=p, q=q)
          for p in np.linspace(0.40, 0.80, 17)
          for q in np.linspace(0.40, 0.80, 17)]),
        ("stretched_exp", _stretched_exp_distribution,
         [dict(tau_factor=t, gamma=g)
          for t in np.linspace(0.10, 0.90, 17)
          for g in np.linspace(0.5, 2.0, 17)]),
        ("exp_decay", _exp_decay_distribution,
         [dict(lam=lam) for lam in np.logspace(-6, -2, 30)]),
        ("logistic", _logistic_distribution,
         [dict(alpha=alpha) for alpha in np.linspace(0.01, 6.0, 30)]),
        ("sym_product_power_law", _sym_product_power_law_distribution,
         [dict(p=p) for p in np.linspace(0.40, 1.20, 41)]),
        ("triangular", _triangular_distribution, [dict()]),
        ("sinusoidal", _sinusoidal_distribution, [dict()]),
    ]

    print("[Sweep] Starting parametric search")
    for name, generator, param_grid in families:
        for params in param_grid:
            if time.time() >= deadline:
                break
            try:
                cand = generator(n, **params)
            except Exception:
                continue
            score = 2.0 * n * _conv_max(cand)
            if score < best_score:
                best_score = score
                best_a = cand.copy()
        if time.time() >= deadline:
            break
    print(f"[Sweep] Best cheap score after sweep = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Smoothed Adam (mirror‑descent on logits)
    # ------------------------------------------------------------------
    logits = _logits_from_distribution(best_a) + 0.02 * np.random.randn(n)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    base_alpha = 0.30
    eps_factor = 0.08
    eps_factor_min = 5e-5
    eps_decay = 0.9994

    it = 0
    best_adam_a = best_a.copy()
    best_adam_score = best_score
    last_improve = 0

    while time.time() < deadline and it < 1_200_000:
        it += 1

        a = _softmax(logits)                      # probability vector
        max_conv, grad_a = _conv_max_and_grad(a, eps_factor)
        cheap = 2.0 * n * max_conv

        if cheap < best_adam_score - 1e-12:
            best_adam_score = cheap
            best_adam_a = a.copy()
            last_improve = it

        # mirror‑descent (Adam) on logits
        mu = np.dot(grad_a, a)
        grad_z = a * (grad_a - mu)

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step = -m_hat / (np.sqrt(v_hat) + eps_adam)

        # back‑tracking line‑search (up to 4 halvings)
        accepted = False
        trial_alpha = alpha_t
        for _ in range(4):
            new_logits = logits + trial_alpha * step
            new_logits = np.clip(new_logits, -50.0, 50.0)
            a_new = _softmax(new_logits)
            new_score = 2.0 * n * _conv_max(a_new)
            if new_score <= cheap:
                logits = new_logits
                accepted = True
                break
            trial_alpha *= 0.5
        if not accepted:
            logits += alpha_t * step
            logits = np.clip(logits, -50.0, 50.0)

        # occasional jitter & decay
        if it % 4000 == 0 and np.random.rand() < 0.15:
            logits += 0.02 * np.random.randn(n)
            m *= 0.6
            v *= 0.6

        eps_factor = max(eps_factor_min, eps_factor * eps_decay)

        if it - last_improve > 250_000:
            # restart from the best we have seen so far
            print(f"[Adam] Stagnation at iter {it}, restarting.")
            logits = _logits_from_distribution(best_adam_a) + 0.03 * np.random.randn(n)
            m.fill(0.0)
            v.fill(0.0)
            eps_factor = 0.08
            last_improve = it

    best_a = best_adam_a
    best_score = best_adam_score
    print(f"[Adam] Finished after {it} iterations, best cheap = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Exact sub‑gradient descent (exponentiated‑gradient on simplex)
    # ------------------------------------------------------------------
    # initialise logits from the best Adam result
    logits = _logits_from_distribution(best_a) + 0.01 * np.random.randn(n)
    it2 = 0
    best_sub_a = best_a.copy()
    best_sub_score = best_score
    last_improve = 0
    base_lr = 0.5
    lr_decay = 0.9995
    lr = base_lr

    while time.time() < deadline and it2 < 300_000:
        it2 += 1

        a = np.exp(logits)
        a /= a.sum()
        max_conv, subgrad = _conv_max_and_subgrad(a)
        cheap = 2.0 * n * max_conv

        if cheap < best_sub_score - 1e-12:
            best_sub_score = cheap
            best_sub_a = a.copy()
            last_improve = it2

        # exponentiated‑gradient step
        lr = lr * lr_decay      # slowly decay the step‑size
        step = -lr * subgrad
        # prevent overflow / underflow
        step = np.clip(step, -20.0, 20.0)

        logits = np.log(a + 1e-30) + step   # log‑space update
        # keep logits bounded for numerical stability
        logits = np.clip(logits, -60.0, 60.0)

        # occasional restart if we hit a wall
        if it2 - last_improve > 80_000:
            print(f"[SubGrad] No improvement for {it2} it., restarting.")
            logits = _logits_from_distribution(best_sub_a) + 0.02 * np.random.randn(n)
            lr = base_lr
            last_improve = it2

    best_a = best_sub_a
    best_score = best_sub_score
    print(f"[SubGrad] Finished after {it2} iterations, best cheap = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Simulated‑annealing random walk (tiny mass transfers)
    # ------------------------------------------------------------------
    a = best_a.copy()
    cur_score = best_score
    temperature = 0.2                     # initial “accept‑even‑if‑worse” temp
    decay_temp = 0.9999

    while time.time() < deadline:
        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        if i == j:
            continue

        # choose a fraction of the source mass; the fraction shrinks with temperature
        frac = min(0.2, temperature) * np.random.rand()
        delta = frac * a[i]
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = 2.0 * n * _conv_max(a_new)

        if new_score < cur_score:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
        else:
            # Metropolis acceptance
            prob = math.exp(-(new_score - cur_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                a = a_new
                cur_score = new_score

        temperature *= decay_temp   # cool down slowly

    # ------------------------------------------------------------------
    # 5. Final exact evaluation
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        exact = evaluate_sequence(final_seq)   # provided by the platform
        print(f"Final exact score = {exact:.6f}")
    except Exception as exc:
        exact = float("inf")
        print(f"Error during final evaluation: {exc}")

    return final_seq
