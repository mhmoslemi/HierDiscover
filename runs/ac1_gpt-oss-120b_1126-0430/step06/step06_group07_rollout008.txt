"""
A fast simulated‑annealing / mirror‑descent hybrid for the
step‑function “minimum max‑convolution” problem.

The algorithm proceeds in three phases:

1. **Broad parametric sweep** – builds a library of candidate families
   (beta, exponential, power‑law, Dirichlet, mixed power‑law) and keeps
   the best one according to the cheap objective
   ``2·n·max(conv(a,a))`` (the exact evaluation is independent of scaling).

2. **Hybrid local optimisation** – starting from the best candidate we run
   a long simulated‑annealing loop that repeatedly swaps entries (random
   swaps and a *targeted* swap that attacks the currently maximal convolution
   entry).  The cheap score is recomputed with an FFT after every move.
   The temperature is slowly cooled; occasional Gaussian perturbations are
   added to escape plateaus.

3. **Fine‑grained mirror descent** – after the annealing schedule stabilises
   we finish with a short exponentiated‑gradient mirror‑descent phase on the
   simplex to polish the amplitudes while keeping the ordering fixed.

The routine respects a user supplied time budget (default 1000 s) and
always returns a feasible non‑negative list.  It also prints progress
information so the runner can see intermediate bounds.
"""

import time
import math

def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # all entries non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using a power‑of‑two FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa).real[: 2 * a.size - 1]
    return conv


def _conv_max_fft(a: np.ndarray, fft_len: int) -> tuple[float, int, np.ndarray]:
    """Maximum value of the autocorrelation a*a, its index and the full array."""
    conv = _conv_full_fft(a, fft_len)
    idx = int(conv.argmax())
    return float(conv[idx]), idx, conv

        j = kmax - rng
        g = np.zeros_like(a)
        mask_eq = (rng == j)
        g[rng[~mask_eq]] = 2.0 * a[j[~mask_eq]]
        g[rng[mask_eq]] = a[rng[mask_eq]]

        eta = eta0 / math.sqrt(iter_cnt)
        log_a = np.log(a + eps) - eta * g
        a = np.exp(log_a)
        s = a.sum()
        if s <= 0.0 or not np.isfinite(s):
            a = best_iter_a.copy()
        else:
            a /= s

    # --------------------------------------------------------------
    # 3.  Final exact evaluation
    # --------------------------------------------------------------
    final_seq = best_iter_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    return final_seq
