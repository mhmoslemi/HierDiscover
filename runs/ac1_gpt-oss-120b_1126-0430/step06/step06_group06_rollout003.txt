def gradient_refine(a0, n, fft_len2, deadline):
    eps = 1e-12
    y = np.log(a0 + eps)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    b1=0.9; b2=0.999; eps_adam=1e-8
    lr=0.1
    beta_smooth=5.0
    best_a = a0.copy()
    best_val = cheap_score(best_a)
    iter_cnt = 0
    last_improve = 0

    while time.time() < deadline:
        iter_cnt += 1
        # compute a from y
        max_y = np.max(y)  # for numerical stability
        exp_y = np.exp(y - max_y)
        a = exp_y / exp_y.sum()
        # compute FFT of a padded to fft_len2
        a_fft = np.fft.rfft(a, n=fft_len2)
        # convolution a*a
        conv = np.fft.irfft(a_fft * a_fft, n=fft_len2)[:2*n-1]
        # smooth objective approx
        # compute softmax weights w
        # Use stable log-sum-exp approach
        beta_conv = beta_smooth * conv
        max_bc = np.max(beta_conv)
        w = np.exp(beta_conv - max_bc)
        w_sum = w.sum()
        w /= w_sum
        # compute convolution w * a
        w_fft = np.fft.rfft(w, n=fft_len2)
        conv_wa = np.fft.irfft(w_fft * a_fft, n=fft_len2)[:(2*n-1)+n-1]  # length 3n-2
        g = 2 * conv_wa[n-1 : n-1+n]  # take slice
        # compute gradient wrt y
        mu = np.dot(a, g)
        grad_y = a * (g - mu)
        # Adam update
        m = b1 * m + (1-b1) * grad_y
        v = b2 * v + (1-b2) * (grad_y ** 2)
        m_hat = m / (1 - b1**iter_cnt)
        v_hat = v / (1 - b2**iter_cnt)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # compute cheap score for best comparison
        cheap = 2.0 * n * conv.max()
        if cheap < best_val:
            best_val = cheap
            best_a = a.copy()
            last_improve = iter_cnt
            # optionally print
        # adjust beta and lr schedule
        if iter_cnt % 2000 == 0:
            beta_smooth = min(beta_smooth * 1.2, 100.0)
        # restarts
        if iter_cnt - last_improve > 40000:
            # random perturb around best
            y = np.log(best_a + eps) + 0.1 * np.random.randn(n)
            last_improve = iter_cnt
            print("[Grad] restart after stagnation")
    return best_a, best_val
```

#!/usr/bin/env python
"""
Propose a non‑negative step function (list of heights) that minimises

        2·n·max_k (a*a)[k] / (Σ a_i)²

The algorithm consists of three phases:
  1. Quick exploration of several parametric families (uniform,
     power‑law, exponential, beta, stretched‑exponential, Dirichlet).
  2. Smooth “soft‑max” gradient refinement using Adam on the log‑space
     (exponentiated) representation of the simplex.  A temperature
     parameter β is increased during optimisation so that the smooth
     objective approaches the true max‑norm.
  3. A cheap (1+1)‑ES hill‑climber on the exact cheap score for a final
     polish.

All phases respect the 1000 s budget and the routine always returns a
feasible sequence (non‑negative, Σa_i>0.01).  The implementation works
with any length n (default 5000‑20000) and uses only NumPy + SciPy‑like
optimisation primitives – no external solvers are required.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1, dtype=np.float64)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        # all non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution (a*a) using real FFT, length = 2·len(a)-1."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len).real[: 2 * a.size - 1]
    return conv


def _cheap_score(a: np.ndarray, n: int, fft_len: int) -> float:
    """Cheap objective (assumes Σa = 1): 2·n·max(a*a)."""
    conv = _conv_full_fft(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families – all return a normalised probability vector
# ----------------------------------------------------------------------


def _uniform_seq(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _power_law_seq(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = i ** (-p)
    seq /= seq.sum()
    return seq


def _exponential_seq(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    seq = np.exp(-lam * i)
    seq /= seq.sum()
    return seq


def _power_law_exp_seq(n: int, p: float, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    seq = (i + 1.0) ** (-p) * np.exp(-lam * i)
    seq /= seq.sum()
    return seq


def _stretched_exponential_seq(n: int, lam: float, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    seq = np.exp(-lam * (i ** gamma))
    seq /= seq.sum()
    return seq


def _beta_seq(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(1, n + 1, dtype=np.float64) / n)
    seq = (x ** alpha) * ((1.0 - x) ** beta)
    s = seq.sum()
    if s == 0.0:
        return None
    seq /= s
    return seq


def _dirichlet_seq(n: int, conc: float) -> np.ndarray:
    g = np.random.gamma(conc, 1.0, size=n)
    seq = g / g.sum()
    return seq


# ----------------------------------------------------------------------
# Gradient refinement (Adam on log‑softmax) ---------------------------


def _adam_refine(
    a0: np.ndarray,
    n: int,
    fft_len2: int,
    deadline: float,
    best_cheap: float,
    best_vec: np.ndarray,
) -> tuple[np.ndarray, float]:
    """
    Refine a probability vector a0 using Adam on the log‑softmax
    parametrisation.  A smooth soft‑max approximation of max(conv) with a
    temperature β is minimised.  The function returns the best vector
    (with respect to the exact cheap score) found before the deadline.
    """
    eps = 1e-30
    beta = 5.0                # start with a modest temperature
    beta_max = 150.0          # final temperature (approaches true max)
    beta_inc = 1.05           # multiplicative increase each schedule step
    lr = 0.12                 # base learning rate for Adam (log‑space)
    b1, b2 = 0.9, 0.999
    eps_adam = 1e-8

    # initialise Adam moments
    y = np.log(a0 + eps)      # unconstrained log‑parameters
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0
    last_imp = 0
    best_local = best_vec.copy()
    best_local_val = best_cheap

    conv_len = 2 * n - 1
    conv_len2 = 3 * n - 2

    while time.time() < deadline - 2.0:
        t += 1

        # ---- softmax to get a (sum = 1, all >0) ----
        y_max = np.max(y)
        exp_y = np.exp(y - y_max)
        a = exp_y / exp_y.sum()

        # ---- FFT of a (zero‑padded to fft_len2) ----
        a_fft = np.fft.rfft(a, n=fft_len2)

        # ---- linear convolution a*a ----
        conv_full = np.fft.irfft(a_fft * a_fft, n=fft_len2)
        conv = conv_full[:conv_len]                # length 2n‑1

        # ---- smooth max via soft‑max (temperature β) ----
        beta_conv = beta * conv
        bc_max = np.max(beta_conv)
        w = np.exp(beta_conv - bc_max)             # stability shift
        w_sum = w.sum()
        w /= w_sum                                 # weights sum to 1

        # ---- convolution w * a (needed for gradient) ----
        w_fft = np.fft.rfft(w, n=fft_len2)
        wa_full = np.fft.irfft(w_fft * a_fft, n=fft_len2)
        wa = wa_full[:conv_len2]                   # length 3n‑2

        # gradient of the smooth objective w.r.t. a:
        # g_i = 2 * sum_j w_{i+j} a_j  (i = 0…n‑1)
        g = 2.0 * wa[n - 1 : n - 1 + n]

        # ---- gradient w.r.t. log‑parameters (mirror of simplex) ----
        mu = np.dot(a, g)
        grad_y = a * (g - mu)                      # shape (n,)

        # ---- Adam update ----
        m = b1 * m + (1.0 - b1) * grad_y
        v = b2 * v + (1.0 - b2) * (grad_y ** 2)
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- cheap exact score (max of linear conv) ----
        cheap = float(2.0 * n * conv.max())
        if cheap < best_local_val - 1e-12:
            best_local_val = cheap
            best_local = a.copy()
            last_imp = t
            # occasional log for progress
            if t % 5000 == 0:
                print(f"[Adam] iter {t:08d} new best cheap = {cheap:.6f}")

        # ---- schedule: increase β slowly (approaches true max) ----
        if t % 2000 == 0 and beta < beta_max:
            beta = min(beta * beta_inc, beta_max)

        # ---- restart if stagnated for a long time ----
        if t - last_imp > 120000:
            # random perturbation around the best seen so far
            y = np.log(best_local + eps) + 0.05 * np.random.randn(n)
            last_imp = t
            print(f"[Adam] restart after {t} iters without improvement")

    return best_local, best_local_val


# ----------------------------------------------------------------------
# (1+1) Evolution Strategy fine‑tuner -----------------------------------
# ----------------------------------------------------------------------


def _es_fine_tune(
    a_start: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    best_cheap: float,
    best_vec: np.ndarray,
) -> tuple[np.ndarray, float]:
    """
    Simple (1+1)‑ES on the exact cheap score.  Multiplicative
    log‑space mutations are used; successful steps shrink the step‑size.
    """
    sigma = 0.13          # mutation strength (log‑space)
    fail_cnt = 0
    best_local = best_vec.copy()
    best_local_val = best_cheap
    while time.time() < deadline - 1.0:
        cand = best_local * np.exp(sigma * np.random.randn(n))
        cand = _simplex_projection(cand)
        val = _cheap_score(cand, n, fft_len)
        if val < best_local_val:
            best_local_val = val
            best_local = cand
            sigma = max(sigma * 0.85, 1e-5)
            fail_cnt = 0
            print(f"[ES] improved cheap = {val:.6f}  sigma={sigma:.5f}")
        else:
            fail_cnt += 1
            if fail_cnt > 250:
                sigma = min(sigma * 1.30, 0.6)
                fail_cnt = 0
        # occasional random “restart” from a fresh Dirichlet if we are stuck
        if fail_cnt == 0 and np.random.rand() < 0.0005:
            pert = _dirichlet_seq(n, np.random.uniform(0.001, 2.0))
            best_local = _simplex_projection(best_local + 0.02 * pert)
    return best_local, best_local_val

# Main entry point required by the platform ---------------------------

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimizes
    ``evaluate_sequence``.  The routine adapts its schedule to the available
    time budget and always returns a feasible sequence.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 6.0  # safety margin for the final evaluation

    # --------------------------------------------------------------
    # Choose a working length.
    # --------------------------------------------------------------
    try:
        prev = globals()["height_sequence_1"]
        prev_arr = np.asarray(prev, dtype=np.float64)
        if prev_arr.size > 0:
            prev_arr = np.maximum(prev_arr, 0.0)
            s = prev_arr.sum()
            if s > 0.0:
                prev_arr /= s
            else:
                prev_arr = None
    except Exception:
        prev_arr = None

    # Base length: at least 5 000, not larger than 20 000 (still fast)
    base_n = max(len(prev_arr) if prev_arr is not None else 0, 5000)
    n = int(min(max(base_n, 5000), 20000))
    print(f"[Info] Working length n = {n}")

    # Prepare FFT lengths
    fft_len = 1 << ((2 * n - 1).bit_length())          # >= 2·n‑1
    fft_len2 = 1 << ((3 * n - 2).bit_length())        # >= 3·n‑2 (used by Adam)

    # --------------------------------------------------------------
    # Phase 0 – quick exploration of parametric families
    # --------------------------------------------------------------
    best_a = _uniform_seq(n)
    best_val = _cheap_score(best_a, n, fft_len)

    # keep previous construction if compatible
    if prev_arr is not None and prev_arr.size == n:
        v = _cheap_score(prev_arr, n, fft_len)
        if v < best_val:
            best_a, best_val = prev_arr.copy(), v

    # Helper to update best
    def _maybe_update(cand):
        nonlocal best_a, best_val
        val = _cheap_score(cand, n, fft_len)
        if val < best_val:
            best_a, best_val = cand, val
            print(f"[Init] new best cheap = {val:.6f}")

    # Power‑law sweep
    for p in np.linspace(0.2, 2.2, 101):
        if time.time() >= deadline:
            break
        _maybe_update(_power_law_seq(n, p))

    # Exponential sweep
    for lam in np.linspace(0.0, 5.0, 121):
        if time.time() >= deadline:
            break
        _maybe_update(_exponential_seq(n, lam))

    # Mixed power‑law × exponential (coarse grid)
    for p in np.linspace(0.2, 1.5, 13):
        for lam in np.linspace(0.0, 3.0, 14):
            if time.time() >= deadline:
                break
            _maybe_update(_power_law_exp_seq(n, p, lam))

    # Stretched exponential (γ in a modest set)
    for gamma in [0.5, 0.8, 1.0, 1.2, 1.5, 2.0]:
        for lam in np.linspace(0.0, 4.0, 31):
            if time.time() >= deadline:
                break
            _maybe_update(_stretched_exponential_seq(n, lam, gamma))

    # Beta‑shaped families (grid search)
    alphas = np.linspace(0.2, 8.0, 26)
    betas = np.linspace(0.2, 8.0, 26)
    for α in alphas:
        for β in betas:
            if time.time() >= deadline:
                break
            seq = _beta_seq(n, α, β)
            if seq is not None:
                _maybe_update(seq)

    # Random Dirichlet seeds (wide range of concentrations)
    for conc in [0.001, 0.005, 0.01, 0.05, 0.1, 0.5, 1.0, 2.0, 5.0]:
        if time.time() >= deadline:
            break
        _maybe_update(_dirichlet_seq(n, conc))

    # Additional random hybrid families
    for _ in range(150):
        if time.time() >= deadline:
            break
        p = np.random.uniform(0.1, 2.0)
        lam = np.random.exponential(scale=0.5)
        _maybe_update(_power_law_exp_seq(n, p, lam))

    print(f"[Init] best cheap score after parametric scan = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – Adam gradient refinement on soft‑max of the max‑norm
    # --------------------------------------------------------------
    best_a, best_val = _adam_refine(
        best_a, n, fft_len2, deadline, best_val, best_a
    )
    print(f"[Adam] best cheap after gradient = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – (1+1) ES hill‑climbing on exact cheap score
    # --------------------------------------------------------------
    best_a, best_val = _es_fine_tune(
        best_a, n, fft_len, deadline, best_val, best_a
    )
    print(f"[ES] final cheap score = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – final exact evaluation
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)  # noqa: F821 (provided by the platform)
        print(f"[Result] exact evaluation = {final_score:.6f}")
    except Exception as exc:      # defensive – should never happen
        final_score = float("inf")
        print(f"[Error] final evaluation failed: {exc}")

    # --------------------------------------------------------------
    # Return the best sequence discovered.
    # --------------------------------------------------------------
    return final_seq
