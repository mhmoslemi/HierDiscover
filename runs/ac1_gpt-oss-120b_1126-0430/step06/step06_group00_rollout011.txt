def _max_conv_and_shift_grad(a):
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    conv = np.fft.irfft(a_fft * a_fft, L)[:2*n-1]

    # gradient of max_conv w.r.t a:

"""
Optimisation of a non‑negative step‑function (probability vector) for the
evaluation metric 2·n·max(conv(a,a)) / (sum(a)²).

The algorithm works in three stages:
1. Initialise a from the exact discretisation of the arcsine (Beta(½,½))
   distribution – the continuous optimum for the problem.
2. Run a long mirror‑descent/Adam optimisation on the simplex.  The loss is
   a smoothed max‑convolution obtained with a soft‑max weighting; the
   smoothing constant is multiplied by a decaying factor so that the update
   gradually targets the true max.
3. When the smoothing becomes very small we switch to the exact sub‑gradient
   of the max‑convolution (a shifted copy of the current vector).  The
   optimiser continues with a smaller stepsize, allowing a fine‑grained
   polishing of the distribution.

A larger support (n≈5·10⁴) reduces the discretisation error of the arcsine
approximation; the optimiser is allowed to run until the time budget (≈1000 s)
is exhausted.  The best vector seen during the run is returned as a plain
list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (must be top‑level)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max that returns a probability vector (sum=1)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _fft_len_for_corr(n: int) -> int:
    """Smallest power‑of‑two ≥ (3·n‑2). Sufficient for all needed FFTs."""
    return 1 << ((3 * n - 2).bit_length())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Smoothed maximum of the linear self‑convolution a*a and its gradient.

    Parameters
    ----------
    a : np.ndarray
        Probability vector (non‑negative, sum ≈ 1).
    eps_factor : float
        Multiplicative factor for the smoothing constant:
        eps = max(eps_factor * max(conv), 1e‑12).

    Returns
    -------
    max_conv : float
        Maximum entry of the linear self‑convolution.
    grad_a : np.ndarray
        Gradient of 2·n·smooth_max with respect to a (already multiplied by 4·n).
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    a_fft = np.fft.rfft(a, L)

    # linear self‑convolution (length 2·n‑1)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # soft‑max weights over the convolution entries
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv          # for numerical stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or np.isnan(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # cross‑correlation a ⋆ w (length 3·n‑2, we need first n entries)
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # gradient (the factor 4·n comes from the derivative of 2·n·max)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a


def _conv_max_and_grad_exact(a: np.ndarray):
    """
    Compute the exact maximum self‑convolution and its (sub)gradient.

    Returns
    -------
    max_conv : float         – maximum entry of a*a
    grad_a   : np.ndarray    – gradient of 2·n·max_conv w.r.t. a (i.e. 4·n·shift)
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_idx = int(conv.argmax())
    max_conv = float(conv[max_idx])

    # Build the shift vector: for each j, contribution = a[max_idx - j] if valid
    shift = np.zeros_like(a)
    k = max_idx
    if k < n:
        # indices 0 .. k correspond to a[k], a[k-1], ..., a[0]
        shift[: k + 1] = a[k::-1]
    else:
        offset = k - (n - 1)            # first non‑zero index
        length = n - offset             # how many entries are non‑zero
        shift[offset:] = a[:length][::-1]

    # grad of max_conv is 2 * shift, gradient of 2·n·max_conv is 4·n * shift
    grad_a = 4.0 * n * shift
    return max_conv, grad_a

        n = external_n
        print(f"[Info] Reusing external length n = {n}")

    # ------------------------------------------------------------------
    # 1. Initialise from the exact arcsine CDF (area discretisation)
    # ------------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)
    x0 = idx / n
    x1 = (idx + 1) / n

    def arcsine_cdf(x):
        # arcsine CDF: F(x) = (2/π)·arcsin(√x)   for x∈[0,1]
        return (2.0 / np.pi) * np.arcsin(np.sqrt(np.clip(x, 0.0, 1.0)))

    a = arcsine_cdf(x1) - arcsine_cdf(x0)
    a_sum = a.sum()
    if a_sum == 0.0 or np.isnan(a_sum):
        raise RuntimeError("Failed to initialise a valid arcsine distribution.")
    a /= a_sum                     # guarantee sum=1

    # ------------------------------------------------------------------
    # 2. Initial evaluation (cheap version)
    # ------------------------------------------------------------------
    best_a = a.copy()
    # compute initial max‑convolution and its gradient (using the smooth version)
    init_eps = 0.2
    max_conv, grad_a = _conv_max_and_grad(best_a, init_eps)
    best_score = 2.0 * n * max_conv
    _log_progress(0, best_score, start_time)

    # ------------------------------------------------------------------
    # 3. Mirror‑descent / Adam optimisation on the simplex
    # ------------------------------------------------------------------
    # Parametrise the simplex via logits z (a = softmax(z))
    z = np.log(best_a + 1e-16)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    # Learning‑rate schedule parameters
    base_alpha = 0.8          # initial magnitude for the Adam step
    eps_factor = 0.2          # start smoothing factor (multiplied with max_conv)
    eps_decay = 0.9998        # exponential decay per iteration
    eps_factor_min = 5e-7     # when we switch to exact gradient

    it = 0
    max_iters = 2_000_000     # we will also stop by wall‑clock time

    while time.time() < deadline and it < max_iters:
        it += 1

        # ---- Current probability vector (always sums to 1) ----
        a_cur = _softmax(z)

        # ---- Choose gradient computation mode -----------------
        if eps_factor > eps_factor_min:
            # Soft‑max smooth approximation
            max_conv, grad_a = _conv_max_and_grad(a_cur, eps_factor)
        else:
            # Exact sub‑gradient of the true max‑convolution
            max_conv, grad_a = _conv_max_and_grad_exact(a_cur)

        cheap_score = 2.0 * n * max_conv

        # Keep the best candidate seen so far
        if cheap_score < best_score:
            best_score = cheap_score
            best_a = a_cur.copy()
            # Log only on improvement or periodically
            if it % 2000 == 0 or cheap_score < 1.02:
                _log_progress(it, best_score, start_time)

        # ---- Mirror‑descent gradient on logits (natural gradient) ----
        # grad_a already equals ∂J/∂a (J = 2·n·max_conv)
        mu = np.dot(grad_a, a_cur)            # 〈∇J, a〉
        grad_z = a_cur * (grad_a - mu)        # ∇_z J

        # ---- Adam update (bias‑corrected) ----
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # Inverse‑sqrt decay of the step size
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step = - alpha_t * m_hat / (np.sqrt(v_hat) + eps_adam)

        z = z + step
        # Clip logits to keep the exponentials numerically safe
        z = np.clip(z, -80.0, 80.0)

        # ---- Decay smoothing factor ----
        eps_factor = max(eps_factor_min, eps_factor * eps_decay)

        # ---- Occasional random jitter to escape plateaus ----
        if it % 5000 == 0 and np.random.rand() < 0.15:
            z += 0.01 * np.random.randn(n)

        # ---- Early stopping if we already beat the target ----
        if best_score < 1.503 and it > 5000:
            print("[Info] Target crossed – terminating early.")
            break

    # ------------------------------------------------------------------
    # 4. Return the best sequence (plain Python list)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()

    # Optional exact evaluation for logging (platform evaluates anyway)
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[Info] Final exact score = {final_score:.6f}")
    except Exception as exc:
        print(f"[Warning] Evaluation error (ignored): {exc}")

    return final_seq
