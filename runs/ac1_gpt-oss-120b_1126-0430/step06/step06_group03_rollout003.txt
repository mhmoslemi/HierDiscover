"""
Hybrid multi‑stage optimiser for the self‑convolution bound.

Stage 0 – initialise: use the provided `height_sequence_1` if available,
otherwise start from a uniform distribution on a moderately large size
(≈25 000 entries).  All vectors are normalised to sum = 1, because the
objective 2·n·max(a*a)/sum(a)² is invariant under scaling.

Stage 1 – coarse parametric sweep: a handful of families (power‑law,
offset‑power‑law, double‑power‑law, stretched‑exponential, exponential,
logistic, symmetric product‑power‑law, inverse‑sum‑power‑law and a
piece‑wise uniform “edge‑blocks’’ family) are evaluated on a short grid.
The best few candidates are retained.

Stage 2 – accelerated mirror‑descent (Adam) on the simplex:
starting from each retained candidate we run an exponential‑gradient
descent on the logits of the probability vector.  The objective is the
smooth‑max of the self‑convolution; the smoothing constant ε = eps_factor·max
is decreased geometrically (continuation) so the algorithm first sees a
very smooth landscape and finally optimises the true max.  The learning‑rate
decays like 1/√t and a back‑tracking line‑search guarantees monotone
improvement.  Stagnation triggers a restart from the best point seen so far.

Stage 3 – simulated‑annealing random walk:
the best sequence from stage 2 is refined by many cheap coordinate moves
(transfer a small mass from one index to another).  Moves that improve the
cheap score are always accepted; worsening moves are accepted with the
Metropolis probability exp(−Δ/T) where the temperature T is slowly cooled.

Finally the best sequence is converted to a plain list and handed to
`evaluate_sequence`.  All numbers are non‑negative, bounded by 1000 and the
sum is > 0.01, satisfying the evaluator’s checks.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
#  Utility functions (no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max returning a probability vector."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    s = e.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / z.size)
    return e / s


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑its that reproduce `a` after a soft‑max (adds a tiny epsilon)."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """Smallest power‑of‑two ≥ 3·n‑2 (enough for all needed convolutions)."""
    return 1 << ((3 * n - 2).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Returns (max_conv, grad_a) for the smoothed objective.

    The gradient corresponds to d/da [2·n·smooth_max(a*a)] where the
    smoothing constant is ε = eps_factor·max_conv.
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    eps = max(max_conv * eps_factor, 1e-12)

    # soft‑max weights over the convolution entries
    shifted = conv - max_conv          # ≤ 0 for stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # correlation a ⋆ w (size n)
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # derivative of 2·n·smooth_max (the factor 4·n comes from the
    # derivative of the quadratic convolution)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a

#  Parametric families of distributions
# ----------------------------------------------------------------------


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}   (normalised)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _power_law_offset_distribution(n: int, p: float, offset: float) -> np.ndarray:
    """a_i ∝ (i+offset)^{-p}   (normalised)."""
    idx = np.arange(offset, offset + n, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _double_power_law_distribution(n: int, p: float, q: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}·(n-i)^{-q}   (normalised)."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) ** (-p)) * ((n - i) ** (-q))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _stretched_exp_distribution(n: int, tau_factor: float, gamma: float) -> np.ndarray:
    """a_i ∝ exp( - (i/(tau_factor·n))^{gamma} )   (normalised)."""
    tau = max(1e-12, tau_factor * n)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(- (i / tau) ** gamma)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _exp_decay_distribution(n: int, lam: float) -> np.ndarray:
    """a_i ∝ e^{-lam·i}+e^{-lam·(n‑1‑i)}   (normalised)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _logistic_distribution(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ 1/(1+e^{α·(i‑mid)})   (normalised)."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = 1.0 / (1.0 + np.exp(alpha * (i - mid)))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _sym_product_power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ ((i+1)*(n‑i))^{-p}   (symmetric, normalised)."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) * (n - i)) ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _inverse_sum_power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p} + (n‑i)^{-p}   (high at the edges, low in the middle)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) + (n - i) ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _edge_block_distribution(n: int, frac: float) -> np.ndarray:
    """
    Uniform mass on the leftmost and rightmost blocks.
    `frac` ∈ (0,0.5] is the fraction of entries belonging to each block.
    The two blocks receive equal total mass.
    """
    w = max(min(frac, 0.5), 1e-6)
    block = int(round(w * n))
    if block == 0:
        block = 1
    a = np.zeros(n, dtype=np.float64)
    a[:block] = 0.5 / block
    a[-block:] = 0.5 / block
    # already normalised (sum = 1)
    return a

def _adam_optimize(start_a: np.ndarray, n: int, deadline: float,
                   best_score_initial: float = None,
                   max_iters: int = 1_500_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (Adam) on the simplex, starting from `start_a`.  Returns
    the best sequence and its cheap score found before `deadline`.
    """
    # initialise logits with a tiny random perturbation
    logits = _logits_from_distribution(start_a) + 0.02 * np.random.randn(n)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    # hyper‑parameters (tuned for speed/quality)
    base_alpha = 0.55            # learning‑rate for logits
    eps_factor = 0.30            # start smoothing proportion
    eps_factor_min = 5e-7
    eps_decay = 0.9995

    it = 0
    last_improve = 0
    best_a = start_a.copy()
    # cheap score = 2·n·max_conv (sum_a = 1)
    max_conv, _ = _conv_max_and_grad(start_a, eps_factor)
    best_score = 2.0 * n * max_conv
    if best_score_initial is not None and best_score_initial < best_score:
        best_score = best_score_initial
        best_a = start_a.copy()

    report_interval = 50000

    while time.time() < deadline and it < max_iters:
        it += 1
        a = _softmax(logits)                     # stays on the simplex

        max_conv, grad_a = _conv_max_and_grad(a, eps_factor)
        cheap_score = 2.0 * n * max_conv

        # record improvement
        if cheap_score < best_score - 1e-13:
            best_score = cheap_score
            best_a = a.copy()
            last_improve = it
            if it % report_interval == 0:
                print(f"[Adam] iter {it:07d} – new best cheap = {best_score:.8f}")

        # sub‑gradient w.r.t. logits (mirror‑descent)
        mu = np.dot(grad_a, a)                      # ⟨∇f(a), a⟩
        grad_z = a * (grad_a - mu)                  # ∇_z f where z = logits

        # Adam bookkeeping
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step size schedule (inverse‑sqrt)
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step = - m_hat / (np.sqrt(v_hat) + eps_adam)

        # back‑tracking line‑search (≤ 4 halvings)
        accepted = False
        trial_alpha = alpha_t
        for _ in range(4):
            new_logits = logits + trial_alpha * step
            # keep logits in a safe interval to avoid overflow
            new_logits = np.clip(new_logits, -50.0, 50.0)
            a_new = _softmax(new_logits)
            new_score = 2.0 * n * _conv_max(a_new)
            if new_score <= cheap_score:
                logits = new_logits
                accepted = True
                break
            trial_alpha *= 0.5
        if not accepted:
            # fall back to a small step even if it does not improve
            logits += alpha_t * step
            logits = np.clip(logits, -50.0, 50.0)

        # occasional random jitter to escape plateaus
        if it % 6000 == 0 and np.random.rand() < 0.20:
            logits += 0.03 * np.random.randn(n)
            m *= 0.6
            v *= 0.6

        # decay smoothing
        eps_factor = max(eps_factor_min, eps_factor * eps_decay)

        # restart if no progress for a long time
        if it - last_improve > 300_000:
            print(f"[Adam] stagnation after {it} iters – restarting.")
            logits = _logits_from_distribution(best_a) + 0.03 * np.random.randn(n)
            m.fill(0.0)
            v.fill(0.0)
            eps_factor = 0.30
            last_improve = it

    return best_a, best_score


def _simulated_annealing(start_a: np.ndarray, n: int, deadline: float,
                         start_score: float) -> tuple[np.ndarray, float]:
    """
    Simple Metropolis‑type random walk that transfers a small amount of mass
    between two positions.  The temperature decays geometrically.
    """
    a = start_a.copy()
    cheap_score = start_score

    # temperature schedule
    T = 1e-3
    T_min = 1e-7
    decay = 0.99995

    # statistics
    total_moves = 0
    accepted = 0

    while time.time() < deadline:
        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        if i == j:
            continue
        # propose moving a modest fraction of mass from i to j
        delta_frac = np.random.rand() * 0.04  # up to 4 % of a[i]
        delta = delta_frac * a[i]
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = 2.0 * n * _conv_max(a_new)

        improve = new_score <= cheap_score
        if improve or (np.random.rand() < math.exp(-(new_score - cheap_score) / max(T, 1e-12))):
            a = a_new
            cheap_score = new_score
            accepted += 1
            # occasional log
            if accepted % 5000 == 0:
                print(f"[Anneal] moves={total_moves} accepted={accepted} score={cheap_score:.8f}")

        total_moves += 1
        T = max(T_min, T * decay)

    print(f"[Anneal] finished – total moves={total_moves}, accepted={accepted}, final cheap={cheap_score:.8f}")
    return a, cheap_score
