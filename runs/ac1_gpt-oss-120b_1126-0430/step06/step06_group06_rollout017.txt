def propose_candidate(seed=12345, budget_s=1000):
   np.random.seed(seed)
   start = time.time()
   deadline = start + budget_s - 5

   # Choose n.
   try: prev = globals()["height_sequence_1"]; n = len(prev)
   except: n = 20000
   n = min(max(n, 20000), 25000) # use big n

   fft_len = 1 << ((2*n -1).bit_length())
   
   # cheap_score definition
   # using _conv_full_fft

   # candidate families
   best_a = _uniform_seq(n)
   best_val = cheap_score(best_a)

   # Additional families:

   # exponential small lam
   for lam in np.logspace(-5, -1, 30):  # 1e-5 to 0.1
       cand = _exponential_seq(n, lam)
       val = cheap_score(cand)
       if val < best_val:
           best_a, best_val = cand, val

   # power law small p
   for p in np.linspace(0.0, 2.0, 41): 
       cand = _power_law_seq(n, p)
       if cand is None: continue
       val = cheap_score(cand)
       if val < best_val:
           best_a, best_val = cand, val

   # power law * exp random combos:
   for _ in range(200):
       p = np.random.uniform(0.0, 2.0)
       lam = np.logspace(-5, -1, 1)[0] * np.random.exponential(0.5) ??? Not needed.
   # etc.

   # custom families
   # inv_sqrt
   i = np.arange(n, dtype=np.float64)+1
   seq = 1.0/np.sqrt(i) + 1.0/np.sqrt(n - i + 1)
   seq /= seq.sum()
   val = cheap_score(seq); update.

   # arcsine
   x = (i - 0.5)/n
   seq = 1.0/np.sqrt(x * (1 - x) + 1e-12)
   seq /= seq.sum()
   # update.

   # triangular shape
   seq = np.concatenate([np.arange(1, n//2+1), np.arange(n//2, 0, -1)])[:n]
   seq = seq.astype(np.float64)
   seq /= seq.sum()
   # update.

   # Dirichlet seeds
   for conc in [0.0005,0.001,0.005,0.01,0.05,0.1]:
      cand = _dirichlet_seq(n, conc)
      val = cheap_score(cand)
      if val < best_val: best_a, best_val = cand, val

   # store best, print.

   # Phase 1 mirror descent with Adam
   a = best_a.copy()
   best_iter_a = a.copy()
   best_iter_val = best_val

   m = np.zeros_like(a)
   v = np.zeros_like(a)
   beta1 = 0.9
   beta2 = 0.999
   eps = 1e-30
   t = 0

   while time.time() < deadline - 10:
       t += 1
       conv = _conv_full_fft(a, fft_len)
       maxb = conv.max()
       cheap = 2.0 * n * maxb

       if cheap < best_iter_val - 1e-12:
           best_iter_val = cheap
           best_iter_a = a.copy()
           print(f"[Adam {t}] new best cheap {cheap:.6f}")

       # active indices: those within relative tolerance
       # tolerance may be 1e-12 * maxb
       tol = maxb * 1e-12
       active = np.where(conv >= maxb - tol)[0]

       # compute subgradient g
       g = np.zeros_like(a)
       for k in active:
           lo = max(0, k - (n - 1))
           hi = min(k, n - 1)
           rng = np.arange(lo, hi + 1, dtype=np.int64)
           g[rng] += 2.0 * a[k - rng]
       g /= max(1, active.size)

       # Adam update in log-space
       m = beta1 * m + (1 - beta1) * g
       v = beta2 * v + (1 - beta2) * (g * g)
       m_hat = m / (1 - beta1 ** t)
       v_hat = v / (1 - beta2 ** t)

       log_a = np.log(a + eps) - (0.1 / (np.sqrt(v_hat) + eps)) * m_hat
       a = np.exp(log_a)

       # project onto simplex
       a = _simplex_projection(a)

   # Phase 2: (1+1)-ES
   sigma = 0.05
   fail = 0
   while time.time() < deadline - 3:
       cand = best_iter_a * np.exp(sigma * np.random.randn(n))
       cand = _simplex_projection(cand)
       val = cheap_score(cand)
       if val < best_iter_val:
           best_iter_val = val
           best_iter_a = cand
           sigma = max(sigma * 0.85, 1e-5)
           fail = 0
           print(f"[ES] improved cheap {val:.6f}, sigma {sigma:.5f}")
       else:
           fail += 1
           if fail > 200:
               sigma = min(sigma * 1.2, 0.5)
               fail = 0

   final_seq = best_iter_a.tolist()
   # final evaluation
   try:
       final_score = evaluate_sequence(final_seq)
       print(f"Final exact score {final_score:.6f}")
   except Exception as e:
       print(f"Evaluation error {e}")
   return final_seq

#!/usr/bin/env python
"""
propose_candidate
-----------------
Search for a non‑negative step function (list of heights) that minimises

        2·n·max_k (a*a)[k] / (Σ a_i)²

The routine works in three stages:

1. **Broad parametric sweep**  
   Exhaustively evaluates many handcrafted families (uniform,
   power‑law, exponential, mixed, inverse‑power, arcsine‑type, Dirichlet
   etc.).  The families are deliberately extended to the extreme
   parameter regimes that were not explored in the previous baseline
   (e.g. very small exponential rates λ≈c/n with c≈2 and β‑shapes with
   exponents < 0.2).  The best candidate of this stage is used as a
   warm‑start.

2. **Mirror‑descent with Adam‑style adaptive steps**  
   The cheap objective 2·n·max(conv) (the sum is normalised to 1) is
   non‑smooth; we use a sub‑gradient built from the active indices of
   the convolution and perform an exponentiated‑gradient (mirror) update
   in log‑space.  Adam momentum (β₁=0.9, β₂=0.999) makes the optimisation
   considerably faster and helps escape shallow plateaus.

3. **(1+1)‑ES fine‑tuning**  
   Multiplicative Gaussian perturbations in log‑space are applied to the
   best vector found so far.  Successful steps shrink the mutation
   strength, failures enlarge it – a classic 1/5th‑success rule.

All operations respect the 1000 s budget.  The final vector is projected
onto the probability simplex (Σa_i = 1, a_i ≥ 0), which guarantees that
`evaluate_sequence` returns exactly the cheap score.  The routine returns
the best sequence found before the time limit.

"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the probability simplex {x ≥ 0, Σx = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:                     # all entries non‑positive
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full linear convolution a*a using a real FFT.
    Returns a vector of length 2·n‑1 (the “valid” part of the
    convolution).
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len).real[: 2 * a.size - 1]
    return conv


def _conv_max_fft(a: np.ndarray, fft_len: int) -> float:
    """Maximum entry of the linear convolution a*a."""
    conv = _conv_full_fft(a, fft_len)
    return float(conv.max())


# ----------------------------------------------------------------------
# Parametric families (return a normalised probability vector)
# ----------------------------------------------------------------------


def _uniform_seq(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _power_law_seq(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = i ** (-p)
    seq /= seq.sum()
    return seq


def _exponential_seq(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    seq = np.exp(-lam * i)
    seq /= seq.sum()
    return seq


def _power_law_exp_seq(n: int, p: float, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    seq = (i + 1.0) ** (-p) * np.exp(-lam * i)
    seq /= seq.sum()
    return seq


def _stretched_exponential_seq(n: int, lam: float, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    seq = np.exp(-lam * (i ** gamma))
    seq /= seq.sum()
    return seq


def _dirichlet_seq(n: int, conc: float) -> np.ndarray:
    g = np.random.gamma(conc, 1.0, size=n)
    seq = g / g.sum()
    return seq


def _inv_power_seq(n: int, p: float) -> np.ndarray:
    """Symmetric inverse‑power shape: a_i ∝ (i+1)^{-p} + (n-i)^{-p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    r = np.arange(n, 0, -1, dtype=np.float64)
    seq = i ** (-p) + r ** (-p)
    seq /= seq.sum()
    return seq


def _arc_seq(n: int, gamma: float) -> np.ndarray:
    """Arcsine‑type shape: a_i ∝ ( x (1‑x) )^{-γ},
    where x = (i‑0.5)/n ∈ (0,1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    x = (i - 0.5) / n
    eps = 1e-12
    seq = (x * (1.0 - x) + eps) ** (-gamma)
    seq /= seq.sum()
    return seq


def _triangular_seq(n: int) -> np.ndarray:
    """A simple symmetric triangle, peak in the centre."""
    half = n // 2
    inc = np.arange(1, half + 1, dtype=np.float64)
    if n % 2:
        peak = np.array([half + 1], dtype=np.float64)
        seq = np.concatenate([inc, peak, inc[::-1]])
    else:
        seq = np.concatenate([inc, inc[::-1]])
    seq /= seq.sum()
    return seq


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that
    minimises ``evaluate_sequence``.  The routine respects the time
    budget and always returns a feasible sequence.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0            # safety margin

    # --------------------------------------------------------------
    # Choose a working length (the larger the better, up to the limit
    # imposed by the time budget).  If a previous solution exists we
    # reuse its length, otherwise we aim for the maximum allowed.
    # --------------------------------------------------------------
    try:
        prev = globals()["height_sequence_1"]
        n = max(len(prev), 20000)
        prev_arr = np.asarray(prev, dtype=np.float64)
        if prev_arr.sum() > 0.0:
            prev_arr /= prev_arr.sum()
        else:
            prev_arr = None
    except Exception:
        prev_arr = None
        n = 25000                                     # max feasible length
    n = int(min(max(n, 20000), 25000))               # 20 k ≤ n ≤ 25 k
    fft_len = 1 << ((2 * n - 1).bit_length())       # next power of two ≥ 2·n‑1

    # --------------------------------------------------------------
    # Cheap score (assumes Σa_i = 1)
    # --------------------------------------------------------------
    def cheap_score(vec: np.ndarray) -> float:
        maxb = _conv_max_fft(vec, fft_len)
        return 2.0 * n * maxb

    # --------------------------------------------------------------
    # Phase 0 – exhaustive scan over enlarged families
    # --------------------------------------------------------------
    best_a = _uniform_seq(n)
    best_val = cheap_score(best_a)

    # include the previous solution if it matches the current length
    if prev_arr is not None and prev_arr.size == n:
        val = cheap_score(prev_arr)
        if val < best_val:
            best_a, best_val = prev_arr.copy(), val

    # ----- Uniform (already the baseline) -----
    # ----- Power‑law ------------------------------------------------
    for p in np.linspace(0.0, 3.0, 61):
        if time.time() >= deadline:
            break
        cand = _power_law_seq(n, p)
        val = cheap_score(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # ----- Exponential – sweep λ = c / n, 0.1 ≤ c ≤ 5.0 -----
    for c in np.linspace(0.2, 5.0, 49):
        if time.time() >= deadline:
            break
        lam = c / n
        cand = _exponential_seq(n, lam)
        val = cheap_score(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # ----- Power‑law × Exponential (randomised grid) -------------
    for _ in range(300):
        if time.time() >= deadline:
            break
        p = np.random.uniform(0.0, 2.0)
        c = np.random.uniform(0.2, 5.0)
        lam = c / n
        cand = _power_law_exp_seq(n, p, lam)
        val = cheap_score(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # ----- Inverse‑power families (U‑shaped) --------------------
    for p in np.linspace(0.3, 2.5, 23):
        if time.time() >= deadline:
            break
        cand = _inv_power_seq(n, p)
        val = cheap_score(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # ----- Arcsine‑type families (γ controls edge concentration) --
    for gamma in [0.15, 0.20, 0.25, 0.30, 0.35]:
        if time.time() >= deadline:
            break
        cand = _arc_seq(n, gamma)
        val = cheap_score(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # ----- Triangular – just for completeness --------------------
    cand = _triangular_seq(n)
    val = cheap_score(cand)
    if val < best_val:
        best_a, best_val = cand, val

    # ----- Random Dirichlet seeds (very spiky) -----------------
    for conc in [0.0005, 0.001, 0.005, 0.01, 0.05, 0.1]:
        if time.time() >= deadline:
            break
        cand = _dirichlet_seq(n, conc)
        val = cheap_score(cand)
        if val < best_val:
            best_a, best_val = cand, val

    print(f"[Init] best cheap score after parametric sweep = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – mirror‑descent with Adam‑style adaptive steps
    # --------------------------------------------------------------
    a = best_a.copy()
    best_iter_a = a.copy()
    best_iter_val = best_val
    eps = 1e-30
    beta1, beta2 = 0.9, 0.999
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    last_improve = 0

    while time.time() < deadline - 8.0:
        t += 1
        conv = _conv_full_fft(a, fft_len)
        maxb = conv.max()
        cheap = 2.0 * n * maxb

        # record improvement
        if cheap < best_iter_val - 1e-12:
            best_iter_val = cheap
            best_iter_a = a.copy()
            last_improve = t
            print(f"[Adam {t:07d}] NEW best cheap = {cheap:.6f}")

        # active set – indices whose value is within a tiny relative
        # tolerance of the maximum (typically just one or a few)
        rel_tol = 1e-12
        active = np.where(conv >= maxb * (1.0 - rel_tol))[0]

        # sub‑gradient of the max‑norm (averaged over the active set)
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            rng = np.arange(lo, hi + 1, dtype=np.int64)
            g[rng] += 2.0 * a[k - rng]
        g /= float(active.size)

        # Adam update in log‑space (exponentiated‑gradient)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        log_a = np.log(a + eps) - (0.2 / (np.sqrt(v_hat) + eps)) * m_hat
        a = np.exp(log_a)
        a = _simplex_projection(a)

        # occasional restart if we have not improved for a long time
        if t - last_improve > 120_000:
            perturb = 0.01 * np.random.rand(n)
            a = _simplex_projection(best_iter_a + perturb)
            last_improve = t
            print(f"[Adam {t:07d}] Restart after stagnation.")

    # --------------------------------------------------------------
    # Phase 2 – simple (1+1)‑ES hill‑climbing on the cheap score
    # --------------------------------------------------------------
    sigma = 0.08         # mutation strength (log‑space)
    fail_cnt = 0
    while time.time() < deadline - 2.0:
        cand = best_iter_a * np.exp(sigma * np.random.randn(n))
        cand = _simplex_projection(cand)
        val = cheap_score(cand)
        if val < best_iter_val:
            best_iter_val = val
            best_iter_a = cand
            sigma = max(sigma * 0.85, 1e-5)
            fail_cnt = 0
            print(f"[ES] improved cheap = {val:.6f}, sigma={sigma:.5f}")
        else:
            fail_cnt += 1
            if fail_cnt > 250:
                sigma = min(sigma * 1.25, 0.6)
                fail_cnt = 0

        # once in a while inject a fresh Dirichlet restart
        if np.random.rand() < 0.001:
            fresh = _dirichlet_seq(n, np.random.uniform(0.0005, 0.1))
            cand = _simplex_projection(best_iter_a + 0.02 * fresh)
            # the (1+1)ES only mutates best_iter_a, so we keep best_iter_a

    # --------------------------------------------------------------
    # Phase 3 – final exact evaluation
    # --------------------------------------------------------------
    final_seq = best_iter_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:      # defensive – should never happen
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # --------------------------------------------------------------
    # Return the best sequence found.
    # --------------------------------------------------------------
    return final_seq
