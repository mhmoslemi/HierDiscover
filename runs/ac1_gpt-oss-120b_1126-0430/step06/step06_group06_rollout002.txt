#!/usr/bin/env python
"""
Propose a non‑negative step function (list of heights) that minimises

        2·n·max_k (a*a)[k] / (Σ a_i)²

Algorithm
---------
0. Determine a working length n (≥5 k, ≤25 k).  If a previous
   solution is available, use its length; otherwise start with 5 000.
1. Exhaustively scan several parametric families (uniform, power‑law,
   exponential, mixed, stretched‑exponential, beta, Dirichlet) and keep
   the best normalised candidate according to the cheap objective
   ``2·n·max(conv)``.
2. Refine the best candidate with a momentum‑based exponentiated‑gradient
   (Adam on log‑probabilities) that uses the true sub‑gradient of the
   max‑convolution.  The step size decays like 1/√t and we periodically
   restart from the current best to avoid stagnation.
3. Perform a lightweight simulated‑annealing / (1+1)‑ES hill‑climb that
   transfers a small amount of mass between two random coordinates.
   Successful moves shrink the mutation strength, failed moves slowly
   increase the temperature and the mutation strength.
4. Return the best sequence found and evaluate it with the supplied
   ``evaluate_sequence``.  All numbers are non‑negative and the sum is
   normalised to 1, therefore the exact evaluation is safe.

The routine respects a 1000 s time budget (leaving a few seconds for the
final exact evaluation).  It prints progress information that will be
captured by the judges.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, Σ x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution (a*a) using real FFT, length 2·n‑1."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len).real[: 2 * a.size - 1]
    return conv


def _conv_max_fft(a: np.ndarray, fft_len: int) -> float:
    """Maximum coefficient of a*a."""
    conv = _conv_full_fft(a, fft_len)
    return float(conv.max())


# ----------------------------------------------------------------------
# Parametric families (all return a normalised probability vector)
# ----------------------------------------------------------------------


def _uniform_seq(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _power_law_seq(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = i ** (-p)
    seq /= seq.sum()
    return seq


def _exponential_seq(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    seq = np.exp(-lam * i)
    seq /= seq.sum()
    return seq


def _power_law_exp_seq(n: int, p: float, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    seq = (i + 1.0) ** (-p) * np.exp(-lam * i)
    seq /= seq.sum()
    return seq


def _stretched_exponential_seq(n: int, lam: float, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    seq = np.exp(-lam * (i ** gamma))
    seq /= seq.sum()
    return seq


def _beta_seq(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(1, n + 1, dtype=np.float64) / n)
    seq = (x ** alpha) * ((1.0 - x) ** beta)
    s = seq.sum()
    if s == 0.0:
        return None
    seq /= s
    return seq


def _dirichlet_seq(n: int, conc: float) -> np.ndarray:
    g = np.random.gamma(conc, 1.0, size=n)
    seq = g / g.sum()
    return seq


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that
    minimises ``evaluate_sequence``.  The routine adapts its schedule to
    the available time budget and always returns a feasible sequence.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the final exact evaluation
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # Choose a working length.
    # --------------------------------------------------------------
    try:
        prev = globals()["height_sequence_1"]
        n = max(len(prev), 5000)
        prev_arr = np.asarray(prev, dtype=np.float64)
        if prev_arr.sum() > 0.0:
            prev_arr /= prev_arr.sum()
        else:
            prev_arr = None
    except Exception:
        prev_arr = None
        n = 5000

    n = int(min(max(n, 5000), 25000))
    fft_len = 1 << ((2 * n - 1).bit_length())      # next power of two ≥ 2·n‑1

    # --------------------------------------------------------------
    # Helper for the cheap objective (assumes Σa_i = 1)
    # --------------------------------------------------------------
    def cheap_score(vec: np.ndarray) -> float:
        maxb = _conv_max_fft(vec, fft_len)
        return 2.0 * n * maxb

    # --------------------------------------------------------------
    # Phase 0 – exhaustive scan of parametric families
    # --------------------------------------------------------------
    best_a = _uniform_seq(n)
    best_val = cheap_score(best_a)

    if prev_arr is not None and prev_arr.size == n:
        val = cheap_score(prev_arr)
        if val < best_val:
            best_a, best_val = prev_arr.copy(), val

    # Power‑law sweep
    for p in np.linspace(0.2, 2.0, 91):
        if time.time() >= deadline:
            break
        cand = _power_law_seq(n, p)
        val = cheap_score(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # Exponential sweep
    for lam in np.linspace(0.0, 5.0, 101):
        if time.time() >= deadline:
            break
        cand = _exponential_seq(n, lam)
        val = cheap_score(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # Mixed power‑law × exponential
    for p in np.linspace(0.2, 1.5, 14):
        for lam in np.linspace(0.0, 3.0, 16):
            if time.time() >= deadline:
                break
            cand = _power_law_exp_seq(n, p, lam)
            val = cheap_score(cand)
            if val < best_val:
                best_a, best_val = cand, val

    # Stretched exponential
    for gamma in [0.5, 0.9, 1.2, 1.5, 2.0]:
        for lam in np.linspace(0.0, 4.0, 41):
            if time.time() >= deadline:
                break
            cand = _stretched_exponential_seq(n, lam, gamma)
            val = cheap_score(cand)
            if val < best_val:
                best_a, best_val = cand, val

    # Beta‑shaped families
    x = (np.arange(1, n + 1, dtype=np.float64) / n)
    for α in np.linspace(0.2, 8.0, 25):
        for β in np.linspace(0.2, 8.0, 25):
            if time.time() >= deadline:
                break
            seq = (x ** α) * ((1.0 - x) ** β)
            s = seq.sum()
            if s == 0.0:
                continue
            cand = seq / s
            val = cheap_score(cand)
            if val < best_val:
                best_a, best_val = cand, val

    # Random Dirichlet seeds (wide range of concentrations)
    for conc in [0.001, 0.005, 0.01, 0.05, 0.1, 0.5, 1.0, 2.0, 5.0]:
        if time.time() >= deadline:
            break
        cand = _dirichlet_seq(n, conc)
        val = cheap_score(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # Additional randomized power‑law × exponential seeds
    for _ in range(250):
        if time.time() >= deadline:
            break
        p = np.random.uniform(0.1, 2.0)
        lam = np.random.exponential(scale=0.5)
        cand = _power_law_exp_seq(n, p, lam)
        val = cheap_score(cand)
        if val < best_val:
            best_a, best_val = cand, val

    print(f"[Phase 0] Initial best cheap score = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – Momentum exponentiated‑gradient (Adam) refinement
    # --------------------------------------------------------------
    a = best_a.copy()
    log_a = np.log(a + 1e-30)
    m = np.zeros_like(a)           # first moment
    v = np.zeros_like(a)           # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-30
    t = 0
    last_improve_iter = 0
    best_iter_a = a.copy()
    best_iter_val = best_val

    while time.time() < deadline - 10.0:
        t += 1
        conv = _conv_full_fft(a, fft_len)
        maxb = conv.max()
        cheap = 2.0 * n * maxb

        if cheap < best_iter_val - 1e-12:
            best_iter_val = cheap
            best_iter_a = a.copy()
            last_improve_iter = t
            print(f"[Iter {t:07d}] NEW best cheap = {cheap:.6f}")

        # active set (indices where conv is essentially maximal)
        active = np.where(conv >= maxb * (1.0 - 1e-12))[0]

        # sub‑gradient of the max‑convolution
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            g[idx] += 2.0 * a[k - idx]
        g /= float(active.size)

        # Adam on the log‑probabilities (mirror descent with entropy)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # step size decays like 0.6/√t (empirically reasonable)
        eta = 0.6 / (math.sqrt(t) + 1.0)

        log_a -= eta * m_hat / (np.sqrt(v_hat) + eps)
        a = np.exp(log_a)

        s = a.sum()
        if not np.isfinite(s) or s <= 0.0:
            # restart from the best seen so far
            a = best_iter_a.copy()
            log_a = np.log(a + eps)
            continue

        a /= s

        # occasional random jitter if we have been idle for a while
        if t - last_improve_iter > 180_000:
            jitter = 0.02 * np.random.rand(n)
            a = _simplex_projection(best_iter_a + jitter)
            log_a = np.log(a + eps)
            last_improve_iter = t
            print(f"[Iter {t:07d}] Restart after stagnation.")

        # safety break if we are really out of time
        if time.time() >= deadline - 10.0:
            break

    # --------------------------------------------------------------
    # Phase 2 – Simulated‑annealing / (1+1)‑ES hill climbing (cheap score)
    # --------------------------------------------------------------
    sigma = 0.12      # mutation strength in log‑space
    temperature = 0.02
    fail_cnt = 0
    a = best_iter_a.copy()           # start from the best after Phase 1

    while time.time() < deadline - 2.0:
        # propose a multiplicative Gaussian perturbation in log‑space
        cand = a * np.exp(sigma * np.random.randn(n))
        cand = _simplex_projection(cand)

        cand_val = cheap_score(cand)

        if cand_val < best_iter_val:
            best_iter_val = cand_val
            best_iter_a = cand.copy()
            a = cand
            sigma = max(sigma * 0.85, 1e-4)
            temperature = max(temperature * 0.95, 5e-4)
            fail_cnt = 0
            print(f"[ES] improved cheap = {cand_val:.6f}  sigma={sigma:.5f}")
        else:
            fail_cnt += 1
            # accept probabilistically (simulated annealing)
            if np.random.rand() < math.exp(-(cand_val - best_iter_val) / max(temperature, 1e-12)):
                a = cand
            # adapt mutation strength / temperature on long failures
            if fail_cnt > 250:
                sigma = min(sigma * 1.30, 0.6)
                temperature = min(temperature * 1.10, 0.2)
                fail_cnt = 0

        # occasional small random Dirichlet perturbation to escape plateaus
        if t % 200_000 == 0:
            pert = _dirichlet_seq(n, np.random.uniform(0.001, 2.0))
            a = _simplex_projection(best_iter_a + 0.02 * pert)

    # --------------------------------------------------------------
    # Phase 3 – final exact evaluation
    # --------------------------------------------------------------
    final_seq = best_iter_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:      # defensive – should never happen
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # --------------------------------------------------------------
    # Return the best sequence found.
    # --------------------------------------------------------------
    return final_seq
