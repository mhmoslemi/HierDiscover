def gradient_multiple(a, conv, top_k, temperature):
    # conv is array length 2n-1
    idxs = np.argpartition(-conv, top_k)[:top_k]  # largest top_k
    values = conv[idxs]
    if temperature > 0:
        weights = np.exp(values / temperature)
        weights /= weights.sum()
    else:
        # pure max
        weights = np.zeros_like(values)
        weights[0] = 1.0
    g = np.zeros_like(a)
    n = a.shape[0]
    for idx, w in zip(idxs, weights):
        lo = max(0, idx - (n - 1))
        hi = min(idx, n - 1)
        rng = np.arange(lo, hi + 1, dtype=np.int64)
        j = idx - rng
        eq = (rng == j)
        g[rng[~eq]] += w * 2.0 * a[j[~eq]]
        g[rng[eq]] += w * a[rng[eq]]
    return g

    n = a.size
    idxs = np.argpartition(-conv, top_k)[:top_k]
    vals = conv[idxs]
    max_val = vals.max()
    # use temperature as fraction of max value
    T = temperature * max_val if temperature > 0 else 0
    if T > 0:
        w = np.exp(vals / T)
        w = w / w.sum()
    else:
        w = np.zeros_like(vals)
        w[0] = 1.0

        hi = min(idx, n - 1)

        eq = (rng == j)
        g[rng[~eq]] += weight * 2.0 * a[j[~eq]]
        g[rng[eq]] += weight * a[rng[eq]]
    return g

#!/usr/bin/env python
"""
Propose a non‑negative step function (list of heights) that minimises
``evaluate_sequence``.  The algorithm consists of three stages:

1. **Broad parametric sweep** – generates many families
   (exponential, power‑law, beta, inverse‑sin, random Dirichlet) and keeps
   the best cheap candidate.
2. **Projected (additive) / exponentiated mirror descent** – iteratively
   reduces the maximum autocorrelation entry by using the exact sub‑gradient
   of the max (or a soft‑max over the top few lags) and projects back onto
   the probability simplex.
3. **Stochastic local search** – random swaps and small mass‑transfer moves
   are performed with an annealing schedule, yielding a final polishing
   phase.  The final exact ``evaluate_sequence`` value is printed and the
   best sequence is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    # Find the rho index (largest i such that u_i - cssv_i/(i+1) > 0)
    rho_arr = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        # all non‑positive – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho_arr[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using a power‑of‑two FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa).real[: 2 * a.size - 1]
    return conv


def _conv_max_fft(a: np.ndarray, fft_len: int) -> tuple[float, int]:
    """Maximum value of the autocorrelation a*a and its index."""
    conv = _conv_full_fft(a, fft_len)
    idx = int(conv.argmax())
    return float(conv[idx]), idx


def _cheap_score(a: np.ndarray, fft_len: int) -> float:
    """Cheap upper‑bound: 2·n·max(conv) (sum a is assumed to be 1)."""
    maxb, _ = _conv_max_fft(a, fft_len)
    n = a.shape[0]
    return 2.0 * n * maxb


def _gradient_max(a: np.ndarray, conv: np.ndarray, kmax: int) -> np.ndarray:
    """Exact sub‑gradient of the maximal convolution entry."""
    n = a.shape[0]
    g = np.zeros_like(a)
    lo = max(0, kmax - (n - 1))
    hi = min(kmax, n - 1)
    rng = np.arange(lo, hi + 1, dtype=np.int64)
    j = kmax - rng
    eq = (rng == j)
    # off‑diagonal contributions (weight 2)
    g[rng[~eq]] = 2.0 * a[j[~eq]]
    # diagonal contribution (weight 1)
    g[rng[eq]] = a[rng[eq]]
    return g


def _gradient_top(a: np.ndarray, conv: np.ndarray, top_k: int = 3,
                  temperature: float = 0.05) -> np.ndarray:
    """Weighted gradient built from the top‑k convolution entries."""
    n = a.shape[0]
    # pick top‑k indices (unsorted)
    idxs = np.argpartition(-conv, top_k)[:top_k]
    vals = conv[idxs]
    # smooth weights (soft‑max)
    max_val = vals.max()
    # avoid division by zero – temperature relative to max
    T = max(temperature * max_val, 1e-12)
    w = np.exp(vals / T)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.zeros_like(w)
        w[0] = 1.0
    else:
        w = w / w_sum
    g = np.zeros_like(a)
    for idx, weight in zip(idxs, w):
        lo = max(0, idx - (n - 1))
        hi = min(idx, n - 1)
        rng = np.arange(lo, hi + 1, dtype=np.int64)
        j = idx - rng
        eq = (rng == j)
        g[rng[~eq]] += weight * 2.0 * a[j[~eq]]
        g[rng[eq]] += weight * a[rng[eq]]
    return g


# ----------------------------------------------------------------------
# Core search routine required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that
    minimises ``evaluate_sequence``.  The routine adapts its schedule to the
    available time budget and always returns a feasible sequence.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0  # keep a small safety margin
    eps = 1e-30

    # --------------------------------------------------------------
    # Decide on a working length.
    # --------------------------------------------------------------
    # Try to reuse a previously found sequence if it exists.
    try:
        prev = globals().get('height_sequence_1')
        prev_arr = None
        if prev is not None:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.size > 0 and prev_arr.sum() > 0.0:
                prev_arr = prev_arr / prev_arr.sum()
        if prev_arr is not None:
            n = max(prev_arr.size, 5000)
        else:
            n = 5000
    except Exception:
        n = 5000
    n = int(min(max(n, 5000), 20000))
    fft_len = 1 << ((2 * n - 1).bit_length())  # next power of two ≥ 2·n‑1

    # --------------------------------------------------------------
    # Initial parametric families – cheap evaluation only
    # --------------------------------------------------------------
    best_a = None
    best_val = float('inf')

    def try_candidate(a_candidate: np.ndarray, label: str = ''):
        nonlocal best_a, best_val
        if time.time() >= deadline:
            return
        # Normalise to sum 1 (the evaluation is homogeneous)
        a = a_candidate.astype(np.float64)
        s = a.sum()
        if s <= 0:
            return
        a /= s
        val = _cheap_score(a, fft_len)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Init] New best ({label}) cheap = {val:.6f}")

    # Uniform baseline
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    try_candidate(uniform, 'uniform')

    # Previous construction if available and same length
    if prev_arr is not None and prev_arr.size == n:
        try_candidate(prev_arr, 'prev')

    # ----- Exponential (left) ------------------------------------
    for lam in np.linspace(0.0, 10.0, 41):
        if time.time() >= deadline:
            break
        a = np.exp(-lam * np.arange(n, dtype=np.float64))
        try_candidate(a, f'exp_left lam={lam:.2f}')

    # ----- Symmetric exponential (U‑shaped) -----------------------
    for lam in np.linspace(0.0, 10.0, 41):
        if time.time() >= deadline:
            break
        idx = np.arange(n, dtype=np.float64)
        a = np.exp(-lam * np.minimum(idx, n - 1 - idx))
        try_candidate(a, f'exp_sym lam={lam:.2f}')

    # ----- Power‑law (left) --------------------------------------
    for p in np.linspace(0.01, 3.0, 61):
        if time.time() >= deadline:
            break
        a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
        try_candidate(a, f'pow_left p={p:.3f}')

    # ----- Symmetric power‑law (U‑shaped) ------------------------
    for p in np.linspace(0.01, 3.0, 61):
        if time.time() >= deadline:
            break
        i = np.arange(n, dtype=np.float64)
        a = np.minimum(i + 1, n - i) ** (-p)
        try_candidate(a, f'pow_sym p={p:.3f}')

    # ----- Beta families (U‑shaped when α,β<1) --------------------
    x = (np.arange(1, n + 1, dtype=np.float64) / n)
    alphas = np.concatenate((
        np.linspace(0.01, 0.2, 5),
        np.linspace(0.2, 1.0, 9),
        np.linspace(1.0, 5.0, 5)
    ))
    betas = alphas.copy()
    for alpha in alphas:
        for beta in betas:
            if time.time() >= deadline:
                break
            a = (x ** alpha) * ((1.0 - x) ** beta)
            s = a.sum()
            if s == 0.0:
                continue
            try_candidate(a, f'beta a={alpha:.2f} b={beta:.2f}')

    # ----- Inverse‑sin power family (edge‑heavy) ---------------
    # a_i ∝ (sin(pi*(i+0.5)/n) + eps)^{-p}
    for p in np.linspace(0.5, 3.0, 26):
        if time.time() >= deadline:
            break
        i = np.arange(n, dtype=np.float64)
        sin_vals = np.sin(np.pi * (i + 0.5) / n) + eps
        a = sin_vals ** (-p)
        try_candidate(a, f'isinv_sin p={p:.2f}')

    # ----- Random Dirichlet seeds --------------------------------

   evaluated with a cheap proxy ``2·n·max(conv)``.  The best candidate
   is kept.

   candidate we repeatedly compute the full convolution, build a

   multiplicative (exponentiated‑gradient) or additive projected gradient
   step with a diminishing step size.  The iterate with the lowest cheap
   score is retained.

   mixture of random swaps (permutations) and tiny mass‑transfer moves

4. **Final evaluation** – the best sequence found is passed to the

The routine is time‑aware (default budget 1000 s) and always returns a
feasible sequence with non‑negative entries that sum to a positive

def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]

    rho_arr = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho_arr[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    """Full linear convolution a*a using a power‑of‑two FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa).real[: 2 * a.size - 1]
    return conv

    """Maximum value of the autocorrelation a*a and its index."""
    conv = _conv_full_fft(a, fft_len)
    idx = int(conv.argmax())
    return float(conv[idx]), idx

    """Cheap upper‑bound: 2·n·max(conv); a is assumed to sum to 1."""
    maxb, _ = _conv_max_fft(a, fft_len)

    """Exact sub‑gradient of the maximal convolution entry."""
    n = a.shape[0]

    hi = min(kmax, n - 1)
    rng = np.arange(lo, hi + 1, dtype=np.int64)
    j = kmax - rng
    eq = (rng == j)
    g[rng[~eq]] = 2.0 * a[j[~eq]]
    g[rng[eq]] = a[rng[eq]]
    return g


def _gradient_top(a: np.ndarray, conv: np.ndarray, top_k: int = 3,
                  temperature: float = 0.05) -> np.ndarray:
    """Weighted gradient built from the top‑k convolution entries."""
    n = a.shape[0]
    idxs = np.argpartition(-conv, top_k)[:top_k]
    vals = conv[idxs]
    max_val = vals.max()
    T = max(temperature * max_val, 1e-12)
    w = np.exp(vals / T)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.zeros_like(w)
        w[0] = 1.0
    else:
        w = w / w_sum
    g = np.zeros_like(a)
    for idx, weight in zip(idxs, w):
        lo = max(0, idx - (n - 1))
        hi = min(idx, n - 1)
        rng = np.arange(lo, hi + 1, dtype=np.int64)
        j = idx - rng
        eq = (rng == j)
        g[rng[~eq]] += weight * 2.0 * a[j[~eq]]
        g[rng[eq]] += weight * a[rng[eq]]
    return g
