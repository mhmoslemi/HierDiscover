#!/usr/bin/env python
"""
Propose a non‑negative step function that minimizes
        2·n·max(a*a) / (Σ a)^2

The search consists of three stages:

1️⃣  **Initialisation** – a fast sweep over a few parametric families
    (geometric, power‑law, mixed) on a moderately large length
    (8 k – 20 k).  The best candidate (lowest cheap score
    ``2·n·max(conv)``) is kept as a start point.

2️⃣  **Smoothed mirror‑descent** – we minimise a smooth approximation of
    the max‑convolution using an Adam optimiser on the simplex.
    The max is replaced by a softmax (temperature β slowly increased
    from 1 → 200).  Gradients are obtained analytically:
        ∂/∂a_i  Σ_k w_k (a*a)_k = 2·Σ_k w_k a_{k‑i}
    which is a (linear) correlation that we evaluate by FFT.
    After each Adam step we project back onto the probability simplex.

3️⃣  **Final refinement** – a cheap (1+1)‑ES hill‑climb on the exact
    cheap score to polish the result, followed by a single call to the
    supplied ``evaluate_sequence`` for the final exact score.

The routine adapts its schedule to the remaining time budget,
always returns a feasible sequence and never exceeds the 1000 s limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        # all non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full_fft(a: np.ndarray, b: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full linear convolution (a * b) using real FFT.
    Length of result = len(a) + len(b) - 1.
    """
    fa = np.fft.rfft(a, fft_len)
    fb = np.fft.rfft(b, fft_len)
    conv = np.fft.irfft(fa * fb, fft_len).real
    # Truncate to linear convolution length
    out_len = a.shape[0] + b.shape[0] - 1
    return conv[:out_len]


def _conv_full_fft_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Convenience: full convolution a*a."""
    return _conv_full_fft(a, a, fft_len)


def _conv_max_fft(a: np.ndarray, fft_len: int) -> float:
    """Maximum coefficient of a*a."""
    conv = _conv_full_fft_self(a, fft_len)
    return float(conv.max())


# ----------------------------------------------------------------------
# Parametric families (all return a normalised probability vector)
# ----------------------------------------------------------------------


def _uniform_seq(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exponential_seq(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    seq = np.exp(-lam * i)
    seq /= seq.sum()
    return seq


def _power_law_seq(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = i ** (-p)
    seq /= seq.sum()
    return seq


def _power_law_exp_seq(n: int, p: float, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    seq = (i + 1.0) ** (-p) * np.exp(-lam * i)
    seq /= seq.sum()
    return seq


def _dirichlet_seq(n: int, conc: float) -> np.ndarray:
    g = np.random.gamma(conc, 1.0, size=n)
    seq = g / g.sum()
    return seq

# Core optimiser
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that
    minimises ``evaluate_sequence``.  The routine adapts its schedule to
    the available time budget and always returns a feasible sequence.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # leave a safety margin

    # --------------------------------------------------------------
    # Choose a working length.  Prefer the length of a previous run,
    # otherwise pick a reasonably large size.
    # --------------------------------------------------------------
    try:
        prev = globals()["height_sequence_1"]
        prev_arr = np.asarray(prev, dtype=np.float64)
        n0 = prev_arr.size
    except Exception:
        prev_arr = None
        n0 = None

    # Target length: 12 k – 20 k (still fast enough for many FFTs)
    n = int(min(max(n0 or 12000, 8000), 20000))
    print(f"[Info] Working length n = {n}")

    # Pre‑compute FFT lengths
    fft_len_self = 1 << ((2 * n - 1).bit_length())                # for a*a
    fft_len_corr = 1 << ((3 * n - 2).bit_length())                # for correlation w*a

    # --------------------------------------------------------------
    # Cheap score (assumes Σa_i = 1): 2·n·max(conv)
    # --------------------------------------------------------------
    def cheap_score(vec: np.ndarray) -> float:
        maxb = _conv_max_fft(vec, fft_len_self)
        return 2.0 * n * maxb

    # --------------------------------------------------------------
    # Phase 0 – quick exploration of simple families
    # --------------------------------------------------------------
    best_a = _uniform_seq(n)
    best_val = cheap_score(best_a)

    # try a few geometric families (lam in [0,5])
    for lam in np.linspace(0.0, 5.0, 31):
        if time.time() >= deadline:
            break
        cand = _exponential_seq(n, lam)
        val = cheap_score(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # power‑law sweep
    for p in np.linspace(0.2, 2.0, 41):
        if time.time() >= deadline:
            break
        cand = _power_law_seq(n, p)
        val = cheap_score(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # mixed power‑law × exponential (coarse)
    for _ in range(120):
        if time.time() >= deadline:
            break
        p = np.random.uniform(0.2, 1.5)
        lam = np.random.exponential(scale=1.0)
        cand = _power_law_exp_seq(n, p, lam)
        val = cheap_score(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # random Dirichlet seeds (wide concentration range)
    for conc in [0.001, 0.005, 0.01, 0.05, 0.1, 0.5, 1.0, 2.0, 5.0]:
        if time.time() >= deadline:
            break
        cand = _dirichlet_seq(n, conc)
        val = cheap_score(cand)
        if val < best_val:
            best_a, best_val = cand, val

    print(f"[Init] best cheap score = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – smoothed mirror‑descent (Adam) on the true objective
    # --------------------------------------------------------------
    a = best_a.copy()
    best_iter_a = a.copy()
    best_iter_val = best_val

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12
    lr = 0.05  # base learning rate for Adam

    # smoothing schedule for soft‑max
    beta_start = 1.0
    beta_end = 200.0

    iter_cnt = 0
    max_iters = 1_000_000  # generous upper bound; loop stops on time anyway
    last_improve = 0

    while time.time() < deadline - 4.0 and iter_cnt < max_iters:
        iter_cnt += 1

        # ---- convolution a*a -------------------------------------------------
        conv = _conv_full_fft_self(a, fft_len_self)          # length 2n‑1
        maxb = conv.max()
        cheap = 2.0 * n * maxb

        # record best cheap score ever seen
        if cheap < best_iter_val - 1e-12:
            best_iter_val = cheap
            best_iter_a = a.copy()
            last_improve = iter_cnt
            print(f"[Iter {iter_cnt:07d}] NEW BEST cheap = {cheap:.6f}")

        # ---- soft‑max weights over conv (smooth approximation of max) ---------
        # increase β linearly with iteration (capped)
        beta = beta_start + (beta_end - beta_start) * min(1.0,
                                                          iter_cnt / 250_000.0)
        # stable soft‑max
        maxc = conv.max()
        w_unnorm = np.exp(beta * (conv - maxc))
        w = w_unnorm / w_unnorm.sum()          # shape (2n‑1,)

        # ---- gradient of smooth_max w.r.t. a (via correlation) --------------
        # correlation w ⋆ a (positive lags 0..n‑1)
        L_corr = fft_len_corr
        fw = np.fft.rfft(w, L_corr)
        fa = np.fft.rfft(a, L_corr)
        corr_full = np.fft.irfft(fw * np.conj(fa), L_corr).real
        # linear correlation length = len(w)+len(a)-1 = 3n‑2
        corr_full = corr_full[: (len(w) + len(a) - 1)]
        offset = len(a) - 1
        corr_pos = corr_full[offset: offset + n]   # shape (n,)
        grad_smooth = 2.0 * corr_pos                # ∂/∂a of Σ_k w_k conv_k

        # ---- gradient of the actual objective (sum(a)=1 → denominator constant)----
        grad = 2.0 * n * grad_smooth                # shape (n,)

        # ---- Adam update ----------------------------------------------------
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** iter_cnt)
        v_hat = v / (1 - beta2 ** iter_cnt)

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step

        # ---- project back onto simplex ---------------------------------------
        a = _simplex_projection(a_new)

        # ---- occasional restart if stagnation --------------------------------
        if iter_cnt - last_improve > 200_000:
            # small random perturbation around the current best
            pert = 0.02 * np.random.rand(n)
            a = _simplex_projection(best_iter_a + pert)
            last_improve = iter_cnt
            print(f"[Iter {iter_cnt:07d}] Restart after stagnation.")

        # occasional logging
        if iter_cnt % 50000 == 0:
            print(f"[Iter {iter_cnt:07d}] cheap = {cheap:.6f}, beta = {beta:.2f}")

    # --------------------------------------------------------------
    # Phase 2 – simple (1+1)‑ES hill‑climbing on the cheap score
    # --------------------------------------------------------------
    sigma = 0.10   # mutation strength in log‑space
    fail_cnt = 0
    while time.time() < deadline - 2.0:
        cand = best_iter_a * np.exp(sigma * np.random.randn(n))
        cand = _simplex_projection(cand)
        val = cheap_score(cand)
        if val < best_iter_val:
            best_iter_val = val
            best_iter_a = cand
            sigma = max(sigma * 0.85, 1e-4)
            fail_cnt = 0
            print(f"[ES] improved cheap = {val:.6f}  sigma={sigma:.5f}")
        else:
            fail_cnt += 1
            if fail_cnt > 200:
                sigma = min(sigma * 1.30, 0.6)
                fail_cnt = 0

    # --------------------------------------------------------------
    # Phase 3 – final exact evaluation
    # --------------------------------------------------------------
    final_seq = best_iter_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:      # defensive – should never happen
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # --------------------------------------------------------------
    # Return the best sequence found.
    # --------------------------------------------------------------
    return final_seq
