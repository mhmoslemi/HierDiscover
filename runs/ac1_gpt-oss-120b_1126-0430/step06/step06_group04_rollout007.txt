def _proj_grad_descent(a, time_limit, rng):
    start = time.time()
    a = np.asarray(a, dtype=float)
    a = a / a.sum()
    n = a.shape[0]

    best_obj = _objective(a)

    while time.time() - start < time_limit:

        k_max = int(np.argmax(conv))
        w = np.zeros_like(conv)
        w[k_max] = 1.0
        grad = 2.0 * _correlation_fft(w, a)
        # Try line search
        improved = False
        for _ in range(12):
            a_candidate = a - eta * grad

"""
Advanced step‑function optimiser.

The routine works in four stages:

1. **Initialisation** – If an earlier construction ``height_sequence_1`` is
   available it is used (normalised to sum = 1); otherwise a quick
   analytical shape search provides a starting vector.

2. **Adam on the simplex** – A soft‑max (log‑it) parametrisation of the
   probability simplex is optimised with an Adam optimiser.  The smooth‑max
   temperature grows from 0.5 to 1e9, giving an excellent warm‑start for the
   exact max‑problem.

3. **Deterministic projected‑gradient descent** – Using the exact
   (hard‑max) gradient a line‑search in the simplex is performed.  The step
   size is adapted by back‑tracking; occasional random jitter avoids stagnation.

4. **Targeted simulated‑annealing + final PG** – A fast SA that moves mass
   from indices contributing heavily to the current maximal autocorrelation
   to random elsewhere, followed by a short but aggressive PG pass.

All phases respect the global time budget (default 1000 s).  The function
``propose_candidate`` returns a plain ``list[float]`` ready for
``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return conv_full[: 2 * n - 1]


def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation of a with weight vector w:
        g[i] = Σ_t w[i+t]·a[t]   for i=0,…,n‑1
    Implemented via FFT (real‑valued).
    """
    n = a.shape[0]
    m = w.shape[0]                     # = 2n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Stable soft‑max."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


def _objective(a: np.ndarray) -> float:
    """
    Exact objective   2·n·max(conv(a,a))  assuming Σ a = 1.
    Returns a Python float.
    """
    n = a.size
    conv = _conv_self_fft(a)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of vector v onto the probability simplex
    {x ≥ 0, Σ x = 1}.
    Implements the O(n log n) algorithm of Duchi et al.
    """
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries non‑positive – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape pre‑search (cheap analytical candidates)
# ----------------------------------------------------------------------


def _shape_candidates(n: int, rng: np.random.Generator):
    """Generate a modest set of analytical shape candidates (sum‑to‑1)."""
    cand = []

    # uniform
    cand.append(np.full(n, 1.0 / n))

    idx = np.arange(n, dtype=float)

    # exponential decays
    for r in np.linspace(0.90, 0.9997, 12):
        a = r ** idx
        a = a / a.sum()
        cand.append(a)
        cand.append(a[::-1])

    # Gaussian windows
    centre = (n - 1) / 2.0
    for sigma in np.linspace(0.10 * n, 0.30 * n, 8):
        a = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
        a = a / a.sum()
        cand.append(a)

    # sparse Dirichlet‑like (few non‑zeros)
    k = max(2, int(0.02 * n))
    for _ in range(4):
        a = np.zeros(n, dtype=float)
        pos = rng.choice(n, size=k, replace=False)
        vals = rng.random(k)
        a[pos] = vals
        a = a / a.sum()
        cand.append(a)

    # pure random Dirichlet
    a = rng.random(n)
    a = a / a.sum()
    cand.append(a)

    return cand


def _best_initial_sequence(n: int, rng: np.random.Generator, time_limit: float):
    """
    Scan the analytical candidates and keep the one with the smallest exact
    objective.  Abort if the allocated time expires.
    """
    start = time.time()
    best_a = None
    best_obj = float("inf")
    for a in _shape_candidates(n, rng):
        if time.time() - start > time_limit:
            break
        obj = _objective(a)
        if obj < best_obj:
            best_obj, best_a = obj, a
    if best_a is None:
        a = rng.random(n)
        best_a = a / a.sum()
        best_obj = _objective(best_a)
    return best_a, best_obj


# ----------------------------------------------------------------------
# Stage 2 – Adam optimiser on the simplex (soft‑max parametrisation)
# ----------------------------------------------------------------------


def _run_opt(n: int, time_limit: float, init_a: np.ndarray | None = None,
             seed: int = 0) -> tuple[list[float], float]:
    """
    Adam optimiser (log‑it) for the smoothed max‑objective.
    Returns the best sequence (as list) and its exact objective.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # initialise
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
        a = a / a.sum()
    else:
        a = rng.random(n)
        a = a / a.sum()

    x = np.log(a + 1e-16)               # logits
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # temperature (smooth‑max) schedule
    beta_start = 0.5
    beta_end = 1e9            # sharper final max
    lr_start = 0.30
    lr_end = 0.0002

    best_a = a.copy()
    best_obj = _objective(best_a)

    it = 0
    while time.time() - start < time_limit:
        it += 1
        a = _softmax(x)

        conv = _conv_self_fft(a)
        M = conv.max()
        obj = float(2.0 * n * M)

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[Adam n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        # schedule (time‑based)
        t = min((time.time() - start) / time_limit, 1.0)
        beta = beta_start * np.exp(np.log(beta_end / beta_start) * t)
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t)

        # smooth‑max weights
        if beta > 5e5:
            k_max = int(np.argmax(conv))
            w = np.zeros_like(conv)
            w[k_max] = 1.0
        else:
            shifted = conv - M               # ≤0
            w = np.exp(beta * shifted)
            s = w.sum()
            if s == 0.0:
                w = np.ones_like(conv) / conv.size
            else:
                w /= s

        # gradient of (smooth) max w.r.t. a
        g_a = 2.0 * _correlation_fft(w, a)

        # natural gradient in log‑it space
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # small jitter to shake out of plateaus
        if it % 2500 == 0:
            jitter = rng.normal(scale=0.012, size=n)
            x += jitter

        # periodic restart from the current best
        if it % 60000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.008, size=n)
            m.fill(0.0)
            v.fill(0.0)

    # compress very small entries (helps later steps)
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = _objective(best_a)
        print(f"[compress] length → {best_a.size}, O={best_obj:.6f}")

    # cheap local moves (optional, cheap hill‑climb)
    remaining = time_limit - (time.time() - start)
    if remaining > 1.0:
        a_loc = best_a.copy()
        obj_loc = best_obj
        n_loc = a_loc.size
        while time.time() - start < time.time() + remaining * 0.3:
            i, j = rng.choice(n_loc, size=2, replace=False)
            delta = a_loc[i] * rng.random() * 0.025
            if delta <= 0.0:
                continue
            a_new = a_loc.copy()
            a_new[i] -= delta
            a_new[j] += delta
            obj_new = _objective(a_new)
            if obj_new < obj_loc:
                a_loc, obj_loc = a_new, obj_new
                if obj_new < best_obj:
                    best_a, best_obj = a_new, obj_new
        print(f"[Adam‑local] post‑Adam O={best_obj:.6f}")

    return best_a.tolist(), best_obj


# ----------------------------------------------------------------------
# Stage 3 – Deterministic projected‑gradient descent (exact max)
# ----------------------------------------------------------------------


def _proj_grad_descent(a0: np.ndarray, time_limit: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Projected‑gradient descent using the exact (hard) max‑gradient.
    A back‑tracking line search adapts the step size; occasional
    jitter prevents stagnation.
    Returns the best vector found and its exact objective.
    """
    start = time.time()
    a = np.asarray(a0, dtype=float)
    a = a / a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_obj = _objective(a)
    eta = 0.5
    it = 0

    while time.time() - start < time_limit:
        it += 1
        conv = _conv_self_fft(a)
        k_max = int(np.argmax(conv))
        w = np.zeros_like(conv)
        w[k_max] = 1.0
        grad = 2.0 * _correlation_fft(w, a)          # gradient of max(conv)

        improved = False
        # try a few back‑tracking steps
        for _ in range(9):
            a_candidate = a - eta * grad
            a_candidate = _project_simplex(a_candidate)
            obj_candidate = _objective(a_candidate)
            if obj_candidate < best_obj - 1e-14:
                a = a_candidate
                best_a = a_candidate.copy()
                best_obj = obj_candidate
                eta = min(eta * 1.2, 2.0)   # modestly increase after success
                improved = True
                # progress report every few thousand iterations
                if it % 5000 == 0:
                    print(f"[PG it={it}] improved O={best_obj:.6f}")
                break
            else:
                eta *= 0.5
        if not improved:
            # add a tiny random jitter and reset step size
            a += rng.normal(scale=0.0015, size=n)
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            eta = 0.5

    return best_a, best_obj


# ----------------------------------------------------------------------
# Stage 4 – Targeted simulated annealing + final PG
# ----------------------------------------------------------------------


def _run_sa(init_seq: list[float], time_limit: float,
            seed: int = 0) -> tuple[list[float], float]:
    """
    Fast targeted simulated‑annealing followed by a short PG pass.
    Returns the best sequence (as list) and its exact objective.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    a = np.asarray(init_seq, dtype=float)
    if a.sum() < 1e-12:
        a = np.ones_like(a) / a.size
    else:
        a = a / a.sum()
    n = a.shape[0]

    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())
    best_seq = a.copy()

    # SA schedule (slow cooling)
    T = best_obj * 0.25 + 1e-9
    T_min = best_obj * 5e-7 + 1e-12
    decay = 0.998

    it = 0
    while time.time() - start < time_limit * 0.65:
        it += 1
        k_max = int(np.argmax(conv))

        # contributions to the current max
        i_vals = np.arange(n)
        j_vals = k_max - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i_vals[mask]] * a[j_vals[mask]]
        total_contrib = contrib.sum()

        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = rng.choice(n, p=probs)
        else:
            i = rng.integers(n)

        # pick a destination distinct from i
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # move a modest fraction of mass
        delta = a[i] * rng.random() * 0.09   # up to 9 %
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.clip(a_new, 0.0, None)
        a_new /= a_new.sum()

        conv_new = _conv_self_fft(a_new)
        obj_new = float(2.0 * n * conv_new.max())

        if obj_new < best_obj - 1e-14:
            a, conv, best_obj, best_seq = a_new, conv_new, obj_new, a_new.copy()
            print(f"[SA it={it}] new best O={best_obj:.6f}")
        else:
            # Metropolis acceptance
            if rng.random() < math.exp(-(obj_new - best_obj) / T):
                a, conv = a_new, conv_new

        if it % 3000 == 0:
            T = max(T * decay, T_min)

        if it % 9000 == 0:
            # global jitter
            noise = rng.normal(scale=0.009, size=n)
            a = np.clip(a + noise, 0.0, None)
            a /= a.sum()
            conv = _conv_self_fft(a)

    # -----------------------------------------------------------------
    # final projected‑gradient pass (exact max)
    remaining = time_limit - (time.time() - start)
    if remaining > 0.3:
        a = best_seq.copy()
        a, best_obj = _proj_grad_descent(a, remaining, rng)
        best_seq = a.copy()
        best_obj = float(2.0 * n * _conv_self_fft(a).max())
        print(f"[SA‑PG] final O={best_obj:.6f}")

    return best_seq.tolist(), best_obj

def propose_candidate(seed: int = 42, budget_s: int = 1000, **_ignore) -> list[float]:
    """
    Optimise a non‑negative step‑function under the evaluation metric.
    Returns a list suitable for ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1️⃣  Initialise – use a historic construction if present
    try:
        init = np.array(height_sequence_1, dtype=float)  # type: ignore
        if init.size == 0:
            raise ValueError
        init = np.clip(init, 0.0, None)
        if init.sum() < 1e-12:
            init = np.ones_like(init) / init.size
        else:
            init = init / init.sum()
        print(f"[init] using historic sequence (len={init.size})")
    except Exception:
        init = None
        print("[init] no historic sequence – will generate a shape")

    # ------------------------------------------------------------------
    # 2️⃣  Choose a support length – longer supports give more freedom.
    #     We explore three lengths, giving the first (smallest) most time.
    lengths = [2000, 4000, 8000]

    best_seq = None
    best_obj = float("inf")

    for idx, n in enumerate(lengths):
        now = time.time()
        remaining = deadline - now
        if remaining <= 4.0:
            break

        # allocate time: first length gets the bulk, later ones share the rest
        if idx == 0:
            alloc = remaining * 0.55
        else:
            alloc = (remaining * 0.35) / (len(lengths) - idx)

        # ------------------------------------------------------------------
        # 2a – initial shape (fast)
        shape_time = min(4.0, alloc * 0.08)
        if init is not None and init.size == n:
            init_a = init.copy()
            init_obj = _objective(init_a)
        else:
            init_a, init_obj = _best_initial_sequence(n, rng, shape_time)
        print(f"[n={n}] shape‑init O={init_obj:.6f}")

        # ------------------------------------------------------------------
        # 2b – Adam optimisation
        a_adam, obj_adam = _run_opt(
            n=n,
            time_limit=alloc * 0.55,
            init_a=init_a,
            seed=seed + idx * 12345,
        )
        if obj_adam < best_obj:
            best_obj, best_seq = obj_adam, np.asarray(a_adam, dtype=float)
            print(f"[global] new overall best O={best_obj:.6f} (n={n})")

        # ------------------------------------------------------------------
        # 2c – deterministic projected‑gradient descent (exact max)
        pg_time = alloc * 0.25
        a_pg, obj_pg = _proj_grad_descent(best_seq, pg_time, rng)
        if obj_pg < best_obj:
            best_obj, best_seq = obj_pg, a_pg.copy()
            print(f"[global] PG improved O={best_obj:.6f} (n={n})")

        # ------------------------------------------------------------------
        # 2d – optional quick SA (if some budget left)
        sa_time = alloc - (pg_time + (alloc * 0.55))
        if sa_time > 5.0:
            seq_sa, obj_sa = _run_sa(best_seq.tolist(), sa_time,
                                     seed=seed + idx * 54321)
            if obj_sa < best_obj:
                best_obj, best_seq = obj_sa, np.asarray(seq_sa, dtype=float)
                print(f"[global] SA improved O={best_obj:.6f} (n={n})")

        # break early if we are already very close to target
        if best_obj <= 1.503 and (deadline - time.time()) > 5.0:
            # give a little extra polishing before exiting
            extra = min(30.0, deadline - time.time() - 0.5)
            if extra > 0:
                best_seq, best_obj = _proj_grad_descent(best_seq, extra, rng)
                print(f"[final‑polish] O={best_obj:.6f}")
            break

    # ------------------------------------------------------------------
    # 3️⃣  Final polishing if any time left (short SA + PG)
    remaining = deadline - time.time()
    if best_seq is not None and remaining > 6.0:
        seq_ref, val_ref = _run_sa(best_seq.tolist(), remaining,
                                   seed=seed + 999999)
        if val_ref < best_obj:
            best_obj, best_seq = val_ref, np.asarray(seq_ref, dtype=float)
            print(f"[final] after SA+PG O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Sanity checks and final return
    if best_seq is None or best_seq.size == 0:
        best_seq = np.ones(200, dtype=float) / 200.0
        best_obj = _objective(best_seq)

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    if best_seq.sum() < 0.01:
        best_seq = np.ones_like(best_seq) / best_seq.size

    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_seq.tolist()
