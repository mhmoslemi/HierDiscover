#!/usr/bin/env python
"""
Non‑negative step‑function optimiser – sub‑gradient / top‑K smooth version.

The algorithm improves the previous Adam‑soft‑max optimiser by
* using a cheap sub‑gradient of the (smoothed) max‑convolution,
* restricting the smoothing to the few largest convolution entries
  (top‑K soft‑max) – this keeps the per‑iteration cost low,
* performing mirror‑descent on the simplex via logits and Adam,
* adding several restarts, jitter phases and a single up‑sampling stage.

The routine respects the 1000 s time budget and returns the best
sequence found (list of non‑negative heights).
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures or lambdas
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max, returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    s = e.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / z.size)
    return e / s


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-15) -> np.ndarray:
    """Log‑it that reproduces `a` after a soft‑max (adds tiny epsilon)."""
    return np.log(a + eps)


def _fft_len_for_conv(n: int) -> int:
    """Smallest power‑of‑two ≥ 2·n‑1 (enough for linear self‑convolution)."""
    return 1 << ((2 * n - 2).bit_length())


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    s = new_a.sum()
    if s == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= s
    return new_a


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{‑p}, normalised."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _sym_product_power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ ((i+1)*(n‑i))^{‑p}, symmetric, normalised."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) * (n - i)) ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _exp_decay_distribution(n: int, lam: float) -> np.ndarray:
    """a_i ∝ e^{‑lam·i}+e^{‑lam·(n‑1‑i)}, symmetric exponential."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _logistic_distribution(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ 1/(1+e^{α·(i‑mid)}), normalised; heavy on the left side."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = 1.0 / (1.0 + np.exp(alpha * (i - mid)))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _triangular_distribution(n: int) -> np.ndarray:
    """Symmetric triangle, peak at the centre, normalised."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _conv_and_topk(a: np.ndarray, topk: int = 5):
    """
    Compute the full linear self‑convolution a*a via FFT,
    return (conv, max_val, argmax, top_indices, top_values).
    """
    n = a.shape[0]
    L = _fft_len_for_conv(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_val = float(conv.max())
    argmax = int(conv.argmax())

    if topk >= conv.size:
        idxs = np.arange(conv.size)
    else:
        idxs = np.argpartition(conv, -topk)[-topk:]
    top_vals = conv[idxs]

    return conv, max_val, argmax, idxs, top_vals


def _smooth_grad_topk(a: np.ndarray, conv: np.ndarray,
                      idxs: np.ndarray, top_vals: np.ndarray,
                      eps_factor: float = 0.15) -> np.ndarray:
    """
    Compute a (sub‑)gradient of the smoothed max‑convolution.
    Only the entries in `idxs` (the top‑k indices) receive non‑zero weight.
    """
    n = a.shape[0]
    max_val = float(conv.max())
    eps = max(max_val * eps_factor, 1e-12)

    # soft‑max on the selected top‑k values
    shifted = top_vals - max_val               # ≤ 0
    w_raw = np.exp(shifted / eps)
    w_sum = w_raw.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w_raw, 1.0 / w_raw.size)
    else:
        w = w_raw / w_sum

    # gradient = 4·n·(a ⋆ w)  where '⋆' denotes correlation.
    grad = np.zeros_like(a)
    for wk, s in zip(w, idxs):
        # indices of a that contribute to grad for this shift
        start = max(0, s - (n - 1))
        end = min(n - 1, s)
        # j runs from start .. end
        j = np.arange(start, end + 1, dtype=np.intp)
        # corresponding a indices are s - j
        a_slice = a[s - j]
        grad[j] += wk * a_slice

    grad *= 4.0 * n
    return grad
