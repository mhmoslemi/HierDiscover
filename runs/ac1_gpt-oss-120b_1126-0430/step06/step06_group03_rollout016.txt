    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]

def _subgradient_max(a):
    # compute convex subgradient at argmax positions.
    n = a.shape[0]
    conv, max_val = _conv_and_max(a)
    # tolerance for floating equality
    tol = 1e-12 * max_val if max_val != 0 else 1e-16
    active = np.where(conv >= max_val - tol)[0]

        idx = np.arange(lo, hi+1)
        idx2 = k - idx
        # add contributions a[idx2] + a[idx]

def _biased_random_move(a, grad, delta_factor=0.001):
    n = a.shape[0]
    # top 10% indices
    top_k = max(1, int(n * 0.01))
    low_k = max(1, int(n * 0.01))
    # compute sorted indices
    sorted_desc = np.argsort(-grad)  # descending
    sorted_asc = np.argsort(grad)    # ascending
    i = sorted_desc[np.random.randint(top_k)]
    j = sorted_asc[np.random.randint(low_k)]
    if i == j:
        return None, None
    delta = delta_factor * a[i]
    if delta == 0.0:
        return None, None
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return a_new, delta

"""
Hybrid optimizer for the self‑convolution bound.

The routine follows three phases:

1️⃣  **Coarse parametric sweep** – a handful of analytic families (power‑law,
    geometric, stretched exponential, …) are sampled on a grid.  The cheap
    objective `2·n·max(a*a)` (with ∑a=1) is used to pick a good start.

2️⃣  **Adam mirror‑descent on the simplex** – the distribution is represented
    by logits `z` (soft‑max gives a probability vector).  A smooth‐max
    approximation of the convolution peak is differentiated analytically
    (via FFT) and Adam updates are applied with a back‑tracking line‑search.
    The schedule for the smoothing constant `eps` as well as for the step
    size is deliberately aggressive, allowing many > 300 k iterations within
    the 1000 s budget.

3️⃣  **Gradient‑biased random walk** – the exact sub‑gradient of the *true*
    max‑convolution is inexpensive to compute (only the active indices are
    needed).  Mass is transferred from coordinates with large sub‑gradient
    (they contribute most to the current peak) to those with small
    sub‑gradient.  Only moves that improve the cheap score are kept.

Finally the best candidate is evaluated with the provided `evaluate_sequence`
function and returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max → probability vector (sum ≈ 1)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    s = e.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / z.size)
    return e / s


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑its reproducing `a` after a soft‑max."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """Smallest power‑of‑two ≥ 3·n‑2 (enough for all needed convolutions)."""
    return 1 << ((3 * n - 2).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Returns (max_conv, grad_a) where

        grad_a = ∇_a [ 2·n·smooth_max(a*a) ]

    `eps_factor` multiplies the smoothing constant ε = eps_factor·max_conv.
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv                     # ≤ 0 for stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # Gradient of 2·n·smooth_max (the factor 4·n comes from derivative of a*a)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a


def _conv_and_full(a: np.ndarray):
    """Return (full convolution array, maximum value)."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return conv, float(conv.max())


def _subgradient_max(a: np.ndarray):
    """
    Exact sub‑gradient of  max_k (a*a)[k]  w.r.t. `a`.
    For each active index k (where the maximum is attained) the derivative is

        ∂/∂a_j  (a*a)[k] = a_{k‑j} + a_j   (if the index is in range).

    The returned vector is the sum of these contributions over all active k.
    """
    n = a.shape[0]
    conv, max_val = _conv_and_full(a)

    # indices where the maximum is attained (tolerance for floating noise)
    tol = 1e-12 * max_val if max_val != 0 else 1e-16
    active = np.where(conv >= max_val - tol)[0]

    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        idx2 = k - idx
        # contribution a_{k‑i} + a_i for every i in [lo, hi]
        grad[idx] += a[idx2] + a[idx]
    return grad


# ----------------------------------------------------------------------
# Parametric families of distributions
# ----------------------------------------------------------------------


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}   (normalised)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _power_law_offset_distribution(n: int, p: float, offset: float) -> np.ndarray:
    """a_i ∝ (i+offset)^{-p}   (normalised)."""
    idx = np.arange(offset, offset + n, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _double_power_law_distribution(n: int, p: float, q: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}·(n‑i)^{-q}   (normalised)."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) ** (-p)) * ((n - i) ** (-q))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _stretched_exp_distribution(n: int, tau_factor: float, gamma: float) -> np.ndarray:
    """a_i ∝ exp( - (i/(tau·n))^{gamma} )   (normalised)."""
    tau = max(1e-12, tau_factor * n)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(- (i / tau) ** gamma)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _exp_decay_distribution(n: int, lam: float) -> np.ndarray:
    """a_i ∝ e^{-lam·i}+e^{-lam·(n‑1‑i)}   (normalised)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _geom_one_sided_distribution(n: int, lam: float) -> np.ndarray:
    """a_i ∝ e^{-lam·i}   (normalised, not symmetric)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _logistic_distribution(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ 1/(1+e^{α·(i‑mid)})   (normalised)."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = 1.0 / (1.0 + np.exp(alpha * (i - mid)))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _sym_product_power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ ((i+1)*(n‑i))^{-p}   (symmetric, normalised)."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) * (n - i)) ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) improves the
    `evaluate_sequence` score.  The routine respects the supplied time budget
    (~1000 s) and uses a hybrid grid‑search + Adam optimiser on the probability
    simplex, followed by a cheap gradient‑biased random walk.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0   # a short safety margin

    # ------------------------------------------------------------------
    # 0. Problem size and fallback initial distribution
    # ------------------------------------------------------------------
    try:
        init_seq = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        n = max(20000, init_seq.shape[0])
        a0 = np.maximum(init_seq, 0.0)
        s = a0.sum()
        if s == 0.0:
            a0 = np.full_like(a0, 1.0 / a0.shape[0])
        else:
            a0 /= s
        print(f"[Init] Using provided sequence, length = {n}")
    except Exception:
        n = 25000
        a0 = np.full(n, 1.0 / n)
        print(f"[Init] No provided sequence – start from uniform, length = {n}")

    best_seq = a0.copy()
    best_score = 2.0 * n * _conv_max(best_seq)

    # ------------------------------------------------------------------
    # 1. Coarse parametric sweep
    # ------------------------------------------------------------------
    families = [
        ("power_law", _power_law_distribution,
         [dict(p=p) for p in np.linspace(0.1, 3.0, 30)]),
        ("power_law_offset", _power_law_offset_distribution,
         [dict(p=p, offset=off)
          for p in np.linspace(0.2, 2.0, 15)
          for off in np.linspace(0.5, 5.0, 15)]),
        ("double_power_law", _double_power_law_distribution,
         [dict(p=p, q=q)
          for p in np.linspace(0.4, 2.0, 15)
          for q in np.linspace(0.4, 2.0, 15)]),
        ("stretched_exp", _stretched_exp_distribution,
         [dict(tau_factor=t, gamma=g)
          for t in np.linspace(0.1, 0.9, 15)
          for g in np.linspace(0.5, 2.5, 15)]),
        ("exp_decay", _exp_decay_distribution,
         [dict(lam=lam) for lam in np.logspace(-7, -2, 30)]),
        ("geom_one_sided", _geom_one_sided_distribution,
         [dict(lam=lam) for lam in np.logspace(-7, -2, 30)]),
        ("logistic", _logistic_distribution,
         [dict(alpha=alpha) for alpha in np.linspace(0.01, 6.0, 30)]),
        ("sym_product_power_law", _sym_product_power_law_distribution,
         [dict(p=p) for p in np.linspace(0.4, 1.5, 30)]),
    ]

    print("[Sweep] Starting parametric search")
    for name, generator, param_dicts in families:
        for params in param_dicts:
            if time.time() >= deadline:
                break
            try:
                cand = generator(n, **params)
            except Exception:
                continue
            score = 2.0 * n * _conv_max(cand)
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
        if time.time() >= deadline:
            break

    print(f"[Sweep] Best cheap score after sweep = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Mirror‑descent (Adam) on the simplex
    # ------------------------------------------------------------------
    logits = _logits_from_distribution(best_seq) + 0.02 * np.random.randn(n)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    base_alpha = 0.55      # larger base step for faster progress
    eps_factor = 0.20      # start with a relatively smooth max
    eps_factor_min = 5e-6
    eps_decay = 0.9999

    report_every = 50000
    it = 0
    last_improve = 0
    best_seq_sofar = best_seq.copy()
    best_score_sofar = best_score

    max_iters = 800_000  # generous upper bound – the time limit will stop us earlier

    while time.time() < deadline and it < max_iters:
        it += 1

        a = _softmax(logits)         # probability vector, sum ≈ 1
        max_conv, grad_a = _conv_max_and_grad(a, eps_factor)
        cheap_score = 2.0 * n * max_conv

        # retain the best ever seen
        if cheap_score < best_score_sofar - 1e-14:
            best_score_sofar = cheap_score
            best_seq_sofar = a.copy()
            last_improve = it
            if it % report_every == 0:
                print(f"[Iter {it:07d}] NEW BEST cheap = {cheap_score:.8f}")

        # ---- mirror‑descent step on logits (Adam) ----
        mu = np.dot(grad_a, a)
        grad_z = a * (grad_a - mu)

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step = - m_hat / (np.sqrt(v_hat) + eps_adam)

        # back‑tracking line‑search (up to 6 halvings)
        accepted = False
        trial_alpha = alpha_t
        for _ in range(6):
            new_logits = logits + trial_alpha * step
            new_logits = np.clip(new_logits, -50.0, 50.0)
            a_new = _softmax(new_logits)
            new_score = 2.0 * n * _conv_max(a_new)
            if new_score <= cheap_score:
                logits = new_logits
                accepted = True
                break
            trial_alpha *= 0.5
        if not accepted:
            logits += alpha_t * step
            logits = np.clip(logits, -50.0, 50.0)

        # occasional jitter to escape plateaus
        if it % 3500 == 0 and np.random.rand() < 0.12:
            logits += 0.02 * np.random.randn(n)
            m *= 0.6
            v *= 0.6

        # decay the smoothing constant
        eps_factor = max(eps_factor_min, eps_factor * eps_decay)

        # restart if stagnating for a long time
        if it - last_improve > 300_000:
            print(f"[Iter {it}] Stagnation – restarting from best so far")
            logits = _logits_from_distribution(best_seq_sofar) + 0.03 * np.random.randn(n)
            m.fill(0.0)
            v.fill(0.0)
            eps_factor = 0.20
            last_improve = it

    best_seq = best_seq_sofar
    best_score = best_score_sofar
    print(f"[Adam] Finished after {it} iterations, best cheap = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Gradient‑biased cheap random walk (coordinate moves)
    # ------------------------------------------------------------------
    a = best_seq.copy()
    cheap_score = best_score

    while time.time() < deadline:
        # compute exact sub‑gradient of the (non‑smoothed) max
        grad = _subgradient_max(a)
        # pick source index among the 2 % largest gradient components
        top_k = max(1, int(0.02 * n))
        src_candidates = np.argpartition(-grad, top_k - 1)[:top_k]
        # pick destination among the 2 % smallest gradient components
        bot_k = max(1, int(0.02 * n))
        dst_candidates = np.argpartition(grad, bot_k - 1)[:bot_k]

        i = np.random.choice(src_candidates)
        j = np.random.choice(dst_candidates)
        if i == j:
            continue
        delta = 0.001 * a[i]          # move 0.1 % of the source mass
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = 2.0 * n * _conv_max(a_new)
        if new_score < cheap_score:
            a = a_new
            cheap_score = new_score
            best_seq = a.copy()
            best_score = cheap_score
            # very sparse progress output
            if np.random.rand() < 0.001:
                print(f"[Walk] Improved cheap = {cheap_score:.8f}")

    # ------------------------------------------------------------------
    # 4. Final exact evaluation with evaluate_sequence()
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        exact_score = evaluate_sequence(final_seq)  # provided by the platform
        print(f"Final exact score = {exact_score:.6f}")
    except Exception as exc:
        exact_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    return final_seq
