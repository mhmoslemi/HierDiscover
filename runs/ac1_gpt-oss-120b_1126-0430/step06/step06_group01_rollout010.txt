def propose_candidate(seed=42, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2

    # Determine base length if external variable exists
    try:
        base_n = len(height_sequence_1)
    except Exception:
        base_n = 20000

    target_score = 1.503

    best_score = float('inf')
    best_seq = None

    n = base_n
    # Set max n limit maybe 200k
    max_n = 200000

    while time.time() < deadline and n <= max_n:
        # generate arcsine distribution for n
        i = np.arange(n, dtype=np.float64)
        x0 = i / n
        x1 = (i + 1) / n
        sqrt0 = np.sqrt(np.clip(x0, 0.0, 1.0))
        sqrt1 = np.sqrt(np.clip(x1, 0.0, 1.0))
        F0 = (2.0 / np.pi) * np.arcsin(sqrt0)
        F1 = (2.0 / np.pi) * np.arcsin(sqrt1)
        a = F1 - F0
        a_sum = a.sum()
        if a_sum <= 0: 
            raise RuntimeError
        a /= a_sum

        # fast evaluation
        max_conv = _conv_max(a)
        score = 2.0 * n * max_conv
        print(f"Arcsine n={n} score approx {score:.8f}")

        # if this is best, store
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"New best score {best_score:.8f}")

        # early break if below target margin
        if best_score <= target_score:
            print(f"Target reached with n={n}")
            break

        # increase n: ratio maybe 1.5 or add 5000 until within budget
        # Let's try multiplicative increase of 1.3 and round to int.
        n = int(n * 1.3) + 1 # ensure increase

    # Optionally refine best_seq with gradient descent if time left
    if best_seq is not None and time.time() < deadline:
        # refine using Adam on logits for limited iterations
        a = best_seq
        n = a.shape[0]
        # initialize logits from a
        z = _logits_from_distribution(a)
        m = np.zeros_like(z)
        v = np.zeros_like(z)
        beta1, beta2 = 0.9, 0.999
        eps_adam = 1e-8
        base_alpha = 0.6
        eps_factor = 0.12
        eps_factor_min = 5e-4
        eps_factor_decay = 0.9995

        it = 0
        max_iters = 100000
        while time.time() < deadline and it < max_iters:
            it += 1
            a_cur = _softmax(z)
            max_conv, grad_a = _conv_max_and_grad(a_cur, eps_factor)
            cheap_score = 2.0 * n * max_conv

            if cheap_score < best_score:
                best_score = cheap_score
                best_seq = a_cur.copy()
                if it % 1000 == 0:
                    print(f"Iter {it} improved score {best_score:.8f}")

            # compute gradient in logits space
            mu = np.dot(grad_a, a_cur)
            grad_z = a_cur * (grad_a - mu)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            alpha_t = base_alpha / (math.sqrt(it) + 1.0)
            step = - alpha_t * m_hat / (np.sqrt(v_hat) + eps_adam)
            z = z + step
            z = np.clip(z, -70.0, 70.0)

            eps_factor = max(eps_factor_min, eps_factor * eps_factor_decay)

#!/usr/bin/env python
"""
Fast arcsine‑bin construction with optional Adam polishing.

The continuous arcsine density f(x)=1/(π√(x(1−x))) satisfies (f*f)(y)=½
on [0,2].  Discretising its CDF on a uniform grid gives a probability
vector a whose self‑convolution is almost constant, so the evaluation
metric 2·n·max(a*a)/(∑a)² ≈ 1.  For modest n (≈2·10⁴) the discretisation
error makes the score ≈1.51, but the error decays like O(1/√n).  By
gradually increasing n we can drive the metric under the target 1.503
even without any optimisation.  When time permits we finish with a
short Adam run on the logits (soft‑max) representation to shave a few
extra digits.

The function returns a plain Python list of non‑negative heights that
minimises the provided ``evaluate_sequence`` metric within the 1000 s
budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top level, no closures
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Logits that map to the probability vector a via soft‑max."""
    return np.log(a + eps)


def _fft_len_for_conv(n: int) -> int:
    """
    Smallest power‑of‑two ≥ 2·n‑1,
    sufficient for a linear self‑convolution of length (2·n‑1).
    """
    return 1 << ((2 * n - 1).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a (FFT, no gradient)."""
    n = a.shape[0]
    L = _fft_len_for_conv(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Smoothed maximum of a*a and its gradient w.r.t. a.

    Parameters
    ----------
    a : np.ndarray
        Probability vector (non‑negative, sum = 1).
    eps_factor : float
        Multiplicative factor for the smoothing constant:
        eps = eps_factor * max(conv).

    Returns
    -------
    max_conv : float
        Maximum entry of the self‑convolution.
    grad_a : np.ndarray
        Gradient of 4·n·smooth_max w.r.t. a (already multiplied by 4·n).
    """
    n = a.shape[0]
    L = _fft_len_for_conv(n)

    a_fft = np.fft.rfft(a, L)

    # linear self‑convolution (length 2·n‑1)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # smoothed soft‑max weights (log‑sum‑exp trick)
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum

    # cross‑correlation a ⋆ w (linear, length 3·n‑2, but we need only first n)
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # gradient (4·n multiplied)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, sum x_i = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Input must be 1‑D")
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # find rho
    rho = np.nonzero(u - cssv / ind > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimises
    the evaluation metric.  The algorithm proceeds in three phases:

    1. Pick a length n (starting from the supplied external length or a
       default).  Generate the exact arcsine bin probabilities for this n.
       If the cheap FFT‑based score is still above the target we increase n
       (geometric growth) until the score drops below a safe margin.

    2. If time permits we run a short Adam optimisation on the logits
       (soft‑max parameterisation) to polish the distribution further.

    3. The best sequence observed is returned as a plain Python list.
    """
    np.random.seed(seed)
    start = time.time()
    # keep a small safety margin for final logging
    deadline = start + budget_s - 2.0

    # ------------------------------------------------------------------
    # 0. Choose an initial length.
    # ------------------------------------------------------------------

        print(f"[init] Using external length n = {n}")
    except Exception:
        n = 20000
        print(f"[init] No external sequence – using default length n = {n}")

    # ------------------------------------------------------------------
    # 1. Search for a suitable n with a cheap FFT‑based score.
    # ------------------------------------------------------------------
    target_score = 1.503        # we aim to be comfortably below this
    safety_margin = 0.01        # stop when we are at least this far below
    best_score = float("inf")
    best_seq = None
    max_n = 200000              # hard upper bound to keep evaluation cheap

    while time.time() < deadline and n <= max_n:
        # --- exact arcsine discretisation --------------------------------
        i = np.arange(n, dtype=np.float64)
        x0 = i / n
        x1 = (i + 1) / n

        sqrt0 = np.sqrt(np.clip(x0, 0.0, 1.0))
        sqrt1 = np.sqrt(np.clip(x1, 0.0, 1.0))
        F0 = (2.0 / np.pi) * np.arcsin(sqrt0)
        F1 = (2.0 / np.pi) * np.arcsin(sqrt1)

        a = F1 - F0               # bin probabilities, sum ≈ 1
        a_sum = a.sum()
        if not np.isfinite(a_sum) or a_sum <= 0.0:
            raise RuntimeError("Failed to generate arcsine bins.")
        a /= a_sum                # enforce exact normalisation

        # --- cheap score using FFT (no clipping, fast) ------------------
        max_conv = _conv_max(a)
        cheap_score = 2.0 * n * max_conv
        print(f"[n={n}] cheap score ≈ {cheap_score:.8f}")

        if cheap_score < best_score:
            best_score = cheap_score
            best_seq = a.copy()
            print(f"   New best score {best_score:.8f}")

        # stop condition: comfortably below target
        if best_score <= target_score - safety_margin:
            print("[search] Desired score reached, exiting size search.")
            break

        # increase n geometrically – keep it a multiple of 1000 for nicer logs
        n = int(n * 1.35) + 1
        n = (n // 1000) * 1000  # round down to nearest 1000
        if n <= 0:
            n = 1000

    if best_seq is None:
        raise RuntimeError("Failed to generate any candidate sequence.")

    # ------------------------------------------------------------------
    # 2. Optional short Adam polishing on the logits (if time permits).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a = best_seq
        n = a.shape[0]
        print(f"[polish] Starting Adam polishing for n={n}")

        # initialise logits from current distribution
        z = _logits_from_distribution(a)
        m = np.zeros_like(z)
        v = np.zeros_like(z)
        beta1, beta2 = 0.9, 0.999
        eps_adam = 1e-8
        base_alpha = 0.6
        eps_factor = 0.12           # smoothing constant multiplier
        eps_factor_min = 5e-4
        eps_factor_decay = 0.9995

        it = 0
        max_iters = 120000          # generous bound, will be cut by deadline

        while time.time() < deadline and it < max_iters:
            it += 1
            a_cur = _softmax(z)

            max_conv, grad_a = _conv_max_and_grad(a_cur, eps_factor)
            cheap_score = 2.0 * n * max_conv

            if cheap_score < best_score:
                best_score = cheap_score
                best_seq = a_cur.copy()
                if it % 2000 == 0 or cheap_score < target_score:
                    print(f"[adam] iter {it:6d} improved score = {best_score:.8f}")

            # Mirror‑descent gradient on the logits
            mu = np.dot(grad_a, a_cur)
            grad_z = a_cur * (grad_a - mu)

            # Adam bias‑corrected update
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            # step‑size schedule (inverse‑sqrt)
            alpha_t = base_alpha / (math.sqrt(it) + 1.0)
            step = - alpha_t * m_hat / (np.sqrt(v_hat) + eps_adam)
            z = z + step
            # keep logits bounded – prevents overflow in exp()
            z = np.clip(z, -70.0, 70.0)

            # tighten smoothing
            eps_factor = max(eps_factor_min, eps_factor * eps_factor_decay)

            # occasional jitter to escape plateaus
            if it % 5000 == 0 and np.random.rand() < 0.07:
                z += 0.004 * np.random.randn(n)

        print(f"[polish] Adam finished – best cheap score {best_score:.8f}")

    # ------------------------------------------------------------------
    # 3. Return the best sequence as a plain Python list.
    # ------------------------------------------------------------------

        print(f"[final] Exact evaluation score = {exact_score:.8f}")
    except Exception as exc:
        print(f"[final] Exact evaluation failed: {exc}")

    return final_seq
