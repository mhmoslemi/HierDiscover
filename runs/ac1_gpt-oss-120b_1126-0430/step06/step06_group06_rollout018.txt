#!/usr/bin/env python
"""
Propose a non‑negative step function (list of heights) that minimises
the supplied ``evaluate_sequence`` objective.

The algorithm consists of three phases:

1. **Broad parametric sweep** – generate many candidate vectors from
   families (uniform, power‑law, exponential, mixed, beta, linear,
   Gaussian, mixtures of exponentials) and keep the best cheap score
   (2·n·max(conv) with Σa=1).

2. **Projected Adam mirror‑descent** – optimise the true nonsmooth
   objective on the probability simplex.  The update uses the
   sub‑gradient of the current maximal convolution entry and an Adam‑style
   adaptive step size.  After each step the vector is re‑normalised onto
   the simplex (no O(n log n) projection needed).

3. **(1+1)‑ES hill‑climbing** – multiplicative Gaussian mutations in
   log‑space with an adaptive mutation strength σ, projected back onto
   the simplex.  This refines the solution locally.

All phases respect the 1000 s budget and always return a feasible
(non‑negative, Σa_i > 0.01) sequence.  The routine prints progress
information and the final exact score.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex utilities – O(n) normalisation (no sorting)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex (fallback – O(n log n))."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _renorm_to_simplex(a: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the simplex by simple normalisation."""
    s = a.sum()
    if s <= 0.0 or not np.isfinite(s):
        # fall back to uniform distribution
        return np.full_like(a, 1.0 / a.size)
    return a / s


# ----------------------------------------------------------------------
# Fast convolution utilities (real FFT)
# ----------------------------------------------------------------------


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length = 2·n‑1) using real FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len).real[: 2 * a.size - 1]
    return conv


def _conv_max_fft(a: np.ndarray, fft_len: int) -> tuple[float, int]:
    """Maximum coefficient of a*a and its index."""
    conv = _conv_full_fft(a, fft_len)
    idx = int(conv.argmax())
    return float(conv[idx]), idx


# ----------------------------------------------------------------------
# Parametric families (all return a normalised probability vector)
# ----------------------------------------------------------------------


def _uniform_seq(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _power_law_seq(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = i ** (-p)
    s = seq.sum()
    if s == 0.0:
        return None
    seq /= s
    return seq


def _exponential_seq(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    seq = np.exp(-lam * i)
    s = seq.sum()
    if s == 0.0:
        return None
    seq /= s
    return seq


def _power_law_exp_seq(n: int, p: float, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    seq = (i + 1.0) ** (-p) * np.exp(-lam * i)
    s = seq.sum()
    if s == 0.0:
        return None
    seq /= s
    return seq


def _stretched_exponential_seq(n: int, lam: float, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    seq = np.exp(-lam * (i ** gamma))
    s = seq.sum()
    if s == 0.0:
        return None
    seq /= s
    return seq


def _beta_seq(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(1, n + 1, dtype=np.float64) / n)
    seq = (x ** alpha) * ((1.0 - x) ** beta)
    s = seq.sum()
    if s == 0.0:
        return None
    seq /= s
    return seq


def _linear_seq(n: int, slope: float) -> np.ndarray:
    """Decreasing linear profile a_i = max(0, 1 - slope * i)."""
    i = np.arange(n, dtype=np.float64)
    seq = np.maximum(0.0, 1.0 - slope * i)
    s = seq.sum()
    if s == 0.0:
        return None
    seq /= s
    return seq


def _gaussian_seq(n: int, mu: float, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    seq = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    s = seq.sum()
    if s == 0.0:
        return None
    seq /= s
    return seq


def _mix_exponential_seq(n: int, alpha: float, lam1: float, lam2: float) -> np.ndarray:
    """Mixture α·exp(-lam1·i) + (1-α)·exp(-lam2·i)."""
    i = np.arange(n, dtype=np.float64)
    seq = alpha * np.exp(-lam1 * i) + (1.0 - alpha) * np.exp(-lam2 * i)
    s = seq.sum()
    if s == 0.0:
        return None
    seq /= s
    return seq


def _dirichlet_seq(n: int, conc: float) -> np.ndarray:
    g = np.random.gamma(conc, 1.0, size=n)
    seq = g / g.sum()
    return seq


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that
    minimises ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    # reserve a few seconds for the final exact evaluation
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # Choose a working length.
    # --------------------------------------------------------------
    try:
        prev = globals()["height_sequence_1"]
        n = max(len(prev), 6000)
        prev_arr = np.asarray(prev, dtype=np.float64)
        if prev_arr.sum() > 0.0:
            prev_arr = prev_arr / prev_arr.sum()
        else:
            prev_arr = None
    except Exception:
        prev_arr = None
        n = 6000

    # clamp to allowed range
    n = int(min(max(n, 5000), 20000))          # 5 k ≤ n ≤ 20 k
    fft_len = 1 << ((2 * n - 1).bit_length())   # next power of two ≥ 2·n‑1

    # --------------------------------------------------------------
    # Cheap objective (assumes Σa_i = 1) : 2·n·max(conv)
    # --------------------------------------------------------------
    def cheap_score(vec: np.ndarray) -> float:
        maxb, _ = _conv_max_fft(vec, fft_len)
        return 2.0 * n * maxb

    # --------------------------------------------------------------
    # Phase 0 – broad parametric sweep
    # --------------------------------------------------------------
    best_a = _uniform_seq(n)
    best_val = cheap_score(best_a)

    if prev_arr is not None and prev_arr.size == n:
        val = cheap_score(prev_arr)
        if val < best_val:
            best_a, best_val = prev_arr.copy(), val

    # --- power‑law -------------------------------------------------
    for p in np.linspace(0.2, 2.0, 91):
        if time.time() >= deadline:
            break
        cand = _power_law_seq(n, p)
        if cand is None:
            continue
        val = cheap_score(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # --- exponential (fine grid, very small λ) --------------------
    for lam in np.linspace(0.0001, 0.0050, 51):
        if time.time() >= deadline:
            break
        cand = _exponential_seq(n, lam)
        if cand is None:
            continue
        val = cheap_score(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # --- mixed power‑law × exponential (fine grid) ---------------
    for p in np.linspace(0.2, 1.5, 14):
        for lam in np.linspace(0.0001, 0.0030, 31):
            if time.time() >= deadline:
                break
            cand = _power_law_exp_seq(n, p, lam)
            if cand is None:
                continue
            val = cheap_score(cand)
            if val < best_val:
                best_a, best_val = cand, val

    # --- stretched exponential (various γ) -----------------------
    for gamma in [0.5, 0.9, 1.2, 1.5, 2.0]:
        for lam in np.linspace(0.0005, 0.0045, 41):
            if time.time() >= deadline:
                break
            cand = _stretched_exponential_seq(n, lam, gamma)
            if cand is None:
                continue
            val = cheap_score(cand)
            if val < best_val:
                best_a, best_val = cand, val

    # --- beta families (coarse grid) -------------------------------
    x = (np.arange(1, n + 1, dtype=np.float64) / n)
    for alpha in np.linspace(0.2, 8.0, 21):
        for beta in np.linspace(0.2, 8.0, 21):
            if time.time() >= deadline:
                break
            seq = (x ** alpha) * ((1.0 - x) ** beta)
            s = seq.sum()
            if s == 0.0:
                continue
            cand = seq / s
            val = cheap_score(cand)
            if val < best_val:
                best_a, best_val = cand, val

    # --- linear decreasing profiles ---------------------------------
    for slope in np.linspace(0.0, 2.0 / (n - 1), 25):
        if time.time() >= deadline:
            break
        cand = _linear_seq(n, slope)
        if cand is None:
            continue
        val = cheap_score(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # --- Gaussian bumps --------------------------------------------
    for sigma in np.linspace(0.05 * n, 0.30 * n, 13):
        if time.time() >= deadline:
            break
        cand = _gaussian_seq(n, mu=0.5 * n, sigma=sigma)
        if cand is None:
            continue
        val = cheap_score(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # --- mixture of two exponentials -------------------------------
    for _ in range(200):
        if time.time() >= deadline:
            break
        alpha = np.random.rand()
        lam1 = np.random.exponential(scale=0.002)
        lam2 = np.random.exponential(scale=0.002)
        cand = _mix_exponential_seq(n, alpha, lam1, lam2)
        if cand is None:
            continue
        val = cheap_score(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # --- random Dirichlet seeds ------------------------------------
    for conc in [0.001, 0.005, 0.01, 0.05, 0.1, 0.5, 1.0, 2.0]:
        if time.time() >= deadline:
            break
        cand = _dirichlet_seq(n, conc)
        val = cheap_score(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # --- randomised power‑law + exponential ------------------------
    for _ in range(300):
        if time.time() >= deadline:
            break
        p = np.random.uniform(0.1, 2.0)
        lam = np.random.exponential(scale=0.0015)
        cand = _power_law_exp_seq(n, p, lam)
        if cand is None:
            continue
        val = cheap_score(cand)
        if val < best_val:
            best_a, best_val = cand, val

    print(f"[Init] best cheap score = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – Adam mirror‑descent on the true (nonsmooth) objective
    # --------------------------------------------------------------
    a = best_a.copy()
    best_iter_a = a.copy()
    best_iter_val = best_val
    last_improve = 0
    iter_cnt = 0

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    base_lr = 0.5                 # base learning‑rate for log‑variables
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment

    while time.time() < deadline - 2.5 and iter_cnt < 2_000_000:
        iter_cnt += 1

        # re‑normalise from log‑space (exponentiated‑gradient step)
        a = np.exp(a)                     # a holds log‑variables at start of loop
        a = _renorm_to_simplex(a)         # enforce Σa = 1, a ≥ 0

        # full convolution and cheap score
        conv = _conv_full_fft(a, fft_len)
        maxb = conv.max()
        cheap = 2.0 * n * maxb

        # record improvement
        if cheap < best_iter_val - 1e-12:
            best_iter_val = cheap
            best_iter_a = a.copy()
            last_improve = iter_cnt
            print(f"[Iter {iter_cnt:08d}] NEW BEST cheap = {cheap:.6f}")

        # active set (indices whose value is within a tiny relative tolerance)
        rel_tol = 1e-6
        active = np.where(conv >= maxb * (1.0 - rel_tol))[0]

        # sub‑gradient of the max‑convolution (average over active set)
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            rng = np.arange(lo, hi + 1, dtype=np.int64)
            g[rng] += 2.0 * a[k - rng]
        g /= float(active.size)

        # Adam update on the *log* variables (stored in a for the next loop)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** iter_cnt)
        v_hat = v / (1.0 - beta2 ** iter_cnt)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        # move in log‑space (log‑variables are stored in a)
        a = np.log(a + eps) - step

        # occasional restart if stagnation persists
        if iter_cnt - last_improve > 150_000:
            perturb = 0.02 * np.random.rand(n)
            a = np.log(_renorm_to_simplex(best_iter_a + perturb) + eps)
            last_improve = iter_cnt
            print(f"[Iter {iter_cnt:08d}] Restart after stagnation.")

    # --------------------------------------------------------------
    # Phase 2 – simple (1+1)‑ES hill‑climbing on the cheap score
    # --------------------------------------------------------------
    sigma = 0.12           # mutation strength (log‑space)
    fail_cnt = 0
    while time.time() < deadline - 1.0:
        cand = best_iter_a * np.exp(sigma * np.random.randn(n))
        cand = _simplex_projection(cand)          # cheap projection (O(n log n))
        val = cheap_score(cand)
        if val < best_iter_val:
            best_iter_val = val
            best_iter_a = cand
            sigma = max(sigma * 0.85, 1e-5)
            fail_cnt = 0
            print(f"[ES] improved cheap = {val:.6f}   sigma={sigma:.5f}")
        else:
            fail_cnt += 1
            if fail_cnt > 300:
                sigma = min(sigma * 1.30, 0.8)
                fail_cnt = 0

        # periodic mild restart from a random Dirichlet
        if iter_cnt % 180_000 == 0:
            pert = _dirichlet_seq(n, np.random.uniform(0.001, 2.0))
            a = np.log(_renorm_to_simplex(best_iter_a + 0.02 * pert) + eps)

    # --------------------------------------------------------------
    # Phase 3 – final exact evaluation
    # --------------------------------------------------------------
    final_seq = best_iter_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # --------------------------------------------------------------
    # Return the best sequence found.
    # --------------------------------------------------------------
    return final_seq
