"""
Step‑function optimiser (v3).

The optimiser proceeds in three phases:

1. **Analytic pre‑search** – a handful of inexpensive analytic
   candidates (uniform, exponential, Gaussian, sinusoid, sparse
   Dirichlet) are generated for each support length and the best one
   (lowest exact objective) is kept.

2. **Deterministic annealing on the simplex** – the sequence is
   represented by logits ``x`` (soft‑max parametrisation).  For a
   schedule of increasing inverse‑temperatures ``beta`` we minimise the
   smooth max   `logsumexp(beta·conv) / beta`  using L‑BFGS‑B
   (exact gradient via the convolution).  After each stage a tiny
   jitter is added to help escape shallow basins.  Several random
   restarts are performed.

3. **Fine‐grained refinement** – a targeted simulated‑annealing
   (mass moves from indices that contribute most to the current peak
   of the autocorrelation) followed by a few exact projected‑gradient
   steps with a simple back‑tracking line‑search.

All work respects the global time budget (default 1000 s).  The entry
point is ``propose_candidate`` which returns a plain list of
non‑negative floats suitable for ``evaluate_sequence``.
"""

import time
import math
import numpy as np
import scipy.optimize
import scipy.special


# ----------------------------------------------------------------------
# Low‑level utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return conv_full[: 2 * n - 1]


def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation of a with weight vector w:
        g[i] = Σ_t w[i+t]·a[t]   for i=0,…,n‑1
    Implemented via FFT (real‑valued).
    """
    n = a.shape[0]
    m = w.shape[0]                     # = 2n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Stable soft‑max."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) assuming Σ a = 1.
    Returns a Python float.
    """
    n = a.size
    conv = _conv_self_fft(a)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of vector v onto the probability simplex
    {x ≥ 0, Σ x = 1}.
    Uses the O(n log n) algorithm of Duchi et al.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape pre‑search utilities
# ----------------------------------------------------------------------


def _shape_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """
    Produce a short list of analytically motivated starting sequences.
    Each candidate is normalised to sum‑to‑1.
    """
    cand: list[np.ndarray] = []

    # Uniform
    cand.append(np.full(n, 1.0 / n))

    # Exponential decay (forward and reversed)
    r_vals = np.linspace(0.90, 0.9997, 13)
    idx = np.arange(n, dtype=float)
    for r in r_vals:
        a = r ** idx
        a = a / a.sum()
        cand.append(a)
        cand.append(a[::-1])

    # Gaussian windows centred in the middle
    centre = (n - 1) / 2.0
    sigma_vals = np.linspace(0.10 * n, 0.40 * n, 8)
    for sigma in sigma_vals:
        a = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
        a = a / a.sum()
        cand.append(a)

    # Half‑sine window (positive only)
    a = np.sin(np.pi * (idx + 0.5) / n)
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    cand.append(a)

    # Sparse Dirichlet‑like (few non‑zero entries)
    k = max(2, int(0.03 * n))
    for _ in range(4):
        a = np.zeros(n, dtype=float)
        pos = rng.choice(n, size=k, replace=False)
        vals = rng.random(k)
        a[pos] = vals
        a = a / a.sum()
        cand.append(a)

    # Pure random Dirichlet (α=1)
    a = rng.random(n)
    a = a / a.sum()
    cand.append(a)

    return cand


def _best_initial_sequence(n: int,
                           rng: np.random.Generator,
                           time_limit: float) -> tuple[np.ndarray, float]:
    """
    Scan a modest set of shape candidates and return the one with the
    smallest exact objective, respecting a per‑call time budget.
    """
    start = time.time()
    best_a = None
    best_obj = float("inf")
    for a in _shape_candidates(n, rng):
        if time.time() - start > time_limit:
            break
        obj = _objective(a)
        if obj < best_obj:
            best_obj, best_a = obj, a
    # Fallback – random Dirichlet
    if best_a is None:
        a = rng.random(n)
        best_a = a / a.sum()
        best_obj = _objective(best_a)
    return best_a, best_obj

# Smoothed max & its gradients (used by L‑BFGS)
# ----------------------------------------------------------------------


def _smooth_max_and_grad(a: np.ndarray, beta: float):
    """
    Returns (smooth_max, gradient_wrt_a, weight_vector).

    * ``smooth_max`` is the smoothed approximation of ``max(conv(a,a))``:
          smooth = max + logsumexp(beta*(conv-max))/beta
    * ``gradient_wrt_a`` is 2·correlation(weight, a)
    * ``weight`` is the soft‑max (or a one‑hot when beta is huge)
    """
    conv = _conv_self_fft(a)
    if beta < 1e5:
        max_val = conv.max()
        shifted = conv - max_val            # ≤ 0
        w = np.exp(beta * shifted)
        sum_w = w.sum()
        if sum_w == 0.0:
            w = np.ones_like(conv) / conv.size
            smooth = conv.mean()
        else:
            w = w / sum_w
            smooth = max_val + math.log(sum_w) / beta
    else:
        # Hard max – avoid overflow
        max_idx = int(np.argmax(conv))
        max_val = conv[max_idx]
        w = np.zeros_like(conv)
        w[max_idx] = 1.0
        smooth = max_val
    # gradient of smooth_max w.r.t. a
    g_a = 2.0 * _correlation_fft(w, a)
    return smooth, g_a, w


def _obj_and_grad_logit(x: np.ndarray, beta: float):
    """
    Objective and gradient for L‑BFGS.

    Parameterisation: a = softmax(x); objective = 2·n·smooth_max(conv(a,a)).
    Returns (objective, gradient_wrt_x).
    """
    a = _softmax(x)
    smooth, g_a, _ = _smooth_max_and_grad(a, beta)
    n = a.size
    f = float(2.0 * n * smooth)
    dot = np.dot(a, g_a)
    g_x = a * (g_a - dot)          # mirroring (natural) gradient in log‑it space
    return f, g_x


def _func_logit(x: np.ndarray, beta: float) -> float:
    """Scalar objective for minimise (beta is passed via args)."""
    f, _ = _obj_and_grad_logit(x, beta)
    return f


def _jac_logit(x: np.ndarray, beta: float) -> np.ndarray:
    """Gradient for minimise (beta is passed via args)."""
    _, g = _obj_and_grad_logit(x, beta)
    return g

# Deterministic annealing optimiser (L‑BFGS on logits)
# ----------------------------------------------------------------------


def _optimize_logit(a_init: np.ndarray,
                    time_limit: float,
                    beta_schedule: list[float],
                    seed: int = 0) -> tuple[np.ndarray, float]:
    """
    Runs L‑BFGS on the log‑it simplex for each beta in ``beta_schedule``.
    Returns the best sequence (as numpy array) and its exact objective.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # initialise logits
    x = np.log(a_init + 1e-16)

    best_a = a_init.copy()
    best_obj = _objective(best_a)

    for beta in beta_schedule:
        if time.time() - start > time_limit:
            break

        # allocate a modest number of L‑BFGS iterations (depends on remaining time)
        remaining = time_limit - (time.time() - start)
        # allow roughly 30 iterations per second of remaining budget
        maxiter = max(5, int(min(200, remaining * 30)))

        try:
            res = scipy.optimize.minimize(
                _func_logit,
                x,
                args=(beta,),
                method="L-BFGS-B",
                jac=_jac_logit,
                options={"maxiter": maxiter, "ftol": 1e-12, "disp": False},
            )
        except Exception:
            # fallback: abort optimisation for this beta
            break

        x = res.x
        a = _softmax(x)
        obj = _objective(a)
        if obj < best_obj:
            best_obj, best_a = obj, a.copy()
            print(f"[beta={beta:.1e}] new best O={best_obj:.6f}")

        # occasional jitter to shake out of plateaus
        jitter = rng.normal(scale=0.008, size=a.shape)
        x += jitter

    return best_a, best_obj


# ----------------------------------------------------------------------
# Targeted simulated annealing + projected‑gradient refinement
# ----------------------------------------------------------------------


def _run_sa(init_seq: list[float],
            time_limit: float,
            seed: int = 0) -> tuple[list[float], float]:
    """
    Simulated‑annealing refinement followed by a few exact projected‑gradient
    steps.  Returns (sequence, objective).
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    a = np.asarray(init_seq, dtype=float)
    if a.sum() < 1e-12:
        a = np.ones_like(a) / a.size
    else:
        a = a / a.sum()
    n = a.shape[0]

    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())
    best_seq = a.copy()

    # SA schedule
    T = best_obj * 0.25 + 1e-8
    T_min = best_obj * 1e-7 + 1e-12
    decay = 0.997

    it = 0
    while time.time() - start < time_limit * 0.70:
        it += 1
        k_max = int(np.argmax(conv))

        # contribution distribution for the current worst lag
        i_vals = np.arange(n)
        j_vals = k_max - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i_vals[mask]] * a[j_vals[mask]]
        total_contrib = contrib.sum()

        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = rng.choice(n, p=probs)
        else:
            i = rng.integers(n)

        # destination index (different from i)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # move a modest fraction of mass
        delta = a[i] * rng.random() * 0.09  # up to 9 % of a[i]
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.clip(a_new, 0.0, None)
        a_new /= a_new.sum()

        conv_new = _conv_self_fft(a_new)
        obj_new = float(2.0 * n * conv_new.max())

        if obj_new < best_obj - 1e-14:
            a, conv, best_obj, best_seq = a_new, conv_new, obj_new, a_new.copy()
            print(f"[SA it={it}] new best O={best_obj:.6f}")
        else:
            # Metropolis acceptance
            if rng.random() < math.exp(-(obj_new - best_obj) / T):
                a, conv = a_new, conv_new

        if it % 2500 == 0:
            T = max(T * decay, T_min)

        if it % 8000 == 0:
            # global jitter every few thousand steps
            noise = rng.normal(scale=0.010, size=n)
            a = np.clip(a + noise, 0.0, None)
            a /= a.sum()
            conv = _conv_self_fft(a)

    # -----------------------------------------------------------------
    # Exact projected‑gradient descent with back‑tracking line‑search
    remaining = time_limit - (time.time() - start)
    if remaining > 0.5:
        a = best_seq.copy()
        n = a.size
        it_g = 0
        eta = 0.3
        while time.time() - start < time_limit:
            it_g += 1
            conv = _conv_self_fft(a)
            k_max = int(np.argmax(conv))
            w = np.zeros_like(conv)
            w[k_max] = 1.0
            g = 2.0 * _correlation_fft(w, a)        # exact max‑gradient

            # simple back‑tracking line‑search
            improved = False
            for _ in range(12):
                a_new = _project_simplex(a - eta * g)
                obj_new = float(2.0 * n * _conv_self_fft(a_new).max())
                if obj_new < best_obj - 1e-14:
                    a, best_obj, best_seq = a_new, obj_new, a_new.copy()
                    improved = True
                    break
                eta *= 0.5
            if not improved:
                eta = max(eta * 0.7, 1e-6)
            if it_g % 300 == 0:
                eta = min(eta * 1.3, 0.5)

    return best_seq.tolist(), best_obj


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **_ignore) -> list[float]:
    """
    Optimises a non‑negative step function under the given evaluation
    function.  The routine respects a total time budget (default 1000 s)
    and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # safety margin; the evaluator itself is cheap
    deadline = time.time() + budget_s - 0.5

    # support lengths to explore – longer lengths give more flexibility
    lengths = [2000, 4000, 8000, 15000]

    best_seq: np.ndarray | None = None
    best_obj = float("inf")

    # -----------------------------------------------------------------
    # 1️⃣  per‑length pre‑search + deterministic annealing (L‑BFGS)
    # -----------------------------------------------------------------
    for idx, n in enumerate(lengths):
        now = time.time()
        remaining = deadline - now
        if remaining <= 5.0:
            break

        # allocate time for this length
        if idx == 0:
            run_time = remaining * 0.12
        elif idx == len(lengths) - 1:
            run_time = remaining * 0.55
        else:
            run_time = remaining * 0.33 / (len(lengths) - 2)

        # ---- shape pre‑search (quick) --------------------------------
        shape_time = min(4.0, run_time * 0.05)
        init_a, _ = _best_initial_sequence(n, rng, shape_time)

        # ---- multiple restarts of deterministic annealing ------------
        beta_sched = [0.5, 1, 2, 5, 10, 20, 50, 100, 200,
                      500, 1000, 5000, 20000, 1e5, 1e6]

        # we run a few random restarts (including the best shape)
        restarts = 4
        per_restart_time = run_time * 0.85 / restarts

        for r in range(restarts):
            if time.time() >= deadline - 0.5:
                break
            if r == 0:
                a0 = init_a
            else:
                a0 = rng.random(n)
                a0 = a0 / a0.sum()
            seq, val = _optimize_logit(a0,
                                       time_limit=per_restart_time,
                                       beta_schedule=beta_sched,
                                       seed=seed + idx * 10000 + r * 123)
            if val < best_obj:
                best_obj, best_seq = val, np.asarray(seq, dtype=float)
                print(f"[global] new overall best O={best_obj:.6f} (n={n})")

    # -----------------------------------------------------------------
    # 2️⃣  Final refinement (SA + exact PGD)
    # -----------------------------------------------------------------
    remaining = deadline - time.time()
    if best_seq is not None and remaining > 6.0:
        seq_ref, val_ref = _run_sa(best_seq.tolist(),
                                   time_limit=remaining,
                                   seed=seed + 999999)
        if val_ref < best_obj:
            best_obj, best_seq = val_ref, np.asarray(seq_ref, dtype=float)
            print(f"[final] after SA+PG new best O={best_obj:.6f}")

    # -----------------------------------------------------------------
    # 3️⃣  Fallback – should never be needed
    # -----------------------------------------------------------------
    if best_seq is None or best_seq.size == 0:
        best_seq = np.ones(200, dtype=float) / 200.0

    # -----------------------------------------------------------------
    # sanity check (diagnostic only)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    # enforce evaluator caps
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    if best_seq.sum() < 0.01:
        best_seq = np.ones_like(best_seq) / best_seq.size

    return best_seq.tolist()
