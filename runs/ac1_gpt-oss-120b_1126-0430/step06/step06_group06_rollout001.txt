"""
Propose a non‑negative step function (list of heights) that minimises

        2·n·max_k (a*a)[k] / (Σ a_i)²

using a three‑phase strategy:

1️⃣  Initialise the distribution with a broad sweep of parametric families
    (uniform, power‑law, exponential, mixed, beta, sine‑shaped, Dirichlet).
    The best candidate (measured by the cheap surrogate
    `2·n·max(conv)`) is kept.

2️⃣  Refine the candidate with an Adam‑based exponentiated‑gradient (mirror
    descent) on the true non‑smooth objective.  At each iteration we
    compute the full convolution via a real FFT, identify the active
    indices that realise the current maximum, build the corresponding
    sub‑gradient and perform an Adam update in log‑space, followed by a
    Euclidean projection onto the probability simplex.

3️⃣  Finish with a (1+1) Evolution Strategy that perturbs the best
    vector in log‑space with an adaptive mutation strength.  The cheap
    score is used for fast acceptance decisions; after the budget is
    exhausted the exact `evaluate_sequence` is called once.

The routine adapts its schedule to the available time budget (default
1000 s) and always returns a feasible sequence (non‑negative,
Σa_i > 0.01).  The implementation is fully self‑contained, uses only
NumPy and the supplied ``evaluate_sequence`` function and respects the
no‑closure / top‑level‑function constraints.

def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        # all non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution (a*a) using real FFT, length 2·n‑1."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.size - 1]


def _conv_max_fft(a: np.ndarray, fft_len: int) -> tuple[float, int]:
    """Maximum coefficient of a*a and its index."""
    conv = _conv_full_fft(a, fft_len)
    idx = int(conv.argmax())
    return float(conv[idx]), idx


def _interpolate_seq(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a distribution by linear interpolation and renormalise."""
    old_n = old_seq.size
    if old_n == new_n:
        return old_seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n, dtype=np.float64)
    x_new = np.linspace(0.0, 1.0, new_n, dtype=np.float64)
    new_seq = np.interp(x_new, x_old, old_seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s == 0.0:
        return np.full(new_n, 1.0 / new_n, dtype=np.float64)
    return new_seq / s

# Parametric families (all return a normalised probability vector)
# ----------------------------------------------------------------------


def _uniform_seq(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _power_law_seq(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = i ** (-p)
    seq /= seq.sum()
    return seq


def _exponential_seq(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    seq = np.exp(-lam * i)
    seq /= seq.sum()
    return seq


def _power_law_exp_seq(n: int, p: float, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    seq = (i + 1.0) ** (-p) * np.exp(-lam * i)
    seq /= seq.sum()
    return seq


def _stretched_exponential_seq(n: int, lam: float, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    seq = np.exp(-lam * (i ** gamma))
    seq /= seq.sum()
    return seq


def _beta_seq(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(1, n + 1, dtype=np.float64) / n)
    seq = (x ** alpha) * ((1.0 - x) ** beta)
    s = seq.sum()
    if s == 0.0:
        return None
    seq /= s
    return seq


def _sine_seq(n: int, power: float = 1.0) -> np.ndarray:
    """Sine‑shaped distribution a_i ∝ sin(π·(i+0.5)/n)^power."""
    i = np.arange(n, dtype=np.float64) + 0.5
    seq = np.sin(np.pi * i / n) ** power
    seq /= seq.sum()
    return seq


def _arcsine_seq(n: int, eps: float = 1e-6) -> np.ndarray:
    """Arcsine‑like distribution a_i ∝ 1/√(x·(1‑x))."""
    x = (np.arange(1, n + 1, dtype=np.float64) - 0.5) / n
    # Avoid division by zero at the boundaries
    x = np.clip(x, eps, 1.0 - eps)
    seq = 1.0 / np.sqrt(x * (1.0 - x))
    seq /= seq.sum()
    return seq


def _dirichlet_seq(n: int, conc: float) -> np.ndarray:
    g = np.random.gamma(conc, 1.0, size=n)
    seq = g / g.sum()
    return seq

            break
        cand = _dirichlet_seq(n, conc)
        val = cheap_score(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # Additional random power‑law + exponential mixes
    for _ in range(150):
        if time.time() >= deadline:
            break
        p = np.random.uniform(0.1, 2.0)
        lam = np.random.exponential(scale=0.5)
        cand = _power_law_exp_seq(n, p, lam)
        val = cheap_score(cand)
        if val < best_val:
            best_a, best_val = cand, val

    print(f"[Init] best cheap score = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – Adam based exponentiated‑gradient (mirror descent)
    # --------------------------------------------------------------
    a = best_a.copy()
    best_iter_a = a.copy()
    best_iter_val = best_val
    last_improve_iter = 0
    iter_cnt = 0

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps_log = 1e-30
    eps_den = 1e-12
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    base_lr = 0.6   # base learning rate in log‑space

    # Main refinement loop – we stop a few seconds before the deadline.
    while time.time() < deadline - 3.0:
        iter_cnt += 1

        # Full convolution via FFT
        conv = _conv_full_fft(a, fft_len)
        maxb = conv.max()
        cheap = 2.0 * n * maxb   # cheap surrogate (Σa_i = 1)

        # Record best seen so far
        if cheap < best_iter_val - 1e-12:
            best_iter_val = cheap
            best_iter_a = a.copy()
            last_improve_iter = iter_cnt
            print(f"[Iter {iter_cnt:07d}] NEW BEST cheap = {cheap:.6f}")

        # Identify active constraints (indices achieving the max)
        # A tiny relative tolerance picks exactly the binding ones.
        rel_tol = 1e-12
        active = np.where(conv >= maxb * (1.0 - rel_tol))[0]

        # Sub‑gradient of max(conv) at the active set.
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            rng = np.arange(lo, hi + 1, dtype=np.int64)
            # contribution 2*a[k - rng] (the derivative of a_i*a_{k-i})
            g[rng] += 2.0 * a[k - rng]
        # Average over the active set to avoid overly large steps.
        if active.size > 0:
            g /= float(active.size)

        # Adam update in log‑space (mirror descent with entropy regulariser)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        # Bias‑corrected moments
        m_hat = m / (1.0 - beta1 ** iter_cnt)
        v_hat = v / (1.0 - beta2 ** iter_cnt)

        # Adaptive step size – we decay it very slowly (∝ 1/√t)
        lr = base_lr / math.sqrt(iter_cnt)

        # Exponentiated‑gradient step
        log_a = np.log(a + eps_log)
        log_a -= lr * m_hat / (np.sqrt(v_hat) + eps_den)
        a = np.exp(log_a)

        # Projection onto the probability simplex
        s = a.sum()
        if s <= 0.0 or not np.isfinite(s):
            # Revert to the best known iterate if numerical trouble appears
            a = best_iter_a.copy()
            continue
        a /= s

        # Occasional random restart if stagnation lasts too long
        if iter_cnt - last_improve_iter > 120_000:
            perturb = 0.02 * np.random.rand(n)
            a = _simplex_projection(best_iter_a + perturb)
            last_improve_iter = iter_cnt
            print(f"[Iter {iter_cnt:07d}] Restart after stagnation.")

    # --------------------------------------------------------------
    # Phase 2 – simple (1+1)‑ES hill‑climbing on the cheap score
    # --------------------------------------------------------------
    sigma = 0.12  # mutation strength in log‑space
    fail_cnt = 0
    while time.time() < deadline - 2.0:
        # Log‑space multiplicative perturbation
        cand = best_iter_a * np.exp(sigma * np.random.randn(n))
        cand = _simplex_projection(cand)
        val = cheap_score(cand)
        if val < best_iter_val:
            best_iter_val = val
            best_iter_a = cand
            sigma = max(sigma * 0.85, 1e-5)   # shrink on success
            fail_cnt = 0
            print(f"[ES] improved cheap = {val:.6f}  sigma={sigma:.5f}")
        else:
            fail_cnt += 1
            if fail_cnt > 250:
                sigma = min(sigma * 1.30, 0.6)   # enlarge after many failures
                fail_cnt = 0

        # Periodic small restart to escape plateaus
        if iter_cnt % 150_000 == 0:
            pert = _dirichlet_seq(n, np.random.uniform(0.001, 2.0))
            a = _simplex_projection(best_iter_a + 0.02 * pert)
            # No immediate impact – exploration is driven by the ES step.

    # --------------------------------------------------------------
    # Phase 3 – final exact evaluation
    # --------------------------------------------------------------
    final_seq = best_iter_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:      # defensive – should never happen
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # --------------------------------------------------------------
    # Return the best sequence found.
    # --------------------------------------------------------------
    return final_seq
