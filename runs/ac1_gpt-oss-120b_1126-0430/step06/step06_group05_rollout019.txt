"""
Non‑negative step‑function optimiser (smooth‑max Adam + final SA).

The optimiser works in two phases:
* Phase 1 – “soft‑max Adam”: minimise a smoothed version of the
  maximal self‑convolution using the log‑sum‑exp (soft‑max) as a
  surrogate.  The gradient of the smooth loss can be written as a
  cross‑correlation of the current sequence with the soft‑max weights,
  which is evaluated with a single FFT.  An exponentially decaying
  temperature drives the surrogate from a very smooth average to the
  true hard max.
* Phase 2 – optional “hard‑max Adam”: a short refinement with the exact
  sub‑gradient of the current max‑convolution entry (as in the reference
  implementation) to polish the solution.
* Phase 3 – simulated‑annealing (SA) fine‑tuning of a few‐percent of
  mass moves.

All updates are performed in the log‑it (exponentiated‑gradient) space
so the simplex constraint (non‑negative entries summing to one) is
automatically respected.  The routine respects the overall time budget
and returns the best sequence found.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return conv_full[: 2 * n - 1]


def _conv_fft_full(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution a*b (both real) using FFT."""
    n = a.shape[0]
    m = b.shape[0]
    L = _next_pow2(n + m - 1)
    af = np.fft.rfft(a, n=L)
    bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(af * bf, n=L)
    return conv[: n + m - 1]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Project logits onto the probability simplex."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


# ----------------------------------------------------------------------
# Phase 1 – smooth‑max Adam optimiser
# ----------------------------------------------------------------------


def _run_opt_smoothmax(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
) -> Tuple[List[float], float]:
    """
    Adam optimiser on the soft‑max surrogate of max(conv(a,a)).
    Returns the best sequence (list) and its hard‑max objective value.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # initialise a on the simplex
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
    else:
        a = rng.random(n)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    a = a / s if s > 1e-12 else np.ones(n) / n

    # log‑it variables for exponentiated‑gradient updates
    x = np.log(a + 1e-16)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # schedules
    lr_start, lr_end = 0.5, 0.001               # learning‑rate decay
    T_start, T_end = 1.0, 1e-5                 # temperature (soft‑max) decay

    best_a = a.copy()
    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())

    it = 0
    while time.time() - start < time_limit:
        it += 1

        # current point on the simplex
        a = _softmax(x)

        # self‑convolution (hard‑max value is needed for the real objective)
        conv = _conv_self_fft(a)
        max_conv = conv.max()
        obj = float(2.0 * n * max_conv)

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[smooth n={n} it={it:07d}] new best O={best_obj:.6f}")

        # ------------------------------------------------------------------
        # soft‑max weights over the convolution entries
        # ------------------------------------------------------------------
        shift = max_conv                     # for numerical stability
        # temperature schedule (exponential decay)
        elapsed = time.time() - start
        t_frac = min(elapsed / time_limit, 1.0)
        T = T_start * np.exp(np.log(T_end / T_start) * t_frac)

        # w[k] = exp(conv[k]/T) / sum exp(...): a smooth approximation of the max
        exp_vals = np.exp((conv - shift) / T)
        w = exp_vals / exp_vals.sum()        # shape (2n‑1,)

        # gradient of the smooth loss = cross‑correlation of a and w
        w_rev = w[::-1]                      # reversed soft‑max weights
        conv_rev = _conv_fft_full(a, w_rev)  # length 3n‑2
        offset = len(w) - 1                  # = 2n‑2
        # g_i = 2 * sum_j a_j * w[i+j]  (cross‑correlation)
        g = 2.0 * conv_rev[offset - np.arange(n)]

        # natural gradient on the simplex (mirrored update)
        dot = np.dot(a, g)
        g_x = a * (g - dot)

        # learning‑rate schedule
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t_frac)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # Adam step in log‑it space
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape plateaus
        if it % 5000 == 0:
            jitter = rng.normal(scale=0.01, size=n)
            x += jitter

        # occasional full restart from the current best
        if it % 80000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.005, size=n)
            m.fill(0.0)
            v.fill(0.0)

    # ------------------------------------------------------------------
    # prune negligible entries (they do not help the objective)
    # ------------------------------------------------------------------
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = float(2.0 * best_a.size * _conv_self_fft(best_a).max())
        print(f"[smooth compress] reduced length → {best_a.size}, O={best_obj:.6f}")

    return best_a.tolist(), best_obj

def _run_opt_hardmax(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
) -> Tuple[List[float], float]:
    """
    Adam optimiser that always uses the exact max‑convolution gradient.
    Returns the best sequence (list) and its hard‑max objective.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
    else:
        a = rng.random(n)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    a = a / s if s > 1e-12 else np.ones(n) / n

    x = np.log(a + 1e-16)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    lr_start, lr_end = 0.5, 0.001

    best_a = a.copy()
    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())

    it = 0
    while time.time() - start < time_limit:
        it += 1
        a = _softmax(x)

        conv = _conv_self_fft(a)
        M = conv.max()
        obj = float(2.0 * n * M)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[hard n={n} it={it:07d}] new best O={best_obj:.6f}")

        k_max = int(np.argmax(conv))
        # exact sub‑gradient of the k‑th entry
        idx = np.arange(n)
        j = k_max - idx
        mask = (j >= 0) & (j < n)
        g_a = np.zeros_like(a)
        if mask.any():
            g_a[mask] = 2.0 * a[j[mask]]

        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)

        elapsed = time.time() - start
        t = min(elapsed / time_limit, 1.0)
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t)

        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if it % 5000 == 0:
            jitter = rng.normal(scale=0.01, size=n)
            x += jitter
        if it % 80000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.005, size=n)
            m.fill(0.0)
            v.fill(0.0)

    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = float(2.0 * best_a.size * _conv_self_fft(best_a).max())
        print(f"[hard compress] reduced length → {best_a.size}, O={best_obj:.6f}")

    return best_a.tolist(), best_obj

def _run_sa(
    init_seq: List[float],
    time_limit: float,
    seed: int = 0,
) -> Tuple[List[float], float]:
    """Small‑scale SA that moves a few percent of mass between positions."""
    start = time.time()
    rng = np.random.default_rng(seed)
    a = np.asarray(init_seq, dtype=float)
    if a.sum() < 1e-12:
        a = np.ones_like(a) / a.size
    else:
        a = a / a.sum()
    n = a.shape[0]

    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())
    best_seq = a.copy()

    T = best_obj * 0.1 + 1e-6
    T_min = best_obj * 1e-6 + 1e-9
    decay = 0.995
    it = 0

    while time.time() - start < time_limit:
        it += 1
        k_max = int(np.argmax(conv))

        i_vals = np.arange(n)
        j_vals = k_max - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i_vals[mask]] * a[j_vals[mask]]
        total_contrib = contrib.sum()

        if total_contrib > 0:
            probs = contrib / total_contrib
            i = rng.choice(n, p=probs)
        else:
            i = rng.integers(n)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.clip(a_new, 0.0, None)
        a_new /= a_new.sum()

        conv_new = _conv_self_fft(a_new)
        obj_new = float(2.0 * n * conv_new.max())

        if obj_new < best_obj - 1e-12:
            a, conv, best_obj, best_seq = a_new, conv_new, obj_new, a_new.copy()
            print(f"[SA it={it}] new best O={best_obj:.6f}")
        else:
            prob = np.exp(-(obj_new - best_obj) / T) if T > 0 else 0.0
            if rng.random() < prob:
                a, conv = a_new, conv_new

        if it % 2000 == 0:
            T = max(T * decay, T_min)
        if it % 10000 == 0:
            noise = rng.normal(scale=0.01, size=n)
            a = np.clip(a + noise, 0.0, None)
            a /= a.sum()
            conv = _conv_self_fft(a)

    thresh = 1e-12
    mask = best_seq > thresh
    if mask.sum() < best_seq.size:
        comp = best_seq[mask]
        comp /= comp.sum()
        comp_obj = float(2.0 * comp.size * _conv_self_fft(comp).max())
        if comp_obj < best_obj:
            best_seq, best_obj = comp, comp_obj
            print(f"[SA compress] reduced length → {comp.size}, O={best_obj:.6f}")

    return best_seq.tolist(), best_obj
