    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0
    try: n = len(height_sequence_1) except: n=20000

    # arcsine init
    # ...
    best_a = a.copy()
    best_score = 2.0 * n * _conv_max(best_a)   # approximate

    # Adam parameters
    z = _logits_from_distribution(best_a)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_alpha = 0.7   # maybe larger

    eps_factor = 0.12
    eps_factor_min = 5e-5  # smaller for final
    eps_factor_decay = 0.9995

    # L2 phase config
    phase1_iters = 50000   # may adjust based on time
    lambda_l2 = 0.5         # weight of L2 gradient

    best_score = 2.0 * n * _conv_max(best_a)
    # For tracking
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        a_cur = _softmax(z)
        # compute lengths
        L = _fft_len_for_conv(n)
        a_fft = np.fft.rfft(a_cur, L)

        # conv
        conv = np.fft.irfft(a_fft * a_fft, L)[:2*n-1]
        max_conv = float(conv.max())
        # compute score
        score = 2.0 * n * max_conv
        if score < best_score:
            best_score = score
            best_a = a_cur.copy()
            if it % 1000 == 0:
                print(f"Iter {it}: best_score {best_score:.8f}")

        # compute gradient components
        # L2 part
        if it <= phase1_iters:
            # compute uniform target u
            u = 1.0 / (2*n - 1)
            e = conv - u
            e_fft = np.fft.rfft(e, L)
            corr_err = np.fft.irfft(np.conj(a_fft) * e_fft, L)[:n]
            grad_L2 = 4.0 * corr_err
        else:
            grad_L2 = np.zeros_like(a_cur)

        # Max part
        eps = max(max_conv * eps_factor, 1e-12)
        shifted = conv - max_conv
        w = np.exp(shifted / eps)
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum
        w_fft = np.fft.rfft(w, L)
        corr_max = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]
        grad_max = 4.0 * n * corr_max

        # total gradient
        grad_a = grad_max + lambda_l2 * grad_L2

        # gradient w.r.t logits

        grad_z = a_cur * (grad_a - mu)

        v = beta2 * v + (1 - beta2) * (grad_z ** 2)

        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step = - alpha_t * m_hat / (np.sqrt(v_hat) + eps_adam)
        z = z + step
        # clip logits

        # update eps_factor schedule

"""
Advanced two‑phase Adam optimiser for the max‑convolution problem.

* Phase 1 minimises the L2 deviation of the self‑convolution from the
  perfectly uniform target (1/(2 n‑1) at every position).  This quickly
  flattens the convolution profile.
* Phase 2 switches to a smoothed L∞ loss (soft‑max of the convolution) to
  drive the true evaluation metric 2·n·max(a*a) down.

The optimiser works on the simplex by parametrising the vector with
logits and a soft‑max, guaranteeing non‑negativity and a unit sum.
A modest amount of random jitter is added occasionally to help escape
plateaus.  The routine respects the 1000 s time budget, prints occasional
progress, and finally returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all defined at top‑level (no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Logits that map to the probability vector ``a`` via soft‑max."""
    return np.log(a + eps)


def _fft_len_for_conv(n: int) -> int:
    """Smallest power of two ≥ 2·n‑1 (sufficient for a linear self‑convolution)."""
    return 1 << ((2 * n - 1).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a (no gradient)."""
    n = a.shape[0]
    L = _fft_len_for_conv(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())
