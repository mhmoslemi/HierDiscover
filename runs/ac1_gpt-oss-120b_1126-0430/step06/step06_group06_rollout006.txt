    if n > 25000: n = 25000

    # helper functions defined earlier top-level
    ...

    best_a = None
    best_score = np.inf

    # random restart count
    restart_idx = 0
    while time.time() < deadline and restart_idx < 10:
        # initialize x
        if restart_idx == 0 and prev_arr is not None and prev_arr.size == n:
            a0 = prev_arr / prev_arr.sum()
        else:
            # Dirichlet draw

        x = np.log(a0 + eps)

        v = np.zeros_like(x)

        beta2 = 0.999
        eps_adam = 1e-8
        alpha = 0.1   # base stepsize
        # initialize sigma

        a = np.exp(x)
        a /= a.sum()
        conv = _conv_full_fft(a, L)[:2*n-1]   # use function that returns full
        maxc = conv.max()
        sigma = maxc * 0.2  # start 20% of max
        sigma_min = maxc * 1e-4  # lower bound
        # iteration loop
        iter_i = 0
        while time.time() < deadline:
            iter_i += 1
            # compute a each iteration from x
            a = np.exp(x)
            sum_a = a.sum()
            a /= sum_a   # ensure simplex
            # conv of a*a
            a_fft = np.fft.rfft(a, n=L)
            conv_full = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]

            cheap = 2.0 * n * maxc
            # record best

                best_a = a.copy()

#!/usr/bin/env python
"""
Propose a non‑negative step function (list of heights) that minimises
        2·n·max_k (a*a)[k] / (Σ a_i)²

Algorithm
---------
* Choose a working length n (5 k – 25 k).  If a previous construction
  ``height_sequence_1`` exists we reuse its length, otherwise we start
  with n≈8 k.
* Initialise a population of candidates using a few simple parametric
  families (uniform, power‑law, exponential, beta) and a Dirichlet
  draw.  Keep the best of these as a warm‑start.
* Refine the best candidate with a *smooth* optimisation:
  - Represent the probability vector a as a soft‑max of free variables
    x (a_i = exp(x_i)/Σexp(x) ).
  - Replace the non‑smooth max‑norm by a log‑sum‑exp smooth
    approximation with a temperature σ, decreasing σ gradually.
  - Compute the gradient of the smoothed objective analytically via
    FFT‑based convolutions (O(n log n) per iteration).
  - Perform an Adam update on the log‑variables x and re‑normalise
    implicitly by the soft‑max.
  - Restart from a fresh Dirichlet draw if progress stalls.
* After the allotted time (≈ 1000 s) return the best simplex‑feasible
  vector found.  The returned list is guaranteed to be non‑negative,
  have Σa_i>0.01 and to respect the evaluation budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_full_fft(a: np.ndarray, L: int) -> np.ndarray:
    """
    Full linear convolution of a with itself using real FFT.
    Returns an array of length 2·len(a)‑1 (the valid linear part).
    """
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    n = a.shape[0]
    return conv[: 2 * n - 1]


def _conv_one_fft(u: np.ndarray, v: np.ndarray, L: int) -> np.ndarray:
    """
    Linear convolution of u and v (both real) via real FFT.
    Returns the full linear convolution (length len(u)+len(v)-1).
    """
    fu = np.fft.rfft(u, n=L)
    fv = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(fu * fv, n=L)
    return conv


def _uniform_seq(n: int) -> np.ndarray:
    """Uniform probability vector."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _power_law_seq(n: int, p: float) -> np.ndarray:
    """Discrete power‑law a_i ∝ i^{‑p} (i starts at 1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = i ** (-p)
    seq /= seq.sum()
    return seq


def _exponential_seq(n: int, lam: float) -> np.ndarray:
    """Discrete exponential a_i ∝ exp(‑λ i)."""
    i = np.arange(n, dtype=np.float64)
    seq = np.exp(-lam * i)
    seq /= seq.sum()
    return seq


def _beta_seq(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑shaped family a_i ∝ x_i^{α} (1‑x_i)^{β}, x_i = i/n."""
    x = (np.arange(1, n + 1, dtype=np.float64) / n)
    seq = (x ** alpha) * ((1.0 - x) ** beta)
    s = seq.sum()
    if s == 0.0:
        return None
    seq /= s
    return seq


def _dirichlet_seq(n: int, conc: float) -> np.ndarray:
    """Dirichlet draw with concentration `conc` (scalar)."""
    g = np.random.gamma(conc, 1.0, size=n)
    seq = g / g.sum()
    return seq


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that
    minimises ``evaluate_sequence``.  The routine adapts its schedule to
    the available time budget and always returns a feasible sequence.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the final exact evaluation
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # Determine working length n
    # --------------------------------------------------------------
    try:
        prev = globals()["height_sequence_1"]
        prev_arr = np.asarray(prev, dtype=np.float64)
    except Exception:
        prev_arr = None

    if prev_arr is not None and prev_arr.size >= 5000:
        n = max(min(prev_arr.size, 25000), 5000)
    else:
        n = 8000                       # default if no previous data
    n = max(5000, min(25000, n))

    # FFT size – needs to accommodate convolution of length up to 3·n‑2
    L = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Cheap scoring function (assumes Σa_i = 1)
    # --------------------------------------------------------------
    def cheap_score(vec: np.ndarray) -> float:
        """Return 2·n·max(conv) for a probability vector."""
        conv = _conv_full_fft(vec, L)
        return 2.0 * n * conv.max()

    # --------------------------------------------------------------
    # Phase 0 – simple parametric initialisation
    # --------------------------------------------------------------
    best_a = None
    best_val = np.inf

    init_candidates = []

    # Uniform
    init_candidates.append(_uniform_seq(n))

    # A few hand‑crafted families
    for p in (0.5, 1.0, 1.5):
        init_candidates.append(_power_law_seq(n, p))
    for lam in (0.1, 0.5, 1.0, 2.0):
        init_candidates.append(_exponential_seq(n, lam))
    for α in (0.5, 2.0):
        for β in (0.5, 2.0):
            seq = _beta_seq(n, α, β)
            if seq is not None:
                init_candidates.append(seq)

    # Random Dirichlet draws (wide range of concentrations)
    for conc in (0.001, 0.01, 0.1, 0.5, 1.0, 2.0):
        init_candidates.append(_dirichlet_seq(n, conc))

    # Include previous construction if length matches
    if prev_arr is not None and prev_arr.size == n:
        tmp = prev_arr.copy()
        if tmp.sum() > 0.0:
            tmp /= tmp.sum()
            init_candidates.append(tmp)

    # Evaluate the cheap scores of all candidates
    for cand in init_candidates:
        if time.time() >= deadline:
            break
        val = cheap_score(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
    print(f"[Init] best cheap score = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – smooth Adam optimisation on the simplex (log‑space)
    # --------------------------------------------------------------
    max_restarts = 5
    restart = 0
    eps = 1e-30

    while time.time() < deadline and restart < max_restarts:
        # initialise log‑probabilities x
        if restart == 0 and best_a is not None:
            a0 = best_a.copy()
        else:
            conc = np.random.choice([0.001, 0.005, 0.01, 0.05,
                                     0.1, 0.5, 1.0, 2.0])
            a0 = _dirichlet_seq(n, conc)

        # avoid degenerate start
        a0 = np.clip(a0, eps, None)
        a0 /= a0.sum()
        x = np.log(a0 + eps)

        # Adam moments
        m = np.zeros_like(x)
        v = np.zeros_like(x)
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-8
        alpha = 0.5                # base learning rate for log‑variables

        # temperature schedule (relative to current max‐convolution)
        sigma_factor = 0.2
        sigma_min_factor = 1e-5

        it = 0
        last_improve_it = 0
        while time.time() < deadline:
            it += 1

            # ----- obtain probability vector from logits -----
            a = np.exp(x)
            s = a.sum()
            if s <= 0.0 or not np.isfinite(s):
                # re‑initialise from current best if something went wrong
                a = best_a.copy() if best_a is not None else _uniform_seq(n)
                x = np.log(a + eps)
                continue
            a /= s

            # ----- convolution a*a (linear) -----
            a_fft = np.fft.rfft(a, n=L)
            conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
            maxc = conv_full.max()
            cheap = 2.0 * n * maxc

            # record global improvement
            if cheap < best_val - 1e-12:
                best_val = cheap
                best_a = a.copy()
                last_improve_it = it
                print(f"[Restart {restart}] iter {it} NEW best cheap = {cheap:.6f}")

            # ----- soft‑max approximation of the max (smooth) -----
            sigma = maxc * sigma_factor
            if sigma < 1e-12:
                sigma = 1e-12
            w = np.exp((conv_full - maxc) / sigma)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # ----- gradient wrt a (smooth surrogate) -----
            a_rev = a[::-1]
            w_fft = np.fft.rfft(w, n=L)
            a_rev_fft = np.fft.rfft(a_rev, n=L)
            conv_wa_rev = np.fft.irfft(w_fft * a_rev_fft, n=L)
            # slice n entries centred at index n‑1 (see analysis)
            g_a = 2.0 * conv_wa_rev[n - 1:n - 1 + n]

            # ----- map gradient to log‑space (soft‑max Jacobian) -----
            dot = np.dot(g_a, a)
            grad_x = a * (g_a - dot)

            # ----- Adam update on logits -----
            m = beta1 * m + (1.0 - beta1) * grad_x
            v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            step = alpha * m_hat / (np.sqrt(v_hat) + eps_adam)
            x -= step

            # ----- temperature decay -----
            sigma_factor = max(sigma_factor * 0.9999, sigma_min_factor)

            # ----- simple stagnation detection -----
            if it - last_improve_it > 200_000:
                # random perturbation and restart the log‑variables
                perturb = 0.02 * np.random.rand(n)
                x = np.log(_simplex_projection(best_a + perturb) + eps)
                m.fill(0.0)
                v.fill(0.0)
                last_improve_it = it
                print(f"[Restart {restart}] Iter {it} – stagnation restart")

            # occasional light checkpoint (prevent overrunning the deadline)
            if it % 5000 == 0 and time.time() >= deadline:
                break

        restart += 1

    # --------------------------------------------------------------
    # Phase 2 – final exact evaluation
    # --------------------------------------------------------------
    if best_a is None:
        best_a = _uniform_seq(n)

    final_seq = best_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    return final_seq
