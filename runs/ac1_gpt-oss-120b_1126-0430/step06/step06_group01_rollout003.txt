#!/usr/bin/env python
"""
A fast projected‑gradient / Adam optimiser for the max‑convolution
objective.

The goal is to minimise  

    2·n·max(a*a) / (sum a)²

over non‑negative vectors a (the sum is forced to 1, which makes the
objective scale‑invariant).  The problem is convex:  max_i (a*a)_i is a
convex quadratic form and the feasible set is a simplex.  We therefore
use a smooth approximation of the max (soft‑max with an adaptive
temperature) and perform Adam updates on the logits of a soft‑max
parameterisation.  The algorithm runs for as long as the time budget
allows (default 1000 s) and keeps the best vector seen.

The implementation is completely self‑contained, uses only NumPy and
no closures or lambdas, and respects the required function signature.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable softmax, returns a probability vector (sum=1)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Inverse of soft‑max (up to an additive constant)."""
    return np.log(a + eps)


def _fft_len_for_conv(n: int) -> int:
    """Smallest power‑of‑two ≥ 2·n‑1 (sufficient for a linear self‑conv)."""
    return 1 << ((2 * n - 1).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a (no gradient)."""
    n = a.shape[0]
    L = _fft_len_for_conv(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Smoothed max of a*a and its gradient w.r.t. a.

    Returns
    -------
    max_conv : float
        Maximum entry of the linear self‑convolution.
    grad_a : np.ndarray
        Gradient of 2·n·max_conv (the objective used in the competition)
        with respect to a.  Already multiplied by 2·n.
    """
    n = a.shape[0]
    L = _fft_len_for_conv(n)

    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # smooth the max with a temperature proportional to max_conv
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv               # improves numerical stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # cross‑correlation a ⋆ w (linear, length n)
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # gradient of 2·n·max_conv = 4·n·corr
    grad_a = 4.0 * n * corr
    return max_conv, grad_a


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Returned array sums to 1 and is non‑negative.
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection only works on 1‑D arrays")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w

# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimises the
    evaluation metric.

    Strategy
    --------
    1.  Choose a problem size n (use the length of the supplied
        ``height_sequence_1`` if it exists, otherwise default to 32768).
        The chosen n keeps the FFT length at 65536, which is fast.
    2.  Initialise the vector with the exact discrete arcsine law
        (the known near‑optimal construction).
    3.  Optimise the vector on the probability simplex using Adam on the
        logits of a soft‑max parametrisation.  The objective is a smoothed
        max of the self‑convolution; the smoothing temperature is reduced
        exponentially.
    4.  The optimisation runs for as many iterations as the time budget
        permits, keeping the best vector seen (lowest cheap metric).
    5.  The final vector is returned as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for final printing / return
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # 0. Determine problem size
    # ------------------------------------------------------------------
    try:
        # If the previous run exported a sequence we reuse its length.
        n = len(globals()["height_sequence_1"])  # type: ignore[name-defined]
        print(f"Using external length n = {n}")
    except Exception:
        # 32768 is the largest n that still fits into a 65536‑point FFT.
        n = 32768
        print(f"No external length – using default n = {n}")

    # ------------------------------------------------------------------
    # 1. Build the exact discrete arcsine distribution (our starting point)
    # ------------------------------------------------------------------
    i = np.arange(n, dtype=np.float64)
    x0 = i / n
    x1 = (i + 1) / n

    sqrt0 = np.sqrt(np.clip(x0, 0.0, 1.0))
    sqrt1 = np.sqrt(np.clip(x1, 0.0, 1.0))
    F0 = (2.0 / np.pi) * np.arcsin(sqrt0)
    F1 = (2.0 / np.pi) * np.arcsin(sqrt1)

    a0 = F1 - F0                     # bin probabilities, sum ≈ 1
    a0_sum = a0.sum()
    if not np.isfinite(a0_sum) or a0_sum <= 0.0:
        raise RuntimeError("Failed to generate arcsine bins.")
    a0 /= a0_sum                     # enforce exact normalisation

    # ------------------------------------------------------------------
    # 2. Optimisation parameters (Adam + soft‑max)
    # ------------------------------------------------------------------
    # Initialise logits from the starting distribution
    z = _logits_from_distribution(a0)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_alpha = 0.8                     # initial Adam step‑size factor
    eps_factor = 0.25                     # initial temperature factor
    eps_factor_min = 5e-6                # lower bound for the temperature
    eps_decay = 0.9995                   # exponential decay per iteration

    best_a = a0.copy()
    max_conv, _ = _conv_max_and_grad(best_a, eps_factor)
    best_score = 2.0 * n * max_conv
    print(f"Initial cheap score = {best_score:.8f}")

    iteration = 0
    max_iters = 600_000   # safety upper bound; loop also stops on time

    # ------------------------------------------------------------------
    # 3. Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline and iteration < max_iters:
        iteration += 1

        # ---- current probability vector
        a_cur = _softmax(z)

        # ---- smooth max and its gradient
        max_conv, grad_a = _conv_max_and_grad(a_cur, eps_factor)
        cheap_score = 2.0 * n * max_conv

        # ---- keep the best vector seen so far
        if cheap_score < best_score:
            best_score = cheap_score
            best_a = a_cur.copy()
            if iteration % 2000 == 0:
                print(f"Iter {iteration:6d}: improved score = {best_score:.8f}")

        # ---- Adam update on logits (mirror descent on the simplex)
        mu = np.dot(grad_a, a_cur)            # <grad, a>
        grad_z = a_cur * (grad_a - mu)        # chain rule for soft‑max

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

        # bias‑corrected moments
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        # step‑size schedule (inverse‑sqrt)
        alpha_t = base_alpha / (math.sqrt(iteration) + 1.0)

        step = -alpha_t * m_hat / (np.sqrt(v_hat) + eps_adam)
        z += step
        # keep logits bounded (avoids overflow in the exp)
        z = np.clip(z, -80.0, 80.0)

        # ---- temperature (smoothing) decay
        eps_factor = max(eps_factor_min, eps_factor * eps_decay)

        # ---- occasional random jitter to escape flat regions
        if iteration % 5000 == 0 and np.random.rand() < 0.07:
            z += 0.005 * np.random.randn(n)

        # ---- periodic logging of the temperature and step size
        if iteration % 10000 == 0:
            elapsed = time.time() - start_time
            print(f"[{elapsed:5.1f}s] iter {iteration:,d} | "
                  f"score={cheap_score:.8f} | eps_factor={eps_factor:.2e}")

    # ------------------------------------------------------------------
    # 4. Final exact evaluation (for information only)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        exact_score = evaluate_sequence(final_seq)
        print(f"Exact evaluation after optimisation: {exact_score:.8f}")
    except Exception as exc:
        print(f"Exact evaluation failed: {exc}")

    return final_seq
