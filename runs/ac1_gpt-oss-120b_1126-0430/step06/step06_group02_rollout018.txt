def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import time, numpy as np, math
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0  # keep safety
    # Determine n: we can try n = 10000 initially.
    # If we have enough time after scanning, we may upsample later.
    n = 10000
    best_seq = None
    best_score = float('inf')
    # scanning exponent range
    exp_vals = np.concatenate([np.linspace(-4.0, -0.2, 39), np.linspace(0.2, 4.0, 39)])
    for p in exp_vals:
        if time.time() >= deadline:
            break
        a = _increase_power_law_distribution(n, p)
        # conv max
        max_conv = _conv_max(a)
        cheap = 2.0 * n * max_conv
        if cheap < best_score:
            best_score = cheap
            best_seq = a.copy()
            # print progress
            print(f"[Init] p={p:.3f}, cheap score={cheap:.6f}")
    # optional extra step: perform few Adam iterations on logits to refine
    # Convert best_seq to logits:
    z = _logits_from_distribution(best_seq) + 0.001 * np.random.randn(n)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_alpha = 0.25
    eps_factor = 0.2
    eps_factor_min = 1e-5
    eps_factor_decay = 0.9995
    it = 0
    last_improve_it = 0
    best_score = best_score
    best_seq = best_seq
    while time.time() < deadline and it < 300000:
        it += 1
        a = _softmax(z)
        max_conv, grad_a = _conv_max_and_grad(a, eps_factor)
        cheap = 2.0 * n * max_conv
        if cheap < best_score - 1e-12:
            best_score = cheap
            best_seq = a.copy()
            last_improve_it = it
            if it % 50000 == 0:
                print(f"[Iter {it}] New best cheap={cheap:.6f}")
        # gradient w.r.t logits
        mu = np.dot(grad_a, a)
        grad_z = a * (grad_a - mu)
        m = beta1 * m + (1-beta1) * grad_z
        v = beta2 * v + (1-beta2) * (grad_z**2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step = - m_hat / (np.sqrt(v_hat) + eps_adam)
        # line search
        accepted=False
        trial_alpha = alpha_t
        for _ in range(5):
            z_new = z + trial_alpha * step
            z_new = np.clip(z_new, -50, 50)
            a_new = _softmax(z_new)
            new_score = 2.0 * n * _conv_max(a_new)
            if new_score <= cheap:
                z = z_new
                accepted=True
                break
            trial_alpha *= 0.5
        if not accepted:
            z = z + alpha_t * step
            z = np.clip(z, -50, 50)
        eps_factor = max(eps_factor_min, eps_factor * eps_factor_decay)
        if it - last_improve_it > 200000:
            # restart jitter
            print(f"[Iter {it}] Restart due to stagnation.")
            z = _logits_from_distribution(best_seq) + 0.01 * np.random.randn(n)
            m.fill(0.0); v.fill(0.0); eps_factor = 0.2; last_improve_it=it
        # optional upsampling when time permits
        if it % 150000 == 0 and n < 20000 and time.time() + 30 < deadline:
            new_n = min(20000, int(n*1.5))
            print(f"[Iter {it}] Upsampling {n}->{new_n}")
            best_seq = _interpolate_distribution(best_seq, new_n)
            z = _logits_from_distribution(best_seq) + 0.005*np.random.randn(new_n)
            m = np.zeros_like(z); v = np.zeros_like(z)
            n = new_n
            # reset other variables
            it = 0
            last_improve_it = 0
            eps_factor = 0.2
    # final evaluation using evaluate_sequence
    final_seq = best_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as e:
        final_score = float('inf')
        print(f"Evaluation error: {e}")
    return final_seq

#!/usr/bin/env python
"""
Simple‑search optimiser for the step‑function benchmark.

The algorithm proceeds in two stages.

1. **Coarse parametric sweep** – we generate monotone power‑law
   distributions a_i ∝ (i+1)^p for a wide range of exponents p
   (both increasing and decreasing shapes).  For each candidate we
   compute a cheap upper‑bound 2·n·max(a*a) using an FFT‑based
   convolution.  The best‑scoring distribution becomes the starting
   point.

2. **Brief Adam refinement** – the chosen distribution is turned into
   logits (the inverse of a soft‑max) and a short Adam optimisation
   (mirror descent on the simplex) is performed.  The gradient is
   obtained from a smooth approximation of the max‑operator, exactly as
   in the reference implementation, but the initial point is now much
   closer to the global optimum, so only a few thousand updates are
   required.

The whole routine respects the supplied time budget (default 1000 s) and
returns the best integer‑free list of heights found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max, returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    s = e.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / z.size)
    return e / s


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it that reproduces ``a`` after a soft‑max."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """Smallest power‑of‑two ≥ 3·n‑2 (enough for all needed convolutions)."""
    return 1 << ((3 * n - 2).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Returns (max_conv, grad_a) where

        grad_a = ∇_a [ 2·n·smooth_max(a*a) ]

    `eps_factor` multiplies the smoothing constant ε = eps_factor·max_conv.
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    # FFT of a (zero‑padded)
    a_fft = np.fft.rfft(a, L)

    # Linear self‑convolution a*a
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # Smoothed soft‑max weights
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv                # ≤ 0 for numerical stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # Correlation a ⋆ w
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # Gradient (already multiplied by 4·n)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    s = new_a.sum()
    if s == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= s
    return new_a


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{p}  (p may be negative – decreasing shape)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** p
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _exp_decay_distribution(n: int, lam: float) -> np.ndarray:
    """a_i ∝ e^{‑lam·i}+e^{‑lam·(n‑1‑i)}, symmetric exponential."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _logistic_distribution(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ 1/(1+e^{α·(i‑mid)}), left‑heavy logistic."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = 1.0 / (1.0 + np.exp(alpha * (i - mid)))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) improves the
    `evaluate_sequence` score.  The routine respects the given time budget.
    """
    np.random.seed(seed)
    start = time.time()
    # keep a small safety margin for the final exact evaluation
    deadline = start + budget_s - 2.0

    # --------------------------------------------------------------
    # 0. Choose a modest support size (fast FFTs) and initialise.
    # --------------------------------------------------------------
    n = 8000                     # a good trade‑off between resolution and speed
    best_a = None
    best_score = float("inf")

    # -----------------------------------------------------------------
    # 0a. Scan a family of monotone power‑law shapes (both directions).
    # -----------------------------------------------------------------
    # Exponents p > 0 give increasing shapes (mass at the right end);
    # p < 0 give decreasing shapes (mass at the left end).  The optimum
    # for our problem lies in the increasing regime, but we also scan the
    # decreasing side for completeness.
    print("[Init] Scanning monotone power‑law family.")
    exp_vals = np.concatenate((
        np.linspace(-4.0, -0.3, 38),   # decreasing (left‑heavy)
        np.linspace(0.3, 4.0, 38)      # increasing (right‑heavy)
    ))
    for p in exp_vals:
        if time.time() >= deadline:
            break
        cand = _power_law_distribution(n, p)
        max_conv = _conv_max(cand)
        cheap_score = 2.0 * n * max_conv
        if cheap_score < best_score:
            best_score = cheap_score
            best_a = cand.copy()
            print(f"   p={p:6.3f} → cheap = {cheap_score:.6f}")

    # -----------------------------------------------------------------
    # 0b. Add a few classic families (exp‑decay, logistic) for safety.
    # -----------------------------------------------------------------
    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_score = 2.0 * n * _conv_max(best_a)

    # exponential decay (symmetric)
    for lam in np.logspace(-6, -2, 15):
        if time.time() >= deadline:
            break
        cand = _exp_decay_distribution(n, lam)
        cheap_score = 2.0 * n * _conv_max(cand)
        if cheap_score < best_score:
            best_score, best_a = cheap_score, cand
            print(f"   exp lam={lam:.2e} → cheap = {cheap_score:.6f}")

    # logistic (left‑heavy)
    for alpha in np.linspace(0.1, 5.0, 30):
        if time.time() >= deadline:
            break
        cand = _logistic_distribution(n, alpha)
        cheap_score = 2.0 * n * _conv_max(cand)
        if cheap_score < best_score:
            best_score, best_a = cheap_score, cand
            print(f"   logistic α={alpha:.3f} → cheap = {cheap_score:.6f}")

    # --------------------------------------------------------------
    # 1. Initialise Adam on the simplex (mirror descent via logits)
    # --------------------------------------------------------------
    # Slight random jitter to avoid exact symmetry issues.
    z = _logits_from_distribution(best_a) + 0.01 * np.random.randn(n)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    base_alpha = 0.45          # base learning‑rate for logits
    eps_factor = 0.20          # ε = eps_factor·max_conv (smooth max)
    eps_factor_min = 5e-5
    eps_factor_decay = 0.9995

    best_seq = _softmax(z)
    best_score = 2.0 * n * _conv_max(best_seq)
    last_improve_iter = 0
    it = 0

    # --------------------------------------------------------------
    # 2. Short optimisation loop (few‑hundred‑thousand updates)
    # --------------------------------------------------------------
    while time.time() < deadline and it < 400_000:
        it += 1

        # current point on simplex
        a = _softmax(z)

        # cheap evaluation + smoothed gradient
        max_conv, grad_a = _conv_max_and_grad(a, eps_factor)
        cheap_score = 2.0 * n * max_conv

        # keep the best ever seen
        if cheap_score < best_score - 1e-12:
            best_score = cheap_score
            best_seq = a.copy()
            last_improve_iter = it
            if it % 50000 == 0:
                print(f"[Iter {it:07d}] NEW BEST cheap = {cheap_score:.6f}")

        # gradient w.r.t. logits (mirror descent)
        mu = np.dot(grad_a, a)               # <grad, a>
        grad_z = a * (grad_a - mu)

        # Adam moment updates (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step‑size (inverse‑sqrt decay)
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step_dir = - m_hat / (np.sqrt(v_hat) + eps_adam)

        # simple back‑tracking line search (up to 5 halvings)
        accepted = False
        trial_alpha = alpha_t
        for _ in range(5):
            z_new = z + trial_alpha * step_dir
            z_new = np.clip(z_new, -50.0, 50.0)
            a_new = _softmax(z_new)
            new_score = 2.0 * n * _conv_max(a_new)
            if new_score <= cheap_score:
                z = z_new
                accepted = True
                break
            trial_alpha *= 0.5

        if not accepted:
            # take a modest step anyway – prevents stagnation.
            z = z + alpha_t * step_dir
            z = np.clip(z, -50.0, 50.0)

        # occasional jitter to shake out of plateaux
        if (it % 3000 == 0) and (np.random.rand() < 0.12):
            z += 0.02 * np.random.randn(n)
            m *= 0.6
            v *= 0.6

        # reduce smoothing gradually
        eps_factor = max(eps_factor_min, eps_factor * eps_factor_decay)

        # restart if we have been stuck for a while
        if it - last_improve_iter > 250_000:
            print(f"[Iter {it}] Stagnation – jitter restart.")
            z = _logits_from_distribution(best_seq) + 0.03 * np.random.randn(n)
            m.fill(0.0)
            v.fill(0.0)
            eps_factor = 0.20
            last_improve_iter = it

        # one optional up‑sampling stage (only if time permits)
        if (it % 150_000 == 0) and (n < 20000) and (time.time() + 30.0 < deadline):
            new_n = min(20000, int(n * 1.4))
            print(f"[Iter {it}] Upsampling support: {n} → {new_n}")
            best_seq = _interpolate_distribution(best_seq, new_n)
            z = _logits_from_distribution(best_seq) + 0.005 * np.random.randn(new_n)
            m = np.zeros_like(z)
            v = np.zeros_like(z)
            n = new_n
            best_score = 2.0 * n * _conv_max(best_seq)
            it = 0
            last_improve_iter = 0
            eps_factor = 0.20

    # --------------------------------------------------------------
    # 3. Final exact evaluation with the provided evaluate_sequence()
    # --------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # --------------------------------------------------------------
    # 4. Return the best sequence discovered.
    # --------------------------------------------------------------
    return final_seq
