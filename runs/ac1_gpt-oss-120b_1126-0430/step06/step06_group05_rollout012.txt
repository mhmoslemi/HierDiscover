def _run_opt_smooth_adam(n, time_limit, init_a, seed, alpha_start=0.5, alpha_end=200.0):

    start = time.time()
    rng = np.random.default_rng(seed)

    # set up a (logits) as before (softmax of logit)
    if init_a is None: a = rng.random(n)
    else: a = np.array(init_a) of length n, else upsample.

    Normalize a.

    x = np.log(a + 1e-16)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    best_a = a.copy()
    best_obj = evaluate_sequence(a.tolist()) # compute hard metric

    it = 0

    while time.time() - start < time_limit:
        it += 1

        # get a = softmax(x)
        a = np.exp(x - np.max(x))
        a = a / a.sum()

        # compute conv = self convolution via FFT
        conv = _conv_self_fft(a)
        # compute smoothed objective at current alpha
        # schedule alpha based on elapsed fraction
        elapsed = time.time() - start
        frac = min(elapsed / time_limit, 1.0)
        alpha = alpha_start * (alpha_end/alpha_start) ** frac  # exponential increase
        # approx max via log-sum-exp
        # compute L = logsumexp(alpha*conv)/alpha
        # can compute: maxconv = conv.max()
        # shift: d = conv - maxconv
        maxc = conv.max()
        # compute logsumexp scaled:
        lse = maxc * alpha + np.log(np.exp(alpha*(conv - maxc)).sum())
        L = lse / alpha
        obj_smooth = 2 * n * L

        # compute softmax weights w = exp(alpha*(conv - maxc)) / sum_exp
        exp_alpha = np.exp(alpha * (conv - maxc))
        sum_exp = exp_alpha.sum()
        w = exp_alpha / sum_exp

        # compute gradient of L wrt a: grad_a = 2 * (a_rev * w)[slice] (as derived)
        a_rev = a[::-1]
        L_fft_len = _next_pow2(len(a_rev) + len(w) - 1)
        Af = np.fft.rfft(a_rev, n=L_fft_len)
        Wf = np.fft.rfft(w, n=L_fft_len)
        corr = np.fft.irfft(Af * Wf, n=L_fft_len)
        # relevant slice:
        start_idx = n - 1
        grad_a = 2.0 * corr[start_idx : start_idx + n]

        # gradient w.r.t logits:
        dot = np.dot(a, grad_a)
        g_x = a * (grad_a - dot)

        # Adam update (learning rate schedule)
        # lr schedule could be constant or decaying. Use similar schedule as before
        lr_start = 0.5
        lr_end = 0.001
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * frac)

        # moments
        m = beta1 * m + (1-beta1) * g_x
        v = beta2 * v + (1-beta2) * (g_x**2)

        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)

        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # Occasionally compute hard objective and store best
        if it % 1000 == 0 or time.time() - start > time_limit - 1.0:
            # compute hard obj
            hard_obj = evaluate_sequence(a.tolist())
            if hard_obj < best_obj:
                best_obj = hard_obj
                best_a = a.copy()
                print(f"[n={n} it={it}] new best hard O={best_obj:.6f} (alpha={alpha:.2f})")

        # occasional jitter restart similar to earlier
        if it % 5000 == 0:
            jitter = rng.normal(scale=0.01, size=n)
            x += jitter
        if it % 80000 == 0:
            x = np.log(best_a + 1e-16) + rng.normal(scale=0.005, size=n)
            m.fill(0); v.fill(0)

    # after loop, compress negligible entries
    # similar as earlier
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = evaluate_sequence(best_a.tolist())
        print(f"[compress] reduced length -> {best_a.size}, O={best_obj:.6f}")

    return best_a.tolist(), best_obj

def upsample_sequence(seq, target_n):
    x_old = np.linspace(0,1, len(seq), endpoint=False)
    x_new = np.linspace(0,1, target_n, endpoint=False)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.clip(new_seq, 0, None)
    new_seq = new_seq / new_seq.sum()
    return new_seq

"""
Annealed Adam optimiser for the max‑autocorrelation problem.

The optimiser works directly on the simplex by mirror‑descent in
log‑probability space.  The objective “max(conv)” is approximated by a
smooth log‑sum‑exp with a temperature α that is increased
exponentially during the run (annealing).  The exact gradient of the
smoothed objective is obtained from two FFT‑based convolutions, so each
iteration costs O(n log n).  Several phases of increasing support length
are performed (coarse → fine), each using a time‑budget.  The final
solution is optionally refined by a short simulated‑annealing local
search.

The routine respects the time budget of the challenge, prints progress
and returns a list of non‑negative floats suitable for ``evaluate_sequence``.
"""
import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# FFT convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    # keep only the first 2n‑1 entries (full linear convolution)
    return conv_full[: 2 * n - 1]


def _conv_rev_weight_fft(a_rev: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Convolution of ``a_rev`` (length n) with ``w`` (length 2n‑1) using FFT.
    Returns the full linear convolution (length 3n‑2).
    """
    n = a_rev.shape[0]
    m = w.shape[0]          # = 2n‑1
    L = _next_pow2(n + m - 1)
    Af = np.fft.rfft(a_rev, n=L)
    Wf = np.fft.rfft(w, n=L)
    conv_full = np.fft.irfft(Af * Wf, n=L)
    return conv_full


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _smooth_max_and_grad(a: np.ndarray, alpha: float) -> Tuple[float, np.ndarray]:
    """
    Returns (smooth_max, grad_a) where:

    * smooth_max = (log ∑_k exp(alpha·conv_k)) / alpha,
      i.e. a smooth approximation of max(conv).
    * grad_a = ∂ smooth_max / ∂ a  (shape ``a``).

    The gradient is computed analytically by convolving the softmax
    weights with the reversed ``a`` (see derivation in the analysis).
    """
    n = a.shape[0]
    # self‑convolution
    conv = _conv_self_fft(a)                 # length 2n‑1
    # log‑sum‑exp in a numerically stable way
    maxc = conv.max()
    exp_alpha = np.exp(alpha * (conv - maxc))
    sum_exp = exp_alpha.sum()
    # smooth approximation of the max
    smooth_max = (maxc * alpha + np.log(sum_exp)) / alpha

    # softmax weights (probability distribution over k)
    w = exp_alpha / sum_exp                 # length 2n‑1, sum(w)=1

    # gradient ∂smooth_max/∂a = 2 * (a_rev * w)[n‑1 : n‑1+n]
    a_rev = a[::-1]
    conv_rev_w = _conv_rev_weight_fft(a_rev, w)  # length 3n‑2
    start = n - 1
    grad_a = 2.0 * conv_rev_w[start : start + n]
    return smooth_max, grad_a, conv


def _run_opt_annealed_adam(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
) -> Tuple[List[float], float]:
    """
    Adam optimiser that minimises the hard max of the self‑convolution
    by gradually annealing a smooth approximation.

    Returns the best sequence (as a Python list) and its hard objective
    ``2·n·max(conv)``.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # Initialise a on the simplex
    if init_a is not None and init_a.size == n:
        a = np.asarray(init_a, dtype=float).copy()
    else:
        a = rng.random(n)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n, dtype=float) / n
    else:
        a = a / s

    # Mirror‑descent variables (log‑space)
    x = np.log(a + 1e-16)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # Learning‑rate schedule (exponential decay)
    lr_start = 0.7
    lr_end = 0.001

    # Annealing schedule for the soft‑max temperature
    alpha_start = 0.5
    alpha_end = 2000.0

    best_a = a.copy()
    # Hard objective (sum(a)=1 ⇒ O = 2·n·max(conv))
    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())
    print(f"[n={n}] initial hard O={best_obj:.6f}")

    it = 0
    last_improve_it = 0
    while time.time() - start < time_limit:
        it += 1

        # Current point on the simplex
        a = np.exp(x - np.max(x))
        a = a / a.sum()

        # Fraction of the allotted time that has elapsed
        elapsed = time.time() - start
        frac = min(elapsed / time_limit, 1.0)

        # Learning‑rate (exponential decay)
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * frac)

        # Annealing temperature (exponential increase)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac

        # Smooth max and its gradient
        smooth_max, grad_a, conv = _smooth_max_and_grad(a, alpha)

        # Mirror‑gradient in log‑space
        dot = np.dot(a, grad_a)
        g_x = a * (grad_a - dot)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # Update log‑variables
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # Occasionally evaluate the hard objective (cost‑free as we have conv)
        if it % 2000 == 0 or time.time() - start > time_limit - 1.0:
            hard_obj = float(2.0 * n * conv.max())
            if hard_obj < best_obj - 1e-14:
                best_obj = hard_obj
                best_a = a.copy()
                last_improve_it = it
                print(f"[n={n} it={it:07d}] new best hard O={best_obj:.6f} α={alpha:.2f}")

        # Jitter to escape flat plateaus
        if it % 5000 == 0:
            jitter = rng.normal(scale=0.01, size=n)
            x += jitter

        # Full restart from the current best every so often
        if it % 80000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.005, size=n)
            m.fill(0.0)
            v.fill(0.0)

        # Simple stagnation detection – if no improvement for a long while,
        # re‑initialise around the best point.
        if it - last_improve_it > 200000:
            # add a moderate random perturbation
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.02, size=n)
            last_improve_it = it
            print(f"[n={n}] stagnation reset at it={it}")

    # ------------------------------------------------------------------
    # Compress negligible entries (they waste length without helping)
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        conv = _conv_self_fft(best_a)
        best_obj = float(2.0 * best_a.size * conv.max())
        print(f"[compress] reduced length → {best_a.size}, hard O={best_obj:.6f}")

    return best_a.tolist(), best_obj

def _run_sa(
    init_seq: List[float],
    time_limit: float,
    seed: int = 0,
) -> Tuple[List[float], float]:
    """Local refinement by mass‑transfer moves with a Metropolis schedule."""
    start = time.time()
    rng = np.random.default_rng(seed)
    a = np.asarray(init_seq, dtype=float).copy()
    if a.sum() < 1e-12:
        a = np.ones_like(a) / a.size
    else:
        a = a / a.sum()
    n = a.shape[0]

    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())
    best_seq = a.copy()

    # Temperature schedule (starts high, decays geometrically)
    T = best_obj * 0.2 + 1e-6
    T_min = best_obj * 1e-6 + 1e-9
    decay = 0.995
    it = 0

    while time.time() - start < time_limit:
        it += 1
        # locate the current maximal convolution entry
        k_max = int(np.argmax(conv))

        # contribution of each index to that entry
        i_vals = np.arange(n)
        j_vals = k_max - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i_vals[mask]] * a[j_vals[mask]]
        total_contrib = contrib.sum()

        # pick a donor proportional to its contribution (or uniformly)
        if total_contrib > 0:
            probs = contrib / total_contrib
            i = rng.choice(n, p=probs)
        else:
            i = rng.integers(n)

        # pick a distinct recipient uniformly
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # transfer a modest fraction of mass
        delta = a[i] * rng.random() * 0.07  # up to 7 % of donor
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.clip(a_new, 0.0, None)
        a_new /= a_new.sum()

        conv_new = _conv_self_fft(a_new)
        obj_new = float(2.0 * n * conv_new.max())

        # Metropolis acceptance
        if obj_new < best_obj - 1e-14:
            a, conv, best_obj, best_seq = a_new, conv_new, obj_new, a_new.copy()
            print(f"[SA it={it}] new best hard O={best_obj:.6f}")
        else:
            prob = np.exp(-(obj_new - best_obj) / T) if T > 0 else 0.0
            if rng.random() < prob:
                a, conv = a_new, conv_new

        # Temperature decay
        if it % 2500 == 0:
            T = max(T * decay, T_min)

        # Global jitter every now and then
        if it % 15000 == 0:
            noise = rng.normal(scale=0.01, size=n)
            a = np.clip(a + noise, 0.0, None)
            a /= a.sum()
            conv = _conv_self_fft(a)

    # Final compression of near‑zero entries
    thresh = 1e-12
    mask = best_seq > thresh
    if mask.sum() < best_seq.size:
        comp = best_seq[mask]
        comp = comp / comp.sum()
        conv = _conv_self_fft(comp)
        comp_obj = float(2.0 * comp.size * conv.max())
        if comp_obj < best_obj:
            best_seq, best_obj = comp, comp_obj
            print(f"[SA compress] reduced length → {comp.size}, hard O={best_obj:.6f}")

    return best_seq.tolist(), best_obj

def _upsample_sequence(seq: np.ndarray, target_n: int) -> np.ndarray:
    """Linear interpolation from ``len(seq)`` to ``target_n`` (periodic interpretation)."""
    old_n = seq.shape[0]
    if old_n == target_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, target_n, endpoint=False)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.clip(new_seq, 0.0, None)
    new_seq = new_seq / new_seq.sum()
    return new_seq

            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.02, size=n)
            last_improve_it = it
            print(f"[n={n}] stagnation reset at it={it}")

    # ------------------------------------------------------------------
    # prune negligible entries
    # ------------------------------------------------------------------
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        conv = _conv_self_fft(best_a)
        best_obj = float(2.0 * best_a.size * conv.max())
        print(f"[compress] reduced length → {best_a.size}, hard O={best_obj:.6f}")

    return best_a.tolist(), best_obj


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (unchanged except for a few parameters)
# ----------------------------------------------------------------------


def _run_sa(
    init_seq: List[float],
    time_limit: float,
    seed: int = 0,
) -> Tuple[List[float], float]:
    """Refine a sequence by small mass‑transfer moves with a Metropolis schedule."""
    start = time.time()
    rng = np.random.default_rng(seed)
    a = np.asarray(init_seq, dtype=float).copy()
    if a.sum() < 1e-12:
        a = np.ones_like(a) / a.size
    else:
        a = a / a.sum()
    n = a.shape[0]

    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())
    best_seq = a.copy()

    # temperature schedule
    T = best_obj * 0.2 + 1e-6
    T_min = best_obj * 1e-6 + 1e-9
    decay = 0.995
    it = 0

    while time.time() - start < time_limit:
        it += 1
        k_max = int(np.argmax(conv))

        i_vals = np.arange(n)
        j_vals = k_max - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i_vals[mask]] * a[j_vals[mask]]
        total_contrib = contrib.sum()

        if total_contrib > 0:
            probs = contrib / total_contrib
            i = rng.choice(n, p=probs)
        else:
            i = rng.integers(n)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.07
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.clip(a_new, 0.0, None)
        a_new /= a_new.sum()

        conv_new = _conv_self_fft(a_new)
        obj_new = float(2.0 * n * conv_new.max())

        if obj_new < best_obj - 1e-14:
            a, conv, best_obj, best_seq = a_new, conv_new, obj_new, a_new.copy()
            print(f"[SA it={it}] new best hard O={best_obj:.6f}")
        else:
            prob = np.exp(-(obj_new - best_obj) / T) if T > 0 else 0.0
            if rng.random() < prob:
                a, conv = a_new, conv_new

        if it % 2500 == 0:
            T = max(T * decay, T_min)

        if it % 15000 == 0:
            noise = rng.normal(scale=0.01, size=n)
            a = np.clip(a + noise, 0.0, None)
            a /= a.sum()
            conv = _conv_self_fft(a)

    # final compression of tiny entries
    thresh = 1e-12
    mask = best_seq > thresh
    if mask.sum() < best_seq.size:
        comp = best_seq[mask]
        comp = comp / comp.sum()
        conv = _conv_self_fft(comp)
        comp_obj = float(2.0 * comp.size * conv.max())
        if comp_obj < best_obj:
            best_seq, best_obj = comp, comp_obj
            print(f"[SA compress] reduced length → {comp.size}, hard O={best_obj:.6f}")

    return best_seq.tolist(), best_obj


# ----------------------------------------------------------------------
# Upsampling helper
# ----------------------------------------------------------------------


def _upsample_sequence(seq: np.ndarray, target_n: int) -> np.ndarray:
    """Linear interpolation from current length to ``target_n`` (periodic)."""
    old_n = seq.shape[0]
    if old_n == target_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, target_n, endpoint=False)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.clip(new_seq, 0.0, None)
    new_seq = new_seq / new_seq.sum()
    return new_seq


# ----------------------------------------------------------------------
# Main entry point required by the challenge
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **_ignore,
) -> List[float]:
    """
    Optimises a non‑negative step‑function for the given evaluation metric.
    The method consists of:

    * a multi‑phase annealed Adam optimisation on increasingly large
      supports (≈6 k → 12 k → 20 k);
    * a short simulated‑annealing refinement;
    * a final sanity check using ``evaluate_sequence``.

    The routine respects the supplied time budget and returns the best
    sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Initialise from a previously discovered construction if available.
    # ------------------------------------------------------------------
    init_seq = None
    if 'height_sequence_1' in globals():
        seq0 = np.asarray(globals()['height_sequence_1'], dtype=float)
        seq0 = np.clip(seq0, 0.0, None)
        if seq0.sum() < 0.01:
            seq0 = np.ones_like(seq0)
        seq0 = seq0 / seq0.sum()
        init_seq = seq0.tolist()

    # ------------------------------------------------------------------
    # Phase schedule (support sizes and time allocation)
    # ------------------------------------------------------------------
    support_sizes = [6000, 12000, 20000]           # feel free to adjust
    # 85 % of the remaining budget for Adam phases, rest for SA
    main_budget = (deadline - time.time()) * 0.85
    sa_budget = deadline - time.time() - main_budget

    # allocate time proportionally to FFT‑complexity n·log₂(n)
    complexities = [n * np.log2(max(n, 2)) for n in support_sizes]
    total_c = sum(complexities)
    phase_budgets = [main_budget * (c / total_c) for c in complexities]

    best_seq = None
    best_obj = float('inf')

    # ------------------------------------------------------------------
    # Run phases sequentially, up‑sampling the result each time.
    # ------------------------------------------------------------------
    for idx, (n, phase_time) in enumerate(zip(support_sizes, phase_budgets)):
        if time.time() >= deadline:
            break
        print(f"\n=== Phase {idx + 1}: n={n}, time budget {phase_time:.1f}s ===")
        if init_seq is not None:
            init_a = _upsample_sequence(np.asarray(init_seq, dtype=float), n)
        else:
            init_a = None

        seq, obj = _run_opt_annealed_adam(
            n=n,
            time_limit=phase_time,
            init_a=init_a,
            seed=seed + idx * 12345,
        )
        if obj < best_obj:
            best_obj, best_seq = obj, np.asarray(seq, dtype=float)
            print(f"[global] after phase {idx + 1} best hard O={best_obj:.6f} (n={n})")

        # warm‑start next phase with the current best
        init_seq = seq

    # ------------------------------------------------------------------
    # Simulated‑annealing refinement if time permits
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0 and best_seq is not None:
        print(f"\n=== Simulated‑annealing refinement (≈{remaining:.1f}s left) ===")
        seq_ref, obj_ref = _run_sa(best_seq.tolist(), time_limit=remaining, seed=seed + 99999)
        if obj_ref < best_obj:
            best_obj, best_seq = obj_ref, np.asarray(seq_ref, dtype=float)
            print(f"[final] after SA new best hard O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check – call the supplied evaluator.
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_seq.tolist()
