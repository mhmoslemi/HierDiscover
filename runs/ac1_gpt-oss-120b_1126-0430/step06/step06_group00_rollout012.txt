def _conv_max_and_hard_grad(a):
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[:2*n-1]
    max_idx = np.argmax(conv)
    # compute gradient
    # Create array grad of zeros
    grad = np.zeros_like(a)
    # for j such that 0 <= max_idx - j < n => i.e., j in [max_idx - (n-1), max_idx]
    start_j = max(0, max_idx - (n-1))
    end_j = min(n-1, max_idx)
    # For these j, grad[j] = 2 * a[max_idx - j]
    # We can slice:
    # indices j: start_j..end_j inclusive.
    # set grad[start_j:end_j+1] = 2 * a[max_idx - np.arange(start_j, end_j+1)]
    gradslice = 2 * a[max_idx - np.arange(start_j, end_j+1)]
    grad[start_j:end_j+1] = gradslice
    max_conv = conv[max_idx]
    return max_conv, grad

"""
Propose a step‑function (non‑negative sequence) that minimizes the metric

    2·n·max(a*a) / (sum(a)**2)

using a long Adam optimisation on the simplex with a decreasing
soft‑max smoothing of the max‑convolution.  The algorithm starts from the
exact discretisation of the arcsine distribution (known to be a strong
continuous candidate) and then performs many thousands of gradient steps.
The smoothing parameter is driven to very small values, effectively
optimising the true max‑convolution.  After the main phase a cheap
local random‑swap refinement is applied while time permits.

The implementation is fully self‑contained, uses only NumPy and the
provided ``evaluate_sequence`` for final logging, and respects the 1000 s
budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max that returns a probability vector (sum = 1)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _fft_len_for_corr(n: int) -> int:
    """
    Smallest power‑of‑two ≥ 3·n‑2.
    This length is sufficient for linear convolution (2·n‑1) and for the
    correlation of length 3·n‑2 used in the gradient computation.
    """
    return 1 << ((3 * n - 2).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a (via FFT)."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Smoothed max of a*a and its gradient w.r.t. a.

    Parameters
    ----------
    a : np.ndarray
        Non‑negative vector (implicitly on the simplex, sum≈1).
    eps_factor : float
        Multiplicative factor for the smoothing constant:
        eps = max(eps_factor * max(conv), 1e‑12).

    Returns
    -------
    max_conv : float
        Hard maximum of the convolution (used for the cheap score).
    grad_a : np.ndarray
        Gradient of 2·n·smooth_max w.r.t. a (already multiplied by 4·n).
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    a_fft = np.fft.rfft(a, L)

    # linear self‑convolution (length 2·n‑1)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # soft‑max weights over the convolution entries
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv               # for numeric stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or np.isnan(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # cross‑correlation a ⋆ w (first n entries are needed)
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # gradient (the factor 4·n comes from the derivative of 2·n·max)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a


def _conv_max_and_hard_grad(a: np.ndarray):
    """
    Exact max of a*a and gradient of that max (sub‑gradient) w.r.t. a.

    Returns
    -------
    max_conv : float
        Maximum entry of the convolution.
    grad_a : np.ndarray
        Gradient of 2·n·max_conv w.r.t. a (i.e. 4·n times the sub‑gradient).
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_idx = int(np.argmax(conv))
    max_conv = float(conv[max_idx])

    # Sub‑gradient of the hard max:
    # (a*a)_i = Σ_j a_j a_{i-j}; d/d a_k = a_{i-k} + a_{i-(i-k)} = 2·a_{i-k} (if in range)
    grad = np.zeros_like(a)
    low = max_idx - (n - 1)          # smallest j that contributes
    high = min(max_idx, n - 1)       # largest j that contributes

    j_start = max(0, low)
    j_end = high                     # inclusive

    # vector of indices in a that appear in the gradient
    idx = max_idx - np.arange(j_start, j_end + 1)
    grad[j_start:j_end + 1] = 2.0 * a[idx]

    # Include the coefficient 4·n as in the smoothed version
    grad_a = 4.0 * n * grad
    return max_conv, grad_a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """

    evaluation metric.  The routine runs for up to ``budget_s`` seconds,
    performing a long Adam optimisation on the probability simplex
    (softmax‑parameterised logits).  The smoothing of the max‑convolution is
    driven to a tiny value, turning the gradient into a true sub‑gradient of
    the hard max.  A cheap random‑swap local search is performed with any
    remaining time.
    """
    np.random.seed(seed)
    start_time = time.time()
    # a small safety margin for final logging / cleanup
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # 0. Choose the length of the sequence
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
        print(f"[Info] Using external length: n = {n}")
    except Exception:
        n = 20000
        print(f"[Info] No external length – falling back to n = {n}")

    # ------------------------------------------------------------------
    # 1. Initialise from the exact arcsine CDF (area discretisation)
    # ------------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)
    x0 = idx / n
    x1 = (idx + 1) / n

    # arcsine CDF: F(x) = (2/π)·arcsin(√x) for x∈[0,1]
    def arcsine_cdf(x):
        return (2.0 / np.pi) * np.arcsin(np.sqrt(np.clip(x, 0.0, 1.0)))

    a0 = arcsine_cdf(x1) - arcsine_cdf(x0)
    a0_sum = a0.sum()
    if a0_sum == 0.0 or np.isnan(a0_sum):
        raise RuntimeError("Failed to initialise a valid arcsine distribution.")
    a0 /= a0_sum                     # now Σ a = 1

    # ------------------------------------------------------------------
    # 2. Adam optimisation on the simplex (logits = log a)
    # ------------------------------------------------------------------
    best_a = a0.copy()
    best_score = 2.0 * n * _conv_max(best_a)

    # logits for the soft‑max parametrisation
    z = np.log(best_a + 1e-16)
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    # Learning‑rate schedule parameters
    base_alpha = 1.2          # initial step size for logits
    eps_factor = 0.35         # initial smoothing factor (relative to max)
    eps_factor_min = 1e-8     # drive to a very tiny smoothing constant
    eps_decay = 0.9995        # multiplicative decay per iteration

    max_iters = 600_000        # absolute safeguard; real stop is time‑budget

    # Helper for periodic logging
    def _log_progress(it, score):
        elapsed = time.time() - start_time
        print(f"[Iter {it:6d}] best_score = {score:.8f}  elapsed = {elapsed:.1f}s")

    it = 0
    no_improve = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # Current probability vector (always sums to 1)
        a_cur = _softmax(z)

        # Choose gradient: smoothed while eps_factor > eps_factor_min,
        # otherwise use the exact sub‑gradient of the hard max.
        if eps_factor > eps_factor_min:
            max_conv, grad_a = _conv_max_and_grad(a_cur, eps_factor)
        else:
            max_conv, grad_a = _conv_max_and_hard_grad(a_cur)

        cheap_score = 2.0 * n * max_conv

        # Track the best solution seen so far
        if cheap_score < best_score - 1e-12:
            best_score = cheap_score
            best_a = a_cur.copy()
            no_improve = 0
            if it % 2000 == 0 or cheap_score < 1.02:
                _log_progress(it, best_score)
        else:
            no_improve += 1

        # Natural (mirror) gradient on the simplex
        mu = np.dot(grad_a, a_cur)          # 〈∇,a〉
        grad_z = a_cur * (grad_a - mu)      # ∇_z J

        # Adam bias‑corrected update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # Inverse‑sqrt decay of the step size
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step = - alpha_t * m_hat / (np.sqrt(v_hat) + eps_adam)

        z = z + step
        # Keep logits within a safe numeric range
        z = np.clip(z, -70.0, 70.0)

        # Reduce smoothing gradually
        eps_factor = max(eps_factor_min, eps_factor * eps_decay)

        # Occasional random jitter to escape plateaus
        if it % 5000 == 0 and np.random.rand() < 0.15:
            z += 0.01 * np.random.randn(n)

        # If we have gone many iterations without improvement,
        # shrink the base step size a little to encourage convergence.
        if no_improve > 5000:
            base_alpha *= 0.9
            no_improve = 0

    # ------------------------------------------------------------------
    # 3. Final cheap random‑swap local search (uses any remaining time)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        # Simple hill‑climbing: move a tiny mass between random bins.
        a_local = best_a.copy()
        swaps = 0
        while time.time() < deadline and swaps < 200_000:
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            # Transfer a small fraction of mass from i to j
            delta = 0.0005 * a_local[i]   # 0.05 % of the source bin
            if delta <= 0.0:
                continue
            a_candidate = a_local.copy()
            a_candidate[i] -= delta
            a_candidate[j] += delta
            # No need to re‑normalise – total mass stays 1 and non‑negativity holds.
            score = 2.0 * n * _conv_max(a_candidate)
            if score < best_score - 1e-12:
                best_score = score
                best_a = a_candidate.copy()
                a_local = a_candidate
                _log_progress(it + swaps, best_score)
            swaps += 1

    # ------------------------------------------------------------------
    # 4. Return the best sequence (plain Python list)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()

    # Optional final exact evaluation for logging only.
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[Info] Exact evaluation (platform will also compute) = {final_score:.8f}")
    except Exception as exc:   # defensive: the platform may not expose it yet
        print(f"[Warning] evaluate_sequence raised an exception: {exc}")

    print(f"[Info] Best (approximate) score achieved = {best_score:.8f}")
    return final_seq
