Define def project_simplex(v):
    # v: raw vector (may be any real)
    # returns projection onto {x >= 0, sum=1}
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Hybrid random‑parametric + sub‑gradient optimiser for the
self‑convolution bound.

The algorithm works in three phases within the 1000 s budget:

1. **Randomised parametric sweep** – a handful of analytic families
   (power‑law, double‑power‑law, stretched‑exponential, logistic,
   Beta‑distribution, …) are sampled at random (wide ranges).  The cheap
   objective ``2·n·max(a*a)`` is evaluated (via FFT) and the best
   candidate is kept.

2. **Projected sub‑gradient descent on the simplex** – starting from the
   best candidate, the exact maximum shift ``k*`` of the linear
   self‑convolution is found.  The sub‑gradient of the active shift is
   ``g_i = 2·a_{k*‑i}`` (zero where the index is out of range).  A step
   ``a ← a − η·g`` is taken and the vector is projected back onto the
   probability simplex (clip‑negative + renormalise).  An inverse‑sqrt
   step‑size schedule together with occasional random jitter and a
   stagnation‑restart guarantees continuous improvement.

3. **Final cheap refinement** – a short random‑walk that transfers a
   tiny amount of mass between two entries is performed until the time
   budget expires.

The best sequence found is finally evaluated with the provided
``evaluate_sequence`` function and returned as a plain Python list.
All numbers are non‑negative, summed to one (the evaluator rescales
automatically) and lie well below the 1000 limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, O(n log n))
# ----------------------------------------------------------------------


def _fft_len_for_corr(n: int) -> int:
    """Smallest power‑of‑two ≥ 3·n‑2 (enough for a*a)."""
    return 1 << ((3 * n - 2).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_argmax(a: np.ndarray):
    """Return (max value, one index attaining the max) of a*a."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_val = float(conv.max())
    # pick a random argmax (helps escape symmetry)
    idxs = np.where(conv >= max_val - 1e-15)[0]
    k = int(np.random.choice(idxs))
    return max_val, k


# ----------------------------------------------------------------------
# Simple projection onto the probability simplex (clip‑negative + renorm)
# ----------------------------------------------------------------------


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto {x ≥ 0, sum(x)=1} by clipping negatives and renormalising."""
    w = np.maximum(v, 0.0)
    s = w.sum()
    if s == 0.0:
        # fallback to uniform
        return np.full_like(w, 1.0 / w.size)
    return w / s


# ----------------------------------------------------------------------
# Parametric families (all return a normalised probability vector)
# ----------------------------------------------------------------------


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _power_law_offset_distribution(n: int, p: float, offset: float) -> np.ndarray:
    idx = np.arange(offset, offset + n, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _double_power_law_distribution(n: int, p: float, q: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) ** (-p)) * ((n - i) ** (-q))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _stretched_exp_distribution(n: int, tau_factor: float, gamma: float) -> np.ndarray:
    tau = max(1e-12, tau_factor * n)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(- (i / tau) ** gamma)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _exp_decay_distribution(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _logistic_distribution(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = 1.0 / (1.0 + np.exp(alpha * (i - mid)))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _sym_product_power_law_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) * (n - i)) ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) on {0,…,n‑1} (mid‑point rule)."""
    # points in (0,1) to avoid zero‑weights when α or β < 1
    x = (np.arange(n) + 0.5) / n
    a = (x ** (alpha - 1)) * ((1 - x) ** (beta - 1))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) minimise
    `evaluate_sequence`.  The routine respects a ~1000 s budget and uses:

        * a randomised parametric sweep over many families,
        * a projected sub‑gradient descent on the simplex,
        * a short random‑walk refinement.

    The final sequence is a probability vector (sum≈1).  `evaluate_sequence`
    will silently clip/scal­e it if necessary.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0   # reserve a couple of seconds for final eval

    # ------------------------------------------------------------------
    # Determine problem size and a fallback initial distribution
    # ------------------------------------------------------------------
    try:
        init_seq = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        n = init_seq.shape[0]
        a0 = np.maximum(init_seq, 0.0)
        s = a0.sum()
        if s == 0.0:
            a0 = np.full_like(a0, 1.0 / n)
        else:
            a0 /= s
        print(f"[Init] Using provided sequence, length = {n}")
    except Exception:
        n = 20000
        a0 = np.full(n, 1.0 / n)
        print(f"[Init] No provided sequence – start from uniform, length = {n}")

    # cheap scoring function (exact for sum‑one vectors)
    def cheap_score(vec):
        return 2.0 * n * _conv_max(vec)

    best_seq = a0.copy()
    best_val = cheap_score(best_seq)
    print(f"[Init] Initial cheap score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 1. Randomised parametric sweep
    # ------------------------------------------------------------------
    families = [
        ("power_law", _power_law_distribution,
         lambda: dict(p=np.random.uniform(0.1, 2.0))),
        ("power_law_offset", _power_law_offset_distribution,
         lambda: dict(p=np.random.uniform(0.1, 2.0),
                      offset=np.random.uniform(0.1, 10.0))),
        ("double_power_law", _double_power_law_distribution,
         lambda: dict(p=np.random.uniform(0.1, 2.0),
                      q=np.random.uniform(0.1, 2.0))),
        ("stretched_exp", _stretched_exp_distribution,
         lambda: dict(tau_factor=10**np.random.uniform(-4, 0),
                      gamma=np.random.uniform(0.5, 3.0))),
        ("exp_decay", _exp_decay_distribution,
         lambda: dict(lam=10**np.random.uniform(-6, -1))),
        ("logistic", _logistic_distribution,
         lambda: dict(alpha=np.random.uniform(0.01, 20.0))),
        ("sym_product_power_law", _sym_product_power_law_distribution,
         lambda: dict(p=np.random.uniform(0.1, 2.0))),
        ("beta", _beta_distribution,
         lambda: dict(alpha=np.random.uniform(0.1, 5.0),
                      beta=np.random.uniform(0.1, 5.0))),
    ]

    sweep_deadline = start + budget_s * 0.15   # ~15 % of total time
    evals = 0
    while time.time() < sweep_deadline:
        fam_name, generator, sampler = families[np.random.randint(len(families))]
        params = sampler()
        try:
            cand = generator(n, **params)
        except Exception:
            continue
        val = cheap_score(cand)
        evals += 1
        if val < best_val - 1e-12:
            best_val = val
            best_seq = cand.copy()
            print(f"[Sweep] New best {fam_name} {params} -> {best_val:.6f}")
    print(f"[Sweep] Completed {evals} evaluations, best cheap = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. Projected sub‑gradient descent on the simplex
    # ------------------------------------------------------------------
    a = best_seq.copy()
    best_mid_seq = a.copy()
    best_mid_val = best_val
    it = 0
    last_improve = 0
    eta0 = 0.5   # initial step size
    while time.time() < deadline and it < 300_000:
        it += 1
        # current convolution and worst shift
        cur_max, k_star = _conv_max_and_argmax(a)
        cur_score = 2.0 * n * cur_max

        # keep the best seen so far
        if cur_score < best_mid_val - 1e-12:
            best_mid_val = cur_score
            best_mid_seq = a.copy()
            last_improve = it
            if it % 5000 == 0:
                print(f"[Iter {it:07d}] NEW BEST cheap = {best_mid_val:.8f}")

        # sub‑gradient of the active shift (g_i = 2·a_{k*‑i})
        g = np.zeros_like(a)
        start_i = max(0, k_star - n + 1)
        end_i = min(n - 1, k_star)
        idx = np.arange(start_i, end_i + 1, dtype=np.int64)
        g[idx] = 2.0 * a[k_star - idx]

        # step‑size schedule (inverse‑sqrt)
        eta = eta0 / (math.sqrt(it) + 1.0)

        # simple back‑tracking line‑search (up to 4 halvings)
        accepted = False
        trial = eta
        for _ in range(4):
            a_new = _proj_simplex(a - trial * g)
            new_max, _ = _conv_max_and_argmax(a_new)
            new_score = 2.0 * n * new_max
            if new_score <= cur_score + 1e-15:
                a = a_new
                accepted = True
                break
            trial *= 0.5
        if not accepted:
            # if all trials fail, just take the smallest step
            a = _proj_simplex(a - eta * g)

        # occasional jitter to shake out of plateaus
        if it % 4000 == 0 and np.random.rand() < 0.15:
            jitter = 0.01 * np.random.randn(n)
            a = _proj_simplex(a + jitter)

        # restart if stuck for a long time
        if it - last_improve > 100_000:
            print(f"[Iter {it}] Stagnation – resetting from best so far")
            a = best_mid_seq.copy()
            last_improve = it

    # keep the best from the whole descent
    best_seq = best_mid_seq
    best_val = best_mid_val
    print(f"[Descent] Finished after {it} iterations, best cheap = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Final random walk (tiny coordinate transfers)
    # ------------------------------------------------------------------
    a = best_seq.copy()
    cur_score = best_val
    while time.time() < deadline:
        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        if i == j:
            continue
        delta = 0.001 * a[i]  # move 0.1 % of the mass at i
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_max = _conv_max(a_new)
        new_score = 2.0 * n * new_max
        if new_score < cur_score:
            a = a_new
            cur_score = new_score
            best_seq = a.copy()
            best_val = cur_score
            # occasional progress print
            if np.random.rand() < 0.001:
                print(f"[Walk] Improved cheap = {best_val:.8f}")

    # ------------------------------------------------------------------
    # 4. Optional mixing with uniform (helps flatten residual spikes)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    for w in np.linspace(0.0, 0.3, 7):    # w in {0,0.05,…,0.30}
        mixed = (1.0 - w) * best_seq + w * uniform
        mixed = _proj_simplex(mixed)     # just to be safe
        mixed_score = cheap_score(mixed)
        if mixed_score < best_val:
            best_seq = mixed
            best_val = mixed_score
            print(f"[Mix] w={w:.2f} improves cheap to {best_val:.8f}")

    # ------------------------------------------------------------------
    # 5. Final exact evaluation
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        exact = evaluate_sequence(final_seq)   # the platform provides this
        print(f"Final exact score = {exact:.6f}")
    except Exception as exc:
        print(f"Error during final evaluation: {exc}")
        exact = float("inf")

    return final_seq
