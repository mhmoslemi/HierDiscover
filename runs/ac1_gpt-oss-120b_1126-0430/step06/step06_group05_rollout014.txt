"""
Power‑law step‑function optimiser.

The evaluation metric is
    O(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

For a non‑negative sequence a with sum=1 the denominator cancels out, and we
only have to minimise  max(conv(a,a)).  For a power‑law distribution

    a_i ∝ (i+1)^p ,  i = 0 … n‑1

the central convolution entry (index n‑1) can be evaluated analytically and
behaves like   C(p)/n   with   C(p) = (p+1)^2·B(p+1,p+1)   where B is the beta
function.  Hence

    O ≈ 2·C(p)          (independent of n for large n).

Choosing p ≥ 1 gives O < 2, and already p=1 yields O≈4/3≈1.33.  Larger p drive
the bound down rapidly (p=3 gives O≈0.23, p=5 gives O≈0.03, …).

The optimiser builds a few candidate power‑law sequences (and a symmetric
product shape for sanity) and returns the one with the smallest observed
objective value.  The whole procedure finishes in a fraction of a second,
well inside the 1000 s budget.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Helper – compute a normalised power‑law (or product) sequence
# ----------------------------------------------------------------------


def _make_power_sequence(n: int, p: int) -> np.ndarray:
    """Return a normalised array a_i ∝ (i+1)^p."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(idx, p)
    a /= a.sum()
    return a


def _make_product_sequence(n: int) -> np.ndarray:
    """
    Symmetric “product” shape a_i ∝ (i+1)*(n-i).
    This is a quadratic, symmetric distribution that historically gives good
    autocorrelation behaviour.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i * (n + 1 - i)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **_ignore,
) -> List[float]:
    """
    Construct a deterministic power‑law sequence (or a simple quadratic
    sequence) that yields a very low value of ``evaluate_sequence``.
    The function tries a handful of exponents and returns the best
    performing sequence.
    """
    # The optimiser is essentially instantaneous; we just keep a tiny
    # safety margin for the overall time budget.
    start = time.time()
    deadline = start + budget_s - 0.5

    # ------------------------------------------------------------------
    # Choose a reasonable length.
    # 12 k is a good compromise: large enough for the asymptotic bound
    # to be tight, but still cheap to convolve.
    # If the remaining budget is tiny we fall back to a short 3 k vector.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    n = 12000 if remaining > 30 else 3000
    print(f"[setup] length = {n}")

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Candidate families
    # ------------------------------------------------------------------
    candidates = []

    # power‑law exponents (integers)
    for p in (1, 2, 3, 4, 5, 6, 8, 10):
        candidates.append((_make_power_sequence(n, p), f"power p={p}"))

    # one symmetric quadratic shape for reference
    candidates.append((_make_product_sequence(n), "product quadratic"))

    # ------------------------------------------------------------------
    # Evaluate each candidate and keep the best.
    # ------------------------------------------------------------------
    for seq_arr, label in candidates:
        # protect against the 1000 s deadline
        if time.time() > deadline:
            print("[timeout] stopping early while scanning candidates")
            break

        seq_list = seq_arr.tolist()
        try:
            score = evaluate_sequence(seq_list)
        except Exception as exc:  # pragma: no cover
            print(f"[error] evaluate_sequence failed for {label}: {exc}")
            continue

        print(f"[candidate] {label:15s} → O = {score:.6f}")

        if score < best_score:
            best_score, best_seq = score, seq_list
            print(f"  ** new global best (O={best_score:.6f})")

    # ------------------------------------------------------------------
    # Final sanity check – call the supplied evaluator once more.
    # ------------------------------------------------------------------
    if best_seq is None:
        # Should never happen, but fall back to a uniform distribution.
        best_seq = np.ones(n, dtype=float) / n
        best_seq = best_seq.tolist()
        best_score = evaluate_sequence(best_seq)
        print("[fallback] using uniform sequence")

    try:
        final_score = evaluate_sequence(best_seq)
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_seq
