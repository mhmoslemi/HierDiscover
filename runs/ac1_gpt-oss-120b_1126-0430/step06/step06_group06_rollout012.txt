def de_optimize_segments(n, seg_of_i, seg_len, fft_len, deadline, seed):
    import numpy as np
    from scipy.optimize import differential_evolution
    np.random.seed(seed)
    K = len(seg_len)
    bounds = [(0.0, 1.0)] * K
    
    def obj_raw(w):
        a = segment_to_seq(w, seg_of_i, seg_len)
        return cheap_score(a, n, fft_len)
    
    # callback
    def callback(xk, convergence):
        # stop if time near deadline
        return time.time() >= deadline
    
    # Set maxiter: dynamic based on time left; we can attempt a moderate number
    # Let maxiter = 100, but if time left small, lower.
    # estimate time left:
    time_left = deadline - time.time()
    if time_left < 10:
        maxiter = 0
    else:
        # approximate 100 evaluations per second; set maxiter accordingly:
        maxeval = int(10 * time_left)  # allow 10 evaluations per sec
        # DE uses popsize*maxiter evaluations roughly
        # solve for maxiter: popsize*maxiter ~ maxeval
        popsize = 15
        maxiter = max(1, maxeval // (popsize * K))
        maxiter = min(maxiter, 200)  # cap
    if maxiter <= 0:
        # fallback: random init
        w0 = np.random.dirichlet(np.ones(K))
        a0 = segment_to_seq(w0, seg_of_i, seg_len)
        return a0, cheap_score(a0, n, fft_len)
    
    result = differential_evolution(
        func=obj_raw,
        bounds=bounds,
        strategy='best1bin',
        maxiter=maxiter,
        popsize=15,
        tol=1e-7,
        mutation=(0.5, 1),
        recombination=0.7,
        seed=seed,
        polish=False,
        callback=callback,
        disp=False,
        updating='deferred',
    )
    w_opt = result.x
    a_opt = segment_to_seq(w_opt, seg_of_i, seg_len)
    val_opt = cheap_score(a_opt, n, fft_len)

#!/usr/bin/env python
"""
Propose a non‑negative step function (list of heights) that minimises

        2·n·max_k (a*a)[k] / (Σ a_i)²

Algorithm
---------
1. Determine a working length n (≈5 k – 25 k).
2. Run a fast parametric sweep (exponential / power‑law) to get an initial
   candidate.
3. Reduce the dimensionality by fixing a piecewise‑constant structure with
   K≈30 segments.  Optimise the segment masses with SciPy's differential
   evolution (DE) while respecting the simplex constraint.
4. Refine the full‑resolution vector with a mirror‑descent (exponentiated‑gradient)
   method that uses the sub‑gradient of the max‑convolution.  Momentum and
   occasional random restarts help escape plateaus.
5. Finish with a simple (1+1)‑ES hill‑climber on the cheap objective.
6. Evaluate the final vector with the supplied ``evaluate_sequence`` and
   return it.

All helper functions are defined at top level (no closures) and the routine
adapts to the remaining time budget, guaranteeing a feasible return within
the 1000 s limit.
"""

# ----------------------------------------------------------------------
# Imports
# ----------------------------------------------------------------------
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·n‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len).real[: 2 * a.size - 1]
    return conv


def _conv_max_fft(a: np.ndarray, fft_len: int) -> tuple[float, int]:
    """Maximum coefficient of a*a and its index."""
    conv = _conv_full_fft(a, fft_len)
    idx = int(conv.argmax())
    return float(conv[idx]), idx


# ----------------------------------------------------------------------
# Parametric families (all return a normalised probability vector)
# ----------------------------------------------------------------------
def _uniform_seq(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _power_law_seq(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = i ** (-p)
    seq /= seq.sum()
    return seq


def _exponential_seq(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    seq = np.exp(-lam * i)
    seq /= seq.sum()
    return seq


def _power_law_exp_seq(n: int, p: float, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    seq = (i + 1.0) ** (-p) * np.exp(-lam * i)
    seq /= seq.sum()
    return seq


def _stretched_exponential_seq(n: int, lam: float, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    seq = np.exp(-lam * (i ** gamma))
    seq /= seq.sum()
    return seq


def _beta_seq(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(1, n + 1, dtype=np.float64) / n)
    seq = (x ** alpha) * ((1.0 - x) ** beta)
    s = seq.sum()
    if s == 0.0:
        return None
    seq /= s
    return seq


def _dirichlet_seq(n: int, conc: float) -> np.ndarray:
    g = np.random.gamma(conc, 1.0, size=n)
    seq = g / g.sum()
    return seq


# ----------------------------------------------------------------------
# Objective helpers
# ----------------------------------------------------------------------
def cheap_score(seq: np.ndarray, n: int, fft_len: int) -> float:
    """Cheap surrogate: 2·n·max(conv) assuming Σseq = 1."""
    mx, _ = _conv_max_fft(seq, fft_len)
    return 2.0 * n * mx


def segment_to_sequence(w: np.ndarray, seg_of_i: np.ndarray,
                         seg_len: np.ndarray) -> np.ndarray:
    """Map a segment‑mass vector w (sum 1) to a full length‑n step function."""
    w = np.maximum(w, 0.0)
    total = w.sum()
    if total <= 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / total
    heights = w[seg_of_i] / seg_len[seg_of_i]
    return heights


# ----------------------------------------------------------------------
# Phase 0 – quick parametric sweep
# ----------------------------------------------------------------------
def parametric_search(n: int, fft_len: int, deadline: float) -> tuple[np.ndarray, float]:
    best_a = _uniform_seq(n)
    best_val = cheap_score(best_a, n, fft_len)

    # exponential family (coarse)
    for lam in np.linspace(0.0, 5.0, 31):
        if time.time() >= deadline:
            break
        a = _exponential_seq(n, lam)
        val = cheap_score(a, n, fft_len)
        if val < best_val:
            best_a, best_val = a, val

    # power‑law family
    for p in np.linspace(0.2, 2.0, 31):
        if time.time() >= deadline:
            break
        a = _power_law_seq(n, p)
        val = cheap_score(a, n, fft_len)
        if val < best_val:
            best_a, best_val = a, val

    # mixed power‑law × exponential
    for p in np.linspace(0.2, 1.0, 9):
        for lam in np.linspace(0.0, 3.0, 10):
            if time.time() >= deadline:
                break
            a = _power_law_exp_seq(n, p, lam)
            val = cheap_score(a, n, fft_len)
            if val < best_val:
                best_a, best_val = a, val

    # random Dirichlet seeds
    for conc in [0.001, 0.005, 0.01, 0.05, 0.1, 0.5, 1.0, 2.0]:
        if time.time() >= deadline:
            break
        a = _dirichlet_seq(n, conc)
        val = cheap_score(a, n, fft_len)
        if val < best_val:
            best_a, best_val = a, val

    print(f"[Parametric] best cheap = {best_val:.6f}")
    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 1 – segment based Differential Evolution
# ----------------------------------------------------------------------
def de_optimize_segments(n: int, seg_of_i: np.ndarray,
                         seg_len: np.ndarray, fft_len: int,
                         deadline: float, seed: int) -> tuple[np.ndarray, float]:
    """Run DE on the K‑segment simplex (no explicit constraints)."""
    from scipy.optimize import differential_evolution  # local import

    K = seg_len.size
    bounds = [(0.0, 1.0)] * K
    np.random.seed(seed)

    def obj_raw(w):
        a = segment_to_sequence(w, seg_of_i, seg_len)
        return cheap_score(a, n, fft_len)

    # callback stops DE as soon as we run out of budget
    def de_callback(xk, convergence):
        return time.time() >= deadline

    # A modest number of generations – cheap enough and still informative
    maxiter = 50
    result = differential_evolution(
        func=obj_raw,
        bounds=bounds,
        strategy='best1bin',
        maxiter=maxiter,
        popsize=15,
        tol=1e-7,
        mutation=(0.5, 1),
        recombination=0.7,
        seed=seed,
        polish=False,
        callback=de_callback,
        disp=False,
        updating='deferred',
    )
    w_opt = result.x
    a_opt = segment_to_sequence(w_opt, seg_of_i, seg_len)
    val_opt = cheap_score(a_opt, n, fft_len)
    print(f"[DE] best cheap = {val_opt:.6f}")
    return a_opt, val_opt


# ----------------------------------------------------------------------
# Phase 2 – mirror‑descent (exponentiated‑gradient) refinement
# ----------------------------------------------------------------------
def mirror_descent_refine(a0: np.ndarray, n: int,
                         fft_len: int, deadline: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_val = cheap_score(a, n, fft_len)
    it = 0
    last_improve = 0
    eta0 = 0.7
    mu = 0.6                # momentum factor
    vel = np.zeros_like(a)
    eps = 1e-30

    while time.time() < deadline - 5.0:
        it += 1
        conv = _conv_full_fft(a, fft_len)
        maxb = conv.max()
        cheap = 2.0 * n * maxb
        if cheap < best_val - 1e-12:
            best_val = cheap
            best_a = a.copy()
            last_improve = it
            print(f"[MD {it}] new best cheap = {cheap:.6f}")

        # active set (tight constraints)
        rel_tol = 1e-12
        active = np.where(conv >= maxb * (1.0 - rel_tol))[0]

        # sub‑gradient of the max‑convolution
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            rng = np.arange(lo, hi + 1, dtype=int)
            g[rng] += 2.0 * a[k - rng]
        g /= float(active.size)

        # momentum‑augmented exponentiated‑gradient step
        eta = eta0 / math.sqrt(it)
        vel = mu * vel + eta * g
        log_a = np.log(a + eps) - vel
        a = np.exp(log_a)
        s = a.sum()
        if s > 0.0 and np.isfinite(s):
            a /= s
        else:
            a = best_a.copy()
            vel = np.zeros_like(a)

        # restart if no progress for a long while
        if it - last_improve > 200_000:
            perturb = 0.02 * np.random.rand(n)
            a = _simplex_projection(best_a + perturb)
            vel = np.zeros_like(a)
            last_improve = it
            print(f"[MD {it}] restart after stagnation")

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 3 – (1+1)‑ES hill‑climbing on the cheap objective
# ----------------------------------------------------------------------
def es_hill_climb(a0: np.ndarray, n: int,
                  fft_len: int, deadline: float) -> tuple[np.ndarray, float]:
    best_a = a0.copy()
    best_val = cheap_score(best_a, n, fft_len)
    sigma = 0.12
    fail_cnt = 0
    it = 0

    while time.time() < deadline - 5.0:
        it += 1
        cand = best_a * np.exp(sigma * np.random.randn(n))
        cand = _simplex_projection(cand)
        val = cheap_score(cand, n, fft_len)
        if val < best_val:
            best_val = val
            best_a = cand
            sigma = max(sigma * 0.85, 1e-4)
            fail_cnt = 0
            print(f"[ES {it}] improved cheap = {val:.6f} sigma={sigma:.5f}")
        else:
            fail_cnt += 1
            if fail_cnt > 200:
                sigma = min(sigma * 1.30, 0.6)
                fail_cnt = 0

    return best_a, best_val
