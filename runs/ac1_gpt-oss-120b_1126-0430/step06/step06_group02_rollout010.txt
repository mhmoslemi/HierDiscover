#!/usr/bin/env python
"""
Hybrid optimiser for the max‑convolution problem.

The optimiser works in three stages:

1️⃣  A quick scan of several parametric families (Beta‑like, product‑power‑law,
    exponential, logistic) on a modest grid (≈ 5 000 points).  
    The best distribution from this scan is kept as a warm‑start.

2️⃣  Adam‑style mirror‑descent on the probability simplex (logits + soft‑max)
    using a smoothed soft‑max approximation of the ‑max‑operator.
    The smoothing factor ε = eps_factor·max_conv is decreased exponentially,
    while the Adam learning‑rate uses an inverse‑sqrt schedule.
    Jitter restarts are performed whenever the optimiser stalls.

3️⃣  A single up‑sampling step to a larger support (≈ 20 000 points) followed
    by a longer fine‑tuning phase with a much smaller ε.  The final candidate
    is evaluated with the supplied ``evaluate_sequence`` routine.

All helper routines are top‑level, no closures or lambdas are used.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max that returns a probability vector (sums to 1)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    s = e.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / z.size)
    return e / s


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it that reproduces ``a`` after a soft‑max (handles zeros)."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """Smallest power‑of‑two ≥ 3·n‑2 (sufficient for all needed linear convolutions)."""
    return 1 << ((3 * n - 2).bit_length())


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    s = new_a.sum()
    if s == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= s
    return new_a


def _normalize(a: np.ndarray) -> np.ndarray:
    """Project onto the simplex (non‑negative & sum = 1)."""
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / a.size
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p)
    return _normalize(a)


def _sym_product_power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ ((i+1)*(n-i))^{-p}, symmetric, heavy on the edges."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) * (n - i)) ** (-p)
    return _normalize(a)


def _exp_decay_distribution(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) + exp(-lam·(n‑1‑i)), symmetric exponential."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    return _normalize(a)


def _logistic_distribution(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ 1/(1+exp(alpha·(i‑mid))) – left‑heavy."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = 1.0 / (1.0 + np.exp(alpha * (i - mid)))
    return _normalize(a)


def _beta_distribution(n: int, a_shape: float, b_shape: float) -> np.ndarray:
    """a_i ∝ (i+1)^{a‑1}·(n‑i)^{b‑1} (Beta‑type shape, a,b>0)."""
    i = np.arange(n, dtype=np.float64)
    # Add a tiny epsilon to avoid 0**negative.
    eps = 1e-12
    a = (i + 1 + eps) ** (a_shape - 1.0) * (n - i + eps) ** (b_shape - 1.0)
    return _normalize(a)


def _mixed_beta_distribution(n: int, params):
    """Mixture of several Beta shapes. ``params`` is a list of (w,a,b)."""
    a = np.zeros(n, dtype=np.float64)
    for w, a_shape, b_shape in params:
        a += w * _beta_distribution(n, a_shape, b_shape)
    return _normalize(a)


# ----------------------------------------------------------------------
# Convolution utilities (FFT‑based, O(n log n))
# ----------------------------------------------------------------------


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Returns (max_conv, grad_a) where

        grad_a = ∇_a [ 2·n·smooth_max(a*a) ]

    ``eps_factor`` multiplies the smoothing constant ε = eps_factor·max_conv.
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    # FFT of a (zero‑padded)
    a_fft = np.fft.rfft(a, L)

    # Linear self‑convolution a*a
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # Smoothed soft‑max weights (log‑sum‑exp trick)
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv                     # ≤ 0, stabilises exp
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # Correlation a ⋆ w  (gradient of smooth max)
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # Gradient (already multiplied by 4·n, the factor needed for the objective)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) improves the
    ``evaluate_sequence`` score.  The routine respects the given time budget.
    """
    np.random.seed(seed)
    start = time.time()
    # keep a couple of seconds for the final exact evaluation
    deadline = start + budget_s - 3.0

    # ------------------------------------------------------------------
    # 0. Light parametric scan on a modest size (≈ 5000 points)
    # ------------------------------------------------------------------
    n_small = 5000
    best_small = None
    best_score = float("inf")
    print("[Stage 0] Parametric scan (n =", n_small, ")")

    # Helper to try a candidate and keep the best
    def _try_candidate(cand):
        nonlocal best_small, best_score
        score = 2.0 * n_small * _conv_max(cand)
        if score < best_score:
            best_score, best_small = score, cand.copy()

    # Power‑law
    for p in np.linspace(0.3, 2.0, 30):
        if time.time() >= deadline:
            break
        _try_candidate(_power_law_distribution(n_small, p))

    # Symmetric product‑power‑law
    for p in np.linspace(0.3, 2.0, 30):
        if time.time() >= deadline:
            break
        _try_candidate(_sym_product_power_law_distribution(n_small, p))

    # Exponential decay
    for loglam in np.linspace(-7, -2, 15):
        if time.time() >= deadline:
            break
        lam = 10.0 ** loglam
        _try_candidate(_exp_decay_distribution(n_small, lam))

    # Logistic (left‑heavy)
    for alpha in np.linspace(0.1, 5.0, 25):
        if time.time() >= deadline:
            break
        _try_candidate(_logistic_distribution(n_small, alpha))

    # Beta shapes (both symmetric and asymmetric)
    for a_shape in np.linspace(0.2, 2.5, 15):
        if time.time() >= deadline:
            break
        # symmetric (a = b)
        _try_candidate(_beta_distribution(n_small, a_shape, a_shape))
        for b_shape in np.linspace(0.2, 2.5, 5):
            if time.time() >= deadline:
                break
            _try_candidate(_beta_distribution(n_small, a_shape, b_shape))

    # If nothing succeeded (should not happen) fall back to uniform
    if best_small is None:
        best_small = np.full(n_small, 1.0 / n_small)
        best_score = 2.0 * n_small * _conv_max(best_small)

    # ------------------------------------------------------------------
    # 1. Optional warm‑start from global variable (if present)
    # ------------------------------------------------------------------
    try:
        glob = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        best_small = _interpolate_distribution(glob, n_small)
        best_score = 2.0 * n_small * _conv_max(best_small)
        print("[Stage 0] Warm‑started from global sequence.")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 2. Adam optimisation on the logit (mirror descent on the simplex)
    # ------------------------------------------------------------------
    # First optimisation phase (moderate size)
    n = n_small
    a = best_small.copy()
    z = _logits_from_distribution(a) + 0.01 * np.random.randn(n)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    base_alpha = 0.45          # base learning‑rate for logits
    eps_factor = 0.20          # initial smoothing factor (ε = eps_factor·max_conv)
    eps_factor_min = 5e-5
    eps_decay = 0.9990

    best_seq = _softmax(z)
    best_score = 2.0 * n * _conv_max(best_seq)
    last_improve = 0
    it = 0

    # main loop – runs until deadline
    while time.time() < deadline:
        it += 1

        # current distribution on the simplex
        a = _softmax(z)

        # cheap evaluation + smoothed gradient
        max_conv, grad_a = _conv_max_and_grad(a, eps_factor)
        cheap_score = 2.0 * n * max_conv

        # keep best ever seen
        if cheap_score < best_score - 1e-12:
            best_score = cheap_score
            best_seq = a.copy()
            last_improve = it

        # sub‑gradient w.r.t. logits (mirror‑descent formula)
        mu = np.dot(grad_a, a)            # <grad, a>
        grad_z = a * (grad_a - mu)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step‑size schedule (inverse sqrt decay)
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step_dir = - m_hat / (np.sqrt(v_hat) + eps_adam)

        # Try a short back‑tracking line‑search (max 5 halvings)
        accepted = False
        trial_alpha = alpha_t
        for _ in range(5):
            z_new = z + trial_alpha * step_dir
            z_new = np.clip(z_new, -50.0, 50.0)           # protect from overflow
            a_new = _softmax(z_new)
            new_score = 2.0 * n * _conv_max(a_new)
            if new_score <= cheap_score:
                z = z_new
                accepted = True
                break
            trial_alpha *= 0.5
        if not accepted:
            # take a small step anyway (helps to escape plateaux)
            z += alpha_t * step_dir
            z = np.clip(z, -50.0, 50.0)

        # occasional random jitter (helps to get out of plateaux)
        if (it % 2500 == 0) and (np.random.rand() < 0.12):
            z += 0.02 * np.random.randn(n)
            m *= 0.6
            v *= 0.6

        # gradually tighten smoothing
        eps_factor = max(eps_factor_min, eps_factor * eps_decay)

        # restart if stuck for a long time
        if it - last_improve > 250_000:
            print(f"[Iter {it}] Restart due to stagnation (best score {best_score:.6f})")
            z = _logits_from_distribution(best_seq) + 0.03 * np.random.randn(n)
            m.fill(0.0)
            v.fill(0.0)
            eps_factor = 0.20
            last_improve = it
            it = 0  # reset counter to give the fresh start a fair chance

        # ------------------------------------------------------------------
        # 3. Single up‑sampling stage (once) when we still have enough time.
        # ------------------------------------------------------------------
        # Target a larger support (~20 000) for the final fine‑tuning.
        if n < 20000 and (time.time() + 45.0) < deadline:
            new_n = min(20000, int(n * 1.8))
            print(f"[Iter {it}] Upsampling: {n} → {new_n}")
            best_seq = _interpolate_distribution(best_seq, new_n)
            # re‑initialise optimizer on the enlarged grid
            z = _logits_from_distribution(best_seq) + 0.005 * np.random.randn(new_n)
            m = np.zeros_like(z)
            v = np.zeros_like(z)
            n = new_n
            best_score = 2.0 * n * _conv_max(best_seq)
            # reset the stagnation helpers
            it = 0
            last_improve = 0
            eps_factor = 0.20
            continue

    # ------------------------------------------------------------------
    # 4. Final exact evaluation with the provided evaluate_sequence()
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[Result] Final exact score = {final_score:.6f}")
    except Exception as exc:
        final_score = float("inf")
        print(f"[Result] Evaluation error: {exc}")

    # ------------------------------------------------------------------
    # Return the best sequence discovered.
    # ------------------------------------------------------------------
    return final_seq
