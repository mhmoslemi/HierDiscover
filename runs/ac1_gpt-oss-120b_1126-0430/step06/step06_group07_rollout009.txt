#!/usr/bin/env python
"""
Propose a non‑negative step function (list of heights) that minimises
``evaluate_sequence``.  The search proceeds in three phases:

1. **Broad parametric sweep** – enumerates many analytic families
   (uniform, exponential, power‑law, symmetric U‑shapes, beta,
   Dirichlet, triangular, Hann, sine‑squared, cosine‑ramps, etc.)
   and keeps the best candidate according to a cheap score
   ``2·n·max(conv)`` (the true objective with the sum normalised to 1).

2. **Smooth‑max mirror descent** – minimises a smooth approximation of the
   ``max`` (soft‑max with a temperature β) using an exponentiated‑gradient
   (mirror descent) update on the probability simplex.  The gradient is
   computed exactly via FFTs.  β is increased slowly so that the algorithm
   first follows the average convolution and then focuses on the true
   peak.  A decaying stepsize η≈η₀/√t and occasional restarts avoid
   stagnation.

3. **Permutation / transfer local search** – starting from the best
   iterate, random swaps and tiny mass‑transfer moves are accepted either
   if they improve the cheap score or with a simulated‑annealing probability.
   This refines the discrete ordering of the heights.

The routine respects the supplied time budget (default 1000 s) and always
returns a feasible sequence.  All helper functions are top‑level; no
closures or lambdas are used.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using a power‑of‑two FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa).real[: 2 * a.size - 1]
    return conv


def _conv_max_fft(a: np.ndarray, fft_len: int) -> float:
    """Maximum value of the autocorrelation a*a."""
    conv = _conv_full_fft(a, fft_len)
    return float(conv.max())

# Core search routine required by the platform
# ----------------------------------------------------------------------

        a = np.exp(log_a)
        a_sum = a.sum()
        if a_sum <= 0.0 or not np.isfinite(a_sum):
            a = best_iter_a.copy()
            continue
        a /= a_sum

        # slowly increase soft‑max temperature
        if iter_cnt % 2000 == 0 and beta < beta_max:
            beta = min(beta * 1.05, beta_max)

        # occasional restart if stuck for a long time
        if iter_cnt - last_improve > 200_000:
            perturb = np.random.rand(n) * 0.001
            a = _simplex_projection(best_iter_a + perturb)
            last_improve = iter_cnt
            print(f"[Iter {iter_cnt:07d}] Restart after stagnation.")

    # --------------------------------------------------------------
    # 5. Permutation / transfer local search (simulated annealing)
    # --------------------------------------------------------------
    a = best_iter_a.copy()
    best_swap_a = a.copy()
    best_swap_val = best_iter_val
    temperature = 1.0
    cooling = 0.99999

    while time.time() < deadline:
        # choose move type: 0 = swap, 1 = small mass transfer
        move_type = np.random.randint(0, 2)
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        if move_type == 0:
            # swap two entries
            a[i], a[j] = a[j], a[i]
            performed_transfer = False
        else:
            # transfer a small amount from i to j
            eps_transfer = min(a[i], np.random.rand() * 0.005)  # up to 0.5 % of a[i]
            if eps_transfer <= 0.0:
                continue
            a[i] -= eps_transfer
            a[j] += eps_transfer
            performed_transfer = True

        conv = _conv_full_fft(a, fft_len)
        cheap = 2.0 * n * conv.max()
        accept = False
        if cheap < best_swap_val - 1e-12:
            accept = True
        else:
            delta = cheap - best_swap_val
            prob = math.exp(-delta / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            if cheap < best_swap_val - 1e-12:
                best_swap_val = cheap
                best_swap_a = a.copy()
                print(f"[Swap] NEW best cheap = {cheap:.6f}")
        else:
            # revert the move
            if move_type == 0:
                a[i], a[j] = a[j], a[i]            # undo swap
            else:
                a[i] += eps_transfer
                a[j] -= eps_transfer                # undo transfer

        temperature = max(temperature * cooling, 1e-6)

    # --------------------------------------------------------------
    # 6. Final exact evaluation
    # --------------------------------------------------------------
    final_seq = best_swap_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:                      # defensive – should never happen
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    return final_seq
