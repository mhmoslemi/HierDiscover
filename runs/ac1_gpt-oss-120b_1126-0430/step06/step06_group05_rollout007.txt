    c = conv * beta

    import time, numpy as np

    start = time.time()
    deadline = start + budget_s - 0.5
    rng = np.random.default_rng(seed)

    # Determine target length based on budget.
    # Use 15000 if main budget > 400 seconds else 12000

    init_seq = None

        seq0 = seq0 / seq0.sum() if seq0.sum() > 0 else np.ones_like(seq0) / seq0.size
        init_seq = seq0

    def upsample(seq, n):
        if seq.size == n:
            return seq.copy()
        x_old = np.linspace(0.0, 1.0, seq.size, endpoint=False)
        x_new = np.linspace(0.0, 1.0, n, endpoint=False)
        seq_new = np.interp(x_new, x_old, seq)
        seq_new = np.clip(seq_new, 0.0, None)
        if seq_new.sum() == 0:
            seq_new = np.ones(n) / n
        else:
            seq_new = seq_new / seq_new.sum()
        return seq_new

    # time allocation
    total_budget = deadline - time.time()
    # allocate
    smoothed_budget = total_budget * 0.6
    hardmax_budget = total_budget * 0.25
    sa_budget = total_budget - smoothed_budget - hardmax_budget

    target_n = 15000 if smoothed_budget > 40 else 12000

    # Run smoothed Adam
    seq_s, obj_s = _run_opt_smooth(target_n, smoothed_budget,
                                   init_a=upsample(init_seq, target_n) if init_seq is not None else None,
                                   seed=seed+11)
    best_seq = np.asarray(seq_s)
    best_obj = obj_s
    print(f"[stage] smoothed Adam best O={best_obj:.6f}")

    # Run exact max Adam for refinement if time left
    remaining = deadline - time.time()
    if remaining > 5.0:

"""
Multi‑stage step‑function optimiser.

The optimiser works in three phases within the 1000 s budget:

1. **Smoothed‑max Adam** – the objective “max(conv)” is replaced by a
   soft‑max with a temperature that rises from 0.1 to ≈20.  The exact
   gradient of the smoothed objective is obtained in O(n log n) by
   FFT‑based convolution, which gives a much richer signal than the
   hard‑max sub‑gradient and helps escape plateaus.

2. **Hard‑max Adam** – a short refinement with the exact max‑gradient
   (the code from the previous run) is performed to sharpen the final
   solution.

3. **Targeted simulated‑annealing** – a final local‑search phase moves
   a small fraction of mass between entries, accepting worsening moves
   with a Metropolis schedule.

All phases respect the simplex constraint via exponentiated‑gradient
updates (log‑it parametrisation).  The best sequence found is returned
as a plain list of non‑negative floats.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Helper functions – FFT utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return conv_full[: 2 * n - 1]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces a probability vector)."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


def _grad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """
    Exact gradient of the k‑th convolution entry w.r.t. ``a``.
    Returns 2·a_{k−i} where the index is in range, otherwise 0.
    """
    n = a.shape[0]
    idx = np.arange(n)
    j = k - idx
    mask = (j >= 0) & (j < n)
    g = np.zeros_like(a)
    if mask.any():
        g[mask] = 2.0 * a[j[mask]]
    return g

def _run_opt_hardmax(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
) -> Tuple[List[float], float]:
    """Adam optimiser that always uses the exact max‑convolution gradient."""
    start = time.time()
    rng = np.random.default_rng(seed)

    # initialise a on the simplex
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
    else:
        a = rng.random(n)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n, dtype=float) / n
    else:
        a = a / s

    # logits for exponentiated (mirror) gradient updates
    x = np.log(a + 1e-16)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # learning‑rate schedule (exponential decay)
    lr_start = 0.5
    lr_end = 0.001

    best_a = a.copy()
    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())

    it = 0
    while time.time() - start < time_limit:
        it += 1

        # current point on the simplex
        a = _softmax(x)

        conv = _conv_self_fft(a)
        M = conv.max()
        obj = float(2.0 * n * M)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[hardmax n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        # index of the maximal convolution entry
        k_max = int(np.argmax(conv))

        # exact sub‑gradient of that entry
        g_a = _grad_max_conv(a, k_max)

        # natural gradient in log‑it space (mirrored / exponentiated)
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)

        # time‑based learning‑rate (exponential)
        elapsed = time.time() - start
        t = min(elapsed / time_limit, 1.0)
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # Adam step in log‑it space
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # jitter / restart
        if it % 5000 == 0:
            jitter = rng.normal(scale=0.01, size=n)
            x += jitter
        if it % 80000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.005, size=n)
            m.fill(0.0)
            v.fill(0.0)

    # prune negligible entries
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = float(2.0 * best_a.size * _conv_self_fft(best_a).max())
        print(f"[hardmax compress] reduced length → {best_a.size}, O={best_obj:.6f}")

    return best_a.tolist(), best_obj

def _run_sa(
    init_seq: List[float],
    time_limit: float,
    seed: int = 0,
) -> Tuple[List[float], float]:
    """Refine a sequence by small mass‑transfer moves with a Metropolis schedule."""
    start = time.time()
    rng = np.random.default_rng(seed)
    a = np.asarray(init_seq, dtype=float)
    if a.sum() < 1e-12:
        a = np.ones_like(a) / a.size
    else:
        a = a / a.sum()
    n = a.shape[0]

    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())
    best_seq = a.copy()

    # temperature schedule
    T = best_obj * 0.1 + 1e-6
    T_min = best_obj * 1e-6 + 1e-9
    decay = 0.995
    it = 0

    while time.time() - start < time_limit:
        it += 1
        k_max = int(np.argmax(conv))

        i_vals = np.arange(n)
        j_vals = k_max - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i_vals[mask]] * a[j_vals[mask]]
        total_contrib = contrib.sum()

        if total_contrib > 0:
            probs = contrib / total_contrib
            i = rng.choice(n, p=probs)
        else:
            i = rng.integers(n)

        # distinct destination uniformly
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.clip(a_new, 0.0, None)
        a_new /= a_new.sum()

        conv_new = _conv_self_fft(a_new)
        obj_new = float(2.0 * n * conv_new.max())

        if obj_new < best_obj - 1e-12:
            a, conv, best_obj, best_seq = a_new, conv_new, obj_new, a_new.copy()
            print(f"[SA it={it}] new best O={best_obj:.6f}")
        else:
            prob = np.exp(-(obj_new - best_obj) / T) if T > 0 else 0.0
            if rng.random() < prob:
                a, conv = a_new, conv_new

        if it % 2000 == 0:
            T = max(T * decay, T_min)
        if it % 10000 == 0:
            noise = rng.normal(scale=0.01, size=n)
            a = np.clip(a + noise, 0.0, None)
            a /= a.sum()
            conv = _conv_self_fft(a)

    # final compression
    thresh = 1e-12
    mask = best_seq > thresh
    if mask.sum() < best_seq.size:
        comp = best_seq[mask]
        comp /= comp.sum()
        comp_obj = float(2.0 * comp.size * _conv_self_fft(comp).max())
        if comp_obj < best_obj:
            best_seq, best_obj = comp, comp_obj
            print(f"[SA compress] reduced length → {comp.size}, O={best_obj:.6f}")

    return best_seq.tolist(), best_obj

def _run_opt_smooth(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,

    """
    Adam optimiser with a soft‑max approximation of the max‑convolution.
    Returns the best sequence (list) and its objective value.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # initialise a on the simplex
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)

            a = rng.random(n)

    a = np.clip(a, 0.0, None)

    if s < 1e-12:

    else:
        a = a / s

    x = np.log(a + 1e-16)         # log‑it parametrisation
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # LR schedule
    lr_start = 0.5
    lr_end = 0.001

    # soft‑max temperature schedule
    beta_start = 0.1
    beta_end = 20.0

    # FFT length that accommodates both the convolution (2n‑1) and the
    # gradient convolution (n + (2n‑1)‑1 = 3n‑2)
    L = _next_pow2(3 * n)

    # pre‑compute phase factor for the FFT of the reversed vector
    k = np.arange(L // 2 + 1)
    phase = np.exp(-2j * np.pi * (n - 1) * k / L)

    # track best solution
    conv_full = np.fft.irfft(np.fft.rfft(a, n=L) * np.fft.rfft(a, n=L), n=L)[: 2 * n - 1]
    best_obj = float(2.0 * n * conv_full.max())
    best_a = a.copy()

    it = 0
    while time.time() - start < time_limit:
        it += 1

        # current a (simplex point)
        a = _softmax(x)

        # FFT of a
        Af = np.fft.rfft(a, n=L)

        # full self‑convolution (linear)
        conv_full = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]

        # objective value
        M = conv_full.max()
        obj = float(2.0 * n * M)
        if obj < best_obj - 1e-14:

        c = conv_full * beta

        A_rev_f = np.conj(Af) * phase

        grad_full = np.fft.irfft(A_rev_f * Wf, n=L)
