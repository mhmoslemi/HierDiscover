def _project_onto_simplex(v):
    # Duchi algorithm
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

#!/usr/bin/env python
"""
Optimise a non‑negative step‑function for the max‑convolution metric.

The algorithm proceeds as follows

1.  Choose a large vector length (up to 65536, the largest n that still
    fits into a 2**16 = 65536‑point FFT).  This dramatically reduces the
    discretisation error of the continuous arcsine construction,
    because the evaluation metric is asymptotically 2·n·max(conv) → 1.

2.  Initialise the vector by the *exact* arcsine bin probabilities:
    a_j = F((j+1)/n) – F(j/n)   with   F(x)=2/π·arcsin(√x).

3.  Refine the initial point with a **mirrored‑gradient Adam** optimiser
    on the simplex (parameterise the probabilities with a soft‑max of
    logits).  The gradient of a smoothed max of the self‑convolution is
    computed with FFTs, and a small amount of stochastic jitter is added
    occasionally to escape plateaus.

4.  The optimizer runs until the time budget (≈1000 s) is exhausted.
    The best sequence seen (lowest value of the cheap score
    2·n·max(conv)) is kept and finally evaluated with the provided
    ``evaluate_sequence`` function.

The whole routine respects the 1000 s wall‑clock limit, performs no
I/O and uses only NumPy and the standard library.  It is deliberately
self‑contained so that the evaluation platform can import ``propose_candidate``
directly.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – top level, no closures
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max returning a probability vector."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Logits that map to ``a`` via the soft‑max."""
    return np.log(a + eps)


def _fft_len_for_conv(n: int) -> int:
    """Smallest power‑of‑two ≥ 2·n‑1, sufficient for a linear self‑convolution."""
    return 1 << ((2 * n - 1).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a (no gradient)."""
    n = a.shape[0]
    L = _fft_len_for_conv(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Smoothed max of a*a and its gradient w.r.t. the probability vector a.

    Returns
    -------
    max_conv : float
        Maximum entry of the linear self‑convolution.
    grad_a : np.ndarray
        Gradient of 2·n·smooth_max w.r.t. a (already multiplied by 2·n).
    """
    n = a.shape[0]
    L = _fft_len_for_conv(n)

    a_fft = np.fft.rfft(a, L)

    # linear self‑convolution (length 2·n‑1)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # smoothed soft‑max weights (log‑sum‑exp trick)
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv               # improve stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum

    # cross‑correlation a ⋆ w (linear, length 3·n‑2)
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # gradient of 2·n·smooth_max (the factor 2·n is already incorporated)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex {x ≥ 0, Σx = 1}.
    Implements the algorithm of Duchi et al. (2008).  Returns a new array.
    """
    n = v.size
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical edge‑case: return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w
