"""
Non‑negative step‑function optimiser – multi‑peak Adam + greedy annealing.

The optimiser works in three stages:

1.  A “soft‑max Adam” that uses a weighted sum of the top‑K convolution
    entries (instead of only the single maximum).  The weighting is
    controlled by a temperature ρ that grows from 0 to a large value,
    turning the objective from a smooth average into the hard max.
    This helps escape plateaus that the original hard‑max Adam
    occasionally gets stuck in.

2.  Optional quick restarts: the best sequence obtained so far is
    repeatedly refined for a short time budget, each time starting from
    the current best (with a little jitter).  The restarts allow the
    optimiser to jump out of local minima.

3.  A cheap simulated‑annealing (SA) post‑processing that moves small
    fractions of mass between indices.  As in the reference code this
    stage can clean up remaining irregularities.

The routine respects the overall time budget, prints progress, and
finally returns a list of non‑negative floats suitable for
``evaluate_sequence``.
"""

import time
from typing import List, Tuple

import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return conv_full[: 2 * n - 1]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (maps logits to the simplex)."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


# ----------------------------------------------------------------------
# Gradient of a single convolution entry (hard‑max sub‑gradient)
# ----------------------------------------------------------------------


def _grad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """
    Exact sub‑gradient of the k‑th convolution entry w.r.t. ``a``.
    Returns 2·a_{k‑i} for indices where it is defined, otherwise 0.
    """
    n = a.shape[0]
    idx = np.arange(n)
    j = k - idx
    mask = (j >= 0) & (j < n)
    g = np.zeros_like(a)
    if mask.any():
        g[mask] = 2.0 * a[j[mask]]
    return g


# ----------------------------------------------------------------------
# Hard‑max Adam with multi‑peak (top‑K) gradient
# ----------------------------------------------------------------------


def _run_opt_multi_peak_adam(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
    top_k: int = 5,
) -> Tuple[List[float], float]:
    """
    Adam optimiser that uses a weighted sum of the top‑K convolution
    entries as the sub‑gradient.  The temperature ρ is increased from
    ``rho_start`` to ``rho_end`` (linear schedule) so the method
    gradually turns from a smooth objective to the hard max.
    Returns the best sequence and its objective value (2·n·max_conv).
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # initialise a on the simplex
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
    else:
        a = rng.random(n)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n, dtype=float) / n
    else:
        a /= s

    # log‑it representation (mirror‑descent on the simplex)
    x = np.log(a + 1e-16)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # learning‑rate schedule (exponential decay)
    lr_start = 0.5
    lr_end = 0.001

    # temperature schedule for the soft‑max smoothing
    rho_start = 0.0
    rho_end = 80.0

    best_a = a.copy()
    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())

    it = 0
    while time.time() - start < time_limit:
        it += 1

        # current point on the simplex
        a = _softmax(x)

        # convolution and objective
        conv = _conv_self_fft(a)
        M = conv.max()
        obj = float(2.0 * n * M)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        # ----- multi‑peak gradient -------------------------------------------------
        # pick the top‑K entries (K may be reduced if the array is tiny)
        K = min(top_k, conv.size)
        # np.argpartition gives the K largest unsorted; we keep them as is
        top_idxs = np.argpartition(-conv, K - 1)[:K]

        # linear schedule for ρ
        elapsed = time.time() - start
        t = min(elapsed / time_limit, 1.0)
        rho = rho_start + (rho_end - rho_start) * t

        # weights for the selected indices (soft‑max with temperature ρ)
        diffs = conv[top_idxs] - M          # ≤ 0
        w_raw = np.exp(rho * diffs)        # 1 for the maximal entry, <1 otherwise
        w_sum = w_raw.sum()
        if w_sum == 0.0:                    # numerical under‑flow – fall back to uniform
            weights = np.full_like(w_raw, 1.0 / K)
        else:
            weights = w_raw / w_sum

        # aggregate sub‑gradient
        g_a = np.zeros_like(a)
        for wk, idx in zip(weights, top_idxs):
            if wk == 0.0:
                continue
            g_a += wk * _grad_max_conv(a, int(idx))

        # natural gradient in log‑it space (mirror‑descent)
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)

        # learning‑rate (exponential decay)
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # Adam step in log‑it space
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to shake out of plateaus
        if it % 5000 == 0:
            jitter = rng.normal(scale=0.01, size=n)
            x += jitter

        # occasional full restart from the current best (helps escape)
        if it % 80000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.005, size=n)
            m.fill(0.0)
            v.fill(0.0)

    # ------------------------------------------------------------------
    # prune negligible entries (they increase n without helping)
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = float(2.0 * best_a.size * _conv_self_fft(best_a).max())
        print(f"[compress] reduced length → {best_a.size}, O={best_obj:.6f}")

    return best_a.tolist(), best_obj

def _run_sa(
    init_seq: List[float],
    time_limit: float,
    seed: int = 0,
) -> Tuple[List[float], float]:
    """Refine a sequence by mass‑transfer moves with a Metropolis schedule."""
    start = time.time()
    rng = np.random.default_rng(seed)
    a = np.asarray(init_seq, dtype=float)
    if a.sum() < 1e-12:
        a = np.ones_like(a) / a.size
    else:
        a = a / a.sum()
    n = a.shape[0]

    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())
    best_seq = a.copy()

    # temperature schedule
    T = best_obj * 0.1 + 1e-6
    T_min = best_obj * 1e-6 + 1e-9
    decay = 0.995
    it = 0

    while time.time() - start < time_limit:
        it += 1
        k_max = int(np.argmax(conv))

        # contribution of each index to the maximal entry
        i_vals = np.arange(n)
        j_vals = k_max - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i_vals[mask]] * a[j_vals[mask]]
        total_contrib = contrib.sum()

        if total_contrib > 0:
            probs = contrib / total_contrib
            i = rng.choice(n, p=probs)
        else:
            i = rng.integers(n)

        # choose a distinct destination uniformly
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # transfer a modest fraction of mass from i to j
        delta = a[i] * rng.random() * 0.05  # up to 5 % of a[i]
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.clip(a_new, 0.0, None)
        a_new /= a_new.sum()

        conv_new = _conv_self_fft(a_new)
        obj_new = float(2.0 * n * conv_new.max())

        # Metropolis acceptance
        if obj_new < best_obj - 1e-12:
            a, conv, best_obj, best_seq = a_new, conv_new, obj_new, a_new.copy()
            print(f"[SA it={it}] new best O={best_obj:.6f}")
        else:
            prob = np.exp(-(obj_new - best_obj) / T) if T > 0 else 0.0
            if rng.random() < prob:
                a, conv = a_new, conv_new

        # temperature decay
        if it % 2000 == 0:
            T = max(T * decay, T_min)

        # occasional global jitter
        if it % 10000 == 0:
            noise = rng.normal(scale=0.01, size=n)
            a = np.clip(a + noise, 0.0, None)
            a /= a.sum()
            conv = _conv_self_fft(a)

    # final compression of tiny entries
    thresh = 1e-12
    mask = best_seq > thresh
    if mask.sum() < best_seq.size:
        comp = best_seq[mask]
        comp /= comp.sum()
        comp_obj = float(2.0 * comp.size * _conv_self_fft(comp).max())
        if comp_obj < best_obj:
            best_seq, best_obj = comp, comp_obj
            print(f"[SA compress] reduced length → {comp.size}, O={best_obj:.6f}")

    return best_seq.tolist(), best_obj


# ----------------------------------------------------------------------
# Main entry point – the function that will be called externally
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **_ignore,
) -> List[float]:
    """
    Find a non‑negative step‑function minimizing the evaluation metric.

    The routine:
    * picks a target length (≈12 k for the full budget);
    * runs the multi‑peak Adam optimiser;
    * performs a few short restarts if time permits;
    * finishes with a greedy simulated‑annealing refinement.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Initialise from the previously discovered construction, if any.
    # ------------------------------------------------------------------
    init_seq = None
    if "height_sequence_1" in globals():
        seq0 = np.asarray(height_sequence_1, dtype=float)
        seq0 = np.clip(seq0, 0.0, None)
        if seq0.sum() < 0.01:
            seq0 = np.ones_like(seq0)
        seq0 = seq0 / seq0.sum()
        init_seq = seq0

    # ------------------------------------------------------------------
    # Choose a target length depending on the remaining time.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 120.0:
        target_n = 12000
    else:
        target_n = 8000

    # Upsample any initial sequence to the target length.
    if init_seq is not None and init_seq.size != target_n:
        x_old = np.linspace(0.0, 1.0, init_seq.size, endpoint=False)
        x_new = np.linspace(0.0, 1.0, target_n, endpoint=False)
        init_a = np.interp(x_new, x_old, init_seq)
        init_a = np.clip(init_a, 0.0, None)
        init_a = init_a / init_a.sum()
    else:
        init_a = init_seq

    # ------------------------------------------------------------------
    # Main optimisation – multi‑peak Adam.
    # ------------------------------------------------------------------
    main_budget = (deadline - time.time()) * 0.78
    seq, obj = _run_opt_multi_peak_adam(
        n=target_n,
        time_limit=main_budget,
        init_a=init_a,
        seed=seed + 123,
        top_k=5,
    )
    best_seq = np.asarray(seq, dtype=float)
    best_obj = obj
    print(f"[global] after Adam best O={best_obj:.6f} (n={target_n})")

    # ------------------------------------------------------------------
    # Quick restarts (short Adam runs) while we still have time.
    # ------------------------------------------------------------------
    while time.time() < deadline - 20.0:
        # allocate a small chunk (≈30 s or whatever fits)
        chunk = min(30.0, deadline - time.time() - 10.0)
        if chunk <= 5.0:
            break
        seq2, obj2 = _run_opt_multi_peak_adam(
            n=target_n,
            time_limit=chunk,
            init_a=best_seq.tolist(),
            seed=seed + rng.integers(1_000_000),
            top_k=5,
        )
        if obj2 < best_obj:
            best_seq = np.asarray(seq2, dtype=float)
            best_obj = obj2
            print(f"[restart] improved O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # If time permits, refine with simulated‑annealing.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        seq_ref, obj_ref = _run_sa(best_seq.tolist(), time_limit=remaining, seed=seed + 9999)
        if obj_ref < best_obj:
            best_seq = np.asarray(seq_ref, dtype=float)
            best_obj = obj_ref
            print(f"[final] after SA new best O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check – call the supplied evaluator.
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_seq.tolist()
