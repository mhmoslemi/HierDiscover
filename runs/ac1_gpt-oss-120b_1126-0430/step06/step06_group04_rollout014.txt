"""
Optimiser for the step‑function evaluation problem.

The objective (for a sequence a with Σa = 1) is

    O(a) = 2·n·max_k (a * a)[k] ,

i.e. minimise the largest entry of the self‑convolution.
The algorithm proceeds in four stages:

1. **Rich shape catalogue** – a variety of analytically inspired families
   (exponential, power‑law, Beta‑like, arcsine, double‑end clusters,
   sparse Dirichlet) are sampled for a given support length n.
   The best of these cheap candidates (exact objective) is kept.

2. **Adam on the simplex** – the chosen candidate is refined by Adam on the
   soft‑max (log‑it) parametrisation of the probability simplex.
   The smooth‑max is implemented by a temperature‑controlled soft‑max;
   the temperature grows from 0.5 to 1e9, i.e. we converge to the exact
   max.

3. **Targeted mass‑transfer Annealing** – a fast simulated‑annealing /
   greedy mass‑transfer loop focuses on the indices that contribute most
   to the current maximal convolution entry and moves a small fraction of
   mass to a random “quiet’’ location.

4. **Projected‑gradient polishing** – a few exact‑gradient steps with
   projection onto the simplex finish the optimisation.

The optimiser tries several support sizes (2000 … 12000).  Longer
supports give more degrees of freedom, allowing the convolution to be
flattened towards the theoretical lower bound O≈1.  The whole routine
honours the supplied time budget (default 1000 s) and returns the best
sequence found.  All numbers are non‑negative, sum to 1 and are clipped
to the limits accepted by ``evaluate_sequence``.
"""

import time

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return conv_full[: 2 * n - 1]


def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation of a with a weight vector w:
        g[i] = Σ_t w[i+t]·a[t]   for i=0,…,n‑1
    Implemented via FFT (real‑valued).
    """
    n = a.shape[0]
    m = w.shape[0]                     # = 2n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Stable soft‑max."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) with Σa = 1."""
    n = a.size
    conv = _conv_self_fft(a)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σ x = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Input must be 1‑D")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _shape_candidates(n: int, rng: np.random.Generator):

    sum‑to‑1.

    cand = []
    idx = np.arange(n, dtype=np.float64)

    # 1. Uniform
    cand.append(np.full(n, 1.0 / n))

    # 2. Exponential decay (forward & reversed)
    for r in np.linspace(0.90, 0.9999, 12):

        cand.append(a)

    for p in np.linspace(0.5, 4.5, 9):

        a = a / a.sum()

        cand.append(a[::-1])

def _best_initial_sequence(n: int, rng: np.random.Generator, time_limit: float):
    """
    Scan a modest set of analytical shapes and return the one with the
    smallest exact objective.  The scan aborts if time runs out.
    """
    start = time.time()
    best_a = None
    best_obj = float("inf")
    for a in _shape_candidates(n, rng):
        if time.time() - start > time_limit:
            break
        obj = _objective(a)
        if obj < best_obj:
            best_obj, best_a = obj, a
    # Fallback: a random Dirichlet
    if best_a is None:
        a = rng.random(n)
        best_a = a / a.sum()
        best_obj = _objective(best_a)
    return best_a, best_obj

    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # logits
    x = np.log(init_a + 1e-16)
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # schedule parameters (more aggressive than the previous version)
    beta_start = 0.5
    beta_end = 1e9         # sharper final max
    lr_start = 0.3
    lr_end = 5e-4

    best_a = init_a.copy()
    best_obj = _objective(best_a)

    it = 0
    while time.time() - start < remaining_time:
        it += 1
        a = _softmax(x)                     # stays on simplex

        conv = _conv_self_fft(a)
        M = conv.max()
        obj = float(2.0 * n * M)

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a.copy()
            # occasional logging
            if it % 1000 == 0:
                print(f"[n={n} it={it}] new best O={best_obj:.6f}")

        # --- annealing schedule (time‑based) -----
        t = min((time.time() - start) / remaining_time, 1.0)
        beta = beta_start * np.exp(np.log(beta_end / beta_start) * t)
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t)

        # smooth‑max weights
        if beta > 5e8:
            # hard max
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            shifted = conv - M                # ≤ 0
            w = np.exp(beta * shifted)
            s = w.sum()
            if s == 0.0:
                w = np.ones_like(conv) / conv.size
            else:
                w /= s

        # gradient of smooth‑max wrt a
        g_a = 2.0 * _correlation_fft(w, a)

        # natural gradient in log‑it space
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape plateaus
        if it % 2500 == 0:
            jitter = rng.normal(scale=0.02, size=n)
            x += jitter

        # periodic restart from current best
        if it % 50000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.008, size=n)
            m.fill(0.0)
            v.fill(0.0)

    return best_a.tolist()


# ----------------------------------------------------------------------
# Stage 3 – targeted simulated annealing + mass‑transfer moves
# ----------------------------------------------------------------------


def _anneal_and_transfer(
    init_seq: list[float],
    remaining_time: float,
    rng: np.random.Generator,
) -> np.ndarray:
    """
    Fast simulated annealing that moves mass away from the indices that
    contribute most to the current maximal convolution entry.
    """
    start = time.time()
    a = np.asarray(init_seq, dtype=float)
    a = a / a.sum()
    n = a.size

    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())
    best_a = a.copy()

    # SA schedule
    T = best_obj * 0.25 + 1e-8
    T_min = best_obj * 1e-6 + 1e-12
    decay = 0.996

    it = 0
    while time.time() - start < remaining_time * 0.85:
        it += 1
        k_max = int(np.argmax(conv))

        # contribution vector for k_max
        i_range = np.arange(n)
        j_range = k_max - i_range
        mask = (j_range >= 0) & (j_range < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i_range[mask]] * a[j_range[mask]]
        total = contrib.sum()
        if total > 0.0:
            probs = contrib / total
            i = rng.choice(n, p=probs)
        else:
            i = rng.integers(n)

        # destination (different from i)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # move a modest fraction of mass from i to j
        delta = a[i] * rng.random() * 0.07   # up to 7 % of a[i]
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.clip(a_new, 0.0, None)
        a_new /= a_new.sum()

        conv_new = _conv_self_fft(a_new)
        obj_new = float(2.0 * n * conv_new.max())

        if obj_new < best_obj - 1e-15:
            a, conv = a_new, conv_new
            best_obj, best_a = obj_new, a_new.copy()
        else:
            # Metropolis acceptance
            if rng.random() < math.exp(-(obj_new - best_obj) / T):
                a, conv = a_new, conv_new

        # temperature decay
        if it % 3000 == 0:
            T = max(T * decay, T_min)

        # occasional global jitter
        if it % 8000 == 0:
            noise = rng.normal(scale=0.01, size=n)
            a = np.clip(a + noise, 0.0, None)
            a /= a.sum()
            conv = _conv_self_fft(a)

    # ------------------------------------------------------------------
    # Final projected‑gradient polishing (exact gradient)
    # ------------------------------------------------------------------
    a = best_a.copy()
    n = a.size
    eta0 = 0.3
    it_g = 0
    while time.time() - start < remaining_time:
        it_g += 1
        conv = _conv_self_fft(a)
        k_max = int(np.argmax(conv))

        # exact max‑gradient (hard one‑hot weight)
        w = np.zeros_like(conv)
        w[k_max] = 1.0
        g = 2.0 * _correlation_fft(w, a)

        eta = eta0 * (0.994 ** it_g)
        a_new = a - eta * g
        a_new = _project_simplex(a_new)
        obj_new = float(2.0 * n * _conv_self_fft(a_new).max())

        if obj_new < best_obj - 1e-15:
            a, best_obj, best_a = a_new, obj_new, a_new.copy()
        else:
            eta0 *= 0.97
            if eta0 < 2e-4:
                break

    return best_a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **_ignore) -> list[float]:
    """
    Main optimisation routine.
    Tries several support sizes, rich analytic shape guesses,
    Adam refinement, and a targeted annealing + gradient polishing.
    Returns a list of non‑negative floats ready for ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # Support sizes we will explore (larger n gives more freedom)
    lengths = [2000, 4000, 8000, 12000]

    best_seq = None
    best_obj = float("inf")

    for idx, n in enumerate(lengths):
        now = time.time()
        if now >= deadline - 2.0:
            break

        # allocate a time budget for this length
        remaining = deadline - now
        # first length gets a larger share, later ones less
        if idx == 0:
            alloc = remaining * 0.55
        else:
            alloc = (remaining * 0.35) / (len(lengths) - idx)

        print(f"[stage] n={n}, allocated {alloc:.1f}s")

        # ---- Stage 1 : shape pre‑search ----
        shape_time = min(6.0, alloc * 0.12)   # a few seconds for a rich sweep
        init_a, init_obj = _best_initial_sequence(n, rng, shape_time)
        print(f"[init n={n}] shape‑obj={init_obj:.6f}")

        # ---- Stage 2 : Adam optimisation ----
        adam_time = alloc * 0.55
        seq_adam = _adam_opt(init_a, n, adam_time, rng, seed + idx * 12345)
        obj_adam = _objective(np.asarray(seq_adam, dtype=float))
        print(f"[adam n={n}] O={obj_adam:.6f}")

        if obj_adam < best_obj:
            best_obj, best_seq = obj_adam, np.asarray(seq_adam, dtype=float)
            print(f"[global] new overall best O={best_obj:.6f} (n={n})")

        # ---- Optional quick pruning (remove tiny entries) ----
        thresh = 1e-12
        mask = best_seq > thresh
        if mask.sum() < best_seq.size:
            best_seq = best_seq[mask]
            best_seq /= best_seq.sum()
            best_obj = _objective(best_seq)
            print(f"[compress] length→{best_seq.size}, O={best_obj:.6f}")

        if time.time() >= deadline - 1.0:
            break

    # ------------------------------------------------------------------
    # Stage 3 : intensive annealing + projected gradient on the best seq
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if best_seq is not None and remaining > 8.0:
        print(f"[final] starting SA+PG stage, {remaining:.1f}s left")
        seq_refined = _anneal_and_transfer(best_seq.tolist(),
                                           remaining,
                                           rng)
        obj_refined = _objective(np.asarray(seq_refined, dtype=float))
        print(f"[final] SA+PG O={obj_refined:.6f}")
        if obj_refined < best_obj:
            best_obj, best_seq = obj_refined, np.asarray(seq_refined, dtype=float)

    # ------------------------------------------------------------------
    # Safety fallback (should never be hit)
    # ------------------------------------------------------------------
    if best_seq is None or best_seq.size == 0:
        best_seq = np.ones(200, dtype=float) / 200.0
        best_obj = _objective(best_seq)

    # Final sanity check (only for logging; the harness will call evaluate_sequence)
    try:
        from __main__ import evaluate_sequence  # the evaluator is injected by the harness
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    # Clamp to the limits required by the evaluator
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Guard against an accidentally tiny sum (should not happen)
    if best_seq.sum() < 0.01:
        best_seq = np.ones_like(best_seq) / best_seq.size

    return best_seq.tolist()
