"""
Hybrid exact‑gradient mirror descent for the self‑convolution bound.

The algorithm works in three phases:

1.  Initialise a probability vector a (sum = 1) from the supplied
    `height_sequence_1` if present, otherwise use a uniform distribution.
    The vector is already non‑negative and lies on the simplex.

2.  Repeatedly minimise the cheap objective
        cheap(a) = 2·n·max(a*a)
    using an **exact sub‑gradient** of the `max` (no soft‑max smoothing).
    The sub‑gradient of maxₖ (a*a)[k] is
        ∇ₐ max = 2· Σ_{k∈A} a_{k‑i},
    where A is the set of shifts attaining the current maximum.
    We perform an exponentiated‑gradient (mirror‑descent w.r.t. KL)
    update:
        aᵢ ← aᵢ·exp(‑η·∇ᵢ) ,   then renormalise.
    The stepsize η decays like η₀/√t and a simple back‑tracking line‑search
    guarantees monotone improvement of the cheap score.

3.  A cheap random‑walk phase (coordinate transfers) runs in the remaining
    time to escape tiny plateaus.

The best vector encountered is finally returned as a plain list; the
provided `evaluate_sequence` will rescale it if needed and compute the
exact bound.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def _fft_len_for_corr(n: int) -> int:
    """
    Return the smallest power‑of‑two ≥ 3·n‑2, sufficient for a self‑convolution
    of length 2·n‑1 via FFT.
    """
    return 1 << ((3 * n - 2).bit_length())


def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """
    Compute the linear self‑convolution a*a using FFT of length L.
    Returns the full convolution of length 2·n‑1 (truncated from the FFT result).
    """
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)
    n = a.shape[0]
    return conv[: 2 * n - 1]


def _max_convolution(a: np.ndarray, L: int) -> float:
    """Return max(a*a) using the FFT helper."""
    conv = _conv_self_fft(a, L)
    return float(conv.max())


def _max_and_active_indices(a: np.ndarray, L: int):
    """
    Compute max(a*a) and the list of shift indices where the maximum is attained.
    Returns (max_value, np.ndarray of active indices).
    """
    conv = _conv_self_fft(a, L)
    max_val = conv.max()
    # tolerance to collect all essentially‑maximal shifts
    tol = max(1e-15, max_val * 1e-12)
    active = np.where(conv >= max_val - tol)[0]
    return max_val, active, conv


def _subgradient_max(a: np.ndarray, active_shifts: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of maxₖ (a*a)[k] for the given active shift set.
    For each i, grad[i] = 2· Σ_{k∈A} a_{k‑i} (with out‑of‑range indices ignored).
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in active_shifts:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # vector of i positions
        i = np.arange(i_start, i_end + 1, dtype=np.int64)
        j = k - i                     # partner indices
        grad[i] += 2.0 * a[j]
    return grad


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex (non‑negative, sum = 1).
    """
    if v.ndim != 1:
        raise ValueError("simplex projection only defined for 1‑D arrays")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative -> uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w
