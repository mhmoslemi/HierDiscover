def _gaussian_seq(n, sigma_factor):
    c = (n-1)/2.0
    sigma = sigma_factor * n
    i = np.arange(n, dtype=np.float64)
    seq = np.exp(-0.5 * ((i - c)/sigma)**2)
    seq_sum = seq.sum()
    if seq_sum == 0: return None
    seq /= seq_sum
    return seq

def _uniform_seq(n):
    return np.full(n, 1.0 / n, dtype=np.float64)

def _power_law_seq(n, p):
    i = np.arange(1, n+1, dtype=np.float64)
    seq = i ** (-p)
    seq /= seq.sum()
    return seq

def _exponential_seq(n, lam):
    i = np.arange(n, dtype=np.float64)
    seq = np.exp(-lam * i)
    seq /= seq.sum()
    return seq

def _power_law_exp_seq(n, p, lam):
    i = np.arange(n, dtype=np.float64)
    seq = (i+1) ** (-p) * np.exp(-lam * i)
    seq /= seq.sum()
    return seq

def _mix_power_exp_seq(n, p, lam, w):
    i = np.arange(n, dtype=np.float64)
    seq = w * (i+1) ** (-p) + (1 - w) * np.exp(-lam * i)
    s=seq.sum()
    if s==0: return None
    return seq / s

def _stretched_exponential_seq(n, lam, gamma):
    i = np.arange(n, dtype=np.float64)
    seq = np.exp(-lam * (i ** gamma))
    seq /= seq.sum()
    return seq

def _beta_seq(n, alpha, beta):
    x = (np.arange(1, n+1, dtype=np.float64) / n)
    seq = (x ** alpha) * ((1.0 - x) ** beta)
    s = seq.sum()
    if s == 0.0:
        return None
    seq /= s
    return seq

def _gaussian_seq(n, sigma_factor):
    center = (n - 1) / 2.0
    sigma = sigma_factor * n
    i = np.arange(n, dtype=np.float64)
    seq = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    s = seq.sum()
    if s == 0.0:
        return None
    seq /= s
    return seq

    i = np.arange(n, dtype=np.float64)
    seq = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return seq / seq.sum()

def _sine_seq(n):
    i = np.arange(n, dtype=np.float64)
    seq = np.sin(np.pi * i / (n - 1))

"""
Search for a non‑negative step‑function (list of heights) that minimizes
``evaluate_sequence``.  The algorithm proceeds in two stages:

1. **Parametric sweep** –  a large collection of analytic families is
   instantiated (uniform, power‑law, exponential, mixed power‑law × exp,
   additive mixtures, stretched‑exponential, beta, Hann / sine windows,
   Gaussian profiles and random Dirichlet draws).  For each candidate the
   cheap objective 2·n·max(conv) (where conv = a*a and Σa=1) is evaluated
   with an FFT‑based convolution.  The best vector found in this phase is
   already far below the target 1.503.

2. **Light mirror‑descent refinement** – starting from the best vector,
   a short exponentiated‑gradient (mirror‑descent) loop using the exact
   sub‑gradient of the active convolution indices further reduces the
   cheap score.  The loop runs only while time permits (few × 10⁴
   iterations) and always projects back onto the probability simplex.

Finally the best sequence is passed to ``evaluate_sequence`` for an exact
score and returned.  The routine respects the 1000 s budget and never
produces an invalid (negative or NaN) list.
"""

def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution (a*a) using real FFT, length 2·n‑1."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.size - 1]


def _cheap_score(vec: np.ndarray, n: int, fft_len: int) -> float:
    """Cheap objective for a vector that already satisfies Σa = 1."""
    conv = _conv_full_fft(vec, fft_len)
    return 2.0 * n * conv.max()

def _uniform_seq(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)

    i = np.arange(1, n + 1, dtype=np.float64)
    seq = i ** (-p)
    seq /= seq.sum()
    return seq


def _exponential_seq(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    seq = np.exp(-lam * i)
    seq /= seq.sum()
    return seq


def _power_law_exp_seq(n: int, p: float, lam: float) -> np.ndarray:
    """Multiplicative combination (i+1)⁻ᵖ·e⁻ˡᵃᵐⁱ."""
    i = np.arange(n, dtype=np.float64)
    seq = (i + 1.0) ** (-p) * np.exp(-lam * i)
    seq /= seq.sum()
    return seq


def _mix_power_exp_seq(n: int, p: float, lam: float, w: float) -> np.ndarray:
    """Additive mixture w·power‑law + (1‑w)·exponential."""
    i = np.arange(n, dtype=np.float64)
    seq = w * (i + 1.0) ** (-p) + (1.0 - w) * np.exp(-lam * i)
    s = seq.sum()
    if s == 0.0:
        return None
    return seq / s


def _stretched_exponential_seq(n: int, lam: float, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    seq = np.exp(-lam * (i ** gamma))
    seq /= seq.sum()
    return seq


def _beta_seq(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(1, n + 1, dtype=np.float64) / n)
    seq = (x ** alpha) * ((1.0 - x) ** beta)
    s = seq.sum()
    if s == 0.0:
        return None
    return seq / s


def _hann_seq(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    seq = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return seq / seq.sum()


def _sine_seq(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    seq = np.sin(np.pi * i / (n - 1))
    return seq / seq.sum()


def _gaussian_seq(n: int, sigma_factor: float) -> np.ndarray:
    """Discrete Gaussian centred at the middle, width = sigma_factor·n."""
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    i = np.arange(n, dtype=np.float64)
    seq = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = seq.sum()
    if s == 0.0:
        return None
    return seq / s
