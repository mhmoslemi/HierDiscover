#!/usr/bin/env python
"""
Non‑negative step‑function optimiser – improved version.

The optimiser proceeds in three stages:

1. **Broad parametric sweep** – a handful of classic window families
   (power‑law, symmetric product‑law, exponential, logistic,
   Hann / Hamming / Blackman, Gaussian, triangular and cosine‑power)
   are instantiated for a moderate support size (≈ 6000).  The cheapest
   candidate (2·n·max(a*a) with a normalised to unit L¹) is retained.

2. **Mirror‑descent on the simplex** – the best candidate is turned into
   logits (a = softmax(z/τ)) and refined using Adam.  The objective
   (smooth‑max of the self‑convolution) is differentiable via the
   softmax‑weight trick.  A temperature τ slowly cools the softmax,
   an ε‑smoothing factor for the max operator is decreased
   geometrically, and occasional random jitter / restarts help escape
   plateaux.

3. **Progressive up‑sampling** – once the optimiser has spent a
   substantial fraction of the time budget on the current support,
   the distribution is interpolated to a larger grid (up to ≈ 25000
   points) and the Adam‑loop continues.  The final best distribution
   is handed to the supplied ``evaluate_sequence`` routine.

All helper functions are top‑level, there are no closures or lambdas,
and the routine respects the given time budget (default 1000 s)."""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max (temperature = 1)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    s = e.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / z.size)
    return e / s


def _softmax_temp(z: np.ndarray, temp: float) -> np.ndarray:
    """Soft‑max with a temperature (temp > 0)."""
    if temp <= 0:
        temp = 1e-12
    z_scaled = z / temp
    return _softmax(z_scaled)


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it that reproduces `a` after a soft‑max."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """Smallest power‑of‑two ≥ 3·n‑2 (enough for all needed convolutions)."""
    return 1 << ((3 * n - 2).bit_length())


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    s = new_a.sum()
    if s == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= s
    return new_a


# ----------------------------------------------------------------------
# Parametric families (all return a non‑negative vector, **not** normalised)
# ----------------------------------------------------------------------


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    return a


def _sym_product_power_law_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) * (n - i)) ** (-p)
    return a


def _exp_decay_distribution(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    return a


def _logistic_distribution(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = 1.0 / (1.0 + np.exp(alpha * (i - mid)))
    return a


def _hann_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a


def _hamming_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return a


def _blackman_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    return a


def _triangular_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = 1.0 - np.abs(i - mid) / (mid + 1.0)
    a = np.maximum(a, 0.0)
    return a


def _gaussian_distribution(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mid) / sigma) ** 2)
    return a


def _cosine_power_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    # Shifted sine that is zero at the ends
    a = np.sin(np.pi * (i + 0.5) / n) ** p
    return a


# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Returns (max_conv, grad_a) where

        grad_a = ∇_a [ 2·n·smooth_max(a*a) ]

    `eps_factor` multiplies the smoothing constant ε = eps_factor·max_conv.
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    a_fft = np.fft.rfft(a, L)

    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv                     # ≤ 0
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # factor 4·n stems from the derivative of 2·n·smooth_max
    grad_a = 4.0 * n * corr
    return max_conv, grad_a


# ----------------------------------------------------------------------
# Evaluation helper (cheap – uses unit‑sum normalisation)
# ----------------------------------------------------------------------


def _cheap_score(a: np.ndarray) -> float:
    """
    Returns the evaluation value 2·n·max(a*a) for a fully normalised
    distribution (sum(a) == 1).  The function is cheap because it only
    needs a single FFT.
    """
    total = a.sum()
    if total <= 0.0:
        return float("inf")
    a = a / total
    n = a.shape[0]
    return 2.0 * n * _conv_max(a)

    # 1.  Initialise Adam state on logits (mirror descent on simplex)

    # Slight jitter to avoid exact symmetry

    beta1, beta2 = 0.9, 0.999
