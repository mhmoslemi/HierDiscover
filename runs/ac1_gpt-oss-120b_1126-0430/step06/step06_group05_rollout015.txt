"""
Non‑negative step‑function optimiser (smooth‑max Adam + hard‑max Adam + SA).

The optimiser consists of three stages:

1.  A short hard‑max Adam run to obtain a reasonable starting point.
2.  A smooth‑max Adam run with a temperature that is exponentially
    decreased; this uses a fully differentiable (soft‑max) approximation
    of the maximal self‑convolution, allowing the optimiser to escape
    plateaus that trap the plain hard‑max version.
3.  A targeted simulated‑annealing refinement that makes small mass‑transfer
    moves while the temperature is gradually cooled.

All updates are performed in log‑it (exponentiated‑gradient) space, which
keeps the variables on the simplex without an explicit projection.
The schedule for the learning‑rate and the temperature are both time‑based,
so the routine adapts automatically to the available time budget.
The best sequence found during the whole run is returned.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full self‑convolution a*a using a real FFT.
    Result length = 2*len(a)-1.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return conv_full[: 2 * n - 1]


def _conv_fft_general(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Convolution of two real vectors using FFT.
    Returns the full convolution of length len(a)+len(b)-1.
    """
    n = a.shape[0]
    m = b.shape[0]
    L = _next_pow2(n + m - 1)
    af = np.fft.rfft(a, n=L)
    bf = np.fft.rfft(b, n=L)
    conv_full = np.fft.irfft(af * bf, n=L)
    return conv_full[: n + m - 1]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


# ----------------------------------------------------------------------
# Hard‑max Adam optimiser (as in the reference)
# ----------------------------------------------------------------------


def _run_opt_hardmax(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
) -> Tuple[List[float], float]:
    """Adam optimiser that uses the exact gradient of the maximal entry."""
    start = time.time()
    rng = np.random.default_rng(seed)

    # initialise on the simplex
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
    else:
        a = rng.random(n)

    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n, dtype=float) / n
    else:
        a = a / s

    x = np.log(a + 1e-16)           # logits
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    lr_start, lr_end = 0.5, 0.001

    best_a = a.copy()
    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())

    it = 0
    while time.time() - start < time_limit:
        it += 1
        a = _softmax(x)

        conv = _conv_self_fft(a)
        M = conv.max()
        obj = float(2.0 * n * M)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[hardmax n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        k_max = int(np.argmax(conv))
        # exact sub‑gradient of the entry with index k_max
        idx = np.arange(n)
        j = k_max - idx
        mask = (j >= 0) & (j < n)
        g_a = np.zeros_like(a)
        if mask.any():
            g_a[mask] = 2.0 * a[j[mask]]

        # mirror gradient (exponentiated‑gradient)
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)

        # learning‑rate schedule (exponential decay)
        elapsed = time.time() - start
        t = min(elapsed / time_limit, 1.0)
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t)

        # Adam update in log‑it space
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter / restart
        if it % 5000 == 0:
            x += rng.normal(scale=0.01, size=n)
        if it % 80000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.005, size=n)
            m.fill(0.0)
            v.fill(0.0)

    # prune negligible entries
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = float(2.0 * best_a.size * _conv_self_fft(best_a).max())
        print(f"[hardmax compress] length → {best_a.size}, O={best_obj:.6f}")

    return best_a.tolist(), best_obj

# Smooth‑max Adam optimiser
# ----------------------------------------------------------------------


def _run_opt_smoothmax(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
) -> Tuple[List[float], float]:
    """
    Adam optimisation on a smooth approximation of the max‑convolution.
    Temperature T is exponentially decreased from T_start to T_end.
    The gradient of the soft‑max of the convolution entries is used,
    which is a weighted sum of the exact sub‑gradients.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # initialise
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
    else:
        a = rng.random(n)

    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n, dtype=float) / n
    else:
        a = a / s

    x = np.log(a + 1e-16)           # logits
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    lr_start, lr_end = 0.4, 0.0005

    best_a = a.copy()
    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())

    it = 0
    while time.time() - start < time_limit:
        it += 1
        a = _softmax(x)

        conv = _conv_self_fft(a)
        M = conv.max()
        obj = float(2.0 * n * M)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[smoothmax n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        # --- temperature schedule ---
        elapsed = time.time() - start
        t_frac = min(elapsed / time_limit, 1.0)
        # start temperature a little larger than typical convolution values
        T_start = max(conv.max() * 5.0, 1e-4)
        T_end = 1e-12
        T = T_start * (T_end / T_start) ** t_frac

        # smooth‑max weights (log‑sum‑exp trick)
        w = np.exp((conv - conv.max()) / T)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.ones_like(w) / w.size
        else:
            w /= w_sum

        # gradient of the smooth‑max:
        # g_i = 2 * sum_j a_j * w_{i+j}
        a_rev = a[::-1]
        conv_aw = _conv_fft_general(a_rev, w)                     # length = 3n-2
        g_a = 2.0 * conv_aw[n - 1 : n - 1 + n]                    # slice of length n

        # mirror gradient
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)

        # learning‑rate schedule (exponential)
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t_frac)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to improve exploration
        if it % 4000 == 0:
            x += rng.normal(scale=0.008, size=n)
        if it % 60000 == 0:
            # restart from the current best with slight noise
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.004, size=n)
            m.fill(0.0)
            v.fill(0.0)

    # prune negligible entries
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = float(2.0 * best_a.size * _conv_self_fft(best_a).max())
        print(f"[smoothmax compress] length → {best_a.size}, O={best_obj:.6f}")

    return best_a.tolist(), best_obj

def _run_sa(
    init_seq: List[float],
    time_limit: float,
    seed: int = 0,
) -> Tuple[List[float], float]:
    """Refine a sequence by small mass‑transfer moves with a Metropolis schedule."""
    start = time.time()
    rng = np.random.default_rng(seed)
    a = np.asarray(init_seq, dtype=float)
    if a.sum() < 1e-12:
        a = np.ones_like(a) / a.size
    else:
        a = a / a.sum()
    n = a.shape[0]

    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())
    best_seq = a.copy()

    # temperature schedule
    T = best_obj * 0.1 + 1e-8
    T_min = best_obj * 1e-6 + 1e-12
    decay = 0.995
    it = 0

    while time.time() - start < time_limit:
        it += 1
        k_max = int(np.argmax(conv))

        # contribution of each index to the maximal entry
        i_vals = np.arange(n)
        j_vals = k_max - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i_vals[mask]] * a[j_vals[mask]]
        total_contrib = contrib.sum()

        if total_contrib > 0:
            probs = contrib / total_contrib
            i = rng.choice(n, p=probs)
        else:
            i = rng.integers(n)

        # choose a distinct destination uniformly
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # transfer a modest fraction of mass from i to j
        delta = a[i] * rng.random() * 0.07   # up to 7 % of a[i]
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.clip(a_new, 0.0, None)
        a_new /= a_new.sum()

        conv_new = _conv_self_fft(a_new)
        obj_new = float(2.0 * n * conv_new.max())

        # Metropolis acceptance
        if obj_new < best_obj - 1e-12:
            a, conv, best_obj, best_seq = a_new, conv_new, obj_new, a_new.copy()
            print(f"[SA it={it}] new best O={best_obj:.6f}")
        else:
            prob = np.exp(-(obj_new - best_obj) / T) if T > 0 else 0.0
            if rng.random() < prob:
                a, conv = a_new, conv_new

        # temperature decay
        if it % 2500 == 0:
            T = max(T * decay, T_min)

        # occasional global jitter
        if it % 12000 == 0:
            noise = rng.normal(scale=0.01, size=n)
            a = np.clip(a + noise, 0.0, None)
            a /= a.sum()
            conv = _conv_self_fft(a)

    # final compression of tiny entries
    thresh = 1e-12
    mask = best_seq > thresh
    if mask.sum() < best_seq.size:
        comp = best_seq[mask]
        comp /= comp.sum()
        comp_obj = float(2.0 * comp.size * _conv_self_fft(comp).max())
        if comp_obj < best_obj:
            best_seq, best_obj = comp, comp_obj
            print(f"[SA compress] reduced length → {comp.size}, O={best_obj:.6f}")

    return best_seq.tolist(), best_obj

def propose_candidate(

    **_ignore,
) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises the supplied
    evaluation metric.  The routine consists of a short hard‑max Adam
    phase, a longer smooth‑max Adam phase (temperature annealing) and
    a final simulated‑annealing refinement.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Initialise from a previously discovered construction if available.
    init_seq = None
    if 'height_sequence_1' in globals():
        seq0 = np.asarray(height_sequence_1, dtype=float)
        seq0 = np.clip(seq0, 0.0, None)
        if seq0.sum() < 0.01:
            seq0 = np.ones_like(seq0)
        seq0 = seq0 / seq0.sum()
        init_seq = seq0

    # ------------------------------------------------------------------
    # Choose a target length (the algorithm works well with ~12k entries)
    remaining = deadline - time.time()
    target_n = 12000 if remaining > 40.0 else 8000

    # Upsample any initial sequence to the target length.
    if init_seq is not None and init_seq.size != target_n:
        x_old = np.linspace(0.0, 1.0, init_seq.size, endpoint=False)
        x_new = np.linspace(0.0, 1.0, target_n, endpoint=False)
        init_a = np.interp(x_new, x_old, init_seq)
        init_a = np.clip(init_a, 0.0, None)
        init_a = init_a / init_a.sum()
    else:
        init_a = init_seq

    # ------------------------------------------------------------------
    # Stage 1 – brief hard‑max Adam to obtain a reasonable start
    remaining = deadline - time.time()
    stage1_budget = min(remaining * 0.15, 30.0)   # at most 30 s, at least 15 %
    seq_hard, obj_hard = _run_opt_hardmax(
        n=target_n,
        time_limit=stage1_budget,
        init_a=init_a,
        seed=seed + 101,
    )
    best_seq = np.asarray(seq_hard, dtype=float)
    best_obj = obj_hard
    print(f"[stage1] after hard‑max O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – smooth‑max Adam with temperature annealing
    remaining = deadline - time.time()
    stage2_budget = remaining * 0.70
    seq_smooth, obj_smooth = _run_opt_smoothmax(
        n=target_n,
        time_limit=stage2_budget,
        init_a=best_seq,

    )
    if obj_smooth < best_obj:

    # ------------------------------------------------------------------

    remaining = deadline - time.time()
    if remaining > 8.0:
        seq_sa, obj_sa = _run_sa(best_seq.tolist(), time_limit=remaining, seed=seed + 303)

    # Final sanity check – call the supplied evaluator.
    try:

        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_seq.tolist()
