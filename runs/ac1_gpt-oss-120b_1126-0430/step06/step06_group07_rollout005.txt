assistantfinal**Solution Overview**

The function `propose_candidate` builds a non‑negative step function (a list of heights) that tries to minimise  

```
2·n·max(conv(a,a)) / (sum(a)²)      ( n = len(a) )
```

The optimisation is performed in three independent stages that together respect the
time budget:

1. **Broad parametric sweep** – a large collection of families (exponential,
   power‑law, triangular, cosine, Gaussian‑mixture, Beta, Dirichlet…) is
   generated for several lengths `n`.  Every candidate is normalised to a
   simplex (`sum = 1`) and its cheap score `2·n·max(conv)` (computed by FFT) is
   evaluated.  The best candidate so far is retained.

2. **Exact sub‑gradient (projected) mirror descent** – starting from the best
   candidate, a true sub‑gradient of the max‑convolution entry is computed and a
   projected (or exponentiated) gradient step is taken on the simplex.
   Multiple random restarts are performed until the time budget for this phase
   is exhausted.

3. **Permutation‑local search** – the best vector from phase 2 is refined by
   simulated‑annealing style swaps of two positions.  The cheap score is used
   for fast acceptance decisions; the temperature is slowly cooled.

Finally the exact `evaluate_sequence` function is called once on the best vector
found and that vector is returned.

All helper utilities are defined at module level, there are no closures or
lambdas, and the implementation works with up to two CPU cores (the heavy work
is performed by NumPy/FFT which already uses multi‑threading).

```python
#!/usr/bin/env python
"""
Optimise a non‑negative step function (list of heights) for the objective

    2·n·max(conv(a,a)) / (sum(a)²),

using a three‑stage strategy:
    • wide parametric sweep over many families and lengths,
    • exact sub‑gradient projected (mirror) descent with several restarts,
    • permutation‑swap simulated‑annealing refinement.
The routine respects a time budget (default 1000 s) and always returns a
feasible sequence.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a real vector onto the probability simplex (non‑negative, sum=1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all entries non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using a power‑of‑two real FFT."""
    # real FFT is faster & uses less memory
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    # keep only the first 2·n‑1 entries (full linear convolution)
    return conv[: 2 * a.size - 1]


def _conv_max_fft(a: np.ndarray, fft_len: int) -> tuple[float, int]:
    """Maximum value of the autocorrelation a*a and its index (full convolution)."""
    conv = _conv_full_fft(a, fft_len)
    idx = int(conv.argmax())
    return float(conv[idx]), idx


def _subgradient(a: np.ndarray, kmax: int) -> np.ndarray:
    """
    Exact sub‑gradient of the maximal convolution entry.
    For a given offset kmax, the gradient w.r.t. a_i is
        2·a_{kmax-i}   (if the counterpart index exists)
    and 0 otherwise.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    lo = max(0, kmax - (n - 1))
    hi = min(kmax, n - 1)
    i = np.arange(lo, hi + 1, dtype=np.int64)
    j = kmax - i
    g[i] = 2.0 * a[j]
    return g


def _cheap_score(vec: np.ndarray, n: int, fft_len: int) -> float:
    """Score under the simplifying assumption sum(vec)=1 : 2·n·max(conv)."""
    maxb, _ = _conv_max_fft(vec, fft_len)
    return 2.0 * n * maxb


# ----------------------------------------------------------------------
# Candidate generators – each returns a normalised (sum=1) ndarray
# ----------------------------------------------------------------------


def _gen_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a /= a.sum()
    return a


def _gen_sym_exponential(n: int, lam: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * np.minimum(idx, n - 1 - idx))
    a /= a.sum()
    return a


def _gen_power_law(n: int, p: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a /= a.sum()
    return a


def _gen_sym_power_law(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.minimum(i + 1, n - i) ** (-p)
    a /= a.sum()
    return a


def _gen_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(1, n + 1, dtype=np.float64) / n)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    a /= s
    return a


def _gen_triangular(n: int, width_frac: float) -> np.ndarray:
    support = max(1, int(round(width_frac * n)))
    centre = (n - 1) / 2.0
    idx = np.arange(n, dtype=np.float64)
    dist = np.abs(idx - centre)
    a = np.maximum(0.0, 1.0 - dist / (support / 2.0))
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _gen_cosine(n: int, width_frac: float) -> np.ndarray:
    support = max(1, int(round(width_frac * n)))
    centre = (n - 1) / 2.0
    idx = np.arange(n, dtype=np.float64)
    dist = np.abs(idx - centre)
    mask = dist <= (support / 2.0)
    a = np.zeros(n, dtype=np.float64)
    a[mask] = 0.5 * (1.0 + np.cos(np.pi * dist[mask] / (support / 2.0)))
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _gen_gaussian_mixture(n: int, k: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    centres = np.linspace(0, n - 1, k)
    a = np.zeros(n, dtype=np.float64)
    for c in centres:
        a += np.exp(-0.5 * ((np.arange(n) - c) / sigma) ** 2)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Sub‑gradient (projected) descent – works on the simplex
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         fft_len: int,
                         n: int,
                         deadline: float,
                         max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Exact projected sub‑gradient descent on the simplex.
    Returns the best vector encountered and its cheap score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _cheap_score(a, n, fft_len)

    # momentum variables (optional)
    momentum = np.zeros_like(a)
    beta = 0.9
    last_improve = 0

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = _conv_full_fft(a, fft_len)
        maxb = conv.max()
        cheap = 2.0 * n * maxb

        if cheap < best_val - 1e-12:
            best_val = cheap
            best_a = a.copy()
            last_improve = it

        # sub‑gradient at the maximal offset
        kmax = int(conv.argmax())
        g = _subgradient(a, kmax)

        # simple (RMS‑prop‑like) adaptive step size
        eta = 0.7 / math.sqrt(it)
        momentum = beta * momentum + (1.0 - beta) * g
        a_next = a - eta * momentum
        a = _simplex_projection(a_next)

        # occasional random perturbation if stagnant
        if it - last_improve > 150_000:
            pert = np.random.rand(n) * 0.001
            a = _simplex_projection(best_a + pert)
            last_improve = it

    return best_a, best_val

    # ------------------------------------------------------------------
    a = best_descent_vec.copy()
    best_swap_vec = a.copy()
    best_swap_val = best_descent_val
    temperature = 1.0
    cooling = 0.99995

    while time.time() < deadline:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        a[i], a[j] = a[j], a[i]

        maxb, _ = _conv_max_fft(a, fft_len)
        cheap = 2.0 * n * maxb

        if cheap < best_swap_val - 1e-12:
            best_swap_val = cheap
            best_swap_vec = a.copy()
            print(f"[Swap] NEW BEST cheap = {cheap:.6f}")
        else:
            # annealed acceptance of worse swaps
            delta = cheap - best_swap_val
            if np.random.rand() >= math.exp(-delta / max(temperature, 1e-12)):
                # reject – revert the swap
                a[i], a[j] = a[j], a[i]

        temperature = max(temperature * cooling, 1e-6)

    # ------------------------------------------------------------------
    # 4. Final exact evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_swap_vec.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:          # defensive – should never happen
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    return final_seq
```
