def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import numpy as np, math, time
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0

    # Determine n
    try:
        n = len(height_sequence_1)
        print(f"Using external length n = {n}")
    except Exception:
        n = 20000
        print(f"No external sequence – using default length n = {n}")

    # Precompute FFT length
    L = _fft_len_for_conv(n)

    # Build arcsine initial distribution
    i = np.arange(n, dtype=np.float64)
    x0 = i / n
    x1 = (i + 1) / n
    sqrt0 = np.sqrt(np.clip(x0, 0.0, 1.0))
    sqrt1 = np.sqrt(np.clip(x1, 0.0, 1.0))
    F0 = (2.0 / np.pi) * np.arcsin(sqrt0)
    F1 = (2.0 / np.pi) * np.arcsin(sqrt1)
    a0 = F1 - F0
    a0_sum = a0.sum()
    a0 = a0 / a0_sum   # normalize to sum=1

    best_a = a0.copy()
    a_cur = a0.copy()
    # Setup Adam variables for logits representation
    # Use logits = log(a) + small epsilon
    eps = 1e-12
    logits = np.log(a_cur + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_alpha = 0.8  # learning rate factor

    # Phase 1: minimize L2 error of convolution vs uniform
    it = 0
    best_score = 2.0 * n * _conv_max(best_a)  # raw cheap score
    print(f"Initial cheap score = {best_score:.8f}")

    # Phase 1 time split: allocate up to 0.85 of budget
    phase1_deadline = start + 0.85 * (budget_s - 2.0)
    while time.time() < phase1_deadline:
        it += 1
        # compute current distribution from logits
        a_cur = _softmax(logits)

        # compute FFT of a
        a_fft = np.fft.rfft(a_cur, L)

        # convolution self
        conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]

        # uniform target u = average of conv
        u_val = conv.mean()
        err = conv - u_val

        # gradient of squared error L2
        err_fft = np.fft.rfft(err, L)
        grad_err = 4.0 * np.fft.irfft(a_fft * err_fft, L)[: n]

        # optional smoothing max gradient? skip or combine later
        # Mirror descent gradient for logits
        mu = np.dot(grad_err, a_cur)
        grad_z = a_cur * (grad_err - mu)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step = - alpha_t * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits = logits + step
        logits = np.clip(logits, -60.0, 60.0)

        # Compute cheap score
        max_conv = conv.max()
        cheap_score = 2.0 * n * max_conv
        if cheap_score < best_score:
            best_score = cheap_score
            best_a = a_cur.copy()
            if it % 5000 == 0:
                print(f"Phase1 iter {it}: improved cheap_score = {best_score:.8f}")

    # Phase 2: refine using smoothed max gradient (like original)
    # Switch to max-based gradient to directly lower max.
    # Use same Adam but compute grad_max via _conv_max_and_grad
    # Use warm start of logits from phase1 final state (or from best)
    # Provide epsilon factor schedule similar to previous code
    eps_factor = 0.12
    eps_factor_min = 5e-4
    eps_factor_decay = 0.9993

    max_iters_phase2 = 50000
    # continue loop until deadline
    it2 = 0
    while time.time() < deadline and it2 < max_iters_phase2:
        it2 += 1
        a_cur = _softmax(logits)
        max_conv, grad_max = _conv_max_and_grad(a_cur, eps_factor)

        cheap_score = 2.0 * n * max_conv
        if cheap_score < best_score:
            best_score = cheap_score
            best_a = a_cur.copy()
            if it2 % 2000 == 0:
                print(f"Phase2 iter {it2}: improved cheap_score = {best_score:.8f}")

        # Mirror descent gradient on logits
        mu = np.dot(grad_max, a_cur)
        grad_z = a_cur * (grad_max - mu)

        # Adam update same parameters as before (reuse m,v)
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** (it + it2))
        v_hat = v / (1.0 - beta2 ** (it + it2))

        alpha_t = base_alpha / (math.sqrt(it + it2) + 1.0)
        step = - alpha_t * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits = logits + step
        logits = np.clip(logits, -60.0, 60.0)

        eps_factor = max(eps_factor_min, eps_factor * eps_factor_decay)

        # occasional jitter
        if it2 % 5000 == 0 and np.random.rand() < 0.07:
            logits += 0.004 * np.random.randn(n)

        # Early stop if score below target
        if best_score < 1.505:
            print(f"Early stopping: score reached {best_score:.8f}")
            break

    final_seq = best_a.tolist()
    try:
        exact_score = evaluate_sequence(final_seq)
        print(f"Exact evaluation: {exact_score:.8f}")
    except Exception as e:
        print(f"Exact evaluation failed: {e}")

    return final_seq

"""
Propose a low‑score step‑function.

Algorithm
---------
1.  Initialise a probability vector by discretising the arcsine law on a uniform
    grid – this gives a convolution that is almost flat.
2.  Phase 1 (long) – minimise the L2‑distance between the self‑convolution and
    the uniform target on the simplex.
    * Parameterise the distribution by soft‑max logits.
    * Use Adam on the simplex with the exact gradient
          ∇L = 4·(a ⟲ (a*a – mean(a*a)))       (⟲ = cross‑correlation).
    * FFT‑based convolutions give O(n log n) per iteration.
3.  Phase 2 (short) – polish the result with the smooth‑max gradient used in
    the reference solution, starting from the best vector found in Phase 1.
4.  Return the best sequence observed (as a plain list).  The final exact
    score is printed but does not affect the returned value.

The code respects the 1000 s budget, works for any length *n* (default 20000
if no external sequence is provided) and only uses NumPy.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max returning a probability vector."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Logits that map to the probability vector a via soft‑max."""
    return np.log(a + eps)


def _fft_len_for_conv(n: int) -> int:
    """
    Smallest power‑of‑two ≥ 2·n‑1,
    sufficient for a linear self‑convolution of length (2·n‑1).
    """
    return 1 << ((2 * n - 1).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a (no gradient)."""
    n = a.shape[0]
    L = _fft_len_for_conv(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Compute the smoothed maximum of a*a and the corresponding gradient.

    Parameters
    ----------
    a : np.ndarray
        Probability vector (non‑negative, sum = 1).
    eps_factor : float
        Multiplicative factor for the smoothing constant:
        eps = eps_factor * max(conv).

    Returns
    -------
    max_conv : float
        Maximum entry of the self‑convolution.
    grad_a : np.ndarray
        Gradient of 4·n·smooth_max w.r.t. a (already multiplied by 4·n).
    """
    n = a.shape[0]
    L = _fft_len_for_conv(n)

    a_fft = np.fft.rfft(a, L)

    # linear self‑convolution (length 2·n‑1)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # smoothed soft‑max weights (log‑sum‑exp trick)
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum

    # cross‑correlation a ⋆ w (linear, length n)
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # gradient (4·n multiplied)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a
