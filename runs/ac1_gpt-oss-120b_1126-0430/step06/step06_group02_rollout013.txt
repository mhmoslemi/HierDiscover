Stage 1 – mirror‑descent on the probability simplex (soft‑max logits) using
          Adam.  The objective is the smoothed maximum of the self‑convolution
          `a*a`.  The smoothing parameter ε = eps_factor·max_conv is decreased
          geometrically.  The learning‑rate decays like 1/√t, a back‑tracking
          line‑search is performed and occasional jitter/restraints help
          escape plateaux.  When a stagnation is detected the optimiser is
          re‑initialised around the best point found so far.

Stage 2 – one up‑sampling step (up to ≤ 20 000 points) is performed once the
          optimiser has made reasonable progress.  The distribution is
          linearly interpolated, the logits are recomputed and optimisation
          continues on the finer grid.

Stage 3 – a cheap simulated‑annealing refinement runs on the best sequence
          discovered by Adam.  Proposals are generated by adding Gaussian
          noise to the logits, re‑soft‑maxing, and are accepted either if they
          improve the exact `2·n·max(a*a)` value or with a temperature‑dependent
          probability.  This local search often pushes the bound a little
          further.

The routine respects a 1000 s budget (leaving a small safety margin for the
final call to `evaluate_sequence`).  The best sequence seen is finally
evaluated and returned as a plain list of non‑negative floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures or lambdas
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    s = e.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / z.size)
    return e / s


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it that reproduces `a` after a soft‑max."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """Smallest power‑of‑two ≥ 3·n‑2 (enough for all needed convolutions)."""
    return 1 << ((3 * n - 2).bit_length())


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    s = new_a.sum()
    if s == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= s
    return new_a


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{‑p}, normalised."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _exp_decay_distribution(n: int, lam: float) -> np.ndarray:
    """a_i ∝ e^{‑lam·i}+e^{‑lam·(n‑1‑i)}, symmetric exponential."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _logistic_distribution(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ 1/(1+e^{α·(i‑mid)}), normalised; left‑heavy."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = 1.0 / (1.0 + np.exp(alpha * (i - mid)))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Returns (max_conv, grad_a) where

        grad_a = ∇_a [ 2·n·smooth_max(a*a) ]

    `eps_factor` multiplies the smoothing constant ε = eps_factor·max_conv.
    The gradient is obtained via FFT‑based correlation.
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    # FFT of a (zero‑padded)
    a_fft = np.fft.rfft(a, L)

    # Linear self‑convolution a*a
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # Smoothed soft‑max weights
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv                # ≤ 0 for numerical stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # Correlation a ⋆ w   (same length as a)
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # Gradient (already multiplied by 4·n, i.e. ∇[2·n·smooth_max])
    grad_a = 4.0 * n * corr
    return max_conv, grad_a


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) improves the
    `evaluate_sequence` score.  The routine respects the given time budget.
    """
    np.random.seed(seed)
    start = time.time()
    # keep a small safety margin for the final exact evaluation
    deadline = start + budget_s - 2.0

    # --------------------------------------------------------------
    # 0. Initialise – choose a modest support size and a good start.
    # --------------------------------------------------------------
    n = 8000                     # initial support (fast enough for many iters)
    best_a = None
    best_score = float("inf")

    # Warm‑start from a global sequence if the variable exists
    try:
        global height_sequence_1   # noqa: F821   (provided by the platform)
        init_global = np.array(height_sequence_1, dtype=np.float64)
        best_a = _interpolate_distribution(init_global, n)
        best_score = 2.0 * n * _conv_max(best_a)
        print("[Init] Warm‑started from global sequence (down‑sampled).")
    except Exception:
        # No global – scan a few parametric families (denser than before)
        print("[Init] Scanning parametric families.")

        # Power‑law – exponents from 0.2 to 3.0
        for p in np.linspace(0.2, 3.0, 28):
            if time.time() >= deadline:
                break
            cand = _power_law_distribution(n, p)
            score = 2.0 * n * _conv_max(cand)
            if score < best_score:
                best_score, best_a = score, cand

        # Symmetric exponential decay (helps if a left‑heavy tail is useful)
        for loglam in np.linspace(-6, -1, 30):
            if time.time() >= deadline:
                break
            lam = 10.0 ** loglam
            cand = _exp_decay_distribution(n, lam)
            score = 2.0 * n * _conv_max(cand)
            if score < best_score:
                best_score, best_a = score, cand

        # Logistic left‑heavy (steepness up to 12)
        for alpha in np.linspace(0.1, 12.0, 35):
            if time.time() >= deadline:
                break
            cand = _logistic_distribution(n, alpha)
            score = 2.0 * n * _conv_max(cand)
            if score < best_score:
                best_score, best_a = score, cand

        # Fallback – uniform
        if best_a is None:
            best_a = np.full(n, 1.0 / n)
            best_score = 2.0 * n * _conv_max(best_a)

    # Small random jitter before optimisation (logits are the optimisation variables)
    z = _logits_from_distribution(best_a) + 0.01 * np.random.randn(n)

    # --------------------------------------------------------------
    # 1. Adam state on logits (mirror descent on the simplex)
    # --------------------------------------------------------------
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    base_alpha = 0.6           # a bit more aggressive than the previous code
    eps_factor = 0.25          # start with a relatively strong smoothing
    eps_factor_min = 5e-7
    eps_factor_decay = 0.9994

    report_every = 25000
    best_seq = _softmax(z)
    best_score = 2.0 * n * _conv_max(best_seq)
    last_improve_iter = 0
    it = 0

    # --------------------------------------------------------------
    # 2. Optimisation loop (Adam + occasional up‑sampling)
    # --------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # ----- Current point on the simplex -----
        a = _softmax(z)                 # always sums to 1

        # ----- Evaluate current max‑convolution and its gradient -----
        max_conv, grad_a = _conv_max_and_grad(a, eps_factor)
        cheap_score = 2.0 * n * max_conv

        # ----- Keep the best ever seen (cheap objective) -----
        if cheap_score < best_score - 1e-12:
            best_score = cheap_score
            best_seq = a.copy()
            last_improve_iter = it
            if it % report_every == 0:
                print(f"[Iter {it:07d}] NEW BEST cheap = {cheap_score:.6f}")

        # ----- Mirror‑descent gradient on logits -----
        mu = np.dot(grad_a, a)          # ⟨grad, a⟩
        grad_z = a * (grad_a - mu)

        # ----- Adam moment updates (bias‑corrected) -----
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # ----- Adaptive step‑size (inverse‑sqrt decay) -----
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step_dir = - m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- Simple back‑tracking line search (up to 5 halvings) -----
        accepted = False
        trial_alpha = alpha_t
        for _ in range(5):
            z_new = z + trial_alpha * step_dir
            # Clamp logits – prevents overflow in the exponential
            z_new = np.clip(z_new, -50.0, 50.0)
            a_new = _softmax(z_new)
            new_score = 2.0 * n * _conv_max(a_new)
            if new_score <= cheap_score:
                z = z_new
                accepted = True
                break
            trial_alpha *= 0.5

        if not accepted:
            # If no improvement found, still move a little (helps escape plateaux)
            z += alpha_t * step_dir
            z = np.clip(z, -50.0, 50.0)

        # ----- Random jitter (occasionally) -----
        if (it % 4000 == 0) and (np.random.rand() < 0.18):
            z += 0.025 * np.random.randn(n)
            m *= 0.5
            v *= 0.5

        # ----- Reduce the smoothing factor gradually -----
        eps_factor = max(eps_factor_min, eps_factor * eps_factor_decay)

        # ----- Restart if stagnation lasts too long -----
        if it - last_improve_iter > 250_000:
            print(f"[Iter {it}] Stagnation – re‑initialising around best point.")
            z = _logits_from_distribution(best_seq) + 0.03 * np.random.randn(n)
            m.fill(0.0)
            v.fill(0.0)
            eps_factor = 0.25
            last_improve_iter = it

        # ----- One up‑sampling stage (once) -----
        # Trigger when enough iterations have passed and we still have time.
        if (it == 180_000) and (n < 20000) and (time.time() + 30.0 < deadline):
            new_n = min(20000, int(n * 1.6))
            print(f"[Iter {it}] Upsampling support: {n} → {new_n}")
            best_seq = _interpolate_distribution(best_seq, new_n)
            z = _logits_from_distribution(best_seq) + 0.005 * np.random.randn(new_n)
            m = np.zeros_like(z)
            v = np.zeros_like(z)
            n = new_n
            # Re‑compute the cheap score for the new length
            best_score = 2.0 * n * _conv_max(best_seq)
            # Reset counters after the drastic change
            it = 0
            last_improve_iter = 0
            eps_factor = 0.25
            continue   # start the next loop with the enlarged problem

    # --------------------------------------------------------------
    # 3. Simulated‑annealing refinement (exact objective)
    # --------------------------------------------------------------
    # Use the best sequence found by Adam as the starting point.
    cur_seq = best_seq.copy()
    cur_score = 2.0 * n * _conv_max(cur_seq)
    print(f"[SA] Starting refinement – score = {cur_score:.6f}")

    # Temperature schedule: start relatively high to allow uphill moves,
    # then decay to near zero.
    sa_end = deadline - 0.7   # leave a little time for final evaluation
    while time.time() < sa_end:
        # progressively shrink the proposal noise
        remaining = max(1e-3, sa_end - time.time())
        frac = remaining / (deadline - start)
        sigma = 0.03 * frac          # noise amplitude shrinks with time
        temp = 0.001 * frac          # temperature shrinks as well

        # Propose a new point by perturbing the logits
        z_prop = _logits_from_distribution(cur_seq) + sigma * np.random.randn(n)
        z_prop = np.clip(z_prop, -50.0, 50.0)
        prop_seq = _softmax(z_prop)
        prop_score = 2.0 * n * _conv_max(prop_seq)

        Δ = prop_score - cur_score
        accept = False
        if Δ < 0:
            accept = True
        else:
            # Metropolis criterion
            if temp > 0 and np.random.rand() < math.exp(-Δ / temp):
                accept = True

        if accept:
            cur_seq = prop_seq
            cur_score = prop_score
            # Keep the global best if this move is a new record
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_seq = cur_seq.copy()
                print(f"[SA] NEW GLOBAL BEST = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Final exact evaluation with the provided evaluate_sequence()
    # --------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:   # safety net – should never happen
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # --------------------------------------------------------------
    # 5. Return the best sequence discovered.
    # --------------------------------------------------------------
    return final_seq
