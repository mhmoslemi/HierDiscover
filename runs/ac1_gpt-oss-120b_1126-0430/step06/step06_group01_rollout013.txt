#!/usr/bin/env python
"""
Improved arcsine‑bin optimizer.

The metric to minimise is

    2·n·max(a∗a) / (∑a)² ,

which for a normalised (∑a=1) vector reduces to 2·n·max(a∗a).
The continuous arcsine law gives a perfectly flat convolution, but the
discrete binning is imperfect.  This routine starts from the exact
discrete arcsine bins (or from the provided external sequence) and
performs a long Adam‑optimisation on the probability simplex.
Key upgrades relative to the reference implementation:

* much larger iteration budget (the whole time‑limit is used);
* a smoother annealing schedule for the “soft‑max” temperature;
* a larger initial learning‑rate (base_alpha=2.0) with a sqrt‑decay;
* aggressive temperature decay (initial eps_factor≈0.8 → 5e‑6);
* occasional random jitter to escape plateaux;
* periodic logging of the best score seen so far.

The optimiser works entirely with FFT‑based convolutions (O(n log n))
and never calls the expensive ``evaluate_sequence`` inside the loop.
Only the final candidate is evaluated exactly.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max, returns a probability vector (sums to 1)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Logits that map to ``a`` via the soft‑max."""
    return np.log(a + eps)


def _fft_len_for_conv(n: int) -> int:
    """Smallest power‑of‑two ≥ 2·n‑1 (linear self‑convolution length)."""
    return 1 << ((2 * n - 1).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a."""
    n = a.shape[0]
    L = _fft_len_for_conv(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Smoothed max of a*a and its gradient w.r.t. a (already multiplied by 4·n).

    Parameters
    ----------
    a : np.ndarray
        Probability vector, non‑negative, sum = 1.
    eps_factor : float
        Multiplicative factor for the smoothing constant:
        eps = eps_factor * max(conv).

    Returns
    -------
    max_conv : float
        Maximum entry of the self‑convolution.
    grad_a : np.ndarray
        Gradient of 4·n·smooth_max w.r.t. a.
    """
    n = a.shape[0]
    L = _fft_len_for_conv(n)

    a_fft = np.fft.rfft(a, L)

    # linear self‑convolution (length 2·n‑1)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # smoothed soft‑max weights (log‑sum‑exp trick)
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv                # improve stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum

    # cross‑correlation a ⋆ w (linear, length 3·n‑2)
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # gradient (already multiplied by 4·n)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, ∑ x_i = 1}.
    """
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    the evaluation metric.  The routine:

    1. Builds the exact discrete arcsine distribution on a uniform grid
       (or uses the external ``height_sequence_1`` if it exists).
    2. Optimises the distribution with a long Adam run on the simplex,
       employing an aggressive temperature‑annealing schedule.
    3. Returns the best sequence found (as a plain Python list).
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for final logging / evaluation
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # 0. Length of the vector.
    # ------------------------------------------------------------------
    try:
        external_seq = height_sequence_1  # type: ignore[name-defined]
        n = len(external_seq)
        print(f"Using external length n = {n}")
        a0 = np.array(external_seq, dtype=np.float64)
        a0 = np.maximum(a0, 0.0)
        if a0.sum() <= 0.0:
            raise ValueError("External sequence sums to zero.")
        a0 /= a0.sum()
    except Exception:
        # default length – a value that is large enough to give a good
        # discretisation but still cheap for many FFT‑iterations.
        n = 20000
        print(f"No external sequence – using default length n = {n}")

        # ------------------------------------------------------------------
        # 1. Exact discretisation of the arcsine law (bin probabilities).
        # ------------------------------------------------------------------
        i = np.arange(n, dtype=np.float64)
        x0 = i / n
        x1 = (i + 1) / n

        sqrt0 = np.sqrt(np.clip(x0, 0.0, 1.0))
        sqrt1 = np.sqrt(np.clip(x1, 0.0, 1.0))
        F0 = (2.0 / np.pi) * np.arcsin(sqrt0)
        F1 = (2.0 / np.pi) * np.arcsin(sqrt1)

        a0 = F1 - F0                     # bin masses, sum ≈ 1
        a_sum = a0.sum()
        if not np.isfinite(a_sum) or a_sum <= 0.0:
            raise RuntimeError("Failed to generate arcsine bins.")
        a0 /= a_sum                      # ensure exact normalisation (sum = 1)

    # ------------------------------------------------------------------
    # 2. Initialise Adam on the logits (soft‑max parametrisation).
    # ------------------------------------------------------------------
    z = _logits_from_distribution(a0)
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    # Adam hyper‑parameters (tuned for a long run)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    # Learning‑rate schedule
    base_alpha = 2.0          # larger than the reference 0.6
    # Temperature (smoothing) schedule
    eps_factor = 0.8          # start fairly smooth
    eps_factor_min = 5e-6
    eps_factor_decay = 0.99993   # very slow decay – many iterations needed

    best_a = a0.copy()
    best_score = float('inf')
    best_iter = 0

    it = 0
    # ------------------------------------------------------------------
    # Main optimisation loop.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # Current probability vector
        a_cur = _softmax(z)

        # Smoothed max and its gradient
        max_conv, grad_a = _conv_max_and_grad(a_cur, eps_factor)

        # cheap (exactly the objective for sum=1)
        cheap_score = 2.0 * n * max_conv

        if cheap_score < best_score:
            best_score = cheap_score
            best_a = a_cur.copy()
            best_iter = it
            if it % 5000 == 0:
                print(f"Iter {it:6d}: new best score = {best_score:.8f}")

        # --------------------------------------------------------------
        # Adam update on the logits – mirrors gradient onto the simplex.
        # --------------------------------------------------------------
        mu = np.dot(grad_a, a_cur)                     # ∇F·a
        grad_z = a_cur * (grad_a - mu)                 # mirrored gradient

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step size (sqrt decay, works well with Adam)
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)

        step = -alpha_t * m_hat / (np.sqrt(v_hat) + eps_adam)
        z = z + step

        # keep logits in a safe range for exp()
        z = np.clip(z, -80.0, 80.0)

        # --------------------------------------------------------------
        # Temperature annealing
        # --------------------------------------------------------------
        eps_factor = max(eps_factor_min, eps_factor * eps_factor_decay)

        # --------------------------------------------------------------
        # Occasional random jitter – helps to escape plateaux.
        # --------------------------------------------------------------
        if it % 8000 == 0 and np.random.rand() < 0.12:
            jitter = 0.008 * np.random.randn(n)
            z += jitter
            # keep the jitter‑affected logits bounded again
            z = np.clip(z, -80.0, 80.0)

        # --------------------------------------------------------------
        # Early‑exit safety: if we are within a few seconds of the deadline,
        # break out to avoid overshooting.
        # --------------------------------------------------------------
        if it % 5000 == 0 and time.time() > deadline - 5.0:
            break

    # ------------------------------------------------------------------
    # 3. Final evaluation (exact, using the provided evaluate_sequence).
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        exact_score = evaluate_sequence(final_seq)
        print(f"Finished after {it:,d} iterations (best at {best_iter:,d}).")
        print(f"Best cheap score (2·n·max_conv) = {best_score:.8f}")
        print(f"Exact evaluation (after optimisation) = {exact_score:.8f}")
    except Exception as exc:
        print(f"Exact evaluation failed: {exc}")

    return final_seq
