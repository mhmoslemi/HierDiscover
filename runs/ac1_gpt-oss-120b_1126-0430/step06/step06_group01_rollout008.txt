#!/usr/bin/env python
"""
step‑function optimiser

* builds several analytically‑motivated initial distributions
  (exact arcsine bins, midpoint arcsine density, Beta(p,p) family)
* picks the best of them as a start point
* performs a long Adam‑style mirror‑descent on the probability simplex
  (log‑softmax parametrisation) with a smoothly‑approximated L∞ loss
* enforces symmetry of the vector at every step (the optimal solution is
  symmetric, this halves the effective dimension and improves convergence)
* runs until the time budget (≈1000 s) is exhausted and finally returns the
  best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable softmax; returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    s = e.sum()
    if s == 0:
        # fallback to uniform distribution
        return np.full_like(z, 1.0 / z.size)
    return e / s


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Logits that map to a via the softmax (inverse of the softmax, up to a constant)."""
    return np.log(a + eps)


def _fft_len_for_conv(n: int) -> int:
    """Smallest power‑of‑two ≥ 2·n‑1, sufficient for a linear self‑convolution."""
    return 1 << ((2 * n - 1).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a (no gradient)."""
    n = a.shape[0]
    L = _fft_len_for_conv(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Compute a smoothed maximum of a*a and its gradient w.r.t. a.

    Parameters
    ----------
    a : np.ndarray
        Probability vector (sum = 1, non‑negative).
    eps_factor : float
        Multiplicative factor that determines the smoothing constant:
        eps = max(eps_factor * max(conv), 1e-12).

    Returns
    -------
    max_conv : float
        Maximum entry of the self‑convolution.
    grad_a : np.ndarray
        Gradient of 2·n·smooth_max w.r.t. a (already multiplied by 2·n).
    """
    n = a.shape[0]
    L = _fft_len_for_conv(n)

    a_fft = np.fft.rfft(a, L)

    # linear self‑convolution (length 2·n‑1)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # smooth maximum via log‑sum‑exp
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv          # improve numerical stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum

    # gradient: 2 * (a ⋆ w).  We compute the (linear) cross‑correlation.
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # 2·n factor (the outer 2·n from the evaluation function) is folded in.
    grad_a = 2.0 * n * (2.0 * corr)   # the inner 2 comes from d(conv)/da = 2·a⋆w
    return max_conv, grad_a


def _beta_distribution(n: int, p: float) -> np.ndarray:
    """Return a probability vector proportional to the Beta(p,p) density evaluated at bin mid‑points."""
    i = np.arange(n, dtype=np.float64)
    mid = (i + 0.5) / n                       # mid‑points of the bins
    # Beta(p,p) pdf: x^{p-1} (1-x)^{p-1} / B(p,p)
    # Normalising constant via the Gamma function
    from math import gamma
    B = gamma(p) * gamma(p) / gamma(2.0 * p)
    pdf = np.power(mid, p - 1.0) * np.power(1.0 - mid, p - 1.0) / B
    a = pdf / n                               # approximate mass = pdf * Δx, Δx = 1/n
    a_sum = a.sum()
    if a_sum == 0 or not np.isfinite(a_sum):
        a = np.full(n, 1.0 / n)
    else:
        a = a / a_sum
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimises the
    evaluation metric 2·n·max(a*a)/(sum a)².

    The routine:
    1. builds a few analytically‑motivated starting vectors;
    2. picks the best as a start;
    3. runs a long Adam mirror‑descent on the simplex with a
       smoothly‑approximated L∞ loss;
    4. enforces symmetry at every iteration;
    5. returns the best sequence found before the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for final logging / return
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # 0. Determine the length of the vector.
    # ------------------------------------------------------------------
    try:
        # If a previous solution exists we reuse its length.
        n = len(height_sequence_1)  # type: ignore[name-defined]
        print(f"Using external length n = {n}")
    except Exception:
        n = 20000
        print(f"No external sequence – using default length n = {n}")

    # ------------------------------------------------------------------
    # 1. Build a few analytical candidates.
    # ------------------------------------------------------------------
    i = np.arange(n, dtype=np.float64)

    # ---- (a) exact arcsine bin probabilities (CDF differences) ----
    x0 = i / n
    x1 = (i + 1) / n
    sqrt0 = np.sqrt(np.clip(x0, 0.0, 1.0))
    sqrt1 = np.sqrt(np.clip(x1, 0.0, 1.0))
    F0 = (2.0 / np.pi) * np.arcsin(sqrt0)
    F1 = (2.0 / np.pi) * np.arcsin(sqrt1)
    arcsine_bins = F1 - F0
    arcsine_bins = np.maximum(arcsine_bins, 0.0)
    arcsine_bins /= arcsine_bins.sum()

    # ---- (b) arcsine density at bin mid‑points (mid‑point rule) ----
    mid = (i + 0.5) / n
    eps = np.finfo(float).eps
    pdf_mid = 1.0 / (np.pi * np.sqrt(np.clip(mid * (1.0 - mid), eps, None)))
    arcsine_mid = pdf_mid / n
    arcsine_mid = np.maximum(arcsine_mid, 0.0)
    arcsine_mid /= arcsine_mid.sum()

    # ---- (c) scan a small family of Beta(p,p) distributions ----
    best_beta = arcsine_bins           # fallback
    best_beta_score = np.inf
    for p in np.linspace(0.35, 0.65, 7):   # 7 points in a plausible interval
        a_beta = _beta_distribution(n, p)
        score = 2.0 * n * _conv_max(a_beta)
        # keep the best beta candidate
        if score < best_beta_score:
            best_beta_score = score
            best_beta = a_beta
    # ------------------------------------------------------------------
    # 2. Pick the best among the analytical candidates (and maybe the
    #    external sequence if it exists).
    # ------------------------------------------------------------------
    candidates = [(arcsine_bins, "arcsine_bins"),
                  (arcsine_mid,  "arcsine_mid"),
                  (best_beta,    "beta_opt")]
    # try to include an external sequence if present
    try:
        external = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        external = np.maximum(external, 0.0)
        if external.sum() > 0:
            external = external / external.sum()
            candidates.append((external, "external"))
    except Exception:
        pass

    best_a = None
    best_score = np.inf
    for vec, name in candidates:
        score = 2.0 * n * _conv_max(vec)
        print(f"Initial {name:12s}  score = {score:.8f}")
        if score < best_score:
            best_score = score
            best_a = vec.copy()
    print(f"Chosen start vector – score = {best_score:.8f}")

    # ------------------------------------------------------------------
    # 3. Adam mirror‑descent on the simplex (soft‑max parametrisation)
    #    with explicit symmetry enforcement.
    # ------------------------------------------------------------------
    # initialise logits from the best start vector
    z = _logits_from_distribution(best_a)
    # enforce symmetry on logits (distribution will be symmetric automatically)
    z = (z + z[::-1]) / 2.0

    m = np.zeros_like(z)      # first moment (Adam)
    v = np.zeros_like(z)      # second moment (Adam)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_alpha = 0.8          # base learning‑rate factor
    eps_factor = 0.3          # initial smoothing factor for the soft‑max over the convolution
    eps_factor_min = 1e-5
    eps_factor_decay = 0.9995

    it = 0
    max_iters = 300000        # upper bound – we will stop earlier when time runs out

    while time.time() < deadline and it < max_iters:
        it += 1

        # ---- probability vector (automatically symmetric) ----
        a = _softmax(z)            # sum(a) == 1, a_i >= 0

        # ---- evaluate smoothed max‑convolution and its gradient ----
        max_conv, grad_a = _conv_max_and_grad(a, eps_factor)

        cheap_score = 2.0 * n * max_conv   # exact objective because sum(a)=1

        if cheap_score < best_score:
            best_score = cheap_score
            best_a = a.copy()
            if it % 1000 == 0:
                elapsed = time.time() - start_time
                print(f"Iter {it:6d} – improved score = {best_score:.8f} (t={elapsed:.1f}s)")

        # ---- mirror‑descent step on logits ----
        mu = np.dot(grad_a, a)                # <grad_a , a>
        grad_z = a * (grad_a - mu)            # ∇_z  (mirror map)

        # Adam bias‑corrected update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step = - alpha_t * m_hat / (np.sqrt(v_hat) + eps_adam)

        z = z + step

        # ---- enforce symmetry of the logits and keep them numerically safe ----
        z = (z + z[::-1]) / 2.0
        z = np.clip(z, -70.0, 70.0)          # avoids overflow/underflow in exp

        # ---- slowly tighten the smoothing of the max ----
        eps_factor = max(eps_factor_min, eps_factor * eps_factor_decay)

        # ---- occasional tiny random jitter to escape plateaus ----
        if it % 4000 == 0 and np.random.rand() < 0.1:
            jitter = 0.002 * np.random.randn(n)
            jitter = (jitter + jitter[::-1]) / 2.0   # keep symmetry
            z += jitter
            z = (z + z[::-1]) / 2.0

        # ---- optional early‑stop if we have already crossed a good threshold ----
        if it % 20000 == 0:
            elapsed = time.time() - start_time
            print(f"Iter {it:6d} – current best = {best_score:.8f} (t={elapsed:.1f}s)")
            if best_score < 1.02:
                print("Reached a very good bound – exiting early.")
                break

    # ------------------------------------------------------------------
    # 4. Final exact evaluation and return.
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        exact_score = evaluate_sequence(final_seq)
        print(f"Exact evaluation (final) = {exact_score:.8f}")
    except Exception as exc:
        print(f"Exact evaluation failed: {exc}")

    return final_seq
