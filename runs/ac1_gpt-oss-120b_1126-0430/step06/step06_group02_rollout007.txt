def project_simplex(v):
    # from: https://github.com/joriskala/pytorch-simplex-projection
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max() - 1  # zero-based index
    theta = cssv[ rho ] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

class AdamLogits:
    def __init__(self, a_init, base_alpha=0.5, beta1=0.9,beta2=0.999, eps=1e-8):
        self.z = np.log(a_init + 1e-16)
        self.m = np.zeros_like(self.z)
        self.v = np.zeros_like(self.z)
        self.base_alpha = base_alpha
        self.beta1 = beta1
        self.beta2 = beta2
        self.eps = eps
        self.it = 0
    def step(self, eps_factor):
        # compute a via softmax
        a = _softmax(self.z)
        max_conv, grad_a = smoothed_conv_max_and_grad(a, eps_factor)
        # grad wrt logits
        mu = np.dot(grad_a, a)
        grad_z = a * (grad_a - mu)
        self.it += 1
        self.m = self.beta1 * self.m + (1 - self.beta1) * grad_z
        self.v = self.beta2 * self.v + (1 - self.beta2) * (grad_z ** 2)
        m_hat = self.m / (1 - self.beta1 ** self.it)
        v_hat = self.v / (1 - self.beta2 ** self.it)
        alpha_t = self.base_alpha / (np.sqrt(self.it) + 1.0)
        step = - m_hat / (np.sqrt(v_hat) + self.eps)
        # line search acceptance
        return a, max_conv, grad_a, step
    def apply_step(self, step, trial_alpha):
        # accept step with trial_alpha
        self.z = self.z + trial_alpha * step
        np.clip(self.z, -50, 50, out=self.z)

#!/usr/bin/env python
"""
Robust step‑function optimiser.

The optimiser works on a modest support (≈ 4 000‑25 000 points) and
uses a two‑stage Adam on the soft‑max (mirror‑descent on the simplex)
combined with a slowly vanishing smoothing of the max‑convolution.
Occasional up‑sampling, random jitter and a stagnation‑restart give
the algorithm many chances to escape plateaux.  The best distribution
seen is finally evaluated with the supplied ``evaluate_sequence``.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures or lambdas
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max, returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    s = e.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / z.size)
    return e / s


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it that reproduces ``a`` after a soft‑max."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """Smallest power‑of‑two ≥ 3·n‑2 (enough for all needed convolutions)."""
    return 1 << ((3 * n - 2).bit_length())


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    s = new_a.sum()
    if s == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= s
    return new_a


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{‑p}, normalised."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _sym_product_power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ ((i+1)*(n‑i))^{‑p}, symmetric, normalised."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) * (n - i)) ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _exp_decay_distribution(n: int, lam: float) -> np.ndarray:
    """a_i ∝ e^{‑lam·i}+e^{‑lam·(n‑1‑i)}, symmetric exponential."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _logistic_distribution(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ 1/(1+e^{α·(i‑mid)}), normalised; heavy on the left side."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = 1.0 / (1.0 + np.exp(alpha * (i - mid)))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _random_piecewise_distribution(n: int, knots: int = 8) -> np.ndarray:
    """Random piece‑wise linear distribution with ``knots`` break‑points."""
    xs = np.sort(np.random.rand(knots) * (n - 1))
    xs = np.concatenate(([0.0], xs, [n - 1.0]))
    ys = np.random.rand(knots + 2)  # positive heights
    # linear interpolation on integer grid
    xi = np.arange(n, dtype=np.float64)
    a = np.interp(xi, xs, ys)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Returns (max_conv, grad_a) where

        grad_a = ∇_a [ 2·n·smooth_max(a*a) ]

    `eps_factor` multiplies the smoothing constant ε = eps_factor·max_conv.
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    # FFT of a (zero‑padded)
    a_fft = np.fft.rfft(a, L)

    # Linear self‑convolution a*a
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # Smoothed soft‑max weights
    eps = max(max_conv * eps_factor, 1e-15)
    shifted = conv - max_conv                # ≤ 0 for numerical stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # Correlation a ⋆ w
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # Gradient (already multiplied by 4·n)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection only defined for 1‑D arrays.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive, return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:          # pragma: no cover – safety net
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # --------------------------------------------------------------
    # 6. Return the best sequence discovered.
    # --------------------------------------------------------------
    return final_seq
