Let's think about projection onto simplex: standard algorithm:

def project_to_simplex(v):
    # v: np array
    # returns projection onto probability simplex
    # implementation from Duchi et al.
    u = np.sort(v)[::-1]
    sv = np.cumsum(u)
    rho = np.where(u > (sv - 1) / (np.arange(1, len(v)+1)))[0][-1]
    theta = (sv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _subgradient_descent(a_init, time_deadline, max_iter=...):

    a = a_init.copy()

    best_a = a.copy()
    best_score = cheap_score = 2*n*_conv_max(a)

    start_time = time.time()
    it = 0
    step0 = 1.0
    while time.time() < time_deadline and it < max_iter:
        it += 1

        L = _fft_len_for_corr(n)
        a_fft = np.fft.rfft(a, L)

        conv = np.fft.irfft(a_fft * a_fft, L)[:2*n-1]
        max_val = conv.max()
        cheap_score = 2.0 * n * max_val

        # Update best
        if cheap_score < best_score - 1e-12:
            best_score = cheap_score
            best_a = a.copy()

        # compute gradient via correlation on argmax set
        # tolerance to find all near max
        tol = 1e-12
        max_inds = np.flatnonzero(conv >= max_val - tol)
        # compute w_fft analytically:
        freqs = np.arange(L // 2 + 1)
        if max_inds.size == 1:
            k = max_inds[0]
            w_fft = np.exp(-2j * np.pi * freqs * k / L)
        else:
            # average of exponentials
            exp_mat = np.exp(-2j * np.pi * freqs[:, None] * max_inds / L)
            w_fft = exp_mat.mean(axis=1)
        # correlation
        corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]
        grad = a + corr  # subgradient (positive)
        # step size
        eta = step0 / (np.sqrt(it) + 1.0)
        # additive update
        a = a - eta * grad
        # project onto simplex (nonnegative, sum=1)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0/n)
        else:
            a /= s

        # occasional random jitter
        if it % 5000 == 0 and np.random.rand() < 0.1:
            # small random noise
            a += 0.001 * np.random.randn(n)
            a = np.maximum(a, 0.0)
            a /= a.sum()
    return best_a, best_score

def _beta_distribution(n, alpha, beta):
    i = np.arange(n, dtype=np.float64) + 0.5  # centers
    x = i / n
    a = np.power(x, alpha - 1) * np.power(1 - x, beta - 1)

def _asym_double_power_law_distribution(n, p1, p2, off1, off2):
    i = np.arange(n, dtype=np.float64)
    left = (i + off1 + 1) ** (-p1)   # i+1 shifted
    right = (n - 1 - i + off2 + 1) ** (-p2)  # (n-i) shift
    a = left * right
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a

"""
Hybrid optimizer for the self‑convolution bound.

The search proceeds in four stages:

1. **Coarse parametric sweep** – a collection of analytic families
   (power‑law, offset power‑law, double‑power‑law, stretched‑exponential,
   exponential, logistic, symmetric product‑power‑law, Beta,
   asymmetric double‑power‑law) is sampled on a grid.
   The cheap score `2·n·max(a*a)` (where `max` is the peak of the linear
   self‑convolution) is used to keep the best candidate.

2. **Multiplicative mirror descent on the simplex** – starting from the best
   candidate, we perform many iterations of an exponentiated‑gradient update
   that minimizes the smoothed max‑convolution.  The update uses only two
   FFTs per iteration (self‑convolution and the correlation with the soft‑max
   weights).  The smoothing constant ε decays slowly; a small random jitter
   is injected periodically to escape plateaus.  Restarts are performed if
   no improvement is seen for a long while.

3. **Cheap random walk** – the remaining time is spent on tiny coordinate
   transfers that are accepted only if they improve the cheap score.

4. **Exact evaluation** – the final probability vector (sum ≈ 1, non‑negative)
   is converted to a plain Python list and passed to the provided
   `evaluate_sequence` function.

The algorithm respects a 1000 s time budget, prints progress and returns
the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max, returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    s = e.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / z.size)
    return e / s


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it that reproduces `a` after a soft‑max."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """Smallest power‑of‑two ≥ 3·n‑2 (enough for all needed convolutions)."""
    return 1 << ((3 * n - 2).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Returns (max_conv, grad_a) where

        grad_a = ∇_a [ 2·n·smooth_max(a*a) ]

    `eps_factor` multiplies the smoothing constant ε = eps_factor·max_conv.
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv                     # ≤ 0 for stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # Gradient of 2·n·smooth_max (the factor 4·n comes from the derivative of a*a)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a


# ----------------------------------------------------------------------
# Parametric families of distributions
# ----------------------------------------------------------------------


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}   (normalised)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _power_law_offset_distribution(n: int, p: float, offset: float) -> np.ndarray:
    """a_i ∝ (i+offset)^{-p}   (normalised)."""
    idx = np.arange(offset, offset + n, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _double_power_law_distribution(n: int, p: float, q: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}·(n-i)^{-q}   (normalised)."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) ** (-p)) * ((n - i) ** (-q))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _stretched_exp_distribution(n: int, tau_factor: float, gamma: float) -> np.ndarray:
    """
    a_i ∝ exp( - (i/(tau_factor·n))^{gamma} )   (normalised).
    `tau_factor` ∈ (0,1] stretches the decay length.
    """
    tau = max(1e-12, tau_factor * n)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(- (i / tau) ** gamma)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _exp_decay_distribution(n: int, lam: float) -> np.ndarray:
    """a_i ∝ e^{-lam·i}+e^{-lam·(n‑1‑i)}   (normalised)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _logistic_distribution(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ 1/(1+e^{α·(i‑mid)})   (normalised)."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = 1.0 / (1.0 + np.exp(alpha * (i - mid)))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _sym_product_power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ ((i+1)*(n‑i))^{-p}   (symmetric, normalised)."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) * (n - i)) ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(α,β) density on the discrete grid (centres of bins)."""
    i = np.arange(n, dtype=np.float64) + 0.5  # bin centres
    x = i / n
    eps = 1e-12
    a = (x + eps) ** (alpha - 1) * (1.0 - x + eps) ** (beta - 1)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _asym_double_power_law_distribution(
    n: int, p1: float, p2: float, off1: float, off2: float
) -> np.ndarray:
    """a_i ∝ (i+off1+1)^{-p1}·(n‑i+off2)^{-p2}   (normalised)."""
    i = np.arange(n, dtype=np.float64)
    left = (i + off1 + 1.0) ** (-p1)
    right = (n - i + off2) ** (-p2)
    a = left * right
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    `evaluate_sequence` bound.  The routine respects a CPU‑time budget,
    performs a parametric sweep, a fast exponentiated‑gradient descent on the
    probability simplex, and a final cheap random walk.
    Returns a list of heights.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0   # safety margin for final evaluation

    # --------------------------------------------------------------
    # Initialise problem size (use a modest n for speed)
    # --------------------------------------------------------------
    try:
        # The platform may provide an initial sequence.
        init_seq = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        # Down‑sample if the provided sequence is larger than our target size.
        target_n = 12000
        if init_seq.shape[0] > target_n:
            idx = np.linspace(0, init_seq.shape[0] - 1, target_n, dtype=int)
            init_seq = init_seq[idx]
        n = init_seq.shape[0]
        a0 = np.maximum(init_seq, 0.0)
        s = a0.sum()
        if s == 0.0:
            a0 = np.full_like(a0, 1.0 / n)
        else:
            a0 /= s
        print(f"[Init] Using provided sequence, length = {n}")
    except Exception:
        n = 12000
        a0 = np.full(n, 1.0 / n, dtype=np.float64)
        print(f"[Init] No provided sequence – start from uniform, length = {n}")

    best_seq = a0.copy()
    best_score = 2.0 * n * _conv_max(best_seq)

    # --------------------------------------------------------------
    # 0. Coarse parametric sweep (grid search)
    # --------------------------------------------------------------
    families = [
        ("power_law", _power_law_distribution,
         [dict(p=p) for p in np.linspace(0.40, 0.80, 41)]),
        ("power_law_offset", _power_law_offset_distribution,
         [dict(p=p, offset=off)
          for p in np.linspace(0.40, 0.80, 21)
          for off in np.linspace(0.5, 5.0, 21)]),
        ("double_power_law", _double_power_law_distribution,
         [dict(p=p, q=q)
          for p in np.linspace(0.40, 0.80, 17)
          for q in np.linspace(0.40, 0.80, 17)]),
        ("stretched_exp", _stretched_exp_distribution,
         [dict(tau_factor=t, gamma=g)
          for t in np.linspace(0.10, 0.90, 17)
          for g in np.linspace(0.5, 2.0, 17)]),
        ("exp_decay", _exp_decay_distribution,
         [dict(lam=lam) for lam in np.logspace(-6, -2, 30)]),
        ("logistic", _logistic_distribution,
         [dict(alpha=alpha) for alpha in np.linspace(0.01, 6.0, 30)]),
        ("sym_product_power_law", _sym_product_power_law_distribution,
         [dict(p=p) for p in np.linspace(0.40, 1.20, 41)]),
        ("beta", _beta_distribution,
         [dict(alpha=alpha, beta=beta)
          for alpha in np.linspace(0.2, 2.0, 15)
          for beta in np.linspace(0.2, 2.0, 15)]),
        ("asym_double_power", _asym_double_power_law_distribution,
         [dict(p1=p1, p2=p2, off1=off1, off2=off2)
          for p1 in np.linspace(0.2, 0.8, 7)
          for p2 in np.linspace(0.2, 0.8, 7)
          for off1 in (0.0, 1.0, 3.0)
          for off2 in (0.0, 1.0, 3.0)])
    ]

    print("[Sweep] Starting parametric search")
    for name, generator, param_grid in families:
        for params in param_grid:
            if time.time() >= deadline:
                break
            try:
                cand = generator(n, **params)
            except Exception:
                continue
            score = 2.0 * n * _conv_max(cand)
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                # Uncomment the next line for verbose per‑candidate logging
                # print(f"[Sweep] New best {name} {params} -> {score:.6f}")
        if time.time() >= deadline:
            break
    print(f"[Sweep] Best cheap score after sweep = {best_score:.6f}")

    # --------------------------------------------------------------
    # 1. Multiplicative mirror descent (exponentiated gradient)
    # --------------------------------------------------------------
    a = best_seq.copy()
    best_seq_sofar = a.copy()
    best_score_sofar = best_score
    it = 0
    last_improve = 0

    # hyper‑parameters
    base_eta = 0.20                     # learning‑rate for multiplicative step
    eps_factor = 0.12                   # initial smoothing factor ε = eps_factor·max_conv
    eps_decay = 0.9995
    eps_factor_min = 5e-5
    max_iters = 1_500_000
    jitter_interval = 5000
    jitter_prob = 0.12
    jitter_scale = 0.001
    restart_patience = 300_000

    while time.time() < deadline and it < max_iters:
        it += 1

        max_conv, grad_a = _conv_max_and_grad(a, eps_factor)
        cheap_score = 2.0 * n * max_conv

        # record best seen
        if cheap_score < best_score_sofar - 1e-12:
            best_score_sofar = cheap_score
            best_seq_sofar = a.copy()
            last_improve = it

        # exponentiated‑gradient update (mirror descent w.r.t. KL)
        eta = base_eta / (math.sqrt(it) + 1.0)
        factor = np.exp(-eta * grad_a)          # positive multiplicative factor
        # avoid underflow/overflow extremes
        factor = np.clip(factor, 1e-12, 1e12)
        a = a * factor
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # occasional random jitter to break symmetry
        if it % jitter_interval == 0 and np.random.rand() < jitter_prob:
            a += jitter_scale * np.random.randn(n) * a
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum == 0.0:
                a = np.full_like(a, 1.0 / n)
            else:
                a /= a_sum

        # decay smoothing constant
        eps_factor = max(eps_factor_min, eps_factor * eps_decay)

        # restart if stagnation detected
        if it - last_improve > restart_patience:
            print(f"[Iter {it}] Stagnation – restart from best so far")
            a = best_seq_sofar.copy()
            a += 0.005 * np.random.randn(n) * a
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum == 0.0:
                a = np.full_like(a, 1.0 / n)
            else:
                a /= a_sum
            eps_factor = 0.12
            last_improve = it

    best_seq = best_seq_sofar
    best_score = best_score_sofar
    print(f"[Descent] Finished after {it} iterations, best cheap = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2. Final cheap random walk (coordinate transfers)
    # --------------------------------------------------------------
    a = best_seq.copy()
    cheap_score = best_score

    while time.time() < deadline:
        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        if i == j:
            continue
        delta = 0.001 * a[i]               # move 0.1 % of mass at i
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = 2.0 * n * _conv_max(a_new)
        if new_score < cheap_score:
            a = a_new
            cheap_score = new_score
            best_seq = a.copy()
            best_score = cheap_score
            # very sparse progress prints
            if np.random.rand() < 0.001:
                print(f"[Walk] Improved cheap = {cheap_score:.8f}")

    # --------------------------------------------------------------
    # 3. Exact evaluation with the provided evaluate_sequence()
    # --------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        exact_score = evaluate_sequence(final_seq)   # provided by the platform
        print(f"Final exact score = {exact_score:.6f}")
    except Exception as exc:
        print(f"Error during final evaluation: {exc}")

    return final_seq
