    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0

    # Determine n
    try:
        n0 = len(height_sequence_1)
    except Exception:
        n0 = 0
    n = max(n0, 40000)   # make it at least 40000
    print(f"Using n = {n}")

    # Build arcsine bins
    i = np.arange(n, dtype=np.float64)
    x0 = i / n
    x1 = (i+1) / n
    sqrt0 = np.sqrt(np.clip(x0, 0.0, 1.0))
    sqrt1 = np.sqrt(np.clip(x1, 0.0, 1.0))
    F0 = (2.0/np.pi) * np.arcsin(sqrt0)
    F1 = (2.0/np.pi) * np.arcsin(sqrt1)
    a = F1 - F0
    a_sum = a.sum()
    a /= a_sum   # ensure sum=1

    best_a = a.copy()
    max_conv = _conv_max(best_a)
    best_score = 2.0 * n * max_conv
    print(f"Initial score (2·n·max_conv) = {best_score:.8f}")

    # Adam optimization
    # convert to logits
    eps_small = 1e-16
    z = np.log(best_a + eps_small)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    base_alpha = 1.0   # slightly larger
    eps_factor = 0.2
    eps_factor_min = 1e-6
    eps_factor_decay = 0.9995

    it = 0
    max_iters = 200000

    while time.time() < deadline and it < max_iters:
        it += 1
        a_cur = _softmax(z)

        max_conv_cur, grad_a = _conv_max_and_grad(a_cur, eps_factor)
        cur_score = 2.0 * n * max_conv_cur

        if cur_score < best_score:
            best_score = cur_score
            best_a = a_cur.copy()
            if it % 1000 == 0:
                print(f"Iter {it}: improved score = {best_score:.8f}")

        # gradient on logits
        mu = np.dot(grad_a, a_cur)
        grad_z = a_cur * (grad_a - mu)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        alpha_t = base_alpha / (np.sqrt(it) + 1.0)
        step = - alpha_t * m_hat / (np.sqrt(v_hat) + eps_adam)
        z = z + step
        # clip for numerical safety
        z = np.clip(z, -80.0, 80.0)

#!/usr/bin/env python
"""
Step‑function construction for the max‑convolution metric.

The continuous arcsine law on [0,1] has a constant self‑convolution,
hence its exact discretisation gives a near‑optimal discrete step
function.  The discretisation error decays roughly like 1/√n, so a
larger number of bins quickly beats the target bound 1.503.

Algorithm
---------
1. Choose a sufficiently large length `n` (default 40 000, or the
   length of an externally supplied `height_sequence_1` if it exists
   and is larger).
2. Build the exact arcsine bin probabilities
        a_j = F((j+1)/n) – F(j/n),   F(x)=2/π·arcsin(√x)
   and normalise them to sum to 1.
3. Compute the initial score `2·n·max(a*a)`.  If it is already below the
   target we return the distribution immediately.
4. Otherwise run a short Adam optimisation on the simplex
   (parameterise the vector by soft‑max logits).  The gradient of the
   smooth‑max of the self‑convolution is obtained analytically via FFT.
   The optimiser uses a decaying smoothing factor, a decaying step‑size
   and occasional jitter to escape plateaus.
5. Return the best sequence found.  A final exact evaluation with the
   provided `evaluate_sequence` function is logged only for information.

The whole procedure respects the 1000 s wall‑clock budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Return logits that map to the probability vector a via soft‑max."""
    return np.log(a + eps)


def _fft_len_for_conv(n: int) -> int:
    """
    Smallest power‑of‑two ≥ 2·n‑1,
    sufficient for a linear self‑convolution of length (2·n‑1).
    """
    return 1 << ((2 * n - 1).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a (no gradient)."""
    n = a.shape[0]
    L = _fft_len_for_conv(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Compute the smoothed maximum of a*a and the corresponding gradient.

    Parameters
    ----------
    a : np.ndarray
        Probability vector (non‑negative, sum = 1).
    eps_factor : float
        Multiplicative factor for the smoothing constant:
        eps = eps_factor * max(conv).

    Returns
    -------
    max_conv : float
        Maximum entry of the self‑convolution.
    grad_a : np.ndarray
        Gradient of 4·n·smooth_max w.r.t. a (already multiplied by 4·n).
    """
    n = a.shape[0]
    L = _fft_len_for_conv(n)

    a_fft = np.fft.rfft(a, L)

    # linear self‑convolution (length 2·n‑1)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # smoothed soft‑max weights (log‑sum‑exp trick)
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv               # improve stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum

    # cross‑correlation a ⋆ w (linear, length 3·n‑2) – we only need first n entries
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[: n]

    # gradient (4·n multiplied)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that minimises the evaluation metric.

    The construction is based on the exact discretisation of the arcsine law.
    If the naïve discretisation already beats the target, we return it.
    Otherwise we polish it with a short Adam optimisation on the simplex.
    """
    np.random.seed(seed)
    start = time.time()
    # safety margin for final logging / return
    deadline = start + budget_s - 2.0

    # ------------------------------------------------------------------
    # 0. Choose the length of the vector.
    # ------------------------------------------------------------------
    try:
        n_external = len(height_sequence_1)  # type: ignore[name-defined]
        print(f"External sequence detected with length {n_external}.")
    except Exception:
        n_external = 0
        print("No external sequence detected.")

    # We want a length large enough to diminish discretisation error.
    # 40 000 is a good compromise between speed of the final evaluation
    # (np.convolve) and the 1/√n error decay.
    n = max(n_external, 40000)
    print(f"Using length n = {n}")

    # ------------------------------------------------------------------
    # 1. Exact discretisation of the arcsine distribution (bin probabilities).
    # ------------------------------------------------------------------
    i = np.arange(n, dtype=np.float64)
    x0 = i / n
    x1 = (i + 1) / n

    sqrt0 = np.sqrt(np.clip(x0, 0.0, 1.0))
    sqrt1 = np.sqrt(np.clip(x1, 0.0, 1.0))
    F0 = (2.0 / np.pi) * np.arcsin(sqrt0)
    F1 = (2.0 / np.pi) * np.arcsin(sqrt1)

    a = F1 - F0                     # bin probability masses, sum ≈ 1
    a_sum = a.sum()
    if not np.isfinite(a_sum) or a_sum <= 0.0:
        raise RuntimeError("Failed to generate arcsine bins.")
    a /= a_sum                      # enforce exact normalisation (probability vector)

    # ------------------------------------------------------------------
    # 2. Initial (cheap) evaluation.
    # ------------------------------------------------------------------
    max_conv = _conv_max(a)
    best_score = 2.0 * n * max_conv
    best_a = a.copy()
    print(f"Initial score (2·n·max_conv) = {best_score:.8f}")

    # ------------------------------------------------------------------
    # 3. If we are already below the target, skip optimisation.
    # ------------------------------------------------------------------
    TARGET = 1.503
    if best_score <= TARGET:
        print(f"Initial construction already beats target {TARGET:.3f}.")
        final_seq = best_a.tolist()
        try:
            exact = evaluate_sequence(final_seq)
            print(f"Exact evaluation: {exact:.8f}")
        except Exception as exc:
            print(f"Exact evaluation failed: {exc}")
        return final_seq

    # ------------------------------------------------------------------
    # 4. Adam refinement on the simplex.
    #    We run while time permits, decaying the smoothing factor to
    #    obtain essentially the true sub‑gradient of the max.
    # ------------------------------------------------------------------
    # Initialise logits (soft‑max parameterisation)
    z = _logits_from_distribution(best_a)
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_alpha = 1.0          # larger initial step‑size than the reference code
    eps_factor = 0.2          # start with a relatively strong smoothing
    eps_factor_min = 1e-6     # final smoothing – essentially exact max
    eps_factor_decay = 0.9995

    it = 0
    max_iters = 300000  # generous upper bound; loop will stop earlier by deadline

    while time.time() < deadline and it < max_iters:
        it += 1

        a_cur = _softmax(z)

        max_conv_cur, grad_a = _conv_max_and_grad(a_cur, eps_factor)
        cur_score = 2.0 * n * max_conv_cur

        # Keep the best solution seen so far
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_cur.copy()
            if it % 1000 == 0:
                print(f"Iter {it:6d}: improved score = {best_score:.8f}")

        # Gradient w.r.t. the logits (mirrored‑descent form)
        mu = np.dot(grad_a, a_cur)
        grad_z = a_cur * (grad_a - mu)

        # Adam bias‑corrected update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # Inverse‑sqrt step‑size schedule (similar to the reference)
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step = - alpha_t * m_hat / (np.sqrt(v_hat) + eps_adam)

        z = z + step
        # keep logits bounded so that exp(z) stays in a safe range
        z = np.clip(z, -80.0, 80.0)

        # Decay the smoothing constant
        eps_factor = max(eps_factor_min, eps_factor * eps_factor_decay)

        # Tiny random jitter every few thousand steps – helps to escape flat plateaus
        if it % 10000 == 0 and np.random.rand() < 0.1:
            z += 0.001 * np.random.randn(n)

        # Early stop if we have already reached the required bound
        if best_score <= TARGET:
            print(f"Target achieved at iteration {it}, score={best_score:.8f}")
            break

    # ------------------------------------------------------------------
    # 5. Return the best sequence.  One final exact evaluation is logged.
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        exact_score = evaluate_sequence(final_seq)
        print(f"Exact evaluation (after optimisation): {exact_score:.8f}")
    except Exception as exc:
        print(f"Exact evaluation failed: {exc}")

    return final_seq
