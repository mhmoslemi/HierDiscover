def _conv_max_and_index(a):

        run_opt(a0, time_limit=remaining * 0.8)  # allocate 80% time

        # generate random start near current best
        noise = 0.01 * np.random.randn(n)
        a_start = best_a + noise
        a_start = np.clip(a_start, 0, None)

        remaining = deadline - time.time()
        if remaining < 5:

        run_opt(a_start, time_limit=remaining * 0.5)  # give some time

    final_seq = best_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[Info] Final evaluated score = {final_score:.6f}")
    except Exception as e:
        print("[Warning] evaluate_sequence error:", e)
    return final_seq

"""
Propose a step‑function (non‑negative sequence) that minimises the evaluation metric.
The algorithm:
1. Initialise from the arcsine CDF (the known near‑optimal continuous solution).
2. Run a long Adam‑style mirror‑descent on the simplex using a smoothed max‑convolution
   (soft‑max of the autocorrelation) with a progressively shrinking smoothing constant.
3. When time permits, perform several short restarts from the current best state
   with random perturbations to escape possible plateaus.
4. Finally, a brief exact‐gradient fine‑tuning phase based on the true max‑convolution.
The whole search respects the 1000 s budget and always returns the best sequence found.
"""

import time
import math
import numpy as np

def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector that sums to 1."""
    z = z - np.max(z)
    e = np.exp(z)
    return e / e.sum()

def _fft_len_for_corr(n: int) -> int:
    """Smallest power of two ≥ 3·n‑2 (sufficient for linear convolution)."""
    return 1 << ((3 * n - 2).bit_length())

def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Smoothed max of a*a and its gradient w.r.t. a.
    Returns (max_conv, grad_a) where grad_a is the gradient of 2·n·max_conv.
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    a_fft = np.fft.rfft(a, L)

    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or np.isnan(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    grad_a = 4.0 * n * corr
    return max_conv, grad_a

def _conv_max_and_index(a: np.ndarray):
    """Linear self‑convolution via FFT, returning (max value, index of first max)."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_val = float(conv.max())
    max_idx = int(np.argmax(conv))
    return max_val, max_idx

def _grad_max_exact(a: np.ndarray, k: int):
    """Exact sub‑gradient of max_{k} a*a at index k (for a with sum=1)."""
    n = a.shape[0]
    grad = np.zeros_like(a)
    i_low = max(0, k - (n - 1))
    i_high = min(n - 1, k)
    if i_low <= i_high:
        idx = np.arange(i_low, i_high + 1, dtype=int)
        grad[idx] = 4.0 * n * a[k - idx]
    return grad

class _Optimizer:
    """Container for the iterative optimisation; avoids closures."""
    def __init__(self, n: int, deadline: float):
        self.n = n
        self.deadline = deadline
        self.best_a = None          # ndarray, current best sequence (sum=1)
        self.best_score = float('inf')
        self.start = time.time()

    def _update_best(self, a: np.ndarray, score: float):
        if score < self.best_score:
            self.best_score = score
            self.best_a = a.copy()
            elapsed = time.time() - self.start
            print(f"[Info] New best {score:.6f} (elapsed {elapsed:.1f}s)")

    def run_from(self, a_init: np.ndarray, time_fraction: float = 0.5):
        """
        Perform a long Adam optimisation from `a_init`.
        The optimisation uses a soft‑max (smooth) approximation of the max‑convolution.
        """
        end_time = min(self.deadline, time.time() + (self.deadline - time.time()) * time_fraction)

        n = self.n
        # logits parametrisation (mirror‑descent on the simplex)
        z = np.log(a_init + 1e-16)
        m = np.zeros_like(z)
        v = np.zeros_like(z)
        beta1, beta2 = 0.9, 0.999
        eps_adam = 1e-8

        base_alpha = 1.2          # initial step size for logits
        eps_factor = 0.5          # start smoothing
        eps_factor_min = 5e-4
        eps_decay = 0.9997        # exponential decay of smoothing

        it = 0
        while time.time() < end_time:
            it += 1
            a = _softmax(z)

            # smoothed max‑convolution and gradient w.r.t. a
            max_conv, grad_a = _conv_max_and_grad(a, eps_factor)

            score = 2.0 * n * max_conv
            self._update_best(a, score)

            # mirror‑descent gradient on logits
            mu = np.dot(grad_a, a)
            grad_z = a * (grad_a - mu)

            # Adam moment updates (bias‑corrected)
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            # inverse‑sqrt step‑size decay
            alpha_t = base_alpha / (math.sqrt(it) + 1.0)
            step = - alpha_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            z = z + step
            z = np.clip(z, -80.0, 80.0)

            # shrink smoothing constant
            eps_factor = max(eps_factor_min, eps_factor * eps_decay)

            # occasional random jitter to shake out of plateaus
            if it % 5000 == 0 and np.random.rand() < 0.2:
                z += 0.02 * np.random.randn(n)

        # ------------------------------------------------------------------
        # Short exact‑gradient fine‑tuning phase (no smoothing)
        # ------------------------------------------------------------------
        for _ in range(2000):
            if time.time() >= self.deadline:
                break
            a = _softmax(z)
            max_conv, max_idx = _conv_max_and_index(a)
            score = 2.0 * n * max_conv
            self._update_best(a, score)

            grad_a = _grad_max_exact(a, max_idx)
            mu = np.dot(grad_a, a)
            grad_z = a * (grad_a - mu)

            # simple diminishing‑step mirror descent (no Adam)
            lr = 0.4 / (math.sqrt(it + 1) + 1.0)
            it += 1
            z = z - lr * grad_z
            z = np.clip(z, -80.0, 80.0)

    def hill_climb(self, steps: int = 2000):
        """
        Tiny stochastic local search around the current best. Each step moves a
        small amount of mass between two random positions and accepts the move
        if it improves the exact metric.
        """
        a_best = self.best_a
        if a_best is None:
            return
        n = self.n
        for s in range(steps):
            if time.time() >= self.deadline:
                break
            # propose a slight random perturbation
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            delta = 1e-4 * (np.random.rand() - 0.5)  # tiny signed transfer
            # enforce non‑negativity
            if a_best[i] + delta < 0 or a_best[j] - delta < 0:
                continue
            a_new = a_best.copy()
            a_new[i] += delta
            a_new[j] -= delta
            # re‑normalise to avoid drift (the sum is unchanged by construction)
            max_conv, _ = _conv_max_and_index(a_new)
            score = 2.0 * n * max_conv
            if score < self.best_score:
                self.best_score = score
                self.best_a = a_new
                a_best = a_new
                elapsed = time.time() - self.start
                print(f"[HillClimb] Improved to {score:.6f} (step {s}, elapsed {elapsed:.1f}s)")

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list representing a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine follows the strategy explained in the
    module docstring.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0   # safety margin

    # --------------------------------------------------------------
    # 0. Choose the length of the sequence
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
        print(f"[Info] Using external length: n = {n}")
    except Exception:
        n = 20000
        print(f"[Info] No external length – defaulting to n = {n}")

    # --------------------------------------------------------------
    # 1. Initialise from the exact arcsine CDF
    # --------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)
    x0 = idx / n
    x1 = (idx + 1) / n

    def arcsine_cdf(x):
        return (2.0 / np.pi) * np.arcsin(np.sqrt(np.clip(x, 0.0, 1.0)))

    a0 = arcsine_cdf(x1) - arcsine_cdf(x0)
    a0_sum = a0.sum()
    if a0_sum == 0.0 or np.isnan(a0_sum):
        raise RuntimeError("Failed to initialise a valid arcsine distribution.")
    a0 = a0 / a0_sum                # now sum(a0) == 1

    # --------------------------------------------------------------
    # 2. Optimisation driver
    # --------------------------------------------------------------
    opt = _Optimizer(n=n, deadline=deadline)
    # initialise best with the plain arcsine sequence
    max_conv0, _ = _conv_max_and_index(a0)
    init_score = 2.0 * n * max_conv0
    opt._update_best(a0, init_score)

    # Main long run (about 80 % of the remaining budget)
    remaining = deadline - time.time()
    if remaining > 20.0:
        opt.run_from(a0, time_fraction=0.8)

    # --------------------------------------------------------------
    # 3. Random‑restart fine‑tuning (if time permits)
    # --------------------------------------------------------------
    while time.time() < deadline - 5.0:
        # perturb the current best a little and re‑optimise briefly
        noise = 0.02 * np.random.randn(n)
        a_start = opt.best_a + noise
        a_start = np.clip(a_start, 0.0, None)
        if a_start.sum() == 0.0:
            continue
        a_start = a_start / a_start.sum()
        # give the restart a modest share of the remaining budget
        opt.run_from(a_start, time_fraction=0.2)

    # --------------------------------------------------------------
    # 4. Optional tiny hill‑climbing phase (very cheap moves)
    # --------------------------------------------------------------
    opt.hill_climb(steps=3000)

    # --------------------------------------------------------------
    # 5. Return the best sequence found
    # --------------------------------------------------------------
    final_seq = opt.best_a.tolist() if opt.best_a is not None else a0.tolist()

    # Optional final exact evaluation (only for logging)
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[Info] Final exact score = {final_score:.6f}")
    except Exception as exc:
        print(f"[Warning] evaluation error (ignored): {exc}")

    return final_seq

"""
Propose a step‑function (non‑negative sequence) that minimises the
``evaluate_sequence`` metric.

The algorithm proceeds as follows:

1. **Initialisation** – Build the exact arcsine CDF discretisation,
   the known near‑optimal continuous solution.

2. **Long Adam optimisation** – Optimise a mirror‑descent (soft‑max)
   parametrisation of the simplex using a smoothed max‑convolution.
   The smoothing constant starts relatively large (≈0.5) and decays
   exponentially, allowing the optimiser to explore broadly at first
   and to focus on the true max later.  Adam with bias‑correction and
   an inverse‑sqrt learning‑rate schedule is employed.

3. **Exact‑gradient fine‑tuning** – After the smoothing reaches its
   minimum, a short phase uses the exact sub‑gradient of the
   max‑convolution (derived from the index of the maximum) for further
   improvement.

4. **Random‑restart polishing** – While time permits, the current best
   sequence is perturbed with Gaussian noise and the whole optimisation
   (steps 2 + 3) is repeated on the perturbed start, giving the algorithm
   a chance to escape local plateaus.

5. **Hill‑climbing** – Finally, cheap random mass‑transfer moves are
   tried between two bins; moves that improve the exact metric are kept.

All work respects the 1000 s budget (a small safety margin is kept) and
the returned list is guaranteed to be non‑negative and to sum to a
positive value, so ``evaluate_sequence`` will never truncate it to
``np.inf``.  The implementation uses only top‑level helper functions
(no closures) and relies on FFT‑based convolution for efficiency.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Top‑level helper functions (no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector that sums to 1."""
    z = z - np.max(z)
    e = np.exp(z)
    return e / e.sum()


def _fft_len_for_corr(n: int) -> int:
    """Smallest power‑of‑two ≥ 3·n‑2 (sufficient for linear convolution)."""
    return 1 << ((3 * n - 2).bit_length())


def _arcsine_cdf(x):
    """Arcsine CDF  F(x) = (2/π)·arcsin(√x)  for x∈[0,1]."""
    return (2.0 / np.pi) * np.arcsin(np.sqrt(np.clip(x, 0.0, 1.0)))


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Smoothed max of a*a and its gradient w.r.t. a.

    Returns
    -------
    max_conv : float
        Maximum entry of the linear self‑convolution.
    grad_a : np.ndarray
        Gradient of 2·n·max_conv w.r.t. a (already multiplied by 4·n).
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    a_fft = np.fft.rfft(a, L)

    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or np.isnan(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    grad_a = 4.0 * n * corr
    return max_conv, grad_a


def _conv_max_and_index(a: np.ndarray):
    """Linear self‑convolution via FFT, returning (max value, index of first max)."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_val = float(conv.max())
    max_idx = int(np.argmax(conv))
    return max_val, max_idx


def _grad_max_exact(a: np.ndarray, k: int):
    """
    Exact sub‑gradient of the max‑convolution at index k.
    Returns a vector g where g_i = 4·n·a_{k‑i} for indices that lie inside the support.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    i_low = max(0, k - (n - 1))
    i_high = min(n - 1, k)
    if i_low <= i_high:
        idx = np.arange(i_low, i_high + 1, dtype=int)
        grad[idx] = 4.0 * n * a[k - idx]
    return grad


# ----------------------------------------------------------------------
# Optimiser class (avoids closures)
# ----------------------------------------------------------------------


class _Optimizer:
    """Encapsulates the optimisation state and procedures."""

    def __init__(self, n: int, deadline: float):
        self.n = n
        self.deadline = deadline
        self.best_a = None          # ndarray, best sequence (sum = 1)
        self.best_score = float("inf")
        self.start = time.time()

    # ------------------------------------------------------------------
    # bookkeeping
    # ------------------------------------------------------------------
    def _update_best(self, a: np.ndarray, score: float):
        """Record a new best sequence if the score improved."""
        if score < self.best_score:
            self.best_score = score
            self.best_a = a.copy()
            elapsed = time.time() - self.start
            print(f"[Info] New best {score:.6f} (elapsed {elapsed:.1f}s)")

    # ------------------------------------------------------------------
    # main optimisation loop (Adam on the soft‑max of the simplex)
    # ------------------------------------------------------------------
    def run_from(self, a_init: np.ndarray, time_fraction: float = 0.5):
        """
        Optimise starting from `a_init`.  The method first runs a long
        Adam phase on a smoothed max‑convolution, then a short exact‑gradient
        fine‑tuning phase.
        `time_fraction` specifies what fraction of the *remaining* budget
        is devoted to this call.
        """
        now = time.time()
        remaining = self.deadline - now
        if remaining <= 0:
            return
        end_time = now + remaining * time_fraction
        n = self.n

        # logits parametrisation (mirror‑descent on the simplex)
        z = np.log(a_init + 1e-16)
        m = np.zeros_like(z)
        v = np.zeros_like(z)
        beta1, beta2 = 0.9, 0.999
        eps_adam = 1e-8

        base_alpha = 1.2          # initial step size for logits
        eps_factor = 0.5          # start smoothing constant
        eps_factor_min = 5e-4
        eps_decay = 0.9997        # exponential decay of smoothing

        it = 0
        # ----------------------- Adam phase -------------------------
        while time.time() < end_time:
            it += 1
            a = _softmax(z)

            max_conv, grad_a = _conv_max_and_grad(a, eps_factor)
            score = 2.0 * n * max_conv
            self._update_best(a, score)

            # mirror‑descent gradient on logits
            mu = np.dot(grad_a, a)
            grad_z = a * (grad_a - mu)

            # Adam moment updates (bias‑corrected)
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            # inverse‑sqrt decay of the step size
            alpha_t = base_alpha / (math.sqrt(it) + 1.0)
            step = -alpha_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            z = z + step
            z = np.clip(z, -80.0, 80.0)

            # decay smoothing
            eps_factor = max(eps_factor_min, eps_factor * eps_decay)

            # occasional random jitter to shake out of plateaus
            if it % 5000 == 0 and np.random.rand() < 0.2:
                z += 0.02 * np.random.randn(n)

        # ------------------- exact‑gradient fine‑tuning -------------
        for _ in range(2000):
            if time.time() >= self.deadline:
                break
            a = _softmax(z)
            max_conv, max_idx = _conv_max_and_index(a)
            score = 2.0 * n * max_conv
            self._update_best(a, score)

            grad_a = _grad_max_exact(a, max_idx)
            mu = np.dot(grad_a, a)
            grad_z = a * (grad_a - mu)

            # simple diminishing‑step mirror descent (no Adam)
            lr = 0.4 / (math.sqrt(it + 1) + 1.0)
            it += 1
            z = z - lr * grad_z
            z = np.clip(z, -80.0, 80.0)

    # ------------------------------------------------------------------
    # tiny stochastic local search
    # ------------------------------------------------------------------
    def hill_climb(self, steps: int = 2000):
        """Randomly transfer a tiny amount of mass between two bins."""
        if self.best_a is None:
            return
        a_best = self.best_a
        n = self.n
        for s in range(steps):
            if time.time() >= self.deadline:
                break
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            # transfer a tiny signed amount δ
            delta = 1e-4 * (np.random.rand() - 0.5)
            if a_best[i] + delta < 0 or a_best[j] - delta < 0:
                continue
            a_new = a_best.copy()
            a_new[i] += delta
            a_new[j] -= delta
            max_conv, _ = _conv_max_and_index(a_new)
            score = 2.0 * n * max_conv
            if score < self.best_score:
                self.best_score = score
                self.best_a = a_new
                a_best = a_new
                elapsed = time.time() - self.start
                print(f"[HillClimb] Improved to {score:.6f} (step {s}, elapsed {elapsed:.1f}s)")
