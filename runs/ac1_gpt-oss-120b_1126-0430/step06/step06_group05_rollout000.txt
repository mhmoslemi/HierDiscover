"""
Step‑function optimiser (parameter‑grid + hard‑max Adam + SA).

The algorithm works in three phases within the supplied time budget:

1. **Parameter sweep** –  explores a few low‑dimensional families of
   symmetric distributions (Beta‑shaped, inverse‑power, random sparse)
   on several sequence lengths (≈8k–15k).  Computing the objective for a
   candidate only needs a single FFT, so thousands of candidates can be
   examined quickly.  The best candidate of this stage seeds the next
   phase.

2. **Hard‑max Adam** –  starting from the best candidate, an Adam optimiser
   works directly on the exact sub‑gradient of the maximal convolution
   entry (no soft‑max smoothing).  The update is performed in log‑it
   (exponentiated‑gradient) space to stay on the probability simplex.
   Learning‑rate decays exponentially from a relatively large start,
   jitter and occasional full restarts help escape plateaus.

3. **Targeted simulated‑annealing** –  if time remains, a lightweight SA
   refines the solution by small mass‑transfer moves guided by the
   current maximal convolution index.

All intermediate progress and the final evaluation (using the provided
``evaluate_sequence``) are printed.  The returned list satisfies the
non‑negative and normalisation requirements of the evaluator.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# FFT based convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return conv_full[: 2 * n - 1]


def _grad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """
    Exact sub‑gradient of the k‑th convolution entry w.r.t. ``a``.
    Returns 2·a_{k−i} for indices where it is defined, otherwise 0.
    """
    n = a.shape[0]
    idx = np.arange(n)
    j = k - idx
    mask = (j >= 0) & (j < n)
    g = np.zeros_like(a)
    if mask.any():
        g[mask] = 2.0 * a[j[mask]]
    return g


# ----------------------------------------------------------------------
# Objective helper
# ----------------------------------------------------------------------


def _objective(a: np.ndarray) -> float:
    """
    Compute 2·n·max(conv(a,a)) / (sum(a)^2).  The sequence is assumed
    non‑negative and already clamped to the allowed numerical range.
    """
    n = a.shape[0]
    conv = _conv_self_fft(a)
    max_b = conv.max()
    sum_a = a.sum()
    # sum_a is guaranteed > 0 (caller normalises to 1)
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Hard‑max Adam optimiser (same core as the reference code)
# ----------------------------------------------------------------------


def _run_opt_hardmax(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
) -> Tuple[List[float], float]:
    """
    Adam optimiser that uses the exact max‑convolution gradient.
    Returns the best sequence (as a list) and its objective value.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # initialise a on the simplex
    if init_a is not None and init_a.size == n:
        a = np.asarray(init_a, dtype=float)
    else:
        a = rng.random(n)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n, dtype=float) / n
    else:
        a = a / s

    # logits for exponentiated (mirror) gradient updates
    x = np.log(a + 1e-16)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # learning‑rate schedule (exponential decay)
    lr_start = 0.8   # a bit higher than the reference
    lr_end = 0.001

    best_a = a.copy()
    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())
    it = 0
    while time.time() - start < time_limit:
        it += 1

        a = np.exp(x - np.max(x))   # numerically stable softmax
        a = a / a.sum()

        conv = _conv_self_fft(a)
        M = conv.max()
        obj = float(2.0 * n * M)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[Adam n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        # index of the maximal convolution entry
        k_max = int(np.argmax(conv))

        # exact sub‑gradient of that entry
        g_a = _grad_max_conv(a, k_max)

        # natural gradient in log‑it space (mirrored / exponentiated)
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)

        # time‑based learning‑rate (exponential)
        elapsed = time.time() - start
        t = min(elapsed / time_limit, 1.0)
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # Adam step in log‑it space
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter
        if it % 4000 == 0:
            jitter = rng.normal(scale=0.008, size=n)
            x += jitter

        # occasional full restart from the current best
        if it % 70000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.004, size=n)
            m.fill(0.0)
            v.fill(0.0)

    # compress negligible entries (they only increase n without helping)
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = float(2.0 * best_a.size * _conv_self_fft(best_a).max())
        print(f"[Adam compress] reduced length → {best_a.size}, O={best_obj:.6f}")

    return best_a.tolist(), best_obj


# ----------------------------------------------------------------------
# Simple simulated‑annealing refinement (unchanged from the reference)
# ----------------------------------------------------------------------


def _run_sa(
    init_seq: List[float],
    time_limit: float,
    seed: int = 0,
) -> Tuple[List[float], float]:
    """Refine a sequence by small mass‑transfer moves with a Metropolis schedule."""
    start = time.time()
    rng = np.random.default_rng(seed)
    a = np.asarray(init_seq, dtype=float)
    if a.sum() < 1e-12:
        a = np.ones_like(a) / a.size
    else:
        a = a / a.sum()
    n = a.shape[0]

    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())
    best_seq = a.copy()

    # temperature schedule
    T = best_obj * 0.1 + 1e-6
    T_min = best_obj * 1e-6 + 1e-9
    decay = 0.995
    it = 0

    while time.time() - start < time_limit:
        it += 1
        k_max = int(np.argmax(conv))

        # contribution of each index to the maximal entry
        i_vals = np.arange(n)
        j_vals = k_max - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i_vals[mask]] * a[j_vals[mask]]
        total_contrib = contrib.sum()

        if total_contrib > 0:
            probs = contrib / total_contrib
            i = rng.choice(n, p=probs)
        else:
            i = rng.integers(n)

        # pick a distinct destination uniformly
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # transfer a modest fraction of mass from i to j
        delta = a[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.clip(a_new, 0.0, None)
        a_new /= a_new.sum()

        conv_new = _conv_self_fft(a_new)
        obj_new = float(2.0 * n * conv_new.max())

        # Metropolis acceptance
        if obj_new < best_obj - 1e-12:
            a, conv, best_obj, best_seq = a_new, conv_new, obj_new, a_new.copy()
            print(f"[SA it={it}] new best O={best_obj:.6f}")
        else:
            prob = np.exp(-(obj_new - best_obj) / T) if T > 0 else 0.0
            if rng.random() < prob:
                a, conv = a_new, conv_new

        # temperature decay
        if it % 2000 == 0:
            T = max(T * decay, T_min)

        # occasional global jitter
        if it % 9000 == 0:
            noise = rng.normal(scale=0.008, size=n)
            a = np.clip(a + noise, 0.0, None)
            a /= a.sum()
            conv = _conv_self_fft(a)

    # final compression of tiny entries
    thresh = 1e-12
    mask = best_seq > thresh
    if mask.sum() < best_seq.size:
        comp = best_seq[mask]
        comp /= comp.sum()
        comp_obj = float(2.0 * comp.size * _conv_self_fft(comp).max())
        if comp_obj < best_obj:
            best_seq, best_obj = comp, comp_obj
            print(f"[SA compress] reduced length → {comp.size}, O={best_obj:.6f}")

    return best_seq.tolist(), best_obj


# ----------------------------------------------------------------------
# Helper constructors for the parameter sweep
# ----------------------------------------------------------------------


def _beta_seq(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric Beta‑shaped distribution on {0,…,n‑1} (α,β > 0)."""
    i = np.arange(n, dtype=float)
    a = (i + 1) ** (alpha - 1) * (n - i) ** (beta - 1)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.clip(a, 0.0, None)
    total = a.sum()
    if total <= 0:
        a = np.ones(n, dtype=float) / n
    else:
        a = a / total
    return a


def _inv_power_seq(n: int, p: float) -> np.ndarray:
    """U‑shaped inverse‑power sequence a_i ∝ (i+1)^(-p) + (n-i)^(-p)."""
    i = np.arange(n, dtype=float)
    a = (i + 1) ** (-p) + (n - i) ** (-p)
    a = np.clip(a, 0.0, None)
    total = a.sum()
    if total <= 0:
        a = np.ones(n, dtype=float) / n
    else:
        a = a / total
    return a


def _random_sparse_seq(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    """Random support of size m (≤ n) with random positive weights."""
    idx = rng.choice(n, size=m, replace=False)
    vals = rng.random(m) + 1e-6  # avoid exact zero
    a = np.zeros(n, dtype=float)
    a[idx] = vals
    a /= a.sum()
    return a

# Main entry point

def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **_ignore,
) -> List[float]:
    """
    Find a non‑negative step‑function minimising the evaluation metric.

    The routine consists of:
        • a fast grid search over a few parametric families,
        • a hard‑max Adam refinement seeded with the best grid candidate,
        • an optional short simulated‑annealing polish.

    The best sequence found before the time budget expires is returned.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.2  # small safety margin for the final print

    # ------------------------------------------------------------------
    # 1)   Parameter sweep (≈20 % of the budget)
    # ------------------------------------------------------------------
    sweep_budget = budget_s * 0.20
    sweep_deadline = start + sweep_budget

    best_seq = None
    best_obj = float("inf")

    # initialise with any existing construction
    if "height_sequence_1" in globals():
        seq0 = np.asarray(height_sequence_1, dtype=float)
        seq0 = np.clip(seq0, 0.0, None)
        if seq0.sum() < 0.01:
            seq0 = np.ones_like(seq0)
        seq0 = seq0 / seq0.sum()
        obj0 = _objective(seq0)
        best_seq, best_obj = seq0.copy(), obj0
        print(f"[init] starting from height_sequence_1 → O={best_obj:.6f}")

    # also try a plain uniform baseline (useful when no previous sequence)
    for n0 in (8000, 10000, 12000, 15000):
        uni = np.ones(n0, dtype=float) / n0
        obj_uni = _objective(uni)
        if obj_uni < best_obj:
            best_seq, best_obj = uni.copy(), obj_uni
            print(f"[init] uniform n={n0} → O={best_obj:.6f}")

    # candidate lengths (the reference used 12000; we try a few)
    lengths = [8000, 10000, 12000, 15000]

    # beta‑shape parameters (symmetric, so beta=alpha)
    alphas = [0.10, 0.15, 0.20, 0.30, 0.40, 0.50, 0.65, 0.80, 1.00]
    # inverse‑power exponents
    powers = [0.5, 0.75, 1.0, 1.25, 1.5, 2.0]

    # start the sweep
    for n in lengths:
        if time.time() >= sweep_deadline:
            break

        # Beta‑family (α=β)
        for a in alphas:
            seq = _beta_seq(n, a, a)
            obj = _objective(seq)
            if obj < best_obj:
                best_seq, best_obj = seq, obj
                print(f"[grid] n={n:5d} α={a:.2f} → O={obj:.6f}")
            if time.time() >= sweep_deadline:
                break

        if time.time() >= sweep_deadline:
            break

        # Inverse‑power family
        for p in powers:
            seq = _inv_power_seq(n, p)
            obj = _objective(seq)
            if obj < best_obj:
                best_seq, best_obj = seq, obj
                print(f"[grid] n={n:5d} p={p:.2f} → O={obj:.6f}")
            if time.time() >= sweep_deadline:
                break

        if time.time() >= sweep_deadline:
            break

        # Random sparse candidates (20 per length)
        for _ in range(20):
            m = rng.integers(200, min(2000, n // 2))
            seq = _random_sparse_seq(n, m, rng)
            obj = _objective(seq)
            if obj < best_obj:
                best_seq, best_obj = seq, obj
                print(f"[rand] n={n:5d} m={m:4d} → O={obj:.6f}")
            if time.time() >= sweep_deadline:
                break

    # ------------------------------------------------------------------
    # 2)   Hard‑max Adam optimisation (majority of remaining time)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0 and best_seq is not None:
        n_adam = best_seq.shape[0]
        # Give Adam about 75 % of the leftover budget
        adam_budget = remaining * 0.75
        seq_adam, obj_adam = _run_opt_hardmax(
            n=n_adam,
            time_limit=adam_budget,
            init_a=best_seq,
            seed=seed + 12345,
        )
        best_seq = np.asarray(seq_adam, dtype=float)
        best_obj = obj_adam
        print(f"[global] after Adam best O={best_obj:.6f} (n={n_adam})")
        remaining = deadline - time.time()

    # ------------------------------------------------------------------
    # 3)   Optional SA refinement (use any time still left)
    # ------------------------------------------------------------------
    if remaining > 5.0 and best_seq is not None:
        seq_sa, obj_sa = _run_sa(best_seq.tolist(), time_limit=remaining, seed=seed + 99999)
        if obj_sa < best_obj:
            best_seq = np.asarray(seq_sa, dtype=float)
            best_obj = obj_sa
            print(f"[final] after SA new best O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check – report evaluator's score
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_seq.tolist()
