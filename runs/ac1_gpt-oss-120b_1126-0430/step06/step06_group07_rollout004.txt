#!/usr/bin/env python
"""
Propose a non‑negative step function (list of heights) that minimises
``evaluate_sequence``.

The search consists of three stages:

1. **Broad seeding** – a few analytic families (uniform, exponential,
   symmetric exponential, Dirichlet) are evaluated quickly via a cheap
   surrogate ``2·n·max(conv)``.  The best seed is kept.

2. **Exact sub‑gradient mirror descent** – we optimise the current
   best vector on the probability simplex using the *correct* sub‑gradient
   of the maximal autocorrelation entry (the gradient is
   ``g_i = 2·a_{k‑i}`` for the active index ``k``).  An Adam‑style
   exponentiated‑gradient update is employed with a decaying learning
   rate; restarts are performed if progress stalls.

3. **Fine‑grained local search** – starting from the best iterate we
   apply simulated‑annealing moves:
      * random swaps of two positions,
      * tiny mass‑transfer moves between two entries.
   Both moves are accepted if they improve the cheap score or with a
   temperature‑controlled probability otherwise.

The routine respects a user‑supplied time budget (default 1000 s) and
always returns a feasible sequence.  All helper utilities are defined at
module level (no closures or lambdas).  The final sequence is evaluated
with the exact ``evaluate_sequence`` before being returned.
"""

# ----------------------------------------------------------------------
# Imports

import time
import math
import numpy as np

# Helper utilities (top level, no closures)

def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Next power of two greater or equal to x."""
    return 1 << ((x - 1).bit_length())


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len).real[: 2 * a.size - 1]
    return conv


def _conv_max_fft(a: np.ndarray, fft_len: int) -> tuple[float, int]:
    """Maximum entry of a*a and its index."""
    conv = _conv_full_fft(a, fft_len)
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


def _gradient_at_max(a: np.ndarray, kmax: int) -> np.ndarray:
    """
    Exact sub‑gradient of the maximal autocorrelation entry ``kmax``.
    Returns a vector ``g`` with ``g[i] = 2 * a[j]`` where ``j = kmax - i``,
    and zero wherever the index is out of range.
    """
    n = a.shape[0]
    lo = max(0, kmax - (n - 1))
    hi = min(kmax, n - 1)
    idx_i = np.arange(lo, hi + 1, dtype=np.int64)
    idx_j = kmax - idx_i
    g = np.zeros_like(a)
    # correct factor 2 for every contribution (including the diagonal)
    g[idx_i] = 2.0 * a[idx_j]
    return g


def _mass_transfer(a: np.ndarray, i: int, j: int, eps: float) -> tuple[np.ndarray, bool]:
    """
    Transfer a small fraction ``eps`` of the mass at index ``i`` to index ``j``.
    Returns the new vector and a flag indicating whether the operation was feasible.
    """
    if i == j:
        return a, False
    if a[i] <= eps:
        return a, False
    new_a = a.copy()
    delta = eps
    new_a[i] -= delta
    new_a[j] += delta
    return new_a, True

    try:

                    prob = math.exp(-delta / max(temperature, 1e-12))
                    if np.random.rand() < prob:
                        accept_mt = True

                if accept_mt:
                    a = a_mt
                    if cheap_mt < best_swap_val - 1e-12:
                        best_swap_val = cheap_mt
                        best_swap_a = a.copy()
                        print(f"[Transfer] NEW BEST cheap = {cheap_mt:.6f}")
                else:
                    # reject: keep original a
                    pass

        # anneal temperature
        temperature = max(temperature * cooling, 1e-7)

    # --------------------------------------------------------------
    # 3. Final exact evaluation
    # --------------------------------------------------------------
    final_seq = best_swap_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:  # defensive – should never happen
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    return final_seq
