def conv_full(a, b, fft_len):
    # compute linear convolution using FFT length
    fa = np.fft.fft(a, fft_len)
    fb = np.fft.fft(b, fft_len)
    return np.fft.ifft(fa * fb).real[:len(a)+len(b)-1]

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)

def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # all non‑positive → uniform simplex point
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full_fft(a: np.ndarray, b: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of a and b using an FFT of given length."""
    fa = np.fft.fft(a, fft_len)
    fb = np.fft.fft(b, fft_len)
    conv = np.fft.ifft(fa * fb).real[: a.size + b.size - 1]
    return conv


def _conv_max_fft(a: np.ndarray, fft_len: int) -> tuple[float, int]:
    """Maximum entry of a∗a and its index (full linear convolution)."""
    conv = _conv_full_fft(a, a, fft_len)
    idx = int(conv.argmax())
    return float(conv[idx]), idx

    while time.time() < deadline:
        it += 1

        # ----- convolution a∗a -------------------------------------------------
        conv_aa = _conv_full_fft(a, a, fft_len1)          # length 2n‑1
        max_conv = conv_aa.max()
        cheap = 2.0 * n * max_conv

        # record best so far (based on cheap score, i.e. exact objective)
        if cheap < best_iter_val - 1e-12:
            best_iter_val = cheap
            best_iter_a = a.copy()
            last_improve_iter = it
            print(f"[GD {it:07d}] NEW BEST cheap = {cheap:.6f}")

        # ----- soft‑max weights (smooth approximation of max) -----------------
        # stable softmax: subtract the max before exponentiation
        w_raw = np.exp(beta * (conv_aa - max_conv))
        w = w_raw / w_raw.sum()          # shape (2n‑1)

        # ----- gradient of the smoothed objective ----------------------------
        # g = 2 * (w ∗ a_rev)_{n‑1 … n‑1+n‑1}
        a_rev = a[::-1]
        conv_wr = _conv_full_fft(w, a_rev, fft_len2)   # length 3n‑2
        grad = 2.0 * conv_wr[n - 1: n - 1 + n]         # slice of length n

        # ----- exponentiated‑gradient update ---------------------------------
        eta = eta0 / math.sqrt(it)
        log_a = np.log(a + eps) - eta * grad
        a = np.exp(log_a)
        a_sum = a.sum()
        if not np.isfinite(a_sum) or a_sum <= 0.0:
            # fallback to the best iterate seen so far
            a = best_iter_a.copy()
            continue
        a /= a_sum  # bring back to the simplex

        # ----- temperature schedule -----------------------------------------
        if it % beta_update_every == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        # ----- occasional restart if we are stuck for a long time ----------
        if it - last_improve_iter > 300_000:
            perturb = np.random.rand(n) * 0.02
            a = _simplex_projection(best_iter_a + perturb)
            last_improve_iter = it
            print(f"[GD {it:07d}] Restart due to stagnation.")

    # --------------------------------------------------------------
    # 2️⃣  Permutation / swap local search (simulated‑annealing style)
    # --------------------------------------------------------------
    a = best_iter_a.copy()
    best_swap_a = a.copy()
    best_swap_val = best_iter_val
    temperature = 1.0
    cooling = 0.99997

    while time.time() < deadline:
        # propose a random swap
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        a[i], a[j] = a[j], a[i]

        conv_aa = _conv_full_fft(a, a, fft_len1)
        cheap = 2.0 * n * conv_aa.max()
        accept = False
        if cheap < best_swap_val - 1e-12:
            accept = True
        else:
            delta = cheap - best_swap_val
            prob = math.exp(-delta / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            if cheap < best_swap_val - 1e-12:
                best_swap_val = cheap
                best_swap_a = a.copy()
                print(f"[Swap] NEW BEST cheap = {cheap:.6f}")
        else:
            # revert the swap
            a[i], a[j] = a[j], a[i]

        temperature = max(temperature * cooling, 1e-6)

    # --------------------------------------------------------------
    # 3️⃣  Final exact evaluation
    # --------------------------------------------------------------
    final_seq = best_swap_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    return final_seq
