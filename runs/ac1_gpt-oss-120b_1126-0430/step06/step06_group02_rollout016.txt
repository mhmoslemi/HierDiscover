    x = (np.arange(n) + 0.5) / n

    return a

Non‑negative step‑function optimiser (enhanced random‑search + Adam).

The routine first performs a fast random search over several
parametric families (power‑law, exponential, Beta‑distribution,
mixtures) on a modest support (≈ 5 000 points) to find a good
starting shape.  This shape is then refined with mirror‑descent
(Adam) on the simplex using a smoothly‑decreasing ε‑smooth‑max
of the self‑convolution.  Occasional jitter, restarts and a
single up‑sampling stage (to ≈ 20 000 points) allow the
optimizer to escape plateaux.  The best sequence seen is finally
evaluated with ``evaluate_sequence`` and returned.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures or lambdas
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max, returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    s = e.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / z.size)
    return e / s


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it that reproduces `a` after a soft‑max."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """Smallest power‑of‑two ≥ 3·n‑2 (enough for all needed convolutions)."""
    return 1 << ((3 * n - 2).bit_length())


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    s = new_a.sum()
    if s == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= s
    return new_a


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{‑p}, normalised."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _exp_decay_distribution(n: int, lam: float) -> np.ndarray:
    """a_i ∝ e^{‑lam·i}+e^{‑lam·(n‑1‑i)}, symmetric exponential."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta PDF on [0,1] discretised, a_i ∝ x^{α‑1}(1‑x)^{β‑1}."""
    # Use mid‑point rule to avoid exact zeros at the ends.
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Returns (max_conv, grad_a) where

        grad_a = ∇_a [ 2·n·smooth_max(a*a) ]

    `eps_factor` multiplies the smoothing constant ε = eps_factor·max_conv.
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    # FFT of a (zero‑padded)
    a_fft = np.fft.rfft(a, L)

    # Linear self‑convolution a*a
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # Smoothed soft‑max weights
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv                # ≤ 0 for numerical stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # Correlation a ⋆ w
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # Gradient (already multiplied by 4·n)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a

def _uniform_distribution(n: int) -> np.ndarray:

    return a


def _random_mixture(n: int) -> np.ndarray:

    # Choose two families uniformly at random
    families = [_uniform_distribution,

                                             np.random.uniform(0.5, 12.0))]

    f2 = families[np.random.randint(len(families))](n)
    w = np.random.rand()
    a = w * f1 + (1.0 - w) * f2
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a

            if it % report_every == 0:
                print(f"[Iter {it:07d}] NEW BEST cheap = {cheap_score:.6f}")

        # Gradient w.r.t. logits (mirror‑descent on the simplex)
        mu = np.dot(grad_a, a)          # <grad, a>
        grad_z = a * (grad_a - mu)

        # Adam moments (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # Inverse‑square‑root decay for the step size
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step_dir = - m_hat / (np.sqrt(v_hat) + eps_adam)

        # Simple back‑tracking line search (max 6 halvings)
        accepted = False
        trial_alpha = alpha_t
        for _ in range(6):
            z_new = z + trial_alpha * step_dir
            z_new = np.clip(z_new, -50.0, 50.0)      # avoid overflow in exp()
            a_new = _softmax(z_new)
            new_score = 2.0 * n * _conv_max(a_new)
            if new_score <= cheap_score:
                z = z_new
                accepted = True
                break
            trial_alpha *= 0.5

        if not accepted:
            # Take a modest step even if no improvement was found
            z += alpha_t * step_dir
            z = np.clip(z, -50.0, 50.0)

        # Periodic jitter to shake the optimiser out of plateaux
        if (it % 3500 == 0) and (np.random.rand() < 0.12):
            z += 0.025 * np.random.randn(n)
            m *= 0.6
            v *= 0.6

        # Gradually tighten the smooth‑max
        eps_factor = max(eps_factor_min, eps_factor * eps_factor_decay)

        # Restart on long stagnation
        if it - last_improve_iter > 400_000:
            print(f"[Iter {it}] Stagnation detected – jitter restart.")
            z = _logits_from_distribution(best_seq) + 0.04 * np.random.randn(n)
            m.fill(0.0)
            v.fill(0.0)
            eps_factor = 0.20
            last_improve_iter = it

        # One up‑sampling stage (if time permits)
        if (it % 300_000 == 0) and (n < 20000) and (time.time() + 30.0 < deadline):
            new_n = min(20000, n * 2)
            print(f"[Iter {it}] Upsampling support: {n} → {new_n}")
            best_seq = _interpolate_distribution(best_seq, new_n)
            z = _logits_from_distribution(best_seq) + 0.006 * np.random.randn(new_n)
            m = np.zeros_like(z)
            v = np.zeros_like(z)
            n = new_n
            best_score = 2.0 * n * _conv_max(best_seq)
            it = 0
            last_improve_iter = 0
            eps_factor = 0.20
            # continue with the larger problem size

    # ------------------------------------------------------------------
    # 3. Final exact evaluation with the supplied evaluate_sequence()
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # ------------------------------------------------------------------
    # 4. Return the best sequence found.
    # ------------------------------------------------------------------
    return final_seq
