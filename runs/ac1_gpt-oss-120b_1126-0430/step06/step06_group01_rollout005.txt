"""
Arcsine‑bin construction with Adam refinement on the simplex.

1.  Build a high‑resolution discretisation of the continuous arcsine
    density  f(x)=1/(π·√(x(1‑x)))  on a uniform grid of size *n*.
    The resulting vector a (scaled to sum = 1) has an almost‑flat
    self‑convolution, giving an evaluation metric close to the theoretical
    lower bound 1.

2.  Parameterise a as a soft‑max of logits *z* so the simplex constraint
    (non‑negative entries, sum = 1) is automatically satisfied.

3.  Perform a short Adam‑style optimisation on the smoothed max of the
    convolution.  The smoothing parameter ε is gradually reduced so that
    the gradient focuses more and more on the true maximum entry.

4.  Keep the best vector seen and stop as soon as the metric drops below
    the target (≈ 1.503) or the time budget expires.

The algorithm runs in O(L log L) per iteration (FFT length L≈2ⁿ) and
fits comfortably into the 1000 s budget for n≥30 000.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (probability vector)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Logits that map to *a* via soft‑max."""
    return np.log(a + eps)


def _fft_len_for_conv(n: int) -> int:
    """
    Smallest power‑of‑two ≥ 2·n‑1, sufficient for a linear self‑convolution.
    """
    return 1 << ((2 * n - 1).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a (exact)."""
    n = a.shape[0]
    L = _fft_len_for_conv(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Smoothed max of a*a and gradient w.r.t. a (already multiplied by 4·n).

    Parameters
    ----------
    a : np.ndarray
        Probability vector (non‑negative, sum = 1).
    eps_factor : float
        Multiplicative factor for the smoothing constant:
        eps = eps_factor * max(conv).

    Returns
    -------
    max_conv : float
        Maximum entry of the self‑convolution.
    grad_a : np.ndarray
        Gradient of 4·n·smooth_max w.r.t. a.
    """
    n = a.shape[0]
    L = _fft_len_for_conv(n)

    a_fft = np.fft.rfft(a, L)

    # linear self‑convolution (length 2·n‑1)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # smoothed soft‑max weights (log‑sum‑exp trick)
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum

    # cross‑correlation a ⋆ w (linear, length n)
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # gradient (4·n multiplied)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a

    max_conv = _conv_max(best_a)
    best_score = 2.0 * n * max_conv
    print(f"Initial metric (2·n·max_conv) = {best_score:.8f}")

    # ------------------------------------------------------------------
    # 3. Adam‑style refinement on the simplex.
    #    The loop terminates when the budget runs out or the target is met.
    # ------------------------------------------------------------------
    # Logits parametrisation guarantees a stays on the simplex.
    z = _logits_from_distribution(best_a)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_alpha = 0.6

    eps_factor = 0.25            # initial smoothing factor
    eps_factor_min = 5e-5        # final smoothing factor
    eps_factor_decay = 0.9994    # geometric decay per iteration

    it = 0
    max_iters = 200_000
    TARGET = 1.503                # stop as soon as we beat the target

    while time.time() < deadline and it < max_iters:
        it += 1

        a_cur = _softmax(z)

        # Smoothed max and gradient
        max_conv, grad_a = _conv_max_and_grad(a_cur, eps_factor)
        score = 2.0 * n * max_conv

        # Keep the best seen vector
        if score < best_score:
            best_score = score
            best_a = a_cur.copy()
            if it % 1000 == 0:
                print(f"Iter {it:6d}: improved metric = {best_score:.8f}")

        # Stop early if we already satisfy the target.
        if best_score < TARGET:
            print(f"Target reached (metric = {best_score:.8f}) at iteration {it}.")
            break

        # Mirror‑descent gradient on the logits (soft‑max parameterisation)
        mu = np.dot(grad_a, a_cur)
        grad_z = a_cur * (grad_a - mu)

        # Adam bias‑corrected update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # Inverse‑sqrt step‑size schedule
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step = -alpha_t * m_hat / (np.sqrt(v_hat) + eps_adam)
        z = z + step

        # Clip logits to keep exp(z) numerically safe
        z = np.clip(z, -80.0, 80.0)

        # Decay the smoothing constant
        eps_factor = max(eps_factor_min, eps_factor * eps_factor_decay)

        # Occasionally inject tiny random jitter to escape plateaus
        if it % 5000 == 0 and np.random.rand() < 0.07:
            z += 0.0025 * np.random.randn(n)

        # Periodic full (exact) logging (cheap enough)
        if it % 25000 == 0:
            exact_max = _conv_max(best_a)
            exact_metric = 2.0 * n * exact_max
            print(f"Iter {it:6d}: exact best metric = {exact_metric:.8f}")

    # ------------------------------------------------------------------
    # 4. Return the best sequence (plain Python list).
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        exact_val = evaluate_sequence(final_seq)
        print(f"Exact evaluation after optimisation: {exact_val:.8f}")
    except Exception as exc:
        print(f"Exact evaluation failed: {exc}")

    return final_seq
