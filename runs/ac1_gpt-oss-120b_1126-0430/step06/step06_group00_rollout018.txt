def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import time, numpy as np, scipy.optimize
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2

    try:
        n = len(height_sequence_1)
    except:
        n = 20000

    # init arcsine distribution
    idx = np.arange(n, dtype=np.float64)
    x0 = idx / n
    x1 = (idx + 1) / n
    def arcsine_cdf(x):
        return (2./np.pi) * np.arcsin(np.sqrt(np.clip(x,0.,1.)))
    a = arcsine_cdf(x1) - arcsine_cdf(x0)
    a = a / a.sum()

    # create logits
    z = np.log(a + 1e-16)

    # define FFT length
    L = _fft_len_for_corr(n)

    def objective_and_grad(z_vec):
        # compute a
        a_vec = _softmax(z_vec)
        # compute FFT and conv
        a_fft = np.fft.rfft(a_vec, L)
        conv = np.fft.irfft(a_fft * a_fft, L)[:2*n-1]
        max_conv = conv.max()
        # smoothed gradient
        eps_factor = 0.05 # maybe schedule eventually small
        eps = max(max_conv * eps_factor, 1e-12)
        shifted = conv - max_conv
        w = np.exp(shifted / eps)
        w_sum = w.sum()
        w /= w_sum
        w_fft = np.fft.rfft(w, L)
        corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]
        grad_a = 4.0 * n * corr
        # gradient wrt logits
        mu = np.dot(grad_a, a_vec)
        grad_z = a_vec * (grad_a - mu)
        # objective: unsmoothed max for evaluation
        obj = 2.0 * n * max_conv
        return obj, grad_z.astype(np.float64)

    # Run L-BFGS
    res = scipy.optimize.minimize(fun=lambda z: objective_and_grad(z),
                                  x0=z,
                                  method='L-BFGS-B',
                                  jac=True,
                                  options={'maxiter': 2000, 'disp': False})
    best_a = _softmax(res.x)
    final_seq = best_a.tolist()
    # optionally evaluate final
    return final_seq

"""
Propose a step‑function (non‑negative sequence) that minimises the
evaluation metric 2·n·max(conv(a,a))/(sum(a)²).

The algorithm proceeds in three stages:

1. **Initial distribution** –   Either reuse the externally supplied
   `height_sequence_1` (re‑normalised) or construct a discretised
   arcsine distribution (the known near‑optimal continuous solution).

2. **L‑BFGS optimisation on the simplex** –   The sequence is parametrised
   by logits `z` with a soft‑max to guarantee non‑negativity and a unit
   sum.  A smoothed version of the max‑convolution (soft‑max over
   the autocorrelation) provides a differentiable surrogate; its
   gradient is obtained analytically via FFTs.  The bound `eps` of the
   soft‑max is taken as a small fraction of the current maximum
   convolution, which yields a tight approximation while remaining
   numerically stable.  SciPy’s L‑BFGS‑B optimiser quickly drives the
   objective down.

3. **Fine‑grained Adam refinement** –   After the quasi‑Newton phase a
   short Adam run (few × 10⁴ iterations) on the same smooth objective
   removes the residual noise and often yields a modest further
   improvement.

All heavy work (FFT based convolutions) is O(n log n) and easily fits
into the 1000 s budget.  The best sequence found is returned as a plain
Python list; the platform will evaluate it via `evaluate_sequence`.
"""

import time
import math
import numpy as np
from scipy import optimize, special

# ----------------------------------------------------------------------
# Helper functions (top level, no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max; output sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _fft_len_for_corr(n: int) -> int:
    """
    Smallest power‑of‑two ≥ 3·n‑2.
    Sufficient for linear convolution (length 2·n‑1) and correlation
    (length 3·n‑2) used in the gradient computation.
    """
    return 1 << ((3 * n - 2).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a (via FFT)."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Smoothed max of a*a and gradient w.r.t. a.

    Parameters
    ----------
    a : np.ndarray
        Probability vector (sum = 1, non‑negative).
    eps_factor : float
        Multiplicative factor for the smoothing constant:
        eps = max(eps_factor * max(conv), 1e‑12).

    Returns
    -------
    max_conv : float
        Maximum entry of the linear self‑convolution.
    grad_a : np.ndarray
        Gradient of 2·n·smooth_max w.r.t. a (already multiplied by 4·n).
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    a_fft = np.fft.rfft(a, L)

    # linear self‑convolution (length 2·n‑1)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # soft‑max weights over the convolution entries
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv          # for numerical stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or np.isnan(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # cross‑correlation a ⋆ w (length 3·n‑2, we need first n entries)
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # gradient (the factor 4·n comes from the derivative of 2·n·max)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, Σ x_i = 1}.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback (should not happen)
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# L‑BFGS objective (used by scipy.optimize.minimize)
# ----------------------------------------------------------------------


def _lbfgs_obj_and_grad(z: np.ndarray, n: int, L: int, eps_factor: float = 0.05):
    """
    Returns the objective (2·n·max(conv)) and its gradient w.r.t. logits `z`.
    The gradient is derived from a smoothed max (soft‑max with scale `eps`).
    """
    a = _softmax(z)

    # forward FFT of a
    a_fft = np.fft.rfft(a, L)

    # self‑convolution
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # smooth‑max weights
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or np.isnan(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # cross‑correlation a ⋆ w
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # gradient w.r.t. a (already contains factor 4·n)
    grad_a = 4.0 * n * corr

    # pull back through the soft‑max (mirror‑descent on the simplex)
    mu = np.dot(grad_a, a)
    grad_z = a * (grad_a - mu)

    obj = 2.0 * n * max_conv
    return obj, grad_z.astype(np.float64)

# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that minimises the evaluation metric.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a safety margin for final logging / possible cleanup
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # 0. Choose the length of the sequence
    # ------------------------------------------------------------------
    try:
        init_seq = np.array(height_sequence_1, dtype=np.float64)  # noqa: F821
        if init_seq.ndim != 1:
            raise ValueError
        n = init_seq.size
        print(f"[Info] Using externally supplied sequence of length n = {n}")
    except Exception:
        n = 20000
        init_seq = None
        print(f"[Info] No external sequence – using default n = {n}")

    # ------------------------------------------------------------------
    # 1. Initialise – either reuse or arcsine discretisation
    # ------------------------------------------------------------------
    if init_seq is not None:
        a = np.clip(init_seq, 0.0, None)
        s = a.sum()
        if s == 0.0 or np.isnan(s):
            raise RuntimeError("Provided sequence is not usable.")
        a /= s
    else:
        idx = np.arange(n, dtype=np.float64)
        x0 = idx / n
        x1 = (idx + 1) / n

        def arcsine_cdf(x):
            # arcsine CDF F(x) = (2/π)·arcsin(√x), guarded for numeric slip.
            return (2.0 / np.pi) * np.arcsin(np.sqrt(np.clip(x, 0.0, 1.0)))

        a = arcsine_cdf(x1) - arcsine_cdf(x0)
        a_sum = a.sum()
        if a_sum == 0.0 or np.isnan(a_sum):
            raise RuntimeError("Failed to initialise a valid distribution.")
        a /= a_sum

    # Basic sanity checks
    assert np.all(a >= 0.0)
    assert np.isfinite(a).all()
    assert abs(a.sum() - 1.0) < 1e-12

    # ------------------------------------------------------------------
    # 2. Optional quick scan over symmetric Beta(α,α) families
    #    (takes only a fraction of a second and may give a better start)
    # ------------------------------------------------------------------
    best_a = a.copy()
    best_score = 2.0 * n * _conv_max(best_a)

    # try a handful of α values in the range [0.5, 3.0]
    alphas = np.concatenate(
        (
            np.linspace(0.5, 1.0, 9),               # finer near arcsine
            np.linspace(1.1, 3.0, 7)                # broader tail
        )
    )
    for α in alphas:
        if time.time() >= deadline:
            break
        # regularised incomplete beta (SciPy’s betainc returns the regularised CDF)
        cdf_vals = special.betainc(α, α, (np.arange(1, n + 1) / n))
        cdf_prev = special.betainc(α, α, (np.arange(0, n) / n))
        a_beta = cdf_vals - cdf_prev
        a_beta_sum = a_beta.sum()
        if a_beta_sum == 0.0 or np.isnan(a_beta_sum):
            continue
        a_beta /= a_beta_sum
        score = 2.0 * n * _conv_max(a_beta)
        if score < best_score:
            best_score = score
            best_a = a_beta.copy()
            print(f"[Info] Beta(α={α:.3f}) improves best_score to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. L‑BFGS optimisation on the simplex (logits parametrisation)
    # ------------------------------------------------------------------
    # initialise logits from the current best probability vector
    z = np.log(best_a + 1e-16)          # avoid log(0)

    L_fft = _fft_len_for_corr(n)

    # L‑BFGS settings – limit the number of function evaluations to stay
    # well within the global time budget.
    maxfun = 3000  # each call ≈ 0.001–0.003 s, so ~few seconds total

    print("[Info] Starting L‑BFGS optimisation ...")
    res = optimize.minimize(
        fun=_lbfgs_obj_and_grad,
        x0=z,
        args=(n, L_fft, 0.05),          # eps_factor is a modest 5 %
        method="L-BFGS-B",
        jac=True,
        options={"maxfun": maxfun, "disp": False},
    )

    # Grab the solution and evaluate its exact score
    a_lbfgs = _softmax(res.x)
    score_lbfgs = 2.0 * n * _conv_max(a_lbfgs)
    print(f"[Info] L‑BFGS finished: score = {score_lbfgs:.6f}")
    if score_lbfgs < best_score:
        best_score = score_lbfgs
        best_a = a_lbfgs.copy()

    # ------------------------------------------------------------------
    # 4. Short Adam refinement on the simplex (optional, runs only if time)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0:   # need a modest amount of time to be worthwhile
        print("[Info] Starting Adam fine‑tuning ...")
        # Logits for the soft‑max parametrisation
        z = np.log(best_a + 1e-16)
        m = np.zeros_like(z)
        v = np.zeros_like(z)
        beta1, beta2 = 0.9, 0.999
        eps_adam = 1e-8

        base_alpha = 0.4          # a slightly larger initial step
        eps_factor = 0.08         # start with a bit more smoothing
        eps_factor_min = 5e-5
        eps_decay = 0.9997

        it = 0
        max_it = 200_000

        # Helper for fast logging
        def _log(it_, sc_):
            elapsed = time.time() - start_time
            print(f"[Iter {it_:6d}] score = {sc_:.6f}  elapsed = {elapsed:.1f}s")

        while time.time() < deadline and it < max_it:
            it += 1
            a_cur = _softmax(z)

            # Smoothed max‑convolution and its gradient
            max_conv, grad_a = _conv_max_and_grad(a_cur, eps_factor)
            cheap_score = 2.0 * n * max_conv

            if cheap_score < best_score:
                best_score = cheap_score
                best_a = a_cur.copy()
                if it % 5000 == 0:
                    _log(it, best_score)

            # Mirror‑descent gradient on logits
            mu = np.dot(grad_a, a_cur)
            grad_z = a_cur * (grad_a - mu)

            # Adam update (bias‑corrected)
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            alpha_t = base_alpha / (math.sqrt(it) + 1.0)
            step = - alpha_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            z = z + step
            z = np.clip(z, -80.0, 80.0)          # keep logits in a sane range

            eps_factor = max(eps_factor_min, eps_factor * eps_decay)

            # occasional tiny jitter to jump out of flat plateaus
            if it % 10000 == 0 and np.random.rand() < 0.15:
                z += 0.01 * np.random.randn(n)

        # record the best after Adam
        print(f"[Info] Adam finished after {it:,} iterations, best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Return the best sequence (as a plain Python list)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()

    # Optional exact evaluation for logging (the platform will evaluate anyway)
    try:
        final_score = evaluate_sequence(final_seq)   # noqa: F821
        print(f"[Info] Final exact score = {final_score:.6f}")
    except Exception as exc:
        print(f"[Warning] Evaluation error (ignored): {exc}")

    return final_seq
