def _beta_distribution(n, alpha, beta):
    i = np.arange(n) + 0.5

def _tukey_window_distribution(n, alpha):
    # alpha in [0,1], 0 -> Hann, 1->rectangular (uniform)
    if alpha <= 0:
        return _hann_window_distribution(n)
    if alpha >= 1:
        return np.full(n, 1.0/n)
    # compute using standard formula
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    w = np.empty_like(t)
    period = alpha / 2.0
    # first region
    idx1 = t < period
    w[idx1] = 0.5 * (1 + np.cos(np.pi * (2*t[idx1]/alpha - 1)))
    # middle region
    idx2 = (t >= period) & (t <= (1 - period))
    w[idx2] = 1.0
    # last region
    idx3 = t > (1 - period)
    w[idx3] = 0.5 * (1 + np.cos(np.pi * (2*t[idx3]/alpha - 2/alpha + 1)))
    a = w
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0/n)
    else:
        a = a / a_sum
    return a

#!/usr/bin/env python
"""
Non‑negative step‑function optimiser (enhanced search).

The optimiser proceeds in several stages:

1. **Initial support** – a modest length (≈ 5 000) is chosen so that many
   cheap evaluations fit into the time budget.

2. **Parametric sweep** – a large collection of hand‑crafted families
   (power‑law, symmetric product‑law, exponential, logistic, beta,
   sine‑power, Hann/Hamming/Blackman windows, Tukey windows) is scanned
   on a dense grid of parameters.  The best shape (as measured by the
   cheap objective `2·n·max(conv)`) is remembered.

3. **Random Dirichlet draws** – a few hundred random probability vectors
   are generated to diversify the search space.

4. **Mirror‑descent (Adam on logits)** – the best candidate from the
   sweep is refined by Adam on the log‑softmax (mirror descent on the
   probability simplex).  A smoothly decaying ε‑smoothing of the
   max‑operator is used; the smoothing factor is slowly reduced to turn
   the softmax into the exact sub‑gradient of the max.

5. **Upsampling** – once a good shape is found the support is enlarged
   (linear interpolation) to ≈ 15 000 points and a short second Adam
   phase fine‑tunes the finer grid.

All helper functions are top‑level, there are no closures or lambda
expressions, and the total runtime respects the 1000 s budget.  The
final sequence is returned as a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (all top‑level)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max, returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    s = e.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / z.size)
    return e / s


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it that reproduces `a` after a soft‑max (adds a tiny epsilon)."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """Smallest power‑of‑two ≥ 3·n‑2 (enough for linear self‑convolution)."""
    return 1 << ((3 * n - 2).bit_length())


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    s = new_a.sum()
    if s == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= s
    return new_a


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}, normalised."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _sym_product_power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ ((i+1)*(n‑i))^{-p}, symmetric, normalised."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) * (n - i)) ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _exp_decay_distribution(n: int, lam: float) -> np.ndarray:
    """a_i ∝ e^{-lam·i}+e^{-lam·(n‑1‑i)}, symmetric exponential."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _logistic_distribution(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ 1/(1+e^{α·(i‑mid)}), normalised; heavy left‑hand side."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = 1.0 / (1.0 + np.exp(alpha * (i - mid)))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑type density: a_i ∝ (i+½)^{α‑1}(n‑i‑½)^{β‑1}, normalised."""
    i = np.arange(n, dtype=np.float64) + 0.5
    # work in log‑space for stability
    log_a = (alpha - 1.0) * np.log(i) + (beta - 1.0) * np.log(n + 0.5 - i)
    log_a -= np.max(log_a)          # avoid overflow
    a = np.exp(log_a)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _sin_power_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ sin^p(π·(i+½)/n), normalised."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n) ** p
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _hann_window_distribution(n: int) -> np.ndarray:
    """Hann window (raised cosine), normalised."""
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    s = a.sum()
    a /= s
    return a


def _hamming_window_distribution(n: int) -> np.ndarray:
    """Hamming window, normalised."""
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    s = a.sum()
    a /= s
    return a


def _blackman_window_distribution(n: int) -> np.ndarray:
    """Blackman window, normalised."""
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    s = a.sum()
    a /= s
    return a


def _tukey_window_distribution(n: int, alpha: float) -> np.ndarray:
    """
    Tukey window with taper proportion `alpha` ∈ [0,1].
    alpha=0 → Hann, alpha=1 → rectangular (uniform).
    """
    if alpha <= 0.0:
        return _hann_window_distribution(n)
    if alpha >= 1.0:
        return np.full(n, 1.0 / n)

    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)  # in [0,1]
    w = np.empty_like(t)

    period = alpha / 2.0
    # first cosine taper
    idx1 = t < period
    w[idx1] = 0.5 * (1 + np.cos(np.pi * (2 * t[idx1] / alpha - 1)))
    # flat middle
    idx2 = (t >= period) & (t <= 1.0 - period)
    w[idx2] = 1.0
    # last cosine taper
    idx3 = t > 1.0 - period
    w[idx3] = 0.5 * (1 + np.cos(np.pi * (2 * t[idx3] / alpha - 2.0 / alpha + 1)))
    a = w
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Returns (max_conv, grad_a) where
        grad_a = ∇_a [ 2·n·smooth_max(a*a) ].
    The smoothing uses ε = eps_factor * max_conv.
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    # FFT of a (zero‑padded)
    a_fft = np.fft.rfft(a, L)

    # Convolution a*a
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # Smoothed soft‑max weights
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv            # ≤ 0 for stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # Correlation a ⋆ w (size n)
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # Gradient of 2·n·smooth_max = 4·n·corr
    grad_a = 4.0 * n * corr
    return max_conv, grad_a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights (step‑function) that (hopefully)
    minimises the supplied `evaluate_sequence` score.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0  # safety margin for final evaluation

    # --------------------------------------------------------------
    # 0. Choose a modest support size for the bulk of the work.
    # --------------------------------------------------------------
    n = 5000
    best_seq = np.full(n, 1.0 / n)          # fallback uniform
    best_score = float("inf")

    # --------------------------------------------------------------
    # 0a. Warm‑start from a globally supplied sequence if available.
    # --------------------------------------------------------------
    try:
        init_global = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        cand = _interpolate_distribution(init_global, n)
        score = 2.0 * n * _conv_max(cand)
        best_seq, best_score = cand, score
        print("[Init] Warm‑started from global sequence (down‑sampled).")
    except Exception:
        # No global sequence – continue with parametric sweep.
        pass

    # --------------------------------------------------------------
    # 1. Parametric sweep over a variety of families.
    # --------------------------------------------------------------
    def _maybe_update(a: np.ndarray):
        nonlocal best_seq, best_score
        s = 2.0 * n * _conv_max(a)
        if s < best_score:
            best_score, best_seq = s, a.copy()

    print("[Sweep] Scanning parametric families.")
    # Power‑law
    for p in np.linspace(0.1, 2.5, 30):
        if time.time() >= deadline:
            break
        _maybe_update(_power_law_distribution(n, p))

    # Symmetric product‑power‑law
    for p in np.linspace(0.1, 2.5, 30):
        if time.time() >= deadline:
            break
        _maybe_update(_sym_product_power_law_distribution(n, p))

    # Exponential decay (symmetric)
    for loglam in np.linspace(-7, -2, 30):
        if time.time() >= deadline:
            break
        lam = 10.0 ** loglam
        _maybe_update(_exp_decay_distribution(n, lam))

    # Logistic left‑heavy
    for alpha in np.linspace(0.1, 5.0, 30):
        if time.time() >= deadline:
            break
        _maybe_update(_logistic_distribution(n, alpha))

    # Beta family (symmetric)
    for alpha in np.linspace(0.2, 1.8, 30):
        if time.time() >= deadline:
            break
        _maybe_update(_beta_distribution(n, alpha, alpha))

    # Sine‑power family
    for p in np.linspace(0.5, 4.0, 30):
        if time.time() >= deadline:
            break
        _maybe_update(_sin_power_distribution(n, p))

    # Hann, Hamming, Blackman windows
    for fn in (_hann_window_distribution,
               _hamming_window_distribution,
               _blackman_window_distribution):
        if time.time() >= deadline:
            break
        _maybe_update(fn(n))

    # Tukey windows (various tapers)
    for alpha in np.linspace(0.0, 1.0, 25):
        if time.time() >= deadline:
            break
        _maybe_update(_tukey_window_distribution(n, alpha))

    # Random Dirichlet draws (small number, cheap)
    for _ in range(200):
        if time.time() >= deadline:
            break
        rand_vec = np.random.rand(n)
        cand = rand_vec / rand_vec.sum()
        _maybe_update(cand)

    print(f"[Best after sweep] Score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2. Gradient‑descent refinement (Adam on logits, soft‑max mirror).
    # --------------------------------------------------------------
    # Initialise logits from the best distribution.
    z = _logits_from_distribution(best_seq) + 0.01 * np.random.randn(n)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    base_alpha = 0.6          # base step‑size for logits
    eps_factor = 0.5          # initial smoothing factor (ε = eps_factor·max_conv)
    eps_factor_min = 1e-7
    eps_factor_decay = 0.9995

    best_overall_seq = _softmax(z)
    best_overall_score = 2.0 * n * _conv_max(best_overall_seq)
    best_seq = best_overall_seq.copy()
    last_improve = 0
    it = 0

    print("[Optim] Starting Adam refinement.")
    while time.time() < deadline and it < 2_500_000:
        it += 1

        a = _softmax(z)

        # cheap evaluation + smoothed gradient
        max_conv, grad_a = _conv_max_and_grad(a, eps_factor)
        cheap_score = 2.0 * n * max_conv

        # keep best ever seen
        if cheap_score < best_overall_score - 1e-12:
            best_overall_score = cheap_score
            best_overall_seq = a.copy()
            last_improve = it
            if it % 10000 == 0:
                print(f"[Iter {it:07d}] NEW BEST cheap = {cheap_score:.6f}")

        # mirror‑descent gradient w.r.t. logits
        mu = np.dot(grad_a, a)                # <grad, a>
        grad_z = a * (grad_a - mu)

        # Adam moment updates (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step‑size schedule (inverse‑sqrt decay)
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step_dir = - m_hat / (np.sqrt(v_hat) + eps_adam)

        # Back‑tracking line search (up to 5 halvings)
        accepted = False
        trial_alpha = alpha_t
        for _ in range(5):
            z_new = z + trial_alpha * step_dir
            z_new = np.clip(z_new, -50.0, 50.0)
            a_new = _softmax(z_new)
            new_score = 2.0 * n * _conv_max(a_new)
            if new_score <= cheap_score:
                z = z_new
                accepted = True
                break
            trial_alpha *= 0.5

        if not accepted:
            # take a modest step even if not improving
            z = z + alpha_t * step_dir
            z = np.clip(z, -50.0, 50.0)

        # occasional jitter to escape plateaux
        if (it % 2000 == 0) and (np.random.rand() < 0.12):
            z += 0.02 * np.random.randn(n)
            m *= 0.6
            v *= 0.6

        # decay smoothing factor
        eps_factor = max(eps_factor_min, eps_factor * eps_factor_decay)

        # restart if stagnation persists
        if it - last_improve > 250_000:
            print(f"[Iter {it}] Stagnation detected – jitter restart.")
            z = _logits_from_distribution(best_overall_seq) + 0.04 * np.random.randn(n)
            m.fill(0.0)
            v.fill(0.0)
            eps_factor = 0.5
            last_improve = it

        # occasional up‑sampling to a finer grid (once)
        if (it % 300_000 == 0) and (n < 15000) and (time.time() + 30.0 < deadline):
            new_n = min(15000, int(n * 1.5))
            print(f"[Iter {it}] Upsampling support: {n} → {new_n}")
            best_overall_seq = _interpolate_distribution(best_overall_seq, new_n)
            z = _logits_from_distribution(best_overall_seq) + 0.005 * np.random.randn(new_n)
            m = np.zeros_like(z)
            v = np.zeros_like(z)
            n = new_n
            best_overall_score = 2.0 * n * _conv_max(best_overall_seq)
            it = 0
            last_improve = 0
            eps_factor = 0.5
            # continue optimisation on larger support

    # --------------------------------------------------------------
    # 3. Final exact evaluation with the platform's routine.
    # --------------------------------------------------------------
    final_seq = best_overall_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # --------------------------------------------------------------
    # 4. Return the best sequence discovered.
    # --------------------------------------------------------------
    return final_seq
