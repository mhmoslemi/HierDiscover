"""
Step‑function optimizer (v3).

- Extends the previous three‑stage optimizer with a richer
  analytical‑shape pre‑search (Beta‑distribution, sinusoidal
  power families, edge‑peaked “U‑shaped” families) and an
  adaptive multi‑restart Adam routine.
- Uses a more aggressive temperature schedule for the smooth‑max
  approximation and a longer projected‑gradient refinement.
- Dynamically allocates the 1000 s budget over a set of support
  lengths, always keeping a safety margin.
"""

import time
import math
import numpy as np

# ------------------------------------------------------------
# Utility functions (FFT convolution, correlation, softmax,
# simplex projection, objective)
# ------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·len(a)‑1) via real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return conv_full[: 2 * n - 1]

def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation of vectors w (length 2n‑1) and a (length n):
        g[i] = Σ_t w[i+t]·a[t]    for i=0,…,n‑1
    Implemented with real FFT.
    """
    n = a.shape[0]
    m = w.shape[0]                     # = 2n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]

def _softmax(x: np.ndarray) -> np.ndarray:
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()

def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for Σ a = 1."""
    n = a.size
    conv = _conv_self_fft(a)
    return float(2.0 * n * conv.max())

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _uniform_shape(n: int):
    return np.full(n, 1.0 / n, dtype=float)

def _exponential_shapes(n: int, rng: np.random.Generator):
    cand = []
    r_vals = np.linspace(0.90, 0.9999, 15)
    idx = np.arange(n, dtype=float)
    for r in r_vals:
        a = r ** idx
        a = a / a.sum()
        cand.append(a)
        cand.append(a[::-1])
    return cand

def _gaussian_shapes(n: int):
    cand = []
    idx = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    sigma_vals = np.linspace(0.10 * n, 0.40 * n, 8)
    for sigma in sigma_vals:
        a = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
        a = a / a.sum()
        cand.append(a)
    return cand

def _sparse_dirichlet_shapes(n: int, rng: np.random.Generator):
    cand = []
    k = max(2, int(0.03 * n))
    for _ in range(4):
        a = np.zeros(n, dtype=float)
        pos = rng.choice(n, size=k, replace=False)
        vals = rng.random(k)
        a[pos] = vals
        a = a / a.sum()
        cand.append(a)
    return cand

def _random_dirichlet_shape(n: int, rng: np.random.Generator):
    a = rng.random(n)
    return a / a.sum()

def _beta_shape(n: int, alpha: float, beta: float):
    """Discrete Beta(α,β) density on {0,…,n‑1}."""
    idx = np.arange(n, dtype=float) + 0.5
    x = idx / (n + 0.5)          # ∈ (0,1)
    a = (x ** (alpha - 1)) * ((1 - x) ** (beta - 1))
    # Guard against possible underflow/overflow
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        # fallback to uniform
        return _uniform_shape(n)
    return a / a.sum()

def _beta_shapes(n: int):
    cand = []
    alphas = [0.2, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0, 5.0]
    # symmetric
    for a in alphas:
        cand.append(_beta_shape(n, a, a))
    # asymmetric (left‑heavy)
    for a in alphas:
        cand.append(_beta_shape(n, a, 1.0))
    # asymmetric (right‑heavy)
    for b in alphas:
        cand.append(_beta_shape(n, 1.0, b))
    return cand

def _sin_power_shapes(n: int):
    cand = []
    idx = np.arange(n, dtype=float) + 0.5
    x = np.pi * idx / n   # in (0, π)
    exps = [0.5, 1.0, 2.0, 4.0, 8.0]
    for p in exps:
        a = np.sin(x) ** p
        # avoid all‑zero when p large
        if a.max() == 0.0:
            continue
        a = a / a.sum()
        cand.append(a)
    return cand

def _cos_power_shapes(n: int):
    """U‑shaped variants based on 1‑cos."""
    cand = []
    idx = np.arange(n, dtype=float)
    x = np.pi * idx / (n - 1)  # 0..π
    exps = [0.5, 1.0, 2.0, 4.0, 8.0]
    for p in exps:
        a = (1.0 - np.cos(x)) ** p   # 0 at ends, max in centre
        # to obtain edge‑peaked shape, invert
        inv = 1.0 - a / a.max()      # now peaks at edges
        inv = np.maximum(inv, 0.0)
        inv = inv / inv.sum()
        cand.append(inv)
        # also keep the original centre‑peaked version
        a = a / a.sum()
        cand.append(a)
    return cand

def _shape_candidates(n: int, rng: np.random.Generator):
    """Aggregate all analytical families."""
    cand = [_uniform_shape(n)]
    cand.extend(_exponential_shapes(n, rng))
    cand.extend(_gaussian_shapes(n))
    cand.extend(_sparse_dirichlet_shapes(n, rng))
    cand.append(_random_dirichlet_shape(n, rng))
    cand.extend(_beta_shapes(n))
    cand.extend(_sin_power_shapes(n))
    cand.extend(_cos_power_shapes(n))
    return cand

def _best_initial_sequence(n: int, rng: np.random.Generator, time_limit: float):
    """Return the shape with smallest exact objective found within time_limit."""
    start = time.time()
    best_a = None
    best_obj = float("inf")
    for a in _shape_candidates(n, rng):
        if time.time() - start > time_limit:
            break
        obj = _objective(a)
        if obj < best_obj:
            best_obj, best_a = obj, a
    # fallback if nothing evaluated (should not happen)
    if best_a is None:
        a = np.full(n, 1.0 / n, dtype=float)
        best_a = a
        best_obj = _objective(a)
    return best_a, best_obj

def _run_opt(n: int,
             time_limit: float,
             init_a: np.ndarray | None = None,
             seed: int = 0) -> tuple[list[float], float]:
    """Adam optimisation on the log‑it parametrisation of the simplex."""
    start = time.time()
    rng = np.random.default_rng(seed)

    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
        a = a / a.sum()
    else:
        a = rng.random(n)
        a = a / a.sum()

    # logits
    x = np.log(a + 1e-16)
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # Temperature schedule (smooth‑max)
    T_start, T_end = 0.5, 1e6          # same as before but we allow faster growth
    lr_start, lr_end = 0.30, 0.0005   # slightly larger initial LR

    best_a = a.copy()
    best_obj = _objective(best_a)

    it = 0
    while time.time() - start < time_limit:
        it += 1
        # soft‑max to stay on simplex
        a = _softmax(x)

        conv = _conv_self_fft(a)
        M = conv.max()
        obj = float(2.0 * n * M)

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a.copy()
            # optional logging
            print(f"[n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        # schedule based on elapsed fraction
        t = min((time.time() - start) / time_limit, 1.0)
        T = T_start * math.exp(math.log(T_end / T_start) * t)
        lr = lr_start * math.exp(math.log(lr_end / lr_start) * t)

        # smooth‑max weights
        if T > 5e5:
            # hard max – simple one‑hot weight
            w = np.zeros_like(conv)
            w[int(np.argmax(conv))] = 1.0
        else:
            # soft max
            shifted = conv - M   # ≤0
            w = np.exp(T * shifted)
            s = w.sum()
            if s == 0.0:
                w = np.ones_like(conv) / conv.size
            else:
                w /= s

        # gradient of (smooth) max w.r.t. a
        g_a = 2.0 * _correlation_fft(w, a)

        # natural gradient in log‑it space
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # jitter occasionally
        if it % 2500 == 0:
            jitter = rng.normal(scale=0.012, size=n)
            x += jitter

        # hard restart every 50000 iters (if we have that many)
        if it % 50000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.008, size=n)
            m.fill(0.0)
            v.fill(0.0)

    # ----- post‑processing -------------------------------------------------
    # discard near‑zero entries to possibly shrink support
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = _objective(best_a)
        print(f"[compress] trimmed support → {best_a.size}, O={best_obj:.6f}")

    # a cheap random mass‑transfer local search
    remaining = time_limit - (time.time() - start)
    if remaining > 0.5:
        a_loc = best_a.copy()
        obj_loc = best_obj
        n_loc = a_loc.size
        while time.time() - start < time.time() + remaining * 0.25:
            i, j = rng.choice(n_loc, size=2, replace=False)
            delta = a_loc[i] * rng.random() * 0.025
            if delta <= 0.0:
                continue
            a_new = a_loc.copy()
            a_new[i] -= delta
            a_new[j] += delta
            conv_new = _conv_self_fft(a_new)
            obj_new = float(2.0 * n_loc * conv_new.max())
            if obj_new < obj_loc:
                a_loc, obj_loc = a_new, obj_new
                if obj_new < best_obj:
                    best_a, best_obj = a_new, obj_new
        print(f"[local] post‑Adam O={best_obj:.6f}")

    return best_a.tolist(), best_obj

def _run_sa(init_seq: list[float],
            time_limit: float,
            seed: int = 0) -> tuple[list[float], float]:
    """Targeted simulated‑annealing followed by exact projected gradient."""
    start = time.time()
    rng = np.random.default_rng(seed)

    a = np.asarray(init_seq, dtype=float)
    if a.sum() < 1e-12:
        a = np.ones_like(a) / a.size
    else:
        a = a / a.sum()
    n = a.shape[0]

    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())
    best_seq = a.copy()

    # SA schedule
    T = best_obj * 0.25 + 1e-8
    T_min = best_obj * 1e-6 + 1e-12
    decay = 0.996

    it = 0
    while time.time() - start < time_limit * 0.65:
        it += 1
        k_max = int(np.argmax(conv))

        # compute contribution weight for positions that currently affect the peak
        i_vals = np.arange(n)
        j_vals = k_max - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i_vals[mask]] * a[j_vals[mask]]
        total_contrib = contrib.sum()

        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = rng.choice(n, p=probs)
        else:
            i = rng.integers(n)

        # distinct destination
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # move a modest fraction
        delta = a[i] * rng.random() * 0.06
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # re‑normalise
        a_new = np.clip(a_new, 0.0, None)
        a_new /= a_new.sum()

        conv_new = _conv_self_fft(a_new)
        obj_new = float(2.0 * n * conv_new.max())

        if obj_new < best_obj - 1e-14:
            a, conv, best_obj, best_seq = a_new, conv_new, obj_new, a_new.copy()
            print(f"[SA it={it}] improved O={best_obj:.6f}")
        else:
            if rng.random() < math.exp(-(obj_new - best_obj) / T):
                a, conv = a_new, conv_new

        if it % 3000 == 0:
            T = max(T * decay, T_min)

        if it % 9000 == 0:
            # global jitter
            noise = rng.normal(scale=0.007, size=n)
            a = np.clip(a + noise, 0.0, None)
            a /= a.sum()
            conv = _conv_self_fft(a)

    # ---------- projected gradient descent ----------
    remaining = time_limit - (time.time() - start)
    if remaining > 0.3:
        a = best_seq.copy()
        n = a.size
        it_g = 0
        eta0 = 0.30
        while time.time() - start < time.time() + remaining * 0.5:
            it_g += 1
            conv = _conv_self_fft(a)
            k_max = int(np.argmax(conv))
            w = np.zeros_like(conv)
            w[k_max] = 1.0
            g = 2.0 * _correlation_fft(w, a)
            eta = eta0 * (0.997 ** it_g)
            a_new = a - eta * g
            a_new = _project_simplex(a_new)
            obj_new = float(2.0 * n * _conv_self_fft(a_new).max())
            if obj_new < best_obj - 1e-14:
                a, best_obj, best_seq = a_new, obj_new, a_new.copy()
                print(f"[PG it={it_g}] improved O={best_obj:.6f}")
            else:
                # reduce step size if stuck
                eta0 *= 0.98
                if eta0 < 1e-6:
                    break

    return best_seq.tolist(), best_obj

        pos = rng.choice(n, size=k, replace=False)
        vals = rng.random(k)
        a[pos] = vals
        a = a / a.sum()
        cand.append(a)
    return cand


def _random_dirichlet_shape(n: int, rng: np.random.Generator):
    a = rng.random(n)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) density on {0,…,n‑1} (scaled and normalised)."""
    idx = np.arange(n, dtype=float) + 0.5
    x = idx / (n + 0.5)                     # ∈ (0,1)
    a = (x ** (alpha - 1)) * ((1 - x) ** (beta - 1))
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        return _uniform_shape(n)
    return a / a.sum()


def _beta_shapes(n: int):
    cand = []
    alphas = [0.2, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0, 5.0]
    # symmetric
    for a in alphas:
        cand.append(_beta_shape(n, a, a))
    # left‑heavy
    for a in alphas:
        cand.append(_beta_shape(n, a, 1.0))
    # right‑heavy
    for b in alphas:
        cand.append(_beta_shape(n, 1.0, b))
    return cand


def _sin_power_shapes(n: int):
    cand = []
    idx = np.arange(n, dtype=float) + 0.5
    x = np.pi * idx / n                     # (0,π)
    exps = [0.5, 1.0, 2.0, 4.0, 8.0]
    for p in exps:
        a = np.sin(x) ** p
        if a.max() == 0.0:
            continue
        a = a / a.sum()
        cand.append(a)
    return cand


def _cos_power_shapes(n: int):
    """Both centre‑peaked and edge‑peaked versions based on (1‑cos)."""
    cand = []
    idx = np.arange(n, dtype=float)
    x = np.pi * idx / (n - 1)               # 0 … π
    exps = [0.5, 1.0, 2.0, 4.0, 8.0]
    for p in exps:
        centre = (1.0 - np.cos(x)) ** p      # 0 at ends, max in centre
        if centre.max() > 0:
            centre = centre / centre.sum()
            cand.append(centre)
        # edge‑peaked version (invert the centre shape)
        edge = 1.0 - centre / centre.max()
        edge = np.maximum(edge, 0.0)
        edge = edge / edge.sum()
        cand.append(edge)
    return cand


def _shape_candidates(n: int, rng: np.random.Generator):
    """Collect a diverse set of analytically motivated candidates."""
    cand = [_uniform_shape(n)]
    cand.extend(_exponential_shapes(n, rng))
    cand.extend(_gaussian_shapes(n))
    cand.extend(_sparse_dirichlet_shapes(n, rng))
    cand.append(_random_dirichlet_shape(n, rng))
    cand.extend(_beta_shapes(n))
    cand.extend(_sin_power_shapes(n))
    cand.extend(_cos_power_shapes(n))
    return cand


def _best_initial_sequence(n: int, rng: np.random.Generator, time_limit: float):
    """Pick the candidate with the smallest exact objective (quick scan)."""
    start = time.time()
    best_a = None
    best_obj = float("inf")
    for a in _shape_candidates(n, rng):
        if time.time() - start > time_limit:
            break
        obj = _objective(a)
        if obj < best_obj:
            best_obj, best_a = obj, a
    if best_a is None:                     # fallback – should never happen
        best_a = _uniform_shape(n)
        best_obj = _objective(best_a)
    return best_a, best_obj


# ------------------------------------------------------------
# Adam optimiser on the simplex (soft‑max + temperature)
# ------------------------------------------------------------

def _run_opt(n: int,
             time_limit: float,
             init_a: np.ndarray | None = None,
             seed: int = 0) -> tuple[list[float], float]:
    """
    Adam optimisation on the log‑it simplex.
    Returns the best sequence (as list) and its exact objective.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
        a = a / a.sum()
    else:
        a = rng.random(n)
        a = a / a.sum()

    x = np.log(a + 1e-16)      # logits
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # temperature (smooth‑max) schedule
    T_start, T_end = 0.5, 1e6
    lr_start, lr_end = 0.30, 0.0005

    best_a = a.copy()
    best_obj = _objective(best_a)

    it = 0
    while time.time() - start < time_limit:
        it += 1

        a = _softmax(x)

        conv = _conv_self_fft(a)
        M = conv.max()
        obj = float(2.0 * n * M)

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a.copy()
            print(f"[n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        # schedule based on elapsed fraction
        t = min((time.time() - start) / time_limit, 1.0)
        T = T_start * math.exp(math.log(T_end / T_start) * t)
        lr = lr_start * math.exp(math.log(lr_end / lr_start) * t)

        # smooth‑max weights
        if T > 5e5:
            w = np.zeros_like(conv)
            w[int(np.argmax(conv))] = 1.0
        else:
            shifted = conv - M          # ≤ 0
            w = np.exp(T * shifted)
            s = w.sum()
            if s == 0.0:
                w = np.ones_like(conv) / conv.size
            else:
                w /= s

        # gradient of (smooth) max w.r.t. a
        g_a = 2.0 * _correlation_fft(w, a)

        # natural gradient in log‑it space
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape plateaus
        if it % 2500 == 0:
            jitter = rng.normal(scale=0.012, size=n)
            x += jitter

        # hard restart every 50000 iterations (if we ever reach that)
        if it % 50000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.008, size=n)
            m.fill(0.0)
            v.fill(0.0)

    # ----- post‑processing -------------------------------------------------
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = _objective(best_a)
        print(f"[compress] trimmed support → {best_a.size}, O={best_obj:.6f}")

    # cheap random mass‑transfer local search
    remaining = time_limit - (time.time() - start)
    if remaining > 0.5:
        a_loc = best_a.copy()
        obj_loc = best_obj
        n_loc = a_loc.size
        while time.time() - start < time.time() + remaining * 0.25:
            i, j = rng.choice(n_loc, size=2, replace=False)
            delta = a_loc[i] * rng.random() * 0.025
            if delta <= 0.0:
                continue
            a_new = a_loc.copy()
            a_new[i] -= delta
            a_new[j] += delta
            conv_new = _conv_self_fft(a_new)
            obj_new = float(2.0 * n_loc * conv_new.max())
            if obj_new < obj_loc:
                a_loc, obj_loc = a_new, obj_new
                if obj_new < best_obj:
                    best_a, best_obj = a_new, obj_new
        print(f"[local] post‑Adam O={best_obj:.6f}")

    return best_a.tolist(), best_obj


# ------------------------------------------------------------
# Simulated annealing + exact projected gradient (refinement)
# ------------------------------------------------------------

def _run_sa(init_seq: list[float],
            time_limit: float,
            seed: int = 0) -> tuple[list[float], float]:
    """
    Targeted simulated‑annealing followed by a short exact projected‑gradient
    descent (using the true max‑gradient).
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    a = np.asarray(init_seq, dtype=float)
    if a.sum() < 1e-12:
        a = np.ones_like(a) / a.size
    else:
        a = a / a.sum()
    n = a.shape[0]

    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())
    best_seq = a.copy()

    # SA schedule
    T = best_obj * 0.25 + 1e-8
    T_min = best_obj * 1e-6 + 1e-12
    decay = 0.996

    it = 0
    while time.time() - start < time_limit * 0.65:
        it += 1
        k_max = int(np.argmax(conv))

        # contribution of each index to the current peak
        i_vals = np.arange(n)
        j_vals = k_max - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i_vals[mask]] * a[j_vals[mask]]
        total_contrib = contrib.sum()

        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = rng.choice(n, p=probs)
        else:
            i = rng.integers(n)

        # distinct destination index
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # move a modest fraction of mass from i to j
        delta = a[i] * rng.random() * 0.06
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.clip(a_new, 0.0, None)
        a_new /= a_new.sum()

        conv_new = _conv_self_fft(a_new)
        obj_new = float(2.0 * n * conv_new.max())

        if obj_new < best_obj - 1e-14:
            a, conv, best_obj, best_seq = a_new, conv_new, obj_new, a_new.copy()
            print(f"[SA it={it}] improved O={best_obj:.6f}")
        else:
            if rng.random() < math.exp(-(obj_new - best_obj) / T):
                a, conv = a_new, conv_new

        if it % 3000 == 0:
            T = max(T * decay, T_min)

        if it % 9000 == 0:
            # occasional global jitter
            noise = rng.normal(scale=0.007, size=n)
            a = np.clip(a + noise, 0.0, None)
            a /= a.sum()
            conv = _conv_self_fft(a)

    # ------------------------------------------------------------
    # Exact projected‑gradient descent (hard max)
    # ------------------------------------------------------------
    remaining = time_limit - (time.time() - start)
    if remaining > 0.3:
        a = best_seq.copy()
        n = a.size
        it_g = 0
        eta0 = 0.30
        while time.time() - start < time.time() + remaining * 0.5:
            it_g += 1
            conv = _conv_self_fft(a)
            k_max = int(np.argmax(conv))
            w = np.zeros_like(conv)
            w[k_max] = 1.0
            g = 2.0 * _correlation_fft(w, a)
            eta = eta0 * (0.997 ** it_g)
            a_new = a - eta * g
            a_new = _project_simplex(a_new)
            obj_new = float(2.0 * n * _conv_self_fft(a_new).max())
            if obj_new < best_obj - 1e-14:
                a, best_obj, best_seq = a_new, obj_new, a_new.copy()
                print(f"[PG it={it_g}] improved O={best_obj:.6f}")
            else:
                eta0 *= 0.98
                if eta0 < 1e-6:
                    break

    return best_seq.tolist(), best_obj


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------

def propose_candidate(seed: int = 42,
                      budget_s: int = 1000,
                      **_ignore) -> list[float]:
    """
    Optimises a non‑negative step function for the supplied evaluation metric.
    The pipeline consists of an enriched analytical‑shape pre‑search,
    several Adam restarts on the simplex, and a final simulated‑annealing /
    projected‑gradient refinement.
    Returns a plain list of floats ready for ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.6       # safety margin

    # support lengths to try – longer vectors give more flexibility
    lengths = [1500, 3000, 6000, 9000, 12000]

    best_seq = None
    best_obj = float("inf")

    # ------------------------------------------------------------
    # Stage 1 + 2 – shape pre‑search + multiple Adam runs per length
    # ------------------------------------------------------------
    for idx, n in enumerate(lengths):
        if time.time() >= deadline - 5.0:
            break

        remaining = deadline - time.time()
        # allocate time slices
        if idx == 0:
            shape_time = min(4.0, remaining * 0.07)
            adam_time_total = remaining * 0.30
        else:
            shape_time = min(3.0, remaining * 0.05)
            adam_time_total = remaining * 0.20

        # ---- shape pre‑search
        init_a, init_obj = _best_initial_sequence(n, rng, shape_time)
        print(f"[init n={n}] shape‑obj={init_obj:.6f}")

        # ---- several Adam restarts (use the allotted time)
        adam_start = time.time()
        run_counter = 0
        while time.time() - adam_start < adam_time_total and (deadline - time.time()) > 4.0:
            run_counter += 1
            sub_limit = min(45.0, deadline - time.time() - 4.0)
            seq, val = _run_opt(
                n=n,
                time_limit=sub_limit,
                init_a=init_a,
                seed=seed + idx * 1000 + run_counter,
            )
            if val < best_obj:
                best_obj = val
                best_seq = np.asarray(seq, dtype=float)
                print(f"[global] new overall best O={best_obj:.6f} (n={n})")
                # feed the current best to the next run as a warm start
                init_a = best_seq.copy()

    # ------------------------------------------------------------
    # Stage 3 – targeted simulated‑annealing + exact PG refinement
    # ------------------------------------------------------------
    remaining = deadline - time.time()
    if best_seq is not None and remaining > 6.0:
        seq_ref, val_ref = _run_sa(
            best_seq.tolist(),
            remaining,
            seed=seed + 999999,
        )
        if val_ref < best_obj:
            best_obj, best_seq = val_ref, np.asarray(seq_ref, dtype=float)
            print(f"[final] after SA+PG new best O={best_obj:.6f}")

    # ------------------------------------------------------------
    # Safety fallback and final sanity check
    # ------------------------------------------------------------
    if best_seq is None or best_seq.size == 0:
        best_seq = np.full(200, 1.0 / 200.0, dtype=float)
        best_obj = _objective(best_seq)

    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    # Clip to evaluator limits and guarantee a non‑tiny sum
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    if best_seq.sum() < 0.01:
        best_seq = np.ones_like(best_seq) / best_seq.size

    return best_seq.tolist()
