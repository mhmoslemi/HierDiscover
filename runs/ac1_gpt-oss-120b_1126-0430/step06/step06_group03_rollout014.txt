"""
Hybrid optimiser with a multi‑scale strategy.

The previous implementation worked directly on the full size (≈25 000)
sequence, which limits the number of gradient steps that can be taken
inside the 1000 s budget (each step needs an FFT of size ≈2⁁17).  
The new version first searches a *small* proxy problem (n≈4096).  Because
the FFT is much cheaper we can afford many more Adam iterations and a
finer exploration of the parameter space.  After a good proxy distribution
has been found it is interpolated to the target length and refined again
with a short Adam run followed by a cheap random‑walk finalisation.

All helper functions are top‑level (no closures) and the code respects
the required time‑budget, printing progress along the way.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities (no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max (returns a probability vector that sums to 1)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    s = e.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / z.size)
    return e / s


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑its that reproduce `a` after a soft‑max."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """Smallest power‑of‑two ≥ 3·n‑2 (enough for all required convolutions)."""
    return 1 << ((3 * n - 2).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Returns (max_conv, grad_a) where

        grad_a = ∇_a [ 2·n·smooth_max(a*a) ]

    `eps_factor` multiplies the smoothing constant ε = eps_factor·max_conv.
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv                     # ≤ 0 for stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # Gradient of 2·n·smooth_max (the factor 4·n comes from the derivative of a*a)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a


def _cheap_score(a: np.ndarray) -> float:
    """Score used during optimisation: 2·n·max(a*a)."""
    n = a.shape[0]
    return 2.0 * n * _conv_max(a)

def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}   (normalised)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _power_law_offset_distribution(n: int, p: float, offset: float) -> np.ndarray:
    """a_i ∝ (i+offset)^{-p}   (normalised)."""
    idx = np.arange(offset, offset + n, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _double_power_law_distribution(n: int, p: float, q: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}·(n-i)^{-q}   (normalised)."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) ** (-p)) * ((n - i) ** (-q))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _stretched_exp_distribution(n: int, tau_factor: float, gamma: float) -> np.ndarray:
    """
    a_i ∝ exp( - (i/(tau_factor·n))^{gamma} )   (normalised).
    """
    tau = max(1e-12, tau_factor * n)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(- (i / tau) ** gamma)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _exp_decay_distribution(n: int, lam: float) -> np.ndarray:
    """a_i ∝ e^{-lam·i}+e^{-lam·(n‑1‑i)}   (normalised)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _logistic_distribution(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ 1/(1+e^{α·(i‑mid)})   (normalised)."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = 1.0 / (1.0 + np.exp(alpha * (i - mid)))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _sym_product_power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ ((i+1)*(n‑i))^{-p}   (symmetric, normalised)."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) * (n - i)) ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _inv_sqrt_sum_distribution(n: int, p: float = 0.5, weight: float = 1.0) -> np.ndarray:
    """
    a_i ∝ (i+1)^{-p} + weight·(n-i)^{-p}   (normalised).
    For weight=1 the shape is symmetric; varying p allows us to
    explore the whole family around the “arcsine”‑type distribution.
    """
    i = np.arange(n, dtype=np.float64)
    left = (i + 1) ** (-p)
    right = (n - i) ** (-p)
    a = left + weight * right
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a

def _downscale(seq: np.ndarray, target_n: int) -> np.ndarray:
    """Linear interpolation from len(seq) → target_n (keeps total mass =1)."""
    L = seq.shape[0]
    if target_n == L:
        return seq.copy()
    idx = np.linspace(0, L - 1, target_n)
    small = np.interp(idx, np.arange(L), seq)
    small = np.maximum(small, 0.0)
    s = small.sum()
    if s == 0.0:
        small[:] = 1.0 / target_n
    else:
        small /= s
    return small


def _upscale(seq: np.ndarray, target_n: int) -> np.ndarray:
    """Linear interpolation from len(seq) → target_n (preserves sum)."""
    return _downscale(seq, target_n)

        # step‑size schedule (inverse‑sqrt decay)
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step = - m_hat / (np.sqrt(v_hat) + eps_adam)

        # simple back‑tracking line‑search (max 3 halvings)
        accepted = False
        trial_alpha = alpha_t
        for _ in range(3):
            new_logits = logits + trial_alpha * step
            new_logits = np.clip(new_logits, -50.0, 50.0)
            a_new = _softmax(new_logits)
            new_score = 2.0 * n_small * _conv_max(a_new)
            if new_score <= cheap:
                logits = new_logits
                accepted = True
                break
            trial_alpha *= 0.5
        if not accepted:
            logits += alpha_t * step
            logits = np.clip(logits, -50.0, 50.0)

        # occasional jitter to avoid plateaus
        if it % 5000 == 0 and np.random.rand() < 0.12:
            logits += 0.02 * np.random.randn(n_small)
            m *= 0.6
            v *= 0.6

        eps_factor = max(eps_factor_min, eps_factor * eps_decay)

        # restart if no improvement for a long while
        if it - last_improve > 300_000:
            print(f"[Adam‑small] Stagnation at iter {it} – restart")
            logits = _logits_from_distribution(best_small_vec) + 0.03 * np.random.randn(n_small)
            m.fill(0.0)
            v.fill(0.0)
            eps_factor = 0.15
            last_improve = it

    best_small = best_small_vec
    best_small_score = best_small_vec_score
    print(f"[Adam‑small] Finished after {it} iters, best cheap = {best_small_score:.6f}")

    # --------------------------------------------------------------
    # 4. Upscale to full length and brief refinement
    # --------------------------------------------------------------
    a_full = _upscale(best_small, full_n)

    # one short Adam run on the full size (≈10 k iterations)
    logits_full = _logits_from_distribution(a_full) + 0.02 * np.random.randn(full_n)

    m = np.zeros_like(logits_full)
    v = np.zeros_like(logits_full)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    base_alpha = 0.30
    eps_factor = 0.08
    eps_factor_min = 5e-5
    eps_decay = 0.9995

    it_full = 0
    best_full = a_full.copy()
    best_full_score = _cheap_score(best_full)

    # allocate the remaining time (≈40 %)
    while time.time() < deadline and it_full < 30_000:
        it_full += 1
        a = _softmax(logits_full)
        max_conv, grad_a = _conv_max_and_grad(a, eps_factor)
        cheap = 2.0 * full_n * max_conv

        if cheap < best_full_score - 1e-12:
            best_full_score = cheap
            best_full = a.copy()

        mu = np.dot(grad_a, a)
        grad_z = a * (grad_a - mu)

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it_full)
        v_hat = v / (1.0 - beta2 ** it_full)

        alpha_t = base_alpha / (math.sqrt(it_full) + 1.0)
        step = - m_hat / (np.sqrt(v_hat) + eps_adam)

        # line‑search (max 2 halvings)
        accepted = False
        trial_alpha = alpha_t
        for _ in range(2):
            new_logits = logits_full + trial_alpha * step
            new_logits = np.clip(new_logits, -50.0, 50.0)
            a_new = _softmax(new_logits)
            new_score = 2.0 * full_n * _conv_max(a_new)
            if new_score <= cheap:
                logits_full = new_logits
                accepted = True
                break
            trial_alpha *= 0.5
        if not accepted:
            logits_full += alpha_t * step
            logits_full = np.clip(logits_full, -50.0, 50.0)

        eps_factor = max(eps_factor_min, eps_factor * eps_decay)

    # --------------------------------------------------------------
    # 5. Final cheap random walk (coordinate moves)
    # --------------------------------------------------------------
    a = best_full.copy()
    cheap = best_full_score

    while time.time() < deadline:
        i = np.random.randint(0, full_n)
        j = np.random.randint(0, full_n)
        if i == j:
            continue
        delta = 0.001 * a[i]          # move 0.1 % of the source mass
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = 2.0 * full_n * _conv_max(a_new)
        if new_score < cheap:
            a = a_new
            cheap = new_score
            best_full = a.copy()
            best_full_score = cheap
            if np.random.rand() < 0.001:
                print(f"[Walk] improved cheap = {cheap:.6f}")

    # --------------------------------------------------------------
    # 6. Final exact evaluation
    # --------------------------------------------------------------
    final_seq = best_full.tolist()
    try:
        exact_score = evaluate_sequence(final_seq)  # supplied by the platform
        print(f"Final exact score = {exact_score:.6f}")
    except Exception as exc:
        exact_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    return final_seq
