        idx_base = len(w) - 1   # 2n-2

        logp = np.log(p + 1e-30) - eta * g
        p = np.exp(logp)
        s = p.sum()
        if s <= 0.0 or not np.isfinite(s):
            p = best_iter_vec.copy()
            continue
        p /= s

        # schedule alpha (increase)
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_growth)

        # restart if stagnated
        if iter_cnt - last_improve > 400000:
            perturb = np.random.rand(n) * 0.01
            p = _simplex_projection(best_iter_vec + perturb)
            last_improve = iter_cnt
            print(f"[Grad {iter_cnt:07d}] Restart after stagnation.")

    # --------------------------------------------------------------
    # Stage 2: permutation / small mass swaps local search
    # --------------------------------------------------------------
    p = best_iter_vec.copy()
    best_local_vec = p.copy()
    best_local_val = best_iter_val
    temperature = 0.8
    cooling = 0.99995
    attempts = 0

    while time.time() < deadline:
        # propose swap of positions i and j
        i, j = np.random.randint(0, n, size=2)
        if i == j: continue
        p[i], p[j] = p[j], p[i]
        conv = _conv_full_fft(p, fft_len)
        cheap = 2.0 * n * conv.max()
        accept = False
        if cheap < best_local_val - 1e-12:
            accept = True
        else:
            delta = cheap - best_local_val
            prob = math.exp(-delta / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True
        if accept:
            if cheap < best_local_val - 1e-12:
                best_local_val = cheap
                best_local_vec = p.copy()
                print(f"[Swap] New best cheap = {cheap:.6f}")
        else:
            # revert swap
            p[i], p[j] = p[j], p[i]

        temperature = max(temperature * cooling, 1e-6)
        attempts += 1

    # --------------------------------------------------------------
    # Stage 3: final evaluation
    # --------------------------------------------------------------
    final_seq = best_local_vec.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:
        final_score = float('inf')
        print(f"Error during final evaluation: {exc}")

    return final_seq

#!/usr/bin/env python
"""
Search for a non‑negative step function that minimises ``evaluate_sequence``.

Algorithm (≈ 3 stages)
----------------------
1. **Wide parametric sweep** – generate many families (uniform, exponential,
   symmetric exponential, power‑law, symmetric power‑law, beta‑pdf,
   random Dirichlet).  Keep the best normalised vector.
2. **Smoothed mirror descent** – minimise a smooth approximation of the
   max‑convolution:
        f(p) = (1/α)·log(∑_k exp(α·(p⋆p)_k)).
   The gradient of *f* is
        g_i = 2·∑_j w_{i+j}·p_j ,
   where  w are the soft‑max weights of the convolution.
   An exponentiated‑gradient (KL‑mirror) update followed by a simplex
   projection is applied.  The temperature α grows slowly so that the
   optimisation first sees a smooth landscape and later focuses on the true
   maximum.
3. **Permutation‑local search** – starting from the best mirror‑descent
   iterate, randomly swap two positions and accept the move either if it
   improves the cheap objective
        cheap(p)=2·n·max(p⋆p)
   or with a simulated‑annealing probability.  The best vector is finally
   evaluated with the exact ``evaluate_sequence`` routine.

All helper utilities are defined at module level (no closures).  The routine
respect the supplied time budget and always return a feasible sequence.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Small helper to compute the next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using a power‑of‑two FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa).real[: 2 * a.size - 1]
    return conv
