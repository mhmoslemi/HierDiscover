```
def project_to_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

```
import time, math, numpy as np

def _next_pow2(x):
    return 1 << ((x-1).bit_length())

def _conv_self(a, L=None):
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2*n-1)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[:2*n-1]
    return conv

def _grad_max_conv(a, max_inds, L2):
    n = a.shape[0]
    # indicator
    ind = np.zeros(2*n-1, dtype=np.float64)
    ind[max_inds] = 1.0
    # reversed a
    a_rev = a[::-1]
    a_rev_fft = np.fft.rfft(a_rev, L2)
    ind_fft = np.fft.rfft(ind, L2)
    conv2_full = np.fft.irfft(a_rev_fft * ind_fft, L2)
    # take slice
    start = n-1
    grad = 2.0 * conv2_full[start:start+n]
    return grad
```

"""
Sub‑gradient descent on the probability simplex.

We minimise the evaluation metric

    2·n·max(a⊛a) / (∑a)²   (the denominator is scale‑invariant,
    so we keep ∑a = 1).

The algorithm works with the exact (non‑smoothed) maximum of the
self‑convolution.  At each iteration we

1. compute the linear convolution a⊛a by FFT,
2. identify the index (or indices) where the convolution attains its
   maximum,
3. obtain a sub‑gradient of the max‑convolution:
      ∂max = 2·(a reversed) ⊛ 1_Kmax
   where 1_Kmax is the indicator of the maximising positions,
4. take a step opposite to the sub‑gradient,
5. project the result back onto the probability simplex
   (clip negatives, renormalise).

The step size decays as α_t = α₀ / sqrt(t) and we occasionally add a
tiny random jitter to escape plateaux.  The best vector seen so far is
returned.  The whole routine respects the 1000 s budget.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# --------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        S = {x ≥ 0, sum(x) = 1}
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    # Clip negatives first for numerical safety
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        # fallback to uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])
    # Normalise to sum‑1 (this is already the Euclidean projection
    # because all entries are non‑negative after clipping)
    return v / s


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear self‑convolution a⊛a using FFT of length L (power of two).
    Returns the first (2·len(a)‑1) entries.
    """
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)
    return conv[: 2 * a.shape[0] - 1]


def _grad_max_conv(a: np.ndarray, max_inds: np.ndarray, L: int) -> np.ndarray:
    """
    Sub‑gradient of max(a⊛a) w.r.t. a.
    Parameters
    ----------
    a         : np.ndarray, shape (n,)
    max_inds  : 1‑D integer array of indices where a⊛a attains its maximum
    L         : FFT length (≥ 3·n‑2)
    Returns
    -------
    grad : np.ndarray, shape (n,)
        The vector 2·(a_rev ⊛ 1_{max_inds}) restricted to the original
        length (first n entries after appropriate shift).
    """
    n = a.shape[0]
    # indicator of the active positions
    ind = np.zeros(2 * n - 1, dtype=np.float64)
    ind[max_inds] = 1.0

    # reversed a
    a_rev = a[::-1]

    a_rev_fft = np.fft.rfft(a_rev, L)
    ind_fft = np.fft.rfft(ind, L)

    # full cross‑correlation (length 3·n‑2)
    corr_full = np.fft.irfft(a_rev_fft * ind_fft, L)
    start = n - 1
    grad = 2.0 * corr_full[start : start + n]
    return grad


def _arcsine_initialisation(n: int) -> np.ndarray:
    """Exact area‑discretised arcsine distribution on n bins."""
    idx = np.arange(n, dtype=np.float64)
    x0 = idx / n
    x1 = (idx + 1) / n

    def arcsine_cdf(x):
        # clip to avoid domain errors at the boundaries
        return (2.0 / np.pi) * np.arcsin(np.sqrt(np.clip(x, 0.0, 1.0)))

    a = arcsine_cdf(x1) - arcsine_cdf(x0)
    a_sum = a.sum()
    if a_sum == 0.0 or np.isnan(a_sum):
        raise RuntimeError("Failed to initialise a valid distribution.")
    return a / a_sum
