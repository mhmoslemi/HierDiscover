def propose_candidate(seed=1234, budget_s=1000, **kwargs):
    import time, math, numpy as np

    start = time.time()
    deadline = start + budget_s - 2.0

    def softmax(z):
        # stable softmax
        m = np.max(z)
        e = np.exp(z - m)
        e_sum = e.sum()
        if e_sum == 0:
            return np.full_like(z, 1.0 / len(z))
        return e / e_sum

    def logits_from_distribution(a, eps=1e-15):
        return np.log(a + eps)

    def fft_len_for_corr(n):
        return 1 << ((3 * n - 2).bit_length())

    def conv_max(a):
        n = a.shape[0]
        L = fft_len_for_corr(n)
        a_fft = np.fft.rfft(a, L)
        conv = np.fft.irfft(a_fft * a_fft, L)[:2 * n - 1]
        return float(conv.max())

    def conv_max_and_grad(a, eps_factor):
        n = a.shape[0]
        L = fft_len_for_corr(n)
        a_fft = np.fft.rfft(a, L)
        conv = np.fft.irfft(a_fft * a_fft, L)[:2 * n - 1]
        max_conv = float(conv.max())
        eps = max(max_conv * eps_factor, 1e-12)
        shifted = conv - max_conv
        w = np.exp(shifted / eps)
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        w_fft = np.fft.rfft(w, L)
        corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]
        grad_a = 4.0 * n * corr
        return max_conv, grad_a

    # Some helper families
    def power_law(n, p):
        idx = np.arange(1, n + 1, dtype=np.float64)
        a = idx ** (-p)
        s = a.sum()
        if s == 0:
            a[:] = 1.0 / n
        else:
            a /= s
        return a

    def sym_product_power_law(n, p):
        i = np.arange(n, dtype=np.float64)
        a = ((i + 1) * (n - i)) ** (-p)
        s = a.sum()
        if s == 0:
            a[:] = 1.0 / n
        else:
            a /= s
        return a

    def exp_decay(n, lam):
        i = np.arange(n, dtype=np.float64)
        a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
        s = a.sum()
        if s == 0:
            a[:] = 1.0 / n
        else:
            a /= s
        return a

    def logistic(n, alpha):
        i = np.arange(n, dtype=np.float64)
        mid = (n - 1) / 2.0
        a = 1.0 / (1.0 + np.exp(alpha * (i - mid)))
        s = a.sum()
        if s == 0:
            a[:] = 1.0 / n
        else:
            a /= s
        return a

    def beta_family(n, a_param, b_param):
        # a,b >0
        x = np.linspace(0, 1, n)
        a = (x ** (a_param - 1)) * ((1 - x) ** (b_param - 1))
        s = a.sum()
        if s == 0:
            a[:] = 1.0 / n
        else:
            a /= s
        return a

    # Choose support size
    n0 = 5000
    # We'll permit upsampling later
    best_overall_seq = None
    best_overall_score = float('inf')
    n_support = n0

    # initial candidates list function
    init_candidates = []

    # add families param sweep
    # we can store tuples ('type', params)
    # but we may generate on the fly in loops
    # For simplicity we directly generate and evaluate within loops.

    # Determine a list of initial seeds for restarts
    restart_seeds = [seed + i * 12345 for i in range(5)]

    # We'll allocate each restart a fraction of time
    # compute approx remaining time after initialization and final evaluation
    # We'll just run restarts until time ends.

    try:
        init_global = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        # downsample to n_support
        old_n = init_global.shape[0]
        if old_n != n_support:
            old_x = np.linspace(0, 1, old_n)
            new_x = np.linspace(0, 1, n_support)
            a0 = np.interp(new_x, old_x, init_global)
        else:
            a0 = init_global.copy()
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s == 0:
            a0[:] = 1.0 / n_support
        else:
            a0 /= s
        init_candidates.append(a0)
        print("[Init] Global sequence added as candidate.")
    except Exception:
        pass

    # Add systematic families scanning some parameters that are cheap
    # We'll just generate a few param combos quickly
    # Parameter grids
    power_ps = [0.7, 0.9, 1.0, 1.2, 1.4]
    sym_ps = [0.7, 0.9, 1.0, 1.2, 1.4]
    lam_vals = [0.001, 0.005, 0.01, 0.02, 0.05]
    alpha_vals = [0.3, 0.6, 1.0, 2.0, 3.0]
    beta_params = [(0.5,0.5), (0.8,0.8), (1.2,1.2), (0.5,2.0), (2.0,0.5), (2.0,2.0)]

    for p in power_ps:
        init_candidates.append(power_law(n_support, p))
    for p in sym_ps:
        init_candidates.append(sym_product_power_law(n_support, p))
    for lam in lam_vals:
        init_candidates.append(exp_decay(n_support, lam))
    for alpha in alpha_vals:
        init_candidates.append(logistic(n_support, alpha))
    for a_param, b_param in beta_params:
        init_candidates.append(beta_family(n_support, a_param, b_param))

    # Add a few random Dirichlet candidates
    rng = np.random.default_rng(seed)
    for conc in [0.5, 1.0, 2.0]:
        a = rng.gamma(conc, size=n_support)
        a_sum = a.sum()
        if a_sum == 0:
            a[:] = 1.0 / n_support
        else:
            a /= a_sum
        init_candidates.append(a)

    # Evaluate cheap_score for each initial candidate and keep best
    for cand in init_candidates:
        maxc = conv_max(cand)
        cheap = 2.0 * n_support * maxc
        if cheap < best_overall_score:
            best_overall_score = cheap
            best_overall_seq = cand.copy()

    print(f"[Init] Best cheap score after scanning families: {best_overall_score:.6f}")

    # For each restart, run ADAM from a perturbation of best candidate or a fresh initial candidate.
    # We'll use a loop that stops when time runs out.

    # ADAM hyperparameters (common)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_alpha = 0.5
    eps_factor_start = 0.2
    eps_factor_min = 1e-5
    eps_factor_decay = 0.9995

    # Define function to run optimisation for a given initial distribution
    def optimise_from(a_init):
        nonlocal deadline
        n = a_init.shape[0]
        # use logits representation
        z = logits_from_distribution(a_init) + 0.01 * np.random.randn(n)
        m = np.zeros_like(z)
        v = np.zeros_like(z)
        eps_factor = eps_factor_start
        best_local_seq = softmax(z)
        max_conv, _ = conv_max_and_grad(best_local_seq, eps_factor)
        best_local_score = 2.0 * n * max_conv
        best_local_iter = 0
        it = 0
        last_improve = 0
        # iteration loop
        while time.time() < deadline and it < 2_000_000:
            it += 1
            a = softmax(z)
            max_conv, grad_a = conv_max_and_grad(a, eps_factor)
            cheap_score = 2.0 * n * max_conv
            if cheap_score < best_local_score - 1e-12:
                best_local_score = cheap_score
                best_local_seq = a.copy()
                last_improve = it
            # compute subgradient of the smooth max objective w.r.t logits
            mu = np.dot(grad_a, a)    # <grad, a>
            grad_z = a * (grad_a - mu)
            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            alpha_t = base_alpha / (math.sqrt(it) + 1.0)
            step = - m_hat / (np.sqrt(v_hat) + eps_adam)
            # backtracking line search (max 5 halves)
            accepted = False
            trial_alpha = alpha_t
            for _ in range(5):
                z_new = z + trial_alpha * step
                # clamp for numeric stability
                z_new = np.clip(z_new, -50, 50)
                a_new = softmax(z_new)
                new_score = 2.0 * n * conv_max(a_new)
                if new_score <= cheap_score:
                    z = z_new
                    accepted = True
                    break
                trial_alpha *= 0.5
            if not accepted:
                # fallback step
                z = z + alpha_t * step
                z = np.clip(z, -50, 50)

            # Occasionally jitter
            if (it % 2500 == 0) and (np.random.rand() < 0.12):
                z += 0.02 * np.random.randn(n)
                m *= 0.6
                v *= 0.6

            # Reduce smoothing
            eps_factor = max(eps_factor_min, eps_factor * eps_factor_decay)

            # Stagnation restart
            if it - last_improve > 250_000:
                # restart from best local
                z = logits_from_distribution(best_local_seq) + 0.03 * np.random.randn(n)
                m.fill(0.0)
                v.fill(0.0)
                eps_factor = eps_factor_start
                last_improve = it

            # Periodic upsampling if time permits and n < 15000
            if (it % 300_000 == 0) and (n < 15000) and (time.time() + 30.0 < deadline):
                new_n = min(15000, int(n * 1.5))
                print(f"[Upsample] {n} -> {new_n} at iter {it}")
                best_local_seq = np.interp(np.linspace(0, 1, new_n), np.linspace(0, 1, n), best_local_seq)
                best_local_seq = np.maximum(best_local_seq, 0.0)
                s = best_local_seq.sum()
                if s == 0:
                    best_local_seq[:] = 1.0 / new_n
                else:
                    best_local_seq /= s
                z = logits_from_distribution(best_local_seq) + 0.005 * np.random.randn(new_n)
                m = np.zeros_like(z)
                v = np.zeros_like(z)
                n = new_n
                # recompute best local score (no need exact)
                best_local_score = 2.0 * n * conv_max(best_local_seq)

        # return best sequence found and its cheap score
        return best_local_seq, best_local_score

    # Run restarts
    for rs in restart_seeds:
        if time.time() >= deadline:
            break
        np.random.seed(rs)
        # pick initial distribution: either best_overall_seq perturbed, or random candidate
        if best_overall_seq is not None and np.random.rand() < 0.6:
            a0 = best_overall_seq.copy()
            # add small jitter
            a0 = a0 + 0.01 * np.random.randn(n_support)
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()

                a0[:] = 1.0 / n_support
            else:
                a0 /= s
        else:
            # random Dirichlet
            conc = np.random.choice([0.5, 1.0, 2.0])
            a0 = rng.gamma(conc, size=n_support)
            s = a0.sum()
            if s == 0:

            else:
                a0 /= s

        seq, score = optimise_from(a0)
        print(f"[Restart {rs}] best cheap {score:.6f}")
        if score < best_overall_score:

            best_overall_seq = seq.copy()

        seq, score = optimise_from(best_overall_seq)
        print(f"[Final refinement] cheap {score:.6f}")
        if score < best_overall_score:
            best_overall_score = score
            best_overall_seq = seq.copy()

    # Prepare final sequence as list of floats
    final_seq = best_overall_seq.tolist() if best_overall_seq is not None else [1.0]

        print(f"Exact final score: {final_score:.6f}")
    except Exception as e:
        final_score = float('inf')

#!/usr/bin/env python
"""
Multi‑start mirror‑descent optimiser for the step‑function bound.

The routine builds a diverse pool of initial candidates (parametric families,
Beta‑distributions and random Dirichlet draws), then runs several independent
Adam‑style mirror‑descent runs on the probability simplex.  The objective
is the smoothed version of
    2·n·max(a*a)          (with Σa = 1)
and the smoothing parameter ε is gradually reduced.
During optimisation the algorithm occasionally injects jitter,
restarts when progress stalls, and performs a limited up‑sampling
to give the optimiser new degrees of freedom.  After the time budget
is exhausted the best distribution found is evaluated with the supplied
`evaluate_sequence` and returned.

The implementation is self‑contained, uses only NumPy and the standard
library, and respects the 1000 s runtime budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max, returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    s = e.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / z.size)
    return e / s


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Inverse of the softmax (log‑its) for a probability vector."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """Smallest power‑of‑two ≥ 3·n‑2 (enough for a full linear self‑convolution)."""
    return 1 << ((3 * n - 2).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Returns (max_conv, grad_a) where

        grad_a = ∇_a [ 2·n·smooth_max(a*a) ]

    The smooth max uses a soft‑max over the convolution entries with
    temperature ε = eps_factor·max_conv (clamped from below).
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    # FFT of a (zero‑padded)
    a_fft = np.fft.rfft(a, L)

    # Linear self‑convolution a*a
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # Smoothed weights (soft‑max)
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv    # ≤ 0
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # Correlation a ⋆ w (gradient of the smooth max)
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # Gradient of 2·n·smooth_max = 4·n·corr
    grad_a = 4.0 * n * corr
    return max_conv, grad_a


# ----------------------------------------------------------------------
# Parametric families – easy to generate diverse initial points
# ----------------------------------------------------------------------


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{‑p}, normalised to sum = 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _sym_product_power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ ((i+1)*(n‑i))^{‑p}, symmetric, normalised."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) * (n - i)) ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _exp_decay_distribution(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential: a_i ∝ e^{‑lam·i}+e^{‑lam·(n‑1‑i)}."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _logistic_distribution(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ 1/(1+e^{α·(i‑mid)}) – left‑heavy logistic."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = 1.0 / (1.0 + np.exp(alpha * (i - mid)))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _beta_distribution(n: int, a_param: float, b_param: float) -> np.ndarray:
    """Beta‑family on the discrete grid: a_i ∝ x^{a‑1}(1‑x)^{b‑1}."""
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = (x ** (a_param - 1.0)) * ((1.0 - x) ** (b_param - 1.0))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector onto a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    s = new_a.sum()
    if s == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= s
    return new_a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):

        z = _logits_from_distribution(a_start) + 0.01 * np.random.randn(n_loc)

        m = np.zeros_like(z)
        v = np.zeros_like(z)
        beta1, beta2 = 0.9, 0.999
        eps_adam = 1e-8

        base_alpha = 0.45            # base learning‑rate for logits
        eps_factor = 0.15            # initial smoothing factor
        eps_factor_min = 5e-5
        eps_decay = 0.9992

        best_local_score = 2.0 * n_loc * _conv_max(best_local)
        last_improve = 0

    # ------------------------------------------------------------------
    # 2.  Run several independent restarts (different seeds / initialisation)
    # ------------------------------------------------------------------
    restart_seeds = [seed + 12345 * i for i in range(5)]
