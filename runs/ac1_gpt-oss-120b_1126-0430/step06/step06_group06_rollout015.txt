#!/usr/bin/env python
"""
A fast “smooth‑max + Adam” optimiser for the step‑function problem.

The idea:
* Work with the log‑softmax parametrisation
      a_i = exp(z_i) / Σ_j exp(z_j)
  which guarantees non‑negativity and Σ a_i = 1.
* Replace the non‑smooth max   max_k (a*a)[k]   by a smooth
  approximation
      L_T(a) = (1/T)·log Σ_k exp(T·(a*a)[k])          (T – temperature)
  When T → ∞, L_T → max .
* The gradient of L_T w.r.t. the logits z can be written using a single
  additional FFT, so every optimisation step costs only two FFTs.
* Adam (adaptive moment estimation) on the logits gives a fast, stable
  descent.  T is increased slowly so that the optimisation starts
  smooth and finishes with an almost exact max‑objective.
* A cheap random‑parametric scan (power‑law, exponential, beta …) is
  performed first to obtain a good starting point.
* After the Adam phase a tiny (1+1)‑ES fine‑tuner is run – it is cheap
  and often squeezes a few extra digits.

The routine respects the 1000 s budget, prints progress and finally
returns a list of heights that minimises ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – no closures, no lambdas
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Small helper: next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:               # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full_fft(a: np.ndarray, b: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution using a real FFT (full length)."""
    fa = np.fft.rfft(a, fft_len)
    fb = np.fft.rfft(b, fft_len)
    conv = np.fft.irfft(fa * fb, fft_len).real
    return conv[: a.size + b.size - 1]


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Convolution a*a (full) – fast shortcut."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len).real
    return conv[: 2 * a.size - 1]


def _cheap_score(a: np.ndarray, n: int, fft_len: int) -> float:
    """Cheap objective 2·n·max(a*a) – assumes Σa = 1."""
    conv = _conv_self_fft(a, fft_len)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Parametric families (all return a *normalised* probability vector)
# ----------------------------------------------------------------------


def _uniform_seq(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _power_law_seq(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = i ** (-p)
    seq /= seq.sum()
    return seq


def _exponential_seq(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    seq = np.exp(-lam * i)
    seq /= seq.sum()
    return seq


def _power_law_exp_seq(n: int, p: float, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    seq = (i + 1.0) ** (-p) * np.exp(-lam * i)
    seq /= seq.sum()
    return seq


def _beta_seq(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(1, n + 1, dtype=np.float64) / n)
    seq = (x ** alpha) * ((1.0 - x) ** beta)
    s = seq.sum()
    if s == 0.0:
        return None
    seq /= s
    return seq


def _stretched_exponential_seq(n: int, lam: float, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    seq = np.exp(-lam * (i ** gamma))
    seq /= seq.sum()
    return seq


def _dirichlet_seq(n: int, conc: float) -> np.ndarray:
    g = np.random.gamma(conc, 1.0, size=n)
    seq = g / g.sum()
    return seq


# ----------------------------------------------------------------------
# Core optimisation – smooth‑max + Adam
# ----------------------------------------------------------------------


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    fft_len1: int,
    fft_len2: int,
    deadline: float,
) -> np.ndarray:
    """
    Adam optimisation on the logits of a probability vector.
    Returns the best vector seen (in cheap score).
    """
    # hyper‑parameters
    lr = 0.07               # base learning rate
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    # temperature schedule for smooth‑max
    T = 5.0                 # start temperature
    T_mult = 1.001          # multiplicative increase per iteration
    T_max = 2000.0

    # initialise Adam state
    z = np.log(a0 + 1e-30)      # logits (any constant shift is irrelevant)
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    best_a = a0.copy()
    best_val = _cheap_score(best_a, n, fft_len1)
    best_iter = 0
    it = 0

    while time.time() < deadline:
        it += 1

        # ---- softmax to obtain a (numerically stable) ----
        z_shift = z - np.max(z)       # shift for stability
        expz = np.exp(z_shift)
        a = expz / expz.sum()          # Σ a = 1, a_i > 0

        # ---- convolution a*a (full) ----
        conv = _conv_self_fft(a, fft_len1)   # length 2n‑1
        maxc = conv.max()
        cheap = 2.0 * n * maxc

        # keep the best seen vector
        if cheap < best_val - 1e-12:
            best_val = cheap
            best_a = a.copy()
            best_iter = it
            print(f"[Adam {it:07d}] NEW BEST cheap = {cheap:.6f}   T={T:.2f}")

        # ---- smooth‑max weights (softmax over conv) ----
        # use stable log‑sum‑exp
        shifted = conv - maxc                      # ≤0
        exp_terms = np.exp(T * shifted)
        w = exp_terms / exp_terms.sum()            # Σ w = 1

        # ---- gradient of smooth‑max wrt a (via a single FFT) ----
        w_rev = w[::-1]                            # length 2n‑1
        conv2 = _conv_full_fft(a, w_rev, fft_len2)  # length 3n‑2
        # slice the part that corresponds to indices (2n‑2‑i)
        slice_part = conv2[n - 1 : 2 * n - 1]       # length n
        g = 2.0 * slice_part[::-1]                 # gradient ∂L/∂a

        # ---- transform gradient to logits (softmax Jacobian) ----
        avg = np.dot(a, g)
        grad_z = a * (g - avg)                     # ∂L/∂z

        # ---- Adam update on logits ----
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        z -= step

        # ---- temperature increase (smooth → hard) ----
        T = min(T_max, T * T_mult)

        # occasional random perturbation if stuck for a while
        if it - best_iter > 120_000:
            # inject a tiny Gaussian jitter into logits
            jitter = 0.01 * np.random.randn(n)
            z += jitter
            best_iter = it
            print(f"[Adam {it:07d}] Restart jitter after stagnation.")

    return best_a


def _es_hill_climb(a0: np.ndarray, n: int, fft_len: int, deadline: float) -> np.ndarray:
    """
    Very cheap (1+1)‑ES on the cheap objective, starting from a0.
    Returns the best vector found.
    """
    sigma = 0.09          # mutation strength (log‑space)
    fail = 0
    best_a = a0.copy()
    best_val = _cheap_score(best_a, n, fft_len)

    while time.time() < deadline:
        # multiplicative Gaussian perturbation in log‑space
        cand = best_a * np.exp(sigma * np.random.randn(n))
        cand = _simplex_projection(cand)

        val = _cheap_score(cand, n, fft_len)
        if val < best_val:
            best_val = val
            best_a = cand
            sigma = max(sigma * 0.85, 1e-5)
            fail = 0
            print(f"[ES] improved cheap = {val:.6f}  sigma={sigma:.5f}")
        else:
            fail += 1
            if fail > 300:
                sigma = min(sigma * 1.35, 0.6)
                fail = 0

        # occasional small random restart to avoid dead‑ends
        if np.random.rand() < 0.0005:
            pert = _dirichlet_seq(n, np.random.uniform(0.001, 0.5))
            cand = _simplex_projection(best_a + 0.015 * pert)
            cand_val = _cheap_score(cand, n, fft_len)
            if cand_val < best_val:
                best_val = cand_val
                best_a = cand
                print(f"[ES] restart gave cheap = {cand_val:.6f}")

    return best_a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that
    minimises ``evaluate_sequence``.  The routine adapts its schedule to
    the available time budget and always returns a feasible sequence.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a few seconds for the final exact evaluation
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # Choose a working length.
    # --------------------------------------------------------------
    try:
        prev = globals()["height_sequence_1"]
        prev_len = len(prev)
    except Exception:
        prev_len = 0

    # we like a longer vector than the previous 5k – gives more freedom
    n = min(max(prev_len, 15000), 25000)           # 15 k ≤ n ≤ 25 k
    fft_len1 = _next_pow2(2 * n - 1)               # for a*a
    fft_len2 = _next_pow2(3 * n - 2)               # for a * w_rev

    # --------------------------------------------------------------
    # Phase 0 – cheap random‑parametric scan (cheapest possible)
    # --------------------------------------------------------------
    best_a = _uniform_seq(n)
    best_val = _cheap_score(best_a, n, fft_len1)

    # try a handful of hand‑crafted families
    families = []

    # power‑law (decreasing)
    families.append(("power", lambda: _power_law_seq(n, np.random.uniform(0.1, 5.0))))

    # exponential
    families.append(("exp", lambda: _exponential_seq(n, np.random.uniform(0.0, 5.0))))

    # power‑law × exponential
    families.append(("pl_exp", lambda: _power_law_exp_seq(
        n,
        np.random.uniform(0.1, 5.0),
        np.random.uniform(0.0, 3.0),
    )))

    # stretched exponential
    families.append(("stretched", lambda: _stretched_exponential_seq(
        n,
        np.random.uniform(0.0, 4.0),
        np.random.choice([0.5, 0.9, 1.2, 1.5, 2.0]),
    )))

    # beta–shaped
    families.append(("beta", lambda: _beta_seq(
        n,
        np.random.uniform(0.2, 8.0),
        np.random.uniform(0.2, 8.0),
    )))

    # Dirichlet with very small concentration (sparse seeds)
    families.append(("dir_small", lambda: _dirichlet_seq(
        n,
        np.random.choice([0.001, 0.005, 0.01, 0.05, 0.1]),
    )))

    # ---- sampling loop ------------------------------------------------
    for i in range(500):
        if time.time() >= deadline:
            break
        kind, maker = families[i % len(families)]
        cand = maker()
        if cand is None:
            continue
        val = _cheap_score(cand, n, fft_len1)
        if val < best_val:
            best_val = val
            best_a = cand
            print(f"[Init {i:03d}] {kind} improved cheap = {val:.6f}")

    print(f"[Init] Best cheap after sweep = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – Adam on the smooth‑max surrogate
    # --------------------------------------------------------------
    a_adam = _adam_optimize(best_a, n, fft_len1, fft_len2, deadline - 2.0)

    # --------------------------------------------------------------
    # Phase 2 – tiny (1+1)‑ES hill‑climb on the true cheap score
    # --------------------------------------------------------------
    a_es = _es_hill_climb(a_adam, n, fft_len1, deadline - 2.0)

    # --------------------------------------------------------------
    # Phase 3 – final exact evaluation (for logging only)
    # --------------------------------------------------------------
    final_seq = a_es.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:      # defensive – should never happen
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # --------------------------------------------------------------
    # Return the best sequence found.
    # --------------------------------------------------------------
    return final_seq
