"""
Hybrid optimizer for the self‑convolution bound.

The pipeline consists of three phases:

1. **Broad parametric sweep** – a handful of analytic families
   (power‑law, offset power‑law, double power‑law, stretched exponential,
   exponential, logistic, symmetric product‑power‑law and a Beta family)
   are sampled on a coarse grid.  The cheap objective
   ``2·n·max(conv)`` (where ``conv = a*a``) is evaluated using an FFT
   convolution and the best distribution is kept.

2. **Exact sub‑gradient Adam on the simplex** – starting from the best
   distribution the true (non‑smoothed) objective is minimised.
   For a given distribution the index ``k*`` where the self‑convolution
   attains its maximum is found; the sub‑gradient of ``max(conv)`` is
   analytically
   ``g_i = 2·a_{k*‑i}`` (zero outside the support).  The gradient on the
   probability simplex is obtained through the mirror‑descent
   (exponentiated‑gradient) transformation
   ``z ← softmax^{-1}(a)``.  Adam with a simple back‑tracking line‑search
   supplies monotone improvement.

3. **Cheap random walk** – any remaining time is spent on tiny
   coordinate moves (transfer of a small mass between two indices) that
   are accepted only when they improve the exact objective.

The routine respects a ~1000 s wall‑time budget, prints progress
messages and finally returns the best sequence as a plain list of
floats.  ``evaluate_sequence`` (provided by the platform) is called once
at the very end to obtain the official score.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic helpers (no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max, returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    s = e.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / z.size)
    return e / s


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑its that reproduce ``a`` after a soft‑max."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """Smallest power‑of‑two ≥ 3·n‑2 (enough for all needed convolutions)."""
    return 1 << ((3 * n - 2).bit_length())


def _conv_full(a: np.ndarray) -> np.ndarray:
    """Full linear self–convolution a*a (length 2·n‑1) via FFT."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return conv


def _conv_max_and_argmax(a: np.ndarray):
    """Return (max(conv), argmax(conv)) for the self‑convolution."""
    conv = _conv_full(a)
    max_val = float(conv.max())
    # break ties randomly to obtain a sub‑gradient
    max_idxs = np.flatnonzero(conv == max_val)
    arg = int(np.random.choice(max_idxs))
    return max_val, arg


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a (cheap version)."""
    conv = _conv_full(a)
    return float(conv.max())

# Parametric families (generators return a *probability* vector, sum≈1)
# ----------------------------------------------------------------------


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _power_law_offset_distribution(n: int, p: float, offset: float) -> np.ndarray:
    """a_i ∝ (i+offset)^{-p}."""
    idx = np.arange(offset, offset + n, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _double_power_law_distribution(n: int, p: float, q: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}·(n-i)^{-q}."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) ** (-p)) * ((n - i) ** (-q))
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _stretched_exp_distribution(n: int, tau_factor: float, gamma: float) -> np.ndarray:
    """a_i ∝ exp( - (i/(tau_factor·n))^{gamma} )."""
    tau = max(1e-12, tau_factor * n)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(- (i / tau) ** gamma)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _exp_decay_distribution(n: int, lam: float) -> np.ndarray:
    """a_i ∝ e^{-lam·i}+e^{-lam·(n‑1‑i)}."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _logistic_distribution(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ 1/(1+e^{α·(i‑mid)})."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = 1.0 / (1.0 + np.exp(alpha * (i - mid)))
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _sym_product_power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ ((i+1)*(n‑i))^{-p}."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) * (n - i)) ** (-p)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discrete Beta(α,β) law on {0,…,n‑1}:
    a_i ∝ (i/(n‑1))^{α‑1}·(1‑i/(n‑1))^{β‑1}.
    Handles the singularities at the boundaries via a tiny ε.
    """
    eps = 1e-12
    x = np.arange(n, dtype=np.float64) / max(1, n - 1)
    x = np.clip(x, eps, 1 - eps)               # avoid exact 0 or 1 when α<1 or β<1
    a = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)
