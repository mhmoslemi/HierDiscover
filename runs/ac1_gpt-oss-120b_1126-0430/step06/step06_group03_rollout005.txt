"""
Hybrid optimisation for the self‑convolution bound (new version).

The routine consists of four stages:

1. **Initialisation** – obtain a starting probability vector.
   If a global `height_sequence_1` is provided we up‑sample it,
   otherwise we start from a uniform distribution of length `N`.

2. **Broad parametric sweep** – a cheap exhaustive search over several
   analytic families (power‑law, double‑power‑law, stretched‑exponential,
   etc.) with extended parameter ranges.  The cheap score
   `2·N·max(a*a)` is used to keep the best candidate.

3. **Mirror‑descent on the simplex (Adam + exact sub‑gradient)** – the
   current best vector is refined by exponentiated‑gradient (Adam) updates
   on the logits of `softmax`.  The gradient of the max‑convolution is
   obtained *exactly* as a sub‑gradient (no soft‑max smoothing).  A
   back‑tracking line‑search guarantees monotone improvement of the cheap
   score.  Restarts are performed if progress stalls.

4. **Final cheap random walk** – coordinate moves that transfer a tiny
   amount of mass between two entries are accepted only when they improve
   the cheap score.

At the end the best vector (a list of non‑negative heights) is returned
and evaluated with the provided `evaluate_sequence`.  The whole search
respects a time budget of roughly 1000 s.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max, returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    s = e.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / z.size)
    return e / s


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Return log‑it such that `softmax(logits) == a` (up to numerical error)."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """Smallest power‑of‑two ≥ 3·n‑2 (sufficient for all required convolutions)."""
    return 1 << ((3 * n - 2).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _max_conv_and_subgrad(a: np.ndarray):
    """
    Exact max‑convolution and a sub‑gradient of `max_k (a*a)_k`.

    Returns
    -------
    max_val : float
        The maximal entry of the full convolution a*a.
    grad : np.ndarray
        Sub‑gradient of the max‑convolution with respect to a
        (same size as a).  The gradient corresponds to the average of the
        gradients of all active constraints (indices where the maximum is
        attained).
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    # --- convolution a * a -------------------------------------------------
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_val = float(conv.max())

    # --- active set --------------------------------------------------------
    # tolerance small enough to capture ties arising from floating point noise
    tol = max(max_val * 1e-12, 1e-15)
    active = conv >= max_val - tol
    cnt = active.sum()
    if cnt == 0:
        # fallback – should never happen, but keep algorithm safe
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        cnt = 1
    else:
        w = np.empty_like(conv, dtype=np.float64)
        w.fill(0.0)
        w[active] = 1.0 / cnt  # normalized weight for each active index

    # --- sub‑gradient (average of gradients of active constraints) ---------
    # Gradient of (a*a)_k w.r.t a_i is 2 * a_{k-i} (zero outside the support)
    # So averaged gradient = 2 * sum_k w_k * a_{k-i}
    # This is a convolution of a with the reversed weight vector.
    w_rev = w[::-1]                     # reverse the weight vector
    w_fft = np.fft.rfft(w_rev, L)       # FFT of the reversed weight
    grad_full = np.fft.irfft(a_fft * w_fft, L)[: n + (2 * n - 1) - 1]  # length 3n‑2
    centre = (2 * n - 2)                # index of the zero‑shift in the full conv
    grad = 2.0 * grad_full[centre - np.arange(n)]

    return max_val, grad

# Parametric families (analytic shapes)

def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}   (normalised)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _power_law_offset_distribution(n: int, p: float, offset: float) -> np.ndarray:
    """a_i ∝ (i+offset)^{-p}   (normalised)."""
    idx = np.arange(offset, offset + n, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _double_power_law_distribution(n: int, p: float, q: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}·(n-i)^{-q}   (normalised)."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) ** (-p)) * ((n - i) ** (-q))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _stretched_exp_distribution(n: int, tau_factor: float, gamma: float) -> np.ndarray:
    """
    a_i ∝ exp( - (i/(tau_factor·n))^{gamma} )   (normalised).
    """
    tau = max(1e-12, tau_factor * n)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(- (i / tau) ** gamma)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _exp_decay_distribution(n: int, lam: float) -> np.ndarray:
    """a_i ∝ e^{-lam·i}+e^{-lam·(n‑1‑i)}   (normalised)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _logistic_distribution(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ 1/(1+e^{α·(i‑mid)})   (normalised)."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = 1.0 / (1.0 + np.exp(alpha * (i - mid)))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _sym_product_power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ ((i+1)*(n‑i))^{-p}   (symmetric, normalised)."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) * (n - i)) ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a

def _upsample_to_length(seq: np.ndarray, target_len: int) -> np.ndarray:
    """
    Linearly up‑sample `seq` to `target_len` entries and renormalise.
    """
    orig_len = seq.shape[0]
    if orig_len == target_len:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, orig_len)
    x_new = np.linspace(0.0, 1.0, target_len)
    up = np.interp(x_new, x_old, seq)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s == 0.0:
        up[:] = 1.0 / target_len
    else:
        up /= s
    return up

            a_new = _softmax(new_logits)
            new_score = 2.0 * n * _conv_max(a_new)
            if new_score <= cheap_score:
                logits = new_logits
                accepted = True
                break
            trial_alpha *= 0.5
        if not accepted:
            logits = np.clip(logits + alpha_t * step, -50.0, 50.0)

        # occasional random jitter to shake off flat plateaus
        if it % 4000 == 0 and np.random.rand() < 0.20:
            logits += 0.02 * np.random.randn(n)
            m *= 0.6
            v *= 0.6

        # restart if no improvement for a long while
        if it - last_improve > stagnation_patience:
            print(f"[Iter {it}] Stagnation – restarting from current best")
            logits = _logits_from_distribution(best_seq_sofar) + 0.03 * np.random.randn(n)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = it

    best_seq = best_seq_sofar
    best_score = best_score_sofar
    print(f"[Adam] Finished after {it} iterations, best cheap score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Final cheap random walk (tiny coordinate moves)
    # ------------------------------------------------------------------
    a = best_seq.copy()
    cheap_score = best_score

    while time.time() < deadline:
        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        if i == j:
            continue
        delta = 0.001 * a[i]                 # move 0.1 % of mass from i to j
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = 2.0 * n * _conv_max(a_new)
        if new_score < cheap_score:
            a = a_new
            cheap_score = new_score
            best_seq = a.copy()
            best_score = cheap_score
            # very sparse progress reports
            if np.random.rand() < 0.001:
                print(f"[Walk] Improved cheap = {cheap_score:.8f}")

    # ------------------------------------------------------------------
    # 4. Final exact evaluation
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        exact_score = evaluate_sequence(final_seq)  # supplied by the platform
        print(f"Final exact score = {exact_score:.6f}")
    except Exception as exc:
        exact_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    return final_seq
