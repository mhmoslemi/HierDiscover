    i = np.arange(n, dtype=np.float64)
    x0 = i / n
    x1 = (i + 1) / n
    F0 = (2.0/np.pi) * np.arcsin(np.sqrt(np.clip(x0, 0.0, 1.0)))

#!/usr/bin/env python
"""
Optimise a non‑negative step function for the max‑convolution metric.

The algorithm
---------------
* Initialise a probability vector from the exact discretisation of the
  arcsine law (the known asymptotically optimal shape).
* Parametrise the simplex by soft‑max logits and run a long Adam
  optimisation on the objective  2·n·max(a*a) / (sum a)² .
* The gradient is obtained by a smoothed soft‑max over the convolution
  entries; when the smoothing factor becomes very small we switch to the
  exact sub‑gradient (uniform weight on all maximal indices).
* Learning‑rate, smoothing factor and occasional Gaussian jitter are
  scheduled adaptively.  The best vector seen so far is kept and returned.
* The search runs until the time budget (≈ 1000 s) expires or a target
  score (≈ 1.503) is reached.

The implementation avoids closures, uses only numpy (FFT based
convolution), and works for vectors of up to a few × 10⁴ entries within
the allotted time.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max yielding a probability vector (sum = 1)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Logits that map to a via soft‑max (adds a tiny epsilon for safety)."""
    return np.log(a + eps)


def _fft_len_for_conv(n: int) -> int:
    """
    Smallest power‑of‑two ≥ 2·n‑1, sufficient for a linear self‑convolution.
    """
    return 1 << ((2 * n - 1).bit_length())


# ----------------------------------------------------------------------
# Convolution helpers
# ----------------------------------------------------------------------


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a (FFT, no gradient)."""
    n = a.shape[0]
    L = _fft_len_for_conv(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    # small negative artefacts are harmless – max will ignore them
    return float(conv.max())


def _conv_max_and_grad(
    a: np.ndarray, eps_factor: float, *, exact: bool = False
) -> tuple[float, np.ndarray]:
    """
    Compute (max conv, gradient w.r.t. a) for the smoothed (or exact) max.

    Parameters
    ----------
    a : np.ndarray
        Probability vector (non‑negative, sum≈1).
    eps_factor : float
        Multiplicative factor for the smoothing constant (ignored if exact).
    exact : bool
        If True, use the exact sub‑gradient (uniform weight on all maximisers).

    Returns
    -------
    max_conv : float
        Maximum entry of the linear self‑convolution.
    grad_a : np.ndarray
        Gradient of 2·n·max_conv w.r.t. a (already multiplied by the factor 2·n).
    """
    n = a.shape[0]
    L = _fft_len_for_conv(n)

    a_fft = np.fft.rfft(a, L)

    # linear self‑convolution (length 2·n‑1)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    if exact:
        # uniform weight on every index attaining the maximum (within tiny tol)
        tol = max(1e-14 * max_conv, 1e-18)
        mask = (conv >= max_conv - tol).astype(np.float64)
        w = mask / mask.sum()
    else:
        # smoothed soft‑max weights
        eps = max(max_conv * eps_factor, 1e-12)
        shifted = conv - max_conv  # for numerical stability
        w = np.exp(shifted / eps)
        w_sum = w.sum()
        if w_sum == 0.0 or not np.isfinite(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

    # cross‑correlation a ⋆ w (linear, length n)
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # Gradient of 2·n·max_conv (the factor 2 appears because d(conv_k)/da_i = 2*a_{k-i}
    # except on the diagonal where it is 1; using correlation captures the 2·a term.
    grad_a = 2.0 * n * corr * 2.0  # = 4·n·corr
    return max_conv, grad_a


# ----------------------------------------------------------------------
# Problem‑specific initialisation
# ----------------------------------------------------------------------


def _arcsine_bins(n: int) -> np.ndarray:
    """
    Exact discretisation of the arcsine law on a uniform grid of size n.
    Returns a probability vector (sum = 1).
    """
    i = np.arange(n, dtype=np.float64)
    x0 = i / n
    x1 = (i + 1) / n
    # CDF: F(x) = (2/π)·arcsin(√x)
    F0 = (2.0 / np.pi) * np.arcsin(np.sqrt(np.clip(x0, 0.0, 1.0)))
    F1 = (2.0 / np.pi) * np.arcsin(np.sqrt(np.clip(x1, 0.0, 1.0)))
    a = F1 - F0
    total = a.sum()
    if not np.isfinite(total) or total <= 0.0:
        raise RuntimeError("Failed to generate arcsine bins.")
    a /= total
    return a

        # Inverse‑sqrt step‑size schedule (slightly decreasing)
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step = -alpha_t * m_hat / (np.sqrt(v_hat) + eps_adam)

        z = z + step
        # Keep logits in a safe range to avoid overflow in exp
        z = np.clip(z, -60.0, 60.0)

        # Decay the smoothing factor (cannot go below eps_factor_min)
        eps_factor = max(eps_factor_min, eps_factor * eps_decay)

        # Occasional random jitter to escape plateaus
        if it % 5000 == 0 and np.random.rand() < jitter_prob:
            z += jitter_scale * np.random.randn(n)

        # Periodic cheap logging (without cluttering output)
        if it % 10000 == 0:
            elapsed = time.time() - start_time
            print(
                f"[{elapsed:5.1f}s] iter {it:6d}, score {score:.8f}, "
                f"best {best_score:.8f}"
            )

    # ------------------------------------------------------------------
    # Final exact evaluation (for logging only)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        exact_score = evaluate_sequence(final_seq)
        print(f"Exact evaluation (after optimisation): {exact_score:.8f}")
    except Exception as exc:
        print(f"Exact evaluation failed: {exc}")

    return final_seq
