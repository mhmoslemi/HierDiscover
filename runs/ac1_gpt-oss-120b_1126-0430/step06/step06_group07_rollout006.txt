#!/usr/bin/env python
"""
Search for a non‑negative step function (list of heights) that minimises
``evaluate_sequence``.  The algorithm consists of three stages:

1. **Broad initial sweep** – generate a handful of parametric families
   (uniform, exponential, power‑law, Dirichlet) and keep the best
   vector (the cheap score assumes Σa_i = 1, i.e. 2·n·max(conv)).
2. **Smoothed mirror descent** – minimise a smooth approximation of the
   max‑convolution using an exponentiated‑gradient update on the simplex.
   The smooth objective is a soft‑max (log‑sum‑exp) of the convolution
   entries; its gradient is computed efficiently by FFT‑based
   correlation.
3. **Local improvement** – simulated‑annealing style random swaps of
   positions combined with small continuous perturbations, always
   projected back onto the simplex.  The best vector found is finally
   evaluated with the true ``evaluate_sequence`` function.

The routine respects the supplied time budget (default 1000 s) and
always returns a feasible sequence.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using a power‑of‑two FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa).real[: 2 * a.size - 1]
    return conv


def _conv_max_fft(a: np.ndarray, fft_len: int) -> float:
    """Maximum value of the autocorrelation a*a."""
    conv = _conv_full_fft(a, fft_len)
    return float(conv.max())


def _softmax_weights(conv: np.ndarray, eps: float) -> np.ndarray:
    """Stable soft‑max over the convolution entries."""
    # shift by the maximum for numerical stability
    shift = conv.max()
    exps = np.exp((conv - shift) / eps)
    s = exps.sum()
    if s == 0.0 or not np.isfinite(s):
        return np.full_like(conv, 1.0 / conv.size)
    return exps / s


def _gradient_softmax(a: np.ndarray, conv: np.ndarray, eps: float,
                     fft_len: int) -> np.ndarray:
    """
    Compute the gradient of the soft‑max (log‑sum‑exp) of the convolution
    with respect to a.  Uses FFT‑based correlation for O(N log N) time.
    """
    n = a.size
    w = _softmax_weights(conv, eps)          # length 2n‑1
    # correlation a ⋆ w  (c[i] = Σ_j a_j w[i+j]) → gradient = 2·c[i]
    w_rev = w[::-1]                          # reverse for convolution
    FA = np.fft.fft(a, fft_len)
    FW = np.fft.fft(w_rev, fft_len)
    r = np.fft.ifft(FA * FW).real[: 3 * n - 2]   # linear convolution a * w_rev
    # slice indices n‑1 … 2n‑2, then reverse to obtain c[0…n‑1]
    corr = r[n - 1 : 2 * n - 1][::-1]
    grad = 2.0 * corr
    return grad

# Core search routine required by the platform
# ----------------------------------------------------------------------

                best_swap_val = cheap
                best_swap_a = a.copy()
                print(f"[Swap] NEW BEST cheap = {cheap:.6f}")
        else:
            a = old_a  # revert

        temperature = max(temperature * cooling, 1e-7)

    # --------------------------------------------------------------
    # 3. Final exact evaluation
    # --------------------------------------------------------------
    final_seq = best_swap_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    return final_seq
