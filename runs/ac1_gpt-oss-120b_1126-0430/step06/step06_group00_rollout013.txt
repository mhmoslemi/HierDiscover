"""
Step‑function optimizer for the metric

    2·n·max(a*a) / (sum(a)**2)

The code works in three stages:

1. **Initialisation** – reuse a previously supplied sequence if available,
   otherwise build the exact area‑discretised arcsine distribution
   (the empirically good continuous optimum).

2. **Lightweight parametric sweep** – try a few simple families
   (power‑law shapes, mixtures with the uniform distribution and small
   moving‑average smoothings).  The best candidate from this cheap search
   is taken as the starting point for the main optimisation.

3. **Adam on the simplex** – the objective is 2·n·max(conv) with the sum
   normalised to 1, so we minimise ``max(conv)``.  The gradient of the
   max‑convolution is obtained exactly (a sub‑gradient):
   for the index `k` attaining the maximum we have
   ∂max(conv)/∂a_i = 2·a_{k‑i} (zero when the index is out of range).
   The gradient on the logits (soft‑max parametrisation) is the natural
   gradient on the simplex.  Adam with a diminishing step‑size refines the
   sequence for the remaining time budget.

The optimiser strives to push the bound below the target 1.503.
All helper functions are top‑level, no closures or lambdas are used.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max; returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _fft_len_for_corr(n: int) -> int:
    """Smallest power of two ≥ 3·n‑2 (sufficient for self‑convolution)."""
    return 1 << ((3 * n - 2).bit_length())


def _full_conv(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (full length 2·n‑1)."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return conv


def _maxconv_and_subgrad(a: np.ndarray):
    """
    Returns (max_conv, grad) where `max_conv` is the maximum entry of
    the linear self‑convolution a*a and `grad` is a sub‑gradient of that
    maximum with respect to `a` (i.e. ∂max/∂a_i).
    """
    conv = _full_conv(a)
    k = int(np.argmax(conv))                # index attaining the max
    max_val = float(conv[k])

    n = a.shape[0]
    grad = np.zeros_like(a)

    if k < n:
        # indices 0..k contribute
        grad[: k + 1] = 2.0 * a[k::-1]
    else:
        # indices k-(n-1)..n-1 contribute
        start = k - (n - 1)
        grad[start:] = 2.0 * a[start:][::-1]

    return max_val, grad


def arcsine_cdf(x: np.ndarray) -> np.ndarray:
    """Arcsine CDF on [0,1]: F(x) = (2/π)·arcsin(√x)."""
    return (2.0 / np.pi) * np.arcsin(np.sqrt(np.clip(x, 0.0, 1.0)))


def _init_arcsine(n: int) -> np.ndarray:
    """Exact area‑discretised arcsine distribution (sum = 1)."""
    idx = np.arange(n, dtype=np.float64)
    x0 = idx / n
    x1 = (idx + 1) / n
    a = arcsine_cdf(x1) - arcsine_cdf(x0)
    a_sum = a.sum()
    if a_sum == 0.0 or np.isnan(a_sum):
        raise RuntimeError("Arcsine initialisation failed.")
    return a / a_sum


def _shape_powerlaw(n: int, eps: float, beta: float) -> np.ndarray:
    """
    Simple symmetric power‑law shape:
        a_i ∝ ((i+ε+0.5)*(n‑1‑i+ε+0.5))^(‑β)

    for ε ≥ 0 and β > 0.  The vector is normalised to sum‑1.
    """
    i = np.arange(n, dtype=np.float64)
    den = (i + eps + 0.5) * (n - 1 - i + eps + 0.5)
    # avoid underflow/overflow – clamp to a small positive number
    den = np.maximum(den, 1e-30)
    a = den ** (-beta)
    a_sum = a.sum()
    if a_sum == 0.0 or np.isnan(a_sum):
        return np.full(n, 1.0 / n)
    return a / a_sum


def _smooth_moving_average(a: np.ndarray, w: int) -> np.ndarray:
    """Apply a simple moving‑average smoothing with window width w (odd)."""
    if w <= 1:
        return a.copy()
    kernel = np.ones(w, dtype=np.float64) / w
    # mode='same' keeps the length; padding with zeros at the borders
    a_s = np.convolve(a, kernel, mode='same')
    # re‑normalise – the convolution may slightly change the total mass
    s = a_s.sum()
    if s == 0.0 or np.isnan(s):
        return a.copy()
    return a_s / s

        print(f"[Iter {it_num:6d}] best_score = {score:.6f}   elapsed = {elapsed:.1f}s")

    while time.time() < deadline and it < max_iters:
        it += 1

        # current probability vector (sums to 1 by construction)
        a_cur = _softmax(z)

        # exact max‑convolution and its sub‑gradient
        max_c, grad_a = _maxconv_and_subgrad(a_cur)

        # objective value (cheap – no extra evaluation needed)
        cur_score = 2.0 * n * max_c

        if cur_score < best_score:
            best_score, best_a = cur_score, a_cur.copy()
            if it % 2000 == 0 or cur_score < 1.02:
                _log(it, best_score)

        # Gradient of the full objective J = 2·n·max_c
        grad_J = 2.0 * n * grad_a

        # Mirror‑descent (natural) gradient on logits
        mu = np.dot(grad_J, a_cur)          # 〈grad_J , a〉
        grad_z = a_cur * (grad_J - mu)      # ∇_z J

        # Adam updates (bias‑corrected)
        m1 = beta1 * m1 + (1.0 - beta1) * grad_z
        v1 = beta2 * v1 + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m1 / (1.0 - beta1 ** it)
        v_hat = v1 / (1.0 - beta2 ** it)

        # Typical Adam step size with a mild decay
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step = -alpha_t * m_hat / (np.sqrt(v_hat) + eps_adam)

        z = z + step
        # keep logits numerically safe
        z = np.clip(z, -60.0, 60.0)

        # occasional tiny random jitter to shake out of flat plateaus
        if it % 10000 == 0 and np.random.rand() < 0.12:
            z += 0.02 * np.random.randn(n)

        # early exit if already very close to the theoretical lower bound
        if best_score < 1.01 and it > 5000:
            print("[Info] Near‑optimal score reached – terminating early.")
            break

    # ------------------------------------------------------------------
    # 4. Final logging and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()

    # Optional exact evaluation (the platform will evaluate anyway)
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[Info] Final exact score = {final_score:.6f}")
    except Exception as exc:
        print(f"[Warning] evaluate_sequence failed (ignored): {exc}")

    return final_seq
