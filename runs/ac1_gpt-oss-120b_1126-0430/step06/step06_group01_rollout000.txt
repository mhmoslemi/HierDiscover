def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    np.random.seed(seed)
    start=time.time()
    deadline = start + budget_s -2  # safety

    # Determine n
    try:
        n = len(height_sequence_1)
        print(f"Using external length n = {n}")
    except Exception:
        n = 50000
        print(f"No external sequence – using default length n = {n}")

    # Choose candidate beta shape parameters to search
    beta_candidates = [0.35,0.4,0.45,0.5,0.55,0.6,0.65]
    best_a=None
    best_score=np.inf
    for beta in beta_candidates:
        a = _beta_bins(n,beta)
        max_conv = _conv_max(a)
        score = 2.0 * n * max_conv
        print(f"beta={beta:.3f}: initial score {score:.6f}")
        if score < best_score:
            best_score = score
            best_a = a
    print(f"Best beta {beta:.3f} initial best_score {best_score:.6f}")

    # Optional quick refine of best candidate using Adam
    # softmax param:
    z = _logits_from_distribution(best_a)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_alpha = 0.8
    eps_factor = 0.1
    eps_factor_min = 5e-4
    eps_factor_decay = 0.9995

    # Decide max_iters based on remaining time: allow up to 200000 or until deadline
    max_iters = int(3e5)
    it=0
    while time.time() < deadline and it < max_iters:
        it+=1
        a_cur = _softmax(z)
        max_conv, grad_a = _conv_max_and_grad(a_cur, eps_factor)
        score = 2.0 * n * max_conv
        if score < best_score:
            best_score = score
            best_a = a_cur.copy()
            if it % 1000 == 0:
                print(f"Iter {it:6d}: new best score {best_score:.6f}")
            # early stop if already below target
            if best_score <= 1.503 + 1e-6:
                print(f"Target reached at iteration {it}, score {best_score:.6f}")
                break

        # grad on logits
        mu = np.dot(grad_a, a_cur)
        grad_z = a_cur * (grad_a - mu)
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z**2)
        m_hat = m / (1.0 - beta1**it)
        v_hat = v / (1.0 - beta2**it)
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step = - alpha_t * m_hat / (np.sqrt(v_hat) + eps_adam)
        z = z + step
        # clip
        z = np.clip(z, -80.0, 80.0)
        # decay eps_factor
        eps_factor = max(eps_factor_min, eps_factor * eps_factor_decay)

        # jitter occasional

"""
Propose a step‑function sequence that minimizes
    2·n·max(a*a) / (sum a)^2

The algorithm:
  1. Pick a moderate size n (default 50000).  Larger n reduces the
     discretisation error of the continuous arcsine law, whose
     self‑convolution is perfectly uniform.
  2. Initialise the vector from a family of symmetric beta‑distributions
     (density ∝ x^{β‑1}(1‑x)^{β‑1}).  β = 0.5 gives the classic arcsine law.
     A tiny grid search over β values finds the best start for the given n.
  3. Optimise the probability vector on the simplex using Adam on a
     *smoothed* max‑convolution objective.
        – The vector is represented by logits z and transformed with softmax
          (guaranteeing non‑negativity and sum‑to‑1).
        – The smoothed max is a log‑sum‑exp of the convolution entries,
          yielding a differentiable surrogate.
        – Gradients are obtained by a single FFT‑based convolution per
          iteration (O(n log n)).
  4. The optimisation runs until the time budget expires (or the target
     score 1.503 is reached).  Random jitter and a slowly decreasing
     smoothing constant help escape plateaus.
  5. The best sequence found is returned as a plain Python list.

All helper functions are top‑level (no closures) and the code respects the
1000 s time limit.
"""

import time
import math
import numpy as np
from scipy import special as spc

# ----------------------------------------------------------------------
# Numerical helpers
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max; returns a probability vector (sum = 1)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Logits that map to a via soft‑max (used as Adam parameters)."""
    return np.log(a + eps)


def _fft_len_for_conv(n: int) -> int:
    """Smallest power‑of‑two ≥ 2·n‑1 (sufficient for linear self‑convolution)."""
    return 1 << ((2 * n - 1).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a using FFT."""
    n = a.shape[0]
    L = _fft_len_for_conv(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Smoothed maximum of the self‑convolution and its gradient w.r.t. a.

    Parameters
    ----------
    a : np.ndarray
        Probability vector (non‑negative, sum = 1).
    eps_factor : float
        Multiplicative factor for the smoothing constant (eps = eps_factor * max).

    Returns
    -------
    max_conv : float
        Exact maximum of a*a (for reporting).
    grad_a : np.ndarray
        Gradient of 4·n·smooth_max with respect to a (already multiplied by 4·n).
    """
    n = a.shape[0]
    L = _fft_len_for_conv(n)

    a_fft = np.fft.rfft(a, L)

    # Linear self‑convolution (length 2·n‑1)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # Smoothing (log‑sum‑exp)
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum

    # Correlation a ⋆ w (length n)
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # Gradient (already includes factor 4·n)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a


def _beta_bins(n: int, beta: float) -> np.ndarray:
    """
    Discretise a symmetric Beta(β,β) distribution on a uniform grid.

    Returns a probability vector of length n (sum = 1).
    """
    # grid points: 0, 1/n, 2/n, …, 1
    pts = np.arange(n + 1, dtype=np.float64) / n
    cdf = spc.betainc(beta, beta, pts)  # regularised incomplete beta = CDF
    a = cdf[1:] - cdf[:-1]               # bin probabilities
    # Clip tiny negatives caused by round‑off and renormalise.
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total <= 0.0 or not np.isfinite(total):
        raise RuntimeError(f"Beta bins failed for beta={beta}")
    a /= total
    return a

    """
    Return a list of non‑negative heights that (approximately) minimise the
    evaluation metric.  The routine follows the strategy described in the
    module docstring.
    """
    np.random.seed(seed)
    start_time = time.time()
    # Keep a small safety margin for printing / final evaluation.
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------

    try:

    # 1. Quick search over a few beta‑shapes to find a good start.
    # ------------------------------------------------------------------
    beta_candidates = [0.35, 0.40, 0.45, 0.50, 0.55, 0.60, 0.65]
    best_score = np.inf
    best_a = None
    best_beta = None

    for beta in beta_candidates:

        max_conv = _conv_max(a)
        score = 2.0 * n * max_conv
        print(f"beta = {beta:5.3f} → initial score = {score:.6f}")
        if score < best_score:
            best_score = score
            best_a = a
            best_beta = beta

    print(f"Chosen beta = {best_beta:.3f} with initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Adam optimisation on the simplex (soft‑max parametrisation).
    # ------------------------------------------------------------------
    # Initialise logits.
    z = _logits_from_distribution(best_a)
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_alpha = 0.8          # base learning‑rate scale
    eps_factor = 0.10         # initial smoothing factor
    eps_factor_min = 5e-4
    eps_factor_decay = 0.9995

    iteration = 0
    # Upper bound on iterations – we also stop when time runs out.
    max_iterations = 300_000

    while time.time() < deadline and iteration < max_iterations:
        iteration += 1

        # Current probability vector.
        a_cur = _softmax(z)

        # Smoothed max + gradient.
        max_conv, grad_a = _conv_max_and_grad(a_cur, eps_factor)
        score = 2.0 * n * max_conv

        # Track best solution.
        if score < best_score:
            best_score = score
            best_a = a_cur.copy()
            if iteration % 1000 == 0:
                print(f"Iter {iteration:6d}: new best score = {best_score:.8f}")

            # Early exit if target is already met.
            if best_score <= 1.503 + 1e-9:
                print(f"Target reached (score ≤ 1.503) at iter {iteration}.")
                break

        # Gradient w.r.t. logits (mirrored‑descent on the simplex).
        mu = np.dot(grad_a, a_cur)
        grad_z = a_cur * (grad_a - mu)

        # Adam biased‑corrected moments.
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        # Step size schedule (inverse‑sqrt).
        alpha_t = base_alpha / (math.sqrt(iteration) + 1.0)
        step = -alpha_t * m_hat / (np.sqrt(v_hat) + eps_adam)
        z += step

        # Keep logits in a safe numeric range.
        z = np.clip(z, -80.0, 80.0)

        # Reduce smoothing gradually.
        eps_factor = max(eps_factor_min, eps_factor * eps_factor_decay)

        # Occasional random jitter to escape shallow plateaus.
        if iteration % 10000 == 0 and np.random.rand() < 0.05:
            z += 0.01 * np.random.randn(n)

    # ------------------------------------------------------------------
    # 3. Return the best sequence as a plain Python list.
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        exact_score = evaluate_sequence(final_seq)
        print(f"Exact evaluation (final) = {exact_score:.8f}")
    except Exception as exc:
        print(f"Exact evaluation failed: {exc}")

    return final_seq
