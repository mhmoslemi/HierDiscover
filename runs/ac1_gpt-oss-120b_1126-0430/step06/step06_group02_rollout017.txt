#!/usr/bin/env python
"""
Non‑negative step‑function optimiser – LP + Adam hybrid.

The optimiser works in two phases.

1. **Linear‑programming boost** – a current distribution `f` is used to
   build a Toeplitz matrix `A` such that `A@g = f*g` (full linear
   convolution).  Solving the LP

        maximise   sum(g)
        subject to A @ g ≤ max(f*f)   (component‑wise)
                   g ≥ 0

   yields a new candidate `g` that can carry more mass without increasing
   the current maximal convolution value.  The distribution is updated
   by a convex combination `f ← (1‑t)·f + t·g` (t≈0.2) and renormalised.
   A handful of such LP updates dramatically reduces the cheap objective
   `2·n·max(f*f)`.

2. **Adam refinement** – the best distribution from phase‑1 is up‑sampled
   (linear interpolation) to a larger support and then optimised with
   Adam on the **soft‑max** parametrisation of the simplex.  The smooth
   max of the convolution is used to obtain a stable gradient
   (`_conv_max_and_grad`).  A simple learning‑rate schedule, occasional
   jitter and a restart‑on‑stagnation strategy finish the optimisation.

The whole routine respects the given time budget (default 1000 s) and
returns a list of non‑negative heights that (hopefully) improves the
`evaluate_sequence` score.
"""

import time
import math
import numpy as np
import scipy.sparse as sp
import cvxpy as cp

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector (sums to 1)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    s = e.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / z.size)
    return e / s


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it that reproduces `a` after a soft‑max."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """Smallest power‑of‑two ≥ 3·n‑2 (enough for all needed convolutions)."""
    return 1 << ((3 * n - 2).bit_length())


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    s = new_a.sum()
    if s == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= s
    return new_a


def _exp_decay_distribution_unsym(n: int, lam: float) -> np.ndarray:
    """a_i ∝ e^{‑lam·i}, normalised (strictly decreasing)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Returns (max_conv, grad_a) where

        grad_a = ∇_a [ 2·n·smooth_max(a*a) ]

    `eps_factor` multiplies the smoothing constant ε = eps_factor·max_conv.
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    # FFT of a (zero‑padded)
    a_fft = np.fft.rfft(a, L)

    # Linear self‑convolution a*a
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # Smoothed soft‑max weights
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv                # ≤ 0 for numerical stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # Correlation a ⋆ w
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # Gradient (already multiplied by 4·n)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a


def _build_conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build a sparse matrix A (shape (2n‑1, n)) such that A @ g = f * g
    (full linear convolution).  `f` must be a 1‑D non‑negative array.
    """
    n = f.shape[0]
    # Each column i contains the vector f placed at rows i … i+n‑1
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    data = np.empty(n * n, dtype=np.float64)
    pos = 0
    for i in range(n):
        rows[pos:pos + n] = np.arange(i, i + n, dtype=np.int32)
        cols[pos:pos + n] = i
        data[pos:pos + n] = f
        pos += n
    # Trim (pos == n*n)
    rows = rows[:pos]
    cols = cols[:pos]
    data = data[:pos]
    return sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


# ----------------------------------------------------------------------
# Main optimiser
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) improves the
    `evaluate_sequence` score.  The routine respects the given time budget.
    """
    np.random.seed(seed)
    start = time.time()
    # Safety margin for the final evaluation call
    deadline = start + budget_s - 5.0

    # --------------------------------------------------------------
    # 0. Initial support size and starting distribution
    # --------------------------------------------------------------
    n0 = 2000                     # small enough for fast LP solves
    best_a = None

    # Warm‑start from a globally supplied sequence if available
    try:
        init_global = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        best_a = _interpolate_distribution(init_global, n0)
        print("[Init] Warm‑started from global sequence (down‑sampled).")
    except Exception:
        # No global – use a simple decreasing exponential as seed
        lam = 0.0005               # λ ≈ 0.5 · n⁻¹, a reasonable starting point
        best_a = _exp_decay_distribution_unsym(n0, lam)
        print("[Init] Seeded with unsymmetric exponential distribution.")

    # Work always with a probability vector (∑ a = 1) – the evaluation
    # is scale‑invariant.
    best_a = best_a / best_a.sum()
    best_score = 2.0 * n0 * _conv_max(best_a)   # cheap objective, Σa=1 ⇒ denominator = 1

    # --------------------------------------------------------------
    # 1. LP‑based bound‑improving loop
    # --------------------------------------------------------------
    mix_step = 0.2               # convex‑combination weight
    outer_it = 0

    while time.time() < deadline and outer_it < 15:
        outer_it += 1
        n = best_a.shape[0]

        # Current maximal self‑convolution
        cur_max = _conv_max(best_a)

        # Build convolution matrix A for the current f = best_a
        A = _build_conv_matrix(best_a)

        # Linear programme:
        # maximise   1ᵀ·g
        # subject to  A·g ≤ cur_max · 1   (element‑wise)
        #             g ≥ 0
        g = cp.Variable(n, nonneg=True)
        constraints = [A @ g <= cur_max * np.ones(2 * n - 1, dtype=np.float64)]
        objective = cp.Maximize(cp.sum(g))
        prob = cp.Problem(objective, constraints)

        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2500, verbose=False)
        except Exception as exc:
            print(f"[LP {outer_it}] Solver error: {exc}")
            break

        if g.value is None:
            print(f"[LP {outer_it}] No solution returned.")
            break

        g_vec = np.asarray(g.value).flatten()
        if g_vec.sum() == 0.0:
            print(f"[LP {outer_it}] Trivial solution, stop.")
            break

        # Normalise g to a probability vector
        g_vec /= g_vec.sum()

        # Convex update and renormalise
        new_a = (1.0 - mix_step) * best_a + mix_step * g_vec
        new_a /= new_a.sum()

        # Cheap score of the candidate
        new_score = 2.0 * n * _conv_max(new_a)

        if new_score < best_score:
            best_score = new_score
            best_a = new_a
            print(f"[LP {outer_it}] improved score → {best_score:.6f}")
        else:
            # No improvement – shrink the step size
            mix_step *= 0.8
            print(f"[LP {outer_it}] no improvement, reduce mixing to {mix_step:.4f}")
            if mix_step < 1e-4:
                print("[LP] mixing step too small, terminating LP phase.")
                break

    # --------------------------------------------------------------
    # 2. Upsampling + Adam‑style refinement on the simplex
    # --------------------------------------------------------------
    # Choose a larger support if time permits
    remaining = deadline - time.time()
    target_n = min(20000, max(best_a.shape[0] * 2, int(best_a.shape[0] * 1.5)))
    if remaining > 200 and target_n > best_a.shape[0]:
        print(f"[Upsample] Interpolating from {best_a.shape[0]} → {target_n}")
        best_a = _interpolate_distribution(best_a, target_n)

    n = best_a.shape[0]

    # Initialise logits for mirror‑descent on the simplex
    z = _logits_from_distribution(best_a) + 0.01 * np.random.randn(n)

    # Adam state
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    base_alpha = 0.55          # base learning‑rate for logits
    eps_factor = 0.25           # ε = eps_factor·max_conv  (larger at start)
    eps_factor_min = 5e-5
    eps_decay = 0.9993

    # Track the best sequence seen during Adam's run
    best_seq = _softmax(z)
    best_seq_score = 2.0 * n * _conv_max(best_seq)
    last_improve_iter = 0
    it = 0

    while time.time() < deadline and it < 2_500_000:
        it += 1

        a = _softmax(z)

        # Smoothed max and its gradient
        max_conv, grad_a = _conv_max_and_grad(a, eps_factor)
        cheap_score = 2.0 * n * max_conv

        # Keep best ever seen
        if cheap_score < best_seq_score - 1e-12:
            best_seq_score = cheap_score
            best_seq = a.copy()
            last_improve_iter = it
            if it % 5000 == 0:
                print(f"[Adam {it:07d}] NEW BEST cheap = {cheap_score:.6f}")

        # Gradient w.r.t. logits (mirror descent)
        mu = np.dot(grad_a, a)               # <grad, a>
        grad_z = a * (grad_a - mu)

        # Adam moment updates (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # Step‑size schedule (inverse‑sqrt decay)
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step_dir = - m_hat / (np.sqrt(v_hat) + eps_adam)

        # Simple back‑tracking line search (up to 5 halvings)
        accepted = False
        trial_alpha = alpha_t
        for _ in range(5):
            z_new = z + trial_alpha * step_dir
            z_new = np.clip(z_new, -50.0, 50.0)
            a_new = _softmax(z_new)
            new_score = 2.0 * n * _conv_max(a_new)
            if new_score <= cheap_score:
                z = z_new
                accepted = True
                break
            trial_alpha *= 0.5

        if not accepted:
            # No improvement – still take a modest step.
            z = z + alpha_t * step_dir
            z = np.clip(z, -50.0, 50.0)

        # Occasionally add jitter to escape plateaux
        if (it % 4000 == 0) and (np.random.rand() < 0.12):
            z += 0.015 * np.random.randn(n)
            m *= 0.6
            v *= 0.6

        # Gradually reduce the smoothing factor
        eps_factor = max(eps_factor_min, eps_factor * eps_decay)

        # Restart if we have been stuck for a long time
        if it - last_improve_iter > 250_000:
            print(f"[Adam {it}] Stagnation – restarting around best.")
            z = _logits_from_distribution(best_seq) + 0.025 * np.random.randn(n)
            m.fill(0.0)
            v.fill(0.0)
            eps_factor = 0.25
            last_improve_iter = it

    # --------------------------------------------------------------
    # 3. Final exact evaluation and return
    # --------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    return final_seq
