#!/usr/bin/env python
"""
Non‑negative step‑function optimiser (variant).

The optimiser works on a moderately large support (≈ 20 000 points)
and combines three ideas:

1. **Warm‑start** – use the global sequence if it exists, otherwise
   scan a few simple parametric families (power‑law, one‑sided
   exponential, logistic, linear) and keep the best (lowest value of
   2·n·max(a∗a)).

2. **Mirror‑descent on the simplex** – the coefficients are stored as
   soft‑max logits.  At each iteration a smoothed ∞‑norm of the
   self‑convolution is evaluated with a soft‑max weighting, its gradient
   is obtained via FFTs and an Adam update is performed on the logits.
   The smoothing constant ε is multiplied by a decaying factor so the
   algorithm gradually focuses on the true maximum.

3. **Exploration tricks** – occasional jitter, learning‑rate decay,
   line‑search, and a stagnation‑restart guard keep the search from
   getting stuck in plateaux.

The routine respects the given time budget, returns the best sequence
found and finally evaluates it with the supplied ``evaluate_sequence``
function.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top‑level, no closures
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max, returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    s = e.sum()
    if s == 0.0 or not np.isfinite(s):
        return np.full_like(z, 1.0 / z.size)
    return e / s


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it that reproduces `a` after a soft‑max."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """Smallest power‑of‑two ≥ 3·n‑2 (enough for all needed FFTs)."""
    return 1 << ((3 * n - 2).bit_length())


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    s = new_a.sum()
    if s == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= s
    return new_a


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{‑p}, normalised."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _exp_asym_distribution(n: int, lam: float) -> np.ndarray:
    """a_i ∝ e^{‑lam·i}, normalised (strictly decreasing)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _logistic_distribution(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ 1/(1+e^{α·(i‑mid)}), left‑heavy, normalised."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = 1.0 / (1.0 + np.exp(alpha * (i - mid)))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _linear_decreasing_distribution(n: int) -> np.ndarray:
    """a_i ∝ (n‑i), i‑th entry larger on the left, normalised."""
    a = np.arange(n, 0, -1, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Returns (max_conv, grad_a) where

        grad_a = ∇_a [ 2·n·smooth_max(a*a) ]

    `eps_factor` multiplies the smoothing constant ε = eps_factor·max_conv.
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    # FFT of a (zero‑padded)
    a_fft = np.fft.rfft(a, L)

    # Linear self‑convolution a*a
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # Smoothed soft‑max weights
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv                # ≤ 0 for numerical stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # Correlation a ⋆ w   (cross‑correlation, length n)
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # Gradient (already multiplied by 4·n as in the original script)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a

    # Power‑law a_i ∝ (i+1)^‑p
    if time.time() < deadline:
        for p in np.linspace(0.2, 2.0, 30):
            if time.time() >= deadline:
                break
            cand = _power_law_distribution(n, p)
            score = 2.0 * n * _conv_max(cand)
            if score < best_score:
                best_score, best_a = score, cand

    # One‑sided exponential a_i ∝ exp(‑λ·i)
    if time.time() < deadline:
        for loglam in np.linspace(-5, -1, 25):  # λ in [1e‑5, 1e‑1]
            if time.time() >= deadline:
                break
            lam = 10.0 ** loglam
            cand = _exp_asym_distribution(n, lam)
            score = 2.0 * n * _conv_max(cand)
            if score < best_score:
                best_score, best_a = score, cand

    # Logistic left‑heavy
    if time.time() < deadline:
        for alpha in np.linspace(0.1, 5.0, 30):
            if time.time() >= deadline:
                break
            cand = _logistic_distribution(n, alpha)
            score = 2.0 * n * _conv_max(cand)
            if score < best_score:
                best_score, best_a = score, cand

    # Linear decreasing (deterministic shape)
    if time.time() < deadline:
        cand = _linear_decreasing_distribution(n)
        score = 2.0 * n * _conv_max(cand)
        if score < best_score:
            best_score, best_a = score, cand

    # Fallback – uniform distribution
    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_score = 2.0 * n * _conv_max(best_a)

    # Small random jitter before the optimisation loop
    z = _logits_from_distribution(best_a) + 0.01 * np.random.randn(n)

    # --------------------------------------------------------------
    # 1. Adam state on logits (mirror descent on the simplex)
    # --------------------------------------------------------------
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    base_alpha = 0.5          # base learning‑rate for the logits
    eps_factor = 0.20          # ε = eps_factor·max_conv (large at start)
    eps_factor_min = 5e-7
    eps_decay = 0.9995
    report_every = 50000

    # --------------------------------------------------------------
    # 2. Optimisation loop
    # --------------------------------------------------------------
    best_seq = _softmax(z)
    best_score = 2.0 * n * _conv_max(best_seq)
    last_improve_iter = 0
    it = 0

    while time.time() < deadline and it < 5_000_000:
        it += 1

        # ----- Current distribution on the simplex -----
        a = _softmax(z)

        # ----- Smoothed max of the self‑convolution and its gradient -----
        max_conv, grad_a = _conv_max_and_grad(a, eps_factor)
        cheap_score = 2.0 * n * max_conv

        # ----- Keep the best sequence ever observed -----
        if cheap_score < best_score - 1e-12:
            best_score = cheap_score
            best_seq = a.copy()
            last_improve_iter = it
            if it % report_every == 0:
                print(f"[Iter {it:07d}] NEW BEST cheap = {cheap_score:.6f}")

        # ----- Mirror‑descent gradient (logits) -----
        mu = np.dot(grad_a, a)               # <grad, a>
        grad_z = a * (grad_a - mu)

        # ----- Adam moment updates (bias‑corrected) -----
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # ----- Step size (inverse‑sqrt decay) -----
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step_dir = - m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- Simple back‑tracking line search (≤ 3 halvings) -----
        accepted = False
        trial_alpha = alpha_t
        for _ in range(3):
            z_new = z + trial_alpha * step_dir
            # Clamp logits to avoid overflow in the exp()
            z_new = np.clip(z_new, -50.0, 50.0)
            a_new = _softmax(z_new)
            new_score = 2.0 * n * _conv_max(a_new)
            if new_score <= cheap_score:
                z = z_new
                accepted = True
                break
            trial_alpha *= 0.5

        if not accepted:
            # no improvement – still take a modest step
            z = z + alpha_t * step_dir
            z = np.clip(z, -50.0, 50.0)

        # ----- Occasional jitter to break out of plateaux -----
        if (it % 5000 == 0) and (np.random.rand() < 0.10):
            z += 0.02 * np.random.randn(n)
            m *= 0.6
            v *= 0.6

        # ----- Gradually tighten the smoothing of the max -----
        eps_factor = max(eps_factor_min, eps_factor * eps_decay)

        # ----- Restart if we have been stuck for a long while -----
        if it - last_improve_iter > 300_000:
            print(f"[Iter {it}] Stagnation detected – restarting with jitter.")
            z = _logits_from_distribution(best_seq) + 0.03 * np.random.randn(n)
            m.fill(0.0)
            v.fill(0.0)
            eps_factor = 0.20
            last_improve_iter = it

    # --------------------------------------------------------------
    # 3. Final exact evaluation with the provided ``evaluate_sequence``.
    # --------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # --------------------------------------------------------------
    # 4. Return the best sequence discovered.
    # --------------------------------------------------------------
    return final_seq
