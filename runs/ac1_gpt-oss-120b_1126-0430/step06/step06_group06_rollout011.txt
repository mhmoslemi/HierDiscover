        conv_new = _conv_full_fft(a_new, conv_fft_len)
        maxb_new = conv_new.max()
        score_new = 2.0 * n * maxb_new

        # accept if improvement, else maybe keep old
        if score_new < score:
            a = a_new
        else:
            # maybe try smaller step
            a = _simplex_projection(a - (eta*0.2) * g)  # try reduced step
            # optionally do nothing

        # occasional random perturbation to avoid stagnation
        if no_improve > 5000:
            # perturb in log space
            log_a = np.log(a + 1e-30)
            log_a += 0.1 * np.random.randn(n)
            a = np.exp(log_a)
            a = a / a.sum()
            no_improve = 0
            print(f"[Iter {iter_cnt}] Restart due to stagnation")
        # occasional restart from best + small noise
        if iter_cnt % 50000 == 0:
            perturb = 0.02 * np.random.rand(n)
            a = _simplex_projection(best_a + perturb)
            print(f"[Iter {iter_cnt}] periodic perturbation")

    # Phase 2: final (1+1) ES hill climbing on best_a
    cur = best_a.copy()
    cur_score = best_val
    sigma = 0.05
    fail = 0
    while time.time() < deadline - 2.0:
        cand = cur * np.exp(sigma * np.random.randn(n))
        cand = _simplex_projection(cand)
        cand_score = cheap_score_vec(cand)
        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            sigma = max(sigma * 0.85, 1e-5)
            fail = 0
            print(f"[ES] improved to {cand_score:.6f}, sigma={sigma:.5f}")
        else:
            fail += 1
            if fail > 200:
                sigma = min(sigma * 1.3, 0.5)
                fail = 0

    final_seq = cur.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"Final exact score = {final_score:.6f}")

    return final_seq

#!/usr/bin/env python
"""
Propose a non‑negative step function (list of heights) that minimises

        2·n·max_k (a*a)[k] / (Σ a_i)²

The code explores a large collection of parametric families, picks the
best one and refines it with a smooth‑max gradient descent (soft‑max
over the convolution) followed by a (1+1)‑ES hill‑climber.  All steps
respect the simplex ∑a_i=1, a_i≥0 and run inside the 1000 s time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using real FFT, length 2·n‑1."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len).real[: 2 * a.size - 1]
    return conv


def _conv_max_fft(a: np.ndarray, fft_len: int) -> tuple[float, int]:
    """Maximum coefficient of a*a and its index."""
    conv = _conv_full_fft(a, fft_len)
    idx = int(conv.argmax())
    return float(conv[idx]), idx


# ----------------------------------------------------------------------
# Parametric families (all return a normalised probability vector)
# ----------------------------------------------------------------------


def _uniform_seq(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _power_law_seq(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = i ** (-p)
    seq /= seq.sum()
    return seq


def _exponential_seq(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    seq = np.exp(-lam * i)
    seq /= seq.sum()
    return seq


def _power_law_exp_seq(n: int, p: float, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    seq = (i + 1.0) ** (-p) * np.exp(-lam * i)
    seq /= seq.sum()
    return seq


def _stretched_exponential_seq(n: int, lam: float, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    seq = np.exp(-lam * (i ** gamma))
    seq /= seq.sum()
    return seq


def _beta_seq(n: int, alpha: float, beta: float) -> np.ndarray | None:
    x = (np.arange(1, n + 1, dtype=np.float64) / n)
    seq = (x ** alpha) * ((1.0 - x) ** beta)
    s = seq.sum()
    if s == 0.0:
        return None
    seq /= s
    return seq


def _dirichlet_seq(n: int, conc: float) -> np.ndarray:
    g = np.random.gamma(conc, 1.0, size=n)
    seq = g / g.sum()
    return seq


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that
    minimises ``evaluate_sequence``.  The routine adapts its schedule to
    the available time budget and always returns a feasible sequence.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 5.0

    # -------------------- choose working length ------------------------
    try:
        prev = globals()["height_sequence_1"]
        n = max(len(prev), 5000)
        prev_arr = np.asarray(prev, dtype=np.float64)
        if prev_arr.sum() > 0.0:
            prev_arr = prev_arr / prev_arr.sum()
        else:
            prev_arr = None
    except Exception:
        prev_arr = None
        n = 5000

    n = min(max(n, 5000), 25000)                 # enforce 5 k ≤ n ≤ 25 k
    conv_fft_len = 1 << ((2 * n - 1).bit_length())   # for a*a
    grad_fft_len = 1 << ((3 * n - 2).bit_length())   # for gradient

    # -------------------- cheap score (sum already 1) -----------------
    def cheap_score(vec: np.ndarray) -> float:
        """Exact value of the objective when Σa = 1."""
        maxb, _ = _conv_max_fft(vec, conv_fft_len)
        return 2.0 * n * maxb

    # -------------------- Phase 0 – sweep families ---------------------
    best_a = _uniform_seq(n)
    best_val = cheap_score(best_a)

    # previous construction if same length
    if prev_arr is not None and prev_arr.size == n:
        val = cheap_score(prev_arr)
        if val < best_val:
            best_a = prev_arr.copy()
            best_val = val

    # 1) power‑law
    for p in np.linspace(0.1, 2.5, 60):
        if time.time() >= deadline:
            break
        cand = _power_law_seq(n, p)
        val = cheap_score(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # 2) exponential
    for lam in np.linspace(0.0, 4.0, 50):
        if time.time() >= deadline:
            break
        cand = _exponential_seq(n, lam)
        val = cheap_score(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # 3) mixed power‑law × exponential
    for p in np.linspace(0.1, 1.5, 16):
        for lam in np.linspace(0.0, 3.0, 16):
            if time.time() >= deadline:
                break
            cand = _power_law_exp_seq(n, p, lam)
            val = cheap_score(cand)
            if val < best_val:
                best_a, best_val = cand, val

    # 4) stretched exponential
    for gamma in [0.5, 0.8, 1.0, 1.2, 1.5, 2.0]:
        for lam in np.linspace(0.0, 4.0, 40):
            if time.time() >= deadline:
                break
            cand = _stretched_exponential_seq(n, lam, gamma)
            val = cheap_score(cand)
            if val < best_val:
                best_a, best_val = cand, val

    # 5) Beta family (α,β in log‑space)
    alphas = np.logspace(-2, 0.8, 12)   # 0.01 … ≈6.3
    betas  = np.logspace(-2, 0.8, 12)
    for α in alphas:
        for β in betas:
            if time.time() >= deadline:
                break
            cand = _beta_seq(n, α, β)
            if cand is None:
                continue
            val = cheap_score(cand)
            if val < best_val:
                best_a, best_val = cand, val

    # 6) sinusoidal‑power family (smooth, unimodal)
    i = np.arange(n, dtype=np.float64)
    for power in np.linspace(0.5, 5.0, 20):
        if time.time() >= deadline:
            break
        seq = np.sin(np.pi * (i + 0.5) / n) ** power
        seq /= seq.sum()
        val = cheap_score(seq)
        if val < best_val:
            best_a, best_val = seq, val

    # 7) tent (piecewise‑linear) shape
    half = n // 2
    tent = np.concatenate([np.arange(1, half + 1),
                           np.arange(half, 0, -1)[: n - half]])
    tent = tent.astype(np.float64)
    tent /= tent.sum()
    val = cheap_score(tent)
    if val < best_val:
        best_a, best_val = tent, val

    # 8) Dirichlet draws (various concentrations)
    for conc in [0.001, 0.005, 0.01, 0.05, 0.1, 0.5, 1.0, 2.0, 5.0]:
        if time.time() >= deadline:
            break
        cand = _dirichlet_seq(n, conc)
        val = cheap_score(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # 9) random uniform draws (purely stochastic)
    for _ in range(300):
        if time.time() >= deadline:
            break
        cand = np.random.rand(n)
        cand /= cand.sum()
        val = cheap_score(cand)
        if val < best_val:
            best_a, best_val = cand, val

    print(f"[Init] best cheap score = {best_val:.6f}")

    # -------------------- Phase 1 – smooth‑max gradient descent -------
    a = best_a.copy()
    best_iter_a = a.copy()
    best_iter_val = best_val
    iter_cnt = 0
    no_improve = 0
    beta = 0.5               # starts very smooth
    beta_inc = 0.02          # slowly sharpen

    max_iters = 500_000

    while time.time() < deadline - 5.0 and iter_cnt < max_iters:
        iter_cnt += 1

        # full convolution
        conv = _conv_full_fft(a, conv_fft_len)
        maxb = conv.max()
        cur_val = 2.0 * n * maxb

        # record improvement
        if cur_val < best_iter_val - 1e-12:
            best_iter_val = cur_val
            best_iter_a = a.copy()
            no_improve = 0
            print(f"[Iter {iter_cnt:07d}] NEW BEST cheap = {cur_val:.6f}")

        else:
            no_improve += 1

        # increase smoothing parameter
        beta = min(beta + beta_inc, 20.0)

        # soft‑max weights over the convolution entries
        # (subtract max for numerical stability)
        exp_w = np.exp(beta * (conv - conv.max()))
        w = exp_w / exp_w.sum()                     # size 2·n‑1, sums to 1

        # gradient = 2 * (w * a_rev)  (convolution)
        a_rev = a[::-1]                              # length n
        fw = np.fft.rfft(w, grad_fft_len)
        fa_rev = np.fft.rfft(a_rev, grad_fft_len)
        g_full = np.fft.irfft(fw * fa_rev, grad_fft_len)
        # slice the central part that corresponds to ∂/∂a_i
        g = 2.0 * g_full[n - 1: n - 1 + n]

        # step‑size schedule (larger at the beginning, decays like 1/√t)
        eta = 0.6 / math.sqrt(iter_cnt)

        # projected gradient step
        a_proposed = _simplex_projection(a - eta * g)

        # evaluate proposed point (cheap because we already have the
        # convolution function – a single FFT call)
        conv_prop = _conv_full_fft(a_proposed, conv_fft_len)
        val_prop = 2.0 * n * conv_prop.max()

        # accept if we improve; otherwise keep the current iterate
        if val_prop < cur_val:
            a = a_proposed
        else:
            # occasional fallback: smaller step
            a_small = _simplex_projection(a - (eta * 0.3) * g)
            conv_small = _conv_full_fft(a_small, conv_fft_len)
            if 2.0 * n * conv_small.max() < cur_val:
                a = a_small

        # restart heuristic if we have been stagnant for a while
        if no_improve > 8000:
            # small random jitter in log‑space, re‑projected
            log_a = np.log(a + 1e-30)
            log_a += 0.12 * np.random.randn(n)
            a = np.exp(log_a)
            a = a / a.sum()
            no_improve = 0
            print(f"[Iter {iter_cnt:07d}] Restart after stagnation")

        # periodic gentle perturbation to keep exploration alive
        if iter_cnt % 50000 == 0:
            a = _simplex_projection(a + 0.015 * np.random.rand(n))

    # -------------------- Phase 2 – (1+1)‑ES hill‑climb ---------------
    cur = best_iter_a.copy()
    cur_val = best_iter_val
    sigma = 0.07
    fail = 0

    while time.time() < deadline - 2.0:
        cand = cur * np.exp(sigma * np.random.randn(n))
        cand = _simplex_projection(cand)
        cand_val = cheap_score(cand)
        if cand_val < cur_val:
            cur, cur_val = cand, cand_val
            sigma = max(sigma * 0.85, 1e-5)
            fail = 0
            print(f"[ES] improved cheap = {cand_val:.6f}  sigma = {sigma:.5f}")
        else:
            fail += 1
            if fail > 250:
                sigma = min(sigma * 1.30, 0.6)
                fail = 0

    # -------------------- Final evaluation ----------------------------
    final_seq = cur.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    return final_seq
