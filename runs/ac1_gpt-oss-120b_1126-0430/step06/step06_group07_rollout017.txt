#!/usr/bin/env python
"""
Propose a non‑negative step function (list of heights) that minimises
``evaluate_sequence``.  The algorithm consists of three stages:

1. **Broad parametric sweep** – generates a large set of candidate
   families (exponential, symmetric exponential, power‑law,
   symmetric power‑law, beta, Dirichlet, and mixtures) and keeps the
   best according to the cheap surrogate ``2·n·max(conv)``.
2. **Exact sub‑gradient exponentiated‑gradient descent** – starting from
   the best candidate, iteratively updates the vector on the simplex
   using a correct sub‑gradient of the maximal convolution entry.
   Several random restarts and a simple learning‑rate schedule are
   employed; the current global best is tracked throughout.
3. **Permutation‑local search** – a simulated‑annealing style random‑swap
   search on the ordering of the entries refines the solution further.
   Finally the exact ``evaluate_sequence`` is called on the best vector.

All helper utilities are defined at module level (no closures or lambdas)
and the routine respects a user‑supplied time‑budget (default 1000 s)."""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using a power‑of‑two FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa).real[: 2 * a.size - 1]
    return conv


def _conv_max_fft(a: np.ndarray, fft_len: int) -> tuple[float, int]:
    """Maximum value of the autocorrelation a*a and its index."""
    conv = _conv_full_fft(a, fft_len)
    idx = int(conv.argmax())
    return float(conv[idx]), idx


def _cheap_score(vec: np.ndarray, n: int, fft_len: int) -> float:
    """Score 2·n·max(conv) – cheap because Σa_i=1 is assumed."""
    maxb, _ = _conv_max_fft(vec, fft_len)
    return 2.0 * n * maxb


def _maxconv_and_subgrad(vec: np.ndarray, n: int, fft_len: int):
    """
    Return (max_conv_value, subgradient) for the function
    f(a)=max_k (a*a)[k]   (the subgradient is taken at the
    first maximal index).  The gradient is the exact sub‑gradient:
    g_i = 2·a_{k_max-i} for all i where the index is valid.
    """
    conv = _conv_full_fft(vec, fft_len)
    maxb = float(conv.max())
    kmax = int(conv.argmax())
    lo = max(0, kmax - (n - 1))
    hi = min(kmax, n - 1)
    idx = np.arange(lo, hi + 1, dtype=np.int64)
    grad = np.zeros_like(vec)
    # correct diagonal contribution (i=j) is also 2·a_i
    grad[idx] = 2.0 * vec[kmax - idx]
    return maxb, grad

# Core search routine required by the platform

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that
    minimises ``evaluate_sequence``.  The routine adapts its schedule
    to the available time budget and always returns a feasible sequence.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the final exact evaluation
    deadline = start_time + budget_s - 5.0
    eps = 1e-30

    # --------------------------------------------------------------
    # Decide on a working length (n).  Use the previous construction if
    # present, otherwise fall back to a default of 5000.
    # --------------------------------------------------------------
    try:
        prev = height_sequence_1          # may be defined globally
        n = max(len(prev), 5000)
        prev_arr = np.asarray(prev, dtype=np.float64)
        if prev_arr.sum() > 0.0:
            prev_arr = prev_arr / prev_arr.sum()
        else:
            prev_arr = None
    except Exception:
        prev_arr = None
        n = 5000

    # clamp n into a reasonable interval for a 1000 s budget
    n = int(min(max(n, 5000), 20000))

    # FFT length – next power of two ≥ 2·n‑1
    fft_len = 1 << ((2 * n - 1).bit_length())

    # --------------------------------------------------------------
    # Helper wrappers that capture the current n/fft_len
    # --------------------------------------------------------------

    def cheap_score(v):
        return _cheap_score(v, n, fft_len)

    def maxconv_and_grad(v):
        return _maxconv_and_subgrad(v, n, fft_len)

    # --------------------------------------------------------------
    # 0.  Broad parametric sweep – keep the best cheap candidate
    # --------------------------------------------------------------
    best_a = None
    best_val = float("inf")

    # uniform baseline
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    val = cheap_score(uniform)
    best_a, best_val = uniform, val
    print(f"[Init] uniform cheap = {val:.6f}")

    # previous construction (if same length)
    if prev_arr is not None and prev_arr.size == n:
        val = cheap_score(prev_arr)
        if val < best_val:
            best_a, best_val = prev_arr.copy(), val
            print(f"[Init] previous cheap = {val:.6f}")

    # ----- exponential (left) ------------------------------------
    for lam in np.geomspace(1e-5, 10.0, num=80):
        if time.time() >= deadline:
            break
        a = np.exp(-lam * np.arange(n, dtype=np.float64))
        a /= a.sum()
        val = cheap_score(a)
        if val < best_val:
            best_a, best_val = a, val
            print(f"[Init] exp(lam={lam:.5g}) cheap = {val:.6f}")

    # ----- symmetric exponential (U‑shaped) -----------------------
    for lam in np.geomspace(1e-5, 10.0, num=80):
        if time.time() >= deadline:
            break
        idx = np.arange(n, dtype=np.float64)
        a = np.exp(-lam * np.minimum(idx, n - 1 - idx))
        a /= a.sum()
        val = cheap_score(a)
        if val < best_val:
            best_a, best_val = a, val
            print(f"[Init] sym‑exp(lam={lam:.5g}) cheap = {val:.6f}")

    # ----- power‑law (left) --------------------------------------
    for p in np.geomspace(0.01, 5.0, num=60):
        if time.time() >= deadline:
            break
        a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
        a /= a.sum()
        val = cheap_score(a)
        if val < best_val:
            best_a, best_val = a, val
            print(f"[Init] power(p={p:.5g}) cheap = {val:.6f}")

    # ----- symmetric power‑law (U‑shaped) ------------------------
    for p in np.geomspace(0.01, 5.0, num=60):
        if time.time() >= deadline:
            break
        i = np.arange(n, dtype=np.float64)
        a = np.minimum(i + 1, n - i) ** (-p)
        a /= a.sum()
        val = cheap_score(a)
        if val < best_val:
            best_a, best_val = a, val
            print(f"[Init] sym‑power(p={p:.5g}) cheap = {val:.6f}")

    # ----- beta families (U‑shaped when α,β<1) --------------------
    x = np.arange(1, n + 1, dtype=np.float64) / n
    alphas = np.concatenate((
        np.geomspace(0.01, 0.2, num=5),
        np.geomspace(0.2, 1.0, num=9),
        np.geomspace(1.0, 5.0, num=5)
    ))
    betas = alphas.copy()
    for alpha in alphas:
        for beta in betas:
            if time.time() >= deadline:
                break
            a = (x ** alpha) * ((1.0 - x) ** beta)
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = cheap_score(a)
            if val < best_val:
                best_a, best_val = a, val
                print(f"[Init] beta(a={alpha:.3g},b={beta:.3g}) cheap = {val:.6f}")

    # ----- random Dirichlet seeds --------------------------------
    concs = [0.001, 0.01, 0.1, 0.5, 1.0, 2.0, 5.0]
    for conc in concs:
        if time.time() >= deadline:
            break
        g = np.random.gamma(conc, 1.0, size=n)
        a = g / g.sum()
        val = cheap_score(a)
        if val < best_val:
            best_a, best_val = a, val
            print(f"[Init] dirichlet(conc={conc}) cheap = {val:.6f}")

    # ----- mixture of exponential + uniform ----------------------
    for _ in range(30):
        if time.time() >= deadline:
            break
        lam = np.random.uniform(0.0, 5.0)
        mix = np.random.rand()  # weight of exponential component
        exp_part = np.exp(-lam * np.arange(n, dtype=np.float64))
        a = mix * exp_part + (1.0 - mix) * np.ones(n)
        a /= a.sum()
        val = cheap_score(a)
        if val < best_val:
            best_a, best_val = a, val
            print(f"[Init] mix(exp lam={lam:.3f},mix={mix:.3f}) cheap = {val:.6f}")

    print(f"[Init] best cheap after sweep = {best_val:.6f}")

    # --------------------------------------------------------------
    # 1. Exact sub‑gradient exponentiated‑gradient descent (with restarts)
    # --------------------------------------------------------------
    global_best_a = best_a.copy()
    global_best_val = best_val
    eta0 = 0.8                     # base stepsize (tuned empirically)
    max_total_iters = 3_000_000   # hard upper bound, rarely reached
    iter_counter = 0
    stagnation_counter = 0
    last_improve_iter = 0

    # Outer restart loop – each restart gets a fresh random seed (unless
    # it is the very first iteration where we start from the best sweep)
    restart_idx = 0
    while time.time() < deadline and restart_idx < 15:
        restart_idx += 1
        if restart_idx == 1 and best_a is not None:
            a = best_a.copy()
            a += np.random.rand(n) * 1e-5    # tiny perturbation
            a = _simplex_projection(a)
        else:
            # random Dirichlet start (choose concentration at random)
            conc = np.random.choice([0.001, 0.01, 0.1, 0.5, 1.0, 2.0, 5.0])
            g = np.random.gamma(conc, 1.0, size=n)
            a = g / g.sum()

        local_best_a = a.copy()
        maxb, _ = maxconv_and_grad(a)
        local_best_val = cheap_score(a)

        # Inner optimisation loop
        while time.time() < deadline and iter_counter < max_total_iters:
            iter_counter += 1
            maxb, grad = maxconv_and_grad(a)
            cheap = 2.0 * n * maxb

            # track the best ever seen
            if cheap < global_best_val - 1e-12:
                global_best_val = cheap
                global_best_a = a.copy()
                print(f"[Iter {iter_counter:07d}] NEW GLOBAL BEST cheap = {cheap:.6f}")

            # classic exponentiated‑gradient update (mirror descent on simplex)
            eta = eta0 / math.sqrt(iter_counter)
            log_a = np.log(a + eps) - eta * grad
            a = np.exp(log_a)
            s = a.sum()
            if s <= 0.0 or not np.isfinite(s):
                # fallback to the last known good point
                a = local_best_a.copy()
                continue
            a /= s

            # occasional local restart if no improvement for a while
            if iter_counter - last_improve_iter > 150_000:
                # perturbation around the best seen so far
                pert = np.random.rand(n) * 1e-4
                a = _simplex_projection(global_best_a + pert)
                last_improve_iter = iter_counter
                print(f"[Iter {iter_counter:07d}] Restart (stagnation).")
                continue

            # keep track of recent improvement to decide on early exit
            if cheap < local_best_val - 1e-12:
                local_best_val = cheap
                local_best_a = a.copy()
                last_improve_iter = iter_counter

            # early exit from the inner loop if we are close to the deadline
            if time.time() > deadline - 3.0:
                break

        # after inner loop we already have ``global_best_a`` possibly improved
        # proceed to the next restart (if time permits)

    # --------------------------------------------------------------
    # 2. Permutation / swap local search (simulated‑annealing style)
    # --------------------------------------------------------------
    a = global_best_a.copy()
    best_swap_a = a.copy()
    best_swap_val = cheap_score(a)
    temperature = 1.0
    cooling = 0.99997

    while time.time() < deadline:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # perform swap
        a[i], a[j] = a[j], a[i]

        cheap = cheap_score(a)
        accept = False
        if cheap < best_swap_val - 1e-12:
            accept = True
        else:
            delta = cheap - best_swap_val
            prob = math.exp(-delta / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            if cheap < best_swap_val - 1e-12:
                best_swap_val = cheap
                best_swap_a = a.copy()
                print(f"[Swap] NEW BEST cheap = {cheap:.6f}")
        else:
            # revert the swap
            a[i], a[j] = a[j], a[i]

        temperature = max(temperature * cooling, 1e-6)

    # --------------------------------------------------------------
    # 3. Final exact evaluation
    # --------------------------------------------------------------
    final_seq = best_swap_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    return final_seq
