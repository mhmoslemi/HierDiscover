def _run_greedy(init_seq, time_limit, seed=0):
    start = time.time()
    rng = np.random.default_rng(seed)
    a = np.asarray(init_seq, dtype=float)
    if a.sum() < 1e-12:
        a = np.ones_like(a) / a.size
    else:
        a = a / a.sum()
    n = a.shape[0]
    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())
    best_seq = a.copy()
    it = 0
    while time.time() - start < time_limit:
        it += 1
        conv = _conv_self_fft(a)
        M = conv.max()
        obj = float(2.0 * n * M)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_seq = a.copy()
            print(f"[Greedy it={it} O={best_obj:.6f}]")
        k_star = int(np.argmax(conv))
        # contributions
        i_vals = np.arange(n)
        j_vals = k_star - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i_vals[mask]] * a[j_vals[mask]]
        # choose source i with largest contribution
        if contrib.max() > 0:
            i = int(np.argmax(contrib))
        else:
            # no contributions? random
            i = rng.integers(n)
        # choose destination j with zero contribution if possible
        zero_mask = ~mask
        if zero_mask.any():
            zero_idxs = np.flatnonzero(zero_mask)
            j = int(rng.choice(zero_idxs))
        else:
            # all contribute, pick smallest contribution
            j = int(np.argmin(contrib))
        if i == j:
            # nothing to transfer
            continue
        # amount to transfer
        delta = a[i] * 0.03  # 3% of source mass
        if delta <= 0:
            continue
        # ensure we don't create negative
        a[i] -= delta
        a[j] += delta
        # renormalize occasionally to avoid drift
        if it % 500 == 0:
            a = np.clip(a, 0.0, None)
            a_sum = a.sum()
            if a_sum > 0:
                a /= a_sum
    # final compression
    thresh = 1e-12
    mask = best_seq > thresh
    if mask.sum() < best_seq.size:
        best_seq = best_seq[mask]
        best_seq = best_seq / best_seq.sum()
        best_obj = float(2.0 * best_seq.size * _conv_self_fft(best_seq).max())
        print(f"[Greedy compress] n={best_seq.size} O={best_obj:.6f}")
    return best_seq.tolist(), best_obj

"""
Non‑negative step‑function optimiser (multipeak Adam + greedy refinement).

The programme runs three stages inside the allotted time budget:

1. **Multipeak Adam** – a mirror‑descent Adam optimiser that uses the exact
   sub‑gradient of *all* convolution entries whose value is within a small
   tolerance of the current maximum.  This provides a smoother descent
   direction than the single‑peak version used previously and helps avoid
   getting stuck in plateaus.

2. **Greedy hill‑climbing** – a deterministic local‑search that repeatedly
   moves a small fraction of mass from the index that contributes most to
   the maximal convolution entry towards an index that does not contribute
   (or contributes very little).  The step is accepted only when it improves
   the objective, and the procedure stops as soon as the target bound
   (≈1.503) is reached.

3. **Optional simulated‑annealing** – a short refinement phase identical
   to the one from the reference implementation; it is only entered if
   time remains after the greedy phase.

All stages respect the simplex constraint (non‑negative entries, sum = 1)
by operating in log‑it space and using a soft‑max projection.  The
algorithm automatically adapts the sequence length to the available time,
trying larger supports (up to 50 k) when the budget permits.

The returned list is ready for ``evaluate_sequence``.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full self‑convolution a*a using a real FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return conv_full[: 2 * n - 1]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


def _grad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """
    Exact gradient of the k‑th convolution entry w.r.t. ``a``.
    Returns 2·a_{k‑i} for indices where it is defined, otherwise 0.
    """
    n = a.shape[0]
    idx = np.arange(n)
    j = k - idx
    mask = (j >= 0) & (j < n)
    g = np.zeros_like(a)
    if mask.any():
        g[mask] = 2.0 * a[j[mask]]
    return g


# ----------------------------------------------------------------------
# Multipeak Adam optimiser (hard‑max with a tolerance set)
# ----------------------------------------------------------------------


def _run_opt_multipeak(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
    target_obj: float = 1.503,
) -> Tuple[List[float], float]:
    """
    Adam optimiser that uses the exact sub‑gradient of *all* convolution
    entries within a relative tolerance of the current maximum.
    Returns the best sequence (as a list) and its objective value
    2·n·max(conv(a,a)).
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # initialise a on the simplex
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
    else:
        a = rng.random(n)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n, dtype=float) / n
    else:
        a = a / s

    # log‑it representation (mirror descent)
    x = np.log(a + 1e-16)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # learning‑rate schedule (exponential decay)
    lr_start = 0.6
    lr_end = 0.001

    best_a = a.copy()
    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())

    it = 0
    while time.time() - start < time_limit:
        it += 1

        # current point on the simplex
        a = _softmax(x)

        conv = _conv_self_fft(a)
        M = conv.max()
        obj = float(2.0 * n * M)

        # record improvement
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[Adam n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        # stop early if we have reached the target bound
        if best_obj <= target_obj:
            print("[Adam] target reached, exiting early")
            break

        # indices whose convolution value is very close to M
        delta = max(1e-12, M * 0.005)          # 0.5 % tolerance
        near_mask = conv >= (M - delta)
        # limit to at most 20 entries to keep gradient cheap
        if near_mask.sum() > 20:
            # quick selection of 20 largest entries
            top_idxs = np.argpartition(-conv, 20)[:20]
        else:
            top_idxs = np.flatnonzero(near_mask)

        # sum of exact sub‑gradients for the selected indices
        g_a = np.zeros_like(a)
        for k in top_idxs:
            g_a += _grad_max_conv(a, int(k))

        # natural gradient in log‑it space (mirror method)
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)

        # time‑based learning‑rate (exponential decay)
        elapsed = time.time() - start
        t = min(elapsed / time_limit, 1.0)
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # Adam step in log‑it space
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape flat regions
        if it % 4000 == 0:
            jitter = rng.normal(scale=0.01, size=n)
            x += jitter

        # occasional full restart from the current best (helps escape)
        if it % 70000 == 0:
            x = np.log(best_a + 1e-16) + rng.normal(scale=0.005, size=n)
            m.fill(0.0)
            v.fill(0.0)

    # ------------------------------------------------------------------
    # prune negligible entries (they increase n without helping)
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = float(2.0 * best_a.size * _conv_self_fft(best_a).max())
        print(f"[Adam compress] reduced length → {best_a.size}, O={best_obj:.6f}")

    return best_a.tolist(), best_obj


# ----------------------------------------------------------------------
# Greedy hill‑climbing refinement
# ----------------------------------------------------------------------


def _run_greedy(
    init_seq: List[float],
    time_limit: float,
    seed: int = 0,
    target_obj: float = 1.503,
) -> Tuple[List[float], float]:
    """
    Deterministic local search that repeatedly transfers a small fraction
    of mass from the most harmful index (largest contribution to the maximal
    convolution entry) to a harmless index.  Only moves that improve the
    objective are kept.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    a = np.asarray(init_seq, dtype=float)
    if a.sum() < 1e-12:
        a = np.ones_like(a) / a.size
    else:
        a = a / a.sum()
    n = a.shape[0]

    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())
    best_seq = a.copy()
    it = 0

    while time.time() - start < time_limit:
        it += 1
        conv = _conv_self_fft(a)
        M = conv.max()
        obj = float(2.0 * n * M)

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_seq = a.copy()
            print(f"[Greedy it={it:07d}] new best O={best_obj:.6f}")

        if best_obj <= target_obj:
            print("[Greedy] target reached, exiting early")
            break

        k_star = int(np.argmax(conv))

        # contributions of each index to the maximal entry
        i_vals = np.arange(n)
        j_vals = k_star - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i_vals[mask]] * a[j_vals[mask]]

        # source = index with largest contribution
        if contrib.max() > 0:
            i = int(np.argmax(contrib))
        else:
            i = rng.integers(n)

        # destination = index that does *not* contribute (or contributes minimally)
        zero_mask = ~mask
        if zero_mask.any():
            zero_idxs = np.flatnonzero(zero_mask)
            j = int(rng.choice(zero_idxs))
        else:
            j = int(np.argmin(contrib))

        if i == j:
            continue

        # transfer a modest fraction of mass from i to j
        delta = a[i] * 0.03   # 3 % of the source mass
        if delta <= 0.0:
            continue
        a[i] -= delta
        a[j] += delta

        # renormalise occasionally (prevents numerical drift)
        if it % 500 == 0:
            a = np.clip(a, 0.0, None)
            s = a.sum()
            if s > 0:
                a /= s

    # final compression of tiny entries
    thresh = 1e-12
    mask = best_seq > thresh
    if mask.sum() < best_seq.size:
        best_seq = best_seq[mask]
        best_seq = best_seq / best_seq.sum()
        best_obj = float(2.0 * best_seq.size * _conv_self_fft(best_seq).max())
        print(f"[Greedy compress] reduced length → {best_seq.size}, O={best_obj:.6f}")

    return best_seq.tolist(), best_obj


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (kept unchanged from reference)
# ----------------------------------------------------------------------


def _run_sa(
    init_seq: List[float],
    time_limit: float,
    seed: int = 0,
) -> Tuple[List[float], float]:
    """Refine a sequence by small mass‑transfer moves with a Metropolis schedule."""
    start = time.time()
    rng = np.random.default_rng(seed)
    a = np.asarray(init_seq, dtype=float)
    if a.sum() < 1e-12:
        a = np.ones_like(a) / a.size
    else:
        a = a / a.sum()
    n = a.shape[0]

    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())
    best_seq = a.copy()

    # temperature schedule
    T = best_obj * 0.1 + 1e-6
    T_min = best_obj * 1e-6 + 1e-9
    decay = 0.995
    it = 0

    while time.time() - start < time_limit:
        it += 1
        k_max = int(np.argmax(conv))

        # contribution of each index to the maximal entry
        i_vals = np.arange(n)
        j_vals = k_max - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i_vals[mask]] * a[j_vals[mask]]
        total_contrib = contrib.sum()

        if total_contrib > 0:
            probs = contrib / total_contrib
            i = rng.choice(n, p=probs)
        else:
            i = rng.integers(n)

        # choose a distinct destination uniformly
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # transfer a modest fraction of mass from i to j
        delta = a[i] * rng.random() * 0.05  # up to 5 % of a[i]
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.clip(a_new, 0.0, None)
        a_new /= a_new.sum()

        conv_new = _conv_self_fft(a_new)
        obj_new = float(2.0 * n * conv_new.max())

        # Metropolis acceptance
        if obj_new < best_obj - 1e-12:
            a, conv, best_obj, best_seq = a_new, conv_new, obj_new, a_new.copy()
            print(f"[SA it={it}] new best O={best_obj:.6f}")
        else:
            prob = np.exp(-(obj_new - best_obj) / T) if T > 0 else 0.0
            if rng.random() < prob:
                a, conv = a_new, conv_new

        # temperature decay
        if it % 2000 == 0:
            T = max(T * decay, T_min)

        # occasional global jitter
        if it % 10000 == 0:
            noise = rng.normal(scale=0.01, size=n)
            a = np.clip(a + noise, 0.0, None)
            a /= a.sum()
            conv = _conv_self_fft(a)

    # final compression of tiny entries
    thresh = 1e-12
    mask = best_seq > thresh
    if mask.sum() < best_seq.size:
        comp = best_seq[mask]
        comp /= comp.sum()
        comp_obj = float(2.0 * comp.size * _conv_self_fft(comp).max())
        if comp_obj < best_obj:
            best_seq, best_obj = comp, comp_obj
            print(f"[SA compress] reduced length → {comp.size}, O={best_obj:.6f}")

    return best_seq.tolist(), best_obj

def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **_ignore,
) -> List[float]:
    """
    Find a non‑negative step‑function minimizing the evaluation metric.
    The routine splits the given time budget into three phases:
      1) multipeak Adam optimisation (≈60 % of time),
      2) greedy hill‑climbing (≈30 %),
      3) optional simulated annealing (≈remaining time).
    The target bound is 1.503; the search stops early if it is reached.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.2  # tiny safety margin

    # ------------------------------------------------------------------
    # Initialise from a previously discovered construction if available.
    init_seq = None
    if "height_sequence_1" in globals():
        seq0 = np.asarray(height_sequence_1, dtype=float)
        seq0 = np.clip(seq0, 0.0, None)
        if seq0.sum() < 0.01:
            seq0 = np.ones_like(seq0)
        seq0 = seq0 / seq0.sum()
        init_seq = seq0

    # ------------------------------------------------------------------
    # Choose a target support length based on the remaining budget.
    remaining = max(0.0, deadline - time.time())
    # Heuristic: larger budget → larger support (up to 50 k)
    base_time_for_12k = 250.0               # empirical time for n≈12000
    scale = max(1.0, remaining / base_time_for_12k)
    target_n = int(min(50000, max(8000, 12000 * scale)))
    print(f"[setup] target_n={target_n}, remaining={remaining:.1f}s")

    # Upsample any initial sequence to the target length.
    if init_seq is not None and init_seq.size != target_n:
        x_old = np.linspace(0.0, 1.0, init_seq.size, endpoint=False)
        x_new = np.linspace(0.0, 1.0, target_n, endpoint=False)
        init_a = np.interp(x_new, x_old, init_seq)
        init_a = np.clip(init_a, 0.0, None)
        init_a = init_a / init_a.sum()
    else:
        init_a = init_seq

    # ------------------------------------------------------------------
    # Phase 1: multipeak Adam optimisation
    t_now = time.time()
    main_budget = (deadline - t_now) * 0.60
    seq, obj = _run_opt_multipeak(
        n=target_n,
        time_limit=main_budget,
        init_a=init_a,
        seed=seed + 123,
    )
    best_seq = np.asarray(seq, dtype=float)
    best_obj = obj
    print(f"[global] after Adam best O={best_obj:.6f} (n={target_n})")

    # ------------------------------------------------------------------
    # Phase 2: greedy hill‑climbing
    t_now = time.time()
    greedy_budget = (deadline - t_now) * 0.30
    if greedy_budget > 2.0:  # need a minimal amount of time
        seq_g, obj_g = _run_greedy(
            best_seq.tolist(),
            greedy_budget,
            seed=seed + 456,
        )
        if obj_g < best_obj:
            best_obj, best_seq = obj_g, np.asarray(seq_g, dtype=float)
            print(f"[global] after Greedy new best O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Phase 3: short simulated‑annealing refinement (if time permits)
    t_now = time.time()
    remaining = deadline - t_now
    if remaining > 1.0:
        seq_sa, obj_sa = _run_sa(best_seq.tolist(), remaining, seed=seed + 999)
        if obj_sa < best_obj:
            best_obj, best_seq = obj_sa, np.asarray(seq_sa, dtype=float)
            print(f"[global] after SA new best O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check – call the supplied evaluator.
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_seq.tolist()
