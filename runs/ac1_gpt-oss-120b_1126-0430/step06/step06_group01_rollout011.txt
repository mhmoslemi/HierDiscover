def propose_candidate(seed=42, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0

    # Determine n
    try:
        n_ex = len(height_sequence_1)  # external.
        n = min(32768, n_ex)
    except Exception:
        n = 32768
    print(f"Using n = {n}")

    # compute arcsine distribution
    i = np.arange(n, dtype=np.float64)
    x0 = i / n
    x1 = (i+1) / n
    sqrt0 = np.sqrt(np.clip(x0, 0.0, 1.0))
    sqrt1 = np.sqrt(np.clip(x1, 0.0, 1.0))
    F0 = (2.0 / np.pi) * np.arcsin(sqrt0)
    F1 = (2.0 / np.pi) * np.arcsin(sqrt1)
    a = F1 - F0
    a_sum = a.sum()
    a = a / a_sum  # ensure sum=1

    # Initialize logits
    z = _logits_from_distribution(a)
    # enforce symmetry
    z = (z + z[::-1]) / 2.0

    # Adam buffers
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_alpha = 0.6  # can be maybe 0.7
    eps_factor = 0.2  # start smoothing factor
    eps_factor_min = 1e-6
    decay = 0.9997

    best_score = float('inf')
    best_a = None
    it = 0

    # optional early evaluation of initial
    max_conv0, _ = _conv_max_and_grad(a, eps_factor)  # but eps_factor may be large modifies metric; better compute exact metric via evaluate? but we just compute cheap_score.
    best_score = 2.0 * n * max_conv0
    best_a = a.copy()
    print(f"Initial cheap score = {best_score:.8f}")

    while time.time() < deadline:
        it += 1
        a_cur = _softmax(z)  # shape (n,)

        # enforce symmetry (optional for a_cur)
        a_cur = (a_cur + a_cur[::-1]) / 2.0

        max_conv, grad_a = _conv_max_and_grad(a_cur, eps_factor)

        cheap_score = 2.0 * n * max_conv

        if cheap_score < best_score:
            best_score = cheap_score
            best_a = a_cur.copy()
            if it % 1000 == 0:
                print(f"Iter {it:6d}: improved score = {best_score:.8f}")

        # compute gradient wrt logits
        mu = np.dot(grad_a, a_cur)
        grad_z = a_cur * (grad_a - mu)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step = - alpha_t * m_hat / (np.sqrt(v_hat) + eps_adam)
        z = z + step

        # enforce symmetry on logits
        z = (z + z[::-1]) / 2.0

        # clip logits
        z = np.clip(z, -60.0, 60.0)

        # decay epsilon factor
        eps_factor = max(eps_factor_min, eps_factor * decay)

        # occasional jitter
        if it % 10000 == 0 and np.random.rand() < 0.05:
            jitter = np.random.randn(n) * 0.002
            z = z + jitter
            # re-symmetrize after jitter
            z = (z + z[::-1]) / 2.0

        # check time every e.g., 1000 iterations
        if it % 2000 == 0 and time.time() >= deadline:
            break

    final_seq = best_a.tolist()
    # final exact evaluation
    try:
        exact = evaluate_sequence(final_seq)
        print(f"Exact evaluation: {exact:.8f}")
    except Exception as e:
        print("Evaluation failed", e)
    return final_seq

#!/usr/bin/env python
"""
Propose a step‑function (non‑negative height sequence) that minimises

    2·n·max(a*a) / (sum a)²

The code builds a symmetric arcsine distribution on a uniform grid (the known
continuous optimum) and then refines it on the simplex using Adam on the
log‑softmax parameters.  The optimisation uses a smoothed version of the
max‑convolution (soft‑max over the convolution entries) whose temperature is
decreased steadily.  After each Adam update the logits are symmetrised, which
keeps the candidate distribution symmetric – a property of the true optimum.
The best sequence seen during the timed run is returned.

The implementation is self‑contained, uses only NumPy and the provided
`evaluate_sequence` function, and respects the 1000 s time budget.
"""

import time
import math
import numpy as np

# ------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ------------------------------------------------------------

def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max, returns a probability vector (sum = 1)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑softmax inverse: gives logits that map to `a` via soft‑max."""
    return np.log(a + eps)


def _fft_len_for_conv(n: int) -> int:
    """
    Smallest power‑of‑two ≥ 2·n‑1.
    Sufficient for a linear self‑convolution of length (2·n‑1).
    """
    return 1 << ((2 * n - 1).bit_length())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Compute the (smoothed) maximum of the self‑convolution a*a and its gradient.

    Parameters
    ----------
    a : np.ndarray
        Probability vector (non‑negative, sum = 1).
    eps_factor : float
        Multiplicative factor for the smoothing constant:
        eps = max(eps_factor * max_conv, 1e-12).

    Returns
    -------
    max_conv : float
        Maximum entry of the linear self‑convolution.
    grad_a : np.ndarray
        Gradient of 2·n·smooth_max w.r.t. a (already multiplied by 2·n).
    """
    n = a.shape[0]
    L = _fft_len_for_conv(n)

    # FFT of a
    a_fft = np.fft.rfft(a, L)

    # Linear self‑convolution (length 2·n‑1)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # Smoothed soft‑max weights
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv               # improve stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum

    # Gradient of smooth max: 2·n·(a ⋆ w)
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]
    grad_a = 2.0 * n * corr                # gradient of 2·n·smooth_max
    return max_conv, grad_a

    Return a list of non‑negative heights that (approximately) minimise the
    evaluation metric.  The routine respects the time budget, prints progress,
    and finally returns the best sequence it discovered.
    """
    np.random.seed(seed)
    start = time.time()
    # small safety margin for the final print/evaluation
    deadline = start + budget_s - 2.0

    # --------------------------------------------------------
    # 0. Choose the length of the vector.
    # --------------------------------------------------------

        n = 32768

    # --------------------------------------------------------
    # 1. Exact arcsine discretisation (probability vector, sum = 1).
    # --------------------------------------------------------
    i = np.arange(n, dtype=np.float64)
    x0 = i / n
    x1 = (i + 1) / n

    # CDF of the arcsine law: F(x) = (2/π)·arcsin(√x)
    sqrt0 = np.sqrt(np.clip(x0, 0.0, 1.0))
    sqrt1 = np.sqrt(np.clip(x1, 0.0, 1.0))
    F0 = (2.0 / np.pi) * np.arcsin(sqrt0)
    F1 = (2.0 / np.pi) * np.arcsin(sqrt1)

    a = F1 - F0                     # bin masses, sum ≈ 1
    a_sum = a.sum()
    if not np.isfinite(a_sum) or a_sum <= 0.0:
        raise RuntimeError("Failed to generate arcsine bins.")
    a /= a_sum                      # enforce exact normalisation (sum = 1)

    # --------------------------------------------------------
    # 2. Initialise Adam on the log‑softmax parameters.
    # --------------------------------------------------------
    z = _logits_from_distribution(a)          # logits → soft‑max = a
    # enforce symmetry of the logits (the optimum is symmetric)

    m = np.zeros_like(z)          # first moment

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_alpha = 0.6               # base learning‑rate factor

    eps_factor_min = 1e-6

    # --------------------------------------------------------

    best_a = a.copy()
    max_conv0, _ = _conv_max_and_grad(a, eps_factor)
    best_score = 2.0 * n * max_conv0
    print(f"Initial cheap score (2·n·max_conv) = {best_score:.8f}")

    it = 0
    max_iters = 500_000  # generous upper bound; loop exits on timeout

        # Current probability vector (symmetric by construction)
        a_cur = _softmax(z)

        # Smoothed max‑convolution and its gradient w.r.t. a
        max_conv, grad_a = _conv_max_and_grad(a_cur, eps_factor)
        cheap_score = 2.0 * n * max_conv

            best_a = a_cur.copy()
            if it % 1000 == 0:

        mu = np.dot(grad_a, a_cur)
        grad_z = a_cur * (grad_a - mu)

        # Adam bias‑corrected update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)

        # Step‑size schedule (≈ 1/√it)

        step = -alpha_t * m_hat / (np.sqrt(v_hat) + eps_adam)
        z = z + step

        # Enforce symmetry after the update
        z = (z + z[::-1]) / 2.0

        # Keep logits in a numerically safe range
        z = np.clip(z, -60.0, 60.0)

        eps_factor = max(eps_factor_min, eps_factor * eps_decay)

            jitter = 0.004 * np.random.randn(n)
            z = z + jitter
            z = (z + z[::-1]) / 2.0  # re‑symmetrise

        # Periodically check the wall‑clock (cheap compared to FFT work)
        if it % 5_000 == 0 and time.time() >= deadline:

    # 4. Return the best sequence found (as a plain Python list).
    #    One final exact evaluation is logged – it does not affect the answer.
    # --------------------------------------------------------

    except Exception as exc:
        print(f"Exact evaluation failed: {exc}")

    return final_seq
