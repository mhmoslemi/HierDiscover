def propose_candidate(seed=42, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0    # buffer for final evaluation

    # Determine base n for scanning and optimization.
    n_init = 5000

    best_seq = None
    best_score = float('inf')
    # try using global if exists
    try:
        init_global = np.array(height_sequence_1, dtype=np.float64) # provided global
        if init_global.ndim == 1:
            # downsample to n_init
            cand = _interpolate_distribution(init_global, n_init)
            score = 2 * n_init * _conv_max(cand)
            best_seq, best_score = cand, score
            print("[Init] Warm start from global sequence")
    except Exception:
        pass

    # families scanning function
    def try_candidate(cand):
        nonlocal best_seq, best_score
        # normalize to sum 1
        s = cand.sum()
        if s == 0.0:
            return
        cand = cand / s
        score = 2 * n_init * _conv_max(cand)
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(f"[Scan] New best score {score:.6f}")

    # Define families and parameter ranges:

    # Beta symmetric: alpha in [0.2, 5], maybe 50 steps
    for alpha in np.linspace(0.2,5.0,45):
        if time.time() >= deadline: break
        cand = _beta_distribution(n_init, alpha, alpha)
        try_candidate(cand)

"""
Propose a non‑negative step‑function that minimises
    2·n·max(conv(a,a)) / (sum(a))²

The routine works in three stages:

1. **Broad parametric scan** – generate candidate shapes from several
   families (Beta, sine‑power, Gaussian, triangular, also a few hand‑crafted
   families).  All candidates are normalised to sum = 1 and evaluated
   cheaply with an FFT‑based convolution.  The best candidate so far is kept.

2. **Mirror‑descent (Adam) on logits** – the best candidate is turned into
   soft‑max logits (so the variables live on the simplex).  A smoothed
   max‑operator is used to obtain a differentiable objective and its gradient.
   Adam with a back‑tracking line‑search refines the distribution for many
   thousands of iterations.

3. **Upsampling & final refinement** – if time permits the distribution is
   interpolated to a larger support (up to ≈ 20 000 points) and a second
   Adam phase runs to exploit the finer resolution.

The final sequence (list of floats) is returned and also evaluated with the
provided ``evaluate_sequence`` for reporting.  The whole routine respects the
given time budget (default 1000 s) and never uses closures or lambdas.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max; returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    s = e.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / z.size)
    return e / s


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Logits that reproduce ``a`` after a soft‑max."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """Smallest power‑of‑two ≥ 3·n‑2 (enough for all needed convolutions)."""
    return 1 << ((3 * n - 2).bit_length())


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    s = new_a.sum()
    if s == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= s
    return new_a


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a (FFT‑based)."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Returns (max_conv, grad_a) where

        grad_a = ∇_a [ 2·n·smooth_max(a*a) ]

    ``eps_factor`` multiplies the smoothing constant ε = eps_factor·max_conv.
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    # FFT of a (zero‑padded)
    a_fft = np.fft.rfft(a, L)

    # Linear self‑convolution a*a
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # Smoothed soft‑max weights
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv                # ≤ 0 for stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # Correlation a ⋆ w
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # Gradient (already multiplied by 4·n)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a

def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric or asymmetric beta‑pdf discretised on n points."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n          # avoid exact 0/1
    a = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _sin_power_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ sin(π·x)^p, x∈(0,1), symmetric and zero at the edges."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.sin(np.pi * x) ** p
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _gaussian_distribution(n: int, sigma_frac: float) -> np.ndarray:
    """Discrete centred Gaussian; sigma = sigma_frac·n."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = max(sigma_frac, 1e-12) * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _triangular_distribution(n: int, width_frac: float) -> np.ndarray:
    """Symmetric triangular bump; support width = width_frac·n."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    max_dist = max(width_frac, 1e-12) * centre
    dist = np.abs(i - centre)
    a = np.maximum(0.0, 1.0 - dist / max_dist)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a

        if time.time() >= deadline:
            break
        cand = _gaussian_distribution(n, sigma_frac)
        evaluate_and_maybe_update(cand)

    # ----- Triangular family -----
    for w_frac in np.linspace(0.2, 1.0, 25):
        if time.time() >= deadline:
            break
        cand = _triangular_distribution(n, w_frac)
        evaluate_and_maybe_update(cand)

    # ----- Random perturbations of current best -----
    for _ in range(120):
        if time.time() >= deadline:
            break
        if best_seq is None:
            cand = np.random.rand(n)
        else:
            noise = np.random.randn(n) * 0.02
            cand = best_seq + noise
        cand = np.maximum(cand, 0.0)
        if cand.sum() == 0.0:
            cand = np.full(n, 1.0 / n)
        evaluate_and_maybe_update(cand)

    # ------------------------------------------------------------------
    # 2. Adam‑style mirror‑descent on the simplex (soft‑max logits).
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    # Initialise logits with a small random jitter.
    z = _logits_from_distribution(best_seq) + 0.01 * np.random.randn(n)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    base_alpha = 0.45          # base learning‑rate for logits
    eps_factor = 0.15          # initial smoothing factor
    eps_factor_min = 5e-5
    eps_factor_decay = 0.9993

    best_seq_opt = best_seq.copy()
    best_score_opt = best_score
    last_improve_iter = 0
    it = 0
    report_every = 25000

    while time.time() < deadline and it < 3_000_000:
        it += 1

        # Current point on the simplex.
        a = _softmax(z)

        # Cheaper evaluation + smoothed gradient.
        max_conv, grad_a = _conv_max_and_grad(a, eps_factor)
        cheap_score = 2.0 * n * max_conv

        # Keep the best ever seen.
        if cheap_score < best_score_opt - 1e-12:
            best_score_opt = cheap_score
            best_seq_opt = a.copy()
            last_improve_iter = it
            if it % report_every == 0:
                print(f"[Iter {it:07d}] NEW BEST cheap = {cheap_score:.6f}")

        # Gradient w.r.t. logits (mirror‑descent trick).
        mu = np.dot(grad_a, a)          # <grad_a , a>
        grad_z = a * (grad_a - mu)

        # Adam moment updates (bias‑corrected).
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # Step‑size schedule (inverse‑sqrt decay).
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step_dir = - m_hat / (np.sqrt(v_hat) + eps_adam)

        # Simple back‑tracking line search (up to 5 halvings).
        trial_alpha = alpha_t
        accepted = False
        for _ in range(5):
            z_new = z + trial_alpha * step_dir
            z_new = np.clip(z_new, -50.0, 50.0)      # keep exp() safe
            a_new = _softmax(z_new)
            new_score = 2.0 * n * _conv_max(a_new)
            if new_score <= cheap_score:
                z = z_new
                accepted = True
                break
            trial_alpha *= 0.5

        if not accepted:
            # No improvement – take a modest adam step anyway.
            z = z + alpha_t * step_dir
            z = np.clip(z, -50.0, 50.0)

        # Gradually tighten the smoothing of the max‑operator.
        eps_factor = max(eps_factor_min, eps_factor * eps_factor_decay)

        # Random jitter restart if we are stuck for a long time.
        if it - last_improve_iter > 350_000:
            print(f"[Iter {it}] Stagnation – jitter restart.")
            z = _logits_from_distribution(best_seq_opt) + 0.03 * np.random.randn(n)
            m.fill(0.0)
            v.fill(0.0)
            eps_factor = 0.15
            last_improve_iter = it

        # ------------------------------------------------------------------
        # 2a. Optional up‑sampling once (if enough budget remains).
        # ------------------------------------------------------------------
        time_left = deadline - time.time()
        if n < 20000 and time_left > 30.0 and (it % 250_000 == 0 or it == 500_000):
            new_n = min(20000, int(n * 1.6))
            print(f"[Iter {it}] Upsampling support: {n} → {new_n}")
            # Interpolate the best sequence found so far to the larger grid.
            best_seq_opt = _interpolate_distribution(best_seq_opt, new_n)
            # Re‑initialise optimisation state on the enlarged grid.
            z = _logits_from_distribution(best_seq_opt) + 0.01 * np.random.randn(new_n)
            m = np.zeros_like(z)
            v = np.zeros_like(z)
            n = new_n
            # Reset counters for the enlarged problem.
            it = 0
            last_improve_iter = 0
            eps_factor = 0.15
            best_score_opt = 2.0 * n * _conv_max(best_seq_opt)
            # continue optimisation on the larger grid
            continue

    # ------------------------------------------------------------------
    # 3. Final exact evaluation with the supplied ``evaluate_sequence``.
    # ------------------------------------------------------------------
    final_seq = best_seq_opt.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:
        final_score = None
        print(f"Error during final evaluation: {exc}")

    # ------------------------------------------------------------------
    # 4. Return the best sequence discovered.
    # ------------------------------------------------------------------
    return final_seq
