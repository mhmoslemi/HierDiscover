def _upsample_to_length(seq, n):
    seq = np.asarray(seq, dtype=float)
    m = seq.size
    if m == n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, n)
    seq_new = np.interp(x_new, x_old, seq)
    # ensure non-negative
    seq_new = np.clip(seq_new, 0.0, None)
    # normalize
    total = seq_new.sum()
    if total > 0:
        seq_new /= total
    else:
        seq_new = np.full(n, 1.0 / n)
    return seq_new
```

"""
Propose a step‑function (non‑negative sequence) that minimises
    2·n·max(conv(a,a)) / (sum(a)^2)

The optimiser works in three conceptual phases:

1.  **Shape pre‑search** – a modest collection of analytically motivated
    candidates (uniform, exponential, Gaussian, triangular, cosine,
    beta‑distribution, sparse Dirichlet, fully random) is generated for a
    chosen support size (default 2000).  The candidate with the smallest
    exact objective is taken as the initial point.

2.  **Adam on the simplex** – we optimise the log‑it (soft‑max) parametrisation
    of the probability simplex.  A temperature‑controlled soft‑max (β)
    approximates the max‑operator; β grows exponentially from 0.5 to 1e7,
    while the learning‑rate decays from 0.5 to 0.001.  Adam’s first‑ and
    second‑moment estimates together with occasional jitter and periodic
    restarts keep the search robust.

3.  **Fine‑grained refinement** – a short projected‑gradient descent on the
    exact max‑gradient (hard‑max) followed by a targeted “mass‑transfer”
    simulated‑annealing style move set (move a small fraction of mass from
    indices that contribute most to the current maximal autocorrelation
    entry to randomly chosen destinations).  The procedure is repeated at
    increasingly larger support sizes by up‑sampling the current best shape.

All work respects a global time budget (default 1000 s).  The routine
returns a plain list of non‑negative floats ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear self‑convolution a*a using a real FFT.
    Returns the first (2·len(a)‑1) entries.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return conv_full[: 2 * n - 1]


def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation of a with weight vector w:
        g[i] = Σ_t w[i+t]·a[t]   for i=0,…,len(a)-1
    Implemented via FFT (real‑valued).
    """
    n = a.shape[0]
    m = w.shape[0]                     # = 2n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x ≥ 0, Σx = 1}.
    O(n log n) algorithm (Duchi et al.).
    """
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def compute_objective(a: np.ndarray) -> float:
    """
    Exact objective for a normalised (sum=1) non‑negative vector a:
        O = 2·n·max(conv(a,a))
    """
    a = np.asarray(a, dtype=float)
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    a = a / s
    conv = _conv_self_fft(a)
    return float(2.0 * a.size * conv.max())


# ----------------------------------------------------------------------
# Shape candidates (Stage 1)
# ----------------------------------------------------------------------


def _shape_candidates(n: int, rng: np.random.Generator):
    """Create a modest library of analytical shapes for a given support length."""
    cand = []

    # Uniform
    cand.append(np.full(n, 1.0 / n))

    idx = np.arange(n, dtype=float)

    # Exponential decay (both directions)
    for r in np.linspace(0.90, 0.9997, 12):
        a = r ** idx
        a = a / a.sum()
        cand.append(a)
        cand.append(a[::-1])

    # Gaussian windows (different sigmas)
    centre = (n - 1) / 2.0
    for sigma in np.linspace(0.10 * n, 0.40 * n, 8):
        a = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
        a = a / a.sum()
        cand.append(a)

    # Triangular / tent shape
    tri = np.minimum(idx + 1, n - idx)
    tri = tri / tri.sum()
    cand.append(tri)

    # Raised cosine (zero at ends, peak in centre)
    x = np.linspace(0, 1, n)
    cos = 0.5 * (1 - np.cos(2 * np.pi * x))
    cos = cos / cos.sum()
    cand.append(cos)

    # Beta‑distribution shapes (different α,β)
    alpha_vals = [0.5, 1.0, 2.0, 5.0]
    beta_vals = [0.5, 1.0, 2.0, 5.0]
    for a in alpha_vals:
        for b in beta_vals:
            pdf = (x ** (a - 1)) * ((1 - x) ** (b - 1))
            if pdf.sum() > 0:
                pdf = pdf / pdf.sum()
                cand.append(pdf)

    # Sparse Dirichlet‑like (few non‑zero entries)
    k = max(2, int(0.025 * n))
    for _ in range(3):
        a = np.zeros(n, dtype=float)
        pos = rng.choice(n, size=k, replace=False)
        vals = rng.random(k)
        a[pos] = vals
        a = a / a.sum()
        cand.append(a)

    # Pure random Dirichlet (α=1)
    a = rng.random(n)
    a = a / a.sum()
    cand.append(a)

    # Random Dirichlet with varied concentration
    for conc in [0.1, 0.3, 0.7, 2.0]:
        a = rng.gamma(conc, size=n)
        a = a / a.sum()
        cand.append(a)

    return cand


def _best_initial_sequence(n: int, rng: np.random.Generator, time_limit: float):
    """
    Scan a modest set of analytically motivated shapes and return the best.
    Aborts early if the time budget expires.
    """
    start = time.time()
    best_a = None
    best_obj = float("inf")
    for a in _shape_candidates(n, rng):
        if time.time() - start > time_limit:
            break
        obj = compute_objective(a)
        if obj < best_obj:
            best_obj, best_a = obj, a
    if best_a is None:
        # fallback – simple Dirichlet
        a = rng.random(n)
        a = a / a.sum()
        best_a = a
        best_obj = compute_objective(a)
    return best_a, best_obj

# Adam optimiser on the simplex (Stage 2)
# ----------------------------------------------------------------------


def _run_opt(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
) -> tuple[list[float], float]:
    """
    Adam optimiser on the log‑it simplex.
    Returns the best sequence (as a list) and its exact objective.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # initialise a on the simplex
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.shape[0] != n:
            a = rng.random(n)
    else:
        a = rng.random(n)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-12:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s

    # log‑it parametrisation
    x = np.log(a + 1e-16)
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # schedule limits
    beta_start, beta_end = 0.5, 1e7
    lr_start, lr_end = 0.5, 0.001

    best_a = a.copy()
    best_obj = compute_objective(best_a)

    it = 0
    while time.time() - start < time_limit:
        it += 1

        # stay on the simplex
        a = _softmax(x)

        conv = _conv_self_fft(a)
        M = conv.max()
        obj = 2.0 * n * M

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        # ---- schedules (time‑based) ---------------------------------
        elapsed = time.time() - start
        t = min(elapsed / time_limit, 1.0)

        beta = beta_start * np.exp(np.log(beta_end / beta_start) * t)
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t)

        # ---- smooth‑max weights ------------------------------------
        if beta > 5e5:                              # effectively hard max
            k_max = int(np.argmax(conv))
            w = np.zeros_like(conv)
            w[k_max] = 1.0
        else:
            shifted = conv - M                      # ≤ 0
            w = np.exp(beta * shifted)
            s_w = w.sum()
            if s_w == 0.0:
                w = np.ones_like(conv) / conv.size
            else:
                w /= s_w

        # ---- gradient in the original space -------------------------
        g_a = 2.0 * _correlation_fft(w, a)

        # ---- transform to log‑it space (natural gradient) ----------
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)

        # ---- Adam update -------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---- occasional jitter --------------------------------------
        if it % 2000 == 0:
            jitter = rng.normal(scale=0.01, size=n)
            x += jitter

        # ---- periodic restart from the current best -----------------
        if it % 50000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.005, size=n)
            m.fill(0.0)
            v.fill(0.0)

    # compress near‑zero entries (helps the final objective)
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = compute_objective(best_a)
        print(f"[compress] length → {best_a.size}, O={best_obj:.6f}")

    return best_a.tolist(), best_obj


# ----------------------------------------------------------------------
# Fine‑grained refinement (Stage 3)
# ----------------------------------------------------------------------


def _projected_gradient_refine(a: np.ndarray, time_limit: float):
    """
    Projected‑gradient descent on the exact max‑gradient.
    Returns a refined sequence (list) and its objective.
    """
    start = time.time()
    a = np.asarray(a, dtype=float)
    if a.sum() < 1e-12:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / a.sum()
    n = a.size
    best_a = a.copy()
    best_obj = compute_objective(best_a)

    eta = 0.5                     # initial step size
    it = 0
    while time.time() - start < time_limit:
        it += 1
        conv = _conv_self_fft(best_a)
        k_max = int(np.argmax(conv))

        # hard max weight
        w = np.zeros_like(conv)
        w[k_max] = 1.0
        grad = 2.0 * _correlation_fft(w, best_a)

        a_candidate = best_a - eta * grad
        a_candidate = _project_simplex(a_candidate)

        obj_cand = compute_objective(a_candidate)
        if obj_cand < best_obj - 1e-14:
            best_a, best_obj = a_candidate, obj_cand
            # enlarge step size when we make progress
            eta = min(eta * 1.2, 1.0)
        else:
            eta *= 0.7
            if eta < 1e-6:
                break

    return best_a.tolist(), best_obj


def _targeted_local_moves(a: np.ndarray, time_limit: float, rng: np.random.Generator):
    """
    Light‑weight simulated‑annealing style mass‑transfer moves that
    preferentially pull mass away from indices that contribute most to
    the current maximal autocorrelation entry.
    """
    start = time.time()
    a = np.asarray(a, dtype=float)
    a = np.clip(a, 0.0, None)
    if a.sum() < 1e-12:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / a.sum()
    n = a.size

    conv = _conv_self_fft(a)
    best_obj = compute_objective(a)
    best_a = a.copy()

    # temperature schedule (very gentle – we mostly accept improvements)
    T = best_obj * 0.2 + 1e-9
    T_min = best_obj * 1e-6 + 1e-12
    decay = 0.997

    it = 0
    while time.time() - start < time_limit:
        it += 1
        k_max = int(np.argmax(conv))

        # contribution of each i to the current maximal entry
        i_idx = np.arange(n)
        j_idx = k_max - i_idx
        mask = (j_idx >= 0) & (j_idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i_idx[mask]] * a[j_idx[mask]]
        total = contrib.sum()

        if total > 0:
            probs = contrib / total
            i = rng.choice(n, p=probs)
        else:
            i = rng.integers(n)

        # pick a destination different from i uniformly
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # move a modest fraction of mass
        delta = a[i] * rng.random() * 0.07
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.clip(a_new, 0.0, None)
        a_new /= a_new.sum()

        conv_new = _conv_self_fft(a_new)
        obj_new = compute_objective(a_new)

        if obj_new < best_obj - 1e-14:
            a, conv, best_obj, best_a = a_new, conv_new, obj_new, a_new.copy()
            # cooling when we improve
            T = max(T * decay, T_min)
        else:
            # Metropolis acceptance
            if rng.random() < math.exp(-(obj_new - best_obj) / T):
                a, conv = a_new, conv_new

        if it % 2500 == 0:
            T = max(T * decay, T_min)

        if it % 8000 == 0:
            # occasional global jitter
            noise = rng.normal(scale=0.008, size=n)
            a = np.clip(a + noise, 0.0, None)
            a /= a.sum()
            conv = _conv_self_fft(a)

    return best_a.tolist(), best_obj


def _upsample_to_length(seq: np.ndarray, n: int) -> np.ndarray:
    """
    Linear interpolation up‑sample / down‑sample a 1‑D sequence to length n.
    The result is re‑normalised to sum = 1.
    """
    seq = np.asarray(seq, dtype=float)
    m = seq.size
    if m == n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, n)
    seq_new = np.interp(x_new, x_old, seq)
    seq_new = np.clip(seq_new, 0.0, None)
    total = seq_new.sum()
    if total > 0:
        seq_new /= total
    else:
        seq_new = np.full(n, 1.0 / n)
    return seq_new

# Main entry point

def propose_candidate(seed: int = 42, budget_s: int = 1000, **_ignore) -> list[float]:
    """
    Main optimisation routine.
    Searches sequentially over a base support size and up‑samples the current
    best shape to larger sizes whenever time permits.
    Returns a list of non‑negative heights ready for ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin
    best_seq = None
    best_obj = float("inf")

    # ------------------------------------------------------------------
    # (Optional) start from the previously public construction if available
    # ------------------------------------------------------------------
    init_from_public = None
    try:
        # ``height_sequence_1`` is injected into the global namespace by the harness
        init_from_public = np.asarray(height_sequence_1, dtype=float)
        if init_from_public.ndim != 1 or init_from_public.size < 10:
            init_from_public = None
    except Exception:
        init_from_public = None

    # ------------------------------------------------------------------
    # Base support size – we start with a reasonably large length that
    # gave good results in the reference implementation.
    # ------------------------------------------------------------------
    base_n = 2000
    remaining = deadline - time.time()
    if remaining < 10.0:
        # not enough time – fall back to a simple uniform construction
        best_seq = np.full(200, 1.0 / 200)
        best_obj = compute_objective(best_seq)
        return best_seq.tolist()

    # ---- Stage 1: shape pre‑search -----------------------------------
    shape_time = min(5.0, remaining * 0.06)
    if init_from_public is not None:
        init_a = _upsample_to_length(init_from_public, base_n)
        init_obj = compute_objective(init_a)
    else:
        init_a, init_obj = _best_initial_sequence(base_n, rng, shape_time)
    print(f"[init n={base_n}] shape‑obj={init_obj:.6f}")

    # ---- Stage 2: Adam optimisation -----------------------------------
    remaining = deadline - time.time()
    adam_time = min(remaining * 0.45, 250.0)
    seq_adam, obj_adam = _run_opt(
        n=base_n,
        time_limit=adam_time,
        init_a=init_a,
        seed=seed + 123,
    )
    if obj_adam < best_obj:
        best_obj, best_seq = obj_adam, np.asarray(seq_adam, dtype=float)
        print(f"[global] new overall best O={best_obj:.6f} (n={base_n})")

    # ---- Stage 3a: Projected‑gradient refinement -----------------------
    remaining = deadline - time.time()
    if remaining > 2.0:
        pg_time = remaining * 0.30
        seq_pg, obj_pg = _projected_gradient_refine(best_seq, pg_time)
        if obj_pg < best_obj:
            best_obj, best_seq = obj_pg, np.asarray(seq_pg, dtype=float)
            print(f"[pg] improved O={best_obj:.6f}")

    # ---- Stage 3b: Targeted local moves -------------------------------
    remaining = deadline - time.time()
    if remaining > 1.0:
        seq_loc, obj_loc = _targeted_local_moves(best_seq, remaining, rng)
        if obj_loc < best_obj:
            best_obj, best_seq = obj_loc, np.asarray(seq_loc, dtype=float)
            print(f"[local] improved O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Multi‑scale up‑sampling – try larger supports if time permits
    # ------------------------------------------------------------------
    for target_n in [4000, 8000, 12000]:
        remaining = deadline - time.time()
        if remaining < 5.0:
            break
        # up‑sample current best shape
        a_up = _upsample_to_length(best_seq, target_n)
        # small random jitter to break symmetry when moving to a new size
        a_up += rng.normal(scale=0.001, size=target_n)
        a_up = np.clip(a_up, 0.0, None)
        a_up /= a_up.sum()

        # Adam on the larger support
        adam_time = remaining * 0.45
        seq_up, obj_up = _run_opt(
            n=target_n,
            time_limit=adam_time,
            init_a=a_up,
            seed=seed + 9999 + target_n,
        )
        if obj_up < best_obj:
            best_obj, best_seq = obj_up, np.asarray(seq_up, dtype=float)
            print(f"[up‑scale n={target_n}] new overall best O={best_obj:.6f}")

        # final PG + local moves on this size (if any time left)
        remaining = deadline - time.time()
        if remaining < 2.0:
            break

        pg_time = remaining * 0.30
        seq_pg2, obj_pg2 = _projected_gradient_refine(best_seq, pg_time)
        if obj_pg2 < best_obj:
            best_obj, best_seq = obj_pg2, np.asarray(seq_pg2, dtype=float)

        remaining = deadline - time.time()
        if remaining > 1.0:
            seq_loc2, obj_loc2 = _targeted_local_moves(best_seq, remaining, rng)
            if obj_loc2 < best_obj:
                best_obj, best_seq = obj_loc2, np.asarray(seq_loc2, dtype=float)

    # ------------------------------------------------------------------
    # Final sanity check (only for diagnostic output)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    # Clip to evaluator’s caps and guard against a too‑small sum
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    if best_seq.sum() < 0.01:
        best_seq = np.ones_like(best_seq) / best_seq.size

    return best_seq.tolist()
