def propose_candidate(seed=123, budget_s=1000, **kwargs):
    import time, math, numpy as np
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0
    eps = 1e-30

    # Determine length n:
    try:
        prev = height_sequence_1
        prev_len = len(prev)
    except Exception:
        prev_len = 0
    # choose n between 8000 and 20000 per time
    n = max(prev_len, 8000)
    n = min(max(n, 8000), 20000)

    # compute fft_len
    fft_len = 1 << ((2 * n - 1).bit_length())

    # define cheap score
    def cheap_score(vec):
        conv = np.fft.ifft(np.fft.fft(vec, fft_len)**2).real[:2*n-1]
        maxb = conv.max()
        return float(2.0 * n * maxb)

    # candidate generation
    best_a = None
    best_val = float('inf')

    # Uniform baseline
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    val = cheap_score(uniform)
    best_a, best_val = uniform.copy(), val

    # previous vector if length matches or we can resample
    if prev_len > 0:
        # Resample to new length via linear interpolation
        # Use np.interp to map old indices to new
        old_arr = np.asarray(prev, dtype=np.float64)
        old_arr = np.maximum(old_arr, 0.0)
        old_sum = old_arr.sum()
        if old_sum > 0:
            old_arr = old_arr / old_sum
            # New positions
            old_x = np.linspace(0, 1, old_len)
            new_x = np.linspace(0, 1, n)
            new_arr = np.interp(new_x, old_x, old_arr)
            new_arr = new_arr / new_arr.sum()
            val = cheap_score(new_arr)
            if val < best_val:
                best_a, best_val = new_arr.copy(), val

    # Parameter sweep families
    # exponential left
    for lam in np.linspace(0.0, 12.0, 31):
        if time.time() >= deadline: break
        a = np.exp(-lam * np.arange(n, dtype=np.float64))
        a_sum = a.sum()
        if a_sum == 0: continue
        a /= a_sum
        val = cheap_score(a)
        if val < best_val:
            best_a, best_val = a.copy(), val

    # symmetric exponential (U-case)
    for lam in np.linspace(0.0, 12.0, 31):
        if time.time() >= deadline: break
        idx = np.arange(n, dtype=np.float64)
        a = np.exp(-lam * np.minimum(idx, n - 1 - idx))
        a_sum = a.sum()
        if a_sum==0: continue
        a /= a_sum
        val = cheap_score(a)
        if val < best_val:
            best_a, best_val = a.copy(), val

    # power law left
    for p in np.linspace(0.01, 4.0, 31):
        if time.time() >= deadline: break
        a = (np.arange(1, n+1, dtype=np.float64) ** (-p))
        s = a.sum()
        if s == 0: continue
        a /= s
        val = cheap_score(a)
        if val < best_val:
            best_a, best_val = a.copy(), val

    # symmetric power law
    for p in np.linspace(0.01,4.0,31):
        if time.time() >= deadline: break
        i = np.arange(n, dtype=np.float64)
        a = np.minimum(i+1, n-i) ** (-p)
        s=a.sum()
        if s==0: continue
        a/=s
        val=cheap_score(a)
        if val<best_val:
            best_a, best_val = a.copy(), val

    # beta families
    x = (np.arange(1, n+1, dtype=np.float64) / n)
    alphas = np.concatenate((
        np.linspace(0.01,0.2,5),
        np.linspace(0.2,1.0,5),
        np.linspace(1.0,4.0,5)
    ))
    betas = alphas.copy()
    for a in alphas:
        for b in betas:
            if time.time() >= deadline:break
            y = (x**a) * ((1.0 - x)**b)
            s = y.sum()
            if s==0: continue
            a_vec = y/s
            val = cheap_score(a_vec)
            if val<best_val:
                best_a, best_val = a_vec.copy(), val

    # random Dirichlet seeds
    for conc in [0.001,0.01,0.1,0.5,1.0,2.0,5.0,10.0]:
        if time.time() >= deadline: break
        g = np.random.gamma(conc, 1.0, size=n)
        a = g / g.sum()
        val = cheap_score(a)
        if val < best_val:
            best_a, best_val = a.copy(), val

    print(f"[Init] best cheap = {best_val:.6f}")

    # ------- Subgradient Mirror Descent -------
    a = best_a.copy()
    best_iter_a = a.copy()
    best_iter_val = best_val
    last_improve = 0
    iter_cnt = 0
    eta0 = 1.2  # larger
    max_iters = 5_000_000

    # precompute index array for efficiency
    n_range = np.arange(n, dtype=np.int64)

    while time.time() < deadline and iter_cnt < max_iters:

        # compute conv via FFT

        j = kmax - rng
        g = np.zeros_like(a)
        eq_mask = (rng == j)
        # off-diagonal contributions
        off = rng[~eq_mask]
        g[off] = 2.0 * a[j[~eq_mask]]
        # diagonal
        diag = rng[eq_mask]
        g[diag] = a[diag]

        eta = eta0 / math.sqrt(iter_cnt)
        log_a = np.log(a + eps)
        log_a -= eta * g
        a = np.exp(log_a)
        sum_a = a.sum()
        if sum_a <= 0 or not np.isfinite(sum_a):
            a = best_iter_a.copy()
            continue
        a /= sum_a

        # restart if no improvement in many iters
        if iter_cnt - last_improve > 300_000:
            perturb = np.random.rand(n) * 1e-3
            a = _simplex_projection(best_iter_a + perturb)
            last_improve = iter_cnt
            print(f"[Iter {iter_cnt:07d}] Restart after stagnation.")

        # Periodically recompute cheap for checking (maybe each 1000 iters)
        # Already computed cheap each iter, that's fine.

    # --------- Local search (swap + transfer) ---------
    a = best_iter_a.copy()
    best_loc_a = a.copy()
    best_loc_val = best_iter_val
    temperature = 1.0
    cooling = 0.99995

    while time.time() < deadline:
        move_type = np.random.rand()
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        if move_type < 0.5:
            # swap
            a[i], a[j] = a[j], a[i]
            changed = True
        else:
            # transfer small fraction from i to j
            fraction = np.random.rand() * 0.1
            delta = a[i] * fraction
            a[i] -= delta
            a[j] += delta
            changed = True

        # evaluate cheap (full conv)
        fa = np.fft.fft(a, fft_len)
        conv = np.fft.ifft(fa * fa).real[: 2 * n - 1]
        cheap = 2.0 * n * conv.max()
        accept = False
        if cheap < best_loc_val - 1e-12:
            accept = True
        else:
            delta_val = cheap - best_loc_val
            prob = math.exp(-delta_val / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            if cheap < best_loc_val - 1e-12:
                best_loc_val = cheap
                best_loc_a = a.copy()
                print(f"[SwapTransfer] NEW BEST cheap = {cheap:.6f}")
        else:
            # revert
            if move_type < 0.5:
                a[i], a[j] = a[j], a[i]
            else:
                a[i] += delta
                a[j] -= delta

        temperature = max(temperature * cooling, 1e-6)

    final_seq = best_loc_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:
        final_score = float('inf')
        print(f"Error during final evaluation: {exc}")

    return final_seq

"""
Propose a non‑negative step function (list of heights) that minimises
``evaluate_sequence``.

Algorithm outline
-----------------
1. **Choose a working length** – try to use a longer vector than the
   previous best (up to 20000 entries).  A larger ``n`` gives the optimiser
   more freedom while the cheap score stays O(1).

2. **Broad parametric sweep** – generate many families (exponential,
   symmetric “U”‑exponential, power‑law, symmetric power‑law, beta and
   Dirichlet).  Each candidate is normalised to sum = 1 and evaluated by a
   cheap proxy ``cheap_score = 2·n·max(conv)`` (the expensive ``evaluate_sequence``
   is called only once at the end).  The best candidate is kept.

3. **Exact exponentiated‑gradient mirror descent** – minimise the convex
   objective ``max_k (a*a)_k`` on the simplex.
   The true sub‑gradient at the maximal convolution entry is computed
   in O(n) and an entropic (exponentiated) update is applied with a
   decreasing step‑size ``η = η₀/√t``.  Restarts (small random perturbations)
   are triggered after long stagnation.

4. **Permutation‑aware local search** – starting from the best iterate,
   a simulated‑annealing style random walk explores three types of moves:
   a simple swap, a small mass transfer between two positions, and a short
   block reversal.  Each move is accepted if it improves the cheap score,
   otherwise with a temperature‑controlled probability.

5. **Final exact evaluation** – the best vector found is converted to a
   Python list and fed to ``evaluate_sequence``.  The function always
   returns a feasible (non‑negative) sequence, even if the time budget
   is exhausted.

The routine respects the user‑supplied time budget (default 1000 s) and
prints progress information.  All helper utilities are defined at the

import time

import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using a power‑of‑two FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa).real[: 2 * a.size - 1]
    return conv


def _conv_max_fft(a: np.ndarray, fft_len: int) -> float:
    """Maximum value of the convolution a*a."""
    conv = _conv_full_fft(a, fft_len)
    return float(conv.max())
