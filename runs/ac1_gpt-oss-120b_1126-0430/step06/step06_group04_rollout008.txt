def _shape_candidates(n, rng):
    cand = []
    idx = np.arange(n, dtype=float)
    # Uniform
    cand.append(np.full(n, 1.0/n))
    # Exponential
    r_vals = np.geomspace(0.9, 0.99999, num=12)   # log spaced
    for r in r_vals:
        a = r**idx
        a = a / a.sum()
        cand.append(a)
        cand.append(a[::-1])
    # Symmetric exponential
    for r in r_vals[::3]:
        a = (r**idx + r**(n-1-idx))
        a = a / a.sum()
        cand.append(a)
    # Power law
    p_vals = np.linspace(0.2, 1.5, 6)
    for p in p_vals:
        a = (idx+1)**(-p) + (n-idx)**(-p)
        a = a / a.sum()
        cand.append(a)
    # Linear ramp down
    a = (n - idx)    # descending from n to 1
    a = a / a.sum()
    cand.append(a)
    cand.append(a[::-1])
    # Quadratic ramp
    a = (n - idx)**2
    a = a / a.sum()
    cand.append(a)
    cand.append(a[::-1])
    # Random Dirichlet (alpha = 0.5, 1, 2)
    for alpha in [0.3, 0.5, 1.0, 2.0]:
        a = rng.gamma(alpha, 1.0, size=n)
        a = a / a.sum()
        cand.append(a)
    # Sparse random (few non-zero)
    k = max(2, int(0.02*n))
    for _ in range(3):
        a = np.zeros(n)
        pos = rng.choice(n, size=k, replace=False)
        vals = rng.random(k)
        a[pos] = vals
        a = a / a.sum()
        cand.append(a)
    return cand

def _run_exp_grad(n, time_limit, init_a, seed=0):
    start = time.time()
    rng = np.random.default_rng(seed)
    a = np.asarray(init_a, dtype=float)
    if a.size != n:
        a = rng.random(n)
    a = np.clip(a, 0, None)
    if a.sum() == 0:
        a = np.full(n, 1.0/n)
    else:
        a = a / a.sum()
    best_a = a.copy()
    best_obj = _objective(a)
    it = 0

    # step size schedule
    eta0 = 0.3
    while time.time() - start < time_limit:
        it += 1
        conv = _conv_self_fft(a)
        M = conv.max()
        obj = float(2.0 * n * M)
        if obj < best_obj - 1e-14:
            best_obj, best_a = obj, a.copy()
            print(f"[EG n={n} it={it}] new best O={best_obj:.6f}")

        # compute gradient of max (exact)
        k_max = int(np.argmax(conv))
        w = np.zeros_like(conv)

"""
Advanced step‑function optimiser.

The optimiser explores several support sizes, builds a rich set of
analytically‑motivated initial shapes (including exponentials,
symmetric exponentials, power‑law ramps, linear/quadratic ramps,
Dirichlet draws and sparse spikes), and then refines them with three
complementary numeric phases:

1. Adam on the simplex (log‑it parametrisation) with a temperature‑
   controlled smooth‑max.  The schedule is more aggressive than the
   baseline and includes frequent jitter / restart to escape plateaus.
2. Exponentiated‑gradient descent on the simplex (exact max‑gradient)
   with a simple 1/√t step‑size decay – this often improves the tail
   of Adam’s progress.
3. Targeted simulated‑annealing + projected‑gradient refinement
   (exact max‑gradient) – identical to the previous implementation but
   with a longer schedule and an extra “swap” move.

The routine adapts the allocated wall‑clock time for each support
length, interpolates the current best sequence to larger lengths,
and finally returns the best construction found.  All numbers are
non‑negative and the sequence is normalised to unit sum before the
final call to ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return conv_full[: 2 * n - 1]


def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation of a with weight vector w:
        g[i] = Σ_t w[i+t]·a[t]   for i=0,…,n‑1
    Implemented via FFT (real‑valued).
    """
    n = a.shape[0]
    m = w.shape[0]  # = 2n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Stable soft‑max."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of vector v onto the probability simplex
    {x ≥ 0, Σ x = 1}.
    O(n log n) algorithm of Duchi et al.
    """
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero or negative – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) assuming Σ a = 1.
    Returns a Python float.
    """
    n = a.size
    conv = _conv_self_fft(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Shape candidates (Stage 1)
# ----------------------------------------------------------------------


def _shape_candidates(n: int, rng: np.random.Generator):
    """
    Produce a list of candidate initial sequences for support length n.
    Each candidate is a NumPy array normalised to sum‑to‑1.
    """
    cand = []
    idx = np.arange(n, dtype=float)

    # 1. Uniform
    cand.append(np.full(n, 1.0 / n))

    # 2. Simple exponential decay (both directions)
    r_vals = np.geomspace(0.90, 0.99999, num=14)
    for r in r_vals:
        a = r ** idx
        a = a / a.sum()
        cand.append(a)
        cand.append(a[::-1])

    # 3. Symmetric exponential (mass at both ends)
    for r in r_vals[::3]:
        a = (r ** idx + r ** (n - 1 - idx))
        a = a / a.sum()
        cand.append(a)

    # 4. Power‑law ramps (symmetric)
    p_vals = np.linspace(0.2, 1.6, 6)
    for p in p_vals:
        a = (idx + 1) ** (-p) + (n - idx) ** (-p)
        a = a / a.sum()
        cand.append(a)

    # 5. Linear ramps (descending / ascending)
    a = (n - idx)               # large at left, small at right
    a = a / a.sum()
    cand.append(a)
    cand.append(a[::-1])

    # 6. Quadratic ramps (more aggressive)
    a = (n - idx) ** 2
    a = a / a.sum()
    cand.append(a)
    cand.append(a[::-1])

    # 7. Dirichlet draws with various concentration parameters
    for alpha in (0.3, 0.5, 1.0, 2.0):
        a = rng.gamma(alpha, 1.0, size=n)
        a = a / a.sum()
        cand.append(a)

    # 8. Sparse random (few non‑zero entries)
    k = max(2, int(0.02 * n))
    for _ in range(3):
        a = np.zeros(n, dtype=float)
        pos = rng.choice(n, size=k, replace=False)
        vals = rng.random(k)
        a[pos] = vals
        a = a / a.sum()
        cand.append(a)

    return cand


def _best_initial_sequence(n: int, rng: np.random.Generator, time_limit: float):
    """
    Scan the analytical shape pool and return the candidate with the
    smallest exact objective.  The scan aborts if the time limit is hit.
    """
    start = time.time()
    best_a = None
    best_obj = float("inf")
    for a in _shape_candidates(n, rng):
        if time.time() - start > time_limit:
            break
        obj = _objective(a)
        if obj < best_obj:
            best_obj, best_a = obj, a
    # fallback – a simple Dirichlet draw
    if best_a is None:
        a = rng.random(n)
        best_a = a / a.sum()
        best_obj = _objective(best_a)
    return best_a, best_obj

# Adam optimiser on the simplex (Stage 2A)
# ----------------------------------------------------------------------


def _run_opt(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
) -> tuple[list[float], float]:
    """
    Adam optimisation on the log‑it simplex.
    Returns the best sequence (as a plain list) and its exact objective.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # initialise a (sum‑to‑1)
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
        a = a / a.sum()
    else:
        a = rng.random(n)
        a = a / a.sum()

    # log‑it parametrisation
    x = np.log(a + 1e-16)
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # schedules
    temp_start = 0.2        # smoother start
    temp_end = 1e7          # essentially hard max
    lr_start = 0.30
    lr_end = 0.00015

    best_a = a.copy()
    best_obj = _objective(best_a)

    it = 0
    while time.time() - start < time_limit:
        it += 1
        # stay on simplex
        a = _softmax(x)

        conv = _conv_self_fft(a)
        M = conv.max()
        obj = float(2.0 * n * M)

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[Adam n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        # time‑based schedules
        t = min((time.time() - start) / time_limit, 1.0)
        temp = temp_start * np.exp(np.log(temp_end / temp_start) * t)
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t)

        # smooth‑max weights
        if temp > 5e5:
            # hard max – one‑hot
            w = np.zeros_like(conv)
            w[int(np.argmax(conv))] = 1.0
        else:
            shifted = conv - M
            w = np.exp(temp * shifted)
            s = w.sum()
            if s == 0.0:
                w = np.ones_like(conv) / conv.size
            else:
                w /= s

        # gradient of (smooth) max w.r.t. a
        g_a = 2.0 * _correlation_fft(w, a)

        # natural gradient in log‑it space
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # jitter every so often to shake off plateaus
        if it % 2500 == 0:
            jitter = rng.normal(scale=0.012, size=n)
            x += jitter

        # periodic restart from the current best (helps escape shallow basins)
        if it % 50000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.008, size=n)
            m.fill(0.0)
            v.fill(0.0)

    # prune near‑zero entries before returning (helps the final objective)
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = _objective(best_a)
        print(f"[prune] length → {best_a.size}, O={best_obj:.6f}")

    return best_a.tolist(), best_obj

# Exponentiated‑gradient optimiser (Stage 2B)
# ----------------------------------------------------------------------


def _run_exp_grad(
    n: int,
    time_limit: float,
    init_a: np.ndarray,
    seed: int = 0,
) -> tuple[list[float], float]:
    """
    Exponentiated‑gradient descent on the simplex using the exact
    max‑gradient.  Step‑size η decays as η₀/√t.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    a = np.asarray(init_a, dtype=float)
    if a.size != n:
        a = rng.random(n)
    a = np.clip(a, 0.0, None)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()

    best_a = a.copy()
    best_obj = _objective(best_a)

    it = 0
    eta0 = 0.35
    while time.time() - start < time_limit:
        it += 1
        conv = _conv_self_fft(a)
        M = conv.max()
        obj = float(2.0 * n * M)
        if obj < best_obj - 1e-14:
            best_obj, best_a = obj, a.copy()
            print(f"[EG n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        # exact max‑gradient (one‑hot)
        k_max = int(np.argmax(conv))
        w = np.zeros_like(conv)
        w[k_max] = 1.0
        g = 2.0 * _correlation_fft(w, a)   # gradient of max(conv)

        # step‑size schedule
        eta = eta0 / math.sqrt(it)

        # exponentiated‑gradient update
        a = a * np.exp(-eta * g)
        a = _project_simplex(a)

        # occasional jitter
        if it % 1800 == 0:
            jitter = rng.normal(scale=0.006, size=n)
            a = _project_simplex(a + jitter)

    return best_a.tolist(), best_obj

# Simulated annealing + projected gradient (Stage 3)
# ----------------------------------------------------------------------


def _run_sa(
    init_seq: list[float],
    time_limit: float,
    seed: int = 0,
) -> tuple[list[float], float]:
    """
    Targeted simulated‑annealing refinement followed by a few
    projected‑gradient descent steps (exact max‑gradient).  Two move
    types are employed: (i) mass‑transfer and (ii) mass‑swap.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    a = np.asarray(init_seq, dtype=float)
    if a.sum() < 1e-12:
        a = np.ones_like(a) / a.size
    else:
        a = a / a.sum()
    n = a.shape[0]

    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())
    best_seq = a.copy()

    # SA schedule
    T = best_obj * 0.25 + 1e-9
    T_min = best_obj * 1e-6 + 1e-12
    decay = 0.998

    it = 0
    while time.time() - start < time_limit * 0.66:
        it += 1
        k_max = int(np.argmax(conv))

        # contributions to the current maximal index
        i_vals = np.arange(n)
        j_vals = k_max - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i_vals[mask]] * a[j_vals[mask]]
        total_contrib = contrib.sum()

        # choose source index biased by contribution
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = rng.choice(n, p=probs)
        else:
            i = rng.integers(n)

        # random move type
        if rng.random() < 0.65:          # mass‑transfer (dominant)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            delta = a[i] * rng.random() * 0.08   # up to 8 % of a[i]
            if delta <= 0.0:
                continue
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
        else:                            # mass‑swap
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            # swap a fraction of mass
            frac = rng.random() * 0.10
            delta_i = a[i] * frac
            delta_j = a[j] * frac
            a_new = a.copy()
            a_new[i] = a[i] - delta_i + delta_j
            a_new[j] = a[j] - delta_j + delta_i

        a_new = np.clip(a_new, 0.0, None)
        a_new /= a_new.sum()

        conv_new = _conv_self_fft(a_new)
        obj_new = float(2.0 * n * conv_new.max())

        if obj_new < best_obj - 1e-14:
            a, conv, best_obj, best_seq = a_new, conv_new, obj_new, a_new.copy()
            print(f"[SA it={it}] new best O={best_obj:.6f}")
        else:
            # Metropolis acceptance
            if rng.random() < math.exp(-(obj_new - best_obj) / T):
                a, conv = a_new, conv_new

        if it % 2500 == 0:
            T = max(T * decay, T_min)

        if it % 9000 == 0:
            # global jitter
            noise = rng.normal(scale=0.009, size=n)
            a = np.clip(a + noise, 0.0, None)
            a /= a.sum()
            conv = _conv_self_fft(a)

    # -----------------------------------------------------------------
    # Projected‑gradient descent with exact max‑gradient
    remaining = time_limit - (time.time() - start)
    if remaining > 0.4:
        a = best_seq.copy()
        n = a.size
        it_g = 0
        eta0 = 0.28
        while time.time() - start < time_limit:
            it_g += 1
            conv = _conv_self_fft(a)
            k_max = int(np.argmax(conv))
            w = np.zeros_like(conv)
            w[k_max] = 1.0
            g = 2.0 * _correlation_fft(w, a)   # exact max‑gradient
            eta = eta0 * (0.995 ** it_g)
            a_new = a - eta * g
            a_new = _project_simplex(a_new)
            obj_new = float(2.0 * n * _conv_self_fft(a_new).max())
            if obj_new < best_obj - 1e-14:
                a, best_obj, best_seq = a_new, obj_new, a_new.copy()
                print(f"[PG it={it_g}] improved O={best_obj:.6f}")
            else:
                # aggressive step‑size reduction on stagnation
                eta0 *= 0.97
                if eta0 < 5e-7:
                    break

    return best_seq.tolist(), best_obj

def _optimize_length(
    n: int,
    init_a: np.ndarray,
    total_time: float,
    seed: int,
) -> tuple[list[float], float]:
    """
    Run Adam → Exponentiated‑gradient → SA/PG on the given length.
    The total_time is split (≈50 %/20 %/30 %).
    """
    if total_time < 3.0:
        # not enough budget – just run a quick Adam pass
        return _run_opt(n, total_time, init_a, seed)

    adam_time = total_time * 0.52
    eg_time = total_time * 0.18
    sa_time = total_time - adam_time - eg_time

    seq, val = _run_opt(n, adam_time, init_a, seed)
    # EG refinement
    if eg_time > 0.6:
        seq_eg, val_eg = _run_exp_grad(n, eg_time, np.asarray(seq, dtype=float), seed + 111)
        if val_eg < val:
            seq, val = seq_eg, val_eg

    # SA + PG final polishing
    if sa_time > 0.8:
        seq_sa, val_sa = _run_sa(seq, sa_time, seed + 222)
        if val_sa < val:
            seq, val = seq_sa, val_sa

    return seq, val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **_ignore,
) -> list[float]:
    """
    Optimisation driver.
    Explores several support sizes, builds a rich pool of analytical
    initial shapes (including interpolation from the current best),
    refines each candidate with Adam → EG → SA/PG, and finally returns
    the best sequence discovered.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Support lengths – start with a medium size, then enlarge
    lengths = [2000, 3000, 4000, 6000, 8000]

    best_seq = None
    best_obj = float("inf")

    # --------------------------------------------------------------
    # Iterate over lengths, allocating time adaptively (more to short lengths)
    for idx, n in enumerate(lengths):
        now = time.time()
        if now >= deadline - 1.0:
            break
        remaining = deadline - now
        # weight favouring shorter lengths (inverse of n)
        weight = 1.0 / n
        total_weight = sum(1.0 / L for L in lengths[idx:])
        alloc = remaining * (weight / total_weight) * 0.92   # a bit of buffer
        if alloc < 5.0:      # skip overly short allocations
            continue

        # ----- Stage 1: shape pre‑search -----
        shape_time = min(3.5, alloc * 0.07)
        init_a, init_obj = _best_initial_sequence(n, rng, shape_time)

        # ----- interpolation from best so‑far (if any) -----
        if best_seq is not None:
            interp = np.interp(
                np.linspace(0, best_seq.size - 1, n),
                np.arange(best_seq.size),
                best_seq,
            )
            interp = interp / interp.sum()
            obj_interp = _objective(interp)
            if obj_interp < init_obj:
                init_a, init_obj = interp, obj_interp

        print(f"[len={n}] init obj={init_obj:.6f}")

        # ----- Stage 2 + 3: optimisation pipeline -----
        seq, val = _optimize_length(n, init_a, alloc, seed + idx * 12345)
        if val < best_obj:
            best_obj, best_seq = val, np.asarray(seq, dtype=float)
            print(f"[global] new overall best O={best_obj:.6f} (len={n})")

    # --------------------------------------------------------------
    # Final refinement on the globally best sequence (if time permits)
    remaining = deadline - time.time()
    if best_seq is not None and remaining > 6.0:
        seq_ref, val_ref = _run_sa(best_seq.tolist(), time_limit=remaining, seed=seed + 999999)
        if val_ref < best_obj:
            best_obj, best_seq = val_ref, np.asarray(seq_ref, dtype=float)
            print(f"[final] after SA+PG new best O={best_obj:.6f}")

    # --------------------------------------------------------------
    # Guard against pathological degenerate cases
    if best_seq is None or best_seq.size == 0:
        best_seq = np.full(200, 1.0 / 200.0)

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    if best_seq.sum() < 0.01:
        best_seq = np.ones_like(best_seq) / best_seq.size

    # --------------------------------------------------------------
    # Diagnostic – report the objective as seen by the official evaluator
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_seq.tolist()
