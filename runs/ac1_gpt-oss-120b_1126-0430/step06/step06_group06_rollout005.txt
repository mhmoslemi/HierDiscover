#!/usr/bin/env python
"""
Improved search for a non‑negative step function that minimises

        2·n·max_k (a*a)[k] / (Σ a_i)²

The programme follows three stages:

1. **Exploratory sweep** over a richer set of parametric families
   (uniform, power‑law, exponential, stretched‑exponential, beta,
   Gaussian, Dirichlet).  The best candidate (normalised to sum = 1) is
   retained.

2. **Smooth mirror‑descent** on a differentiable surrogate of the
   infinity‑norm.  We replace the hard max by a soft‑max
   (log‑sum‑exp) with a temperature that is increased during the run.
   The optimisation is performed on the log‑odds variables
   `y = log(a)` (so the simplex constraint is satisfied automatically)
   using an Adam‑style update.  The gradient of the surrogate can be
   expressed with a single extra FFT (correlation of `a` with the
   soft‑max weights), which keeps the per‑iteration cost low.

3. **Local stochastic refinement** – a (1+1)‑ES that perturbs the
   current best vector multiplicatively in log‑space, plus occasional
   convex combinations with fresh Dirichlet samples.  This stage
   finishes the search and supplies the final sequence.

The whole search is guaranteed to return a feasible sequence before the
budget expires.  The implementation is fully self‑contained, uses only
NumPy and the supplied ``evaluate_sequence`` function, and respects the
1 000 s time limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        # all non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution (a*a) using real FFT, length 2·n‑1."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len).real[: 2 * a.size - 1]
    return conv


def _conv_max_fft(a: np.ndarray, fft_len: int) -> tuple[float, int]:
    """Maximum coefficient of a*a and its index."""
    conv = _conv_full_fft(a, fft_len)
    idx = int(conv.argmax())
    return float(conv[idx]), idx


def _weighted_corr_fft(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute 2 * Σ_j a_j * w_{i+j} for all i (0 ≤ i < len(a)).
    `w` is the weight vector of length 2·len(a)-1.
    Returns a vector of length len(a).
    """
    n = a.shape[0]
    # reverse w for correlation
    w_rev = w[::-1]
    L = n + w_rev.size - 1          # = 3·n-2
    fft_len = 1 << ((L).bit_length())
    fa = np.fft.rfft(a, fft_len)
    fw = np.fft.rfft(w_rev, fft_len)
    conv = np.fft.irfft(fa * fw, fft_len).real
    base = w_rev.size - 1           # = 2·n-2
    # indices base, base-1, ..., base-(n-1)
    g = conv[base - np.arange(n)]
    return 2.0 * g


# ----------------------------------------------------------------------
# Parametric families (all return a normalised probability vector)
# ----------------------------------------------------------------------


def _uniform_seq(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _power_law_seq(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = i ** (-p)
    seq /= seq.sum()
    return seq


def _exponential_seq(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    seq = np.exp(-lam * i)
    seq /= seq.sum()
    return seq


def _power_law_exp_seq(n: int, p: float, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    seq = (i + 1.0) ** (-p) * np.exp(-lam * i)
    seq /= seq.sum()
    return seq


def _stretched_exponential_seq(n: int, lam: float, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    seq = np.exp(-lam * (i ** gamma))
    seq /= seq.sum()
    return seq


def _beta_seq(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(1, n + 1, dtype=np.float64) / n)
    seq = (x ** alpha) * ((1.0 - x) ** beta)
    s = seq.sum()
    if s == 0.0:
        return None
    seq /= s
    return seq


def _gaussian_seq(n: int, sigma: float) -> np.ndarray:
    """Centered discrete Gaussian, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    seq = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    seq /= seq.sum()
    return seq


def _dirichlet_seq(n: int, conc: float) -> np.ndarray:
    g = np.random.gamma(conc, 1.0, size=n)
    seq = g / g.sum()
    return seq

# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function (list of heights) that
    minimises ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the final evaluation
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # Determine working length.
    # --------------------------------------------------------------
    try:
        prev = globals()["height_sequence_1"]
        n = max(len(prev), 5000)
        prev_arr = np.asarray(prev, dtype=np.float64)
        if prev_arr.sum() > 0.0:
            prev_arr /= prev_arr.sum()
        else:
            prev_arr = None
    except Exception:
        prev_arr = None
        n = 5000
    # keep n modest (5k–25k) to stay within time budget
    n = min(max(n, 5000), 25000)
    fft_len_conv = 1 << ((2 * n - 1).bit_length())
    fft_len_corr = 1 << ((3 * n - 2).bit_length())

    # --------------------------------------------------------------
    # Cheap objective (assumes Σa_i = 1): 2·n·max(conv)
    # --------------------------------------------------------------
    def cheap_score(vec: np.ndarray) -> float:
        maxb, _ = _conv_max_fft(vec, fft_len_conv)
        return 2.0 * n * maxb

    # --------------------------------------------------------------
    # Phase 0 – parametric sweep (≈15 % of time budget)
    # --------------------------------------------------------------
    best_a = _uniform_seq(n)
    best_val = cheap_score(best_a)

    # include previous construction if dimensions match
    if prev_arr is not None and prev_arr.size == n:
        val = cheap_score(prev_arr)
        if val < best_val:
            best_a, best_val = prev_arr.copy(), val

    # ---------- systematic families ----------
    # Power‑law
    for p in np.linspace(0.2, 2.0, 81):
        if time.time() >= deadline:
            break
        cand = _power_law_seq(n, p)
        val = cheap_score(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # Exponential
    for lam in np.linspace(0.0, 5.0, 81):
        if time.time() >= deadline:
            break
        cand = _exponential_seq(n, lam)
        val = cheap_score(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # Mixed power‑law × exponential
    for p in np.linspace(0.2, 1.5, 11):
        for lam in np.linspace(0.0, 3.0, 11):
            if time.time() >= deadline:
                break
            cand = _power_law_exp_seq(n, p, lam)
            val = cheap_score(cand)
            if val < best_val:
                best_a, best_val = cand, val

    # Stretched exponential
    for gamma in [0.5, 0.9, 1.2, 1.5, 2.0]:
        for lam in np.linspace(0.0, 4.0, 31):
            if time.time() >= deadline:
                break
            cand = _stretched_exponential_seq(n, lam, gamma)
            val = cheap_score(cand)
            if val < best_val:
                best_a, best_val = cand, val

    # Beta‑shaped families (coarse grid)
    x = np.arange(1, n + 1, dtype=np.float64) / n
    for α in np.linspace(0.2, 8.0, 17):
        for β in np.linspace(0.2, 8.0, 17):
            if time.time() >= deadline:
                break
            seq = (x ** α) * ((1.0 - x) ** β)
            s = seq.sum()
            if s == 0.0:
                continue
            cand = seq / s
            val = cheap_score(cand)
            if val < best_val:
                best_a, best_val = cand, val

    # Gaussian family (σ roughly n/10 … n/3)
    for sigma in np.linspace(n / 20.0, n / 4.0, 13):
        if time.time() >= deadline:
            break
        cand = _gaussian_seq(n, sigma)
        val = cheap_score(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # Random Dirichlet seeds (wide concentration range)
    for conc in [0.001, 0.005, 0.01, 0.05, 0.1, 0.5, 1.0, 2.0, 5.0]:
        if time.time() >= deadline:
            break
        cand = _dirichlet_seq(n, conc)
        val = cheap_score(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # Small random perturbations of the current best
    for _ in range(120):
        if time.time() >= deadline:
            break
        pert = 0.02 * np.random.rand(n)
        cand = _simplex_projection(best_a + pert)
        val = cheap_score(cand)
        if val < best_val:
            best_a, best_val = cand, val

    print(f"[Phase 0] best cheap score = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – smooth mirror‑descent (Adam on logits) (≈55 % time)
    # --------------------------------------------------------------
    # we work with logits y where a = softmax(y)
    eps = 1e-30
    y = np.log(best_a + eps)          # initialise logits
    m = np.zeros_like(y)              # Adam first moment
    v = np.zeros_like(y)              # Adam second moment
    beta1, beta2 = 0.9, 0.999
    lr = 0.15                         # base learning rate
    t = 0
    best_iter_a = best_a.copy()
    best_iter_val = best_val
    beta_temp = 1.0                   # temperature for soft‑max (β in exp)
    beta_target = 30.0                # final temperature
    beta_inc_per_iter = (beta_target - beta_temp) / 200_000  # gradual

    while time.time() < deadline - 20.0:
        t += 1
        # ---- compute current a ----
        maxy = np.max(y)                # for numerical stability
        expy = np.exp(y - maxy)
        a = expy / expy.sum()

        # ---- convolution and cheap score ----
        conv = _conv_full_fft(a, fft_len_conv)
        maxb = conv.max()
        cheap = 2.0 * n * maxb

        if cheap < best_iter_val - 1e-12:
            best_iter_val = cheap
            best_iter_a = a.copy()
            print(f"[Iter {t:07d}] NEW best cheap = {cheap:.6f}")

        # ---- soft‑max weights over convolution entries ----
        # temperature β grows slowly → tighter approximation to max
        beta_temp = min(beta_temp + beta_inc_per_iter, beta_target)
        # use stable softmax on (conv - maxb)
        w = np.exp(beta_temp * (conv - maxb))
        w /= w.sum()                    # now Σ w = 1

        # ---- gradient of cheap surrogate w.r.t. a ----
        g_a = _weighted_corr_fft(a, w)   # shape (n,)

        # ---- gradient w.r.t. logits y (softmax Jacobian) ----
        dot = np.dot(a, g_a)
        g_y = a * (g_a - dot)           # dF/dy_i

        # ---- Adam update on y ----
        m = beta1 * m + (1.0 - beta1) * g_y
        v = beta2 * v + (1.0 - beta2) * (g_y ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        y -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional restart if stagnation > 120 k iterations
        if t % 120_000 == 0:
            # blend current best with a fresh Dirichlet sample
            pert = _dirichlet_seq(n, np.random.uniform(0.001, 1.0))
            y = np.log(_simplex_projection(best_iter_a + 0.015 * pert) + eps)

    print(f"[Phase 1] finished after {t} iterations, best cheap = {best_iter_val:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – stochastic (1+1)‑ES + convex combos (≈25 % time)
    # --------------------------------------------------------------
    sigma = 0.12                # mutation strength (log‑space)
    fail_cnt = 0
    it2 = 0
    while time.time() < deadline - 2.0:
        it2 += 1
        # multiplicative Gaussian perturbation in log‑space
        cand = best_iter_a * np.exp(sigma * np.random.randn(n))
        cand = _simplex_projection(cand)
        val = cheap_score(cand)

        if val < best_iter_val:
            best_iter_val = val
            best_iter_a = cand
            sigma = max(sigma * 0.85, 1e-4)
            fail_cnt = 0
            print(f"[ES {it2:06d}] improved cheap = {val:.6f}  sigma={sigma:.5f}")
        else:
            fail_cnt += 1
            if fail_cnt > 300:
                sigma = min(sigma * 1.25, 0.6)
                fail_cnt = 0

        # every 60 k steps try a convex combination with a new Dirichlet sample
        if it2 % 60000 == 0:
            mix = np.random.rand()
            dirv = _dirichlet_seq(n, np.random.uniform(0.001, 2.0))
            cand2 = _simplex_projection(mix * best_iter_a + (1.0 - mix) * dirv)
            val2 = cheap_score(cand2)
            if val2 < best_iter_val:
                best_iter_val = val2
                best_iter_a = cand2
                print(f"[Mix {it2}] improved cheap = {val2:.6f}")

    print(f"[Phase 2] final cheap = {best_iter_val:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – final exact evaluation
    # --------------------------------------------------------------
    final_seq = best_iter_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # --------------------------------------------------------------
    # Return the best sequence found.
    # --------------------------------------------------------------
    return final_seq
