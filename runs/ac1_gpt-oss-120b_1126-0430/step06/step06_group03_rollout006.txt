def _beta_distribution(n, alpha, beta):
    i = np.arange(n, dtype=np.float64)
    a = (i+1)**(alpha-1) * (n-i)**(beta-1)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0/n
    else:
        a /= s
    return a

"""
Hybrid optimisation for the self‑convolution bound with extra parameter
refinement and multi‑restart Adam.  The routine

1. loads an initial sequence (the provided ``height_sequence_1`` or a uniform
   distribution),
2. runs a fast parametric sweep over several families (including a beta
   distribution) and keeps the best candidate,
3. performs a cheap local optimisation of the family parameters by random
   perturbations,
4. runs Adam on the probability simplex (multiple restarts if stagnation is
   detected) for as long as the time budget allows,
5. finishes with a targeted random‑walk that moves mass from high‑gradient
   positions to low‑gradient ones,
6. finally evaluates the exact score with ``evaluate_sequence`` and returns the
   best sequence found.

All numbers are non‑negative, clipped to the allowed range and normalised to
unit sum (the score is scale‑invariant).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max, returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    s = e.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / z.size)
    return e / s


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑its that reproduce `a` after a soft‑max."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """Smallest power‑of‑two ≥ 3·n‑2 (enough for all needed convolutions)."""
    return 1 << ((3 * n - 2).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Returns (max_conv, grad_a) where

        grad_a = ∇_a [ 2·n·smooth_max(a*a) ]

    ``eps_factor`` multiplies the smoothing constant ε = eps_factor·max_conv.
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv                     # ≤ 0 for stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # Gradient of 2·n·smooth_max (the factor 4·n comes from the derivative of a*a)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a


# ----------------------------------------------------------------------
# Parametric families of distributions
# ----------------------------------------------------------------------


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}   (normalised)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _power_law_offset_distribution(n: int, p: float, offset: float) -> np.ndarray:
    """a_i ∝ (i+offset)^{-p}   (normalised)."""
    idx = np.arange(offset, offset + n, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _double_power_law_distribution(n: int, p: float, q: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}·(n-i)^{-q}   (normalised)."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) ** (-p)) * ((n - i) ** (-q))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _stretched_exp_distribution(n: int, tau_factor: float, gamma: float) -> np.ndarray:
    """
    a_i ∝ exp( - (i/(tau_factor·n))^{γ} )   (normalised).
    """
    tau = max(1e-12, tau_factor * n)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(- (i / tau) ** gamma)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _exp_decay_distribution(n: int, lam: float) -> np.ndarray:
    """a_i ∝ e^{-lam·i}+e^{-lam·(n‑1‑i)}   (normalised)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _logistic_distribution(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ 1/(1+e^{α·(i‑mid)})   (normalised)."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = 1.0 / (1.0 + np.exp(alpha * (i - mid)))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _sym_product_power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ ((i+1)*(n‑i))^{-p}   (symmetric, normalised)."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) * (n - i)) ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^{α−1}·(n−i)^{β−1} (normalised)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (alpha - 1.0) * (n - i) ** (beta - 1.0)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):

    `evaluate_sequence` score.  The routine respects the supplied time budget
    (~1000 s) and uses a hybrid parametric sweep → local parameter refinement →
    multi‑restart Adam → targeted random walk.  The final candidate is evaluated

    """

    start = time.time()

        a0 = np.full(n, 1.0 / n)
        print(f"[Init] No provided sequence – start from uniform, length = {n}")

    best_seq = a0.copy()
    best_score = 2.0 * n * _conv_max(best_seq)   # cheap score = 2·n·max_conv

    # ------------------------------------------------------------------
    # 1. Coarse parametric sweep
    # ------------------------------------------------------------------
    families = [
        ("power_law", _power_law_distribution,
         [dict(p=p) for p in np.linspace(0.30, 1.00, 71)]),
        ("power_law_offset", _power_law_offset_distribution,
         [dict(p=p, offset=off)
          for p in np.linspace(0.30, 1.00, 35)
          for off in np.linspace(0.1, 6.0, 35)]),
        ("double_power_law", _double_power_law_distribution,
         [dict(p=p, q=q)
          for p in np.linspace(0.30, 1.00, 35)
          for q in np.linspace(0.30, 1.00, 35)]),
        ("stretched_exp", _stretched_exp_distribution,
         [dict(tau_factor=t, gamma=g)
          for t in np.linspace(0.05, 0.95, 25)
          for g in np.linspace(0.3, 3.0, 25)]),
        ("exp_decay", _exp_decay_distribution,
         [dict(lam=lam) for lam in np.logspace(-8, -1, 40)]),
        ("logistic", _logistic_distribution,

        ("sym_product_power_law", _sym_product_power_law_distribution,

    ]

    best_family = None
    best_params = None

    for name, generator, param_dicts in families:
        for params in param_dicts:
            if time.time() >= deadline:
                break
            try:
                cand = generator(n, **params)
            except Exception:
                continue
            score = 2.0 * n * _conv_max(cand)
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                best_family = (name, generator)
                best_params = params.copy()
                # print(f"[Sweep] New best {name} {params} -> {score:.6f}")
        if time.time() >= deadline:
            break

    print(f"[Sweep] Best cheap score after sweep = {best_score:.6f}")

    # 2. Local refinement of the best parametric family (if any)
    # ------------------------------------------------------------------
    if best_family is not None and best_params is not None:

        # reasonable bounds for each parameter (loose but safe)

            "offset": (0.01, 10.0), "tau_factor": (1e-3, 1.0),
            "gamma": (0.1, 5.0), "lam": (1e-10, 1.0),
            "alpha": (0.01, 10.0), "beta": (0.01, 10.0),

                    new_params[k] = v
                    continue

                no_improve_cnt = 0

        u = np.sort(v)[::-1]
        sv = np.cumsum(u)
        rho = np.where(u > (sv - 1) / (np.arange(1, len(v) + 1)))[0][-1]
        theta = (sv[rho] - 1) / (rho + 1)
        w = np.maximum(v - theta, 0.0)
        return w

    v_adam = np.zeros_like(logits)

    eps_factor = 0.20           # large smoothing at the beginning
    eps_factor_min = 5e-5

    it = 0
    last_improve = 0
    best_seq_sofar = best_seq.copy()
    best_score_sofar = best_score
    report_every = 25000

    while time.time() < deadline - 5.0 and it < 2_000_000:
        it += 1

        # current probability vector
        a = _softmax(logits)

        # smoothed max and its gradient
        max_conv, grad_a = _conv_max_and_grad(a, eps_factor)
        cheap_score = 2.0 * n * max_conv

        # record best
        if cheap_score < best_score_sofar - 1e-12:
            best_score_sofar = cheap_score
            best_seq_sofar = a.copy()
            last_improve = it
            if it % report_every == 0:
                print(f"[Adam {it:07d}] NEW BEST cheap = {cheap_score:.8f}")

        # mirror‑descent step in log‑it space
        mu = np.dot(grad_a, a)
        grad_z = a * (grad_a - mu)

        # Adam bias‑corrected moments
        m = beta1 * m + (1.0 - beta1) * grad_z
        v_adam = beta2 * v_adam + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v_adam / (1.0 - beta2 ** it)

        # step size schedule – inverse‑sqrt decay
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step = - m_hat / (np.sqrt(v_hat) + eps_adam)

        # back‑tracking line‑search (up to 5 halvings)
        trial_alpha = alpha_t
        accepted = False
        for _ in range(5):
            new_logits = logits + trial_alpha * step
            new_logits = np.clip(new_logits, -50.0, 50.0)
            a_new = _softmax(new_logits)
            new_score = 2.0 * n * _conv_max(a_new)
            if new_score <= cheap_score:
                logits = new_logits
                accepted = True
                break
            trial_alpha *= 0.5
        if not accepted:
            # take a (possibly non‑monotone) step anyway
            logits = logits + alpha_t * step
            logits = np.clip(logits, -50.0, 50.0)

        # occasional jitter to escape plateaus
        if it % 3000 == 0 and np.random.rand() < 0.20:
            logits += 0.03 * np.random.randn(n)
            m *= 0.5
            v_adam *= 0.5

        # decay smoothing constant
        eps_factor = max(eps_factor_min, eps_factor * eps_decay)

        # restart if stagnation for a long while
        if it - last_improve > 300_000:
            print(f"[Adam {it}] Stagnation – restarting from best so far")
            logits = _logits_from_distribution(best_seq_sofar) + 0.04 * np.random.randn(n)
            m.fill(0.0)
            v_adam.fill(0.0)
            eps_factor = 0.20
            last_improve = it

    best_seq = best_seq_sofar
    best_score = best_score_sofar
    print(f"[Adam] Finished after {it} iterations, best cheap score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Final cheap random walk (targeted mass transfers)
    # ------------------------------------------------------------------
    a = best_seq.copy()
    cheap_score = best_score

    # for targeted moves we compute a quick gradient approximation:
    def _approx_gradient(a_vec):
        """Gradient of the (unsmoothed) max‑conv – sum of reversed a at max indices."""
        n_loc = a_vec.shape[0]
        L_loc = _fft_len_for_corr(n_loc)
        a_fft_loc = np.fft.rfft(a_vec, L_loc)
        conv_full = np.fft.irfft(a_fft_loc * a_fft_loc, L_loc)[: 2 * n_loc - 1]
        max_val = conv_full.max()
        mask = (conv_full >= max_val - 1e-12).astype(np.float64)
        # gradient = 2 * (mask convolve a) restricted to original size
        mask_fft = np.fft.rfft(mask, L_loc)
        grad_full = np.fft.irfft(np.conj(a_fft_loc) * mask_fft, L_loc)[:n_loc]
        return 2.0 * grad_full

    grad_est = _approx_gradient(a)

    while time.time() < deadline - 1.0:
        # pick source index proportional to its contribution to the gradient
        src_probs = grad_est / (grad_est.sum() + 1e-32)
        i = np.random.choice(n, p=src_probs)
        # pick destination index where gradient is small (inverse weight)
        inv = 1.0 / (grad_est + 1e-8)
        dst_probs = inv / inv.sum()
        j = np.random.choice(n, p=dst_probs)
        if i == j:
            continue
        # propose moving a fraction of the source mass (adaptive magnitude)
        delta = 0.005 * a[i]   # 0.5 % of source mass
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # ensure non‑negativity (tiny numerical noise may appear)
        if (a_new < 0).any():
            continue
        new_score = 2.0 * n * _conv_max(a_new)
        if new_score < cheap_score:
            a = a_new
            cheap_score = new_score
            best_seq = a.copy()
            best_score = cheap_score
            grad_est = _approx_gradient(a)   # update gradient estimate
        else:
            # occasionally accept a worsening move to escape plateaus (simulated‑annealing style)
            if np.random.rand() < 0.001:
                a = a_new
                cheap_score = new_score
                grad_est = _approx_gradient(a)

    # ------------------------------------------------------------------
    # 5. Final exact evaluation with evaluate_sequence()
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        exact_score = evaluate_sequence(final_seq)  # provided by the platform
        print(f"Final exact score = {exact_score:.6f}")
    except Exception as exc:
        exact_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    return final_seq
