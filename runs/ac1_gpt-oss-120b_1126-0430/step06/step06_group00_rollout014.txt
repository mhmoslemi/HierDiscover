"""
Propose a non‑negative step‑function (a list of heights) that minimises

    2·n·max(a*a) / (sum(a)**2)

The algorithm consists of two stages:

1. **Regularised arcsine family** –  The continuous optimum is the arcsine
   distribution.  We introduce a regularisation parameter `α≥0` that removes
   the singularities at the endpoints:
       f_α(x) ∝ 1/√((x+α)(1−x+α)) ,    x∈[0,1] .
   For each `α` we discretise the exact CDF, normalise the vector to sum = 1,
   and evaluate the metric (the sum is 1, so the metric is simply
   `2·n·max(conv)`).  A cheap 1‑D search over `α` (coarse grid + a few
   golden‑section steps) yields a distribution that already improves on the
   plain arcsine discretisation.

2. **Adam refinement on the simplex** –  Starting from the best `α`‑candidate
   we run a projected Adam optimiser on the logits of a soft‑max
   parametrisation (so the vector always stays on the probability simplex).
   The objective is the smoothed max of the autocorrelation; the smoothing
   factor is reduced gradually to obtain a true sub‑gradient when the
   optimisation nears a minimum.  The best vector seen during the optimisation
   is returned.

Both stages are limited by the overall time budget (≈ 1000 s).  The code
contains a few safety checks and progress prints.  The returned list is
compatible with the platform’s `evaluate_sequence` routine.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level helpers (no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    s = e.sum()
    if s == 0.0 or np.isnan(s):
        return np.full_like(z, 1.0 / z.size)
    return e / s


def _fft_len_for_corr(n: int) -> int:
    """
    Smallest power‑of‑two ≥ 3·n‑2.
    This length is sufficient for both linear convolution (2·n‑1) and the
    correlation (3·n‑2) used in the gradient computation.
    """
    return 1 << ((3 * n - 2).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a (computed via FFT)."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Smoothed max of a*a and its gradient w.r.t. a (already multiplied by
    4·n, which is the factor appearing in the derivative of the objective
    2·n·max).

    Parameters
    ----------
    a : np.ndarray
        Probability vector (sum = 1, non‑negative).
    eps_factor : float
        Multiplicative factor for the smoothing constant:
        eps = max(eps_factor * max(conv), 1e‑12).

    Returns
    -------
    max_conv : float
        Maximum entry of the linear self‑convolution.
    grad_a : np.ndarray
        Gradient of 2·n·smooth_max w.r.t. a (including the factor 4·n).
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    a_fft = np.fft.rfft(a, L)

    # linear self‑convolution (length 2·n‑1)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # soft‑max weights over the convolution entries
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv          # for numerical stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or np.isnan(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # cross‑correlation a ⋆ w (length 3·n‑2, we need first n entries)
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # gradient (the factor 4·n comes from d[2·n·max]/da)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a


# ----------------------------------------------------------------------
# Regularised arcsine family
# ----------------------------------------------------------------------


def _distribution_alpha(alpha: float, n: int) -> np.ndarray:
    """
    Return a probability vector of length `n` obtained by discretising the
    regularised arcsine density

        f_α(x) ∝ 1 / sqrt{ (x+α) (1−x+α) } ,   x∈[0,1] .

    The discretisation is performed by exact integration of the CDF over each
    bin interval.  The returned vector sums to exactly 1.
    """
    # parameters for the CDF
    A = 1.0 + 2.0 * alpha
    # lower offset   arcsin( sqrt(α/A) )
    offset = np.arcsin(np.sqrt(alpha / A))
    # upper bound   arcsin( sqrt((1+α)/A) )
    upper = np.arcsin(np.sqrt((1.0 + alpha) / A))
    denom = upper - offset
    if denom == 0.0:
        denom = 1e-12

    idx = np.arange(n, dtype=np.float64)
    x0 = idx / n
    x1 = (idx + 1) / n

    # CDF values at the bin edges
    sqrt0 = np.sqrt(np.clip((x0 + alpha) / A, 0.0, 1.0))
    sqrt1 = np.sqrt(np.clip((x1 + alpha) / A, 0.0, 1.0))

    F0 = (np.arcsin(sqrt0) - offset) / denom
    F1 = (np.arcsin(sqrt1) - offset) / denom

    a = F1 - F0
    # eliminate tiny negative values caused by rounding
    a = np.maximum(a, 0.0)

    total = a.sum()
    if total == 0.0 or np.isnan(total):
        raise RuntimeError("Failed to build a valid α‑distribution.")
    a /= total
    return a


def _metric_alpha(alpha: float, n: int) -> float:
    """
    Compute the evaluation metric for the discretised regularised arcsine
    distribution with parameter `α`.  Since the vector is normalised to sum = 1,
    the metric reduces to `2·n·max(conv)`.
    """
    a = _distribution_alpha(alpha, n)
    max_conv = _conv_max(a)
    return 2.0 * n * max_conv


def _search_best_alpha(n: int, time_deadline: float):
    """

    grid is examined, after that a short golden‑section refinement around the
    best grid point.
    Returns the best α, the associated probability vector, and the metric.
    """

    max_alpha = 0.05
    grid_points = 31  # inclusive both ends
    alphas = np.linspace(0.0, max_alpha, grid_points)

    best_alpha = 0.0
    best_metric = np.inf
    best_vec = None

    for a in alphas:
        if time.time() > time_deadline:
            break
        m = _metric_alpha(a, n)
        if m < best_metric:
            best_metric = m
            best_alpha = a
            best_vec = _distribution_alpha(a, n)

    # ----- golden‑section refinement (10 iterations) --------------------
    phi = (np.sqrt(5.0) - 1.0) / 2.0  # ≈ 0.618
    delta = max_alpha / grid_points

            break
        m1 = lo + (hi - lo) * (1.0 - phi)
        m2 = lo + (hi - lo) * phi
        metric1 = _metric_alpha(m1, n)
        metric2 = _metric_alpha(m2, n)

        if metric1 < metric2:
            hi = m2
            if metric1 < best_metric:
                best_metric = metric1
                best_alpha = m1
                best_vec = _distribution_alpha(m1, n)
        else:
            lo = m1
            if metric2 < best_metric:
                best_metric = metric2
                best_alpha = m2
                best_vec = _distribution_alpha(m2, n)

    return best_alpha, best_vec, best_metric

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise the
    evaluation metric.  The routine first searches a one‑parameter family of
    regularised arcsine distributions, then refines the best candidate with an
    Adam optimiser on the simplex.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Approximate time budget in seconds.

    Returns
    -------
    List[float]
        The best sequence found within the allotted time.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for final logging / possible cleanup
    deadline = start_time + budget_s - 2.0

    # --------------------------------------------------------------
    # 0. Choose length n
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
        print(f"[Info] Using external length: n = {n}")
    except Exception:
        n = 20000
        print(f"[Info] No external length – using default n = {n}")

    # --------------------------------------------------------------
    # 1. Search over the regularised arcsine family
    # --------------------------------------------------------------
    print("[Info] Starting α‑search …")
    alpha_deadline = start_time + (budget_s * 0.15)  # about 15 % of the budget
    best_alpha, best_a, best_metric = _search_best_alpha(n, alpha_deadline)
    print(f"[Info] α‑search finished. best_alpha = {best_alpha:.6g}, "
          f"metric = {best_metric:.6f}")

    # --------------------------------------------------------------
    # 2. Adam refinement on the simplex (logits = log a)
    # --------------------------------------------------------------
    # initialise Adam from the best α‑candidate
    a = best_a.copy()
    best_score = best_metric
    best_vec = a.copy()

    # logits for the soft‑max parametrisation (ensures a stays on the simplex)
    z = np.log(a + 1e-16)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    # learning‑rate schedule
    base_alpha = 0.8          # larger initial step for faster progress
    eps_factor = 0.20          # initial smoothing factor for the soft‑max
    eps_factor_min = 5e-5
    eps_decay = 0.9997

    it = 0
    max_iters = 250_000

    # helper for progress logging
    def _log(it_, score_):
        elapsed = time.time() - start_time
        print(f"[Iter {it_:6d}] metric = {score_:.6f}  elapsed = {elapsed:.1f}s")

    while time.time() < deadline and it < max_iters:
        it += 1

        # probability vector (always sums to 1)
        a_cur = _softmax(z)

        # smoothed max‑convolution and its gradient
        max_conv, grad_a = _conv_max_and_grad(a_cur, eps_factor)

        cheap_score = 2.0 * n * max_conv

        # keep the best candidate seen so far
        if cheap_score < best_score:
            best_score = cheap_score
            best_vec = a_cur.copy()
            if it % 2000 == 0 or cheap_score < 1.02:
                _log(it, best_score)

        # natural‑gradient step on logits
        mu = np.dot(grad_a, a_cur)          # 〈∇, a〉
        grad_z = a_cur * (grad_a - mu)      # ∇_z J

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # inverse‑sqrt decay of the step size
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step = -alpha_t * m_hat / (np.sqrt(v_hat) + eps_adam)

        z = z + step
        # keep logits in a numerically safe range
        z = np.clip(z, -70.0, 70.0)

        # gradually reduce the smoothing of the soft‑max
        eps_factor = max(eps_factor_min, eps_factor * eps_decay)

        # occasional small random jitter to escape plateaus
        if it % 10000 == 0 and np.random.rand() < 0.15:
            jitter = 0.02 * np.random.randn(n)
            z += jitter

        # tiny early‑stop: if we already beat the target we can break early
        if best_score < 1.48 and it > 5000:
            print("[Info] Target passed – exiting optimisation early.")
            break

    # --------------------------------------------------------------
    # 3. Return the best sequence (as a plain Python list)
    # --------------------------------------------------------------
    final_seq = best_vec.tolist()

    # Optional final exact evaluation (only for logging)
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[Info] Final exact score = {final_score:.6f}")
    except Exception as exc:
        print(f"[Warning] Evaluation error (ignored): {exc}")

    return final_seq
