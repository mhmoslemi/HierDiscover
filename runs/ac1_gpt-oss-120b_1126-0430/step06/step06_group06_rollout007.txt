#!/usr/bin/env python
"""
Propose a non‑negative step function (list of heights) that minimises
``evaluate_sequence``.

The algorithm consists of three phases:

1. **Initialisation** – use the previously best construction (if any)
   and a few cheap parametric families to obtain a good starting point.

2. **Soft‑max‑gradient mirror descent** – minimise a smooth surrogate of
   the objective ``2·n·max_k (a*a)[k]``.
   The surrogate replaces the hard max by a soft‑max with an annealing
   temperature β.  In each iteration we:
   * compute the full auto‑convolution via a real FFT,
   * form soft‑max weights w_k,
   * obtain the gradient of the surrogate loss by a cross‑correlation
     (again via FFT),
   * perform a multiplicative (exponentiated‑gradient) update
     ``a ← a·exp(‑η·grad)`` and re‑project onto the probability simplex.
   β is increased slowly so that the optimisation first sees a smooth
   landscape and finally focuses on the true max.

3. **(1+1)‑ES hill‑climbing** – a simple mutation‑selection loop
   (multiplicative Gaussian perturbations in log‑space) that refines the
   final solution.

The whole routine respects the given time budget (default 1000 s) and
always returns a feasible sequence (non‑negative, Σa_i>0.01).  The code
uses only NumPy and runs comfortably within the allowed resources.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        # all non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·n‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len).real[: 2 * a.size - 1]
    return conv


def _softmax_gradient(a: np.ndarray,
                      fft_len: int,
                      beta: float) -> tuple[float, float, np.ndarray]:
    """
    Compute the soft‑max surrogate of max(a*a) and its gradient.

    Returns
    -------
    max_conv : float
        The exact maximum of the auto‑convolution (needed for cheap score).
    cheap   : float
        2·n·max_conv – the quantity minimise when Σa_i = 1.
    grad    : np.ndarray, shape (n,)
        Gradient of the surrogate loss w.r.t. the vector a.
    """
    n = a.shape[0]

    # --- convolution a*a -------------------------------------------------
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len).real[: 2 * n - 1]
    max_conv = conv.max()

    # --- soft‑max weights -------------------------------------------------
    # exp(beta * (c - max)) for numerical stability
    exp_term = np.exp(beta * (conv - max_conv))
    w = exp_term / exp_term.sum()                     # shape (2n‑1,)

    # --- gradient of the surrogate ---------------------------------------
    # The gradient of the smooth max is
    #   ∂/∂a_i  ( (1/β) log Σ_k e^{β conv_k} )
    # = Σ_k w_k * ∂ conv_k / ∂a_i  = Σ_k w_k * 2 * a_{k-i}
    # The term Σ_k w_k * a_{k-i} is a cross‑correlation between a and w.
    w_rev = w[::-1]                                   # length 2n‑1
    fw = np.fft.rfft(w_rev, fft_len)
    # linear cross‑correlation a ⋆ w (length n + (2n‑1) – 1 = 3n‑2)
    cross = np.fft.irfft(fa * fw, fft_len).real[: 3 * n - 2]

    # For index i we need Σ_k w_k a_{k-i} = cross[(2n‑2)‑i]
    idx = (2 * n - 2) - np.arange(n)
    g = 2.0 * cross[idx]                              # 2·Σ_k w_k a_{k-i}

    # Gradient of the cheap score (2·n·max) is 2·n·g
    grad = 2.0 * n * g

    cheap = 2.0 * n * max_conv
    return max_conv, cheap, grad


def _cheap_score(a: np.ndarray, fft_len: int) -> float:
    """Cheap objective (assumes Σa_i = 1): 2·n·max(a*a)."""
    conv = _conv_full_fft(a, fft_len)
    return 2.0 * a.shape[0] * conv.max()


# ----------------------------------------------------------------------
# Parametric families (all return a normalised probability vector)
# ----------------------------------------------------------------------


def _uniform_seq(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _power_law_seq(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = i ** (-p)
    seq /= seq.sum()
    return seq


def _exponential_seq(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    seq = np.exp(-lam * i)
    seq /= seq.sum()
    return seq


def _power_law_exp_seq(n: int, p: float, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    seq = (i + 1.0) ** (-p) * np.exp(-lam * i)
    seq /= seq.sum()
    return seq


def _stretched_exponential_seq(n: int, lam: float, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    seq = np.exp(-lam * (i ** gamma))
    seq /= seq.sum()
    return seq


def _beta_seq(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(1, n + 1, dtype=np.float64) / n)
    seq = (x ** alpha) * ((1.0 - x) ** beta)
    s = seq.sum()
    if s == 0.0:
        return None
    seq /= s
    return seq


def _dirichlet_seq(n: int, conc: float) -> np.ndarray:
    g = np.random.gamma(conc, 1.0, size=n)
    seq = g / g.sum()
    return seq

            if fail_cnt > 200:
                sigma = min(sigma * 1.30, 0.6)
                fail_cnt = 0

        # occasional gentle random perturbation from a fresh Dirichlet
        if iter_cnt % 200_000 == 0:
            pert = _dirichlet_seq(n, np.random.uniform(0.001, 2.0))
            a = _simplex_projection(best_iter_a + 0.02 * pert)

    # --------------------------------------------------------------
    # Phase 3 – final exact evaluation
    # --------------------------------------------------------------
    final_seq = best_iter_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:      # defensive – should never happen
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # --------------------------------------------------------------
    # Return the best sequence found.
    # --------------------------------------------------------------
    return final_seq
```
