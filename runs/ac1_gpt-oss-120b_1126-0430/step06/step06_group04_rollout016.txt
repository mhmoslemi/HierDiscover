def _generate_candidates(n, rng):
    cand = []
    # Uniform
    cand.append(np.full(n, 1.0/n))
    # Linear decreasing (simple)
    linear = (np.arange(n, 0, -1).astype(float))
    linear = linear / linear.sum()
    cand.append(linear)
    # Linear increasing (reverse)
    cand.append(linear[::-1])
    # Exponential r in [0.90,0.94,0.96,0.98,0.99,0.995]
    for r in [0.90, 0.94, 0.96, 0.98, 0.99, 0.995]:
        a = (r ** np.arange(n)).astype(float)
        a = a / a.sum()
        cand.append(a)
        cand.append(a[::-1])
    # Power law p in [0.2,0.5,0.8,1.0,1.2,1.5,2.0]
    for p in [0.2,0.5,0.8,1.0,1.2,1.5,2.0]:
        a = (np.arange(1, n+1) ** (-p)).astype(float)
        a = a / a.sum()
        cand.append(a)
        cand.append(a[::-1])
    # Beta shapes: alpha in [0.3,0.5,0.7,0.9,1.1], beta in [1.1,1.3,1.5,2.0,3.0]
    alphas = [0.3,0.5,0.7,0.9,1.1]
    betas = [1.1,1.3,1.5,2.0,3.0]
    for alpha in alphas:
        for beta in betas:
            idx = np.arange(1, n+1, dtype=float)
            rev = (n - idx + 1)
            a = (idx ** (alpha - 1.0)) * (rev ** (beta - 1.0))
            if np.any(np.isnan(a)) or np.any(np.isinf(a)):
                continue
            a_sum = a.sum()
            if a_sum <= 0:
                continue
            a = a / a_sum
            cand.append(a)
            cand.append(a[::-1])
    # Mixture with uniform: for each existing shape (excluding uniform), create mixture with uniform weight mix in [0.1,0.2,0.3]
    # We'll iterate over a copy of current cand (excluding uniform)
    base_shapes = cand[1:]  # exclude first uniform
    uniform = cand[0]
    for shape in base_shapes:
        for mix in [0.1,0.2,0.3]:
            mixed = (1 - mix) * shape + mix * uniform
            mixed = mixed / mixed.sum()
            cand.append(mixed)

    # Random sparse Dirichlet: k = max(2, int(0.02*n))
    k = max(2, int(0.02 * n))
    for _ in range(5):
        a = np.zeros(n, dtype=float)
        pos = rng.choice(n, size=k, replace=False)
        vals = rng.random(k)
        a[pos] = vals
        total = a.sum()
        if total <= 0:
            continue
        a = a / total
        cand.append(a)

    # Random Dirichlet (full)
    a = rng.random(n)
    a = a / a.sum()
    cand.append(a)

    return cand

    while time.time() - start < time_budget:
        it += 1

        # compute convolution and max index
        conv = _conv_self_fft(a)
        k_max = int(np.argmax(conv))
        max_conv = conv[k_max]
        obj = float(2 * n * max_conv)

        # gradient based step
        # compute gradient of max conv (exact)
        # g = 2 * correlation with one-hot at k_max
        # using O(n) direct method
        g = np.zeros_like(a)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        # indices i from lo to hi
        i = np.arange(lo, hi + 1)
        g[i] = a[k_max - i]
        g *= 2.0

        # try step in direction opposite gradient
        step = 0.5  # initial step scaling factor
        improved = False
        while step > 1e-8:
            a_new = _project_simplex(a - step * g)
            # fast compute objective
            conv_new = _conv_self_fft(a_new)
            obj_new = float(2 * n * conv_new.max())
            if obj_new < obj - 1e-14:
                a = a_new
                obj = obj_new
                improved = True
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_a = a.copy()
                break
            step *= 0.5
        if improved:
            continue

        # SA move: pick i with largest contribution to conv[k_max] (product)
        # compute contributions array
        contrib = np.zeros_like(a)
        # contributions for i in [lo,hi]: a[i] * a[k_max-i]
        contrib[i] = a[i] * a[k_max - i]
        # choose i with max contrib
        i_src = int(np.argmax(contrib))
        # choose destination j different from i_src uniformly
        j = rng.integers(n - 1)
        if j >= i_src:
            j += 1
        # move a fraction of mass from src to dst
        delta = a[i_src] * rng.random() * 0.07  # up to 7%
        if delta <= 0:
            continue
        a_new = a.copy()
        a_new[i_src] -= delta
        a_new[j] += delta
        # Normalize (mass preserved)
        # Ensure non-negative
        a_new = np.clip(a_new, 0.0, None)
        a_new = a_new / a_new.sum()
        # evaluate
        conv_new = _conv_self_fft(a_new)
        obj_new = float(2 * n * conv_new.max())
        # accept if better, else Metropolis
        if obj_new < obj - 1e-14:
            a = a_new
            obj = obj_new
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a.copy()
        else:
            # acceptance probability
            if T > 0 and rng.random() < math.exp(-(obj_new - obj) / T):
                a = a_new
                obj = obj_new
        # update temperature occasionally
        if it % 2000 == 0:
            T = max(T * decay, T_min)
        # occasional jitter to escape plateau
        if it % 5000 == 0:
            noise = rng.normal(scale=0.005, size=n)
            a = np.clip(a + noise, 0.0, None)
            a = a / a.sum()
    return best_a.tolist(), best_obj

"""
Non‑negative step‑function optimiser (v3).

The optimiser works in two phases:

1. **Broad parametric search** – a large set of simple analytic families
   (uniform, linear, exponential, power‑law, Beta‑distribution, sine‑based,
   sparse Dirichlet) is sampled for several support lengths.  For each length
   the best shape (lowest exact objective) is recorded.  The search is
   limited by a per‑length time quota.

2. **Targeted refinement** – the globally best shape found in phase‑1 is
   refined with a hybrid Projected‑Gradient / Simulated‑Annealing scheme.
   The exact max‑convolution gradient is used (no smooth‑max
   approximation).  Mass is moved away from the indices that contribute most
   to the current maximal autocorrelation entry; occasional random jitter
   and an annealing schedule help escape plateaus.

The whole procedure respects the 1000 s budget and returns a plain list of
non‑negative floats suitable for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return conv_full[: 2 * n - 1]


def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) assuming Σ a = 1.
    Returns a Python float.
    """
    n = a.size
    conv = _conv_self_fft(a)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0, Σ x = 1}.
    Implements the O(n log n) algorithm of Duchi et al.
    """
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape generators (all produce a probability vector, i.e. sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _linear_descending(n: int) -> np.ndarray:
    a = np.arange(n, 0, -1, dtype=float)
    return a / a.sum()


def _exponential(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=float)
    return a / a.sum()


def _power_law(n: int, p: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=float) ** (-p))
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    rev = n - i + 1
    a = (i ** (alpha - 1.0)) * (rev ** (beta - 1.0))
    if np.any(np.isnan(a)) or np.any(np.isinf(a)):
        return None
    s = a.sum()
    if s <= 0.0:
        return None
    return a / s


def _sine_shape(n: int, power: float) -> np.ndarray:
    i = np.arange(n, dtype=float) + 0.5
    a = np.sin(np.pi * i / n) ** power
    if a.sum() == 0.0:
        return None
    return a / a.sum()


# ----------------------------------------------------------------------
# Candidate generator (returns a list of ndarray candidates)
# ----------------------------------------------------------------------


def _generate_candidates(n: int, rng: np.random.Generator):
    cand = []

    # ---- deterministic families -------------------------------------------------
    cand.append(_uniform(n))

    cand.append(_linear_descending(n))
    cand.append(_linear_descending(n)[::-1])

    # exponential family
    for r in (0.90, 0.94, 0.96, 0.98, 0.99, 0.995):
        a = _exponential(n, r)
        cand.append(a)
        cand.append(a[::-1])

    # power‑law family
    for p in (0.2, 0.5, 0.8, 1.0, 1.2, 1.5, 2.0):
        a = _power_law(n, p)
        cand.append(a)
        cand.append(a[::-1])

    # Beta (α,β) family – decreasing shapes (α<1, β>1)
    alphas = (0.3, 0.5, 0.7, 0.9, 1.1)
    betas = (1.1, 1.3, 1.5, 2.0, 3.0)
    for α in alphas:
        for β in betas:
            a = _beta_shape(n, α, β)
            if a is not None:
                cand.append(a)
                cand.append(a[::-1])

    # sine‑based family
    for power in (0.5, 1.0, 2.0, 3.0):
        a = _sine_shape(n, power)
        if a is not None:
            cand.append(a)
            cand.append(a[::-1])

    # ---- mixtures with the uniform distribution ---------------------------------
    uniform = cand[0]  # the first entry is uniform
    for base in list(cand[1:]):  # exclude uniform itself
        for mix in (0.1, 0.2, 0.3):
            m = (1.0 - mix) * base + mix * uniform
            cand.append(m / m.sum())

    # ---- random sparse Dirichlet -------------------------------------------------
    k = max(2, int(0.02 * n))
    for _ in range(5):
        a = np.zeros(n, dtype=float)
        pos = rng.choice(n, size=k, replace=False)
        vals = rng.random(k)
        a[pos] = vals
        a = a / a.sum()
        cand.append(a)

    # ---- dense random Dirichlet -------------------------------------------------
    a = rng.random(n)
    cand.append(a / a.sum())

    return cand


# ----------------------------------------------------------------------
# Targeted refinement (projected‑gradient + SA)
# ----------------------------------------------------------------------


def _refine_sequence(seq: np.ndarray,
                     time_budget: float,
                     rng: np.random.Generator):
    """
    Refine the given probability vector using a hybrid projected‑gradient
    and simulated‑annealing scheme.
    """
    a = np.asarray(seq, dtype=float)
    n = a.size
    best_a = a.copy()
    best_obj = _objective(best_a)

    start = time.time()
    it = 0

    # Annealing parameters (used only for the SA part)
    T = best_obj * 0.2 + 1e-12
    T_min = best_obj * 1e-7 + 1e-14
    decay = 0.9995

    while time.time() - start < time_budget:
        it += 1

        # current convolution and max index
        conv = _conv_self_fft(a)
        k_max = int(np.argmax(conv))

        # ----- exact gradient of the hard max -----
        # g[i] = 2 * a[k_max - i] for all valid i
        g = np.zeros_like(a)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        g[idx] = a[k_max - idx]
        g *= 2.0

        # ----- projected gradient step with simple back‑tracking -----
        step = 0.5
        improved = False
        while step > 1e-8:
            a_new = _project_simplex(a - step * g)
            obj_new = _objective(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                improved = True
                a = a_new
                break
            elif obj_new < _objective(a) - 1e-14:
                # accept a non‑global improvement to move forward
                a = a_new
                improved = True
                break
            step *= 0.5

        if improved:
            continue

        # ----- SA move: target the index that contributes most to conv[k_max] -----
        # contributions[i] = a[i] * a[k_max-i] (zero when out‑of‑range)
        contrib = np.zeros_like(a)
        contrib[idx] = a[idx] * a[k_max - idx]
        i_src = int(np.argmax(contrib))

        # pick destination uniformly (different from source)
        j = rng.integers(n - 1)
        if j >= i_src:
            j += 1

        # move a modest fraction of mass
        delta = a[i_src] * rng.random() * 0.07  # ≤ 7 % of a[i_src]
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i_src] -= delta
        a_new[j] += delta
        a_new = np.clip(a_new, 0.0, None)          # protect against negatives
        a_new = a_new / a_new.sum()                # re‑normalise (projection is cheap)

        obj_new = _objective(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Metropolis acceptance
            if T > 0 and rng.random() < math.exp(-(obj_new - _objective(a)) / T):
                a = a_new

        # annealing schedule
        if it % 2000 == 0:
            T = max(T * decay, T_min)

        # occasional global jitter to shake the system
        if it % 5000 == 0:
            noise = rng.normal(scale=0.005, size=n)
            a = np.clip(a + noise, 0.0, None)
            a = a / a.sum()

    return best_a, best_obj

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **_ignore) -> list[float]:
    """
    Optimise a non‑negative step‑function under the objective used by
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1. Parametric pre‑search over a few support lengths
    # ------------------------------------------------------------------
    # chosen lengths – longer supports give more flexibility but cost more
    lengths = [2000, 4000, 6000, 8000, 12000, 20000]

    global_best_seq = None
    global_best_obj = float("inf")
    best_n = None

    for idx, n in enumerate(lengths):
        now = time.time()
        remaining = deadline - now
        if remaining < 6.0:                         # need a few seconds to finish
            break

        # allocate a modest proportion of the remaining budget for this length
        # (≈12 % per length, capped at 30 s)
        shape_time = min(remaining * 0.12, 30.0)
        shape_start = time.time()

        candidates = _generate_candidates(n, rng)
        best_local_a = None
        best_local_obj = float("inf")

        for a in candidates:
            # honour the per‑length time budget
            if time.time() - shape_start > shape_time:
                break
            obj = _objective(a)
            if obj < best_local_obj:
                best_local_obj = obj
                best_local_a = a

        if best_local_a is None:
            continue

        print(f"[init] n={n:5d}  best shape obj={best_local_obj:.6f}")

        if best_local_obj < global_best_obj:
            global_best_obj = best_local_obj
            global_best_seq = best_local_a.copy()
            best_n = n
            print(f"[global] new overall best n={n}  O={global_best_obj:.6f}")

        # early exit if we are already close to the target
        if global_best_obj < 1.45:
            break

    # ------------------------------------------------------------------
    # 2. Targeted refinement of the best candidate found so far
    # ------------------------------------------------------------------
    if global_best_seq is None:
        # very unlikely – fall back to a simple uniform sequence
        n = 2000
        global_best_seq = _uniform(n)
        global_best_obj = _objective(global_best_seq)

    remaining = deadline - time.time()
    if remaining > 5.0:
        refined_seq, refined_obj = _refine_sequence(global_best_seq,
                                                    remaining,
                                                    rng)
        if refined_obj < global_best_obj:
            global_best_obj = refined_obj
            global_best_seq = refined_seq
            print(f"[refine] improved to O={global_best_obj:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check (for debugging – does not affect the returned value)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(global_best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    # ------------------------------------------------------------------
    # Clip to the evaluator's allowed range and guard against tiny sums
    # ------------------------------------------------------------------
    best = np.clip(global_best_seq, 0.0, 1000.0)
    if best.sum() < 0.01:
        best = np.ones_like(best) / best.size

    return best.tolist()
