"""
Smooth‑max Adam optimiser for the non‑negative step‑function problem.

The routine replaces the hard‑max gradient (used in the reference) by a
temperature‑controlled soft‑max (log‑sum‑exp) of the convolution entries.
Starting with a very low temperature gives a gradient that flattens the
entire autocorrelation; the temperature is increased exponentially so the
optimizer finally concentrates on the true maximum.  The optimisation is
performed in log‑it (exponentiated‑gradient) space, which automatically
keeps the coefficients on the simplex.  Several restarts, jitter injections
and a final simulated‑annealing refinement are used within the allotted
time budget.

The function `propose_candidate` follows the required signature and
returns the best sequence found.
"""

import time
import numpy as np
from typing import List, Tuple


# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return conv_full[: 2 * n - 1]


def _conv_full_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using a real FFT."""
    L = _next_pow2(len(x) + len(y) - 1)
    xf = np.fft.rfft(x, n=L)
    yf = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(xf * yf, n=L)
    return conv[: len(x) + len(y) - 1]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces a probability vector)."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


# ----------------------------------------------------------------------
# Optimiser – smooth‑max Adam on the simplex
# ----------------------------------------------------------------------


def _run_opt_smoothmax(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
) -> Tuple[List[float], float]:
    """
    Adam optimiser that minimises the smoothed max of the self‑convolution.
    Returns the best sequence (as a list) and its hard‑max objective
    value 2·n·max(conv(a,a)).
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # initialise a on the simplex
    if init_a is not None and init_a.size == n:
        a = np.asarray(init_a, dtype=float).copy()
    else:
        a = rng.random(n)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    a = a / s if s > 1e-12 else np.ones(n) / n

    # logits for exponentiated‑gradient updates
    x = np.log(a + 1e-16)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # learning‑rate schedule (exponential decay)
    lr_start = 0.6
    lr_end = 0.001

    # temperature (smooth‑max) schedule – log‑linear increase
    tau_start = 0.1      # very smooth
    tau_end = 5000.0     # practically hard

    best_a = a.copy()
    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())

    it = 0
    while time.time() - start < time_limit:
        it += 1

        # simplex point
        a = _softmax(x)

        # linear self‑convolution
        conv = _conv_self_fft(a)
        M = conv.max()
        obj = float(2.0 * n * M)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[smooth n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        # temperature for this iteration (log‑linear schedule)
        elapsed = time.time() - start
        t = min(elapsed / time_limit, 1.0)
        tau = np.exp(np.log(tau_start) + (np.log(tau_end) - np.log(tau_start)) * t)

        # soft‑max weights over convolution entries
        # numerically stable: subtract max before exp
        c_max = conv.max()
        w = np.exp(tau * (conv - c_max))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.ones_like(w) / w.size
        else:
            w /= w_sum

        # gradient of the smoothed objective w.r.t. a
        # grad_a = 2 * (a ⋆ w) where ⋆ denotes cross‑correlation
        w_rev = w[::-1]
        cross = _conv_full_fft(w_rev, a)            # length = 3n‑2
        grad_a = 2.0 * cross[n - 1 : n - 1 + n]      # slice of length n

        # mirror (natural) gradient in log‑it space
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # learning‑rate at this time
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # Adam step in log‑it space
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape plateaus
        if it % 4000 == 0:
            jitter = rng.normal(scale=0.015, size=n)
            x += jitter

        # full restart from the current best every so often
        if it % 70000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.006, size=n)
            m.fill(0.0)
            v.fill(0.0)

    # ------------------------------------------------------------------
    # prune negligible entries (they increase n without helping)
    thresh = 1e-13
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        if best_a.sum() == 0.0:
            best_a = np.ones_like(best_a) / best_a.size
        else:
            best_a = best_a / best_a.sum()
        # recompute objective after compression
        best_obj = float(2.0 * best_a.size * _conv_self_fft(best_a).max())
        print(f"[smooth compress] reduced length → {best_a.size}, O={best_obj:.6f}")

    return best_a.tolist(), best_obj


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (same as the reference, slightly tuned)
# ----------------------------------------------------------------------


def _run_sa(
    init_seq: List[float],
    time_limit: float,
    seed: int = 0,
) -> Tuple[List[float], float]:
    """Refine a sequence by modest mass‑transfer moves with a Metropolis schedule."""
    start = time.time()
    rng = np.random.default_rng(seed)
    a = np.asarray(init_seq, dtype=float)
    if a.sum() < 1e-12:
        a = np.ones_like(a) / a.size
    else:
        a = a / a.sum()
    n = a.shape[0]

    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())
    best_seq = a.copy()

    # temperature schedule (starts relatively hot, cools exponentially)
    T = best_obj * 0.15 + 1e-6
    T_min = best_obj * 1e-6 + 1e-9
    decay = 0.994
    it = 0

    while time.time() - start < time_limit:
        it += 1
        k_max = int(np.argmax(conv))

        # contribution of each index to the maximal entry
        i_vals = np.arange(n)
        j_vals = k_max - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i_vals[mask]] * a[j_vals[mask]]
        total_contrib = contrib.sum()

        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = rng.choice(n, p=probs)
        else:
            i = rng.integers(n)

        # choose a distinct destination uniformly
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # transfer a modest fraction (up to 7 %) of mass from i to j
        delta = a[i] * rng.random() * 0.07
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.clip(a_new, 0.0, None)
        a_new /= a_new.sum()

        conv_new = _conv_self_fft(a_new)
        obj_new = float(2.0 * n * conv_new.max())

        # Metropolis acceptance
        if obj_new < best_obj - 1e-12:
            a, conv, best_obj, best_seq = a_new, conv_new, obj_new, a_new.copy()
            print(f"[SA it={it}] new best O={best_obj:.6f}")
        else:
            prob = np.exp(-(obj_new - best_obj) / T) if T > 0 else 0.0
            if rng.random() < prob:
                a, conv = a_new, conv_new

        # temperature decay
        if it % 2500 == 0:
            T = max(T * decay, T_min)

        # occasional global jitter
        if it % 12000 == 0:
            noise = rng.normal(scale=0.012, size=n)
            a = np.clip(a + noise, 0.0, None)
            a /= a.sum()
            conv = _conv_self_fft(a)

    # final compression of tiny entries
    thresh = 1e-13
    mask = best_seq > thresh
    if mask.sum() < best_seq.size:
        comp = best_seq[mask]
        comp /= comp.sum()
        comp_obj = float(2.0 * comp.size * _conv_self_fft(comp).max())
        if comp_obj < best_obj:
            best_seq, best_obj = comp, comp_obj
            print(f"[SA compress] reduced length → {comp.size}, O={best_obj:.6f}")

    return best_seq.tolist(), best_obj

def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **_ignore,
) -> List[float]:
    """
    Find a non‑negative step‑function minimising the evaluation metric.
    The routine runs several independent smooth‑max Adam optimisations
    (with different seeds and possibly different lengths) and finishes
    with a targeted simulated‑annealing refinement.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.3  # small safety margin

    # ------------------------------------------------------------------
    # Prepare an initial sequence from the global variable if available.
    # ------------------------------------------------------------------
    init_seq: np.ndarray | None = None
    if 'height_sequence_1' in globals():
        seq0 = np.asarray(height_sequence_1, dtype=float)
        seq0 = np.clip(seq0, 0.0, None)
        if seq0.sum() < 0.01:
            seq0 = np.ones_like(seq0)
        seq0 = seq0 / seq0.sum()
        init_seq = seq0

    # ------------------------------------------------------------------
    # Decide how many independent optimisation runs we can afford.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    # At least one run with a relatively large support.
    runs = 1
    # If we have a lot of time, schedule a second run with a different size.
    if remaining > 250.0:
        runs = 2

    best_seq = None
    best_obj = float('inf')
    used_seeds = []

    for r in range(runs):
        # allocate a fair share of the remaining budget
        now = time.time()
        remaining = deadline - now
        if remaining <= 5.0:
            break
        # split remaining roughly 85 % for Adam, 15 % for SA
        adam_budget = remaining * 0.85
        sa_budget = remaining - adam_budget

        # target length – larger for the first run, a bit smaller for the second
        if runs == 1:
            target_n = 18000 if adam_budget > 120.0 else 12000
        else:
            target_n = 18000 if r == 0 else 13000

        # Upsample the initial sequence (if any) to the desired length.
        if init_seq is not None and init_seq.size != target_n:
            x_old = np.linspace(0.0, 1.0, init_seq.size, endpoint=False)
            x_new = np.linspace(0.0, 1.0, target_n, endpoint=False)
            init_a = np.interp(x_new, x_old, init_seq)
            init_a = np.clip(init_a, 0.0, None)
            if init_a.sum() == 0.0:
                init_a = np.ones(target_n) / target_n
            else:
                init_a = init_a / init_a.sum()
        else:
            init_a = init_seq

        run_seed = seed + 1000 * r + 7
        used_seeds.append(run_seed)

        print(f"[run {r + 1}/{runs}] target_n={target_n}, adam_budget={adam_budget:.1f}s")
        seq, obj = _run_opt_smoothmax(
            n=target_n,
            time_limit=adam_budget,
            init_a=init_a,
            seed=run_seed,
        )
        print(f"[run {r + 1}] Adam finished – O={obj:.6f}")

        # keep the best so far (hard‑max objective)
        if obj < best_obj:
            best_obj, best_seq = obj, np.asarray(seq, dtype=float)

        # ------------------------------------------------------------------
        # Simulated‑annealing refinement (if time permits)
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > sa_budget and remaining > 5.0:
            print(f"[run {r + 1}] starting SA refinement (budget {sa_budget:.1f}s)")
            seq_ref, obj_ref = _run_sa(best_seq.tolist(), time_limit=remaining, seed=run_seed + 999)
            if obj_ref < best_obj:
                best_obj, best_seq = obj_ref, np.asarray(seq_ref, dtype=float)
                print(f"[run {r + 1}] SA improved O to {best_obj:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check – call the supplied evaluator.
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_seq.tolist()
