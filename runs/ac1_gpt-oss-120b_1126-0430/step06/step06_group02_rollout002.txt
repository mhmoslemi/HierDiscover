   – computes the exact convolution via an FFT,
   – finds all indices attaining the maximal convolution value,
   – builds the exact (sub‑)gradient of the objective from this *active set*,
   – updates the logits with Adam (mirror descent on the simplex);
4. injects occasional jitter, restarts from the best seen point when stagnation
   occurs and optionally upsamples the support to give more degrees of freedom;
5. after the time budget returns the best distribution (as a plain list) and
   prints its exact `evaluate_sequence` score.

The use of the exact active‑set gradient (instead of a smoothed soft‑max) gives
a clearer descent direction and, together with longer runs and a modest
upsampling stage, typically pushes the bound below the target 1.503.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures or lambdas
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max, returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    s = e.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / z.size)
    return e / s


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it that reproduces `a` after a soft‑max."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """Smallest power‑of‑two ≥ 3·n‑2 (enough for all needed convolutions)."""
    return 1 << ((3 * n - 2).bit_length())


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    s = new_a.sum()
    if s == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= s
    return new_a


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{‑p}, normalised."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _sym_product_power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ ((i+1)*(n‑i))^{‑p}, symmetric, normalised."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) * (n - i)) ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _exp_decay_distribution(n: int, lam: float) -> np.ndarray:
    """a_i ∝ e^{‑lam·i}+e^{‑lam·(n‑1‑i)}, symmetric exponential."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _logistic_distribution(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ 1/(1+e^{α·(i‑mid)}), normalised; heavy on the left side."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = 1.0 / (1.0 + np.exp(alpha * (i - mid)))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_active_grad(a: np.ndarray):
    """
    Compute (max_conv, grad_a) where grad_a = ∇_a[2·n·max(a*a)] using the exact
    active set (all k attaining the maximal convolution value).
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    # FFT of a (zero‑padded)
    a_fft = np.fft.rfft(a, L)

    # Linear self‑convolution a*a
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]

    max_val = float(conv.max())
    # Build an active‑set weight vector w (normalized)
    active = np.isclose(conv, max_val, rtol=1e-12, atol=0.0)
    cnt = int(active.sum())
    if cnt == 0:
        # fallback – single maximum
        w = np.zeros(2 * n - 1, dtype=np.float64)
        w[np.argmax(conv)] = 1.0
        cnt = 1
    else:
        w = active.astype(np.float64) / cnt

    # Correlation a ⋆ w  (gives ∂max_conv/∂a up to factor 2)
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # Gradient of the full objective 2·n·max_conv
    grad_a = 4.0 * n * corr
    return max_val, grad_a


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) improves the
    `evaluate_sequence` score.  The routine respects the given time budget.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0   # small safety margin

    # --------------------------------------------------------------
    # 0. Choose an initial support size and an initial distribution.
    # --------------------------------------------------------------
    n = 8000                     # moderate size – cheap enough for many updates
    best_dist = None
    best_score = float("inf")

    # Try to warm‑start from the global sequence if the platform provides it.
    try:
        # `height_sequence_1` is supplied by the platform when available.
        init_global = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        best_dist = _interpolate_distribution(init_global, n)
        best_score = 2.0 * n * _conv_max(best_dist)
        print("[Init] Warm‑started from global sequence (down‑sampled).")
    except Exception:
        # No global – scan a few parametric families and keep the best cheap score.
        print("[Init] Scanning parametric families for a good start.")
        candidates = []

        # Power‑law
        for p in np.linspace(0.6, 1.4, 17):
            candidates.append(_power_law_distribution(n, p))

        # Symmetric product‑power‑law
        for p in np.linspace(0.6, 1.4, 17):
            candidates.append(_sym_product_power_law_distribution(n, p))

        # Symmetric exponential decay
        for loglam in np.linspace(-6, -2, 15):
            lam = 10.0 ** loglam
            candidates.append(_exp_decay_distribution(n, lam))

        # Logistic (left‑heavy)
        for alpha in np.linspace(0.3, 3.0, 20):
            candidates.append(_logistic_distribution(n, alpha))

        # Uniform fallback (always present)
        candidates.append(np.full(n, 1.0 / n))

        for cand in candidates:
            score = 2.0 * n * _conv_max(cand)
            if score < best_score:
                best_score = score
                best_dist = cand.copy()
        print(f"[Init] Selected start with cheap score = {best_score:.6f}")

    # Slight jitter before optimisation and initialise logits
    z = _logits_from_distribution(best_dist) + 0.02 * np.random.randn(n)

    # --------------------------------------------------------------
    # 1. Adam state on logits (mirror descent on the simplex)
    # --------------------------------------------------------------
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_alpha = 0.55          # base learning‑rate for logits

    # bookkeeping
    global_best_dist = best_dist.copy()
    global_best_score = best_score
    last_improve_iter = 0
    it = 0

    # --------------------------------------------------------------
    # 2. Optimisation loop
    # --------------------------------------------------------------
    while time.time() < deadline and it < 5_000_000:
        it += 1

        # ----- Current point on the simplex -----
        a = _softmax(z)

        # ----- Exact max‑convolution and exact sub‑gradient -----
        max_conv, grad_a = _conv_max_and_active_grad(a)
        cheap_score = 2.0 * n * max_conv

        # ----- Keep best ever seen -----
        if cheap_score < global_best_score - 1e-12:
            global_best_score = cheap_score
            global_best_dist = a.copy()
            last_improve_iter = it
            if it % 20000 == 0:
                print(f"[Iter {it:07d}] NEW BEST cheap = {cheap_score:.6f}")

        # ----- Gradient w.r.t. logits (mirror descent) -----
        mu = np.dot(grad_a, a)               # <grad, a>
        grad_z = a * (grad_a - mu)

        # ----- Adam moment updates (bias‑corrected) -----
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # ----- Step‑size schedule (inverse‑sqrt decay) -----
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step_dir = - m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- Update logits (with clipping for stability) -----
        z = z + alpha_t * step_dir
        z = np.clip(z, -50.0, 50.0)

        # ----- Occasional jitter to escape plateaux -----
        if (it % 4000 == 0) and (np.random.rand() < 0.15):
            z += 0.015 * np.random.randn(n)
            m *= 0.6
            v *= 0.6

        # ----- Restart if stagnation lasts too long -----
        if it - last_improve_iter > 250_000:
            print(f"[Iter {it}] Stagnation detected – restarting from best point.")
            z = _logits_from_distribution(global_best_dist) + 0.03 * np.random.randn(n)
            m.fill(0.0)
            v.fill(0.0)
            last_improve_iter = it

        # ----- Optional up‑sampling stage (once) -----
        # Give the optimiser a chance to refine on a finer grid.
        if (it % 250_000 == 0) and (n < 16000) and (time.time() + 30.0 < deadline):
            new_n = min(16000, int(n * 1.6))
            print(f"[Iter {it}] Upsampling support: {n} → {new_n}")
            global_best_dist = _interpolate_distribution(global_best_dist, new_n)
            best_dist = global_best_dist.copy()
            n = new_n
            # Re‑initialise everything for the new size
            z = _logits_from_distribution(global_best_dist) + 0.005 * np.random.randn(n)
            m = np.zeros_like(z)
            v = np.zeros_like(z)
            # keep the same iteration counter – step‑size will naturally reset
            # (the `it` variable continues counting globally)

    # --------------------------------------------------------------
    # 3. Final exact evaluation with the provided `evaluate_sequence()`
    # --------------------------------------------------------------
    final_seq = global_best_dist.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # --------------------------------------------------------------
    # 4. Return the best sequence discovered.
    # --------------------------------------------------------------
    return final_seq
