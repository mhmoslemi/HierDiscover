                shifted[:lo] = 0.0

            grad = 4.0 * n * (shifted - maxb)
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            a = a - step
            # projection onto simplex
            a = _simplex_projection(a)
            # evaluate cheap score
            cheap = 2.0 * n * maxb
            if cheap < best_local_val - 1e-12:
                best_local_val = cheap
                best_local_a = a.copy()
                stagnation = 0
                print(f"[Adam {t}] new local best cheap = {cheap:.6f}")
            else:
                stagnation += 1
            # occasional restart if stagnated
            if stagnation > 30000:
                # perturb around best_local_a
                pert = best_local_a + np.random.randn(n) * 0.001
                a = _simplex_projection(pert)
                m.fill(0.0)
                v.fill(0.0)
                stagnation = 0
                print(f"[Adam {t}] restart after stagnation.")
        return best_local_a, best_local_val

    global_best_a = best_a
    global_best_val = best_val

    for idx, (val, a0) in enumerate(top):
        if time.time() >= deadline:
            break
        print(f"[Start Adam {idx}] init cheap={val:.6f}")
        a_opt, val_opt = _run_adam(a0, max_iters=200000)
        if val_opt < global_best_val:
            global_best_val = val_opt
            global_best_a = a_opt.copy()
            print(f"[Global] new best cheap after Adam {idx} = {val_opt:.6f}")

    # Permutation local search on global best
    a = global_best_a.copy()
    best_swap_a = a.copy()
    best_swap_val = global_best_val
    temperature = 1.0
    cooling = 0.99995

    while time.time() < deadline:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # swap
        a[i], a[j] = a[j], a[i]
        conv = _conv_full_fft(a, fft_len)
        cheap = 2.0 * n * conv.max()
        accept = False
        if cheap < best_swap_val - 1e-12:
            accept = True
        else:
            delta = cheap - best_swap_val
            prob = math.exp(-delta / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True
        if accept:
            if cheap < best_swap_val - 1e-12:
                best_swap_val = cheap
                best_swap_a = a.copy()
                print(f"[Swap] new best cheap = {cheap:.6f}")
        else:
            # revert
            a[i], a[j] = a[j], a[i]
        temperature = max(temperature * cooling, 1e-6)

    final_seq = best_swap_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:
        final_score = float('inf')
        print(f"Error during final evaluation: {exc}")

    return final_seq

#!/usr/bin/env python
"""
Propose a non‑negative step function (list of heights) that minimises
``evaluate_sequence``.

Algorithm
---------
1. **Initial families** – generate a diverse set of candidate vectors
   (uniform, exponential, power‑law, beta, arcsine, sinusoid, Gaussian,
   random Dirichlet, …) and keep the few best according to the cheap
   surrogate 2·n·max(conv) (the true score when the sum equals 1).

2. **Continuous optimisation** – for each top candidate run an
   Adam‑style mirror‑descent on the simplex.  The exact gradient of the
   cheap surrogate is known analytically:
        ∂f/∂a_i = 4·n·(a_{k*‑i} – max_b)
   where k* is the index attaining the maximal convolution value
   max_b = (a∗a)[k*].  After every step the vector is projected back onto
   the probability simplex.

3. **Permutation local search** – the best vector from the previous
   stage is further improved by simulated‑annealing style random swaps.
   Each swap is accepted if it reduces the cheap score, otherwise with
   a probability that decays with temperature.

4. **Final exact evaluation** – the final candidate is evaluated with
   ``evaluate_sequence`` and returned.

The routine respects a user supplied time budget (default 1000 s) and
always returns a feasible sequence.  All helper functions are defined
at module level (no closures, no lambdas)."""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using a power‑of‑two FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa).real[: 2 * a.size - 1]
    return conv


def _conv_max_fft(a: np.ndarray, fft_len: int) -> tuple[float, int]:
    """Maximum value of the autocorrelation a*a and its index."""
    conv = _conv_full_fft(a, fft_len)
    idx = int(conv.argmax())
    return float(conv[idx]), idx


# ----------------------------------------------------------------------
# Core search routine required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that
    minimises ``evaluate_sequence``.  The routine adapts its schedule
    to the available time budget and always returns a feasible sequence.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin to guarantee the final exact evaluation
    deadline = start_time + budget_s - 5.0
    eps = 1e-30

    # --------------------------------------------------------------
    # Choose a working length (larger length gives a tighter theoretical lower bound)
    # --------------------------------------------------------------
    try:
        prev = height_sequence_1          # may be defined globally
        n0 = len(prev)
        n = max(n0, 8000)                # try to enlarge the previous size
    except Exception:
        n = 8000
    n = min(max(n, 5000), 20000)          # keep it in a reasonable interval

    # FFT length – next power of two ≥ 2·n‑1
    fft_len = 1 << ((2 * n - 1).bit_length())

    # --------------------------------------------------------------
    # Cheap surrogate: 2·n·max(conv)   (valid because we keep sum(a)=1)
    # --------------------------------------------------------------
    def cheap_score(a: np.ndarray) -> float:
        maxb, _ = _conv_max_fft(a, fft_len)
        return 2.0 * n * maxb

    # --------------------------------------------------------------
    # 0. Generate a variety of candidate families
    # --------------------------------------------------------------
    candidates = []

    # Uniform baseline
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # Previous construction, if it has the same length
    try:
        prev_arr = np.asarray(prev, dtype=np.float64)
        if prev_arr.shape[0] == n and prev_arr.sum() > 0.0:
            prev_arr = prev_arr / prev_arr.sum()
            candidates.append(prev_arr.copy())
    except Exception:
        pass

    # Exponential (left‑hand side)
    for lam in np.linspace(0.0, 10.0, 21):
        a = np.exp(-lam * np.arange(n, dtype=np.float64))
        a /= a.sum()
        candidates.append(a)

    # Symmetric exponential (U‑shaped)
    idx = np.arange(n, dtype=np.float64)
    for lam in np.linspace(0.0, 10.0, 21):
        a = np.exp(-lam * np.minimum(idx, n - 1 - idx))
        a /= a.sum()
        candidates.append(a)

    # Power‑law (left)
    for p in np.linspace(0.01, 3.0, 31):
        a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
        a /= a.sum()
        candidates.append(a)

    # Symmetric power‑law (U‑shaped)
    for p in np.linspace(0.01, 3.0, 31):
        a = np.minimum(idx + 1, n - idx) ** (-p)
        a /= a.sum()
        candidates.append(a)

    # Beta family (U‑shaped when α,β<1)
    x = (np.arange(1, n + 1, dtype=np.float64) / (n + 1.0))
    alphas = np.concatenate((
        np.linspace(0.01, 0.2, 5),
        np.linspace(0.2, 1.0, 9),
        np.linspace(1.0, 5.0, 5)
    ))
    betas = alphas.copy()
    for alpha in alphas:
        for beta in betas:
            a = (x ** alpha) * ((1.0 - x) ** beta)
            s = a.sum()
            if s > 0.0:
                a /= s
                candidates.append(a)

    # Arcsine shape: 1 / sqrt(i·(n‑i))
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((i + 1) * (n - i))
    a /= a.sum()
    candidates.append(a)

    # Sine shape (positive half‑wave)
    a = np.sin(np.pi * (i + 0.5) / n)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    candidates.append(a)

    # Cosine shape (U‑shaped)
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
    a /= a.sum()
    candidates.append(a)

    # Gaussian bumps of varying width
    mu = (n - 1) / 2.0
    for sigma in (n / 10.0, n / 6.0, n / 4.0):
        a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
        a /= a.sum()
        candidates.append(a)

    # Random Dirichlet seeds
    for conc in [0.001, 0.01, 0.1, 0.5, 1.0, 2.0, 5.0]:
        g = np.random.gamma(conc, 1.0, size=n)
        a = g / g.sum()
        candidates.append(a)

    # --------------------------------------------------------------
    # Keep the best few from the coarse sweep
    # --------------------------------------------------------------
    best_a = None
    best_val = float("inf")
    top = []                       # (value, vector) pairs
    for a in candidates:
        if time.time() >= deadline:
            break
        val = cheap_score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
        top.append((val, a.copy()))
    top.sort(key=lambda x: x[0])
    top_k = 5
    top = top[:top_k]

    print(f"[Init] n={n}, best cheap={best_val:.6f}")

    # --------------------------------------------------------------
    # 1. Continuous optimisation (Adam on the simplex)
    # --------------------------------------------------------------
    def _run_adam(start_a: np.ndarray, max_iters: int = 200_000):
        """Adam‑style optimisation of the cheap surrogate."""
        a = start_a.copy()
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-8
        lr = 0.05                     # base learning rate
        t = 0
        best_local_a = a.copy()
        best_local_val = cheap_score(a)
        stagnated = 0

        while time.time() < deadline and t < max_iters:
            t += 1

            maxb, kstar = _conv_max_fft(a, fft_len)

            # build a_shifted[i] = a_{kstar‑i} (0 if out of range)
            shifted = np.empty_like(a)
            if kstar < n:
                shifted[:kstar + 1] = a[kstar::-1]
                shifted[kstar + 1:] = 0.0
            else:
                lo = kstar - (n - 1)
                shifted[lo:] = a[n - 1:lo - 1:-1]
                shifted[:lo] = 0.0

            # gradient of 2·n·maxb (sum(a)=1) w.r.t. a_i
            grad = 4.0 * n * (shifted - maxb)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
            a = a - step

            # project back onto the probability simplex
            a = _simplex_projection(a)

            cheap = cheap_score(a)
            if cheap < best_local_val - 1e-12:
                best_local_val = cheap
                best_local_a = a.copy()
                stagnated = 0
                print(f"[Adam {t}] new local best cheap = {cheap:.6f}")
            else:
                stagnated += 1

            # occasional random restart if we stagnate for long
            if stagnated > 30_000:
                pert = best_local_a + np.random.randn(n) * 0.001
                a = _simplex_projection(pert)
                m.fill(0.0)
                v.fill(0.0)
                stagnated = 0
                print(f"[Adam {t}] restart after stagnation.")

        return best_local_a, best_local_val

    global_best_a = best_a
    global_best_val = best_val

    for idx, (val0, a0) in enumerate(top):
        if time.time() >= deadline:
            break
        print(f"[Start Adam {idx}] init cheap={val0:.6f}")
        a_opt, val_opt = _run_adam(a0, max_iters=200_000)
        if val_opt < global_best_val:
            global_best_val = val_opt
            global_best_a = a_opt.copy()
            print(f"[Global] new best cheap after Adam {idx} = {val_opt:.6f}")

    # --------------------------------------------------------------
    # 2. Permutation / swap local search (simulated‑annealing style)
    # --------------------------------------------------------------
    a = global_best_a.copy()
    best_swap_a = a.copy()
    best_swap_val = global_best_val
    temperature = 1.0
    cooling = 0.99995

    while time.time() < deadline:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        a[i], a[j] = a[j], a[i]

        maxb, _ = _conv_max_fft(a, fft_len)
        cheap = 2.0 * n * maxb

        accept = False
        if cheap < best_swap_val - 1e-12:
            accept = True
        else:
            delta = cheap - best_swap_val
            prob = math.exp(-delta / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            if cheap < best_swap_val - 1e-12:
                best_swap_val = cheap
                best_swap_a = a.copy()
                print(f"[Swap] NEW BEST cheap = {cheap:.6f}")
        else:
            # revert the swap
            a[i], a[j] = a[j], a[i]

        temperature = max(temperature * cooling, 1e-6)

    # --------------------------------------------------------------
    # 3. Final exact evaluation
    # --------------------------------------------------------------
    final_seq = best_swap_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:      # defensive – should never happen
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    return final_seq
