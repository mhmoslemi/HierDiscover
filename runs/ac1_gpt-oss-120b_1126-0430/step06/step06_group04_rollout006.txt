def _shape_grid_search(n, rng, time_limit):
   start = time.time()
   best = None
   best_obj = inf
   # define param ranges
   # uniform: evaluate
   shape = np.full(n, 1/n)
   obj = _objective(shape)
   if obj < best_obj: best_obj=obj; best=shape

   # left exponential: for beta in logspace
   betas = np.geomspace(0.80, 0.9999, 20)
   for beta in betas:
        a = beta**np.arange(n)
        a = a / a.sum()
        obj = _objective(a)
        if obj < best_obj: best_obj=obj; best=a
        # reversed
        a_rev = a[::-1]
        obj = _objective(a_rev)
        if obj < best_obj: best_obj=obj; best=a_rev

   # centered exponential
   m = (n-1)/2
   for beta in betas:
        a = beta**np.abs(np.arange(n)-m)
        a = a / a.sum()
        obj = _objective(a)
        if obj < best_obj: best_obj=obj; best=a

   # Gaussian sigma grid
   sigma_vals = np.linspace(0.1*n, 0.5*n, 12)
   for sigma in sigma_vals:
        a = np.exp(-0.5 * ((np.arange(n)-m)/sigma)**2)
        a = a / a.sum()
        obj = _objective(a)
        if obj < best_obj: best_obj=obj; best=a

   # Sine shape
   a = np.sin(np.pi * (np.arange(n)+0.5)/n)
   a = a / a.sum()
   obj = _objective(a)
   if obj < best_obj: best_obj=obj; best=a

   # power law
   p_vals = np.linspace(-2, 3, 12)
   for p in p_vals:
        # avoid zero.
        base = np.arange(1, n+1, dtype=float)
        a = base**p
        a = a / a.sum()
        obj = _objective(a)
        if obj < best_obj: best_obj=obj; best=a
        # reverse
        a_rev = a[::-1]
        obj = _objective(a_rev)
        if obj < best_obj: best_obj=obj; best=a_rev

   # random Dirichlet few attempts
   for _ in range(5):
        a = rng.random(n)
        a = a / a.sum()
        obj = _objective(a)
        if obj < best_obj: best_obj=obj; best=a

   return best, best_obj

def _run_adam(init_seq, time_limit, seed):
    start = time.time()
    rng = np.random.default_rng(seed)

    a0 = np.asarray(init_seq, dtype=float)
    a0 = a0 / a0.sum()
    n = a0.size
    x = np.log(a0 + 1e-16)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    best_a = a0.copy()
    best_obj = _objective(best_a)
    it = 0
    while time.time() - start < time_limit:
        it += 1
        a = _softmax(x)
        conv = _conv_self_fft(a)
        M = conv.max()
        obj = float(2.0 * n * M)
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a.copy()
            print(...)
        # schedule
        t = min((time.time() - start) / time_limit, 1.0)
        # smooth temperature
        beta = 0.5 * np.exp(np.log(1e7 / 0.5) * t)  # final ~1e7
        lr = 0.3 * np.exp(np.log(0.0005 / 0.3) * t)
        # compute w
        if beta > 1e6:
            idx = int(np.argmax(conv))
            w = np.zeros_like(conv)
            w[idx] = 1.0
        else:
            shifted = conv - M
            w = np.exp(beta * shifted)
            s = w.sum()
            if s == 0:
                w = np.ones_like(conv) / conv.size
            else:
                w /= s
        g_a = 2.0 * _correlation_fft(w, a)
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)
        m = 0.9 * m + 0.1 * g_x
        v = 0.999 * v + 0.001 * (g_x**2)
        m_hat = m / (1 - 0.9**it)
        v_hat = v / (1 - 0.999**it)
        x -= lr * m_hat / (np.sqrt(v_hat) + 1e-12)
        # jitter and restarts
        if it % 3000 == 0:
            x += rng.normal(scale=0.01, size=n)
        if it % 60000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.005, size=n)
            m.fill(0)
            v.fill(0)
    # final best
    return best_a.tolist(), best_obj

def _mass_transfer(a_seq, time_limit, seed):
    start = time.time()
    rng = np.random.default_rng(seed)
    a = np.asarray(a_seq, dtype=float).copy()
    n = a.size
    best_a = a.copy()
    best_obj = _objective(a)
    while time.time() - start < time_limit:
        conv = _conv_self_fft(a)
        k_max = int(np.argmax(conv))
        # compute contributions
        i_array = np.arange(n)
        j_array = k_max - i_array
        mask = (j_array >= 0) & (j_array < n)
        contrib = np.zeros(n)
        # Contrib for i where mask True:
        contrib[mask] = a[mask] * a[j_array[mask]]
        total_contrib = contrib.sum()
        # choose i
        if total_contrib > 0:
            probs = contrib / total_contrib
            i = rng.choice(n, p=probs)
        else:
            i = rng.integers(n)
        # choose j different
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        # delta fraction
        delta = a[i] * rng.random() * 0.05
        if delta <= 0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # compute new objective
        obj_new = _objective(a_new)
        if obj_new < best_obj - 1e-14:
            a, best_obj, best_a = a_new, obj_new, a_new.copy()
        else:
            # optionally accept if lower than current? we keep current a unchanged if not improvement
            pass
    return best_a.tolist(), best_obj

def _projected_gradient(a_seq, time_limit, seed):

        conv = _conv_self_fft(a)
        k_max = int(np.argmax(conv))
        w = np.zeros_like(conv)
        w[k_max] = 1.0
        g = 2.0 * _correlation_fft(w, a)
        eta = eta0 * (0.996 ** it)
        a_new = a - eta * g
        a_new = _project_simplex(a_new)
        obj_new = _objective(a_new)

            # reduce step size factor if no improvement maybe
            if obj_new > best_obj:
                eta0 *= 0.99
                if eta0 < 1e-6:
                    break

"""
Optimiser for the non‑negative step‑function problem.

The algorithm proceeds in three phases, repeated for one or more
support lengths:

1. **Analytic shape grid** – a modest set of parameterised families
   (uniform, exponential, centred‑exponential, Gaussian, sine,
   power‑law, sparse random) is sampled.  Each candidate is normalised
   to sum‑to‑1 and evaluated exactly; the best candidate becomes the
   start point for the next phase.

2. **Adam on the simplex** – the log‑it parametrisation of the
   probability simplex is optimised with Adam.  A temperature‑controlled
   smooth‑max approximates the exact ℓ∞‑norm of the autocorrelation;
   the temperature grows from ≈0.5 to ≈10⁷, turning the smooth
   approximation into the hard max.  Learning‑rate and momentum are
   annealed, occasional jitter and restarts keep the search from
   stagnating.

3. **Targeted refinement** – (a) a fast hill‑climber moves mass from
   indices that contribute most to the current maximal autocorrelation
   entry to random other positions; (b) a few projected‑gradient steps
   with the exact sub‑gradient of the ℓ∞‑norm finalise the solution.

All phases respect the global time budget (default 1000 s).  The
routine returns a plain list of non‑negative floats suitable for
``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)
    return conv[: 2 * n - 1]


def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation of a with weight vector w:
        g[i] = Σ_t w[i+t]·a[t]   for i=0,…,n‑1
    Implemented via FFT (real‑valued).
    """
    n = a.shape[0]
    m = w.shape[0]                     # = 2n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr = np.fft.irfft(wf * np.conj(af), n=L)
    return corr[:n]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(a: np.ndarray) -> float:
    """
    Exact objective (assuming Σ a = 1):
        O = 2·n·max(conv(a,a))
    """
    n = a.size
    conv = _conv_self_fft(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Shape families and grid search
# ----------------------------------------------------------------------


def _shape_grid_search(n: int, rng: np.random.Generator,
                       time_limit: float):
    """
    Scan a modest collection of analytic families (uniform, exponential,
    centred‑exponential, Gaussian, sine, power‑law, sparse Dirichlet).
    Return the best normalised sequence and its objective.
    """
    start = time.time()
    best_a = None
    best_obj = float("inf")

    # helper to evaluate and possibly update the incumbent
    def _maybe_update(a: np.ndarray):
        nonlocal best_a, best_obj
        obj = _objective(a)
        if obj < best_obj:
            best_obj, best_a = obj, a.copy()

    # 1. Uniform
    a = np.full(n, 1.0 / n)
    _maybe_update(a)

    # 2. Exponential (left‑to‑right) and its reversal
    betas = np.geomspace(0.90, 0.9999, num=30)
    idx = np.arange(n, dtype=float)
    for beta in betas:
        if time.time() - start > time_limit:
            break
        a = beta ** idx
        a /= a.sum()
        _maybe_update(a)
        a_rev = a[::-1]
        _maybe_update(a_rev)

    # 3. Centre‑exponential (symmetric, peak in the middle)
    centre = (n - 1) / 2.0
    for beta in betas:
        if time.time() - start > time_limit:
            break
        a = beta ** np.abs(idx - centre)
        a /= a.sum()
        _maybe_update(a)

    # 4. Gaussian windows
    sigma_vals = np.linspace(0.10 * n, 0.45 * n, 12)
    for sigma in sigma_vals:
        if time.time() - start > time_limit:
            break
        a = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
        a /= a.sum()
        _maybe_update(a)

    # 5. Sine window (positive half‑sine)
    a = np.sin(np.pi * (idx + 0.5) / n)
    a /= a.sum()
    _maybe_update(a)

    # 6. Power‑law families a_i ∝ (i+1)^p
    p_vals = np.linspace(-2.0, 3.0, 13)
    base = np.arange(1, n + 1, dtype=float)
    for p in p_vals:
        if time.time() - start > time_limit:
            break
        a = base ** p
        a /= a.sum()
        _maybe_update(a)
        a_rev = a[::-1]
        _maybe_update(a_rev)

    # 7. Sparse Dirichlet‑like (few non‑zero entries)
    k = max(2, int(0.02 * n))
    for _ in range(4):
        if time.time() - start > time_limit:
            break
        a = np.zeros(n, dtype=float)
        pos = rng.choice(n, size=k, replace=False)
        vals = rng.random(k)
        a[pos] = vals
        a /= a.sum()
        _maybe_update(a)

    # 8. Pure random Dirichlet (α=1)
    for _ in range(3):
        if time.time() - start > time_limit:
            break
        a = rng.random(n)
        a /= a.sum()
        _maybe_update(a)

    # Fallback in the unlikely case nothing was evaluated
    if best_a is None:
        a = np.full(n, 1.0 / n)
        best_a, best_obj = a, _objective(a)

    return best_a, best_obj


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑it parametrisation)
# ----------------------------------------------------------------------


def _run_adam(init_seq: np.ndarray,
              time_limit: float,
              seed: int = 0):
    """
    Adam optimiser on the simplex using a temperature‑controlled smooth‑max.
    Returns the best sequence (as a NumPy array) and its exact objective.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    n = init_seq.size
    a0 = init_seq.astype(float)
    a0 /= a0.sum()
    x = np.log(a0 + 1e-16)                     # logits
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # schedule endpoints
    beta_start, beta_end = 0.5, 1e7
    lr_start, lr_end = 0.3, 5e-4

    best_a = a0.copy()
    best_obj = _objective(best_a)

    it = 0
    while time.time() - start < time_limit:
        it += 1

        # stay on the simplex
        a = _softmax(x)

        conv = _conv_self_fft(a)
        M = conv.max()
        obj = float(2.0 * n * M)

        if obj < best_obj - 1e-14:
            best_obj, best_a = obj, a.copy()
            print(f"[n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        # progress fraction
        t = min((time.time() - start) / time_limit, 1.0)
        beta = beta_start * np.exp(np.log(beta_end / beta_start) * t)
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t)

        # smooth‑max weights
        if beta > 5e6:                     # essentially hard max
            idx = int(np.argmax(conv))
            w = np.zeros_like(conv)
            w[idx] = 1.0
        else:
            shifted = conv - M               # ≤ 0
            w = np.exp(beta * shifted)
            s = w.sum()
            if s == 0.0:
                w = np.ones_like(conv) / conv.size
            else:
                w /= s

        # gradient of (smooth) max w.r.t. a
        g_a = 2.0 * _correlation_fft(w, a)

        # natural gradient in log‑it space
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape plateaus
        if it % 2500 == 0:
            x += rng.normal(scale=0.012, size=n)

        # periodic restart from the current best
        if it % 60000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.006, size=n)
            m.fill(0.0)
            v.fill(0.0)

    return best_a, best_obj


# ----------------------------------------------------------------------
# Targeted mass‑transfer hill‑climber
# ----------------------------------------------------------------------


def _mass_transfer_hc(init_seq: np.ndarray,
                     time_limit: float,
                     seed: int = 0):
    """
    Fast hill‑climber that repeatedly moves a small fraction of mass away
    from indices that contribute most to the current maximal autocorrelation.
    Returns the best sequence found and its objective.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    a = init_seq.astype(float).copy()
    n = a.size
    best_a = a.copy()
    best_obj = _objective(a)

    while time.time() - start < time_limit:
        conv = _conv_self_fft(a)
        k_max = int(np.argmax(conv))

        # contribution of each i to conv[k_max]
        i_arr = np.arange(n)
        j_arr = k_max - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        contrib = np.zeros(n, dtype=float)
        contrib[mask] = a[i_arr[mask]] * a[j_arr[mask]]
        total = contrib.sum()

        # choose source index i (biased by contribution)
        if total > 0.0:
            probs = contrib / total
            i = rng.choice(n, p=probs)
        else:
            i = rng.integers(n)

        # choose a distinct destination j uniformly
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # move a modest fraction of mass
        delta = a[i] * rng.random() * 0.05   # ≤ 5 % of a[i]
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # (sum unchanged, non‑negativity guaranteed by construction)

        obj_new = _objective(a_new)
        if obj_new < best_obj - 1e-14:
            a, best_a, best_obj = a_new, a_new.copy(), obj_new
            # occasional progress report
            if rng.random() < 0.01:
                print(f"[HC] improved O={best_obj:.6f}")

    return best_a, best_obj


# ----------------------------------------------------------------------
# Projected‑gradient descent on the exact max‑gradient
# ----------------------------------------------------------------------


def _projected_gradient(init_seq: np.ndarray,
                        time_limit: float,
                        seed: int = 0):
    """
    A few projected‑gradient steps using the exact sub‑gradient
    (hard max) of the autocorrelation ℓ∞‑norm.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    a = init_seq.astype(float).copy()
    n = a.size
    best_a = a.copy()
    best_obj = _objective(a)

    it = 0
    eta0 = 0.25
    while time.time() - start < time_limit:
        it += 1
        conv = _conv_self_fft(a)
        k_max = int(np.argmax(conv))
        w = np.zeros_like(conv)
        w[k_max] = 1.0
        g = 2.0 * _correlation_fft(w, a)               # exact max‑gradient
        eta = eta0 * (0.995 ** it)
        a_new = a - eta * g
        a_new = _project_simplex(a_new)
        obj_new = _objective(a_new)
        if obj_new < best_obj - 1e-14:
            a, best_a, best_obj = a_new, a_new.copy(), obj_new
            print(f"[PG it={it}] improved O={best_obj:.6f}")
        else:
            # if no progress, slowly shrink the step‑size base
            eta0 *= 0.99
            if eta0 < 1e-6:
                break

    return best_a, best_obj

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **_ignore) -> list[float]:
    """
    Main optimiser.  Searches a handful of support lengths, starts from
    analytically motivated shapes, refines with Adam, then performs
    targeted hill‑climbing and projected‑gradient descent.
    Returns a plain list of non‑negative floats ready for ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    start_global = time.time()
    deadline = start_global + budget_s - 0.5   # safety margin

    # lengths to try – longer = more flexibility but also more work
    lengths = [2000, 4000, 6000, 8000]

    best_seq = None
    best_obj = float("inf")

    for idx, n in enumerate(lengths):
        now = time.time()
        remaining = deadline - now
        if remaining < 8.0:          # need a little room for final polishing
            break

        # allocate a chunk of the remaining budget to this length
        # the first (largest) length gets a bit more
        if idx == 0:
            chunk = remaining * 0.55
        else:
            chunk = remaining * 0.35 / (len(lengths) - idx)

        # ------------------------------------------------------------------
        # 1. Shape grid search
        shape_time = min(6.0, chunk * 0.07)
        init_a, init_obj = _shape_grid_search(n, rng, shape_time)
        print(f"[init n={n}] shape‑obj={init_obj:.6f}")

        # ------------------------------------------------------------------
        # 2. Adam optimisation
        adam_time = max(5.0, chunk * 0.55)
        seq_adam, obj_adam = _run_adam(init_a, adam_time,
                                       seed + idx * 12345)
        print(f"[Adam n={n}] obj={obj_adam:.6f}")

        # ------------------------------------------------------------------
        # 3. Targeted mass‑transfer hill‑climber
        hc_time = max(4.0, chunk * 0.20)
        seq_hc, obj_hc = _mass_transfer_hc(seq_adam, hc_time,
                                          seed + idx * 54321)
        print(f"[HC n={n}] obj={obj_hc:.6f}")

        # ------------------------------------------------------------------
        # 4. Projected‑gradient refinement (exact max‑gradient)
        pg_time = max(2.0, chunk - (shape_time + adam_time + hc_time))
        seq_pg, obj_pg = _projected_gradient(seq_hc, pg_time,
                                            seed + idx * 98765)
        print(f"[PG n={n}] obj={obj_pg:.6f}")

        if obj_pg < best_obj:
            best_obj, best_seq = obj_pg, np.asarray(seq_pg, dtype=float)
            print(f"[global] new overall best O={best_obj:.6f} (n={n})")

        if time.time() >= deadline - 0.5:
            break

    # ----------------------------------------------------------------------
    # Final sanity check (diagnostic only)
    # ----------------------------------------------------------------------
    if best_seq is None:
        # As a last resort return a uniform sequence of moderate size
        n = 2000
        best_seq = np.full(n, 1.0 / n)

    # Clamp to evaluator's allowed range and guard against sum‑too‑small
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    if best_seq.sum() < 0.01:
        best_seq = np.ones_like(best_seq) / best_seq.size

    # Optional: report the objective as seen by the supplied evaluator
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:       # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_seq.tolist()
