def _gradient_smooth_max(a, conv, tau, fft_len):
    maxc = conv.max()
    # shift for numeric stability
    w = np.exp(tau * (conv - maxc))
    w_sum = w.sum()
    if w_sum == 0 or not np.isfinite(w_sum):
        # fallback to uniform weights?
        w = np.ones_like(conv) / conv.size
    else:
        w /= w_sum

   cheap score (2·n·max(conv) with Σa=1) is kept as a warm‑start.

   max‑convolution using the log‑sum‑exp (temperature τ).  The true gradient
   of this smooth objective is obtained analytically:
       g_i = 2·(w * a_rev)_i – w_{2i}·a_i,
   where w_k ∝ exp(τ·conv_k).  An exponentiated‑gradient (mirror‑descent
   on the simplex) step is performed and the vector is renormalised.
   The temperature τ is increased slowly (τ←τ·2) to sharpen the
   approximation to the true max.  Restarts with small random noise are
   issued if progress stalls.

3. **Permutation / value‑transfer local search** – starting from the best
   iterate, random swaps of two positions and tiny mass transfers are
   accepted by a simulated‑annealing schedule to escape remaining local
   minima.

4. **Final exact evaluation** – the best vector found is passed to the
   provided ``evaluate_sequence`` routine and returned.

The routine respects the user‑supplied time budget (default 1000 s) and
always returns a feasible sequence (non‑negative, finite numbers)."""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using a power‑of‑two FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len).real[: 2 * a.size - 1]
    return conv


def _conv_max_fft(a: np.ndarray, fft_len: int) -> float:
    """Maximum entry of the autocorrelation a*a."""
    conv = _conv_full_fft(a, fft_len)
    return float(conv.max())


def _cheap_score(vec: np.ndarray, n: int, fft_len: int) -> float:
    """Assumes Σvec = 1. Returns 2·n·max(conv)."""
    maxb = _conv_max_fft(vec, fft_len)
    return 2.0 * n * maxb


def _smooth_gradient(a: np.ndarray, conv: np.ndarray, tau: float,
                     fft_len: int) -> np.ndarray:
    """
    Compute the gradient of the smoothed max (log‑sum‑exp) objective.

    Parameters
    ----------
    a : ndarray
        Current probability vector (sum = 1).
    conv : ndarray
        Full convolution a*a (length 2n‑1).
    tau : float
        Temperature for the log‑sum‑exp smoothing (larger → sharper).
    fft_len : int
        FFT length (power of two ≥ 2·n‑1).

    Returns
    -------
    grad : ndarray
        Gradient w.r.t. a (shape n).
    """
    n = a.shape[0]

    # stable softmax over convolution entries
    maxc = conv.max()
    w = np.exp(tau * (conv - maxc))
    w_sum = w.sum()
    if w_sum == 0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # convolution w * a_rev
    a_rev = a[::-1]
    fw = np.fft.rfft(w, n=fft_len)
    fa_rev = np.fft.rfft(a_rev, n=fft_len)
    g_full = np.fft.irfft(fw * fa_rev, n=fft_len).real[: 2 * n - 1]

    idx2i = 2 * np.arange(n)                     # indices of w_{2i}
    grad = 2.0 * g_full[:n] - w[idx2i] * a
    return grad


# ----------------------------------------------------------------------
# Core search routine required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that
    minimises ``evaluate_sequence``.  The routine respects the time budget
    and always returns a feasible sequence.
    """
    np.random.seed(seed)
    start_time = time.time()
    # give a few seconds for the final exact evaluation
    deadline = start_time + budget_s - 5.0
    eps = 1e-30

    # --------------------------------------------------------------
    # Choose a working length.
    # --------------------------------------------------------------
    try:
        prev = height_sequence_1               # may be defined globally
        n = max(len(prev), 5000)
        prev_arr = np.asarray(prev, dtype=np.float64)
        if prev_arr.sum() > 0.0:
            prev_arr = prev_arr / prev_arr.sum()
        else:
            prev_arr = None
    except Exception:
        prev_arr = None
        n = 5000

    # Keep n reasonable for the given budget
    n = min(max(n, 5000), 20000)

    # FFT length – next power of two ≥ 2·n‑1
    fft_len = 1 << ((2 * n - 1).bit_length())

    # --------------------------------------------------------------
    # Initial parametric sweep (cheap_score = 2·n·max(conv))
    # --------------------------------------------------------------
    best_a = None
    best_val = float("inf")

    # uniform baseline
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    val = _cheap_score(uniform, n, fft_len)
    best_a, best_val = uniform, val

    # previous construction, if any and same length
    if prev_arr is not None and prev_arr.size == n:
        val = _cheap_score(prev_arr, n, fft_len)
        if val < best_val:
            best_a, best_val = prev_arr.copy(), val

    # ----- exponential (left) ------------------------------------
    for lam in np.linspace(0.0, 12.0, 49):
        if time.time() >= deadline:
            break
        a = np.exp(-lam * np.arange(n, dtype=np.float64))
        a /= a.sum()
        val = _cheap_score(a, n, fft_len)
        if val < best_val:
            best_a, best_val = a.copy(), val

    # ----- symmetric exponential (U‑shaped) -----------------------
    for lam in np.linspace(0.0, 12.0, 49):
        if time.time() >= deadline:
            break
        idx = np.arange(n, dtype=np.float64)
        a = np.exp(-lam * np.minimum(idx, n - 1 - idx))
        a /= a.sum()
        val = _cheap_score(a, n, fft_len)
        if val < best_val:
            best_a, best_val = a.copy(), val

    # ----- power‑law (left) --------------------------------------
    for p in np.linspace(0.01, 6.0, 61):
        if time.time() >= deadline:
            break
        a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
        a /= a.sum()
        val = _cheap_score(a, n, fft_len)
        if val < best_val:
            best_a, best_val = a.copy(), val

    # ----- symmetric power‑law (U‑shaped) ------------------------
    for p in np.linspace(0.01, 6.0, 61):
        if time.time() >= deadline:
            break
        i = np.arange(n, dtype=np.float64)
        a = np.minimum(i + 1, n - i) ** (-p)
        a /= a.sum()
        val = _cheap_score(a, n, fft_len)
        if val < best_val:
            best_a, best_val = a.copy(), val

    # ----- beta families (U‑shaped when α,β<1) --------------------
    x = (np.arange(1, n + 1, dtype=np.float64) / n)
    alphas = np.concatenate((
        np.linspace(0.01, 0.2, 5),
        np.linspace(0.2, 1.0, 9),
        np.linspace(1.0, 5.0, 5)
    ))
    betas = alphas.copy()
    for alpha in alphas:
        for beta in betas:
            if time.time() >= deadline:
                break
            a = (x ** alpha) * ((1.0 - x) ** beta)
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _cheap_score(a, n, fft_len)
            if val < best_val:
                best_a, best_val = a.copy(), val

    # ----- double exponential (left & right) --------------------
    lam_grid = np.linspace(0.0, 12.0, 13)  # 0,1,…,12
    for lamL in lam_grid:
        for lamR in lam_grid:
            if time.time() >= deadline:
                break
            a = np.exp(-lamL * np.arange(n, dtype=np.float64)) + \
                np.exp(-lamR * np.arange(n - 1, -1, -1, dtype=np.float64))
            a /= a.sum()
            val = _cheap_score(a, n, fft_len)
            if val < best_val:
                best_a, best_val = a.copy(), val

    # ----- cosine‑power (U‑shaped) -------------------------------
    i = np.arange(n, dtype=np.float64)
    cos_sq = np.cos(np.pi * i / (n - 1)) ** 2  # symmetric U‑shape
    for p in np.linspace(0.1, 12.0, 49):
        if time.time() >= deadline:
            break
        shape = cos_sq ** p
        a = shape / shape.sum()
        val = _cheap_score(a, n, fft_len)
        if val < best_val:
            best_a, best_val = a.copy(), val

    # ----- mixtures with uniform -------------------------------
    mixes = np.linspace(0.2, 0.8, 13)  # α in mixture α·shape + (1‑α)·uniform
    for p in np.linspace(0.1, 12.0, 31):
        if time.time() >= deadline:
            break
        shape = cos_sq ** p
        shape /= shape.sum()
        for alpha in mixes:
            a = alpha * shape + (1.0 - alpha) * uniform
            a /= a.sum()
            val = _cheap_score(a, n, fft_len)
            if val < best_val:
                best_a, best_val = a.copy(), val

    # ----- random Dirichlet seeds --------------------------------
    for conc in [0.001, 0.01, 0.1, 0.5, 1.0, 2.0, 5.0]:
        if time.time() >= deadline:
            break
        g = np.random.gamma(conc, 1.0, size=n)
        a = g / g.sum()
        val = _cheap_score(a, n, fft_len)
        if val < best_val:
            best_a, best_val = a.copy(), val

    print(f"[Init] best cheap = {best_val:.6f}")

    # --------------------------------------------------------------
    # 1. Smoothed gradient descent (exponentiated‑gradient on simplex)
    # --------------------------------------------------------------
    a = best_a.copy()
    best_iter_a = a.copy()
    best_iter_val = best_val
    last_improve = 0
    iter_cnt = 0
    eta0 = 0.25               # base step size for exponentiated‑gradient
    tau = 1.0                 # initial temperature
    tau_max = 80.0
    max_iters = 2_000_000

    while time.time() < deadline and iter_cnt < max_iters:
        iter_cnt += 1

        # full convolution and cheap score (sum = 1)
        conv = _conv_full_fft(a, fft_len)
        maxb = conv.max()
        cheap = 2.0 * n * maxb

        if cheap < best_iter_val - 1e-12:
            best_iter_val = cheap
            best_iter_a = a.copy()
            last_improve = iter_cnt
            print(f"[Iter {iter_cnt:07d}] NEW BEST cheap = {cheap:.6f}")

        # compute smooth gradient
        grad = _smooth_gradient(a, conv, tau, fft_len)

        # exponentiated‑gradient (mirror descent on simplex)
        step = eta0 / math.sqrt(iter_cnt)
        log_a = np.log(a + eps) - step * grad
        a = np.exp(log_a)
        sum_a = a.sum()
        if sum_a <= 0.0 or not np.isfinite(sum_a):
            # fallback to the best iterate seen so far
            a = best_iter_a.copy()
            continue
        a /= sum_a

        # temperature schedule – increase every 20000 iterations
        if iter_cnt % 20000 == 0:
            tau = min(tau * 2.0, tau_max)

        # restart if stagnated for a long time
        if iter_cnt - last_improve > 120_000:
            perturb = np.random.rand(n) * 0.01
            a = _simplex_projection(best_iter_a + perturb)
            last_improve = iter_cnt
            print(f"[Iter {iter_cnt:07d}] Restart after stagnation (τ={tau:.2f}).")

    # --------------------------------------------------------------
    # 2. Permutation / value‑transfer local search (simulated‑annealing)
    # --------------------------------------------------------------
    a = best_iter_a.copy()
    best_swap_a = a.copy()
    best_swap_val = best_iter_val
    temperature = 1.0
    cooling = 0.99995

    while time.time() < deadline:
        # ---- random swap of positions ----
        i, j = np.random.randint(0, n, size=2)
        if i != j:
            a[i], a[j] = a[j], a[i]

        conv = _conv_full_fft(a, fft_len)
        cheap = 2.0 * n * conv.max()
        accept = False
        if cheap < best_swap_val - 1e-12:
            accept = True
        else:
            delta = cheap - best_swap_val
            prob = math.exp(-delta / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            if cheap < best_swap_val - 1e-12:
                best_swap_val = cheap
                best_swap_a = a.copy()
                print(f"[Swap] NEW BEST cheap = {cheap:.6f}")
        else:
            # revert swap
            a[i], a[j] = a[j], a[i]

        # ---- tiny mass transfer between two indices (value perturbation) ----
        i, j = np.random.randint(0, n, size=2)
        if i != j and a[i] > 1e-12:
            alpha = np.random.rand() * 0.02   # up to 2 % of a[i]
            delta = alpha * a[i]
            a[i] -= delta
            a[j] += delta
            # keep on simplex automatically (sum unchanged)

            conv = _conv_full_fft(a, fft_len)
            cheap = 2.0 * n * conv.max()
            accept = False
            if cheap < best_swap_val - 1e-12:
                accept = True
            else:
                delta = cheap - best_swap_val
                prob = math.exp(-delta / max(temperature, 1e-12))
                if np.random.rand() < prob:
                    accept = True

            if accept:
                if cheap < best_swap_val - 1e-12:
                    best_swap_val = cheap
                    best_swap_a = a.copy()
                    print(f"[Transfer] NEW BEST cheap = {cheap:.6f}")
            else:
                # revert transfer
                a[i] += delta
                a[j] -= delta

        temperature = max(temperature * cooling, 1e-6)

    # --------------------------------------------------------------
    # 3. Final exact evaluation
    # --------------------------------------------------------------
    final_seq = best_swap_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:  # defensive – should never happen
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    return final_seq
