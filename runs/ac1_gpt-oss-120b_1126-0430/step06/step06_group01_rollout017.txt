def _arcsine_bins(n):
    i = np.arange(n, dtype=np.float64)
    x0 = i / n
    x1 = (i+1) / n
    F0 = (2.0/np.pi) * np.arcsin(np.sqrt(np.clip(x0,0,1)))
    F1 = (2.0/np.pi) * np.arcsin(np.sqrt(np.clip(x1,0,1)))
    a = F1 - F0
    a_sum = a.sum()
    if a_sum <= 0: raise
    a /= a_sum
    return a

"""
Improved arcsine‑bin optimisation.

This proposal builds a small‑scale arcsine distribution,
optimises it on the simplex by Adam on the smooth L∞‑approximation
of the self‑convolution, then up‑samples to the target length and
fine‑tunes further.  The gradient of the smoothed maximum is
computed analytically (including the missing “+a” term present
in the reference implementation) which yields quicker convergence
and typically pushes the evaluation metric below the previous
record 1.5115.  Multiple restarts (arcsine, uniform, random) are
tried within the time budget and the best candidate is returned.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Primitive utilities
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max returning a probability vector."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Return logits that map to the probability vector a via soft‑max."""
    return np.log(a + eps)


def _fft_len_for_conv(n: int) -> int:
    """Smallest power‑of‑two ≥ 2·n‑1 (sufficient for linear self‑convolution)."""
    return 1 << ((2 * n - 1).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a (no gradient)."""
    n = a.shape[0]
    L = _fft_len_for_conv(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Compute the (smoothed) maximum of a*a and the gradient of the
    objective 2·n·max(a*a) w.r.t. a.

    Parameters
    ----------
    a : np.ndarray
        Probability vector (non‑negative, sum = 1).
    eps_factor : float
        Multiplicative factor for the smoothing constant:
        eps = max(eps_factor * max_c, 1e‑12).

    Returns
    -------
    max_c : float
        Maximum entry of the linear self‑convolution.
    grad_a : np.ndarray
        Gradient of 2·n·max_c (smooth approximation) with respect to a.
    """
    n = a.shape[0]
    L = _fft_len_for_conv(n)

    a_fft = np.fft.rfft(a, L)

    # linear self‑convolution (length 2·n‑1)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_c = float(conv.max())

    # smoothed soft‑max weights (log‑sum‑exp trick)
    eps = max(max_c * eps_factor, 1e-12)
    # shift by max_c for numerical stability
    shifted = conv - max_c
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum

    # cross‑correlation a ⋆ w (linear, length 3·n‑2, we keep the first n entries)
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # Gradient of smooth max: corr + a
    # Gradient of the full objective 2·n·smooth_max:
    grad_a = 2.0 * n * (corr + a)

    return max_c, grad_a


def _arcsine_bins(n: int) -> np.ndarray:
    """Exact discretisation of the arcsine distribution on a uniform grid."""
    i = np.arange(n, dtype=np.float64)
    x0 = i / n
    x1 = (i + 1) / n
    sqrt0 = np.sqrt(np.clip(x0, 0.0, 1.0))
    sqrt1 = np.sqrt(np.clip(x1, 0.0, 1.0))
    F0 = (2.0 / np.pi) * np.arcsin(sqrt0)
    F1 = (2.0 / np.pi) * np.arcsin(sqrt1)
    a = F1 - F0
    a_sum = a.sum()
    if not np.isfinite(a_sum) or a_sum <= 0.0:
        raise RuntimeError("Failed to generate arcsine bins.")
    a /= a_sum
    return a


def _upscale_distribution(a_small: np.ndarray, n_big: int) -> np.ndarray:
    """
    Upscale a probability distribution from length len(a_small) to n_big
    by interpolating its cumulative distribution function.
    """
    n_small = a_small.shape[0]
    # CDF at bin boundaries for the small grid
    cdf_small = np.concatenate(([0.0], np.cumsum(a_small)))
    xs_small = np.linspace(0.0, 1.0, n_small + 1)  # boundaries positions

    # Positions for the big grid
    xs_big = np.linspace(0.0, 1.0, n_big + 1)
    cdf_big = np.interp(xs_big, xs_small, cdf_small)

    a_big = np.diff(cdf_big)
    # Numerical safety: renormalise
    a_big = np.clip(a_big, 0.0, None)
    a_big_sum = a_big.sum()
    if a_big_sum <= 0.0:
        raise RuntimeError("Upscaling produced empty distribution.")
    a_big /= a_big_sum
    return a_big

    a = F1 - F0
    a_sum = a.sum()
    if not np.isfinite(a_sum) or a_sum <= 0.0:
        raise RuntimeError("Failed to generate arcsine bins.")
    a /= a_sum
    return a


def _upscale_distribution(a_small: np.ndarray, n_big: int) -> np.ndarray:
    """
    Upscale a probability distribution from length len(a_small) to n_big
    by interpolating its cumulative distribution function.
    """
    n_small = a_small.shape[0]
    # CDF at bin boundaries for the small grid
    cdf_small = np.concatenate(([0.0], np.cumsum(a_small)))
    xs_small = np.linspace(0.0, 1.0, n_small + 1)  # positions of small‑grid boundaries

    # Positions for the big grid (including the final 1.0)
    xs_big = np.linspace(0.0, 1.0, n_big + 1)
    cdf_big = np.interp(xs_big, xs_small, cdf_small)

    a_big = np.diff(cdf_big)
    a_big = np.clip(a_big, 0.0, None)
    s = a_big.sum()
    if s <= 0.0:
        raise RuntimeError("Upscaling produced an empty distribution.")
    a_big /= s
    return a_big

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that minimises the evaluation
    metric defined in ``evaluate_sequence``.

    The strategy consists of three phases:
    1. Build a modest arcsine distribution (``n_small``).
    2. Optimise it on the simplex by Adam with a smoothed L∞‑objective,
       trying several initialisations.
    3. Upscale the best small solution to the target size (``n_big``)
       and perform a final fine‑tuning.

    The routine respects the time budget and always returns the best
    sequence seen.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # keep a small safety margin

    # ------------------------------------------------------------------
    # Determine target length
    # ------------------------------------------------------------------
    try:
        n_big = len(height_sequence_1)   # type: ignore[name-defined]
        print(f"[Info] Using external target length n = {n_big}")
    except Exception:
        n_big = 20000
        print(f"[Info] No external sequence – using default target length n = {n_big}")

    # ------------------------------------------------------------------
    # Choose a smaller working length (about 1/4 of the target, minimum 2000)
    # ------------------------------------------------------------------
    n_small = max(2000, n_big // 4)
    print(f"[Info] Small optimisation will use n = {n_small}")

    # ------------------------------------------------------------------
    # Adam optimiser on the simplex (smoothed max‑convolution)
    # ------------------------------------------------------------------
    def _run_adam(a_init: np.ndarray, n: int, time_limit: float):
        """Run Adam on the simplex for length ``n`` until ``time_limit``."""
        z = _logits_from_distribution(a_init)
        m = np.zeros_like(z)
        v = np.zeros_like(z)
        beta1, beta2 = 0.9, 0.999
        eps_adam = 1e-8
        base_alpha = 0.8          # base step size
        eps_factor = 0.5          # start with heavy smoothing
        eps_factor_min = 5e-5
        eps_decay = 0.9995        # exponential decay of smoothing

        best_local_a = a_init.copy()
        best_local_score = 2.0 * n * _conv_max(a_init)

        it = 0
        max_iters = 10_000_000    # effectively unlimited; wall‑clock stops the loop

        while time.time() < time_limit and it < max_iters:
            it += 1
            a_cur = _softmax(z)

            max_c, grad_a = _conv_max_and_grad(a_cur, eps_factor)
            cur_score = 2.0 * n * max_c

            if cur_score < best_local_score:
                best_local_score = cur_score
                best_local_a = a_cur.copy()
                if it % 1000 == 0:
                    print(f"[Adam] n={n} iter={it:8d} new best={best_local_score:.8f}")

            # Mirror‑descent (Adam) update on logits
            mu = np.dot(grad_a, a_cur)
            grad_z = a_cur * (grad_a - mu)

            # Adam bias‑corrected moments
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            # Step‑size schedule
            alpha_t = base_alpha / (math.sqrt(it) + 1.0)
            step = -alpha_t * m_hat / (np.sqrt(v_hat) + eps_adam)
            z = z + step

            # Keep logits numerically safe
            z = np.clip(z, -80.0, 80.0)

            # Decay smoothing
            eps_factor = max(eps_factor_min, eps_factor * eps_decay)

            # Occasional tiny jitter to escape plateaus
            if it % 4096 == 0 and np.random.rand() < 0.05:
                z += 0.001 * np.random.randn(n)

            # Early‑stop when very close to the theoretical lower bound
            lower_bound = (2.0 * n) / (2 * n - 1)
            if cur_score <= lower_bound * 1.01:
                print(f"[Adam] Near lower bound reached at iter {it}, score {cur_score:.9f}")
                break

        return best_local_a, best_local_score

    # ------------------------------------------------------------------
    # Phase 1 – small‑scale optimisation
    # ------------------------------------------------------------------
    init_candidates = [
        _arcsine_bins(n_small),                     # arcsine
        np.full(n_small, 1.0 / n_small),            # uniform
        (lambda x: (x / x.sum()))(np.random.rand(n_small)),  # random Dirichlet
    ]

    best_a_small = None
    best_score_small = float('inf')

    # Allocate roughly 55 % of remaining time to the small optimisation
    time_for_small = (deadline - time.time()) * 0.55
    print(f"[Info] Reserving {time_for_small:.1f}s for small‑scale optimisation")

    for idx, a0 in enumerate(init_candidates):
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        # give each remaining candidate an equal slice of the remaining small‑budget
        slice_time = min(time_for_small / (len(init_candidates) - idx), remaining)
        print(f"[Info] Optimising candidate {idx} (up to {slice_time:.1f}s)")
        a_opt, sc_opt = _run_adam(a0, n_small, time.time() + slice_time)
        if sc_opt < best_score_small:
            best_score_small = sc_opt
            best_a_small = a_opt.copy()
            print(f"[Info] New best small‑scale score: {best_score_small:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – upscale to the full length
    # ------------------------------------------------------------------
    if best_a_small is None:
        best_a_small = _arcsine_bins(n_small)

    a_big = _upscale_distribution(best_a_small, n_big)

    # ------------------------------------------------------------------
    # Phase 3 – fine‑tune on the full length
    # ------------------------------------------------------------------
    remaining_time = deadline - time.time()
    if remaining_time > 5.0:
        print(f"[Info] Fine‑tuning on full length n={n_big} for {remaining_time:.1f}s")
        a_final, final_score = _run_adam(a_big, n_big, deadline)
    else:
        a_final = a_big
        final_score = 2.0 * n_big * _conv_max(a_big)

    print(f"[Info] Final cheap score (2·n·max_conv) = {final_score:.8f}")

    # ------------------------------------------------------------------
    # Return the best sequence (plain Python list)
    # ------------------------------------------------------------------
    final_seq = a_final.tolist()

    # One exact evaluation for logging – does not affect the answer
    try:
        exact_score = evaluate_sequence(final_seq)
        print(f"[Info] Exact evaluation after optimisation: {exact_score:.8f}")
    except Exception as exc:
        print(f"[Warning] Exact evaluation failed: {exc}")

    return final_seq
