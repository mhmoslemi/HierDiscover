"""
Step‑function optimiser (v3).

The optimiser builds many analytic “shapes”, refines them with a fast
projected‑gradient descent that uses the exact max‑convolution gradient
(O(n) per iteration), runs many short restarts, and finishes with a
targeted simulated‑annealing + final projected‑gradient pass.  The
search is time‑budget aware (default 1000 s) and works on a single
support length (≈2000) – the size that gave the best result in the
previous runs.  The code is deliberately lightweight (no heavy
optimisers) so that thousands of gradient steps fit comfortably inside
the allotted time, giving a better chance to beat the 1.503 target.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full a*a convolution using a real FFT.
    Length of the result is 2·len(a)−1.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return conv_full[: 2 * n - 1]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0, Σ x = 1}.
    O(n log n) algorithm (Duchi et al.).
    """
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) assuming Σ a = 1."""
    n = a.size
    conv = _conv_self_fft(a)
    return float(2.0 * n * conv.max())


def _grad_maxconv(a: np.ndarray, k: int) -> np.ndarray:
    """
    Gradient of conv[k] w.r.t. a (exact, no FFT).
    For a given index k (0 ≤ k < 2n-1) the gradient entry i is:
        2 * a[k-i]  if 0 ≤ k-i < n  else 0.
    """
    n = a.shape[0]
    # vectorised implementation
    idx = np.arange(n)
    src = k - idx
    mask = (src >= 0) & (src < n)
    grad = np.zeros_like(a)
    if np.any(mask):
        grad[mask] = 2.0 * a[src[mask]]
    return grad

# Shape candidate generation (stage 1)
# ----------------------------------------------------------------------


def _shape_candidates(n: int, rng: np.random.Generator):
    """
    Produce a list of analytically‑motivated candidates,
    each normalised to sum‑to‑1.
    """
    cand = []
    idx = np.arange(n, dtype=float)

    # Uniform
    cand.append(np.full(n, 1.0 / n))

    # Exponential decays (forward & reversed)
    for r in np.linspace(0.90, 0.9995, 12):
        a = r ** idx
        a = a / a.sum()
        cand.append(a)
        cand.append(a[::-1])

    # Power‑law (i+1)^{-p}
    for p in np.linspace(0.3, 2.5, 12):
        a = 1.0 / ((idx + 1) ** p)
        a = a / a.sum()
        cand.append(a)

    # Cosine‑type windows (non‑negative)
    # Hann (raised‑cosine)
    a = 0.5 * (1 - np.cos(2 * np.pi * idx / (n - 1)))
    a = a / a.sum()
    cand.append(a)

    # Hamming
    a = 0.54 - 0.46 * np.cos(2 * np.pi * idx / (n - 1))
    a = a / a.sum()
    cand.append(a)

    # Gaussian centred
    centre = (n - 1) / 2.0
    for sigma in np.linspace(0.10 * n, 0.35 * n, 6):
        a = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
        a = a / a.sum()
        cand.append(a)

    # Triangular (peak in the middle)
    a = 1.0 - np.abs(idx - centre) / centre
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    cand.append(a)

    # Sparse Dirichlet‑like (few non‑zero entries)
    k = max(2, int(0.025 * n))
    for _ in range(4):
        a = np.zeros(n, dtype=float)
        pos = rng.choice(n, size=k, replace=False)
        vals = rng.random(k)
        a[pos] = vals
        a = a / a.sum()
        cand.append(a)

    # Pure Dirichlet (α=1)
    a = rng.random(n)
    a = a / a.sum()
    cand.append(a)

    return cand


def _best_initial_sequence(n: int, rng: np.random.Generator, time_limit: float):
    """
    Scan a moderate set of shapes and return the one with the smallest
    exact objective.  The scan is aborted if `time_limit` seconds are
    exceeded.
    """
    start = time.time()
    best_a, best_val = None, float("inf")
    for a in _shape_candidates(n, rng):
        if time.time() - start > time_limit:
            break
        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a
    # fallback: random Dirichlet
    if best_a is None:
        a = rng.random(n)
        best_a = a / a.sum()
        best_val = _objective(best_a)
    return best_a, best_val

# Projected‑gradient descent on the exact max (stage 2)
# ----------------------------------------------------------------------


def _pgd_optimize(
    a0: np.ndarray,
    time_limit: float,
    seed: int = 0,
    eta0: float = 0.30,
):
    """
    Fast projected gradient descent using the exact max‑convolution
    gradient.  Returns the best (a, objective) pair found within the
    allocated time.
    """
    rng = np.random.default_rng(seed)
    n = a0.size
    a = a0.copy()
    best_a = a.copy()
    best_obj = _objective(a)

    it = 0
    start = time.time()
    while time.time() - start < time_limit:
        it += 1
        # 1) compute convolution and max index
        conv = _conv_self_fft(a)
        k_max = int(np.argmax(conv))
        obj = float(2.0 * n * conv[k_max])
        if obj < best_obj - 1e-14:
            best_obj, best_a = obj, a.copy()
            # occasional progress report
            if it % 1000 == 0:
                print(f"[PGD it={it:06d}] new best O={best_obj:.6f}")

        # 2) gradient of the exact max
        g = _grad_maxconv(a, k_max)          # = ∂conv[k_max]/∂a
        g = 2.0 * g                           # because objective has factor 2·n later (we handle later)

        # 3) step size (simple 1/√t schedule)
        eta = eta0 / math.sqrt(it)

        # 4) gradient step + simplex projection
        a = _project_simplex(a - eta * g)

        # 5) occasional jitter to escape plateaus
        if it % 2500 == 0:
            jitter = rng.normal(scale=0.008, size=n)
            a = _project_simplex(a + jitter)

    return best_a.tolist(), best_obj

# Targeted simulated‑annealing + final PG (stage 3)
# ----------------------------------------------------------------------


def _sa_refine(
    init_seq: list[float],
    time_limit: float,
    seed: int = 0,
):
    """
    A lightweight simulated‑annealing routine that moves a modest
    fraction of mass away from indices that contribute most to the
    current maximal autocorrelation entry, followed by a short exact
    projected‑gradient finish.
    """
    rng = np.random.default_rng(seed)
    a = np.asarray(init_seq, dtype=float)
    if a.sum() < 1e-12:
        a = np.ones_like(a) / a.size
    else:
        a = a / a.sum()
    n = a.size

    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())
    best_a = a.copy()

    # SA schedule (geometric cooling)
    T = best_obj * 0.25 + 1e-8
    T_min = best_obj * 1e-6 + 1e-12
    decay = 0.995

    start = time.time()
    it = 0
    while time.time() - start < time_limit * 0.65:
        it += 1
        k_max = int(np.argmax(conv))

        # contributions to the current peak
        i_vals = np.arange(n)
        j_vals = k_max - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i_vals[mask]] * a[j_vals[mask]]
        total = contrib.sum()
        if total > 0.0:
            probs = contrib / total
            i = rng.choice(n, p=probs)
        else:
            i = rng.integers(n)

        # random destination different from i
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # move a small fraction of mass from i to j
        delta = a[i] * rng.random() * 0.07   # up to 7 % of a[i]
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.clip(a_new, 0.0, None)
        a_new /= a_new.sum()

        conv_new = _conv_self_fft(a_new)
        obj_new = float(2.0 * n * conv_new.max())

        if obj_new < best_obj - 1e-14:
            a, conv, best_obj, best_a = a_new, conv_new, obj_new, a_new.copy()
            print(f"[SA it={it}] new best O={best_obj:.6f}")
        else:
            # Metropolis acceptance
            if rng.random() < math.exp(-(obj_new - best_obj) / T):
                a, conv = a_new, conv_new

        if it % 2000 == 0:
            T = max(T * decay, T_min)

        if it % 8000 == 0:
            # global jitter
            noise = rng.normal(scale=0.005, size=n)
            a = _project_simplex(a + noise)
            conv = _conv_self_fft(a)

    # ---- final exact PG (few hundred iterations) ----
    remaining = time_limit - (time.time() - start)
    if remaining > 0.3:
        a = best_a.copy()
        n = a.size
        best_obj = float(2.0 * n * _conv_self_fft(a).max())
        it_g = 0
        eta0 = 0.25
        while time.time() - start < time_limit:
            it_g += 1
            conv = _conv_self_fft(a)
            k_max = int(np.argmax(conv))
            g = _grad_maxconv(a, k_max)
            g = 2.0 * g
            eta = eta0 * (0.995 ** it_g)
            a_new = _project_simplex(a - eta * g)
            obj_new = float(2.0 * n * _conv_self_fft(a_new).max())
            if obj_new < best_obj - 1e-14:
                a, best_obj = a_new, obj_new
                best_a = a_new.copy()
                print(f"[PG2 it={it_g}] improved O={best_obj:.6f}")
            else:
                # shrink step size if no improvement for a while
                if it_g % 500 == 0:
                    eta0 *= 0.95
                    if eta0 < 1e-6:
                        break

    return best_a.tolist(), best_obj

# Main entry point

def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **_ignore,
) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    evaluator’s objective.  The routine adapts its behaviour to the
    remaining time budget, starts from a variety of analytically
    motivated shapes, follows with many short projected‑gradient
    descents, and finishes with a targeted simulated‑annealing
    refinement.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Target support length – previous best was around 2000, we keep it.
    # A few alternative lengths are also tried (if time permits).
    # ------------------------------------------------------------------
    lengths = [1800, 2000, 2200]

    best_seq = None
    best_obj = float("inf")

    # ------------------------------------------------------------------
    # Iterate over candidate lengths (short first)
    # ------------------------------------------------------------------
    for idx, n in enumerate(lengths):
        now = time.time()
        remaining = deadline - now
        if remaining < 8.0:
            break

        # Allocate a chunk of the remaining budget to this length.
        # First length gets the bulk, later ones share the rest.
        if idx == 0:
            allotted = remaining * 0.55
        else:
            allotted = (remaining * 0.35) / (len(lengths) - idx)

        # ---- Stage 1: quick shape scan (≤ 3 s) --------------------
        shape_time = min(3.0, allotted * 0.07)
        init_a, init_obj = _best_initial_sequence(n, rng, shape_time)
        print(f"[init n={n}] shape‑obj={init_obj:.6f}")

        # ---- Stage 2: many short PGD restarts --------------------
        pgd_time = allotted * 0.55
        pgd_start = time.time()
        # we keep the best sequence found for this n
        best_a_n = init_a.copy()
        best_obj_n = init_obj

        # number of restarts is limited by time, not a fixed count
        restart = 0
        while time.time() - pgd_start < pgd_time:
            restart += 1
            # mix of deterministic (best) and random starts
            if restart == 1:
                a0 = best_a_n
            else:
                # jittered version of the current best
                jitter = rng.normal(scale=0.01, size=n)
                a0 = _project_simplex(best_a_n + jitter)

            # each restart gets a modest slice of the remaining PGD budget
            slice_time = min(5.0, pgd_time - (time.time() - pgd_start))
            if slice_time < 0.1:
                break

            a_candidate, obj_candidate = _pgd_optimize(
                a0, time_limit=slice_time, seed=seed + idx * 1000 + restart
            )
            if obj_candidate < best_obj_n:
                best_obj_n, best_a_n = obj_candidate, np.asarray(a_candidate, dtype=float)
                print(f"[n={n} restart={restart}] new local best O={best_obj_n:.6f}")

        # keep the very best across lengths
        if best_obj_n < best_obj:
            best_obj, best_seq = best_obj_n, best_a_n.copy()
            print(f"[global] new overall best O={best_obj:.6f} from n={n}")

        # -----------------------------------------------------------------
        # optional: a tiny SA refinement for this length before moving on
        # (takes only a few seconds)
        # -----------------------------------------------------------------
        if time.time() < deadline - 2.0:
            sa_time = allotted * 0.15
            seq_ref, obj_ref = _sa_refine(
                best_a_n.tolist(), time_limit=sa_time, seed=seed + 9999 + idx
            )
            if obj_ref < best_obj:
                best_obj, best_seq = obj_ref, np.asarray(seq_ref, dtype=float)
                print(f"[global SA] improved O={best_obj:.6f} (n={n})")

        if time.time() >= deadline - 0.5:
            break

    # ------------------------------------------------------------------
    # Final long‑run SA + PG on the global best (uses any remaining time)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if best_seq is not None and remaining > 6.0:
        seq_final, obj_final = _sa_refine(
            best_seq.tolist(), time_limit=remaining, seed=seed + 55555
        )
        if obj_final < best_obj:
            best_obj, best_seq = obj_final, np.asarray(seq_final, dtype=float)
            print(f"[final] after full SA+PG O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Fallback safety
    # ------------------------------------------------------------------
    if best_seq is None or best_seq.size == 0:
        best_seq = np.ones(200, dtype=float) / 200.0
        best_obj = _objective(best_seq)

    # ------------------------------------------------------------------
    # Final sanity check (diagnostic)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    # Clip to evaluator limits and guard against sum‑too‑small
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    if best_seq.sum() < 0.01:
        best_seq = np.ones_like(best_seq) / best_seq.size

    return best_seq.tolist()
