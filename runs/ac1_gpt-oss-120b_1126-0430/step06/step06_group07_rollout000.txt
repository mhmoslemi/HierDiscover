"""
Propose a non‑negative step function that (approximately) minimises
``evaluate_sequence``.

Algorithm
---------
1. **Initial sampling** – generate a modest number of candidate
   probability vectors from several simple families (exponential,
   Dirichlet with various concentrations, random perturbations of the
   previous best).  Keep the best according to the cheap objective
   ``2·n·max(conv)`` (where ``conv`` is the full linear convolution).

2. **Adam‑based mirror descent on the simplex** – parametrize the
   probabilities as a soft‑max of a free vector `p`.  At each iteration
   the sub‑gradient of the currently maximal convolution entry is
   obtained analytically, transformed to the gradient w.r.t. `p` and
   fed to Adam.  After every update the probabilities are recomputed
   by the soft‑max, guaranteeing the simplex constraint.  The learning
   rate decays like ``1/√t`` and restarts (small random perturbation)
   are performed when no improvement has been seen for many steps.

3. **Ordering & local swaps** – the best vector from step 2 is sorted
   in decreasing order (monotone shape is favourable for the
   convolution objective).  A short simulated‑annealing style search
   then performs random swaps of entries, accepting moves that improve
   the cheap score or with a temperature‑controlled probability.

4. **Final exact evaluation** – the best vector found is converted to
   a Python list and passed to ``evaluate_sequence``.  The routine
   always returns a feasible sequence (non‑negative, sum≥0.01).

The implementation respects the time budget (default 1000 s) and
prints progress information.  All helper functions are top‑level, no
closures or lambdas are used.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # all non‑positive -> uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow_two(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using a power‑of‑two FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa).real
    # keep only the first 2·len(a)-1 entries (linear convolution)
    return conv[: 2 * a.size - 1]


def _conv_max_fft(a: np.ndarray, fft_len: int) -> tuple[float, int]:
    """Maximum value of the autocorrelation a*a and its index."""
    conv = _conv_full_fft(a, fft_len)
    idx = int(conv.argmax())
    return float(conv[idx]), idx


def _subgradient_at_max(a: np.ndarray, kmax: int) -> np.ndarray:
    """
    Sub‑gradient of the maximal convolution entry (kmax) w.r.t. a.
    The gradient is zero outside the support of the term.
    """
    n = a.shape[0]
    lo = max(0, kmax - (n - 1))
    hi = min(kmax, n - 1)
    rng = np.arange(lo, hi + 1, dtype=np.int64)
    j = kmax - rng
    g = np.zeros_like(a)
    eq_mask = (rng == j)
    # off‑diagonal contributions: 2 * a_j
    g[rng[~eq_mask]] = 2.0 * a[j[~eq_mask]]
    # diagonal contribution: a_i (since derivative of a_i^2 is 2 a_i,
    # but the factor 2 is already absorbed in the off‑diagonal case)
    g[rng[eq_mask]] = a[rng[eq_mask]]
    return g


def _softmax(p: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    shift = p.max()
    exp_p = np.exp(p - shift)
    s = exp_p.sum()
    if s == 0.0 or not np.isfinite(s):
        # fallback to uniform
        n = p.shape[0]
        return np.full_like(p, 1.0 / n)
    return exp_p / s
