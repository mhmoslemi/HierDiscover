"""
Hybrid optimisation for the self‑convolution bound.

The routine works in three stages

1️⃣ **Broad parametric sweep** – a set of analytic families (power law,
   double power law, stretched exponential, exponential, logistic,
   symmetric product power‑law, and a beta‑distribution) is scanned with
   a cheap score `2·n·max(a*a)`.  The best distribution found becomes the
   seed for the next stage.

2️⃣ **Mirror‑descent (Adam) on the probability simplex** – the seed is
   turned into logits (so that a = softmax(logits) is a probability
   vector).  Adam with a KL‑mirror step minimises the *smooth* maximum of
   the self‑convolution.  The smoothing constant ε is gradually
   reduced (ε = eps_factor·max_conv) so that the optimiser finally sees
   the true max‑convolution.  The loop runs as long as time permits
   (up to about 1.5 M iterations on a 25000‑element vector).

3️⃣ **Guided random walk** – the final candidate is refined by cheap
   coordinate moves that transfer a small amount of mass from indices
   with a large contribution to the current peak towards indices with a
   small contribution.  Only moves that improve the cheap score are
   accepted.

The returned list is automatically normalised (sum ≈ 1 > 0.01) and
contains only non‑negative numbers; ``evaluate_sequence`` will apply the
required clipping before scoring.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    s = e.sum()
    return e / s if s != 0.0 else np.full_like(z, 1.0 / z.size)


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it that reproduces ``a`` after a soft‑max."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """Smallest power‑of‑two ≥ 3·n‑2 (enough for all needed convolutions)."""
    return 1 << ((3 * n - 2).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Returns (max_conv, grad_a) where

        grad_a = ∇_a [ 2·n·smooth_max(a*a) ]

    ``eps_factor`` multiplies the smoothing constant ε = eps_factor·max_conv.
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv                     # ≤ 0 for stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # Gradient of 2·n·smooth_max (the factor 4·n comes from the derivative of a*a)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a


# ----------------------------------------------------------------------
# Parametric families of distributions
# ----------------------------------------------------------------------


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}   (normalised)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    a /= s if s != 0.0 else 1.0 / n
    return a


def _power_law_offset_distribution(n: int, p: float, offset: float) -> np.ndarray:
    """a_i ∝ (i+offset)^{-p}   (normalised)."""
    idx = np.arange(offset, offset + n, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    a /= s if s != 0.0 else 1.0 / n
    return a


def _double_power_law_distribution(n: int, p: float, q: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}·(n-i)^{-q}   (normalised)."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) ** (-p)) * ((n - i) ** (-q))
    s = a.sum()
    a /= s if s != 0.0 else 1.0 / n
    return a


def _stretched_exp_distribution(n: int, tau_factor: float, gamma: float) -> np.ndarray:
    """
    a_i ∝ exp( - (i/(tau_factor·n))^{gamma} )   (normalised).
    `tau_factor` ∈ (0,1] stretches the decay length.
    """
    tau = max(1e-12, tau_factor * n)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(- (i / tau) ** gamma)
    s = a.sum()
    a /= s if s != 0.0 else 1.0 / n
    return a


def _exp_decay_distribution(n: int, lam: float) -> np.ndarray:
    """a_i ∝ e^{-lam·i}+e^{-lam·(n‑1‑i)}   (normalised)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    s = a.sum()
    a /= s if s != 0.0 else 1.0 / n
    return a


def _logistic_distribution(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ 1/(1+e^{α·(i‑mid)})   (normalised)."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = 1.0 / (1.0 + np.exp(alpha * (i - mid)))
    s = a.sum()
    a /= s if s != 0.0 else 1.0 / n
    return a


def _sym_product_power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ ((i+1)*(n‑i))^{-p}   (symmetric, normalised)."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) * (n - i)) ** (-p)
    s = a.sum()
    a /= s if s != 0.0 else 1.0 / n
    return a


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discretised Beta(α,β) on the interior of (0,1).
    Uses mid‑point rule to avoid zero/one singularities.
    """
    xs = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = xs ** (alpha - 1) * (1.0 - xs) ** (beta - 1)
    s = a.sum()
    a /= s if s != 0.0 else 1.0 / n
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) improves the
    ``evaluate_sequence`` score.  The routine respects the supplied time
    budget (≈1000 s) and uses a hybrid parametric sweep + Adam optimiser
    followed by a cheap guided random walk.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # keep a small safety margin

    # ------------------------------------------------------------------
    # 0. Problem size and initial distribution
    # ------------------------------------------------------------------
    try:
        init_seq = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if init_seq.ndim != 1:
            init_seq = init_seq.ravel()
        if init_seq.shape[0] < 20000:
            # pad with zeros to reach the minimum size that we have tuned for
            pad = 20000 - init_seq.shape[0]
            init_seq = np.pad(init_seq, (0, pad))
        n = init_seq.shape[0]
        a0 = np.maximum(init_seq, 0.0)
        s = a0.sum()
        if s == 0.0:
            a0 = np.full_like(a0, 1.0 / n)
        else:
            a0 /= s
        print(f"[Init] Using supplied sequence, length = {n}")
    except Exception:
        n = 25000
        a0 = np.full(n, 1.0 / n)
        print(f"[Init] No supplied sequence – start from uniform, length = {n}")

    best_seq = a0.copy()
    best_score = 2.0 * n * _conv_max(best_seq)

    # ------------------------------------------------------------------
    # 1. Coarse parametric sweep (including a Beta family)
    # ------------------------------------------------------------------
    families = [
        ("power_law", _power_law_distribution,
         [dict(p=p) for p in np.linspace(0.40, 1.20, 41)]),
        ("power_law_offset", _power_law_offset_distribution,
         [dict(p=p, offset=off)
          for p in np.linspace(0.40, 1.20, 21)
          for off in np.linspace(0.5, 5.0, 21)]),
        ("double_power_law", _double_power_law_distribution,
         [dict(p=p, q=q)
          for p in np.linspace(0.40, 1.00, 13)
          for q in np.linspace(0.40, 1.00, 13)]),
        ("stretched_exp", _stretched_exp_distribution,
         [dict(tau_factor=t, gamma=g)
          for t in np.linspace(0.10, 0.90, 13)
          for g in np.linspace(0.5, 2.0, 13)]),
        ("exp_decay", _exp_decay_distribution,
         [dict(lam=lam) for lam in np.logspace(-6, -1, 40)]),
        ("logistic", _logistic_distribution,
         [dict(alpha=alpha) for alpha in np.linspace(0.01, 6.0, 30)]),
        ("sym_product_power_law", _sym_product_power_law_distribution,
         [dict(p=p) for p in np.linspace(0.40, 2.00, 41)]),
        ("beta", _beta_distribution,
         [dict(alpha=a, beta=b)
          for a in np.linspace(0.15, 0.90, 13)
          for b in np.linspace(0.15, 0.90, 13)]),
    ]

    print("[Sweep] Beginning parametric search")
    for name, generator, param_dicts in families:
        for params in param_dicts:
            if time.time() >= deadline:
                break
            try:
                cand = generator(n, **params)
            except Exception:
                continue
            score = 2.0 * n * _conv_max(cand)
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                # Uncomment for verbose output:
                # print(f"[Sweep] New best {name} {params} -> {score:.6f}")
        if time.time() >= deadline:
            break

    # Small mixture with the uniform distribution (often helps flatten the peak)
    uniform = np.full(n, 1.0 / n)
    for w in np.linspace(0.05, 0.45, 9):
        cand = (1.0 - w) * best_seq + w * uniform
        cand /= cand.sum()
        score = 2.0 * n * _conv_max(cand)
        if score < best_score:
            best_score = score
            best_seq = cand.copy()

    print(f"[Sweep] Best cheap score after sweep = {best_score:.8f}")

    # ------------------------------------------------------------------
    # 2. Mirror‑descent (Adam) on the simplex
    # ------------------------------------------------------------------
    # Initialise logits from the best candidate (add a tiny jitter)
    logits = _logits_from_distribution(best_seq) + 0.02 * np.random.randn(n)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    base_alpha = 0.35          # base learning‑rate for logits
    eps_factor = 0.12          # start smoothing constant (ε = eps_factor·max_conv)
    eps_factor_min = 5e-6
    eps_decay = 0.9997

    report_every = 40000
    it = 0
    last_improve = 0
    best_seq_sofar = best_seq.copy()
    best_score_sofar = best_score

    # Upper bound on iteration count – the loop will also stop when time runs out
    max_iters = 1_500_000

    while time.time() < deadline and it < max_iters:
        it += 1

        # probability vector (lies on the simplex)
        a = _softmax(logits)

        # smoothed max‑convolution and its gradient w.r.t. a
        max_conv, grad_a = _conv_max_and_grad(a, eps_factor)
        cheap_score = 2.0 * n * max_conv

        # record best ever seen
        if cheap_score < best_score_sofar - 1e-12:
            best_score_sofar = cheap_score
            best_seq_sofar = a.copy()
            last_improve = it
            if it % report_every == 0:
                print(f"[Iter {it:07d}] NEW BEST cheap = {cheap_score:.8f}")

        # Mirror‑descent (Adam) update on logits
        mu = np.dot(grad_a, a)                 # <grad, a>
        grad_z = a * (grad_a - mu)             # natural gradient in probability space

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        alpha_t = base_alpha / (math.sqrt(it) + 1.0)   # inverse‑sqrt schedule
        step = - m_hat / (np.sqrt(v_hat) + eps_adam)

        # back‑tracking line‑search (max 5 halvings)
        trial_alpha = alpha_t
        accepted = False
        for _ in range(5):
            new_logits = logits + trial_alpha * step
            new_logits = np.clip(new_logits, -50.0, 50.0)
            a_new = _softmax(new_logits)
            new_score = 2.0 * n * _conv_max(a_new)
            if new_score <= cheap_score:
                logits = new_logits
                accepted = True
                break
            trial_alpha *= 0.5
        if not accepted:
            logits += alpha_t * step
            logits = np.clip(logits, -50.0, 50.0)

        # occasional stochastic jitter to escape plateaus
        if it % 5000 == 0 and np.random.rand() < 0.12:
            logits += 0.02 * np.random.randn(n)
            m *= 0.7
            v *= 0.7

        # decay smoothing constant (approaches the true max)
        eps_factor = max(eps_factor_min, eps_factor * eps_decay)

        # restart if we have been idle for too long
        if it - last_improve > 300_000:
            print(f"[Iter {it}] Stagnation – restarting from best so far")
            logits = _logits_from_distribution(best_seq_sofar) + 0.03 * np.random.randn(n)
            m.fill(0.0)
            v.fill(0.0)
            eps_factor = 0.12
            last_improve = it

    best_seq = best_seq_sofar
    best_score = best_score_sofar
    print(f"[Adam] Finished after {it} iterations, best cheap score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Guided cheap random walk (mass transfer moves)
    # ------------------------------------------------------------------
    a = best_seq.copy()
    cheap_score = best_score

    # pre‑compute an index‑order for “high‑impact” and “low‑impact” positions
    # based on the gradient of the *smooth* max (already cheap to obtain)
    while time.time() < deadline:
        # compute the smooth gradient only when we have spare time
        _, grad_a = _conv_max_and_grad(a, eps_factor_min)
        # indices where the gradient (i.e. contribution to the current peak) is large
        high_idx = np.argpartition(-grad_a, int(0.02 * n))[: int(0.02 * n)]
        low_idx = np.argpartition(grad_a, int(0.02 * n))[: int(0.02 * n)]

        i = int(high_idx[np.random.randint(high_idx.size)])
        j = int(low_idx[np.random.randint(low_idx.size)])

        if i == j:
            continue
        delta = 0.001 * a[i]          # shift 0.1 % of mass from i to j
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # ensure numerical non‑negativity
        a_new = np.maximum(a_new, 0.0)
        new_score = 2.0 * n * _conv_max(a_new)
        if new_score < cheap_score:
            a = a_new
            cheap_score = new_score
            best_seq = a.copy()
            best_score = cheap_score
            # occasional progress report
            if np.random.rand() < 0.001:
                print(f"[Walk] Improved cheap = {cheap_score:.8f}")

    # ------------------------------------------------------------------
    # 4. Final exact evaluation
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        exact_score = evaluate_sequence(final_seq)   # platform‑provided
        print(f"Final exact score = {exact_score:.6f}")
    except Exception as exc:
        exact_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    return final_seq
