def propose_candidate(seed=42, budget_s=1000, **_ignore):

    rng = np.random.default_rng(seed)

    deadline = start + budget_s - 0.5

    # Resolve possible initial sequence.
    init_seq = None
    if 'height_sequence_1' in globals():
        # convert to probability
        seq0 = np.asarray(height_sequence_1, dtype=float)
        seq0 = np.clip(seq0, 0.0, None)
        if seq0.sum() < 0.01:
            seq0 = np.ones_like(seq0)
        seq0 = seq0 / seq0.sum()
        init_seq = seq0

    # define phases
    phases = []
    remaining = deadline - time.time()
    # allocate fractions
    # If ample time, three phases: 0.2, 0.4, 0.4.
    # else use only one phase.
    if remaining > 30.0:
        # get lengths
        n1 = 8000
        n2 = 12000
        n3 = 20000
        t1 = remaining * 0.2
        t2 = remaining * 0.4
        t3 = remaining - t1 - t2
        phases = [(n1, t1), (n2, t2), (n3, t3)]
    else:
        # just a single phase with moderate n
        n = max(4000, int(remaining / 0.1))  # guess
        phases = [(min(20000, n), remaining)]

    best_seq = None
    best_obj = float('inf')

    for idx, (n, t_budget) in enumerate(phases):
        # Prepare init: upsample if needed
        if best_seq is not None:
            # upsample best_seq to length n.
            x_old = np.linspace(0.0, 1.0, best_seq.size, endpoint=False)
            x_new = np.linspace(0.0, 1.0, n, endpoint=False)
            init_a = np.interp(x_new, x_old, best_seq)
            init_a = np.clip(init_a, 0.0, None)
            init_a /= init_a.sum()
        else:
            init_a = init_seq

        # Run smoothed Adam.
        seq, obj = _run_opt_smooth_adam(n, t_budget, init_a, seed + idx*12345)
        if obj < best_obj:
            best_obj = obj
            best_seq = np.asarray(seq, dtype=float)
            print(f"[phase {idx}] new best O={best_obj} (n={n})")

    # After all phases, final refinement by SA using remaining time (if any)
    remaining = deadline - time.time()
    if remaining > 5.0 and best_seq is not None:
        seq_ref, obj_ref = _run_sa(best_seq.tolist(), remaining, seed + 9999)
        if obj_ref < best_obj:
            best_seq = np.asarray(seq_ref, dtype=float)
            best_obj = obj_ref

    # Final evaluation

        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score}")

        print("[final] evaluate_sequence call failed:", exc)

def _run_opt_smooth_adam(n, time_limit, init_a=None, seed=0):
    start = time.time()
    rng = np.random.default_rng(seed)
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
    else:
        a = rng.random(n)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / s
    x = np.log(a + 1e-16)

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    L = _next_pow2(3*n - 2)
    conv_len = 2*n - 1
    w_rev_len = 2*n - 1

    # initial tau (smoothness)
    # compute initial M via conv.
    fa = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fa * fa, n=L)[:conv_len]
    M = conv_full.max()
    tau0 = max(M * 5.0, 1e-6)  # ensure not too small
    # schedule: exponential decay final factor ~1e-6.
    final_factor = 1e-6

    best_a = a.copy()
    best_obj = float(2.0 * n * M)
    it = 0

    while time.time() - start < time_limit:
        it += 1

        a = _softmax(x)  # still using softmax for always on simplex.

        # convolution
        fa = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(fa * fa, n=L)[:conv_len]
        M = conv_full.max()
        obj = float(2.0 * n * M)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            print(f"[n={n} it={it}] new best O={best_obj:.6f}")

        # compute smoothing parameter tau as function of elapsed fraction
        elapsed = time.time() - start
        frac = min(elapsed / time_limit, 1.0)
        tau = tau0 * (final_factor ** frac)
        tau = max(tau, 1e-12)

        # compute weights for smooth max
        # subtract M for stability
        w = np.exp((conv_full - M) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.ones_like(w) / w.size
        else:
            w /= w_sum

        # compute gradient of smooth max
        # w_rev = w[::-1]
        w_rev = w[::-1]
        fw = np.fft.rfft(w_rev, n=L)
        conv_rev = np.fft.irfft(fa * fw, n=L)[: (n + w_rev.size - 1)]
        offset = w_rev.size - 1  # = 2n-2
        g = 2.0 * conv_rev[offset - np.arange(n)]

        # natural gradient in log space (mirrored)
        dot = np.dot(a, g)
        g_x = a * (g - dot)

        # learning rate schedule decays from lr_start to lr_end
        lr_start = 0.5
        lr_end = 0.001
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * frac)

        # Adam moments update
        m = beta1 * m + (1 - beta1) * g_x
        v = beta2 * v + (1 - beta2) * (g_x ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        # update logits
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # jitter occasional
        if it % 2000 == 0:
            jitter = rng.normal(scale=0.01, size=n)
            x += jitter

        # restart occasional
        if it % 50000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.005, size=n)
            m.fill(0.0)
            v.fill(0.0)

    # compress small entries
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        # recompute objective after compression
        best_obj = float(2.0 * best_a.size * _conv_self_fft(best_a).max())
        print(f"[compress] reduced length to {best_a.size}, O={best_obj:.6f}")

    return best_a.tolist(), best_obj

"""
Non‑negative step‑function optimiser (smooth‑max Adam + final SA).

The optimiser works on a probability vector a (∑a_i=1).  The objective
to minimise is 2·n·max(conv(a,a)).  We perform an Adam optimisation in
log‑it (exponentiated‑gradient) space while smoothing the max with a
temperature τ that decays from a large value to (effectively) zero.
When τ becomes very small we switch to the exact sub‑gradient of the
maximising convolution entry, obtaining a fast but accurate final
phase.  After the main optimisation a short simulated‑annealing
refinement moves a few per‑cent of mass between coordinates and often
improves the bound further.

The routine adapts the support length to the available time budget
(up to ~20 k entries) and respects the 1000 s limit.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·len(a)-1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return conv_full[: 2 * n - 1]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max of a vector."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


def _grad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """
    Exact gradient of the k‑th convolution entry w.r.t. ``a``.
    Returns 2·a_{k−i} for indices where it is defined, otherwise 0.
    """
    n = a.shape[0]
    idx = np.arange(n)
    j = k - idx
    mask = (j >= 0) & (j < n)
    g = np.zeros_like(a)
    if mask.any():
        g[mask] = 2.0 * a[j[mask]]
    return g


# ----------------------------------------------------------------------
# Smooth‑max Adam optimiser on the simplex
# ----------------------------------------------------------------------


def _run_opt_smooth_adam(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
) -> Tuple[List[float], float]:
    """
    Adam optimiser that minimises 2·n·max(conv(a,a)) using a smooth
    approximation of the max.  The algorithm works in log‑it space, keeps
    a running best point and prunes tiny entries at the end.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # initialise a on the simplex
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
    else:
        a = rng.random(n)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n, dtype=float) / n
    else:
        a = a / s

    # log‑it representation (mirrored gradient)
    x = np.log(a + 1e-16)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # FFT size sufficient for both self‑convolution and the gradient step
    L = _next_pow2(3 * n - 2)          # for a * reversed‑weight convolution
    conv_len = 2 * n - 1                # length of a*a

    # pre‑compute the first objective value
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[:conv_len]
    best_obj = float(2.0 * n * conv.max())
    best_a = a.copy()
    it = 0

    while time.time() - start < time_limit:
        it += 1

        # point on the simplex
        a = _softmax(x)

        # FFT of a (re‑used for both conv and gradient)
        fa = np.fft.rfft(a, n=L)

        # self‑convolution
        conv = np.fft.irfft(fa * fa, n=L)[:conv_len]
        M = conv.max()
        obj = float(2.0 * n * M)

        # keep the best seen so far
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        # time fraction for schedules
        elapsed = time.time() - start
        frac = min(elapsed / time_limit, 1.0)

        # learning‑rate (exponential decay)
        lr_start, lr_end = 0.5, 0.001
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * frac)

        # temperature schedule for the smooth max
        tau0 = max(M * 10.0, 1e-6)      # start fairly smooth
        tau = max(tau0 * (1.0 - frac), 1e-12)   # linear decay to zero

        # ---- gradient of the smooth max ---------------------------------
        if tau > 1e-8:          # smooth regime
            # soft‑max weights over the convolution entries
            w = np.exp((conv - M) / tau)   # subtract M for stability
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.ones_like(w) / w.size
            else:
                w /= w_sum

            # gradient = 2 * sum_j a_j w_{i+j}
            w_rev = w[::-1]                     # reverse for convolution
            fw = np.fft.rfft(w_rev, n=L)
            rev_len = n + w_rev.size - 1        # = 3n‑2
            conv_rev = np.fft.irfft(fa * fw, n=L)[:rev_len]
            offset = w_rev.size - 1             # = 2n‑2
            g = 2.0 * conv_rev[offset - np.arange(n)]
        else:                  # hard‑max regime (sub‑gradient at the argmax)
            k_max = int(np.argmax(conv))
            g = _grad_max_conv(a, k_max)

        # natural gradient in log‑it space (mirrored / exponentiated)
        dot = np.dot(a, g)
        g_x = a * (g - dot)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # Adam step
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape flat plateaux
        if it % 5000 == 0:
            jitter = rng.normal(scale=0.01, size=n)
            x += jitter

        # occasional full restart from the best point seen so far
        if it % 80000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.005, size=n)
            m.fill(0.0)
            v.fill(0.0)

    # ------------------------------------------------------------------
    # prune negligible entries (they increase n without helping)
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = float(2.0 * best_a.size * _conv_self_fft(best_a).max())
        print(f"[compress] reduced length → {best_a.size}, O={best_obj:.6f}")

    return best_a.tolist(), best_obj

def _run_sa(
    init_seq: List[float],
    time_limit: float,
    seed: int = 0,
) -> Tuple[List[float], float]:
    """Refine a sequence by small mass‑transfer moves with a Metropolis schedule."""
    start = time.time()
    rng = np.random.default_rng(seed)
    a = np.asarray(init_seq, dtype=float)
    if a.sum() < 1e-12:
        a = np.ones_like(a) / a.size
    else:
        a = a / a.sum()
    n = a.shape[0]

    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())
    best_seq = a.copy()

    # temperature schedule
    T = best_obj * 0.1 + 1e-6
    T_min = best_obj * 1e-6 + 1e-9
    decay = 0.995
    it = 0

    while time.time() - start < time_limit:
        it += 1
        k_max = int(np.argmax(conv))

        # contribution of each index to the maximal entry
        i_vals = np.arange(n)
        j_vals = k_max - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i_vals[mask]] * a[j_vals[mask]]
        total_contrib = contrib.sum()

        if total_contrib > 0:
            probs = contrib / total_contrib
            i = rng.choice(n, p=probs)
        else:
            i = rng.integers(n)

        # choose a distinct destination uniformly
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # transfer a modest fraction of mass from i to j
        delta = a[i] * rng.random() * 0.05  # up to 5 % of a[i]
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.clip(a_new, 0.0, None)
        a_new /= a_new.sum()

        conv_new = _conv_self_fft(a_new)
        obj_new = float(2.0 * n * conv_new.max())

        # Metropolis acceptance
        if obj_new < best_obj - 1e-12:
            a, conv, best_obj, best_seq = a_new, conv_new, obj_new, a_new.copy()
            print(f"[SA it={it}] new best O={best_obj:.6f}")
        else:
            prob = np.exp(-(obj_new - best_obj) / T) if T > 0 else 0.0
            if rng.random() < prob:
                a, conv = a_new, conv_new

        # temperature decay
        if it % 2000 == 0:
            T = max(T * decay, T_min)

        # occasional global jitter
        if it % 10000 == 0:
            noise = rng.normal(scale=0.01, size=n)
            a = np.clip(a + noise, 0.0, None)
            a /= a.sum()
            conv = _conv_self_fft(a)

    # final compression of tiny entries
    thresh = 1e-12
    mask = best_seq > thresh
    if mask.sum() < best_seq.size:
        comp = best_seq[mask]
        comp /= comp.sum()
        comp_obj = float(2.0 * comp.size * _conv_self_fft(comp).max())
        if comp_obj < best_obj:
            best_seq, best_obj = comp, comp_obj
            print(f"[SA compress] reduced length → {comp.size}, O={best_obj:.6f}")

    return best_seq.tolist(), best_obj
