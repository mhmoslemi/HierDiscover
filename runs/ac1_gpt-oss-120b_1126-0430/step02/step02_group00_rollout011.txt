"""
Propose a non‑negative step‑function sequence that minimises

    O(a) = 2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm consists of two phases.

Phase 1 – smooth optimisation
    * The constraint a_i ≥ 0 and ∑ a_i = 1 is enforced by a soft‑max
      re‑parameterisation a = softmax(y).
    * A smooth surrogate of the max‑convolution,
          f_α(a) = (1/α)·log Σ_k exp(α·conv_k(a)),
      is minimised for a sequence of increasing α (5 → 2000) with
      L‑BFGS‑B (SciPy).  The gradient of f_α is obtained by an FFT‑based
      correlation and the soft‑max Jacobian.
    * After each α‑stage the exact objective O(a) is evaluated and the
      best sequence is retained.

Phase 2 – local refinement
    * A fast hill‑climbing loop moves a random fraction of probability
      mass from one index to another (mass‑transfer moves).  Moves that
      lower O are always accepted; otherwise they are accepted with a
      Metropolis probability that is cooled over time.
    * Every few iterations a sub‑gradient step for the exact maximal
      convolution entry is applied and the result is projected back onto
      the simplex.
    * The loop runs until the 1000 s budget expires.

The final sequence (list of floats) is returned.  All numbers are
non‑negative and the sum is positive, so `evaluate_sequence` will never
return ``inf``.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Optional: SciPy optimiser (if available)
try:
    from scipy.optimize import minimize
except Exception:          # pragma: no cover
    minimize = None

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum(x) = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:                     # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full self‑convolution a*a using an FFT of length L."""
    fft_a = rfft(a, n=L)
    conv_full = irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


def _smooth_gradient(a: np.ndarray, conv: np.ndarray, alpha: float,
                     L: int) -> np.ndarray:
    """
    Gradient of the smooth surrogate (1/α)·log Σ exp(α·conv_k)
    w.r.t. a.  The formula reduces to 2·corr(w, a) where
    w_k ∝ exp(α·conv_k).
    """
    # soft‑max weights
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / exp_scaled.sum()

    # correlation w ⋆ a   (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = rfft(w, n=L)
    fft_a_rev = rfft(a_rev, n=L)
    corr_rev = irfft(fft_w * fft_a_rev, n=L)       # length = 3·n‑2
    n = a.size
    grad = 2.0 * corr_rev[n - 1: n - 1 + n]        # central n entries
    return grad


def _exact_objective(a: np.ndarray, n: int, L: int) -> float:
    """Exact objective O(a) = 2·n·max(conv(a,a)) (∑a = 1)."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for the step‑function that minimises `evaluate_sequence`.
    Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    # ----- initialise ------------------------------------------------
    if 'height_sequence_1' in globals():
        base = np.array(height_sequence_1, dtype=float)
        n = base.size
        a0 = base / base.sum()                     # normalise to simplex
        print(f"[init] using existing sequence, n={n}")
    else:
        n = 4000
        a0 = np.full(n, 1.0 / n)                   # uniform start
        print(f"[init] no prior sequence – using uniform, n={n}")

    # FFT length sufficient for both convolution (2n‑1) and correlation (3n‑2)
    L = 1 << ((3 * n - 2).bit_length())

    # best solution so far
    best_seq = a0.copy()
    best_score = _exact_objective(best_seq, n, L)
    print(f"[start] exact O = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – smooth optimisation with L‑BFGS (if SciPy is available)
    if minimize is not None:
        # start from soft‑max parametrisation
        eps = 1e-12
        y = np.log(best_seq + eps)                 # initial logits

        alphas = [5.0, 20.0, 80.0, 300.0, 800.0, 2000.0]
        for alpha in alphas:
            # remaining time check
            if time.time() >= deadline:
                break
            max_iter = 200
            print(f"[phase‑1] α={alpha:.1f} start (max_iter={max_iter})")

            # objective + gradient for scipy.minimize
            def fun_and_grad(y_vec):
                a = _softmax(y_vec)
                conv = _conv_fft(a, L)

                # ----- smooth surrogate -----
                s = alpha * conv
                s_max = np.max(s)
                exp_s = np.exp(s - s_max)
                sum_exp = np.exp(s_max) * exp_s.sum()   # = Σ exp(α·conv)
                log_sum_exp = s_max + np.log(exp_s.sum())

                f = log_sum_exp / alpha                # (1/α)·log Σ exp

                # gradient w.r.t. a (2·corr(w,a))
                grad_a = _smooth_gradient(a, conv, alpha, L)

                # gradient w.r.t. logits y (soft‑max Jacobian)
                dot = np.dot(a, grad_a)
                grad_y = a * (grad_a - dot)
                return f, grad_y

            res = minimize(fun_and_grad, y, method='L-BFGS-B',
                           jac=True,

            y = res.x
            a = _softmax(y)
            cur_score = _exact_objective(a, n, L)
            if cur_score < best_score:
                best_score = cur_score

    else:                                          # pragma: no cover

    # Phase 2 – local refinement (hill‑climbing + sub‑gradient)
    print(f"[phase‑2] entering local refinement, best O={best_score:.6f}")

            new_obj = 2.0 * n * conv_new.max()

            # Metropolis acceptance
            accept = new_obj < cur_obj
            if not accept:

                accept = np.random.rand() < prob

            if accept:
                a = a_new
                conv = conv_new
                cur_obj = new_obj
                no_improve_counter = 0
                if cur_obj < best_score:
                    best_score = cur_obj
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] New global best O={best_score:.6f} (iter={iter_local})")
            else:
                no_improve_counter += 1
        else:
            no_improve_counter += 1

        # ---- sub‑gradient step on the active convolution entry ---------
        if iter_local % subgrad_interval == 0:
            max_val = conv.max()
            ks = np.flatnonzero(conv == max_val)
            g = np.zeros_like(a)
            for k in ks:
                i_start = max(0, k - (n - 1))
                i_end = min(n - 1, k)
                i_idx = np.arange(i_start, i_end + 1, dtype=int)
                j_idx = k - i_idx
                g[i_idx] += 2.0 * a[j_idx]
            g /= len(ks)                     # average over ties

            eta = 0.07 * (1.0 - iter_local / max_iter_local)
            a = a - eta * g
            a = _project_to_simplex(a)       # restore feasibility
            conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_score:
                best_score = cur_obj
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Sub‑grad new best O={best_score:.6f}")

        # ---- temperature decay -----------------------------------------
        T = temperature0 * (1.0 - iter_local / max_iter_local)

        # ---- occasional restart if stuck --------------------------------
        if no_improve_counter >= 5_000:
            # re‑initialise a small jitter around the current best
            a = best_seq.copy()
            jitter = np.random.randn(n) * 0.001
            a = np.clip(a + jitter, 0, None)
            a = _project_to_simplex(a)
            conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max()
            no_improve_counter = 0

        iter_local += 1

    # ------------------------------------------------------------------
    # Final sanity check with the provided evaluation function
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:                               # pragma: no cover
        print("[final] evaluate_sequence failed:", e)

    return best_seq.tolist()
