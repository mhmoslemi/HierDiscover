"""
Propose a non‑negative step‑function sequence that minimizes  

    O(a) = 2·n·max(conv(a,a)) / (sum(a)^2)  

(the sum factor cancels out, so we may constrain sum(a)=1 and minimize  
2·n·max(conv).)

Algorithm
---------
* **Phase 1 – Adaptive Adam on a smoothed log‑sum‑exp surrogate**  
  - The surrogate is  fα(a) = (2·n/α)·log∑k e^{α·conv_k}.  
  - Its gradient can be written as 2·corr(w,a) where w_k ∝ e^{α·conv_k}.  
  - An Adam optimiser (first/second‑moment adaptation) is applied to this
    gradient, followed by Euclidean projection onto the probability simplex.
  - The smoothing parameter α is annealed linearly from 5 up to 5000, which
    gradually focuses the optimisation on the true maximum.
  - Occasionally a tiny random jitter is added and very small entries are
    pruned to keep the support compact.

* **Phase 2 – Stochastic local refinement**  
  - Starting from the best candidate of Phase 1 we repeatedly perform
    small “mass‑transfer” moves (move a random fraction of probability mass
    from one index to another).  Moves are accepted if they improve the
    exact objective, otherwise with a Metropolis probability that slowly
    cools down.
  - Every few iterations a pure sub‑gradient step for the current maximal
    convolution entry is taken (the same direction used in the reference
    implementation).
  - Random restarts with a little jitter ensure exploration of distant
    basins.

Both phases respect the overall time budget (default 1000 s).  The
function returns the best sequence found as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        # all non‑positive – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full self‑convolution (length 2·n‑1)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return a*a (full convolution) using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Exact objective for a simplex‑normalised vector
def _objective_simplex(a: np.ndarray, n: int, L: int) -> float:
    """2·n·max(conv(a,a)) (a is assumed to sum to 1)."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Sub‑gradient of the maximal convolution entry (as used in the reference code)
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Return the sub‑gradient of max(conv) w.r.t. a."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)  # possibly several maximisers
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    # average over all maximising k's
    if ks.size > 0:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
# Gradient of the log‑sum‑exp (soft‑max) surrogate of the max
def _smooth_gradient(a: np.ndarray, conv: np.ndarray, alpha: float, L: int) -> np.ndarray:
    """
    Compute ∇_a ( (2·n/α)·log∑_k e^{α·conv_k} ).
    The formula reduces to 2·corr(w, a) where w_k ∝ e^{α·conv_k}.
    """
    # soft‑max weights
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)          # for numerical stability
    w = exp_scaled / exp_scaled.sum()

    # correlation w ⋆ a   (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)   # length ≥ 3·n‑2
    n_len = a.size
    grad = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]
    return grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):

    Search for a step‑function sequence that minimises the provided evaluation
    function.  Returns the sequence as a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()

    deadline = start_time + budget_s - 0.3

    # ------------------------------------------------------------------

        base = np.array(height_sequence_1, dtype=float)
        n = base.size
        a = base / base.sum()                      # simplex normalisation
        print(f"[init] using supplied sequence, n={n}")
    else:
        # default size – a few thousand entries give a good trade‑off
        n = 4600
        a = np.full(n, 1.0 / n)                    # uniform start
        print(f"[init] no supplied sequence – using uniform, n={n}")

    # FFT length – next power of two larger than the longest convolution/correlation
    L = 1 << ((3 * n - 2).bit_length())

    # ------------------------------------------------------------------
    # Phase 1 – Adam on the smooth surrogate
    best_seq = a.copy()
    best_score = _objective_simplex(best_seq, n, L)
    print(f"[phase‑1] start O={best_score:.6f}")

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    lr0 = 0.5                      # base learning rate (will be decayed)
    alpha_start = 5.0
    alpha_end = 5000.0

    max_iter_phase1 = 300_000      # generous upper bound; actual stop is time‑driven
    it = 0

    # allocate roughly 55 % of the total budget to phase 1

    return best_seq.tolist()
