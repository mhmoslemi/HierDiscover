    # Additional random perturb starting from arcsine sequence for comparison

"""
Step‑function optimizer for the convolution bound.

The objective (used by `evaluate_sequence`) is
    score = 2·n·max_k (a * a)_k / (∑ a_i)^2 .
The score is invariant under scaling of the whole vector, therefore we fix
∑ a_i = 1 during the search and minimise the maximal self‑convolution entry.
The algorithm consists of three phases:

1. **Parameter sweep** – generate a few simple families (power‑law,
   exponential, combined) and keep the one with the smallest score.
2. **Exponentiated‑gradient descent** – repeatedly locate the index k where the
   convolution attains its maximum, form the sub‑gradient
        g_i = a_{k-i}   (zero out of range),
   and update the distribution by a multiplicative rule
        a_i ← a_i·exp(−η·g_i) ,
   followed by a projection onto the probability simplex (renormalisation).
   The step‑size η decays like η₀/√t.
   This pushes mass from positions that strongly contribute to the peak
   towards positions with tiny contribution, dramatically flattening the
   convolution.
3. **Final scaling** – the optimiser works with a unit sum; before returning we
   rescale the vector to the conventional sum √(2·n) (any positive scale is
   acceptable because the score is scale‑invariant).

The whole routine respects the 1000 s time budget, runs in pure NumPy and
does not use any external files or network resources.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast evaluation (identical to the supplied evaluate_sequence but without
# the safety checks).  Returns the exact score used by the harness.
# ----------------------------------------------------------------------
def _fast_evaluation(seq: np.ndarray) -> float:
    n = seq.size
    if n == 0:
        return float("inf")
    # FFT length = next power of two >= 2*n-1
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    max_b = float(conv.max())
    sum_a = float(seq.sum())
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Catalan (arcsine) coefficients – provided for a quick baseline.
# ----------------------------------------------------------------------
def _catalan_sequence(length: int) -> np.ndarray:
    seq = np.empty(length, dtype=float)
    a = 1.0
    seq[0] = a
    for i in range(1, length):
        a *= (2.0 * i - 1) / (2.0 * i)
        seq[i] = a
    return seq


# ----------------------------------------------------------------------
# Projection onto the probability simplex (sum = 1, all entries ≥ 0).
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto {x ≥ 0, sum x = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerically degenerate – fall back to uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].sum() / rho
    w = np.maximum(v - theta, 0.0)
    # due to round‑off the sum may deviate a bit – renormalise
    w_sum = w.sum()
    if w_sum == 0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum


# ----------------------------------------------------------------------
# Main optimisation routine.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights whose evaluation score is well
    below the historic π/2 bound.  The method first scans a few parametric
    families for a good starting point and then runs an exponentiated‑gradient
    descent with simplex projection.
    """
    np.random.seed(seed)

    # safety margin for the harness (it stops a few seconds before the hard limit)
    deadline = time.time() + budget_s - 5.0

    # length of the step function – a few thousand balances quality vs speed
    n = 5000
    fft_len = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # 1)  Rough parameter sweep – keep the best primitive shape.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    # power‑law family a_i ∝ (i+1)^{-p}
    for p in np.linspace(0.25, 2.0, 15):
        a = np.arange(1, n + 1, dtype=float) ** (-p)
        a /= a.sum()                     # enforce ∑a = 1
        score = _fast_evaluation(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[init] power p={p:.3f} → score={score:.8f}")

    # exponential decay a_i ∝ exp(-λ·i)
    for lam in np.linspace(0.0004, 0.012, 15):
        a = np.exp(-lam * np.arange(n, dtype=float))
        a /= a.sum()
        score = _fast_evaluation(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[init] exp λ={lam:.5f} → score={score:.8f}")

    # combined power‑law × exponential
    for p in (0.5, 0.8, 1.0):
        for lam in (0.001, 0.003, 0.006):
            a = (np.arange(1, n + 1, dtype=float) ** (-p)) * np.exp(-lam * np.arange(n))
            a /= a.sum()
            score = _fast_evaluation(a)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[init] combo p={p}, λ={lam:.5f} → score={score:.8f}")

    # Catalan (arcsine) seed – classic baseline
    cat = _catalan_sequence(n)
    cat = cat / cat.sum()
    cat_score = _fast_evaluation(cat)
    if cat_score < best_score:
        best_score = cat_score
        best_seq = cat.copy()
        print(f"[init] Catalan (arcsine) → score={cat_score:.8f}")

    # ------------------------------------------------------------------
    # 2)  Exponentiated‑gradient descent on the simplex.
    # ------------------------------------------------------------------
    a = best_seq.copy()               # current iterate (unit sum)
    best_seq = a.copy()
    best_score = _fast_evaluation(a)

    eta0 = 0.8                        # initial step size
    it = 0

    while time.time() < deadline:
        # ----- convolution and peak location -----
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_val = float(conv.max())
        # break ties randomly
        max_idxs = np.flatnonzero(conv >= max_val - 1e-12)
        k = int(np.random.choice(max_idxs))

        # ----- sub‑gradient for the maximal entry -----
        g = np.zeros_like(a)
        low = max(0, k - n + 1)
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        g[idx] = a[k - idx]                     # w_i = a_{k-i}

        # ----- multiplicative (exponentiated) update -----
        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * g)

        # ----- projection onto the probability simplex -----
        a = _proj_simplex(a)

        # ----- occasional tiny random jitter to avoid stagnation -----
        if (it & 0x3FF) == 0:                    # every 1024 steps
            jitter_i = np.random.randint(n)
            a[jitter_i] *= 1.01
            a = _proj_simplex(a)

        # ----- evaluate and keep the best seen so far -----
        cur_score = _fast_evaluation(a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"Iter {it:6d}: improved score = {best_score:.8f}, max conv = {max_val:.8f}")

        it += 1

    # ------------------------------------------------------------------
    # 3)  Final scaling – any positive scale is fine; we use √(2·n) to match
    #      the convention used in the reference implementation.
    # ------------------------------------------------------------------
    scale = np.sqrt(2.0 * n) / best_seq.sum()    # should be ≈√(2·n) because sum≈1
    best_seq = best_seq * scale
    # Clip to the limits enforced by evaluate_sequence (just to be safe)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Finished. Best score obtained: {best_score:.8f}")
    return best_seq.tolist()
