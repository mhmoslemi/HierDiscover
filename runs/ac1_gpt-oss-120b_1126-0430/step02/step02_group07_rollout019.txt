"""
Propose a step‑function (list of non‑negative heights) that minimises

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

The approach combines several ideas:

1.  Extensive random sampling of several parametric families
    (power‑laws, power‑law·exponential, mixtures of power‑laws).  All
    candidates are normalised to a probability vector (sum = 1) and
    evaluated with the cheap internal scorer ``_compute_score``.

2.  A short differential‑evolution run for the most promising families.
    This is a global optimiser that works directly on the few
    parameters of the families (2‑5 dimensions) and therefore explores
    the space far beyond the hand‑crafted grid used previously.

3.  A smooth‑max gradient‑descent refinement.  The objective
    ``max(conv(a,a))`` is approximated by a log‑sum‑exp (temperature
    λ≈20).  The gradient is obtained analytically via a correlation,
    the step is projected onto the probability simplex and the process
    is repeated with a linearly decaying learning‑rate.

4.  A final fast hill‑climbing pass identical to the one from the
    previous baseline (move a small amount of mass from the index that
    contributes most to the current peak towards the index that
    contributes least).

All helpers are top‑level functions, there are no closures,
and the whole search respects the supplied time‑budget (default
950 s, leaving a safety margin).  The best sequence found is returned
as a plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import differential_evolution

# ----------------------------------------------------------------------
# Basic utilities (pure NumPy, no external state)
# ----------------------------------------------------------------------


def _project_to_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


def _compute_score(arr):
    """
    Fast score: 2*n*max(conv(arr,arr)) / (sum(arr)**2)
    ``arr`` may be any non‑negative 1‑D numpy array.
    """
    total = arr.sum()
    if total <= 0:
        return float("inf")
    conv = np.convolve(arr, arr)
    max_val = conv.max()
    n = arr.shape[0]
    return 2.0 * n * max_val / (total * total)


def _make_powerlaw(alpha, shift, n):
    """a_i = (i+shift)^(-alpha) (no normalisation)."""
    idx = np.arange(n, dtype=float)
    return np.power(idx + shift, -alpha)


def _make_powerlaw_exp(alpha, shift, beta, n):
    """a_i = (i+shift)^(-alpha) * exp(-beta*i)."""
    idx = np.arange(n, dtype=float)
    return np.power(idx + shift, -alpha) * np.exp(-beta * idx)


def _make_powerlaw_cut(alpha, shift, gamma, n):
    """a_i = (i+shift)^(-alpha) * (1 - i/(n-1))^gamma, zero at the end."""
    idx = np.arange(n, dtype=float)
    tail = 1.0 - idx / (n - 1)
    tail = np.maximum(tail, 0.0)  # avoid negative due to round‑off
    return np.power(idx + shift, -alpha) * np.power(tail, gamma)


def _make_mix2(params, n):
    """
    params = (alpha1, shift1, alpha2, shift2, w)
    w∈[0,1] is the weight of the first component.
    """
    a1 = _make_powerlaw(params[0], params[1], n)
    a2 = _make_powerlaw(params[2], params[3], n)
    w = params[4]
    return w * a1 + (1.0 - w) * a2


def _make_mix3(params, n):
    """
    params = (a1, s1, a2, s2, a3, s3, w1, w2)
    w1,w2∈[0,1] and w1+w2≤1, the third weight is 1‑w1‑w2.
    """
    a1 = _make_powerlaw(params[0], params[1], n)
    a2 = _make_powerlaw(params[2], params[3], n)
    a3 = _make_powerlaw(params[4], params[5], n)
    w1, w2 = params[6], params[7]
    w3 = 1.0 - w1 - w2
    return w1 * a1 + w2 * a2 + w3 * a3


def _smooth_max_gradient(a, lam):
    """
    Gradient of the smoothed max (log‑sum‑exp) of conv(a,a).
    Returns a vector of the same length as ``a``.
    """
    conv = np.convolve(a, a)
    m = conv.max()
    # soft‑max weights, shifted by the maximum for numerical stability
    exps = np.exp(lam * (conv - m))
    p = exps / exps.sum()
    # d/d a_i of Σ_k p_k * Σ_j a_j a_{k-j}
    # = 2 * Σ_k p_k a_{k-i}  → correlation of p with a
    corr = np.correlate(p, a, mode="full")
    start = len(p) - 1
    grad = 2.0 * corr[start : start + len(a)]
    return grad

        if src == tgt or cur[src] <= 0:
            src, tgt = rng.choice(n, size=2, replace=False)

        frac = 0.07 * max(0.0, 1.0 - it / max_iters) + 0.02
        step = cur[src] * rng.random() * frac
        if step <= 0:
            continue

        new = cur.copy()
        new[src] -= step
        new[tgt] += step
        # re‑normalise (cheap O(n))
        new /= new.sum()
        new_conv = np.convolve(new, new)
        new_score = 2.0 * n * new_conv.max() / (new.sum() ** 2)

        if new_score < cur_score - 1e-12:
            # sanity check with the official evaluator
            real = evaluate_sequence(new.tolist())
            if real < cur_score - 1e-12:
                cur = new
                cur_conv = new_conv
                cur_score = real
                if real < best_score - 1e-12:
                    best_score = real
                    best_seq = new.tolist()
                    print(f"[refine] iter={it} new best={real:.6f}")
        if it % 1000 == 0:
            # occasional re‑projection to avoid drift
            cur = _project_to_simplex(cur)
            cur_conv = np.convolve(cur, cur)

    print(f"[final] best score found: {best_score:.6f}")
    return best_seq
