"""
Propose a step‑function (non‑negative list of floats) that minimises
``evaluate_sequence``.  The optimiser combines three complementary
techniques:

1. **Exponentiated‑gradient descent on a smoothed max‑convolution**.
   We approximate ``max(conv(a,a))`` by a log‑sum‑exp with a
   temperature β and perform a mirror‑descent update
   a_i ← a_i·exp(−η·∇_i).  After each update the vector is renormalised
   onto the simplex, which guarantees non‑negativity and sum‑one.

2. **Targeted mass transfers**.
   In every few iterations we locate the index k of the current maximal
   convolution entry and move a fraction of the mass belonging to the
   pair (i, k−i) that contributes most to this peak towards a position
   whose partner for the same offset is (almost) zero.  This explicitly
   lowers the dominant peak.

3. **Exploratory random transfers**.
   Small stochastic moves (moving a random fraction of one coordinate to
   another) together with a simple Metropolis acceptance rule keep the
   search from getting stuck.

The routine respects the 1000 s budget, prints progress and finally
returns the best sequence found (as a plain list of floats).  All
operations stay on the probability simplex, so the scaling invariance of
the original bound is respected.
"""

import time
import math
import numpy as np

# -------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _normalize(v: np.ndarray) -> np.ndarray:
    """Clamp to ≥0 and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        # fallback to uniform distribution
        n = v.shape[0]
        return np.full_like(v, 1.0 / n)
    return v / s


def _objective(a: np.ndarray) -> float:
    """
    Fast objective used during the search:
        O = 2·n·max(conv(a,a))   (the sum of a is 1).
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    return 2.0 * n * conv.max()


# -------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smoothed objective
        f(a) = (1/β)·log Σ_k exp(β·conv_k(a))
    where conv = a * a (discrete convolution).
    The formula can be written using two ordinary convolutions.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)             # length 2n‑1
    max_c = conv.max()
    # log‑sum‑exp stabilisation
    w = np.exp(beta * (conv - max_c))
    p = w / w.sum()                      # a probability vector on the offsets

    # convolution of a with p  (for the a_{i‑k} part)
    conv2 = np.convolve(a, p)            # length 3n‑2
    # convolution of a with reversed p (for the a_{k‑i} part)
    conv_rev = np.convolve(a, p[::-1])   # length 3n‑2

    # extract the two needed slices
    # term1_i = Σ_j a_j·p_{i+j} = conv_rev[ n‑1 : 2n‑1 ][::-1]
    term1 = conv_rev[n - 1 : 2 * n - 1][::-1]
    # term2_i = Σ_j a_j·p_{i‑j} = conv2[0:n]
    term2 = conv2[:n]

    grad = term1 + term2                 # ∇ f(a) (no extra factor of 2 needed)
    return grad

def _targeted_move(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Perform a deterministic move that directly reduces the current maximal
    convolution entry.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    k = int(conv.argmax())               # offset with maximal value
    # indices that actually contribute to this offset
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    idx = np.arange(i_start, i_end + 1, dtype=int)
    jdx = k - idx
    contrib = a[idx] * a[jdx]

    # pick the index of the larger member of the most harmful pair
    src = idx[np.argmax(contrib)]
    src_comp = k - src                    # the partner index in the max‑offset pair

    # choose a target where the partner for offset k is (almost) zero
    # partner weight for a candidate t is a[k‑t] if the index exists
    partner = np.full(n, np.inf, dtype=float)
    valid = (k - np.arange(n) >= 0) & (k - np.arange(n) < n)
    partner[valid] = a[k - np.arange(n)[valid]]

    # forbid moving mass back to the offending pair
    mask = np.ones(n, dtype=bool)
    mask[src] = False
    if 0 <= src_comp < n:
        mask[src_comp] = False

    cand = np.where(mask)[0]
    if cand.size == 0:
        # fall back to a completely random destination
        tgt = rng.integers(n)
    else:
        # we would like a partner weight as small as possible (ideally zero)
        # ``partner`` may contain ``inf`` for out‑of‑range indices – treat them as best.
        # Replace ``inf`` by a large finite number for argmin.
        partner_c = partner[cand].copy()
        inf_mask = np.isinf(partner_c)
        if inf_mask.any():
            # these are the best candidates (no partner at all)
            tgt = cand[inf_mask][rng.integers(inf_mask.sum())]
        else:
            tgt = cand[np.argmin(partner_c)]

    # move a moderate fraction of the source mass to the target
    if a[src] == 0.0:
        return a
    frac = 0.30  # 30 % of the source weight
    delta = a[src] * frac
    a_new = a.copy()
    a_new[src] -= delta
    a_new[tgt] += delta
    return _normalize(a_new)


# -------------------------------------------------------------
def propose_candidate(seed: int = 2024, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the bound
    returned by ``evaluate_sequence``.  Returns the sequence as a list of
    floats.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # -----------------------------------------------------------------
    # Length of the sequence – a few thousand gives a good trade‑off.
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=float)
        a = _normalize(a)
        n = a.size
    else:
        n = 5000
        a = np.full(n, 1.0 / n, dtype=float)   # start from uniform
        # break symmetry a little
        a += rng.normal(scale=1e-4, size=n)
        a = _normalize(a)

    best_seq = a.copy()
    best_score = _objective(best_seq)
    print(f"[init] n={n}, objective={best_score:.6f}")

    # -----------------------------------------------------------------
    # Hyper‑parameters
    beta = 200.0                # temperature for the smooth max (log‑sum‑exp)
    eta0 = 1.0                  # initial step‑size for the exponentiated update
    T0 = 0.02                   # temperature for the Metropolis rule (random moves)
    max_iters = 300_000        # hard cap on the number of iterations
    no_improve = 0
    max_no_improve = 10_000

    iter_cnt = 0
    cur_score = best_score
    cur_a = a.copy()
    last_restart = 0

    while time.time() < deadline and iter_cnt < max_iters:
        iter_cnt += 1

        # -------------------------------------------------------------
        # Choose an operation
        op = rng.random()
        if op < 0.10:
            # exponentiated‑gradient descent on the smoothed objective
            grad = _smooth_gradient(cur_a, beta)
            eta = eta0 / math.sqrt(iter_cnt)
            cur_a = cur_a * np.exp(-eta * grad)
            cur_a = _normalize(cur_a)
        elif op < 0.30:
            # a targeted move that directly attacks the current peak
            cur_a = _targeted_move(cur_a, rng)
        else:
            # generic stochastic mass transfer (Metropolis acceptance)
            i, j = rng.integers(n, size=2)
            if i != j and cur_a[i] > 0.0:
                # move a random fraction (up to 20 %) of the source mass
                frac = rng.random() * 0.2
                delta = cur_a[i] * frac
                a_new = cur_a.copy()
                a_new[i] -= delta
                a_new[j] += delta
                a_new = _normalize(a_new)

                new_score = _objective(a_new)

                # Metropolis style acceptance
                T = T0 * (1.0 - iter_cnt / max_iters)
                if new_score < cur_score or rng.random() < math.exp((cur_score - new_score) / max(T, 1e-12)):
                    cur_a = a_new
                    cur_score = new_score
                    no_improve = 0
                else:
                    no_improve += 1
                # early update of the best‑so‑far
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = cur_a.copy()
                    print(f"[iter {iter_cnt}] new global best O={best_score:.6f}")

                # skip the rest of the loop body – we already handled the scores
                continue

        # -------------------------------------------------------------
        # After a deterministic move we recompute the exact objective
        cur_score = _objective(cur_a)

        if cur_score < best_score:
            best_score = cur_score
            best_seq = cur_a.copy()
            print(f"[iter {iter_cnt}] new global best O={best_score:.6f}")
            no_improve = 0
        else:
            no_improve += 1

        # -------------------------------------------------------------
        # Periodic restart if we have been stuck for a while
        if no_improve >= max_no_improve:
            # re‑initialise from the best solution with a tiny random kick
            cur_a = best_seq.copy()
            cur_a += rng.normal(scale=5e-4, size=n)
            cur_a = _normalize(cur_a)
            cur_score = _objective(cur_a)
            no_improve = 0
            last_restart = iter_cnt
            print(f"[restart] iter={iter_cnt}, re‑initialised from best (score={best_score:.6f})")

    # -----------------------------------------------------------------
    # Final evaluation with the provided ``evaluate_sequence`` function
    final_seq = best_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:  # pragma: no cover
        print("[final] evaluation failed:", e)

    return final_seq
