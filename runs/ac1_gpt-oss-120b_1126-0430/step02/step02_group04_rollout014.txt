def propose_candidate(seed=0, budget_s=1000):
    import time, numpy as np
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5
    n = 8000

    # initial candidates: previous best if exists, arcsin shape, sqrt shape, random
    candidates = []
    if 'height_sequence_1' in globals():
        a_prev = np.array(height_sequence_1, dtype=float)
        if a_prev.size == n:
            candidates.append(_proj_simplex(a_prev))
        else:
            # maybe interpolate/resample?
            a_resampled = np.interp(np.linspace(0, len(a_prev)-1, n), np.arange(len(a_prev)), a_prev)
            candidates.append(_proj_simplex(a_resampled))

    # arcsin shape: f_i ~ 1/ sqrt(x_i * (1-x_i)), normalized
    x = np.arange(n)/ (n-1)
    eps = 1e-12
    arcs = 1.0/ np.sqrt(np.maximum(x* (1-x), eps))
    arcs = _proj_simplex(arcs)
    candidates.append(arcs)

    # sqrt shape (semi-circular)
    s = np.sqrt(x * (1-x))
    s = _proj_simplex(s)
    candidates.append(s)

    # random simplex
    cand = np.random.rand(n)
    candidates.append(_proj_simplex(cand))

    # choose best initial candidate by exact objective
    best_seq = None
    best_score = np.inf
    for a in candidates:
        score = _eval_score(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
    a = best_seq.copy()
    cur_score = best_score
    best_seq = a.copy()
    best_score = cur_score

    # optimization parameters
    step = 0.2  # initial step size
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)  # first moment
    v = np.zeros_like(a)  # second moment
    t = 0
    no_improve = 0

    while time.time() < deadline:
        # compute convolution and max
        b = _convolve_fft(a, a)
        maxb = b.max()
        # compute mask of max entries (using tolerance)
        mask = (b >= maxb - 1e-12).astype(float)
        # gradient of max (subgradient)
        grad = 2.0 * _convolve_fft(a, mask[::-1])
        # because convolution length is n + n - 1 = 2n-1, we need proper slice as before
        idx = (len(b)-1) - np.arange(n)
        grad = grad[idx]

        # Adam update
        t += 1
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * (grad*grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step_size = step / (np.sqrt(v_hat) + eps)
        a_candidate = a - step_size * m_hat

        # enforce nonnegativity & simplex + symmetry
        a_candidate = np.maximum(a_candidate, 0.0)
        # enforce symmetry: average with reversed
        a_candidate = (a_candidate + a_candidate[::-1]) / 2.0
        a_candidate = _proj_simplex(a_candidate)

        cand_score = _eval_score(a_candidate)

        if cand_score < cur_score:
            a = a_candidate
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = a.copy()
                print(f"[iter {t}] new best {best_score:.6f}")
                no_improve = 0
            else:
                no_improve += 1
        else:
            # if no improvement, maybe accept occasional uphill with probability
            no_improve += 1
            # also adjust step
            step *= 0.95

        # restart if stuck long
        if no_improve > 5000:
            # perturb best
            a = best_seq + 0.001*np.random.randn(n)
            a = np.maximum(a, 0)
            a = (a + a[::-1]) / 2.0

            cur_score = _eval_score(a)
            m = np.zeros_like(a)
            v = np.zeros_like(a)
            t = 0
            step = 0.2
            no_improve = 0
            print(f"[restart] new starting score {cur_score:.6f}")

    print(f"Final best score {best_score:.6f}")
    return best_seq.tolist()

"""
Propose a step‑function (list of non‑negative heights) that minimises
``evaluate_sequence``.  The objective is   2·n·max(a∗a) / (∑a)² .
It is scale‑invariant, so we keep every candidate on the probability simplex
(∑a = 1, a_i ≥ 0).

The optimiser is a simple projected gradient method with a *exact* sub‑gradient
of the max‑convolution (instead of the smooth‑max used previously).  The
gradient is
    g = 2·conv(a, M_rev)[(len(b)‑1)‑i] ,
where ``M`` is the indicator of the maximal entries of the convolution
``b = a∗a``.  After each gradient step we

* project onto the simplex (Euclidean projection – Duchi et al. 2008),
* optionally perform a back‑tracking line search guaranteeing monotone
  improvement,
* keep the best sequence seen so far.

To avoid getting trapped in a local minimum we periodically
*restart* the search by adding a tiny random perturbation around the best
solution found so far.  Several simple initialisations are tried
(previous best, Catalan‑like, arcsine, semi‑circular and a random point);
the one with the lowest exact objective starts the optimisation.

The whole routine respects the 1000 s time budget, prints progress and
returns the best sequence as a plain ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, sum=1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u > cssv / np.arange(1, n + 1))[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of real vectors using a real FFT."""
    n = x.size + y.size - 1
    size = 1 << (n - 1).bit_length()      # next power of two
    X = np.fft.rfft(x, size)
    Y = np.fft.rfft(y, size)
    Z = X * Y
    conv = np.fft.irfft(Z, size)[:n]
    return conv.real


def _eval_score(a: np.ndarray) -> float:
    """Exact evaluation of ``evaluate_sequence`` for a non‑negative simplex vector."""
    n = a.size
    b = _convolve_fft(a, a)
    max_b = b.max()
    return float(2.0 * n * max_b / (a.sum() ** 2))


def _catalan_sequence(n: int) -> np.ndarray:
    """Catalan‑like seed a_i = C(2i,i)/4**i, normalised to sum 1."""
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * (i - 1) + 1) / (2.0 * i)
    a /= a.sum()
    return a


def _arcsine_sequence(n: int) -> np.ndarray:
    """Beta(0.5,0.5) – 1/√(x(1‑x)) – normalised to sum 1."""
    x = np.linspace(0.0, 1.0, n)
    eps = 1e-12
    a = 1.0 / np.sqrt(np.maximum(x * (1.0 - x), eps))
    a = _proj_simplex(a)
    return a


def _semicircle_sequence(n: int) -> np.ndarray:
    """Semi‑circular shape sqrt(x(1‑x)), normalised to sum 1."""
    x = np.linspace(0.0, 1.0, n)
    a = np.sqrt(x * (1.0 - x))
    a = _proj_simplex(a)
    return a


def _geometric_sequence(n: int, rate: float) -> np.ndarray:
    """Geometric decay a_i ∝ r**i, normalised to sum 1."""
    i = np.arange(n, dtype=float)
    a = rate ** i
    a = _proj_simplex(a)
    return a


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a non‑negative height sequence that minimises ``evaluate_sequence``.
    Returns the best sequence found before the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0    # small safety margin

    # -------------------------------------------------------------------
    # problem size – if a previous solution exists we adopt its length,
    # otherwise we use the default size 8000 (fast enough for FFTs).
    # -------------------------------------------------------------------
    default_n = 8000
    if 'height_sequence_1' in globals():
        prev = np.asarray(globals()['height_sequence_1'], dtype=float)
        n = max(default_n, prev.size)           # make sure we have enough points
    else:
        n = default_n

    # -------------------------------------------------------------------
    # build a small pool of starting points
    # -------------------------------------------------------------------
    candidates = [_catalan_sequence(n), _arcsine_sequence(n), _semicircle_sequence(n)]

    # several geometric decays
    for rate in np.linspace(0.95, 0.30, 6):
        candidates.append(_geometric_sequence(n, rate))

    # a random simplex point
    rnd = np.random.rand(n)
    candidates.append(_proj_simplex(rnd))

    # if we have a previous solution, resample it to the current length
    if 'height_sequence_1' in globals():
        prev = np.asarray(globals()['height_sequence_1'], dtype=float)
        if prev.size != n:
            # linear interpolation to desired length
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            prev_resampled = np.interp(x_new, x_old, prev)
            candidates.append(_proj_simplex(prev_resampled))
        else:
            candidates.append(_proj_simplex(prev))

    # -------------------------------------------------------------------
    # choose the best initial candidate according to the exact objective
    # -------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    for a in candidates:
        sc = _eval_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
    a = best_seq.copy()
    cur_score = best_score
    print(f"[init] n={n}, initial score={cur_score:.6f}")

    # -------------------------------------------------------------------
    # optimisation hyper‑parameters
    # -------------------------------------------------------------------
    step = 0.4                 # initial step size (will be reduced if needed)
    beta = 0.9                 # momentum coefficient
    momentum = np.zeros_like(a)
    iteration = 0
    no_improve = 0
    max_no_improve = 5000

    # -------------------------------------------------------------------
    while time.time() < deadline:
        # --- convolution and maximal entry ---
        b = _convolve_fft(a, a)
        max_b = b.max()

        # --- sub‑gradient of the max‑convolution (exact) ---
        # mask of all positions attaining the maximum (within tiny tol)
        tol = max_b * 1e-12
        mask = (b >= max_b - tol).astype(float)
        # conv of a with the reversed mask
        conv_full = _convolve_fft(a, mask[::-1])
        idx = (len(b) - 1) - np.arange(n)   # slice that yields the correct gradient
        grad = 2.0 * conv_full[idx]

        # --- momentum update ---
        momentum = beta * momentum + (1.0 - beta) * grad

        # --- back‑tracking line search (guaranteed monotone improvement) ---
        improved = False
        local_step = step
        while local_step > 1e-9:
            a_candidate = a - local_step * momentum
            a_candidate = np.maximum(a_candidate, 0.0)        # non‑negativity
            a_candidate = _proj_simplex(a_candidate)         # project onto simplex
            cand_score = _eval_score(a_candidate)
            if cand_score < cur_score:
                a = a_candidate
                cur_score = cand_score
                improved = True
                break
            local_step *= 0.5

        # --- bookkeeping -------------------------------------------------
        if improved:
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{iteration:06d}] new best {best_score:.6f} (step={local_step:.3e})")
                no_improve = 0
            else:
                no_improve += 1
        else:
            # no improvement – shrink the step size a little
            step *= 0.8
            no_improve += 1

        # occasional global decay of step size
        if iteration % 500 == 0 and iteration > 0:
            step = max(step * 0.9, 1e-5)

        # restart if we have been stuck for a while
        if no_improve > max_no_improve:
            a = best_seq + 0.005 * np.random.randn(n)
            a = np.maximum(a, 0.0)
            a = _proj_simplex(a)
            cur_score = _eval_score(a)
            momentum.fill(0.0)
            step = 0.4
            no_improve = 0
            print(f"[{iteration:06d}] restart around best (score={cur_score:.6f})")

        iteration += 1

    print(f"[done] best score found = {best_score:.6f}")
    # return plain list as required by the harness
    return best_seq.tolist()
