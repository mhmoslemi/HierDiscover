"""
Propose a non‑negative step‑function sequence that minimizes the bound

    O = 2·n·max(conv(a,a))   (with sum(a)=1),

by combining:
  1. Analytic “U‑shaped’’ initialisations (inverse power, exponential,
     triangular, uniform) on several lengths.
  2. Projected sub‑gradient descent on the non‑smooth objective
     max(conv) (the sub‑gradient is computed analytically).
  3. A short random hill‑climbing phase after each descent.
  4. Many random restarts (different lengths, Dirichlet initialisations)
     until the time budget (≈1000 s) is exhausted.

All intermediate vectors are kept on the probability simplex using the
Duchi‑et‑al. projection.  The best sequence found is finally evaluated
with the provided ``evaluate_sequence`` function.
"""

import time
import math
import numpy as np

# -------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x ≥ 0, sum(x)=1} (Duchi et al.)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# -------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a with sum(a)=1."""
    n = a.size
    conv = np.convolve(a, a)
    return 2.0 * n * conv.max()


# -------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) while a lies on the simplex.
    Returns a vector of the same shape as a.
    """
    n = a.size
    conv = np.convolve(a, a)
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)      # all arg‑max indices
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]            # contribution from each pair
    g /= max(1, len(ks))                     # average if several maxima
    return g


# -------------------------------------------------------------
# Analytic shape generators (all return non‑negative arrays, not yet normalised)
def shape_inverse_pow(n: int, p: float) -> np.ndarray:
    """U‑shaped profile |i‑mid|^p, zero at the centre, maximal at the ends."""
    i = np.arange(n, dtype=float)
    mid = (n - 1) / 2.0
    w = (np.abs(i - mid) / mid) ** p
    return w


def shape_exponential(n: int, lam: float = 0.01) -> np.ndarray:
    """Exponential decay from the centre to the ends."""
    i = np.arange(n, dtype=float)
    mid = (n - 1) / 2.0
    w = np.exp(-lam * np.abs(i - mid))
    return w


def shape_triangular(n: int) -> np.ndarray:
    """Standard triangular (peak in the centre, zero at the ends)."""
    i = np.arange(n, dtype=float)
    mid = (n - 1) / 2.0
    w = 1.0 - np.abs(i - mid) / mid
    w = np.maximum(w, 0.0)
    return w


def shape_uniform(n: int) -> np.ndarray:
    """Uniform vector."""
    return np.ones(n, dtype=float)

    # Choose a set of lengths to explore.

    if 'height_sequence_1' in globals():
        length_options.append(len(height_sequence_1))
    length_options = sorted(set(length_options))

    best_seq = None
    best_score = float('inf')

    # ---------------------------------------------------------
    # 1️⃣   Analytic initial candidates (U‑shapes, exponential, uniform)
    print("[phase] evaluating analytic prototypes")
    for n in length_options:
        # uniform
        a = shape_uniform(n)
        a /= a.sum()
        score = _objective(a)
        if score < best_score:
            best_score, best_seq = score, a.copy()
            print(f"[init] uniform n={n}  O={score:.6f}")

        # triangular (centre‑peak) – usually worse, but kept for completeness
        a = shape_triangular(n)
        if a.sum() > 0:
            a /= a.sum()
            score = _objective(a)
            if score < best_score:
                best_score, best_seq = score, a.copy()
                print(f"[init] triangular n={n}  O={score:.6f}")

        # inverse‑power “U” shapes, several exponents
        for p in (0.5, 1.0, 2.0, 3.0):
            a = shape_inverse_pow(n, p)
            if a.sum() == 0:

            a /= a.sum()
            score = _objective(a)

        # exponential decay from centre

        if score < best_score:

            print(f"[init] exponential n={n}  O={score:.6f}")

    # Helper to update the global best (no closure – done inline)
    # ---------------------------------------------------------
    # 2️⃣   Random restarts with projected sub‑gradient descent + hill‑climbing
    print("[phase] stochastic optimisation")
    while time.time() < deadline:

        n = int(rng.choice(length_options))

            a = np.clip(a, 0.0, None)
