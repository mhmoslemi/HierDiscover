"""
Projected gradient descent on a smooth approximation of the
max‑convolution objective.

The objective to minimise (for a unit‑sum non‑negative sequence a)
    O(a) = 2·n·max_k (a * a)[k]

where * denotes the linear convolution.  The max is replaced by a
log‑sum‑exp smoothing, whose gradient can be written using a single
correlation.  Each iteration:

1. Compute the full convolution conv = a * a.
2. Form a soft‑max weight vector w that concentrates on the
   indices where conv is large.
3. Gradient of the smoothed objective is
       g = 2 * correlate(a, w)            (slice of length n)
4. Take a step a ← a – η·g, then project back onto the probability
   simplex (non‑negative, sum = 1).
5. Accept the step if the true objective improves, otherwise shrink η.
6. Occasionally inject a tiny random perturbation to escape plateaus.

The routine runs for up to the supplied time budget (default 1000 s),
keeps the best sequence ever seen and finally returns it as a plain
list of floats.
"""

import time
import math
import numpy as np

# ---------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ---------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a already normalised to sum=1."""
    n = a.size
    conv = np.convolve(a, a)
    return 2.0 * n * np.max(conv)

    used by `evaluate_sequence`.  Returns the best sequence as a list
    of floats.
    """

    a += np.random.randn(a.size) * 1e-3

    n = a.size

    best_score = _objective(best_seq)
    cur_score = best_score

    print(f"[init] n={n}, objective={best_score:.6f}")

    # -----------------------------------------------------------------
    # Hyper‑parameters (they evolve during the run)
    step = 0.2                     # initial step size for gradient descent
    step_min = 1e-5
    eps_factor = 0.25              # smoothing = eps_factor * max(conv)
    eps_min = 1e-9

    iter_no_improve = 0
    max_no_improve = 3000
    iteration = 0

    while time.time() < deadline and iteration < 200_000:
        iteration += 1

        # ---- current objective ------------------------------------------------
        conv = np.convolve(a, a)

            best_seq = a.copy()
