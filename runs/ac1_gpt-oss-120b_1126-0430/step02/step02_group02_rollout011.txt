#!/usr/bin/env python
"""
Propose a step‑function (list of non‑negative heights) that minimizes the
evaluation ratio used in the challenge.

The algorithm proceeds in three stages:

1. **Initialisation** – a power‑law vector (a_i ∝ (i+1)^‑p) is scanned on a
   fine grid and the best exponent is kept (sum normalised to 1).

2. **Linear‑programming redistribution** – given the current vector f,
   we compute M = max_k (f*f)[k].  A linear program
       maximise   Σ b_j   subject to   (f*b)[k] ≤ M   (∀k) ,  b_j ≥ 0
   is solved.  The optimal b is scaled to Σ b = 1 giving a new candidate g.
   If g improves the objective we replace f by g; otherwise we mix
   f←(1‑t)f+t g (t≈0.5).  Several outer LP iterations are performed.

3. **Projected sub‑gradient descent** – after each LP step (and finally
   until the time budget expires) we apply a simple projected sub‑gradient
   descent on the non‑convex objective  max_k (a*a)[k] under the simplex
   constraint Σ a = 1, a ≥ 0.  The best sequence encountered is kept.

All heavy work uses NumPy; the LP uses SciPy’s HiGHS solver with a sparse
Toeplitz constraint matrix, which is fast for n≈2000.  The routine respects
the 1000 s budget, prints progress, and returns the best sequence found.
"""

import time
import math
import numpy as np

# Optional SciPy imports (LP support)
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog
    from scipy.sparse import diags
    _HAVE_SCIPY = True
except Exception:                     # pragma: no cover
    _HAVE_SCIPY = False

# Helper utilities – top‑level, no closures

def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (sum = 1, v ≥ 0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float) -> np.ndarray:
    """Return a normalised power‑law vector a_i ∝ (i+1)^‑p (so Σ a_i = 1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p)
    a /= a.sum()
    return a


def _cheap_score(a: np.ndarray) -> float:
    """
    Fast evaluation of the objective for a vector with Σ a = 1.
    Returns 2·n·max_k (a*a)[k].
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    return 2.0 * n * conv.max()


def _downsample(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linearly interpolate a longer sequence to a shorter length."""
    old_len = len(seq)
    if new_len >= old_len:
        return seq.copy()
    x_old = np.arange(old_len)
    x_new = np.linspace(0, old_len - 1, new_len)
    return np.interp(x_new, x_old, seq)


def _build_conv_matrix(f: np.ndarray):
    """
    Build the (2n‑1)×n sparse matrix A that represents the convolution
    constraints (f * b)[k] = Σ_i f_i b_{k‑i}.
    The k‑th row of A contains f_{k‑j} on column j (where defined).
    """
    n = f.shape[0]
    offsets = np.arange(n)                     # 0 … n‑1
    # each diagonal d contains the constant f[d]
    data = [np.full(n, f[d]) for d in range(n)]
    A = diags(data, offsets, shape=(2 * n - 1, n), format='csr')
    return A


def _solve_lp_max_sum(f: np.ndarray, M: float):
    """
    Solve   maximise Σ b_j   subject to   (f * b)[k] ≤ M   (∀k),  b≥0.
    Returns the optimal b (array of length n) or None on failure.
    """
    if not _HAVE_SCIPY:               # pragma: no cover
        return None
    n = f.shape[0]
    A = _build_conv_matrix(f)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # maximise → minimise –cᵀb
    bounds = [(0, None)] * n
    # A tiny tolerance avoids numerical infeasibility
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def _subgradient_step(a: np.ndarray, eta: float) -> np.ndarray:
    """
    One projected sub‑gradient step for the objective max_k (a*a)[k].
    Returns the projected new vector (still on the simplex).
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    k = int(conv.argmax())
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    idx = np.arange(lo, hi + 1, dtype=np.int64)
    g = np.zeros_like(a)
    g[idx] = 2.0 * a[k - idx]
    a_next = a - eta * g
    return _simplex_projection(a_next)

    while time.time() < deadline and sub_iter < 200_000:
        sub_iter += 1
        eta = eta0 / math.sqrt(sub_iter)
        f_candidate = _subgradient_step(f, eta)
        cand_score = _cheap_score(f_candidate)
        if cand_score < best_score - 1e-9:
            best_score = cand_score
            best_seq = f_candidate.copy()
            print(f"[Final sub {sub_iter}] NEW BEST cheap score = {best_score:.6f}")
        if cand_score < _cheap_score(f) - 1e-12:
            f = f_candidate
        if sub_iter % 1000 == 0 and np.random.rand() < 0.05:
            perturb = 0.02 * np.random.rand(n)
            f = _simplex_projection(f + perturb)

    # --------------------------------------------------------------
    # 5. Exact evaluation and return.
    # --------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:     # defensive – should never happen
        final_score = float('inf')
        print(f"Error during final evaluation: {exc}")

    return final_seq
