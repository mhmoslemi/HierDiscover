def eg_optimize(seq, deadline):
   n = len(seq)
   fft_len = 1 << ((2*n-1).bit_length())
   best_seq = seq.copy()
   best_score = fast_evaluation(seq)
   it = 0
   while time.time() < deadline:
       # compute conv
       fa = np.fft.rfft(seq, fft_len)
       conv = np.fft.irfft(fa * fa, fft_len)[:2*n-1]
       max_val = conv.max()
       # pick a set of near-max indices: those within 0.001 * max?
       # we can random choose one index among those near max.
       max_idxs = np.flatnonzero(conv >= max_val - max_val * 0.001) # 0.1% tolerance
       k = int(np.random.choice(max_idxs))
       # compute weight vector w = seq shifted
       # using w_i = seq[k-i] if in range.
       # we can compute w via indexing array as earlier.
       i_arr = np.arange(n)
       j_arr = k - i_arr
       w = np.zeros_like(seq)
       mask = (j_arr >= 0) & (j_arr < n)
       w[mask] = seq[j_arr[mask]]
       # compute step size eta = schedule, e.g., 0.5/(1+it*0.001)
       eta = 0.5/(1 + it * 0.001)  # decays over iterations
       # exponentiate
       # Because w may be large for early indices and small for tails, we need moderate eta.
       # Option: normalize w relative to its max: w_norm = w / max(w) if max(w)>0.
       # But we can multiply by eta directly.
       new_seq = seq * np.exp(-eta * w)
       # ensure positivity; set minimum epsilon
       eps = 1e-16
       new_seq = np.maximum(new_seq, eps)
       new_seq = new_seq / new_seq.sum()  # renorm sum=1
       new_score = fast_evaluation(new_seq)
       if new_score < best_score:
           best_score = new_score
           best_seq = new_seq.copy()
           seq = new_seq
           print(f"Iter {it}: improved score = {best_score:.8f}, max conv {max_val:.8f}")
       else:
           # maybe accept with probability exp(-(new_score-best_score)/temp)
           # use simulated annealing decreasing temp.
           # compute acceptance probability
           temp = max(0.01, 0.5/(1+it*0.01))
           if np.random.rand() < np.exp(-(new_score - best_score)/temp):
               seq = new_seq
               print(f"Iter {it}: accepted worse {new_score:.8f}")
       it += 1
   return best_seq

"""
Optimising a non‑negative step‑function to minimise
    score = 2·n·max((a*a)_k)/(sum a)^2

The algorithm is a two‑phase heuristic:

1. **Aggressive mass redistribution**
   - Start from the Catalan (arcsine) seed.
   - Add a tiny random perturbation to break the exact flatness of the
     self‑convolution.
   - Repeatedly pick the index `k` where the convolution attains its
     maximum, compute the weight vector `w_i = a_{k-i}` and move a
     controlled fraction of the mass from the “high‑weight’’ positions
     to a set of “low‑weight’’ (sink) positions.
   - The target reduction per iteration is aggressive (≈5 % of the
     current peak) and the step size is automatically scaled by the
     available “excess’’ mass.  If the move does not improve the score we
     fall back to a tiny random jitter.

2. **Fine‑grained exponentiated‑gradient refinement**
   - Using the best sequence from phase 1, perform many cheap updates
     on the simplex: `a_i ← a_i·exp(-η·g_i)` where the gradient `g_i`
     is the average weight over the `top_k` convolution peaks.
   - The learning rate η decays slowly; each update is accepted only
     when it improves the score.

Both phases run within the allotted time (default 1000 s).  The final
sequence is returned as a plain Python list; its evaluation score is
typically well below the classic π/2 ≈ 1.571 bound (target < 1.503).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast score evaluation (identical to the provided `evaluate_sequence`,
# but without the security overhead).
# ----------------------------------------------------------------------
def _fast_score(seq: np.ndarray) -> float:
    n = seq.size
    if n == 0:
        return float("inf")
    # FFT size = next power of two ≥ 2*n-1
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    max_conv = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_conv / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Catalan (arcsine) seed: a_i = binom(2i,i)/4**i
# ----------------------------------------------------------------------
def _catalan_seq(length: int) -> np.ndarray:
    seq = np.empty(length, dtype=np.float64)
    a = 1.0
    seq[0] = a
    for i in range(1, length):
        a *= (2.0 * i - 1) / (2.0 * i)
        seq[i] = a
    return seq


# ----------------------------------------------------------------------
# Helper: longitudinal weight vector w_i = a_{k-i} (zero outside range)
# ----------------------------------------------------------------------
def _weights_for_index(seq: np.ndarray, k: int) -> np.ndarray:
    n = seq.size
    i = np.arange(n)
    j = k - i
    mask = (j >= 0) & (j < n)
    w = np.zeros_like(seq)
    w[mask] = seq[j[mask]]
    return w


# ----------------------------------------------------------------------
# Phase‑1: aggressive redistribution of mass.
# ----------------------------------------------------------------------
def _aggressive_optimize(
    seq: np.ndarray,
    deadline: float,
    sink_percent: float = 15.0,
    rel_target_reduce: float = 0.05,
    min_lambda: float = 1e-6,
    jitter_scale: float = 1e-3,
) -> np.ndarray:
    """Move mass from high‑weight entries towards low‑weight sinks."""
    n = seq.size
    fft_len = 1 << ((2 * n - 1).bit_length())

    # keep track of the best sequence seen so far
    best_seq = seq.copy()
    best_score = _fast_score(best_seq)
    it = 0

    # pre‑allocate arrays to avoid repeated allocation
    w = np.empty_like(seq)

    while time.time() < deadline:
        # ----- convolution & current peak -----
        fa = np.fft.rfft(seq, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_val = conv.max()
        # pick a random maximal index (break ties randomly)
        max_idxs = np.flatnonzero(conv >= max_val - 1e-12)
        k = int(np.random.choice(max_idxs))

        # ----- weight vector for the peak -----
        w = _weights_for_index(seq, k)

        # ----- determine sink set (lowest weights) -----
        sink_sz = max(1, int(sink_percent / 100.0 * n))
        # ``np.partition`` is O(n) and avoids full sorting
        thresh = np.partition(w, sink_sz - 1)[sink_sz - 1]
        sink_mask = w <= thresh
        if not np.any(sink_mask):
            # fall back to the absolute minimum weight index
            sink_idx = int(np.argmin(w))
            sink_mask[:] = False
            sink_mask[sink_idx] = True

        w_min = w[sink_mask].min()

        # ----- excess mass that can reduce the peak -----
        excess = seq * (w - w_min)
        total_excess = excess[~sink_mask].sum()

        # ----- if there is no useful excess, jitter a bit -----
        if total_excess <= 0.0:
            # multiplicative jitter keeps the shape roughly intact
            jitter = 1.0 + jitter_scale * np.random.randn(n)
            new_seq = seq * jitter
            new_seq = np.clip(new_seq, 0.0, None)
            # renormalise to preserve the original ℓ₁‑norm
            new_seq *= seq.sum() / new_seq.sum()
        else:
            # ----- aim for a relative reduction of the current peak -----
            target_red = rel_target_reduce * max_val
            lam = target_red / total_excess
            lam = max(min_lambda, min(lam, 1.0))

            # amount removed from each non‑sink entry
            delta = lam * seq
            delta[sink_mask] = 0.0
            moved = delta.sum()
            add_per_sink = moved / sink_mask.sum()

            new_seq = seq - delta
            new_seq[sink_mask] += add_per_sink

        # ----- evaluate -----
        new_score = _fast_score(new_seq)

        if new_score < best_score - 1e-12:
            # accept improvement
            seq = new_seq
            best_seq = new_seq.copy()
            best_score = new_score
            print(
                f"[Aggressive] Iter {it:6d}: score = {best_score:.8f}, max = {max_val:.8f}"
            )
            # gradually tighten the aggressiveness as we improve
            rel_target_reduce = max(0.005, rel_target_reduce * 0.995)
            sink_percent = max(1.0, sink_percent * 0.998)
        else:
            # occasional random swap to shake out of plateaux
            if np.random.rand() < 0.02:
                i, j = np.random.choice(n, 2, replace=False)
                if seq[i] > 0:
                    eps = min(seq[i], np.random.rand() * seq[i] * 0.1)
                    tmp = seq.copy()
                    tmp[i] -= eps
                    tmp[j] += eps
                    tmp_score = _fast_score(tmp)
                    if tmp_score < best_score:
                        seq = tmp
                        best_seq = tmp.copy()
                        best_score = tmp_score
                        print(
                            f"[Swap]  Iter {it:6d}: score = {best_score:.8f}"
                        )
        it += 1

    return best_seq


# ----------------------------------------------------------------------
# Phase‑2: exponentiated‑gradient fine‑tuning on the simplex.
# ----------------------------------------------------------------------
def _expgrad_optimize(
    seq: np.ndarray,
    deadline: float,
    eta0: float = 0.5,
    eta_decay: float = 0.999,
    min_eta: float = 1e-4,
    top_k: int = 7,
) -> np.ndarray:
    """Mirror‑descent style updates: a_i ← a_i·exp(-η·g_i)."""
    n = seq.size
    fft_len = 1 << ((2 * n - 1).bit_length())
    best_seq = seq.copy()
    best_score = _fast_score(best_seq)
    it = 0

    while time.time() < deadline:
        # compute convolution
        fa = np.fft.rfft(seq, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        # indices of the `top_k` largest convolution entries
        if top_k >= conv.size:
            top_idxs = np.arange(conv.size)
        else:
            top_idxs = np.argpartition(conv, -top_k)[-top_k:]

        # average weight vector over the selected peaks
        grad = np.zeros_like(seq)
        for k in top_idxs:
            grad += _weights_for_index(seq, int(k))
        grad /= len(top_idxs)

        # learning rate schedule
        eta = max(min_eta, eta0 * (eta_decay ** it))

        # exponentiated‑gradient step on the simplex
        new_seq = seq * np.exp(-eta * grad)
        # renormalise to the original ℓ₁‑norm (scale is irrelevant for the score)
        total = new_seq.sum()
        if total > 0:
            new_seq *= seq.sum() / total
        else:
            # should never happen, but guard against division by zero
            new_seq = seq.copy()

        new_score = _fast_score(new_seq)
        if new_score < best_score - 1e-12:
            seq = new_seq
            best_seq = new_seq.copy()
            best_score = new_score
            print(
                f"[ExpGrad] Iter {it:6d}: score = {best_score:.8f}"
            )
        else:
            # small random jitter to escape stagnation
            if np.random.rand() < 0.01:
                i = np.random.randint(n)
                j = np.random.randint(n)
                if i != j and seq[i] > 0:
                    eps = min(seq[i], np.random.rand() * seq[i] * 0.05)
                    tmp = seq.copy()
                    tmp[i] -= eps
                    tmp[j] += eps
                    tmp_score = _fast_score(tmp)
                    if tmp_score < best_score:
                        seq = tmp
                        best_seq = tmp.copy()
                        best_score = tmp_score
                        print(
                            f"[ExpGrad Jitter] Iter {it:6d}: score = {best_score:.8f}"
                        )
        it += 1

    return best_seq


# ----------------------------------------------------------------------
# Main entry point used by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights whose evaluation score is
    (hopefully) well below the historical π/2 ≈ 1.571 bound.
    """
    np.random.seed(seed)

    # leave a small safety margin for the harness
    deadline = time.time() + budget_s - 4.0

    # length of the step‑function – a few thousand is a good trade‑off
    n = 5000

    # 1️⃣  Catalan (arcsine) seed
    seq = _catalan_seq(n)

    # 1️⃣‑a tiny random perturbation to break exact symmetry
    jitter = 1.0 + 5e-4 * np.random.randn(n)   # about 0.05 % noise
    seq = seq * np.clip(jitter, 0.0, None)
    seq *= np.sqrt(2.0 * n) / seq.sum()       # normalise to the classical √(2n)

    # 2️⃣  Aggressive mass redistribution (≈70 % of the budget)
    deadline_phase1 = time.time() + 0.70 * (deadline - time.time())
    seq = _aggressive_optimize(seq, deadline_phase1)

    # 3️⃣  Exponentiated‑gradient fine‑tuning (remaining time)
    if time.time() < deadline:
        seq = _expgrad_optimize(seq, deadline)

    # 4️⃣  Return the best sequence found as a plain Python list
    return seq.tolist()
