"""
propose_candidate

Search for a non‑negative step‑function (list of heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

where conv is the full discrete convolution.  The algorithm consists of
three stages:

1. **Broad parametric sampling** – many candidate vectors are drawn from
   several families (power‑law, exponential, power‑law×exponential,
   beta‑shaped, mixtures).  Each candidate is projected onto the simplex
   (sum = 1) and evaluated with a fast FFT based convolution.  The
   best few candidates are kept.

2. **Projected sub‑gradient descent** – each of the retained candidates
   is refined by repeatedly moving a small amount of mass from the index
   that contributes most to the current worst convolution value to the
   index that contributes least.  The step size shrinks during the run,
   the vector is kept on the probability simplex and the improvement
   (exact score via ``evaluate_sequence``) is recorded.

3. **Linear‑programming polishing** – an LP (solved with SciPy’s HiGHS
   solver) computes a direction that maximises total mass while keeping
   the convolution under the current maximum.  A convex combination of
   the current vector and the LP solution typically lowers the objective.
   A few LP polishing steps are attempted if time permits.

The whole procedure runs for the allotted time (default 950 s) and
returns the best sequence found.
"""

import time
import numpy as np

# ---------------------------------------------------------
# Helper utilities (no closures, top‑level definitions)
# ---------------------------------------------------------

def _project_to_simplex(v):
    """Project v onto {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


def _conv_full_fft(a):
    """Full linear convolution of a with itself using FFT (real input)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _fast_score(vec):
    """
    Cheap evaluation of the objective for a vector that already sums to 1.
    Returns 2 * n * max(conv(vec, vec)).
    """
    conv = _conv_full_fft(vec)
    return 2.0 * vec.shape[0] * conv.max()


def _add_candidate(raw, top_k, best_cands, best_seq, best_score):
    """
    Normalise a raw (non‑negative) vector, evaluate its cheap score,
    update the global best and the list of top‑k candidates.
    """
    total = raw.sum()
    if total <= 0:
        return best_seq, best_score
    norm = raw / total
    # cheap evaluation – no clipping occurs because values are ≤ 1
    score = _fast_score(norm)

    # global improvement
    if score < best_score:
        best_score = score
        best_seq = norm.copy()
        print(f"[global] improved to {best_score:.6f}")

    # keep the k best candidates (sorted by score)
    if len(best_cands) < top_k:
        best_cands.append((score, norm.copy()))
        best_cands.sort(key=lambda x: x[0])
    else:
        if score < best_cands[-1][0]:
            best_cands[-1] = (score, norm.copy())
            best_cands.sort(key=lambda x: x[0])

    return best_seq, best_score


def _subgradient_refine(base_vec, budget_s, start, global_best):
    """
    Perform the hill‑climbing / sub‑gradient refinement described in the
    original solution but with a slightly more aggressive schedule.
    Returns the best vector found during the refinement and its score.
    """
    n = base_vec.shape[0]
    cur = base_vec.copy()
    cur_score = _fast_score(cur)
    it = 0
    max_iters = 200_000
    best_vec, best_score = global_best

    while time.time() - start < budget_s - 20 and it < max_iters:
        it += 1

        # full convolution to locate the worst shift
        conv = _conv_full_fft(cur)
        max_idx = int(np.argmax(conv))
        shift = max_idx - (n - 1)          # shift s where conv is maximal

        # contribution of each index to that shift
        contrib = np.zeros_like(cur)
        if shift >= 0:
            s = shift
            contrib[: n - s] += cur[: n - s] * cur[s:]
            contrib[s:] += cur[s:] * cur[: n - s]
        else:
            s = -shift
            contrib[s:] += cur[s:] * cur[: n - s]
            contrib[: n - s] += cur[: n - s] * cur[s:]

        src = int(np.argmax(contrib))
        tgt = int(np.argmin(contrib))
        if src == tgt or cur[src] <= 0:
            src, tgt = np.random.choice(n, size=2, replace=False)

        # step size that shrinks linearly
        frac = 0.5 * (1.0 - it / max_iters) + 0.05       # 0.05 … 0.55
        step = cur[src] * np.random.random() * frac
        if step <= 0:
            continue

        new = cur.copy()
        new[src] -= step
        new[tgt] += step
        new = _project_to_simplex(new)

        new_score = _fast_score(new)
        if new_score < cur_score - 1e-12:
            cur = new
            cur_score = new_score
            # record a new global best if we beat it
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_vec = new.copy()
                print(f"[refine] it={it} new best={best_score:.6f}")

        # occasional re‑projection for numerical safety
        if it % 5000 == 0:
            cur = _project_to_simplex(cur)

    return best_vec, best_score


def _lp_polish(cur, budget_s, start):
    """
    One or two HiGHS LP steps that try to find a feasible direction g
    with larger total mass while keeping f∗g ≤ M (M = max conv of f).
    The mixture (1‑t)·f + t·g is projected back onto the simplex.
    """
    try:
        from scipy.optimize import linprog
    except Exception:                       # SciPy not available → skip LP
        return cur

    n = cur.shape[0]
    max_iters = 3
    for lp_it in range(max_iters):
        if time.time() - start > budget_s - 5:
            break
        # current “worst’’ value
        M = _fast_score(cur) / (2.0 * n)

        # ensure strict positivity for the convolution matrix
        f = np.maximum(cur, 1e-12)
        f /= f.sum()

        # build the convolution matrix A (size (2n‑1)×n)
        rows = 2 * n - 1
        A = np.zeros((rows, n), dtype=np.float64)
        for i in range(n):
            A[i : i + n, i] = f[i]

        # maximise sum(b)  → minimise -sum(b)
        res = linprog(
            c=-np.ones(n),
            A_ub=A,
            b_ub=np.full(rows, M),
            bounds=[(0, None)] * n,
            method="highs",
        )
        if not res.success:
            break

        g = res.x
        if g.sum() == 0:
            break
        g /= g.sum()

        # convex combination and projection
        t = 0.5
        new = (1.0 - t) * cur + t * g
        new = _project_to_simplex(new)

        # accept if it really improves the true objective
        new_score = _fast_score(new)
        if new_score < _fast_score(cur) - 1e-12:
            cur = new
            print(f"[LP] iteration {lp_it} improved to {new_score:.6f}")

    return cur

def propose_candidate(seed=123, budget_s=950, **kwargs):
    """
    Return a list of non‑negative heights (the step function) that
    minimises the provided ``evaluate_sequence`` value within the given
    time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # -----------------------------------------------------
    # Problem size – use the reference sequence if available
    # -----------------------------------------------------
    try:
        n = len(height_sequence_1)   # provided by the evaluation harness
        baseline = np.asarray(height_sequence_1, dtype=float)
    except Exception:
        n = 2244
        baseline = np.full(n, 1.0 / n)

    baseline = _project_to_simplex(baseline)
    best_seq = baseline.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, baseline score={best_score:.6f}")

    # -----------------------------------------------------
    # 1. Broad parametric sampling
    # -----------------------------------------------------
    top_k = 20                               # keep this many candidates
    best_cands = []                          # list of (score, vector)

    idx = np.arange(n, dtype=float)

    # ---- power‑law -------------------------------------------------
    for _ in range(2000):
        if time.time() - start > budget_s - 300:
            break
        alpha = rng.uniform(0.05, 2.0)
        shift = rng.uniform(0.0, 10.0)
        raw = np.power(idx + shift, -alpha)
        best_seq, best_score = _add_candidate(raw, top_k, best_cands,
                                              best_seq, best_score)

    # ---- exponential -----------------------------------------------
    for _ in range(1000):
        if time.time() - start > budget_s - 300:
            break
        beta = rng.uniform(1e-5, 0.2)
        raw = np.exp(-beta * idx)
        best_seq, best_score = _add_candidate(raw, top_k, best_cands,
                                              best_seq, best_score)

    # ---- power‑law × exponential ------------------------------------
    for _ in range(1500):
        if time.time() - start > budget_s - 300:
            break
        alpha = rng.uniform(0.05, 2.0)
        shift = rng.uniform(0.0, 10.0)
        beta = rng.uniform(1e-5, 0.2)
        raw = np.power(idx + shift, -alpha) * np.exp(-beta * idx)
        best_seq, best_score = _add_candidate(raw, top_k, best_cands,
                                              best_seq, best_score)

    # ---- beta‑shaped (i+1)^p * (n-i)^q -----------------------------
    for _ in range(1500):
        if time.time() - start > budget_s - 300:
            break
        p = rng.uniform(-1.0, 2.0)
        q = rng.uniform(-1.0, 2.0)
        raw = np.power(idx + 1, p) * np.power(n - idx, q)
        raw = np.maximum(raw, 0.0)
        best_seq, best_score = _add_candidate(raw, top_k, best_cands,
                                              best_seq, best_score)

    # ---- two‑component power‑law mixtures ---------------------------
    for _ in range(2000):
        if time.time() - start > budget_s - 300:
            break
        a1 = rng.uniform(0.05, 2.0)
        s1 = rng.uniform(0.0, 10.0)
        a2 = rng.uniform(0.05, 2.0)
        s2 = rng.uniform(0.0, 10.0)
        w = rng.random()
        raw = w * np.power(idx + s1, -a1) + (1 - w) * np.power(idx + s2, -a2)
        best_seq, best_score = _add_candidate(raw, top_k, best_cands,
                                              best_seq, best_score)

    print(f"[sampling] best candidate after sampling: {best_score:.6f}")

    # -----------------------------------------------------
    # 2. Sub‑gradient / hill‑climbing refinement
    # -----------------------------------------------------
    # Refine the global best and a few other top candidates.
    to_refine = [cand[1] for cand in best_cands[:5]]
    to_refine.append(best_seq)   # ensure the current global best is refined

    # keep the best vector seen so far together with its score
    global_best_vec = best_seq.copy()
    global_best_score = best_score

    for start_vec in to_refine:
        vec, scr = _subgradient_refine(start_vec, budget_s, start,
                                       (global_best_vec, global_best_score))
        if scr < global_best_score:
            global_best_vec = vec
            global_best_score = scr

    # -----------------------------------------------------
    # 3. Linear‑programming polishing (optional)
    # -----------------------------------------------------
    polished = _lp_polish(global_best_vec, budget_s, start)
    polished_score = _fast_score(polished)

    if polished_score < global_best_score - 1e-12:
        global_best_vec = polished
        global_best_score = polished_score
        print(f"[polish] score after LP = {global_best_score:.6f}")

    # final verification with the official evaluator
    final_score = evaluate_sequence(global_best_vec.tolist())
    print(f"[final] best score returned: {final_score:.6f}")

    return global_best_vec.tolist()
