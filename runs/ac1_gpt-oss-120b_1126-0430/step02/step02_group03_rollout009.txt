Function project_simplex(v):
    # ensures non-negative and sum equals 1.

Implementation using method from Duchi et al. 2008:

    u = np.sort(v)[::-1]   # descending
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
propose_candidate
-----------------

A fast iterative sub‑gradient / mirror‑descent optimizer that starts from the
Catalan (1‑/√(1‑z)) distribution and repeatedly reduces the maximal entry of
the self‑convolution while keeping the vector on the probability simplex.
The algorithm runs as long as the time budget allows, performs many thousands
of inexpensive FFT‑based updates (O(n log n) each) and keeps the best point
found.  Simple random restarts are added if time remains.

The objective that we minimise is

    O(a) = 2·n·max_k (a * a)[k]          (∑a = 1, a_i ≥ 0)

which is exactly the value returned by ``evaluate_sequence`` after the
normalisation step.  The optimisation uses the exact sub‑gradient of the
max‑convolution term (the gradient of the active quadratic form) together
with a projection onto the simplex.  Momentum (Nesterov‑style) and a
diminishing step size improve convergence.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: real FFT based linear convolution (full “linear” mode)
# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT.  Result length = 2*len(a)-1."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Projection of a vector onto the probability simplex (sum = 1, non‑negative)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = 1}."""
    n = v.size
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)                      # 1‑based index
    cond = u - cssv / ind > 0
    if not np.any(cond):                           # numerical safety
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]                             # last true index
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Catalan‑type coefficients, normalised to unit L¹‑norm
# ----------------------------------------------------------------------
def _catalan_normalised(n: int) -> np.ndarray:
    """First n terms a_i = C(2i,i)/4**i, normalised so that Σ a_i = 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)   # a_i = a_{i‑1}·(2i‑1)/(2i)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Objective (exactly the value returned by evaluate_sequence for Σa=1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    return float(2.0 * n * np.max(_fft_convolve(a)))


# ----------------------------------------------------------------------
# Core optimiser – projected sub‑gradient / mirror descent
# ----------------------------------------------------------------------
def _optimize(a0: np.ndarray,
              deadline: float,
              max_iters: int = 500_000,
              verbose: bool = True) -> tuple[np.ndarray, float]:
    """
    Refine a candidate vector a0 (already normalised to sum‑1) by
    projected sub‑gradient descent on the objective 2·n·max(a*a).

    Returns the best vector discovered and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]

    best = a.copy()
    best_val = _objective(best)

    # momentum parameters (classical heavy‑ball)
    beta = 0.9
    momentum = np.zeros_like(a)

    t = 0
    while time.time() < deadline and t < max_iters:
        # current convolution and the active index
        conv = _fft_convolve(a)
        k = int(np.argmax(conv))
        max_c = conv[k]

        # current objective
        cur_val = 2.0 * n * max_c
        if cur_val < best_val:
            best_val = cur_val
            best = a.copy()
            if verbose:
                print(f"[improve] iter {t:06d}  obj {best_val:.9f}")

        # sub‑gradient of the active quadratic form (g = 2·J_k·a)
        grad = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        grad[idx] = 2.0 * a[k - idx]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # diminishing step‑size (standard 1/√t schedule)
        lr = 0.5 / np.sqrt(t + 1)

        # additive update + projection onto simplex
        a = a - lr * momentum
        a = np.maximum(a, 0.0)                     # keep non‑negative
        a = _project_simplex(a)                    # enforce Σa = 1

        # occasional tiny random perturbation to escape plateaus
        if t % 5_000 == 0 and t > 0:
            pert = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + pert, 0.0)
            a = _project_simplex(a)

        t += 1

    return best, best_val
