Adam‑based smooth‑max optimization for the auto‑convolution
objective  2 n·max_k (a*a)[k]  with a on the probability simplex.

def _fft_len_for_conv(n):
    # we need length for convolutions up to length 3n-2
    L = 3 * n - 2
    return 1 << ((L).bit_length())

def _full_convolution(a, b, fft_len):
    """Return full convolution a*b using FFT length fft_len."""
    fa = np.fft.fft(a, fft_len)
    fb = np.fft.fft(b, fft_len)
    conv = np.fft.ifft(fa * fb).real
    # truncate to length len(a)+len(b)-1
    return conv[:len(a) + len(b) - 1]

def _softmax_weights(conv, tau):
    """Return softmax weights w_k = exp(tau*conv_k)/sum."""
    # stability: subtract max
    m = conv.max()
    exp_term = np.exp(tau * (conv - m))
    return exp_term / exp_term.sum()

def _gradient_softmax_max(a, conv, tau, fft_len):
    """
    Gradient of f_tau(a) = (1/tau) log Σ exp(tau * (a*a)[k])
    w.r.t the logits (before simplex projection).
    Returns gradient wrt a (not logits) first.
    """
    # softmax over convolution entries
    w = _softmax_weights(conv, tau)   # shape (2n-1,)
    # compute g = Σ_k w_k ∂conv_k/∂a
    # ∂conv_k/∂a_i = a_{k-i} + a_{i-k}
    # we compute using two convolutions: w * a and w * rev(a)

    start = len(a) - 1

    return g

    v = beta2 * v + (1.0 - beta2) * (grad * grad)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    x_new = x - lr * m_hat / (np.sqrt(v_hat) + eps)
    return x_new, m, v

#!/usr/bin/env python
"""
Adam‑based optimisation of a smooth‑max approximation of the
convolution objective

    F(a) = 2·n·max_k (a*a)[k]          with a_i ≥ 0 , Σ a_i = 1

The max is approximated by
    f_τ(a) = (1/τ)·log Σ_k exp( τ·(a*a)[k] )
which becomes the true max as τ → ∞.
The distribution a is represented by logits x (soft‑max
parameterisation) so that the simplex constraint is satisfied
automatically.
The gradient of f_τ w.r.t. the logits is obtained analytically
using two FFT‑based convolutions (w*a and w*rev(a)).
Adam updates are applied to the logits; every few hundred steps
the exact cheap score 2·n·max(conv) is computed and the best
candidate is stored.
Occasional random “shakes’’ are added to avoid plateaus.
The routine starts from the previously discovered construction
(if the global variable `height_sequence_1` exists) or from a
power‑law seed, and finally returns the best sequence found
within the time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – top level only
# ----------------------------------------------------------------------

def _fft_len_for_n(n: int) -> int:
    """
    Return a power‑of‑two size that can hold the full convolution
    of two vectors of lengths up to 3·n‑2 (the longest convolution
    needed in the algorithm).
    """
    target = 3 * n - 2
    return 1 << ((target).bit_length())


def _full_convolution(a: np.ndarray, b: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*b using FFT of the given size."""
    fa = np.fft.fft(a, fft_len)
    fb = np.fft.fft(b, fft_len)
    conv = np.fft.ifft(fa * fb).real
    return conv[:len(a) + len(b) - 1]


def _softmax_weights(conv: np.ndarray, tau: float) -> np.ndarray:
    """Soft‑max over the convolution entries, stable for large τ."""
    m = conv.max()
    exp_term = np.exp(tau * (conv - m))
    return exp_term / exp_term.sum()


def _gradient_softmax_max(a: np.ndarray, conv: np.ndarray,
                         tau: float, fft_len: int) -> np.ndarray:
    """
    Gradient of the smooth‑max loss f_τ(a) = (1/τ)·log Σ exp(τ·conv)
    with respect to the vector a (still on the simplex).
    Returns a vector of length len(a).
    """
    # soft‑max weights for each convolution entry
    w = _softmax_weights(conv, tau)                      # shape (2n‑1,)

    # w * a  (full convolution, length 3n‑2)
    conv_wa = _full_convolution(w, a, fft_len)[:3 * len(a) - 2]
    term2 = conv_wa[:len(a)]                            # (w*a)[i]

    # w * rev(a)
    rev_a = a[::-1]
    conv_wr = _full_convolution(w, rev_a, fft_len)[:3 * len(a) - 2]
    start = len(a) - 1
    term1 = conv_wr[start:start + len(a)]               # (w*rev(a))[i+n‑1]

    # ∂f/∂a = term1 + term2
    return term1 + term2


def _adam_step(x: np.ndarray, grad: np.ndarray,
               m: np.ndarray, v: np.ndarray, t: int,
               lr: float = 0.12,
               beta1: float = 0.9, beta2: float = 0.999,
               eps: float = 1e-8):
    """Perform a single Adam update of the logits x."""
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad * grad)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    x_new = x - lr * m_hat / (np.sqrt(v_hat) + eps)
    return x_new, m, v
