def propose_candidate(seed=123, budget_s=950, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    
    try: n = len(height_sequence_1)
    except: n = 2244
    
    # generate a set of initial candidates to have a good start
    # We'll sample same families as earlier but fewer
    best_seq = None
    best_score = float('inf')
    idx = np.arange(n, dtype=float)
    # define function to evaluate candidate: uses evaluate_sequence
    def consider(raw, label):
        nonlocal best_seq, best_score
        raw = np.maximum(raw, 0)
        if raw.sum() == 0: return
        norm = raw / raw.sum()
        score = evaluate_sequence(norm.tolist())
        if score < best_score:
            best_score = score
            best_seq = norm
            print(f"[{label}] init improved to {score:.6f}")

    # baseline if height_sequence_1 provided
    try:
        arr = np.array(height_sequence_1, dtype=float)
        arr = np.maximum(arr, 0)
        norm = arr / arr.sum()
        best_seq = norm
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[baseline] score {best_score:.6f}")
    except:
        pass

    # sample random power law etc.
    for _ in range(2000):
        alpha = rng.uniform(0.15, 1.5)
        shift = rng.uniform(1e-3, 4.0)
        raw = np.power(idx + shift, -alpha)
        consider(raw, "pl")
        if time.time() - start > budget_s * 0.1:
            break

    # include random exponential
    for _ in range(1000):
        beta = rng.uniform(1e-5, 0.15)
        raw = np.exp(-beta * idx)
        consider(raw, "exp")
        if time.time() - start > budget_s * 0.2:
            break

    # fallback uniform if not set
    if best_seq is None:
        best_seq = np.full(n, 1.0/n)
        best_score = evaluate_sequence(best_seq.tolist())

    # start gradient descent from best_seq
    cur = best_seq.copy()
    # Ensure sum=1
    cur = cur / cur.sum()

    # convolution length
    fft_len = 1
    while fft_len < 2*n:
        fft_len <<= 1

    # Precomputed occasional full evaluation
    def conv_fft(a):
        A = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(A * A, fft_len)
        return conv[:2*n-1]

    cur_conv = conv_fft(cur)
    cur_max = cur_conv.max()
    cur_score = 2.0 * n * cur_max

    best_seq = cur.copy()
    best_score = cur_score

    print(f"[init descent] score {cur_score:.6f}")

    # parameters
    max_iters = 200000
    last_improve_iter = 0

    for it in range(max_iters):
        # time check
        if time.time() - start > budget_s - 5:
            break

        # max index
        max_idx = int(np.argmax(cur_conv))
        # subgradient for max_idx
        lo = max(0, max_idx - n + 1)
        hi = min(n - 1, max_idx)
        idx_range = np.arange(lo, hi + 1)
        g = np.zeros_like(cur)
        if idx_range.size:
            g[idx_range] = 2.0 * cur[max_idx - idx_range]

        # step schedule
        step = 0.5 / np.sqrt(it + 1)  # initial 0.5, decaying
        new = cur - step * g
        # project
        new = _project_to_simplex(new)

        # compute conv of new
        new_conv = conv_fft(new)
        new_max = new_conv.max()
        new_score = 2.0 * n * new_max

        # decide acceptance
        if new_score < cur_score - 1e-12:
            cur = new
            cur_conv = new_conv
            cur_score = new_score
            # update global best
            if new_score < best_score - 1e-12:
                best_seq = new.copy()
                best_score = new_score
                print(f"[iter {it}] new best {best_score:.6f}")
                last_improve_iter = it
        else:
            # occasional random jitter to escape
            if rng.random() < 0.001:
                cur = new

"""
Propose a candidate step‑function (list of non‑negative heights) that
minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of three stages:

1. **Fast initial sampling** – generate a few simple families
   (power‑law, exponential) and keep the best one (according to the
   provided `evaluate_sequence`).  If the harness supplies a previous
   construction (`height_sequence_1`) it is used as the starting point.

2. **Projected sub‑gradient descent on the simplex** – the objective
   depends only on the maximal convolution value `t = max(conv(a,a))`.
   For the current index `k*` that attains `t` the sub‑gradient w.r.t.
   `a_i` is `2 * a_{k*‑i}` (zero when the index is out of range).  An
   additive step `a ← a – η·g` is taken and the result is projected back
   onto the probability simplex (non‑negative, sum = 1) using the
   Duchi‑et‑al. algorithm.  The step size decays as `η = 0.2 / sqrt(t)`.
   Convolutions are computed with an FFT (complexity `O(n log n)`) to
   keep each iteration cheap.

3. **Periodic exact evaluation & perturbations** – every few thousand
   iterations the best vector is checked with the official evaluator
   (to avoid drift), and occasional random perturbations are applied
   when progress stalls.

The routine respects the 1000 s time budget, prints progress messages
and finally returns the best sequence found as a plain Python list of
floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, sum_i x_i = 1}.
    Implements the algorithm from
    Duchi, Shalev‑Shwartz, Singer & Chandra (2008).
    """
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback to uniform distribution if something goes wrong
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute the discrete convolution a * a using FFT.
    Returns a real array of length 2*len(a)-1.
    """
    A = np.fft.rfft(a, fft_len)
    conv_full = np.fft.irfft(A * A, fft_len)
    return conv_full[: 2 * a.shape[0] - 1]

# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: float = 950, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) improves the
    current upper bound for the evaluation function.
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # Problem size (the harness may provide a previous construction)
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # supplied by the evaluation harness
    except Exception:
        n = 2244                            # fallback length used in the description

    # ------------------------------------------------------------------
    # 1. Fast initial sampling to obtain a decent starting point
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    idx = np.arange(n, dtype=float)

    def _consider(raw_vec: np.ndarray, label: str):
        """Normalise, evaluate and possibly update the global best."""
        nonlocal best_seq, best_score
        raw_vec = np.maximum(raw_vec, 0.0)
        total = raw_vec.sum()
        if total <= 0.0:
            return
        norm = raw_vec / total
        score = evaluate_sequence(norm.tolist())
        if score < best_score:
            best_score = score
            best_seq = norm.copy()
            print(f"[{label}] initial best → {score:.6f}")

    # use the supplied baseline if it exists
    try:
        baseline = np.array(height_sequence_1, dtype=float)
        _consider(baseline, "baseline")
    except Exception:
        pass

    # a small grid of power‑law shapes
    alphas = np.linspace(0.2, 1.5, 25)

    for a in alphas:

            raw = np.power(idx + s, -a)
            _consider(raw, "grid")

    # a few random power‑law draws
    for _ in range(1500):

        s = rng.uniform(1e-3, 4.0)

    # ------------------------------------------------------------------
    # 2. Projected sub‑gradient descent on the simplex
    # ------------------------------------------------------------------
    cur = best_seq.copy()
    cur = cur / cur.sum()                     # ensure exact normalisation

    # FFT length – smallest power of two ≥ 2n
    fft_len = 1
    while fft_len < 2 * n:

    cur_conv = _conv_fft(cur, fft_len)
    cur_max = cur_conv.max()

    best_seq = cur.copy()
    best_score = cur_score

    max_iters = 200_000
    last_improve = 0

    for it in range(max_iters):
        # Abort if we are out of the allotted budget (keep a small safety margin)
        if time.time() - start > budget_s - 5.0:
            break

        # index where the current convolution attains its maximum
        k_star = int(np.argmax(cur_conv))

        # ---- sub‑gradient of max(conv) at k_star ----
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        g = np.zeros_like(cur)
        if lo <= hi:
            i_range = np.arange(lo, hi + 1)
            g[i_range] = 2.0 * cur[k_star - i_range]

        # ---- step size (diminishing) ----
        step = 0.2 / np.sqrt(it + 1.0)   # 0.2 is a moderate initial magnitude

        # ---- gradient step + projection onto the simplex ----
        cand = cur - step * g
        cand = _project_to_simplex(cand)

        # ---- cheap evaluation via FFT (sum is guaranteed to be 1) ----
        cand_conv = _conv_fft(cand, fft_len)
        cand_max = cand_conv.max()
        cand_score = 2.0 * n * cand_max

        # ---- accept if we made progress ----
        improved = False
        if cand_score < cur_score - 1e-12:
            improved = True
        elif rng.random() < 0.001:  # very occasional uphill move
            improved = True

        if improved:
            cur = cand
            cur_conv = cand_conv
            cur_score = cand_score

            if cand_score < best_score - 1e-12:
                best_seq = cand.copy()
                best_score = cand_score
                last_improve = it
                print(f"[iter {it}] new best ≈ {best_score:.6f}")

        # ---- occasional random perturbation when stuck ----
        if it - last_improve > 20_000 and rng.random() < 0.01:
            noise = rng.normal(scale=0.015, size=n)
            cur = _project_to_simplex(cur + noise)
            cur_conv = _conv_fft(cur, fft_len)
            cur_score = 2.0 * n * cur_conv.max()
            last_improve = it

        # ---- re‑evaluate with the official function every few thousand iters ----
        if it % 5_000 == 0:
            real = evaluate_sequence(best_seq.tolist())
            if real < best_score:
                best_score = real
                best_seq = best_seq.copy()
                print(f"[exact] verified best → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Final verification & return
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    if final_score < best_score:
        best_score = final_score
        print(f"[final] official score improved to {best_score:.6f}")
    else:
        print(f"[final] official score = {best_score:.6f}")

    return best_seq.tolist()
