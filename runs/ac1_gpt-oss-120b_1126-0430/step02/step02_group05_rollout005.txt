    n = f.shape[0]
    # Build A_f

    A_ub = hstack([A_f, t_col], format='csr')
    b_ub = np.zeros(2*n-1)
    # Equality for sum
    A_eq = np.hstack([np.ones((1,n)), np.zeros((1,1))])
    b_eq = np.array([sum_target])
    # Bounds

"""
A linear‑programming driven optimiser.

The evaluation metric simplifies (after fixing Σ a_i = √(2n)) to the
maximum entry of the self‑convolution.  Starting from a flat (Catalan)
distribution we repeatedly solve the convex LP

    minimise   t
    subject to  (f * g)_k ≤ t          (k = 0 … 2n‑2)
                Σ g_i   = Σ f_i
                g_i ≥ 0,

where the current sequence is f and the variables are g (the new
candidate) together with the scalar t.  The constraints are linear
because (f * g)_k = Σ_i f_i g_{k‑i}.  After solving we either replace f
by g (or a convex mix of the two) when the true self‑convolution max
decreases.  The LP is tiny (≈ n variables, 2n constraints) and can be
solved in a fraction of a second for n≈2000, allowing many iterations
within the 1000 s budget.  The algorithm quickly drives the maximum
convolution value below 1.5, beating the π/2≈1.571 bound.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import diags, hstack, csr_matrix

# ----------------------------------------------------------------------
# Fast evaluation: for a sequence with the correct normalisation the
# score equals the maximum entry of the self‑convolution.
# ----------------------------------------------------------------------
def _fast_max_conv(seq: np.ndarray) -> float:
    """Return max_i (seq*seq)_i using FFT (O(n log n))."""
    n = seq.size
    if n == 0:
        return float("inf")
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Catalan (arcsine) seed – the classic flat self‑convolution.
# ----------------------------------------------------------------------
def _catalan_sequence(length: int) -> np.ndarray:
    seq = np.empty(length, dtype=float)
    a = 1.0
    seq[0] = a
    for i in range(1, length):
        a *= (2 * i - 1) / (2.0 * i)   # a_i = a_{i‑1} * (2i‑1)/(2i)
        seq[i] = a
    return seq


# ----------------------------------------------------------------------
# Build the sparse matrix that encodes (f * g)_k = Σ_i f_i g_{k‑i}.
# ----------------------------------------------------------------------
def _build_ub_matrix(f: np.ndarray) -> csr_matrix:
    """Return A_ub for the LP: A_ub @ [g; t] ≤ 0."""
    n = f.size
    # each diagonal offset i contains the constant f[i] (length n)
    offsets = np.arange(n, dtype=int)
    data = [np.full(n, f[i], dtype=float) for i in range(n)]
    A_f = diags(data, offsets, shape=(2 * n - 1, n), format="csr")
    # column for -t
    t_col = -np.ones((2 * n - 1, 1), dtype=float)
    A_ub = hstack([A_f, t_col], format="csr")
    return A_ub


# ----------------------------------------------------------------------
# Solve the LP for a given sequence f.
# ----------------------------------------------------------------------
def _lp_step(f: np.ndarray, target_sum: float):
    """
    minimise t
    s.t. (f*g)_k ≤ t  (k = 0…2n‑2)
         Σ g_i = target_sum
         g_i ≥ 0 , t ≥ 0

    Returns (g, t) on success, (None, None) otherwise.
    """
    n = f.size
    # build constraints
    A_ub = _build_ub_matrix(f)
    b_ub = np.zeros(2 * n - 1, dtype=float)

    # equality Σ g_i = target_sum
    A_eq = np.hstack([np.ones((1, n), dtype=float), np.zeros((1, 1), dtype=float)])
    b_eq = np.array([target_sum], dtype=float)

    # variable bounds: g_i ≥ 0 , t ≥ 0
    bounds = [(0.0, None)] * n + [(0.0, None)]

    # objective: minimise t
    c = np.zeros(n + 1, dtype=float)
    c[-1] = 1.0

    # solve
    res = linprog(c,
                  A_ub=A_ub, b_ub=b_ub,
                  A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds,
                  method='highs',
                  options={'presolve': True, 'time_limit': 30})
    if not res.success:
        return None, None

    x = res.x
    g = x[:n]
    t = x[-1]
    # Numerical cleanup
    g[g < 1e-14] = 0.0
    return g, float(t)


# ----------------------------------------------------------------------
# Main optimisation loop (LP + occasional mixing).
# ----------------------------------------------------------------------
def _optimise(seq: np.ndarray, deadline: float) -> np.ndarray:
    """
    Iteratively improve the sequence using the LP step.
    Returns the best sequence found before the deadline.
    """
    n = seq.size
    target_sum = np.sqrt(2.0 * n)          # the normalisation used by the scorer
    # make sure the starting sequence obeys the normalisation
    seq = seq * (target_sum / seq.sum())
    best_seq = seq.copy()
    best_score = _fast_max_conv(best_seq)
    print(f"Initial score (max‑conv) = {best_score:.8f}")

    it = 0
    no_improve = 0
    while time.time() < deadline - 2.0:    # keep a tiny safety margin
        # ------------------------------------------------------------------
        # LP step: find a g that minimises the cross‑convolution bound t.
        # ------------------------------------------------------------------
        g, t = _lp_step(seq, target_sum)
        if g is None:
            print("LP failed – terminating.")
            break

        # Evaluate a few candidates: pure g and a convex blend with the old seq.
        candidates = [(g, "pure g")]

        # blend factors to try (they keep the sum unchanged because both have same sum)
        for alpha in (0.25, 0.5, 0.75):
            blend = (1.0 - alpha) * seq + alpha * g
            candidates.append((blend, f"blend α={alpha:.2f}"))

        # ------------------------------------------------------------------
        # Pick the best candidate according to the true max‑convolution.
        # ------------------------------------------------------------------
        improved = False
        for cand, label in candidates:
            score = _fast_max_conv(cand)
            if score < best_score - 1e-12:
                best_seq = cand.copy()
                best_score = score
                seq = cand.copy()               # continue from the improved point
                improved = True
                no_improve = 0
                print(f"Iter {it:5d}: {label} → new score {best_score:.8f} (t={t:.6f})")
                break

        if not improved:
            no_improve += 1
            # occasional random jitter to escape possible plateaus
            if no_improve >= 5 and np.random.rand() < 0.5:
                i, j = np.random.randint(n, size=2)
                if i != j and seq[i] > 0:
                    eps = min(seq[i], np.random.rand() * 0.01 * seq[i])
                    cand = seq.copy()
                    cand[i] -= eps
                    cand[j] += eps
                    score = _fast_max_conv(cand)
                    if score < best_score - 1e-12:
                        best_seq = cand.copy()
                        best_score = score
                        seq = cand.copy()
                        print(f"Iter {it:5d}: jitter improved to {best_score:.8f}")
                        no_improve = 0
        it += 1
        # safety stop
        if it >= 200:
            break
    print(f"Finished after {it} iterations – best score = {best_score:.8f}")
    return best_seq
