def convolve_fft_real(x, y, L):
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    Z = X * Y
    result = np.fft.irfft(Z, n=L)
    return result[:len(x)+len(y)-1]

"""
Propose a step‑function sequence that minimizes the bound

    O = 2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm works on the simplex (sum(a)=1) so that the denominator
is 1 and the objective reduces to 2·n·max(conv(a,a)).  It combines:

1. **Fast FFT‑based convolutions** – O(n log n) per iteration.
2. **Smoothed‑max gradient descent** – a log‑sum‑exp approximation of
   the max‑convolution yields a differentiable surrogate; the gradient
   is obtained by a second convolution, then the iterate is projected
   back onto the probability simplex.
3. **Simulated‑annealing type random moves** – occasional mass‑transfer
   steps help escape shallow plateaus.
4. **Multi‑restart & adaptive schedules** – step size and smoothing
   parameter are slowly adapted; after a long stall the algorithm is
   re‑initialised from the current best solution with a small random
   perturbation.

The routine respects the 1000 s budget, prints progress, and finally
returns the best sequence found (as a plain list of floats).  The
sequence can be directly fed to the supplied `evaluate_sequence` function.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x≥0, sum(x)=1}."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Small utility – next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray, L: int) -> np.ndarray:
    """
    Convolution of real vectors a and b using FFT of length L.
    Returns the (len(a)+len(b)-1)-long result (real‑valued).
    """
    A = np.fft.rfft(a, n=L)
    B = np.fft.rfft(b, n=L)
    C = A * B
    conv = np.fft.irfft(C, n=L)
    length = a.shape[0] + b.shape[0] - 1
    return conv[:length]


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for sum(a)=1."""
    n = a.size
    conv = np.convolve(a, a)            # small enough for the occasional use
    return 2.0 * n * np.max(conv)

    Search for a non‑negative step‑function that minimises the bound.
    Returns a list of floats (the sequence).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.3   # safety margin

    # --------------------------------------------------------------
    # 1) Initialise a feasible sequence (sum = 1, non‑negative)
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=float)
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-12:
            a = np.random.rand(a.size)
        a = a / a.sum()
        n = a.size
        print(f"[init] Using provided sequence of length {n}")
    else:
        # a moderate length – large enough for a good bound, small enough to be fast
        n = 2500
        a = np.random.rand(n)
        a = _project_to_simplex(a)
        print(f"[init] Random start, length = {n}")

    # Pre‑computed FFT length (must accommodate convolutions of length 3n‑2)
    L = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # 2) Main optimisation loop
    best_a = a.copy()
    best_obj = _objective(best_a)
    print(f"[init] objective = {best_obj:.6f}")

    # Adaptive parameters
    lam = 5.0            # smoothing parameter for log‑sum‑exp (starts low)
    eta = 0.5            # step size for gradient descent
    iter_cnt = 0
    no_improve_cnt = 0
    max_no_improve = 8000   # restart after this many non‑improving steps

    while time.time() < deadline and iter_cnt < 300_000:
        iter_cnt += 1

        # ----- (a) compute self‑convolution via FFT -----------------
        A_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(A_fft * A_fft, n=L)[: 2 * n - 1]
        max_conv = conv_full.max()
        obj = 2.0 * n * max_conv

        # ----- (b) record a new best if we have one ----------------
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            no_improve_cnt = 0

        else:

        # slowly increase smoothing (focus more on the real max)
