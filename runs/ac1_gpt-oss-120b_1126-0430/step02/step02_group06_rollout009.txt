        alpha = max_alpha * min(1.0, it / (2000.0)) # ramp up over 2000 iters
        if alpha < 1.0:
            alpha = 1.0

        # compute p = softmax(alpha * conv)
        max_act = np.max(alpha * conv)
        exp_vals = np.exp(alpha * conv - max_act)
        p = exp_vals / exp_vals.sum()

        # compute gradient
        a_rev = a[::-1]
        fp = np.fft.rfft(p, m)
        fa_rev = np.fft.rfft(a_rev, m)
        conv2 = np.fft.irfft(fp * fa_rev, m)[:3*n-2]
        g = 2.0 * conv2[n-1:n-1+n]

        dot_ag = np.dot(a, g)
        grad_y = a * (g - dot_ag)

        # Adam update
        mt = beta1 * mt + (1 - beta1) * grad_y
        vt = beta2 * vt + (1 - beta2) * (grad_y ** 2)
        mt_hat = mt / (1 - beta1 ** it)
        vt_hat = vt / (1 - beta2 ** it)
        step = lr * mt_hat / (np.sqrt(vt_hat) + eps_adam)
        y = y - step

        # evaluate score
        score = 2.0 * n * conv.max()
        if score < best_score - 1e-12:
            best_score = score
            best_seq = a.copy()
            if it % 200 == 0:
                print(f"[opt n={n}] iter {it} score {best_score:.6f}")

        # occasional random jitter
        if (it & 0x1FF) == 0:  # every 512 steps
            jitter_idx = np.random.randint(n)
            a_j = best_seq[jitter_idx]
            # small perturbation and renormalize
            perturb = (np.random.rand() - 0.5) * 0.001
            a_new = best_seq.copy()
            a_new[jitter_idx] = max(0.0, a_j + perturb)
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new / a_new.sum()
            y = np.log(a_new + eps)  # reset y to this new a for momentum reset?
            # optionally keep mt, vt as is.

    return best_seq, best_score

"""
Optimising a non‑negative step function for the metric

    score = 2·n·max(conv(seq,seq)) / (sum(seq)²)

The metric is scale‑invariant, so we fix sum(seq)=1 and minimise the
maximal self‑convolution.  The code proceeds in three phases:

1.  A quick optimisation on a small problem (n≈800) using Adam on the
    log‑probability (soft‑max) parametrisation.  A smooth approximation of
    the max‑operator (log‑sum‑exp) yields a well‑behaved gradient.

2.  The best small‑scale solution is interpolated to a larger length
    (n≈8000) and refined for the remaining time with the same optimiser.
    The learning‑rate schedule and the “temperature’’ α (which controls
    how sharply the soft‑max approximates the true maximum) are annealed.

3.  A cheap stochastic local‑search (random mass transfers) finishes the
    run, trying to squeeze out any remaining improvement.

All heavy work uses FFT‑based convolutions; the routine respects the
1000 s budget and prints intermediate scores.  The returned list satisfies
the security checks of ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast evaluation (identical to evaluate_sequence but without checks).
# ----------------------------------------------------------------------
def _fast_evaluation(seq: np.ndarray) -> float:
    """Return the metric for a non‑negative sequence with sum(seq)=1."""
    n = seq.size
    if n == 0:
        return np.inf
    # FFT length = next power‑of‑two ≥ 2·n‑1
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    max_conv = conv.max()
    # sum(seq) == 1 -> score = 2·n·max_conv
    return float(2.0 * n * max_conv)


# ----------------------------------------------------------------------
# Catalan (arcsine) seed – the coefficients of (1‑z)^‑½.
# ----------------------------------------------------------------------
def _catalan_sequence(length: int) -> np.ndarray:
    """First ``length`` Catalan‑like coefficients (non‑negative, decreasing)."""
    seq = np.empty(length, dtype=float)
    a = 1.0
    seq[0] = a
    for i in range(1, length):
        a *= (2.0 * i - 1) / (2.0 * i)
        seq[i] = a
    return seq


# ----------------------------------------------------------------------
# Linear interpolation from a short to a long sequence.
# ----------------------------------------------------------------------
def _interpolate(seq_small: np.ndarray, length_big: int) -> np.ndarray:
    """Upsample ``seq_small`` to ``length_big`` by linear interpolation."""
    if seq_small.size == length_big:
        return seq_small.copy()
    x_small = np.linspace(0.0, 1.0, seq_small.size)
    x_big = np.linspace(0.0, 1.0, length_big)
    big = np.interp(x_big, x_small, seq_small)
    # clip tiny negatives caused by interpolation noise
    big = np.maximum(big, 0.0)
    # re‑normalise to sum = 1
    total = big.sum()
    if total == 0.0:
        big[:] = 1.0 / length_big
    else:
        big /= total
    return big


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (soft‑max parametrisation).
# The loss is a smoothed max‑convolution given by log‑sum‑exp.
# ----------------------------------------------------------------------
def _adam_optimize(seq_init: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Refine ``seq_init`` (already normalised to sum = 1) until ``deadline``.
    Returns the best sequence found and its score.
    """
    n = seq_init.size
    # initialise log‑probability vector (mirror descent variable)
    eps = 1e-12
    y = np.log(seq_init + eps)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.30

    # FFT length that can hold convolutions of size up to 3·n‑2
    m = 1 << ((3 * n - 2).bit_length())

    # bookkeeping
    mt = np.zeros_like(y)
    vt = np.zeros_like(y)
    best_seq = np.exp(y - np.max(y))
    best_seq /= best_seq.sum()
    best_score = _fast_evaluation(best_seq)

    it = 0
    while time.time() < deadline:
        it += 1

        # ----- soft‑max probabilities -----
        max_y = np.max(y)
        a = np.exp(y - max_y)
        a_sum = a.sum()
        a = a / a_sum                     # a ≥ 0, ∑a = 1

        # ----- self‑convolution (FFT) -----
        fa = np.fft.rfft(a, m)
        conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]   # length 2n‑1

        # ----- smooth max via log‑sum‑exp (temperature α) -----
        # α grows from a small value to a large one (≈20·n) as iterations proceed.
        max_alpha = n * 20.0
        alpha = max_alpha * min(1.0, it / 2000.0)
        if alpha < 1.0:
            alpha = 1.0
        scaled = alpha * conv
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        p = exp_scaled / exp_scaled.sum()                # p is a probability vector on 2n‑1 points

        # ----- gradient of the smooth max w.r.t. a -----
        a_rev = a[::-1]
        fp = np.fft.rfft(p, m)
        fa_rev = np.fft.rfft(a_rev, m)
        conv2 = np.fft.irfft(fp * fa_rev, m)[: 3 * n - 2]   # length 3n‑2
        g = 2.0 * conv2[n - 1 : n - 1 + n]                  # shape (n,)

        # sub‑gradient on the log‑probability variables
        dot_ag = np.dot(a, g)
        grad_y = a * (g - dot_ag)

        # ----- Adam update -----
        mt = beta1 * mt + (1.0 - beta1) * grad_y
        vt = beta2 * vt + (1.0 - beta2) * (grad_y ** 2)
        mt_hat = mt / (1.0 - beta1 ** it)
        vt_hat = vt / (1.0 - beta2 ** it)
        step = lr * mt_hat / (np.sqrt(vt_hat) + eps_adam)
        y -= step

        # ----- evaluate the true score (fast version) -----
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            if it % 200 == 0:
                print(f"[adam] n={n:5d} iter={it:6d} score={best_score:.6f}")

        # occasional tiny random jitter to escape plateaus
        if (it & 0x1FF) == 0:      # every 512 iterations
            jdx = np.random.randint(n)
            perturb = (np.random.rand() - 0.5) * 1e-3
            a_j = a.copy()
            a_j[jdx] = max(0.0, a_j[jdx] + perturb)
            a_j /= a_j.sum()
            y = np.log(a_j + eps)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Stochastic local search – random mass transfers.
# ----------------------------------------------------------------------
def _local_search(seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Simple random walk on the simplex; accepts only improvements."""
    n = seq.size
    best_seq = seq.copy()
    best_score = _fast_evaluation(best_seq)

    it = 0
    while time.time() < deadline:
        it += 1
        src = np.random.randint(n)
        if best_seq[src] < 1e-12:
            continue
        dst = np.random.randint(n)
        if src == dst:
            continue
        # transfer a random fraction (up to 10 %) of the source mass
        frac = np.random.rand() * 0.10
        eps = frac * best_seq[src]
        new_seq = best_seq.copy()
        new_seq[src] -= eps
        new_seq[dst] += eps
        new_seq = np.maximum(new_seq, 0.0)
        new_seq /= new_seq.sum()
        cur_score = _fast_evaluation(new_seq)
        if cur_score < best_score:
            best_seq = new_seq
            best_score = cur_score
            print(f"[local] iter={it:6d} score={best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Returns a list of non‑negative heights of a step function that (hopefully)
    beats the π/2 ≈ 1.5708 barrier.  The routine uses three stages:
    (1) small‑scale Adam optimisation, (2) up‑sampling + large‑scale Adam,
    (3) a cheap stochastic local search.  The total running time respects
    the supplied ``budget_s`` (default 1000 s).
    """
    np.random.seed(seed)
    start = time.time()
    # safety margin: leave a few seconds for the harness
    total_deadline = start + max(0.0, budget_s - 5)

    # -------------------- Phase 1: small problem --------------------
    n_small = 800
    seq_small = _catalan_sequence(n_small).astype(float)
    seq_small /= seq_small.sum()                     # normalise to sum = 1

    deadline1 = start + (budget_s - 5) * 0.20       # ~20 % of the budget
    if deadline1 > total_deadline:
        deadline1 = total_deadline
    best_small, score_small = _adam_optimize(seq_small, deadline1)
    print(f"[phase1] n={n_small} best score={score_small:.6f}")

    # -------------------- Phase 2: large problem --------------------
    n_big = 8000
    seq_big = _interpolate(best_small, n_big)
    deadline2 = start + (budget_s - 5) * 0.90       # ~70 % of the budget
    if deadline2 > total_deadline:
        deadline2 = total_deadline
    best_big, score_big = _adam_optimize(seq_big, deadline2)
    print(f"[phase2] n={n_big} best score={score_big:.6f}")

    # keep the best of the two phases as a starting point for local search
    if score_small < score_big:
        cur_best_seq = best_small
        cur_best_score = score_small
    else:
        cur_best_seq = best_big
        cur_best_score = score_big

    # -------------------- Phase 3: stochastic refinement --------------------
    deadline3 = total_deadline
    if time.time() < deadline3:
        cur_best_seq, cur_best_score = _local_search(cur_best_seq, deadline3)
        print(f"[final] best score after local search = {cur_best_score:.6f}")

    # Return the best sequence as a plain Python list (evaluate_sequence will clip
    # values >1000 and enforce non‑negativity – our sequence already satisfies both).
    return cur_best_seq.tolist()
