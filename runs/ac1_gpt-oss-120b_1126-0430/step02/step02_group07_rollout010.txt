import numpy as np
import time
from scipy import sparse
from scipy.optimize import linprog

def _normalize_target(vec, target):
    """Scale a non‑negative vector to have L1 norm exactly ``target``."""
    total = vec.sum()
    if total == 0:
        # If all zero fall back to uniform vector
        n = vec.shape[0]
        return np.full(n, target / n)
    return vec * (target / total)

def _conv_matrix(f):
    """
    Build the (2n‑1)×n convolution matrix C such that
    ``C @ b == np.convolve(f, b)``.
    The matrix is stored as a CSR sparse matrix for efficiency.
    """
    n = f.shape[0]
    # Each diagonal –k (k=0..n‑1) contains the constant f[k].
    offsets = -np.arange(n)
    # Broadcast f into an (n,n) array; each row i becomes the diagonal i.
    data = np.broadcast_to(f[:, None], (n, n))
    # Construct sparse matrix
    return sparse.diags(data, offsets, shape=(2 * n - 1, n), format='csr')

def _max_convolution(vec):
    """Return max value of v*v convolution."""
    return np.convolve(vec, vec).max()

def propose_candidate(seed=123, budget_s=950, **kwargs):

            maximise   Σ b_j

        The optimal b is rescaled to the same L1 norm as f and mixed
        into f with a carefully chosen step size.

    2.  A lightweight hill‑climbing refinement that moves a tiny
        amount of mass from indices that contribute most to the current
        maximal convolution entry towards indices that contribute the
        least.  This smoothes the remaining peaks.

    observed for a while.  The returned sequence is normalised to sum
    to 1, which makes the value of the official evaluator equal to the
    internal maximal convolution M.

    no_improve_counter = 0

    for it in range(1, max_outer_iters + 1):
        # time check
        if time.time() - start > budget_s - 5:

        # --------------------------------------------------------------

        t_sched = max(t_end, t_start * (1.0 - it / max_outer_iters))
        # Evaluate a handful of candidate t’s around the schedule.
        candidate_ts = [t_sched * 0.5, t_sched, min(1.0, t_sched * 2.0), 0.1, 0.3]

        if it % 5 == 0:

            conv_f = np.convolve(f, f)

Propose a step‑function that minimises ``evaluate_sequence``.

The algorithm has two stages:

1. **LP‑mass maximisation** –  following the idea from the statement.
   For a vector `f` with  L1‑norm  √(2 n) we compute
   `M = max(conv(f,f))`.  We then solve the linear programme

        maximise   Σ b_j
        subject to b_j ≥ 0,
                   conv(f,b) ≤ M      (one inequality per convolution index)

   The solution `b` is rescaled to the same L1‑norm as `f`
   (denoted `g`).  A convex combination `f←(1‑t)f+t g` is taken,
   where the mixing weight `t` is chosen by a tiny line‑search.
   This step strictly reduces the bound `M` in practice.

2. **Fine‑grained hill‑climbing** –  after every few LP steps we move a
   small amount of mass from the index that contributes most to the
   current maximal convolution entry to the index that contributes the
   least.  This smooths the remaining peaks.

The loop runs until the time budget (≈ 950 s) is exhausted or no
substantial improvement occurs.  The returned list is normalised to
sum to 1, which makes the official evaluator’s score exactly the final
maximal convolution `M`.  The whole search stays within the allowed

def _normalize_target(vec: np.ndarray, target: float) -> np.ndarray:
    """
    Scale a non‑negative vector so that its L1 norm equals ``target``.
    If the input is all zeros, a uniform vector of the required norm is
    returned (this should never happen for a sensible start).
    """
    total = vec.sum()

    for it in range(1, max_outer_iters + 1):
        # ----- time guard ------------------------------------------------
        if time.time() - start > budget_s - 5:
            break

        # ----- current bound ------------------------------------------------
        M = _max_convolution(f)

        # ----- 1) LP: maximise Σ b_j  subject to conv(f,b) ≤ M -------------
        A_ub = _conv_matrix(f)                     # (2n‑1) × n sparse matrix
        b_ub = np.full(2 * n - 1, M, dtype=float)

        # linprog minimises cᵀx, so we minimise –Σb
        c = -np.ones(n, dtype=float)

        # HiGHS interior‑point solver – fast for the size we have
        res = linprog(c,
                      A_ub=A_ub,
                      b_ub=b_ub,
                      bounds=(0, None),
                      method="highs")
        if res.status != 0:
            print(f"[iter {it}] LP failed (status={res.status}) – stop")
            break

        b_opt = res.x
        sum_b = b_opt.sum()
        if sum_b <= 0.0:
            print(f"[iter {it}] LP produced zero mass – stop")
            break

        # ----- 2) Rescale LP solution to the target L1 norm ----------------
        g = b_opt * (target_sum / sum_b)

        # ----- 3) Choose mixing weight via a tiny line‑search ---------------
        # Weight schedule that shrinks linearly
        t_sched = max(t_end, t_start * (1.0 - it / max_outer_iters))
        candidate_ts = [t_sched * 0.5, t_sched, min(1.0, t_sched * 2.0), 0.1, 0.3]

        best_local_f = None
        best_local_M = np.inf
        for t in candidate_ts:
            if not (0.0 < t <= 1.0):
                continue
            f_candidate = (1.0 - t) * f + t * g
            f_candidate = _normalize_target(f_candidate, target_sum)
            M_candidate = _max_convolution(f_candidate)
            if M_candidate < best_local_M:
                best_local_M = M_candidate
                best_local_f = f_candidate

        # ----- 4) Accept the best convex combination if it improves -------
        if best_local_M < M - 1e-12:
            f = best_local_f
            M = best_local_M
            no_improve = 0
            if M < best_M - 1e-12:
                best_M = M
                best_f = f.copy()
                best_score = M
                print(f"[iter {it}] new best M={M:.6f}")
        else:
            no_improve += 1
            # If we have stagnated for a while, inject a tiny random jiggle
            if no_improve >= 10:
                src, tgt = rng.choice(n, size=2, replace=False)
                delta = f[src] * rng.random() * 0.01
                f[src] -= delta
                f[tgt] += delta
                f = _normalize_target(f, target_sum)
                no_improve = 0
                continue

        # ----- 5) Light hill‑climbing refinement every few outer steps -----
        if it % 5 == 0:
            conv_f = np.convolve(f, f)
            peak_idx = int(np.argmax(conv_f))

            # contribution of each element to that peak
            idx_arr = np.arange(n)
            rev = peak_idx - idx_arr
            mask = (rev >= 0) & (rev < n)
            contrib = np.zeros_like(f)
            contrib[mask] = f[idx_arr[mask]] * f[rev[mask]]

            src = int(np.argmax(contrib))
            tgt = int(np.argmin(contrib))
            if src == tgt or f[src] <= 0.0:
                src, tgt = rng.choice(n, size=2, replace=False)

            # move a very small amount of mass
            step = f[src] * rng.random() * 0.02
            f[src] -= step
            f[tgt] += step
            f = _normalize_target(f, target_sum)

    # ------------------------------------------------------------------
    # Return the best sequence found, normalised to sum = 1 (required by
    # the official evaluator).  The score returned by the evaluator
    # equals the internal maximal convolution M.
    # ------------------------------------------------------------------
    final_seq = (best_f / target_sum).tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"[final] best score reported by evaluator {final_score:.6f}")
    return final_seq
