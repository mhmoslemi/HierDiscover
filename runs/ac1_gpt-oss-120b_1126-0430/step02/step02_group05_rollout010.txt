        corr_full = np.fft.irfft(Fa * np.conj(Fw), fft_len)
        start = L - 1

        # exponentiated gradient update
        seq = seq * np.exp(-eta * grad)

        # renorm to sum_target, ensure nonnegativity
        seq = np.maximum(seq, 0.0)
        seq_sum = seq.sum()
        if seq_sum == 0:

        else:

        if score < best_score - 1e-12:
            best_score = score
            best_seq = seq.copy()
            print(f"Iter {it:6d}: improved score {best_score:.8f}, max {max_val:.8f}")
        # parameter schedule

        it += 1
    return best_seq

"""
Propose a step‑function (non‑negative sequence) with a low
evaluate_sequence() score.

Algorithm
---------
1.  Start from the discrete Catalan (arcsine) coefficients – the classic
    construction whose self–convolution is flat and whose score equals
    π/2 ≈ 1.571.

2.  Phase A – Exponentiated‑gradient descent.
    * The objective is the maximal self‑convolution value.
      Because the score is scale‑invariant we keep the L1‑norm fixed.
    * At each iteration we compute the self‑convolution b = a*a.
      A soft‑max (temperature β) converts b into a probability vector
      w that concentrates on the current peak(s).
    * The gradient of the smoothed objective is
          g = 2·corr(a, w)          (corr = linear cross‑correlation)
      which gives the direction in which moving mass reduces the peak.
    * We perform a mirror‑descent (exponentiated‑gradient) step  
          a ← a·exp(−η·g)           (· is element‑wise)
      and renormalise to the original L1‑norm.
    * β is increased slowly (making the soft‑max sharper) and η is
      decreased, allowing large early moves and fine later tuning.
    * The best sequence seen is stored.

3.  Phase B – Fine‑tuning with the “sink” heuristic from the reference
    implementation.  It moves a modest fraction of the mass from the
    “high‑impact’’ positions (large weight w_i = a_{k‑i}) to a small
    tail where the weight is tiny, thereby shrinking the maximal peak.
    The parameters are chosen more aggressively than the original code.

Both phases run until the time budget (≈ 1000 s) is exhausted.  The
final sequence is returned as a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast (un‑checked) evaluation – identical to evaluate_sequence but
# without the security checks.  Uses FFT for O(n log n) convolution.
# ----------------------------------------------------------------------
def _fast_evaluation(seq: np.ndarray) -> float:
    n = seq.shape[0]
    if n == 0:
        return float("inf")
    # FFT length: next power of two ≥ 2·n‑1
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Catalan (arcsine) coefficients: a_0 = 1,
# a_{i+1} = a_i * (2*i+1)/(2*i+2)
# ----------------------------------------------------------------------
def _catalan_sequence(length: int) -> np.ndarray:
    seq = np.empty(length, dtype=float)
    a = 1.0
    seq[0] = a
    for i in range(1, length):
        # a_i = a_{i-1} * (2i-1)/(2i)
        a *= (2 * i - 1) / (2.0 * i)
        seq[i] = a
    return seq


# ----------------------------------------------------------------------
# Phase A – exponentiated‑gradient optimisation.
# ----------------------------------------------------------------------
def _optimise_eg(seq: np.ndarray, deadline: float) -> np.ndarray:
    """Mirror‑descent on the smoothed max‑convolution."""
    n = seq.shape[0]
    L = 2 * n - 1                # length of the self‑convolution
    # FFT length that can handle both convolution (2n‑1) and
    # correlation (n+L‑1 = 3n‑2) without wrap‑around.
    fft_len = 1 << ((3 * n - 2).bit_length())

    sum_target = seq.sum()       # the L1‑norm must stay constant
    best_seq = seq.copy()
    best_score = _fast_evaluation(best_seq)

    # hyper‑parameters (will be annealed)
    beta = 5.0                    # soft‑max temperature
    eta = 1.0                     # step size for the EG update
    it = 0

    while time.time() < deadline:
        # --- self‑convolution ------------------------------------------------
        Fa = np.fft.rfft(seq, fft_len)
        conv = np.fft.irfft(Fa * Fa, fft_len)[:L]
        max_val = conv.max()

        # --- soft‑max over convolution entries ------------------------------
        # w is a probability vector concentrating on the current peak(s)
        w = np.exp(beta * (conv - max_val))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # --- gradient = 2·corr(seq, w) --------------------------------------
        Fw = np.fft.rfft(w, fft_len)
        corr_full = np.fft.irfft(Fa * np.conj(Fw), fft_len)
        start = L - 1                     # index where shift = 0 starts
        grad = 2.0 * corr_full[start:start + n]

        # --- exponentiated‑gradient step ------------------------------------
        seq = seq * np.exp(-eta * grad)   # element‑wise
        # keep non‑negative (exp never produces negative)
        seq = np.maximum(seq, 0.0)

        # --- renormalise to the original L1 norm ---------------------------
        total = seq.sum()
        if total == 0.0:                  # unlikely, but guard against it
            seq = np.full(n, sum_target / n)
        else:
            seq *= (sum_target / total)

        # --- evaluate --------------------------------------------------------
        score = _fast_evaluation(seq)

        if score < best_score - 1e-12:
            best_score = score
            best_seq = seq.copy()
            print(f"EG iter {it:6d}: improved score = {best_score:.8f}, "
                  f"max conv = {max_val:.8f}")

        # --- simple schedules ------------------------------------------------
        if it % 200 == 0 and it > 0:
            beta = min(beta * 1.3, 200.0)   # sharpen the soft‑max
            eta = max(eta * 0.9, 0.001)     # smaller steps later

        # occasional random jitter to escape plateaux
        if np.random.rand() < 0.01:
            idx = np.random.randint(n)
            jitter = (np.random.rand() - 0.5) * seq[idx] * 0.05
            seq[idx] = max(0.0, seq[idx] + jitter)
            seq *= (sum_target / seq.sum())

        it += 1

    return best_seq


# ----------------------------------------------------------------------
# Phase B – the “sink” heuristic from the reference implementation,
# but with more aggressive parameters.
# ----------------------------------------------------------------------
def _optimise_sink(seq: np.ndarray, deadline: float,
                   sink_percent: float = 2.0,
                   rel_target_reduce: float = 0.02) -> np.ndarray:
    """Iteratively move mass from high‑impact to low‑impact positions."""
    n = seq.size
    fft_len = 1 << ((2 * n - 1).bit_length())

    best_seq = seq.copy()
    best_score = _fast_evaluation(best_seq)
    it = 0

    while time.time() < deadline:
        # current self‑convolution
        fa = np.fft.rfft(seq, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_val = conv.max()
        # pick a random index among the tied maxima
        max_idxs = np.flatnonzero(conv >= max_val - 1e-12)
        k = int(np.random.choice(max_idxs))

        # weight vector w_i = a_{k-i} (zero out‑of‑range entries)
        i_arr = np.arange(n)
        j_arr = k - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        w = np.zeros_like(seq)
        w[mask] = seq[j_arr[mask]]

        # choose sink set – positions with smallest weight
        thresh = np.percentile(w, sink_percent)
        sink_mask = w <= thresh
        sink_idxs = np.flatnonzero(sink_mask)
        if sink_idxs.size == 0:
            sink_idxs = np.array([int(np.argmin(w))])
            sink_mask = np.zeros_like(w, dtype=bool)
            sink_mask[sink_idxs[0]] = True

        w_min = w[sink_idxs].min()
        excess = seq * (w - w_min)
        total_excess = excess.sum() - excess[sink_mask].sum()

        if total_excess <= 0:
            # nothing helpful – tiny random jitter
            jitter_i = np.random.randint(n)
            jitter = (np.random.rand() - 0.5) * 1e-4
            new_seq = seq.copy()
            new_seq[jitter_i] = max(0.0, new_seq[jitter_i] + jitter)
        else:
            target_reduction = rel_target_reduce * max_val
            lam = target_reduction / total_excess
            lam = max(min(lam, 1.0), 1e-6)

            delta = lam * seq
            delta[sink_mask] = 0.0
            moved = delta.sum()
            add_per_sink = moved / sink_idxs.size

            new_seq = seq - delta
            new_seq[sink_idxs] += add_per_sink

        new_score = _fast_evaluation(new_seq)

        if new_score < best_score - 1e-12:
            seq = new_seq
            best_seq = new_seq.copy()
            best_score = new_score
            print(f"Sink iter {it:6d}: improved score = {best_score:.8f}, "
                  f"max = {max_val:.8f}")
        else:
            # occasional random shake‑up
            if np.random.rand() < 0.02:
                i, j = np.random.randint(n, size=2)
                if i != j and seq[i] > 0:
                    eps = min(seq[i], np.random.rand() * seq[i] * 0.1)
                    tmp = seq.copy()
                    tmp[i] -= eps
                    tmp[j] += eps
                    tmp_score = _fast_evaluation(tmp)
                    if tmp_score < best_score:
                        seq = tmp
                        best_seq = tmp.copy()
                        best_score = tmp_score
                        print(f"Sink shake {it:6d}: improved score = {best_score:.8f}")
        it += 1

    return best_seq


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights whose evaluation score
    is hoped to be well below the π/2 ≈ 1.571 bound.
    """
    np.random.seed(seed)
    # leave a small safety margin for the harness
    deadline = time.time() + budget_s - 5.0

    # length ≈ few thousand provides a good trade‑off between quality and speed
    n = 5000

    # 1. start from the Catalan / arcsine seed (sum is O(√n))
    seq = _catalan_sequence(n)

    # 2. Phase A – exponentiated‑gradient descent (≈ 70 % of the budget)
    eg_deadline = time.time() + (budget_s * 0.70) - 2.0
    seq = _optimise_eg(seq, eg_deadline)

    # 3. Phase B – aggressive sink‑heuristic fine‑tuning (remaining time)
    sink_deadline = time.time() + (budget_s * 0.30) - 2.0
    seq = _optimise_sink(seq, sink_deadline,
                         sink_percent=2.0,
                         rel_target_reduce=0.02)

    # 4. Return the best sequence discovered (plain Python list)
    return seq.tolist()
