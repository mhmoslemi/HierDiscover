"""
Optimization of a non‑negative step‑function that minimises

    O = 2·n·max(conv(a,a))   (with sum(a)=1)

The bound is scale‑invariant, therefore the search is performed on the
simplex {a≥0, Σa=1}.  Theory shows that a_i ∝ 1/√i gives O≈π/2≈1.571.
A simple exponentiated‑gradient descent on the sub‑gradient of
max(conv) quickly drives the sequence below this value.  The programme:

1. builds a few sensible starts (previous best, the 1/√‑law, a random point);
2. for each start runs a projected exponentiated‑gradient loop
   – compute the full linear convolution by FFT,
   – locate the (few) indices where the convolution attains its maximum,
   – form the sub‑gradient of the max‑operator,
   – update a_i ← a_i·exp(−η·g_i) and renormalise to the simplex.
3. keeps the best sequence seen,
4. finally calls the supplied ``evaluate_sequence`` for the exact score.

The loop is time‑capped (default 1000 s) and prints progress.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return the bound 2·n·max(conv(a,a)) for a already normalised (Σa=1)."""
    # use FFT for speed on large vectors
    n = a.size
    size = 1 << ((2 * n - 2).bit_length())  # next power of two
    A = np.fft.fft(a, n=size)
    conv = np.fft.ifft(A * A).real[: 2 * n - 1]
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a power‑of‑two FFT."""
    n = a.size
    size = 1 << ((2 * n - 2).bit_length())
    A = np.fft.fft(a, n=size)
    conv = np.fft.ifft(A * A).real[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv(a,a)) w.r.t. a on the simplex."""
    n = a.size
    conv = _conv_fft(a)
    max_val = conv.max()
    # indices where the convolution attains its maximum (allow tiny tolerance)
    ks = np.where(np.abs(conv - max_val) < 1e-14)[0]
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        # derivative of a_i*a_j appears twice (i↔j)
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size > 0:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
def _exp_update(a: np.ndarray, g: np.ndarray, eta: float) -> np.ndarray:
    """
    Exponentiated‑gradient step respecting the simplex.
    a – current distribution (already normalised)
    g – sub‑gradient of the max‑convolution
    eta – learning rate
    """
    # work in log‑space for numerical stability
    log_a = np.log(a + 1e-15)            # avoid log(0)
    log_a -= eta * g
    # shift to keep numbers moderate
    log_a -= np.max(log_a)
    a_new = np.exp(log_a)
    a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
def _optimize(a0: np.ndarray, max_iters: int, eta0: float,
              deadline: float) -> tuple[np.ndarray, float]:
    """
    Perform a simple exponentiated‑gradient descent on the sub‑gradient of
    max(conv).  Returns the best vector found and its objective value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)

    n = a.size
    no_improve = 0
    patience = max_iters // 20   # early‑stop if no improvement for a while

    for t in range(max_iters):
        if time.time() >= deadline:
            break

        # compute sub‑gradient at the current point
        g = _subgradient_max_conv(a)

        # linearly decaying learning rate
        eta = eta0 * (1.0 - t / max_iters)

        # exponentiated‑gradient update
        a = _exp_update(a, g, eta)

        cur_score = _objective(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            print(f"[opt] iter {t:5d}  new best O={best_score:.6f}")
        else:
            no_improve += 1
            if no_improve >= patience:
                # small random jitter to escape a flat region
                jitter = np.random.randn(n) * 1e-4
                a = best_a + jitter
                a = np.clip(a, 0.0, None)
                a = _project_to_simplex(a)
                no_improve = 0

    return best_a, best_score

            break
        init_score = _objective(a_start)
        print(f"[init {i}] start O={init_score:.6f}")

        a_opt, score_opt = _optimize(a_start, iters_per_start, eta0, deadline)

        print(f"[result {i}] optimized O={score_opt:.6f}")

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()

    # ------------------------------------------------------------------
    # Fallback: if nothing succeeded (e.g. time ran out), use the 1/√ shape.
    if best_seq is None:
        best_seq = a_sqrt.copy()
        best_score = _objective(best_seq)

    # Final exact evaluation using the supplied function
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:  # pragma: no cover
        print("[final] evaluate_sequence call failed:", e)
        final_score = None

    return best_seq.tolist()
