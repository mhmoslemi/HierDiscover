def simplex_projection(v):
    """Project v onto the probability simplex (sum=1, v>=0)."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

from scipy.optimize import linprog
import numpy as np
import scipy.sparse as sp

def lp_step(seq_fixed):
    n = len(seq_fixed)
    conv_fixed = np.convolve(seq_fixed, seq_fixed)
    # number of constraints = 2*n - 1
    m = 2*n - 1
    
    # Build sparse matrix A_ub (m x (n+1))
    # We'll fill rows for each k:
    rows = []
    cols = []
    data = []
    for k in range(m):
        lo = max(0, k-(n-1))
        hi = min(k, n-1)
        # i from lo..hi, j = k-i
        # coefficient for a_i = 2 * seq_fixed[j]
        for i in range(lo, hi+1):
            j = k - i
            rows.append(k)
            cols.append(i)
            data.append(2.0 * seq_fixed[j])
        # coefficient for M
        rows.append(k)
        cols.append(n)  # last variable index
        data.append(-1.0)
    A_ub = sp.csr_matrix((data, (rows, cols)), shape=(m, n+1))
    b_ub = conv_fixed  # shape (m,)
    
    # Equality: sum a_i = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: a_i >=0, no upper bound, M >=0
    bounds = [(0, None)] * n + [(0, None)]
    
    c = np.zeros(n+1)
    c[-1] = 1.0  # minimize M
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True, 'time_limit': 30})
    if res.success:
        a_new = res.x[:n]
        M_new = res.x[-1]  # not needed
        return a_new, M_new
    else:
        return None, None

"""
Low‑peak step function generator.

The algorithm consists of three stages:

1. **Seed** – a Catalan (discrete arcsine) distribution, normalised to sum = 1.
2. **Successive convex approximation (SCA)** – each iteration linearises the
   quadratic constraints `a*a <= M` around the current point and solves a
   linear program that minimises the bound `M`.  The LP variables are the
   probabilities `a_i` (non‑negative, sum = 1) and the auxiliary bound `M`.
   A convex combination of the old and the LP solution is used if the raw
   LP step does not improve the true objective.
3. **Fine‑tuning** – a projected exponentiated‑gradient descent on the simplex
   (the same routine used previously) is run for the remaining time.

All operations stay on the probability simplex, so the evaluation metric
`2*n*max(conv)/sum(a)^2` is invariant to scaling.  The routine respects the
global 1000 s budget and prints intermediate scores.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Fast evaluation (identical to evaluate_sequence but without the security
# checks and using an FFT based convolution).  The result equals the score
# returned by evaluate_sequence for a normalised sequence (sum == 1).
# ----------------------------------------------------------------------
def _fast_evaluation(seq: np.ndarray) -> float:
    n = seq.size
    if n == 0:
        return np.inf
    # FFT size = next power of two >= 2*n-1
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Catalan (arcsine) seed.
# a_0 = 1, a_{i+1} = a_i * (2*i+1) / (2*i+2)
# ----------------------------------------------------------------------
def _catalan_sequence(length: int) -> np.ndarray:
    seq = np.empty(length, dtype=float)
    a = 1.0
    seq[0] = a
    for i in range(1, length):
        a *= (2 * i - 1) / (2.0 * i)
        seq[i] = a
    return seq


# ----------------------------------------------------------------------
# Successive Convex Approximation (linearised LP) step.
# Returns a new probability vector and the bound M found by the LP.
# ----------------------------------------------------------------------
def _sca_lp_step(a_fixed: np.ndarray):
    """
    Linearise the quadratic constraints a*a <= M around ``a_fixed`` and
    solve the resulting LP:
        minimise M
        subject to 2*(a * a_fixed)_k - M <= (a_fixed * a_fixed)_k   for all k
        a_i >= 0 ,  sum_i a_i = 1 ,  M >= 0 .
    """
    n = a_fixed.size
    conv_fixed = np.convolve(a_fixed, a_fixed)          # length 2*n-1
    m = 2 * n - 1

    # Build the sparse inequality matrix.
    # Row k corresponds to constraint for convolution entry k.
    rows = []
    cols = []
    data = []

    # Coefficients for a_i : 2 * a_fixed[j] where j = k - i.
    for k in range(m):
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        # i runs from lo..hi, j = k-i
        for i in range(lo, hi + 1):
            j = k - i
            rows.append(k)
            cols.append(i)
            data.append(2.0 * a_fixed[j])
        # coefficient for the auxiliary variable M (last column)
        rows.append(k)
        cols.append(n)      # index of M
        data.append(-1.0)

    A_ub = sp.csr_matrix((data, (rows, cols)), shape=(m, n + 1))
    b_ub = conv_fixed

    # Equality constraint: sum_i a_i = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    # Objective: minimise M
    c = np.zeros(n + 1)
    c[-1] = 1.0

    # Bounds: a_i >= 0, M >= 0
    bounds = [(0.0, None)] * n + [(0.0, None)]

    # Solve the LP with HiGHS (fast and robust)
    res = opt.linprog(
        c,
        A_ub=A_ub,
        b_ub=b_ub,
        A_eq=A_eq,
        b_eq=b_eq,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )

    if not res.success:
        # LP should always be feasible; if not, signal failure.
        return None, None

    a_new = res.x[:n]
    M_new = res.x[-1]

    # Numerical clean‑up: enforce non‑negativity and exact normalisation.
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s <= 0.0:
        return None, None
    a_new /= s
    return a_new, M_new


# ----------------------------------------------------------------------
# Exponentiated‑gradient refinement on the simplex (same as before, but
# slightly re‑parameterised for clarity).
# ----------------------------------------------------------------------
def _eg_refine(seq: np.ndarray, deadline: float) -> np.ndarray:
    """
    Mirror‑descent on the probability simplex using the sub‑gradient of the
    maximal self‑convolution.  ``seq`` must already be normalised (sum = 1).
    The routine runs until ``deadline`` (unix timestamp) is reached.
    """
    n = seq.size
    best_seq = seq.copy()
    best_score = _fast_evaluation(best_seq)

    fft_len = 1 << ((2 * n - 1).bit_length())
    it = 0
    while time.time() < deadline:
        # convolution and its maximal value
        fa = np.fft.rfft(seq, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        M = conv.max()

        # active indices (those attaining the maximum up to tolerance)
        tol = 1e-12
        active = np.where(conv >= M - tol)[0]

        # sub‑gradient of the max‑operator:
        # g_i = 2 * sum_{k in active} a_{k-i}
        g = np.zeros_like(seq)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            i_range = np.arange(lo, hi + 1)
            j_range = k - i_range
            g[i_range] += seq[j_range]
        g *= 2.0

        # step‑size schedule (decays as 1/sqrt(t))
        eta = 0.8 / np.sqrt(it + 1)

        # multiplicative update (exponentiated gradient)
        seq = seq * np.exp(-eta * g)

        # projection onto the simplex (ensures non‑negativity & sum = 1)
        seq = np.maximum(seq, 1e-15)
        seq_sum = seq.sum()
        if seq_sum == 0.0:
            seq = np.full(n, 1.0 / n)
        else:
            seq /= seq_sum

        # evaluate
        score = _fast_evaluation(seq)
        if score < best_score - 1e-12:
            best_seq = seq.copy()
            best_score = score
            if it % 400 == 0:
                print(f"[EG] iter {it:5d}  score {best_score:.8f}")

        # occasional tiny random jitter to escape tiny plateaus
        if np.random.rand() < 0.001:
            idx = np.random.randint(n)
            seq[idx] = max(0.0, seq[idx] + np.random.randn() * 1e-3)
            seq /= seq.sum()

        it += 1
    return best_seq


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Build a non‑negative step function with a low evaluation score.
    The routine works within ``budget_s`` seconds (default 1000) and returns
    a plain Python list of floats.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0          # safety margin for the harness

    # ------------------------------------------------------------------
    # 1.  Choose a moderate length – large enough for a good approximation,
    #     small enough for the LP to stay fast.
    # ------------------------------------------------------------------
    n = 2000

    # ------------------------------------------------------------------
    # 2.  Initialise with the Catalan (discrete arcsine) coefficients.
    # ------------------------------------------------------------------
    seq = _catalan_sequence(n).astype(float)
    seq /= seq.sum()                         # normalise to a probability vector

    # Small random jitter to break symmetry (helps the LP escape trivial solutions)
    seq += 1e-4 * (np.random.rand(n) - 0.5)
    seq = np.maximum(seq, 0.0)
    seq /= seq.sum()

    best_seq = seq.copy()
    best_score = _fast_evaluation(best_seq)
    print(f"[init] n={n}  score={best_score:.8f}")

    # ------------------------------------------------------------------
    # 3.  Successive Convex Approximation (LP) phase – use roughly 40 % of the budget.
    # ------------------------------------------------------------------
    sca_deadline = start + 0.45 * (budget_s - 5.0)
    sca_iter = 0
    while time.time() < sca_deadline and sca_iter < 12:
        a_lp, M_lp = _sca_lp_step(seq)
        if a_lp is None:
            break

        cand_score = _fast_evaluation(a_lp)

        if cand_score < best_score - 1e-12:
            seq = a_lp
            best_seq = a_lp.copy()
            best_score = cand_score
            print(f"[SCA] iter {sca_iter:2d}  improved score {best_score:.8f}")
        else:
            # try a convex combination (0.5/0.5) – often smooths the transition
            combo = 0.5 * seq + 0.5 * a_lp
            combo = np.maximum(combo, 0.0)
            combo /= combo.sum()
            combo_score = _fast_evaluation(combo)
            if combo_score < best_score - 1e-12:
                seq = combo
                best_seq = combo.copy()
                best_score = combo_score
                print(f"[SCA] iter {sca_iter:2d}  combo improved score {best_score:.8f}")
            else:
                # No progress – exit early
                print(f"[SCA] iter {sca_iter:2d}  no improvement, stopping")
                break

        sca_iter += 1

    # ------------------------------------------------------------------
    # 4.  Fine‑tune the result with exponentiated‑gradient descent.
    # ------------------------------------------------------------------
    seq = _eg_refine(seq, deadline)
    final_score = _fast_evaluation(seq)
    if final_score < best_score:
        best_seq = seq.copy()
        best_score = final_score
    print(f"[final] best score = {best_score:.8f}")

    # Return as a plain Python list
    return best_seq.tolist()
