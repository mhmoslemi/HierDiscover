"""
Optimise the step‑function heights for the `evaluate_sequence` target.

The algorithm consists of three stages:

1. **Initialisation** – a small pool of plausible seeds (power‑law,
   exponential decay, uniform, random Dirichlet).  All seeds are
   normalised to sum = 1 (the objective is scale‑invariant).

2. **Multi‑active Adam descent** – for each seed we run a projected Adam
   optimiser.  At every iteration the full autocorrelation
   `c = a * a` (via an FFT) is computed, the indices whose value is
   within a tiny fraction of the current maximum are collected (the
   “active set”) and their sub‑gradients are summed.  The Adam step is
   applied and the iterate is projected onto the probability simplex.
   The step size schedule and momentum help to escape plateaus.

3. **Final evaluation** – the best vector found (by the cheap
   `2·n·max(c)` bound) is passed to the provided `evaluate_sequence`
   function for the exact score and returned.

The whole routine respects the 1000 s budget, prints occasional progress
messages and returns a list of non‑negative heights.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex (sum = 1, all entries ≥ 0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # all non‑positive -> uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length = fft_len)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.size - 1]
    return conv


def _conv_max_and_arg(a: np.ndarray, fft_len: int):
    """Return (max value, arg‑max) of the autocorrelation of a."""
    conv = _conv_full_fft(a, fft_len)
    maxb = conv.max()
    arg = int(conv.argmax())
    return maxb, arg, conv


def _subgradient_multi(a: np.ndarray, active_idxs: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the maximum convolution entry summed over the active set.
    For each active index k the contribution is:
        g_i += 2·a_{k‑i}  (when the index is valid).
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in active_idxs:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i = np.arange(lo, hi + 1, dtype=np.int64)
        g[i] += 2.0 * a[k - i]
    return g


def _power_law_seq(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}, normalised to sum = 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    seq = idx ** (-p)
    seq /= seq.sum()
    return seq


def _exponential_seq(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i), normalised to sum = 1."""
    idx = np.arange(n, dtype=np.float64)
    seq = np.exp(-lam * idx)
    seq /= seq.sum()
    return seq


def _candidate_initialisers(n: int):
    """Yield a handful of promising starting vectors."""
    # power‑law family (broad sweep)
    for p in (0.45, 0.50, 0.55, 0.60, 0.65, 0.70):
        yield _power_law_seq(n, p)
    # exponential family
    for lam in (0.001, 0.0025, 0.005, 0.01):
        yield _exponential_seq(n, lam)
    # uniform
    yield np.full(n, 1.0 / n)
    # a few random Dirichlet samples (very rough, may help escape)
    for _ in range(3):
        yield np.random.dirichlet(np.ones(n))


# ----------------------------------------------------------------------
# Core optimisation routine (Adam with multi‑active sub‑gradient)
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray,
                   n: int,
                   fft_len: int,
                   deadline: float,
                   verbose: bool = False):
    """
    Projected Adam optimisation of 2·n·max(a*a) for a single seed.
    Returns the best vector (list) and its cheap score.
    """
    a = a0.copy()
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    # learning‑rate (tuned for this scale)
    lr = 0.5

    best_a = a.copy()
    best_score = float("inf")
    t = 0
    # we cap the active‑set size to keep O(n) work per iteration
    K_MAX = 8
    # relative window for the active set (indices whose value is within
    # this fraction of the maximum are considered)
    REL_THR = 0.9995

    while time.time() < deadline:
        t += 1

        # full autocorrelation via FFT (cheap O(n log n))
        conv = _conv_full_fft(a, fft_len)
        maxb = conv.max()
        cheap_score = 2.0 * n * maxb

        # keep the best seen so far
        if cheap_score < best_score:
            best_score = cheap_score
            best_a = a.copy()
            if verbose:
                print(f"[Iter {t:06d}] NEW BEST cheap = {cheap_score:.6f}")

        # ----- active set -------------------------------------------------
        # grab the K_MAX largest entries (sub‑gradient points)
        # `np.argpartition` is O(n); we then sort the tiny slice.
        if K_MAX >= conv.shape[0]:
            active = np.arange(conv.shape[0])
        else:
            cand = np.argpartition(-conv, K_MAX)[:K_MAX]
            # sort them descending (not strictly necessary)
            active = cand[np.argsort(-conv[cand])]

        # optionally shrink to those within a tiny relative band
        # (helps keep the set small when many entries are almost equal)
        thr_val = maxb * REL_THR
        active = active[conv[active] >= thr_val]

        # ----- multi‑active sub‑gradient ---------------------------------
        g = _subgradient_multi(a, active)

        # ----- Adam update ------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a = _simplex_projection(a_new)

        # occasional tiny random jitter to shake out of plateaus
        if t % 2000 == 0 and np.random.rand() < 0.15:
            jitter = 0.01 * np.random.rand(n)
            a = _simplex_projection(a + jitter)

    return best_a, best_score
