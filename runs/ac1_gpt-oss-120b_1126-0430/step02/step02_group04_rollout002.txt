"""
Exponentiated‑gradient descent on the exact L∞‑norm of the auto‑convolution.

The objective for a non‑negative vector a with sum = 1 is  
    F(a) = 2·n·max_k (a * a)_k   (the factor 2·n comes from evaluate_sequence).

We minimise F by mirror‑descent on the probability simplex using the
Kullback‑Leibler geometry (a.k.a. exponentiated gradient).  The (sub‑)gradient
of the true maximum is obtained by a convolution with a binary mask that
selects all positions attaining the current maximal convolution value.
Each iteration requires two FFT‑based convolutions (a*a and a*mask_rev) and
runs in O(n log n).  The algorithm starts from a few sensible seeds
(Catalan‑like, geometric decays, a random Dirichlet and, if available,
the previous best `height_sequence_1`).  It adapts the step‑size,
perturbs the current iterate when progress stalls and keeps the best
vector seen before the time budget expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------


def _catalan_sequence(n: int) -> np.ndarray:
    """First n Catalan‑like coefficients a_i = C(2i,i)/4**i, normalised to sum = 1."""
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * (i - 1) + 1) / (2.0 * i)
    a /= a.sum()
    return a


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx = 1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u > cssv / np.arange(1, n + 1))[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of real vectors x and y using a real FFT."""
    n = x.size + y.size - 1
    size = _next_pow2(n)
    X = np.fft.rfft(x, size)
    Y = np.fft.rfft(y, size)
    Z = X * Y
    conv = np.fft.irfft(Z, size)[:n]
    return conv.real


def _eval_score(a: np.ndarray) -> float:
    """Exact value of evaluate_sequence for a non‑negative simplex vector a."""
    n = a.size
    b = _convolve_fft(a, a)
    max_b = b.max()
    return float(2.0 * n * max_b)  # sum(a)=1 is guaranteed

    """

    # small safety margin to guarantee returning before the external timeout
    deadline = time.time() + budget_s - 4.0

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    n = 8000

    candidates = []

    decay = 0.9997                 # exponential decay per iteration

        a_new = np.exp(log_a)
