    """Return log‑it representation that yields ``a`` when passed through soft‑max."""
    return np.log(a + eps)

    """Smallest power‑of‑two ≥ 2·n‑1, good for real FFTs."""
    return 1 << ((2 * n - 1).bit_length())

def _conv_max_fft(a: np.ndarray, fft_len: int) -> tuple[float, int]:
    """Maximum entry of a*a using an FFT of length ``fft_len``."""
    # real‑valued FFT (rfft) is about twice as fast as complex FFT
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.size - 1]
    idx = int(conv.argmax())
    return float(conv.max()), idx

def _subgradient_max_conv(a: np.ndarray, max_idx: int) -> np.ndarray:
    """
    Sub‑gradient of ``max_k (a*a)[k]`` at the active index ``max_idx``.
    For the active offset k we have (∂/∂a_i) (a*a)[k] = 2·a_{k‑i}
    whenever the index is valid.
    """
    n = a.shape[0]
    g = np.zeros_like(a)

    lo = max(0, max_idx - (n - 1))
    hi = min(max_idx, n - 1)
    i = np.arange(lo, hi + 1, dtype=np.int64)
    g[i] = 2.0 * a[max_idx - i]
    return g

    """Linear interpolation of a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)

    new_a = np.maximum(new_a, 0.0)

    if total == 0.0:

    else:
        new_a /= total
    return new_a

The program searches for a non‑negative step‑function (a list of heights) that minimises

```
evaluate_sequence(seq) = 2·n·max_k (seq * seq)[k] / (sum(seq))² .
```

Because the objective is *scale‑invariant* we may restrict ourselves to probability
vectors (`∑ a_i = 1 , a_i ≥ 0`).  
For a probability vector `a` the score reduces to the cheap proxy  

```
score(a) = 2·n·max_k (a*a)[k]                (1)
```

The term `max_k (a*a)[k]` is a convex piece‑wise quadratic function.  
We optimise it on the probability simplex using **Adam** (adaptive‑momentum
stochastic optimisation) together with a **soft‑max parametrisation** of the
simplex.  The soft‑max automatically enforces `a_i ≥ 0` and `∑ a_i = 1`,
so no explicit projection is required.

Key ingredients

* **FFT based convolution** – `max_k (a*a)[k]` is obtained in `O(n log n)` time.
* **Sub‑gradient** – for the active index `k*` the gradient w.r.t. `a` is  
  `g_i = 2·a_{k*‑i}` (zero if the index is out of range).
* **Mirror‑gradient** – the gradient in the log‑it space (the space of the
  soft‑max parameters) is  
  `∇_z L = a ⊙ (g - μ)` where `μ = a·g`.  This is the natural gradient for the
  KL‑divergence on the simplex.
* **Adam with back‑tracking line‑search** – a few halvings of the step size are
  tried to guarantee a descent step.
* **Random perturbations** – occasional small noise helps escape plateaus.
* **Progressive support growth** – we start with a modest length (≈ 5 000),
  optimise, then linearly interpolate the best distribution to a larger size
  (up to 20 000) and continue optimisation.  Larger support generally yields a
  smaller asymptotic bound.
* **Fallback to the globally supplied sequence** (`height_sequence_1`) if it
  exists, otherwise a power‑law scan for a good initial point.

The algorithm respects the 1000 s wall‑clock budget, keeps the best sequence
encountered (according to the cheap proxy (1)), and finally evaluates it with
the exact `evaluate_sequence` before returning it.

        n = 5000
        use_global = False

    n_max = 20000  # maximum support we will ever grow to

    # ---------------------------------------------------------------
    # 1. Initial distribution.
    # ---------------------------------------------------------------
    if use_global and init_global is not None:
        a = init_global.copy()
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / a.shape[0])
        else:
            a /= s
        print(f"Initial sequence taken from global variable, length = {n}")
    else:
        # Scan a grid of power‑law exponents, keep the best cheap score.
        p_grid = np.linspace(0.55, 0.70, 81)  # modest grid
        best_a = None
        best_score = float("inf")
        fft_len = _fft_len_for_n(n)
        for p in p_grid:
            if time.time() >= deadline:
                break
            cand = _power_law_distribution(n, p)
            maxb, _ = _conv_max_fft(cand, fft_len)
            score = 2.0 * n * maxb            # cheap proxy (∑a = 1)
            if score < best_score:
                best_score = score
                best_a = cand
                best_p = p
        if best_a is None:
            best_a = _power_law_distribution(n, 0.60)
            best_score = 2.0 * n * _conv_max_fft(best_a, _fft_len_for_n(n))[0]
            best_p = 0.60
        a = best_a
        print(f"Power‑law initialise: best p = {best_p:.5f}, cheap score = {best_score:.6f}")

    # ---------------------------------------------------------------
    # 2. Log‑it parametrisation + Adam state.
    # ---------------------------------------------------------------
    eps = 1e-16
    z = _logits_from_distribution(a, eps)

    m = np.zeros_like(z)      # first‑moment estimate
    v = np.zeros_like(z)      # second‑moment estimate
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    base_alpha = 0.2          # base learning‑rate (scaled later)

    # ---------------------------------------------------------------
    # 3. Main optimisation loop.
    # ---------------------------------------------------------------
    best_seq = a.copy()
    best_score = 2.0 * n * _conv_max_fft(a, _fft_len_for_n(n))[0]

    it = 0
    last_improve_iter = 0
    report_every = 500

    while time.time() < deadline and it < 1_000_000:
        it += 1

        # probability vector from logits
        a = _softmax(z)

        # cheap convolution via FFT
        fft_len = _fft_len_for_n(n)
        maxb, max_idx = _conv_max_fft(a, fft_len)
        cheap_score = 2.0 * n * maxb

        # keep the best ever seen
        if cheap_score < best_score:
            best_score = cheap_score
            best_seq = a.copy()
            last_improve_iter = it
            if it % report_every == 0:
                print(f"[Iter {it:07d}] NEW BEST cheap = {cheap_score:.6f}")

        # sub‑gradient of the active convolution entry
        g_a = _subgradient_max_conv(a, max_idx)

        # gradient in the log‑it space (mirror descent with KL divergence)
        mu = np.dot(g_a, a)
        grad_z = a * (g_a - mu)

        # Adam moment updates (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # direction (negative Adam step)
        step_dir = - m_hat / (np.sqrt(v_hat) + eps_adam)

        # basic step‑size schedule: inverse‑sqrt decay
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)

        # -----------------------------------------------------------
        # Back‑tracking line search (up to 4 halvings)
        # -----------------------------------------------------------
        accepted = False
        trial_alpha = alpha_t
        for _ in range(4):
            z_new = z + trial_alpha * step_dir
            # keep logits in a safe range to avoid overflow in exp
            z_new = np.clip(z_new, -50.0, 50.0)
            a_new = _softmax(z_new)
            maxb_new, _ = _conv_max_fft(a_new, _fft_len_for_n(n))
            new_score = 2.0 * n * maxb_new
            if new_score <= cheap_score:
                z = z_new
                accepted = True
                break
            trial_alpha *= 0.5

        if not accepted:
            # no descent found – still take a modest step to keep moving
            z += alpha_t * step_dir
            z = np.clip(z, -50.0, 50.0)

        # -----------------------------------------------------------
        # Random perturbations (helps escape plateaus)
        # -----------------------------------------------------------
        if it % 5000 == 0 and np.random.rand() < 0.2:
            perturb = 0.01 * np.random.randn(n)
            z += perturb
            # damp momentum to avoid stale direction
            m *= 0.5
            v *= 0.5

        # -----------------------------------------------------------
        # Progressive enlargement of the support (if time permits)
        # -----------------------------------------------------------
        if (it % 20000 == 0) and (n < n_max) and (time.time() + 30 < deadline):
            new_n = min(n_max, int(n * 1.5) + 1)
            print(f"[Iter {it:07d}] Expanding support: {n} → {new_n}")
            # Interpolate the best distribution obtained so far
            a = _interpolate_distribution(best_seq, new_n)
            # Re‑initialise logits and Adam state for the larger problem
            z = _logits_from_distribution(a, eps)
            m = np.zeros_like(z)
            v = np.zeros_like(z)
            n = new_n
            # Reset counters for the new problem
            it = 0
            last_improve_iter = 0
            # Update best_score for the new length
            best_score = 2.0 * n * _conv_max_fft(a, _fft_len_for_n(n))[0]
            best_seq = a.copy()
            continue  # jump to the next iteration with the new size

        # -----------------------------------------------------------
        # Early stop if stagnation is observed (only after we hit n_max)
        # -----------------------------------------------------------
        if n >= n_max and it - last_improve_iter > 200_000:
            print(f"No improvement for {it - last_improve_iter} iterations – stopping early.")
            break

    # ---------------------------------------------------------------
    # 4. Final exact evaluation (uses the provided evaluate_sequence)
    # ---------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:   # defensive – should never happen
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # ---------------------------------------------------------------
    # 5. Return the best sequence found.
    # ---------------------------------------------------------------
    return final_seq
