"""
Gradient descent on a smoothed ℓ∞‑norm of the self‑convolution.
-----------------------------------------------------------------
* Initialise the sequence with the (truncated) Catalan distribution,
  which is asymptotically optimal (score → π/2).

* At each iteration a smooth approximation of the maximum
  `max_k (a*a)_k` is used:
        f(a) = (1/λ)·log Σ_k exp(λ·c_k) ,  c = a*a .
  Its gradient can be written in closed form and evaluated with two
  FFT‑based convolutions, i.e. in O(n log n) time.

* A simple projected gradient step (onto the probability simplex) is
  performed, the step size is adapted, and the smoothing parameter λ
  is slowly increased so that the optimisation focuses more on the
  true maximum.

* The loop runs until the time budget (≈1000 s) is exhausted.
  The best sequence seen so far is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based linear convolution (full “linear” mode)
# ----------------------------------------------------------------------
def _fft_convolve_full(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a*b using a real FFT.
    Result length = len(a) + len(b) - 1.
    """
    n = a.shape[0]
    m = b.shape[0]
    size = 1 << ((n + m - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    conv = np.fft.irfft(fa * fb, size)
    return conv[: n + m - 1]


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for the self‑convolution a*a."""
    return _fft_convolve_full(a, a)


# ----------------------------------------------------------------------
# Catalan numbers (a_i = C(2i,i)/4**i) – the classic construction
# ----------------------------------------------------------------------
def _catalan_normalised(n: int) -> np.ndarray:
    """First n Catalan‑type numbers a_i = C(2i,i)/4**i,
    normalised so that sum_i a_i = 1.
    The recurrence a_{i} = a_{i-1}·(2i‑1)/(2i) is numerically stable.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Gradient of the smooth‑max of the self‑convolution
# ----------------------------------------------------------------------
def _smooth_max_grad(a: np.ndarray, lam: float):
    """
    Returns (grad, conv) where
    * conv = a*a (full convolution)
    * grad  = ∇_a  ( 1/λ·log Σ_k exp(λ·conv_k) )
    The gradient is computed using the identity
        ∂conv_k/∂a_j = 2·a_{k-j}   (with out‑of‑range entries = 0)
    and two FFT‑based convolutions.
    """
    n = a.shape[0]

    # full self‑convolution
    conv = _fft_convolve_self(a)                     # length 2n‑1
    max_val = conv.max()

    # soft‑max weights (stable exponentiation)
    w = np.exp(lam * (conv - max_val))
    w /= w.sum()                                       # Σ_k w_k = 1

    # reversed version of a (needed for the gradient formula)
    rev_a = a[::-1]

    # gradient = 2 * ( w * rev_a )  evaluated at the centre
    grad_full = _fft_convolve_full(w, rev_a)          # length 3n‑2
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]        # slice of length n
    return grad, conv


# ----------------------------------------------------------------------
# Project onto the simplex {x ≥ 0, Σ_i x_i = 1}
# ----------------------------------------------------------------------
def _project_simplex(x: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    # Shift by the smallest value so that we never deal with negatives
    # before the final projection – this is a cheap but stable trick.
    # For our use‑case the standard scaling works fine.
    x = np.maximum(x, 0.0)
    s = x.sum()
    if s > 0.0:
        return x / s
    # degenerate case – fall back to a uniform distribution
    n = x.shape[0]
    return np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence``.
    The routine runs a limited‑time projected gradient descent starting
    from the Catalan construction.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # --------------------------------------------------------------
    # 1) problem size – larger n gives a better asymptotic bound,
    #    but also larger FFTs.  n≈20000 is a sweet spot.
    # --------------------------------------------------------------
    n = 20000

    # --------------------------------------------------------------
    # 2) start from the Catalan distribution (known to be near‑optimal)
    # --------------------------------------------------------------
    a = _catalan_normalised(n)

    # --------------------------------------------------------------
    # 3) initialise optimisation parameters
    # --------------------------------------------------------------
    best_a = a.copy()
    conv = _fft_convolve_self(a)
    best_score = 2.0 * n * conv.max()
    print(f"[init] n={n}, initial score ≈ {best_score:.9f}")

    step_size = 0.5            # initial gradient step (will be adapted)
    lam = 5.0                  # smoothing parameter – will be increased
    iter_no = 0
    no_improve = 0

    # --------------------------------------------------------------
    # 4) main optimisation loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        iter_no += 1

        # compute gradient of the smoothed max
        grad, conv = _smooth_max_grad(a, lam)

        # proposed move (gradient descent)
        a_candidate = a - step_size * grad
        a_candidate = _project_simplex(a_candidate)

        # evaluate – we use the analytical expression (sum is 1)
        conv_candidate = _fft_convolve_self(a_candidate)
        cand_score = 2.0 * n * conv_candidate.max()

        # accept if better, otherwise shrink the step size
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
            a = a_candidate
            step_size *= 1.02          # gentle acceleration
            lam = min(lam * 1.02, 200.0)   # focus more on the true max
            no_improve = 0
            print(f"[iter {iter_no:06d}] improved score {best_score:.9f}")
        else:
            # no improvement – backtrack and reduce step
            step_size *= 0.97
            lam = max(lam * 0.99, 5.0)
            a = a_candidate            # still move (stochastic descent)
            no_improve += 1

        # occasional random jitter to escape shallow plateaux
        if no_improve > 500:
            jitter = np.random.randn(n) * 1e-5
            a = _project_simplex(a + jitter)
            no_improve = 0

        # safety stop if step becomes too tiny
        if step_size < 1e-8:
            break

    # --------------------------------------------------------------
    # 5) final report
    # --------------------------------------------------------------
    final_conv = _fft_convolve_self(best_a)
    final_score = 2.0 * n * final_conv.max()
    print(f"[final] best score ≈ {final_score:.9f} after {iter_no} iterations")
    # Return a plain Python list – the harness will call evaluate_sequence().
    return best_a.tolist()
