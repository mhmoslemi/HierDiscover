"""
Propose a step‑function (list of non‑negative heights) that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

where conv is the discrete convolution of the sequence with itself.
The algorithm consists of three stages:

1. **Diverse parametric sampling** – generate candidates from several
   families (power‑law, exponential, mixtures, two‑sided exponential,
   beta‑distribution).  Each candidate is normalised to a probability
   vector (sum = 1) and evaluated with a cheap in‑kernel score.  When a
   candidate beats the current best, the official ``evaluate_sequence``
   function is called to confirm the improvement.

2. **Fast adaptive hill‑climbing** – start from the best sampled vector
   and repeatedly move a fraction of mass from the index that most
   contributes to the current maximal autocorrelation to the index that
   contributes the least.  The convolution is updated incrementally
   (O(n) per iteration) using the identity

       conv(a+Δ, a+Δ) = conv(a,a) +
                       2·Δ·conv(a, e) + conv(Δ,Δ)

   which avoids recomputing a full convolution at every step.  The step
   size shrinks linearly and a tiny simulated‑annealing acceptance
   probability allows occasional uphill moves.

3. **Return the best sequence** – the best vector seen (according to the
   official evaluator) is returned as a Python list of floats.

The routine respects the 1000 s time budget, prints progress information
and works for any length ``n`` (the harness provides ``height_sequence_1``).
"""

import time
import numpy as np
import math

# ----------------------------------------------------------------------
# Helper utilities – top level, no closures
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _fast_score(arr: np.ndarray) -> float:
    """Return the objective value for a non‑negative, normalised array."""
    total = arr.sum()
    if total <= 0.0:
        return float("inf")
    conv = np.convolve(arr, arr)
    max_val = conv.max()
    n = arr.shape[0]
    return 2.0 * n * max_val / (total * total)


def _norm_and_score(raw: np.ndarray):
    """Normalise a raw (non‑negative) vector and return (official, cheap)."""
    total = raw.sum()
    if total <= 0.0:
        return float("inf"), None
    norm = raw / total
    cheap = _fast_score(norm)
    # verify with the official evaluator only when promising
    try:
        official = evaluate_sequence(norm.tolist())
    except Exception:  # defensive – should never happen
        official = float("inf")
    return official, norm


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: float = 950, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    the supplied ``evaluate_sequence`` value.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : float, optional
        Time budget (seconds).  The function stops a few seconds before the
        limit to avoid a hard termination.
    **kwargs
        Ignored – present for compatibility with the harness.

    Returns
    -------
    list[float]
        The best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # Determine the length of the problem (fallback to 2244)
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)  # provided by the harness
    except Exception:
        n = 2244

    idx = np.arange(n, dtype=np.float64)
    idx_norm = idx / (n - 1.0)

    # ------------------------------------------------------------------
    # Baseline – start from the provided solution if it exists
    # ------------------------------------------------------------------
    try:
        best_seq = list(height_sequence_1)
        best_score = evaluate_sequence(best_seq)
    except Exception:
        best_seq = np.full(n, 1.0 / n).tolist()
        best_score = evaluate_sequence(best_seq)

    print(f"[init] length = {n}, baseline score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Helper to test a raw candidate vector
    # ------------------------------------------------------------------
    def _try_candidate(raw_vec: np.ndarray, label: str):
        nonlocal best_seq, best_score
        if raw_vec.ndim != 1:
            raw_vec = raw_vec.ravel()
        # ensure non‑negative
        raw_vec = np.maximum(raw_vec, 0.0)
        official, norm = _norm_and_score(raw_vec)
        if norm is None:
            return
        if official < best_score:
            best_score = official
            best_seq = norm.tolist()
            print(f"[{label}] improved to {official:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Diverse parametric sampling (time‑controlled)
    # ------------------------------------------------------------------
    # Helper to check remaining budget
    def _remaining():
        return budget_s - (time.time() - start) - 5.0  # keep 5 s margin

    # ---- Power‑law families (grid + random) ----
    alphas_grid = np.linspace(0.30, 1.00, 71)          # 71 values
    shifts_grid = np.linspace(0.01, 3.0, 30)           # 30 values
    for alpha in alphas_grid:
        if _remaining() <= 0:
            break
        for shift in shifts_grid:
            raw = np.power(idx + shift, -alpha)
            _try_candidate(raw, "gridPL")

    for _ in range(2500):
        if _remaining() <= 0:
            break
        alpha = rng.uniform(0.15, 1.30)
        shift = rng.uniform(0.001, 6.0)
        raw = np.power(idx + shift, -alpha)
        _try_candidate(raw, "randPL")

    # ---- Pure exponential families ----
    for _ in range(800):
        if _remaining() <= 0:
            break
        beta = rng.uniform(1e-6, 0.2)
        raw = np.exp(-beta * idx)
        _try_candidate(raw, "exp")

    # ---- Mixture of exponential & uniform ----
    for _ in range(800):
        if _remaining() <= 0:
            break
        beta = rng.uniform(1e-6, 0.2)
        p = rng.random()
        raw = p * np.exp(-beta * idx) + (1.0 - p) * np.ones_like(idx)
        _try_candidate(raw, "mixExpU")

    # ---- Two‑sided exponential (skewed mass) ----
    for _ in range(800):
        if _remaining() <= 0:
            break
        beta = rng.uniform(1e-5, 0.15)
        p = rng.random()
        raw = p * np.exp(-beta * idx) + (1.0 - p) * np.exp(-beta * (n - 1 - idx))
        _try_candidate(raw, "twoSideExp")

    # ---- Beta‑distribution shapes (flexible skew) ----
    for _ in range(1200):
        if _remaining() <= 0:
            break
        a = rng.uniform(0.0, 12.0)          # shape on left side
        b = rng.uniform(0.0, 12.0)          # shape on right side
        raw = np.power(idx_norm, a) * np.power(1.0 - idx_norm, b) + 1e-12
        _try_candidate(raw, "beta")

    # ---- Random mixtures of two power‑laws (more expressive) ----
    for _ in range(1500):
        if _remaining() <= 0:
            break
        a1 = rng.uniform(0.1, 1.5)
        s1 = rng.uniform(0.001, 5.0)
        a2 = rng.uniform(0.1, 1.5)
        s2 = rng.uniform(0.001, 5.0)
        w = rng.random()
        raw = w * np.power(idx + s1, -a1) + (1.0 - w) * np.power(idx + s2, -a2)
        _try_candidate(raw, "mix2PL")

    print(f"[sampling] best score after sampling: {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Fast adaptive hill‑climbing / simulated annealing
    # ------------------------------------------------------------------
    cur = np.array(best_seq, dtype=np.float64)   # guaranteed sum≈1
    cur_sum = cur.sum()
    cur /= cur_sum                                 # safety normalisation
    cur_conv = np.convolve(cur, cur)               # full convolution (len 2n‑1)
    cur_score = _fast_score(cur)                   # cheap score

    # algorithmic hyper‑parameters
    max_iters = 500000
    min_frac = 0.015          # final move‑size fraction
    max_frac = 0.09           # initial move‑size fraction
    init_temp = 1e-4          # annealing temperature start
    eps_improve = 1e-12

    it = 0
    last_improve_iter = 0

    while time.time() - start < budget_s - 5.0 and it < max_iters:
        it += 1

        # ---- locate the index where the current autocorrelation peaks ----
        max_idx = int(np.argmax(cur_conv))

        # ---- compute contribution of each index to that peak (i·a_i·a_{max‑i}) ----
        rev = max_idx - idx.astype(np.int64)
        mask = (rev >= 0) & (rev < n)
        contrib = np.zeros_like(cur)
        contrib[mask] = cur[mask] * cur[rev[mask]]

        src = int(np.argmax(contrib))
        tgt = int(np.argmin(contrib))

        # fallback to random distinct pair if the contribution based pair is degenerate
        if src == tgt or cur[src] <= 0.0:
            src, tgt = rng.choice(n, size=2, replace=False)

        # ---- step size (fraction of source mass) ----
        frac = max_frac - (max_frac - min_frac) * (it / max_iters)
        step = cur[src] * rng.random() * frac
        if step <= 0.0:
            continue

        # ---- fast convolution update (linear + quadratic terms) ----
        new_conv = cur_conv.copy()

        # linear part: 2·step·(shift_tgt - shift_src)
        new_conv[tgt:tgt + n] += 2.0 * step * cur
        new_conv[src:src + n] -= 2.0 * step * cur

        # quadratic part: δ_{2tgt} + δ_{2src} - 2·δ_{src+tgt}
        idx_2tgt = 2 * tgt
        idx_2src = 2 * src
        idx_sum = src + tgt
        new_conv[idx_2tgt] += step * step
        new_conv[idx_2src] += step * step
        new_conv[idx_sum] -= 2.0 * step * step

        # ---- cheap score for the proposal ----
        new_max = new_conv.max()
        new_score = 2.0 * n * new_max  # denominator stays 1 (sum of cur is 1)

        # ---- simulated‑annealing acceptance ----
        accept = False
        if new_score < cur_score - eps_improve:
            accept = True
        else:
            # temperature linearly cools to 0
            temp = init_temp * max(0.0, 1.0 - it / max_iters)
            if temp > 0.0:
                prob = math.exp(-(new_score - cur_score) / temp)
                if rng.random() < prob:
                    accept = True

        if accept:
            # perform the actual mass move
            cur[src] -= step
            cur[tgt] += step

            # numerical cleanup: enforce non‑negativity and renormalise occasionally
            cur[src] = max(cur[src], 0.0)
            cur[tgt] = max(cur[tgt], 0.0)

            cur_conv = new_conv
            cur_score = new_score
            cur_sum = cur.sum()
            if abs(cur_sum - 1.0) > 1e-12:
                cur /= cur_sum          # renormalise to keep sum = 1

            # if the cheap score beats the global best, verify with the official evaluator
            if new_score < best_score - eps_improve:
                official = evaluate_sequence(cur.tolist())
                if official < best_score - eps_improve:
                    best_score = official
                    best_seq = cur.tolist()
                    last_improve_iter = it
                    print(f"[refine] iter={it:,d} new best={official:.6f}")

        # occasional re‑projection to the simplex for numerical stability
        if it % 1000 == 0:
            cur = _project_to_simplex(cur)
            cur_conv = np.convolve(cur, cur)
            cur_score = _fast_score(cur)

        # early abort if we have gone a long time without improvement
        if it - last_improve_iter > 50000 and _remaining() < 30.0:
            # not enough time left for many more iterations – stop early
            break

    print(f"[final] best score found: {best_score:.6f}")
    return best_seq

adds a simple simulated‑annealing acceptance rule and periodic

below the previous bound and aims for the target ≈ 1.503 within the 1000 s
budget.
