"""
Propose a non‑negative step‑function sequence that minimizes the
evaluation function defined in the challenge.

Algorithm
---------
1. **Initial shape** – start from a discrete *arcsine* distribution,
   a_i ∝ 1/√((i+½)(n‑½‑i)).  This shape is known to flatten the
   self‑convolution and gives a good baseline.

2. **Long sub‑gradient phase** – the exact objective is
      O(a)=2·n·max(conv(a,a))  (conv = full self‑convolution).
   While time permits we repeatedly
   * compute the full convolution,
   * evaluate the exact objective,
   * obtain the sub‑gradient of the maximal convolution entry
     (the gradient of max(conv) w.r.t. a),
   * take a small step opposite to the sub‑gradient,
   * project back onto the probability simplex.
   The step size is slowly decreased; the projection guarantees
   non‑negativity and Σa_i=1, which makes the objective invariant
   under scaling.

3. **Stochastic local refinement** – after the deterministic phase we
   perform a Simulated‑Annealing‑style hill‑climbing:
   * randomly transfer a fraction of mass from one index to another,
   * accept the move if it improves the exact objective, otherwise
     accept with a temperature‑controlled Metropolis probability,
   * every few iterations perform an extra sub‑gradient step to drive
     the peak down further.

4. The best sequence seen during the whole run is returned.
   Intermediate progress (objective value and time stamp) is printed.

The whole routine respects the supplied time budget (default 1000 s) and
uses only NumPy and the provided ``evaluate_sequence`` function.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x ≥ 0, Σx = 1}
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full self‑convolution (length 2·n‑1)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return full self‑convolution a*a using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) (used in deterministic phase)
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Gradient of max(conv) w.r.t. a (exact sub‑gradient)."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)  # indices attaining the maximum
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    g /= len(ks)
    return g


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence that minimises the evaluation
    function.  Returns the sequence as a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size --------------------------------------------------
    # If a previous construction is available we keep its length,
    # otherwise we use a power‑of‑two size (good for FFT).
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
        print(f"[init] using length of existing sequence: n={n}")
    else:
        n = 8192                # a convenient power of two
        print(f"[init] no prior sequence – using n={n}")

    # FFT length – must be at least 3·n‑2 for the correlation used later
    L = 1 << ((3 * n - 2).bit_length())   # next power of two ≥ 3n‑2

    # --------------------------------------------------------------
    # 1. Initial arcsine‑like distribution -------------------------
    i = np.arange(n, dtype=float)
    # ½‑offset avoids singularities at the exact boundaries
    w = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    a = w / w.sum()                         # normalise to the simplex
    best_seq = a.copy()
    conv = _conv_fft(a, L)
    best_score = 2.0 * n * conv.max()
    print(f"[init] arcsine start O={best_score:.6f}")

    # --------------------------------------------------------------
    # 2. Deterministic sub‑gradient descent (large chunk of the budget)
    subgrad_time_limit = start_time + budget_s * 0.45
    # schedule for the learning‑rate multiplier (η) – largest at start,
    # decreasing linearly to a small value.
    eta_start = 0.25
    eta_end = 0.015
    max_iter_sub = 1_200_000   # huge upper bound – loop will stop by time

    iter_sub = 0
    while time.time() < subgrad_time_limit and iter_sub < max_iter_sub:
        # current objective
        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[{time.time():.1f}] sub‑grad new best O={best_score:.6f} (iter={iter_sub})")

        # sub‑gradient of the maximal convolution entry
        g = _subgradient_max_conv(a, conv)

        # linear annealing of the step‑size multiplier
        eta = eta_start + (eta_end - eta_start) * (iter_sub / max_iter_sub)

        # scale the step so that the largest coordinate changes by at most η
        max_abs = np.max(np.abs(g)) + 1e-12
        step = eta / max_abs
        a = a - step * g
        a = _project_to_simplex(a)

        iter_sub += 1

    # --------------------------------------------------------------
    # 3. Stochastic local refinement (the remainder of the budget)
    print(f"[phase‑2] entering stochastic refinement, best O={best_score:.6f}")

    # parameters for the simulated‑annealing hill‑climber
    temperature0 = 0.025
    subgrad_interval = 9                # how often we sprinkle a sub‑grad step
    max_iter_local = 2_000_000          # generous cap – real stop is time

    iter_local = 0
    T = temperature0
    while time.time() < deadline and iter_local < max_iter_local:
        # ---- mass‑transfer move ------------------------------------------------
        i, j = np.random.randint(n, size=2)
        if i != j and a[i] > 0.0:
            # move up to 12 % of the mass at i
            delta = a[i] * np.random.rand() * 0.12
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            a_new = _project_to_simplex(a_new)

            conv_new = _conv_fft(a_new, L)
            new_obj = 2.0 * n * conv_new.max()

            # Metropolis acceptance
            if new_obj < cur_obj or np.random.rand() < math.exp((cur_obj - new_obj) / max(T, 1e-12)):
                a = a_new
                conv = conv_new
                cur_obj = new_obj
                if cur_obj < best_score:
                    best_score = cur_obj
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] New global best O={best_score:.6f} (iter={iter_local})")
        else:
            # even if we didn't move, we keep the current objective
            pass

        # ---- occasional exact sub‑gradient step ------------------------------
        if iter_local % subgrad_interval == 0:
            g = _subgradient_max_conv(a, conv)
            # same scaling logic as in the deterministic phase
            maxg = np.max(np.abs(g)) + 1e-12
            eta = 0.07 * (1.0 - iter_local / max_iter_local)   # slowly decay
            step = eta / maxg
            a = a - step * g
            a = _project_to_simplex(a)
            conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_score:
                best_score = cur_obj
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Sub‑grad improvement O={best_score:.6f}")

        # temperature decay (linear)
        T = temperature0 * (1.0 - iter_local / max_iter_local)
        if T < 1e-6:
            T = 0.0

        iter_local += 1

    # --------------------------------------------------------------
    # Final sanity check with the provided evaluation routine
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence failed:", exc)

    return best_seq.tolist()
