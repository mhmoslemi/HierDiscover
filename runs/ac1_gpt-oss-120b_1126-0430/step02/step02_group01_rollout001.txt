"""
Propose a step‑function sequence that minimizes the bound

    O = 2·n·max(conv(a,a)) / (sum(a)^2)

using a fast projected Adam optimizer on a smooth approximation of the
max‑convolution.  The algorithm works as follows:

1.  Choose a length `n` (default 6000) and start from a random probability
    vector `a` (non‑negative, sum = 1).

2.  At each iteration we evaluate the full linear convolution `c = a*a`
    using an FFT (O(L log L) with L≈2ⁿ).  The peak `max(c)` gives the
    current bound `O = 2·n·max(c)`.

3.  To obtain a smooth surrogate for the non‑differentiable `max`,
    a soft‑max with temperature `α` is formed:
        p_k = exp(α·c_k) / Σ_j exp(α·c_j).

    For `α→∞` this approaches the indicator of the worst index.

4.  The gradient of the smooth max w.r.t. `a` can be written as a
    (linear) cross‑correlation:
        ∇_a  f = 2·(a ⋆ p) .
    The correlation is also computed by FFTs, so each iteration requires
    only a few FFTs.

5.  An Adam update (`m`, `v` moments) with a decreasing stepsize is
    applied, after which `a` is projected back onto the probability
    simplex (`_project_to_simplex`).  The temperature `α` is increased
    linearly during the run, turning the smooth objective into the true
    max.

6.  The best sequence encountered is stored.  After the time budget
    expires the sequence is returned and its exact score is printed via
    the provided `evaluate_sequence` function.

The method explores the continuous space efficiently, works for
thousands of coefficients, and stays well within the 1000 s limit.

def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σ x = 1} (Duchi et al.)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        # All entries are zero – return the uniform distribution.
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# -------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return the evaluation objective 2·n·max(conv(a,a)) for Σa = 1."""
    n = a.size
    conv = np.convolve(a, a)                 # linear (full) convolution
    return 2.0 * n * conv.max()


# -------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power‑of‑two ≥ x."""
    return 1 << (x - 1).bit_length()


# -------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    evaluate_sequence(...).  Returns the sequence as a list of floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ---------------------------------------------------------
    # Problem size – a larger n gives more flexibility.
    n = 6000                     # feel free to increase (memory permitting)

    # Random start on the simplex.
    a = np.random.rand(n)
    a = _project_to_simplex(a)

    # Pre‑allocate FFT length (covers a (n) + p (2n‑1) – 1 = 3n‑2).
    L = _next_pow2(3 * n - 2)

    # Adam hyper‑parameters.
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # Temperature schedule for the smooth max (soft‑max parameter α).
    α_start = 2.0
    α_end = 200.0

    # Learning‑rate schedule.
    η0 = 0.30                     # initial “raw” step size

    # Keep the best solution seen so far.
    best_a = a.copy()
    best_obj = _objective(best_a)
    print(f"[init] n={n}, O={best_obj:.6f}")

    # Counters for progress / early‑stop handling.
    iter_no_improve = 0
    max_no_improve = 25000
    t = 0
    max_iters = 200_000          # hard cap – the time budget will stop us earlier

    while time.time() < deadline and t < max_iters:
        # -----------------------------------------------------
        # 1) Convolution a*a (full) via FFT.
        fft_a = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(fft_a * fft_a, n=L)[: 2 * n - 1]

        # Current objective.
        cur_max = conv_full.max()
        cur_obj = 2.0 * n * cur_max

        # Record improvement.
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            iter_no_improve = 0
            print(f"[iter {t:07d}] new global best O={best_obj:.6f}")
        else:
            iter_no_improve += 1

        # -----------------------------------------------------
        # 2) Form smooth max via soft‑max weights.
        α = α_start + (α_end - α_start) * (t / max_iters)   # linear schedule
        # shift for numerical stability.
        shift = conv_full.max()
        exps = np.exp(α * (conv_full - shift))
        p = exps / exps.sum()          # shape (2n‑1,)

        # -----------------------------------------------------
        # 3) Gradient of the smooth max:  ∇ = 2·(a ⋆ p).
        fft_p = np.fft.rfft(p, n=L)
        # Linear cross‑correlation (a ⋆ p)[i] = Σ_j a[j]·p[i+j]
        corr_full = np.fft.irfft(fft_a * np.conj(fft_p), n=L)
        grad = 2.0 * corr_full[:n]     # first n entries correspond to i=0..n‑1

        # -----------------------------------------------------
        # 4) Adam update + simplex projection.
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        η = η0 / math.sqrt(t)          # simple decay
        step = η * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # Enforce feasibility.
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a)

        # -----------------------------------------------------
        # 5) Periodic random perturbation if we are stuck.
        if iter_no_improve >= max_no_improve:
            # Small random jitter centred at the current best.
            a = best_a + 0.01 * np.random.randn(n)
            a = np.clip(a, 0.0, None)
            a = _project_to_simplex(a)
            iter_no_improve = 0
            print(f"[restart] iteration {t}, re‑initialised from best.")

    # ---------------------------------------------------------
    # Final evaluation using the exact (potentially throttled) routine.
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:                     # pragma: no cover
        print("[final] evaluate_sequence call failed:", e)

    return best_a.tolist()
