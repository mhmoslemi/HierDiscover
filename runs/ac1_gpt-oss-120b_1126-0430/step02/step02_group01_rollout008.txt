def _linear_convolve(x, y):
    n = len(x) + len(y) - 1
    N = 1 << (n - 1).bit_length()
    X = np.fft.rfft(x, N)
    Y = np.fft.rfft(y, N)
    Z = X * Y
    conv = np.fft.irfft(Z, N)
    return conv[:n]

"""
Propose a step‑function (non‑negative vector with sum = 1) that minimises

    O = 2·n·max(conv(a,a))

where conv is the linear convolution.  

The algorithm consists of two complementary phases:

1. **Mirror‑descent flattening** –  
   We minimise the squared deviation of the self‑convolution from the
   uniform target `1/(2n‑1)`.  The gradient of this smooth loss can be
   computed by convolutions and we perform an exponential (multiplicative
   weights) update, which is a mirror‑descent step on the simplex.
   After each step the vector is renormalised to the probability simplex.

2. **Stochastic refinement** –  
   Periodically a small random mass transfer between two coordinates is
   attempted and accepted if it improves the objective.  If no improvement
   is observed for a long stretch the current solution is perturbed a
   little and the optimisation continues.

Both phases run until the allotted time (~1000 s) expires.  The best
sequence encountered is returned and its exact score is printed via the
provided `evaluate_sequence` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _linear_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear (non‑circular) convolution of two real vectors using FFT.
    Returns a real array of length len(x)+len(y)-1.
    """
    n = len(x) + len(y) - 1
    N = 1 << ((n - 1).bit_length())           # next power of two
    X = np.fft.rfft(x, N)
    Y = np.fft.rfft(y, N)
    Z = X * Y
    conv = np.fft.irfft(Z, N)
    return conv[:n]


# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project a non‑negative vector onto the probability simplex
    {x≥0, sum(x)=1}.  For vectors that are already non‑negative,
    simple normalisation is sufficient.
    """
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return v / s


# ----------------------------------------------------------------------
def _conv_and_grad(a: np.ndarray):
    """
    Compute a*a (linear convolution) and the gradient of the squared‑error
    loss   L = Σ_k (conv_k - 1/(2n‑1))²   w.r.t. a.
    Returns (conv, grad) where conv is a vector of length 2n‑1 and grad
    has length n.
    """
    n = a.size
    conv = _linear_convolve(a, a)                     # length 2n‑1
    avg = 1.0 / (2 * n - 1)
    e = conv - avg                                    # deviation from uniform target
    rev_a = a[::-1]
    grad_full = _linear_convolve(e, rev_a)            # length 3n‑2
    start = n - 1
    grad = 4.0 * grad_full[start:start + n]           # slice to length n
    return conv, grad


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) (sum(a) is assumed to be 1)."""
    n = a.size
    conv = _linear_convolve(a, a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Search for a step‑function with minimal bound. Returns a plain list."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ----- problem size -------------------------------------------------
    n = 2500                     # a length that gives a good trade‑off speed/quality

    # ----- initialisation (arcsine‑shaped distribution) -----------------
    idx = np.arange(n, dtype=float)
    a = 1.0 / np.sqrt((idx + 0.5) * (n - idx - 0.5))
    a = _simplex_projection(a)

    best_seq = a.copy()
    best_obj = _objective(a)
    print(f"[init] n={n}, objective={best_obj:.6f}")

    # ----- main optimisation loop ----------------------------------------
    iteration = 0
    eta_initial = 0.12            # step‑size for the exponentiated update
    no_improve = 0
    max_no_improve = 25000        # restart after this many non‑improving steps

    while time.time() < deadline:
        iteration += 1

        # current convolution and objective
        conv, grad = _conv_and_grad(a)
        cur_obj = 2.0 * n * conv.max()

        # keep global best
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_seq = a.copy()
            no_improve = 0
            print(f"[iter {iteration}] new best O={best_obj:.6f}")
        else:
            no_improve += 1

        # ---- mirror‑descent (exponential) update -----------------------
        # avoid overflow / underflow by clipping the gradient
        grad = np.clip(grad, -25.0, 25.0)

        # decreasing step size
        eta = eta_initial / (1.0 + iteration * 1e-5)

        a_candidate = a * np.exp(-eta * grad)
        a_candidate = _simplex_projection(a_candidate)

        # evaluate candidate
        cand_obj = _objective(a_candidate)

        # accept if improvement (strictly better)
        if cand_obj < cur_obj:
            a = a_candidate
        else:
            # occasional stochastic move to escape flat regions
            if iteration % 400 == 0:
                i, j = np.random.randint(0, n, size=2)
                if i != j and a[i] > 0.0:
                    delta = a[i] * np.random.rand() * 0.20   # move up to 20 % of a[i]
                    a2 = a.copy()
                    a2[i] -= delta
                    a2[j] += delta
                    a = _simplex_projection(a2)

        # ---- restart if stuck -------------------------------------------
        if no_improve >= max_no_improve:
            print(f"[restart] iteration {iteration}, rebuilding from best.")
            a = best_seq.copy()
            a += np.random.randn(n) * 0.004
            a = _simplex_projection(a)
            no_improve = 0

    # ----- final exact evaluation ----------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence failed:", exc)

    return best_seq.tolist()
