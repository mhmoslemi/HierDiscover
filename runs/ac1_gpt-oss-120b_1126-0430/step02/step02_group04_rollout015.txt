def propose_candidate(seed=123, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5
    n = 12000

    # Phase 1: coarse grid
    p_vals = np.linspace(-0.9, 3.5, 15)  # 15 values
    q_vals = np.linspace(-0.9, 3.5, 15)
    best_score = np.inf
    best_params = (0,0)
    best_seq = None
    for p in p_vals:
        for q in q_vals:
            a = _beta_sequence(n, p, q)
            score = _eval_score(a)
            if score < best_score:
                best_score = score
                best_params = (p,q)
                best_seq = a.copy()
    print(f"[grid] best after coarse grid: p={best_params[0]:.3f}, q={best_params[1]:.3f}, score={best_score:.6f}")

    # Phase 2: local search
    p, q = best_params
    step = 0.3
    while time.time() < deadline - 200:  # reserve time for gradient descent
        for _ in range(30):
            dp = np.random.randn() * step
            dq = np.random.randn() * step
            p_new = p + dp
            q_new = q + dq
            # ensure > -0.99
            if p_new <= -0.95 or q_new <= -0.95:
                continue
            a = _beta_sequence(n, p_new, q_new)
            score = _eval_score(a)
            if score < best_score:
                best_score = score
                best_params = (p_new, q_new)
                best_seq = a.copy()
                p, q = p_new, q_new
                print(f"[local] new best p={p:.3f}, q={q:.3f}, score={score:.6f}")
        step *= 0.9  # shrink step size
        if step < 1e-4:
            break

    # Phase 3: gradient descent refine
    a = best_seq.copy()
    cur_score = best_score
    lr = 0.1
    alpha = 15.0
    iteration = 0
    no_improve = 0

    while time.time() < deadline:
        b = _convolve_fft(a, a)
        max_b = b.max()
        # increase alpha gradually
        if iteration % 1000 == 0 and iteration > 0:
            alpha = min(alpha * 1.15, 5000)
        # compute softmax weights
        w = np.exp(alpha * (b - max_b))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(b) / len(b)
        else:
            w /= w_sum
        w_rev = w[::-1]
        conv_full = _convolve_fft(a, w_rev)
        idx = (len(b) - 1) - np.arange(n)
        grad = 2.0 * conv_full[idx]

        # momentum
        if iteration == 0:
            momentum = np.zeros_like(a)
        else:
            momentum = 0.9 * momentum + grad

        # line search
        improved = False
        lr_local = lr
        while lr_local > 1e-8:
            a_candidate = _proj_simplex(a - lr_local * momentum)
            cand_score = _eval_score(a_candidate)
            if cand_score < cur_score:
                a = a_candidate
                cur_score = cand_score
                improved = True
                break
            lr_local *= 0.5

        if improved:
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{iteration:06d}] new best {best_score:.6f} lr={lr_local:.5f} α={alpha:.2f}")
                no_improve = 0
            else:
                no_improve += 1
        else:
            lr *= 0.8
            no_improve += 1

        if iteration % 500 == 0 and iteration > 0:
            lr = max(lr * 0.9, 1e-5)

        if no_improve > 3000:
            # random perturb around best
            a = best_seq + 0.01 * np.random.randn(n)
            a = np.maximum(a, 0)
            a = _proj_simplex(a)
            cur_score = _eval_score(a)
            lr = 0.1
            alpha = 15.0
            no_improve = 0
            print(f"[{iteration}] restart after stall (score={cur_score:.6f})")

        iteration += 1

    print(f"[done] best score {best_score:.6f}")
    return best_seq.tolist()

def _beta_sequence(n, p, q):
    # x in (0,1) range centered at half-step
    i = np.arange(n)
    x = (i + 0.5) / n  # avoid 0 and 1
    a = np.power(x, p) * np.power(1 - x, q)
    # if p or q near zero with negative exponent, ensure no zeros (some may be inf)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    # clip tiny negative cause rounding
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0:
        # fallback uniform
        a = np.ones(n)
        total = n
    a /= total
    return a

"""
Propose a candidate non‑negative step‑function minimizing `evaluate_sequence`.

The algorithm works in three stages:
1. **Rich initialisation** – we evaluate a collection of analytic families
   (Catalan, geometric, power‑law * exponential, β‑distributions) on a
   moderate problem size (`n≈12000`).  The best among them (according to the
   exact objective `2·n·max(conv)/sum²`) becomes the start point.
2. **Local continuous refinement** – a projected smooth‑max gradient descent
   on the probability simplex is performed.  The smooth‑max uses a soft‑max
   weighting of the convolution entries, the gradient is obtained by a
   convolution with the reversed weight vector, and the iterate is projected
   back onto the simplex via the Euclidean projection of Duchi et al.
   Momentum (Nesterov‑like) and an adaptive learning‑rate/back‑tracking line
   search guarantee monotone improvement.
3. **Stochastic restarts** – if the optimiser stalls for thousands of
   iterations we perturb the current best solution and continue.

All heavy convolutions are carried out with real FFTs (`O(n log n)`).  The
routine keeps track of the best sequence found so far and returns it before
the supplied time‑budget (default 1000 s) expires.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, sum=1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u > cssv / np.arange(1, n + 1))[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via real FFT."""
    n = x.size + y.size - 1
    size = 1 << (n - 1).bit_length()  # next power of two
    X = np.fft.rfft(x, size)
    Y = np.fft.rfft(y, size)
    Z = X * Y
    conv = np.fft.irfft(Z, size)[:n]
    return conv.real


def _eval_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv)/(sum(a)**2) for a non‑negative vector."""
    n = a.size
    b = _convolve_fft(a, a)
    max_b = b.max()
    sum_a = a.sum()
    # sum_a is guaranteed >0 because we keep a on the simplex
    return float(2.0 * n * max_b / (sum_a ** 2))


def _catalan_sequence(n: int) -> np.ndarray:
    """Catalan‑like decreasing sequence, normalised to sum = 1."""
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * (i - 1) + 1) / (2.0 * i)
    a /= a.sum()
    return a


def _geom_sequence(n: int, r: float) -> np.ndarray:
    """Geometric (exponential) decay a_i ∝ r**i, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    a = r ** i

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
