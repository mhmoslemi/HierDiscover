    if best_seq is None:
        # fallback – a uniform distribution (should never happen)
        best_seq = np.ones(n, dtype=float) / n
        best_score = _score(best_seq)
        print("[fallback] using uniform sequence")

    print(f"[done] best exponent p={best_p}, best score={best_score:.9f}")
    # return a plain Python list as required by the harness
    return best_seq.tolist()
