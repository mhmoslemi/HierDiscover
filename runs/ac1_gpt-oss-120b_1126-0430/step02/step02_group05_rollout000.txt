"""
Gradient‐balancing optimizer for the autoconvolution max‑norm problem.

The algorithm works with a *zero‑sum* update direction that moves mass
away from positions that contribute most to the current peak(s) of the
self‑convolution and toward positions that contribute the least.
For a chosen active set K (indices where the convolution attains its
maximum) the exact sub‑gradient is

    g_i = 2 * Σ_{k∈K} f_{k‑i}   (with out‑of‑range entries ignored).

We then form a zero‑mean direction   Δ = –(g – mean(g)).
All components of Δ that are negative correspond to “sources” (high‑impact
positions) and the positive ones to “sinks”.  The largest feasible step
size that keeps the sequence non‑negative is

    ε_max = min_i { f_i / –Δ_i  :  Δ_i < 0 } .

A fraction of ε_max (by default 0.5) is taken, the update is applied,
the sequence is (re‑)normalised to keep the L1‑norm unchanged (the
objective is scale‑invariant) and the new score is evaluated.  If the
move does not improve the score we backtrack by repeatedly halving ε
until an improvement is found or ε becomes tiny – in the latter case a
small random jitter is performed to escape possible plateaus.

The optimizer is run several times from different sensible seeds
(Catalan, geometric and uniform).  The best sequence found before the
time budget expires is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast evaluation (identical to the provided evaluate_sequence but without
# the heavy security checks).  Returns the same score that the harness
# expects.
# ----------------------------------------------------------------------
def _fast_evaluation(seq: np.ndarray) -> float:
    n = seq.size
    if n == 0:
        return float("inf")
    # FFT size: next power of two ≥ 2*n‑1
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Simple families of initial sequences.
# ----------------------------------------------------------------------
def _catalan_sequence(length: int) -> np.ndarray:
    """Catalan / arcsine coefficients a_0 = 1, a_{i+1}=a_i*(2i+1)/(2i+2)."""
    seq = np.empty(length, dtype=float)
    a = 1.0
    seq[0] = a
    for i in range(1, length):
        a *= (2 * i - 1) / (2.0 * i)   # a_i = a_{i-1} * (2i-1)/(2i)
        seq[i] = a
    return seq


def _geometric_sequence(length: int, r: float = 0.9) -> np.ndarray:
    """Descending geometric progression, normalised to unit L1."""
    seq = np.power(r, np.arange(length, dtype=float))
    seq /= seq.sum()
    return seq


def _uniform_sequence(length: int) -> np.ndarray:
    """Flat sequence (unit L1)."""
    seq = np.ones(length, dtype=float)
    seq /= seq.sum()
    return seq


# ----------------------------------------------------------------------
# Compute the sub‑gradient g_i = 2 * Σ_{k∈K} f_{k‑i} for a given active set K.
# ----------------------------------------------------------------------
def _gradient_for_active(seq: np.ndarray, active: np.ndarray) -> np.ndarray:
    n = seq.size
    g = np.zeros_like(seq)
    # For each active convolution index k we add the reversed slice of f.
    for k in active:
        start = max(0, k - (n - 1))
        end = min(k, n - 1)                # inclusive
        # slice f[start : end+1] and reverse it -> f_{k‑i}
        seg = seq[start : end + 1][::-1]
        g[start : end + 1] += 2.0 * seg
    return g


# ----------------------------------------------------------------------
# One optimisation run: gradient‑balancing descent with backtracking.
# ----------------------------------------------------------------------
def _optimise_one_start(seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Perform the gradient‑balancing descent on `seq` until `deadline`.
    Returns the best sequence encountered and its score.
    """
    n = seq.size
    best_seq = seq.copy()
    best_score = _fast_evaluation(best_seq)

    it = 0
    eps_active = 1e-12                # tolerance for picking the active set
    min_step_fraction = 0.05          # fraction of ε_max we start with
    max_backtrack = 8                 # how many halvings we allow

    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1) current convolution and active set
        # --------------------------------------------------------------
        m = 1 << ((2 * n - 1).bit_length())
        fa = np.fft.rfft(seq, m)
        conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
        max_val = conv.max()
        # include values that are within a tiny tolerance of the max
        active = np.flatnonzero(conv >= max_val - eps_active)

        # --------------------------------------------------------------
        # 2) gradient for the active set
        # --------------------------------------------------------------
        g = _gradient_for_active(seq, active)

        # zero‑mean direction (sum Δ = 0)
        avg = g.mean()
        delta = -(g - avg)

        # --------------------------------------------------------------
        # 3) compute maximal feasible step size that keeps all entries ≥0
        # --------------------------------------------------------------
        neg = delta < 0.0
        if not np.any(neg):
            # already flat – nothing to improve
            break
        eps_max = np.min(seq[neg] / -delta[neg])
        # start with a safe fraction of that bound
        eps = eps_max * min_step_fraction

        # --------------------------------------------------------------
        # 4) backtracking line search
        # --------------------------------------------------------------
        improved = False
        backtrack = 0
        while eps > 0 and backtrack <= max_backtrack:
            new_seq = seq + eps * delta
            # numerical safety: clip tiny negatives to zero
            new_seq = np.maximum(new_seq, 0.0)

            # re‑normalise to keep the L1‑norm unchanged (objective is scale‑invariant)
            sum_seq = seq.sum()
            sum_new = new_seq.sum()
            if sum_new == 0:
                # degenerate – abort this direction
                break
            new_seq = new_seq * (sum_seq / sum_new)

            new_score = _fast_evaluation(new_seq)

            if new_score < best_score - 1e-12:
                # accept
                seq = new_seq
                best_seq = new_seq.copy()
                best_score = new_score
                improved = True
                print(
                    f"Iter {it:6d}: score={best_score:.8f}, max_conv={max_val:.8f}, "
                    f"step={eps:.3e}, active={len(active)}"
                )
                break
            else:
                # try a smaller step
                eps *= 0.5
                backtrack += 1

        if not improved:
            # --------------------------------------------------------------
            # 5) occasional random jitter to escape plateaus
            # --------------------------------------------------------------
            if np.random.rand() < 0.03:
                i, j = np.random.randint(n, size=2)
                if i != j and seq[i] > 0:
                    eps_jit = np.random.rand() * seq[i] * 0.2
                    tmp = seq.copy()
                    tmp[i] -= eps_jit
                    tmp[j] += eps_jit
                    # keep sum exact
                    tmp_score = _fast_evaluation(tmp)
                    if tmp_score < best_score:
                        seq = tmp
                        best_seq = tmp.copy()
                        best_score = tmp_score
                        print(
                            f"Iter {it:6d}: jitter improved score={best_score:.8f}"
                        )
        it += 1

    return best_seq, best_score

        _geometric_sequence(n, r=0.92),
        _uniform_sequence(n),
        np.abs(np.random.randn(n))        # random positive vector
    ]
    # Normalise each starter to unit L1 (scale does not matter for the metric)
    starters = [s / s.sum() for s in starters]

    # ------------------------------------------------------------------
    # Run the optimiser from each starter, keep the best overall
    # ------------------------------------------------------------------
    global_best_seq = None
    global_best_score = float("inf")

    for idx, start in enumerate(starters):
        # allocate roughly equal time slices, but honour the global deadline
        sub_deadline = min(deadline, time.time() + (budget_s / len(starters)))
        print(f"=== Run {idx + 1}/{len(starters)}: starting optimisation ===")
        best_seq, best_score = _optimise_one_start(start.copy(), sub_deadline)
        print(
            f"Run {idx + 1} finished: best score = {best_score:.8f}"
        )
        if best_score < global_best_score:
            global_best_score = best_score
            global_best_seq = best_seq.copy()

        if time.time() >= deadline:
            break

    # Final sanity check: evaluate through the official function (if imported)
    try:
        # The harness will import evaluate_sequence; calling it here is safe.
        from __main__ import evaluate_sequence  # type: ignore
        official_score = evaluate_sequence(global_best_seq.tolist())
        print(f"Official evaluation score: {official_score:.8f}")
    except Exception:
        # If the harness does not expose evaluate_sequence yet, ignore.
        pass

    return global_best_seq.tolist()
