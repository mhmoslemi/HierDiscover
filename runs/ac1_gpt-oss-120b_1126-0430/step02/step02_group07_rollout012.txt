def _hill_climb(a, max_iters, time_end, rng):
    n = len(a)
    cur = a.copy()
    cur_conv = np.convolve(cur, cur)
    cur_score = 2.0 * n * cur_conv.max()
    best_a = cur.copy()
    best_score = cur_score
    max_idx = np.argmax(cur_conv)
    for it in range(max_iters):
        if time.time() > time_end: break
        # contributions
        idx_arr = np.arange(n)
        rev = max_idx - idx_arr
        mask = (rev >= 0) & (rev < n)
        contrib = np.zeros_like(cur)
        # need to compute both directions? Actually each i contributes cur[i]*cur[rev[i]]
        contrib[mask] = cur[idx_arr[mask]] * cur[rev[mask]]

        src = int(np.argmax(contrib))
        tgt = int(np.argmin(contrib))
        if src == tgt or cur[src] <= 0:
            src, tgt = rng.choice(n, size=2, replace=False)

        frac = 0.05  # step fraction of src mass
        step = cur[src] * frac * rng.random()
        if step <= 0:

        new = cur.copy()
        new[src] -= step
        new[tgt] += step
        new /= new.sum()
        # compute conv
        new_conv = np.convolve(new, new)
        new_score = 2.0 * n * new_conv.max()
        if new_score < cur_score:
            cur = new
            cur_conv = new_conv
            cur_score = new_score
            max_idx = int(np.argmax(cur_conv))
            if new_score < best_score:
                best_score = new_score
                best_a = new.copy()
        # else optionally accept with probability
    return best_a, best_score

"""
Propose a step‑function (list of non‑negative heights) that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

where `conv` is the discrete self‑convolution.  The optimisation is
performed in two stages:

1. **Smoothed max‑gradient descent** – the non‑smooth objective is
   approximated by a log‑sum‑exp (soft‑max) with a temperature β.
   Using FFT‑based convolutions we compute the gradient of the smoothed
   objective efficiently and update the distribution with a
   multiplicative (exponentiated‑gradient) step, re‑normalising onto the
   probability simplex after each iteration.  β is increased
   monotonically, turning the smooth surrogate into the true max.

2. **Local hill‑climbing refinement** – a very cheap greedy move transfers
   a small amount of mass from the index contributing most to the
   current maximal convolution entry to the index contributing least.
   The step size decays with the iteration count.

The procedure runs for the allotted time (default 950 s) and returns
the best sequence found.  It starts from the provided baseline
`height_sequence_1` and a few random restarts to avoid local minima.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_to_simplex(v):
    """Project a vector onto the probability simplex {x≥0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


def _objective_from_conv(conv, n):
    """Exact objective value given a convolution vector (sum of a = 1)."""
    return 2.0 * n * conv.max()


def _real_convolve_fft(a, b, size):
    """
    Compute the full convolution a*b using a real FFT of length `size`.
    The returned array has length len(a)+len(b)-1.
    """
    A = np.fft.rfft(a, size)
    B = np.fft.rfft(b, size)
    C = A * B
    conv = np.fft.irfft(C, size)
    return conv[: len(a) + len(b) - 1]


def _smooth_gradient(a, beta, size_fft):
    """
    For a probability vector a (sum = 1) compute:
        * conv = a * a                (full convolution)
        * grad = ∂/∂a ( soft‑max_β(conv) )
    The gradient is derived analytically and evaluated with FFTs.
    Returns (conv, grad).
    """
    n = a.shape[0]

    # FFT of a (re‑used for both conv and correlation)
    A = np.fft.rfft(a, size_fft)

    # full self‑convolution (length 2n‑1)
    conv_full = np.fft.irfft(A * A, size_fft)
    conv = conv_full[: 2 * n - 1]

    # soft‑max weights
    mx = conv.max()
    w = np.exp(beta * (conv - mx))          # max(w) = 1
    Z = w.sum()

    # correlation of a with w (via convolution with reversed w)
    w_rev = w[::-1]
    B_rev = np.fft.rfft(w_rev, size_fft)
    corr_full = np.fft.irfft(A * B_rev, size_fft)
    corr = corr_full[: 3 * n - 2]           # length n + (2n‑1) – 1

    # indices that pick the needed slice for the gradient
    idx = 2 * n - 2 - np.arange(n)
    grad = (2.0 / Z) * corr[idx]

    return conv, grad


def _run_gradient_descent(init_a, n, size_fft, time_deadline,
                         rng,
                         max_iters=5000,
                         lr_start=15.0,
                         lr_end=0.5,
                         beta_start=1.0,
                         beta_end=20000.0):
    """
    Perform the smoothed‑max gradient descent from a starting vector.

    Returns the best vector (as a NumPy array) and its exact objective.
    """
    a = init_a.copy()
    a = _project_to_simplex(a)          # safety

    # initial convolution and objective
    A = np.fft.rfft(a, size_fft)
    conv_full = np.fft.irfft(A * A, size_fft)
    conv = conv_full[: 2 * n - 1]
    best_obj = _objective_from_conv(conv, n)
    best_a = a.copy()

    for it in range(max_iters):
        # time check
        if time.time() > time_deadline:
            break

        # linear schedules for lr and beta
        frac = it / max_iters
        lr = lr_start * (1.0 - frac) + lr_end * frac
        beta = beta_start + (beta_end - beta_start) * frac

        # gradient of the smooth surrogate
        conv, grad = _smooth_gradient(a, beta, size_fft)

        # multiplicative (exponentiated‑gradient) update
        a = a * np.exp(-lr * grad)

        # renormalise onto the simplex
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        # recompute convolution for the *new* a (needed for objective)
        A = np.fft.rfft(a, size_fft)
        conv_full = np.fft.irfft(A * A, size_fft)
        conv = conv_full[: 2 * n - 1]

        cur_obj = _objective_from_conv(conv, n)

        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            # occasional progress report
            print(f"[gd] it={it} obj={cur_obj:.6f}")

    return best_a, best_obj


def _hill_climb(a, n, time_deadline, rng, max_iters=20000):
    """
    Very cheap greedy refinement: move a small fraction of mass from the
    index that contributes most to the current maximal convolution entry
    to the index that contributes the least.
    """
    cur = a.copy()
    cur_conv = np.convolve(cur, cur)
    cur_obj = _objective_from_conv(cur_conv, n)
    best = cur.copy()
    best_obj = cur_obj

    for it in range(max_iters):
        if time.time() > time_deadline:
            break

        # locate the (one) index where the convolution attains its maximum
        max_idx = int(np.argmax(cur_conv))

        # contribution of each position to that peak
        idx_arr = np.arange(n)
        rev = max_idx - idx_arr
        mask = (rev >= 0) & (rev < n)
        contrib = np.zeros_like(cur)
        contrib[mask] = cur[idx_arr[mask]] * cur[rev[mask]]

        src = int(np.argmax(contrib))
        tgt = int(np.argmin(contrib))

        if src == tgt or cur[src] <= 0.0:
            src, tgt = rng.choice(n, size=2, replace=False)

        # move a decaying fraction of the source mass
        step_frac = 0.08 * max(0.0, 1.0 - it / max_iters) + 0.02
        step = cur[src] * rng.random() * step_frac
        if step <= 0.0:
            continue

        new = cur.copy()
        new[src] -= step
        new[tgt] += step
        new /= new.sum()

        new_conv = np.convolve(new, new)
        new_obj = _objective_from_conv(new_conv, n)

        # accept only improving moves (tiny chance to accept worse)
        if new_obj < cur_obj - 1e-12 or rng.random() < 0.001:
            cur = new
            cur_conv = new_conv
            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best = new.copy()
                print(f"[hc] it={it} new_best={new_obj:.6f}")

        # occasional re‑projection (numerical safety)
        if it % 500 == 0:
            cur = _project_to_simplex(cur)
            cur_conv = np.convolve(cur, cur)
            cur_obj = _objective_from_conv(cur_conv, n)

    return best, best_obj


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed=123, budget_s=950, **kwargs):
    """
    Return a list of non‑negative heights (step‑function) that minimizes the
    given evaluation function.  The routine respects the supplied time
    budget (seconds) and uses a combination of smoothed gradient descent
    and greedy hill climbing.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s

    # ------------------------------------------------------------------
    # Length of the problem (use the global baseline if present)
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)  # provided by the harness
    except Exception:
        n = 2244                     # safe default

    # ------------------------------------------------------------------
    # Baseline – start from the provided construction (if any)
    # ------------------------------------------------------------------
    try:
        baseline = np.array(height_sequence_1, dtype=float)
        if baseline.sum() == 0:
            baseline = np.full(n, 1.0 / n)
    except Exception:
        baseline = np.full(n, 1.0 / n)

    baseline = _project_to_simplex(baseline)
    baseline_obj = _objective_from_conv(np.convolve(baseline, baseline),
                                        n)
    best_a = baseline.copy()
    best_obj = baseline_obj
    print(f"[init] n={n}, baseline objective={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Pre‑compute FFT size (must accommodate a + w_rev convolution)
    # ------------------------------------------------------------------
    max_len = 3 * n - 2                     # length of a * w_rev
    size_fft = 1 << ((max_len - 1).bit_length())

    # ------------------------------------------------------------------
    # Helper to try a new candidate (updates the global best)
    # ------------------------------------------------------------------
    def _maybe_update(candidate_a, candidate_obj, label):
        nonlocal best_a, best_obj
        if candidate_obj < best_obj - 1e-12:
            best_obj = candidate_obj
            best_a = candidate_a.copy()
            print(f"[{label}] improved to {candidate_obj:.6f}")

    # ------------------------------------------------------------------
    # 1. Gradient‑descent from several starting points
    # ------------------------------------------------------------------
    starts = []

    # a) the baseline itself
    starts.append(baseline)

    # b) a few random Dirichlet‑style points
    for _ in range(3):
        raw = rng.exponential(scale=1.0, size=n)
        starts.append(_project_to_simplex(raw))

    # c) random power‑law shapes
    idx = np.arange(n, dtype=float)
    for _ in range(3):
        alpha = rng.uniform(0.2, 1.5)
        shift = rng.uniform(0.01, 4.0)
        raw = np.power(idx + shift, -alpha)
        starts.append(_project_to_simplex(raw))

    # d) mixtures of two power‑law components
    for _ in range(3):
        alpha1 = rng.uniform(0.2, 1.5)
        shift1 = rng.uniform(0.01, 4.0)
        alpha2 = rng.uniform(0.2, 1.5)
        shift2 = rng.uniform(0.01, 4.0)
        p = rng.random()
        raw = p * np.power(idx + shift1, -alpha1) + (1 - p) * np.power(idx + shift2, -alpha2)
        starts.append(_project_to_simplex(raw))

    # run optimiser for each start (stop if time is exhausted)
    for sidx, start_vec in enumerate(starts):
        if time.time() > deadline - 5:
            break
        label = f"gd{sidx}"
        a_opt, obj_opt = _run_gradient_descent(start_vec, n, size_fft,
                                               deadline, rng)
        _maybe_update(a_opt, obj_opt, label)

    # ------------------------------------------------------------------
    # 2. Final local hill‑climbing refinement (on the current best)
    # ------------------------------------------------------------------
    if time.time() < deadline - 2:
        a_refined, obj_refined = _hill_climb(best_a, n, deadline, rng)
        _maybe_update(a_refined, obj_refined, "final_hc")

    # ------------------------------------------------------------------
    # Return the best sequence found (as a plain Python list)
    # ------------------------------------------------------------------
    print(f"[final] best objective = {best_obj:.6f}")
    return best_a.tolist()
