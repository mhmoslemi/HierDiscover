def propose_candidate(seed=42, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5

    n = 2000  # can try 2000
    
    # initialization: Catalan
    f = catalan_seq(n)
    f = f / f.sum()
    
    best_f = f.copy()
    best_score = _fast_evaluation(best_f)
    print("init score:", best_score)
    
    # precompute indices for building conv matrix: For each row k, a list of columns i where f_i contributes.
    # We'll rebuild each iteration.
    
    iteration = 0
    while time.time() < deadline:
        # compute M
        M = max_conv(f)  # via FFT
        # Build A matrix: shape (2n-1, n)
        A = np.zeros((2*n-1, n), dtype=float)
        for i in range(n):
            A[i:i+n, i] = f
        # LP: maximize sum(g) subject to A @ g <= M, g >= 0
        c = -np.ones(n)  # minimize -sum(g)
        bounds = [(0, None)] * n
        # b vector: M repeated for each row
        b_ub = np.full(2*n-1, M)
        # Solve LP
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            print("LP failed at iteration", iteration)
            break
        g = res.x
        # scale to sum = 1
        g_sum = g.sum()
        if g_sum <= 0:
            # random fallback
            g = np.full(n, 1.0/n)
            g_sum = 1.0
        g = g / g_sum # now sum 1
        # Combine
        t = 0.2  # maybe adapt decreasing
        f_new = (1 - t) * f + t * g
        # Normalize sum to 1 (should be already)
        f_new = f_new / f_new.sum()
        # Evaluate
        score = _fast_evaluation(f_new)
        if score < best_score:
            best_score = score
            best_f = f_new.copy()
            print(f"iter {iteration} score {score:.6f}")
        # Accept new f (always moving but maybe revert if no improvement)
        f = f_new
        iteration += 1
        # optionally adjust t, break if t too small
        if iteration > 200:
            break
    # Return best sequence as list
    return best_f.tolist()

"""
low‑score step‑function search
--------------------------------
1. Start from the Catalan (arcsine) sequence, normalised to sum = 1.
2. Iteratively improve the shape by solving a linear programme:
   maximise the ℓ₁‑mass of a new distribution `g` while keeping the
   convolution `f * g` below the current maximal self‑convolution `M`.
   This pushes mass to indices where `f` is small.
   The result is scaled back to sum = 1 and mixed with the current `f`.
3. After the LP phase a few greedy mass‑transfer moves are performed to
   fine‑tune the peak of the self‑convolution.
The routine respects the 1000 s budget, prints intermediate scores and
returns the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Fast evaluation (identical to evaluate_sequence but without the security
# checks and using an FFT based convolution).
# ----------------------------------------------------------------------
def _fast_evaluation(seq: np.ndarray) -> float:
    """Return the same metric as evaluate_sequence but much faster."""
    n = seq.size
    if n == 0:
        return np.inf
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Catalan (arcsine) seed – coefficients of (1‑z)^‑1/2.
# ----------------------------------------------------------------------
def _catalan_sequence(length: int) -> np.ndarray:
    """Return the first ``length`` Catalan‑like coefficients."""
    seq = np.empty(length, dtype=float)
    a = 1.0
    seq[0] = a
    for i in range(1, length):
        a *= (2 * i - 1) / (2.0 * i)
        seq[i] = a
    return seq


# ----------------------------------------------------------------------
# Helpers for the LP refinement
# ----------------------------------------------------------------------
def _conv_max(seq: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of ``seq``."""
    n = seq.size
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    return float(conv.max())


def _build_conv_matrix(f: np.ndarray, out: np.ndarray) -> None:
    """In‑place fill ``out`` with the convolution matrix of ``f``.
    out has shape (2*n-1, n) and on row k, column i stores f[k-i] if
    0 ≤ k-i < n, otherwise 0.
    """
    n = f.size
    out.fill(0.0)
    for i in range(n):
        out[i:i + n, i] = f


def _lp_refine(seq: np.ndarray, deadline: float) -> np.ndarray:
    """Iteratively improve ``seq`` by linear‑programming steps."""
    n = seq.size
    best_seq = seq.copy()
    best_score = _fast_evaluation(best_seq)

    # pre‑allocate the convolution matrix (dense, but re‑used)
    A = np.empty((2 * n - 1, n), dtype=np.float64)

    it = 0
    stagnation = 0

    while time.time() < deadline:
        M = _conv_max(seq)

        # Build the linear constraints A·g ≤ M
        _build_conv_matrix(seq, A)

        # LP: maximise  Σ g_i   (i.e. minimise  –Σ g_i)
        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': max(0.0, deadline - time.time())})

        if not res.success:
            # something went wrong – abort the LP phase
            print(f"[LP] iteration {it} failed: {res.message}")
            break

        g = res.x
        g_sum = g.sum()
        if g_sum == 0.0:
            break
        # Normalise to sum = 1 (the metric is scale‑invariant)
        g = g / g_sum

        # Decide whether to take g outright or to blend it with the current seq
        score_g = _fast_evaluation(g)

        if score_g < best_score:
            # Direct replacement gives a strict improvement – accept it
            new_seq = g
        else:
            # Gentle convex combination (step size decays with √iteration)
            t = 0.2 / np.sqrt(it + 1)
            if t > 0.5:
                t = 0.5
            new_seq = (1.0 - t) * seq + t * g
            new_seq = new_seq / new_seq.sum()

        score_new = _fast_evaluation(new_seq)

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_seq = new_seq.copy()
            print(f"[LP] iter {it:3d}  score {best_score:.6f}")
            stagnation = 0
        else:
            stagnation += 1

        # Prepare for the next iteration
        seq = new_seq
        it += 1

        # Stop after a reasonable number of iterations or if stuck
        if it >= 250 or stagnation >= 30:
            break

    return best_seq


# ----------------------------------------------------------------------
# Simple greedy local move – moves a small fraction of mass from the
# index that contributes most to the current maximal convolution entry to
# an index that contributes almost nothing.
# ----------------------------------------------------------------------
def _greedy_refine(seq: np.ndarray, deadline: float) -> np.ndarray:
    """Fast greedy improvement (few hundred iterations max)."""
    n = seq.size
    best_seq = seq.copy()
    best_score = _fast_evaluation(best_seq)

    # FFT length does not change
    fft_len = 1 << ((2 * n - 1).bit_length())

    it = 0
    while time.time() < deadline and it < 2000:
        fa = np.fft.rfft(seq, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        k = int(conv.argmax())

        # contribution weights of each position to the maximal entry k
        i_arr = np.arange(n)
        j_arr = k - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        w = np.zeros_like(seq)
        w[mask] = seq[j_arr[mask]]

        # source = index with largest product seq[i] * w[i]
        contrib = seq * w
        src = int(np.argmax(contrib))
        sink = int(np.argmin(w))

        if seq[src] < 1e-12:
            break

        eps = 0.05 * seq[src]               # move 5 % of the source mass
        new_seq = seq.copy()
        new_seq[src] -= eps
        new_seq[sink] += eps

        score = _fast_evaluation(new_seq)

        if score < best_score - 1e-12:
            seq = new_seq
            best_seq = new_seq.copy()
            best_score = score
            if it % 200 == 0:
                print(f"[greedy] iter {it:4d}  score {best_score:.6f}")
        else:
            # occasional tiny jitter to escape flat plateaus
            if np.random.rand() < 0.02:
                jitter_idx = np.random.randint(n)
                jitter = np.random.randn() * 1e-3
                new_seq = seq.copy()
                new_seq[jitter_idx] = max(0.0, new_seq[jitter_idx] + jitter)
                seq = new_seq
        it += 1

    return best_seq


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights aiming at a low evaluation score.
    The routine works within ``budget_s`` seconds (default 1000).
    It builds a Catalan seed, improves it with a linear‑programming phase
    and finishes with a short greedy fine‑tuning.
    """
    np.random.seed(seed)
    start = time.time()
    # keep a small safety margin for the harness
    deadline = start + budget_s - 5.0

    # ------------------------------------------------------------------
    # 1.  Initialise – Catalan (arcsine) coefficients, normalised to sum = 1.
    # ------------------------------------------------------------------
    n = 2000                      # a size that balances quality and speed
    seq = _catalan_sequence(n).astype(float)
    seq /= seq.sum()              # sum = 1  (scale does not matter for the metric)

    init_score = _fast_evaluation(seq)
    print(f"[init] length={n}  score={init_score:.6f}")

    # ------------------------------------------------------------------
    # 2.  Linear‑programming refinement (takes the bulk of the time budget).
    # ------------------------------------------------------------------
    lp_deadline = start + 0.78 * (budget_s - 5.0)   # leave ~20 % for final polishing
    if time.time() < lp_deadline:
        seq = _lp_refine(seq, lp_deadline)
        lp_score = _fast_evaluation(seq)
        print(f"[after LP] score={lp_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Short greedy polishing (uses any remaining time).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq = _greedy_refine(seq, deadline)
        final_score = _fast_evaluation(seq)
        print(f"[final] best score={final_score:.6f}")
    else:
        final_score = _fast_evaluation(seq)
        print(f"[final] best score={final_score:.6f}")

    # Return the best sequence as a plain Python list.
    return seq.tolist()
