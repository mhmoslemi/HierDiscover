"""
Propose a step‑function (list of non‑negative heights) that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of three phases:

1. **Initialization** – start from the provided reference sequence
   (``height_sequence_1``) and normalise it to a probability vector.

2. **Iterated linear‑programming refinement** – given the current vector
   `f` with maximal convolution value `M = max(f*f)`,
   we solve the linear programme

        maximise   Σ g_i
        subject to  (f * g)[k] ≤ M   for all k = 0 … 2n‑2
                    g_i ≥ 0

   The programme returns a vector `g` with sum larger than one.
   After scaling `g` to sum = 1 we update `f ← (1‑t)·f + t·g`
   (with a slowly decreasing mixing factor `t`).  This is the
   “alternating LP” scheme described in the problem statement and
   provably yields a non‑increasing objective value.

3. **Fine‑grained hill‑climbing** – once the LP phase stalls,
   a cheap coordinate‑wise hill‑climber moves a small amount of mass
   from the index that contributes most to the current maximal
   convolution entry to the index that contributes least.
   The step size shrinks over time and occasional random moves
   help escape plateaus.

The whole search respects the 1000 s budget and returns the best
sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _evaluate(arr):
    """Fast wrapper around the provided evaluate_sequence."""
    return evaluate_sequence(arr.tolist())


def _project_to_simplex(v):
    """Project a non‑negative vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _build_conv_constraint_matrix(f):
    """
    Construct the sparse matrix A (size (2n‑1) × n) such that
          (A @ g)[k] = (f * g)[k]   for all k.
    The i‑th column of A contains the reversed copy of f shifted by i.
    """
    # Use a dense representation of the index pattern (i, j) with i+j = k.
    n = f.shape[0]
    i_idx = np.repeat(np.arange(n, dtype=np.int32), n)      # i repeated n times
    j_idx = np.tile(np.arange(n, dtype=np.int32), n)        # j cycling
    rows = i_idx + j_idx                                    # k = i + j
    cols = i_idx                                            # column = i
    data = f[j_idx]                                         # value = f[j]
    from scipy import sparse
    return sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def _lp_step(f, M, i_idx, j_idx):
    """
    Solve the linear programme that maximises Σ g_i under
    (f * g)[k] ≤ M for all k.  Returns a normalised vector g (sum = 1)
    or None if the solver fails.
    """
    import scipy.optimize as opt
    import scipy.sparse as sp

    n = f.shape[0]

    # Build the sparse constraint matrix using the pre‑computed index arrays.
    # data = f[j_idx]  (re‑computed each call)
    data = f[j_idx]
    rows = i_idx + j_idx
    cols = i_idx
    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    b = np.full(2 * n - 1, M, dtype=float)

    # maximise Σ g_i  <=> minimise –Σ g_i
    c = -np.ones(n, dtype=float)

    # Bounds: g_i ≥ 0
    bounds = [(0, None)] * n

    # Let HiGHS run a bit fast – we set a generous iteration limit.
    try:
        res = opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                         method='highs', options={'presolve': True,
                                                  'time_limit': 30.0})
    except Exception:          # fall back to the old interior‑point method
        try:
            res = opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                             method='interior-point', options={'maxiter': 500})
        except Exception:
            return None

    if not res.success or res.x is None:
        return None

    g_raw = res.x
    s = g_raw.sum()
    if s <= 0:
        return None
    return g_raw / s


def _hill_climb(arr, cur_score, rng, time_limit):
    """
    Simple hill‑climbing routine that moves mass from the index
    contributing most to the maximal convolution entry to the index
    contributing least.  The routine stops after `time_limit` seconds.
    Returns the best vector seen and its score.
    """
    n = arr.shape[0]
    best_vec = arr.copy()
    best_sc = cur_score
    cur = arr.copy()
    cur_sc = cur_score
    conv_cur = np.convolve(cur, cur)

    start = time.time()
    it = 0
    max_iters = 200_000
    while time.time() - start < time_limit and it < max_iters:
        it += 1

        max_k = int(np.argmax(conv_cur))

        # contribution of each index to conv[max_k]
        i = np.arange(n)
        j = max_k - i
        mask = (j >= 0) & (j < n)
        contrib = np.zeros_like(cur)
        contrib[mask] = cur[i[mask]] * cur[j[mask]]

        src = int(np.argmax(contrib))
        tgt = int(np.argmin(contrib))

        if src == tgt or cur[src] <= 0:
            src, tgt = rng.choice(n, size=2, replace=False)

        # step size shrinks linearly with iteration count
        frac = 0.10 * max(0.0, 1.0 - it / max_iters) + 0.02   # between 0.02 and 0.12
        step = cur[src] * rng.random() * frac
        if step <= 0:
            continue

        new = cur.copy()
        new[src] -= step
        new[tgt] += step
        new = _project_to_simplex(new)       # keep sum = 1 and non‑negative

        new_sc = _evaluate(new)
        if new_sc < best_sc - 1e-12:
            best_sc = new_sc
            best_vec = new.copy()
            print(f"[hill] iter={it} new best={new_sc:.6f}")

        if new_sc < cur_sc - 1e-12 or rng.random() < 0.001:
            cur = new
            cur_sc = new_sc
            conv_cur = np.convolve(cur, cur)

    return best_vec, best_sc

def propose_candidate(seed=123, budget_s=950, **kwargs):
    """
    Return a list of heights (non‑negative, sum = 1) that minimises the
    evaluation function.  The routine first performs an LP‑based
    refinement of the supplied reference sequence and then finishes
    with a fast hill‑climber.
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # Determine problem size and initialise from the reference sequence
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)   # provided by the harness
        init = np.asarray(height_sequence_1, dtype=float)
    except Exception:
        # fallback – uniform distribution of a reasonable length
        n = 2244
        init = np.full(n, 1.0 / n, dtype=float)

    # normalise to a probability simplex (sum = 1)
    init = np.maximum(init, 0.0)
    init_sum = init.sum()
    if init_sum <= 0:
        init = np.full(n, 1.0 / n, dtype=float)
    else:
        init = init / init_sum

    best_seq = init.copy()
    best_score = _evaluate(best_seq)
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Pre‑compute index arrays used for the constraint matrix
    # ------------------------------------------------------------------
    i_idx = np.repeat(np.arange(n, dtype=np.int32), n)      # i repeated n times
    j_idx = np.tile(np.arange(n, dtype=np.int32), n)        # j cycling

    # ------------------------------------------------------------------
    # Phase 1 – iterated LP refinement
    # ------------------------------------------------------------------
    f = best_seq.copy()
    it_lp = 0
    max_lp_iters = 30          # a modest number – each iteration is expensive
    while time.time() - start < budget_s - 30 and it_lp < max_lp_iters:
        it_lp += 1

        conv_f = np.convolve(f, f)
        M = conv_f.max()
        # Solve the LP (may return None if something goes wrong)
        g = _lp_step(f, M, i_idx, j_idx)
        if g is None:
            print("[lp] LP failed – exiting LP phase")
            break

        # Evaluate the raw candidate g (its own self‑conv)
        score_g = _evaluate(g)
        if score_g < best_score:
            best_score = score_g
            best_seq = g.copy()
            print(f"[lp] iter={it_lp} improved via g: {score_g:.6f}")

        # Mix the new vector into f.  Mixing factor decays slowly.
        t = 0.5 * max(0.1, 1.0 - it_lp / max_lp_iters)   # starts at 0.5, ends ~0.05
        f = (1.0 - t) * f + t * g
        f = _project_to_simplex(f)   # re‑project to avoid drift
        cur_score = _evaluate(f)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = f.copy()
            print(f"[lp] iter={it_lp} improved via mix: {cur_score:.6f}")

    print(f"[lp] finished after {it_lp} iterations, best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – cheap hill‑climbing refinement (uses the previous code)
    # ------------------------------------------------------------------
    hill_time = max(0.0, budget_s - (time.time() - start) - 5.0)
    if hill_time > 5.0:
        print(f"[hill] starting hill‑climbing for ~{hill_time:.1f}s")
        best_vec, best_sc = _hill_climb(best_seq, best_score, rng, hill_time)
        best_seq = best_vec
        best_score = best_sc

    print(f"[final] best score achieved: {best_score:.6f}")
    return best_seq.tolist()
