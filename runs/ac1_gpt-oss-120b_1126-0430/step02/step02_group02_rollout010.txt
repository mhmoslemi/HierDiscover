#!/usr/bin/env python
"""
Propose a step‑function (non‑negative heights) with a low evaluation score.

Algorithm
---------
1. Choose a working length `n` (default 20000, or the length of
   `height_sequence_1` if it exists and is large enough).
2. Generate a portfolio of candidate shapes (power‑law, exponential,
   power‑exponential, plateau+exponential, triangle, random Dirichlet).
   Each candidate is normalised to sum‑to‑one and its cheap score
   `2·n·max(a*a)` is computed via FFT.  The best few candidates are kept.
3. Starting from the best candidate, run a *smoothed* gradient descent:
   * The true objective is `max_k (a*a)[k]`; we approximate it by the
     smooth function  `ε·log Σ_k exp(conv_k/ε)`.
   * The gradient of this smooth objective can be written as a
     weighted cross‑correlation and is computed with two FFTs per
     iteration.
   * An Adam‑style update with a diminishing step size is applied,
     followed by projection onto the probability simplex.
   * The smoothing parameter `ε` is gradually reduced (∝ 1/√t).
4. The best exact score (using the provided `evaluate_sequence`) is
   reported and the corresponding sequence is returned.
The routine respects the 1000 s budget and prints occasional progress
messages.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – top level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (sum=1, non‑negative)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # All entries are non‑positive – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Return the linear autocorrelation a*a (full length 2·n‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len).real[: 2 * a.size - 1]
    return conv


def _cheap_score(a: np.ndarray, conv_fft_len: int) -> float:
    """Score 2·n·max(a*a) assuming Σ a = 1 (scale‑invariant)."""
    conv = _max_convolution_fft(a, conv_fft_len)
    return float(2.0 * a.size * conv.max())


# ----------------------------------------------------------------------
# Shape generators – all return a normalised (sum=1) ndarray
# ----------------------------------------------------------------------


def _shape_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    seq = idx ** (-p)
    return seq / seq.sum()


def _shape_exponential(n: int, beta: float) -> np.ndarray:
    """a_i ∝ exp(-beta·i)."""
    idx = np.arange(n, dtype=np.float64)
    seq = np.exp(-beta * idx)
    return seq / seq.sum()


def _shape_power_exponential(n: int, p: float, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}·exp(-beta·i)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    seq = (idx ** (-p)) * np.exp(-beta * (idx - 1))
    return seq / seq.sum()


def _shape_plateau_exponential(n: int, M: int, beta: float) -> np.ndarray:
    """
    First M entries equal, then exponential tail.
    a_i = A·[1 (i<M) else exp(-beta·(i-M))], normalised.
    """
    if M < 0:
        M = 0
    if M > n:
        M = n
    tail_len = n - M
    if beta == 0.0:
        tail = np.ones(tail_len, dtype=np.float64)
    else:
        tail = np.exp(-beta * np.arange(tail_len, dtype=np.float64))
    seq = np.concatenate([np.ones(M, dtype=np.float64), tail])
    return seq / seq.sum()


def _shape_triangle(n: int, L: int) -> np.ndarray:
    """Triangular shape: a_i = max(L-i,0)."""
    if L <= 0:
        return np.full(n, 1.0 / n)
    idx = np.arange(n, dtype=np.float64)
    tri = np.maximum(L - idx, 0.0)
    if tri.sum() == 0:
        return np.full(n, 1.0 / n)
    return tri / tri.sum()


def _shape_random_dirichlet(n: int, alpha: float = 0.5) -> np.ndarray:
    """Random Dirichlet vector with concentration `alpha`."""
    seq = np.random.gamma(alpha, 1.0, size=n)
    return seq / seq.sum()
