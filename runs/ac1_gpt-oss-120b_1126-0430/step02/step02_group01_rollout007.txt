    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0

    # Determine n: use if height_sequence_1 exists -> n0 = len(height_sequence_1)
    # We'll use n = max(n0*2, 5000) maybe to increase length.
    # Option: set n = 8000.

    else:

    # If initial length differs from n, we need to resize:

    eta_end = 0.001
    tau_start = None
    tau_end = 1e-5

        tau = tau_start * (1 - progress) + tau_end * progress
        # compute softmax weights
        # shifted so max is 0 for numerical stability
        shifted = conv - max_val
        # exp with temperature
        # avoid overflow: if tau very small, only largest entry ~0 will dominate; we can treat as argmax.
        if tau < 1e-12:

        max_new = conv_new.max()

    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[final] evaluate_sequence O={final_score:.6f}")
    return best_seq.tolist()

"""
propose_candidate
-----------------
A gradient‑descent based optimizer that flattens the self‑convolution
of a non‑negative step function.  The algorithm works directly on the
simplex (∑a_i = 1, a_i ≥ 0) by iteratively:

1. Computing the full convolution a*a.
2. Forming a smooth approximation of the max‑entry using a soft‑max
   (temperature τ that anneals to zero).
3. Obtaining the gradient of this smooth max w.r.t. a via a
   cross‑correlation (two fast convolutions).
4. Taking a momentum‑aided projected gradient step and occasionally
   applying a tiny smoothing filter to keep the shape regular.
5. Keeping the best sequence seen and performing occasional random
   perturbations to escape plateaus.

The routine respects a 1000 s time budget, prints progress, and finally
returns the best coefficient list found.  It works with an optional
starting point supplied through the global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np


# -------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        # all entries are zero – return the uniform simplex point
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# -------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective used for reporting: O = 2·n·max(conv(a,a))
       (the denominator ∑a is 1 because a lives on the simplex)."""
    n = a.size
    conv = np.convolve(a, a)
    return 2.0 * n * conv.max()

    elif n0 > target_n:
        a = a0[:target_n]
    else:
        a = a0.copy()
    a = a / a.sum()                     # guarantee simplex
    n = a.size

    best_seq = a.copy()

    print(f"[init] n={n}, objective={best_score:.6f}")

    # -----------------------------------------------------------------
    # Optimisation hyper‑parameters
    max_iters = 150_000
    iter_cnt = 0

    eta_start = 0.20          # initial learning rate

    tau_start_factor = 0.30   # initial temperature is this fraction of max(conv)
    tau_end = 1e-6

    momentum = np.zeros_like(a)
    beta = 0.85                # momentum decay

        # ---- temperature (soft‑max smoothing) ---------------------------------
        progress = iter_cnt / max_iters
        tau_start = max_val * tau_start_factor
        tau = tau_start * (1.0 - progress) + tau_end * progress

        # soft‑max over convolution entries
