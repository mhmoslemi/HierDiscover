def lp_improve(a, M, active_idx):
    # Build A_sub matrix with only active constraints.
    # active_idx: indices of conv where conv close to M.
    # shape (len(active_idx), n)
    n = a.shape[0]
    rows = []
    for k in active_idx:
        # row_j = a_{k-j} if within range else 0
        row = np.zeros(n)
        j_start = max(0, k - (n-1))
        j_end = min(n-1, k)
        # positions j from j_start..j_end
        # a_{k-j} index in a: let i = k - j (so i runs from k - j_end .. k - j_start)
        # compute vectorized:
        j_range = np.arange(j_start, j_end+1)
        i_range = k - j_range
        row[j_range] = a[i_range]
        rows.append(row)
    A_sub = np.vstack(rows) if rows else np.empty((0, n))
    # objective: maximize sum(g) -> minimize -sum(g)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    # b_ub vector:
    b_ub = np.full(A_sub.shape[0], M)
    # Solve
    res = linprog(c, A_ub=A_sub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        g = res.x
    else:
        # fallback: return a (no improvement)
        return a
    # Normalize g
    if g.sum() == 0:
        return a
    g_norm = g / g.sum()
    # Combine
    new_a = 0.5 * a + 0.5 * g_norm
    return new_a

"""
Proposes a step‑function (non‑negative vector on the simplex) that
minimises the supplied ``evaluate_sequence`` objective.

The method combines three ideas:

1. **Projected Adam on a smooth approximation of the max‑convolution**  
   The convolution b = a∗a is turned into a soft‑max
   w_k ∝ exp(α·(b_k−max(b))) and the gradient of the
   weighted sum is ∇ = 2·(a ⊛ w_rev).  Adam’s adaptive steps
   (first/second moments) together with projection onto the
   probability simplex give a fast, robust descent.

2. **Active‑set linear‑programming refinement**  
   Every few hundred iterations the entries attaining (or almost
   attaining) the current maximum are collected.  Using only those
   rows of the Toeplitz convolution matrix we solve the LP

        maximise   Σ g_i
        subject to a∗g ≤ M   (active rows only)
                    g_i ≥ 0

   The solution g is normalised and blended with the current a.
   Because the LP respects the active constraints, the maximal
   convolution value M can often be reduced dramatically in a
   single step.

3. **Multiple initial seeds and occasional random restarts**  
   Catalan‑like, geometric decay, a random simplex point and (if
   present) the previously found ``height_sequence_1`` are used as
   seeds.  The best of them is taken forward; if progress stalls a
   small random perturbation is applied.

All heavy work (convolutions) uses an FFT for O(n log n) cost.
The routine runs until the supplied time budget (≈ 1000 s) expires
and returns the best sequence found as a plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _catalan_sequence(n: int) -> np.ndarray:
    """Catalan‑like coefficients a_i = C(2i,i)/4**i, normalised to sum 1."""
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * (i - 1) + 1) / (2.0 * i)
    a /= a.sum()
    return a


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u > cssv / np.arange(1, n + 1))[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via a real FFT."""
    n = x.size + y.size - 1
    size = 1 << (n - 1).bit_length()          # next power of two
    X = np.fft.rfft(x, size)
    Y = np.fft.rfft(y, size)
    Z = X * Y
    conv = np.fft.irfft(Z, size)[:n]
    return conv.real


def _eval_score(a: np.ndarray) -> float:
    """Exact value of ``evaluate_sequence`` for a non‑negative simplex vector."""
    n = a.size
    b = _convolve_fft(a, a)
    return float(2.0 * n * b.max())   # sum(a) == 1 ⇒ denominator = 1


def _softmax_weights(b: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max over the convolution entries (higher weight on the current max)."""
    max_b = b.max()
    w = np.exp(alpha * (b - max_b))
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full_like(w, 1.0 / w.size)
    return w / w_sum


def _grad_smooth_max(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Gradient of the smooth max: 2·(a ⊛ w_rev) restricted to the original length."""
    n = a.size
    w_rev = w[::-1]
    conv_full = _convolve_fft(a, w_rev)          # length = 3·n‑2
    idx = (2 * n - 2) - np.arange(n)              # map to gradient components
    return 2.0 * conv_full[idx]


def _lp_improve(a: np.ndarray, M: float, active_idx: np.ndarray) -> np.ndarray:
    """
    One active‑set LP refinement.
    maximise Σ g_i   s.t.   (a∗g)_k ≤ M   for k∈active_idx,   g_i ≥ 0.
    Returns the blended vector (0.5·a + 0.5·g_norm) projected onto the simplex.
    """
    n = a.shape[0]
    if active_idx.size == 0:
        return a

    # Limit the number of active constraints to keep the LP cheap.
    max_cons = 200
    if active_idx.shape[0] > max_cons:
        active_idx = np.random.choice(active_idx, size=max_cons, replace=False)

    # Build the Toeplitz rows for the chosen k‑indices.
    rows = np.zeros((active_idx.size, n), dtype=float)
    for r, k in enumerate(active_idx):
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        j_range = np.arange(j_start, j_end + 1)
        i_range = k - j_range
        rows[r, j_range] = a[i_range]

    # Add diagonal constraints a_i·g_i ≤ M to guarantee boundedness.
    diag_idx = np.where(a > 1e-12)[0]
    extra = np.zeros((diag_idx.size, n), dtype=float)
    extra[np.arange(diag_idx.size), diag_idx] = a[diag_idx]

    A_ub = np.vstack((rows, extra))
    b_ub = np.full(A_ub.shape[0], M)

    # Objective: maximise Σ g_i  ⇔ minimise -Σ g_i
    c = -np.ones(n)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
    except Exception:
        return a

    if not res.success:
        return a

    g = res.x
    if g.sum() == 0.0:
        return a
    g_norm = g / g.sum()

    # Simple averaging; other mixing rates could be used.
    a_new = 0.5 * a + 0.5 * g_norm
    return _proj_simplex(a_new)

    sequence (as a plain Python list) found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0    # safety margin for final return

    # ------------------------------------------------------------------
    # Problem size and initial pool
    # ------------------------------------------------------------------
    # Prefer the length of the previously supplied sequence, if any.
    if 'height_sequence_1' in globals():
        seq0 = np.maximum(np.asarray(height_sequence_1, dtype=float), 0.0)
        if seq0.size < 2000:            # too short → fall back to default size
            n = 8000
        else:
            n = seq0.size
        seq0 = _proj_simplex(seq0)       # normalise to sum 1
        candidates = [seq0]
    else:
        n = 8000
        candidates = [_catalan_sequence(n)]

    # geometric / exponential families
    for p in np.linspace(0.5, 4.0, 7):
        eps = p / n
        r = 1.0 - eps
        a = eps * (r ** np.arange(n))
        a = _proj_simplex(a)
        candidates.append(a)

    # a random simplex point
    rnd = np.random.rand(n)
    candidates.append(_proj_simplex(rnd))

    # ------------------------------------------------------------------
    # Choose the best starter according to the exact objective
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    for a in candidates:
        sc = _eval_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
    a = best_seq.copy()
    cur_score = best_score
    print(f"[init] n={n}, initial best score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Optimisation hyper‑parameters (Adam + smooth‑max)
    # ------------------------------------------------------------------
    lr = 0.04                # base learning rate for Adam
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros(n)          # first moment
    v = np.zeros(n)          # second moment
    t_iter = 1
    alpha = 1.0              # soft‑max sharpening factor
    max_alpha = 5000.0

    # ------------------------------------------------------------------
    # Main loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1) Evaluate current objective and keep global best
        # --------------------------------------------------------------
        b = _convolve_fft(a, a)
        max_b = b.max()
        cur_score = 2.0 * n * max_b          # sum(a)=1 ⇒ denominator = 1
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{t_iter:07d}] new global best {best_score:.6f}")

        # --------------------------------------------------------------
        # 2) Gradient of smooth‑max and Adam update
        # --------------------------------------------------------------
        # increase the soft‑max temperature slowly (focus on the true max)
        if t_iter % 500 == 0:
            alpha = min(alpha * 1.15, max_alpha)

        w = _softmax_weights(b, alpha)
        grad = _grad_smooth_max(a, w)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_candidate = _proj_simplex(a - step)

        cand_score = _eval_score(a_candidate)
        if cand_score < cur_score:
            a = a_candidate
            cur_score = cand_score
        else:
            # If no improvement, shrink the learning rate a little
            lr = max(lr * 0.7, 1e-6)

        # --------------------------------------------------------------
        # 3) Periodic active‑set LP refinement
        # --------------------------------------------------------------
        if t_iter % 300 == 0:
            # Tiny tolerance to collect only the (almost) tight constraints
            tol = max(max_b * 1e-8, 1e-12)
            active_idx = np.where(b >= max_b - tol)[0]
            a_lp = _lp_improve(a, max_b, active_idx)
            lp_score = _eval_score(a_lp)
            if lp_score < cur_score:
                a = a_lp
                cur_score = lp_score
                # reset Adam moments to avoid stale direction after a jump
                m.fill(0.0)
                v.fill(0.0)
                print(f"[{t_iter:07d}] LP refinement improved to {lp_score:.6f}")

        # --------------------------------------------------------------
        # 4) Occasional random perturbation if progress stalls
        # --------------------------------------------------------------
        if t_iter % 2000 == 0:
            # simple restart around the current best
            a = best_seq + 0.02 * np.random.randn(n)
            a = np.maximum(a, 0.0)
            a = _proj_simplex(a)
            cur_score = _eval_score(a)
            m.fill(0.0)
            v.fill(0.0)
            lr = 0.04
            alpha = 1.0
            print(f"[{t_iter:07d}] random restart (score={cur_score:.6f})")

        t_iter += 1

    print(f"[done] best score found = {best_score:.6f}")
    # Return the best sequence (plain list, as required by the harness)
    return best_seq.tolist()
