def iterate_lp(f, t, deadline):
    while time() < deadline:
        n = len(f)
        conv_f = np.convolve(f,f)
        M = conv_f.max()
        # build A matrix
        A = np.zeros((2*n-1, n), dtype=float)
        for k in range(2*n-1):
            i_low = max(0, k-(n-1))
            i_high = min(k, n-1)
            # For each j in [i_low, i_high], i = k - j
            # So A[k, j] = f[i] = f[k-j]
            # Use vector slice assignment:
            A[k, i_low:i_high+1] = f[k - np.arange(i_low, i_high+1)]
        # solve LP
        res = linprog(c=-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
        if not res.success:
            break  # cannot solve
        b = res.x
        sum_b = b.sum()
        if sum_b <= 1e-12:
            break
        g = b / sum_b
        # mixing
        f_new = (1 - t) * f + t * g
        # normalize sum to 1 (should already hold)
        # evaluate
        score = _fast_evaluation(f_new)
        # maybe keep best
        if score < best_score:

        # optionally adapt t (maybe decreasing)
        f = f_new
        # maybe reduce t gradually e.g., t = max(0.1, t*0.99)

def proj_simplex(v):
    # ensure v >= 0 after projection (simplex)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w
```

import numpy as np, time, sys
from scipy.optimize import linprog

def _fast_eval(seq):
    n = len(seq)
    if n == 0: return np.inf
    m = 1 << ((2*n - 1).bit_length())
    fft = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fft*fft, m)[:2*n - 1]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01: return np.inf
    return float(2.0 * n * max_b / (sum_a*sum_a))

def _catalan_seq(n):
    seq = np.empty(n, dtype=float)
    a = 1.0
    seq[0] = a
    for i in range(1,n):
        a *= (2*i-1)/(2*i)
        seq[i] = a
    return seq

def _conv_matrix(f):
    n = len(f)
    # First column: f0,...,f_{n-1}, 0,...,0 (length 2n-1)
    c = np.concatenate([f, np.zeros(n-1, dtype=float)])
    # First row: f0, 0,...,0 (length n)
    r = np.concatenate([np.array([f[0]], dtype=float), np.zeros(n-1, dtype=float)])
    # Use scipy.linalg.toeplitz
    from scipy.linalg import toeplitz
    return toeplitz(c, r)

def _lp_step(f, M):
    n = len(f)
    A = _conv_matrix(f)
    b_ub = np.full(2*n-1, M, dtype=float)
    # bounds for each variable 0 <= x <= None
    bounds = [(0, None)] * n
    # Solve max sum(b) -> minimize -sum(b)
    c = -np.ones(n, dtype=float)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None
    return res.x

def _project_simplex(v):
    # Project onto simplex {x: x>=0, sum x=1}
    # Implementation from literature
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0  # safety margin

    # Choose length
    n = 300
    # initialize with Catalan shape
    seq = _catalan_seq(n).astype(float)
    seq /= seq.sum()
    best_seq = seq.copy()
    best_score = _fast_eval(seq)
    print(f"[init] n={n}, score={best_score:.6f}")

    # Phase 1: linear program driven refinement (first 50% of time)
    lp_deadline = start + 0.5 * (budget_s - 5.0)
    t_mix = 0.5
    it = 0
    while time.time() < lp_deadline:
        conv = np.convolve(seq, seq)
        M = conv.max()
        # Build convolution matrix A
        n = seq.size
        # Reuse function
        from scipy.linalg import toeplitz
        c = np.concatenate([seq, np.zeros(n-1, dtype=float)])
        r = np.concatenate([np.array([seq[0]], dtype=float), np.zeros(n-1, dtype=float)])
        A = toeplitz(c, r)
        b_ub = np.full(2*n-1, M, dtype=float)
        # Solve LP
        res = linprog(-np.ones(n), A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
        if not res.success:
            print("[LP] solve failed")
            break
        b_opt = res.x
        sum_b = b_opt.sum()
        if sum_b <= 1e-12:
            break
        g = b_opt / sum_b   # normalized to sum 1
        # evaluate candidates
        cand_seq = g
        score_g = _fast_eval(cand_seq)
        mixed = (1 - t_mix) * seq + t_mix * g
        score_mix = _fast_eval(mixed)
        # Choose best among current, g, mixed
        # Option 1: replace seq with whichever gives lowest score
        candidate = seq
        cand_score = best_score
        if score_g < cand_score:
            candidate = cand_seq
            cand_score = score_g
        if score_mix < cand_score:
            candidate = mixed
            cand_score = score_mix
        # Update seq
        seq = candidate
        # Keep best global
        if cand_score < best_score:
            best_score = cand_score
            best_seq = seq.copy()
            print(f"[LP] iter {it:4d}  score {best_score:.6f}")
        # Occasionally random jitter to help escape stagnation
        if np.random.rand() < 0.01:
            j = np.random.randint(n)
            seq[j] = max(0.0, seq[j] + np.random.randn() * 0.001)
            seq /= seq.sum()
        it += 1

    # Phase 2: exponentiated‑gradient fine tuning (remaining time)
    eg_deadline = deadline
    seq = best_seq.copy()   # start from best found
    n = seq.size
    best_score = _fast_eval(seq)
    best_seq = seq.copy()
    it = 0
    while time.time() < eg_deadline:
        # compute convolution and active set
        conv = np.convolve(seq, seq)
        M = conv.max()
        active = np.where(conv >= M - 1e-12)[0]
        # subgradient (factor 2)
        g = np.zeros_like(seq)
        # For each active k compute contribution
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            i_range = np.arange(lo, hi + 1)
            j_range = k - i_range
            g[i_range] += seq[j_range]
        # factor 2 for correct gradient
        g = 2 * g
        eta = 0.5 / np.sqrt(it + 1)
        seq = seq * np.exp(-eta * g)
        # projection onto simplex
        seq = _project_simplex(seq)
        # evaluate
        score = _fast_eval(seq)
        if score < best_score - 1e-12:
            best_score = score
            best_seq = seq.copy()
            if it % 200 == 0:
                print(f"[EG] iter {it:5d}  score {best_score:.6f}")
        # occasional jitter
        if np.random.rand() < 0.001:
            idx = np.random.randint(n)
            seq[idx] = max(0.0, seq[idx] + np.random.randn() * 1e-3)
            seq = _project_simplex(seq)
        it += 1

    final_score = _fast_eval(best_seq)
    print(f"[final] best score={final_score:.6f}")
    return best_seq.tolist()

"""
Hybrid linear‑program + exponentiated‑gradient optimisation.

The metric 2·n·max(a*a)/(∑a)² is scale‑invariant, therefore we keep
∑a = 1 throughout.  Starting from the Catalan (discrete arcsine)
distribution we perform two phases:

1.  LP‑phase (≈ 40 % of the time budget).  For the current vector f we
    maximise Σb subject to the linear constraints (f∗b)[k] ≤ M,
    where M = max(f∗f).  The optimiser pushes mass into indices where
    f is small, thus flattening the autocorrelation.  The normalised
    optimiser output g = b/Σb replaces f whenever it improves the score
    (or is mixed with f).

2.  EG‑phase (the remaining time).  Using the sub‑gradient of the
    max‑convolution we run exponentiated‑gradient descent on the simplex,
    with an explicit factor‑2 in the sub‑gradient and a simple simplex
    projection.  Small random jitter is added occasionally.

Both phases respect the 1000 s wall‑clock limit.  The best sequence found
is returned as a plain list of non‑negative floats.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.linalg import toeplitz

# -------------------------------------------------------------
# Fast evaluation – identical to the supplied evaluate_sequence
# but without the security checks.
# -------------------------------------------------------------
def _fast_eval(seq: np.ndarray) -> float:
    n = seq.size
    if n == 0:
        return np.inf
    # FFT size = next power of two ≥ 2·n‑1
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return np.inf
    # metric is scale‑invariant → we keep sum=1, so 2·n·max_b
    return float(2.0 * n * max_b / (sum_a * sum_a))


# -------------------------------------------------------------
# Catalan (arcsine) seed – the coefficients of (1‑z)^‑½.
# -------------------------------------------------------------
def _catalan_seq(length: int) -> np.ndarray:
    seq = np.empty(length, dtype=float)
    a = 1.0
    seq[0] = a
    for i in range(1, length):
        a *= (2 * i - 1) / (2.0 * i)
        seq[i] = a
    return seq


# -------------------------------------------------------------
# Build the (2·n‑1)×n convolution matrix A such that
#    (f ∗ b)[k] = (A @ b)[k] .
# -------------------------------------------------------------
def _conv_matrix(f: np.ndarray) -> np.ndarray:
    n = f.size
    # first column = f followed by n‑1 zeros, first row = f[0] then zeros
    c = np.concatenate([f, np.zeros(n - 1, dtype=float)])
    r = np.concatenate([np.array([f[0]], dtype=float), np.zeros(n - 1, dtype=float)])
    return toeplitz(c, r)


# -------------------------------------------------------------
# Projection of a vector onto the probability simplex {x ≥ 0, Σx = 1}.
# -------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    # Find the rho (last index where the condition holds)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# -------------------------------------------------------------
# Main entry point required by the evaluation harness.
# -------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights that (hopefully) yields a low
    evaluation score.  The routine works within ``budget_s`` seconds,
    performing an LP‑driven mass‑redistribution phase followed by an
    exponentiated‑gradient refinement on the simplex.
    """
    np.random.seed(seed)
    start = time.time()
    # keep a small safety margin
    deadline = start + budget_s - 5.0

    # ---------------------------------------------------------
    # Choose a modest length – the constant in the objective is
    # essentially independent of n, so n≈300 gives a good speed/quality
    # trade‑off.
    # ---------------------------------------------------------
    n = 300

    # Initial seed (Catalan) normalised to sum = 1
    seq = _catalan_seq(n).astype(float)
    seq /= seq.sum()
    best_seq = seq.copy()
    best_score = _fast_eval(seq)
    print(f"[init] n={n}, score={best_score:.6f}")

    # ---------------------------------------------------------
    # Phase 1 – linear‑program driven refinement.
    # ---------------------------------------------------------
    lp_deadline = start + 0.40 * (budget_s - 5.0)   # 40 % of total time
    mix_factor = 0.5                                 # mixing weight for (f,g)
    it_lp = 0
    while time.time() < lp_deadline:
        # Current max‑convolution value
        conv = np.convolve(seq, seq)
        M = conv.max()

        # Build convolution matrix A (shape (2·n‑1, n))
        A = _conv_matrix(seq)

        # Linear program:
        #   maximise Σ b_j    ⇔ minimise -Σ b_j
        #   s.t. A @ b ≤ M   (component‑wise)
        #        b ≥ 0
        ub = np.full(2 * n - 1, M, dtype=float)
        bounds = [(0.0, None)] * n
        res = linprog(-np.ones(n), A_ub=A, b_ub=ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if not res.success:
            # In the unlikely event that the LP fails, abort the LP phase.
            print("[LP] Solver failure – exiting LP phase")
            break

        b_opt = res.x
        sum_b = b_opt.sum()
        if sum_b <= 1e-12:          # numeric safety
            break

        # Normalise to obtain a feasible probability vector
        g = b_opt / sum_b

        # Evaluate three candidates: keep old seq, replace by g, or mix.
        score_g = _fast_eval(g)
        mixed = (1.0 - mix_factor) * seq + mix_factor * g
        score_mix = _fast_eval(mixed)

        # Pick the best of the three.
        candidate_seq = seq
        candidate_score = best_score
        if score_g < candidate_score:
            candidate_seq = g
            candidate_score = score_g
        if score_mix < candidate_score:
            candidate_seq = mixed
            candidate_score = score_mix

        seq = candidate_seq
        # Record global best
        if candidate_score < best_score - 1e-12:
            best_score = candidate_score
            best_seq = seq.copy()
            if it_lp % 10 == 0:
                print(f"[LP] iter {it_lp:4d}  score {best_score:.6f}")

        # Small random jitter occasionally to escape tiny plateaus
        if np.random.rand() < 0.01:
            j = np.random.randint(n)
            seq[j] = max(0.0, seq[j] + np.random.randn() * 1e-3)
            seq = _project_simplex(seq)

        it_lp += 1

    # ---------------------------------------------------------
    # Phase 2 – exponentiated‑gradient fine‑tuning.
    # ---------------------------------------------------------
    seq = best_seq.copy()   # start from the best LP result
    best_score = _fast_eval(seq)
    best_seq = seq.copy()
    it_eg = 0
    while time.time() < deadline:
        # Convolution and active (maximal) indices
        conv = np.convolve(seq, seq)
        M = conv.max()
        active = np.where(conv >= M - 1e-12)[0]

        # Sub‑gradient of the max‑convolution (factor 2 for correct derivative)
        g = np.zeros_like(seq)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            i_range = np.arange(lo, hi + 1)
            j_range = k - i_range
            g[i_range] += seq[j_range]
        g *= 2.0

        # Exponentiated‑gradient update
        eta = 0.5 / np.sqrt(it_eg + 1.0)
        seq = seq * np.exp(-eta * g)

        # Project back onto the simplex
        seq = _project_simplex(seq)

        # Evaluate
        score = _fast_eval(seq)
        if score < best_score - 1e-12:
            best_score = score
            best_seq = seq.copy()
            if it_eg % 200 == 0:
                print(f"[EG] iter {it_eg:5d}  score {best_score:.6f}")

        # Occasional jitter to avoid stagnation
        if np.random.rand() < 0.001:
            idx = np.random.randint(n)
            seq[idx] = max(0.0, seq[idx] + np.random.randn() * 1e-3)
            seq = _project_simplex(seq)

        it_eg += 1

    final_score = _fast_eval(best_seq)
    print(f"[final] best score={final_score:.6f}")
    return best_seq.tolist()
