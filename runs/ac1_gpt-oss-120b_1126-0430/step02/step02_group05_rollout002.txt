def solve_lp(f, M):
    n = len(f)
    # Build Toeplitz matrix.
    col = np.concatenate([f, np.zeros(n-1, dtype=f.dtype)])
    row = np.concatenate([np.array([f[0]], dtype=f.dtype), np.zeros(n-1, dtype=f.dtype)])
    A = scipy.linalg.toeplitz(col, row)  # shape (2*n-1, n)
    # Upper bounds: M per row.
    b = np.full(2*n-1, M, dtype=float)
    # objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    # bounds: (0, None)
    bounds = [(0, None)] * n
    # linprog
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True, 'time_limit': 60})
    if not res.success:
        print("LP failed:", res.message)
        return None
    return res.x

"""
Hybrid gradient-descent / linear-programming optimizer for the step‑function
score 2*n*max(a*a)/(sum a)^2.  Starts from the Catalan (arcsine) seed,
runs a fast mass‑transfer descent that reduces the maximal self‑convolution
using several peak indices at once, and periodically solves a linear program
to obtain an optimal direction (maximise sum(g) under the current peak bound).
The convex combination of the current sequence and the LP solution is
accepted only when it lowers the score.  The loop stops before the time
budget expires and returns the best sequence found.
"""

import time
import numpy as np
import scipy.optimize
import scipy.linalg

def _fast_evaluation(seq: np.ndarray) -> float:
    n = seq.size
    if n == 0:
        return float('inf')
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[:2 * n - 1]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))

def _catalan_sequence(length: int) -> np.ndarray:
    seq = np.empty(length, dtype=float)
    a = 1.0
    seq[0] = a
    for i in range(1, length):
        a *= (2 * i - 1) / (2.0 * i)
        seq[i] = a
    return seq

def _weights_for_index(seq: np.ndarray, k: int) -> np.ndarray:
    n = seq.size
    i_arr = np.arange(n)
    j_arr = k - i_arr
    mask = (j_arr >= 0) & (j_arr < n)
    w = np.zeros_like(seq)
    w[mask] = seq[j_arr[mask]]
    return w

def _aggregate_weights(seq: np.ndarray, max_conv: np.ndarray, tol: float = 1e-12, top_frac: float = 0.001) -> np.ndarray:
    # Identify indices where conv value is within top_frac fraction of the maximum.
    max_val = max_conv.max()
    # Choose a tolerance: either absolute fraction of max or using top fraction
    # Here we pick a fixed number of largest indices, e.g. at most 10.
    # Compute indices sorted descending
    # We want to include all k where conv >= max_val - tol*max_val maybe.
    # For robust selection, we take those within 0.1% of the maximum.
    threshold = max_val - max_val * top_frac
    top_idxs = np.flatnonzero(max_conv >= threshold)
    # If too many, keep only first e.g. 10
    if top_idxs.size > 10:
        # keep the ones with largest values (already sorted? not exactly)
        # We'll select first 10 arbitrarily
        top_idxs = top_idxs[:10]
    # Aggregate weight vector
    w_agg = np.zeros_like(seq)
    for k in top_idxs:
        w_agg += _weights_for_index(seq, k)
    return w_agg

def _lp_step(seq: np.ndarray, cur_max: float) -> np.ndarray | None:
    """
    Solve the LP:
        maximise   sum(g)
        subject to 0 <= g,
                    (seq * g)_k <= cur_max   for all k
    Return the solution g (numpy array) or None on failure.
    """
    n = seq.size
    # Build Toeplitz matrix for convolution constraints.
    # Using scipy.linalg.toeplitz, which produces a dense matrix.
    # For memory, use float32? Keep float64.
    col = np.concatenate([seq, np.zeros(n - 1, dtype=seq.dtype)])
    row = np.concatenate([np.array([seq[0]], dtype=seq.dtype), np.zeros(n - 1, dtype=seq.dtype)])
    A = scipy.linalg.toeplitz(col, row)  # shape (2n-1, n)
    # Upper bounds
    b = np.full(2 * n - 1, cur_max, dtype=float)
    # Objective: maximise sum(g)  <=> minimise -sum(g)
    c = -np.ones(n, dtype=float)
    # Bounds for variables
    bounds = [(0, None)] * n
    # Solve via HiGHS (default)
    try:
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    except Exception as e:
        print("LP exception:", e)
        return None
    if not res.success:
        print("LP failed:", res.message)
        return None
    g = res.x
    return g

def _try_update(seq: np.ndarray, new_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float, bool]:
    """Evaluate new_seq and return (seq, score, improved)."""
    new_score = _fast_evaluation(new_seq)
    if new_score < best_score - 1e-12:
        return new_seq, new_score, True
    else:
        return seq, best_score, False

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises the provided evaluation.
    Uses a hybrid gradient‑descent / LP scheme.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0  # leave a safety margin

    # Choose length; larger length gives more degrees of freedom but also
    # heavier LPs.  2000 is a good trade‑off for the time budget.
    n = 2000
    seq = _catalan_sequence(n)

    best_seq = seq.copy()
    best_score = _fast_evaluation(seq)
    print(f"Initial score: {best_score:.8f}")

    it = 0
    # parameters
    sink_percent = 5.0         # higher sink region than before
    rel_target_reduce = 0.02   # aim to cut peak by 2% per iteration
    min_lambda = 1e-6
    lp_interval = 30           # call LP every 30 iterations
    line_search_ts = [0.05, 0.1, 0.2, 0.3, 0.4]  # candidate mixing coefficients

    while time.time() < deadline:
        # compute convolution via FFT
        n = seq.size
        fft_len = 1 << ((2 * n - 1).bit_length())
        fa = np.fft.rfft(seq, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[:2 * n - 1]
        max_val = conv.max()

        # -- gradient‑like mass transfer step ---------------------------------
        # aggregate several top peaks to get a smoother weight vector
        w = _aggregate_weights(seq, conv, top_frac=0.001)

        # sink set based on low aggregated weight
        thresh = np.percentile(w, sink_percent)
        sink_mask = w <= thresh
        sink_idxs = np.flatnonzero(sink_mask)
        if sink_idxs.size == 0:
            sink_idxs = np.array([int(np.argmin(w))])
            sink_mask = np.zeros_like(w, dtype=bool)
            sink_mask[sink_idxs[0]] = True

        w_min = w[sink_idxs].min()
        excess = seq * (w - w_min)
        total_excess = excess.sum() - excess[sink_mask].sum()  # exclude sinks

        if total_excess > 0:
            target_reduction = rel_target_reduce * max_val
            lam = target_reduction / total_excess
            lam = max(min(lam, 1.0), min_lambda)
            # amount removed from each non‑sink entry proportionally to its contribution
            delta = lam * seq
            delta[sink_mask] = 0.0
            moved_mass = delta.sum()
            add_per_sink = moved_mass / sink_idxs.size
            new_seq = seq.copy()
            new_seq -= delta
            new_seq[sink_idxs] += add_per_sink
        else:
            # tiny random jitter
            jitter_i = np.random.randint(n)
            jitter = (np.random.rand() - 0.5) * 1e-4
            new_seq = seq.copy()
            new_seq[jitter_i] = max(0.0, new_seq[jitter_i] + jitter)

        # evaluate
        seq, best_score, improved = _try_update(seq, new_seq, best_score)
        if improved:
            best_seq = seq.copy()
            print(f"Iter {it:5d} (grad): improved score = {best_score:.8f}, max = {max_val:.8f}")

        # occasional LP step
        if it % lp_interval == 0 and it > 0:
            # solve LP using current max bound
            g = _lp_step(seq, max_val)
            if g is not None:
                sum_seq = seq.sum()
                sum_g = g.sum()
                if sum_g > 0:
                    # scale g to have the same total mass as seq
                    g_scaled = g * (sum_seq / sum_g)
                    # try several mixing ratios
                    best_local_seq = seq
                    best_local_score = best_score
                    for t in line_search_ts:
                        cand = (1 - t) * seq + t * g_scaled
                        cand_score = _fast_evaluation(cand)
                        if cand_score < best_local_score:
                            best_local_seq = cand
                            best_local_score = cand_score
                    if best_local_score < best_score - 1e-12:
                        seq = best_local_seq
                        best_seq = seq.copy()
                        best_score = best_local_score
                        print(f"Iter {it:5d} (LP): improved score = {best_score:.8f}")
        # occasional random shake‑up
        if np.random.rand() < 0.01:
            i, j = np.random.randint(n, size=2)
            if i != j and seq[i] > 0:
                eps = min(seq[i], np.random.rand() * seq[i] * 0.1)
                tmp = seq.copy()
                tmp[i] -= eps
                tmp[j] += eps
                tmp_score = _fast_evaluation(tmp)
                if tmp_score < best_score:
                    seq = tmp
                    best_seq = tmp.copy()
                    best_score = tmp_score
                    print(f"Iter {it:5d}: random shake improved score = {best_score:.8f}")

        it += 1

    print(f"Finished after {it} iterations, best score {best_score:.8f}")
    # Return list for evaluation harness
    return best_seq.tolist()

"""
Hybrid optimizer for the step‑function score.
Starts from the Catalan (arcsine) seed, performs a fast mass‑transfer
gradient‑like descent that reduces the maximal self‑convolution using
several peak indices at once, and periodically solves a linear program
(maximise ∑g subject to f*g ≤ current peak) to obtain a better descent
direction.  Convex combinations of the current sequence and the LP
solution are accepted only when they improve the score.  The loop runs
until the time budget is exhausted and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Try to import SciPy – needed for the LP step.  If unavailable we simply
# skip the LP part and fall back to the gradient descent only.
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
    import scipy.linalg as _la
    _SCIPY_AVAILABLE = True
except Exception:               # pragma: no cover
    _SCIPY_AVAILABLE = False

# ----------------------------------------------------------------------
# Fast evaluation – identical to evaluate_sequence but without the security
# checks.  Uses FFT based self‑convolution.
# ----------------------------------------------------------------------
def _fast_evaluation(seq: np.ndarray) -> float:
    n = seq.size
    if n == 0:
        return float("inf")
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))

# ----------------------------------------------------------------------
# Catalan (arcsine) coefficients: a_0 = 1,
# a_{i+1} = a_i * (2*i+1)/(2*i+2)
# ----------------------------------------------------------------------
def _catalan_sequence(length: int) -> np.ndarray:
    seq = np.empty(length, dtype=float)
    a = 1.0
    seq[0] = a
    for i in range(1, length):
        a *= (2 * i - 1) / (2.0 * i)          # a_i = a_{i-1} * (2i‑1)/(2i)
        seq[i] = a
    return seq

# ----------------------------------------------------------------------
# Helper: weight vector w_i = a_{k-i} for a given convolution index k.
# ----------------------------------------------------------------------
def _weights_for_index(seq: np.ndarray, k: int) -> np.ndarray:
    n = seq.size
    i_arr = np.arange(n)
    j_arr = k - i_arr
    mask = (j_arr >= 0) & (j_arr < n)
    w = np.zeros_like(seq)
    w[mask] = seq[j_arr[mask]]
    return w

# ----------------------------------------------------------------------
# Aggregate weights over the highest convolution entries.
# ----------------------------------------------------------------------
def _aggregate_weights(seq: np.ndarray,
                       conv: np.ndarray,
                       top_frac: float = 0.001) -> np.ndarray:
    """
    Return a weight vector that is the sum of w_i = a_{k-i}
    over a small set of convolution indices where conv is near its maximum.
    """
    max_val = conv.max()
    # keep indices whose value is within `top_frac` of the maximum
    thresh = max_val - max_val * top_frac
    cand = np.flatnonzero(conv >= thresh)

    # limit to at most ten indices for speed
    if cand.size > 10:
        cand = cand[:10]

    w_agg = np.zeros_like(seq)
    for k in cand:
        w_agg += _weights_for_index(seq, int(k))
    return w_agg

# ----------------------------------------------------------------------
# Linear‑programming step: maximise Σ g_j  s.t. 0 ≤ g,  f * g ≤ cur_max.
# Returns the optimal g (numpy array) or None on failure.
# ----------------------------------------------------------------------
def _lp_step(seq: np.ndarray, cur_max: float):
    if not _SCIPY_AVAILABLE:                     # pragma: no cover
        return None
    n = seq.size

    # Toeplitz matrix of the convolution operator.
    # Row k contains (seq)_{k‑j} when 0 ≤ k‑j < n, otherwise 0.
    col = np.concatenate([seq, np.zeros(n - 1, dtype=seq.dtype)])
    row = np.concatenate([np.array([seq[0]], dtype=seq.dtype),
                          np.zeros(n - 1, dtype=seq.dtype)])
    A = _la.toeplitz(col, row)               # shape (2n‑1, n)

    b = np.full(2 * n - 1, cur_max, dtype=float)
    c = -np.ones(n, dtype=float)             # minimise –Σ g_j
    bounds = [(0, None)] * n

    try:
        res = _opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                           method='highs',
                           options={'presolve': True})
    except Exception:                        # pragma: no cover
        return None
    if not res.success:
        return None
    g = res.x
    # Numerical noise may introduce tiny negatives – clip them.
    g = np.clip(g, 0.0, None)
    return g

# ----------------------------------------------------------------------
# Helper: try to replace the current sequence with a candidate.
# Returns (new_seq, new_score, improved_flag).
# ----------------------------------------------------------------------
def _try_update(seq: np.ndarray,
                cand: np.ndarray,
                best_score: float) -> tuple[np.ndarray, float, bool]:
    cand_score = _fast_evaluation(cand)
    if cand_score < best_score - 1e-12:
        return cand, cand_score, True
    return seq, best_score, False

# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that lowers the evaluation
    score.  A hybrid gradient/Lp algorithm is employed under the given
    time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    # Length of the sequence – a compromise between quality and speed.
    n = 2000
    seq = _catalan_sequence(n)

    best_seq = seq.copy()
    best_score = _fast_evaluation(seq)
    print(f"Initial score: {best_score:.8f}")

    it = 0
    # Parameters for the gradient‑like step.
    sink_percent = 5.0           # percentage of positions used as sinks
    rel_target_reduce = 0.02     # aim to cut the current peak by 2 % each step
    min_lambda = 1e-6
    lp_interval = 30             # solve an LP every <lp_interval> iterations
    line_search_ts = [0.05, 0.1, 0.2, 0.3, 0.4]  # mixing ratios with the LP solution

    while time.time() < deadline:
        # --------------------------------------------------------------
        # Current self‑convolution and its maximum.
        # --------------------------------------------------------------
        n = seq.size
        fft_len = 1 << ((2 * n - 1).bit_length())
        fa = np.fft.rfft(seq, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_val = conv.max()

        # --------------------------------------------------------------
        # Gradient‑like mass‑transfer step (uses several top peaks).
        # --------------------------------------------------------------
        w = _aggregate_weights(seq, conv, top_frac=0.001)

        # Identify a sink set: indices whose aggregated weight is small.
        thresh = np.percentile(w, sink_percent)
        sink_mask = w <= thresh
        sink_idxs = np.flatnonzero(sink_mask)
        if sink_idxs.size == 0:
            sink_idxs = np.array([int(np.argmin(w))])
            sink_mask = np.zeros_like(w, dtype=bool)
            sink_mask[sink_idxs[0]] = True

        w_min = w[sink_idxs].min()
        diff = w - w_min                     # contribution of each position to the peak
        # Exclude sink positions from the excess computation.
        excess = seq * diff
        total_excess = excess.sum() - excess[sink_mask].sum()

        if total_excess > 0:
            # Desired absolute reduction of the current peak.
            target_reduction = rel_target_reduce * max_val
            # Simple uniform removal (fraction λ of every non‑sink entry).
            lam = target_reduction / total_excess
            lam = max(min(lam, 1.0), min_lambda)

            delta = lam * seq
            delta[sink_mask] = 0.0
            moved = delta.sum()
            add_per_sink = moved / sink_idxs.size

            new_seq = seq.copy()
            new_seq -= delta
            new_seq[sink_idxs] += add_per_sink
        else:
            # No useful excess – apply a tiny random jitter.
            jitter_i = np.random.randint(n)
            jitter = (np.random.rand() - 0.5) * 1e-4
            new_seq = seq.copy()
            new_seq[jitter_i] = max(0.0, new_seq[jitter_i] + jitter)

        # Accept the gradient step if it improves the score.
        seq, best_score, improved = _try_update(seq, new_seq, best_score)
        if improved:
            best_seq = seq.copy()
            print(f"Iter {it:5d} (grad): improved score = {best_score:.8f},  max = {max_val:.8f}")

        # --------------------------------------------------------------
        # Linear‑programming step – performed periodically.
        # --------------------------------------------------------------
        if it % lp_interval == 0 and it > 0 and _SCIPY_AVAILABLE:
            g = _lp_step(seq, max_val)
            if g is not None and g.sum() > 0:
                sum_seq = seq.sum()
                sum_g = g.sum()
                # Scale g to have the same total mass as the current sequence.
                g_scaled = g * (sum_seq / sum_g)
                g_scaled = np.clip(g_scaled, 0.0, None)

                # Try a few mixing coefficients and keep the best.
                best_local_seq = seq
                best_local_score = best_score
                for t in line_search_ts:
                    cand = (1 - t) * seq + t * g_scaled
                    cand_score = _fast_evaluation(cand)
                    if cand_score < best_local_score:
                        best_local_seq = cand
                        best_local_score = cand_score
                if best_local_score < best_score - 1e-12:
                    seq = best_local_seq
                    best_seq = seq.copy()
                    best_score = best_local_score
                    print(f"Iter {it:5d} (LP): improved score = {best_score:.8f}")

        # --------------------------------------------------------------
        # Small random shake‑up to escape possible plateaux.
        # --------------------------------------------------------------
        if np.random.rand() < 0.01:
            i, j = np.random.randint(n, size=2)
            if i != j and seq[i] > 0:
                eps = min(seq[i], np.random.rand() * seq[i] * 0.1)
                tmp = seq.copy()
                tmp[i] -= eps
                tmp[j] += eps
                tmp_score = _fast_evaluation(tmp)
                if tmp_score < best_score:
                    seq = tmp
                    best_seq = tmp.copy()
                    best_score = tmp_score
                    print(f"Iter {it:5d}: random shake improved score = {best_score:.8f}")

        it += 1

    print(f"Finished after {it} iterations, best score {best_score:.8f}")
    # Return a plain Python list – the evaluation harness expects this.
    return best_seq.tolist()
