    # arr is numpy array normalized to sum=1.
    # Compute objective quickly.
    total = arr.sum()

    conv = np.convolve(arr, arr)
    max_val = conv.max()
    n = arr.shape[0]

"""
Propose a low‑value step‑function for the given objective.

The algorithm consists of three stages:

1. **Broad parametric sampling** – generate many candidates from
   power‑law, exponential and two‑component mixtures, keep the best.

2. **Linear‑programming guided refinement** – for the current sequence
   `f` compute `M = max(conv(f,f))`.  Solve the LP

        maximise    Σ b_j
        subject to  (f * b)_k ≤ M    for k = 0 … 2n‑2
                    b_j ≥ 0

   (the convolution matrix is built as a sparse Toeplitz matrix).
   The LP returns a direction `b`; after normalising `b` to a probability
   vector `g` we mix `f ← (1‑t) f + t g` with a few trial step sizes `t`.
   Any improvement is kept and the process repeats a limited number of
   times.

3. **Local hill‑climbing** – a cheap gradient‑like move that shifts a
   small amount of mass from the index contributing most to the current
   maximal convolution value to the index contributing least.
   The move size shrinks slowly; the routine runs until the time budget
   is exhausted.

All numbers are forced to be non‑negative and the sequence is always
normalised to sum = 1, which makes the objective scale‑invariant.
The function returns the best sequence found before the time budget ends.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import diags

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _project_to_simplex(v):
    """Project a vector onto the probability simplex {x ≥ 0, Σ x = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


def _fast_score(arr):
    """
    Cheap evaluation of 2 * n * max(conv(a, a)) / (sum(a)**2)
    for a non‑negative ndarray `arr` (not normalised).
    """
    total = arr.sum()
    if total <= 0:
        return float('inf')
    conv = np.convolve(arr, arr)
    return 2.0 * arr.shape[0] * conv.max() / (total * total)


def _norm_and_score(raw):
    """Normalise a raw vector (non‑negative) and obtain the official score."""
    total = raw.sum()
    if total <= 0:
        return float('inf'), None
    norm = raw / total
    # official (slow) evaluator – required for final comparison
    score = evaluate_sequence(norm.tolist())
    return score, norm


def _build_convolution_matrix(f):
    """
    Build the (2n‑1) × n sparse matrix `A` that satisfies
        (A @ b)[k] = (f * b)[k]   for all k,
    i.e. each column `j` contains the vector `f` placed on the diagonal
    offset `j`.  The matrix is CSR‑formatted and suitable for `linprog`.
    """
    n = len(f)
    # one constant diagonal per offset = 0 … n‑1
    data = [np.full(n, val, dtype=np.float64) for val in f]
    offsets = np.arange(n, dtype=int)
    A = diags(data, offsets, shape=(2 * n - 1, n), format='csr')
    return A

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed=123, budget_s=950, **kwargs):
    """
    Generate a step‑function (list of heights) that minimises the
    supplied ``evaluate_sequence`` value.

    Parameters
    ----------
    seed : int, optional
        Random seed.
    budget_s : float, optional
        Time budget in seconds (default 950 s, leaving a margin for the
        harness).  Any excess time is ignored.
    **kwargs
        Ignored – kept for compatibility.

    Returns
    -------
    list[float]
        The best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)   # provided by the harness
    except Exception:
        n = 2244                     # fallback length

    # ------------------------------------------------------------------
    # Baseline – use the supplied construction (if any) and normalise it
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=float)
        if base.sum() <= 0:
            raise ValueError
    except Exception:
        base = np.full(n, 1.0 / n, dtype=float)

    base = base / base.sum()                 # sum = 1
    best_seq = base.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1. Broad sampling of parametric families
    # ------------------------------------------------------------------
    idx = np.arange(n, dtype=float)

            if new_score < cur_score - 1e-12:

                cur_score = new_score
                if new_score < best_score - 1e-12:
                    best_score = new_score
                    best_seq = new.copy()

    # ------------------------------------------------------------------
    # 3. Light hill‑climbing refinement (fast cheap score)

    cur = best_seq.copy()
    cur_score = best_score
    cur_conv = np.convolve(cur, cur)
    max_hc_iters = 80000
    it = 0

        it += 1
        # index of the current maximal convolution entry
