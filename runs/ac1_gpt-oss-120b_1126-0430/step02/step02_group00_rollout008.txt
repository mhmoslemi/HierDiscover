```
def _pocs_iter(a, L):
    # compute FFT
    A = np.fft.rfft(np.concatenate([a, np.zeros(L - len(a))]))
    # magnitude constant target
    const = np.sqrt(np.sum(a * a))
    # new spectrum
    A_new = const * np.exp(1j * np.angle(A))
    # inverse transform
    a_full = np.fft.irfft(A_new, n=L)
    a_new = a_full[:a.shape[0]]
    # enforce nonnegativity and simplex
    a_new = np.maximum(a_new, 0.0)
    a_new = _project_to_simplex(a_new)
    return a_new
```

"""...
"""
import time
import math
import numpy as np

# Helper: projection onto the probability simplex
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >= 0, sum(x)=1}."""
    n = v.shape[0]
    # sort v decreasingly
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        # uniform distribution if all non‑positive
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Full linear convolution via FFT
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length L (L >= 2·len(a)-1)."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    # keep only the linear part
    return conv_full[: 2 * a.size - 1]

# Objective for a simplex‑normalised vector (sum = 1)
def _objective(a: np.ndarray, n: int, L: int) -> float:
    """Exact evaluation 2·n·max(conv(a,a)) with unit‑sum normalisation."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()

# Sub‑gradient of the maximal convolution entry
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. a (used for a single peak)."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)  # indices where the peak occurs
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    g /= max(1, len(ks))
    return g

# Gradient of the soft‑max (log‑sum‑exp) surrogate
def _smooth_gradient(a: np.ndarray, conv: np.ndarray, alpha: float, L: int) -> np.ndarray:
    """
    Gradient of f_α(a) = (2·n/α)·log∑_k e^{α·conv_k}
    which reduces to 2·corr(w,a) where w_k ∝ e^{α·conv_k}.
    """
    # soft‑max weights
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)  # stabilised
    w = exp_scaled / exp_scaled.sum()

    # correlation w ⋆ a  (central n entries)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n = a.size
    grad = 2.0 * corr_rev[n - 1 : n - 1 + n]
    return grad

# One iteration of the Gerchberg–Saxton / POCS scheme
def _pocs_step(a: np.ndarray, L: int) -> np.ndarray:
    """
    Perform a single alternating‑projection step that enforces a flat
    magnitude spectrum (|FFT| = const).  The result is projected back
    onto the probability simplex.
    """
    n = a.size
    # zero‑pad to length L
    a_full = np.concatenate([a, np.zeros(L - n, dtype=a.dtype)])
    A = np.fft.rfft(a_full)
    # target magnitude: sqrt(∑ a_i²) (Parseval)
    const = np.sqrt(np.dot(a, a))
    A_new = const * np.exp(1j * np.angle(A))
    a_full_new = np.fft.irfft(A_new, n=L)
    a_new = a_full_new[:n]
    # enforce non‑negativity and simplex
    a_new = np.maximum(a_new, 0.0)
    a_new = _project_to_simplex(a_new)
    return a_new

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (probability distribution)
    that minimises   O = 2·n·max(conv(a,a))   with ∑a_i = 1.

    The producer combines three complementary heuristics:
      1. Alternating‑projection (Gerchberg–Saxton) to flatten the
         Fourier magnitude, which tends to flatten the autocorrelation.
      2. Smooth log‑sum‑exp gradient descent (soft‑max surrogate) for
         fine‑grained reduction of the maximal convolution entry.
      3. Stochastic mass‑transfer hill‑climbing with a tiny Metropolis
         acceptance probability and occasional sub‑gradient pushes.

    The routine is fully time‑bounded – it stops slightly before the
    supplied budget to guarantee a return.  The best sequence seen is
    returned as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # -----------------------------------------------------------------
    # Choose problem size
    if 'height_sequence_1' in globals():
        base = np.array(height_sequence_1, dtype=float)
        n = base.size
        a0 = base / base.sum()
        print(f"[init] using existing sequence, n={n}")
    else:
        n = 4000

"""
Propose a non‑negative step‑function sequence that minimizes

    O(a) = 2·n·max(conv(a,a))   with   sum(a) = 1.

The search combines three complementary heuristics:

1. **Alternating projection (Gerchberg–Saxton / POCS)**
   – Enforce a flat magnitude spectrum (|FFT| = const) which tends to
     flatten the self‑convolution.
   – Project the result back onto the probability simplex.
   – Several random restarts explore different attractors.

2. **Smooth log‑sum‑exp gradient descent**
   – Replace the hard max by a soft‑max (log‑sum‑exp) surrogate.
   – Compute its gradient using FFT‑based correlation.
   – Adaptive step size with back‑tracking; the smoothing parameter α
     is annealed from small to large.

3. **Stochastic hill‑climbing + occasional sub‑gradient pushes**
   – Random “mass‑transfer’’ moves (move a fraction of probability mass
     from one index to another) with Metropolis acceptance.
   – Every few iterations a sub‑gradient step targeting the current
     maximal convolution entry is applied.
   – Temperature is slowly cooled.

The routine respects a 1000 s time budget, prints progress, and finally
returns the best sequence (as a plain Python list).  It works with any
starting sequence stored in the global variable `height_sequence_1`.
"""

import time
import math
import numpy as np

# -----------------------------------------------------------------
# Helper: Euclidean projection onto the simplex {x >= 0, sum(x)=1}
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:               # all non‑positive ⇒ uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# -----------------------------------------------------------------
# Linear self‑convolution using FFT (full linear result)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return a*a (linear convolution) using an FFT of length L (L≥2·len(a)‑1)."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# -----------------------------------------------------------------
# Exact objective for a simplex‑normalised vector
def _objective(a: np.ndarray, n: int, L: int) -> float:
    """2·n·max(conv(a,a))  (a is assumed to sum to 1)."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# -----------------------------------------------------------------
# Sub‑gradient of the maximal convolution entry (used in local refinement)
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. a."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    g /= max(1, len(ks))
    return g


# -----------------------------------------------------------------
# Gradient of the soft‑max (log‑sum‑exp) surrogate of max(conv)
def _smooth_gradient(a: np.ndarray, conv: np.ndarray, alpha: float, L: int) -> np.ndarray:
    """
    Compute ∇_a ( (2·n/α)·log∑_k e^{α·conv_k} ).
    It reduces to 2·corr(w, a) with w_k ∝ e^{α·conv_k}.
    """
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)           # stabilise
    w = exp_scaled / exp_scaled.sum()

    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n = a.size
    grad = 2.0 * corr_rev[n - 1 : n - 1 + n]            # central n entries
    return grad


# -----------------------------------------------------------------
# One iteration of the Gerchberg–Saxton / POCS scheme

    """
    Enforce a flat magnitude spectrum (|FFT| = const) and project back onto
    the probability simplex.
    """
    n = a.size
    a_full = np.concatenate([a, np.zeros(L - n, dtype=a.dtype)])
    A = np.fft.rfft(a_full)
    const = np.sqrt(np.dot(a, a))                     # Parseval: sqrt(∑a_i²)
    A_new = const * np.exp(1j * np.angle(A))
    a_full_new = np.fft.irfft(A_new, n=L)
    a_new = a_full_new[:n]
    a_new = np.maximum(a_new, 0.0)
    a_new = _project_to_simplex(a_new)
    return a_new
