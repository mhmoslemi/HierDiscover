"""
Propose a step‑function (list of non‑negative heights) that minimises
``evaluate_sequence``.
The code builds a Catalan‑like seed (a_i ∝ C(2i,i)/4^i) which is known to be
asymptotically optimal (objective → π/2).  It then refines the seed with an
Adam‑style projected gradient descent that works on the *exact* max‑convolution
value (sub‑gradient of the ℓ∞‑norm).  The optimiser runs until the 1000 s
budget expires (or the improvement stalls) and returns the best sequence
found.

The algorithm is fully deterministic apart from a few tiny random perturbations
used when the search gets stuck.  All heavy work (convolutions) is performed
with an FFT‑based O(n log n) routine, allowing many thousands of iterations
inside the time limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def catalan_seed(n: int) -> np.ndarray:
    """
    Catalan‑like seed:
        a_i ∝ C(2i,i) / 4**i   (≈ 1/√(π i) )
    Normalised to sum 1.
    """
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    # recurrence  C(2i, i) / 4**i   =>   a_{i+1}=a_i * (2*i+1)/(2*(i+1))
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * (i - 1) + 1.0) / (2.0 * i)
    a /= a.sum()
    return a


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x ≥ 0, sum(x)=1}
    Implementation of the O(n) algorithm of Duchi et al. (2008).
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u > cssv / np.arange(1, n + 1))[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution using a real FFT (rfft/irfft).  Length of the result is
    len(x)+len(y)-1.
    """
    n = x.size + y.size - 1
    size = 1 << (n - 1).bit_length()            # next power of two
    X = np.fft.rfft(x, size)
    Y = np.fft.rfft(y, size)
    Z = X * Y
    conv = np.fft.irfft(Z, size)[:n]
    return conv.real


def eval_score(a: np.ndarray) -> float:
    """
    Exact value of ``evaluate_sequence`` for a non‑negative simplex vector.
    (The external harness will call ``evaluate_sequence``; we use this fast
    exact version for internal comparisons.)
    """
    n = a.size
    b = convolve_fft(a, a)
    return float(2.0 * n * b.max() / (a.sum() ** 2))


# ----------------------------------------------------------------------
# Optimiser (Adam on the ℓ∞‑norm of the auto‑convolution)
# ----------------------------------------------------------------------
def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of   max_k (a * a)_k   with respect to a.
    We take the average over all indices where the maximum is attained.
    """
    n = a.size
    b = convolve_fft(a, a)                 # length 2n‑1
    max_val = b.max()
    # indicator of the active set (within machine epsilon)
    mask = (b == max_val).astype(float)
    mask_cnt = mask.sum()
    if mask_cnt == 0:
        mask = np.ones_like(b) / b.size
    else:
        mask /= mask_cnt                     # average over the active set

    # gradient g_i = 2 * Σ_k mask_k * a_{k‑i}
    # This is a cross‑correlation which we compute via convolution
    mask_rev = mask[::-1]
    full = convolve_fft(a, mask_rev)       # length 3n‑2
    # pick the slice that corresponds to indices i (see analysis in docstring)
    idx = (b.size - 1) - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a sequence of non‑negative heights that minimises
    ``evaluate_sequence``.
    Returns the best list found before the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # a small safety margin

    # ------------------------------------------------------------------
    # problem size ------------------------------------------------------
    # A larger n reduces the O(1/n) discretisation error, but also makes each
    # iteration more expensive.  n≈12000 is a good compromise.
    n = 12000

    # ------------------------------------------------------------------
    # initialise with the Catalan‑like seed (known to be essentially optimal)
    a = catalan_seed(n)
    best_a = a.copy()
    best_score = eval_score(a)
    cur_score = best_score
    print(f"[init] n={n}  score={cur_score:.9f}")

    # ------------------------------------------------------------------
    # Adam hyper‑parameters (projected onto the simplex each step)
    lr = 0.08                     # base learning rate
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # bookkeeping -------------------------------------------------------
    no_improve = 0
    start = time.time()

    while time.time() < deadline:
        # ------- sub‑gradient of the ℓ∞‑norm --------------------------------
        grad = _subgradient_maxconv(a)

        # ------- Adam update (un‑constrained) -----------------------------
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # ------- projected Adam step ---------------------------------------
        a_candidate = proj_simplex(a - step)

        cand_score = eval_score(a_candidate)

        # ------- simple acceptance rule (monotone improvement) -------------
        if cand_score < cur_score:
            a = a_candidate
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{t:07d}]  new best {best_score:.9f}")
                no_improve = 0
            else:
                no_improve += 1
        else:
            # no improvement – shrink learning‑rate a little
            lr *= 0.85
            no_improve += 1

        # ------- occasional random restart if we are stuck -----------------
        if no_improve > 4000:
            # perturb the current best point
            pert = 0.02 * np.random.randn(n)
            a = best_a + pert
            a = np.maximum(a, 0.0)
            a = proj_simplex(a)
            cur_score = eval_score(a)
            lr = 0.08
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0
            print(f"[restart]  score={cur_score:.9f}")

        # ------------------------------------------------------------------
        # occasional learning‑rate decay to keep steps reasonable
        if t % 5000 == 0:
            lr = max(lr * 0.9, 1e-5)

    print(f"[done] best score found = {best_score:.9f}")
    # Return a plain Python list as required by the harness
    return best_a.tolist()
