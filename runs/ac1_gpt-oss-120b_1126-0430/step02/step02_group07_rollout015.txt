def _softmax(x, epsilon):

def _gradient_descent(a, time_budget, start_time):
    n = len(a)
    best_vec = a.copy()
    best_score = _compute_score(best_vec)
    it = 0
    max_iters = 200000
    # schedule params
    while time.time() - start_time < time_budget and it < max_iters:
        it += 1
        # compute conv
        conv = np.convolve(a, a)
        max_c = conv.max()
        # compute softmax weights
        eps = max_c * 0.05  # might adjust
        if eps < 1e-12:
            eps = 1e-12
        # softmax stable
        x = conv - conv.max()
        w = np.exp(x / eps)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / len(w))
        else:
            w /= w_sum
        # compute gradient
        rev_a = a[::-1]
        grad_full = np.convolve(rev_a, w, mode='full')
        start = n - 1
        grad = 2.0 * grad_full[start:start + n]
        # step size schedule
        eta = 0.5 / np.sqrt(it)   # start 0.5, shrink
        # propose
        a_new = a - eta * grad
        # projection to simplex
        a_new = _project_to_simplex(a_new)
        # cheap score
        score_new = _compute_score(a_new)
        # acceptance: if better or with small prob
        if score_new < best_score:
            best_score = score_new
            best_vec = a_new.copy()
        if score_new < _compute_score(a):  # compare with current a
            a = a_new
        else:
            # occasional acceptance
            if np.random.rand() < 0.001:
                a = a_new
        # periodic evaluate with official function to keep accurate best_score
        if it % 200 == 0:
            real_score = evaluate_sequence(a.tolist())
            if real_score < best_score:
                best_score = real_score
                best_vec = a.copy()
                print(f"[grad] iter={it} official best {real_score:.6f}")
    return best_vec, best_score

def _hill_climb(a, time_budget, start_time):
    n = len(a)
    cur = a.copy()
    cur_score = _compute_score(cur)
    cur_conv = np.convolve(cur, cur)
    max_iters = 300000
    it = 0
    while time.time() - start_time < time_budget and it < max_iters:
        it += 1
        max_idx = int(np.argmax(cur_conv))
        i_arr = np.arange(n)
        rev = max_idx - i_arr
        mask = (rev >= 0) & (rev < n)
        contrib = np.zeros_like(cur)
        contrib[mask] = cur[i_arr[mask]] * cur[rev[mask]]
        src = int(np.argmax(contrib))
        tgt = int(np.argmin(contrib))
        if src == tgt or cur[src] <= 0:
            src, tgt = np.random.choice(n, size=2, replace=False)
        # step fraction: start high then shrink
        frac = 0.3 * max(0.0, 1.0 - it / max_iters) + 0.01
        step = cur[src] * np.random.random() * frac
        if step <= 0:
            continue
        new = cur.copy()
        new[src] -= step
        new[tgt] += step
        new = _project_to_simplex(new)
        new_score = _compute_score(new)
        if new_score < cur_score - 1e-12:
            cur = new
            cur_conv = np.convolve(cur, cur)
            cur_score = new_score
        # periodic official evaluation
        if it % 5000 == 0:
            real_score = evaluate_sequence(cur.tolist())
            if real_score < cur_score - 1e-12:
                cur_score = real_score
                print(f"[hill] iter={it} improved official score {real_score:.6f}")
    return cur, cur_score

def _softmax_weights(conv, eps):
    max_val = np.max(conv)
    exp_terms = np.exp((conv - max_val) / eps)
    sum_exp = np.sum(exp_terms)
    if sum_exp == 0:
        return np.full_like(conv, 1/conv.size)
    return exp_terms / sum_exp

"""
Propose a low‑value step‑function for the objective
    2 * n * max(conv(a,a)) / (sum(a)**2)

The search consists of three stages:

1.  Broad random sampling from several parametric families (power‑law,
    exponential, mixtures, Dirichlet).  Every sample is normalised to
    sum = 1 and evaluated with the official ``evaluate_sequence``.
    The best sample becomes the starting point.

2.  Gradient‑descent on a smooth approximation of the max‑convolution.
    For the current vector `a` we build a soft‑max weight vector `w`
    over the full convolution `c = a*a`.  The gradient of the smoothed
    objective is
        ∇ = 2 * (rev(a) * w)    (full convolution, slice of length n)
    A diminishing step size is used and after every step the vector is
    projected onto the probability simplex.

3.  Targeted hill‑climbing refinement – repeatedly move a small amount
    of mass from the index that contributes most to the current maximal
    convolution value to the index that contributes the least.  The
    step‑size shrinks linearly with the number of iterations.

All stages respect the time budget (default 950 s).  The best sequence
found is returned as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _project_to_simplex(v):
    """
    Project a vector onto the probability simplex {x ≥ 0, sum(x) = 1}.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


def _compute_score(arr):
    """
    Cheap evaluation of the objective for a non‑negative array whose sum is 1.
    Returns ``float('inf')`` for an empty or non‑positive sum.
    """
    total = arr.sum()
    if total <= 0:
        return float('inf')
    conv = np.convolve(arr, arr)
    max_val = conv.max()
    n = arr.shape[0]
    # because total == 1, the formula simplifies to 2*n*max_val
    return 2.0 * n * max_val


def _norm_and_score(raw):
    """
    Normalise a raw (non‑negative) vector to sum = 1, evaluate it with the
    official ``evaluate_sequence`` and return the pair (score, normalised).
    """
    total = raw.sum()
    if total <= 0:
        return float('inf'), None
    norm = raw / total
    score = evaluate_sequence(norm.tolist())
    return score, norm


def _softmax_weights(conv, eps):
    """
    Stable soft‑max over the convolution values.
    ``eps`` must be > 0; a larger ``eps`` yields a smoother distribution.
    """
    max_val = np.max(conv)
    # shift for numerical stability
    shifted = (conv - max_val) / eps
    exp_vals = np.exp(shifted)
    s = exp_vals.sum()
    if s == 0:
        return np.full_like(conv, 1.0 / conv.size)
    return exp_vals / s


def _gradient(a, w):
    """
    Gradient of the smoothed max‑convolution.
    a : (n,) probability vector
    w : (2n‑1,) weight vector (soft‑max over the convolution)
    Returns a (n,) gradient.
    """
    rev_a = a[::-1]
    full = np.convolve(rev_a, w, mode='full')
    n = a.shape[0]
    start = n - 1
    # slice of length n
    grad = 2.0 * full[start:start + n]
    return grad


def _gradient_descent(start_vec, time_budget, start_time):
    """
    Smooth‑max gradient descent on the probability simplex.
    Returns the best vector found and its official score.
    """
    n = start_vec.shape[0]
    a = start_vec.copy()
    best_vec = a.copy()
    best_score = _compute_score(best_vec)   # cheap score (sum==1)
    it = 0
    max_iters = 200_000

    while time.time() - start_time < time_budget and it < max_iters:
        it += 1

        # full convolution and its maximum
        conv = np.convolve(a, a)
        max_c = conv.max()

        # soft‑max smoothing parameter – linearly decreasing
        eps = max_c * max(0.02, 0.22 * (1.0 - it / max_iters))

        w = _softmax_weights(conv, eps)

        # gradient step
        grad = _gradient(a, w)

        # step size schedule (large at start, shrink ~1/√it)
        eta = 0.8 / np.sqrt(it)

        a_new = a - eta * grad
        a_new = _project_to_simplex(a_new)

        # cheap score (sum == 1)
        new_score = _compute_score(a_new)

        # accept if improvement, occasionally accept worse to escape plateaus
        if new_score < best_score:
            best_score = new_score
            best_vec = a_new.copy()
        if new_score < _compute_score(a) - 1e-12 or np.random.rand() < 0.001:
            a = a_new

        # occasional official evaluation for reliability
        if it % 250 == 0:
            real_score = evaluate_sequence(a.tolist())
            if real_score < best_score:
                best_score = real_score
                best_vec = a.copy()
                print(f"[grad] iter={it} official best={real_score:.6f}")

    return best_vec, best_score


def _hill_climb(start_vec, time_budget, start_time):
    """
    Fine‑grained hill climbing using the contribution‑based source/target
    move described in the original baseline.  The step‑size shrinks linearly.
    """
    n = start_vec.shape[0]
    cur = start_vec.copy()
    cur_score = _compute_score(cur)
    cur_conv = np.convolve(cur, cur)
    it = 0
    max_iters = 300_000
    while time.time() - start_time < time_budget and it < max_iters:
        it += 1
        # index of the current maximal convolution value
        max_idx = int(np.argmax(cur_conv))

        # contribution of each index to that peak
        i_arr = np.arange(n)
        rev = max_idx - i_arr
        mask = (rev >= 0) & (rev < n)
        contrib = np.zeros_like(cur)
        contrib[mask] = cur[i_arr[mask]] * cur[rev[mask]]

        src = int(np.argmax(contrib))
        tgt = int(np.argmin(contrib))
        if src == tgt or cur[src] <= 0:
            src, tgt = np.random.choice(n, size=2, replace=False)

        # step size – start large, shrink to ~0.01
        step_frac = 0.20 * max(0.0, 1.0 - it / max_iters) + 0.01
        step = cur[src] * np.random.rand() * step_frac
        if step <= 0:
            continue

        new = cur.copy()
        new[src] -= step
        new[tgt] += step
        new = _project_to_simplex(new)

        new_score = _compute_score(new)
        if new_score < cur_score - 1e-12:
            cur = new
            cur_conv = np.convolve(cur, cur)
            cur_score = new_score

        # periodic official check
        if it % 5000 == 0:
            real_score = evaluate_sequence(cur.tolist())
            if real_score < cur_score - 1e-12:
                cur_score = real_score
                print(f"[hill] iter={it} official best={real_score:.6f}")

    return cur, cur_score

def propose_candidate(seed=123, budget_s=950, **kwargs):
    """
    Search for a step‑function (list of heights) that minimises the provided
    ``evaluate_sequence`` value within the allotted time budget.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : float, optional
        Time budget in seconds (default 950 s, leaving a small safety margin).
    **kwargs :
        Ignored – kept for compatibility with the evaluation harness.

    Returns
    -------
    list[float]
        The best sequence discovered.
    """
    np.random.seed(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # Determine problem size (the harness provides ``height_sequence_1``)
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)   # provided by the outer harness
    except Exception:
        n = 2244                       # fallback length from the description

    idx = np.arange(n, dtype=float)

    # ------------------------------------------------------------------
    # Initialise with the supplied baseline (if any)
    # ------------------------------------------------------------------
    try:
        init_raw = np.array(height_sequence_1, dtype=float)
        init_raw = np.maximum(0.0, init_raw)                # protect against negatives
        init_raw = np.minimum(1000.0, init_raw)             # clip to allowed range
        init_seq = init_raw / init_raw.sum()
        best_seq = init_seq.copy()
        best_score = evaluate_sequence(best_seq.tolist())
    except Exception:
        best_seq = np.full(n, 1.0 / n)
        best_score = evaluate_sequence(best_seq.tolist())

    print(f"[init] n={n}, baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Helper to normalise a raw vector, evaluate it and possibly update
    # the global best.  ``label`` is only for logging.
    # ------------------------------------------------------------------
    def _try_raw(raw, label):
        nonlocal best_seq, best_score
        score, norm_vec = _norm_and_score(raw)
        if norm_vec is None:
            return
        if score < best_score:
            best_score = score
            best_seq = norm_vec
            print(f"[{label}] improved to {score:.6f}")

    # ------------------------------------------------------------------
    # 1. Broad parametric sampling
    # ------------------------------------------------------------------
    # a) grid of power‑laws
    alphas = np.linspace(0.10, 1.60, 71)
    shifts = np.linspace(0.01, 5.0, 30)
    for alpha in alphas:
        if time.time() - start > budget_s - 200:
            break
        for shift in shifts:
            raw = np.power(idx + shift, -alpha)
            _try_raw(raw, "gridPL")

    # b) random power‑laws
    for _ in range(2000):
        if time.time() - start > budget_s - 200:
            break
        alpha = np.random.uniform(0.05, 1.70)
        shift = np.random.uniform(0.001, 6.0)
        raw = np.power(idx + shift, -alpha)
        _try_raw(raw, "rndPL")

    # c) random exponentials
    for _ in range(1500):
        if time.time() - start > budget_s - 200:
            break
        beta = np.random.uniform(1e-5, 0.2)
        raw = np.exp(-beta * idx)
        _try_raw(raw, "exp")

    # d) mixtures of two power‑laws
    for _ in range(2000):
        if time.time() - start > budget_s - 200:
            break
        a1 = np.random.uniform(0.05, 1.70)
        s1 = np.random.uniform(0.001, 6.0)
        a2 = np.random.uniform(0.05, 1.70)
        s2 = np.random.uniform(0.001, 6.0)
        p = np.random.rand()
        raw = p * np.power(idx + s1, -a1) + (1.0 - p) * np.power(idx + s2, -a2)
        _try_raw(raw, "mixPL")

    # e) Dirichlet‑type samples (highly variable shapes)
    for _ in range(1000):
        if time.time() - start > budget_s - 200:
            break
        beta = np.random.uniform(0.0, 2.0)
        conc = np.exp(-beta * idx) + 0.01               # ensure positivity
        gam = np.random.gamma(conc, 1.0)
        raw = gam / gam.sum()
        # raw is already normalised
        score = _compute_score(raw)
        if score < best_score:
            best_score = score
            best_seq = raw.copy()
            print(f"[dirich] improved to {score:.6f}")

    print(f"[sampling] best score after sampling: {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Smooth‑max gradient descent (takes the majority of remaining time)
    # ------------------------------------------------------------------
    elapsed = time.time() - start
    remaining = budget_s - elapsed - 5.0          # keep a tiny safety margin
    if remaining > 10.0:
        # split the remaining budget: 60 % for gradient descent, 40 % for hill climb
        gd_budget = remaining * 0.60
        hc_budget = remaining - gd_budget
        print(f"[gradient] start, budget {gd_budget:.1f}s")
        vec_gd, gd_score = _gradient_descent(best_seq, gd_budget, start)
        # official check (the gradient routine already did occasional checks)
        real_score = evaluate_sequence(vec_gd.tolist())
        if real_score < best_score:
            best_score = real_score
            best_seq = vec_gd.copy()
            print(f"[gradient] new best official score {real_score:.6f}")

        # ------------------------------------------------------------------
        # 3. Focused hill‑climbing refinement (uses the rest of the budget)
        # ------------------------------------------------------------------
        print(f"[hillclimb] start, budget {hc_budget:.1f}s")
        vec_hc, hc_score = _hill_climb(best_seq, hc_budget, start)
        real_score = evaluate_sequence(vec_hc.tolist())
        if real_score < best_score:
            best_score = real_score
            best_seq = vec_hc.copy()
            print(f"[hillclimb] new best official score {real_score:.6f}")

    # ------------------------------------------------------------------
    # Final reporting
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[final] best score found: {final_score:.6f}")
    return best_seq.tolist()
