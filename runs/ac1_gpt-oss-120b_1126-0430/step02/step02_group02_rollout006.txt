def build_conv_matrix(f):
    n = len(f)
    m = 2*n - 1
    A = np.empty((m, n), dtype=np.float64)
    # can't allocate full then fill columnwise - just start with zeros then assign.
    A[:] = 0.0
    for i in range(n):
        A[i:i+n, i] = f
    return A
```

"""
A two‑stage optimiser for the step‑function bound.

Stage 1 – a quick scan of power‑law shapes (a_i ∝ (i+1)^‑p) gives a

Stage 2 – an iterative linear‑program refinement proposed in the problem
statement:
    * For the current vector f we compute M = max_k (f*f)[k].
    * Keeping M fixed we solve a linear program
          maximise   Σ g_j
          subject to (f*g)_k ≤ M   (k = 0 … 2n−2)
                     g_j ≥ 0
      The constraints are linear because convolution is linear in g.
    * The LP furnishes a new direction g.  A small convex combination of
      f and the normalised g (chosen by a tiny line‑search) strictly
      decreases M, and the process is repeated.

The cheap objective (Σa_i = 1) is  2·n·max_k (a*a)[k] .
All heavy work uses FFTs (O(n log n)) or a single LP per iteration
(≈ O(n²) memory, but n is kept modest ≈ 2500, well within the limits).

The best vector seen during the run is finally evaluated with the
provided ``evaluate_sequence`` and returned.

from scipy.optimize import linprog

def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (sum=1, v_i ≥ 0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float) -> np.ndarray:
    """Return the normalised power‑law vector a_i ∝ (i+1)^‑p  (Σa_i = 1)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    seq = idx ** (-p)
    seq /= seq.sum()
    return seq


def _convolution_max(a: np.ndarray, fft_len: int) -> float:
    """Maximum entry of the linear convolution a * a using FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa).real[: 2 * a.size - 1]
    return float(conv.max())


def _subgradient_max_convolution(a: np.ndarray, max_idx: int) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)[k] at the active index ``max_idx``.
    For the active offset k we have (∂/∂a_i) (a*a)[k] = 2·a_{k‑i}
    whenever the index is valid.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    lo = max(0, max_idx - (n - 1))
    hi = min(max_idx, n - 1)
    i = np.arange(lo, hi + 1, dtype=np.int64)
    g[i] = 2.0 * a[max_idx - i]
    return g


def _subgradient_refine(a: np.ndarray, max_iters: int,
                        deadline: float, fft_len: int) -> np.ndarray:
    """
    A limited projected sub‑gradient descent on the cheap objective
    2·n·max_k (a*a)[k]  with the simplex constraint Σa_i = 1.
    """
    n = a.shape[0]
    eta0 = 0.5
    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        # convolution and active index
        fa = np.fft.fft(a, fft_len)
        conv = np.fft.ifft(fa * fa).real[: 2 * n - 1]
        max_idx = int(conv.argmax())
        # sub‑gradient
        g = _subgradient_max_convolution(a, max_idx)
        # diminishing step size
        eta = eta0 / sqrt(t)
        # gradient step + projection
        a_candidate = a - eta * g
        a = _simplex_projection(a_candidate)
    return a


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1)×n convolution matrix A such that A·g = f*g.
    Each column i contains the vector f shifted by i positions.
    """
    n = f.shape[0]
    # first column: f followed by n‑1 zeros
    c = np.concatenate([f, np.zeros(n - 1, dtype=np.float64)])
    # first row: f[0] followed by n‑1 zeros
    r = np.concatenate([f[:1], np.zeros(n - 1, dtype=np.float64)])
    A = toeplitz(c, r)          # shape (2n‑1, n)
    return A

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the evaluation
    score.  The routine respects the given time budget (≈ 1000 s) and
    returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0      # a small safety margin

    # --------------------------------------------------------------
    # 0. Choose a working length.
    #    If a previous construction is available we down‑sample it,
    #    otherwise we work with a modest default n≈2500.
    # --------------------------------------------------------------
    try:
        old_seq = np.array(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]  # noqa: F821
        old_seq = np.maximum(old_seq, 0.0)
        target_n = 2500
        if old_seq.size > target_n:
            factor = old_seq.size // target_n
            seq = old_seq[: factor * target_n].reshape(target_n, factor).mean(axis=1)
        else:
            seq = old_seq.copy()
            target_n = old_seq.size
    except Exception:
        # fallback – no previous sequence known
        target_n = 2500
        seq = None

    n = target_n
    fft_len = 1 << ((2 * n - 1).bit_length())

    # --------------------------------------------------------------
    # 1. Coarse power‑law scan to obtain a decent starting point.
    # --------------------------------------------------------------
    p_grid = np.linspace(0.55, 0.68, 81)    # dense enough around the known optimum
    best_f = None
    best_score = float("inf")
    for p in p_grid:
        if time.time() >= deadline:
            break
        a = _power_law_sequence(n, p)
        maxb = _convolution_max(a, fft_len)
        score = 2.0 * n * maxb                # cheap surrogate (sum=1)
        if score < best_score:
            best_score = score
            best_f = a.copy()
    if best_f is None:                        # should never happen
        best_f = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = 2.0 * n * _convolution_max(best_f, fft_len)

    print(f"[Init] Power‑law scan best cheap score = {best_score:.6f}")

    # If we have a down‑sampled old sequence, use it as a second starter
    if seq is not None:
        # normalise the old sequence to sum = 1
        seq = seq / seq.sum()
        maxb_old = _convolution_max(seq, fft_len)
        score_old = 2.0 * n * maxb_old
        if score_old < best_score:
            best_f = seq.copy()
            best_score = score_old
            print(f"[Init] Using down‑sampled previous sequence, cheap score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2. A brief projected sub‑gradient refinement (helps the LP stage).
    # --------------------------------------------------------------
    f = best_f.copy()
    f = _simplex_projection(f)          # safety (should already be on simplex)
    f = _subgradient_refine(f, max_iters=2000,
                            deadline=deadline,
                            fft_len=fft_len)

    # --------------------------------------------------------------
    # 3. Iterative LP‑based improvement (the core of the method).
    # --------------------------------------------------------------
    best_seq = f.copy()
    best_iter_score = 2.0 * n * _convolution_max(f, fft_len)

    iteration = 0
    max_lp_iters = 20                     # usually far fewer are needed
    t_candidates = [0.1, 0.2, 0.3, 0.5, 0.8]

    while time.time() < deadline and iteration < max_lp_iters:
        iteration += 1

        # current peak of the autocorrelation
        M = _convolution_max(f, fft_len)

        # build the convolution matrix A ( (f*g)_k = A·g )
        A = _build_conv_matrix(f)                     # shape (2n‑1, n)
        b = np.full(A.shape[0], M, dtype=np.float64)  # right‑hand side for all k
        c = -np.ones(n, dtype=np.float64)             # maximise sum(g)  ⇔ minimise ‑∑g
        bounds = [(0.0, None)] * n

        # solve the LP
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'presolve': True})
        if not res.success:
            print(f"[Iter {iteration}] LP failed: {res.message}")
            break

        g = res.x
        sum_g = g.sum()

        # If the LP cannot increase the ℓ₁‑norm we are (approximately) at a
        # stationary point – terminate the LP loop.
        if sum_g <= 1.0 + 1e-9:
            print(f"[Iter {iteration}] No ℓ₁‑gain (sum(g)={sum_g:.6e}), stopping LP.")
            break

        # Normalise the LP solution to the simplex.
        g_norm = g / sum_g

        # Try a few mixing factors and keep the best resulting M.
        best_local_f = None
        best_local_M = None
        for t in t_candidates:
            cand = (1.0 - t) * f + t * g_norm
            cand_max = _convolution_max(cand, fft_len)
            if (best_local_M is None) or (cand_max < best_local_M):
                best_local_M = cand_max
                best_local_f = cand

        # Update f with the best convex combination found.
        f = best_local_f
        # Ensure exact simplex feasibility (tiny numerical drift possible)
        f = _simplex_projection(f)

        # Cheap bound for the new iterate
        cheap_score = 2.0 * n * best_local_M
        if cheap_score < best_iter_score:
            best_iter_score = cheap_score
            best_seq = f.copy()
            print(f"[Iter {iteration:02d}] NEW cheap bound = {cheap_score:.6f}")

        # occasional exact evaluation for monitoring
        if iteration % 5 == 0 or time.time() > deadline - 4.0:
            try:
                exact = evaluate_sequence(best_seq.tolist())
                print(f"    Exact evaluation (iter {iteration}) = {exact:.6f}")
                # keep the exact best if it happens to be lower
                if exact < best_iter_score:
                    best_iter_score = exact
            except Exception as exc:
                print(f"    Exact evaluation error: {exc}")

    # --------------------------------------------------------------
    # 4. Final polishing with a short sub‑gradient run (optional).
    # --------------------------------------------------------------
    remaining_time = deadline - time.time()
    if remaining_time > 5.0:
        f = _subgradient_refine(best_seq, max_iters=5000,
                                deadline=deadline,
                                fft_len=fft_len)
        final_cheap = 2.0 * n * _convolution_max(f, fft_len)
        if final_cheap < best_iter_score:
            best_seq = f.copy()
            best_iter_score = final_cheap
            print(f"[Polish] Improved cheap bound = {final_cheap:.6f}")

    # --------------------------------------------------------------
    # 5. Final exact evaluation and return.
    # --------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:
        final_score = float("inf")
        print(f"Final evaluation failed: {exc}")

    return final_seq
