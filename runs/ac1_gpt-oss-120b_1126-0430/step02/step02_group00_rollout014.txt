# Helper: Euclidean projection onto the probability simplex {x>=0, sum=1}
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex using the algorithm of
    Wang & Carreira‑Perpiñán (2013)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1, dtype=float)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:                     # all non‑positive
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution via FFT (returns length 2·n‑1)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length L (L ≥ 2·len(a)-1)."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) (used for exact max‑convolution objective)
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Return a sub‑gradient of max(conv) w.r.t. a."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)   # indices where max is attained
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        # each pair (i,j) contributes 2*a_j to ∂max/∂a_i
        g[i_idx] += 2.0 * a[j_idx]
    g /= float(len(ks))                    # average if several maximisers
    return g


# ----------------------------------------------------------------------
# Gradient of the soft‑max (log‑sum‑exp) surrogate of max(conv)
def _smooth_gradient(a: np.ndarray, conv: np.ndarray, alpha: float, L: int) -> np.ndarray:
    """
    Compute ∇_a  ( (2·n/α)·log Σ_k exp(α·conv_k) )
    which reduces to 2·corr(w, a) with w_k ∝ exp(α·conv_k).
    """
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)       # stability shift
    w = exp_scaled / exp_scaled.sum()              # soft‑max weights, length 2n‑1

    # correlation w ⋆ a  (g_i = 2· Σ_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)  # length L
    n = a.size
    grad = 2.0 * corr_rev[n - 1 : n - 1 + n]         # central n entries
    return grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (non‑negative sequence) that minimises the
    evaluation function.  Returns the sequence as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # 1.  Initialise a probability vector
    if 'height_sequence_1' in globals():
        base = np.array(height_sequence_1, dtype=float)
        n = base.size
        a = base / base.sum()
        print(f"[init] using supplied sequence, n={n}")
    else:
        n = 4096                     # a power of two is convenient for FFT
        a = np.full(n, 1.0 / n)
        print(f"[init] no prior sequence – using uniform, n={n}")

    # FFT length: need to accommodate both convolution (2n‑1) and correlation (3n‑2)
    L = 1 << ((3 * n - 2).bit_length())   # next power of two

    # Keep best ever seen
    conv = _conv_fft(a, L)
    best_score = 2.0 * n * conv.max()
    best_seq = a.copy()
    print(f"[init] O={best_score:.9f}")

    # ------------------------------------------------------------------
    # 2. Phase 1 – smooth optimisation (Adam‑style momentum)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    m = np.zeros_like(a)          # 1st moment
    v = np.zeros_like(a)          # 2nd moment
    alpha_start, alpha_end = 1.0, 4000.0
    max_iter_phase1 = 250_000
    phase1_deadline = start_time + budget_s * 0.55
    it = 0
    while it < max_iter_phase1 and time.time() < phase1_deadline:
        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_score - 1e-12:
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[phase‑1][{time.time():.1f}] new best O={best_score:.9f} (iter={it})")

        # linear schedule for α
        frac = it / max_iter_phase1
        alpha = alpha_start + (alpha_end - alpha_start) * frac

        # gradient of smooth surrogate
        grad = _smooth_gradient(a, conv, alpha, L)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        direction = m_hat / (np.sqrt(v_hat) + eps)

        # scale step so that the largest coordinate change is ≤ 0.4
        max_abs = np.max(np.abs(direction)) + 1e-16
        step = 0.4 / max_abs
        cand = a - step * direction
        cand = _project_to_simplex(cand)

        # replace a with the candidate (no line‑search – Adam already adapts)
        a = cand
        it += 1

    # ------------------------------------------------------------------
    # 3. Phase 2 – exact sub‑gradient descent + local hill‑climbing
    print(f"[phase‑2] entering, best O={best_score:.9f}")
    # reuse the best vector as starting point for phase 2
    a = best_seq.copy()
    conv = _conv_fft(a, L)
    cur_obj = 2.0 * n * conv.max()
    it_local = 0
    no_improve_counter = 0
    # parameters
    jitter_prob = 0.001
    mass_transfer_frac = 0.06   # up to 6 % of the source mass per move
    metropolis_temp = 0.008

    while time.time() < deadline:
        # ----- sub‑gradient step with maximal feasible step ----------
        g = _subgradient_max_conv(a, conv)            # sub‑gradient of max(conv)
        # direction that preserves the sum (Σ d_i = 0)
        d = -g + (2.0 / n) * np.ones_like(a)

        # maximal η keeping a+η d ≥ 0
        neg = d < 0
        if np.any(neg):
            eta_max = np.min(a[neg] / (-d[neg]))
            # use 90 % of the feasible distance to stay safely inside the simplex
            eta = 0.90 * eta_max
        else:
            eta = 0.0

        if eta > 1e-12:
            a_new = a + eta * d
            a_new = _project_to_simplex(a_new)   # correct possible tiny drift
            conv_new = _conv_fft(a_new, L)
            new_obj = 2.0 * n * conv_new.max()
            if new_obj < cur_obj - 1e-12:
                a = a_new
                conv = conv_new
                cur_obj = new_obj
                no_improve_counter = 0
                if cur_obj < best_score:
                    best_score = cur_obj
                    best_seq = a.copy()
                    print(f"[sub‑grad][{time.time():.1f}] new global best O={best_score:.9f}")
            else:
                no_improve_counter += 1
        else:
            no_improve_counter += 1

        # ----- random mass‑transfer moves (hill‑climbing) ---------------
        # try a few independent moves per outer iteration
        for _ in range(2):
            i, j = np.random.randint(n, size=2)
            if i == j or a[i] <= 0.0:
                continue
            delta = a[i] * np.random.rand() * mass_transfer_frac
            a_cand = a.copy()
            a_cand[i] -= delta
            a_cand[j] += delta
            # (sum is unchanged and all entries stay ≥0)
            conv_cand = _conv_fft(a_cand, L)
            cand_obj = 2.0 * n * conv_cand.max()
            # Metropolis acceptance
            if cand_obj < cur_obj or np.random.rand() < math.exp((cur_obj - cand_obj) / max(metropolis_temp, 1e-16)):
                a = a_cand
                conv = conv_cand
                cur_obj = cand_obj
                no_improve_counter = 0
                if cur_obj < best_score:
                    best_score = cur_obj
                    best_seq = a.copy()
                    print(f"[transfer][{time.time():.1f}] new global best O={best_score:.9f}")

        # ----- occasional jitter to escape flat plateaus --------------
        if np.random.rand() < jitter_prob:
            a += np.random.randn(n) * 0.001
            a = np.clip(a, 0.0, None)
            a = _project_to_simplex(a)
            conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max()
            # no immediate best‑check – jitter is tiny, will be caught later

        # ----- restart if stuck for a while --------------------------
        it_local += 1
        if no_improve_counter > 5000:
            # perturb the current best and continue from there
            a = best_seq.copy()
            a += np.random.randn(n) * 0.004
            a = np.clip(a, 0.0, None)
            a = _project_to_simplex(a)
            conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max()
            no_improve_counter = 0
            print(f"[restart][{time.time():.1f}] re‑initialised after {it_local} iterations")

        # safety break if time ran out
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # 4. Final sanity check with the supplied evaluator
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.9f}")
    except Exception as e:
        print("[final] evaluate_sequence failed:", e)

    return best_seq.tolist()
