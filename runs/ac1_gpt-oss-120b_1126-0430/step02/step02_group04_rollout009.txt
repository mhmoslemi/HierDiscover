def permutation_hill_climb(a, best_score, time_budget):

   n = len(a)

"""
A fast multi‑stage optimiser for the “minimise the max of a*a” problem.

Stage 1 – diverse seeds
    * Catalan (the historic seed)
    * Exponential, power‑law and classic window shapes (sine, Hann)
    * Random simplex points
    * For each shape also try a few random shuffles and a
      *Sidon‑set* placement of the largest entries (the large terms are forced
      to have distinct pairwise sums, which dramatically reduces the worst
      convolution coefficient).

Stage 2 – sub‑gradient descent on the current maximal coefficient
    * At each iteration the index k attaining max(a∗a) is found.
    * The exact sub‑gradient of  J(a)=2·n·max(a∗a)/∥a∥₁²  at that index is
          ∇J = (4 n/‖a‖₁²)·g − (4 n·max/‖a‖₁³)·1,
      where g_i = 2·a_{k−i} (zero when the index is out of range).
    * A diminishing step size (η₀/√t) and Euclidean projection onto the
      probability simplex keep the iterates feasible.
    * Occasional tiny Gaussian jitter helps escaping plateaus.

Stage 3 – permutation hill‑climb
    * Keeping the multiset of values fixed, random swaps are tried.
    * Each swap is accepted only if it improves the objective (a simple
      greedy local search).  A time budget limits the number of swaps.

All heavy work (convolutions, FFTs) is O(n log n) with n≈8000, so many
iterations fit easily into the 1000 s time limit.  The best sequence found
before the deadline is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u > cssv / np.arange(1, n + 1))[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using a real FFT."""
    n = x.size + y.size - 1
    size = 1 << (n - 1).bit_length()
    X = np.fft.rfft(x, size)
    Y = np.fft.rfft(y, size)
    Z = X * Y
    conv = np.fft.irfft(Z, size)[:n]
    return conv.real


def _eval_score(a: np.ndarray) -> float:
    """Exact value of the supplied evaluate_sequence (scale‑invariant)."""
    n = a.size
    b = _convolve_fft(a, a)
    return float(2.0 * n * b.max() / (a.sum() ** 2))


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------
def _catalan_sequence(n: int) -> np.ndarray:
    """Catalan‑like seed a_i = C(2i,i)/4**i, normalised to sum 1."""
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * (i - 1) + 1) / (2.0 * i)
    a /= a.sum()
    return a


def _exp_decay(n: int, r: float) -> np.ndarray:
    """Geometric decay a_i = r**i, normalised."""
    a = r ** np.arange(n, dtype=float)
    return a / a.sum()


def _powerlaw_decay(n: int, p: float) -> np.ndarray:
    """Power‑law a_i = 1/(i+1)^p, normalised."""
    a = 1.0 / (np.arange(1, n + 1, dtype=float) ** p)
    return a / a.sum()


def _sine_window(n: int) -> np.ndarray:
    """a_i = sin(pi·i/(n+1)), normalised."""
    i = np.arange(1, n + 1, dtype=float)
    a = np.sin(np.pi * i / (n + 1))
    return a / a.sum()


def _hann_window(n: int) -> np.ndarray:
    """Classic Hann window."""
    i = np.arange(n, dtype=float)
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
    return a / a.sum()


# ----------------------------------------------------------------------
# Sidon‑set arrangement
# ----------------------------------------------------------------------
def _greedy_sidon_set(n: int, target: int) -> list:
    """Greedy construction of a Sidon set (distinct pairwise sums)."""
    S = []
    sums = set()
    for i in range(n):
        conflict = False
        for s in S:
            if (i + s) in sums:
                conflict = True
                break
        if not conflict:
            S.append(i)
            # update sums with the new element
            for s in S:
                sums.add(i + s)
            if len(S) >= target:
                break
    return S


def _arrange_sidon(desc_vals: np.ndarray, n: int, k: int) -> np.ndarray:
    """
    Place the largest `k` values of `desc_vals` on a Sidon set,
    fill the remaining slots with the rest (preserving order).
    """
    # ensure k does not exceed length
    k = min(k, desc_vals.size)
    # obtain a Sidon set of size k (or as large as possible)
    S = _greedy_sidon_set(n, k)
    if len(S) < k:                     # fallback – just use a random placement
        a = np.zeros(n, dtype=float)
        perm = np.random.permutation(n)
        a[perm[:k]] = desc_vals[:k]
        a[perm[k:]] = desc_vals[k:]
        return _proj_simplex(a)

    # shuffle the top‑k values before placing them
    top_vals = desc_vals[:k].copy()
    np.random.shuffle(top_vals)

    a = np.zeros(n, dtype=float)
    for pos, val in zip(S, top_vals):
        a[pos] = val

    # Fill the remaining positions with the rest of the values
    rest_vals = desc_vals[k:]
    remaining_idx = [i for i in range(n) if a[i] == 0.0]
    a[remaining_idx[:rest_vals.size]] = rest_vals
    # any still‑empty slots stay zero – the simplex projection will renormalise
    return _proj_simplex(a)


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         max_iter: int,
                         lr0: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent on the current maximal convolution coefficient."""
    a = a0.copy()
    n = a.size
    cur_score = _eval_score(a)

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        # full convolution → locate the maximal entry
        b = _convolve_fft(a, a)
        max_b = b.max()
        k = int(np.argmax(b))

        # g_i = 2·a_{k‑i} for valid i
        g = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx_i = np.arange(lo, hi + 1, dtype=int)
        g[idx_i] = 2.0 * a[k - idx_i]

        sum_a = a.sum()
        # exact sub‑gradient of J(a)=2 n max/‖a‖₁²
        grad = (4.0 * n / (sum_a ** 2)) * g - (4.0 * n * max_b / (sum_a ** 3)) * np.ones_like(a)

        # diminishing step size
        lr = lr0 / np.sqrt(it + 1.0)
        a_candidate = a - lr * grad
        a_candidate = _proj_simplex(a_candidate)

        cand_score = _eval_score(a_candidate)
        if cand_score < cur_score:
            a = a_candidate
            cur_score = cand_score
        else:
            # simple back‑off if we are stuck
            lr0 *= 0.9

        # occasional tiny jitter to leave flat regions
        if it % 200 == 0 and it > 0:
            jitter = 0.001 * np.random.randn(n)
            a_jit = a + jitter
            a_jit = np.maximum(a_jit, 0.0)
            a_jit = _proj_simplex(a_jit)
            s_jit = _eval_score(a_jit)
            if s_jit < cur_score:
                a = a_jit
                cur_score = s_jit

    return a, cur_score


def _permutation_hill_climb(a0: np.ndarray,
                            cur_score: float,
                            deadline: float,
                            max_swaps: int = 30000) -> tuple[np.ndarray, float]:
    """Greedy local search on the ordering (swap two positions)."""
    a = a0.copy()
    best = cur_score
    n = a.size
    attempts = 0

    while time.time() < deadline and attempts < max_swaps:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        b = a.copy()
        b[i], b[j] = b[j], b[i]
        s = _eval_score(b)
        if s < best:
            a = b
            best = s
        attempts += 1
    return a, best


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a non‑negative height sequence that minimises the supplied
    ``evaluate_sequence``.  Returns the best sequence found before the
    time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0          # a safety margin

    n = 8000                                         # problem size

    # ------------------------------------------------------------------
    # 1. generate a diverse pool of seeds
    # ------------------------------------------------------------------
    seeds = []

    # classic Catalan seed
    seeds.append(_catalan_sequence(n))

    # exponential families
    for r in np.linspace(0.6, 0.9995, 5):
        a = _exp_decay(n, r)
        seeds.append(a)
        # a few random permutations of the same multiset
        for _ in range(2):
            b = a.copy()
            np.random.shuffle(b)
            seeds.append(b)

    # power‑law families
    for p in [0.5, 1.0, 1.5, 2.0]:
        a = _powerlaw_decay(n, p)
        seeds.append(a)
        for _ in range(2):
            b = a.copy()
            np.random.shuffle(b)
            seeds.append(b)

    # sine and Hann windows (smooth shapes)
    for gen in (_sine_window, _hann_window):
        a = gen(n)
        seeds.append(a)
        b = a.copy()
        np.random.shuffle(b)
        seeds.append(b)

    # a pure random simplex point
    rnd = np.random.rand(n)
    seeds.append(_proj_simplex(rnd))

    # Sidon‑set placements for a handful of the above shapes
    for base in seeds[:12]:
        desc = np.sort(base)[::-1]                     # descending order
        k = int(np.sqrt(n) * 2)                        # size of the Sidon set
        a_sidon = _arrange_sidon(desc, n, k)
        seeds.append(a_sidon)

    # ------------------------------------------------------------------
    # 2. pick the best initial candidate
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    for a in seeds:
        sc = _eval_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
    print(f"[init] best initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. iterative refinement: sub‑gradient descent + permutation search
    # ------------------------------------------------------------------
    while time.time() < deadline - 30:               # keep ~30 s for final touches
        # ----- sub‑gradient descent on the current best -----
        a_ref, sc_ref = _subgradient_descent(
            best_seq,
            max_iter=5000,
            lr0=0.2,
            deadline=deadline - 30,
        )
        if sc_ref < best_score:
            best_score = sc_ref
            best_seq = a_ref.copy()
            print(f"[sgd] improved score = {best_score:.6f}")

        # ----- greedy permutation hill‑climb -----
        a_perm, sc_perm = _permutation_hill_climb(
            best_seq,
            best_score,
            deadline=deadline - 30,
            max_swaps=25000,
        )
        if sc_perm < best_score:
            best_score = sc_perm
            best_seq = a_perm.copy()
            print(f"[swap] improved score = {best_score:.6f}")

        # ----- occasional tiny random restart around the best -----
        jitter = 0.01 * np.random.randn(n)
        a_jit = best_seq + jitter
        a_jit = np.maximum(a_jit, 0.0)
        a_jit = _proj_simplex(a_jit)
        sc_jit = _eval_score(a_jit)
        if sc_jit < best_score:
            best_score = sc_jit
            best_seq = a_jit.copy()
            print(f"[jitter] improved score = {best_score:.6f}")

    print(f"[final] best score found = {best_score:.6f}")
    # Return as plain Python list, as required by the harness
    return best_seq.tolist()
