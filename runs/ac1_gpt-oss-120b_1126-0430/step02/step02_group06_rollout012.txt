def project_simplex(v):
    v = np.asarray(v)
    # shift to nonnegative via subtracting tau
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _project_to_simplex(v):
    v = np.asarray(v)
    n = v.shape[0]
    # shift to ensure sum=1
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1  # sum -1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        # all entries <= 0, return uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]  # last true index
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a low‑score step function.

The algorithm works on the simplex (non‑negative coefficients summing to one).
It minimises a smooth approximation of the maximal self‑convolution using
projected gradient descent with an increasing “soft‑max’’ temperature.
The gradient is computed efficiently in the Fourier domain, and after each
gradient step the vector is projected back onto the probability simplex.
A simple restart/perturbation scheme is added to avoid plateaus.

The method starts from the Catalan (arcsine) seed, which is already a strong
baseline, and then refines it for the full time budget (default 1000 s).
The returned list is guaranteed to be non‑negative and its sum is 1, which
makes the objective scale‑invariant.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Fast evaluation (identical to evaluate_sequence but without the security
# checks) – used only for intermediate reporting, the final score will be
# computed by the external ``evaluate_sequence`` function.
# ----------------------------------------------------------------------
def _fast_evaluation(seq: np.ndarray) -> float:
    """Return the same metric as evaluate_sequence but much faster."""
    n = seq.size
    if n == 0:
        return np.inf
    # Length for linear convolution (full size)
    L = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, L)
    conv = np.fft.irfft(fa * fa, L)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Catalan (arcsine) seed – the coefficients of (1‑z)^‑1/2.
# ----------------------------------------------------------------------
def _catalan_sequence(length: int) -> np.ndarray:
    """First ``length`` Catalan‑like coefficients, normalised later."""
    seq = np.empty(length, dtype=float)
    a = 1.0
    seq[0] = a
    for i in range(1, length):
        a *= (2.0 * i - 1) / (2.0 * i)
        seq[i] = a
    return seq


# ----------------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex.
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto {x≥0, sum(x)=1}."""
    n = v.shape[0]
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # Find the last index where the condition holds
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # Numerical safety: ensure the sum is exactly 1
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(v, 1.0 / n)
    else:
        w /= w_sum
    return w


# ----------------------------------------------------------------------
# Main search routine.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Return a list of non‑negative heights with a low evaluation score.

    The routine works completely within the supplied time budget.
    It starts from the Catalan seed, then performs a smoothed
    projected‑gradient descent on the simplex.  A soft‑max temperature
    (``beta``) is increased gradually so that the algorithm first
    smooths the max‑operator and later focuses on the true peak.
    The best sequence seen during the run is returned.
    """
    np.random.seed(seed)
    start = time.time()
    # Give a small safety margin for the harness
    deadline = start + budget_s - 5.0

    # --------------------------------------------------------------
    # 1.  Choose the length of the step function.
    # --------------------------------------------------------------
    n = 5000                     # a good trade‑off between quality and speed

    # --------------------------------------------------------------
    # 2.  Initialise from the Catalan (arcsine) seed and normalise.
    # --------------------------------------------------------------
    seq = _catalan_sequence(n).astype(float)
    seq = np.maximum(seq, 0.0)
    seq /= seq.sum()             # now sum(seq) == 1

    # A tiny random perturbation helps avoid pathological symmetry.
    seq += 1e-6 * np.random.rand(n)
    seq = _project_to_simplex(seq)

    # --------------------------------------------------------------
    # 3.  Pre‑compute FFT length that can accommodate all linear
    #     convolutions (self‑convolution length = 2n‑1,
    #     correlation length = n + (2n‑1) = 3n‑1).
    # --------------------------------------------------------------
    L = 1 << ((3 * n - 2).bit_length())

    # --------------------------------------------------------------
    # 4.  Hyper‑parameters
    # --------------------------------------------------------------
    beta = 5.0                    # starting smoothness (soft‑max temperature)
    beta_max = 200.0              # final temperature
    beta_inc = (beta_max - beta) / max(1, int(0.9 * (budget_s - 5)))  # linear increase
    eta0 = 0.7                    # initial step size for the projected gradient
    restart_interval = 2000      # how often to inject a random perturbation
    no_improve_limit = 5000      # stop early if no improvement for many steps

    best_seq = seq.copy()
    best_score = _fast_evaluation(best_seq)
    print(f"[init] n={n}  score={best_score:.6f}")

    it = 0
    last_improve = 0

    while time.time() < deadline:
        # ----------------------------------------------------------
        # a)  Compute self‑convolution (full) via FFT
        # ----------------------------------------------------------
        fa = np.fft.fft(seq, L)
        conv_full = np.real(np.fft.ifft(fa * fa))
        conv = conv_full[: 2 * n - 1]                     # length 2n‑1

        # ----------------------------------------------------------
        # b)  Build a soft‑max weight vector over convolution entries
        # ----------------------------------------------------------
        max_conv = conv.max()
        # subtract the maximum to avoid overflow
        exp_weights = np.exp(beta * (conv - max_conv))
        p = exp_weights / exp_weights.sum()               # sum(p) == 1

        # ----------------------------------------------------------
        # c)  Gradient of the smoothed objective:
        #     g = 2 * (a ⋆ p)   where ⋆ denotes linear cross‑correlation.
        # ----------------------------------------------------------
        fp = np.fft.fft(p, L)
        # cross‑correlation a ⋆ p = ifft(fft(a) * conj(fft(p)))
        corr_full = np.real(np.fft.ifft(fa * np.conj(fp)))
        # valid lags 0 … n‑1 appear at the beginning of the linear result
        g = 2.0 * corr_full[:n]

        # ----------------------------------------------------------
        # d)  Projected gradient descent step
        # ----------------------------------------------------------
        eta = eta0 / np.sqrt(it + 1.0)          # diminishing step size
        new_seq = seq - eta * g
        new_seq = _project_to_simplex(new_seq)

        # ----------------------------------------------------------
        # e)  Evaluate (fast) and keep the best seen so far
        # ----------------------------------------------------------
        cur_score = _fast_evaluation(new_seq)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = new_seq.copy()
            last_improve = it
            if it % 500 == 0:
                print(f"[iter {it:6d}] score={best_score:.6f}  beta={beta:.1f}")

        # ----------------------------------------------------------
        # f)  Prepare for next iteration
        # ----------------------------------------------------------
        seq = new_seq
        it += 1
        # increase the temperature slowly
        if beta < beta_max:
            beta = min(beta_max, beta + beta_inc)

        # ----------------------------------------------------------
        # g)  Random restart / perturbation to shake off stagnation
        # ----------------------------------------------------------
        if it % restart_interval == 0:
            # inject a small random jitter across all coordinates
            jitter = 1e-4 * np.random.randn(n)
            seq = _project_to_simplex(np.maximum(seq + jitter, 0.0))

        # ----------------------------------------------------------
        # h)  Early exit if no improvement for a long stretch
        # ----------------------------------------------------------
        if it - last_improve > no_improve_limit:
            # reset with a fresh random perturbation around the best point
            seq = best_seq + 1e-4 * np.random.randn(n)
            seq = _project_to_simplex(np.maximum(seq, 0.0))
            last_improve = it

    # --------------------------------------------------------------
    # 5.  Final reporting
    # --------------------------------------------------------------
    final_score = _fast_evaluation(best_seq)
    print(f"[final] best score={final_score:.6f} after {it} iterations")
    # Return as plain Python list – evaluate_sequence will perform the
    # final, fully‑checked evaluation.
    return best_seq.tolist()
```

This implementation stays within the allowed 1000 s, works entirely on the simplex, uses only `numpy` and standard library, and follows the required interface. The combination of a soft‑max smoothing of the max‑operator, efficient FFT‑based gradient computation, and a simple restart/perturbation scheme typically yields a noticeable improvement over the baseline Catalan/EG method.
