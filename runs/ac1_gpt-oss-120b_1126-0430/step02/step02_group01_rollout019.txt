Implementation details of LP:

```
def solve_lp(a, M):
    n = len(a)
    rows = 2*n - 1
    A = np.zeros((rows, n))
    for i in range(n):
        A[i:i+n, i] = a
    b = np.full(rows, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    else:
        return None
```

"""step_function_optimizer.py

A search for a non‑negative step function that minimises

    O = 2·n·max(conv(a,a)) / (sum(a)**2)

using a simple linear‑programming based “mass‑inflation’’ phase
followed by a stochastic hill‑climbing refinement.
The LP is solved with SciPy’s high‑performance HiGHS solver
(and the code gracefully falls back to pure hill‑climbing if SciPy is absent).

The algorithm:
1.  Initialise a highly peaked vector (all mass on a single coordinate).
    Its convolution peak is huge, leaving a large slack for the LP.
2.  Repeatedly:
        – compute the current peak M = max(conv(a,a))
        – solve    max   sum(g)
                    s.t. conv(a,g) ≤ M   (element‑wise)
                         g ≥ 0
          The solution g can carry far more total mass than a while
          respecting the same convolution bound.
        – normalise g to keep the total sum equal to the chosen
          normalisation S = √(2·n) (the same normalisation used in the
          original paper).  The mixture a←(1‑α)a+α g (with several α
          candidates) never raises the peak and typically lowers it.
        – keep the best mixture as the new “a’’.
    The LP phase stops when no increase in total mass is possible.
3.  A lightweight simulated‑annealing / sub‑gradient hill‑climber
    further reduces the peak.
4.  The best vector found is returned as a plain list of floats.

The whole routine respects a 1000 s wall‑clock budget
(the “budget_s” argument) and prints progress information.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import – used for the linear program.
try:
    from scipy.optimize import linprog
    _SCIPY_AVAILABLE = True
except Exception:               # pragma: no cover
    _SCIPY_AVAILABLE = False


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return the evaluation objective  O = 2·n·max(conv(a,a)) / sum(a)² ."""
    n = a.size
    conv = np.convolve(a, a)
    max_conv = conv.max()
    s = a.sum()
    if s == 0.0:                         # safety – should never happen
        return float('inf')
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
def _build_conv_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the matrix A (size (2n‑1)×n) such that
        (A @ g)[k] == (np.convolve(a, g))[k]
    for all k.  This is a (banded) Toeplitz matrix.
    """
    n = a.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=float)
    # each column i contributes a[i] to rows i … i+n‑1
    for i in range(n):
        A[i:i + n, i] = a
    return A


# ----------------------------------------------------------------------
def _solve_lp(a: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the LP

        max   sum(g)
        s.t.  conv(a, g) ≤ M   (element‑wise)
              g ≥ 0

    Returns the optimal g (raw, not normalised) or None on failure.
    """
    if not _SCIPY_AVAILABLE:               # pragma: no cover
        return None
    n = a.size
    A = _build_conv_matrix(a)
    b = np.full(2 * n - 1, M, dtype=float)

    c = -np.ones(n, dtype=float)          # maximise sum(g)  → minimise -sum(g)
    bounds = [(0.0, None) for _ in range(n)]

    # HiGHS is fast and reliable for dense problems of this size.
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True})
    if res.success:
        g = np.asarray(res.x, dtype=float)
        # Numerical safety – enforce non‑negativity
        g[g < 0] = 0.0
        return g
    return None


# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) w.r.t. a on the probability simplex.
    Used only for a cheap occasional descent step in the hill‑climbing phase.
    """
    n = a.size
    conv = np.convolve(a, a)
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    g /= len(ks)
    return g

                    cur_score = new_score
                    iter_no_improve = 0
                    if cur_score < best_score:
                        best_score = cur_score
                        best_seq = cur_seq.copy()
                        print(f"[HC {iter_cnt}] New global best O={best_score:.6f}")
                else:
                    iter_no_improve += 1

            # ---- occasional sub‑gradient descent on the peak -------------------------
            if iter_cnt % 11 == 0:
                g_sub = _subgradient_max_conv(cur_seq)
                eta = 0.06 * (1.0 - iter_cnt / max_iters)
                cur_seq = cur_seq - eta * g_sub
                cur_seq = np.clip(cur_seq, 0.0, None)
                cur_seq = cur_seq * (S / cur_seq.sum())
                cur_score = _objective(cur_seq)
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = cur_seq.copy()
                    print(f"[SG {iter_cnt}] New global best O={best_score:.6f}")

            # ---- temperature schedule --------------------------------------------------
            T = T0 * (1.0 - iter_cnt / max_iters)

            # ---- restart if stuck -------------------------------------------------------
            if iter_no_improve >= max_no_improve:
                cur_seq = best_seq.copy()
                cur_seq += np.random.randn(n) * 0.004
                cur_seq = np.clip(cur_seq, 0.0, None)
                cur_seq = cur_seq * (S / cur_seq.sum())
                cur_score = _objective(cur_seq)
                iter_no_improve = 0
                print(f"[restart] after {iter_cnt} iters – re‑initialised from best.")

        # keep the best sequence found for this n
        if best_score < float('inf'):
            print(f"[summary n={n}] best O={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final check with the provided evaluate_sequence (exact scoring)
    if best_seq is None:                     # fallback – return a simple uniform vector
        n = 300
        best_seq = np.full(n, 1.0 / n, dtype=float)
    else:
        best_seq = best_seq.copy()

    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:                   # pragma: no cover
        print("[final] evaluate_sequence call failed:", e)

    return best_seq.tolist()
