def proj_simplex(v, s):
    # Duchi et al algorithm
    n = v.shape[0]
    if s < 0:
        raise ValueError
    # shift to ensure positivity
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.arange(1, n+1)
    cond = u - cssv / rho_arr > 0
    # find last true
    rho = np.where(cond)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w

This is O(n log n) due to sort.

def _exp_step(seq, w, eta, S):
    grad = 2.0 * w
    # compute scaling factor
    maxg = grad.max()
    # shift to avoid overflow
    exp_factor = np.exp(-eta * (grad - maxg))
    new_seq = seq * exp_factor
    # renorm to sum S
    total = new_seq.sum()
    if total <= 0:
        return seq
    new_seq = new_seq * (S / total)
    return new_seq

"""
Propose a step‑function with a lower evaluation score.

The strategy is a fast first‑order optimisation on the L‑∞
self‑convolution objective.

1.  Generate a few sensible initial guesses (Catalan / arcsine,
    a handful of Beta‑shaped sequences and, if present,
    the historic `height_sequence_1`).  All are scaled so that the
    maximum entry stays ≤ 1000 (the security clamp of the evaluator).

2.  Pick the best of those seeds using a lightweight FFT based
    score computation (`_fast_score`).

3.  Main optimisation loop (time‑bounded):
    * Compute the self‑convolution `b = a*a` with FFT.
    * Choose a random index `k` among the (near) maximal entries.
    * The gradient of `b_k = Σ a_i a_{k-i}` w.r.t. `a` is
      `∇b_k = 2·w` with `w_i = a_{k-i}` (zero out‑of‑range).
    * Perform an *exponentiated‑gradient* update
          a' = a · exp(‑η·∇b_k)      (· is element‑wise)
      and project the result back onto the simplex
      `{a ≥ 0, Σ a = S}` (`S` is the initial L¹‑norm).  This
      automatically moves mass from indices that contribute a lot
      to the current peak towards those that contribute little.
    * If the new sequence improves the score, keep it; otherwise
      fall back to a small “sink‑move’’ that mimics the original
      algorithm (mass transfer to a few indices with the smallest
      weight).  Every few hundred iterations a random jitter is
      performed to escape flat plateaux.

The whole routine runs for up to 1000 s (a 5 s safety margin) and
returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution / score (identical to the provided evaluate_sequence
# but without the security checks).  Uses FFT for O(n log n) work.
# ----------------------------------------------------------------------
def _fast_score(seq: np.ndarray) -> float:
    """Return the evaluation score for a non‑negative sequence."""
    n = seq.size
    if n == 0:
        return float("inf")
    # next power of two for FFT
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Classical Catalan (arcsine) seed.
# ----------------------------------------------------------------------
def _catalan_sequence(length: int) -> np.ndarray:
    seq = np.empty(length, dtype=float)
    a = 1.0
    seq[0] = a
    for i in range(1, length):
        a *= (2 * i - 1) / (2.0 * i)          # a_i = a_{i‑1}·(2i‑1)/(2i)
        seq[i] = a
    return seq


# ----------------------------------------------------------------------
# Beta(p,p) shape (U‑shaped for p<1).  Mid‑point rule discretisation.
# ----------------------------------------------------------------------
def _beta_sequence(length: int, p: float) -> np.ndarray:
    # points in (0,1)
    x = (np.arange(length, dtype=float) + 0.5) / length
    with np.errstate(divide="ignore"):
        log_pdf = (p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x)
    pdf = np.exp(log_pdf)                       # un‑normalised
    return pdf


# ----------------------------------------------------------------------
# Project a vector onto the simplex {x ≥ 0, Σx = s}.
# Duchi et al. (2008) algorithm – O(n log n).
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray, s: float) -> np.ndarray:
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Weight vector w_i = a_{k‑i} (zero outside the range).
# ----------------------------------------------------------------------
def _weight_vector(seq: np.ndarray, k: int) -> np.ndarray:
    n = seq.shape[0]
    idx = k - np.arange(n)
    mask = (idx >= 0) & (idx < n)
    w = np.zeros_like(seq)
    w[mask] = seq[idx[mask]]
    return w


# ----------------------------------------------------------------------
# One exponentiated‑gradient step with simplex projection.
# ----------------------------------------------------------------------
def _exp_gradient_step(seq: np.ndarray, w: np.ndarray, eta: float,
                       target_sum: float) -> np.ndarray:
    """
    seq   – current non‑negative vector (already on the simplex)
    w     – weight vector for the chosen peak (a_{k‑i})
    eta   – step size (>0)
    target_sum – the L¹‑norm we keep fixed
    """
    grad = 2.0 * w                     # ∇b_k
    # multiplicative update
    factor = np.exp(-eta * grad)       # shrink indices with large weight
    new_seq = seq * factor
    # enforce the simplex (renormalise)
    s = new_seq.sum()
    if s == 0.0:
        # should not happen, fall back to unchanged sequence
        return seq
    new_seq = new_seq * (target_sum / s)
    # safety clamp – keep everything ≤ 1000 (the evaluator silently caps)
    if new_seq.max() > 1000.0:
        new_seq = np.minimum(new_seq, 1000.0)
        # re‑project to keep the sum exactly target_sum
        new_seq = _proj_simplex(new_seq, target_sum)
    return new_seq


# ----------------------------------------------------------------------
# A small “sink‑move’’ similar to the original method – used when the
# exponentiated update does not improve the score.
# ----------------------------------------------------------------------
def _sink_move(seq: np.ndarray, w: np.ndarray, max_val: float,
               target_sum: float, sink_percent: float = 3.0,
               rel_target_reduce: float = 0.02) -> np.ndarray:
    n = seq.size
    thresh = np.percentile(w, sink_percent)
    sink_mask = w <= thresh
    sink_idxs = np.flatnonzero(sink_mask)
    if sink_idxs.size == 0:
        sink_idxs = np.array([int(np.argmin(w))])
        sink_mask = np.zeros_like(w, dtype=bool)
        sink_mask[sink_idxs[0]] = True

    w_min = w[sink_idxs].min()
    excess = seq * (w - w_min)
    total_excess = excess[~sink_mask].sum()
    if total_excess <= 0:
        return seq

    target_reduction = rel_target_reduce * max_val
    lam = target_reduction / total_excess
    lam = max(min(lam, 1.0), 1e-6)

    delta = lam * seq
    delta[sink_mask] = 0.0
    moved = delta.sum()
    add_per_sink = moved / sink_idxs.size

    new_seq = seq - delta
    new_seq[sink_idxs] += add_per_sink

    # project back onto the simplex (to correct tiny numerical drift)
    new_seq = _proj_simplex(new_seq, target_sum)
    return new_seq

        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
    print(f"Initial best score: {best_score:.8f}")

    # ------------------------------------------------------------------
    # 4.  Optimisation loop (projected exponentiated gradient + fall‑backs)
    # ------------------------------------------------------------------
    it = 0
    eta0 = 0.5                     # initial step‑size for the EG update
    no_improve = 0
    max_no_improve = 200          # trigger a sink‑move after this many failures
    n = best_seq.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())   # cached FFT length

    while time.time() < deadline:
        # ---------- self‑convolution ----------
        fa = np.fft.rfft(best_seq, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_val = conv.max()

        # ---------- pick a random maximal index ----------
        # tolerance to capture a flat plateau
        tol = max(max_val * 1e-12, 1e-15)
        max_idxs = np.flatnonzero(conv >= max_val - tol)
        k = int(np.random.choice(max_idxs))

        # ---------- weight vector for the chosen peak ----------
        w = _weight_vector(best_seq, k)

        # ---------- exponentiated‑gradient step ----------
        eta = eta0 / math.sqrt(it + 1)
        cand_seq = _exp_gradient_step(best_seq, w, eta, target_sum)
        cand_score = _fast_score(cand_seq)

        if cand_score < best_score - 1e-12:
            best_seq = cand_seq
            best_score = cand_score
            no_improve = 0
            print(f"Iter {it:6d}: EG improved → {best_score:.8f}")
        else:
            # ---------- fallback: sink move ----------
            no_improve += 1
            if no_improve >= max_no_improve:
                cand_seq = _sink_move(best_seq, w, max_val, target_sum,
                                      sink_percent=2.0, rel_target_reduce=0.02)
                cand_score = _fast_score(cand_seq)
                if cand_score < best_score - 1e-12:
                    best_seq = cand_seq
                    best_score = cand_score
                    print(f"Iter {it:6d}: sink move improved → {best_score:.8f}")
                no_improve = 0

        # ---------- occasional random shuffle to escape plateaux ----------
        if it % 100 == 0 and it > 0:
            i, j = np.random.randint(n, size=2)
            if i != j and best_seq[i] > 0:
                eps = min(best_seq[i], np.random.rand() * best_seq[i] * 0.05)
                tmp = best_seq.copy()
                tmp[i] -= eps
                tmp[j] += eps
                tmp_score = _fast_score(tmp)
                if tmp_score < best_score:
                    best_seq = tmp
                    best_score = tmp_score
                    print(f"Iter {it:6d}: random swap improved → {best_score:.8f}")

        it += 1

    print(f"Finished after {it} iterations. Best score: {best_score:.8f}")
    # Return a plain Python list – the harness will perform the final
    # evaluation using the official `evaluate_sequence` function.
    return best_seq.tolist()
