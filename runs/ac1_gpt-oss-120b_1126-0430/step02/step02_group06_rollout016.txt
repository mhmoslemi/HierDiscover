"""
Propose a low‑score step‑function.

Algorithm
---------
* Start from the discrete arcsine (Catalan) seed – already close to the
  known theoretical optimum (π/2 ≈ 1.5708).
* Repeatedly solve a linear programme

      maximise   1·g
      subject to  f * g ≤ M               (entry‑wise)
                  g ≥ 0

  where * is the ordinary discrete convolution,
  f is the current sequence and M = max(f*f).
  The LP finds a direction g that can carry *more* total mass than
  f while still respecting the same peak bound M.
* Rescale g to have the prescribed ℓ¹‑norm ( √(2 n) ) and move a
  moderate step towards it.  The step size is halved until the peak of
  the self‑convolution does not increase.
* The process strictly reduces the objective (the maximal convolution)
  and is iterated until the time budget is exhausted.
* In the remaining time a few epochs of exponentiated‑gradient descent
  (mirror descent on the simplex) are used for fine‑tuning.

The whole routine stays well within the 1000 s budget, prints progress
and finally returns the best sequence discovered.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Fast evaluation (identical to evaluate_sequence but without the security
# checks and using an FFT based convolution).
# ----------------------------------------------------------------------
def _fast_evaluation(seq: np.ndarray) -> float:
    """Return the same metric as evaluate_sequence but much faster."""
    n = seq.size
    if n == 0:
        return np.inf
    # FFT size = next power of two >= 2*n-1
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Catalan (arcsine) seed.
# a_0 = 1, a_{i+1} = a_i * (2*i+1) / (2*i+2)
# ----------------------------------------------------------------------
def _catalan_sequence(length: int) -> np.ndarray:
    """Return the first ``length`` Catalan‑like coefficients."""
    seq = np.empty(length, dtype=float)
    a = 1.0
    seq[0] = a
    for i in range(1, length):
        a *= (2 * i - 1) / (2.0 * i)
        seq[i] = a
    return seq


# ----------------------------------------------------------------------
# Linear‑programming based improvement step.
# ----------------------------------------------------------------------
def _lp_step(seq: np.ndarray, target_sum: float, deadline: float) -> tuple[np.ndarray, bool]:
    """
    Perform one LP‑based improvement step.
    Returns (new_seq, improved_flag).  ``improved_flag`` is True iff the
    maximal self‑convolution decreased.
    """
    n = seq.size
    # current maximal self‑convolution
    m_fft = 1 << ((2 * n - 1).bit_length())
    ff = np.fft.rfft(seq, m_fft)
    conv = np.fft.irfft(ff * ff, m_fft)[: 2 * n - 1]
    M = conv.max()

    # Build the convolution matrix A with A @ g = conv(seq, g)
    # A has shape (2n‑1, n); column i contains ``seq[i]`` on rows i … i+n‑1.
    rows = 2 * n - 1
    A = np.empty((rows, n), dtype=float)
    A.fill(0.0)
    for i in range(n):
        A[i:i + n, i] = seq[i]

    # Linear programme: maximise sum(g)  ⇔ minimise -sum(g)
    c = -np.ones(n, dtype=float)                       # objective
    bounds = [(0.0, None)] * n                         # g ≥ 0
    b_ub = np.full(rows, M, dtype=float)              # f * g ≤ M
    # solve
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        # LP failed – give up the LP step.
        return seq, False

    g = res.x
    sum_g = g.sum()
    if sum_g == 0.0:
        return seq, False

    # Rescale g to have the required ℓ¹‑norm.
    lam = target_sum / sum_g
    g = g * lam

    # --------------------------------------------------------------
    # Take a convex combination (1‑t)·seq + t·g.
    # Reduce t if the new maximal convolution would increase.
    # --------------------------------------------------------------
    t = 0.5                     # initial step size
    best_seq = seq.copy()
    improved = False
    while t > 1e-6 and time.time() < deadline:
        cand = (1.0 - t) * seq + t * g
        # keep strict positivity and renormalise (numerical drift)
        cand = np.maximum(cand, 1e-12)
        cand = cand / cand.sum() * target_sum

        # evaluate new maximal self‑convolution
        ff_c = np.fft.rfft(cand, m_fft)
        conv_c = np.fft.irfft(ff_c * ff_c, m_fft)[: 2 * n - 1]
        M_new = conv_c.max()

        if M_new <= M + 1e-12:            # non‑increase accepted
            best_seq = cand
            improved = True
            break
        t *= 0.5                           # backtrack

    return best_seq, improved


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) fine‑tuning.
# ----------------------------------------------------------------------
def _eg_refine(seq: np.ndarray, target_sum: float,
               deadline: float) -> np.ndarray:
    """
    Multiplicative (exponentiated‑gradient) descent on the simplex.
    ``seq`` is assumed non‑negative and already normalised to have
    ℓ¹‑norm = ``target_sum``.
    """
    n = seq.size
    best_seq = seq.copy()
    best_score = _fast_evaluation(best_seq)

    # FFT length used for all convolutions in this phase
    fft_len = 1 << ((2 * n - 1).bit_length())

    # Adam‑style state for the mirror descent
    m_vec = np.zeros_like(seq)
    v_vec = np.zeros_like(seq)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    lr = 0.12
    it = 0

    while time.time() < deadline:
        # current convolution and its maximum
        fa = np.fft.rfft(seq, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        M = conv.max()

        # weights derived from a smooth approximation of the max‑operator
        # (log‑sum‑exp with temperature α)
        alpha = 8.0
        exp_term = np.exp(alpha * (conv - M))
        w = exp_term / exp_term.sum()        # shape (2n‑1,)

        # sub‑gradient of the smooth max w.r.t. the sequence
        # g_i = 2 * Σ_k w_k * seq_{k‑i}
        w_rev = w[::-1]
        # convolution of w_rev with seq gives the needed sums
        conv_rev = np.fft.irfft(
            np.fft.rfft(w_rev, fft_len) * np.fft.rfft(seq, fft_len),
            fft_len)[: 3 * n - 2]
        g = 2.0 * conv_rev[2 * n - 2 - np.arange(n)]

        # mirror‑descent (exponentiated gradient) update
        dot = np.dot(seq, g)
        grad = seq * (g - dot)                     # ∇_x L = f·(g−⟨g,f⟩)

        it += 1
        m_vec = beta1 * m_vec + (1 - beta1) * grad
        v_vec = beta2 * v_vec + (1 - beta2) * (grad * grad)
        m_hat = m_vec / (1 - beta1 ** it)
        v_hat = v_vec / (1 - beta2 ** it)

        seq = seq * np.exp(-lr * m_hat / (np.sqrt(v_hat) + eps))

        # re‑project onto the simplex (positivity + fixed ℓ¹‑norm)
        seq = np.maximum(seq, 1e-15)
        seq = seq / seq.sum() * target_sum

        # occasional evaluation
        if it % 250 == 0:
            score = _fast_evaluation(seq)
            if score < best_score - 1e-12:
                best_score = score
                best_seq = seq.copy()
                print(f"[EG] iter {it:5d}  score {best_score:.6f}")

    return best_seq


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights aiming at a low evaluation score.

    The routine works within ``budget_s`` seconds (default 1000).  It starts
    from the Catalan (arcsine) distribution, repeatedly improves it by a
    linear‑programming step, and finishes with a few epochs of
    exponentiated‑gradient descent.  The best sequence discovered is returned.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0                 # safety margin

    # Length – a few thousand gives a good trade‑off between quality and speed.
    n = 2000                                          # moderate size (≈ 2 k)

    # Target ℓ¹‑norm (the metric is scale‑invariant, we keep √(2 n) for convenience)
    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # 1.  Initialise with the Catalan (arcsine) seed and normalise.
    # ------------------------------------------------------------------
    seq = _catalan_sequence(n).astype(float)
    seq = seq / seq.sum() * target_sum

    print(f"[init] n={n}  initial score={_fast_evaluation(seq):.6f}")

    # ------------------------------------------------------------------
    # 2.  Linear‑programming refinement – allocate a fixed portion of the budget.
    # ------------------------------------------------------------------
    lp_deadline = start + 0.45 * (budget_s - 5.0)      # about half the time
    lp_iter = 0
    while time.time() < lp_deadline:
        seq, improved = _lp_step(seq, target_sum, lp_deadline)
        lp_iter += 1
        if not improved:
            # No improvement – stop the LP phase early.
            break
        if lp_iter % 5 == 0:
            cur_score = _fast_evaluation(seq)
            print(f"[LP]  iter {lp_iter:3d}  score {cur_score:.6f}")

    cur_score = _fast_evaluation(seq)
    print(f"[after LP] iterations={lp_iter}  score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Exponentiated‑gradient fine‑tuning on the remaining time.
    # ------------------------------------------------------------------
    seq = _eg_refine(seq, target_sum, deadline)

    final_score = _fast_evaluation(seq)
    print(f"[final] best score={final_score:.6f}")

    # Return the best sequence as a plain Python list.
    return seq.tolist()
