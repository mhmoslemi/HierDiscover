    best_seq = None
    best_score = np.inf
    for a in candidates:
        sc = _eval_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
    a = best_seq.copy()
    cur_score = best_score
    print(f"[init, n={n_small}] best initial score = {cur_score:.6f}")

    # --------------------------------------------------------------
    # Stage 1 – Adam optimiser on the medium size problem
    # --------------------------------------------------------------
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.07                      # base learning rate for Adam
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    alpha = 5.0                    # soft‑max temperature
    stagnation = 0
    iteration = 0

    while time.time() < deadline:
        # ---- evaluate current iterate ------------------------------------------------
        b = _convolve_fft(a, a)
        max_b = b.max()
        cur_score = float(2.0 * n_small * max_b)   # denominator == 1

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{iteration:06d}] new best (n={n_small}) score={best_score:.6f} α={alpha:.1f}")
            stagnation = 0
        else:
            stagnation += 1

        # ---- soft‑max weights ---------------------------------------------------------
        w = np.exp(alpha * (b - max_b))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.ones_like(b) / b.size
        else:
            w = w / w_sum

        # ---- gradient of the smooth max (see analysis) -------------------------------
        w_rev = w[::-1]
        conv_full = _convolve_fft(a, w_rev)                     # length = 3n‑2
        idx = (len(b) - 1) - np.arange(n_small)                 # slice to size n
        grad = 2.0 * conv_full[idx]

        # ---- Adam update ---------------------------------------------------------------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = _proj_simplex(a)

        # ---- schedules ----------------------------------------------------------------
        if iteration % 1000 == 0:
            alpha = min(alpha * 1.03, 3000.0)
        if iteration % 3000 == 0:
            lr = max(lr * 0.95, 0.01)

        # ---- restart if we have been stuck for a long time ---------------------------
        if stagnation > 6000:
            a = best_seq + 0.01 * np.random.randn(n_small)
            a = np.maximum(a, 0.0)
            a = _proj_simplex(a)
            m.fill(0.0)
            v.fill(0.0)
            lr = 0.07
            alpha = 5.0
            stagnation = 0
            print(f"[{iteration:06d}] restart (score={_eval_score(a):.6f})")

        # ---- leave enough time for the refined stage ---------------------------------
        if time.time() > deadline - 250.0:
            break

    # --------------------------------------------------------------
    # Stage 2 – up‑sample to the large size and continue optimisation
    # --------------------------------------------------------------
    a_big = _downsample(best_seq, n_big)
    a_big = _proj_simplex(a_big)
    best_big = a_big.copy()
    best_big_score = _eval_score(a_big)

    # re‑initialise Adam state for the large problem
    m = np.zeros_like(a_big)
    v = np.zeros_like(a_big)
    lr = 0.05
    alpha = 10.0
    stagnation = 0
    iteration = 0

    while time.time() < deadline:
        b = _convolve_fft(a_big, a_big)
        max_b = b.max()
        cur_score = float(2.0 * n_big * max_b)

        if cur_score < best_big_score:
            best_big_score = cur_score
            best_big = a_big.copy()
            print(f"[{iteration:06d}] refined best (n={n_big}) score={best_big_score:.6f} α={alpha:.1f}")
            stagnation = 0
        else:
            stagnation += 1

        # soft‑max weights
        w = np.exp(alpha * (b - max_b))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.ones_like(b) / b.size
        else:
            w = w / w_sum

        # gradient
        w_rev = w[::-1]
        conv_full = _convolve_fft(a_big, w_rev)
        idx = (len(b) - 1) - np.arange(n_big)
        grad = 2.0 * conv_full[idx]

        # Adam update
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_big = a_big - step
        a_big = _proj_simplex(a_big)

        # schedules
        if iteration % 800 == 0:
            alpha = min(alpha * 1.02, 5000.0)
        if iteration % 2000 == 0:
            lr = max(lr * 0.92, 0.005)

        # stagnation handling
        if stagnation > 8000:
            a_big = best_big + 0.005 * np.random.randn(n_big)
            a_big = np.maximum(a_big, 0.0)
            a_big = _proj_simplex(a_big)
            m.fill(0.0)
            v.fill(0.0)
            lr = 0.05
            alpha = 10.0
            stagnation = 0
            print(f"[{iteration:06d}] restart big (score={_eval_score(a_big):.6f})")

    # --------------------------------------------------------------
    # Final polishing – a few hard‑max sub‑gradient steps
    # --------------------------------------------------------------
    final_deadline = time.time() + min(30.0, deadline - time.time())
    a = best_big.copy()
    step_base = 0.02
    it = 0
    while time.time() < final_deadline:
        b = _convolve_fft(a, a)
        max_b = b.max()
        k = np.argmax(b)

        w = np.zeros_like(b)
        w[k] = 1.0
        grad = 2.0 * _convolve_fft(a, w[::-1])[(len(b) - 1) - np.arange(n_big)]

        decay = step_base / np.sqrt(it + 1.0)
        a = a - decay * grad
        a = _proj_simplex(a)

        it += 1
        if it % 200 == 0:
            cur = _eval_score(a)
            if cur < best_big_score:
                best_big_score = cur
                best_big = a.copy()
                print(f"[final {it:04d}] improved to {best_big_score:.6f}")

    print(f"[done] best score achieved = {best_big_score:.6f} (n={n_big})")
    return best_big.tolist()
