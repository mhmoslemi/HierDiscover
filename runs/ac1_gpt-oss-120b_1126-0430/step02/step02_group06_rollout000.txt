    seq = _catalan_sequence(n).astype(float)
    seq /= seq.sum()

    # Setup schedule for beta and eta (learning rate)
    beta = 1.0
    beta_max = 30.0 (or 100)
    beta_inc_per_iter = (beta_max - beta)/(max_iters)

    eta0 = 0.5
    maybe decreasing 1/sqrt(iter+1)

    best_seq = seq.copy()
    best_score = evaluate_sequence(seq.tolist())

    it = 0
    while time.time() < deadline:
        # evaluation
        score = evaluate_sequence(seq.tolist())
        if score < best_score - 1e-12:
            best_score = score
            best_seq = seq.copy()
            print(f"[iter {it}] new best {best_score:.6f}")

        # compute self-convolution
        fft_len_self = 1 << ((2*n - 1).bit_length())
        F = np.fft.rfft(seq, fft_len_self)
        conv_self = np.fft.irfft(F*F, fft_len_self)[:2*n-1]

        # compute w via softmax with current beta
        conv_max = conv_self.max()
        # Use stable softmax
        exps = np.exp(beta * (conv_self - conv_max))
        w = exps / exps.sum()

        # compute gradient
        # preparation for conv_rev: w convolve seq_rev
        seq_rev = seq[::-1]
        fft_len_grad = 1 << (( (2*n-1) + n -1 ).bit_length())
        W = np.fft.rfft(w, fft_len_grad)
        F_rev = np.fft.rfft(seq_rev, fft_len_grad)
        conv_rev = np.fft.irfft(W * F_rev, fft_len_grad)[:3*n-2]
        grad = 2.0 * conv_rev[n-1: n-1 + n]  # shape n

        # step-size schedule
        eta = eta0 / np.sqrt(it + 1)

        # multiplicative update
        seq = seq * np.exp(-eta * grad)
        seq = np.maximum(seq, 1e-20)  # avoid zero
        seq /= seq.sum()

        # optionally increase beta gradually
        beta = min(beta_max, beta * 1.001)  # slight increment each iter maybe

        it += 1
    return best_seq.tolist()

"""
Gradient‑softmax descent for the max‑self‑convolution problem.

We minimise the quantity
    2·n·max(conv(f,f))
for a non‑negative step function f of length n (the sum of f can be
normalised to 1 because the objective is scale‑invariant).

The algorithm works on the simplex using a mirror‑descent (exponentiated
gradient) update.  The sub‑gradient of the (smooth) maximum is obtained
by a soft‑max weighting of the entries of the self‑convolution; the
resulting gradient is a convolution of those weights with the reversed
sequence, which can be evaluated efficiently with FFTs.

A continuation scheme gradually sharpens the soft‑max (β grows) and the
step‑size decays as 1/√t.  The best sequence encountered during the
budget is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _catalan_sequence(length: int) -> np.ndarray:
    """Catalan‑like (arcsine) seed: a₀=1, a_{i+1}=a_i·(2i‑1)/(2i)."""
    seq = np.empty(length, dtype=np.float64)
    a = 1.0
    seq[0] = a
    for i in range(1, length):
        a *= (2 * i - 1) / (2.0 * i)
        seq[i] = a
    return seq


def _self_convolution(seq: np.ndarray, fft_len: int) -> np.ndarray:
    """Return f∗f for a real sequence using an FFT of given length."""
    F = np.fft.rfft(seq, fft_len)
    conv = np.fft.irfft(F * F, fft_len)
    return conv[: 2 * seq.size - 1]   # keep the exact length


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Smooth approximation of the max‑operator via a soft‑max."""
    m = conv.max()
    # stability: subtract the max before exponentiating
    exp_vals = np.exp(beta * (conv - m))
    w = exp_vals / exp_vals.sum()
    return w


def _gradient(seq: np.ndarray, weight: np.ndarray,
              grad_fft_len: int) -> np.ndarray:
    """
    Gradient of the soft‑max approximation of max(conv(seq,seq))
    w.r.t. the sequence (on the simplex).

    Formula:   g_i = 2·∑_j weight_{i+j}·seq_j,
    which is a convolution of the weight vector with the reversed sequence.
    """
    n = seq.size
    # reverse seq for the cross‑correlation trick
    seq_rev = seq[::-1]
    W = np.fft.rfft(weight, grad_fft_len)
    F_rev = np.fft.rfft(seq_rev, grad_fft_len)
    conv_rev = np.fft.irfft(W * F_rev, grad_fft_len)
    conv_rev = conv_rev[: 3 * n - 2]               # exact length
    # slice to obtain g_i = 2·conv_rev[i + n‑1]
    g = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return g


def _objective_score(seq: np.ndarray) -> float:
    """Exact objective value 2·n·max(f∗f) (sum(seq) is assumed = 1)."""
    n = seq.size
    # reuse self‑convolution (FFT) for speed
    fft_len = 1 << ((2 * n - 1).bit_length())
    conv = _self_convolution(seq, fft_len)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights with a low evaluation score.

    The routine runs for ``budget_s`` seconds (default 1000).  It starts
    from the Catalan/arcsine seed, then performs a soft‑max‑based
    exponentiated‑gradient descent on the simplex, gradually sharpening
    the soft‑max (parameter β) and decaying the learning rate.
    The best sequence seen during the run is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # 1. Problem size and initialisation
    # ------------------------------------------------------------------
    n = 3000                         # a few thousand is a good trade‑off
    seq = _catalan_sequence(n).astype(np.float64)
    seq /= seq.sum()                 # normalise to the simplex

    # pre‑compute FFT lengths (constant for the whole run)
    self_fft_len = 1 << ((2 * n - 1).bit_length())
    grad_fft_len = 1 << ((3 * n - 2).bit_length())

    # ------------------------------------------------------------------
    # 2. Hyper‑parameters for the continuation scheme
    # ------------------------------------------------------------------
    beta = 0.5                       # start with a smooth approximation
    beta_max = 40.0                  # final sharpness (≈ hard max)
    beta_growth = 1.001              # multiplicative increase per iteration

    eta0 = 0.4                       # initial step‑size (will decay)

    # ------------------------------------------------------------------
    # 3. Main optimisation loop
    # ------------------------------------------------------------------
    best_seq = seq.copy()
    best_score = _objective_score(best_seq)

    it = 0
    while time.time() < deadline:
        # ----- self‑convolution and current objective -----
        conv_self = _self_convolution(seq, self_fft_len)
        cur_score = 2.0 * n * conv_self.max()

        # keep the best ever seen
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = seq.copy()
            print(f"[iter {it:6d}] new best score = {best_score:.6f}")

        # ----- soft‑max weights for the gradient -----
        w = _softmax_weights(conv_self, beta)

        # ----- gradient of the smoothed objective -----
        grad = _gradient(seq, w, grad_fft_len)

        # ----- learning‑rate schedule (1/√t) -----
        eta = eta0 / math.sqrt(it + 1)

        # ----- mirror descent (exponentiated‑gradient) update -----
        seq = seq * np.exp(-eta * grad)

        # avoid exact zeros – keep a tiny floor for numerical stability
        np.maximum(seq, 1e-18, out=seq)

        # re‑project onto the simplex
        seq_sum = seq.sum()
        if seq_sum == 0.0:                # should never happen, but be safe
            seq = np.full(n, 1.0 / n)
        else:
            seq /= seq_sum

        # ----- continuation: increase β slowly -----
        beta = min(beta_max, beta * beta_growth)

        # ----- occasional random jitter to escape flat plateaus -----
        if np.random.rand() < 0.001:
            idx = np.random.randint(n)
            seq[idx] = max(0.0, seq[idx] + np.random.randn() * 1e-3)
            seq /= seq.sum()

        it += 1

    # ------------------------------------------------------------------
    # 4. Return the best sequence found
    # ------------------------------------------------------------------
    print(f"[final] best score after {it} iterations = {best_score:.6f}")
    return best_seq.tolist()
