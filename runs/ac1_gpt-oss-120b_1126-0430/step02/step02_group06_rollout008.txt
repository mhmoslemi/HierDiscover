def _smooth_gradient_descent(seq, deadline):
    n = seq.size
    conv_len = 2*n - 1
    fft_len = 1 << ((n + conv_len - 1).bit_length())
    best_seq = seq.copy()
    best_score = _fast_evaluation(seq)
    it = 0
    # hyperparameters
    eta0 = 0.5       # initial step size maybe
    beta0 = 0.5
    beta_inc = 0.01   # per iteration
    beta_max = 50.0

    while time.time() < deadline:
        # compute conv and max using FFT (like _fast_evaluation)
        Fa = np.fft.rfft(seq, fft_len)
        conv = np.fft.irfft(Fa * Fa, fft_len)[:conv_len]
        # compute smoothed weights
        M = conv.max()
        # beta schedule
        beta = min(beta_max, beta0 + beta_inc * it)
        # compute weight vector w
        # To avoid overflow: subtract max
        exp_arg = beta * (conv - M)   # max term becomes exp(0)=1
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0:  # avoid divide by zero
            w = np.full_like(w, 1.0 / conv_len)
        else:
            w /= w_sum

        # compute gradient g = 2 * correlation(seq, w) for i=0..n-1
        # correlation via FFT
        # compute FFT of w padded to fft_len
        Fw = np.fft.rfft(w, fft_len)
        # correlation = irfft(Fa * conj(Fw))
        corr = np.fft.irfft(Fa * np.conj(Fw), fft_len)  # length = fft_len
        g = 2.0 * corr[:n]  # first n entries

        # step size schedule
        eta = eta0 / np.sqrt(it + 1)

        # multiplicative update
        seq = seq * np.exp(-eta * g)

        # renormalise to sum=1
        sum_seq = seq.sum()
        if sum_seq == 0:
            # reinitialize uniform
            seq = np.full(n, 1.0 / n)
        else:
            seq /= sum_seq

        # evaluate score
        score = _fast_evaluation(seq)
        if score < best_score - 1e-12:
            best_score = score
            best_seq = seq.copy()
            # maybe log
            if it % 200 == 0:
                print(f"[sgd] it {it} score {best_score:.6f}")
        # maybe random jitter
        if np.random.rand() < 0.001:
            idx = np.random.randint(n)
            seq[idx] = max(0.0, seq[idx] + np.random.randn() * 1e-3)
            seq /= seq.sum()
        it += 1
    return best_seq, best_score

"""
Propose a low‑score step‑function.

The algorithm consists of three stages:

1. **Initial pool** – a few simple seeds (Catalan/arcsine, uniform,
   several Beta‑like power‑law patterns and a few random vectors) are
   normalised to sum = 1 and the best of them is selected.

2. **Smoothed exponential‑gradient descent** – we minimise a smooth
   approximation of the maximum self‑convolution.
   For a sequence ``a`` we compute the convolution ``c = a*a``,
   form a soft‑max weight ``w_k ∝ exp(β·c_k)`` (β grows slowly),
   and take the gradient of the soft‑max:
   ``∇_i = 2·∑_j a_j·w_{i+j}``.  An exponentiated‑gradient update
   ``a_i ← a_i·exp(−η·∇_i)`` keeps the variables non‑negative; a final
   normalisation projects back onto the simplex.

   The step‑size η follows an inverse‑square‑root schedule and the
   smoothing parameter β is increased from a tiny value to a moderate
   constant, which gives a well‑behaved descent direction while still
   focusing on the peak of the convolution.

3. **Multiple short runs** – the descent is performed several times
   on the current best sequence as well as on fresh random restarts.
   The whole search respects the supplied time budget (default 1000 s)
   and prints intermediate scores.  The best sequence is returned as a
   plain Python list; it is already non‑negative and normalised, so the
   external ``evaluate_sequence`` function can be called safely.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution (full, length 2·n‑1) using a cached FFT length.
# ----------------------------------------------------------------------
def _full_convolution(seq: np.ndarray, fft_len: int) -> np.ndarray:
    """Return the full self‑convolution of ``seq`` (length 2·n‑1)."""
    Fa = np.fft.rfft(seq, fft_len)
    conv = np.fft.irfft(Fa * Fa, fft_len)[: 2 * seq.size - 1]
    return conv


# ----------------------------------------------------------------------
# Fast evaluation – exact replica of evaluate_sequence (without checks).
# ----------------------------------------------------------------------
def _fast_evaluation(seq: np.ndarray) -> float:
    """
    Compute the metric used by ``evaluate_sequence``.
    The sequence is assumed to be a 1‑D float array.
    """
    n = seq.size
    if n == 0:
        return np.inf
    # FFT size = next power of two ≥ 2·n‑1
    m = 1 << ((2 * n - 1).bit_length())
    Fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(Fa * Fa, m)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Catalan / arcsine seed (the classic (1‑z)^‑1/2 coefficients).
# ----------------------------------------------------------------------
def _catalan_sequence(length: int) -> np.ndarray:
    a = np.empty(length, dtype=float)
    cur = 1.0
    a[0] = cur
    for i in range(1, length):
        cur *= (2 * i - 1) / (2.0 * i)
        a[i] = cur
    return a


# ----------------------------------------------------------------------
# Power‑law (Beta‑like) family a_i ∝ (i+1)^α·(n‑i)^α, α<0.
# ----------------------------------------------------------------------
def _powerlaw_sequence(length: int, alpha: float) -> np.ndarray:
    i = np.arange(length, dtype=float)
    # shift by +1 to avoid zero‑to‑zero when α<0
    seq = ((i + 1) * (length - i)) ** alpha
    seq[seq < 0] = 0.0
    total = seq.sum()
    if total == 0.0:
        # fallback to uniform
        seq = np.full(length, 1.0 / length)
    else:
        seq /= total
    return seq


# ----------------------------------------------------------------------
# Correlation a ⋆ w (first n entries) – needed for the gradient.
# ----------------------------------------------------------------------
def _correlation_first_n(a: np.ndarray, w: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Return (a ⋆ w)[0:n] where `⋆` is the discrete cross‑correlation:
        (a ⋆ w)[ℓ] = Σ_i a[i]·w[i+ℓ] .
    Both inputs are zero‑padded to ``fft_len`` before the transform.
    """
    Fa = np.fft.rfft(a, fft_len)
    Fw = np.fft.rfft(w, fft_len)
    corr = np.fft.irfft(Fa * np.conj(Fw), fft_len)
    return corr[: a.size]


# ----------------------------------------------------------------------
# Smoothed exponential‑gradient descent on the simplex.
# ----------------------------------------------------------------------
def _smooth_exponential_gradient(seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    minimise  log‑sum‑exp of the self‑convolution.
    Returns the best sequence seen (as ndarray) and its score.
    """
    n = seq.size
    conv_len = 2 * n - 1
    # FFT length that can hold a (n) and w (2n‑1) simultaneously
    fft_len = 1 << ((n + conv_len - 1).bit_length())

    best_seq = seq.copy()
    best_score = _fast_evaluation(best_seq)

    # hyper‑parameters
    eta0 = 0.6          # initial step size
    beta0 = 0.5         # start smoothing
    beta_inc = 0.005    # how fast we increase the smoothing
    beta_max = 30.0

    it = 0
    while time.time() < deadline:
        # 1) convolution of the current sequence
        Fa = np.fft.rfft(seq, fft_len)
        conv = np.fft.irfft(Fa * Fa, fft_len)[:conv_len]

        # 2) soft‑max weights over the convolution entries
        M = conv.max()
        beta = min(beta_max, beta0 + beta_inc * it)
        # subtract the maximum for numerical stability
        w = np.exp(beta * (conv - M))
        w_sum = w.sum()
        if w_sum == 0.0:                     # should never happen, but be safe
            w = np.full_like(w, 1.0 / conv_len)
        else:
            w /= w_sum

        # 3) gradient of the smooth max (2·a ⋆ w)
        grad = 2.0 * _correlation_first_n(seq, w, fft_len)

        # 4) step‑size schedule
        eta = eta0 / np.sqrt(it + 1.0)

        # 5) exponentiated‑gradient step and projection onto the simplex
        seq = seq * np.exp(-eta * grad)
        s = seq.sum()
        if s == 0.0 or not np.isfinite(s):
            # re‑initialise if something went wrong
            seq = np.full(n, 1.0 / n)
        else:
            seq /= s

        # optional tiny random jitter to avoid getting stuck on exact plateaux
        if np.random.rand() < 0.001:
            idx = np.random.randint(n)
            seq[idx] = max(0.0, seq[idx] + np.random.randn() * 1e-4)
            seq /= seq.sum()

        # 6) evaluate
        score = _fast_evaluation(seq)
        if score < best_score - 1e-12:
            best_score = score
            best_seq = seq.copy()
            if it % 200 == 0:
                print(f"[sgd] iter {it:6d}   best score {best_score:.6f}")

        it += 1

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Return a list of heights (non‑negative, sum>0) that aims to minimise
    ``evaluate_sequence``.  The routine works inside the given time budget
    (default 1000 s) and prints intermediate diagnostics.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # a small safety margin

    # Length of the step function – 8000 gives a good trade‑off.
    n = 8000

    # ------------------------------------------------------------------
    # Build an initial pool of candidates.
    # ------------------------------------------------------------------
    candidate_pool = []

    # 1) Catalan (arcsine) seed
    cand = _catalan_sequence(n).astype(float)
    cand /= cand.sum()
    candidate_pool.append(cand)

    # 2) Uniform distribution
    candidate_pool.append(np.full(n, 1.0 / n))

    # 3) Power‑law (Beta‑like) seeds for a few alphas
    alphas = np.linspace(-0.8, -0.3, 7)          # from more endpoint‑heavy to flatter
    for a in alphas:
        candidate_pool.append(_powerlaw_sequence(n, a))

    # 4) Random points (few)
    for _ in range(4):
        r = np.random.rand(n)
        r /= r.sum()
        candidate_pool.append(r)

    # Optional: include the globally provided reference if present
    try:
        # ``height_sequence_1`` is a global defined by the harness
        if isinstance(height_sequence_1, list) and len(height_sequence_1) > 0:
            arr = np.array(height_sequence_1, dtype=float)
            if arr.size >= n:
                # truncate / pad
                arr = arr[:n]
            else:
                pad = np.zeros(n - arr.size)
                arr = np.concatenate([arr, pad])
            arr = np.maximum(arr, 0.0)
            if arr.sum() == 0.0:
                arr = np.full(n, 1.0 / n)
            else:
                arr /= arr.sum()
            candidate_pool.append(arr)
    except NameError:
        pass

    # ------------------------------------------------------------------
    # Pick the best among the raw seeds.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    for seq in candidate_pool:
        sc = _fast_evaluation(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
    print(f"[init] best raw score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Main optimisation loop – we keep launching short gradient runs.
    # ------------------------------------------------------------------
    # Give roughly 85 % of the remaining budget to the optimiser.
    optimise_deadline = deadline
    runs_without_improvement = 0

    while time.time() < optimise_deadline:
        # each run gets a few seconds (or the remainder if we are close)
        now = time.time()
        run_time = min(30.0, optimise_deadline - now)   # at most 30 s per run
        if run_time <= 0.0:
            break

        # start from the current best (occasionally also from a fresh random start)
        start_seq = best_seq.copy()
        if runs_without_improvement >= 4:         # try a random restart after a while
            start_seq = np.random.rand(n)
            start_seq /= start_seq.sum()
            runs_without_improvement = 0

        refined_seq, refined_score = _smooth_exponential_gradient(start_seq, now + run_time)

        if refined_score < best_score - 1e-12:
            best_score = refined_score
            best_seq = refined_seq.copy()
            runs_without_improvement = 0
            print(f"[opt] new global best = {best_score:.6f}")
        else:
            runs_without_improvement += 1

    final_score = _fast_evaluation(best_seq)
    print(f"[final] best score after optimisation = {final_score:.6f}")

    # Return a plain Python list, as required by the harness.
    return best_seq.tolist()
