#!/usr/bin/env python
"""
Propose a step‑function (a list of non‑negative heights) with a low
evaluation score.

Algorithm
---------
1. Choose a length n (default 20000 – still cheap for the exact
   evaluation).  If a previous solution ``height_sequence_1`` exists it
   is up‑sampled and used as an additional warm‑start.
2. Scan a grid of power‑law exponents p and also a few mixtures with
   the uniform distribution.  The cheapest sequence (according to the
   cheap FFT based evaluation) seeds the optimisation.
3. Optimise the shape with Adam on a *smooth* approximation of the
   max‑convolution:
   * The convolution a∗a is computed with an FFT (O(n log n)).
   * A soft‑max (with a temperature λ that grows during the run) turns
     the max into a smooth function.
   * The gradient of this smooth objective can be obtained by a second
     convolution (again via FFT).
   * After the Adam step the vector is projected onto the probability
     simplex (Σa = 1, a_i ≥ 0).
4. Periodically record the best cheap score and keep the associated
   vector.  When the time budget is almost exhausted a final exact
   evaluation with ``evaluate_sequence`` is performed and the best
   sequence is returned.
"""

import time
import numpy as np
from math import sqrt

# ----------------------------------------------------------------------
# Helper functions – all defined at top level (no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (sum=1, non‑negative)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float) -> np.ndarray:
    """Return the normalised power‑law vector a_i ∝ (i+1)^‑p (sum = 1)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    seq = idx ** (-p)
    seq /= seq.sum()
    return seq


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using FFT (real output)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa).real[: 2 * a.size - 1]
    return conv
