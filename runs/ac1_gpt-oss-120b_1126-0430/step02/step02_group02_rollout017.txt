#!/usr/bin/env python
"""
Propose a non‑negative step‑function (list of heights) that minimizes

    2·n·max_k (a*a)[k] / (∑a_i)²

where a*a is the linear convolution of the vector with itself.
The algorithm consists of three phases:

1. **Coarse parametric search** – families
   (i+γ)⁻ᵖ and (i+1)⁻ᵖ·e^{-λi} are generated for many
   (p,γ,λ).  For each candidate the cheap FFT‑based score
   2·n·max(conv) is evaluated and the best vector is kept.
2. **Local refinement** – a projected sub‑gradient descent
   (only the active convolution index is used) starting
   from the best parametric candidate.  The step‑size follows
   η₀/√t and the iterate is always projected onto the
   probability simplex.
3. **Exact evaluation** – the best vector found is finally
   passed to the provided ``evaluate_sequence`` function.

The whole routine respects the 1000 s budget, prints occasional
progress information and returns the best sequence it discovered.
"""

import time
import numpy as np
from math import sqrt

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (sum=1, non‑negative)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _convolution_max(a: np.ndarray, fft_len: int) -> float:
    """Maximum entry of the linear convolution a*a using a real FFT."""
    # Use rfft/irfft which are a bit faster for real data.
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    # Only the first 2*len(a)-1 entries correspond to the linear convolution.
    conv = conv[: 2 * a.size - 1]
    return float(conv.max())


def _ratio_from_max(n: int, max_conv: float) -> float:
    """Exact evaluation ratio for a vector normalised to sum=1."""
    return 2.0 * n * max_conv


def _subgradient_max_convolution(a: np.ndarray, max_idx: int) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)[k] at the active index ``max_idx``.
    For the active shift k we have
        ∂/∂a_i (a*a)[k] = 2·a_{k-i}
    whenever the index is valid.
    """
    n = a.shape[0]
    g = np.zeros_like(a)

    lo = max(0, max_idx - (n - 1))
    hi = min(max_idx, n - 1)

    i = np.arange(lo, hi + 1, dtype=np.int64)
    g[i] = 2.0 * a[max_idx - i]
    return g


def _seq_powerlaw_offset(n: int, p: float, gamma: float) -> np.ndarray:
    """a_i ∝ (i+1+γ)⁻ᵖ, then normalised to sum = 1."""
    idx = np.arange(n, dtype=np.float64) + 1.0 + gamma
    seq = np.power(idx, -p)
    seq /= seq.sum()
    return seq


def _seq_powerlaw_exponential(n: int, p: float, lam: float) -> np.ndarray:
    """a_i ∝ (i+1)⁻ᵖ · e^{-λ i}, then normalised."""
    i = np.arange(n, dtype=np.float64)
    seq = np.power(i + 1.0, -p) * np.exp(-lam * i)
    seq /= seq.sum()
    return seq


def _seq_combined(n: int, p: float, gamma: float, lam: float) -> np.ndarray:
    """a_i ∝ (i+1+γ)⁻ᵖ · e^{-λ i}, then normalised."""
    idx = np.arange(n, dtype=np.float64) + 1.0 + gamma
    seq = np.power(idx, -p) * np.exp(-lam * np.arange(n, dtype=np.float64))
    seq /= seq.sum()
    return seq


def _cheap_score(seq: np.ndarray, n: int, fft_len: int) -> float:
    """Return the cheap objective 2·n·max(conv) (sum of seq is 1)."""
    maxb = _convolution_max(seq, fft_len)
    return _ratio_from_max(n, maxb)

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) beats the target
    bound 1.503.  The routine respects the given time budget.
    """
    # ------------------------------------------------------------------
    # 0. Basic set‑up
    # ------------------------------------------------------------------
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 4.0          # keep a small safety margin

    # ------------------------------------------------------------------
    # 1. Choose a working length.
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)  # type: ignore[name-defined]
        if n < 3000:
            n = 3000
    except Exception:
        n = 20000

    fft_len = 1 << ((2 * n - 1).bit_length())   # next power of two

    # ------------------------------------------------------------------
    # 2. Initialise with any provided sequence (if any)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    try:
        seq0 = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if seq0.sum() > 0:
            seq0 = seq0 / seq0.sum()
            sc0 = _cheap_score(seq0, n, fft_len)
            best_seq = seq0.copy()
            best_score = sc0
            print(f"Initial sequence from height_sequence_1: score = {sc0:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Coarse parametric search (offset power‑law)
    # ------------------------------------------------------------------
    phase_deadline = time.time() + min(120.0, (deadline - time.time()) * 0.25)

    p_grid = np.linspace(0.55, 0.70, 31)          # step ≈0.005
    gamma_grid = np.arange(0.0, 51.0, 5.0)        # 0,5,…,50

    for p in p_grid:
        if time.time() >= phase_deadline:
            break
        for gamma in gamma_grid:
            seq = _seq_powerlaw_offset(n, p, gamma)
            sc = _cheap_score(seq, n, fft_len)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                best_p, best_gamma = p, gamma
                print(f"[Coarse] new best: p={p:.4f} gamma={gamma:.1f} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 4. Refine around the best (p,γ) found above
    # ------------------------------------------------------------------
    if best_seq is not None:
        phase_deadline = time.time() + min(120.0, (deadline - time.time()) * 0.20)

        p_fine = np.linspace(max(0.45, best_p - 0.02),
                             min(0.80, best_p + 0.02), 41)
        gamma_fine = np.arange(max(0.0, best_gamma - 5.0),
                               best_gamma + 5.5, 0.5)

        for p in p_fine:
            if time.time() >= phase_deadline:
                break
            for gamma in gamma_fine:
                seq = _seq_powerlaw_offset(n, p, gamma)
                sc = _cheap_score(seq, n, fft_len)
                if sc < best_score:
                    best_score = sc
                    best_seq = seq.copy()
                    best_p, best_gamma = p, gamma
                    print(f"[Fine] new best: p={p:.5f} gamma={gamma:.2f} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 5. Exponential‑decay family (adds an extra degree of freedom)
    # ------------------------------------------------------------------
    phase_deadline = time.time() + min(120.0, (deadline - time.time()) * 0.15)

    lam_grid = np.linspace(0.0, 0.0015, 31)     # up to 1.5·10⁻³
    for p in np.linspace(0.55, 0.70, 16):
        if time.time() >= phase_deadline:
            break
        for lam in lam_grid:
            seq = _seq_powerlaw_exponential(n, p, lam)
            sc = _cheap_score(seq, n, fft_len)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                best_p, best_lam = p, lam
                print(f"[Exp] new best: p={p:.4f} lam={lam:.6f} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 6. Combined offset + exponential family (small local search)
    # ------------------------------------------------------------------
    if best_seq is not None:
        phase_deadline = time.time() + min(120.0, (deadline - time.time()) * 0.15)

        # narrow neighbourhood around the best parameters obtained so far
        p_center = best_p if 'best_p' in locals() else 0.60
        gamma_center = best_gamma if 'best_gamma' in locals() else 0.0
        lam_center = best_lam if 'best_lam' in locals() else 0.0

        p_grid2 = np.linspace(max(0.45, p_center - 0.01),
                              min(0.80, p_center + 0.01), 21)
        gamma_grid2 = np.arange(max(0.0, gamma_center - 2.0),
                                gamma_center + 2.5, 0.25)
        lam_grid2 = np.linspace(max(0.0, lam_center - 0.0005),
                                lam_center + 0.00055, 21)

        for p in p_grid2:
            if time.time() >= phase_deadline:
                break
            for gamma in gamma_grid2:
                for lam in lam_grid2:
                    seq = _seq_combined(n, p, gamma, lam)
                    sc = _cheap_score(seq, n, fft_len)
                    if sc < best_score:
                        best_score = sc
                        best_seq = seq.copy()
                        best_p, best_gamma, best_lam = p, gamma, lam
                        print(f"[Comb] new best: p={p:.5f} g={gamma:.2f} lam={lam:.6f} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 7. Random local perturbations (cheap hill‑climbing)
    # ------------------------------------------------------------------
    phase_deadline = time.time() + min(60.0, (deadline - time.time()) * 0.10)

    if best_seq is not None:
        a = best_seq.copy()
        for _ in range(2000):
            if time.time() >= phase_deadline:
                break
            # small random shake, keep positivity
            perturb = 0.01 * np.random.rand(n)
            cand = _simplex_projection(a + perturb)
            sc = _cheap_score(cand, n, fft_len)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                a = cand
                print(f"[Perturb] improved score = {sc:.6f}")

    # ------------------------------------------------------------------
    # 8. Projected sub‑gradient descent starting from the best candidate
    # ------------------------------------------------------------------
    if best_seq is None:
        # Fallback – uniform distribution (should never happen)
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _cheap_score(best_seq, n, fft_len)

    a = best_seq.copy()
    best_iter_seq = a.copy()
    best_iter_score = best_score

    iter_cnt = 0
    eta0 = 0.8                     # initial step‑size factor (tuned empirically)

    while time.time() < deadline and iter_cnt < 800_000:
        iter_cnt += 1

        # --- convolution and active index ---------------------------------
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
        maxb = float(conv.max())
        max_idx = int(conv.argmax())

        # --- sub‑gradient -------------------------------------------------
        g = _subgradient_max_convolution(a, max_idx)

        # --- step size (diminishing) --------------------------------------
        eta = eta0 / sqrt(iter_cnt)

        # --- gradient step + simplex projection ----------------------------
        a_candidate = a - eta * g
        a = _simplex_projection(a_candidate)

        # --- occasional evaluation & random shake -------------------------
        if iter_cnt % 500 == 0 or time.time() > deadline - 5:
            sc = 2.0 * n * maxb
            if sc < best_iter_score:
                best_iter_score = sc
                best_iter_seq = a.copy()
                print(f"[Iter {iter_cnt:07d}] new best (cheap) score = {sc:.6f}")

            # small random perturbation to escape plateaus
            if np.random.rand() < 0.12:
                perturb = 0.02 * np.random.rand(n)
                a = _simplex_projection(a + perturb)

    # ------------------------------------------------------------------
    # 9. Final exact evaluation (uses the provided ``evaluate_sequence``)
    # ------------------------------------------------------------------
    final_seq = best_iter_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:      # defensive – should never happen
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # ------------------------------------------------------------------
    # 10. Return the best sequence found.
    # ------------------------------------------------------------------
    return final_seq
