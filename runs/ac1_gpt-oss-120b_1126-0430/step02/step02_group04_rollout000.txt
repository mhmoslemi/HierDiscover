"""
Propose a step‑function (non‑negative heights) that minimises
evaluate_sequence.  The algorithm works as follows:

1.  Choose a moderate length (n = 8000) that still allows fast FFT‑based
    convolutions.
2.  Build a small pool of diverse initial points:
      – a Catalan‑type vector,
      – several geometric decays,
      – several symmetric Beta‑distribution shapes (α = β < 1) which put
        most mass near the ends of the interval,
      – a random point on the simplex.
   The best of these (according to the exact objective) is kept.
3.  Run a projected Adam optimiser on the *smooth‑max* surrogate of the
    true objective.  For the current vector a we compute the self‑convolution
    b = a∗a, build soft‑max weights
        w_k ∝ exp(α·(b_k – max(b))),
    and use the gradient of Σ_k w_k·b_k,
        ∇a = 2·(a ⊛ w_rev)                (⊛ = linear convolution,
                                            w_rev = w reversed).
    After each Adam step the iterate is projected back onto the probability
    simplex (non‑negative, sum = 1).
4.  The temperature α is increased slowly so that the optimisation
    gradually concentrates on the true maximum.  The learning rate is
    adapted automatically by Adam; occasional reductions are applied when
    no improvement is seen.
5.  If the search stalls for many iterations a random perturbation around
    the best point is performed (restart).
6.  The loop runs until the time budget (≈ 1000 s) is exhausted; the best
    sequence found is returned as a plain Python list.

The objective is scale‑invariant, therefore the simplex constraint (∑a_i = 1)
is enforced throughout.  All heavy work (convolution, projection) is
implemented with NumPy and FFTs, giving an O(n log n) per‑iteration cost.

# ----------------------------------------------------------------------
# Linear‑algebra utilities

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, sum=1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    # Duchi et al. (2008) – O(n log n) due to sorting.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u > cssv / np.arange(1, n + 1))[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution of real vectors x and y using real FFTs.
    Returns a vector of length len(x) + len(y) – 1.
    """
    n = x.size + y.size - 1
    size = 1 << (n - 1).bit_length()            # next power of two
    X = np.fft.rfft(x, size)
    Y = np.fft.rfft(y, size)
    Z = X * Y
    conv = np.fft.irfft(Z, size)[:n]
    return conv.real


def _eval_score(a: np.ndarray) -> float:
    """Exact value of evaluate_sequence for a non‑negative simplex vector a."""
    n = a.size
    b = _convolve_fft(a, a)
    max_b = b.max()
    # sum(a) == 1 because we keep a on the simplex
    return float(2.0 * n * max_b)

# Candidate generators
# ----------------------------------------------------------------------
def _catalan_sequence(n: int) -> np.ndarray:
    """Catalan‑like decreasing sequence, normalised to sum = 1."""
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * (i - 1) + 1) / (2.0 * i)
    a /= a.sum()
    return a


def _geometric_sequence(n: int, rate: float) -> np.ndarray:
    """Geometric decay a_i = C·r^i, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    a = rate ** i
    a /= a.sum()
    return a


def _beta_shape(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density sampled at the mid‑points of n bins,
    normalised to sum = 1.  For α < 1 the mass concentrates near the edges.
    """
    x = (np.arange(n, dtype=float) + 0.5) / n          # points in (0,1)
    # log‑space formulation avoids overflow for small α
    log_f = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    f = np.exp(log_f)
    # Numerical safety: replace NaNs (possible at 0/1) by a large number
    f = np.where(np.isfinite(f), f, 0.0)
    f /= f.sum()
    return f

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # --- projection back onto the simplex ----------------------------------
        a_candidate = _proj_simplex(a - step)
        cand_score = _eval_score(a_candidate)

        # --- accept / reject ----------------------------------------------------
        if cand_score < cur_score:
            a = a_candidate
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{t:06d}] new best {best_score:.6f} "
                      f"(lr={lr:.5f}, α={alpha_temp:.2f})")
                no_improve = 0
            else:
                no_improve += 1
        else:
            # no improvement – damp the learning rate a little
            lr *= 0.93
            no_improve += 1

        # occasional learning‑rate decay to keep steps sensible
        if t % 500 == 0:
            lr = max(lr * 0.9, 1e-5)

        # --- restart if we are stuck for a long time ----------------------------
        if no_improve > 4000:
            # perturb the best solution and restart the Adam state
            perturb = 0.02 * np.random.randn(n)
            a = best_seq + perturb
            a = np.maximum(a, 0.0)
            a = _proj_simplex(a)
            cur_score = _eval_score(a)
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            lr = 0.12
            alpha_temp = 10.0
            no_improve = 0
            print(f"[restart] score={cur_score:.6f}")

    print(f"[done] best score found = {best_score:.6f}")
    # Return as plain Python list – the surrounding harness expects this format.
    return best_seq.tolist()
