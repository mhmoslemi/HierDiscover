```python
"""
Exponentiated‑gradient optimisation of a non‑negative step‑function
to minimise 2·n·max(a*a)/(sum(a)^2).  Starts from the Catalan
(arcsine) seed, then repeatedly:
  * compute the self‑convolution,
  * turn the convolution values into a soft‑max probability vector,
  * compute the gradient of the smoothed max w.r.t. the coefficients,
  * perform an exponentiated‑gradient step and project back onto the
    probability simplex.
The loop runs until the time budget (≈1000 s) is exhausted and the
best sequence seen is returned.
"""

import time
import numpy as np

def _fast_evaluation(seq: np.ndarray) -> float:
    """Fast version of evaluate_sequence (no security checks)."""
    n = seq.size
    if n == 0:
        return float("inf")
    # FFT length: next power of two >= 2*n-1
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))

def _self_convolution(a: np.ndarray) -> np.ndarray:
    """Return a*a (full convolution) using FFT."""
    n = a.size
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    return conv

def _correlation(p: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute g_i = Σ_k p_k a_{k-i} for i=0..len(a)-1.
    This is the (full) correlation of p with a.
    """
    n = a.size
    pl = p.size
    m = 1 << ((pl + n - 1).bit_length())
    fp = np.fft.rfft(p, m)
    fa_rev = np.fft.rfft(a[::-1], m)
    conv = np.fft.irfft(fp * fa_rev, m)[: pl + n - 1]
    start = pl - 1
    return conv[start:start + n]

def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x ≥ 0, sum(x)=1}.
    Implementation of the algorithm by Duchi et al. (2008).
    """
    if v.ndim != 1:
        raise ValueError("v must be a 1‑dimensional array")
    n = v.shape[0]
    # sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    # find the rho where u_i - (cssv_i / i) > 0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # everything is zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def _catalan_sequence(length: int) -> np.ndarray:
    """Return the Catalan / arcsine coefficients of given length."""
    seq = np.empty(length, dtype=float)
    a = 1.0
    seq[0] = a
    for i in range(1, length):
        a *= (2 * i - 1) / (2.0 * i)   # a_i = a_{i-1} * (2i-1)/(2i)
        seq[i] = a
    return seq

def _optimize_exponential_gradient(a: np.ndarray, deadline: float) -> np.ndarray:
    """
    Exponentiated‑gradient descent on the smoothed max convolution.
    Returns the best sequence encountered before the deadline.
    """
    n = a.size
    best_seq = a.copy()
    best_score = _fast_evaluation(best_seq)
    # hyper‑parameters (tuned empirically)
    eta = 1.0                # learning‑rate for exponentiated update
    mu = 5.0                 # initial inverse temperature for soft‑max
    mu_max = 30.0
    mu_inc = (mu_max - mu) / 2000.0   # reach mu_max after ~2000 iterations
    it = 0
    # main loop
    while time.time() < deadline:

"""
Exponentiated‑gradient optimisation of a step‑function.

The evaluation metric is
    E(a) = 2·n·max(a*a) / (sum(a)^2)

because the metric is scale‑invariant we can work with a probability
vector (sum = 1).  Starting from the Catalan (arcsine) seed we repeatedly

  * compute the self‑convolution,
  * turn the convolution values into a soft‑max distribution,
  * compute the gradient of the smoothed max w.r.t. the coefficients,
  * perform an exponentiated‑gradient update and project back onto the
    simplex.

The temperature of the soft‑max is slowly increased so that the
algorithm focuses more on the true maximum as the optimisation proceeds.
The best sequence seen before the time budget expires is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast version of the provided evaluate_sequence (no security checks)
# ----------------------------------------------------------------------
def _fast_evaluation(seq: np.ndarray) -> float:
    n = seq.size
    if n == 0:
        return float("inf")
    m = 1 << ((2 * n - 1).bit_length())          # next power of two ≥ 2·n‑1
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Self‑convolution via FFT (full length 2·n‑1)
# ----------------------------------------------------------------------
def _self_convolution(a: np.ndarray) -> np.ndarray:
    n = a.size
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Correlation g_i = Σ_k p_k a_{k‑i}   (i = 0 … n‑1)
# ----------------------------------------------------------------------
def _correlation(p: np.ndarray, a: np.ndarray) -> np.ndarray:
    n = a.size
    pl = p.size
    m = 1 << ((pl + n - 1).bit_length())
    fp = np.fft.rfft(p, m)
    fa_rev = np.fft.rfft(a[::-1], m)
    conv = np.fft.irfft(fp * fa_rev, m)[: pl + n - 1]
    start = pl - 1
    return conv[start:start + n]


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {x≥0, Σx = 1}
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex using the algorithm of Duchi et al."""
    if v.ndim != 1:
        raise ValueError("Input must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]               # descending sort
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):               # all entries are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Catalan (arcsine) coefficients, un‑normalised
# ----------------------------------------------------------------------
def _catalan_sequence(length: int) -> np.ndarray:
    seq = np.empty(length, dtype=float)
    a = 1.0
    seq[0] = a
    for i in range(1, length):
        a *= (2.0 * i - 1.0) / (2.0 * i)     # a_i = a_{i‑1}·(2i‑1)/(2i)
        seq[i] = a
    return seq


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser
# ----------------------------------------------------------------------
def _optimize_exponential_gradient(a: np.ndarray, deadline: float) -> np.ndarray:
    """
    Perform exponentiated‑gradient descent on the smoothed max of the
    self‑convolution.  Returns the best sequence observed before the deadline.
    """
    n = a.size
    best_seq = a.copy()
    best_score = _fast_evaluation(best_seq)

    # hyper‑parameters (tuned for speed / progress)
    eta = 1.0                # learning‑rate for the exponentiated step
    mu = 5.0                 # inverse temperature for the soft‑max
    mu_max = 50.0
    mu_inc = (mu_max - mu) / 2000.0   # raise mu gradually
    it = 0

    while time.time() < deadline:
        # ---- self‑convolution and current score ----
        b = _self_convolution(a)
        max_b = b.max()
        cur_score = 2.0 * n * max_b          # sum(a) == 1 after projection

        # ---- soft‑max over convolution entries ----
        max_val = max_b
        exp_terms = np.exp(mu * (b - max_val))   # safe: max term => 1
        p = exp_terms / exp_terms.sum()          # probabilities, sum to 1

        # ---- gradient of the smooth maximum ----
        g = _correlation(p, a)   # shape (n,)

        # ---- exponentiated‑gradient update + projection ----
        a_new = a * np.exp(-eta * g)
        a_new = _simplex_projection(a_new)

        # ---- evaluate and keep the best ever seen ----
        new_score = _fast_evaluation(a_new)
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_seq = a_new.copy()
            print(f"Iter {it:5d}: improved score = {best_score:.8f}")

        # ---- advance the iterate (always move forward) ----
        a = a_new

        # ---- schedule adjustments ----
        mu = min(mu_max, mu + mu_inc)   # increase focus on the true maximum
        if it % 500 == 0 and it > 0:
            eta *= 0.9                # slowly decay the step size

        # occasional tiny jitter to escape possible plateaus
        if np.random.rand() < 0.005:
            idx = np.random.randint(n)
            eps = (np.random.rand() - 0.5) * 1e-3
            a[idx] = max(0.0, a[idx] + eps)
            a = _simplex_projection(a)

        it += 1

    return best_seq


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Produce a non‑negative height sequence that (hopefully) beats the
    classic π/2 ≈ 1.571 bound.  The optimiser runs until the allotted
    time budget is exhausted and the best sequence found is returned.
    """
    np.random.seed(seed)

    # give a small safety margin for the harness
    deadline = time.time() + budget_s - 5.0

    # length – a few thousand gives a good trade‑off between quality and speed
    n = 5000

    # 1. Catalan seed, normalised to a probability vector (sum = 1)
    seq = _catalan_sequence(n)
    seq = seq / seq.sum()

    # 2. Optimise with exponentiated‑gradient descent
    best_seq = _optimize_exponential_gradient(seq, deadline)

    # 3. Return as a plain Python list
    return best_seq.tolist()
