"""
propose_candidate
-----------------
Generate a very long step‑function based on the Catalan (central‑binomial)
distribution and then improve it with a fast projected sub‑gradient method.
The sub‑gradient is taken w.r.t. the maximal entry of the linear
self‑convolution (which is exactly the quantity evaluated by
``evaluate_sequence``).  Momentum and a simple learning‑rate schedule allow
thousands of cheap FFT‑based updates within the 1000 s time budget, usually
yielding a modest but consistent improvement over the plain Catalan sequence.
The returned list is normalised to unit sum, which is the natural scale for the
objective (the score equals ``2·n·max(conv)`` in that case).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast linear convolution using a real FFT (full linear mode)
# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    size = 1 << (2 * n - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Catalan‑type coefficients, normalised to unit L¹‑norm
# ----------------------------------------------------------------------
def _catalan_normalised(n: int) -> np.ndarray:
    """First n Catalan‑type numbers a_i = C(2i,i)/4**i,
    normalised so that sum_i a_i = 1.
    Recurrence a_i = a_{i‑1}·(2i‑1)/(2i) is numerically stable.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Project a vector onto the (non‑negative) simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Clamp to non‑negative and renormalise to sum to 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 0.0:
        v /= s
    else:
        # fall back to a uniform point if everything became zero
        v[:] = 1.0 / v.size
    return v


# ----------------------------------------------------------------------
# Heavy‑duty projected sub‑gradient optimiser
# ----------------------------------------------------------------------
def _optimise_subgradient(a: np.ndarray, deadline: float) -> np.ndarray:
    """Iteratively minimise ``max(conv)`` with a momentum‑based sub‑gradient.
    The routine runs until the time limit is reached and returns the best
    candidate found.
    """
    n = a.size
    best = a.copy()
    best_score = 2.0 * n * np.max(_fft_convolve(a))

    # hyper‑parameters
    lr = 0.03                     # initial learning rate
    beta = 0.9                    # momentum coefficient
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # current convolution and its worst index
        conv = _fft_convolve(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient of the active convolution entry
        grad = np.zeros_like(a)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        # derivative of (a*a)_k w.r.t. a_i is 2·a_{k‑i}
        grad[idx] = 2.0 * a[k_max - idx]

        # momentum update
        momentum = beta * momentum - lr * grad
        a_new = a + momentum
        a_new = _project_simplex(a_new)

        # evaluate the new candidate (fast via FFT)
        new_max = np.max(_fft_convolve(a_new))
        new_score = 2.0 * n * new_max

        if new_score < best_score:
            best_score = new_score
            best = a_new.copy()
            # a small gain → slightly increase step‑size to stay aggressive
            lr *= 1.02
        else:
            # no gain → gently shrink the step‑size
            lr *= 0.98

        # keep learning‑rate within reasonable bounds
        if lr < 1e-7:
            lr = 1e-7
        if lr > 0.5:
            lr = 0.5

        a = a_new

        # occasional progress report (avoids too much output)
        if it % 500 == 0:
            print(f"[subgrad] iter {it:05d}  score {best_score:.9f}  lr {lr:.2e}")

    return best


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Return a sequence of non‑negative heights that (approximately) minimises
    ``evaluate_sequence``.  The implementation builds a large Catalan
    distribution, refines it with thousands of projected sub‑gradient steps,
    and finally returns the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # a safety margin for the harness

    # Choose a length that is big enough for the asymptotic regime but still
    # cheap to work with.  30 000 gives a good trade‑off (≈3 × 10⁻⁵ per
    # convolution entry, sub‑gradient steps are <0.01 s each).
    n = 30000

    # 1) Initialise with the Catalan distribution (near‑optimal analytically)
    a = _catalan_normalised(n)

    # 2) Heavily optimise the maximal convolution entry using sub‑gradients.
    #    The loop runs for as long as the time budget permits.
    if time.time() < deadline - 5.0:
        a = _optimise_subgradient(a, deadline)

    # 3) Final sanity check (the score printed is the same quantity that
    #    evaluate_sequence returns for a unit‑sum sequence).
    final_score = 2.0 * n * np.max(_fft_convolve(a))
    print(f"[final] n={n}, tentative score ≈ {final_score:.9f}")

    return a.tolist()
