"""
Optimiser for the step‑function evaluation task.

The score is   2 * n * max(conv(a,a)) / (sum(a))**2 .
Because the metric is scale‑invariant we can fix  sum(a)=1  and simply
minimise  2·n·max(conv).

The algorithm proceeds in three stages:

1.  **Exploratory grid search** – we test a family of Beta‑type
    distributions  a_i ∝ (i+½)^{α‑1}·(n‑i‑½)^{β‑1}  with several (α,β)
    pairs.  These shapes include the Catalan distribution
    (α≈½,β≈1) and the symmetric arcsine law (α=β≈½).  The best
    candidate from this cheap scan is kept.

2.  **Linear‑programming boost** – given the current sequence `f`,
    we solve a linear programme that finds a non‑negative vector `g`
    with maximal ℓ₁‑mass under the constraint  f∗g ≤ M  (where
    M is the present maximal self‑convolution).  Adding mass in this
    fashion can only lower the final score after normalisation.

3.  **Smooth max‑gradient descent** – finally we refine the best
    sequence by gradient descent on a smooth approximation of the
    max‑operator (soft‑max with an increasing temperature).  The
    gradient of  softmax(conv)  w.r.t. the coefficients is a simple
    cross‑correlation that can be evaluated with FFTs.  An Adam‑style
    update together with projection onto the probability simplex yields
    rapid convergence.

All heavy work uses real FFTs (O(n log n)) and the whole search is
limited by the supplied time budget (default 1000 s).  The function
`propose_candidate` complies with the required signature and returns
the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities – full linear convolution using a real FFT
# ----------------------------------------------------------------------
def _fft_convolve_full(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a * b using a real FFT.
    The result length is len(a)+len(b)-1.
    """
    n = a.shape[0] + b.shape[0] - 1
    # size = next power‑of‑two for speed
    size = 1 << (n - 1).bit_length()
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    conv = np.fft.irfft(fa * fb, size)
    return conv[:n]


def _fft_convolve(a: np.ndarray) -> np.ndarray:
    """Convenience wrapper for a * a (self‑convolution)."""
    return _fft_convolve_full(a, a)


# ----------------------------------------------------------------------
# Families of candidate shapes
# ----------------------------------------------------------------------
def _catalan_normalised(n: int) -> np.ndarray:
    """Catalan‑type coefficients a_i = C(2i,i)/4**i, normalised to L1‑norm 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)   # recurrence a_i = a_{i-1}*(2i-1)/(2i)
    a /= a.sum()
    return a


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric Beta‑type density on {0,…,n‑1}.
    a_i ∝ (i+½)^{α‑1}·(n‑i‑½)^{β‑1}   and normalised to unit sum.
    """
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n                 # lies in (0,1)
    # avoid NaNs when alpha or beta is 1 (exponent 0)
    with np.errstate(divide='ignore', invalid='ignore'):
        a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.where(np.isfinite(a), a, 0.0)   # replace possible inf/NaN by 0
    total = a.sum()
    if total == 0.0:                     # fall back to uniform if something went wrong
        a[:] = 1.0 / n
    else:
        a /= total
    return a


# ----------------------------------------------------------------------
# Linear‑programming step (uses SciPy if available)
# ----------------------------------------------------------------------
def _lp_step(f: np.ndarray, M: float) -> np.ndarray | None:
    """Given a probability vector f (sum=1) and its current bound M,
    solve  max Σ g_i   s.t.  g≥0  and (f∗g)_k ≤ M.
    Returns the optimiser g (not normalised) or None if the LP fails.
    """
    try:
        from scipy.optimize import linprog
    except Exception:                      # SciPy missing – skip LP phase
        return None

    n = f.shape[0]
    rows = 2 * n - 1
    # Build the convolution matrix A where A[k,i] = f_{k-i} (zero if out of range)
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # column indices i_start … i_end receive the reversed slice of f
        A[k, i_start:i_end + 1] = f[k - np.arange(i_start, i_end + 1)]
    # Linear programme: maximise Σ g_i  → minimise - Σ g_i
    c = -np.ones(n, dtype=np.float64)
    b_ub = M * np.ones(rows, dtype=np.float64)
    bounds = [(0.0, None) for _ in range(n)]

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True, 'verbose': False})
    if not res.success:
        return None
    return res.x


# ----------------------------------------------------------------------
# Smooth‑max gradient step (soft‑max with Adam)
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, beta: float) -> np.ndarray:
    """Gradient of  Lβ(a)= (1/β)·log Σ exp(β·conv)  w.r.t. a.
    Returns ∂L/∂a (same shape as a).  Uses FFT for speed.
    """
    n = a.shape[0]
    conv = _fft_convolve(a)                 # length 2n‑1
    # soft‑max weights
    max_c = conv.max()
    exp_term = np.exp(beta * (conv - max_c))
    w = exp_term / exp_term.sum()           # probability vector, length 2n‑1
    # gradient = 2·(w⋆a)  (cross‑correlation)
    a_rev = a[::-1]
    corr = _fft_convolve_full(w, a_rev)     # length 3n‑2
    grad = 2.0 * corr[n - 1:n - 1 + n]       # slice the needed n components
    return grad


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the time budget and
    returns the best sequence discovered (plain Python list).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    # ------------------------------------------------------------------
    # 1. Choose problem size – a few‑thousand entries give a good trade‑off
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # 2. Baseline – Catalan distribution (known asymptotic constant π/2)
    # ------------------------------------------------------------------
    a = _catalan_normalised(n)
    best = a.copy()
    best_score = 2.0 * n * np.max(_fft_convolve(a))

    # ------------------------------------------------------------------
    # 3. Grid search over Beta‑type families (quick, cheap evaluations)
    # ------------------------------------------------------------------
    alphas = np.linspace(0.3, 1.5, 13)   # 0.3 … 1.5
    betas  = np.linspace(0.3, 1.5, 13)

    for alpha in alphas:
        for beta in betas:
            if time.time() >= deadline:
                break
            cand = _beta_distribution(n, alpha, beta)
            score = 2.0 * n * np.max(_fft_convolve(cand))
            if score < best_score:
                best_score = score
                best = cand.copy()
                print(f"[grid] α={alpha:.3f} β={beta:.3f} → score {best_score:.9f}")

    # ------------------------------------------------------------------
    # 4. Linear‑programming boost – tries to increase total mass while
    #    respecting the current maximal convolution bound.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        f = best.copy()
        for lp_iter in range(8):                     # a handful of LP refinements
            if time.time() >= deadline:
                break
            M = np.max(_fft_convolve(f))
            g = _lp_step(f, M)
            if g is None:
                break
            sum_g = g.sum()
            if sum_g <= 1.0001:                      # no slack gained
                break
            # Normalise g to a probability vector (sum = 1)
            g_norm = g / sum_g
            # Mix the old and the new candidate (α=0.5 gives a safe convex step)
            mix = 0.5
            f_new = (1.0 - mix) * f + mix * g_norm
            f_new = np.maximum(f_new, 0.0)
            f_new /= f_new.sum()
            new_score = 2.0 * n * np.max(_fft_convolve(f_new))
            if new_score < best_score:
                best_score = new_score
                best = f_new.copy()
                print(f"[lp] iter {lp_iter} improved to {best_score:.9f}")
            f = f_new

    # ------------------------------------------------------------------
    # 5. Smooth‑max gradient descent (Adam) – final polishing step
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        a = best.copy()
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        lr = 0.15
        beta1, beta2 = 0.9, 0.999
        eps = 1e-12
        for it in range(2000):
            if time.time() >= deadline:
                break
            # temperature (β) slowly ramps up – the objective approaches the true max
            beta_tmp = min(200.0, 1.0 + it * 0.05)
            grad = _softmax_gradient(a, beta_tmp)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** (it + 1))
            v_hat = v / (1.0 - beta2 ** (it + 1))
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            a = a - step
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 0.0:
                a /= s
            else:                                   # degenerate – fall back to uniform
                a = np.full_like(a, 1.0 / n)

            cur_score = 2.0 * n * np.max(_fft_convolve(a))
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[gd] iter {it:04d}  score {best_score:.9f}")

    # ------------------------------------------------------------------
    # 6. Return the best candidate found
    # ------------------------------------------------------------------
    print(f"[final] n={n}, best score ≈ {best_score:.9f}")
    return best.tolist()
