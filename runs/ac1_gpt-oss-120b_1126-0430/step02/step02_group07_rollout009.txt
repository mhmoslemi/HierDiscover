def apply_move(cur, conv, src, tgt, delta):
    # cur: current vector (length n)
    # conv: current conv (length 2n-1)
    # Returns new_cur, new_conv, new_max (or just new_conv)
    n = len(cur)
    new_cur = cur.copy()
    new_cur[src] -= delta
    new_cur[tgt] += delta
    # conv updates
    new_conv = conv.copy()
    # subtract 2*delta*old_cur from positions src:src+n

    new_conv[tgt:tgt+n] += 2.0 * delta * cur

    new_conv[2*src] += delta * delta
    new_conv[2*tgt] += delta * delta
    new_conv[src + tgt] -= 2.0 * delta * delta
    return new_cur, new_conv

"""
Propose a sequence of non‑negative heights that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm works in three phases:

1. **Initial sampling** – generate a large number of random probability
   vectors (Dirichlet draws, power‑law, exponential families).  Keep the best
   one as a seed.

2. **Fast simulated‑annealing / hill‑climbing** – the current distribution
   `a` is kept together with its full self‑convolution `c = a * a`.
   A move consists of transferring a small amount `δ` of mass from an index
   that contributes most to the current maximal convolution value to a
   low‑mass index.  The convolution update is performed *incrementally* in
   O(n) time using the identity

        (a+Δ)*(a+Δ) = a*a + 2·a*Δ + Δ*Δ  ,

   where `Δ` has only two non‑zero entries.  This makes it possible to run
   many (hundreds of thousands) of iterations within the time budget.
   The temperature slowly decreases; worse moves are occasionally accepted
   to escape plateaus.

3. **Return the best vector** – after the time limit the best sequence
   encountered (as a Python list of floats) is returned.

All numbers are kept non‑negative and the sum is forced to 1, so the scalar
invariant of the objective is respected.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level, no closures
# ----------------------------------------------------------------------


def _project_to_simplex(v):
    """Project a vector onto the probability simplex {x ≥ 0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


def _score_from_conv(conv, n):
    """
    Compute the official score from a convolution array.
    The vector is assumed to be normalised (sum = 1), so the denominator is 1.
    """
    return float(2.0 * n * conv.max())


def _apply_move(cur, conv, src, tgt, delta):
    """
    Return the updated (new_cur, new_conv) after moving `delta` mass from `src`
    to `tgt`.  Both `cur` and `conv` are *not* modified in‑place.
    """
    n = cur.shape[0]
    # --- new probability vector
    new_cur = cur.copy()
    new_cur[src] -= delta
    new_cur[tgt] += delta

    # --- new convolution using the identity
    new_conv = conv.copy()
    # 2 * cur * Δ  (linear part)
    new_conv[src:src + n] -= 2.0 * delta * cur
    new_conv[tgt:tgt + n] += 2.0 * delta * cur
    # Δ * Δ  (quadratic part)
    new_conv[2 * src] += delta * delta
    new_conv[2 * tgt] += delta * delta
    new_conv[src + tgt] -= 2.0 * delta * delta

    return new_cur, new_conv


def _random_dirichlet(rng, n, concentration):
    """
    Sample a random point on the simplex using a Dirichlet distribution.
    `concentration` may be a scalar or an array of size n.
    """
    sample = rng.gamma(shape=concentration, scale=1.0, size=n)
    total = sample.sum()
    if total <= 0:
        return np.full(n, 1.0 / n)
    return sample / total


def _initial_sampling(rng, n, time_limit, start):
    """
    Perform a quick random search over several parametric families.
    Returns (best_vector, best_score) found within the allocated time.
    """
    best_vec = None
    best_score = np.inf

    # helper
    def _try(vec, label):
        nonlocal best_vec, best_score
        cur = np.maximum(vec, 0.0)
        cur_sum = cur.sum()
        if cur_sum == 0:
            return
        cur /= cur_sum
        # fast score (exact, no evaluator call needed)
        conv = np.convolve(cur, cur)
        score = _score_from_conv(conv, n)
        if score < best_score:
            best_score = score
            best_vec = cur.copy()
            print(f"[{label}] new best {best_score:.6f}")

    # 1) Dirichlet draws with varying concentration
    for _ in range(3000):
        if time.time() - start > time_limit:
            break
        conc = rng.uniform(0.1, 5.0)       # scalar concentration
        vec = _random_dirichlet(rng, n, conc)
        _try(vec, "dir")

    # 2) Power‑law families a_i ∝ (i+shift)^{-α}
    idx = np.arange(n, dtype=float)
    for _ in range(2000):
        if time.time() - start > time_limit:
            break
        alpha = rng.uniform(0.2, 1.5)
        shift = rng.uniform(0.001, 5.0)
        raw = np.power(idx + shift, -alpha)
        _try(raw, "pl")

    # 3) Exponential families a_i ∝ exp(-β i)
    for _ in range(1500):
        if time.time() - start > time_limit:
            break
        beta = rng.uniform(1e-5, 0.2)
        raw = np.exp(-beta * idx)
        _try(raw, "exp")

    # 4) Mixture of two power‑laws
    for _ in range(1500):
        if time.time() - start > time_limit:
            break
        a1 = rng.uniform(0.2, 1.5)
        s1 = rng.uniform(0.001, 5.0)
        a2 = rng.uniform(0.2, 1.5)
        s2 = rng.uniform(0.001, 5.0)
        p = rng.random()
        raw = (p * np.power(idx + s1, -a1) +
               (1 - p) * np.power(idx + s2, -a2))
        _try(raw, "mixpl")

    # 5) Simple uniform (fallback)
    if best_vec is None:
        best_vec = np.full(n, 1.0 / n)
        best_score = _score_from_conv(np.convolve(best_vec, best_vec), n)

    return best_vec, best_score

def propose_candidate(seed=123, budget_s=950, **kwargs):
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The function runs for at most ``budget_s`` seconds and returns the best
    list of heights found.
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # Determine the length of the problem (provided by the harness)
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)  # type: ignore[name-defined]  # noqa: F821

    # --------------------------------------------------------------
    try:

        cur = np.maximum(cur, 0.0)
        cur_sum = cur.sum()
        if cur_sum == 0:
            raise ValueError
        cur /= cur_sum
        best_seq = cur.copy()
        best_score = _score_from_conv(np.convolve(cur, cur), n)
        print(f"[init] baseline length={n}, score={best_score:.6f}")
    except Exception:
        # fallback to uniform
        cur = np.full(n, 1.0 / n, dtype=float)
        best_seq = cur.copy()
        best_score = _score_from_conv(np.convolve(cur, cur), n)
        print(f"[init] uniform fallback length={n}, score={best_score:.6f}")

    # --------------------------------------------------------------
    # Quick random sampling to get a better starting point
    # --------------------------------------------------------------
    sample_time = budget_s * 0.08  # reserve ~8 % of the budget for sampling
    vec, scr = _initial_sampling(rng, n, sample_time, start)
    if scr < best_score:
        cur = vec
        best_seq = vec.copy()
        best_score = scr
        print(f"[init] improved by sampling to {best_score:.6f}")

    # --------------------------------------------------------------
    # Pre‑compute convolution of the current vector
    # --------------------------------------------------------------
    conv = np.convolve(cur, cur)          # length 2n-1
    cur_score = best_score                # same as best_score now
    max_iters = 10_000_000                # safety cap (will be limited by time)

    # Annealing parameters
    T0 = cur_score * 0.02                  # initial temperature (≈2 % of score)
    min_frac = 1e-5                        # minimal move fraction
    max_frac = 0.25                        # maximal move fraction at start
    big_move_prob = 0.003                  # probability of trying a very big transfer
    big_frac = 0.7                         # up‑to 70 % of source mass in a big move

    it = 0
    while time.time() - start < budget_s - 2 and it < max_iters:
        it += 1

        # ---------------------------------------------------------
        # 1) pick source index – the constituent that most contributes
        #    to the current peak of the convolution
        # ---------------------------------------------------------
        max_idx = int(np.argmax(conv))
        i = np.arange(n)
        j = max_idx - i
        mask = (j >= 0) & (j < n)
        contrib = np.zeros(n, dtype=float)
        contrib[mask] = cur[i[mask]] * cur[j[mask]]
        src = int(np.argmax(contrib))

        # If the chosen source has essentially no mass, pick a random one
        if cur[src] <= 1e-12:
            src = int(rng.integers(n))

        # ---------------------------------------------------------
        # 2) pick a target – usually an index with very small mass
        # ---------------------------------------------------------
        # we test a few random candidates and keep the best improvement
        num_targets = 5
        best_tgt = None
        best_tgt_score = np.inf
        best_tgt_delta = None

        # move size schedule – linearly shrinking fraction
        elapsed = time.time() - start
        progress = elapsed / budget_s
        frac = max(min_frac, max_frac * (1.0 - progress))

        for _ in range(num_targets):
            tgt = int(rng.integers(n))
            if tgt == src:
                continue

            # Occasionally try a "big" move
            if rng.random() < big_move_prob:
                dst_frac = big_frac
            else:
                dst_frac = frac

            # step is a random proportion of the source mass
            delta = cur[src] * dst_frac * rng.random()
            if delta <= 1e-15:
                continue

            # simulate the move
            new_cur, new_conv = _apply_move(cur, conv, src, tgt, delta)
            # renormalisation is not required – mass is conserved
            new_score = _score_from_conv(new_conv, n)

            if new_score < best_tgt_score:
                best_tgt_score = new_score
                best_tgt = tgt
                best_tgt_delta = delta

        # If we could not find any improving target, fall back to a random move
        if best_tgt is None:
            # simple random move
            tgt = int(rng.integers(n - 1))
            if tgt >= src:
                tgt += 1
            delta = cur[src] * frac * rng.random()
            if delta <= 0:
                continue
            new_cur, new_conv = _apply_move(cur, conv, src, tgt, delta)
            new_score = _score_from_conv(new_conv, n)
        else:
            tgt = best_tgt
            delta = best_tgt_delta
            new_cur = cur.copy()
            new_cur[src] -= delta
            new_cur[tgt] += delta
            new_conv = conv.copy()
            # linear part
            new_conv[src:src + n] -= 2.0 * delta * cur
            new_conv[tgt:tgt + n] += 2.0 * delta * cur
            # quadratic part
            new_conv[2 * src] += delta * delta
            new_conv[2 * tgt] += delta * delta
            new_conv[src + tgt] -= 2.0 * delta * delta
            new_score = _score_from_conv(new_conv, n)

        # ---------------------------------------------------------
        # 3) acceptance test (simulated annealing)
        # ---------------------------------------------------------
        accept = False
        if new_score < cur_score - 1e-14:
            accept = True
        else:
            # temperature linearly decreasing from T0 to 0
            T = T0 * (1.0 - progress)
            if T > 0 and rng.random() < np.exp(-(new_score - cur_score) / T):
                accept = True

        if accept:
            cur = new_cur
            conv = new_conv
            cur_score = new_score

            # maintain sum≈1 (numerical drift)
            if abs(cur.sum() - 1.0) > 1e-12:
                cur /= cur.sum()
                conv = np.convolve(cur, cur)

            if cur_score < best_score - 1e-14:
                best_score = cur_score
                best_seq = cur.copy()
                print(f"[iter {it}] new best {best_score:.6f}")

        # occasional global re‑normalisation (numerical safety)
        if it % 2000 == 0:
            cur = _project_to_simplex(cur)
            conv = np.convolve(cur, cur)

    # ------------------------------------------------------------------
    # Final sanity check with the official evaluator (rare but safe)
    # ------------------------------------------------------------------
    try:
        official = evaluate_sequence(best_seq.tolist())
        if official < best_score:
            best_score = official
            best_seq = np.array(best_seq, dtype=float)  # already fine
    except Exception:
        pass

    print(f"[final] best score = {best_score:.6f} after {it} iterations")
    # Return as a plain Python list (required by the harness)
    return best_seq.tolist()
