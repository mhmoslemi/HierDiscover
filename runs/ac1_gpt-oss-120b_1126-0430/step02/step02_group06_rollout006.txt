"""
Projected‑gradient optimisation of the max‑self‑convolution.
----------------------------------------------------------------
* Starts from the Catalan (arcsine) seed, which already yields the
  classic bound π/2 ≈ 1.5708.
* Minimises a smooth approximation of the objective
      max_k (a*a)_k  ≈  (1/γ)·log Σ_k exp(γ·(a*a)_k)
  using a simple projected (Adam‑style) gradient descent on the
  probability simplex.  The smoothing parameter γ is slowly increased,
  turning the objective from a smooth average into the true max.
* The gradient is obtained analytically:
      ∂/∂a_i  Σ_k w_k (a*a)_k  =  2·Σ_j a_j·w_{i+j},
  where w are the soft‑max weights.  This convolution is computed
  efficiently with a single FFT per iteration.
* After the gradient phase the routine falls back to the fast greedy
  mass‑transfer refinement used previously; this never worsens the
  score and can clean up remaining plateaus.
* The whole procedure respects the 1000 s budget and returns the best
  sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast evaluation (identical to evaluate_sequence but without the security
# checks and using an FFT based convolution).
# ----------------------------------------------------------------------
def _fast_evaluation(seq: np.ndarray) -> float:
    """Return the same metric as evaluate_sequence but much faster."""
    n = seq.size
    if n == 0:
        return np.inf
    # FFT size = next power of two >= 2*n-1
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Catalan (arcsine) seed.
# a_0 = 1, a_{i+1} = a_i * (2*i+1) / (2*i+2)
# ----------------------------------------------------------------------
def _catalan_sequence(length: int) -> np.ndarray:
    """Return the first ``length`` Catalan‑like coefficients."""
    seq = np.empty(length, dtype=float)
    a = 1.0
    seq[0] = a
    for i in range(1, length):
        a *= (2.0 * i - 1.0) / (2.0 * i)
        seq[i] = a
    return seq


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex (sum = 1, ≥0).
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x≥0, Σx=1}."""
    # Implementation of the algorithm from
    # "Efficient Projections onto the L1‑Ball for Learning in High Dimensions"
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero after projection
        return np.full_like(v, 1.0 / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Projected gradient (Adam) optimiser on the smooth max objective.
# ----------------------------------------------------------------------
def _smooth_max_optimize(seq: np.ndarray, deadline: float) -> np.ndarray:
    """
    Perform projected gradient descent on the softened max objective.
    ``seq`` is assumed to be a probability vector (sum==1, non‑negative).
    The function returns the best sequence seen before the deadline.
    """
    n = seq.size
    m = 2 * n - 1                     # length of the self‑convolution
    # FFT length that can accommodate both a*a (2n‑1) and a*w_rev (3n‑2)
    L = 1 << ((3 * n - 2).bit_length())

    # Adam moments
    m1 = np.zeros_like(seq)
    m2 = np.zeros_like(seq)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_seq = seq.copy()
    best_score = _fast_evaluation(best_seq)

    it = 0
    while time.time() < deadline:
        # FFT of the current sequence (zero‑padded to length L)
        A = np.fft.rfft(seq, L)

        # ----- self‑convolution (a*a) -----
        conv_full = np.fft.irfft(A * A, L)
        s = conv_full[:m]                     # length 2n‑1

        # ----- soft‑max weights -----
        s_max = s.max()
        # slowly raise the smoothing parameter
        gamma = 2.0 + 0.001 * it
        exp_vals = np.exp(gamma * (s - s_max))
        w = exp_vals / exp_vals.sum()          # Σ w = 1

        # ----- gradient of the smoothed objective -----
        w_rev = w[::-1]                        # reverse for convolution
        W = np.fft.rfft(w_rev, L)
        grad_full = np.fft.irfft(A * W, L)     # a * w_rev
        # extract the n components needed for the gradient
        # (see derivation in analysis)
        grad = 2.0 * grad_full[m - 1 - np.arange(n)]

        # ----- Adam update on the simplex -----
        it += 1
        m1 = beta1 * m1 + (1.0 - beta1) * grad
        m2 = beta2 * m2 + (1.0 - beta2) * (grad * grad)
        m1_hat = m1 / (1.0 - beta1 ** it)
        m2_hat = m2 / (1.0 - beta2 ** it)

        # step size – larger initially, decays as 1/√t
        lr = 0.5 / np.sqrt(it)
        step = lr * m1_hat / (np.sqrt(m2_hat) + eps)

        seq = seq - step
        # enforce the probability simplex
        seq = _project_simplex(seq)

        # occasional evaluation (every ~200 iterations)
        if it % 200 == 0:
            cur_score = _fast_evaluation(seq)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = seq.copy()
                print(f"[smooth‑GD] iter {it:6d}  score {best_score:.6f}")

        # tiny random jitter to avoid getting stuck on a perfectly flat plateau
        if np.random.rand() < 0.001:
            idx = np.random.randint(n)
            seq[idx] = max(0.0, seq[idx] + np.random.randn() * 1e-3)
            seq = _project_simplex(seq)

    return best_seq


# ----------------------------------------------------------------------
# Simple greedy refinement – identical to the one used previously.
# ----------------------------------------------------------------------
def _refine(seq: np.ndarray, deadline: float) -> np.ndarray:
    """Greedy local improvement; never raises the objective."""
    n = seq.size
    best_seq = seq.copy()
    best_score = _fast_evaluation(best_seq)

    fft_len = 1 << ((2 * n - 1).bit_length())
    it = 0
    while time.time() < deadline:
        fa = np.fft.rfft(seq, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        k = int(conv.argmax())
        # contribution weights for each position to that maximal entry
        i_arr = np.arange(n)
        j_arr = k - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        w = np.zeros_like(seq)
        w[mask] = seq[j_arr[mask]]

        contrib = seq * w
        src = int(np.argmax(contrib))
        sink = int(np.argmin(w))

        if seq[src] < 1e-12:
            break

        eps = 0.05 * seq[src]          # move 5 % of the source mass
        new_seq = seq.copy()
        new_seq[src] -= eps
        new_seq[sink] += eps

        score = _fast_evaluation(new_seq)
        if score < best_score - 1e-12:
            seq = new_seq
            best_seq = new_seq.copy()
            best_score = score
            if it % 1000 == 0:
                print(f"[greedy] iter {it:6d}  score {best_score:.6f}")
        else:
            # occasional random jitter
            if np.random.rand() < 0.01:
                jitter_idx = np.random.randint(n)
                jitter = np.random.randn() * 0.001
                new_seq = seq.copy()
                new_seq[jitter_idx] = max(0.0, new_seq[jitter_idx] + jitter)
                seq = new_seq
        it += 1
    return best_seq


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights aiming at a low evaluation score.

    The routine works within ``budget_s`` seconds (default 1000).  It builds
    a Catalan (arcsine) seed, performs a smoothed gradient descent,
    finishes with a greedy refinement and finally returns the best sequence
    discovered.
    """
    np.random.seed(seed)
    start = time.time()
    # safety margin – keep a few seconds free for the final printout
    deadline = start + budget_s - 3

    # Length – a few thousand gives a good trade‑off between quality and speed.
    n = 5000

    # 1. Initial seed (Catalan / arcsine) and normalisation.
    seq = _catalan_sequence(n).astype(float)
    seq /= seq.sum()                     # ‑‑ sum = 1 (scale‑invariant)

    print(f"[init] length={n}  initial score={_fast_evaluation(seq):.6f}")

    # 2. Smoothed gradient descent (≈ 70 % of the budget).
    grad_deadline = start + int(0.70 * (budget_s - 3))
    if grad_deadline > time.time():
        seq = _smooth_max_optimize(seq, grad_deadline)
        cur_score = _fast_evaluation(seq)
        print(f"[after GD] score={cur_score:.6f}")

    # 3. Greedy refinement for the remaining time.
    seq = _refine(seq, deadline)
    final_score = _fast_evaluation(seq)
    print(f"[final] best score={final_score:.6f}")

    # Return as plain Python list.
    return seq.tolist()
