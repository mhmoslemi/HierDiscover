"""
Propose a non‑negative step‑function sequence that (approximately) minimizes

    O(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The search consists of three stages:

1. **Initialisation** – use the best known sequence (`height_sequence_1`) if it
   exists, otherwise start from a uniform distribution.  The vector is
   projected onto the probability simplex (sum = 1) so that the denominator
   of `O` is fixed.

2. **Phase 1 – Adam optimisation on a smooth surrogate** – we minimise the
   log‑sum‑exp approximation of the max‑convolution:

        f_α(a) = (2·n/α)·log(∑_k e^{α·conv_k})

   The gradient of `f_α` can be written as a correlation and is computed
   efficiently with FFTs.  An Adam‑style adaptive stepsize is used, together
   with an annealing schedule for `α` (linear from ≈1 to ≈2000).  After each
   iteration the exact objective `O` is evaluated and the best sequence so far
   is stored.

3. **Phase 2 – Simulated‑annealing hill‑climbing + sub‑gradient polishing** –
   random “mass‑transfer’’ moves (move a fraction of probability mass from
   index *i* to index *j*) are attempted; they are accepted if they improve
   the exact objective, otherwise with a Metropolis probability that decays
   with a temperature schedule.  Every few iterations a sub‑gradient step
   for the true max‑convolution entry is performed.  If no improvement is
   observed for many steps we jump to a fresh restart around the current best.

All computations use FFT‑based convolutions (`O(n log n)`).  The algorithm
fits comfortably into the 1000 s budget and prints progress updates.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# -------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        # all non‑positive – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# -------------------------------------------------------------
# FFT‑based full self‑convolution (length 2·n‑1)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return a*a (full convolution) using an FFT of length L."""
    fft_a = rfft(a, n=L)
    conv_full = irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# -------------------------------------------------------------
# Exact objective for a (not necessarily normalised) vector
def _objective(a: np.ndarray, n: int, L: int) -> float:
    """Compute O = 2·n·max(conv(a,a)) / (sum(a)^2)."""
    conv = _conv_fft(a, L)
    s = a.sum()
    if s == 0.0:
        return np.inf
    return 2.0 * n * conv.max() / (s * s)


# -------------------------------------------------------------
# Sub‑gradient of the maximal convolution entry (as used in the reference code)
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. a."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)  # typically a single index
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        # derivative w.r.t a[i] is a[j] (and double when i==j)
        g[i_idx] += 2.0 * a[j_idx]
    g /= len(ks)  # average if several maximisers
    return g


# -------------------------------------------------------------
# Gradient of the smoothed log‑sum‑exp surrogate
def _smooth_gradient(a: np.ndarray, conv: np.ndarray, alpha: float, L: int) -> np.ndarray:
    """
    Compute ∇_a ( (2·n/α)·log∑_k e^{α·conv_k} ).
    The formula reduces to 2·corr(w, a) where w_k ∝ e^{α·conv_k}.
    """
    # soft‑max weights
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)               # for numerical stability
    w = exp_scaled / exp_scaled.sum()

    # correlation w ⋆ a (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = rfft(w, n=L)
    fft_a_rev = rfft(a_rev, n=L)
    corr_rev = irfft(fft_w * fft_a_rev, n=L)               # length >= 3·n‑2
    n_len = a.size
    grad = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]
    return grad
