"""
Optimise a non‑negative step‑function (probability vector) to minimise

    O(a) = 2·n·max(conv(a,a))      with   sum(a)=1

The algorithm proceeds in three stages:

1️⃣  **Flattening (L2‑error minimisation)**  
    Optimise the squared error ∑_k (conv_k – 1/(2n‑1))² using Adam on a
    soft‑max (log‑sum‑exp) parametrisation a_i = exp(z_i)/∑exp(z).  The
    gradient of this objective is obtained by a single FFT‑based
    correlation, guaranteeing O(n log n) per iteration.

2️⃣  **Smooth max‑approximation**  
    Switch to a log‑sum‑exp surrogate of the max and perform projected
    gradient descent with an annealing temperature α (the code from the
    reference is reused with tighter step‑size control).

3️⃣  **Directed local refinement**  
    A hill‑climbing loop that repeatedly transfers a small fraction of
    mass from indices with a large sub‑gradient of the current peak to
    indices with a small sub‑gradient.  Accepted moves lower the exact
    objective; otherwise they are accepted with a tiny Metropolis
    probability.  Periodically a pure sub‑gradient step on the active
    maximal convolution entry is performed.

All phases respect the 1000 s budget; the best sequence found is printed
and returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full self‑convolution via FFT (returns length 2·n‑1 array)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Exact objective for a simplex‑normalised vector
def _objective_simplex(a: np.ndarray, n: int, L: int) -> float:
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Sub‑gradient of the maximal convolution entry (used in phase 3)
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)  # should be a few indices
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    g /= len(ks)
    return g


# ----------------------------------------------------------------------
# Gradient of the smooth log‑sum‑exp surrogate (phase 2)
def _smooth_gradient(a: np.ndarray, conv: np.ndarray, alpha: float, L: int) -> np.ndarray:
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)          # stability
    w = exp_scaled / exp_scaled.sum()                # soft‑max weights
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n = a.size
    grad = 2.0 * corr_rev[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Gradient of the L2‑flattening objective (phase 1)
def _flatten_gradient(a: np.ndarray, conv: np.ndarray, L: int) -> np.ndarray:
    """∇_a ∑_k (conv_k – c)²  where c = 1/(2n‑1)."""
    n = a.shape[0]
    c = 1.0 / (2 * n - 1)
    d = conv - c                      # length 2n‑1
    a_rev = a[::-1]
    fft_d = np.fft.rfft(d, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_d * fft_a_rev, n=L)   # length L
    grad = 4.0 * corr_rev[n - 1 : n - 1 + n]
    return grad

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):

    Search for a step‑function sequence that minimises the evaluation function.
    Returns the sequence as a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # reserve a tiny safety margin
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Initialise
    if 'height_sequence_1' in globals():
        base = np.array(height_sequence_1, dtype=float)
        n = base.size
        a0 = base / base.sum()                     # normalise to simplex
        print(f"[init] using existing sequence, n={n}")
    else:
        n = 4000
        a0 = np.full(n, 1.0 / n)
        print(f"[init] no prior sequence – using uniform, n={n}")

    # FFT length sufficient for both convolution (2n‑1) and correlation (3n‑2)
    L = 1 << ((3 * n - 2).bit_length())             # next power of two

    # --------------------------------------------------------------
    # Phase 1 – L2 flattening using Adam on the soft‑max parametrisation
    best_seq = a0.copy()
    best_score = _objective_simplex(best_seq, n, L)
    print(f"[phase‑1‑flatten] start O={best_score:.6f}")

    # Adam state
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.25
    m = np.zeros(n)
    v = np.zeros(n)
    beta1_pow = 1.0
    beta2_pow = 1.0
    # initialise logits (soft‑max)
    z = np.log(best_seq + 1e-12)          # avoid log(0)

    # allocate ~45 % of total time to flattening
    flat_deadline = start_time + budget_s * 0.45
    it_flat = 0
    while time.time() < flat_deadline:
        # soft‑max to obtain a
        exp_z = np.exp(z - np.max(z))     # shift for numerical stability
        a = exp_z / np.sum(exp_z)         # simplex guaranteed

        # full convolution and exact objective
        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[{time.time():.1f}] Phase‑1 new best O={best_score:.6f}")

        # gradient of flattening loss
        grad = _flatten_gradient(a, conv, L)      # ∂F/∂a

        # convert to gradient w.r.t. logits (soft‑max chain rule)
        grad_z = a * (grad - np.dot(grad, a))

        # Adam update
        it_flat += 1
        beta1_pow *= beta1
        beta2_pow *= beta2
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # adjust logits
        z = z - step

        # occasionally inject a tiny random jitter to avoid stagnation
        if np.random.rand() < 0.001:
            z += np.random.randn(n) * 0.001

    # --------------------------------------------------------------
    # Phase 2 – smooth log‑sum‑exp surrogate (as in the reference code)
    print(f"[phase‑2‑smooth] entering, best O={best_score:.6f}")
    best_seq2 = best_seq.copy()
    best_obj2 = best_score

    # hyper‑parameters for the log‑sum‑exp phase
    alpha_start = 5.0
    alpha_end = 2000.0
    max_iter_phase2 = 200_000
    it2 = 0
    phase2_deadline = start_time + budget_s * 0.75

    while time.time() < phase2_deadline and it2 < max_iter_phase2:
        conv = _conv_fft(best_seq2, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj2:
            best_obj2 = cur_obj
            best_seq2 = best_seq2.copy()
            print(f"[{time.time():.1f}] Phase‑2 new best O={best_obj2:.6f}")

        cand = _project_to_simplex(cand)

        conv_cand = _conv_fft(cand, L)
        cand_obj = 2.0 * n * conv_cand.max()

            best_seq2 = cand
        else:

    print(f"[phase‑3‑local] start, best O={best_score:.6f}")

    max_iter_per_restart = 70_000
    no_improve_limit = 3000
    temperature0 = 0.015
    subgrad_interval = 9

                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] Sub‑grad new best O={best_score:.6f}")

            # restart if stuck
            if no_improve >= no_improve_limit:
                break

            it_loc += 1

        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:
        print("[final] evaluate_sequence failed:", e)

    return best_seq.tolist()
