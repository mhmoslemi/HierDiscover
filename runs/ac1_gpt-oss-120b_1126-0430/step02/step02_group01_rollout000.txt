function gradient_smooth(a,beta):

    n = len(a)

    # compute convolution a*a (full) using FFT for speed
    Lconv = 1 << (2*n-1).bit_length() # next power of 2
    # Actually compute conv via FFT: conv = np.fft.irfft(np.fft.rfft(a, Lconv) * np.fft.rfft(a, Lconv), n=Lconv)[:2*n-1]
    fft_a = np.fft.rfft(a, Lconv)
    conv = np.fft.irfft(fft_a * fft_a, n=Lconv)[:2*n-1]

    # compute w weights
    max_c = conv.max()
    exp_vals = np.exp(beta * (conv - max_c))  # safe
    w = exp_vals / exp_vals.sum()  # length 2n-1

    # compute correlation of w with a: g_i = 2 * sum_{t=0}^{n-1} w[i+t] * a[t]  for i in[0,n-1]

"""
Propose a step‑function sequence that minimises the bound

    O = 2·n·max(conv(a,a))   (with sum(a)=1, a_i ≥ 0)

The algorithm uses a *projected smooth gradient descent* on the
max‑convolution.  The non‑smooth max is approximated by a
log‑sum‑exp (soft‑max) with an annealing parameter β that grows
during the run.  In each iteration we

1. compute the full self‑convolution of the current candidate a
   (FFT based, O(n log n));
2. build the soft‑max weights w_k ∝ exp(β·(conv_k−max(conv)));
3. obtain the gradient of the soft‑max with respect to a via a
   correlation (again via FFT);
4. take a projected gradient step on the probability simplex.

The step‑size η and the smoothing β are decreased / increased
respectively according to a simple schedule.  Occasionally a tiny
random perturbation is added to escape plateaus and a best‑so‑far
solution is kept.  The routine respects the 1000 s budget, prints
progress and finally returns the best sequence found.

The implementation is completely self‑contained (except for the
provided ``evaluate_sequence`` function) and makes no use of
closures or lambdas.
"""

import time
import math
import numpy as np

# -------------------------------------------------------------
# Helper: projection onto the simplex {x ≥ 0, sum(x)=1}
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of a vector onto the probability simplex."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:          # all non‑positive -> uniform vector
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# -------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small utility returning the next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# -------------------------------------------------------------
def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using FFT (real‐valued)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv


# -------------------------------------------------------------
def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute g_i = Σ_t w[i+t]·a[t]   for i=0,…,n‑1
    (a correlation of w with a, without reversing a).
    Implemented via FFT.
    """
    n = a.shape[0]
    m = w.shape[0]                # = 2n‑1
    L = _next_pow2(m + n - 1)     # length for linear convolution
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    # first n entries correspond to the required lags 0…n‑1
    return corr_full[:n]


# -------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) (sum(a)=1)."""
    n = a.size
    conv = _conv_self_fft(a)
    return 2.0 * n * conv.max()


# -------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` bound.  Returns the best sequence as a
    plain list of floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # keep a small safety margin

    # -----------------------------------------------------------------
    # Choose an initial length and a starting point.
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=float)
        a = a / a.sum()
        n = a.size
    else:
        # a reasonably large length (the problem favours a few thousand)
        n = 4600
        a = np.random.rand(n)
        a = _project_to_simplex(a)

    # Keep the globally best solution.
    best_seq = a.copy()
    best_score = _objective(best_seq)
    print(f"[init] n={n}, objective={best_score:.6f}")

    # -----------------------------------------------------------------
    # Parameters for the annealing‑gradient descent.
    #   β – smoothing of the max (starts small, grows to a large value)
    #   η – step‑size (starts moderate, decays)
    beta_start = 4.0
    beta_end = 4000.0
    eta_start = 0.45
    eta_end = 0.02

    # A rough estimate of how many iterations we may afford.
    # (The actual number will be limited by the time budget.)
    max_iters_est = 200_000

    iter_cnt = 0
    while time.time() < deadline and iter_cnt < 500_000:
        iter_cnt += 1

        # -------------------------------------------------------------
        # 1) current objective and best‑so‑far update
        conv = _conv_self_fft(a)
        M = conv.max()
        cur_score = 2.0 * n * M
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[iter {iter_cnt:06d}] new best O={best_score:.6f}")

        # -------------------------------------------------------------
        # 2) annealing schedule (linear interpolation on the fraction)
        fraction = min(iter_cnt / max_iters_est, 1.0)
        beta = beta_start + (beta_end - beta_start) * fraction
        eta = eta_start + (eta_end - eta_start) * (fraction - 1.0)   # linear decay

        # -------------------------------------------------------------
        # 3) soft‑max weights for the max‑convolution
        #    w_k ∝ exp(β·(conv_k − max(conv)))  → max entry gets weight 1,
        #    all others get exponentially smaller weight.
        shifted = conv - M                     # ≤ 0
        # Subtract a constant to avoid overflow (max is 0 already)
        exp_vals = np.exp(beta * shifted)
        w = exp_vals / exp_vals.sum()          # length 2n‑1, sums to 1

        # -------------------------------------------------------------
        # 4) gradient of the smooth max (2·correlation)
        g = 2.0 * _correlation_fft(w, a)       # shape (n,)

        # -------------------------------------------------------------
        # 5) projected gradient step on the simplex
        a_new = a - eta * g
        a_new = _project_to_simplex(a_new)

        # -------------------------------------------------------------

        if iter_cnt % 4000 == 0:
            jitter = np.random.randn(n) * 0.001
            a_new += jitter
            a_new = _project_to_simplex(a_new)

        a = a_new

        # -------------------------------------------------------------

            # the current best with a small random perturbation.
            a = best_seq.copy()
            a += np.random.randn(n) * 0.002
            a = _project_to_simplex(a)
            print(f"[restart] iter {iter_cnt}, re‑initialised from best.")

    try:

        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_seq.tolist()
