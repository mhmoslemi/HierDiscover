* Choose a moderate length (≈15 000).  Larger n gives a better
  asymptotic bound while remaining feasible for the exact O(n²) test.
* Initialise with a normalised power‑law a_i ∝ (i+1)^‑p.
  The exponent p is scanned on a fine grid.
* Refine the vector with an accelerated projected sub‑gradient method:
    – convolution (a*a) is obtained by FFT (O(n log n)).
    – the sub‑gradient at the active offset k* (where the convolution
      attains its maximum) is g_i = 2·a_{k*‑i} (zero outside the support).
    – a heavy‑ball momentum term v ← μ·v – η·g is used.
    – after the momentum step the iterate is projected onto the
      probability simplex (Σ a_i = 1, a_i ≥ 0).
* If no improvement is seen for many iterations the current iterate is
  re‑initialised with a small random perturbation of the best‑found vector.
* The cheapest (FFT‑based) score 2·n·max_conv is monitored and the best
  vector is kept.  After the time budget a final exact evaluation using the
  provided ``evaluate_sequence`` is performed and the best sequence is
  returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – must be defined at top level (no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (sum=1, non‑negative)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all entries non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float) -> np.ndarray:
    """Return a normalised power‑law vector a_i ∝ (i+1)^‑p (Σa_i = 1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = i ** (-p)
    seq /= seq.sum()
    return seq


def _conv_max_fft(a: np.ndarray, fft_len: int) -> tuple[float, int]:
    """Return (max value, index) of the autocorrelation a*a using FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa).real[: 2 * a.size - 1]
    max_idx = int(conv.argmax())
    return float(conv[max_idx]), max_idx


def _subgradient_max_convolution(a: np.ndarray, max_idx: int) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)[k] at the active offset ``max_idx``.
    For the active shift k we have (∂/∂a_i) (a*a)[k] = 2·a_{k‑i}
    whenever the index is valid.
    """
    n = a.shape[0]
    lo = max(0, max_idx - (n - 1))
    hi = min(max_idx, n - 1)
    idx = np.arange(lo, hi + 1, dtype=np.int64)
    g = np.zeros_like(a)
    g[idx] = 2.0 * a[max_idx - idx]
    return g


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    ``evaluate_sequence``.  The routine respects the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # 0. Choose a working length.
    #    If a previous construction exists, reuse its length (but enforce a
    #    minimum of 5000 to stay in the asymptotic regime).
    # ------------------------------------------------------------------
    try:
        prev = height_sequence_1               # type: ignore[name-defined]
        n = max(len(prev), 5000)
        prev_arr = np.asarray(prev, dtype=np.float64)
        if prev_arr.sum() == 0:
            prev_arr = None
        else:
            prev_arr = prev_arr / prev_arr.sum()
    except Exception:
        prev_arr = None
        n = 15000                               # default working size

    # FFT length – next power of two ≥ 2n‑1 for cheap convolutions.
    fft_len = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # 1. Scan a fine grid of power‑law exponents.
    # ------------------------------------------------------------------
    p_grid = np.linspace(0.55, 0.68, 81)          # dense around known optimum
    best_a = None
    best_score = float("inf")
    best_p = None

    for p in p_grid:
        if time.time() >= deadline:
            break
        a = _power_law_sequence(n, p)
        maxb, _ = _conv_max_fft(a, fft_len)
        score = 2.0 * n * maxb                     # cheap proxy (exact for Σa=1)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            best_p = p

    # Also consider the previous construction (if any) as a candidate.
    if prev_arr is not None and prev_arr.size == n:
        maxb, _ = _conv_max_fft(prev_arr, fft_len)
        score = 2.0 * n * maxb
        if score < best_score:
            best_score = score
            best_a = prev_arr.copy()
            best_p = "prev"

    # Fallback
    if best_a is None:
        best_a = _power_law_sequence(n, 0.60)
        maxb, _ = _conv_max_fft(best_a, fft_len)
        best_score = 2.0 * n * maxb
        best_p = 0.60

    print(f"Initial power‑law scan finished: best p = {best_p}, cheap score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Refine with accelerated projected sub‑gradient descent.
    # ------------------------------------------------------------------
    a = best_a.copy()                # current iterate (always Σa = 1)
    a_prev = a.copy()
    v = np.zeros_like(a)             # momentum buffer

    best_iter_a = a.copy()
    best_iter_score = best_score

    # hyper‑parameters
    eta0 = 0.6                        # initial step‑size factor
    mu = 0.85                         # momentum coefficient (0 < mu < 1)

    iter_count = 0
    last_improve = 0

    while time.time() < deadline and iter_count < 1_200_000:
        iter_count += 1

        # ----- convolution and active index -----
        maxb, max_idx = _conv_max_fft(a, fft_len)

        # ----- sub‑gradient at the active offset -----
        g = _subgradient_max_convolution(a, max_idx)

        # ----- heavy‑ball update + projection -----
        eta = eta0 / math.sqrt(iter_count)
        v = mu * v - eta * g
        a_candidate = a + v
        a_candidate = _simplex_projection(a_candidate)

        a_prev = a
        a = a_candidate

        # ----- keep track of the best cheap score -----
        if (iter_count % 500 == 0) or (time.time() > deadline - 5):
            cheap_score = 2.0 * n * maxb
            if cheap_score < best_iter_score:
                best_iter_score = cheap_score
                best_iter_a = a.copy()
                last_improve = iter_count
                print(f"[Iter {iter_count:07d}] NEW BEST cheap score = {cheap_score:.6f}")

        # ----- occasional random perturbation to escape plateaus -----
        if iter_count - last_improve > 80_000:
            # perturb the current best vector slightly and restart momentum
            perturb = 0.02 * np.random.rand(n)
            a = _simplex_projection(best_iter_a + perturb)
            a_prev = a.copy()
            v = np.zeros_like(a)
            last_improve = iter_count
            print(f"[Iter {iter_count:07d}] Restarted after stagnation (perturb).")

    # ------------------------------------------------------------------
    # 3. Final exact evaluation (uses the provided ``evaluate_sequence``)
    # ------------------------------------------------------------------
    final_seq = best_iter_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:      # defensive – should never happen
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # ------------------------------------------------------------------
    # 4. Return the best sequence found.
    # ------------------------------------------------------------------
    return final_seq
