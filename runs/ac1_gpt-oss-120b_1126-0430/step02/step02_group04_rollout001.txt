"""
propose_candidate
=================

A fast projected‑Adam optimiser for the “minimum peak of the self‑convolution”
problem.

The algorithm works on a symmetric probability vector `a` (a step‑function)
with ∑a = 1 and a_i ≥ 0.  Symmetry a_i = a_{n‑1‑i} is known to be optimal for the
continuous version of the problem, it reduces the dimensionality and greatly
stabilises the optimisation.

The objective 2·n·max(a∗a) is smooth‑approximated by a soft‑max:
    w_k ∝ exp(α·(b_k−max(b)))   with  b = a∗a.
The gradient of the smooth‑max is
    g = 2·(a ⊛ w_rev)   (⊛ = linear convolution, w_rev = w reversed).

An Adam‐style update (first‑ and second‑moment estimates) is applied, after
which we
   * enforce non‑negativity,
   * enforce symmetry by averaging with the reversed vector,
   * project back onto the probability simplex (Euclidean projection).

The smoothing parameter α is slowly increased, the step‑size is decayed when
no improvement is observed and a random perturbation is injected after a long
stall.  Several simple seeds (Catalan‑like, geometric decays, random simplex)
are evaluated and the best one is taken as the starting point.

The whole routine runs for the allotted time (≈1000 s) and returns the best
sequence discovered as a plain list of Python floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, sum=1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u > cssv / np.arange(1, n + 1))[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using an FFT of power‑of‑2 size."""
    n = x.size + y.size - 1
    size = 1 << (n - 1).bit_length()          # next power of two
    X = np.fft.rfft(x, size)
    Y = np.fft.rfft(y, size)
    Z = X * Y
    conv = np.fft.irfft(Z, size)[:n]
    return conv.real


def _eval_score(a: np.ndarray) -> float:
    """Exact value returned by the supplied evaluate_sequence for a simplex a."""
    n = a.size
    b = _convolve_fft(a, a)
    max_b = b.max()
    return float(2.0 * n * max_b / (a.sum() ** 2))


def _catalan_sequence(n: int) -> np.ndarray:
    """Catalan‑like seed: a_i = C(2i,i)/4**i, normalised to sum 1."""
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * (i - 1) + 1) / (2.0 * i)
    a /= a.sum()
    return a


def _geometric_sequence(n: int, rate: float) -> np.ndarray:
    """Exponential decay a_i ∝ r^i with r = 1 - rate/n, normalised."""
    r = 1.0 - rate / n
    a = (r ** np.arange(n))
    a /= a.sum()
    return a


def _random_simplex(n: int) -> np.ndarray:
    """Draw a random point on the probability simplex."""
    rnd = np.random.rand(n)
    return _proj_simplex(rnd)


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a non‑negative height sequence that minimises ``evaluate_sequence``.
    Returns the best sequence found before the time budget expires.
    """
    np.random.seed(seed)
    safety = 5.0                       # seconds reserved for clean exit
    deadline = time.time() + budget_s - safety

    # --------------------------------------------------------------
    # Problem size – 8000 gives a good trade‑off between quality and speed
    # --------------------------------------------------------------
    n = 8000

    # --------------------------------------------------------------
    # Build a pool of simple seeds and pick the best one
    # --------------------------------------------------------------
    seeds = [_catalan_sequence(n)]
    for rate in np.linspace(0.5, 4.0, 9):
        seeds.append(_geometric_sequence(n, rate))
    seeds.append(_random_simplex(n))

    best_seq = None
    best_score = np.inf
    for s in seeds:
        sc = _eval_score(s)
        if sc < best_score:
            best_score = sc
            best_seq = s.copy()
    print(f"[init] n={n}, initial best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Adam hyper‑parameters
    # --------------------------------------------------------------
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.12                # base learning rate (will be adapted)
    alpha = 10.0             # soft‑max “temperature” multiplier
    max_alpha = 5000.0

    # optimisation state
    a = best_seq.copy()
    m = np.zeros_like(a)     # first moment
    v = np.zeros_like(a)     # second moment
    t_iter = 0
    no_improve = 0
    cur_score = best_score

    # --------------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        t_iter += 1

        # ---- convolution and soft‑max weights ---------------------------------
        b = _convolve_fft(a, a)                # length = 2n‑1
        max_b = b.max()

        # occasionally increase the focus on the true maximum
        if t_iter % 2000 == 0 and t_iter > 0:
            alpha = min(alpha * 1.07, max_alpha)

        # soft‑max (weights sum to 1, dominant entry gets weight 1)
        w = np.exp(alpha * (b - max_b))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of the smooth‑max (see analysis) -----------------------
        w_rev = w[::-1]
        conv_full = _convolve_fft(a, w_rev)    # length = 3n‑2
        idx = (len(b) - 1) - np.arange(n)      # slice that yields the gradient
        grad = 2.0 * conv_full[idx]

        # ---- Adam moment updates ---------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)

        # ---- tentative update -------------------------------------------------
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step

        # ---- projection back onto the feasible set ----------------------------
        a_new = np.maximum(a_new, 0.0)                 # non‑negative
        a_new = (a_new + a_new[::-1]) * 0.5           # enforce symmetry
        a_new = _proj_simplex(a_new)                  # simplex (keeps symmetry)

        # ---- evaluate ---------------------------------------------------------
        new_score = _eval_score(a_new)

        improved = new_score < cur_score
        if improved:
            a = a_new
            cur_score = new_score

            # keep a global best
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{t_iter:07d}] NEW BEST {best_score:.6f} "
                      f"(lr={lr:.5f}, α={alpha:.2f})")
            no_improve = 0
        else:
            # no progress – shrink learning rate a little
            lr = max(lr * 0.85, 1e-6)
            no_improve += 1

        # ---- periodic learning‑rate decay ------------------------------------
        if t_iter % 500 == 0:
            lr = max(lr * 0.93, 1e-6)

        # ---- restart if we have been stuck for a long time -------------------
        if no_improve > 5000:
            print(f"[{t_iter:07d}] Restart (stalled).")
            a = best_seq + 0.01 * np.random.randn(n)
            a = np.maximum(a, 0.0)
            a = (a + a[::-1]) * 0.5
            a = _proj_simplex(a)
            cur_score = _eval_score(a)
            m.fill(0.0)
            v.fill(0.0)
            lr = 0.12
            alpha = 10.0
            no_improve = 0

    # ----------------------------------------------------------------------
    print(f"[done] best score found = {best_score:.6f}")
    # Return as plain Python list, as required by the evaluation harness
    return best_seq.tolist()
