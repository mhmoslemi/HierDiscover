"""
Propose a non‑negative step‑function (list of heights) with a low
evaluate_sequence score.

Algorithm
---------
1. Choose a working length `n`.  If a previous solution
   (`height_sequence_1`) exists we reuse its length, otherwise we use
   `n = 8000`.  The vector is always normalised to sum to 1 – the
   evaluation function is scale‑invariant.

2. Initialise the candidate with a power‑law shape a_i ∝ (i+1)^‑p,
   scanning a fine grid of exponents `p` and keeping the best (cheap
   FFT based) score.

3. Refine the candidate by *exponentiated‑gradient* descent on the
   smoothed objective

        F_α(a) = 2·n·log( Σ_k exp( α·(a*a)[k] ) ) / α

   (which converges to the true objective as α → ∞).  The gradient of
   the soft‑max is computed efficiently with three FFTs per iteration.
   The update is

        a_i ← a_i·exp(−η·∇F_α(a)_i) ,  then renormalise to Σa_i = 1.

   The step size η and the smoothing parameter α are increased
   gradually; a small amount of random perturbation is added from time
   to time to escape plateaus.

4. The best (lowest cheap) sequence encountered is finally evaluated
   with the provided ``evaluate_sequence`` and returned.

The routine respects the 1000 s time budget, prints progress messages,
and requires only NumPy (no external optimisation packages).
"""

# Helper utilities (top‑level, no closures)

def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex Σx=1, x≥0."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float) -> np.ndarray:
    """Return the normalised power‑law vector a_i ∝ (i+1)^‑p  (Σa=1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p)
    s = a.sum()
    if s == 0.0:
        a.fill(1.0 / n)
    else:
        a /= s
    return a


def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """
    Compute a*a (full linear convolution) via FFT.
    Returns a real array of length 2·len(a)‑1.
    """
    fa = np.fft.fft(a, L)
    conv = np.real(np.fft.ifft(fa * fa))
    n = a.shape[0]
    return conv[: 2 * n - 1]


def _gradient_via_fft(a: np.ndarray, w: np.ndarray, L: int) -> np.ndarray:
    """
    Gradient of the soft‑max weighted convolution:
        g_i = 2·∑_k w_k a_{k‑i}
    Computed by a single convolution in the Fourier domain.
    """
    n = a.shape[0]
    # reverse a for correlation
    a_rev = a[::-1]
    A_rev = np.fft.fft(a_rev, L)
    W = np.fft.fft(w, L)
    g_full = np.real(np.fft.ifft(W * A_rev))
    # extract the n relevant entries (indices n‑1 … 2n‑2)
    return 2.0 * g_full[n - 1 : n - 1 + n]

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) beats the
    target bound 1.503.  The routine respects the given time budget.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # 0. Length of the working vector
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)            # type: ignore[name-defined]
    except Exception:
        n = 8000                              # a reasonably large size

    # FFT length large enough for 3·n (needed for gradient) and a power of 2
    L = 1 << ((3 * n - 2).bit_length())

    # ------------------------------------------------------------------
    # 1. Initial power‑law scan (cheap, uses FFT)
    # ------------------------------------------------------------------
    p_grid = np.linspace(0.55, 0.70, 81)       # fine around the known optimum
    best_a = None
    best_score = float("inf")
    best_p = None
    for p in p_grid:
        if time.time() >= deadline:
            break
        a = _power_law_sequence(n, p)
        conv = _conv_fft(a, L)
        maxb = conv.max()
        score = 2.0 * n * maxb                # Σa = 1  ⇒ exact cheap score
        if score < best_score:
            best_score = score
            best_a = a.copy()
            best_p = p
    if best_a is None:                        # fall‑back, should never happen
        best_a = _power_law_sequence(n, 0.60)
        best_score = 2.0 * n * _conv_fft(best_a, L).max()
        best_p = 0.60

    print(f"[Init] n={n}, best p={best_p:.6f}, cheap score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Exponentiated‑gradient refinement with soft‑max smoothing
    # ------------------------------------------------------------------
    a = best_a.copy()                         # current iterate (Σa = 1)
    best_iter_a = a.copy()
    best_iter_score = best_score

    iter_cnt = 0
    eta0 = 0.30                               # initial step‑size factor
    alpha = 5.0                               # start with a very smooth max
    alpha_target = 2000.0
    alpha_inc = (alpha_target - alpha) / 8000.0   # reach α_target after ≈8000 iters
    # exponential‑gradient has no explicit momentum; we keep a simple schedule

    while time.time() < deadline and iter_cnt < 200_000:
        iter_cnt += 1

        # ----- convolution and cheap score ------------------------------
        conv = _conv_fft(a, L)
        maxb = conv.max()
        cheap_score = 2.0 * n * maxb

        if cheap_score < best_iter_score:
            best_iter_score = cheap_score
            best_iter_a = a.copy()
            print(f"[Iter {iter_cnt:06d}] NEW BEST cheap score = {cheap_score:.6f}")

        # ----- soft‑max weights (smooth approximation of the max) -------
        # stability: shift by maxb so that max exponent is 0
        w = np.exp(alpha * (conv - maxb))
        w_sum = w.sum()
        if w_sum == 0.0:                      # numerical safeguard
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed objective ----------------------
        grad = _gradient_via_fft(a, w, L)      # shape (n,)

        # ----- exponentiated‑gradient update ---------------------------
        eta = eta0 / sqrt(iter_cnt)           # diminishing schedule
        # clip the exponent to avoid overflow/underflow
        exponent = -eta * grad
        exponent = np.clip(exponent, -30.0, 30.0)
        a = a * np.exp(exponent)
        # Renormalise to the simplex (Σa = 1)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= a_sum

        # ----- slowly increase smoothing (move towards true max) -------
        if alpha < alpha_target:
            alpha = min(alpha_target, alpha + alpha_inc)

        # ----- occasional random perturbation to escape plateaus -------
        if iter_cnt % 2500 == 0 and np.random.rand() < 0.12:
            # add a tiny Dirichlet‑type noise and renormalise
            noise = 0.015 * np.random.rand(n)
            a = a + noise
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            a /= a_sum

    # ------------------------------------------------------------------
    # 3. Final exact evaluation (uses the provided evaluate_sequence)
    # ------------------------------------------------------------------
    final_seq = best_iter_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[Final] exact score = {final_score:.6f}")
    except Exception as exc:                  # defensive – should never happen
        final_score = float("inf")
        print(f"[Final] evaluation error: {exc}")

    return final_seq
