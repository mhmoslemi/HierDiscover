"""
Optimization of a step function to minimize the evaluation function.
We combine a parametric arcsine‑like initial family with Adam projected
gradient descent on the exact L∞ convolution objective.
"""

import time
import numpy as np

def _proj_simplex(v):
    """Projection onto the probability simplex."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u > cssv / np.arange(1, n+1))[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _convolve_fft(x, y):
    """Linear convolution using a real FFT."""
    n = x.size + y.size - 1
    size = 1 << (n - 1).bit_length()
    X = np.fft.rfft(x, size)
    Y = np.fft.rfft(y, size)
    Z = X * Y
    conv = np.fft.irfft(Z, size)[:n]
    return conv.real

def _eval_score(a):
    """Exact evaluation when a is already a non‑negative simplex vector."""
    n = a.size
    b = _convolve_fft(a, a)
    max_b = b.max()
    return float(2.0 * n * max_b / (a.sum() ** 2))

def _catalan_sequence(n):
    """Catalan‑like coefficients a_i = C(2i,i)/4**i, normalised to sum 1."""
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i-1] * (2.0 * (i-1) + 1) / (2.0 * i)
    a /= a.sum()
    return a

def _arc_shape(n, gamma, beta):
    """Arcsine‑like shape: a_i ∝ ((i+β)*(n‑1‑i+β))^{-γ}."""
    i = np.arange(n, dtype=float)
    denom = (i + beta) * (n - 1 - i + beta)
    # ensure positivity, avoid zero
    a = np.power(denom, -gamma)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones(n) / n
    else:
        a = a / total
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Uses a simple parametric family (arcsine‑like) to get a good start and
    then runs a projected Adam optimiser on the exact L∞ objective.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin for cleanup

    # problem size – a few thousand points are enough to capture the shape
    n = 4000

    # ------------------------------------------------------------------
    # 1️⃣  Build a pool of initial candidates and keep the best one.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # Catalan seed
    cand = _catalan_sequence(n)
    sc = _eval_score(cand)
    if sc < best_score:
        best_seq, best_score = cand.copy(), sc

    # Random simplex point
    rnd = np.random.rand(n)
    cand = _proj_simplex(rnd)
    sc = _eval_score(cand)
    if sc < best_score:
        best_seq, best_score = cand.copy(), sc

    # Grid over the arcsine‑like family
    gammas = np.linspace(0.35, 0.85, 11)   # 0.35 … 0.85
    betas = np.linspace(0.5, 3.0, 6)      # 0.5 … 3.0
    for gamma in gammas:
        for beta in betas:
            cand = _arc_shape(n, gamma, beta)
            sc = _eval_score(cand)
            if sc < best_score:
                best_seq, best_score = cand.copy(), sc

    # Keep the best candidate as the current iterate
    a = best_seq.copy()
    cur_score = best_score

    print(f"[init] n={n}, initial best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Adam optimiser on the exact L∞ objective.
    # ------------------------------------------------------------------
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.05               # base learning rate
    iteration = 0
    no_improve = 0

    while time.time() < deadline:
        # --- convolution of a with itself ------------------------------
        b = _convolve_fft(a, a)
        max_b = b.max()

        # --- sub‑gradient of the infinity norm -----------------------
        # Choose all entries that are (practically) maximal.
        # A tiny tolerance avoids numerical glitches.
        tol = max_b * 1e-12 + 1e-15
        idx_max = np.where(b >= max_b - tol)[0]
        w = np.zeros_like(b)
        w[idx_max] = 1.0 / len(idx_max)

        # gradient = 2 * (a ⊛ w_rev) restricted to the original length
        w_rev = w[::-1]
        conv_full = _convolve_fft(a, w_rev)          # length = 3n‑2
        idx_slice = (len(b) - 1) - np.arange(n)      # maps to size n
        grad = 2.0 * conv_full[idx_slice]

        # --- Adam update ---------------------------------------------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = _proj_simplex(a_new)          # keep feasibility

        # --- evaluate -------------------------------------------------
        new_score = _eval_score(a_new)

        # --- accept if we made progress -------------------------------
        if new_score < cur_score:
            cur_score = new_score
            a = a_new
            if new_score < best_score:
                best_score = new_score
                best_seq = a_new.copy()
                print(f"[{iteration:06d}] new best {best_score:.6f} (lr={lr:.4g})")
            no_improve = 0
        else:
            # keep moving – Adam already performed a step – but reset
            # the momentum if we have been stuck for a while.
            a = a_new
            no_improve += 1

        # --- learning‑rate schedule ----------------------------------
        if iteration % 2500 == 0:
            lr *= 0.9
            lr = max(lr, 1e-5)

        # --- occasional random restart to escape flat plateaus --------
        if no_improve > 5000:
            # perturb around the best solution found so far
            a = best_seq + 0.02 * np.random.randn(n)
            a = np.maximum(a, 0.0)
            a = _proj_simplex(a)
            cur_score = _eval_score(a)
            m.fill(0.0)
            v.fill(0.0)
            lr = 0.05
            no_improve = 0
            print(f"[{iteration:06d}] restart around best (score={cur_score:.6f})")

    print(f"[done] best score found = {best_score:.6f}")
    # Return a plain Python list as required by the harness.
    return best_seq.tolist()

"""
Optimise a non‑negative step‑function to minimise
``evaluate_sequence``.  

The algorithm proceeds in three stages:

1️⃣  Build a small library of “hand‑crafted” seeds (Catalan,
    uniform, exponential, arcsine‑like) and a set of random
    mixtures of them.  The best seed (by the exact objective
    2·n·max(a∗a)) is kept.

2️⃣  Run a projected Adam optimiser directly on the L∞‑objective.
    The sub‑gradient of max(a∗a) is obtained from the indices that
    attain the current maximum.  After each Adam step the vector is
    projected onto the probability simplex (non‑negative, sum = 1).

3️⃣  If the optimiser stalls a small random perturbation around the
    best solution is performed (restart).  The loop runs until the
    time budget (≈ 1000 s) is exhausted and the best sequence found
    is returned as a plain list of floats.

The whole search is performed with FFT‑based convolutions (O(n log n))
so that even a few thousand iterations are cheap.  The default problem
size is n = 4000 – large enough to capture the asymptotic shape but
still very fast to handle.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, sum=1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    # Fast O(n log n) projection (Duchi et al., 2008)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u > cssv / np.arange(1, n + 1))[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of real vectors x and y using a real FFT."""
    n = x.size + y.size - 1
    size = 1 << (n - 1).bit_length()          # next power of two
    X = np.fft.rfft(x, size)
    Y = np.fft.rfft(y, size)
    Z = X * Y
    conv = np.fft.irfft(Z, size)[:n]
    return conv.real


def _eval_score(a: np.ndarray) -> float:
    """Exact value of ``evaluate_sequence`` for a non‑negative simplex vector."""
    n = a.shape[0]
    b = _convolve_fft(a, a)
    max_b = b.max()
    # sum(a) is 1 by construction, but we keep the formula for safety
    return float(2.0 * n * max_b / (a.sum() ** 2))


def _catalan_sequence(n: int) -> np.ndarray:
    """Catalan‑like coefficients a_i = C(2i,i)/4**i, normalised to sum = 1."""
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * (i - 1) + 1) / (2.0 * i)
    a /= a.sum()
    return a


def _geometric_sequence(n: int, r: float) -> np.ndarray:
    """Truncated geometric distribution (sum = 1)."""
    a = (1.0 - r) * (r ** np.arange(n, dtype=float))
    a_sum = a.sum()
    if a_sum == 0:
        a = np.ones(n) / n
    else:
        a /= a_sum
    return a


def _arc_shape(n: int, gamma: float, beta: float) -> np.ndarray:
    """Arcsine‑like shape: a_i ∝ ((i+β)*(n‑1‑i+β))^{-γ}."""
    i = np.arange(n, dtype=float)
    denom = (i + beta) * (n - 1 - i + beta)
    a = np.power(denom, -gamma)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones(n) / n
    else:
        a /= total
    return a


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence (list of floats) found before the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # tiny safety margin

    # --------------------------------------------------------------
    # Problem size – a few thousand points give a good trade‑off.
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # 1️⃣  Build a pool of candidate seeds.
    # --------------------------------------------------------------
    candidates = []

    # Catalan seed
    candidates.append(_catalan_sequence(n))

    # Uniform seed
    candidates.append(np.ones(n) / n)

    # A geometric decay (moderately slow)
    candidates.append(_geometric_sequence(n, r=0.9995))

    # Arcsine‑like shape (γ=0.5, β=0.5 – the classical arcsine pdf)
    candidates.append(_arc_shape(n, gamma=0.5, beta=0.5))

    # --------------------------------------------------------------
    # Random mixtures of the above basis vectors.
    # --------------------------------------------------------------
    basis = candidates.copy()
    n_basis = len(basis)
    # generate a few hundred random convex combinations
    for _ in range(1200):
        w = np.random.rand(n_basis)
        mix = np.zeros(n, dtype=float)
        for j in range(n_basis):
            mix += w[j] * basis[j]
        mix /= mix.sum()
        candidates.append(mix)

    # --------------------------------------------------------------
    # Choose the best initial point according to the exact objective.
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    for a in candidates:
        sc = _eval_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
    a = best_seq.copy()
    cur_score = best_score
    print(f"[init] n={n}, initial best score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Projected Adam optimiser on the L∞‑objective.
    # --------------------------------------------------------------
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.05                     # base learning rate for Adam
    iteration = 0
    no_improve = 0

    while time.time() < deadline:
        # ---- convolution a∗a -------------------------------------------------
        b = _convolve_fft(a, a)
        max_b = b.max()

        # ---- sub‑gradient of max(b) (indices attaining the maximum) ----------
        # a tiny tolerance avoids missing numerically equal entries
        tol = max_b * 1e-12 + 1e-15
        max_idx = np.where(b >= max_b - tol)[0]
        w = np.zeros_like(b)
        w[max_idx] = 1.0 / len(max_idx)

        # gradient of the (non‑smooth) max = 2·(a ⊛ w_rev) restricted to length n
        w_rev = w[::-1]
        conv_full = _convolve_fft(a, w_rev)                # length = 3n‑2
        idx_slice = (len(b) - 1) - np.arange(n)            # map to size n
        grad = 2.0 * conv_full[idx_slice]

        # ---- Adam update ----------------------------------------------------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = _proj_simplex(a_new)          # enforce feasibility

        # ---- evaluate -------------------------------------------------------
        new_score = _eval_score(a_new)

        # ---- accept if improvement -----------------------------------------
        if new_score < cur_score:
            cur_score = new_score
            a = a_new
            if new_score < best_score:
                best_score = new_score
                best_seq = a_new.copy()
                print(f"[{iteration:06d}] new best {best_score:.6f} (lr={lr:.4g})")
            no_improve = 0
        else:
            # keep moving (Adam already performed a step) but count stagnation
            a = a_new
            no_improve += 1

        # ---- learning‑rate decay -------------------------------------------
        if iteration % 2500 == 0:
            lr = max(lr * 0.9, 1e-5)

        # ---- occasional restart to break out of flat plateaus --------------
        if no_improve > 5000:
            # perturb around the best solution so far
            a = best_seq + 0.02 * np.random.randn(n)
            a = np.maximum(a, 0.0)
            a = _proj_simplex(a)
            cur_score = _eval_score(a)
            m.fill(0.0)
            v.fill(0.0)
            lr = 0.05
            no_improve = 0
            print(f"[{iteration:06d}] restart around best (score={cur_score:.6f})")

    print(f"[done] best score found = {best_score:.6f}")
    # return as plain Python list, as required by the harness
    return best_seq.tolist()
