from math import sqrt

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the probability simplex:
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def power_law_sequence(n: int, p: float) -> np.ndarray:
    """Return a normalised power‑law vector a_i ∝ (i+1)⁻ᵖ with Σa_i = 1."""
    seq = np.fromiter(((i + 1) ** (-p) for i in range(n)), dtype=np.float64)
    seq /= seq.sum()
    return seq


def convolution_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a (real part) using FFT, length 2·n‑1."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa).real[: 2 * a.size - 1]
    return conv


def weighted_gradient(a: np.ndarray, w: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Gradient of  Σ_k w_k (a*a)[k] w.r.t. a.
    Implements   ∇ = 2 * (w ⋆ a)   (cross‑correlation).
    """
    n = a.shape[0]
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    corr = np.fft.ifft(fw * fa_rev).real
    # slice the part that corresponds to indices 0 … n‑1
    grad = 2.0 * corr[n - 1 : n - 1 + n]
    return grad


def subgradient_max(a: np.ndarray, max_idx: int) -> np.ndarray:
    """
    Sub‑gradient of the active convolution entry
    (a*a)[max_idx]  w.r.t. a.
    """
    n = a.shape[0]
    lo = max(0, max_idx - (n - 1))
    hi = min(max_idx, n - 1)
    i = np.arange(lo, hi + 1, dtype=np.int64)
    g = np.zeros_like(a)
    g[i] = 2.0 * a[max_idx - i]
    return g

    # 2. Scan a fine grid of exponents p for a power‑law warm‑start.
    # ------------------------------------------------------------------
    p_grid = np.linspace(0.55, 0.68, 141)          # dense grid around known optimum
    best_seq = None
    best_score = float("inf")
    best_p = None

    for p in p_grid:
        if time.time() >= deadline:
            break
        a = power_law_sequence(n, p)
        conv = convolution_fft(a, fft_len)
        max_b = float(conv.max())
        score = 2.0 * n * max_b                     # exact score because Σa = 1
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            best_p = p

    if best_seq is None:
        # should never happen – fallback to p = 0.60
        best_seq = power_law_sequence(n, 0.60)
        best_score = 2.0 * n * convolution_fft(best_seq, fft_len).max()
        best_p = 0.60

    print(f"Initial power‑law scan: best p = {best_p:.5f}, cheap score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Smoothed max‑gradient descent with Adam and simplex projection.
    # ------------------------------------------------------------------
    a = best_seq.copy()                # current iterate (always Σa = 1)
    best_iter_seq = a.copy()
    best_iter_score = best_score

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    iter_count = 0
    beta0 = 0.2                     # start of soft‑max temperature
    beta_max = 8.0                  # final temperature (sharp approx. of max)
    lr = 0.001                      # base learning rate for Adam

    while time.time() < deadline:
        iter_count += 1

        # ---- convolution and active information --------------------------------
        conv = convolution_fft(a, fft_len)
        max_b = conv.max()

        # ---- temperature schedule ------------------------------------------------
        progress = (time.time() - start) / (deadline - start)   # 0 … 1
        beta = beta0 + (beta_max - beta0) * progress

        # ---- compute a descent direction ----------------------------------------
        if beta <= 0.0:
            # pure sub‑gradient (fallback)
            max_idx = int(conv.argmax())
            grad = subgradient_max(a, max_idx)
        else:
            # soft‑max weighting of all convolution entries
            # numerical stability: subtract the maximal value before exponentiation
            w = np.exp(beta * (conv - max_b))
            w_sum = w.sum()
            if w_sum <= 0.0 or not np.isfinite(w_sum):
                # fall back to the active sub‑gradient
                max_idx = int(conv.argmax())
                grad = subgradient_max(a, max_idx)
            else:
                w /= w_sum
                grad = weighted_gradient(a, w, fft_len)

        # ---- Adam update --------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iter_count)
        v_hat = v / (1.0 - beta2 ** iter_count)

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_candidate = a - step                     # move opposite to gradient
        a = simplex_projection(a_candidate)        # stay on the simplex

        # ---- occasional cheap evaluation ----------------------------------------
        if iter_count % 500 == 0:
            conv_check = convolution_fft(a, fft_len)
            score = 2.0 * n * conv_check.max()
            if score < best_iter_score:
                best_iter_score = score
                best_iter_seq = a.copy()
                print(f"[Iter {iter_count:06d}] NEW BEST cheap score = {score:.6f}")

        # ---- random jitter to escape plateaus -----------------------------------
        if iter_count % 5000 == 0 and np.random.rand() < 0.15:
            perturb = 1e-3 * np.random.rand(n)
            a = simplex_projection(a + perturb)

        # ---- early stop if we are about to exceed the deadline --------------------
        if time.time() > deadline - 5.0:
            break

    except Exception as exc:                     # defensive – should never happen
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # ------------------------------------------------------------------
    # 5. Return the best sequence found.
    # ------------------------------------------------------------------
    return final_seq
