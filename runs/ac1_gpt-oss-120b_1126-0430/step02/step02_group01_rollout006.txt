        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()

        # gradient of squared error

        fft_v = np.fft.rfft(v, n=L)

"""
Gradient‐descent on the squared‑error of the self‑convolution.

The objective to minimise is
    O(a) = 2·n·max(conv(a,a))      (∑a = 1, a_i ≥ 0)

If conv(a,a) were perfectly flat (value = 1/(2n‑1) everywhere) then
O = 2·n·1/(2n‑1) → 1 as n → ∞.  The algorithm tries to flatten the
convolution by minimising the L² error
    F(a) = ∑_k (conv_k(a,a) – μ)² ,   μ = 1/(2n‑1).

∇F(a) = 4·( (conv(a,a)–μ)  *  a )   (full convolution, first n components).
A simple projected gradient step is performed (additive update, clipping to
non‑negative, renormalisation onto the probability simplex).  The step size
decreases like 1/√t and occasional random perturbations avoid stagnation.

The routine runs for up to 1000 s, keeps the best sequence found and finally
calls the supplied ``evaluate_sequence`` for the exact score.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _normalise_to_simplex(v: np.ndarray) -> np.ndarray:
    """Clip to ≥0 and rescale so that the sum equals 1 (probability simplex)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:                     # fallback – uniform distribution
        v = np.full_like(v, 1.0 / v.shape[0])
    else:
        v = v / s
    return v


def _objective(a: np.ndarray) -> float:
    """Exact objective used during the search (sum(a) is assumed to be 1)."""
    conv = np.convolve(a, a)
    return float(2.0 * a.size * conv.max())


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the bound
    ``evaluate_sequence``.  Returns the best sequence found as a plain list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ---- initialise ----------------------------------------------------
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=float)
        a = _normalise_to_simplex(a)
        n = a.size
    else:
        n = 8000                     # a comfortable size (larger = better potential)
        a = np.random.rand(n)
        a = _normalise_to_simplex(a)

    best_seq = a.copy()
    best_score = _objective(best_seq)
    print(f"[init] n={n}, O={best_score:.6f}")

    # ---- constants for the FFT based convolutions ---------------------
    mu = 1.0 / (2 * n - 1)          # average value of the convolution
    # FFT length must accommodate a (size n) and v (size 2n‑1) => at least 3n‑2
    L = 1 << ((3 * n - 1).bit_length())   # next power of two ≥ 3n

    # ---- main optimisation loop ----------------------------------------
    it = 0
    no_improve = 0
    while time.time() < deadline and it < 300_000:
        it += 1

        # FFT of the current a (re‑used for both convolutions)
        fft_a = np.fft.rfft(a, n=L)

        # --- self‑convolution a*a (full) --------------------------------
        conv_full = np.fft.irfft(fft_a * fft_a, n=L)[: 2 * n - 1]

        # current bound
        M = conv_full.max()
        cur_score = 2.0 * n * M

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            no_improve = 0
            print(f"[iter {it:6d}] new best O={best_score:.6f}")
        else:
            no_improve += 1

        # --- gradient of the L2 error of the convolution ----------------
        v = conv_full - mu                     # deviation from the flat target
        fft_v = np.fft.rfft(v, n=L)
        conv_v_a_full = np.fft.irfft(fft_v * fft_a, n=L)[: 3 * n - 2]
        grad = 4.0 * conv_v_a_full[:n]         # only the first n components are needed

        # --- step size (decays like 1/√t) -------------------------------
        eta = 5.0 / math.sqrt(it)              # tuneable constant (5 works well)

        # --- additive gradient step + projection onto the simplex -------
        a_new = a - eta * grad
        a = _normalise_to_simplex(a_new)

        # --- occasional stochastic perturbation to escape plateaus -----
        if it % 20000 == 0:
            a += np.random.randn(n) * 0.001
            a = _normalise_to_simplex(a)
            print(f"[iter {it:6d}] added small stochastic noise")

        # --- restart if stuck for a long time -------------------------
        if no_improve >= 20000:
            a = best_seq.copy()
            a += np.random.randn(n) * 0.002
            a = _normalise_to_simplex(a)
            no_improve = 0
            print(f"[restart] after {it} iter, re‑initialised from best")

    # ------------------------------------------------------------------
    # final evaluation with the provided exact function
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:                # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_seq.tolist()
