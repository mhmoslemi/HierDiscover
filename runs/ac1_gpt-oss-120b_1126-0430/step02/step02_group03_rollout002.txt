def propose_candidate(seed=0, budget_s=1000):
    import time, numpy as np, scipy.sparse as sp, scipy.optimize as opt

    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0

    n = 2000  # can increase if time permits
    # initial shape: Catalan
    a = _catalan_normalised(n)

    best_a = a.copy()
    best_score = 2.0 * n * np.max(np.convolve(a, a))
    print("[init] score:", best_score)

    iteration = 0
    t_initial = 0.3

    while time.time() < deadline:
        iteration += 1
        # compute convolution
        conv = np.convolve(a, a)
        M = conv.max()
        score = 2.0 * n * M
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[iter {iteration}] new best score {best_score:.9f}")

        # Build LP constraints
        rows = 2*n - 1
        # using sparse matrix
        A = sp.lil_matrix((rows, n), dtype=np.float64)
        # fill
        for j in range(n):
            A[j:j + n, j] = a
        A = A.tocsr()

        # solve LP
        c = -np.ones(n)  # maximize sum(g)
        bounds = [(0, None)] * n
        b_ub = np.full(rows, M, dtype=np.float64)
        # use high-speed solver
        res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            g = res.x
            # ensure nonnegative
            # res may produce small negative due to tolerance: clip
            g = np.maximum(g, 0.0)
            # normalize g to sum 1
            sum_g = g.sum()
            if sum_g > 0:
                g = g / sum_g
            else:
                # fallback: random simplex point
                g = np.random.rand(n)
                g = g / g.sum()
        else:
            print("[LP] failed, breaking")
            break

        # mixing factor t decreasing
        t = max(0.1, t_initial / (1 + iteration * 0.05))
        a = (1 - t) * a + t * g
        # re-normalize to sum 1 (should be already)
        a = a / a.sum()

        # Optional early exit if improvement negligible
        if time.time() > deadline - 5.0:
            break
    # final best
    final_score = 2.0 * n * np.max(np.convolve(best_a, best_a))
    print("[final] best score approx", final_score)
    return best_a.tolist()

convolution.  After each LP we normalise g to the same ℓ¹‑norm as a and
take a convex combination a←(1‑t)a+t g, decreasing t when the step does not
lower M.  This is exactly the scheme described in the reference
“state‑of‑the‑art’’ paragraph, but here we implement it with SciPy’s
high‑performance LP solver and a simple back‑tracking line search.
A few cheap projected sub‑gradient refinements are performed when time
permits.

The routine runs for at most ``budget_s`` seconds and returns the best
sequence discovered.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt


# ----------------------------------------------------------------------
# Fast linear convolution (real FFT) – used for the score and for the
# optional sub‑gradient refinement.
# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Catalan numbers C(2i,i)/4**i, normalised to ℓ¹‑norm 1.
# ----------------------------------------------------------------------
def _catalan_normalised(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Build the (2n‑1)×n matrix that maps g↦a∗g (linear convolution).
# Row k, column j contains a[k‑j] if 0≤k‑j<n, otherwise 0.
# ----------------------------------------------------------------------
def _conv_matrix(a: np.ndarray) -> sp.csr_matrix:
    n = a.shape[0]
    rows = 2 * n - 1
    # each diagonal  (offset i) is constant a[i]
    diags = [np.full(n, a[i], dtype=np.float64) for i in range(n)]
    offsets = np.arange(n, dtype=int)
    return sp.diags(diags, offsets, shape=(rows, n), format="csr")


# ----------------------------------------------------------------------
# Tiny projected sub‑gradient refinement (optional, cheap).
# ----------------------------------------------------------------------
def _refine_subgradient(a: np.ndarray, iters: int, deadline: float) -> np.ndarray:
    n = a.size
    lr = 0.02
    best = a.copy()
    best_score = 2.0 * n * np.max(_fft_convolve(a))

    for it in range(iters):
        if time.time() >= deadline:
            break
        conv = _fft_convolve(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient of max(conv) w.r.t. a
        grad = np.zeros_like(a)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        grad[idx] = 2.0 * a[k_max - idx]

        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 0.0:
            a_new /= s
        else:
            a_new = np.random.rand(n)
            a_new /= a_new.sum()

        new_score = 2.0 * n * np.max(_fft_convolve(a_new))
        if new_score < best_score:
            best_score = new_score
            best = a_new.copy()
            print(f"[refine] iter {it:04d}  score {best_score:.9f}")
            lr *= 1.02
        else:
            lr *= 0.98
        a = a_new
    return best


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list:
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    The method uses an LP‑based iterative improvement of the Catalan
    distribution and a tiny sub‑gradient polishing step.

    Parameters
    ----------
    seed : int, optional
        Random seed.
    budget_s : int, optional
        Time budget in seconds (default 1000).

    Returns
    -------
    list[float]
        The best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # 1. problem size – larger n gives more freedom but makes the LP
    #    slower.  For the allotted time n≈2500 works well.
    # ------------------------------------------------------------------
    n = 2500
    target_sum = np.sqrt(2.0 * n)                     # we keep Σa = √(2n)

    # ------------------------------------------------------------------
    # 2. initialise with the (scaled) Catalan sequence.
    # ------------------------------------------------------------------
    a = _catalan_normalised(n) * target_sum          # Σa = target_sum

    # keep the best sequence seen so far (score = max convolution)
    conv = _fft_convolve(a)
    M = conv.max()
    best_a = a.copy()
    best_score = M
    print(f"[init] n={n}, score≈{best_score:.9f}")

    # ------------------------------------------------------------------
    # 3. main LP‑based improvement loop.
    # ------------------------------------------------------------------
    t = 0.25                     # initial mixing weight
    min_t = 0.02
    iteration = 0

    while time.time() < deadline:
        iteration += 1

        # ----- evaluate current candidate -----
        conv = _fft_convolve(a)
        M = conv.max()
        if M < best_score:
            best_score = M
            best_a = a.copy()
            print(f"[iter {iteration:04d}] new best M={best_score:.9f}")

        # ----- build linear constraints a∗g ≤ M -----
        A = _conv_matrix(a)                     # (2n‑1)×n sparse matrix
        rows = 2 * n - 1
        b_ub = np.full(rows, M, dtype=np.float64)

        # ----- solve LP: maximise Σ g subject to A·g ≤ M, g ≥ 0 -----
        c = -np.ones(n)                         # minimise –Σg  ⇒ maximise Σg
        bounds = [(0, None)] * n
        try:
            res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                               method='highs', options={'presolve': True})
        except Exception as e:
            print("[LP] exception:", e)
            break

        if not res.success:
            print("[LP] failed:", res.message)
            break

        g = res.x
        # Clip possible tiny negatives caused by numeric tolerances
        g[g < 0.0] = 0.0
        g_sum = g.sum()
        if g_sum == 0.0:
            # fallback – random point on the simplex
            g = np.random.rand(n)
            g /= g.sum()
        else:
            # rescale to the same ℓ¹‑norm as a
            g *= target_sum / g_sum

        # ----- line‑search / backtracking on the mixing factor -----
        accepted = False
        t_try = t
        while t_try >= min_t:
            a_candidate = (1.0 - t_try) * a + t_try * g
            # re‑normalise explicitly (numerical drift)
            a_candidate *= target_sum / a_candidate.sum()
            M_candidate = _fft_convolve(a_candidate).max()
            if M_candidate < M:
                a = a_candidate
                M = M_candidate
                accepted = True
                # good step – we can be a little bolder next time
                t = min(0.5, t_try * 1.1)
                break
            else:
                # step too aggressive – shrink t_try
                t_try *= 0.5
        if not accepted:
            # couldn't improve – lower the global step size
            t = max(min_t, t * 0.7)
            # If the step size is already tiny, give up early
            if t <= min_t and iteration > 10:
                print("[loop] step size exhausted, terminating early")
                break

        # optional quick exit if no more time
        if time.time() > deadline - 5.0:
            break

    # ------------------------------------------------------------------
    # 4. optional projected sub‑gradient polishing (cheap, improves a few 1e‑4)
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        a = _refine_subgradient(best_a, iters=150, deadline=deadline)

        # re‑evaluate after polishing
        final_M = _fft_convolve(a).max()
        if final_M < best_score:
            best_score = final_M
            best_a = a.copy()
            print(f"[polish] improved to M={best_score:.9f}")

    # ------------------------------------------------------------------
    # 5. report and return the best sequence found
    # ------------------------------------------------------------------
    print(f"[final] best M≈{best_score:.9f} (score = {best_score:.9f})")
    return best_a.tolist()
