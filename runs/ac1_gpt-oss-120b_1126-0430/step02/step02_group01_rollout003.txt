def fft_convolve(x, y, size=None):
    n = len(x) + len(y) - 1
    sz = 1 << (n - 1).bit_length()  # next power of two
    if size is not None: sz = max(sz, size)
    fx = np.fft.rfft(x, sz)
    fy = np.fft.rfft(y, sz)
    conv = np.fft.irfft(fx * fy, sz)[:n]
    return conv

    n = x.size + y.size - 1
    size = 1 << (n - 1).bit_length()
    X = np.fft.rfft(x, size)

    conv = np.fft.irfft(X * Y, size)[:n]

def _smooth_gradient(a: np.ndarray, beta: float) -> tuple[np.ndarray, float]:
    """Return gradient of smooth approximation of max(conv) and current max conv."""
    conv = _fft_convolve_real(a, a)
    max_c = conv.max()
    # compute softmax weights
    w = np.exp(beta * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum
    # compute gradient
    rev_w = w[::-1]
    conv_rev = _fft_convolve_real(a, rev_w)
    L = 2 * a.size - 2
    grad = 2.0 * conv_rev[L - np.arange(a.size)]
    return grad, max_c
```

"""
Search for a non‑negative step‑function (probability vector) minimizing

    O = 2·n·max(conv(a,a))    with    sum(a)=1 , a_i ≥ 0

The evaluation performed by ``evaluate_sequence`` is exactly this expression
once the vector is normalised to sum 1.  The algorithm consists of two
phases:

* **Phase 1 – Smoothed gradient descent**  
  We approximate the non‑smooth max with a softmax (temperature β).  An
  Adam optimiser updates the vector and after each update a projection onto
  the probability simplex keeps the feasibility constraints.  β is slowly
  increased, so the method first moves quickly (large smoothing) and later
  behaves like a sub‑gradient descent for the true max.

* **Phase 2 – Stochastic hill‑climbing**  
  Starting from the best vector found in phase 1 we perform a simulated‑
  annealing style random mass transfer between coordinates and occasional
  exact sub‑gradient steps on the true max‑convolution.  This fine‑tunes
  the solution.

All helper functions are top‑level; the search stops after the allotted
time budget (default 1000 s) and returns the best sequence found.
Progress and the final exact score obtained from ``evaluate_sequence`` are
printed.
"""

import time
import math
import numpy as np

# -------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, Σ x_i = 1}
    Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        # all entries are zero or negative – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# -------------------------------------------------------------
def _fft_convolve_real(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Real‑valued convolution using FFT.
    Returns the linear convolution of x and y (length len(x)+len(y)-1).
    """
    n = x.size + y.size - 1
    size = 1 << (n - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:n]
    # Numerical noise may produce tiny negative values – clip them
    conv = np.clip(conv, 0.0, None)
    return conv


# -------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) assuming sum(a)=1.
    """
    conv = _fft_convolve_real(a, a)
    return 2.0 * a.size * conv.max()


# -------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, beta: float) -> tuple[np.ndarray, float]:
    """
    Gradient of a smoothed version of max(conv(a,a)).
    Returns (grad, max_conv) where
        grad = ∂/∂a  ( 1/β log Σ_k exp(β·conv_k) )
    and conv_k are the entries of a*a.
    """
    n = a.size
    conv = _fft_convolve_real(a, a)          # length 2n-1
    max_c = conv.max()
    # soft‑max weights (stable because we subtract max_c)
    w = np.exp(beta * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # gradient: g_i = 2 * Σ_k w_k * a_{k-i}
    rev_w = w[::-1]
    conv_rev = _fft_convolve_real(a, rev_w)   # length 3n-2
    L = 2 * n - 2
    grad = 2.0 * conv_rev[L - np.arange(n)]
    return grad, max_c


# -------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the true max‑convolution at a.
    Averages the gradients of all indices attaining the maximum.
    """
    n = a.size
    conv = _fft_convolve_real(a, a)
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)
    subg = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        jdx = k - idx
        subg[idx] += 2.0 * a[jdx]
    subg /= len(ks)
    return subg


# -------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises the bound.
    Returns the best sequence found as a plain list of floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # a small safety margin

    # ----- initialise -------------------------------------------------
    init_seq = globals().get('height_sequence_1')
    if init_seq is not None:
        a = np.asarray(init_seq, dtype=float)
        if a.size == 0:
            raise ValueError("height_sequence_1 is empty")
        a = a / a.sum()
        n = a.size
        print("[init] Using provided height_sequence_1")
    else:
        # pick a length that gives enough degrees of freedom but stays fast
        n = 5000
        a = np.random.rand(n)
        a = _project_to_simplex(a)
        print(f"[init] Random start, n={n}")

    best_seq = a.copy()
    best_obj = _objective(best_seq)
    print(f"[init] Objective = {best_obj:.6f}")

    # ----- Phase 1: Smoothed gradient descent (Adam) -----------------
    beta = 0.5
    beta_max = 200.0
    beta_factor = 1.2
    lr = 0.05                     # base learning rate for Adam
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    t = 0
    iter_g = 0

    while time.time() < deadline:
        iter_g += 1
        t += 1
        grad, max_c = _smooth_gradient(a, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = _project_to_simplex(a_new)

        # evaluate new point
        obj_new = _objective(a_new)

        # accept if we made progress (or very slight deterioration)
        if obj_new <= best_obj * 1.001:
            a = a_new
            if obj_new < best_obj:
                best_obj = obj_new
                best_seq = a.copy()
                print(f"[grad {iter_g}] new best O={best_obj:.6f}")

        # increase β every few hundred iterations
        if iter_g % 300 == 0:
            old_beta = beta
            beta = min(beta * beta_factor, beta_max)
            if beta != old_beta:
                print(f"[beta] increased to {beta:.2f}")

        # stop early if we have already reached a very good bound
        if best_obj < 1.50 and (deadline - time.time()) > 30.0:
            # keep a little time for the stochastic phase
            pass

        if time.time() >= deadline:
            break

    # ----- Phase 2: Stochastic hill‑climbing -------------------------
    cur_a = best_seq.copy()
    cur_obj = best_obj
    T0 = 0.04
    T = T0
    iter_hc = 0
    max_iter_hc = 300_000
    no_improve = 0

    while time.time() < deadline and iter_hc < max_iter_hc:
        iter_hc += 1

        # --- random mass transfer ------------------------------------
        i, j = np.random.randint(n, size=2)
        if i != j and cur_a[i] > 0.0:
            delta = cur_a[i] * np.random.rand() * 0.12
            cand = cur_a.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = _project_to_simplex(cand)

            cand_obj = _objective(cand)

            # Metropolis acceptance
            if cand_obj < cur_obj or np.random.rand() < math.exp((cur_obj - cand_obj) / max(T, 1e-12)):
                cur_a = cand
                cur_obj = cand_obj
                no_improve = 0
                if cur_obj < best_obj:
                    best_obj = cur_obj
                    best_seq = cur_a.copy()
                    print(f"[hc {iter_hc}] new best O={best_obj:.6f}")
            else:
                no_improve += 1

        # --- occasional exact sub‑gradient step ----------------------
        if iter_hc % 13 == 0:
            subg = _subgradient_max_conv(cur_a)
            eta = 0.018 * (1.0 - iter_hc / max_iter_hc)   # slowly decaying step
            cur_a = cur_a - eta * subg
            cur_a = np.clip(cur_a, 0.0, None)
            cur_a = _project_to_simplex(cur_a)
            cur_obj = _objective(cur_a)
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_seq = cur_a.copy()
                print(f"[sg {iter_hc}] new best O={best_obj:.6f}")

        # --- temperature decay ---------------------------------------
        T = T0 * max(0.0, 1.0 - iter_hc / max_iter_hc)

        # --- restart if stuck ----------------------------------------
        if no_improve >= 5000:
            cur_a = best_seq.copy()
            cur_a += np.random.randn(n) * 0.003
            cur_a = np.clip(cur_a, 0.0, None)
            cur_a = _project_to_simplex(cur_a)
            cur_obj = _objective(cur_a)
            no_improve = 0
            print(f"[restart] after {iter_hc} iterations, re‑initialised")

    # ----- final exact evaluation ------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:  # pragma: no cover
        print("[final] evaluate_sequence call failed:", e)

    return best_seq.tolist()
