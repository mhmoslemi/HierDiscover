"""
Propose a candidate step‑function (list of non‑negative heights) that minimises
evaluate_sequence.

Algorithm
---------
1. Work with a moderate size n_small (≈2000) where a linear‑programming
   formulation is tractable.  Initialise the sequence with the known
   “≈1/√i’’ shape (the analytic solution giving a value π/2 ≈ 1.5708).
2. Iteratively improve the small‑size sequence by the classic “balancing”
   linear‑program:

      • Given a current f (∑f = √(2·n_small)), compute M = max(f∗f).
      • Solve the LP   maximise 1ᵀg   s.t.  g ≥ 0  and  (f∗g)_k ≤ M  ∀k .
        The constraints are linear in g (a Toeplitz matrix built from f).
      • Scale g to the same ℓ₁‑norm √(2·n_small) and replace
        f ← (1‑t)·f + t·g  (t starts at 0.5 and shrinks slowly).
      • Repeat until M stops decreasing (≈10 iterations).

   The LP is solved with SciPy’s high‑performance “highs’’ method; the
   constraint matrix is constructed once and reused (only the data values
   change each iteration).

3. Upsample the improved small‑size vector to the target size n_large
   (8000) by linear interpolation and renormalise it to a probability
   simplex (∑a = 1).  The scaling does not alter the objective because
   evaluate_sequence is homogeneous of degree 0.

4. Refine the large‑size vector with a projected sub‑gradient method:
   – compute the exact sub‑gradient of max(f∗f) (weights are 1 on all
     maximal convolution entries),
   – take a step with a slowly decaying learning rate,
   – project back onto the simplex (Euclidean projection, Duchi et al.),
   – occasional random “jitter’’ restores diversity when stagnation is
     detected.

   This stage finishes the optimisation and yields the final sequence.

The whole routine respects the 1000 s time budget, prints progress and
returns the best list of floats found.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, sum=1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    # Duchi et al. (2008) O(n) algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u > cssv / np.arange(1, n + 1))[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (efficient for large n)."""
    n = x.size + y.size - 1
    size = 1 << (n - 1).bit_length()        # next power of two
    X = np.fft.rfft(x, size)
    Y = np.fft.rfft(y, size)
    Z = X * Y
    conv = np.fft.irfft(Z, size)[:n]
    return conv.real


def _eval_score(a: np.ndarray) -> float:
    """Fast exact evaluation of the objective (sum of a assumed ≈1)."""
    n = a.size
    b = _convolve_fft(a, a)
    max_b = b.max()
    return float(2.0 * n * max_b / (a.sum() ** 2))


def _initial_sqrt_shape(n: int) -> np.ndarray:
    """
    Initialise a_i ∝ 1/√(i+1) and normalise to ℓ₁‑norm = 1.
    This shape attains the theoretical value π/2 ≈ 1.5708.
    """
    i = np.arange(1, n + 1, dtype=float)   # i = 1 … n
    a = 1.0 / np.sqrt(i)
    a /= a.sum()
    return a


def _precompute_constraint_structure(n: int):
    """
    Return (row_indices, col_indices) for the Toeplitz constraint matrix
    that represents (f * g)[k] ≤ M.
    The pattern (row, col) is the same for all f; only the data vector changes.
    """
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    pos = 0
    for j in range(n):
        # column j contributes to rows j … j+n‑1
        rows[pos:pos + n] = np.arange(j, j + n, dtype=np.int32)
        cols[pos:pos + n] = j
        pos += n
    return rows, cols


def _lp_step(f: np.ndarray, rows, cols, M, t):
    """
    Perform one LP‑based improvement of the vector f.

    Parameters
    ----------
    f : np.ndarray, shape (n,)
        Current sequence, already scaled to sum = sqrt(2·n).
    rows, cols : pre‑computed index arrays for the constraint matrix.
    M : float
        Current maximal convolution value (max(f∗f)).
    t : float (0 < t ≤ 1)
        Mixing parameter for the convex update f ← (1‑t)·f + t·g.

    Returns
    -------
    f_new : np.ndarray
        Updated sequence (still with the same ℓ₁‑norm).
    M_new : float
        Maximal convolution after the update.
    success : bool
        Whether the LP succeeded.
    """
    n = f.size
    data = np.repeat(f, n)                     # f_i repeated n times
    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    # LP: maximise sum(g)  ⇔ minimise -∑g  (bounds g≥0)
    c = -np.ones(n, dtype=float)

    # RHS: M for each convolution entry
    b_ub = np.full(2 * n - 1, M, dtype=float)

    # Solve using the high‑performance HiGHS interior‑point method
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                  method='highs', options={'presolve': True, 'time_limit': 30})

    if not res.success:
        return f, M, False

    g0 = res.x
    # Rescale g0 to the same ℓ₁‑norm as f (√(2n))
    target_norm = np.sqrt(2.0 * n)
    g = g0 * (target_norm / g0.sum())

    # Convex combination
    f_new = (1.0 - t) * f + t * g

    # Compute new maximal convolution
    b = _convolve_fft(f_new, f_new)
    M_new = b.max()
    return f_new, M_new, True


def _upscale(sequence_small: np.ndarray, n_large: int) -> np.ndarray:
    """Linear interpolation from length |sequence_small| to n_large."""
    n_small = sequence_small.size
    x_small = np.linspace(0, n_small - 1, n_small)
    x_large = np.linspace(0, n_small - 1, n_large)
    return np.interp(x_large, x_small, sequence_small)

# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:

    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    # ----- Phase 1: LP optimisation on a small problem -----------------
    n_small = 2048                     # small enough for a dense LP
    a_small = _initial_sqrt_shape(n_small)      # sum = 1
    # Scale to the “√(2n)’’ norm required by the LP formulation
    a_small = a_small * (np.sqrt(2.0 * n_small) / a_small.sum())

    rows, cols = _precompute_constraint_structure(n_small)

    # LP parameters
    max_lp_iters = 12
    t = 0.5
    print("[LP] start optimisation, n_small =", n_small)
    for lp_iter in range(max_lp_iters):
        # current maximal convolution

        M = b.max()

        if not ok:

        # Progress report

            # shrink mixing factor slowly (larger steps early, finer later)

    # ----- Phase 2: Upscale to the target size ------------------------
    n_large = 8000
    a_large = _upscale(a_small, n_large)

    cur_seq = a_large.copy()

    # hyper‑parameters
    lr = 0.15                 # initial step size
