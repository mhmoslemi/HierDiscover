"""
Improved step‑function search.

The method starts from the Catalan (discrete arcsine) seed and first performs
a few iterations of a *linear‑programming* (LP) step that is exactly the one
described in the original state‑of‑the‑art paper:  for the current
vector f we compute M = max(f*f) and then solve

    maximise 1ᵀg
    s.t.      f * g ≤ M ,   g ≥ 0 .

Because the convolution f*g is linear in g this is a genuine LP.
The optimiser pushes as much mass as possible into g while respecting the
same convolution bound M; after normalising g we move a fraction t towards it.
Repeating this step strictly decreases the objective 2·n·max(f*f).

After a limited number of LP refinements we fall back to the fast greedy
mass‑transfer routine and finish with the exponentiated‑gradient (EG)
refinement that was already used in the previous baseline.  All three phases
run within the allotted 1000 s budget and the best sequence found is returned.

The code is fully self‑contained; it only relies on NumPy, SciPy and the
already‑imported ``evaluate_sequence`` for the final security‑checked score.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.linalg
import scipy.optimize


# ----------------------------------------------------------------------
# Fast evaluation – identical to the harness but without the security checks.
# ----------------------------------------------------------------------
def _fast_evaluation(seq: np.ndarray) -> float:
    """Return the same metric as evaluate_sequence but much faster."""
    n = seq.size
    if n == 0:
        return np.inf
    # FFT size = next power of two ≥ 2·n‑1
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Catalan (discrete arcsine) seed.
# ----------------------------------------------------------------------
def _catalan_sequence(length: int) -> np.ndarray:
    """Return the first ``length`` Catalan‑like coefficients."""
    seq = np.empty(length, dtype=float)
    a = 1.0
    seq[0] = a
    for i in range(1, length):
        a *= (2 * i - 1) / (2.0 * i)         # a_{i+1} = a_i * (2i‑1)/(2i+2)
        seq[i] = a
    return seq


# ----------------------------------------------------------------------
# Linear‑programming refinement step.
# ----------------------------------------------------------------------
def _lp_step(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   maximise 1ᵀg   s.t.   f * g ≤ M , g ≥ 0.
    Returns the optimal g (unnormalised) or ``None`` on failure.
    """
    n = f.size
    # Build the (2n‑1)×n convolution matrix A such that A @ g == f * g.
    # First column = f padded with zeros, first row = [f₀, 0, …, 0].
    first_col = np.concatenate([f, np.zeros(n - 1)])
    first_row = np.concatenate([f[:1], np.zeros(n - 1)])
    A_dense = scipy.linalg.toeplitz(first_col, first_row)   # shape (2n‑1, n)
    A = sp.csr_matrix(A_dense)                               # sparse for the LP solver

    # LP: minimise -∑g  (i.e. maximise ∑g)
    c = -np.ones(n)                                          # objective
    bounds = [(0, None)] * n                                 # g ≥ 0
    b_ub = np.full(2 * n - 1, M)                             # f * g ≤ M

    # use HiGHS (fast, can handle sparse A)
    res = scipy.optimize.linprog(
        c,
        A_ub=A,
        b_ub=b_ub,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )
    if not res.success:
        return None
    return res.x


# ----------------------------------------------------------------------
# Simple greedy refinement – mass transfer from a “bad’’ index to a “good’’ one.
# ----------------------------------------------------------------------
def _refine(seq: np.ndarray, deadline: float) -> np.ndarray:
    """Greedy local improvement; stops when the deadline is reached."""
    n = seq.size
    best_seq = seq.copy()
    best_score = _fast_evaluation(best_seq)

    # pre‑compute FFT length (does not change)
    fft_len = 1 << ((2 * n - 1).bit_length())

    it = 0
    while time.time() < deadline:
        # convolution and its maximum
        fa = np.fft.rfft(seq, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        k = int(conv.argmax())
        max_val = conv[k]

        # contribution weights for each position to that maximal entry
        i_arr = np.arange(n)
        j_arr = k - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        w = np.zeros_like(seq)
        w[mask] = seq[j_arr[mask]]

        # source = index with largest contribution, sink = index with almost
        # no contribution to this maximal entry
        contrib = seq * w
        src = int(np.argmax(contrib))
        sink = int(np.argmin(w))

        if seq[src] < 1e-12:
            break

        # move a modest fraction of the mass (5 % of the source amount)
        eps = 0.05 * seq[src]
        new_seq = seq.copy()
        new_seq[src] -= eps
        new_seq[sink] += eps

        score = _fast_evaluation(new_seq)

        if score < best_score - 1e-12:
            seq = new_seq
            best_seq = new_seq.copy()
            best_score = score
        else:
            # occasional random jitter to escape plateaus
            if np.random.rand() < 0.01:
                jitter_idx = np.random.randint(n)
                jitter = np.random.randn() * 0.001
                new_seq = seq.copy()
                new_seq[jitter_idx] = max(0.0, new_seq[jitter_idx] + jitter)
                seq = new_seq
        it += 1
    return best_seq


# ----------------------------------------------------------------------
# Exponentiated‑gradient refinement on the simplex.
# ----------------------------------------------------------------------
def _eg_refine(seq: np.ndarray, deadline: float) -> np.ndarray:
    """
    Exponentiated‑gradient descent that directly minimises the maximal
    self‑convolution.  ``seq`` is assumed to be non‑negative and normalised
    to have sum = 1 (the metric is scale‑invariant).
    """
    n = seq.size
    best_seq = seq.copy()
    best_score = _fast_evaluation(best_seq)

    fft_len = 1 << ((2 * n - 1).bit_length())
    # Adam‑like parameters for the mirror‑descent on the simplex
    alpha = 0.1                # base step size
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    m = np.zeros_like(seq)      # first moment
    v = np.zeros_like(seq)      # second moment
    it = 1

    while time.time() < deadline:
        # convolution and its maximal value
        fa = np.fft.rfft(seq, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        M = conv.max()

        # active indices (those attaining the maximum up to a tiny tolerance)
        tol = 1e-12
        active = np.where(conv >= M - tol)[0]

        # sub‑gradient of the max‑operator:
        g = np.zeros_like(seq)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            i_range = np.arange(lo, hi + 1)
            j_range = k - i_range
            g[i_range] += seq[j_range]

        # gradient of the objective (2·n·M) w.r.t. a
        grad_a = 2.0 * n * 2.0 * g                     # = 4·n·g

        # mirror‑descent on simplex (exponentiated‑gradient / Adam)
        m = beta1 * m + (1 - beta1) * grad_a
        v = beta2 * v + (1 - beta2) * (grad_a * grad_a)

        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        delta = -alpha * m_hat / (np.sqrt(v_hat) + eps)

        # multiplicative update in the exponential domain
        seq = seq * np.exp(delta)

        # keep positivity, renormalise to the simplex
        seq = np.maximum(seq, 1e-15)
        seq /= seq.sum()

        score = _fast_evaluation(seq)
        if score < best_score - 1e-12:
            best_seq = seq.copy()
            best_score = score

        # tiny random jitter occasionally
        if np.random.rand() < 0.001:
            idx = np.random.randint(n)
            seq[idx] = max(0.0, seq[idx] + np.random.randn() * 1e-3)
            seq /= seq.sum()
        it += 1

    return best_seq

    # 1.  Initial seed (Catalan / arcsine coefficients) and normalisation.
    # ------------------------------------------------------------------
    n = 1200                     # moderate size – enough to get a good constant,
                                 # still cheap for the LPs.
    seq = _catalan_sequence(n).astype(float)
    seq /= seq.sum()             # normalise to sum = 1 (scale‑invariant)

    best_seq = seq.copy()
    best_score = _fast_evaluation(seq)
    print(f"[init] n={n}  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2.  LP refinement – each iteration can only lower max(f*f).
    # ------------------------------------------------------------------
    mix_frac = 0.3               # how much of the new g to mix in
    lp_iter = 0
    max_lp_iters = 25            # keep the LP work bounded
    while time.time() < deadline and lp_iter < max_lp_iters:
        # current maximal self‑convolution
        m_fft = 1 << ((2 * n - 1).bit_length())
        fa = np.fft.rfft(seq, m_fft)
        conv = np.fft.irfft(fa * fa, m_fft)[: 2 * n - 1]
        M = conv.max()

        # solve the linear programme
        g = _lp_step(seq, M)
        if g is None:
            print("[LP] solver failed – aborting LP phase")
            break
        sum_g = g.sum()
        if sum_g <= 0.0:
            print("[LP] got zero‑mass solution – aborting")
            break

        # normalise the LP solution and blend it with the current sequence
        g_norm = g / sum_g
        seq = (1.0 - mix_frac) * seq + mix_frac * g_norm
        seq /= seq.sum()                     # enforce exact normalisation

        cur_score = _fast_evaluation(seq)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = seq.copy()
            print(f"[LP {lp_iter:02d}]  score={best_score:.6f}")

        # slowly reduce the mixing fraction – later steps become finer
        mix_frac = max(0.01, mix_frac * 0.93)
        lp_iter += 1

    # ------------------------------------------------------------------
    # 3.  Greedy refinement – cheap local moves.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq = _refine(best_seq, deadline)
        cur_score = _fast_evaluation(seq)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = seq.copy()
            print(f"[greedy] after refinement  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Exponentiated‑gradient refinement for the remaining time.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq = _eg_refine(best_seq, deadline)
        cur_score = _fast_evaluation(seq)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = seq.copy()
            print(f"[EG] final refinement  score={best_score:.6f}")

    print(f"[final] best score={best_score:.6f}")
    return best_seq.tolist()
