"""
Propose a non‑negative step‑function sequence that minimizes

        O(a) = 2·n·max(conv(a,a))          (a is projected to the simplex)

The algorithm consists of three phases:

1. **Initialisation** – build several cheap candidates (uniform,
   exponential decay, random sorted, the previous best if available) and
   keep the one with the smallest exact objective.

2. **Smoothed gradient descent** – minimise a log‑sum‑exp surrogate of the
   max‑convolution using an Adam optimiser.  After each update the vector is
   projected onto the probability simplex.  The smoothing parameter α is
   annealed from 5 to a large value (≈3000) so that the surrogate gradually
   becomes the true max.

3. **Targeted local refinement** – repeatedly locate the shift k where the
   convolution attains its maximum, then perform a small “mass‑transfer’’
   move from a heavy coefficient that participates in this peak to a light
   coefficient that does not.  Occasionally a sub‑gradient step (as used in
   the reference code) and random swap moves are applied.  All moves are
   accepted if they improve the exact objective, otherwise Metropolis‑type
   uphill moves are allowed.

The whole search respects the 1000 s budget (the function stops a little
before the deadline).  The best sequence found is returned as a plain
Python list and a final call to ``evaluate_sequence`` is printed for sanity
checking.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper – Euclidean projection onto the probability simplex {x ≥ 0, sum(x)=1}
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex."""
    n = v.shape[0]
    # Sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:                     # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full linear convolution (length 2·n‑1)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return a*a (full linear convolution) using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Exact objective for a simplex‑normalised vector (sum = 1)
def _objective_simplex(a: np.ndarray, n: int, L: int) -> float:
    """2·n·max(conv(a,a)) (assumes sum(a)=1)."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Sub‑gradient of the maximal convolution entry (used for occasional
# corrective steps in the local phase).
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)   # typically a single index
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    g /= len(ks)                           # average if several maximisers
    return g


# ----------------------------------------------------------------------
# Gradient of the soft‑max (log‑sum‑exp) surrogate of the max.
def _smooth_gradient(a: np.ndarray, conv: np.ndarray, alpha: float, L: int) -> np.ndarray:
    """∇_a ( (2·n/α)·log∑_k e^{α·conv_k} )."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)            # stabilise
    w = exp_scaled / exp_scaled.sum()                   # soft‑max weights

    # correlation w ⋆ a (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)     # length ≥ 3·n‑2
    n = a.size
    # central n entries correspond to indices n‑1 … 2·n‑2
    grad = 2.0 * corr_rev[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser step (projected onto simplex afterwards)
def _adam_step(a: np.ndarray,
               grad: np.ndarray,
               m: np.ndarray,
               v: np.ndarray,
               t: int,
               lr: float = 0.2,
               beta1: float = 0.9,
               beta2: float = 0.999,
               eps: float = 1e-8) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """Perform one Adam update and project the result onto the simplex."""
    t += 1
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad * grad)

    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)

    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    a_new = a - step
    a_new = _project_to_simplex(a_new)
    return a_new, m, v, t


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function (non‑negative, sum normalised) that minimises
    the provided evaluation function.  Returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # a safety margin

    # ------------------------------------------------------------------
    # Determine problem size and initialise a few candidates
    if 'height_sequence_1' in globals():
        base = np.array(height_sequence_1, dtype=float)
        n = base.size
        print(f"[init] using existing sequence, n={n}")
    else:
        n = 4000
        print(f"[init] no prior sequence – using n={n}")

    # A power‑of‑two FFT length sufficient for both convolution (2n‑1) and
    # correlation (3n‑2) used in the smooth gradient.
    L = 1 << ((3 * n - 2).bit_length())

    # ----- build a few simple starting points -------------------------
    candidates = []

    # (1) previous best, if available
    if 'height_sequence_1' in globals():
        seq = np.array(height_sequence_1, dtype=float)
        seq = seq / seq.sum()
        candidates.append(seq)

    # (2) uniform
    candidates.append(np.full(n, 1.0 / n))

    # (3) exponential decay (r close to 1 → gentle decay)
    r = 0.99
    expo = (1.0 - r) * (r ** np.arange(n))
    expo = expo / expo.sum()
    candidates.append(expo)

    # (4) random descending vector
    rnd = np.random.rand(n)
    rnd = np.sort(rnd)[::-1]               # descending order
    rnd = rnd / rnd.sum()
    candidates.append(rnd)

    # Evaluate the cheap candidates and keep the best one
    best_seq = None
    best_score = np.inf
    for idx, seq in enumerate(candidates):
        sc = _objective_simplex(seq, n, L)
        print(f"[init] candidate {idx} → O={sc:.6f}")
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
    print(f"[init] best initial O={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – smoothed Adam optimiser (log‑sum‑exp surrogate)
    alpha_start = 5.0
    alpha_end = 3000.0
    max_iter_phase1 = 300_000          # generous – will be cut by time
    m = np.zeros_like(best_seq)        # Adam first moment
    v = np.zeros_like(best_seq)        # Adam second moment
    t_adam = 0

    phase1_deadline = start_time + budget_s * 0.55
    it1 = 0
    while time.time() < phase1_deadline and it1 < max_iter_phase1:
        conv = _conv_fft(best_seq, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = best_seq.copy()
            print(f"[phase‑1] new best O={best_score:.6f} (iter={it1})")

        # linear annealing of α
        frac = min(1.0, it1 / max_iter_phase1)
        alpha = alpha_start + (alpha_end - alpha_start) * frac

        grad = _smooth_gradient(best_seq, conv, alpha, L)

        # Adam update + projection
        best_seq, m, v, t_adam = _adam_step(best_seq, grad, m, v, t_adam)

        # occasional back‑track if objective got worse (rare)
        if it1 % 5000 == 0 and it1 > 0:
            conv_check = _conv_fft(best_seq, L)
            obj_check = 2.0 * n * conv_check.max()
            if obj_check > cur_obj:          # went uphill
                # revert to the previous point (simple safety)
                best_seq = _project_to_simplex(best_seq + np.random.randn(n) * 1e-4)

        it1 += 1

    print(f"[phase‑1] finished after {it1} iters, best O={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – targeted local refinement (mass‑transfer + sub‑gradient)
    print("[phase‑2] entering local refinement")
    max_restarts = 0
    max_iter_per_restart = 120_000
    no_improve_limit = 6_000
    temperature0 = 0.02
    subgrad_interval = 9

    while time.time() < deadline:
        # ----- initialise a new restart -------------------------------------------------
        if np.random.rand() < 0.5:
            a = best_seq.copy()
            a += np.random.randn(n) * 0.003
            a = np.clip(a, 0, None)
            a = _project_to_simplex(a)
        else:
            a = np.random.rand(n)
            a = _project_to_simplex(a)

        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        it_local = 0
        no_improve = 0
        T = temperature0

        while time.time() < deadline and it_local < max_iter_per_restart:
            # ---------- mass‑transfer move -------------------------------------------------
            # locate the shift where the convolution is maximal
            k_max = int(conv.argmax())
            # indices that actually contribute to this peak
            i_start = max(0, k_max - (n - 1))
            i_end = min(n - 1, k_max)
            # pick a "heavy" index inside the contributing window
            i_candidates = np.arange(i_start, i_end + 1)
            weights_i = a[i_candidates] * a[k_max - i_candidates]   # contribution of each i
            if weights_i.sum() == 0:
                # fall back to uniform random move
                i = np.random.randint(n)
                j = np.random.randint(n)
            else:
                # bias the choice towards larger contributions
                probs = weights_i / weights_i.sum()
                i = np.random.choice(i_candidates, p=probs)
                # pick a destination far from the window (to diminish overlap)
                # we simply pick any index outside the window; if none exist we pick random.
                outside = np.setdiff1d(np.arange(n), i_candidates, assume_unique=True)
                if outside.size == 0:
                    j = np.random.randint(n)
                else:
                    j = np.random.choice(outside)

            if i != j and a[i] > 0:
                delta = a[i] * np.random.rand() * 0.10          # up to 10 % of a[i]
                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta
                a_new = _project_to_simplex(a_new)

                conv_new = _conv_fft(a_new, L)
                new_obj = 2.0 * n * conv_new.max()

                # Metropolis acceptance
                if new_obj < cur_obj or np.random.rand() < math.exp((cur_obj - new_obj) / max(T, 1e-12)):
                    a = a_new
                    conv = conv_new
                    cur_obj = new_obj
                    no_improve = 0
                    if cur_obj < best_score:
                        best_score = cur_obj
                        best_seq = a.copy()
                        print(f"[{time.time():.1f}] new global best O={best_score:.6f} (restart={max_restarts}, iter={it_local})")
                else:
                    no_improve += 1
            else:
                no_improve += 1

            # ---------- occasional sub‑gradient push ------------------------------------
            if it_local % subgrad_interval == 0:
                g = _subgradient_max_conv(a, conv)
                eta = 0.07 * (1.0 - it_local / max_iter_per_restart)   # decaying step
                a = a - eta * g
                a = _project_to_simplex(a)
                conv = _conv_fft(a, L)
                cur_obj = 2.0 * n * conv.max()
                if cur_obj < best_score:
                    best_score = cur_obj
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] sub‑grad new best O={best_score:.6f}")

            # ---------- random swap move (helps escaping permutations) ---------------
            if it_local % 5000 == 0 and np.random.rand() < 0.3:
                i_swap, j_swap = np.random.choice(n, size=2, replace=False)
                a_swap = a.copy()
                a_swap[i_swap], a_swap[j_swap] = a_swap[j_swap], a_swap[i_swap]
                # no need to re‑project – swapping preserves sum and non‑negativity
                conv_swap = _conv_fft(a_swap, L)
                obj_swap = 2.0 * n * conv_swap.max()
                if obj_swap < cur_obj or np.random.rand() < 0.05:
                    a = a_swap
                    conv = conv_swap
                    cur_obj = obj_swap
                    if cur_obj < best_score:
                        best_score = cur_obj
                        best_seq = a.copy()
                        print(f"[{time.time():.1f}] swap new best O={best_score:.6f}")

            # ---------- temperature decay -----------------------------------------------
            T = temperature0 * (1.0 - it_local / max_iter_per_restart)

            # ---------- restart if stalled ------------------------------------------------
            if no_improve >= no_improve_limit:
                break

            it_local += 1

        max_restarts += 1
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Final sanity check with the provided evaluator
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:        # pragma: no cover
        print("[final] evaluate_sequence failed:", e)

    return best_seq.tolist()
