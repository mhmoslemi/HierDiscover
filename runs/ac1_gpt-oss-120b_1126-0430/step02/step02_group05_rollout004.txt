def gradient(a, tau):
    # compute convolution c = a * a (real)
    N_conv = next_pow2(2*n-1)
    fa = np.fft.rfft(a, N_conv)
    conv = np.fft.irfft(fa * fa, N_conv)[:2*n-1]
    # compute softmax weights
    # logsumexp
    # from scipy.special import logsumexp
    log_w = tau*conv
    max_log = np.max(log_w)
    w_unnorm = np.exp(log_w - max_log)
    Z = np.sum(w_unnorm)
    w = w_unnorm / Z
    # compute cross-correlation a with w
    N_corr = next_pow2(n + (2*n-1) - 1)
    # use complex FFT
    A = np.fft.fft(a, N_corr)
    W = np.fft.fft(w, N_corr)
    cross = np.fft.ifft(A * np.conj(W))
    grad = 2 * np.real(cross[:n])
    return grad, conv.max()

```
def _gradient(a, tau):
    n = a.shape[0]
    L = 2*n-1
    m = 1 << ((2*n-1).bit_length())
    fa = np.fft.rfft(a, m)
    conv = np.fft.irfft(fa * fa, m)[:L]   # c_k
    # compute softmax weights
    scaled = tau * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / np.sum(exp_scaled)
    # cross-correlation gradient
    N_corr = 1 << ((n + L - 1).bit_length())
    A = np.fft.fft(a, N_corr)
    W = np.fft.fft(w, N_corr)
    cross = np.fft.ifft(A * np.conj(W))
    grad = 2.0 * np.real(cross[:n])
    return grad, conv.max()
```

def _step_move(a, tau, rel_target):
    n = a.shape[0]
    fft_len = 1 << ((2*n-1).bit_length())
    # compute conv
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa*fa, fft_len)[:2*n-1]
    max_val = conv.max()
    # choose random max index
    max_idxs = np.flatnonzero(conv >= max_val - 1e-12)
    k = np.random.choice(max_idxs)
    # compute weight vector w = a_{k - i}
    i_arr = np.arange(n)
    j_arr = k - i_arr
    w = np.zeros_like(a)
    mask = (j_arr >= 0) & (j_arr < n)
    w[mask] = a[j_arr[mask]]
    # select sink: small weights
    sink_percent = 2.0 # percentage
    sink_thresh = np.percentile(w, sink_percent)
    sink_mask = w <= sink_thresh
    sink_idxs = np.flatnonzero(sink_mask)
    if sink_idxs.size == 0:
        sink_idxs = np.array([int(np.argmin(w))])
        sink_mask = np.zeros_like(w, dtype=bool); sink_mask[sink_idxs[0]] = True
    # compute source indexes high weight excluding sink
    source_percent = 20.0  # we will choose top 20% by weight
    source_thresh = np.percentile(w, 100 - source_percent)
    source_mask = w >= source_thresh
    source_mask = np.logical_and(source_mask, ~sink_mask)  # exclude sinks
    source_idxs = np.flatnonzero(source_mask)
    if source_idxs.size == 0:
        source_mask = ~sink_mask
        source_idxs = np.flatnonzero(source_mask)
    # compute total excess amount
    w_min = w[sink_idxs].min()
    excess = a[source_idxs] * (w[source_idxs] - w_min)
    total_excess = excess.sum()
    if total_excess <= 0:
        # fallback jitter
        jitter_i = np.random.randint(n)
        jitter = (np.random.rand() - 0.5) * 1e-4
        new_a = a.copy()
        new_a[jitter_i] = max(0.0, new_a[jitter_i] + jitter)
        new_a = new_a / new_a.sum()
        return new_a
    # desired reduce
    target_reduction = rel_target * max_val
    lam = target_reduction / total_excess
    lam = max(min(lam, 1.0), 1e-6)

"""
Step‑function optimizer for the convolution bound.

The algorithm starts from the classic Catalan (arcsine) seed,
normalises it to sum 1 and then repeatedly reduces the highest
entry of the self‑convolution.  For the current peak `k` we compute
the “weight’’ vector w_i = a_{k‑i}.  Mass is moved *only* from indices
with large weight to a small set of “sink’’ indices with the smallest
weight.  The amount transferred is chosen so that the peak is
reduced by a fixed relative amount (≈1 %).  After each move the vector
is re‑normalised (projection onto the probability simplex).  Random
jitter steps are injected occasionally to escape plateaux.

All convolutions are performed with FFTs for speed, allowing tens of
thousands of iterations within the 1000 s budget.  The best sequence
found (lowest value of the evaluation function) is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast evaluation – identical to the provided `evaluate_sequence` but
# without the security checks.  Uses an FFT based self‑convolution.
# ----------------------------------------------------------------------
def _fast_evaluation(seq: np.ndarray) -> float:
    """Return the score 2·n·max(a*a) for a sequence with sum(seq)=1."""
    n = seq.size
    if n == 0:
        return float("inf")
    # FFT size = next power of two ≥ 2*n‑1
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b)   # sum(seq) == 1


# ----------------------------------------------------------------------
# Catalan (arcsine) coefficients.
# a_{i+1} = a_i * (2*i+1)/(2*i+2)
# ----------------------------------------------------------------------
def _catalan_sequence(length: int) -> np.ndarray:
    seq = np.empty(length, dtype=np.float64)
    a = 1.0
    seq[0] = a
    for i in range(1, length):
        a *= (2.0 * i - 1.0) / (2.0 * i)   # a_i = a_{i‑1} * (2i‑1)/(2i)
        seq[i] = a
    return seq


# ----------------------------------------------------------------------
# Helper: compute the weight vector w_i = a_{k‑i} for a given convolution
# index k (zero for out‑of‑range indices).
# ----------------------------------------------------------------------
def _weight_vector(a: np.ndarray, k: int) -> np.ndarray:
    n = a.size
    i = np.arange(n)
    j = k - i
    mask = (j >= 0) & (j < n)
    w = np.zeros_like(a)
    w[mask] = a[j[mask]]
    return w


# ----------------------------------------------------------------------
# Perform a single “peak‑reduction’’ step.
# The current self‑convolution is examined, a maximal entry `k` is
# selected, mass is transferred from high‑weight positions to the
# low‑weight sink set, and the result is projected back onto the simplex.
# ----------------------------------------------------------------------
def _peak_reduction_step(a: np.ndarray,
                         rel_reduce: float = 0.01,
                         sink_percent: float = 3.0,
                         source_percent: float = 20.0) -> np.ndarray:
    """
    Return a new probability vector after moving a modest amount of mass
    from high‑weight indices to low‑weight (sink) indices.
    """
    n = a.size

    # ----- self‑convolution -------------------------------------------------
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]

    # ----- pick a (random) maximal index ----------------------------------
    max_val = conv.max()
    max_idxs = np.flatnonzero(conv >= max_val - 1e-12)
    k = int(np.random.choice(max_idxs))

    # ----- weight vector for the chosen peak -------------------------------
    w = _weight_vector(a, k)

    # ----- choose sink indices (smallest weights) -------------------------
    sink_thresh = np.percentile(w, sink_percent)
    sink_mask = w <= sink_thresh
    sink_idxs = np.flatnonzero(sink_mask)
    if sink_idxs.size == 0:                     # fallback – absolute minimum
        sink_idxs = np.array([int(np.argmin(w))])
        sink_mask = np.zeros_like(w, dtype=bool)
        sink_mask[sink_idxs[0]] = True

    w_min = w[sink_idxs].min()

    # ----- choose source indices (largest weights, excluding sinks) -------
    source_thresh = np.percentile(w, 100.0 - source_percent)
    source_mask = (w >= source_thresh) & (~sink_mask)
    source_idxs = np.flatnonzero(source_mask)
    if source_idxs.size == 0:                  # fall back to all non‑sinks
        source_mask = ~sink_mask
        source_idxs = np.flatnonzero(source_mask)

    # ----- total “excess’’ mass that would help the chosen peak ------------
    excess = a[source_idxs] * (w[source_idxs] - w_min)
    total_excess = excess.sum()

    if total_excess <= 0.0:
        # Nothing helpful – do a tiny random jitter.
        jitter_i = np.random.randint(n)
        jitter = (np.random.rand() - 0.5) * 1e-4
        new_a = a.copy()
        new_a[jitter_i] = max(0.0, new_a[jitter_i] + jitter)
        new_a /= new_a.sum()
        return new_a

    # ----- we want to cut the peak by a relative amount --------------------
    target_reduction = rel_reduce * max_val
    lam = target_reduction / total_excess
    lam = max(min(lam, 1.0), 1e-6)          # keep lam reasonable

    # ----- amount removed from each source index ---------------------------
    delta = lam * a[source_idxs]           # mass to move away from each source
    moved_mass = delta.sum()

    # ----- distribute the moved mass uniformly among the sinks -------------
    add_per_sink = moved_mass / sink_idxs.size

    new_a = a.copy()
    new_a[source_idxs] -= delta
    new_a[sink_idxs] += add_per_sink

    # ----- enforce non‑negativity and re‑normalise (projection onto simplex)-
    new_a = np.maximum(new_a, 0.0)
    new_a /= new_a.sum()
    return new_a


# ----------------------------------------------------------------------
# Main optimisation driver.
# ----------------------------------------------------------------------
def _optimise(a: np.ndarray,
              deadline: float,
              rel_reduce: float = 0.01,
              sink_percent: float = 3.0,
              source_percent: float = 20.0) -> np.ndarray:
    """
    Iteratively apply the peak‑reduction step until the time budget runs out.
    Returns the best sequence (lowest score) encountered.
    """
    best_a = a.copy()
    best_score = _fast_evaluation(best_a)
    it = 0

    while time.time() < deadline:
        a_candidate = _peak_reduction_step(
            a,
            rel_reduce=rel_reduce,
            sink_percent=sink_percent,
            source_percent=source_percent,
        )
        cand_score = _fast_evaluation(a_candidate)

        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best_a = a_candidate.copy()
            a = a_candidate
            print(f"Iter {it:6d}: improved score = {best_score:.8f}")
        else:
            # occasional random shake‑up (helps escape plateaux)
            if np.random.rand() < 0.02:
                i, j = np.random.randint(a.size, size=2)
                if i != j and a[i] > 1e-12:
                    eps = min(a[i], np.random.rand() * a[i] * 0.1)
                    tmp = a.copy()
                    tmp[i] -= eps
                    tmp[j] += eps
                    tmp /= tmp.sum()
                    tmp_score = _fast_evaluation(tmp)
                    if tmp_score < best_score:
                        a = tmp
                        best_a = tmp.copy()
                        best_score = tmp_score
                        print(f"Iter {it:6d}: random shake improved score = {best_score:.8f}")
        # slowly tighten the reduction factor as the optimisation proceeds
        if it % 500 == 0 and rel_reduce > 0.001:
            rel_reduce *= 0.9

        it += 1

    return best_a


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness.
# ----------------------------------------------------------------------

    """
    Return a non‑negative step‑function (list of heights) whose

    below the classic π/2 ≈ 1.5708 bound.
    """
    np.random.seed(seed)

    # keep a safety margin for the harness
    deadline = time.time() + budget_s - 5.0

    # sequence length – a compromise between quality and speed
    n = 5000

    # ------------------------------------------------------------------
    # Initialise with the Catalan (arcsine) sequence and normalise so that
    # the total sum equals 1 (the metric is scale‑invariant).
    # ------------------------------------------------------------------
    try:
        # Use a previously discovered construction if available.
        init = np.array(height_sequence_1, dtype=np.float64)

    # Run the optimisation loop.
    # ------------------------------------------------------------------
