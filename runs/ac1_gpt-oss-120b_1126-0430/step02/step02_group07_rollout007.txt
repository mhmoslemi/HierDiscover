"""
Propose a step‑function that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

where conv is the discrete self‑convolution.  The algorithm works in three
stages:

1. **Broad sampling** – generate many candidates from a few parametric
   families (power‑law, exponential, mixtures).  Each candidate is normalised
   to the probability simplex and evaluated with the official
   ``evaluate_sequence``.  The best candidate is kept.

2. **Projected gradient / simulated‑annealing refinement** – starting from the
   best sample, repeatedly move a small amount of mass from the index that
   contributes most to the current maximal convolution entries to the index
   that contributes the least.  The move size shrinks with the iteration
   count, a simple temperature schedule allows occasional uphill moves,
   and the vector is projected onto the simplex after each update.
   Convolutions are computed with an FFT (O(n log n)) for speed.

3. **Return the best sequence** – the best vector found (as a plain Python
   list of floats) is returned.  All numbers are non‑negative and sum to 1,
   which makes the objective scale‑invariant.
"""

import time
import numpy as np

# Helper utilities – must stay at top level (no closures)
# ----------------------------------------------------------------------


def _project_to_simplex(v):
    """Project a vector onto the probability simplex {x ≥ 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


def _norm_and_score(raw):
    """Normalise a raw (non‑negative) vector and return (score, normalised)."""
    total = raw.sum()
    if total <= 0:
        return float("inf"), None
    norm = raw / total
    score = evaluate_sequence(norm.tolist())
    return score, norm


def _conv_self_fft(arr, fft_size):
    """Self‑convolution using a real FFT (returns length 2·len(arr)‑1)."""
    a_fft = np.fft.rfft(arr, n=fft_size)
    conv_full = np.fft.irfft(a_fft * a_fft, n=fft_size)
    return conv_full[: 2 * len(arr) - 1]


def _compute_contributions(arr, mask_idx):
    """
    For a set of convolution indices (mask_idx) compute, for each position i,
    the total contribution a_i * a_{k-i} summed over all k in mask_idx.
    """
    n = len(arr)
    contrib = np.zeros(n, dtype=float)
    for k in mask_idx:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i = np.arange(i_start, i_end + 1)
        contrib[i] += arr[k - i]
    return contrib


def _local_search(
    start_vec,
    time_budget,
    start_time,
    fft_size,
    rng,
    max_iters=300_000,
    thresh=0.99,
):
    """
    Projected gradient / simulated‑annealing search that attempts to lower the
    L∞ norm of the self‑convolution.  Returns the best vector and its cheap
    (non‑official) score.
    """
    n = len(start_vec)
    cur = _project_to_simplex(start_vec.copy())
    cur_score = 2.0 * n * np.max(_conv_self_fft(cur, fft_size))
    best = cur.copy()
    best_score = cur_score

    it = 0
    no_improve = 0

    while time.time() - start_time < time_budget and it < max_iters:
        it += 1

        conv = _conv_self_fft(cur, fft_size)
        max_val = conv.max()
        mask_idx = np.where(conv >= max_val * thresh)[0]
        if mask_idx.size == 0:  # safety fallback
            mask_idx = np.array([np.argmax(conv)])

        # contributions to the (near‑)maximal convolution entries
        contrib = _compute_contributions(cur, mask_idx)

        src = int(np.argmax(contrib))
        tgt = int(np.argmin(contrib))
        if src == tgt or cur[src] <= 0.0:
            src, tgt = rng.choice(n, size=2, replace=False)

        # step fraction slowly shrinks from ~0.12 to ~0.015
        frac = 0.12 * (1.0 - it / max_iters) + 0.015
        step = cur[src] * frac * rng.random()
        if step <= 0.0:
            continue

        new = cur.copy()
        new[src] -= step
        new[tgt] += step
        new = _project_to_simplex(new)

        new_score = 2.0 * n * np.max(_conv_self_fft(new, fft_size))

        # deterministic improvement
        improved = new_score < cur_score

        # occasional uphill acceptance (simulated annealing)
        if not improved:
            temperature = max(1e-6, (1.0 - it / max_iters) * cur_score * 0.01)
            if rng.random() < np.exp(-(new_score - cur_score) / temperature):
                improved = True

        if improved:
            cur = new
            cur_score = new_score
            no_improve = 0
        else:
            no_improve += 1

        if cur_score < best_score:
            best = cur.copy()
            best_score = cur_score
            if it % 5_000 == 0:
