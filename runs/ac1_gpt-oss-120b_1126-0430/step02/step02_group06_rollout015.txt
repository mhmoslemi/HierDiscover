def _fast_evaluation(seq):
    n = seq.size
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[:2 * n - 1]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))

def _smooth_optimize(seq, deadline):
    n = seq.size
    best_seq = seq.copy()
    best_score = _fast_evaluation(seq)
    m = 1 << ((3 * n - 2).bit_length())
    it = 0
    beta = 0.5
    beta_max = 10.0
    while time.time() < deadline:
        # compute self-convolution and max
        fa = np.fft.rfft(seq, m)
        conv = np.fft.irfft(fa * fa, m)[:2 * n - 1]
        max_c = conv.max()
        # compute weights w for smooth max
        # scaling for stability
        maxc = max_c
        exp_term = np.exp(beta * (conv - maxc))
        w = exp_term / exp_term.sum()
        # compute gradient
        rev_w = w[::-1]
        fw = np.fft.rfft(rev_w, m)
        corr = np.fft.irfft(fa * fw, m)[:3 * n - 2]
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * corr[idx]
        # step size
        eta = 0.5 / np.sqrt(it + 1)
        # multiplicative update
        seq = seq * np.exp(-eta * grad)
        seq = np.maximum(seq, 1e-15)
        seq /= seq.sum()
        # evaluate metric
        score = _fast_evaluation(seq)
        if score < best_score - 1e-12:
            best_seq = seq.copy()
            best_score = score
            print(f"[smooth] iter {it} score {best_score:.6f}")
        # update beta gradually
        if beta < beta_max:
            beta = min(beta_max, beta + (beta_max - beta) * 0.01)   # simple increase
        # occasional random perturb
        if np.random.rand() < 0.001:
            idx_random = np.random.randint(n)
            seq[idx_random] += np.random.randn() * 1e-3
            seq = np.maximum(seq, 0)
            seq /= seq.sum()
        it += 1
    return best_seq, best_score

"""
A hybrid optimizer for the step‑function evaluation problem.

The pipeline consists of three stages:
1.  A Catalan (arcsine) seed – the coefficients of (1‑z)^‑1/2 – are
    normalised to sum = 1.
2.  A single linear‑programming refinement (the “LP step” from the
    statement) tries to increase the ℓ₁‑norm of a vector  b  that
    respects the current maximal self‑convolution bound.  Mixing the
    seed with the LP solution yields a strictly better sequence.
3.  A smooth‑max gradient descent finishes the search.  The non‑smooth
    max‑operator is replaced by a softmax (with a temperature that is
    ramped up), the gradient is obtained by a correlation computed via
    FFTs, and an exponentiated‑gradient update keeps the variables on
    the probability simplex.

The whole routine respects the 1000 s budget, prints intermediate
scores and finally returns the best sequence found (as a plain Python
list of non‑negative floats).
"""

import time
import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Fast evaluation – identical to evaluate_sequence but without the safety
# checks (used internally).
# ----------------------------------------------------------------------
def _fast_evaluation(seq: np.ndarray) -> float:
    """Return the metric 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    n = seq.size
    if n == 0:
        return np.inf
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Catalan (arcsine) seed.
# a0 = 1,   a_{i+1} = a_i·(2·i+1)/(2·i+2)
# ----------------------------------------------------------------------
def _catalan_sequence(length: int) -> np.ndarray:
    seq = np.empty(length, dtype=float)
    a = 1.0
    seq[0] = a
    for i in range(1, length):
        a *= (2 * i - 1) / (2.0 * i)
        seq[i] = a
    return seq


# ----------------------------------------------------------------------
# Self‑convolution maximum (used by the LP step).
# ----------------------------------------------------------------------
def _conv_max(seq: np.ndarray) -> float:
    n = seq.size
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Build the (2n‑1)×n matrix A with   A_{k,j}=seq_{k‑j}  when the index
# is inside the range 0…n‑1, otherwise 0.
# ----------------------------------------------------------------------
def _build_conv_matrix(seq: np.ndarray) -> sp.csr_matrix:
    n = seq.size
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1, dtype=int)
        vals = seq[k - js]                     # f_{k‑j}
        rows.extend([k] * len(js))
        cols.extend(js.tolist())
        data.extend(vals.tolist())
    return sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


# ----------------------------------------------------------------------
# Linear‑programming refinement (single pass).
# ----------------------------------------------------------------------
def _lp_refine(seq: np.ndarray, deadline: float) -> np.ndarray:
    """
    Perform one LP refinement:
        maximise   Σ b_j
        subject to b_j ≥ 0  and  (seq * b)_k ≤ M   for all k,
    where M = max(seq*seq).  The solution is mixed with the current
    sequence using a fixed mixing factor t=0.5.
    """
    n = seq.size
    M = _conv_max(seq)

    # Build constraint matrix A (A·b ≤ M)
    A = _build_conv_matrix(seq)

    c = -np.ones(n)                     # maximise Σ b_j  ↔ minimise –Σ b_j
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    # Solve the LP (highs is fast and accepts sparse matrices)
    res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
    if not res.success:
        return seq                         # fall back to the input sequence

    b = res.x
    if b.sum() <= 0:
        return seq

    # Normalise the LP solution to sum = 1 and mix.
    g = b / b.sum()
    t = 0.5                               # mixing weight
    new_seq = (1.0 - t) * seq + t * g
    new_seq = np.maximum(new_seq, 1e-15)
    new_seq /= new_seq.sum()
    return new_seq


# ----------------------------------------------------------------------
# Smooth‑max gradient descent (exponentiated‑gradient on the simplex).
# ----------------------------------------------------------------------
def _smooth_gradient_descent(seq: np.ndarray, deadline: float) -> np.ndarray:
    """Minimise the maximal self‑convolution using a soft‑max surrogate."""
    n = seq.size
    best_seq = seq.copy()
    best_score = _fast_evaluation(best_seq)

    # FFT length for convolutions/correlations (fixed throughout).
    fft_len = 1 << ((3 * n - 2).bit_length())

    it = 0
    beta = 0.5                # initial temperature
    beta_max = 12.0           # final temperature (approaches the true max)
    while time.time() < deadline:
        # ----- self‑convolution -----
        fa = np.fft.rfft(seq, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_c = conv.max()

        # ----- soft‑max weights -----
        # stabilise the exponentials
        shift = max_c
        exp_term = np.exp(beta * (conv - shift))
        w = exp_term / exp_term.sum()                     # shape (2n‑1,)

        # ----- gradient of the soft‑max (correlation) -----
        rev_w = w[::-1]
        fw = np.fft.rfft(rev_w, fft_len)
        corr = np.fft.irfft(fa * fw, fft_len)[: 3 * n - 2]
        idx = (2 * n - 2) - np.arange(n)                 # pick the valid slice
        grad = 2.0 * corr[idx]                           # shape (n,)

        # ----- exponentiated‑gradient update (mirrored on the simplex) -----
        eta = 0.7 / np.sqrt(it + 1)                      # decreasing step‑size
        seq = seq * np.exp(-eta * grad)
        seq = np.maximum(seq, 1e-15)                     # keep strict positivity
        seq /= seq.sum()                                 # simplex projection

        # ----- evaluate -----
        score = _fast_evaluation(seq)
        if score < best_score - 1e-12:
            best_seq = seq.copy()
            best_score = score
            print(f"[smooth] iter {it:6d}  score {best_score:.6f}")

        # ----- temperature schedule (slowly tighten the approximation) -----
        if beta < beta_max:
            beta = min(beta_max, beta + (beta_max - beta) * 0.005)

        # ----- occasional random jitter to avoid tiny stagnation -----
        if np.random.rand() < 0.001:
            j = np.random.randint(n)
            seq[j] = max(0.0, seq[j] + np.random.randn() * 1e-3)
            seq /= seq.sum()

        it += 1
    return best_seq


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights aiming at a low evaluation score.

    The routine works within ``budget_s`` seconds (default 1000).  It
    builds a Catalan (arcsine) seed, performs a single linear‑programming
    refinement and finishes with a smooth‑max gradient descent on the
    simplex.  The best sequence discovered is returned as a plain Python
    list.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5          # a small safety margin

    # Length – 2000 gives a good trade‑off between quality and LP size.
    n = 2000

    # 0. Initial seed (Catalan / arcsine coefficients) and normalisation.
    seq = _catalan_sequence(n).astype(float)
    seq /= seq.sum()                         # normalise to sum = 1
    print(f"[init] length={n}  initial score={_fast_evaluation(seq):.6f}")

    # 1. LP refinement (uses roughly the first 15 % of the budget).
    lp_deadline = start + 0.15 * (budget_s - 5)
    if time.time() < lp_deadline:
        seq = _lp_refine(seq, lp_deadline)
        cur_score = _fast_evaluation(seq)
        print(f"[after LP] score={cur_score:.6f}")

    # 2. Smooth‑max gradient descent for the remaining time.
    seq = _smooth_gradient_descent(seq, deadline)
    final_score = _fast_evaluation(seq)
    print(f"[final] best score={final_score:.6f}")

    # Return the best sequence as a plain Python list.
    return seq.tolist()
