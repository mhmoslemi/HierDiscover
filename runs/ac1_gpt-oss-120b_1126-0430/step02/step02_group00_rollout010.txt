"""
propose_candidate
=================

A fast multi‑phase optimiser for the “max‑convolution” problem.

* Phase 0 – careful initialisation.
  * Start from a previously known best sequence (if any).
  * Also try a family of *arcsine*‑shaped distributions
    a_i ∝ ((i+½)(N‑½‑i))⁻ᵖ (p≈0.5).  These are known to give a
    near‑flat self‑convolution, i.e. a low value of the objective.
  * Pick the best of those as the seed.

* Phase 1 – smooth‑max gradient descent.
  * Optimise the log‑sum‑exp surrogate
    f_α(a) = (2·N/α)·log∑_k e^{α·conv_k(a)}.
  * α is increased exponentially from 5 to 8000, so the surrogate
    focuses more and more on the true maximum.
  * After each gradient step we project onto the simplex
    {x≥0, Σx=1}.  The step size is chosen so that the maximum
    coordinate change never exceeds 0.4.
  * The best sequence seen is recorded.

* Phase 2 – simulated‑annealing hill‑climbing + occasional sub‑gradient.
  * Random “mass‑transfer’’ moves: move a random fraction (≤10 %) of the
    mass from a source index to a target index.
  * Random swaps of two entries – a cheap way of changing the ordering
    (important because the convolution depends on the permutation).
  * Every few iterations a true sub‑gradient step for the current peak
    of the convolution is performed.
  * Acceptance follows a Metropolis rule with a temperature that cools
    linearly to zero.

The whole routine respects a 1000 s wall‑clock budget; it prints
periodic progress messages and finally returns the best list of
coefficients it has found.
"""

import time
import math
import numpy as np
from scipy import sparse

# ----------------------------------------------------------------------
# Helper – projection onto the probability simplex {x ≥ 0, Σx = 1}
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:               # all non‑positive – return the uniform point
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full linear convolution (length 2·N‑1)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return the linear convolution a*a using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Exact objective for a simplex‑normalised vector (Σa = 1)
def _objective_simplex(a: np.ndarray, N: int, L: int) -> float:
    """2·N·max(a*a)  (a is assumed to sum to 1)."""
    conv = _conv_fft(a, L)
    return 2.0 * N * conv.max()


# ----------------------------------------------------------------------
# Sub‑gradient of the maximal convolution entry (used in phase 2)
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Return a sub‑gradient of max(conv) w.r.t. a."""
    N = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (N - 1))
        i_end = min(N - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    g /= len(ks)                     # average if several maxima exist
    return g


# ----------------------------------------------------------------------
# Gradient of the log‑sum‑exp surrogate (soft‑max of conv)
def _smooth_gradient(a: np.ndarray, conv: np.ndarray, alpha: float, L: int) -> np.ndarray:
    """
    Compute ∇_a ( (2·N/α)·log∑_k e^{α·conv_k} ).
    The formula collapses to 2·corr(w, a) where w_k ∝ e^{α·conv_k}.
    """
    # stable soft‑max weights
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / exp_scaled.sum()               # shape (2N‑1,)

    # correlation w ⋆ a  (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)   # length ≥ 3N‑2
    N = a.size
    grad = 2.0 * corr_rev[N - 1: N - 1 + N]           # central N entries
    return grad


# ----------------------------------------------------------------------
# Family of arcsine‑like initialisers
def _init_arcsine(N: int, p: float) -> np.ndarray:
    """
    Produce a symmetric distribution a_i ∝ ((i+0.5)*(N-0.5-i))^{-p}.
    p≈0.5 gives the classical arcsine shape.
    """
    i = np.arange(N, dtype=np.float64)
    a = 1.0 / ((i + 0.5) * (N - 0.5 - i)) ** p
    a = a / a.sum()
    return a
