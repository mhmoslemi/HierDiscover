"""
Propose a candidate step‑function (list of non‑negative heights) that minimises
``evaluate_sequence``.

Algorithm
---------
1. Choose a moderate problem size (default n=12000).  
2. Build a diverse pool of initial candidates:
   * uniform,
   * Catalan‑like sequence,
   * several geometric (exponential) decays,
   * a few Beta‑distribution shapes a_i ∝ (x(1‑x))^{p‑1},
   * Hann / raised‑cosine window,
   * a Gaussian window,
   * a random simplex point.
   The best of these (according to the exact objective
   2·n·max(conv) / sum²) is kept.
3. Optimise the chosen seed with a projected Adam optimiser on the
   smooth‑max surrogate of max(conv).  At each iteration:
   * compute the full convolution b = a∗a,
   * form soft‑max weights w_k ∝ exp(α·(b_k−max(b))) (α grows slowly),
   * gradient of the smooth max is 2·(a ⊛ w_rev) restricted to the
     original length,
   * perform an Adam step and project back onto the probability simplex.
   * keep the best point seen so far; occasionally shrink the step‑size or
     restart from the current best if progress stalls.
4. The loop runs until the time budget (≈ 1000 s) is exhausted; the best
   sequence found is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, sum=1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    # Fast projection (Duchi et al., 2008)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u > cssv / np.arange(1, n + 1))[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of real vectors x and y using a real FFT."""
    n = x.size + y.size - 1
    size = 1 << (n - 1).bit_length()      # next power of two
    X = np.fft.rfft(x, size)
    Y = np.fft.rfft(y, size)
    Z = X * Y
    conv = np.fft.irfft(Z, size)[:n]
    return conv.real


def _eval_score(a: np.ndarray) -> float:
    """Exact objective value 2·n·max(a∗a) / (sum a)² for a non‑negative simplex vector."""
    n = a.size
    b = _convolve_fft(a, a)
    return float(2.0 * n * b.max() / (a.sum() ** 2))


def _catalan_sequence(n: int) -> np.ndarray:
    """Catalan‑like coefficients a_i = C(2i,i)/4**i, normalised to sum 1."""
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * (i - 1) + 1) / (2.0 * i)
    a /= a.sum()
    return a


def _geometric_sequence(rate: float, n: int) -> np.ndarray:
    """Exponential decay a_i ∝ r^i with total mass 1."""
    r = max(0.0, min(0.999999, 1.0 - rate / n))
    a = r ** np.arange(n)
    return a / a.sum()


def _beta_sequence(p: float, n: int) -> np.ndarray:
    """
    Symmetric Beta(p,p) density on [0,1] discretised on n points.
    p>0 (p=1 gives the uniform distribution).
    """
    i = np.arange(n, dtype=float) + 0.5
    x = i / n
    # avoid overflow for very small p
    a = (x ** (p - 1)) * ((1.0 - x) ** (p - 1))
    a_sum = a.sum()
    if a_sum == 0.0 or np.isnan(a_sum):
        # fallback to uniform if numerical trouble
        a = np.ones(n, dtype=float)
    a /= a.sum()
    return a


def _hann_sequence(n: int) -> np.ndarray:
    """Raised‑cosine (Hann) window, non‑negative and zero at the ends."""
    i = np.arange(n, dtype=float)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def _gaussian_sequence(sigma_frac: float, n: int) -> np.ndarray:
    """Gaussian‑shaped window centred at the middle of the interval."""
    i = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a

            alpha = min(alpha * 1.15, 5000.0)

        # soft‑max weights (focus on the current maximal entry)
        w = np.exp(alpha * (b - max_b))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smooth max
        w_rev = w[::-1]
        conv_full = _convolve_fft(a, w_rev)   # length 3n‑2
        idx_slice = (len(b) - 1) - np.arange(n)
        grad = 2.0 * conv_full[idx_slice]

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        t = iteration + 1
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + adam_eps)

        # propose new point and project onto the simplex
        a_candidate = _proj_simplex(a - step)
        cand_score = _eval_score(a_candidate)

        improved = False
        if cand_score < cur_score:
            a = a_candidate
            cur_score = cand_score
            improved = True
        else:
            # fallback: simple back‑tracking line search
            lr_local = lr * 0.5
            for _ in range(5):
                a_candidate = _proj_simplex(a - lr_local * grad)
                cand_score = _eval_score(a_candidate)
                if cand_score < cur_score:
                    a = a_candidate
                    cur_score = cand_score
                    improved = True
                    break
                lr_local *= 0.5

        # ---------- bookkeeping ----------
        if improved:
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{iteration:06d}] NEW BEST {best_score:.6f} (α={alpha:.2f}, lr={lr:.5f})")
                no_improve = 0
            else:
                no_improve += 1
        else:
            # no improvement – shrink step size a little
            lr = max(lr * 0.8, 1e-5)
            no_improve += 1

        # periodic decay of the base learning rate
        if iteration % 500 == 0 and iteration > 0:
            lr = max(lr * 0.9, 1e-5)

        # restart if progress stalls
        if no_improve > max_no_improve:
            # small random perturbation around the best point seen
            a = best_seq + 0.015 * np.random.randn(n)
            a = np.maximum(a, 0.0)
            a = _proj_simplex(a)
            cur_score = _eval_score(a)
            lr = 0.02
            alpha = 5.0
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[{iteration:06d}] RESTART (score={cur_score:.6f})")

        iteration += 1

        # occasional progress report
        if iteration % 2000 == 0:
            elapsed = time.time() - start_time
            print(f"[{iteration:06d}] iter, best={best_score:.6f}, cur={cur_score:.6f}, "
                  f"time={elapsed:.1f}s, α={alpha:.2f}")

    print(f"[done] best score found = {best_score:.6f}")
    # Return as a plain Python list, as required by the harness
    return best_seq.tolist()
