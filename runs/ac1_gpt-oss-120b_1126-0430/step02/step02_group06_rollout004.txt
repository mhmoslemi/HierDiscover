def _adam_refine(seq, deadline):
    n = len(seq)
    best_seq = seq.copy()
    best_score = _fast_evaluation(seq)
    # parameters:
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.1
    # log representation to keep positivity
    log_seq = np.log(seq + 1e-15)  # avoid log(0)
    m = np.zeros_like(seq)
    v = np.zeros_like(seq)
    it = 0
    while time.time() < deadline:
        # compute seq from log and renorm
        seq = np.exp(log_seq)
        seq_sum = seq.sum()
        if seq_sum == 0:
            seq = np.full(n, 1.0/n)
            log_seq = np.log(seq)
        else:
            seq /= seq_sum
        # convolution
        fft_len = 1 << ((2*n-1).bit_length())
        fa = np.fft.rfft(seq, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[:2*n-1]
        M = conv.max()
        # get active indices
        tol = 1e-12
        active = np.where(conv >= M - tol)[0]
        # compute subgradient g: g_i = sum_{k in active} seq_{k-i} where valid
        g = np.zeros_like(seq)
        for k in active:
            lo = max(0, k-(n-1))
            hi = min(k, n-1)
            i_range = np.arange(lo, hi+1)
            j_range = k - i_range
            g[i_range] += seq[j_range]
        # Adam updates on log_seq:
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)

        it += 1
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        log_seq = log_seq - step

        # evaluate
        score = _fast_evaluation(seq)  # note: after renorm but before next step? maybe compute after step?
        if score < best_score - 1e-12:
            best_score = score
            best_seq = seq.copy()
        # optionally print progress
        if it % 500 == 0:
            print(f"[Adam] iter {it:6d} score {best_score:.6f}")

    return best_seq

The goal is to minimise the metric

    2·n·max(conv(a,a)) / (sum(a))²

over non‑negative vectors a.  Since the metric is scale‑invariant we fix
∑a = 1 and minimise the peak of the self‑convolution.

The optimiser works in three stages:

1. **Generate a diverse set of seed distributions**  
   – symmetric Beta(α,α) (includes the classic arcsine α=0.5)  
   – a few asymmetric Beta(α,β) mixes  
   – sin‑power shapes  sin(πx)ᵖ  
   – short random Dirichlet draws.  

   All seeds are normalised to sum = 1 and have length nₛₘₐₗₗ≈2000
   (fast to evaluate).

2. **Adam‑style Exponentiated‑Gradient descent on the simplex**  
   The variables are kept in log‑space (`log_a`).  At each iteration we
   compute the convolution via an FFT, locate the *active* (maximal)
   indices and form the sub‑gradient  
   g_i = Σ_{k∈active} a_{k‑i}.  
   Adam’s moment estimates give a per‑coordinate adaptive step size;
   the update is `log_a ← log_a – step` which corresponds to a

   update the vector is re‑normalised to stay on the simplex.

   The routine runs until the allocated time for the current seed

3. **Select the best seed, upscale to the final length (≈5000) and

# ----------------------------------------------------------------------
# Fast evaluation (identical to the supplied evaluate_sequence but without
# the security checks).  The result is a float, not np.inf for valid input.
# ----------------------------------------------------------------------
def _fast_evaluation(seq: np.ndarray) -> float:
    """Return the metric 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    n = seq.size
    if n == 0:
        return float("inf")
    # FFT length: next power of two ≥ 2·n‑1
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------
def _beta_sequence(alpha: float, beta: float, length: int) -> np.ndarray:
    """Symmetric or asymmetric Beta(α,β) density sampled on (0,1)."""
    # midpoint sampling avoids the singular endpoints.
    xs = (np.arange(length, dtype=float) + 0.5) / length
    # compute the (unnormalised) density
    a = np.power(xs, alpha - 1.0, out=np.empty_like(xs))
    b = np.power(1.0 - xs, beta - 1.0, out=np.empty_like(xs))
    seq = a * b
    # replace possible infinities caused by very small exponents
    seq[np.isinf(seq)] = 0.0
    seq = np.maximum(seq, 0.0)
    if seq.sum() == 0.0:
        seq = np.full(length, 1.0 / length)
    else:
        seq /= seq.sum()
    return seq


def _sin_power_sequence(p: float, length: int) -> np.ndarray:
    """Sequence a_i ∝ sin(π·(i+0.5)/length)^p, p may be negative."""
    xs = (np.arange(length, dtype=float) + 0.5) / length
    vals = np.sin(np.pi * xs) ** p
    vals = np.maximum(vals, 0.0)
    if vals.sum() == 0.0:
        vals = np.full(length, 1.0 / length)
    else:
        vals /= vals.sum()
    return vals


def _random_dirichlet(length: int, concentration: float = 0.5) -> np.ndarray:
    """Draw a random point from Dirichlet(concentration,…,concentration)."""
    seq = np.random.gamma(concentration, 1.0, size=length)
    if seq.sum() == 0.0:
        seq = np.full(length, 1.0 / length)
    else:
        seq /= seq.sum()
    return seq


# ----------------------------------------------------------------------
# Adam‑style exponentiated‑gradient minimiser on the simplex.
# ----------------------------------------------------------------------
def _adam_refine(seq: np.ndarray, deadline: float) -> np.ndarray:
    """
    Refine ``seq`` (non‑negative, sum = 1) using an Adam‑style
    multiplicative update on log‑coefficients.  The routine stops when
    ``time.time()`` reaches ``deadline`` and returns the best vector seen.
    """
    n = seq.size
    # initialise log‑variables (add tiny epsilon to avoid log(0))
    eps = 1e-15
    log_seq = np.log(seq + eps)

    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    lr = 0.12          # base learning rate (controls the size of the
                       # multiplicative step)
    m = np.zeros_like(seq)
    v = np.zeros_like(seq)
    t = 0

    # pre‑compute FFT length
    fft_len = 1 << ((2 * n - 1).bit_length())

    best_seq = None
    best_score = float("inf")
    it = 0

    while time.time() < deadline:
        # ----- obtain current probability vector and normalise -----
        seq = np.exp(log_seq)
        sum_seq = seq.sum()
        if sum_seq == 0.0:
            seq = np.full(n, 1.0 / n)
            log_seq = np.log(seq)
        else:
            seq /= sum_seq

        # ----- convolution and active set -----
        fa = np.fft.rfft(seq, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        M = conv.max()
        # treat values within a tiny tolerance as active
        tol = 1e-12
        active = np.where(conv >= M - tol)[0]

        # ----- sub‑gradient of the max‑operator -----
        g = np.zeros_like(seq)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            i_range = np.arange(lo, hi + 1, dtype=int)
            j_range = k - i_range
            g[i_range] += seq[j_range]

        # ----- Adam moment updates -----
        t += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # ----- multiplicative step (log‑space update) -----
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        log_seq -= step

        # optional clipping to keep numbers in a reasonable range
        np.clip(log_seq, -30.0, 30.0, out=log_seq)

        # ----- evaluate and keep the best -----
        score = _fast_evaluation(seq)
        if score < best_score - 1e-12:
            best_score = score
            best_seq = seq.copy()
            if it % 500 == 0:
                print(f"[Adam] iter {it:6d}  best {best_score:.6f}")

        # occasional tiny random jitter to escape stalls
        if np.random.rand() < 0.001:
            j = np.random.randint(n)
            seq[j] = max(0.0, seq[j] + np.random.randn() * 1e-3)
            seq /= seq.sum()
            log_seq = np.log(seq + eps)

        it += 1

    # If nothing improved (unlikely) return the last normalised vector
    if best_seq is None:
        seq = np.exp(log_seq)
        seq /= seq.sum()
        best_seq = seq
    return best_seq

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Produce a low‑score step‑function.

    The algorithm:
      * generate a collection of diverse seeds (Beta, sin‑power, random)
      * run a short Adam‑style refinement on each seed (time‑sliced)

      * upscale it to length 5000 and run a final refinement
    The returned list contains non‑negative numbers whose sum is 1.
    """
    np.random.seed(seed)
    start = time.time()
    # a small safety margin (the harness will penalise >budget_s)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

        seeds.append((f"beta_sym_{a:.2f}", _beta_sequence(a, a, n_small)))

    for a, b in asym_params:

    # sin‑power shapes (p negative → heavier edges, p positive → flatter)

    # a few random Dirichlet draws (concentration < 1 gives spikier vectors)
    for i in range(4):
        seeds.append((f"rand_dir_{i}", _random_dirichlet(n_small, concentration=0.4)))

    # ------------------------------------------------------------------
    # 3.  Run a time‑sliced Adam refinement on each seed.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    total_seeds = len(seeds)

    while seeds and time.time() < deadline:
        # time remaining and a fair share for the remaining seeds
        time_left = deadline - time.time()
        # guarantee at least a few seconds per remaining seed
        per_seed = max(5.0, time_left / (len(seeds) + 1))
        name, seq0 = seeds.pop(0)

        # refine this seed
        seed_deadline = time.time() + per_seed

        if score < best_score:

            best_seq = refined.copy()
            print(f"*** new global best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Upscale the best vector to the final length and do a final pass.
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never happen)
        best_seq = np.full(n_small, 1.0 / n_small)

    # final length (the original challenge used 5000)
    n_big = 5000
    # simple linear interpolation from the small representation to the big one
    x_small = np.arange(n_small, dtype=float)
    x_big = np.linspace(0, n_small - 1, n_big)
    up_seq = np.interp(x_big, x_small, best_seq)

    # enforce non‑negativity and renormalise
    up_seq = np.maximum(up_seq, 0.0)
    if up_seq.sum() == 0.0:
        up_seq = np.full(n_big, 1.0 / n_big)
    else:
        up_seq /= up_seq.sum()

        best_seq = final_seq
        best_score = final_score

    # 5.  Return the best sequence as a plain Python list.
    # ------------------------------------------------------------------
