def _seq_from_params(raw_len, raw_val, n):
    # compute block fractions via softmax of raw_len
    l_frac = np.exp(raw_len)
    l_frac = l_frac / l_frac.sum()
    # approximated integer lengths
    lengths = np.floor(l_frac * n).astype(int)
    # adjust remainder
    diff = n - lengths.sum()
    if diff > 0:
        # add 1 to largest fractions
        add_idx = np.argsort(l_frac)[-diff:]
        lengths[add_idx] += 1
    elif diff < 0:
        # subtract from smallest fractions
        sub_idx = np.argsort(l_frac)[: -diff]
        lengths[sub_idx] -= 1
    # ensure nonzero lengths (set min 1)
    # maybe for any zero length assign 1 and adjust
    # but we can ignore as diff adjust ensures total.
    # compute values
    vals = np.exp(raw_val)
    # to maintain monotonic decreasing, sort vals descending and align with lengths accordingly; but we can assign values after sorting.
    # We'll assign each block a value; we may sort by value descending and also lengths sorted accordingly for monotonic shape.
    order = np.argsort(vals)[::-1]  # descending
    lengths = lengths[order]
    vals = vals[order]

    # Build sequence
    a = np.concatenate([np.full(l, v) for l, v in zip(lengths, vals) if l > 0])

    return a

"""
Hybrid block‑parameter search + Adam‑based smooth‑max gradient descent.

1. **Block‑parameter search** (coarse stage)  
   – Represent a step‑function with a small number `B` of constant blocks.  
   – Block lengths are a softmax of `raw_len` (so they always sum to 1).  
   – Block heights are `exp(raw_val)`.  
   – The full vector of length `n` is built, sorted in decreasing order
     (monotonically non‑increasing step‑function) and projected onto the
     probability simplex (∑a_i = 1, a_i ≥ 0).  
   – Random perturbations of the low‑dimensional parameters are accepted
     if they improve the exact objective `2·n·max(conv(a,a))`.  The
     perturbation scale is slowly reduced, giving a simple stochastic‑hill‑climbing
     optimiser that can explore millions of candidates quickly.

2. **Fine‑tuning** (Adam‑based smooth‑max descent)  
   – Starting from the best block‑based sequence, minimise the smooth
     approximation of the max‑convolution using the log‑sum‑exp trick:
       `soft_max(b) = (1/α)·log Σ exp(α·b)`.  
   – Its gradient is `2·a ⊛ w_rev` where `w_i ∝ exp(α·(b_i−max(b)))`.  
   – Adam (β₁=0.9, β₂=0.999) provides an adaptive step size;
     after each update the iterate is projected back onto the simplex.  
   – The soft‑max temperature `α` is increased slowly so that the optimisation
     gradually concentrates on the true maximum.

The whole routine runs for the allotted time (≈ 1000 s) and returns the
best sequence found.  All computations use FFT‑based convolution
(`O(n log n)`) and stay well within the time budget even for `n≈8000`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    # Fast projection (Duchi et al. 2008)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of real vectors x and y using a real FFT."""
    n = x.size + y.size - 1
    size = 1 << (n - 1).bit_length()          # next power of two
    X = np.fft.rfft(x, size)
    Y = np.fft.rfft(y, size)
    Z = X * Y
    conv = np.fft.irfft(Z, size)[:n]
    return conv.real


def _eval_score(a: np.ndarray) -> float:
    """Exact value of evaluate_sequence for a non‑negative simplex vector a."""
    n = a.size
    b = _convolve_fft(a, a)
    max_b = b.max()
    return float(2.0 * n * max_b / (a.sum() ** 2))


def _seq_from_params(raw_len: np.ndarray,
                     raw_val: np.ndarray,
                     n: int) -> np.ndarray:
    """
    Build a monotone non‑increasing step‑function of length n from
    low‑dimensional parameters.

    Parameters
    ----------
    raw_len : (B,) array
        Unnormalised log‑lengths; after softmax they become block fractions.
    raw_val : (B,) array
        Log‑heights; heights are exp(raw_val).
    n : int
        Desired length of the full sequence.

    Returns
    -------
    a : (n,) ndarray
        Non‑negative vector on the probability simplex.
    """
    B = raw_len.shape[0]

    # --- block fractions (softmax) ------------------------------------
    exp_len = np.exp(raw_len - np.max(raw_len))      # for numerical stability
    frac = exp_len / exp_len.sum()

    # Convert fractions to integer lengths that sum to n
    lengths = np.floor(frac * n).astype(int)
    diff = n - lengths.sum()
    if diff > 0:
        # add one to the `diff` largest fractions
        add_idx = np.argsort(frac)[-diff:]
        lengths[add_idx] += 1
    elif diff < 0:
        # remove one from the `-diff` smallest fractions (ensuring ≥1)
        sub_idx = np.argsort(frac)[: -diff]
        lengths[sub_idx] = np.maximum(lengths[sub_idx] - 1, 1)

    # --- block heights ------------------------------------------------
    heights = np.exp(raw_val)               # always positive

    # Sort blocks by decreasing height to enforce monotonicity
    order = np.argsort(heights)[::-1]
    lengths = lengths[order]
    heights = heights[order]

    # Build full vector
    parts = [np.full(l, h, dtype=float) for l, h in zip(lengths, heights) if l > 0]
    if not parts:
        # fallback to uniform vector if something went wrong
        a = np.full(n, 1.0 / n, dtype=float)
    else:
        a = np.concatenate(parts)
        if a.size > n:
            a = a[:n]
        elif a.size < n:
            # pad with smallest height (last element after sorting)
            pad = np.full(n - a.size, heights[-1], dtype=float)
            a = np.concatenate([a, pad])
    # Project onto the simplex to guarantee Σa = 1 and non‑negativity
    a = _proj_simplex(a)
    return a


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a non‑negative height sequence that minimises
    ``evaluate_sequence``.  Returns the best sequence (plain list)
    found before the time budget expires.

    The algorithm consists of a cheap block‑parameter stochastic search
    followed by an Adam‑based smooth‑max gradient refinement.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # small safety margin

    # ------------------------------------------------------------------
    # Problem size – a compromise between expressiveness and speed
    # ------------------------------------------------------------------
    n = 8000                     # length of the step function

    # ------------------------------------------------------------------
    # Stage 1 – block‑parameter stochastic hill climbing
    # ------------------------------------------------------------------
    B = 6                        # number of constant blocks
    # initial random parameters
    best_raw_len = np.random.randn(B)
    best_raw_val = np.random.randn(B)
    best_seq = _seq_from_params(best_raw_len, best_raw_val, n)
    best_score = _eval_score(best_seq)

    # keep also a few hand‑crafted seeds (Catalan, geometric, random simplex)
    def _catalan_sequence(nn: int) -> np.ndarray:
        a = np.empty(nn, dtype=float)
        a[0] = 1.0
        for i in range(1, nn):
            a[i] = a[i - 1] * (2 * (i - 1) + 1) / (2.0 * i)
        return _proj_simplex(a)

    seeds = [
        _catalan_sequence(n),
        _proj_simplex((1 - 0.9) * (0.9 ** np.arange(n))),      # geometric r=0.9
        _proj_simplex(np.random.rand(n))                     # pure random simplex
    ]
    for s in seeds:
        sc = _eval_score(s)
        if sc < best_score:
            best_score = sc
            best_seq = s.copy()
            # initialise raw parameters to match this seed (roughly)
            # we approximate by fitting a decreasing exponential; not exact.
    print(f"[stage 1‑init] n={n}, score={best_score:.6f}")

    # Hyper‑parameters for the stochastic search
    sigma_len = 0.5          # step size for length logits
    sigma_val = 0.8          # step size for log‑heights
    min_sigma_len = 1e-4
    min_sigma_val = 1e-4
    it = 0
    no_imp = 0

    while time.time() < deadline - 200:   # reserve ~200 s for fine tuning
        it += 1
        # occasional large jump to escape plateaus
        if np.random.rand() < 0.03:
            cand_len = np.random.randn(B)
            cand_val = np.random.randn(B)
        else:
            cand_len = best_raw_len + sigma_len * np.random.randn(B)
            cand_val = best_raw_val + sigma_val * np.random.randn(B)

        cand_seq = _seq_from_params(cand_len, cand_val, n)
        cand_score = _eval_score(cand_seq)

        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            best_raw_len = cand_len.copy()
            best_raw_val = cand_val.copy()
            sigma_len = max(sigma_len * 0.995, min_sigma_len)
            sigma_val = max(sigma_val * 0.995, min_sigma_val)
            no_imp = 0
            print(f"[stage 1] it={it:06d} new best {best_score:.6f}")
        else:
            no_imp += 1
            if no_imp > 5000:
                # small random restart around the current best
                best_raw_len = best_raw_len + 0.2 * np.random.randn(B)
                best_raw_val = best_raw_val + 0.2 * np.random.randn(B)
                no_imp = 0
                print(f"[stage 1] it={it:06d} restart (score={best_score:.6f})")

    # ------------------------------------------------------------------
    # Stage 2 – Adam smooth‑max gradient descent for fine optimisation
    # ------------------------------------------------------------------
    a = best_seq.copy()
    cur_score = best_score

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.02               # base learning rate
    alpha = 20.0            # initial soft‑max sharpness
    iter2 = 0
    no_improve2 = 0

    while time.time() < deadline:
        iter2 += 1
        # convolution of a with itself
        b = _convolve_fft(a, a)
        max_b = b.max()

        # soft‑max weights (focus on the currently maximal entry)
        w = np.exp(alpha * (b - max_b))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.ones_like(b) / b.size
        else:
            w /= w_sum

        # gradient of the smooth max (2·a ⊛ w_rev)
        grad = 2.0 * _convolve_fft(a, w[::-1])

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** iter2)
        v_hat = v / (1.0 - beta2 ** iter2)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = _proj_simplex(a - step)
        cand_score = _eval_score(a_candidate)

        if cand_score < cur_score:
            a = a_candidate
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[adam] it={iter2:06d} new best {best_score:.6f} (α={alpha:.1f})")
            no_improve2 = 0
        else:
            # no improvement → shrink learning rate a little
            lr *= 0.95
            no_improve2 += 1

        # schedule adjustments
        if iter2 % 500 == 0:
            alpha = min(alpha * 1.15, 5000.0)   # sharpen soft‑max slowly
            lr = max(lr * 0.9, 5e-4)

        # occasional restart if we stall too long
        if no_improve2 > 4000:
            a = best_seq + 0.02 * np.random.randn(n)
            a = np.maximum(a, 0.0)
            a = _proj_simplex(a)
            cur_score = _eval_score(a)
            lr = 0.02
            alpha = 20.0
            m.fill(0)
            v.fill(0)
            no_improve2 = 0
            print(f"[adam] it={iter2:06d} restart (score={cur_score:.6f})")

    print(f"[done] best score found = {best_score:.6f}")
    # Return a plain Python list as required by the harness
    return best_seq.tolist()
