"""
Projected‑gradient descent on the Lᵣ norm of the self‑convolution.

We work with a probability vector p (sum = 1).  The evaluation score is

    score(p) = 2·n·max_k (p * p)_k

which does not depend on an overall scaling of the sequence.  Starting from
the Catalan (arcsine) distribution we minimise an Lᵣ norm of the convolution
with r slowly increased (r≈4→12).  The gradient of

    F_r(p) = ( Σ_k (p*p)_k^r )^{1/r}

is
    ∇F_r(p) = 2·r·( ( (p*p)^{r‑1} ) ↔  a )   (↔ = cross‑correlation)

and can be evaluated by two FFTs.  After a gradient step we project back onto
the probability simplex (non‑negative, sum = 1).  A simple back‑tracking line
search guarantees monotone improvement; occasional random swaps help to escape
plateaus.  The loop runs until the time budget expires (≈1000 s) and the best
sequence found is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex Σx_i = 1, x_i ≥ 0
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σ x = 1}."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = v - theta
    w[w < 0] = 0.0
    # re‑normalise to avoid tiny numerical drift
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w.fill(1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast evaluation used during the optimisation loop (no safety checks)
# ----------------------------------------------------------------------
def _fast_score(p: np.ndarray) -> float:
    """Return 2·n·max(p*p) assuming Σ p = 1."""
    n = p.size
    if n == 0:
        return float("inf")
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(p, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Catalan (arcsine) seed – un‑normalised
# ----------------------------------------------------------------------
def _catalan_sequence(length: int) -> np.ndarray:
    seq = np.empty(length, dtype=float)
    a = 1.0
    seq[0] = a
    for i in range(1, length):
        a *= (2 * i - 1) / (2.0 * i)          # a_i = a_{i‑1}·(2i‑1)/(2i)
        seq[i] = a
    return seq


# ----------------------------------------------------------------------
# Gradient of the L_r norm of (p * p)  (see docstring)
# ----------------------------------------------------------------------
def _gradient(p: np.ndarray, r: int, fft_len: int) -> np.ndarray:
    """Compute ∇F_r(p) for r ≥ 2 using FFTs."""
    n = p.size
    # forward FFT of p (used for both convolutions)
    Fa = np.fft.rfft(p, fft_len)

    # (p * p)  – linear convolution, length 2n‑1
    conv_full = np.fft.irfft(Fa * Fa, fft_len)
    conv = conv_full[: 2 * n - 1]

    # (p * p)^{r‑1}
    cp = np.maximum(conv, 0.0) ** (r - 1)

    # cross‑correlation a ↔ (p*p)^{r‑1} = a * reverse(cp)
    rev_cp = cp[::-1]
    Fr = np.fft.rfft(rev_cp, fft_len)
    corr_full = np.fft.irfft(Fa * Fr, fft_len)

    # indexing: L = len(conv)‑1 = 2n‑2
    L = 2 * n - 2
    idx = L - np.arange(n)          # indices L, L‑1, … , L‑(n‑1)
    grad = 2.0 * r * corr_full[idx]
    return grad


# ----------------------------------------------------------------------
# Core optimiser – projected gradient descent with back‑tracking line search
# ----------------------------------------------------------------------
def _optimise(p0: np.ndarray, deadline: float) -> np.ndarray:
    """Improves p0 (probability vector) until the deadline is reached."""
    n = p0.size
    # FFT length large enough for the longest convolution (3n‑2)
    fft_len = 1 << ((3 * n - 2).bit_length())

    p = p0.copy()
    best = p.copy()
    best_score = _fast_score(best)

    # hyper‑parameters
    step = 0.1                     # initial step length
    r = 4                          # start exponent
    r_max = 12                     # final exponent
    r_change_iter = 2000           # how many iterations before r is increased
    iteration = 0

    while time.time() < deadline:
        # compute gradient for the current r
        g = _gradient(p, r, fft_len)

        # ---- back‑tracking line search ------------------------------------
        eta = step
        improved = False
        while eta > 1e-12:
            cand = p - eta * g
            cand = _project_simplex(cand)
            sc = _fast_score(cand)
            if sc < best_score - 1e-14:
                p = cand
                best = cand.copy()
                best_score = sc
                improved = True
                break
            eta *= 0.5

        # ---- if no improvement, perform a tiny random swap -----------------
        if not improved:
            i, j = np.random.randint(n, size=2)
            if i != j and p[i] > 0:
                eps = min(p[i], np.random.rand() * p[i] * 0.05)
                cand = p.copy()
                cand[i] -= eps
                cand[j] += eps
                cand = _project_simplex(cand)
                sc = _fast_score(cand)

                    best_score = sc

        else:
            step *= 0.9

        # slowly increase the exponent r (focus more on the true maximum)
        if iteration % r_change_iter == 0 and r < r_max:

        iteration += 1

        # occasional progress report (useful when run interactively)
