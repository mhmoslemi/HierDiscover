def _fft_convolve_two(a, b):
    n = a.shape[0]; m = b.shape[0]
    size = 1 << ((n + m - 1).bit_length())
    Fa = np.fft.rfft(a, size)
    Fb = np.fft.rfft(b, size)
    conv = np.fft.irfft(Fa * Fb, size)
    return conv[: n + m - 1]

def solve_lp(f, M):
    n = f.shape[0]
    m = 2 * n - 1
    # Build sparse matrix
    data = []
    row_idx = []
    col_idx = []
    for k in range(m):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # for i from i_start to i_end: f_{k-i} coefficient for variable g_i
        # This can be vectorized?
        for i in range(i_start, i_end + 1):
            data.append(f[k - i])    # coefficient for g_i
            row_idx.append(k)
            col_idx.append(i)
    A = scipy.sparse.csr_matrix((data, (row_idx, col_idx)), shape=(m, n))
    b = np.full(m, M, dtype=np.float64)
    c = -np.ones(n)   # maximize sum g => minimize -sum
    bounds = [(0, None)] * n
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        # fallback: return f (no improvement)
        return f.copy()
    g = res.x
    return g

"""
Propose a near‑optimal step‑function sequence.

The algorithm follows the scheme described in the problem statement:
1. Initialise with the Catalan (binomial) distribution, which has the
   attractive property that its infinite self‑convolution is flat.
   After truncation the metric is π/2 ≈ 1.5708.
2. Repeatedly solve a linear programme that maximises the L1‑mass of a
   candidate vector g under the linear constraints   f * g ≤ M,
   where f is the current sequence and M is the current maximum
   convolution entry.  The LP therefore finds a direction that can
   increase the total mass without raising the worst convolution entry.
3. Mix the current sequence f with the normalised LP‑solution g and
   keep the update only if the evaluation metric decreases.
4. After a few LP‑updates a cheap multiplicative‑gradient refinement
   (smooth approximation of the ∞‑norm) is applied to shave off the
   remaining gap.
The whole search runs within the given time budget and returns the best
sequence found (as a plain list of floats).  All coefficients are
non‑negative and the routine never performs any I/O.
"""

import time
import numpy as np
import scipy.optimize
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Fast linear convolution via real FFT (full “linear” mode)
# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray | None = None) -> np.ndarray:
    """Return the full linear convolution of a with b (or a with itself
    if b is None) using a real FFT.  The result has length len(a)+len(b)-1.
    """
    if b is None:
        b = a
    n = a.shape[0]
    m = b.shape[0]
    size = 1 << ((n + m - 1).bit_length())        # next power of two
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    conv = np.fft.irfft(fa * fb, size)
    return conv[: n + m - 1]


# ----------------------------------------------------------------------
# Catalan‑type coefficients, normalised to L¹‑norm = 1
# ----------------------------------------------------------------------
def _catalan_normalised(n: int) -> np.ndarray:
    """First n Catalan‑type numbers a_i = C(2i,i)/4**i,
    normalised so that sum_i a_i = 1.
    Recurrence a_{i} = a_{i-1} * (2i-1)/(2i)  (i≥1) is numerically stable.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * (i - 1) + 1) / (2.0 * i)   # = a_{i-1} * (2i-1)/(2i)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Linear programme: maximise sum(g) subject to f*g ≤ M, g ≥ 0
# ----------------------------------------------------------------------
def _solve_lp(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve   maximise 1·g   s.t.   (f * g)_k ≤ M  for all k,
                              g_i ≥ 0.
    Returns the optimal g (may be the zero vector if LP fails).
    """
    n = f.shape[0]
    m = 2 * n - 1                               # length of full convolution
    # Build sparse constraint matrix A (shape m × n)
    # Row k corresponds to (f * g)_k = Σ_i f_{k-i} g_i   (indices out of range → 0)
    data = []
    rows = []
    cols = []
    for k in range(m):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # vectorised contribution for this row
        coeffs = f[k - np.arange(i_start, i_end + 1)]
        idx = np.arange(i_start, i_end + 1)
        data.append(coeffs)
        rows.append(np.full_like(idx, k))
        cols.append(idx)
    # flatten the lists of arrays
    data = np.concatenate(data)
    rows = np.concatenate(rows)
    cols = np.concatenate(cols)
    A = sp.csr_matrix((data, (rows, cols)), shape=(m, n))

    # Right‑hand side
    b = np.full(m, M, dtype=np.float64)

    # Objective: minimise -sum(g)  (equivalent to maximise sum(g))
    c = -np.ones(n, dtype=np.float64)

    # Variable bounds: g_i ≥ 0
    bounds = [(0, None)] * n

    # Solve with HiGHS (fast interior‑point/simplex hybrid)
    res = scipy.optimize.linprog(
        c,
        A_ub=A,
        b_ub=b,
        bounds=bounds,
        method="highs",
        options={"presolve": True, "time_limit": 30},
    )
    if res.success:
        return res.x
    # Fall back – return a copy of f (no improvement)
    return f.copy()


# ----------------------------------------------------------------------
# Multiplicative gradient refinement (soft‑max approximation of max)
# ----------------------------------------------------------------------
def _smooth_max_refine(a: np.ndarray,
                       max_iters: int,
                       deadline: float,
                       lam: float = 120.0,
                       eta: float = 0.3,
                       t_inc: float = 1.05,
                       t_dec: float = 0.5) -> np.ndarray:
    """
    Perform a few multiplicative‑gradient steps on the smoothed
    max‑convolution objective.
    """
    n = a.shape[0]
    best = a.copy()
    conv = _fft_convolve(a)
    best_score = 2.0 * n * conv.max()

    for _ in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _fft_convolve(a)
        # soft‑max weights over convolution entries
        w = np.exp(lam * conv - lam * conv.max())
        w /= w.sum()

        # gradient of the smooth max = 2 * correlation(w, a)
        # correlation can be obtained via convolution with the reversed w
        w_rev = w[::-1]
        grad_full = _fft_convolve(a, w_rev)                 # length 3n‑2
        shift = 2 * n - 2
        grad = 2.0 * grad_full[shift:shift + n]             # size n

        # multiplicative update (ensures positivity)
        step = eta
        for _ in range(7):
            a_new = a * np.exp(-step * grad)
            s = a_new.sum()
            if s == 0.0:
                # degenerate – fall back to a small random perturbation
                a_new = np.random.random(n)
                a_new /= a_new.sum()
            else:
                a_new /= s
            new_score = 2.0 * n * _fft_convolve(a_new).max()
            if new_score < best_score:
                best_score = new_score
                best = a_new.copy()
                a = a_new
                eta = step * t_inc
                break
            step *= t_dec
        else:
            # no step succeeded – shrink learning rate
            eta *= t_dec
    return best


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence``.
    The routine respects the 1000 s time budget and returns the best
    sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # 1) Choose a moderate length – large enough to have freedom but
    #    small enough for the LP to finish quickly.
    # ------------------------------------------------------------------
    n = 2000

    # ------------------------------------------------------------------
    # 2) Initialise with the Catalan distribution (near‑optimal analytically)
    # ------------------------------------------------------------------
    a = _catalan_normalised(n)

    # ------------------------------------------------------------------
    # 3) Iterative LP‑based mass‑augmentation.
    #    We perform a few (≤8) updates; each iteration solves a linear
    #    programme and mixes the result with the current vector.
    # ------------------------------------------------------------------
    t_mix = 0.5            # convex‑combination weight
    for lp_iter in range(8):
        if time.time() >= deadline:
            break
        conv = _fft_convolve(a)
        M = conv.max()
        # Solve LP: maximise Σ g_i  subject to (a * g) ≤ M
        g = _solve_lp(a, M)
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-12:      # no improvement possible
            break
        # Normalise g to unit L¹‑norm (the same normalization as a)
        g_norm = g / sum_g
        # Convex combination
        a_candidate = (1.0 - t_mix) * a + t_mix * g_norm
        # Evaluate
        score_candidate = 2.0 * n * _fft_convolve(a_candidate).max()
        score_current = 2.0 * n * conv.max()
        if score_candidate < score_current:
            a = a_candidate
            # Slightly increase mixing to move faster
            t_mix = min(0.9, t_mix * 1.10)
            print(f"[LP {lp_iter}] improved score: {score_candidate:.9f}")
        else:
            # Reduce mixing factor – the LP direction was not useful enough
            t_mix *= 0.6

    # ------------------------------------------------------------------
    # 4) Optional smooth‑max gradient refinement (cheap, monotone).
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        a = _smooth_max_refine(
            a,
            max_iters=2000,
            deadline=deadline,
            lam=180.0,
            eta=0.25,
        )

    # ------------------------------------------------------------------
    # 5) Final sanity check and return.
    # ------------------------------------------------------------------
    final_score = 2.0 * n * _fft_convolve(a).max()
    print(f"[final] n={n}, final score ≈ {final_score:.9f}")

    # Convert to plain Python list (the harness will call evaluate_sequence)
    return a.tolist()
