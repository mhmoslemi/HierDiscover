def greedy_sidon(n, L, order):
    used_sum = np.zeros(2*n, dtype=np.int16) # counts per sum
    S = []
    for i in order:
        # compute sums i + j for j in S and i+i.

                can = False
                break
        if can and used_sum[2*i] < L:
            # add i
            for j in S:
                used_sum[i+j] += 1
            used_sum[2*i] += 1
            S.append(i)
    return S

import time
import numpy as np

def _greedy_sidon_set(n, max_per_sum, rng):
    used = np.zeros(2*n, dtype=np.int16)
    S = []
    order = rng.permutation(n)
    for i in order:
        if used[2*i] >= max_per_sum:
            continue
        # Check all sums with existing indices
        if S:
            sums = i + np.array(S, dtype=int)
            if np.any(used[sums] >= max_per_sum):
                continue
        # Accept i
        if S:
            sums = i + np.array(S, dtype=int)
            used[sums] += 1
        used[2*i] += 1
        S.append(int(i))
    return np.array(S, dtype=int), used

def _estimate_score(n, k, max_count):
    # Uniform weights on k points, each weight = 1/k.
    # max(conv) = max_count / k**2
    return 2.0 * n * max_count / (k * k)

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    evaluate_sequence() bound.  The search builds many
    low‑collision “Sidon‑like” supports of size k on a line of
    length n (max per sum = 2).  Uniform weights 1/k on the support give
    a cheap bound O = 2·n·max_count/k²; the best candidate is evaluated
    with the exact evaluate_sequence() function and returned.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5

    best_seq = None
    best_score = float('inf')
    best_est = float('inf')
    attempt = 0

    while time.time() < deadline:
        # Random length for diversity (between 3000 and 6000)
        n = int(rng.integers(3000, 6000))
        max_per_sum = 2          # allow at most two pairs per convolution index
        S, used = _greedy_sidon_set(n, max_per_sum, rng)
        k = S.size
        if k < 2:
            continue

        max_cnt = int(used.max())
        est = _estimate_score(n, k, max_cnt)

        # Keep only promising candidates (strictly better estimate)
        if est < best_est:
            # Build the full sequence (uniform weight on the support)
            a = np.zeros(n, dtype=float)
            a[S] = 1.0 / k

            # Compute the exact value using the provided evaluate_sequence()
            try:
                exact = evaluate_sequence(a.tolist())
            except Exception:
                exact = float('inf')

            print(f"[attempt {attempt}] n={n}, k={k}, max_cnt={max_cnt}, "
                  f"est={est:.6f}, exact={exact:.6f}")

            if exact < best_score:
                best_score = exact
                best_seq = a.tolist()
                best_est = est
                print(f"   ** new global best O={best_score:.6f}")

        attempt += 1

    # If search failed (unlikely) fall back to a uniform sequence of modest size.
    if best_seq is None:
        n = 2000
        a = np.full(n, 1.0 / n, dtype=float)
        best_seq = a.tolist()
        print("[fallback] returned uniform sequence")

    return best_seq

"""
search for a step‑function sequence that minimizes
    O = 2·n·max(conv(a,a)) / (sum(a)^2)

The bound depends only on the relative shape of the coefficients, so we
fix sum(a)=1.  A low value of O is obtained when the auto‑convolution
has few large entries.  We therefore build a support set S ⊂ {0,…,n‑1}
such that each possible sum i+j appears at most `max_per_sum` times
(Low‑collision “Sidon‑like” set).  With uniform weights 1/|S| on this
support the maximum convolution coefficient is

    max(conv) = max_count / |S|² ,

where `max_count` ≤ `max_per_sum`.  Hence the cheap estimate

    O_est = 2·n·max_count / |S|² .

The algorithm repeatedly constructs random supports (greedy insertion),
keeps the best one according to the exact `evaluate_sequence` call,
and finally returns the best sequence found.  The whole routine respects
the 1000 s time budget and never uses closures, lambdas or file I/O.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _greedy_low_collision_set(n: int, max_per_sum: int, rng: np.random.Generator):
    """
    Greedy construction of a set S ⊂ {0,…,n‑1} with at most `max_per_sum`
    pairs (i,j) (including i=j) sharing the same sum i+j.
    Returns the set as a NumPy array and the array of pair‑counts for each sum.
    """
    used = np.zeros(2 * n, dtype=np.int16)          # pair‑count for each possible sum
    S = []                                          # selected indices
    order = rng.permutation(n)                     # random insertion order

    for i in order:
        # check self‑sum
        if used[2 * i] >= max_per_sum:
            continue

        # check sums with already chosen indices
        if S:
            sums = i + np.array(S, dtype=int)
            if np.any(used[sums] >= max_per_sum):
                continue

        # accept i: update pair counts
        if S:
            used[i + np.array(S, dtype=int)] += 1
        used[2 * i] += 1
        S.append(int(i))

    return np.array(S, dtype=int), used


# ----------------------------------------------------------------------
def _estimate_score(n: int, k: int, max_count: int) -> float:
    """
    Cheap estimate of the evaluation bound for a uniform distribution
    on a support of size k with maximal pair‑count `max_count`.
    """
    return 2.0 * n * max_count / (k * k)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The search builds many low‑collision supports,
    evaluates the exact bound for the most promising candidates and
    returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5          # safety margin

    best_seq = None               # best full sequence (list of floats)
    best_score = float('inf')     # exact score from evaluate_sequence
    best_est = float('inf')       # best cheap estimate seen so far
    attempt = 0

    # --------------------------------------------------------------
    # Main stochastic search
    while time.time() < deadline:
        # random length (larger n gives more room for a large support)
        n = int(rng.integers(3000, 6000))
        max_per_sum = 2                               # allow at most two pairs per sum

        S, used = _greedy_low_collision_set(n, max_per_sum, rng)
        k = S.size
        if k < 2:
            continue

        max_cnt = int(used.max())
        est = _estimate_score(n, k, max_cnt)

        # only evaluate exactly if the cheap bound improves on the best known
        if est < best_est:
            a = np.zeros(n, dtype=float)
            a[S] = 1.0 / k                              # uniform weight on the support

            try:
                exact = evaluate_sequence(a.tolist())
            except Exception:
                exact = float('inf')

            print(f"[attempt {attempt}] n={n:4d}, |S|={k:3d}, max_cnt={max_cnt}, "
                  f"est={est:.6f}, exact={exact:.6f}")

            if exact < best_score:
                best_score = exact
                best_seq = a.tolist()
                best_est = est
                print(f"   ** new global best O={best_score:.6f}")

        attempt += 1

    # --------------------------------------------------------------
    # Fallback – should never be needed, but keeps the function safe
    if best_seq is None:
        n = 2000
        a = np.full(n, 1.0 / n, dtype=float)
        best_seq = a.tolist()
        print("[fallback] returning a uniform sequence.")

    return best_seq
