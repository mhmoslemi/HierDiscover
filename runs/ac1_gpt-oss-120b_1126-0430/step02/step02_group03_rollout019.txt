def propose_candidate(seed=42,budget_s=1000):
    import time, numpy as np
    np.random.seed(seed)
    deadline=time.time()+budget_s-1.0

    # Choose n based on budget - bigger can give lower theoretical bound but slower.
    n=20000

    # Prepare Catalan as baseline
    a=_catalan_normalised(n)

    best_a=a.copy()
    best_score=2.0*n*np.max(_fft_convolve(a))

    # Search over a few Beta-like families
    for _ in range(200):
        if time.time()>=deadline: break
        alpha= np.random.uniform(0.1,3.0)
        beta = np.random.uniform(0.1,3.0)
        i = np.arange(n, dtype=np.float64)
        a_candidate = (i+0.5)**(alpha-1) * (n - i - 0.5)**(beta-1)
        a_candidate[a_candidate<0]=0
        a_candidate_sum = a_candidate.sum()
        if a_candidate_sum==0: continue
        a_candidate /= a_candidate_sum
        score = 2.0*n*np.max(_fft_convolve(a_candidate))
        if score < best_score:
            best_score = score
            best_a = a_candidate.copy()

    # start from best_a for local refinement
    a = best_a.copy()
    current_score = best_score

    step = 0.2
    top_k = 5
    iter_count=0
    while time.time()<deadline:
        # compute convolution
        conv=_fft_convolve(a)
        max_val=conv.max()
        # compute gradient using top_k indices
        # choose indices of k with largest values
        if top_k < len(conv):
            # using argpartition for performance
            idx_top = np.argpartition(-conv, top_k-1)[:top_k]
        else:
            idx_top = np.arange(len(conv))
        grad=np.zeros_like(a)
        for k in idx_top:
            i_start = max(0, k - (n-1))
            i_end   = min(k, n-1)
            idx = np.arange(i_start, i_end+1)
            grad[idx] += a[k - idx]   # a_{k-i}
        grad /= len(idx_top)   # average

        # propose new a
        a_new = a - step*grad
        a_new = np.maximum(a_new,0.0)
        s = a_new.sum()
        if s>0:
            a_new/=s
        else:
            # fall back to random simplex point
            a_new = np.random.random(n)
            a_new/=a_new.sum()

        new_score=2.0*n*np.max(_fft_convolve(a_new))

        if new_score < current_score:
            a = a_new
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
                print(f"[improve] iter {iter_count} score {best_score:.9f}")
            # increase step a little to keep momentum
            step *= 1.02
        else:
            # no improvement, reduce step
            step *= 0.95
            # occasional random jitter to escape plateaus
            if np.random.rand() < 0.02:
                i,j = np.random.randint(0,n,2)
                delta = min(a[i], step*0.5)
                a[i] -= delta
                a[j] += delta
                a = np.maximum(a,0)
                a /= a.sum()
        iter_count += 1

    print(f"[final] best score {best_score:.9f}, n={n}")
    return best_a.tolist()
```

"""
Propose a low‑score step‑function by (a) starting from the Catalan
distribution (which gives the asymptotic value π/2 ≈ 1.5708) and a few
random Beta‑like shapes, (b) selecting the best of those candidates,
and (c) performing a long run of projected (sub‑)gradient descent on the
objective 2 n·max(conv) with respect to the probability simplex.
The gradient of the maximal convolution entry is obtained from the
active convolution indices; a small learning rate together with an
adaptive schedule gradually reshapes the sequence, pushing mass away
from the ends and lowering the peak of the self‑convolution.  The
routine respects the 1000 s time budget, prints occasional progress
messages and finally returns the best sequence found as a plain Python
list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast linear convolution (full length) using a real FFT.
# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length) via real FFT."""
    n = a.shape[0]
    # zero‑pad to the next power of two ≥ 2*n‑1 (FFT friendly)
    size = 1 << (2 * n - 1).bit_length()
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Catalan‑type coefficients (the classic construction).
# ----------------------------------------------------------------------
def _catalan_normalised(n: int) -> np.ndarray:
    """First n Catalan numbers a_i = C(2i,i)/4**i, normalised to sum‑1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        # recurrence a_i = a_{i‑1}·(2i‑1)/(2i)
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Beta‑like shapes: a_i ∝ (i+½)^{α‑1}·(n‑i‑½)^{β‑1}.
# ----------------------------------------------------------------------
def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray | None:
    """Beta‑type density on {0,…,n‑1} (shifted by ½ to avoid zeros)."""
    i = np.arange(n, dtype=np.float64)
    # the shift by 0.5 keeps the values finite for α,β < 1
    a = (i + 0.5) ** (alpha - 1.0) * (n - i - 0.5) ** (beta - 1.0)
    if np.any(np.isnan(a)) or np.any(np.isinf(a)):
        return None
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


# ----------------------------------------------------------------------
# Gradient of the maximal convolution entry (sub‑gradient).
# ----------------------------------------------------------------------
def _subgradient(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """Sum of sub‑gradients coming from the provided active convolution indices."""
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        # a_{k-i} contributes to ∂/∂a_i of (a*a)[k]
        g[idx] += a[k - idx]
    # average over the active set – a modest scaling improves stability
    g /= max(1, len(active))
    return g


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a sequence of non‑negative heights that minimises
    ``evaluate_sequence``.  The routine runs for at most *budget_s* seconds
    and returns the best sequence discovered.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : int, optional
        Time budget in seconds (default 1000).

    Returns
    -------
    list[float]
        The best sequence discovered (plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # tiny safety margin

    # ------------------------------------------------------------------
    # 1) Choose a working length.  Large enough to give the Catalan
    #    construction its asymptotic behaviour, small enough to allow
    #    many gradient steps within the time budget.
    # ------------------------------------------------------------------
    n = 12000                     # a compromise between quality and speed

    # ------------------------------------------------------------------
    # 2) Build a quick pool of starting points: Catalan + a few
    #    random Beta‑like shapes.  Keep the best one as the seed for the
    #    subsequent optimisation.
    # ------------------------------------------------------------------
    best_a = _catalan_normalised(n)
    best_score = 2.0 * n * np.max(_fft_convolve(best_a))
    print(f"[init] Catalan start, score {best_score:.9f}")

    # random Beta shapes – a very cheap global search
    for _ in range(150):
        if time.time() >= deadline:
            break
        # draw exponents roughly in the interval (0.4, 2.5); this covers
        # both left‑heavy and more centred profiles.
        alpha = np.random.uniform(0.4, 2.5)
        beta  = np.random.uniform(0.4, 2.5)
        a = _beta_shape(n, alpha, beta)
        if a is None:
            continue
        score = 2.0 * n * np.max(_fft_convolve(a))
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[init] random ({alpha:.2f},{beta:.2f}) → {best_score:.9f}")

    # ------------------------------------------------------------------
    # 3) Local refinement: projected sub‑gradient descent on the
    #    supremum of the convolution.  The step size decays slowly;
    #    occasional random jitter helps escape tiny plateaux.
    # ------------------------------------------------------------------
    a = best_a.copy()
    cur_score = best_score

    # learning‑rate schedule – start relatively aggressive, then cool down
    lr = 0.12                      # initial step size
    min_lr = 5e-5
    decay = 0.9995                 # multiplicative decay per iteration
    max_active = 20                # maximum number of active indices used
    it = 0
    while time.time() < deadline:
        # full convolution (fast via FFT)
        conv = _fft_convolve(a)
        max_val = conv.max()
        # stop if no improvement possible (the bound 1/(2n‑1) is unreachable)
        # but we keep iterating until time runs out.

        # identify a (small) set of indices where the convolution is near its
        # maximum.  Using a tolerance avoids miss‑selecting only a single
        # index when many are practically tied.
        eps = max_val * 1e-12
        active = np.where(conv >= max_val - eps)[0]
        # cap the size for speed – random subset if there are many ties
        if active.size > max_active:
            active = np.random.choice(active, size=max_active, replace=False)

        # sub‑gradient for the current active set
        grad = _subgradient(a, active)

        # propose a new point – simple additive step followed by projection
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)               # keep non‑negative
        s = a_new.sum()
        if s > 0.0:
            a_new /= s                                # project onto the simplex
        else:
            # degenerate – restart from a tiny random simplex point
            a_new = np.random.random(n)
            a_new /= a_new.sum()

        new_score = 2.0 * n * np.max(_fft_convolve(a_new))

        # accept only improvements (the problem is non‑convex, so we keep
        # the best ever found but also allow the current iterate to move
        # forward even without immediate improvement)
        if new_score < cur_score:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
                print(f"[improve] iter {it:06d}  score {best_score:.9f}")
        else:
            # occasional random perturbation to escape flat regions
            if np.random.rand() < 0.003:
                i, j = np.random.randint(0, n, size=2)
                delta = min(a[i], lr * 0.5)
                a[i] -= delta
                a[j] += delta
                a = np.maximum(a, 0.0)
                a /= a.sum()
            # no improvement – simply continue with the current iterate

        # decay the learning rate, but never go below the floor
        lr = max(min_lr, lr * decay)
        it += 1

    # ------------------------------------------------------------------
    # 4) Return the best sequence discovered.
    # ------------------------------------------------------------------
    final_score = 2.0 * n * np.max(_fft_convolve(best_a))
    print(f"[final] n={n}, best score ≈ {final_score:.9f}")
    return best_a.tolist()
```

**Explanation of the approach**

* The Catalan distribution yields the classical bound π/2 ≈ 1.5708.  
* To move below this barrier we explore a broader family of shapes
  (`Beta`‑type densities) and keep the best candidate found by a cheap
  random search.  
* Starting from that candidate we perform a large number of projected
  sub‑gradient steps on the objective `2 n·max(conv)`.  The sub‑gradient
  for a particular convolution index `k` is simply the shifted version
  of the current vector (`a_{k‑i}`), and we aggregate the gradients from
  the few indices that are (almost) maximal.  
* After each step we project back onto the probability simplex
  (non‑negative entries that sum to 1).  An adaptive learning‑rate
  schedule together with occasional random jitter helps to escape tiny
  plateaux.  
* The routine respects the 1000 s wall‑clock limit, prints the progress,
  and finally returns the best sequence discovered as a plain Python
  list, ready for the surrounding harness to evaluate.
