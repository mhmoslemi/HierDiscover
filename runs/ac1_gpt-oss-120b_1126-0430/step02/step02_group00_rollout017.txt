"""
Propose a non‑negative step‑function sequence that minimizes the
evaluation function `evaluate_sequence`.  The algorithm consists of
three stages:

1. **Smart initialisation.**  A few simple parametric shapes
   (uniform, quadratic, triangular, cosine, Gaussian, exponential)
   are generated, normalised to the simplex and the one with the
   smallest exact objective is kept as a starting point.

2. **Accelerated mirror descent on a smooth surrogate.**
   The variables are stored as logits `z` and the actual sequence is
   `a = softmax(z)`.  A log‑sum‑exp surrogate of the max‑convolution
   is minimised using Adam (exponentiated‑gradient / mirror descent).
   The smoothing parameter `α` is annealed from 5 to 2000.
   The Adam update works directly on the logits, thus respecting the
   non‑negativity and sum‑to‑one constraints without an explicit
   projection.

3. **Targeted local refinement.**
   After the smooth phase we switch to the exact objective.  In each
   iteration the offset `k*` that attains the current maximum
   convolution is identified.  A mass‑transfer move removes a random
   fraction of the larger of the two contributing coefficients
   (`a[i]·a[k*-i]`) and adds it to a random remote index.  The move is
   accepted if it improves the objective, otherwise with a Metropolis
   probability (simulated‑annealing).  Every few iterations a
   sub‑gradient step for the exact maximum is performed.  The best
   sequence seen throughout the whole run is returned.

All heavy operations (convolution, correlation) are performed with
FFT using a power‑of‑two length for speed.  The routine respects the
1000 s budget and prints progress information.
"""

import time
import math
import numpy as np

# -------------------------------------------------------------
# FFT‑based full self‑convolution (length 2·n‑1)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return a*a (full convolution) using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# -------------------------------------------------------------
# Exact objective for a simplex‑normalised vector
def _objective_simplex(a: np.ndarray, n: int, L: int) -> float:
    """2·n·max(conv(a,a)) (a is assumed to sum to 1)."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# -------------------------------------------------------------
# Sub‑gradient of the maximal convolution entry
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Return the sub‑gradient of max(conv) w.r.t. a."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)          # could be several indices
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    g /= len(ks)
    return g


# -------------------------------------------------------------
# Smooth gradient (log‑sum‑exp surrogate)
def _smooth_grad_a(a: np.ndarray, conv: np.ndarray, alpha: float, L: int) -> np.ndarray:
    """
    Gradient of (2·n/α)·log∑_k e^{α·conv_k} w.r.t. a.
    The result is 2·corr(w, a) where w_k ∝ e^{α·conv_k}.
    """
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)          # stability
    w = exp_scaled / exp_scaled.sum()                # soft‑max weights

    # correlation w ⋆ a     (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)   # length ≥ 3·n‑2
    n = a.size
    grad = 2.0 * corr_rev[n - 1 : n - 1 + n]
    return grad


# -------------------------------------------------------------

            conv = conv_new
            cur_obj = obj_new
            no_improve = 0
            if cur_obj < best_score:
                best_score = cur_obj
                best_seq = a.copy()
                print(f"[{time.time():.1f}] new global best O={best_score:.6f} (iter={iter_local})")
        else:
            no_improve += 1

        # ---- occasional exact sub‑gradient step ----
        if iter_local % 12 == 0:
            g = _subgradient_max_conv(a, conv)
            eta = 0.07 * (1.0 - iter_local / 500_000)   # decreasing step
            a = a - eta * g
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_score:
                best_score = cur_obj
                best_seq = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improvement O={best_score:.6f}")

        # temperature decay (simulated annealing)
        temperature *= 0.9995
        iter_local += 1

        # if stagnant for a while, inject a mild jitter and continue
        if no_improve >= max_no_improve:
            a = best_seq.copy()
            a += np.random.randn(n) * 0.001
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max()
            no_improve = 0

    # ------------------ final sanity check --------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:          # pragma: no cover
        print("[final] evaluate_sequence failed:", e)

    return best_seq.tolist()
