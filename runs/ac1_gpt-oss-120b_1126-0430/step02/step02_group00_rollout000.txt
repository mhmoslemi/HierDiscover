"""
Hybrid Adam + Simulated‑Annealing optimizer for the step‑function problem.

The algorithm works in two stages:

1️⃣  **Smooth Adam descent** – a soft‑max (log‑sum‑exp) surrogate of the
    max‑convolution is minimised by Adam (first‑order moments).  After each
    update the iterate is projected onto the probability simplex.
    The temperature `α` of the surrogate is linearly increased from 5 to 2000,
    turning the smooth objective gradually into the true `max`.

2️⃣  **Local stochastic refinement** – a simulated‑annealing hill‑climber
    performs random “mass‑transfer’’ moves (and a gradient‑directed move)
    on the simplex, occasionally applying a sub‑gradient step on the
    current maximal convolution entry.  The temperature decays, allowing
    escape from shallow plateaus while converging to a local optimum.

Both phases respect the global time budget (default 1000 s).  The best
sequence found is returned as a plain Python list; a final call to
`evaluate_sequence` prints the exact score for verification.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex using the algorithm of
    Wang & Carreira‑Perpiñán (2013)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # sort descending
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:                        # all entries non‑positive
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Full self‑convolution via FFT (returns length 2·n‑1)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Compute a*a (full convolution) using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]

# ----------------------------------------------------------------------
# Gradient of the soft‑max (log‑sum‑exp) surrogate of max(conv)
def _smooth_gradient(a: np.ndarray, conv: np.ndarray, alpha: float, L: int) -> np.ndarray:
    """∇_a ( (2·n/α)·log Σ_k e^{α·conv_k} ).  Returns a vector of length n."""
    # soft‑max weights w_k ∝ exp(α·conv_k)
    scaled = alpha * conv
    m = np.max(scaled)
    exp_scaled = np.exp(scaled - m)               # stability
    w = exp_scaled / exp_scaled.sum()             # shape (2n‑1,)

    # correlation w ⋆ a (g_i = 2·Σ_j w_{i+j} a_j) using FFT
    a_rev = a[::-1]                               # reversed a
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)   # length ≥ 3·n‑2

    n = a.size
    # central n entries of the correlation correspond to gradient components
    grad = 2.0 * corr_rev[n - 1 : n - 1 + n]
    return grad

# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) (used in local refinement)
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Return a sub‑gradient of max(conv) w.r.t. a."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)   # indices attaining the max
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    g /= len(ks)    # average if several maximisers
    return g

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function sequence that minimises
    `evaluate_sequence`.  Returns the best sequence found as a list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # small safety margin

    # ------------------------------------------------------------------
    # Initialise a starting point
    if 'height_sequence_1' in globals():
        base = np.array(height_sequence_1, dtype=float)
        n = base.size
        a = base / base.sum()                       # simplex normalisation
        print(f"[init] using existing sequence, n={n}")
    else:
        n = 4000
        a = np.full(n, 1.0 / n)                     # uniform start
        print(f"[init] no prior sequence – using uniform, n={n}")

    # FFT length large enough for both convolution (2n‑1) and correlation (3n‑2)
    L = 1 << ((3 * n - 2).bit_length())            # next power of two

    # ------------------------------------------------------------------
    # Phase 1 – Adam descent on a smooth approximation of the max
    best_seq = a.copy()
    conv = _conv_fft(best_seq, L)
    best_score = 2.0 * n * conv.max()
    print(f"[phase‑1] start O={best_score:.6f}")

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(best_seq)
    v = np.zeros_like(best_seq)

    # surrogate temperature schedule
    alpha_start = 5.0
    alpha_end = 2000.0

    # total number of Adam iterations (subject to time)
    max_iter_phase1 = 250_000
    iter1 = 0
    phase1_deadline = start_time + budget_s * 0.60   # 60 % of budget

    while time.time() < phase1_deadline and iter1 < max_iter_phase1:
        # compute current convolution and surrogate gradient
        conv = _conv_fft(best_seq, L)
        cur_obj = 2.0 * n * conv.max()

        # log‑sum‑exp surrogate progress (optional diagnostics)
        if iter1 % 5000 == 0 and cur_obj < best_score:
            best_score = cur_obj
            best_seq = best_seq.copy()
            print(f"[{time.time():.1f}] Phase‑1 new best O={best_score:.6f} (iter={iter1})")

        # anneal α linearly
        frac = iter1 / max_iter_phase1
        alpha = alpha_start + (alpha_end - alpha_start) * frac

        # smooth gradient
        grad = _smooth_gradient(best_seq, conv, alpha, L)

        # Adam moments update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** (iter1 + 1))
        v_hat = v / (1.0 - beta2 ** (iter1 + 1))

        # learning‑rate schedule (decays to 10 % of start)
        base_lr = 0.2
        lr = base_lr * (1.0 - frac) + 0.02 * frac

        # Adam step (project back onto simplex)
        cand = best_seq - lr * m_hat / (np.sqrt(v_hat) + eps)
        cand = _project_to_simplex(cand)

        # cheap acceptance test using the exact objective
        conv_cand = _conv_fft(cand, L)
        cand_obj = 2.0 * n * conv_cand.max()
        if cand_obj < cur_obj:
            best_seq = cand
            best_score = cand_obj
        else:
            # occasional jitter to avoid stagnation
            if np.random.rand() < 0.001:
                jitter = np.random.randn(n) * 1e-3
                best_seq = _project_to_simplex(best_seq + jitter)

        iter1 += 1

    # ------------------------------------------------------------------
    # Phase 2 – stochastic local refinement (simulated annealing + sub‑grad)
    print(f"[phase‑2] entering local refinement, best O={best_score:.6f}")

    # parameters
    max_restarts = 0
    max_iter_per_restart = 120_000
    no_improve_limit = 3_000
    temperature0 = 0.04                 # initial Metropolis temperature
    subgrad_interval = 10
    restart_interval = 15_000

    while time.time() < deadline:
        # ------------------------------------------------------------------
        # initialise a fresh restart (biased towards current best)
        if np.random.rand() < 0.6:
            a_local = best_seq.copy()
            a_local += np.random.randn(n) * 2e-3
            a_local = np.clip(a_local, 0, None)
            a_local = _project_to_simplex(a_local)
        else:
            a_local = np.random.rand(n)
            a_local = _project_to_simplex(a_local)

        conv = _conv_fft(a_local, L)
        cur_obj = 2.0 * n * conv.max()
        iter_loc = 0
        no_improve = 0
        T = temperature0

        while time.time() < deadline and iter_loc < max_iter_per_restart:
            # ---- 1) Random mass‑transfer move ---------------------------------
            i, j = np.random.randint(n, size=2)
            if i != j and a_local[i] > 0:
                delta = a_local[i] * np.random.rand() * 0.12   # up to 12 % of source
                a_new = a_local.copy()
                a_new[i] -= delta
                a_new[j] += delta
                a_new = _project_to_simplex(a_new)

                conv_new = _conv_fft(a_new, L)
                new_obj = 2.0 * n * conv_new.max()

                # Metropolis acceptance
                if new_obj < cur_obj or np.random.rand() < math.exp((cur_obj - new_obj) / max(T, 1e-12)):
                    a_local, conv, cur_obj = a_new, conv_new, new_obj
                    no_improve = 0
                    if cur_obj < best_score:
                        best_score = cur_obj
                        best_seq = a_local.copy()
                        print(f"[{time.time():.1f}] New global best O={best_score:.6f} (restart={max_restarts}, iter={iter_loc})")
                else:
                    no_improve += 1
            else:
                no_improve += 1

            # ---- 2) Gradient‑directed mass move (once per 20 iterations) ---
            if iter_loc % 20 == 0:
                g = _subgradient_max_conv(a_local, conv)
                # indices: we would like to decrease a where g is large,
                # increase where g is small (most negative).
                i_max = np.argmax(g)
                i_min = np.argmin(g)
                if a_local[i_max] > 0:
                    delta = a_local[i_max] * np.random.rand() * 0.08
                    a2 = a_local.copy()
                    a2[i_max] -= delta
                    a2[i_min] += delta
                    a2 = _project_to_simplex(a2)
                    conv2 = _conv_fft(a2, L)
                    obj2 = 2.0 * n * conv2.max()
                    if obj2 < cur_obj or np.random.rand() < math.exp((cur_obj - obj2) / max(T, 1e-12)):
                        a_local, conv, cur_obj = a2, conv2, obj2
                        no_improve = 0
                        if cur_obj < best_score:
                            best_score = cur_obj
                            best_seq = a_local.copy()
                            print(f"[{time.time():.1f}] Gradient‑move best O={best_score:.6f}")

            # ---- 3) Sub‑gradient step on the current maximal entry ----------
            if iter_loc % subgrad_interval == 0:
                g = _subgradient_max_conv(a_local, conv)
                # simple decreasing step size
                eta = 0.07 * (1.0 - iter_loc / max_iter_per_restart)
                a_local = a_local - eta * g
                a_local = _project_to_simplex(a_local)
                conv = _conv_fft(a_local, L)
                cur_obj = 2.0 * n * conv.max()
                if cur_obj < best_score:
                    best_score = cur_obj
                    best_seq = a_local.copy()
                    print(f"[{time.time():.1f}] Sub‑grad new best O={best_score:.6f}")

            # ---- temperature decay -------------------------------------------
            T = temperature0 * (1.0 - iter_loc / max_iter_per_restart)

            # ---- restart condition -------------------------------------------
            if no_improve >= no_improve_limit or iter_loc % restart_interval == 0:
                break

            iter_loc += 1

        max_restarts += 1
        # safety break if out of time
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Final sanity check with the provided evaluator
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:
        print("[final] evaluate_sequence failed:", e)

    return best_seq.tolist()
