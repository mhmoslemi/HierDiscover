#!/usr/bin/env python
"""
propose_candidate
-----------------
Search for a non‑negative step‑function (list of heights) that minimises

    E(a) = 2·n·max_k (a*a)[k] / (sum_i a_i)^2

using a hybrid of three ideas:

1. **Good start** – a dense power‑law scan (a_i ∝ (i+1)^‑p) on a
   moderate length `n` (≈2000) gives a solid initial point.

2. **Projected sub‑gradient descent** – minimise the L∞‑norm of the
   autocorrelation while staying on the probability simplex
   (Σa_i = 1, a_i ≥ 0).  This quickly reduces the worst convolution
   entry.

3. **Linear‑programming “boost”** – for the current iterate `f` we
   solve the LP

        maximise   Σ g_j
        subject to g_j ≥ 0
                   (f * g)[k] ≤ M   for all k                (M = max(f*f))

   The LP finds a direction `g` that spreads the mass without raising
   the current worst convolution value `M`.  A convex combination
   `f ← (1‑t)·f + t·ĝ` (with a small step size `t`) strictly
   decreases the objective.  The LP is solved with SciPy’s HiGHS
   solver; we perform it only occasionally because the problem size
   (`≈ n²` constraints) is the main bottleneck.

The routine respects a 1000 s time budget, prints occasional progress,
keeps the best exact score found, and finally returns the best sequence
as a plain Python list of floats.
"""

import time
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the probability simplex:
        Σ_i w_i = 1,   w_i ≥ 0 .
    Implements the O(n) algorithm of
    "Efficient Projections onto the L1‑ball for Learning in High Dimensions".
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    # rho = largest i such that u_i > cssv_i / i
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:          # all entries non‑positive → uniform point
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float) -> np.ndarray:
    """
    Return a normalised power‑law vector a_i ∝ (i+1)^‑p  (Σ a_i = 1).
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = i ** (-p)
    seq /= seq.sum()
    return seq


def _convolution_max(a: np.ndarray, fft_len: int) -> float:
    """
    Fast computation of max_k (a*a)[k] using an FFT of length `fft_len`.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa).real[: 2 * a.size - 1]
    return float(conv.max())


def _subgradient_max_convolution(a: np.ndarray, max_idx: int) -> np.ndarray:
    """
    Sub‑gradient of the active convolution entry.
    For the active offset k we have
        ∂/∂a_i (a*a)[k] = 2·a_{k‑i}
    whenever the index is valid.
    """
    n = a.shape[0]
    g = np.zeros_like(a)

    lo = max(0, max_idx - (n - 1))
    hi = min(max_idx, n - 1)

    i = np.arange(lo, hi + 1, dtype=np.int64)
    g[i] = 2.0 * a[max_idx - i]
    return g


def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix A (size (2n‑1) × n) such that
        A @ g = np.convolve(f, g)   (full convolution).
    The i‑th column of A is `f` placed starting at row i.
    """
    n = f.size
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f
    return A

# ----------------------------------------------------------------------
