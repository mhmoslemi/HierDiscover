def _subgradient(seq):
    n = seq.shape[0]
    # compute conv using FFT
    m = 1 << ((2*n -1).bit_length())
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[:2*n-1]
    M = conv.max()
    # active indices within tolerance
    tol = 1e-12
    active = np.where(conv >= M - tol)[0]
    # compute subgradient g = 2 * Σ_{k∈active} seq_{k-i}
    g = np.zeros_like(seq)
    # For each active k, iterate i
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # i in [i_start, i_end]
        # for vectorized:
        i_range = np.arange(i_start, i_end+1)
        j_range = k - i_range
        g[i_range] += 2 * seq[j_range]
    return g, M

def _adam_optimize(logit, deadline):
    n = len(logit)
    best_seq = None; best_score=inf
    m = np.zeros_like(logit) ; v = np.zeros_like(logit)
    beta1 = 0.9; beta2 = 0.999; eps = 1e-8
    t = 0
    while time.time() < deadline:
        t+=1
        # compute a = softmax(logit)
        exp_logits = np.exp(logit - np.max(logit))  # stable
        a = exp_logits / exp_logits.sum()
        # compute gradient w.r.t logit
        g, _ = _subgradient(a)  # returns subgradient w.r.t a
        # compute gradient w.r.t logits: a_i * (g_i - g_mean)
        g_mean = np.dot(a, g)
        grad = a * (g - g_mean)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        # learning rate schedule 
        lr = 0.5 / np.sqrt(t)  # decays
        # update
        logit -= lr * m_hat / (np.sqrt(v_hat) + eps)
        # evaluate current a, compute metric
        seq = a
        score = _fast_evaluation(seq)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            # occasionally print progress
            if t % 200 == 0:
                print(f"[Adam] n={len(seq)} iter={t} score={best_score:.6f}")
        # random jitter occasionally:
        if np.random.rand() < 0.001:
            idx = np.random.randint(len(seq))
            logit[idx] += np.random.randn() * 0.5  # random perturbation

def _smooth_grad(seq, alpha):
    n = len(seq)
    # compute conv = seq * seq (full)
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[:2*n-1]
    # compute weights = softmax(alpha * conv)
    # To avoid overflow, subtract max:
    maxc = conv.max()
    exps = np.exp(alpha * (conv - maxc))
    w = exps / exps.sum()
    # compute g via convolution of w and reversed seq
    seq_rev = seq[::-1]
    # convolution length needed
    m2 = 1 << ((len(w) + n - 1).bit_length())
    fw = np.fft.rfft(w, m2)
    fr = np.fft.rfft(seq_rev, m2)
    conv2 = np.fft.irfft(fw * fr, m2)[: len(w) + n - 1]
    # extract slice n-1 : n-1 + n
    g = 2.0 * conv2[n-1 : n-1 + n]   # shape (n,)
    return g, conv.max()  # used for metric compute

def _adam_softmax_optimize(seq_init, time_limit):
    logit = np.log(seq_init + 1e-12)
    n = len(seq_init)
    # Adam parameters
    m = np.zeros_like(logit)
    v = np.zeros_like(logit)
    beta1=0.9; beta2=0.999; eps=1e-8
    best_seq = seq_init.copy()
    best_score = _fast_evaluation(best_seq)
    iter = 0
    start = time.time()
    while time.time() < start + time_limit:
        iter += 1
        # compute a = softmax(logit)
        max_logit = np.max(logit)
        exp_logit = np.exp(logit - max_logit)
        a = exp_logit / exp_logit.sum()
        # compute gradient via smooth max
        # schedule alpha: maybe linearly increase to 200?
        alpha = min(200.0, 5.0 + iter * 0.01)   # increments 0.01 per iter, after 20000 it saturates at 200
        g, max_conv = _smooth_grad(a, alpha)
        # compute gradient w.r.t logit via chain rule: grad = a * (g - np.dot(a,g))
        mean = np.dot(a, g)
        grad = a * (g - mean)
        # Adam updates:
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**iter)
        v_hat = v / (1 - beta2**iter)
        lr = 0.1 / np.sqrt(iter)   # maybe 0.1 initial step.
        logit -= lr * m_hat / (np.sqrt(v_hat) + eps)
        # evaluate score using _fast_evaluation
        score = _fast_evaluation(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            if iter % 100 == 0:
                print(f"[AdamSoft] iter {iter:6d} score {best_score:.6f} alpha {alpha:.2f}")
        # random jitter occasional
        if np.random.rand() < 0.001:
            idx = np.random.randint(n)
            logit[idx] += np.random.randn() * 0.5
    return best_seq, best_score

"""
Improved search for a step‑function with a low evaluation score.

Algorithm
---------
1.  Choose a moderate length (default 4000) and build a few
    parametric families of non‑negative sequences:
      – Catalan / arcsine (the known baseline).
      – Generalised “power‑arcsine’’   a_i ∝ ( (i+½)*(n‑½‑i) )^{-p},
        where p is searched in a small interval.
      – Random Dirichlet draws for extra diversity.

    The best sequence from this fast grid search is kept.

2.  Starting from the best candidate, run a **soft‑max Adam**
    optimisation on the simplex.
    * The objective is a smooth approximation of the
      max‑self‑convolution:
        f_α(a) = (1/α)·log Σ_k exp(α·(a*a)_k) .
      When α → ∞, f_α → max (a*a)_k, so the optimisation
      drives the true metric while keeping gradients well‑behaved.
    * The parameters are the logits of a soft‑max (mirror
      descent on the simplex).  Adam with an adaptive learning
      rate is used and the temperature α is increased slowly.

3.  After the Adam phase, a short **greedy mass‑transfer**
    refinement (identical to the one used in the reference
    solution) is applied to the current best sequence – this
    never increases the objective and often squeezes a little
    more improvement.

All steps respect the overall time budget (default 1000 s) and
the final sequence is returned as a plain Python list of
non‑negative floats (sum≈1, which is scale‑invariant for the
evaluation metric).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast evaluation (identical to evaluate_sequence but without the security
# checks and using an FFT based convolution).
# ----------------------------------------------------------------------
def _fast_evaluation(seq: np.ndarray) -> float:
    """Return the same metric as evaluate_sequence but much faster."""
    n = seq.size
    if n == 0:
        return np.inf
    # FFT size = next power of two >= 2*n-1
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Catalan (arcsine) seed.
# a_0 = 1, a_{i+1} = a_i * (2*i+1) / (2*i+2)
# ----------------------------------------------------------------------
def _catalan_sequence(length: int) -> np.ndarray:
    """Return the first ``length`` Catalan‑like coefficients."""
    seq = np.empty(length, dtype=float)
    a = 1.0
    seq[0] = a
    for i in range(1, length):
        a *= (2 * i - 1) / (2.0 * i)
        seq[i] = a
    return seq


# ----------------------------------------------------------------------
# Generalised power‑arcsine family:  a_i ∝ ((i+0.5)*(n-0.5-i))^{-p}
# ----------------------------------------------------------------------
def _power_arcsine_sequence(n: int, p: float) -> np.ndarray:
    """Return a symmetric sequence of length n with exponent p."""
    i = np.arange(n, dtype=float)
    x = i + 0.5
    y = n - 0.5 - i
    # product is always >0, safe to raise to -p
    a = (x * y) ** (-p)
    # normalise to sum = 1
    a_sum = a.sum()
    if a_sum == 0.0:
        a[:] = 1.0 / n
    else:
        a /= a_sum
    return a


# ----------------------------------------------------------------------
# Smooth‑max gradient (soft‑max of the self‑convolution).
# ----------------------------------------------------------------------
def _smooth_max_gradient(seq: np.ndarray, alpha: float,
                         fft_len_self: int, fft_len_mix: int) -> tuple[np.ndarray, float]:
    """
    Compute the gradient of  f_α(a)= (1/α) log Σ_k exp(α·(a*a)_k)
    w.r.t. the probability vector a (the gradient w.r.t a is returned)
    together with the current max convolution value.
    """
    n = seq.size

    # ---- self‑convolution (a*a) ------------------------------------
    fa = np.fft.rfft(seq, fft_len_self)
    conv = np.fft.irfft(fa * fa, fft_len_self)[: 2 * n - 1]

    # ---- soft‑max weights over the convolution entries -------------
    max_c = conv.max()
    # stable soft‑max
    exp_weights = np.exp(alpha * (conv - max_c))
    w = exp_weights / exp_weights.sum()               # shape (2n‑1,)

    # ---- gradient = 2 * (w  *  a)  (convolution with reversed a) --
    seq_rev = seq[::-1]
    fw = np.fft.rfft(w, fft_len_mix)
    fr = np.fft.rfft(seq_rev, fft_len_mix)
    conv_mix = np.fft.irfft(fw * fr, fft_len_mix)[: (2 * n - 1) + n - 1]
    # slice n‑1 … n‑1+n‑1 => n entries
    g = 2.0 * conv_mix[n - 1: n - 1 + n]

    return g, max_c


# ----------------------------------------------------------------------
# Adam optimisation on the simplex using the smooth‑max surrogate.
# ----------------------------------------------------------------------
def _adam_softmax_optimize(seq_init: np.ndarray,
                           time_limit: float) -> tuple[np.ndarray, float]:
    """
    Adam on the logits of a soft‑max representation.
    Returns the best sequence (sum≈1) discovered within time_limit.
    """
    n = seq_init.size
    # pre‑compute FFT lengths (they do not change)
    fft_len_self = 1 << ((2 * n - 1).bit_length())
    fft_len_mix = 1 << (((2 * n - 1) + n - 1).bit_length())

    # initialise logits (mirror‑descent on the simplex)
    # small epsilon to avoid log(0)
    eps = 1e-12
    logits = np.log(np.maximum(seq_init, eps))

    # Adam state
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    best_seq = np.exp(logits - np.max(logits))
    best_seq /= best_seq.sum()
    best_score = _fast_evaluation(best_seq)

    start = time.time()
    it = 0
    while time.time() - start < time_limit:
        it += 1

        # ---- current probability vector (soft‑max) ------------------
        max_logit = np.max(logits)
        exp_logit = np.exp(logits - max_logit)
        a = exp_logit / exp_logit.sum()               # shape (n,)

        # ---- smooth‑max gradient ------------------------------------
        # alpha grows slowly from ~5 to ~200 (empirically good)
        alpha = min(200.0, 5.0 + it * 0.02)

        g_a, _ = _smooth_max_gradient(a, alpha,
                                      fft_len_self, fft_len_mix)

        # ---- chain rule to obtain gradient w.r.t logits -------------
        avg = np.dot(a, g_a)                           # scalar
        grad = a * (g_a - avg)                         # shape (n,)

        # ---- Adam update -------------------------------------------
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        # step‑size schedule (larger early, decays as 1/√t)
        lr = 0.5 / np.sqrt(it)

        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- evaluate ------------------------------------------------
        score = _fast_evaluation(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            if it % 250 == 0:
                print(f"[Adam] iter {it:6d}  score {best_score:.6f}  alpha {alpha:.1f}")

        # ---- occasional random jitter to escape tiny plateaus -----
        if np.random.rand() < 0.001:
            idx = np.random.randint(n)
            logits[idx] += np.random.randn() * 0.3

    return best_seq, best_score


# ----------------------------------------------------------------------
# Simple greedy refinement – move a little mass from the biggest
# contributing index to an index that contributes almost nothing.
# ----------------------------------------------------------------------
def _refine(seq: np.ndarray, deadline: float) -> np.ndarray:
    """Greedy local improvement; stops when the deadline is reached."""
    n = seq.size
    best_seq = seq.copy()
    best_score = _fast_evaluation(best_seq)

    # pre‑compute FFT length (does not change)
    fft_len = 1 << ((2 * n - 1).bit_length())

    it = 0
    while time.time() < deadline:
        # convolution and its maximum position
        fa = np.fft.rfft(seq, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        k = int(conv.argmax())

        # contribution weights for each position to that maximal entry
        i_arr = np.arange(n)
        j_arr = k - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        w = np.zeros_like(seq)
        w[mask] = seq[j_arr[mask]]

        # source = index with largest contribution product,
        # sink = index with virtually zero contribution
        contrib = seq * w
        src = int(np.argmax(contrib))
        sink = int(np.argmin(w))

        if seq[src] < 1e-12:
            break

        # move a modest fraction (5 %) of source mass
        eps = 0.05 * seq[src]
        new_seq = seq.copy()
        new_seq[src] -= eps
        new_seq[sink] += eps

        score = _fast_evaluation(new_seq)

        if score < best_score - 1e-12:
            seq = new_seq
            best_seq = new_seq.copy()
            best_score = score
            if it % 500 == 0:
                print(f"[greedy] iter {it:6d}  score {best_score:.6f}")
        else:
            # occasional tiny random jitter
            if np.random.rand() < 0.01:
                jitter_idx = np.random.randint(n)
                jitter = np.random.randn() * 0.001
                new_seq = seq.copy()
                new_seq[jitter_idx] = max(0.0, new_seq[jitter_idx] + jitter)
                seq = new_seq
        it += 1
    return best_seq


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights aiming at a low evaluation score.

    The routine works within ``budget_s`` seconds (default 1000).  It
    builds a few parametric candidates (Catalan + a power‑arcsine family),
    keeps the best, refines it with a soft‑max Adam optimiser and finally
    performs a short greedy mass‑transfer sweep.
    """
    np.random.seed(seed)
    start = time.time()
    # keep a small safety margin
    overall_deadline = start + budget_s - 5

    # ------------------------------------------------------------------
    # 1.  Choose a length – 4000 offers a good trade‑off.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 2.  Baseline Catalan (arcsine) sequence.
    # ------------------------------------------------------------------
    catalan_seq = _catalan_sequence(n).astype(float)
    catalan_seq /= catalan_seq.sum()
    best_seq = catalan_seq.copy()
    best_score = _fast_evaluation(best_seq)
    print(f"[init] Catalan length={n}  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Scan a simple one‑parameter family (power‑arcsine).
    # ------------------------------------------------------------------
    # Allocate roughly 15 % of the remaining budget for the grid scan.
    scan_deadline = time.time() + 0.15 * (budget_s - 5)
    p_vals = np.linspace(0.4, 2.0, 33)          # 33 candidates → cheap
    for p in p_vals:
        if time.time() >= scan_deadline:
            break
        seq = _power_arcsine_sequence(n, p)
        sc = _fast_evaluation(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[grid] p={p:.3f}  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  A few random Dirichlet starts (adds diversity).
    # ------------------------------------------------------------------
    # Use another ~10 % of the time for a couple of random starts.
    rand_deadline = time.time() + 0.10 * (budget_s - 5)
    while time.time() < rand_deadline:
        # concentration 0.5 → fairly sparse vectors
        random_seq = np.random.dirichlet(np.full(n, 0.5))
        sc = _fast_evaluation(random_seq)
        if sc < best_score:
            best_score = sc
            best_seq = random_seq.copy()
            print(f"[rand] new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Adam‑softmax refinement (majority of the remaining budget).
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 5.0:               # keep a few seconds for final greedy
        seq_adam, sc_adam = _adam_softmax_optimize(best_seq, remaining * 0.85)
        if sc_adam < best_score:
            best_score = sc_adam
            best_seq = seq_adam.copy()
            print(f"[adam] post‑optim score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Final short greedy refinement (uses whatever time is left).
    # ------------------------------------------------------------------
    final_deadline = overall_deadline
    if time.time() < final_deadline:
        best_seq = _refine(best_seq, final_deadline)
        best_score = _fast_evaluation(best_seq)
        print(f"[final greedy] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 7.  Return the best sequence found.
    # ------------------------------------------------------------------
    print(f"[final] best score={best_score:.6f}  length={n}")
    # ensure sum>0.01 (already true) and convert to list
    return best_seq.tolist()
