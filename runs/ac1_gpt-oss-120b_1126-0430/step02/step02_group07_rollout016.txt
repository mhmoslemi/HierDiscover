def fast_convolve(a):
    n = a.shape[0]
    size = 1 << ((2*n-1).bit_length())
    # compute FFT of size size
    a_fft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(a_fft * a_fft, n=size)
    return conv[:2*n-1]

    n = len(current)

    best_score = score_current
    # optional: define max number of LP rounds

        if time.time() > remaining_time - 5:
            break

        M = conv.max()
        # Build A matrix
        rows = 2 * n - 1

def _gradient_refine(seq, score, time_limit):
    cur = seq.copy()
    cur_score = score
    n = len(cur)
    iter = 0
    max_iters = 200000
    while time.time() < time_limit - 2 and iter < max_iters:
        iter += 1
        conv = fast_convolve(cur)
        max_idx = int(conv.argmax())
        max_val = conv[max_idx]

        # Compute contributions vector for this index:
        lo = max(0, max_idx - (n-1))
        hi = min(max_idx, n-1)
        i_range = np.arange(lo, hi+1)
        contrib = cur[i_range] * cur[max_idx - i_range]
        src = lo + np.argmax(contrib)  # absolute index
        # target minimal contribution (prefer zero)
        tgt = lo + np.argmin(contrib)

        if src == tgt or cur[src] <= 0:
            # fallback random pick
            src, tgt = rng.choice(n, size=2, replace=False)

        # step size fraction of source mass, decreasing over time
        # Use schedule: frac = max(0.02, 0.2 * (1 - iter / max_iters))
        frac = 0.15 * max(0.0, 1.0 - iter / max_iters) + 0.02
        step = cur[src] * rng.random() * frac
        if step <= 0:
            continue
        new = cur.copy()
        new[src] -= step
        new[tgt] += step
        # project
        new = _project_to_simplex(new)

        new_score = _compute_score(new)
        if new_score < cur_score - 1e-12:
            cur = new
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_seq = new.tolist()
                print(f"[grad] iter={iter} new best={new_score:.6f}")

        # occasionally random shake if stagnant
        if iter % 2000 == 0:
            # random perturb small
            idxs = rng.choice(n, size=2, replace=False)
            delta = cur[idxs[0]] * 0.05
            cur[idxs[0]] -= delta
            cur[idxs[1]] += delta
            cur = _project_to_simplex(cur)
            cur_score = _compute_score(cur)
    return best_seq, best_score

def _gradient_refine(seq, score, deadline, rng):
    n = len(seq)
    cur = seq.copy()
    best_seq = cur.copy()
    best_score = score
    max_iters = 150000
    it = 0
    while time.time() < deadline - 2 and it < max_iters:
        it += 1
        conv = _fast_convolve(cur)
        max_idx = int(conv.argmax())
        # contributions for this max_idx
        lo = max(0, max_idx - (n-1))
        hi = min(max_idx, n-1)
        i_range = np.arange(lo, hi+1)
        contrib = cur[i_range] * cur[max_idx - i_range]
        src = lo + int(np.argmax(contrib))
        tgt = lo + int(np.argmin(contrib))
        if src == tgt or cur[src] <= 0:
            src, tgt = rng.choice(n, size=2, replace=False)
        # step size schedule
        frac = 0.15 * max(0.0, 1.0 - it / max_iters) + 0.02
        step = cur[src] * rng.random() * frac
        if step <= 0:
            continue
        new = cur.copy()
        new[src] -= step
        new[tgt] += step
        new = _project_to_simplex(new)
        new_score = _compute_score(new)
        if new_score + 1e-12 < cur_score:
            cur = new
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_seq = new.copy()
                print(f"[grad] iter={it} new best={new_score:.6f}")
        # occasional random rebalancing
        if it % 2000 == 0:
            # add small random shake
            idxs = rng.choice(n, size=2, replace=False)
            delta = cur[idxs[0]] * 0.05
            cur[idxs[0]] -= delta
            cur[idxs[1]] += delta
            cur = _project_to_simplex(cur)
            cur_score = _compute_score(cur)
    return best_seq, best_score

"""
Improved search for a non‑negative step function that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of three phases:

1. **Broad sampling** – generate many candidates from a richer set of
   parametric families (power‑law, symmetric power‑law, beta‑like,
   arcsine‑like, exponentials and mixtures).  Each raw vector is
   normalised to a probability distribution and evaluated with a fast
   FFT‑based convolution.  The best candidate is kept.

2. **Linear‑programming refinement** – for the current best distribution
   *f* compute *M = max(conv(f,f))* and solve the LP

        maximise   Σ b_i
        subject to  conv(f, b) ≤ M   (element‑wise)
                    b_i ≥ 0

   The solution *b* has larger total mass while still respecting the
   convolution bound.  After scaling *b* back to a probability vector
   we mix it with the current vector (50 % each) and repeat a few
   iterations.  This often yields a noticeable drop in the objective.

3. **Gradient‑descent hill‑climbing** – starting from the LP‑improved
   vector we repeatedly move a small amount of mass from the index that
   contributes most to the current maximal convolution entry to the index
   that contributes least.  The step size shrinks with the iteration
   count.  The convolution is recomputed with an FFT each step; the
   vector is projected back onto the simplex to keep it feasible.

All helpers are defined at module level (no closures).  The function
`propose_candidate` returns the best sequence found before the time
budget expires.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities (top level, no closures)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _fast_convolve(a: np.ndarray) -> np.ndarray:
    """Self‑convolution using a power‑of‑two FFT (real input)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def _compute_score(arr: np.ndarray) -> float:
    """Fast evaluation of the objective for a non‑negative vector."""
    total = arr.sum()
    if total <= 0.0:
        return float("inf")
    conv = _fast_convolve(arr)
    max_val = conv.max()
    n = arr.shape[0]
    return 2.0 * n * max_val / (total * total)


def _candidate_improve(
    raw: np.ndarray, label: str, best_seq: np.ndarray, best_score: float
) -> tuple[np.ndarray, float]:
    """Normalise a raw candidate, compute its cheap score and keep it if it improves."""
    total = raw.sum()
    if total <= 0.0:
        return best_seq, best_score
    norm = raw / total
    score = _compute_score(norm)
    if score < best_score - 1e-12:
        best_seq = norm.copy()
        best_score = score
        print(f"[{label}] improved to {score:.6f}")
    return best_seq, best_score


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the matrix A (size (2n‑1)×n) such that A @ b = conv(f, b).
    Row k, column j contains f_{k‑j} when the index is in range, otherwise 0.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        A[j : j + n, j] = f
    return A


def _lp_refine(
    seq: np.ndarray, cur_score: float, deadline: float, rng: np.random.Generator
) -> tuple[np.ndarray, float]:
    """
    Linear‑programming refinement described in the comment header.
    Returns an improved sequence (still normalised) and its score.
    """
    n = seq.shape[0]
    best_seq = seq.copy()
    best_score = cur_score
    cur = seq.copy()

    max_lp_iters = 8
    for it in range(max_lp_iters):
        if time.time() > deadline - 5.0:
            break

        conv = _fast_convolve(cur)
        M = conv.max()

        # Build the constraint matrix A such that A @ b = conv(cur, b)
        A = _build_conv_matrix(cur)

        # Linear program: maximise Σ b_i  <=> minimise -Σ b_i
        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        bounds = [(0.0, None)] * n

        res = linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            print(f"[LP] iteration {it} failed: {res.message}")
            break

        b = res.x
        s = b.sum()
        if s <= 1.000001:  # no increase in total mass → nothing to gain
            print("[LP] sum(b) ≤ 1, terminating LP stage.")
            break

        # Scale back to a probability vector and blend 50 % with the current one
        scale = 1.0 / s
        g = b * scale
        t = 0.5  # mixing weight
        new = (1.0 - t) * cur + t * g
        new = _project_to_simplex(new)

        new_score = _compute_score(new)
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_seq = new.copy()
            print(f"[LP] improved to {new_score:.6f}")

        cur = new

    return best_seq, best_score


def _gradient_refine(
    seq: np.ndarray, cur_score: float, deadline: float, rng: np.random.Generator
) -> tuple[np.ndarray, float]:
    """
    Targeted hill‑climbing: move a fraction of the mass from the index that
    contributes most to the current maximal convolution value towards the
    index that contributes the least.
    """
    n = seq.shape[0]
    cur = seq.copy()
    best_seq = cur.copy()
    best_score = cur_score
    cur_score = cur_score

    max_iters = 150_000
    it = 0
    while time.time() < deadline - 2.0 and it < max_iters:
        it += 1

        conv = _fast_convolve(cur)
        max_idx = int(conv.argmax())

        # indices that actually appear in the current max convolution term
        lo = max(0, max_idx - (n - 1))
        hi = min(max_idx, n - 1)
        idx_range = np.arange(lo, hi + 1)

        # contribution = a_i * a_{max_idx-i}
        contrib = cur[idx_range] * cur[max_idx - idx_range]
        src = lo + int(np.argmax(contrib))
        tgt = lo + int(np.argmin(contrib))

        if src == tgt or cur[src] <= 0.0:
            src, tgt = rng.choice(n, size=2, replace=False)

        # step size slowly shrinks from ~10 % to ~2 %
        frac = 0.15 * max(0.0, 1.0 - it / max_iters) + 0.02
        step = cur[src] * rng.random() * frac
        if step <= 0.0:
            continue

        new = cur.copy()
        new[src] -= step
        new[tgt] += step
        new = _project_to_simplex(new)

        new_score = _compute_score(new)

        if new_score + 1e-12 < cur_score:
            cur = new
            cur_score = new_score

            if new_score < best_score - 1e-12:
                best_score = new_score
                best_seq = new.copy()
                print(f"[grad] iter={it} new best={new_score:.6f}")

        # occasional random shake to escape plateaus
        if it % 2000 == 0:
            i1, i2 = rng.choice(n, size=2, replace=False)
            delta = cur[i1] * 0.05
            cur[i1] -= delta
            cur[i2] += delta
            cur = _project_to_simplex(cur)
            cur_score = _compute_score(cur)

    return best_seq, best_score

def propose_candidate(seed: int = 123, budget_s: float = 950, **kwargs) -> list[float]:
    """
    Search for a step‑function (list of heights) that minimises the provided
    `evaluate_sequence` value.  The routine respects a time budget and
    returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s

    # ------------------------------------------------------------------
    # Determine problem size (fallback if the harness does not provide it)
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)  # provided by the evaluation harness
    except Exception:
        n = 2244

    # ------------------------------------------------------------------
    # Baseline – start from the supplied construction or uniform
    # ------------------------------------------------------------------
    try:
        best_seq = np.array(height_sequence_1, dtype=np.float64)
    except Exception:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = _project_to_simplex(best_seq)  # ensure exact sum = 1
    best_score = _compute_score(best_seq)
    print(f"[init] length = {n}, baseline score = {best_score:.6f}")

    idx = np.arange(n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 1. Monotone power‑law grid search
    # ------------------------------------------------------------------
    alphas = np.linspace(0.30, 1.00, 71)
    shifts = np.linspace(0.01, 3.00, 30)
    for alpha in alphas:
        if time.time() > deadline - 10.0:
            break
        for shift in shifts:
            raw = np.power(idx + shift, -alpha)
            best_seq, best_score = _candidate_improve(raw, "gridPL", best_seq, best_score)

    # ------------------------------------------------------------------
    # 2. Symmetric power‑law families
    # ------------------------------------------------------------------
    for _ in range(2000):
        if time.time() > deadline - 10.0:
            break
        alpha = rng.uniform(0.20, 1.20)
        shift = rng.uniform(0.001, 5.0)
        raw = np.power(idx + shift, -alpha) + np.power((n - 1 - idx) + shift, -alpha)
        best_seq, best_score = _candidate_improve(raw, "symPL", best_seq, best_score)

    # ------------------------------------------------------------------
    # 3. Beta‑like distributions (a_i ∝ i^{p‑1}·(n‑1‑i)^{q‑1})
    # ------------------------------------------------------------------
    for _ in range(1500):
        if time.time() > deadline - 10.0:
            break
        p = rng.uniform(0.1, 2.0)
        q = rng.uniform(0.1, 2.0)
        eps = 1e-6
        raw = np.power(idx + eps, p - 1.0) * np.power((n - 1 - idx) + eps, q - 1.0)
        best_seq, best_score = _candidate_improve(raw, "beta", best_seq, best_score)

    # ------------------------------------------------------------------
    # 4. Arcsine‑like (U‑shaped) family
    # ------------------------------------------------------------------
    for _ in range(1000):
        if time.time() > deadline - 10.0:
            break
        eps = 1e-6
        raw = 1.0 / np.sqrt((idx + eps) * ((n - 1 - idx) + eps))
        best_seq, best_score = _candidate_improve(raw, "arc", best_seq, best_score)

    # ------------------------------------------------------------------
    # 5. Monotone exponential families
    # ------------------------------------------------------------------
    for _ in range(1000):
        if time.time() > deadline - 10.0:
            break
        beta = rng.uniform(1e-5, 0.1)
        raw = np.exp(-beta * idx)
        best_seq, best_score = _candidate_improve(raw, "exp", best_seq, best_score)

    # ------------------------------------------------------------------
    # 6. Symmetric exponential families
    # ------------------------------------------------------------------
    for _ in range(500):
        if time.time() > deadline - 10.0:
            break
        beta = rng.uniform(1e-5, 0.1)
        raw = np.exp(-beta * idx) + np.exp(-beta * (n - 1 - idx))
        best_seq, best_score = _candidate_improve(raw, "symExp", best_seq, best_score)

    # ------------------------------------------------------------------
    # 7. Random mixtures of two power‑law components (symmetrised)
    # ------------------------------------------------------------------
    for _ in range(2000):
        if time.time() > deadline - 10.0:
            break
        alpha1 = rng.uniform(0.20, 1.20)
        shift1 = rng.uniform(0.001, 5.0)
        alpha2 = rng.uniform(0.20, 1.20)
        shift2 = rng.uniform(0.001, 5.0)
        p = rng.random()
        mixed = p * np.power(idx + shift1, -alpha1) + (1.0 - p) * np.power(idx + shift2, -alpha2)
        raw = mixed + mixed[::-1]  # enforce symmetry
        best_seq, best_score = _candidate_improve(raw, "mixSym", best_seq, best_score)

    print(f"[search] best score after sampling: {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Linear‑programming refinement (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline - 10.0:
        best_seq, best_score = _lp_refine(best_seq, best_score, deadline, rng)

    # ------------------------------------------------------------------
    # 3. Gradient / hill‑climbing refinement (final polishing)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        best_seq, best_score = _gradient_refine(best_seq, best_score, deadline, rng)

    print(f"[final] best score found: {best_score:.6f}")
    return best_seq.tolist()
