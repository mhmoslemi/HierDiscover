def _fast_conv(a):
    # compute convolution via FFT
    n = a.shape[0]
    # determine next power of two for length 2n-1
    conv_len = 2*n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    f = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(f*f, n=fft_len)[:conv_len]
    return conv_full

def construct_sidon_set(k, rng):
    S = []
    sums = set()
    attempts = 0
    max_attempts = 10000
    while len(S) < k and attempts < max_attempts:
        cand = rng.integers(0, n)
        if cand in S:
            attempts += 1
            continue
        # compute new sums
        coll = False
        # pairs with existing
        for s in S:
            ssum = s + cand
            if ssum in sums: 
                coll = True
                break
        # also check diagonal sum: cand + cand
        if not coll and (cand + cand) in sums:
            coll = True
        if coll:
            attempts += 1
            continue
        # add
        S.append(cand)
        # update sums
        sums.add(cand + cand)
        for s in S[:-1]:  # previous items
            sums.add(s + cand)
        attempts = 0  # reset attempts after successful addition
    return S

        if score < best_score:

            best_seq = norm.copy()

    # Phase 3: local refinement hill climbing
    cur = best_seq.copy()

    max_iters = 2000000  # maybe limit by time
    # track iteration count
    it = 0

        # find top convolution indices
        # if we want top_m unique largest values
        idx_top = np.argpartition(cur_conv, -top_m)[-top_m:]

        contrib = np.zeros(n)
        i_arr = np.arange(n)

            mask = (rev >= 0) & (rev < n)
            contrib[mask] += cur[i_arr[mask]] * cur[rev[mask]]

        src = int(np.argmax(contrib))
        tgt = int(np.argmin(contrib))
        if src == tgt or cur[src] <= 0:

        step = cur[src] * rng.random() * frac

        new = cur.copy()
        new[src] -= step

            continue

    # final verify

"""
Propose a low‑value step‑function for the evaluation routine.

The algorithm consists of three stages:

1. **Broad random sampling** – many candidate vectors are generated from
   a variety of parametric families (power‑law, beta, Gaussian,
   Laplace, sinusoid, and sparse random support).  Each candidate is
   normalised to a probability vector (sum = 1) and evaluated with a
   fast FFT‑based convolution.  When a candidate beats the current best
   (checked with the official ``evaluate_sequence``), it becomes the new
   incumbent.

2. **Recombination** – the incumbent is mixed with additional random
   candidates (convex combination) to explore the neighbourhood of good
   solutions.

3. **Targeted hill‑climbing / simulated annealing** – starting from the
   best vector, the algorithm repeatedly moves a small amount of mass
   from indices that contribute most to the largest convolution entries
   to indices that contribute least.  The step size decays over time,
   and occasional uphill moves are accepted with a tiny probability to
   escape plateaus.  The convolution is computed with an FFT (O(n log n))
   for speed, allowing many thousands of iterations within the time
   budget.

All numbers are kept non‑negative and the sum is forced to 1 (simplex
projection) to stay in the feasible region.  The final best sequence is
returned as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _project_to_simplex(v):
    """Project a vector onto the simplex {x ≥ 0, sum(x)=1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


def _fast_conv(a, fft_len):
    """Convolution of a with itself using an FFT of length ``fft_len``."""
    f = np.fft.rfft(a, fft_len)
    conv_full = np.fft.irfft(f * f, fft_len)[: 2 * a.shape[0] - 1]
    # tiny negative values may appear because of rounding; clip them.
    return np.maximum(conv_full, 0.0)


def _fast_score(a, fft_len):
    """Fast evaluation of the objective for a (already normalised) array."""
    total = a.sum()
    if total <= 0:
        return float("inf")
    conv = _fast_conv(a, fft_len)
    max_val = conv.max()
    n = a.shape[0]
    return 2.0 * n * max_val / (total * total)


# ----------------------------------------------------------------------
# Parametric families (return a *non‑normalised* vector)
# ----------------------------------------------------------------------


def _family_powerlaw(n, rng):
    alpha = rng.uniform(0.1, 2.0)
    shift = rng.uniform(0.01, 5.0)
    idx = np.arange(n, dtype=float)
    return np.power(idx + shift, -alpha)


def _family_beta(n, rng):
    a = rng.uniform(0.1, 5.0)
    b = rng.uniform(0.1, 5.0)
    i = np.arange(n, dtype=float)
    x = (i + 0.5) / n  # stay away from 0 and 1
    return (x ** (a - 1)) * ((1.0 - x) ** (b - 1))


def _family_gaussian(n, rng):
    mu = rng.uniform(0, n - 1)
    sigma = rng.uniform(0.05 * n, 0.5 * n)
    i = np.arange(n, dtype=float)
    return np.exp(-0.5 * ((i - mu) / sigma) ** 2)


def _family_laplace(n, rng):
    mu = rng.uniform(0, n - 1)
    scale = rng.uniform(0.02 * n, 0.5 * n)
    i = np.arange(n, dtype=float)
    return np.exp(-np.abs(i - mu) / scale)


def _family_sine(n, rng):
    p = rng.uniform(0.5, 5.0)
    i = np.arange(n, dtype=float)
    s = np.sin(np.pi * (i + 1) / (n + 1))  # positive on (0,π)
    return s ** p


def _family_random_support(n, rng):
    """Uniform mass on a random subset of size k (20 ≤ k ≤ 200)."""
    k = int(rng.integers(20, min(n, 200)))
    idx = rng.choice(n, size=k, replace=False)
    raw = np.zeros(n, dtype=float)
    raw[idx] = 1.0 / k
    return raw


def _family_mixture(n, rng):
    """Convex mixture of two independently drawn families."""
    f1 = rng.choice(_ALL_FAMILIES)
    f2 = rng.choice(_ALL_FAMILIES)
    raw1 = f1(n, rng)
    raw2 = f2(n, rng)
    w = rng.random()
    raw = w * raw1 + (1.0 - w) * raw2
    return raw


# List of generator functions (used for random sampling)
_ALL_FAMILIES = [
    _family_powerlaw,
    _family_beta,
    _family_gaussian,
    _family_laplace,
    _family_sine,
    _family_random_support,
]


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed=123, budget_s=950, **kwargs):
    """
    Produce a list of non‑negative heights (length = the length of the
    supplied ``height_sequence_1`` if present) that aims to minimise the
    provided ``evaluate_sequence`` objective.  The routine runs for at
    most ``budget_s`` seconds and returns the best vector discovered.
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # Determine the length of the problem.
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)  # provided by the evaluation harness
        base_seq = list(height_sequence_1)
    except Exception:
        n = 2244
        base_seq = np.full(n, 1.0 / n).tolist()

    # ------------------------------------------------------------------
    # Baseline (always available)
    # ------------------------------------------------------------------
    try:
        best_seq = np.array(base_seq, dtype=float)
        best_seq = best_seq / best_seq.sum()  # normalise just in case
        best_score = evaluate_sequence(best_seq.tolist())
    except Exception:
        best_seq = np.full(n, 1.0 / n, dtype=float)
        best_score = evaluate_sequence(best_seq.tolist())

    print(f"[init] n = {n}, baseline score = {best_score:.6f}")

    # Pre‑compute FFT length for the fast convolution routine.
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    # ------------------------------------------------------------------
    # Helper to normalise a raw vector and (optionally) verify with the
    # official evaluator.
    # ------------------------------------------------------------------
    def _try_candidate(raw, label, verify=True):
        nonlocal best_seq, best_score
        total = raw.sum()
        if total <= 0:
            return
        norm = raw / total
        # fast evaluation first
        fast = _fast_score(norm, fft_len)
        if fast + 1e-12 < best_score:
            # verify with the official (slow) routine
            real = evaluate_sequence(norm.tolist())
            if real + 1e-12 < best_score:
                best_score = real
                best_seq = norm.copy()
                print(f"[{label}] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1. Broad random sampling from many families
    # ------------------------------------------------------------------
    max_samples = 20000
    for i in range(max_samples):
        if time.time() - start > budget_s - 30:
            break
        fam = rng.choice(_ALL_FAMILIES)
        raw = fam(n, rng)
        _try_candidate(raw, "sample")

    # ------------------------------------------------------------------
    # 2. Recombination of the current best with fresh random families
    # ------------------------------------------------------------------
    for i in range(5000):
        if time.time() - start > budget_s - 30:
            break
        # generate an independent random candidate
        fam = rng.choice(_ALL_FAMILIES)
        raw2 = fam(n, rng)
        # convex blend
        t = rng.random()
        cand = (1.0 - t) * best_seq + t * (raw2 / raw2.sum())
        cand = cand / cand.sum()
        _try_candidate(cand, "mix")

    # ------------------------------------------------------------------
    # 3. Targeted hill‑climbing / simulated annealing refinement
    # ------------------------------------------------------------------
    cur = best_seq.copy()
    cur_score = best_score
    cur_conv = _fast_conv(cur, fft_len)

    top_m = 5                # number of largest convolution entries to consider
    max_iters = 2_000_000    # safety cap; real stop is time budget
    it = 0

    while time.time() - start < budget_s - 5 and it < max_iters:
        it += 1

        # --- indices of the top‑m convolution values
        top_idx = np.argpartition(cur_conv, -top_m)[-top_m:]

        # --- compute per‑index contribution to those peaks
        contrib = np.zeros(n, dtype=float)
        i_arr = np.arange(n, dtype=int)
        for k in top_idx:
            rev = k - i_arr
            mask = (rev >= 0) & (rev < n)
            contrib[mask] += cur[i_arr[mask]] * cur[rev[mask]]

        src = int(np.argmax(contrib))
        tgt = int(np.argmin(contrib))
        if src == tgt or cur[src] <= 0.0:
            src, tgt = rng.choice(n, size=2, replace=False)

        # step size that shrinks linearly with elapsed time
        elapsed = time.time() - start
        frac = 0.10 * (1.0 - elapsed / budget_s) + 0.02   # between 0.02 and 0.10
        step = cur[src] * rng.random() * frac
        if step <= 1e-15:
            continue

        new = cur.copy()
        new[src] -= step
        new[tgt] += step
        if new[src] < 0.0:          # numerical safety
            continue

        # raw fast evaluation
        new_conv = _fast_conv(new, fft_len)
        new_score = 2.0 * n * new_conv.max() / (new.sum() * new.sum())

        improved = False
        if new_score + 1e-12 < cur_score:
            improved = True
        elif rng.random() < 0.001:          # occasional uphill move
            improved = True

        if improved:
            # keep the move
            cur = new
            cur_conv = new_conv
            cur_score = new_score

            # update the global best if we beat it
            if new_score + 1e-12 < best_score:
                # verify with official evaluator (rare but cheap)
                real = evaluate_sequence(new.tolist())
                if real + 1e-12 < best_score:
                    best_score = real
                    best_seq = new.copy()
                    print(f"[refine] iter={it} new best={best_score:.6f}")

        # occasional re‑projection to avoid drift from the simplex
        if it % 5000 == 0:
            cur = _project_to_simplex(cur)
            cur_conv = _fast_conv(cur, fft_len)

    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[final] best score = {final_score:.6f}")
    return best_seq.tolist()
