def _corr_gradient(w, a, L):
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)

"""
Propose a non‑negative step‑function sequence that minimizes the evaluation
function  O = 2·n·max(conv(a,a)) / (∑a)² .

The algorithm consists of three stages:

1.  **Variance reduction** – minimise the L² variance of the convolution
    (i.e. make it as flat as possible) using an Adam optimiser.
    The gradient of ϕ(a)=∑(conv_k−μ)²  is  g = 2·( (conv−μ) ⋆ a ),
    where “⋆’’ denotes the (non‑circular) correlation; it is computed with a
    single FFT.

2.  **Soft‑max (log‑sum‑exp) descent** – a smooth surrogate of the max‑norm
    is optimised (α·convₖ → soft‑max).  A projected Adam step with an
    annealed temperature α (5 → 2000) pushes the true maximum down.

3.  **Local refinement** – random “mass‑transfer’’ moves together with an
    exact sub‑gradient step for the current maximal convolution entry
    (the classical direction used in the reference code).  Occasionally an
    uphill move is accepted (Metropolis rule) to escape shallow plateaus.

The search is run repeatedly from several diverse initialisations
(uniform, exponential, harmonic, and random) and the best sequence found
within the allotted 1000 s budget is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simple helpers --------------------------------------------------------
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x>=0, sum(x)=1}."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:                     # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full convolution a*a using FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    # the non‑zero part lives in the first 2·len(a)-1 entries
    return conv_full[: 2 * a.size - 1]


def _objective_simplex(a: np.ndarray, n: int, L: int) -> float:
    """Exact objective 2·n·max(conv) for a normalized to sum(a)=1."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. a (used in phase 3)."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    g /= len(ks)
    return g


def _correlation_grad(w: np.ndarray, a: np.ndarray, L: int) -> np.ndarray:
    """
    Gradient 2·(w⋆a) where w has length 2·n‑1 and a length n.
    Returned vector has length n.
    """
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n = a.size
    grad = 2.0 * corr_rev[n - 1 : n - 1 + n]
    return grad


def _smooth_gradient(a: np.ndarray, conv: np.ndarray, alpha: float, L: int) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate:
        (2·n/α)·log(∑ₖ e^{α·convₖ})
    The formula reduces to 2·corr(w,a) with wₖ ∝ e^{α·convₖ}.
    """
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)   # stability
    w = exp_scaled / exp_scaled.sum()
    return _correlation_grad(w, a, L)


# ----------------------------------------------------------------------
# Initialisation helpers -------------------------------------------------
# ----------------------------------------------------------------------


def _init_exponential(n: int, c: float = 2.0) -> np.ndarray:
    """
    Exponential (geometric) shape: a_i ∝ r^i with
    r = 1 - c/n  (0<r<1).  Normalised to sum 1.
    """
    r = 1.0 - c / n
    if r <= 0.0:
        r = 0.0
    i = np.arange(n, dtype=float)
    a = r ** i
    a = a / a.sum()
    return a


def _init_harmonic(n: int, p: float = 1.0) -> np.ndarray:
    """
    Edge‑heavy harmonic shape:
        a_i ∝ (i+1)^{-p} + (n-i)^{-p}
    Symmetric and normalised.
    """
    i = np.arange(n, dtype=float)
    a = (i + 1) ** (-p) + (n - i) ** (-p)
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Optimisation phases ----------------------------------------------------
# ----------------------------------------------------------------------


def _phase_variance(a: np.ndarray, n: int, L: int,
                    deadline: float, max_iter: int = 30_000) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the variance of the convolution:
        minimise ½·∑(conv_k - μ)²   with μ = mean(conv).
    Returns the (possibly) improved a and its exact objective.
    """
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.5                       # base learning rate
    cur_obj = _objective_simplex(a, n, L)

    it = 0
    while time.time() < deadline and it < max_iter:
        conv = _conv_fft(a, L)
        mu = conv.mean()
        w = conv - mu
        grad = _correlation_grad(w, a, L)          # 2·(w⋆a)
        # Adam update (no bias‑correction – cheap and works well)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        step = lr / (np.sqrt(v) + eps)
        a_new = a - step * m
        a_new = _project_to_simplex(a_new)

        a = a_new
        it += 1

        # occasional exact evaluation
        if it % 1_000 == 0:
            cur_obj = _objective_simplex(a, n, L)
    return a, cur_obj


def _phase_softmax(a: np.ndarray, n: int, L: int,
                  deadline: float,
                  max_iter: int = 150_000,
                  alpha_start: float = 5.0,
                  alpha_end: float = 2000.0) -> tuple[np.ndarray, float]:
    """
    Adam descent on the log‑sum‑exp surrogate of the max‑norm.
    α is linearly annealed from α_start to α_end.
    """
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    base_lr = 0.5

    cur_obj = _objective_simplex(a, n, L)

    it = 0
    while time.time() < deadline and it < max_iter:
        conv = _conv_fft(a, L)
        # linear schedule for α
        frac = min(1.0, it / max_iter)
        alpha = alpha_start + (alpha_end - alpha_start) * frac

        grad = _smooth_gradient(a, conv, alpha, L)

        # Adam step
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        step = base_lr / (np.sqrt(v) + eps)

        a_new = a - step * m
        a_new = _project_to_simplex(a_new)

        a = a_new
        it += 1

        if it % 2_000 == 0:
            cur_obj = _objective_simplex(a, n, L)
    return a, cur_obj


def _phase_hillclimb(a: np.ndarray, n: int, L: int,
                     deadline: float,
                     max_iter: int = 200_000,
                     no_improve_limit: int = 5_000,
                     temperature0: float = 0.02,
                     subgrad_every: int = 7) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing on the exact objective.
    Random mass‑transfer moves are combined with a sub‑gradient step
    targeting the current maximal convolution entry.
    """
    cur_conv = _conv_fft(a, L)
    cur_obj = 2.0 * n * cur_conv.max()
    best_obj = cur_obj
    best_seq = a.copy()

    it = 0
    no_improve = 0
    T = temperature0

    while time.time() < deadline and it < max_iter:
        # ---- random mass transfer ------------------------------------
        i, j = np.random.randint(n, size=2)
        if i != j and a[i] > 0.0:
            delta = a[i] * np.random.rand() * 0.10          # ≤10 % of a[i]
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            a_new = _project_to_simplex(a_new)

            conv_new = _conv_fft(a_new, L)
            new_obj = 2.0 * n * conv_new.max()

            # Metropolis acceptance
            if new_obj < cur_obj or np.random.rand() < math.exp((cur_obj - new_obj) / max(T, 1e-12)):
                a = a_new
                cur_conv = conv_new
                cur_obj = new_obj
                no_improve = 0
                if cur_obj < best_obj:
                    best_obj = cur_obj
                    best_seq = a.copy()
            else:
                no_improve += 1
        else:
            no_improve += 1

        # ---- sub‑gradient step on the current max --------------------
        if it % subgrad_every == 0:
            g = _subgradient_max_conv(a, cur_conv)
            # simple decreasing step size
            eta = 0.07 * (1.0 - it / max_iter)
            a = a - eta * g
            a = _project_to_simplex(a)
            cur_conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * cur_conv.max()
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_seq = a.copy()

        # ---- temperature decay ----------------------------------------
        T = temperature0 * (1.0 - it / max_iter)

        # ---- restart if stuck -----------------------------------------
        if no_improve >= no_improve_limit:
            break

        it += 1

    return best_seq, best_obj
