"""
Candidate generator for the convolution‐minimisation challenge.

Approach
--------
The objective (after the harmless ‑scale invariance) is to minimise

    score = 2·n·max( a * a )          (sum_i a_i = 1 , a_i ≥ 0)

where * denotes the linear convolution.  A family of analytically
tractable “power‑law” step functions

    a_i(α,δ) ∝ (i + δ)^{‑α} ,    0 < α < 1 ,   δ > 0

covers the Catalan construction (α = ½ , δ → 0).  For a given exponent
α and a shift δ the normalisation factor makes the whole vector a
probability mass function; the maximal convolution value is then a
smooth function of (α,δ).  By scanning a modest grid of (α,δ) on a
small problem size (n≈4000) we can locate a pair that gives a
substantially smaller score than the Catalan baseline (≈1.57).  The
best parameters are finally used to generate a long sequence (n≈20000)
and a few cheap projected sub‑gradient steps are applied for a final
polish.

All heavy work (FFT‑based convolution) is performed inside the search,
so the surrounding evaluation harness only incurs a single call to
``evaluate_sequence`` on the returned list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast linear convolution via a real FFT (full linear mode)
# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * a using a real FFT.
    The result has length 2*len(a)-1.
    """
    n = a.shape[0]
    size = 1 << (2 * n - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Metric for a power‑law shape on a given length
# ----------------------------------------------------------------------
def _score_powerlaw(alpha: float, shift_factor: float, n: int) -> float:
    """
    Build a_i ∝ (i + δ)^{-α} with δ = shift_factor·√n,
    normalise to sum‑one and compute the evaluation score
    (2·n·max(conv)).
    """
    delta = max(1e-12, shift_factor * np.sqrt(n))
    i = np.arange(n, dtype=np.float64)
    a = (i + delta) ** (-alpha)               # unnormalised shape
    a_sum = a.sum()
    if a_sum == 0.0:
        return np.inf
    a = a / a_sum
    conv = _fft_convolve(a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Tiny projected sub‑gradient refinement (identical to the one used
# in the Catalan baseline but generic)
# ----------------------------------------------------------------------
def _refine_subgradient(a: np.ndarray, iters: int, deadline: float) -> np.ndarray:
    """Perform a few projected sub‑gradient steps that directly reduce
    the current maximal convolution entry.
    """
    n = a.size
    lr = 0.02
    best = a.copy()
    best_score = 2.0 * n * np.max(_fft_convolve(a))

    for it in range(iters):
        if time.time() >= deadline:
            break

        conv = _fft_convolve(a)
        k_max = int(np.argmax(conv))

        grad = np.zeros_like(a)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        grad[idx] = 2.0 * a[k_max - idx]

        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 0.0:
            a_new /= s
        else:                                   # fall back – should never happen
            a_new = np.random.random(n)
            a_new /= a_new.sum()

        new_score = 2.0 * n * np.max(_fft_convolve(a_new))

        if new_score < best_score:
            best_score = new_score
            best = a_new.copy()
            # expand step size a little on success
            lr *= 1.02
        else:
            lr *= 0.98

        a = a_new

    return best


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine runs for at most *budget_s*
    seconds and returns the best sequence discovered.

    Returns
    -------
    list[float]
        The candidate sequence (plain Python list).
    """
    np.random.seed(seed)
    now = time.time()
    deadline = now + budget_s - 2.0               # safety margin

    # ------------------------------------------------------------------
    # 1) Coarse grid search on a modest size (fast evaluation)
    # ------------------------------------------------------------------
    n_search = 4000                               # trade‑off speed vs quality
    alphas = np.linspace(0.55, 0.99, 12)          # include Catalan (≈0.5)
    shift_factors = np.linspace(0.0, 5.0, 13)     # δ = factor·√n
    best_score = np.inf
    best_alpha = 0.5
    best_shift = 0.0

    print("[search] starting coarse grid (n=%d)" % n_search)

    for alpha in alphas:
        for sf in shift_factors:
            sc = _score_powerlaw(alpha, sf, n_search)
            if sc < best_score:
                best_score = sc
                best_alpha = alpha
                best_shift = sf
    print("[search] coarse best – α=%.4f  shift_factor=%.4f  score≈%.6f"
          % (best_alpha, best_shift, best_score))

    # ------------------------------------------------------------------
    # 2) Refined random search around the current best
    # ------------------------------------------------------------------
    # random perturbations (Gaussian) – stop when time runs out
    while time.time() < deadline - 5.0:
        # random walk in the (α, shift) plane
        alpha = np.clip(
            best_alpha + np.random.normal(scale=0.02), 0.51, 0.999)
        shift = np.clip(
            best_shift + np.random.normal(scale=0.5), 0.0, 10.0)

        sc = _score_powerlaw(alpha, shift, n_search)
        if sc < best_score:
            best_score = sc
            best_alpha = alpha
            best_shift = shift
            print("[search] new best – α=%.4f  shift_factor=%.4f  score≈%.6f"
                  % (best_alpha, best_shift, best_score))

    # ------------------------------------------------------------------
    # 3) Build the final (large) sequence with the best parameters
    # ------------------------------------------------------------------
    n_final = 20000
    delta_final = max(1e-12, best_shift * np.sqrt(n_final))
    i = np.arange(n_final, dtype=np.float64)
    a = (i + delta_final) ** (-best_alpha)
    a_sum = a.sum()
    if a_sum == 0.0:                      # safety fallback – uniform
        a = np.ones(n_final, dtype=np.float64) / n_final
    else:
        a = a / a_sum

    # ------------------------------------------------------------------
    # 4) Optional sub‑gradient polishing (cheap and never hurts)
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        a = _refine_subgradient(a, iters=300, deadline=deadline)

    # ------------------------------------------------------------------
    # 5) Final sanity check and return
    # ------------------------------------------------------------------
    final_score = 2.0 * n_final * np.max(_fft_convolve(a))
    print("[final] n=%d  α=%.4f  shift_factor=%.4f  tentative score≈%.9f"
          % (n_final, best_alpha, best_shift, final_score))

    return a.tolist()
