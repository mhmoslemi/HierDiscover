    fft_self_len = 1 << ((2*n-1).bit_length())
    fft_grad_len = 1 << ((3*n-2).bit_length())

    while time.time() < deadline and iter < max_iter:
        # compute self-convolution
        fft_a = np.fft.rfft(a, fft_self_len)
        conv_full = np.fft.irfft(fft_a * fft_a, n=fft_self_len)
        conv = conv_full[:2*n-1]
        max_val = conv.max()
        O = 2 * n * max_val
        if O < best_O:
            best_O = O
            best_a = a.copy()
        # compute softmax weights
        beta = min(500.0, beta * 1.005)  # slowly increase
        # subtract max to avoid overflow
        exp_vals = np.exp(beta * (conv - max_val))
        p = exp_vals / exp_vals.sum()
        # compute gradient g
        a_rev = a[::-1]
        fft_p = np.fft.rfft(p, fft_grad_len)
        fft_a_rev = np.fft.rfft(a_rev, fft_grad_len)

        # step size schedule

        # update by exponentiated gradient (multiplicative)
        a = a * np.exp(-step * g)

        # random move jitter occasionally
        if iter % 1000 == 0:

            a = np.clip(a, 0.0, None)
            a = _project_to_simplex(a)
        iter += 1
    return best_a, best_O

        else:
            n = 1000

            x = np.arange(n) + 0.5

            a0 = np.random.rand(n)
            a0 = _project_to_simplex(a0)
        else:
            if 'height_sequence_1' in globals():
                h = np.asarray(height_sequence_1, dtype=float)
                if h.size > n:
                    a0 = h[:n]
                else:
                    a0 = np.concatenate([h, np.zeros(n - h.size)])
                a0 = a0 / a0.sum()
            else:
                a0 = np.random.rand(n)
                a0 = _project_to_simplex(a0)

        a_opt, val_opt = _optimize_sequence(a0, end_time)

def _optimize_sequence(a, deadline):
    n = a.shape[0]
    fft_self_len = 1 << ((2*n-1).bit_length())
    fft_grad_len = 1 << ((3*n-2).bit_length())
    best_a = a.copy()
    best_score = _objective(a)
    iter = 0
    max_iter = 200_000
    step0 = 0.5
    beta = 1.0
    # decide when to switch to subgradient
    switch_iter = 5000
    while time.time() < deadline and iter < max_iter:
        # compute self conv
        fft_a = np.fft.rfft(a, fft_self_len)
        conv_full = np.fft.irfft(fft_a * fft_a, n=fft_self_len)
        conv = conv_full[:2*n-1]
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        # gradient computation
        if iter < switch_iter:
            # softmax gradient
            beta = min(500.0, beta * 1.005)   # slowly increase
            # compute softmax of conv
            exp_vals = np.exp(beta * (conv - max_val))
            p = exp_vals / exp_vals.sum()
            a_rev = a[::-1]
            fft_p = np.fft.rfft(p, fft_grad_len)
            fft_a_rev = np.fft.rfft(a_rev, fft_grad_len)
            conv_pr = np.fft.irfft(fft_p * fft_a_rev, n=fft_grad_len)
            g = 2.0 * conv_pr[n-1:n-1+n]
        else:
            # subgradient of max conv
            # identify all indices where conv close to max_val
            # tolerance relative to max_val
            tol = max_val * 1e-12 + 1e-15
            ks = np.where(conv >= max_val - tol)[0]
            g = np.zeros_like(a)
            for k in ks:
                i_start = max(0, k - (n - 1))
                i_end = min(n - 1, k)
                i_idx = np.arange(i_start, i_end + 1, dtype=int)
                j_idx = k - i_idx
                g[i_idx] += 2.0 * a[j_idx]
            if ks.size > 1:
                g /= ks.size
        # step size schedule
        step = step0 / math.sqrt(iter + 1.0)
        # multiplicative update
        a = a * np.exp(-step * g)
        a = _project_to_simplex(a)

        # occasional random perturbation
        if iter % 1000 == 0 and iter > 0:
            a += np.random.randn(n) * 0.001
            a = np.clip(a, 0.0, None)
            a = _project_to_simplex(a)

        # incremental improvement: if stuck after many iterations, break
        # Could also random restart if no improvement in last 2000 iterations
        iter += 1
    return best_a, best_score

"""
propose_candidate
-----------------

Search for a non‑negative step‑function (a probability vector) that
minimises the bound

    O = 2·n·max(conv(a,a))      with   sum(a)=1

where ``conv`` is the ordinary (linear) convolution.
The optimisation consists of several independent runs, each of which
starts from a different random initialisation (uniform, sine‑shaped,
or the previously discovered construction).  Within a run we perform a
projected gradient descent on a smooth “soft‑max” approximation of the
max‑convolution:

    Lβ(a) = (1/β)·log Σk exp(β·convk(a))

For small β the objective is smooth, for large β it approaches
max(conv).  The gradient of Lβ can be written as a convolution of the
soft‑max weights with the reversed vector, which is computed efficiently
by FFTs.  After each gradient step the iterate is projected onto the
probability simplex (non‑negative and unit sum).  A multiplicative
(up‑dates a←a·exp(‑η·∇Lβ)) scheme preserves positivity automatically.
Occasional random jitter and a simple restart mechanism help to escape
flat regions.

The routine keeps the best sequence found over all runs and finally
returns it as a plain ``list[float]``.  The total wall‑clock time is
limited to the supplied budget (≈ 1000 s).  Progress is printed to
standard output.
"""

import time
import math
import numpy as np

# -------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex
    {x ≥ 0, Σ x = 1}.  Implements the algorithm of Duchi et al.
    """
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# -------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective  O = 2·n·max(conv(a,a))   assuming sum(a)=1."""
    n = a.size
    conv = np.convolve(a, a)          # fine for the moderate sizes used
    return 2.0 * n * conv.max()


# -------------------------------------------------------------
def _optimize_sequence(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the soft‑max surrogate of max(conv).
    Returns the best sequence encountered and its objective value.
    """
    a = a0.copy()
    n = a.size

    # FFT lengths (next powers of two) for speed
    fft_self_len = 1 << ((2 * n - 1).bit_length())      # for a*a
    fft_grad_len = 1 << ((3 * n - 2).bit_length())      # for gradient convolution

    best_a = a.copy()
    best_score = _objective(a)

    # hyper‑parameters
    step0 = 0.5               # base step size for multiplicative update
    beta = 1.0                # start of soft‑max temperature
    max_iter = 200_000        # absolute safety cap
    switch_iter = 5_000       # after this we use the true sub‑gradient of max

    last_improve_iter = 0

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        # ---------- self‑convolution (objective) ----------
        fft_a = np.fft.rfft(a, fft_self_len)
        conv_full = np.fft.irfft(fft_a * fft_a, n=fft_self_len)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve_iter = it
            # Visual progress
            print(f"[opt] iter={it:,} O={best_score:.6f}")

        # ---------- gradient of the surrogate ----------
        if it < switch_iter:
            # smooth soft‑max
            beta = min(500.0, beta * 1.005)                     # slowly sharpen
            # subtract max to avoid overflow
            exp_vals = np.exp(beta * (conv - max_val))
            p = exp_vals / exp_vals.sum()                      # soft‑max weights

            # gradient = 2·(p ⋆ a_rev)   (cross‑correlation)
            a_rev = a[::-1]
            fft_p = np.fft.rfft(p, fft_grad_len)
            fft_a_rev = np.fft.rfft(a_rev, fft_grad_len)
            corr = np.fft.irfft(fft_p * fft_a_rev, n=fft_grad_len)
            g = 2.0 * corr[n - 1 : n - 1 + n]                  # slice length n
        else:
            # exact sub‑gradient of max(conv)
            # treat all indices that are numerically maximal
            eps = max_val * 1e-12 + 1e-15
            ks = np.where(conv >= max_val - eps)[0]
            g = np.zeros_like(a)
            for k in ks:
                i_start = max(0, k - (n - 1))
                i_end = min(n - 1, k)
                i_idx = np.arange(i_start, i_end + 1, dtype=int)
                j_idx = k - i_idx
                g[i_idx] += 2.0 * a[j_idx]
            if ks.size > 1:
                g /= ks.size

        # ---------- multiplicative (exponentiated) update ----------
        step = step0 / math.sqrt(it + 1.0)      # diminishing step
        a = a * np.exp(-step * g)              # keep positivity automatically
        a = _project_to_simplex(a)             # enforce Σa = 1

        # ---------- occasional jitter ----------
        if it % 1000 == 0 and it > 0:
            a += np.random.randn(n) * 0.001
            a = np.clip(a, 0.0, None)
            a = _project_to_simplex(a)

        # ---------- simple restart if stagnated ----------
        if it - last_improve_iter > 30_000:
            # restart from the current best with a small random kick
            a = best_a + np.random.randn(n) * 0.002
            a = np.clip(a, 0.0, None)
            a = _project_to_simplex(a)
            last_improve_iter = it
            print(f"[restart] iter={it:,} – re‑initialising from best.")

    return best_a, best_score


# -------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Entry point called by the evaluation harness.

    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    global_best_seq = None

    # expressive power; we also try a shorter length if time runs low.
    lengths = [2000, 1500, 1000]

    while time.time() < deadline:
        # pick a length that fits into the remaining time
        remaining = deadline - time.time()
        if remaining < 30:                     # very little time left – stop
            break
