"""
Propose a non‑negative step‑function (vector of heights) that minimizes

    O(a) = 2·n·max(conv(a,a)) / (∑ a_i)²

The evaluation routine `evaluate_sequence` is homogeneous of degree 0, so we
work under the normalization Σa_i = 1 (probability simplex).  The search has
two stages:

1. **Adam optimisation on a smooth surrogate** – we minimise the
   log‑sum‑exp approximation of the max‑convolution:
       Fα(a) = (2·n/α)·log( Σ_k exp(α·conv_k) ).
   The gradient w.r.t. the probability vector `a` is
       ∇_a Fα = 2·corr(w, a)  with  w_k ∝ exp(α·conv_k).
   To keep the simplex constraint we optimise the logits `x` such that
   `a = softmax(x)`.  An Adam optimiser with a linearly increasing smoothing
   parameter `α` drives the distribution towards a shape with a flat
   auto‑convolution.

2. **Local hill‑climbing + sub‑gradient refinement** – starting from the
   best candidate of stage 1 we repeatedly perform small “mass‑transfer”
   moves (move a random fraction of probability mass from one index to
   another) and accept them if they improve the exact objective or with a
   small Metropolis probability.  Every few iterations a sub‑gradient step
   for the true max‑convolution entry is applied, and occasional smoothing
   of the vector helps to escape narrow local wells.

The algorithm automatically rescales a previously found sequence stored in
the global variable `height_sequence_1` to a larger size (default 15000),
so that more degrees of freedom are available.  All operations use FFTs
(`numpy.fft.rfft`) and run well within the 1000 s budget.

The function returns the best sequence found as a plain Python list of
floats.

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:                     # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution via FFT (length 2·n‑1)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return the linear self‑convolution a*a (full length 2·n‑1)."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Gradient of the log‑sum‑exp (soft‑max) surrogate
def _smooth_gradient(a: np.ndarray, conv: np.ndarray,
                     alpha: float, L: int) -> np.ndarray:
    """
    Gradient of   (2·n/α)·log Σ_k exp(α·conv_k)   w.r.t. a.
    The result equals 2·corr(w, a) where
        w_k ∝ exp(α·conv_k)  (soft‑max weights).
    """
    # soft‑max weights w_k
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)          # stability trick
    w = exp_scaled / exp_scaled.sum()

    # correlation w ⋆ a (g_i = 2·Σ_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)   # length ≥ 3·n‑2
    n = a.shape[0]
    grad = 2.0 * corr_rev[n - 1 : n - 1 + n]          # central n entries
    return grad


# ----------------------------------------------------------------------
# Sub‑gradient of the maximal convolution entry (exact max, not smoothed)
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Return the sub‑gradient of max(conv) w.r.t. a."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)   # indices where max is attained
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    g /= len(ks)                            # average if several maximisers
    return g

        # ----- random mass transfer -----
        i, j = np.random.randint(target_n, size=2)
        if i != j and a[i] > 0.0:
            # transfer up to 10 % of the source mass
            delta = a[i] * np.random.rand() * 0.10
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            # (still on simplex because total mass unchanged)
            conv_new = _conv_fft(a_new, L)
            new_obj = 2.0 * target_n * conv_new.max()

            # Metropolis acceptance (mostly downhill)
            if new_obj < cur_obj:
                accept = True
            else:
                # temperature slowly cools to zero
                T = temperature0 * max(0.0,
                                       (deadline - time.time()) /
                                       (deadline - phase2_start))
                acceptance_prob = math.exp((cur_obj - new_obj) / max(T, 1e-12))
                accept = np.random.rand() < acceptance_prob

            if accept:
                a = a_new
                conv = conv_new
                cur_obj = new_obj
                no_improve = 0
                if cur_obj < best_obj:
                    best_obj = cur_obj
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:6.1f}s] New global best O={best_obj:.6f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        # ----- sub‑gradient step on the true max entry (every few iter) -----
        if (iter1 + 1) % subgrad_interval == 0:
            g = _subgradient_max_conv(a, conv)
            # a simple decreasing step size
            eta = 0.07 * (1.0 - (time.time() - phase2_start) / (deadline - phase2_start))
            a = a - eta * g
            a = np.clip(a, 0.0, None)
            a = _project_to_simplex(a)          # re‑project onto simplex
            conv = _conv_fft(a, L)
            cur_obj = 2.0 * target_n * conv.max()
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.1f}s] Sub‑grad new best O={best_obj:.6f}")

        # ----- gentle smoothing every few thousand steps (helps escape local spikes) -----
        if (iter1 + 1) % 5000 == 0:
            kernel = np.ones(5) / 5.0
            a = np.convolve(a, kernel, mode='same')
            a = _project_to_simplex(a)

        # restart if stuck for a while
        if no_improve >= max_no_improve:
            # re‑initialise around the current best with a tiny perturbation
            a = best_a.copy()
            a += np.random.randn(target_n) * 0.001
            a = np.clip(a, 0.0, None)
            a = _project_to_simplex(a)
            conv = _conv_fft(a, L)
            cur_obj = 2.0 * target_n * conv.max()
            no_improve = 0
            print(f"[{time.time() - start_time:6.1f}s] Restarted local search (stagnation)")

        iter1 += 1

    # ------------------------------------------------------------------
    # 4) Final sanity check with the provided evaluation function
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:                     # pragma: no cover
        print("[final] evaluate_sequence failed:", e)

    return final_seq
