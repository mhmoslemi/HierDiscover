"""
Propose a step‑function that minimises

    2 * n * max(conv(a,a)) / (sum(a)**2)

The algorithm consists of three stages:

1. **Focused parametric sweep** – exhaustive search over pure exponential
   families a_i ∝ exp(-β·i) and over power‑law families
   a_i ∝ (i+Δ)^(-α).  The best candidate from the sweep is kept as a
   warm‑start.

2. **Simple gradient descent on a smooth surrogate** – we minimise the
   soft‑max of the convolution
       φ(a)= (1/λ)·log Σ_k exp(λ·(a∗a)_k)
   (λ grows from 1 to ~20).  The gradient of φ is
       ∇φ_i = 2·Σ_k p_k·a_{k−i}      with  p_k = softmax_λ(conv)_k .
   After each step we project back onto the probability simplex.

3. **Targeted hill‑climbing refinement** – as in the reference code we move
   a small amount of mass from the index that contributes most to the
   current maximal convolution value to the index that contributes the
   least, accepting only improving moves (with a tiny probability of
   accepting a worsening move to escape plateaus).

All three phases respect the time‑budget (default 950 s) and the
non‑negativity/sum‑to‑one constraints required by ``evaluate_sequence``.
The best sequence found is returned as a plain ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (must be defined at top level – no closures)
# ----------------------------------------------------------------------


def _project_to_simplex(v):
    """Project a vector onto the probability simplex {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


def _fast_score(arr):
    """Fast evaluation of the objective for a normalised (sum=1) non‑negative array."""
    total = arr.sum()
    if total <= 0:
        return float('inf')
    conv = np.convolve(arr, arr)
    max_val = conv.max()
    n = arr.shape[0]
    return 2.0 * n * max_val / (total * total)


def _candidate_from_exponential(beta, n):
    """Return a normalised exponential shape a_i ∝ exp(-beta*i)."""
    idx = np.arange(n, dtype=float)
    raw = np.exp(-beta * idx)
    norm = raw / raw.sum()
    return norm


def _candidate_from_powerlaw(alpha, shift, n):
    """Return a normalised power‑law shape a_i ∝ (i+shift)^(-alpha)."""
    idx = np.arange(n, dtype=float)
    raw = np.power(idx + shift, -alpha)
    norm = raw / raw.sum()
    return norm

    Parameters
    ----------
    seed : int, optional
        Seed for NumPy's RNG (default 123).
    budget_s : float, optional
        Time budget in seconds.  The function will stop before this limit.
    **kwargs
        Ignored – kept for compatibility with the evaluation harness.

    Returns
    -------
    list[float]
        The best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # Determine length (use the reference sequence if available)
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # provided by the harness
    except Exception:

    # ------------------------------------------------------------------
    # Baseline – start from the reference sequence if possible
    # ------------------------------------------------------------------
    try:
        best_seq = list(height_sequence_1)
        best_score = evaluate_sequence(best_seq)
    except Exception:
        best_seq = np.full(n, 1.0 / n).tolist()
        best_score = evaluate_sequence(best_seq)

    print(f"[init] n={n}, baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1. Focused parametric sweep (exponential + power‑law)

    # Exponential sweep – β in a log‑space grid that covers the region

        if time.time() - start > budget_s - 30:

        cand = _candidate_from_exponential(beta, n)
        score = evaluate_sequence(cand.tolist())
        if score < best_score:
            best_score = score
            best_seq = cand.tolist()
            print(f"[exp‑sweep] β={beta:.2e} → {score:.6f}")

    # Power‑law sweep – coarse grid first, then a finer random search
    alphas = np.linspace(0.15, 1.30, 30)
    shifts = np.linspace(0.01, 5.0, 30)
    for alpha in alphas:
        for shift in shifts:
            if time.time() - start > budget_s - 30:
                break
            cand = _candidate_from_powerlaw(alpha, shift, n)
            score = evaluate_sequence(cand.tolist())
            if score < best_score:
                best_score = score
                best_seq = cand.tolist()
                print(f"[pl‑grid] α={alpha:.3f}, Δ={shift:.3f} → {score:.6f}")

    # Randomised power‑law samples to explore the neighbourhood
    for _ in range(2000):
        if time.time() - start > budget_s - 30:
            break
        alpha = rng.uniform(0.10, 1.40)
        shift = rng.uniform(0.001, 6.0)
        cand = _candidate_from_powerlaw(alpha, shift, n)
        score = evaluate_sequence(cand.tolist())
        if score < best_score:
            best_score = score
            best_seq = cand.tolist()

    print(f"[sweep] best score after parametric sweep: {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Gradient descent on a smooth surrogate of the max‑convolution
    # ------------------------------------------------------------------
    cur = np.array(best_seq, dtype=float)          # current working vector
    cur = _project_to_simplex(cur)                 # safety normalisation
    max_iters = 6000                               # per‑run limit
    total_iters = 0
    # schedule for λ (soft‑max temperature)
    lam_start, lam_end = 1.0, 20.0

    while time.time() - start < budget_s - 20:
        # Re‑initialise from the current best (occasionally inject randomness)
        if rng.random() < 0.2:
            # a fresh random Dirichlet start
            cur = rng.dirichlet(np.ones(n) * 0.5)
        else:
            cur = np.array(best_seq, dtype=float)
        cur = _project_to_simplex(cur)

        for it in range(max_iters):
            elapsed = time.time() - start
            if elapsed > budget_s - 10:
                break

            # ----- current objective (fast version) -----
            conv = np.convolve(cur, cur)                # O(n²) but cheap for n≈2200
            maxconv = conv.max()
            cur_score = 2.0 * n * maxconv / (cur.sum() ** 2)

            # keep global best (periodic, not every iteration)
            if it % 250 == 0:
                real_score = evaluate_sequence(cur.tolist())
                if real_score < best_score - 1e-12:
                    best_score = real_score
                    best_seq = cur.tolist()
                    print(f"[gd‑improve] iter={total_iters + it} score={real_score:.6f}")

            # ----- soft‑max weight over the convolution -----
            lam = lam_start + (lam_end - lam_start) * (it / max_iters)
            # numerical stability
            shift = conv.max()
            w = np.exp(lam * (conv - shift))
            p = w / w.sum()                         # softmax probabilities, length 2n‑1

            # ----- gradient of the smooth surrogate -----
            # ∇φ_i = 2 * Σ_k p_k * a_{k-i}
            grad = 2.0 * np.correlate(p, cur, mode='valid')   # length n

            # ----- step size (decreasing) -----
            # we control the Euclidean step length rather than the raw η
            base_step = 0.08 * (1.0 - it / max_iters) + 0.02   # between 0.02 and 0.10
            grad_norm = np.linalg.norm(grad)
            if grad_norm == 0:
                break
            eta = base_step / (grad_norm + 1e-12)

            # ----- gradient step + projection -----
            nxt = cur - eta * grad
            nxt = np.maximum(nxt, 0.0)
            nxt = _project_to_simplex(nxt)

            cur = nxt

        total_iters += max_iters
        # after a gradient run we try a short hill‑climb to clean up any
        # remaining peaks (the next phase will also run later on the best)
        if time.time() - start > budget_s - 5:
            break

    # ------------------------------------------------------------------
    # 3. Targeted hill‑climbing refinement (identical to the reference logic)
    # ------------------------------------------------------------------
    cur = np.array(best_seq, dtype=float)
    cur = _project_to_simplex(cur)
    cur_conv = np.convolve(cur, cur)
    max_hc_iters = 150_000

    it = 0
    while time.time() - start < budget_s - 3 and it < max_hc_iters:
        it += 1

        # index of the maximal convolution entry
        max_idx = int(np.argmax(cur_conv))

        # contribution of each coefficient to that entry
        i_arr = np.arange(n)
        rev = max_idx - i_arr
        mask = (rev >= 0) & (rev < n)
        contrib = np.zeros_like(cur)
        contrib[mask] = cur[i_arr[mask]] * cur[rev[mask]]

        src = int(np.argmax(contrib))
        tgt = int(np.argmin(contrib))
        if src == tgt or cur[src] <= 0:
            src, tgt = rng.choice(n, size=2, replace=False)

        # step size – slowly shrinks
        frac = 0.08 * max(0.0, 1.0 - it / max_hc_iters) + 0.02

        new[src] -= step
        new[tgt] += step
        new = _project_to_simplex(new)

        new_conv = np.convolve(new, new)
        new_max = new_conv.max()
        new_score = 2.0 * n * new_max / (new.sum() ** 2)

        accept = False

            accept = True
        elif rng.random() < 0.001:
            accept = True

        if accept:

            cur_conv = new_conv

            # occasional exact evaluation
            if it % 2000 == 0:
                real = evaluate_sequence(cur.tolist())
                if real < best_score - 1e-12:
                    best_score = real
                    best_seq = cur.tolist()

    # Final normalisation and return
    # ------------------------------------------------------------------
    best_arr = np.array(best_seq, dtype=float)

    else:

    final_score = evaluate_sequence(best_arr.tolist())
    print(f"[final] best score = {final_score:.6f}")
    return best_arr.tolist()
