"""
Propose a step‑function sequence that (approximately) minimises

    O = 2·n·max(conv(a,a)) / (sum(a)^2)

using a smooth‑max gradient descent on the simplex.  The method
starts from the best known construction (``height_sequence_1`` if
available) and repeatedly:

* computes the self‑convolution,
* builds a soft‑max weight over the convolution entries (parameter β),
* evaluates the gradient of the smoothed objective,
* takes a projected‑gradient step on the probability simplex,
* occasionally performs a true sub‑gradient step on the exact max,
* injects tiny random perturbations to escape flat regions.

The loop runs until the 1000 s budget expires (with a 0.5 s safety
margin).  The best sequence found is returned as a plain list of
floats; the exact bound is finally reported with ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:          # fallback – uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of   max_k (a * a)[k]   w.r.t. a on the simplex.
    The returned vector has the same shape as a.
    """
    n = a.size
    conv = np.convolve(a, a)
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)          # active convolution indices
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        # each unordered pair (i,j) contributes twice, i=j once
        g[i_idx] += 2.0 * a[j_idx]
    # average over all active ks to keep the step size stable
    g /= len(ks)
    return g


# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, beta: float) -> tuple[np.ndarray, float]:
    """
    Compute the gradient of the smoothed max‑conv objective

        fβ(a) = (1/β) * log( Σ_k exp(β·conv_k(a)) )

    and also return the current max convolution value.
    """
    n = a.size
    conv = np.convolve(a, a)            # length 2n‑1
    max_c = conv.max()
    # soft‑max weights (numerically stable because max_c is subtracted)
    w = np.exp(beta * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0.0:                     # safety – fall back to uniform weights
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # gradient: g_j = 2 * Σ_i w_i * a_{i‑j}
    # Implemented via convolution with the reversed a.
    rev_a = a[::-1]
    conv_wr = np.convolve(w, rev_a)               # length 3n‑2
    # take the first n entries and reverse to align with indices 0..n‑1
    grad = 2.0 * conv_wr[:n][::-1]
    return grad, max_c


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the bound
    used by `evaluate_sequence`.  Returns the best sequence discovered
    within the time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5          # safety margin

    # ---- initialise ----------------------------------------------------
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=float)
        a = np.maximum(a, 0.0)                       # safety
    else:
        # start from a modest length; 3000–5000 works well for the budget
        n = 3500
        a = np.random.rand(n).astype(float)

    # normalise to a probability vector (sum = 1)
    a = _project_to_simplex(a)
    n = a.size

    # bookkeeping
    conv = np.convolve(a, a)
    best_obj = 2.0 * n * conv.max()
    best_seq = a.copy()
    cur_obj = best_obj
    print(f"[init] n={n}, O={best_obj:.6f}")

    # parameters for the schedule
    max_iter = 300_000          # hard cap (the time limit will stop earlier)
    beta_start = 0.5
    beta_end = 30.0
    eta_sub_start = 0.07
    eta_sub_end = 0.005
    eta_smooth_start = 0.25
    eta_smooth_end = 0.02
    no_improve_counter = 0
    max_no_improve = 8_000

    iter_cnt = 0
    while time.time() < deadline and iter_cnt < max_iter:
        iter_cnt += 1

        # linearly increase β and decay step sizes
        beta = beta_start + (beta_end - beta_start) * (iter_cnt / max_iter)
        eta_sub = eta_sub_start + (eta_sub_end - eta_sub_start) * (iter_cnt / max_iter)
        eta_smooth = eta_smooth_start + (eta_smooth_end - eta_smooth_start) * (iter_cnt / max_iter)

        # ----- sub‑gradient step on the exact maximum every few iterations -----
        if iter_cnt % 11 == 0:
            g_sub = _subgradient_max_conv(a)
            a_tmp = a - eta_sub * g_sub
            # keep feasibility
            a_tmp = np.clip(a_tmp, 0.0, None)
            a_tmp = _project_to_simplex(a_tmp)
            a = a_tmp
        else:
            # ----- smooth‑max gradient step -----
            grad, max_c = _smooth_gradient(a, beta)
            a_tmp = a - eta_smooth * grad
            a_tmp = np.clip(a_tmp, 0.0, None)
            a_tmp = _project_to_simplex(a_tmp)
            a = a_tmp

        # occasional random jitter to shake out flat plateaus
        if iter_cnt % 5_000 == 0:
            jitter = np.random.randn(n) * 0.002
            a = a + jitter
            a = np.clip(a, 0.0, None)
            a = _project_to_simplex(a)

        # evaluate current objective
        conv = np.convolve(a, a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj + 1e-12 < best_obj:          # improvement found
            best_obj = cur_obj
            best_seq = a.copy()
            no_improve_counter = 0
            print(f"[iter {iter_cnt}] New best O={best_obj:.6f}")
        else:
            no_improve_counter += 1

        # simple restart if stagnation persists
        if no_improve_counter >= max_no_improve:
            print(f"[restart] Stagnated after {iter_cnt} iters – re‑initialising.")
            a = best_seq.copy()
            a += np.random.randn(n) * 0.003
            a = np.clip(a, 0.0, None)
            a = _project_to_simplex(a)
            no_improve_counter = 0

        # periodic progress report (lightweight)
        if iter_cnt % 10_000 == 0:
            elapsed = time.time() + budget_s - deadline - budget_s
            print(f"[status] iter={iter_cnt}, O={cur_obj:.6f}, best={best_obj:.6f}")

    # ------------------------------------------------------------------

    except Exception as e:                     # pragma: no cover
