def simplex_projection(v):
    """Project onto the probability simplex (non-negative entries sum to 1)."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    # Find rho = max_i { u_i - cssv_i / i > 0 }
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

The algorithm proceeds in four stages:

    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Catalan (arcsine) seed.
# a_0 = 1, a_{i+1} = a_i * (2*i+1) / (2*i+2)
# ----------------------------------------------------------------------
def _catalan_sequence(length: int) -> np.ndarray:
    """Return the first ``length`` Catalan‑like coefficients."""
    seq = np.empty(length, dtype=float)
    a = 1.0
    seq[0] = a
    for i in range(1, length):
        a *= (2 * i - 1) / (2.0 * i)
        seq[i] = a
    return seq


# ----------------------------------------------------------------------
# Power‑law “U‑shaped” families: a_i ∝ ((i+δ)*(n‑1‑i+δ))^{-α}
# ----------------------------------------------------------------------
def _powerlaw_sequence(n: int, alpha: float, delta: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    w = ((i + delta) * (n - 1 - i + delta)) ** (-alpha)
    # guard against infinities (α large, delta=0 → divide by zero)
    w = np.where(np.isfinite(w), w, 0.0)
    seq = w / w.sum()
    return seq


# ----------------------------------------------------------------------
# Projection of a vector onto the probability simplex (ℓ₂ projection)
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum(x)=1}."""
    # Implementation of the algorithm from
    #   Wang & Carreira‑Perpiñán, “Projection onto the probability simplex”
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    # Find the largest rho where u_i - cssv_i / i > 0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient descent on the simplex (additive updates)
# ----------------------------------------------------------------------
def _subgrad_descent(seq: np.ndarray, deadline: float) -> np.ndarray:
    """
    Additive projected sub‑gradient descent that directly minimises the
    maximal self‑convolution.  ``seq`` is assumed to be non‑negative and
    normalised (sum = 1).  The current best sequence is returned.
    """
    n = seq.size
    best_seq = seq.copy()
    best_score = _fast_evaluation(best_seq)

    fft_len = 1 << ((2 * n - 1).bit_length())
    it = 0
    while time.time() < deadline:
        # convolution
        fa = np.fft.rfft(seq, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        M = conv.max()
        # active indices (within a tiny tolerance)
        active = np.where(conv >= M - 1e-12)[0]

        # sub‑gradient of the max‑operator
        g = np.zeros_like(seq)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            i_idx = np.arange(lo, hi + 1)
            j_idx = k - i_idx
            g[i_idx] += seq[j_idx]

        # step‑size schedule (diminishing)
        eta = 0.5 / np.sqrt(it + 1)

        # additive update + projection onto simplex
        v = seq - eta * g
        seq_new = _simplex_projection(v)

        score = _fast_evaluation(seq_new)

        if score < best_score - 1e-12:
            best_score = score
            best_seq = seq_new.copy()
            seq = seq_new
            # occasional progress report
            if it % 2000 == 0:
                print(f"[subgrad] iter {it:6d}  score {best_score:.6f}")
        else:
            # occasional random jitter to escape plateaus
            if np.random.rand() < 0.001:
                idx = np.random.randint(n)
                seq[idx] = max(0.0, seq[idx] + np.random.randn() * 1e-3)
                seq = _simplex_projection(seq)
        it += 1
    return best_seq


# ----------------------------------------------------------------------
# Exponentiated‑gradient refinement on the simplex (multiplicative updates)
# ----------------------------------------------------------------------
def _eg_refine(seq: np.ndarray, deadline: float) -> np.ndarray:
    """
    Exponentiated‑gradient descent that directly minimises the maximal
    self‑convolution.  ``seq`` is assumed to be non‑negative and normalised
    to have sum = 1 (the metric is scale‑invariant).
    """
    n = seq.size
    best_seq = seq.copy()
    best_score = _fast_evaluation(best_seq)

    fft_len = 1 << ((2 * n - 1).bit_length())
    it = 0
    while time.time() < deadline:
        # convolution
        fa = np.fft.rfft(seq, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        M = conv.max()

        # active indices (those attaining the maximum up to tolerance)
        tol = 1e-12
        active = np.where(conv >= M - tol)[0]

        # sub‑gradient of the max‑operator:
        g = np.zeros_like(seq)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            i_range = np.arange(lo, hi + 1)
            j_range = k - i_range
            g[i_range] += seq[j_range]

        # step‑size schedule (larger initially, decays as 1/√t)
        eta = 0.5 / np.sqrt(it + 1)

        # multiplicative update (exponentiated gradient)
        seq = seq * np.exp(-eta * g)

        # keep positivity, renormalise to the simplex
        seq = np.maximum(seq, 1e-15)
        seq_sum = seq.sum()
        if seq_sum == 0.0:                     # extremely unlikely
            seq = np.full(n, 1.0 / n)
        else:
            seq /= seq_sum

        score = _fast_evaluation(seq)
        if score < best_score - 1e-12:
            best_score = score
            best_seq = seq.copy()
            if it % 500 == 0:
                print(f"[EG]   iter {it:6d}  score {best_score:.6f}")

        # very occasional random jitter to avoid tiny stagnation
        if np.random.rand() < 0.001:
            idx = np.random.randint(n)
            seq[idx] = max(0.0, seq[idx] + np.random.randn() * 1e-3)
            seq /= seq.sum()

        it += 1
    return best_seq


# ----------------------------------------------------------------------
# Simple greedy refinement – move mass from the worst contributing index
# to the least contributing one.
# ----------------------------------------------------------------------
def _greedy_refine(seq: np.ndarray, deadline: float) -> np.ndarray:
    """Greedy local improvement; stops when the deadline is reached."""
    n = seq.size
    best_seq = seq.copy()
    best_score = _fast_evaluation(best_seq)

    # pre‑compute FFT length (does not change)
    fft_len = 1 << ((2 * n - 1).bit_length())
    it = 0
    while time.time() < deadline:
        # convolution and its maximum
        fa = np.fft.rfft(seq, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        k = int(conv.argmax())
        max_val = conv[k]

        # contribution weights for each position to that maximal entry
        i_arr = np.arange(n)
        j_arr = k - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        w = np.zeros_like(seq)
        w[mask] = seq[j_arr[mask]]

        # source = index with largest contribution, sink = index with almost no contribution
        contrib = seq * w
        src = int(np.argmax(contrib))
        sink = int(np.argmin(w))

        if seq[src] < 1e-12:
            break

        # move a fraction of the mass (adaptive)
        eps = 0.1 * seq[src]                # more aggressive than the baseline
        new_seq = seq.copy()
        new_seq[src] -= eps
        new_seq[sink] += eps

        score = _fast_evaluation(new_seq)

        if score < best_score - 1e-12:
            seq = new_seq
            best_seq = new_seq.copy()
            best_score = score
            if it % 1000 == 0:
                print(f"[greedy] iter {it:6d}  score {best_score:.6f}")
        else:
            # tiny random jitter to escape flat plateaus
            if np.random.rand() < 0.01:
                jitter_idx = np.random.randint(n)
                jitter = np.random.randn() * 0.001
                new_seq = seq.copy()
                new_seq[jitter_idx] = max(0.0, new_seq[jitter_idx] + jitter)
                seq = new_seq
        it += 1
    return best_seq


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights aiming at a low evaluation score.

    The routine works within ``budget_s`` seconds (default 1000).  It
    builds a diverse set of seeds (including the Catalan arcsine seed,
    several power‑law “U‑shaped” families and random Dirichlet draws),
    keeps the best one, and then improves it by a projected sub‑gradient
    descent, an exponentiated‑gradient descent and a final greedy mass
    transfer.  The best sequence discovered is returned as a plain list.
    """
    np.random.seed(seed)
    start = time.time()
    deadline_total = start + budget_s - 5          # safety margin

    # ------------------------------------------------------------------
    # 1. Choose a length.  A few thousand entries give a good trade‑off.
    # ------------------------------------------------------------------
    n = 5000

    # ------------------------------------------------------------------
    # 2. Build a pool of initial candidates.
    # ------------------------------------------------------------------
    candidates = []

    # uniform
    candidates.append(np.full(n, 1.0 / n))

    # Catalan / arcsine seed
    catalan = _catalan_sequence(n).astype(float)
    candidates.append(catalan / catalan.sum())

    # Power‑law U‑shaped families
    alphas = np.linspace(0.30, 0.70, 9)    # 0.30 … 0.70
    deltas = [0.0, 0.3, 0.5, 1.0]
    for alpha in alphas:
        for delta in deltas:
            seq = _powerlaw_sequence(n, alpha, delta)
            candidates.append(seq)

    # Random Dirichlet draws (different concentration parameters)
    rng = np.random.default_rng(seed + 1)
    for conc in [0.2, 0.5, 1.0]:
        seq = rng.gamma(conc, size=n)
        seq = seq / seq.sum()
        candidates.append(seq)

    # If a historic solution is available, include it as a candidate.
    try:
        if 'height_sequence_1' in globals():
            old = np.array(height_sequence_1, dtype=float)
            if old.size == n:
                candidates.append(old / old.sum())
            else:
                # truncate / pad to the chosen length
                tmp = np.zeros(n, dtype=float)
                l = min(n, old.size)
                tmp[:l] = old[:l]
                candidates.append(tmp / tmp.sum())
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Pick the best seed according to the fast metric.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    for idx, seq in enumerate(candidates):
        score = _fast_evaluation(seq)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
        # occasional log
        if idx % 5 == 0:
            print(f"[seed] candidate {idx:2d}  score {score:.6f}")

    print(f"[init] selected seed score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Projected sub‑gradient descent (additive updates).
    # ------------------------------------------------------------------
    sub_deadline = start + 0.38 * budget_s
    if sub_deadline > deadline_total:
        sub_deadline = deadline_total
    best_seq = _subgrad_descent(best_seq, sub_deadline)
    cur_score = _fast_evaluation(best_seq)
    print(f"[after subgrad] score = {cur_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Exponentiated‑gradient refinement.
    # ------------------------------------------------------------------
    eg_deadline = start + 0.68 * budget_s
    if eg_deadline > deadline_total:
        eg_deadline = deadline_total
    best_seq = _eg_refine(best_seq, eg_deadline)
    cur_score = _fast_evaluation(best_seq)
    print(f"[after EG] score = {cur_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Greedy local mass‑transfer refinement (the final polish).
    # ------------------------------------------------------------------
    greedy_deadline = deadline_total
    best_seq = _greedy_refine(best_seq, greedy_deadline)
    final_score = _fast_evaluation(best_seq)
    print(f"[final] best score = {final_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Return the best sequence as a Python list (the evaluation harness
    #    will apply its own security checks).
    # ------------------------------------------------------------------
    # Slight rescaling to avoid pathological tiny numbers (optional)
    best_seq = np.maximum(best_seq, 0.0)
    return best_seq.tolist()
```

The function `propose_candidate` can be called directly by the evaluation harness.  
It respects the 1000 s budget, works entirely with in‑memory NumPy operations,
and explores several complementary optimisation strategies to push the
evaluation score well below the previous 1.570 ≈ π/2 target.
