Propose a step‑function (non‑negative list) that minimizes

    O(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

where conv is the discrete convolution.  The algorithm is a fully
self‑contained heuristic that makes heavy use of the fact that
`O(a)` is scale‑invariant (the sum can be forced to 1).  It consists of
three phases:

1.  **Initialisation** – start from the provided `height_sequence_1`
    (if any) or from a random simplex point of length ≈ 5000.

2.  **Targeted stochastic optimisation** – a simulated‑annealing loop
    that mixes two kinds of moves:
       * random mass transfers between two positions,
       * *peak‑reduction* moves that explicitly move mass away from the
         index where the convolution attains its current maximum.
    The current objective is recomputed with a cheap NumPy convolution
    (O(n log n)).  An exponential cooling schedule limits uphill
    acceptance.  Every few hundred iterations a cheap sub‑gradient step
    (the exact sub‑gradient of `max(conv)`) is applied.

3.  **Post‑processing** – the best sequence found is optionally
    *pruned*: keeping only the largest entries (preserving their order)
    and re‑normalising.  Each pruned candidate is evaluated with the

The whole routine runs for at most the supplied time budget (default

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto the probability simplex {x≥0, sum(x)=1}."""
    n = v.shape[0]
    # Duchi et al. (2008) – O(n log n) due to sorting
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:                     # all entries are zero
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Fast evaluation of 2·n·max(conv(a,a)) (the denominator is 1)."""
    conv = np.convolve(a, a)
    return 2.0 * a.size * conv.max()


# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) with respect to a.
    Returns a vector of the same size as a.
    """
    n = a.size
    conv = np.convolve(a, a)
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)
    g = np.zeros_like(a)

    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        # each ordered pair (i,j) contributes 2*a[j] to the gradient
        g[i_idx] += 2.0 * a[j_idx]

    g /= len(ks)                     # average over all maximising ks
    return g

                accept = True
            else:
                prob = math.exp((cur_obj - cand_obj) / max(T, 1e-12))
                if np.random.rand() < prob:
                    accept = True

            if accept:
                cur = cand
                cur_obj = cand_obj
                no_improve = 0
                if cand_obj < best_obj:
                    best_obj = cand_obj
                    best = cur.copy()
                    print(f"[peak {iteration}] new global best O={best_obj:.6f}")
            else:
                no_improve += 1

        # ----------------------------------------------------------
        # occasional sub‑gradient descent (smooths the walk)
        if iteration % 97 == 0:
            g = _subgradient_max_conv(cur)
            # step size shrinks linearly
            eta = 0.07 * max(0.0, 1.0 - iteration / 250_000.0)
            cur = cur - eta * g
            cur = np.clip(cur, 0.0, None)
            cur = _project_to_simplex(cur)
            cur_obj = _objective(cur)
            if cur_obj < best_obj:
                best_obj = cur_obj
                best = cur.copy()
                print(f"[sg {iteration}] new global best O={best_obj:.6f}")

        # ----------------------------------------------------------
        # temperature schedule (linear cooling)
        T = T0 * max(0.0, 1.0 - iteration / 250_000.0)

        # ----------------------------------------------------------
        # restart if we have been stuck for a while
        if no_improve >= max_no_improve:
            cur = best.copy()
            cur += np.random.randn(n) * 0.004   # tiny jitter
            cur = np.clip(cur, 0.0, None)
            cur = _project_to_simplex(cur)
            cur_obj = _objective(cur)
            no_improve = 0
            print(f"[restart] iteration {iteration}, re‑initialised from best")

    # --------------------------------------------------------------
    # 2. post‑processing – try to prune tiny entries
    try:
        final_score = evaluate_sequence(best.tolist())
        print(f"[final] raw candidate O={final_score:.6f}")
    except Exception as e:                     # pragma: no cover
        final_score = float('inf')
        print("[final] evaluate_sequence failed:", e)

    # keep a copy that we will possibly replace
    best_seq = best.copy()
    best_score = final_score

    # order indices by descending magnitude, keep only top‑K entries
    # (preserving their original order)
    order = np.argsort(-best_seq)               # descending
    for keep_frac in (0.30, 0.45, 0.60, 0.80):
        keep = int(keep_frac * n)
        keep = max(50, min(keep, n))
        idx = np.sort(order[:keep])             # preserve ordering
        cand_seq = best_seq[idx]
        if cand_seq.sum() < 1e-12:
            continue
        cand_seq = cand_seq / cand_seq.sum()    # re‑normalise
        try:
            sc = evaluate_sequence(cand_seq.tolist())
            if sc < best_score:
                best_score = sc
                best_seq = cand_seq.copy()
                print(f"[prune] keep={keep} -> O={best_score:.6f}")
        except Exception:                       # pragma: no cover
            pass

    print(f"[final] best reported O={best_score:.6f}")
    return best_seq.tolist()
