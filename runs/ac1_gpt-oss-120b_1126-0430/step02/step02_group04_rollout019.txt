def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5
    n=10000
    # define helper functions (conv_fft, eval_score, proj_simplex)
    # build seeds as above
    best_seq = None
    best_score = float('inf')
    for candidate in seeds:
        sc = _eval_score(candidate)
        if sc < best_score:
            best_score = sc
            best_seq = candidate.copy()
    a = best_seq.copy()
    cur_score = best_score

    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9
    beta2=0.999
    eps=1e-8
    lr = 0.1
    alpha = 5.0  # temperature for softmax

    iteration = 0
    no_improve = 0

    while time.time() < deadline:
        b = _convolve_fft(a, a)
        max_b = b.max()
        # softmax weights
        w = np.exp(alpha * (b - max_b))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.ones_like(b) / b.size
        else:
            w /= w_sum
        # gradient w.r.t a:
        # grad = 2 * conv(a, w_rev) then slice as before
        w_rev = w[::-1]
        conv_full = _convolve_fft(a, w_rev)  # length 3n-2
        idx = (len(b)-1) - np.arange(n)
        grad = 2.0 * conv_full[idx]

        # Adam update
        iteration += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad**2)
        m_hat = m / (1 - beta1**iteration)
        v_hat = v / (1 - beta2**iteration)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        sum_a = a_new.sum()
        if sum_a == 0.0:
            a_new = a
        else:
            a_new /= sum_a
        new_score = _eval_score(a_new)
        if new_score < cur_score:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{iteration}] new best {best_score:.6f} (lr={lr:.5f}, α={alpha:.2f})")
                no_improve = 0
        else:
            # if not improved, reduce lr or increase α slowly
            lr *= 0.9
            no_improve += 1

        # schedule
        if iteration % 500 == 0:
            alpha = min(alpha * 1.05, 2000.0)
            if lr < 1e-6:
                lr = 0.1

        if no_improve > 2000:
            # restart perturbation
            a = best_seq + 0.01 * np.random.randn(n)
            a = np.clip(a, 0, None)
            a = a / a.sum()
            cur_score = _eval_score(a)
            m.fill(0); v.fill(0)
            lr = 0.1
            alpha = 5.0
            no_improve = 0
            print(f"[{iteration}] restart (score={cur_score:.6f})")
```

"""
Improved search for a non‑negative step sequence that minimises
evaluate_sequence (i.e. minimise 2·n·max(a∗a)/(∑a)²).

Approach
--------
1. Problem size n is chosen modestly (default 10 000) – large enough
   for a fine discretisation of the continuous optimum, still cheap
   enough for many FFT‑based iterations.
2. A pool of diverse seeds is built:
   * the standard Catalan‑like sequence (as used previously);
   * a symmetric “arcsine’’ seed a_i ∝ 1/√((i+½)(n‑i‑½));
   * several exponential (geometric) families a_i ∝ p·r^i;
   * a random point on the probability simplex;
   * the optional historic best `height_sequence_1` if present.
   The best seed (lowest exact score) becomes the starting point.
3. Optimisation uses an Adam‑style projected gradient descent on the
   smooth approximation of the max–operator.
   * b = a∗a is obtained with a real FFT (O(n log n));
   * soft‑max weights w_k ∝ exp(α·(b_k−max(b))) give a smooth
     surrogate of the max; the gradient of the surrogate is
     g = 2·(a ⊛ w_rev) restricted to the n relevant entries.
   * Adam moments (m, v) are updated, the raw step a←a−η·m̂/√(v̂+ε) is
     taken, then projected back onto the probability simplex
     (non‑negative, sum = 1) by Euclidean projection.
   * The temperature α is increased slowly (so the surrogate
     converges to the true max) and the base learning rate η is
     gradually decayed.  If no improvement is seen for a while the
     algorithm is re‑initialised by adding small Gaussian noise around
     the current best solution.
4. The loop runs until the time budget (≈ 1000 s) is exhausted; the
   best sequence discovered is returned as a plain Python list.

The optimisation is scale‑invariant, therefore all iterates live on the
simplex (∑a_i = 1).  The algorithm is fully deterministic apart from the
random seeds used for initialisation and occasional perturbations.

import numpy as np

def _convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of real vectors x and y using a real FFT."""
    n = x.size + y.size - 1
    size = 1 << (n - 1).bit_length()          # next power of two
    X = np.fft.rfft(x, size)
    Y = np.fft.rfft(y, size)
    Z = X * Y
    conv = np.fft.irfft(Z, size)[:n]
    return conv.real


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x ≥ 0, sum = 1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u > cssv / np.arange(1, n + 1))[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _eval_score(a: np.ndarray) -> float:
    """Exact evaluation of evaluate_sequence for a non‑negative simplex vector."""
    n = a.size
    b = _convolve_fft(a, a)
    max_b = b.max()
    return float(2.0 * n * max_b / (a.sum() ** 2))


def _catalan_sequence(n: int) -> np.ndarray:
    """Catalan‑like seed: a_i = C(2i,i)/4**i, normalised to sum 1."""
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * (i - 1) + 1) / (2.0 * i)
    a /= a.sum()
    return a


def _arcsine_sequence(n: int) -> np.ndarray:
    """Symmetric arcsine‑like seed:
       a_i ∝ 1/√((i+½)(n‑i‑½)).
    """
    idx = np.arange(n, dtype=float)
    a = 1.0 / np.sqrt((idx + 0.5) * (n - idx - 0.5))
    a /= a.sum()
    return a


def _geometric_sequence(n: int, p: float) -> np.ndarray:
    """Geometric (exponential) seed with parameter p (effective decay = p/n)."""
    eps = p / n
    r = 1.0 - eps
    a = eps * (r ** np.arange(n))
    a /= a.sum()
    return a


def _random_simplex(n: int) -> np.ndarray:
    """Random point on the probability simplex."""
    rnd = np.random.rand(n)
    return _proj_simplex(rnd)

def propose_candidate(seed: int = 12345, budget_s: int = 1000) -> list:
    """
    Search for a step‑function (list of non‑negative heights) that
    minimises ``evaluate_sequence`` within the allotted time budget.
    Returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 4.0  # small safety margin

    # ----- problem size -------------------------------------------------

    # ----- Adam hyper‑parameters ----------------------------------------
