The goal is to find a non‑negative vector `a` (the heights of a step function) that minimises  

\[
\text{score}(a)=\frac{2\,n\;\max(a*a)}{(\sum a)^2},
\]

where `*` denotes the linear convolution.  
The score is *scale‑invariant* – we can force `∑a = 1` and minimise simply `2·n·max(a*a)`.

The program follows a three‑stage strategy:

1. **Generate a diverse pool of seeds** – Catalan numbers, geometric decays,
   power‑law decays, Gaussian‑shaped vectors and a few random Dirichlet draws.
   The best seed (according to the exact score) becomes the starting point.

2. **Projected smooth‑max Adam optimisation** –  
   *smooth max* of the convolution is approximated by a soft‑max with a
   temperature `α`.  The gradient of this smooth max is  

   \[
   \nabla = 2\;(a\;\overset{\text{lin}}{*}\;w^{\text{rev}})_{0..n-1},
   \]

   where `w` are the normalised soft‑max weights and `w_rev` is a reversal.  
   Adam is used to obtain an adaptive per‑coordinate step size; after each
   update the vector is projected back onto the probability simplex (non‑negative,
   sum = 1) using the Euclidean projection of Duchi et al.

3. **Stagnation handling** – if the optimiser makes no improvement for a while,
   a small random perturbation around the current best point is performed and the
   optimiser is re‑initialised.  The learning‑rate `lr` and the soft‑max temperature
   `α` are increased slowly so that the optimisation focuses increasingly on the
   true maximum.

All heavy work (convolutions) is performed with a real FFT, giving an
`O(n log n)` cost per iteration.  With `n=8000` the routine runs thousands of
iterations well inside the 1000 s budget.

The function `propose_candidate` returns the best sequence found as a plain
Python list of floats, ready to be handed to `evaluate_sequence`.

```python
#!/usr/bin/env python
# -*- coding: utf-8 -*-

"""
propose_candidate
~~~~~~~~~~~~~~~~~

Search for a non‑negative step‑function (list of heights) that minimises
``evaluate_sequence``.  The algorithm consists of

1.  a rich set of initial candidates (Catalan, geometric, power‑law, Gaussian,
    random Dirichlet);
2.  a projected Adam optimiser that minimises a smooth approximation of the
    ∞‑norm of the convolution;
3.  automatic schedule for the soft‑max temperature and learning rate,
    plus random restarts when progress stalls.

All convolutions are performed with a real FFT, the simplex projection is the
Euclidean projection of Duchi et al.  The routine respects the 1000 s time
budget and returns the best sequence found.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, sum=1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u > cssv / np.arange(1, n + 1))[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two real vectors using a real FFT.
    The result has length len(x)+len(y)-1.
    """
    n = x.size + y.size - 1
    size = 1 << (n - 1).bit_length()      # next power of two
    X = np.fft.rfft(x, size)
    Y = np.fft.rfft(y, size)
    Z = X * Y
    conv = np.fft.irfft(Z, size)[:n]
    return conv.real


def _eval_score(a: np.ndarray) -> float:
    """Exact value of evaluate_sequence for a non‑negative simplex vector."""
    n = a.size
    b = _convolve_fft(a, a)
    max_b = b.max()
    # sum(a) == 1 because we keep the vector on the simplex
    return float(2.0 * n * max_b)


def _catalan_sequence(n: int) -> np.ndarray:
    """Catalan‑like coefficients a_i = C(2i,i)/4**i, normalised to sum 1."""
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * (i - 1) + 1) / (2.0 * i)
    a /= a.sum()
    return a


def _geometric_sequence(n: int, r: float) -> np.ndarray:
    """Geometric decay a_i ∝ r**i (0<r<1), normalised to sum 1."""
    i = np.arange(n)
    a = (1.0 - r) * (r ** i)
    a /= a.sum()
    return a


def _power_sequence(n: int, alpha: float) -> np.ndarray:
    """Power‑law decay a_i ∝ (i+1)**(-alpha), normalised to sum 1."""
    i = np.arange(1, n + 1, dtype=float)
    a = i ** (-alpha)
    a /= a.sum()
    return a


def _gaussian_sequence(n: int, sigma: float) -> np.ndarray:
    """Discrete half‑Gaussian a_i ∝ exp(-i²/(2σ²)), normalised to sum 1."""
    i = np.arange(n, dtype=float)
    a = np.exp(-(i ** 2) / (2.0 * sigma ** 2))
    a /= a.sum()
    return a


def _random_simplex(n: int) -> np.ndarray:
    """Draw a random point on the simplex using a Dirichlet(1) distribution."""
    v = np.random.rand(n)
    return _proj_simplex(v)


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a step‑function (list of non‑negative heights) that
    minimises ``evaluate_sequence``.  The function runs for at most
    ``budget_s`` seconds and returns the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0  # safety margin

    # Problem size – a size that still gives fast FFTs while offering many degrees of freedom
    n = 8000

    # -----------------------------
    # 1. Build a pool of seeds
    # -----------------------------
    candidates = [_catalan_sequence(n)]

    # geometric family
    for r in np.linspace(0.90, 0.9999, 12):
        candidates.append(_geometric_sequence(n, r))

    # power‑law family
    for alpha in np.linspace(0.5, 3.0, 12):
        candidates.append(_power_sequence(n, alpha))

    # half‑Gaussian family
    for sigma in np.linspace(n / 30.0, n / 5.0, 8):
        candidates.append(_gaussian_sequence(n, sigma))

    # a few random Dirichlet draws
    for _ in range(6):
        candidates.append(_random_simplex(n))

    # -------------------------------------------------
    # Pick the best seed according to the exact score
    # -------------------------------------------------
    best_seq = None
    best_score = np.inf
    for a in candidates:
        sc = _eval_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
    a = best_seq.copy()
    cur_score = best_score
    print(f"[init] n={n}, initial best score = {cur_score:.6f}")

    # -------------------------------------------------
    # 2. Projected Adam optimisation of the smooth‑max
    # -------------------------------------------------
    lr = 0.20                 # base learning rate
    alpha = 5.0               # soft‑max temperature
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)     # 1st moment estimate
    v = np.zeros_like(a)     # 2nd moment estimate
    iteration = 0
    no_improve = 0

    while time.time() < deadline:
        iteration += 1

        # ----- convolution and soft‑max weights -----
        b = _convolve_fft(a, a)
        max_b = b.max()

        # slowly increase the focus on the true maximum
        if iteration % 2000 == 0 and iteration > 0:
            alpha = min(alpha * 1.07, 20000.0)

        # soft‑max (more weight on the current maximal entry)
        w = np.exp(alpha * (b - max_b))          # max entry gets weight 1
        w_sum = w.sum()
        if w_sum > 0.0:
            w /= w_sum
        else:           # numerical safeguard – fall back to uniform weights
            w.fill(1.0 / w.size)

        # ----- gradient of the smooth max -----
        w_rev = w[::-1]
        conv_full = _convolve_fft(a, w_rev)        # length = 3n‑2
        idx = (len(b) - 1) - np.arange(n)          # slice needed for gradient
        grad = 2.0 * conv_full[idx]

        # ----- Adam update with back‑tracking line‑search -----
        # update moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias‑corrected moments
        bias_correction1 = 1.0 - beta1 ** iteration
        bias_correction2 = 1.0 - beta2 ** iteration
        m_hat = m / bias_correction1
        v_hat = v / bias_correction2

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        improved = False
        # try the Adam step
        a_candidate = _proj_simplex(a - step)
        cand_score = _eval_score(a_candidate)
        if cand_score < cur_score:
            improved = True
            a = a_candidate
            cur_score = cand_score
        else:
            # back‑track on the learning rate a few times
            lr_local = lr
            for _ in range(6):
                lr_local *= 0.5
                a_candidate = _proj_simplex(a - lr_local * grad)
                cand_score = _eval_score(a_candidate)
                if cand_score < cur_score:
                    improved = True
                    a = a_candidate
                    cur_score = cand_score
                    lr = lr_local          # keep the smaller step size
                    break

        # ----- bookkeeping -----
        if improved:
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{iteration:06d}] new best {best_score:.6f} "
                      f"(lr={lr:.5f}, α={alpha:.2f})")
                no_improve = 0
            else:
                no_improve += 1
        else:
            # no improvement – shrink the step size a little
            lr *= 0.75
            no_improve += 1

        # periodic decay of the learning rate
        if iteration % 500 == 0:
            lr = max(lr * 0.9, 1e-5)

        # restart if we have been stuck for a while
        if no_improve > 6000:
            a = best_seq + 0.01 * np.random.randn(n)
            a = np.clip(a, 0.0, None)
            a = _proj_simplex(a)
            cur_score = _eval_score(a)
            lr = 0.20
            alpha = 5.0
            m.fill(0.0)
            v.fill(0.0)
            iteration = 0
            no_improve = 0
            print(f"[restart] around best (score={cur_score:.6f})")

    print(f"[done] best score found = {best_score:.6f}")
    # Return as a plain Python list, as required by the harness
    return best_seq.tolist()
```
