"""
Propose a step‑function (list of non‑negative heights) with a small
evaluation score.

Algorithm – smoothed sub‑gradient descent on the “log‑sum‑exp”
approximation of the max‑convolution.
--------------------------------------------------------------------
1. Choose a working length n (5000 – 20000).  If a previous construction
   `height_sequence_1` exists we keep its length, otherwise we use 5000.
2. Initialise the probability vector a ( Σa = 1 ) by trying several
   candidates (power‑law, Dirichlet draws and the previous best) and
   keeping the one with the lowest cheap score 2·n·max(a*a) (computed
   by FFT).
3. Iterate while time permits:
   * Compute the full self‑convolution c = a*a (FFT, linear convolution).
   * Approximate the max by a smooth “log‑sum‑exp”:
         w_k ∝ exp( (c_k – max(c))/T )
     where the temperature T slowly decreases (∝ 1/√t) from a modest
     start (≈0.05) down to a floor (1e‑3).
   * The gradient of the smoothed objective is
         ∇F(a)_i = 2·∑_k w_k·a_{k‑i}
     which is exactly the linear convolution of a with the weight
     vector w (computed again by FFT).
   * Perform a projected gradient step:
         a ← Π_simplex( a – η·∇F(a) )
     with a diminishing step size η = η0/√t (η0≈0.5).
   * Occasionally (every few thousand iterations) add a tiny random
     perturbation and keep the best vector seen so far (according to
     the cheap objective 2·n·max(c)).
4. After the time budget the best vector is evaluated with the exact
   ``evaluate_sequence`` routine and returned.

All helper functions are defined at module level (no closures) and the
code respects the 1000 s time limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    # Find the rho index
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # All entries non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float) -> np.ndarray:
    """Return a normalised power‑law vector a_i ∝ (i+1)⁻ᵖ (Σa = 1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p)
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    return a


def _convolution_fft(a: np.ndarray, b: np.ndarray, L: int) -> np.ndarray:
    """
    Linear convolution of a (len n) and b (len m) using FFT of length L.
    Returns the full result of length n+m‑1 (truncated to that length).
    """
    fa = np.fft.fft(a, L)
    fb = np.fft.fft(b, L)
    conv_full = np.fft.ifft(fa * fb).real
    return conv_full[: a.size + b.size - 1]


def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Linear self‑convolution a*a (full length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, L)
    conv_full = np.fft.ifft(fa * fa).real
    n = a.size
    return conv_full[: 2 * n - 1]


def _cheap_score(a: np.ndarray, conv: np.ndarray) -> float:
    """Score = 2·n·max(conv) for a that already satisfies Σa = 1."""
    n = a.size
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) improves upon the
    current best bound.  The routine respects the given time budget.
    """
    # --------------------------------------------------------------
    # 0. Basic set‑up
    # --------------------------------------------------------------
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0          # keep a safety margin

    # --------------------------------------------------------------
    # 1. Choose a working length (n)
    # --------------------------------------------------------------
    try:
        # use the length of the previously submitted construction if available
        height_sequence_1  # noqa: F821  (reference for existence)
        n = len(height_sequence_1)  # type: ignore[name-defined]
        if n < 5000:
            n = 5000
    except Exception:
        n = 5000

    # FFT length – must accommodate linear convolution of length up to 3·n‑2
    L = 1 << ((3 * n - 2).bit_length())

    # --------------------------------------------------------------
    # 2. Assemble a small pool of initial candidates and pick the best
    # --------------------------------------------------------------
    candidates = []

    # a) previous construction (if any)
    try:
        prev = np.array(height_sequence_1[:n], dtype=np.float64)  # type: ignore[name-defined]
        if prev.size < n:
            prev = np.pad(prev, (0, n - prev.size), constant_values=0.0)
        prev = np.maximum(prev, 0.0)
        s = prev.sum()
        if s > 1e-12:
            prev /= s
        else:
            prev = np.full(n, 1.0 / n)
        candidates.append(prev)
    except Exception:
        pass

    # b) a fine grid of power‑law exponents around the known good region
    for p in np.linspace(0.55, 0.70, 31):
        candidates.append(_power_law_sequence(n, p))

    # c) a few random Dirichlet‑type draws (Gamma with shape 0.5 → sparse)
    for _ in range(6):
        rnd = np.random.gamma(shape=0.5, scale=1.0, size=n)
        s = rnd.sum()
        if s > 0:
            rnd /= s
        candidates.append(rnd)

    # --------------------------------------------------------------
    # 3. Pick the best candidate according to the cheap objective
    # --------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    for a in candidates:
        conv = _conv_self_fft(a, L)
        score = _cheap_score(a, conv)
        if score < best_score:
            best_score = score
            best_seq = a.copy()

    if best_seq is None:
        # fallback – uniform distribution (should never happen)
        best_seq = np.full(n, 1.0 / n)
        best_score = _cheap_score(best_seq, _conv_self_fft(best_seq, L))

    print(f"Initial cheap score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Smoothed sub‑gradient descent on the log‑sum‑exp approximation
    # --------------------------------------------------------------
    a = best_seq.copy()          # current iterate (Σa = 1)
    best_iter_seq = a.copy()
    best_iter_score = best_score

    iter_count = 0
    eta0 = 0.5                    # initial step‑size factor
    T0 = 0.05                     # initial temperature for smoothing
    T_min = 0.001                 # floor temperature
    momentum = np.zeros_like(a)  # Nesterov‑type momentum term
    beta = 0.9                    # momentum decay

    while time.time() < deadline:
        iter_count += 1

        # ----- self‑convolution (full) -----
        conv = _conv_self_fft(a, L)
        max_conv = float(conv.max())

        # ----- temperature schedule (gradual cooling) -----
        T = max(T_min, T0 / math.sqrt(iter_count))

        # ----- softmax weights w_k ∝ exp((c_k - max)/T) -----
        # shift by max to avoid overflow
        exp_vals = np.exp((conv - max_conv) / T)
        w = exp_vals / exp_vals.sum()               # length 2n‑1, sums to 1

        # ----- gradient of the smoothed objective -----
        # grad = 2 * (a * w)[:n]  (linear convolution)
        grad_conv = _convolution_fft(a, w, L)[:n]
        grad = 2.0 * grad_conv

        # ----- Nesterov‑like momentum -----
        momentum = beta * momentum + grad

        # ----- diminishing step size -----
        eta = eta0 / math.sqrt(iter_count)

        # ----- gradient descent step + projection onto simplex -----
        a_candidate = a - eta * momentum
        a = _simplex_projection(a_candidate)

        # ----- occasional random perturbation (helps escape plateaus) -----
        if iter_count % 5000 == 0 and np.random.rand() < 0.15:
            perturb = 0.01 * np.random.rand(n)
            a = _simplex_projection(a + perturb)

        # ----- monitor and keep the best cheap score -----
        if iter_count % 2000 == 0 or time.time() > deadline - 2:
            conv_check = _conv_self_fft(a, L)
            score = _cheap_score(a, conv_check)
            if score < best_iter_score:
                best_iter_score = score
                best_iter_seq = a.copy()
                print(f"[Iter {iter_count:07d}] New best cheap score = {score:.6f}")

    # --------------------------------------------------------------
    # 5. Final exact evaluation with the provided ``evaluate_sequence``
    # --------------------------------------------------------------
    final_seq = best_iter_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:  # defensive – should never happen
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # Return the best sequence found (the one we finally evaluated)
    return final_seq
