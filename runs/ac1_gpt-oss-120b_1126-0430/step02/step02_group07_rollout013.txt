"""
Propose a step‑function (list of non‑negative heights) that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of four phases:

1. **Broad parametric sampling** – power‑law, exponential, mixtures,
   and Beta‑like families are sampled randomly and on a coarse grid.
   Every candidate is normalised to a probability vector (sum = 1)
   and evaluated; the best candidate is kept.

2. **Low‑dimensional global optimisation** – a mixture of two
   “truncated‑power‑law * exponential’’ components is optimised with
   SciPy's differential evolution (7 parameters).  This adds flexibility
   beyond the pure families used in step 1.

3. **Gradient‑descent refinement on the simplex** – the current
   probability vector is represented as a soft‑max of log‑weights.
   At each iteration the index `p` of the maximal convolution entry is
   identified, a sub‑gradient of the objective w.r.t. the log‑weights
   is computed analytically and a diminishing step‑size update is
   applied.  The vectors are projected back onto the simplex via the
   soft‑max.  Whenever a new best value is found it is printed.

4. **Targeted hill‑climbing** – as a final polishing step a very
   simple move‑mass procedure (source = index with largest contribution
   to the current peak, target = index with smallest contribution) is
   performed with a line‑search on the step size.

All phases respect the global time budget (`budget_s`).  The best
sequence found is returned as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def _project_to_simplex(v):
    """Project a vector onto the probability simplex {x ≥ 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


def _fast_score(arr):
    """Fast evaluation of the objective for a non‑negative array that sums to 1."""
    total = arr.sum()
    if total <= 0:
        return float("inf")
    conv = np.convolve(arr, arr)
    max_val = conv.max()
    n = arr.shape[0]
    # sum is assumed = 1, but we keep the denominator for safety
    return 2.0 * n * max_val / (total * total)


def _norm_and_score(raw):
    """
    Normalise a raw non‑negative vector to sum‑1 and evaluate it.
    Returns (score, normalised_vector) where score is obtained from the
    official evaluator (more reliable than the fast version).
    """
    total = raw.sum()
    if total <= 0:
        return float("inf"), None
    norm = raw / total
    score = evaluate_sequence(norm.tolist())
    return score, norm

        if fam == 0:          # power‑law
            alpha = rng.uniform(0.20, 2.00)
            shift = rng.uniform(0.001, 6.0)
            raw = np.power(idx + shift, -alpha)
        elif fam == 1:        # exponential
            beta = rng.uniform(1e-5, 0.3)
            raw = np.exp(-beta * idx)
        elif fam == 2:        # product power‑law * exponential
            alpha = rng.uniform(0.20, 2.00)
            shift = rng.uniform(0.001, 6.0)
            beta = rng.uniform(1e-5, 0.3)
            raw = np.power(idx + shift, -alpha) * np.exp(-beta * idx)
        else:                 # Beta‑like shape (p‑1, q‑1) on [0,1]
            p = rng.uniform(0.2, 2.0)
            q = rng.uniform(0.2, 2.0)
            x = idx / (n - 1)
            raw = (x + 1e-12) ** (p - 1) * (1 - x + 1e-12) ** (q - 1)
        _try_candidate(raw, "randFam")

    # ------------------------------------------------------------------
    # 3. Random mixtures of two pure power‑law components
    # ------------------------------------------------------------------
    for _ in range(2500):
        if time.time() - start > budget_s - 30:
            break
        alpha1 = rng.uniform(0.10, 2.00)
        shift1 = rng.uniform(0.001, 6.0)
        alpha2 = rng.uniform(0.10, 2.00)
        shift2 = rng.uniform(0.001, 6.0)
        p = rng.random()
        raw = (p * np.power(idx + shift1, -alpha1) +
               (1 - p) * np.power(idx + shift2, -alpha2))
        _try_candidate(raw, "mix2PL")

    # ------------------------------------------------------------------
    # 4. Differential Evolution on a 2‑component mixture model
    #    a_i = p*f1(i) + (1-p)*f2(i) with
    #    f(i) = (i+shift)^(-alpha) * exp(-beta*i)
    # ------------------------------------------------------------------
    try:
        import scipy.optimize as opt
        have_de = True
    except Exception:
        have_de = False

    if have_de:
        # ------------------------------------------------------------------
        # Build a sequence from the 7‑dimensional parameter vector
        # ------------------------------------------------------------------
        def _make_seq_from_params(par):
            # par = [p, a1, s1, b1, a2, s2, b2]
            p, a1, s1, b1, a2, s2, b2 = par
            # enforce positivity internally
            a1 = max(a1, 1e-8)
            s1 = max(s1, 1e-8)
            b1 = max(b1, 0.0)
            a2 = max(a2, 1e-8)
            s2 = max(s2, 1e-8)
            b2 = max(b2, 0.0)
            shape1 = np.power(idx + s1, -a1) * np.exp(-b1 * idx)
            shape2 = np.power(idx + s2, -a2) * np.exp(-b2 * idx)
            raw = p * shape1 + (1.0 - p) * shape2
            total = raw.sum()
            if total <= 0:
                return np.full(n, 1.0 / n)
            return raw / total

        def _de_objective(par):
            seq = _make_seq_from_params(par)
            # fast surrogate for the objective (exact up to rounding)
            return _fast_score(seq)

        bounds = [
            (0.0, 1.0),        # p
            (0.10, 3.0),       # a1
            (0.0, 8.0),        # s1
            (0.0, 0.5),        # b1
            (0.10, 3.0),       # a2
            (0.0, 8.0),        # s2
            (0.0, 0.5)         # b2
        ]

        # small budget for DE – we keep some time for later refinements
        de_maxiter = 12
        de_popsize = 10
        print("[DE] starting differential evolution")
        try:
            de_res = opt.differential_evolution(
                _de_objective,
                bounds,
                maxiter=de_maxiter,
                popsize=de_popsize,
                seed=seed,
                polish=False,
                updating='deferred',
                disp=False,
                workers=1,
                tol=1e-6
            )
            seq_de = _make_seq_from_params(de_res.x)
            score_de = evaluate_sequence(seq_de.tolist())
            if score_de < best_score:
                best_score = score_de
                best_seq = seq_de.tolist()
                print(f"[DE] improved to {score_de:.6f}")
        except Exception as e:
            print("[DE] exception:", e)

    # ------------------------------------------------------------------
    # 5. Gradient‑descent style refinement on the simplex (soft‑max)
    # ------------------------------------------------------------------
    a = np.array(best_seq, dtype=np.float64)  # current working vector (sum≈1)
    # initialise log‑weights (soft‑max parametrisation)
    eps = 1e-16
    log_w = np.log(a + eps)

    # schedules
    grad_iters = 3000
    lr_start = 0.30
    lr_end = 0.02

    it = 0
    while time.time() - start < budget_s - 8 and it < grad_iters:
        # compute convolution and peak index
        conv = np.convolve(a, a)
        max_val = conv.max()
        p = int(conv.argmax())

        # sub‑gradient of max_b = a*a[p] w.r.t. a_i  ( = 2*a_{p-i} )
        rev = p - np.arange(n)
        mask = (rev >= 0) & (rev < n)
        g = np.zeros_like(a)
        # 2 * a_{p-i} contributes to max_b; objective has factor 2*n
        g[mask] = 4.0 * n * a[rev[mask]]

        # projected gradient on the simplex via soft‑max trick
        avg = np.dot(a, g)
        w_grad = a * (g - avg)

        # learning rate schedule (linear decay)
        lr = lr_start * (1.0 - it / grad_iters) + lr_end

        # gradient step on log‑weights
        log_w -= lr * w_grad
        a = np.exp(log_w)
        a /= a.sum()                     # guarantees sum‑to‑one

        # evaluate current candidate (fast)
        cur_score = 2.0 * n * np.max(np.convolve(a, a))

        # occasionally verify with the official evaluator
        if it % 200 == 0:
            real_score = evaluate_sequence(a.tolist())
            if real_score < best_score:
                best_score = real_score
                best_seq = a.tolist()
                print(f"[grad] iter={it:4d} new best={real_score:.6f}")

        it += 1

    # ------------------------------------------------------------------
    # 6. Final targeted hill‑climbing (pairwise mass transfers)
    # ------------------------------------------------------------------
    cur = np.array(best_seq, dtype=np.float64)
    cur_conv = np.convolve(cur, cur)
    cur_score = best_score
    max_iters = 200000
    it = 0
    while time.time() - start < budget_s - 2 and it < max_iters:
        it += 1
        # peak index of current convolution
        p = int(cur_conv.argmax())
        # contribution of each index to this peak
        rev = p - np.arange(n)
        mask = (rev >= 0) & (rev < n)
        contrib = np.zeros_like(cur)
        contrib[mask] = cur[mask] * cur[rev[mask]]

        src = int(np.argmax(contrib))
        tgt = int(np.argmin(contrib))
        if src == tgt or cur[src] <= 0:
            src, tgt = rng.choice(n, size=2, replace=False)

        # step size (fraction of source mass) – start fairly aggressive
        frac = 0.12 * max(0.0, 1.0 - it / max_iters) + 0.02   # 0.02 .. 0.14
        step = cur[src] * rng.random() * frac
        if step <= 0:
            continue

        new = cur.copy()
        new[src] -= step
        new[tgt] += step
        new /= new.sum()                     # renormalise (soft‑max like)

        new_conv = np.convolve(new, new)
        new_max = new_conv.max()
        new_score = 2.0 * n * new_max

        # accept only improving moves (tiny chance of uphill moves)
        accept = False
        if new_score < cur_score - 1e-12:
            accept = True
        elif rng.random() < 0.001:
            accept = True

        if accept:
            cur = new
            cur_conv = new_conv
            cur_score = new_score
            # verify with the official evaluator only when we improve globally
            real_score = evaluate_sequence(new.tolist())
            if real_score < best_score - 1e-12:
                best_score = real_score
                best_seq = new.tolist()
                print(f"[refine] iter={it} new best={real_score:.6f}")

        # occasional re‑projection to avoid numerical drift
        if it % 1000 == 0:
            cur = _project_to_simplex(cur)
            cur_conv = np.convolve(cur, cur)

    print(f"[final] best score achieved: {best_score:.6f}")
    return best_seq
