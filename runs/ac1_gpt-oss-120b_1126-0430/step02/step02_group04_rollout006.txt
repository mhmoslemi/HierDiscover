def optimize(n, a_init, time_limit):
    start = time.time()
    a = a_init.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9; beta2 = 0.999; eps = 1e-8
    lr = 0.2
    beta = 5.0
    max_beta = 5000.0
    iteration = 0
    best_a = a.copy()
    best_score = eval_score(a)
    cur_score = best_score
    no_improve = 0
    while time.time() - start < time_limit - 0.2:
        # compute convolution and gradient
        b = convolve_fft(a, a)
        max_b = b.max()
        # update smoothing parameter occasionally
        if iteration % 1000 == 0 and iteration > 0:
            beta = min(beta * 1.05, max_beta)
        # compute softmax weights
        exp_vals = np.exp(beta * (b - max_b))
        w_sum = exp_vals.sum()
        if w_sum == 0.0 or np.isnan(w_sum):
            w = np.ones_like(b) / b.size
        else:
            w = exp_vals / w_sum

def generate_initial_candidates(n):
    candidates = []

    # uniform
    candidates.append(np.full(n, 1.0 / n))

    # Catalan-like sequence
    cat = np.empty(n)
    cat[0] = 1.0
    for i in range(1, n):
        cat[i] = cat[i-1] * (2 * (i-1) + 1) / (2.0 * i)
    cat = cat / cat.sum()
    candidates.append(cat)

    # Geometric decays: rates r = 0.999,0.9995 etc appropriate to shape
    for decay in np.linspace(0.9, 0.9999, 5):
        a = np.geomspace(1.0, decay**(n-1), num=n) # careful - maybe simpler: a = decay**np.arange(n)
        a = decay ** np.arange(n)
        a = a / a.sum()
        candidates.append(a)

    # Power law p in [0.2, 2.0]
    for p in np.linspace(0.2, 2.0, 7):
        a = 1.0 / (np.arange(1, n+1) ** p)
        a = a / a.sum()
        candidates.append(a)

    # Beta shapes:
    for alpha in [0.5, 0.7, 1.0, 1.5]:
        for beta in [0.5, 0.7, 1.0, 1.5]:
            x = (np.arange(n) + 0.5) / n
            a = (x ** (alpha - 1)) * ((1 - x) ** (beta - 1))
            # replace infinities by large finite number
            a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
            a = a / a.sum()
            candidates.append(a)

    # random Dirichlet
    for _ in range(3):
        a = np.random.rand(n)
        a = proj_simplex(a)
        candidates.append(a)

    return candidates

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, sum=1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u > cssv / np.arange(1, n + 1))[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of real vectors using a real FFT (O(n log n))."""
    n = x.size + y.size - 1
    size = 1 << (n - 1).bit_length()  # next power of two
    X = np.fft.rfft(x, size)
    Y = np.fft.rfft(y, size)
    Z = X * Y
    conv = np.fft.irfft(Z, size)[:n]
    return conv.real


def _eval_score(a: np.ndarray) -> float:
    """Exact value of ``evaluate_sequence`` for a non‑negative simplex vector."""
    n = a.size
    b = _convolve_fft(a, a)
    max_b = b.max()
    # sum(a) == 1 by construction → denominator = 1
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Candidate generation
# ----------------------------------------------------------------------
def _initial_candidates(n: int) -> list:
    """Return a modest list of diverse simplex vectors of length n."""
    cand = []

    # uniform
    cand.append(np.full(n, 1.0 / n))

    # Catalan‑like (the “Catalan seed” used previously)
    cat = np.empty(n, dtype=float)
    cat[0] = 1.0
    for i in range(1, n):
        cat[i] = cat[i - 1] * (2 * (i - 1) + 1) / (2.0 * i)
    cat /= cat.sum()
    cand.append(cat)

    # geometric decays
    for r in np.linspace(0.95, 0.9999, 5):
        g = r ** np.arange(n, dtype=float)
        g /= g.sum()
        cand.append(g)

    # power‑law (p = exponent)
    for p in np.linspace(0.2, 2.0, 7):
        a = 1.0 / (np.arange(1, n + 1, dtype=float) ** p)
        a /= a.sum()
        cand.append(a)

    # Beta‑shaped families (α,β > 0).  Symmetric cases are enough for a start.
    for alpha in (0.5, 0.7, 1.0, 1.5):
        beta = alpha
        x = (np.arange(n) + 0.5) / n
        a = (x ** (alpha - 1)) * ((1.0 - x) ** (beta - 1))
        a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
        a /= a.sum()
        cand.append(a)

    # few random Dirichlet points
    for _ in range(3):
        rnd = np.random.rand(n)
        cand.append(_proj_simplex(rnd))

    # If a global ``height_sequence_1`` exists and has compatible length, use it
    try:
        global height_sequence_1
        hs = np.asarray(height_sequence_1, dtype=float)
        if hs.size == n:
            hs = np.maximum(hs, 0.0)
            hs /= hs.sum()
            cand.append(hs)
    except Exception:
        pass

    return cand


# ----------------------------------------------------------------------
# Optimisation routine (projected Adam + smooth‑max)
# ----------------------------------------------------------------------
def _optimise(
    n: int,
    a_start: np.ndarray,
    time_budget: float,
    seed: int = 0,
) -> tuple:
    """
    Return (best_vector, best_score) after running projected Adam for at most
    ``time_budget`` seconds on a problem of size ``n``.
    """
    np.random.seed(seed)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.2                # initial learning rate
    beta = 5.0              # soft‑max “temperature”
    max_beta = 5000.0

    a = a_start.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    cur_score = _eval_score(a)
    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    it = 0
    no_improve = 0

    while time.time() - start_time < time_budget - 0.05:
        # ---- convolution and smooth‑max weights ----
        b = _convolve_fft(a, a)
        max_b = b.max()

        # slowly increase the focus on the true maximum
        if it % 1000 == 0 and it > 0:
            beta = min(beta * 1.05, max_beta)

        # soft‑max on the convolution entries
        w = np.exp(beta * (b - max_b))          # max entry gets weight 1
        w_sum = w.sum()
        if w_sum == 0.0 or not np.isfinite(w_sum):
            w = np.ones_like(b) / b.size
        else:
            w /= w_sum

        # ---- gradient of the smooth max ----
        w_rev = w[::-1]
        conv_full = _convolve_fft(a, w_rev)    # length 2n‑1
        grad = 2.0 * conv_full[::-1][:n]        # reversed slice (size n)

        # ---- Adam update with projection onto the simplex ----
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_candidate = _proj_simplex(a - step)

        cand_score = _eval_score(a_candidate)

        if cand_score < cur_score:                     # accept improvement
            a = a_candidate
            cur_score = cand_score
            lr = min(lr * 1.05, 2.0)                  # gently increase step‑size
            no_improve = 0
            if cand_score < best_score:
                best_score = cand_score
                best_a = a_candidate.copy()
                print(f"[{it:07d}] new best {best_score:.6f}  lr={lr:.4f}  β={beta:.2f}")
        else:                                          # reject – shrink step‑size
            lr = max(lr * 0.7, 1e-5)
            no_improve += 1

        # ---- occasional random restart if stuck ----
        if no_improve > 4000:
            a = best_a + 0.01 * np.random.randn(n)
            a = np.maximum(a, 0.0)
            a = _proj_simplex(a)
            cur_score = _eval_score(a)
            m.fill(0.0)
            v.fill(0.0)
            it = 0
            lr = 0.2
            beta = 5.0
            no_improve = 0
            print(f"[restart] perturbation, score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a non‑negative height sequence that minimises
    ``evaluate_sequence``.  The routine runs for at most ``budget_s`` seconds
    and returns the best list of floats found.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0          # safety margin for I/O etc.

    # ------------------------------------------------------------------
    # Phase 1 – moderate size (fast iterations)
    # ------------------------------------------------------------------
    n_small = 8000
    time_small = max(0.0, (deadline - time.time()) * 0.55)

    cand_small = _initial_candidates(n_small)
    best_init = None
    best_init_score = np.inf
    for a in cand_small:
        sc = _eval_score(a)
        if sc < best_init_score:
            best_init_score = sc
            best_init = a.copy()
    print(f"[phase‑1 init] n={n_small}, best initial score={best_init_score:.6f}")

    best_small, best_small_score = _optimise(
        n=n_small,
        a_start=best_init,
        time_budget=time_small,
        seed=seed + 1,
    )

    # ------------------------------------------------------------------
    # Phase 2 – larger size (refinement)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:                        # need a non‑trivial amount of time
        n_big = 18000
        # upscale the best‑so‑far vector to the larger dimension
        x_old = np.linspace(0.0, 1.0, n_small)
        x_new = np.linspace(0.0, 1.0, n_big)
        a_big = np.interp(x_new, x_old, best_small)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        time_big = max(0.0, remaining * 0.95)
        print(f"[phase‑2] n={n_big}, time budget≈{time_big:.1f}s")

        best_big, best_big_score = _optimise(
            n=n_big,
            a_start=a_big,
            time_budget=time_big,
            seed=seed + 2,
        )
    else:
        best_big, best_big_score = None, np.inf

    # ------------------------------------------------------------------
    # Choose the overall best result
    # ------------------------------------------------------------------
    if best_big_score < best_small_score:
        final_seq = best_big
        final_score = best_big_score
    else:
        final_seq = best_small
        final_score = best_small_score

    print(f"[done] final best score = {final_score:.6f} (length={final_seq.size})")
    return final_seq.tolist()
