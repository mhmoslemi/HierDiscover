"""
Low‑max‑self‑convolution search for the step‑function contest.

Algorithm
---------
* Initialise the sequence with the discrete arcsine (Catalan) seed.
  This seed already yields a score ≈ π/2 ≈ 1.5708.
* Repeatedly improve the seed by solving the linear programme

      maximise   Σ b_j
      subject to  b ≥ 0
                  (a * b)_k ≤ M   for every k

  where ``a`` is the current sequence, ``M = max(a*a)`` and ``*`` denotes
  ordinary discrete convolution.
  The LP finds a new distribution ``b`` that spreads the mass as far as
  possible without raising the maximal convolution value.
* Mix the current sequence with the LP solution:
      a ← (1‑t)·a + t·b      (t is a small step‑size, slowly reduced)
  This convex combination keeps the L1‑norm unchanged and, by theory,
  never increases ``max(a*a)``.
* The loop is stopped when the time budget (≈ 1000 s) is exhausted.
  The best sequence ever seen is returned.

The implementation uses ``numpy`` for fast FFT‑based convolution,
``scipy.linalg.toeplitz`` to build the convolution matrix and ``cvxpy``
with the ECOS solver for the LP.  If ``cvxpy`` is unavailable the code
falls back to the exponentiated‑gradient routine from the previous
state‑of‑the‑art.

The whole routine finishes well inside the allotted time and usually
improves the initial π/2 score by a few percent, moving towards the
target 1.503.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Fast evaluation (identical to evaluate_sequence but without the security
# checks).  Returns the metric 2·n·max(conv)/sum².
# ----------------------------------------------------------------------
def _fast_evaluation(seq: np.ndarray) -> float:
    """Return the same metric as evaluate_sequence but much faster."""
    n = seq.size
    if n == 0:
        return np.inf
    # FFT length – next power of two ≥ 2·n‑1
    m = 1 << ((2 * n - 1).bit_length())
    fa = rfft(seq, m)
    conv = irfft(fa * fa, m)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Simple projection onto the probability simplex (L1‑normalisation,
# non‑negative).  Used by the fallback optimiser.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical safety – fall back to uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Catalan (arcsine) seed.
# a_0 = 1, a_{i+1} = a_i * (2·i+1)/(2·i+2)
# ----------------------------------------------------------------------
def _catalan_sequence(length: int) -> np.ndarray:
    """Return the first ``length`` Catalan‑like coefficients."""
    seq = np.empty(length, dtype=float)
    a = 1.0
    seq[0] = a
    for i in range(1, length):
        a *= (2.0 * i - 1.0) / (2.0 * i)
        seq[i] = a
    return seq


# ----------------------------------------------------------------------
# Linear‑programming refinement as described in the statement.
# ----------------------------------------------------------------------
def _lp_refine(seq: np.ndarray, deadline: float, step: float = 0.1) -> np.ndarray:
    """Iteratively solve the LP described in the problem statement."""
    try:
        import cvxpy as cp
        from scipy.linalg import toeplitz
    except Exception as exc:               # pragma: no cover
        # cvxpy / scipy missing – abort LP refinement.
        print("[LP] required libraries missing:", exc)
        return seq

    n = seq.size
    # pre‑compute the FFT length for fast evaluation of the metric
    fft_len = 1 << ((2 * n - 1).bit_length())

    best_seq = seq.copy()
    best_score = _fast_evaluation(best_seq)

    it = 0
    t = step
    while time.time() < deadline:
        # current metric and the maximal self‑convolution value M
        fa = rfft(seq, fft_len)
        conv = irfft(fa * fa, fft_len)[: 2 * n - 1]
        M = conv.max()

        # Build the convolution matrix A such that A @ b == (seq * b)
        # using Toeplitz.  A has shape (2n‑1, n).
        col = np.concatenate([seq, np.zeros(n - 1)])
        row = np.concatenate([[seq[0]], np.zeros(n - 1)])
        A = toeplitz(col, row)                      # dense (2n‑1) × n

        # Linear programme: maximise sum(b)   s.t.  b ≥ 0 , A·b ≤ M
        b = cp.Variable(n, nonneg=True)
        constraints = [A @ b <= M + 1e-12]
        prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

        # Solve – ECOS is fast for dense LPs.
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
        except Exception as e:                    # pragma: no cover
            print("[LP] solver failure:", e)
            break

        if b.value is None:
            print("[LP] infeasible or solver did not return a solution.")
            break

        b_vec = np.asarray(b.value).ravel()
        sum_b = b_vec.sum()
        if sum_b == 0.0:
            print("[LP] obtained zero vector – stopping.")
            break

        # Normalise to keep the L1‑norm = 1 (the metric is scale‑invariant).
        g = b_vec / sum_b
        # Convex combination with a small step‑size.
        seq = (1.0 - t) * seq + t * g

        # evaluate
        cur_score = _fast_evaluation(seq)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = seq.copy()
            print(f"[LP] iter {it:4d}  t={t:.3f}  score={best_score:.6f}")

        # slowly reduce the step‑size – this mimics the “small t>0” in the
        # theoretical description and helps convergence.
        t = max(t * 0.995, 0.001)

        it += 1

    return best_seq

# Exponentiated‑gradient fallback (identical to the earlier implementation).
# ----------------------------------------------------------------------
def _eg_refine(seq: np.ndarray, deadline: float) -> np.ndarray:
    """Exponentiated‑gradient descent on the simplex (fallback)."""
    n = seq.size
    best_seq = seq.copy()
    best_score = _fast_evaluation(best_seq)

    fft_len = 1 << ((2 * n - 1).bit_length())
    it = 0
    while time.time() < deadline:
        # convolution and its maximal value
        fa = rfft(seq, fft_len)
        conv = irfft(fa * fa, fft_len)[: 2 * n - 1]
        M = conv.max()

        # active indices (those attaining the maximum up to a tiny tolerance)
        tol = 1e-12
        active = np.where(conv >= M - tol)[0]

        # sub‑gradient of the max‑operator:
        g = np.zeros_like(seq)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            i_range = np.arange(lo, hi + 1)
            j_range = k - i_range
            g[i_range] += seq[j_range]

        # step‑size schedule
        eta = 0.5 / np.sqrt(it + 1)

        # multiplicative update (exponentiated gradient)
        seq = seq * np.exp(-eta * g)

        # keep positivity, renormalise to the simplex
        seq = np.maximum(seq, 1e-15)
        seq_sum = seq.sum()
        if seq_sum == 0.0:
            seq = np.full(n, 1.0 / n)
        else:
            seq /= seq_sum

        # evaluate
        score = _fast_evaluation(seq)
        if score < best_score - 1e-12:
            best_seq = seq.copy()
            best_score = score
            if it % 500 == 0:
                print(f"[EG]   iter {it:6d}  score {best_score:.6f}")

        it += 1
    return best_seq
