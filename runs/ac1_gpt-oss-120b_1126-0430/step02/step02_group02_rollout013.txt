    n = a.shape[0]
    lo = max_idx - (n-1)
    if lo < 0: lo = 0

def subgradient_max_convolution(a, max_idx):
    n = a.shape[0]
    lo = max(0, max_idx - (n-1))
    hi = min(max_idx, n-1)
    idx = np.arange(lo, hi+1)
    rev_idx = max_idx - idx
    g = np.zeros_like(a)
    g[idx] = 2.0 * a[rev_idx]
    return g

#!/usr/bin/env python
"""
Propose a step‑function (list of non‑negative heights) with a low evaluation score.
Algorithm
---------
1. Choose a length n (here 20000 – a good compromise between asymptotics and speed).
2. Build a diverse library of candidate shapes:
   * Power‑law a_i ∝ (i+1)^{-p}
   * Exponential decay a_i ∝ exp(-λ i)
   * Symmetric power‑law a_i ∝ ((i+1)(n-i))^{-p}
   * Gaussian bump, cosine bump and symmetric Beta PDFs.
   All candidates are normalised to sum to 1.
3. Evaluate each candidate cheaply with an FFT‑based convolution and keep the
   best few (default 12).
4. Refine every kept candidate by a projected Adam optimiser on the simplex.
   In each iteration the active index of the convolution (the one attaining the
   maximum) supplies a sub‑gradient:
       ∂(a*a)[k]/∂a_i = 2·a_{k-i} (when the index is valid).
   The iterate is projected onto the probability simplex after every update.
   Random jitter is occasionally injected to escape plateaus.
5. The best refined sequence is finally evaluated with the exact
   ``evaluate_sequence`` routine and returned.
The function respects the supplied time budget (default 1000 s).
"""

import time
import math

def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the probability simplex {x ≥ 0, Σx = 1}.
    """
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    # find rho
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all entries non‑positive
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 1.0
    a = i ** (-p)
    a /= a.sum()
    return a


def _exponential(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    a /= a.sum()
    return a


def _sym_power_law(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 1.0
    rev = n - np.arange(n, dtype=np.float64)
    a = (i * rev) ** (-p)
    a /= a.sum()
    return a


def _gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def _cosine_bump(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    arg = np.pi * (i - centre) / (n - 1)       # runs from -π to π
    a = np.cos(arg)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric (or asymmetric) Beta PDF on a discrete grid, normalised."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # protect against α<1 or β<1 at the boundaries
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.where(np.isfinite(a), a, 0.0)
    tot = a.sum()
    if tot == 0.0:
        a = np.ones(n, dtype=np.float64) / n
    else:
        a /= tot
    return a


def _conv_max_and_index(a: np.ndarray, fft_len: int):
    """Return (max_convolution_value, index_of_max, full convolution array)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.size - 1]
    maxb = float(conv.max())
    max_idx = int(conv.argmax())
    return maxb, max_idx, conv


def _subgradient_active(a: np.ndarray, active_idxs: np.ndarray):
    """
    Sub‑gradient of the (soft) max of the convolution.
    For each active index k we add 2·a_{k−i} to the gradient component i.
    The result is averaged over the active set.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    # limit size of the active set to keep the cost low
    if active_idxs.size > 20:
        active_idxs = np.random.choice(active_idxs, 20, replace=False)

    for k in active_idxs:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1)
        rev_idx = k - idx
        g[idx] += 2.0 * a[rev_idx]

    if active_idxs.size > 0:
        g /= float(active_idxs.size)
    return g


def _cheap_score(a: np.ndarray, n: int, fft_len: int) -> float:
    """
    Cheap surrogate of ``evaluate_sequence`` for a normalised vector
    (Σa = 1): 2·n·max((a*a)[k]).
    """
    maxb, _, _ = _conv_max_and_index(a, fft_len)
    return 2.0 * n * maxb

    candidate_pool = []   # will hold (array, cheap_score) pairs

    # power‑law
    for p in np.linspace(0.50, 0.90, 41):
        a = _power_law(n, p)
        score = _cheap_score(a, n, fft_len)
        candidate_pool.append((a, score))

    # exponential decay (λ = c/n)
    for c in np.linspace(0.5, 5.0, 19):
        lam = c / n
        a = _exponential(n, lam)
        score = _cheap_score(a, n, fft_len)
        candidate_pool.append((a, score))

    # symmetric power‑law
    for p in np.linspace(0.50, 0.90, 31):
        a = _sym_power_law(n, p)
        score = _cheap_score(a, n, fft_len)
        candidate_pool.append((a, score))

    # Gaussian bumps (σ = r·n)
    for r in np.linspace(0.05, 0.40, 8):
        sigma = r * n
        a = _gaussian(n, sigma)
        score = _cheap_score(a, n, fft_len)
        candidate_pool.append((a, score))

    # cosine bump
    a = _cosine_bump(n)
    candidate_pool.append((a, _cheap_score(a, n, fft_len)))

    # symmetric Beta PDFs
    for alpha in np.linspace(0.6, 3.0, 8):
        a = _beta_distribution(n, alpha, alpha)
        candidate_pool.append((a, _cheap_score(a, n, fft_len)))

    # a few asymmetric Beta combos (coarse grid)
    for alpha in (0.7, 1.5, 2.5):
        for beta in (0.7, 1.5, 2.5):
            a = _beta_distribution(n, alpha, beta)
            candidate_pool.append((a, _cheap_score(a, n, fft_len)))

    # ------------------------------------------------------------------
    # 3.    Keep the most promising candidates
    # ------------------------------------------------------------------
    top_k = 12
    # sort by cheap score (ascending)
    candidate_pool.sort(key=lambda tup: tup[1])
    top_candidates = [pair[0].copy() for pair in candidate_pool[:top_k]]
    print(f"Initial scan finished – best cheap score = {candidate_pool[0][1]:.6f}")

    # ------------------------------------------------------------------
    # 4.    Refine each top candidate with projected Adam
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    # Adam hyper‑parameters (simple, works well on the simplex after projection)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.5                     # base learning rate

    # maximum number of Adam steps per candidate (will stop earlier if out of time)
    max_iter_per = 8000

    for cand_idx, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break

        a = a0.copy()
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        best_local_a = a.copy()
        best_local_score = _cheap_score(a, n, fft_len)

        # Adam iteration counter
        t = 0
        while time.time() < deadline and t < max_iter_per:
            t += 1

            # --- convolution, max entry and active set ---------------------------------
            maxb, max_idx, conv = _conv_max_and_index(a, fft_len)

            # active set: all indices within a tiny tolerance of the maximum
            tol = 1e-12
            active_idxs = np.where(conv >= maxb - tol)[0]

            # --- sub‑gradient ---------------------------------------------------------
            g = _subgradient_active(a, active_idxs)

            # --- Adam update -----------------------------------------------------------
            m = beta1 * m + (1.0 - beta1) * g
            v = beta2 * v + (1.0 - beta2) * (g * g)

            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            step = lr0 / (np.sqrt(v_hat) + eps)
            a = _simplex_projection(a - step * m_hat)

            # --- periodic cheap evaluation ---------------------------------------------
            if t % 200 == 0 or time.time() > deadline - 5.0:
                cur_score = 2.0 * n * maxb          # cheap, exact surrogate
                if cur_score < best_local_score:
                    best_local_score = cur_score
                    best_local_a = a.copy()
                    print(f"[Cand {cand_idx:02d} it {t:05d}] new local best cheap = {cur_score:.6f}")

                # occasional random jitter to try to escape stagnation
                if np.random.rand() < 0.03:
                    a = _simplex_projection(a + 0.02 * np.random.rand(n))

        # end of candidate refinement
        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local_a.copy()
            print(f"*** Global best updated by candidate {cand_idx} → cheap score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.    If we still have time, perform a final fine‑tuning of the best sequence
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        a = best_seq.copy()
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        t = 0
        extra_iters = 20000
        while time.time() < deadline and t < extra_iters:
            t += 1
            maxb, max_idx, conv = _conv_max_and_index(a, fft_len)
            active_idxs = np.where(conv >= maxb - 1e-12)[0]
            g = _subgradient_active(a, active_idxs)

            m = beta1 * m + (1.0 - beta1) * g
            v = beta2 * v + (1.0 - beta2) * (g * g)

            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            step = lr0 / (np.sqrt(v_hat) + eps)
            a = _simplex_projection(a - step * m_hat)

            if t % 500 == 0:
                cur_score = 2.0 * n * maxb
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(f"[Final tune it {t:05d}] new global cheap best = {cur_score:.6f}")

    # ------------------------------------------------------------------
    # 6.    Final exact evaluation and return
    # ------------------------------------------------------------------
    if best_seq is None:                     # should never happen, but be safe
        best_seq = _power_law(n, 0.6)

    final_seq = best_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:                  # defensive – should never happen
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    return final_seq
