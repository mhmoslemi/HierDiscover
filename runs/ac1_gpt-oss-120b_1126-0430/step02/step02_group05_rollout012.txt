"""
Gradient‑descent on the maximal self‑convolution entry.

The objective to minimise is

    E(a) = 2·n·max(a*a) / (sum(a))² .

Scaling the whole vector does not change the value, therefore we keep the
ℓ₁‑norm fixed (sum = 1) for numerical stability.  At each iteration we

1. evaluate the self‑convolution by FFT,
2. locate a peak index k where the maximum is attained,
3. build the weight vector w_i = a_{k-i} (zero outside the array),
4. compute a descent direction d_i = max·a_i – sum·w_i.
   (d = –∇E up to a positive factor.)
5. take a step a ← a + η·d, clamp to non‑negative and renormalise the sum.
   The step size η is chosen so that the change is moderate and no entry
   becomes negative.
6. keep the best sequence seen.

In addition occasional random “shakes’’ are performed to escape plateaux.
The loop runs until the time budget (≈ 1000 s) is exhausted and the
best sequence is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast version of the provided evaluate_sequence (no security checks).
# ----------------------------------------------------------------------
def _fast_evaluation(seq: np.ndarray) -> float:
    """Return the same score as evaluate_sequence for a 1‑D float array."""
    n = seq.size
    if n == 0:
        return float("inf")
    # FFT length = next power of two ≥ 2*n‑1
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Catalan (arcsine) seed – a classic start with flat self‑convolution.
# ----------------------------------------------------------------------
def _catalan_sequence(length: int) -> np.ndarray:
    """Return the first `length` terms a_i = binom(2i,i)/4**i."""
    seq = np.empty(length, dtype=float)
    a = 1.0
    seq[0] = a
    for i in range(1, length):
        a *= (2.0 * i - 1.0) / (2.0 * i)      # a_i = a_{i-1}·(2i‑1)/(2i)
        seq[i] = a
    return seq


# ----------------------------------------------------------------------
# Helper – weight vector w_i = a_{k-i} (zero where the index is out of range).
# ----------------------------------------------------------------------
def _weight_for_index(seq: np.ndarray, k: int) -> np.ndarray:
    n = seq.size
    w = np.zeros_like(seq)
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    if lo <= hi:
        i = np.arange(lo, hi + 1)
        w[lo:hi + 1] = seq[k - i]
    return w


# ----------------------------------------------------------------------
# Core optimiser – gradient descent with occasional random perturbations.
# ----------------------------------------------------------------------
def _optimise(seq: np.ndarray, deadline: float) -> np.ndarray:
    """
    Perform a constrained gradient descent on the score.
    The sequence is kept non‑negative and renormalised to sum = 1 after each step.
    """
    n = seq.size
    target_sum = 1.0                     # keep the ℓ₁ norm fixed for stability
    seq = seq / seq.sum() * target_sum   # normalise initial seed

    best_seq = seq.copy()
    best_score = _fast_evaluation(best_seq)

    it = 0
    # parameters for the adaptive step size
    max_step = 0.5                       # largest admissible relative step
    min_step = 0.001                     # after many iterations we still move a little

    # pre‑compute FFT length (does not change)
    fft_len = 1 << ((2 * n - 1).bit_length())

    start = time.time()
    while time.time() < deadline:
        # ---- compute self‑convolution (FFT) ---------------------------------
        fa = np.fft.rfft(seq, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_val = conv.max()
        # pick a random peak among the (almost) equal maxima
        cand = np.flatnonzero(conv >= max_val - 1e-12)
        k = int(np.random.choice(cand))

        # ---- weight vector for the chosen peak -------------------------------
        w = _weight_for_index(seq, k)

        # ---- descent direction d = max·a − sum·w  (proportional to –∇E) -----
        s = seq.sum()                     # should be ≈ target_sum
        d = max_val * seq - s * w

        # ---- adaptive step size ------------------------------------------------
        # we restrict the step so that the update does not create negatives.
        # The largest admissible η that keeps every component non‑negative:
        #    η ≤ min_i ( a_i / -d_i )  over d_i < 0
        neg = d < 0
        if np.any(neg):
            eta_max = (seq[neg] / (-d[neg])).min()
        else:
            eta_max = np.inf
        # linearly decay the allowed step as the deadline approaches
        remaining = deadline - time.time()
        frac = remaining / (deadline - start)
        step_limit = min_step + (max_step - min_step) * frac
        eta = min(step_limit, eta_max * 0.999)   # a tiny safety margin

        # ---- propose the new sequence -----------------------------------------
        new_seq = seq + eta * d
        # numerical safety – clip tiny negatives to zero
        new_seq = np.maximum(new_seq, 0.0)

        # renormalise the ℓ₁ norm (scale invariance guarantees the score is unchanged)
        new_seq_sum = new_seq.sum()
        if new_seq_sum == 0.0:            # should never happen, but guard against it
            new_seq = seq.copy()
        else:
            new_seq *= (target_sum / new_seq_sum)

        # ---- evaluate -----------------------------------------------------------
        new_score = _fast_evaluation(new_seq)

        # ---- accept if we improve ------------------------------------------------
        if new_score < best_score - 1e-12:
            seq = new_seq
            best_seq = new_seq.copy()
            best_score = new_score
            print(f"Iter {it:6d}: improved score = {best_score:.8f}, max = {max_val:.8f}")
        else:
            # occasional random shake‑up to escape flat regions
            if np.random.rand() < 0.01:
                i, j = np.random.randint(n, size=2)
                if i != j and seq[i] > 0:
                    eps = min(seq[i], np.random.rand() * seq[i] * 0.2)
                    tmp = seq.copy()
                    tmp[i] -= eps
                    tmp[j] += eps
                    tmp = np.maximum(tmp, 0.0)
                    tmp *= (target_sum / tmp.sum())
                    tmp_score = _fast_evaluation(tmp)
                    if tmp_score < best_score:
                        seq = tmp
                        best_seq = tmp.copy()
                        best_score = tmp_score
                        print(f"Iter {it:6d}: random shake improved score = {best_score:.8f}")
        it += 1

    return best_seq


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights whose evaluation score is
    (hopefully) well below the previous π/2 ≈ 1.571 bound.
    """
    np.random.seed(seed)
    # keep a small safety margin for the harness
    deadline = time.time() + budget_s - 5.0

    # length of the step‑function – a few thousand is a good trade‑off
    n = 8000

    # 1. start from the Catalan (arcsine) seed
    seq = _catalan_sequence(n)

    # 2. aggressive gradient descent on the maximal convolution entry
    seq = _optimise(seq, deadline)

    # 3. return the best sequence discovered (plain Python list)
    return seq.tolist()
