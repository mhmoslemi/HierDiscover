    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.1

    # Determine n:

        a = np.full(n, 1/n)

    L = 1 << ((3*n-2).bit_length())

    # Adam parameters
    b1=0.9; b2=0.999; eps=1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # schedule parameters:
    total_iters = 500000   # maybe limited by time
    iter = 0
    best_seq = a.copy(); best_score = _objective_simplex(a,n,L)
    # Phase 1: smoothed gradient descent with Adam
    while time.time() < deadline and iter < total_iters:
        # compute conv and smooth gradient
        conv = _conv_fft(a, L)
        # compute alpha
        frac = iter / total_iters
        alpha = 5 + (2000-5) * frac**0.5   # maybe sqrt schedule

        # Adam update (grad is Euclidean gradient)
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad**2)
        m_hat = m / (1 - b1**(iter+1))

        step = 0.2 / (np.sqrt(v_hat) + eps)   # learning rate scaling
        cand = a - step * m_hat
        cand = _project_to_simplex(cand)
        # evaluate objective
        conv_cand = _conv_fft(cand, L)
        cand_score = 2*n*conv_cand.max()
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"[{time.time():.1f}] Phase1 iter {iter} new best O={best_score:.6f}")

        # Accept candidate anyway for next iter
        a = cand
        iter += 1

    # Phase 2: Subgradient + random moves (like earlier)
    temperature0 = 0.01
    sub_interval = 5
    while time.time() < deadline:
        # start from best_seq or random perturbation
        a = best_seq.copy()
        # optionally jitter
        a += np.random.randn(n) * 0.001
        a = np.clip(a,0,None)
        a = _project_to_simplex(a)
        conv = _conv_fft(a, L)
        cur_obj = 2*n*conv.max()
        local_iter = 0
        no_improve = 0
        T = temperature0
        while time.time() < deadline and local_iter < 200_000:
            # mass transfer
            i,j = np.random.randint(n, size=2)
            if i!=j and a[i]>0:
                delta = a[i] * np.random.rand() * 0.2
                a[i] -= delta; a[j] += delta
                a = _project_to_simplex(a)
                conv = _conv_fft(a, L)
                new_obj = 2*n*conv.max()
                if new_obj < cur_obj or np.random.rand() < math.exp((cur_obj - new_obj)/max(T,1e-12)):
                    cur_obj = new_obj
                    if cur_obj < best_score:
                        best_score = cur_obj
                        best_seq = a.copy()
                        print(f"[{time.time():.1f}] Phase2 new global best O={best_score:.6f}")
                    no_improve=0
                else:
                    no_improve+=1
            else:
                no_improve+=1

            if local_iter % sub_interval == 0:
                g = _subgradient_max_conv(a, conv)
                eta = 0.05 * (1 - local_iter / 200_000)
                a = a - eta * g
                a = _project_to_simplex(a)

                    best_seq = a.copy()

            if no_improve >= 10000:

"""
Propose a non‑negative step‑function sequence that (approximately) minimises

    O(a) = 2·n·max(conv(a,a))    with   sum(a) = 1

The algorithm consists of three stages:

1. **Smoothed gradient descent (Adam)**
   – Optimise the log‑sum‑exp surrogate
       f_α(a) = (2·n/α)·log(∑_k e^{α·conv_k})
   – The gradient is obtained by a single FFT‑based correlation.
   – Adam’s adaptive step sizes and a simple global scaling keep the update
     inside the simplex.  The smoothing parameter α is linearly annealed from
     a small to a large value, gradually turning the surrogate into the true
     max.

2. **Exact sub‑gradient descent with exponentiated updates**
   – The sub‑gradient of the true objective is the average of the gradients
     of all active (maximal) convolution entries.
   – An exponentiated‑gradient (mirror descent w.r.t. KL) update
       a_i ← a_i·exp(‑η·g_i)  ,  a ← a/∑a
     stays on the simplex without an explicit projection.

3. **Stochastic local refinement**
   – Repeated “mass‑transfer’’ moves: move a random fraction of probability
     mass from one index to another.
   – Accept improvements always; otherwise accept with a Metropolis‑type
     probability that decays with a temperature schedule.
   – Occasionally apply a cheap sub‑gradient step to keep the peak down.

The whole routine respects the supplied time budget (default 1000 s) and
returns the best sequence seen as a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex {x ≥ 0, sum(x)=1}
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex using the algorithm of
    Wang & Carreira‑Perpiñán (2013)."""
    n = v.shape[0]
    if n == 0:
        return v
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        # all values are non‑positive → return uniform distribution
        w = np.full_like(v, 1.0 / n)
        return w
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # renormalise to avoid tiny numerical drift
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(v, 1.0 / n)
    else:
        w /= w_sum
    return w


# ----------------------------------------------------------------------
# FFT‑based full self‑convolution (length 2·n‑1)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return a * a (full convolution) using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    # keep only the first 2·n‑1 entries
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp gradient (see the docstring above)
def _smooth_gradient(a: np.ndarray, conv: np.ndarray, alpha: float, L: int) -> np.ndarray:
    """
    Compute the gradient of  (2·n/α)·log∑_k exp(α·conv_k)
    with respect to a.  The result is   2·corr(w, a)   where
    w_k ∝ exp(α·conv_k).
    """
    # soft‑max weights
    scaled = alpha * conv
    # subtract max for numerical stability
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / exp_scaled.sum()

    # correlation w ⋆ a (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n = a.size
    # central n entries correspond to the true correlation
    grad = 2.0 * corr_rev[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) (exact, not smoothed)
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Return a sub‑gradient of   max_k conv_k   w.r.t. a."""
    n = a.size
    max_val = conv.max()
    # all indices attaining the maximum are active
    ks = np.flatnonzero(conv == max_val)
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        # each pair (i,j) contributes 2·a_j to ∂conv_k/∂a_i
        g[i_idx] += 2.0 * a[j_idx]
    g /= max(1, ks.size)  # average over all active constraints
    return g

                a = a - eta * g
                a = _project_to_simplex(a)
                conv = _conv_fft(a, L)
                cur_obj = 2.0 * n * conv.max()
                if cur_obj < best_score:
                    best_score = cur_obj
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] Sub‑grad step improved O={best_score:.6f}")

            # temperature decay
            T = temperature0 * (1.0 - it_local / 150_000)

            # give up on this local run if we have been stagnant for a while
            if stagnation >= no_improve_limit:
                break

            it_local += 1

    # ------------------------------------------------------------------
    # Final sanity check with the provided evaluate_sequence()
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence failed:", exc)

    return best_seq.tolist()
