"""
Gradient‑based refinement for minimizing   2·n·max(conv(a,a))/(sum(a)^2).

The algorithm:
1. Sample many random candidates (Dirichlet‑like, power‐law, exponential,
   mixtures) and keep the best one.
2. Starting from that best candidate run a projected gradient descent on the
   simplex.  The max‑convolution is smoothed with a soft‑max (temperature τ)
   to obtain a differentiable surrogate; the gradient is computed analytically
   and the update is projected back onto the probability simplex.
3. A simple schedule lowers the step size while raising τ, and occasional
   random jitter helps escape plateaus.  The best sequence found within the
   time budget is returned.
"""

import time
import numpy as np

def _project_to_simplex(v):
    """Project a vector onto the probability simplex {x≥0, sum(x)=1}."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback uniform
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


def _smooth_weights(conv, tau):
    """Return probability weights proportional to exp(tau*conv) (soft‑max)."""
    # shift by max for numerical stability
    max_c = np.max(conv)
    expw = np.exp(tau * (conv - max_c))
    total = np.sum(expw)
    if total == 0:
        # should not happen, fallback to uniform
        return np.full_like(conv, 1.0 / conv.shape[0])
    return expw / total


def _gradient(x, conv, tau):
    """
    Compute the gradient of the smooth max objective
    fτ(x)= (1/τ) log Σ_k exp(τ·conv_k(x)).
    Returns a vector of length len(x).
    """
    n = x.shape[0]
    p = _smooth_weights(conv, tau)            # length 2n-1
    # Convolution with reversed p gives g_i = 2 Σ_j x_j p_{i+j}
    p_rev = p[::-1]
    full = np.convolve(x, p_rev)              # length 3n-2
    idx = 2 * n - 2 - np.arange(n)            # indices for i = 0..n-1
    g = 2.0 * full[idx]
    return g

            cur_score = cand_score
            stagnation = 0
            if cand_score < best_score - 1e-12:
                best_score = cand_score
                best_seq = candidate.tolist()
                print(f"[grad] iter {it:6d} new best {best_score:.6f}")
        else:
            stagnation += 1
            # occasional jitter to escape plateaus
            if stagnation % 4000 == 0:
                jitter = rng.uniform(0.0, 0.002, size=n)
                cur = _project_to_simplex(cur + jitter)
                cur_score = evaluate_sequence(cur.tolist())
                stagnation = 0

        # periodic clean‑up (eliminate numerical drift)
        if it % 2500 == 0:
            cur = _project_to_simplex(cur)

    print(f"[final] best score found = {best_score:.6f}")
    return best_seq
