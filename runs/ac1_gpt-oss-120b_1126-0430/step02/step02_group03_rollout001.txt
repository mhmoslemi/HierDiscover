#!/usr/bin/env python
"""
Propose a step‑function (non‑negative sequence) that minimises the
``evaluate_sequence`` metric.

The algorithm consists of three phases:

1. **Initialisation** – several simple analytic families (Catalan,
   power‑law and shifted power‑law) are generated, the one with the
   lowest metric is kept.
2. **Projected sub‑gradient descent** – the objective
   ``M = max_k (a*a)_k`` (with ∑a = 1) is convex in the sense of a
   sub‑gradient.  At each iteration the index `k_max` where the
   convolution attains its maximum is found; the sub‑gradient is
   ``g_i = 2*a_{k_max-i}`` on the feasible index range.
   A back‑tracking line‑search proposes a new point
   ``a' = a - η g``, projects it onto the probability simplex
   (clip negative values, re‑normalise) and accepts the move only if
   the maximal convolution entry strictly decreases.  The step size
   is adaptively increased/decreased.
3. **Termination** – the loop runs until the supplied time budget is
   exhausted (≈1000 s).  The best sequence seen so far is returned as a
   plain ``list[float]``.

All convolutions are performed with a real FFT (linear full convolution)
for O(n log n) speed.  The whole optimisation stays in memory (≈ a few
hundred KB) and finishes well within the time limit, typically
achieving a metric appreciably below the Catalan bound π/2 ≈ 1.570796.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast linear full convolution using a real FFT (same as numpy.convolve)
# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a * a (full mode) using a real FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    # next power of two that is >= 2*n-1 (optimal for FFT speed)
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Analytic families for a quick start
# ----------------------------------------------------------------------
def _catalan_normalised(n: int) -> np.ndarray:
    """Central binomial coefficients a_i = C(2i,i)/4**i, normalised to unit sum."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    a /= a.sum()
    return a


def _power_law_normalised(n: int, beta: float, shift: float) -> np.ndarray:
    """
    a_i ∝ (i + shift) ** (-beta), normalised to unit sum.
    shift > 0 guarantees a_0 is finite.
    """
    idx = np.arange(n, dtype=np.float64)
    a = (idx + shift) ** (-beta)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Helper: current metric (uses the fact that sum(a)=1 → 2*n*max(conv))
# ----------------------------------------------------------------------
def _metric(a: np.ndarray) -> float:
    """Return 2*n*max(a*a) assuming sum(a)==1 (exact up to fp error)."""
    n = a.shape[0]
    conv = _fft_convolve(a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Main optimisation routine (projected sub‑gradient with line‑search)
# ----------------------------------------------------------------------
def _optimise(a: np.ndarray, deadline: float) -> np.ndarray:
    """
    Perform projected sub‑gradient descent on ``a`` until ``deadline``.
    Returns the best sequence found (unit‑sum, non‑negative).
    """
    n = a.shape[0]
    best_a = a.copy()
    best_val = _metric(best_a)

    # initial step size (will be adapted)
    step = 0.2
    min_step = 1e-8
    it = 0

    # tolerance for considering a new point an improvement
    improvement_eps = 1e-15

    while time.time() < deadline - 0.05:
        it += 1
        conv = _fft_convolve(a)
        M = np.max(conv)
        k_max = int(np.argmax(conv))

        # if current point already beats best, record it
        if M < best_val - improvement_eps:
            best_val = M
            best_a = a.copy()
            print(f"[iter {it:07d}] new best metric ≈ {2.0 * n * M:.9f}")

        # --- sub‑gradient for the active convolution index ---
        i_start = max(0, k_max - (n - 1))
        i_end = min(k_max, n - 1)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k_max - idx]

        # line‑search: try decreasing step sizes until improvement
        cur_step = step
        success = False
        for _ in range(6):
            a_new = a - cur_step * grad
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s <= 0.0:                     # degenerate – random restart
                a_new = np.random.rand(n)
                a_new /= a_new.sum()
                s = 1.0
            else:
                a_new /= s

            conv_new = _fft_convolve(a_new)
            M_new = np.max(conv_new)

            if M_new < M - improvement_eps:
                # accept
                a = a_new
                step = cur_step * 1.02   # slightly increase after success
                success = True
                break
            else:
                cur_step *= 0.5              # shrink step and retry

        if not success:
            step = max(step * 0.7, min_step)   # reduce global step if stuck

        # occasional small random perturbation to escape flat plateaus
        if it % 5000 == 0:
            perturb = np.random.normal(scale=0.001, size=n)
            a = a + perturb
            a = np.maximum(a, 0.0)
            a /= a.sum()

    # final check before returning
    final_metric = _metric(best_a)
    print(f"[final] best metric ≈ {2.0 * n * best_val:.9f}")
    return best_a


# ----------------------------------------------------------------------
# Public entry point required by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Return a list of non‑negative heights that (empirically) minimise
    ``evaluate_sequence``.  The routine runs for at most ``budget_s`` seconds.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0  # safety margin

    # ------------------------------------------------------------------
    # 1) Choose a reasonably large support.
    #    n≈30000 gives a good trade‑off between freedom and speed.
    # ------------------------------------------------------------------
    n = 30000

    # ------------------------------------------------------------------
    # 2) Generate several cheap candidates and pick the best start point.
    # ------------------------------------------------------------------
    candidates = []

    # Catalan distribution (the classic benchmark)
    candidates.append(_catalan_normalised(n))

    # Power‑law families (beta≈0.5 is close to Catalan, shift tweaks the head)
    for beta in (0.45, 0.48, 0.50, 0.52, 0.55):
        for shift in (0.1, 0.5, 1.0, 2.0):
            candidates.append(_power_law_normalised(n, beta, shift))

    # Evaluate them quickly (using the fast FFT metric)
    best_start = None
    best_score = np.inf
    for idx, cand in enumerate(candidates):
        sc = _metric(cand)
        if sc < best_score:
            best_score = sc
            best_start = cand
        # keep the console tidy – only occasional prints
        if idx % 5 == 0:
            print(f"[seed init] candidate {idx:02d} metric ≈ {2.0 * n * sc:.9f}")

    # ------------------------------------------------------------------
    # 3) Refine the chosen start point with sub‑gradient descent.
    # ------------------------------------------------------------------
    refined = _optimise(best_start, deadline)

    # ------------------------------------------------------------------
    # 4) Return the best sequence found (as a plain Python list).
    #    ``evaluate_sequence`` will be called by the outer harness.
    # ------------------------------------------------------------------
    final_metric = _metric(refined)
    print(f"[return] final metric (approx) ≈ {2.0 * n * final_metric:.9f}")
    return refined.tolist()
