def update_conv(conv, a, i, j, delta):
    # conv is array length 2n-1 (float)
    # a is current array before move (maybe we need original a for shift). We will build shift_i and shift_j via slicing.
    n = a.shape[0]
    # compute shift_i (length 2n-1) where shift_i[k] = a[k-i] if 0 <= k-i < n else 0
    # This can be done by constructing a zero array and fill via slices.
    # Using vectorized slicing: shift_i = np.zeros_like(conv); shift_i[i:i+n] = a
    # Actually for index offset: for k = i..i+n-1, shift_i[k] = a[k-i]
    # So shift_i[i:i+n] = a
    # Similarly shift_j.
    shift_i = np.zeros_like(conv)
    shift_i[i:i+n] = a
    shift_j = np.zeros_like(conv)
    shift_j[j:j+n] = a
    # apply updates:
    conv_new = conv - 2*delta*shift_i + 2*delta*shift_j
    # Add diagonal and cross terms:
    conv_new[2*i] += delta*delta   # e_i ⊛ e_i term
    conv_new[2*j] += delta*delta   # e_j ⊛ e_j term
    conv_new[i+j] -= 2*delta*delta # e_i ⊛ e_j term (twice)
    return conv_new

"""
Heuristic optimizer for minimizing the evaluation function.
Uses fast incremental convolution updates and
simulated annealing with subgradient descents.
"""

import time
import math
import numpy as np

def _full_conv_fft(a, L):
    """Return full linear convolution a*a using FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]

def _subgradient_max_conv(a, conv):
    """Subgradient of max_{k} (a*a)_k w.r.t a (average if multiple maxima)."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        jdx = k - idx
        g[idx] += 2.0 * a[jdx]
    g /= max(1, len(ks))
    return g

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):

    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    if 'height_sequence_1' in globals():
        a = np.array(height_sequence_1, dtype=float)
        n = a.size
        a_sum = a.sum()

            a = np.full(n, 1.0 / n)
        else:
            a = a / a_sum          # normalise to sum 1

    else:
        n = 5000
        a = np.full(n, 1.0 / n)

    # FFT length sufficient for convolutions and later calculations
    L = 1 << ((3 * n - 2).bit_length())   # next power of two >= 3n-2

    # initial convolution and objective
    conv = _full_conv_fft(a, L)
    cur_obj = 2.0 * n * conv.max()
    best_seq = a.copy()
    best_obj = cur_obj
    print(f"[start] O={cur_obj:.6f}")

    # parameters for simulated annealing
    temperature0 = 0.03
    # thresholds
    no_improve = 0
    no_improve_limit = 200_000   # after this many iterations without a new global best we re‑jitter

    it = 0
    while time.time() < deadline:
        # remaining time fraction
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        # decay for temperature and max transfer fraction
        frac = remaining / budget_s
        T = temperature0 * frac
        max_transfer_frac = 0.001 + 0.199 * frac   # linearly shrink from 0.2 to 0.001

        # pick distinct indices
        i, j = np.random.randint(n, size=2)

            # nothing to move

        if delta <= 0.0:

        conv_new = conv.copy()

        conv_new[i:i+n] -= 2.0 * delta * a
        # add contribution of added mass
        conv_new[j:j+n] += 2.0 * delta * a
        # quadratic (delta^2) corrections

"""
Heuristic optimizer for the step‑function evaluation problem.

The objective (after normalising the sum to 1) is
    O(a) = 2·n·max_k (a * a)_k .
We minimise O by a combination of
* a fast incremental update of the self‑convolution when a small amount of
  mass is moved from one position to another,
* a simulated‑annealing style Metropolis acceptance rule,
* periodic mirror‑descent (exponentiated‑gradient) steps using the exact
  sub‑gradient of the maximal convolution entry,
* occasional re‑jittering when the search stagnates.

All heavy work is done with NumPy (FFT for the occasional full recompute,
vectorised slice updates for the cheap moves).  The routine respects the
1000 s budget, prints progress, and finally returns the best sequence found
as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: full linear convolution using FFT (length 2·n‑1)
def _full_convolution(a: np.ndarray, L: int) -> np.ndarray:
    """Return a*a (full linear convolution) using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]

# ----------------------------------------------------------------------
# Sub‑gradient of max_k (a*a)_k  (averaged when several k are maximal)
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Compute a sub‑gradient of max(conv) w.r.t. a."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        jdx = k - idx
        g[idx] += 2.0 * a[jdx]
    g /= max(1, len(ks))
    return g

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function sequence that minimises the
    evaluation function.  Returns the sequence as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # a small safety margin

    # --------------------------------------------------------------
    # Initialise a – use the provided construction if available,
    # otherwise fall back to a uniform distribution.
    if 'height_sequence_1' in globals():
        a = np.array(height_sequence_1, dtype=float)
        n = a.size
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / total                     # normalise to sum = 1
        print(f"[init] using supplied sequence, n={n}")
    else:
        n = 5000
        a = np.full(n, 1.0 / n)
        print(f"[init] no supplied sequence – using uniform, n={n}")

    # --------------------------------------------------------------
    # Pre‑compute the FFT length – must be at least 3·n‑2 for the
    # correlations used later.
    L = 1 << ((3 * n - 2).bit_length())       # next power of two

    # Initial convolution and objective
    conv = _full_convolution(a, L)
    cur_obj = 2.0 * n * conv.max()
    best_seq = a.copy()
    best_obj = cur_obj
    print(f"[start] O={cur_obj:.6f}")

    # --------------------------------------------------------------
    # Parameters for the annealing / sub‑gradient schedule
    temperature0 = 0.03           # initial Metropolis temperature
    no_improve = 0
    no_improve_limit = 200_000   # after this many rejected moves we jitter
    iteration = 0

    # initial sub‑gradient – also yields a source (high‑impact) and a sink
    g = _subgradient_max_conv(a, conv)
    i_max = int(np.argmax(g))
    i_min = int(np.argmin(g))

    # --------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()
        if remaining <= 0.0:
            break

        # Linear cooling of temperature and of the maximal transfer fraction
        frac_time = remaining / budget_s
        T = temperature0 * frac_time
        max_transfer_frac = 0.001 + 0.199 * frac_time   # from 0.2 → 0.001

        # ----------------------------------------------------------
        # Propose a mass‑transfer move.
        # With probability 0.6 we use the gradient‑guided source/target,
        # otherwise we pick two indices uniformly at random.
        if np.random.rand() < 0.6:
            i = i_max
            j = i_min
            if i == j:
                i = np.random.randint(n)
                j = np.random.randint(n)
                while j == i:
                    j = np.random.randint(n)
        else:
            i = np.random.randint(n)
            j = np.random.randint(n)
            while j == i:
                j = np.random.randint(n)

        if a[i] <= 0.0:                     # nothing to take from i
            iteration += 1
            continue

        # Amount of probability mass moved from i to j
        delta = a[i] * max_transfer_frac * np.random.rand()
        if delta <= 0.0:
            iteration += 1
            continue

        # ----------------------------------------------------------
        # Fast incremental update of the convolution.
        #   conv' = conv - 2·δ·(e_i * a) + 2·δ·(e_j * a)
        #          + δ²·δ_{2i} + δ²·δ_{2j} - 2·δ²·δ_{i+j}
        conv_new = conv.copy()
        conv_new[i:i + n] -= 2.0 * delta * a          # -2·δ·shift_i
        conv_new[j:j + n] += 2.0 * delta * a          # +2·δ·shift_j
        conv_new[2 * i] += delta * delta
        conv_new[2 * j] += delta * delta
        conv_new[i + j] -= 2.0 * delta * delta

        new_max = conv_new.max()
        new_obj = 2.0 * n * new_max

        # ----------------------------------------------------------
        # Metropolis acceptance
        accept = False
        if new_obj < cur_obj:
            accept = True
        else:
            prob = math.exp(-(new_obj - cur_obj) / max(T, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            # commit the move
            a[i] -= delta
            a[j] += delta
            conv = conv_new
            cur_obj = new_obj
            no_improve = 0
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_seq = a.copy()
                print(f"[{time.time():.1f}] New global best O={best_obj:.6f} it={iteration}")
        else:
            no_improve += 1

        # ----------------------------------------------------------
        # Periodic exact sub‑gradient descent (mirror / exponentiated step)
        if iteration % 5000 == 0:
            # sub‑gradient for the current maximal convolution entry
            g = _subgradient_max_conv(a, conv)
            # step size shrinks as time goes on
            eta = 0.07 * frac_time
            # exponentiated‑gradient update (keeps positivity automatically)
            a = a * np.exp(-eta * g)
            s = a.sum()
            if s == 0.0:
                a = np.full_like(a, 1.0 / n)
            else:
                a = a / s
            # recompute the whole convolution (FFT is cheap relative to n ≈ 5 k)
            conv = _full_convolution(a, L)
            cur_obj = 2.0 * n * conv.max()
            # refresh the source / sink indices for the next mass‑transfer phase
            i_max = int(np.argmax(g))
            i_min = int(np.argmin(g))
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Sub‑grad improvement O={best_obj:.6f}")

        # ----------------------------------------------------------
        # If we have been stuck for a long time, jitter the current best
        # and continue from there.
        if no_improve >= no_improve_limit:
            jitter = np.random.randn(n) * 0.005
            a = best_seq + jitter
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            conv = _full_convolution(a, L)
            cur_obj = 2.0 * n * conv.max()
            # recompute a fresh sub‑gradient direction
            g = _subgradient_max_conv(a, conv)
            i_max = int(np.argmax(g))
            i_min = int(np.argmin(g))
            no_improve = 0
            print(f"[{time.time():.1f}] Re‑jitter after stagnation, O={cur_obj:.6f}")

        iteration += 1

    # ------------------------------------------------------------------
    # Final sanity check with the supplied evaluate_sequence()
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:               # pylint: disable=broad-except
        print("[final] evaluate_sequence failed:", exc)

    return best_seq.tolist()
