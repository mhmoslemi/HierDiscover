def _project_simplex(v):
    # ensure sum=1, v_i >=0
    # using sorting
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

assistantfinal**Solution Overview**

The score to minimise is  

\[
F(a)=\frac{2n\max (a*a)}{(\sum a_i)^2},
\qquad a_i\ge 0 .
\]

Since the expression is scale‑invariant we may enforce \(\sum a_i=1\) and work on the
probability simplex.  
The code below builds a strong initial guess (a family of symmetric
“arcsine‑like’’ sequences), picks the best among a few parameter choices and
then refines it with a **projected Adam optimiser** that works on a smooth
approximation of the $\max$‑operator:

*   \(f_\lambda(a)=\frac{1}{\lambda}\log\!\bigl(\sum_k e^{\lambda\,(a*a)_k}\bigr)\)   
    → when \(\lambda\) is large  \(f_\lambda(a)\approx\max(a*a)\).

*   The gradient of \(f_\lambda\) can be written only with convolutions,
    therefore each iteration costs only a few FFTs (\(O(n\log n)\)).

*   After every Adam step the vector is projected onto the simplex
    (non‑negative, sum = 1).

*   A short **greedy mass‑transfer** phase is run at the very end to
    polish any remaining plateau.

The whole routine respects the 1000 s budget, prints intermediate scores and
finally returns the best sequence found.

```python
"""
Projected‑Adam optimiser for the self‑convolution L∞‑objective.

* Initialise with a few arcsine‑like families (a_i ∝ ((i+ε)(n‑1‑i+ε))^α)
  and keep the best shape.
* Refine the shape with Adam on a soft‑max approximation of the maximal
  convolution entry.  The gradient is computed by two FFT‑based convolutions,
  then the iterate is projected onto the probability simplex.
* Finish with a tiny greedy mass‑transfer to escape possible plateaus.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Fast evaluation (identical to the imported `evaluate_sequence`
# but without the security checks).  Used only for logging.
# ----------------------------------------------------------------------
def _fast_evaluation(seq: np.ndarray) -> float:
    """Return the metric used by the competition (2·n·max(conv)/sum²)."""
    n = seq.size
    if n == 0:
        return float("inf")
    # FFT length = next power of two ≥ 2·n‑1
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    max_b = conv.max()
    s = seq.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Catalan (arcsine) seed – useful as a fallback if everything else fails.
# ----------------------------------------------------------------------
def _catalan_sequence(length: int) -> np.ndarray:
    """First `length` coefficients of (1‑z)^‑1/2."""
    seq = np.empty(length, dtype=float)
    a = 1.0
    seq[0] = a
    for i in range(1, length):
        a *= (2 * i - 1) / (2.0 * i)
        seq[i] = a
    return seq


# ----------------------------------------------------------------------
# Simple projection onto the probability simplex (Duchi et al. 2008).
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector `v` onto {x ≥ 0, sum(x)=1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Small greedy refinement – moves a fraction of mass from the index that
# contributes most to the current maximal convolution entry to an index
# that contributes almost nothing.
# ----------------------------------------------------------------------
def _greedy_refine(seq: np.ndarray, deadline: float) -> np.ndarray:
    n = seq.size
    best_seq = seq.copy()
    best_score = _fast_evaluation(best_seq)
    fft_len = 1 << ((2 * n - 1).bit_length())
    it = 0
    while time.time() < deadline:
        # convolution and its maximal entry
        fa = np.fft.rfft(seq, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        k = int(conv.argmax())
        # contribution weight of each index to conv[k]
        i = np.arange(n)
        j = k - i
        mask = (j >= 0) & (j < n)
        contrib = np.zeros_like(seq)
        contrib[mask] = seq[j[mask]]
        src = int(np.argmax(seq * contrib))          # source (large contribution)
        sink = int(np.argmin(contrib + 1e-30))       # sink (almost no contribution)
        if seq[src] < 1e-12:
            break
        eps = 0.08 * seq[src]                        # move 8 % of the source mass
        new_seq = seq.copy()
        new_seq[src] -= eps
        new_seq[sink] += eps
        score = _fast_evaluation(new_seq)
        if score < best_score - 1e-12:
            seq = new_seq
            best_seq = new_seq.copy()
            best_score = score
        it += 1
    return best_seq

# Main entry point.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Returns a list of non‑negative heights (length ≈ 5 000) with a low
    evaluation score.  The routine
        * explores a parametric family of symmetric “arcsine‑like’’ sequences,
        * refines the best candidate with a projected Adam optimiser that
          minimises a smooth approximation of the maximal self‑convolution,
        * finishes with a short greedy mass‑transfer step.
    The whole search is bounded by ``budget_s`` seconds.
    """
    np.random.seed(seed)
    start = time.time()
    # keep a safety margin for the harness
    deadline = start + budget_s - 10.0

    # ------------------------------------------------------------------
    # 1.  Basic parameters
    # ------------------------------------------------------------------
    n = 5000                     # length of the step‑function

    # ------------------------------------------------------------------
    # 2.  Scan a tiny parametric family (symmetric power‑law)
    # ------------------------------------------------------------------
    eps = 0.5                    # shift to avoid division by zero
    best_seq = None
    best_score = float("inf")

    # a)  Catalan / arcsine seed – always a safe baseline
    cat = _catalan_sequence(n).astype(float)
    cat /= cat.sum()
    sc = _fast_evaluation(cat)
    best_seq, best_score = cat.copy(), sc
    print(f"[baseline] Catalan score = {sc:.6f}")

    # b)  One‑parameter family  a_i ∝ ((i+ε)(n‑1‑i+ε))^α
    alphas = np.linspace(-1.2, -0.2, 13)      # coarse sweep
    for alpha in alphas:
        i = np.arange(n, dtype=float)
        a = ((i + eps) * (n - 1 - i + eps)) ** alpha
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum == 0.0:
            continue
        a /= a_sum
        sc = _fast_evaluation(a)
        if sc < best_score:

            print(f"[param sweep] α={alpha: .3f}  score = {sc:.6f}")

    # c)  Two‑parameter random search (α_left, α_right)
    for _ in range(40):
        al = np.random.uniform(-1.5, -0.05)

        a = ((i + eps) ** al) * ((n - 1 - i + eps) ** ar)

        a_sum = a.sum()
        if a_sum == 0.0:

        a /= a_sum
        sc = _fast_evaluation(a)
        if sc < best_score:
            best_seq, best_score = a.copy(), sc

    # ------------------------------------------------------------------
    # 3.  Projected Adam optimisation on a smoothed max‑objective
    # ------------------------------------------------------------------
    seq = best_seq.copy()
    m = np.zeros_like(seq)               # first moment
    v = np.zeros_like(seq)               # second moment
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr_base = 0.12                       # base learning rate
    # start with a modest soft‑max temperature and increase it gradually
    lam = 8.0
    lam_target = 180.0

    fft_len1 = 1 << ((2 * n - 1).bit_length())
    fft_len2 = 1 << ((3 * n - 2).bit_length())

    it = 0
    while time.time() < deadline:
        # ---- convolution a*a ------------------------------------------------
        a_fft = np.fft.rfft(seq, fft_len1)
        conv = np.fft.irfft(a_fft * a_fft, fft_len1)[: 2 * n - 1]

        # ---- soft‑max over the convolution entries ---------------------------
        max_val = conv.max()
        if lam < lam_target:
            lam = min(lam + lam_inc, lam_target)
        # stable soft‑max
        exps = np.exp(lam * (conv - max_val))
        p = exps / exps.sum()                     # p_k ≈ 1 on the maximal index

        # ---- gradient of the smoothed objective ------------------------------
        # w_i = Σ_k p_k a_{k-i}   (convolution of p with reversed a)
        a_rev = seq[::-1]
        p_fft = np.fft.rfft(p, fft_len2)
        rev_fft = np.fft.rfft(a_rev, fft_len2)
        conv2 = np.fft.irfft(p_fft * rev_fft, fft_len2)[: 3 * n - 2]
        w = conv2[n - 1: n - 1 + n]               # length n

        # diagonal correction (indices where k = 2i)
        diag_idx = 2 * np.arange(n)
        diag_term = p[diag_idx] * seq

        grad = 2.0 * w - diag_term                # ∂f/∂a_i

        # ---- Adam update -----------------------------------------------------
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr_base / (np.sqrt(v_hat) + eps_adam)
        seq_new = seq - step * m_hat

        # ---- projection onto the simplex ------------------------------------
        seq = _project_simplex(seq_new)

        # ---- occasional random jitter (helps escape tiny plateaus) ----------
        if np.random.rand() < 0.001:
            idx = np.random.randint(n)
            seq[idx] = max(0.0, seq[idx] + np.random.randn() * 1e-3)

        cur_score = _fast_evaluation(seq)
        if cur_score < best_score - 1e-12:

            best_seq = seq.copy()

            print(f"[status] iter={it:6d}  best={best_score:.6f}  elapsed={elapsed:.1f}s")

    return best_seq.tolist()
```
