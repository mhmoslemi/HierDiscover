def propose_candidate(seed=42,budget_s=1000,**kwargs):

    start = time.time()
    deadline = start + budget_s - 2.0

    # Determine length n

        n = len(height_sequence_1)
        if n < 2000: n = 2000
    except NameError:
        n = 5000

    # try to use existing sequence if available
    if 'height_sequence_1' in globals():
        init = np.array(height_sequence_1, dtype=float)
        if len(init) != n:
            # truncate or pad with power law tail
            if len(init) > n:
                init = init[:n]
            else:
                # pad with small tail following power law exponent ~0.6
                tail_len = n - len(init)
                tail = _power_law_sequence(tail_len, 0.6)
                init = np.concatenate([init, tail])
        # renorm
        init = init / init.sum()
        a = init.copy()
    else:
        a = _power_law_sequence(n, 0.60)

    # initial power law scan for best p anyway
    p_grid = np.linspace(0.55, 0.65, 61)

        maxb = _max_convolution_fft(seq)
        score = 2*n*maxb

    a = best.copy()
    best_score = 2*n*_max_convolution_fft(a)
    best_seq = a.copy()
    print(f"Initial best score {best_score:.6f}")

    # main loop
    iter_count = 0
    eta0 = 0.5
    beta = 1.0
    beta_max = 200.0
    beta_inc = 0.01  # per iteration
    while time.time() < deadline and iter_count < 500000:

        # compute autocorrelation
        conv_self = _fft_convolve(a, a)
        maxb = conv_self.max()
        score = 2*n*maxb

        # update best
        if score < best_score:

            best_seq=a.copy()

        # compute softmax weights
        beta = min(beta_max, beta + beta_inc)  # slowly increase

"""
Propose a step‑function with a low evaluation score.

Algorithm
---------
1. Choose a length `n`.  If a previous construction `height_sequence_1`
   exists we reuse its length, otherwise we take `n = 8000`.
2. Initialise the vector with a power‑law `a_i ∝ (i+1)^-p` and keep the
   best exponent on a cheap FFT‑based evaluation.
3. Optimise the **smoothed** objective  

        J(a) = 2·n·log‑sum‑exp( β·(a*a) ) / β

   (β → ∞ makes `J` equal to the true evaluation 2·n·max(a*a)).
   For a given β the gradient is

        ∇J_i = 4·n·∑_k w_k·a_{k-i},   w_k = exp(β·c_k) / ∑_j exp(β·c_j)

   where `c = a*a`.  The gradient is obtained with two FFT convolutions.
4. Perform additive projected gradient descent.
   The step size is chosen so that the *largest* coordinate changes by
   `η_t = η0 / sqrt(t)` (η0≈0.5).  After each step the vector is projected
   onto the probability simplex (ensuring non‑negativity and unit sum).
5. Increase β slowly (β←min(β_max, β+Δβ)) so that the optimisation
   gradually focuses on the true maximum.
6. Keep the best iterate (lowest cheap score).  After the time budget is
   exhausted we evaluate the best candidate with the exact
   `evaluate_sequence` and return it.

The method uses only FFT‑based convolutions (O(n log n) per iteration),
so many thousands of iterations fit comfortably into the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures

def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (sum = 1, v ≥ 0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all entries ≤ 0
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float) -> np.ndarray:
    """Return a normalised power‑law probability vector a_i ∝ (i+1)^‑p."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    seq = np.power(idx, -p)
    seq /= seq.sum()
    return seq


def _fft_convolve(x: np.ndarray, y: np.ndarray, fft_len: int = None) -> np.ndarray:
    """
    Fast (real) convolution using FFT.
    Returns the full convolution of length len(x)+len(y)-1.
    """
    n = len(x) + len(y) - 1
    if fft_len is None:
        fft_len = 1 << ((n - 1).bit_length())
    fx = np.fft.rfft(x, fft_len)
    fy = np.fft.rfft(y, fft_len)
    conv = np.fft.irfft(fx * fy, fft_len)[:n]
    return conv


def _autocorr(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Compute the full autocorrelation a*a using FFT (real input)."""
    return _fft_convolve(a, a, fft_len)


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


def _subgradient_max_convolution(a: np.ndarray, max_idx: int) -> np.ndarray:
    """
    Sub‑gradient of the active quadratic term (a*a)[max_idx].
    This is the same as 2·a_{max_idx‑i} on the valid index range.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    lo = max(0, max_idx - (n - 1))
    hi = min(max_idx, n - 1)
    i = np.arange(lo, hi + 1, dtype=np.int64)
    g[i] = 2.0 * a[max_idx - i]
    return g
