def propose_candidate(seed=42, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.2
    if 'height_sequence_1' in globals():
        a = np.array(height_sequence_1, dtype=float)
    else:
        n = 4000
        a = np.full(n, 1.0 / n)
    a = np.maximum(a, 0)
    a = a / a.sum()
    n = a.shape[0]
    conv = np.convolve(a, a)
    cur_max = conv.max()
    best_a = a.copy()
    best_max = cur_max
    best_score = 2.0 * n * best_max
    # parameters
    temperature = 0.02
    cooling_rate = 0.9995
    min_temp = 1e-6
    iteration = 0
    recompute_interval = 2000  # recompute conv from scratch to avoid drift
    while time.time() < deadline:
        # pick i, j random, ensure a[i] > 0
        i = np.random.randint(n)
        if a[i] <= 0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue
        # propose delta
        max_frac = min(0.5, temperature * 10)  # larger at high T
        delta = a[i] * np.random.rand() * max_frac
        # incremental update
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        conv_new = conv.copy()
        # shift updates
        conv_new[i:i+n] -= 2.0 * delta * a
        conv_new[j:j+n] += 2.0 * delta * a
        conv_new[2*i] += delta*delta
        conv_new[2*j] += delta*delta
        conv_new[i+j] -= 2.0 * delta*delta
        new_max = conv_new.max()
        # acceptance
        if new_max < cur_max or np.random.rand() < math.exp((cur_max - new_max)/ max(temperature, 1e-12)):
            a = a_new
            conv = conv_new
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()
                best_score = 2.0 * n * best_max
                print(f"[{time.time():.1f}] New best O={best_score:.6f}")
        # update temp
        temperature = max(min_temp, temperature * cooling_rate)
        iteration += 1
        if iteration % recompute_interval == 0:
            conv = np.convolve(a, a)
            cur_max = conv.max()
    # finalize
    final_score = evaluate_sequence(best_a.tolist())
    print("[final] evaluate_sequence reports O=", final_score)
    return best_a.tolist()

"""
propose_candidate
-----------------
Search for a non‑negative step‑function (a list of floats) that minimises

    O(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

where conv(a,a) is the full linear convolution.
The algorithm starts from the best known construction (the global variable
`height_sequence_1` if it exists, otherwise a uniform distribution) and then
performs a long simulated‑annealing / hill‑climbing phase.  Convolution updates
are performed **incrementally** (O(n) per move) using the exact formula

    a' = a + Δ·(e_j – e_i)                (Δ > 0, i ≠ j)
    conv' = conv
            – 2·Δ·shift_i(conv)          (shift_i = a placed at offset i)
            + 2·Δ·shift_j(conv)
            + Δ²·δ_{2i} – 2·Δ²·δ_{i+j} + Δ²·δ_{2j}

so that only a few slices of the convolution array need to be updated.
A simple Metropolis rule with a slowly cooling temperature allows occasional
up‑hill moves, helping to escape flat plateaus.  The best sequence seen so far
is kept and finally evaluated with the provided `evaluate_sequence` function.

The whole search respects the 1000 s budget; progress messages are printed
occasionally.
"""

import time
import math
import numpy as np

# -------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# -------------------------------------------------------------
def _conv_full(a: np.ndarray) -> np.ndarray:
    """Exact full convolution a * a (length 2·n‑1)."""
    return np.convolve(a, a)


# -------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return O = 2·n·max(conv(a,a)) (a is assumed to sum to 1)."""
    n = a.shape[0]
    conv = _conv_full(a)
    return 2.0 * n * conv.max()


# -------------------------------------------------------------
def _apply_mass_transfer(a: np.ndarray,
                         conv: np.ndarray,
                         i: int,
                         j: int,
                         delta: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Return (a_new, conv_new) after moving *delta* mass from index i to index j.
    The update is exact and O(n).
    """
    n = a.shape[0]
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta

    conv_new = conv.copy()
    # shift contributions of the *old* a (before the move)
    # subtract the part coming from position i
    conv_new[i:i + n] -= 2.0 * delta * a
    # add the part coming from position j
    conv_new[j:j + n] += 2.0 * delta * a
    # delta² corrections
    conv_new[2 * i] += delta * delta
    conv_new[2 * j] += delta * delta
    conv_new[i + j] -= 2.0 * delta * delta
    return a_new, conv_new


# -------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function sequence that minimises the evaluation function.
    Returns the best sequence as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # keep a small safety margin

    # -----------------------------------------------------------------
    # Initialise from the previous best construction if available,
    # otherwise start from a uniform distribution.
    if 'height_sequence_1' in globals():
        a = np.array(height_sequence_1, dtype=float)
        print(f"[init] Using provided sequence, length = {a.size}")
    else:
        n = 4000
        a = np.full(n, 1.0 / n, dtype=float)
        print(f"[init] No prior sequence – using uniform, length = {n}")

    # Guard against negatives and normalise to the probability simplex
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a)
    n = a.shape[0]

    # Initial convolution and objective
    conv = _conv_full(a)
    cur_max = conv.max()
    cur_score = 2.0 * n * cur_max
    best_a = a.copy()
    best_score = cur_score

    print(f"[start] length={n}, O={cur_score:.6f}")

    # -----------------------------------------------------------------
    # Simulated‑annealing / hill‑climbing phase.
    # --------------------------------------------------------------
    # Parameters (tuned for a ~5 k long sequence)
    temperature = cur_max * 0.1          # start relatively hot
    cooling_rate = 0.9997                # slow exponential cooling
    min_temperature = 1e-8
    max_frac_at_start = 0.10             # max fraction of a[i] moved in one step
    recompute_every = 2000               # recompute convolution from scratch
    iteration = 0
    no_improve = 0
    max_no_improve = 50000

    while time.time() < deadline:
        # ---------------------------------------------------------
        # Choose two distinct indices; ensure donor has some mass.
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        # Fraction of mass to move – shrinks with temperature.
        max_frac = max_frac_at_start * (temperature / (cur_max * 0.1) + 0.1)
        max_frac = max(1e-4, min(max_frac, max_frac_at_start))
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # Apply the move (exact, O(n))
        a_new, conv_new = _apply_mass_transfer(a, conv, i, j, delta)
        new_max = conv_new.max()

        # Metropolis acceptance
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            # uphill move – probability decreases with temperature
            prob = math.exp((cur_max - new_max) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a = a_new
            conv = conv_new
            cur_max = new_max
            cur_score = 2.0 * n * cur_max
            no_improve = 0
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] New global best O={best_score:.6f} "
                      f"(iter={iteration})")
        else:
            no_improve += 1

        # ---------------------------------------------------------
        # Temperature schedule
        temperature = max(min_temperature, temperature * cooling_rate)

        # Periodically recompute the whole convolution to avoid drift
        if iteration % recompute_every == 0:
            conv = _conv_full(a)
            cur_max = conv.max()
            cur_score = 2.0 * n * cur_max

        # Simple restart if stuck for a long time
        if no_improve >= max_no_improve:
            # random jitter – small Gaussian perturbation followed by projection
            jitter = np.random.randn(n) * 0.001
            a = _project_to_simplex(a + jitter)
            conv = _conv_full(a)
            cur_max = conv.max()
            cur_score = 2.0 * n * cur_max
            no_improve = 0
            print(f"[{time.time():.1f}] Restart (jitter), O={cur_score:.6f}")

        iteration += 1

    # -----------------------------------------------------------------
    # Final sanity check with the provided evaluation function
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:  # pragma: no cover
        print("[final] evaluate_sequence failed:", e)

    return best_a.tolist()
