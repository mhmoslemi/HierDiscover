"""
Propose a step‑function sequence via exponentiated‑gradient descent.

The evaluation metric is   2·n·max(conv(a,a)) / (sum a)² .
Because the objective is homogeneous we fix sum(a)=1 and minimise
   L(a)=2·n·max(conv(a,a)).
The convex hull of feasible sequences is large; a simple gradient
descent on the (non‑smooth) ∞‑norm of the convolution rarely moves far.
We therefore use a smooth proxy: the (log‑)exponential update (mirror
descent on the probability simplex) driven by the sub‑gradient of the
largest convolution entries.  Starting from the Catalan (near‑optimal)
distribution we perform many cheap FFT‑based convolution evaluations,
adapt a learning‑rate and keep the best sequence seen.
The method runs for up to the given time budget (default 1000 s) and
returns a plain Python list that can be fed to `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast linear convolution using real FFT (full linear mode)
# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << (2 * n - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Catalan‑type coefficients, normalised to unit L¹‑norm
# ----------------------------------------------------------------------
def _catalan_normalised(n: int) -> np.ndarray:
    """First n Catalan‑type numbers a_i = C(2i,i)/4**i, normalised so that sum=1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * (i - 1) + 1) / (2.0 * i)   # a_i = a_{i-1}*(2i-1)/(2i)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the simplex
# ----------------------------------------------------------------------
def _exp_gradient_descent(a: np.ndarray,
                          n: int,
                          deadline: float,
                          top_k: int = 3,
                          lr_start: float = 0.5,
                          lr_min: float = 1e-4,
                          lr_max: float = 3.0) -> np.ndarray:
    """
    Refine `a` (already normalised to sum=1) by repeated
    exponentiated‑gradient steps that directly reduce the maximal
    convolution entry.  The loop stops when the time budget expires.
    """
    lr = lr_start
    best_a = a.copy()
    conv = _fft_convolve(a)
    best_score = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1) evaluate current objective
        # --------------------------------------------------------------
        conv = _fft_convolve(a)
        max_val = conv.max()
        score = 2.0 * n * max_val

        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[exp‑gd] iter {it:05d}  score {best_score:.9f}  lr {lr:.5f}")

        # --------------------------------------------------------------
        # 2) compute a sub‑gradient from the largest convolution entries
        # --------------------------------------------------------------
        # indices of the (up to) top_k largest entries
        if top_k == 1:
            k_idxs = [np.argmax(conv)]
        else:
            # np.argpartition is O(n), cheap for the tiny top_k we need
            part = np.argpartition(conv, -top_k)[-top_k:]
            # ensure they are really the largest (order not needed)
            k_idxs = part.tolist()

        grad = np.zeros_like(a)          # shape (n,)

        for k in k_idxs:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            idx = np.arange(i_start, i_end + 1, dtype=int)
            # sub‑gradient of (a*a)_k w.r.t. a_i is 2*a_{k-i}
            grad[idx] += 2.0 * a[k - idx]

        # --------------------------------------------------------------
        # 3) exponentiated‑gradient update on the simplex
        # --------------------------------------------------------------
        # work in log‑space to keep positivity
        log_a = np.log(a + 1e-30) - lr * grad
        a = np.exp(log_a)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:                              # fallback – uniform distribution
            a = np.full(n, 1.0 / n)

        # --------------------------------------------------------------
        # 4) adapt learning rate
        # --------------------------------------------------------------
        if score < best_score + 1e-12:     # improvement (tiny tolerance)
            lr = min(lr * 1.02, lr_max)
        else:
            lr = max(lr * 0.98, lr_min)

        it += 1

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a sequence of non‑negative heights that minimises
    ``evaluate_sequence``.  The routine runs for at most *budget_s* seconds
    and returns the best sequence found.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : int, optional
        Time budget in seconds (default 1000).

    Returns
    -------
    list[float]
        The best sequence discovered (plain Python list).
    """
    np.random.seed(seed)
    # leave a small safety margin for the harness
    deadline = time.time() + budget_s - 2.0

    # ------------------------------------------------------------------
    # 1) Choose a reasonably large support.
    #    n≈12000 gives a good trade‑off between runtime and quality.
    # ------------------------------------------------------------------
    n = 12000

    # ------------------------------------------------------------------
    # 2) Initialise with the Catalan distribution (near‑optimal analytically)
    # ------------------------------------------------------------------
    a = _catalan_normalised(n)

    # ------------------------------------------------------------------
    # 3) Refine by exponentiated‑gradient descent.
    #    The routine keeps the best vector seen.
    # ------------------------------------------------------------------
    a = _exp_gradient_descent(a, n=n, deadline=deadline,
                              top_k=3, lr_start=0.5)

    # ------------------------------------------------------------------
    # 4) Final sanity check and return.
    # ------------------------------------------------------------------
    final_score = 2.0 * n * np.max(_fft_convolve(a))
    print(f"[final] n={n}, final score ≈ {final_score:.9f}")

    return a.tolist()
