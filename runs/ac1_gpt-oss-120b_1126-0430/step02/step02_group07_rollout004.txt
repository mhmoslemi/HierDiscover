"""
propose_candidate

A fast, gradient‑based optimiser that works directly on the probability
(simplex) representation of the step function.  The objective

    2 * n * max(conv(a,a)) / (sum(a)**2)

is scale‑invariant, so we keep the vector normalised to sum = 1.
At each iteration we:

1. Compute the full discrete convolution a*a by means of an FFT
   (O(n log n) per evaluation).
2. Locate the index k where the convolution attains its maximum.
3. The gradient of the maximal convolution term w.r.t. a is
   g_i = 2·a_{k−i} (zero when k−i is out of bounds).  This is the
   exact sub‑gradient of the objective at the current point.
4. Perform an *exponentiated‑gradient* step:
        a_i ← a_i · exp(−η·g_i) ,   then renormalise onto the simplex.
   The step‑size η decreases like η₀/√t, providing large moves early
   and fine adjustment later.
5. Accept the new point if it improves the objective; otherwise
   accept it with a tiny Metropolis probability to escape shallow
   plateaux.
6. Every few thousand iterations we add a tiny random mass‑transfer
   and re‑project onto the simplex to keep the search stochastic.

The loop runs as long as the time budget permits (≈950 s).  The best
vector seen during the run is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures
# ----------------------------------------------------------------------


def _project_to_simplex(v):
    """Project a vector onto the probability simplex {x ≥ 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are equal (or negative); return uniform vector
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


def _conv_fft(a, size):
    """
    Discrete convolution a*a using FFT.
    Returns the full convolution of length 2*len(a)-1.
    """
    A = np.fft.rfft(a, n=size)
    C = np.fft.irfft(A * A, n=size)
    return C[: 2 * len(a) - 1]


def _max_conv_and_grad(a, size):
    """
    Compute the maximal convolution value, its index, and the exact
    sub‑gradient of that value w.r.t. a (assuming sum(a)=1).

    Returns
    -------
    max_val : float
        maximal entry of a*a
    max_idx : int
        index where the maximum occurs (0 … 2n‑2)
    grad    : np.ndarray
        gradient vector (shape = a.shape)
    """
    n = len(a)
    conv = _conv_fft(a, size)
    max_idx = int(conv.argmax())
    max_val = float(conv[max_idx])

    # gradient: g_i = 2 * a_{k_max - i} when the index is valid
    rev = max_idx - np.arange(n)
    mask = (rev >= 0) & (rev < n)
    grad = np.zeros_like(a)
    grad[mask] = 2.0 * a[rev[mask]]
    return max_val, max_idx, grad

# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed=123, budget_s=950, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence``.  The algorithm is a projected
    exponentiated‑gradient descent on the simplex with occasional
    stochastic perturbations.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : float, optional
        Time budget in seconds; the function stops a few seconds before
        this limit.
    **kwargs
        Ignored – kept for compatibility.

    Returns
    -------
    list[float]
        The best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # Determine problem size and initialise a feasible point
    # ------------------------------------------------------------------
    seq0 = globals().get("height_sequence_1")

        except Exception:

    else:

        n = 2244

    else:
        total = init.sum()
        if total <= 0:
            cur = np.full(n, 1.0 / n)
        else:
            cur = init / total

    # ------------------------------------------------------------------
    # Baseline evaluation
    # ------------------------------------------------------------------
    try:

    except Exception:

    best_seq = cur.copy()
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    fft_size = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    no_improve = 0                # counter for stagnation detection

        it += 1

        # --------------------------------------------------------------
        # Gradient step (exponentiated‑gradient)
        # --------------------------------------------------------------
        eta = eta0 / np.sqrt(it)               # decreasing step‑size
        factor = np.exp(-eta * grad)           # element‑wise exponent
        new = cur * factor
        s = new.sum()
        if s == 0.0:
            new = np.full(n, 1.0 / n)
        else:
            new = new / s

        # evaluate the candidate (fast – via FFT)
        max_val_new, _, _ = _max_conv_and_grad(new, fft_size)
        new_score = 2.0 * n * max_val_new

        # --------------------------------------------------------------
        # Acceptance rule
        # --------------------------------------------------------------
        accept = False
        if new_score < cur_score - 1e-12:
            accept = True
        else:
            # tiny Metropolis move to escape flat plateaux
            temperature = 0.001 * max(0.0, 1.0 - it / max_iters)
            if rng.random() < np.exp(-(new_score - cur_score) / (temperature + 1e-12)):
                accept = True

        if accept:
            cur = new
            cur_score = new_score
            # recompute gradient for the next iteration
            max_val, max_idx, grad = _max_conv_and_grad(cur, fft_size)
            no_improve = 0

            # Verify with the official evaluator (rare but safe)
            real_score = evaluate_sequence(cur.tolist())
            if real_score < best_score - 1e-12:
                best_score = real_score
                best_seq = cur.copy()
                print(f"[iter {it}] new best {best_score:.6f}")
        else:
            no_improve += 1
            # keep the old gradient; no need to recompute unless we perturb

        # --------------------------------------------------------------
        # Periodic stochastic perturbation to avoid getting stuck
        # --------------------------------------------------------------
        if it % 5000 == 0:
            # move a tiny random amount of mass from a weighted source to a random target
            src = rng.choice(n, p=cur / cur.sum())
            tgt = rng.integers(0, n)
            if src != tgt and cur[src] > 0.0:
                eps = cur[src] * rng.uniform(0.001, 0.01)
                cur[src] -= eps
                cur[tgt] += eps
                cur = _project_to_simplex(cur)
                # recompute convolution & gradient after the perturbation
                max_val, max_idx, grad = _max_conv_and_grad(cur, fft_size)
                cur_score = 2.0 * n * max_val
                no_improve = 0

        # --------------------------------------------------------------
        # Numerical safety – re‑project onto the simplex occasionally
        # --------------------------------------------------------------
        if it % 2000 == 0:
            cur = _project_to_simplex(cur)

        # optional early‑stop if we have stalled for a long time
        if no_improve > 50_000:
            # inject a small random reshuffle to shake the system
            cur = np.maximum(cur + rng.normal(scale=1e-4, size=n), 0.0)
            cur = _project_to_simplex(cur)
            max_val, max_idx, grad = _max_conv_and_grad(cur, fft_size)
            cur_score = 2.0 * n * max_val
            no_improve = 0

    print(f"[final] best score={best_score:.6f} after {it} iterations")
    return best_seq.tolist()
