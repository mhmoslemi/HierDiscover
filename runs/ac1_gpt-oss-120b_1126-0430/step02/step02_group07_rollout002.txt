def convolve_fft(x):
    n = len(x)
    size = 1
    while size < 2*n:
        size <<= 1
    X = np.fft.rfft(x, size)
    conv = np.fft.irfft(X * X, size)[:2*n-1]
    return conv

propose_candidate – gradient‐descent optimiser for the max‑autocorrelation
objective.

The evaluation metric is
    score = 2 * n * max(conv(a,a)) / (sum(a)**2)

which is scale‑invariant (the sum can be fixed to 1).  The algorithm works

# Helper utilities – must be defined at top level (no closures)

def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, sum(x)=1}."""
    n = v.size
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are equal – return uniform vector
        return np.full(n, 1.0 / n, dtype=v.dtype)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p

    )
    init_vectors.append(tri / tri.sum())

    # 4) Power‑law families (decreasing)
    idx = np.arange(n, dtype=float)
    for alpha in (0.4, 0.6, 0.8, 1.0):
        pl = np.power(idx + 1.0, -alpha)
        init_vectors.append(pl / pl.sum())

    # 5) Mix of two power‑laws (randomly sampled a few times)
    for _ in range(5):
        a1, s1 = rng.uniform(0.2, 1.2), rng.uniform(0.1, 5.0)
        a2, s2 = rng.uniform(0.2, 1.2), rng.uniform(0.1, 5.0)
        p = rng.random()
        mix = p * np.power(idx + s1, -a1) + (1 - p) * np.power(idx + s2, -a2)
        init_vectors.append(mix / mix.sum())

    # 6) The previously best known vector
    init_vectors.append(np.array(best_seq, dtype=float))

    # Pick the best among the above (using the cheap exact score)
    cur = None
    cur_score = np.inf
    for cand in init_vectors:
        # Since sum(cand)=1, score = 2*n*max(conv(cand,cand))
        conv = np.convolve(cand, cand)
        score = 2.0 * n * conv.max()
        if score < cur_score:
            cur_score = score
            cur = cand.copy()
    print(f"[init] selected start score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # FFT parameters (common size for both convolutions)
    # ------------------------------------------------------------------
    # Need size ≥ 3*n-2 to cover a*w_rev convolution
    fft_len = _next_pow2(3 * n)
    # Pre‑allocate arrays to avoid repeated allocations where possible
    # (NumPy creates fresh arrays on each call – acceptable for this scale)

    # ------------------------------------------------------------------
    # Optimisation hyper‑parameters
    # ------------------------------------------------------------------
    max_iters = 200_000                # safety cap
    lam_start = 5.0                    # low temperature – smooth loss
    lam_end = 200.0                    # high temperature – close to max
    step_start = 0.5                   # initial step size
    step_end = 0.001                   # final step size
    improve_tol = 1e-12                # numerical tolerance for improvement

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    it = 0
    last_improve_it = 0
    while time.time() - start_time < budget_s - 5 and it < max_iters:
        it += 1

        # --- 1) convolution a*a (via FFT) --------------------------------
        A_hat = np.fft.rfft(cur, n=fft_len)
        conv_full = np.fft.irfft(A_hat * A_hat, n=fft_len)[: (2 * n - 1)]
        max_conv = conv_full.max()
        cur_score = 2.0 * n * max_conv  # exact score, sum(cur)=1

        # --- 2) compute soft‑max weights over the convolution entries -----
        # Temperature schedule (linear in iteration count)
        lam = lam_start + (lam_end - lam_start) * min(it / 50_000.0, 1.0)

        # Stabilise exponentials by subtracting the maximum (which is 0 after scaling)
        exp_arg = lam * (conv_full - max_conv)          # ≤0
        # Use np.exp safely – for large lam the values can under‑flow, which is fine
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:                    # extreme under‑flow – fallback to hard max
            w = np.zeros_like(w)
            w[conv_full.argmax()] = 1.0
            w_sum = 1.0
        w /= w_sum

        # --- 3) gradient = 2 * correlation(a, w) ------------------------
        w_rev = w[::-1]                     # reversed weight vector
        W_hat = np.fft.rfft(w_rev, n=fft_len)
        corr_full = np.fft.irfft(A_hat * W_hat, n=fft_len)
        # Length of corr_full = n + len(w_rev) - 1 = n + (2n-1) - 1 = 3n-2
        # The gradient entry i uses corr_full[mid - i] where mid = len(w) - 1
        mid = len(w) - 1                    # = 2*n - 2
        grad = 2.0 * corr_full[mid - np.arange(n)]

        # --- 4) step size (linear decay) ---------------------------------
        step = step_start + (step_end - step_start) * (it / max_iters)

        # --- 5) propose update, project onto simplex ----------------------
        new_vec = cur - step * grad
        new_vec = _project_to_simplex(new_vec)

        # --- 6) evaluate new score (fast, using FFT again) ---------------
        A_hat_new = np.fft.rfft(new_vec, n=fft_len)
        conv_new = np.fft.irfft(A_hat_new * A_hat_new, n=fft_len)[: (2 * n - 1)]
        new_score = 2.0 * n * conv_new.max()

        # --- 7) accept if improvement (or occasional random accept) ------
        accepted = False
        if new_score < cur_score - improve_tol:
            accepted = True
        else:
            # tiny probability to escape plateaus
            if rng.random() < 0.001:
                accepted = True

        if accepted:
            cur = new_vec
            cur_score = new_score
            if new_score < best_score - improve_tol:
                best_score = new_score
                best_seq = cur.tolist()
                last_improve_it = it
                print(f"[iter {it:5d}] new best (fast) score = {best_score:.6f}")

        # --- 8) occasional exact evaluation (to guard against numerical drift) ---
        if it % 5000 == 0:
            real_score = evaluate_sequence(cur.tolist())
            if real_score < best_score - improve_tol:
                best_score = real_score
                best_seq = cur.tolist()
                print(f"[iter {it:5d}] exact evaluation improved best to {best_score:.6f}")
            else:
                # keep internal fast score in sync with the official one
                cur_score = real_score

        # --- 9) early stopping if we reach the target --------------------
        if best_score <= 1.5035:
            print(f"[early stop] reached targetish score {best_score:.6f} at iter {it}")
            break

        # --- 10) stagnation handling – random perturbation ----------------
        if it - last_improve_it > 20_000:
            # inject a little noise to potentially escape a flat region
            noise = rng.normal(scale=0.02, size=n)
            cur = _project_to_simplex(cur + noise)
            last_improve_it = it
            print(f"[iter {it:5d}] injected noise to escape stagnation")

    print(f"[final] best score found = {best_score:.6f}")
    return best_seq
