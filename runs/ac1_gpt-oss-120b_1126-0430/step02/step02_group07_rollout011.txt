"""
propose_candidate

Search for a step‑function (non‑negative list) that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of three stages:

1. **Fast random sampling** – generate candidates from a few simple
   parametric families (power‑law, exponential) and keep the best one.
   This gives a good starting point.

2. **Linear‑programming refinement** – repeatedly solve the LP

        max   Σ b_j
        s.t.  (f * b)_k ≤ M    for all k
              b_j ≥ 0

   where * denotes discrete convolution, f is the current sequence,
   and M = max(conv(f,f)).  The LP pushes mass from positions where
   f is large to positions where f is small while keeping the
   convolution peak bounded.  The solution b is normalised and,
   if it yields a smaller peak, replaces f.  Multiple LP iterations
   quickly drive the maximum convolution down.

3. **Targeted hill‑climbing** – a final lightweight local search that
   moves a small amount of mass from the index contributing most to
   the current maximal convolution value to the index contributing the
   least.  This polishes the solution obtained by the LP stage.

All numbers are kept non‑negative and summed to 1, so the objective
reduces to `2 * n * max(conv)`.  The routine respects the 1000‑second
time budget and prints progress diagnostics.
"""

import time
import numpy as np
from scipy.sparse import csr_matrix, diags
from scipy.optimize import linprog


def _project_to_simplex(v):
    """Project a vector onto the probability simplex {x ≥ 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

    start = time.time()

    # ------------------------------------------------------------------
    # Determine length of the problem (use the provided baseline if any)
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)  # provided by the harness
    except Exception:
        n = 2244  # fallback length

    # ------------------------------------------------------------------
    # Baseline sequence (always available)
    # ------------------------------------------------------------------
    try:
        best_seq = list(height_sequence_1)
        best_score = evaluate_sequence(best_seq)
    except Exception:
        best_seq = np.full(n, 1.0 / n).tolist()
        best_score = evaluate_sequence(best_seq)

    print(f"[init] n={n}, baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Helper to try a raw (non‑normalised) vector
    # ------------------------------------------------------------------
    idx = np.arange(n, dtype=float)

    def _try_raw(raw_vec, label):
        nonlocal best_seq, best_score
        total = raw_vec.sum()
        if total <= 0:
            return
        norm = raw_vec / total
        score = evaluate_sequence(norm.tolist())
        if score < best_score:
            best_score = score
            best_seq = norm.tolist()
            print(f"[{label}] improved to {score:.6f}")

    # ------------------------------------------------------------------
    # 1. Random sampling of simple families (fast, cheap)
    # ------------------------------------------------------------------
    for _ in range(2000):
        if time.time() - start > budget_s - 30:
            break
        alpha = rng.uniform(0.20, 1.20)
        shift = rng.uniform(0.001, 5.0)
        raw = np.power(idx + shift, -alpha)
        _try_raw(raw, "randPL")

    for _ in range(500):
        if time.time() - start > budget_s - 30:
            break
        beta = rng.uniform(1e-5, 0.1)
        raw = np.exp(-beta * idx)
        _try_raw(raw, "exp")

    # ------------------------------------------------------------------
    # 2. Linear‑programming refinement
    # ------------------------------------------------------------------
    cur = np.array(best_seq, dtype=float)   # current working vector
    cur_score = best_score

    # Build the sparse pattern matrix P such that (P @ b) = conv(cur, b)
    #   rows = i + j, cols = i   (i,j ∈ [0,n-1])
    i = np.arange(n).reshape(-1, 1)          # shape (n,1)
    j = np.arange(n).reshape(1, -1)          # shape (1,n)
    rows = (i + j).ravel()                   # length n*n
    cols = np.repeat(np.arange(n), n)        # length n*n
    data = np.ones_like(rows, dtype=float)
    pattern = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    max_lp_iters = 30
    lp_iter = 0
    while time.time() - start < budget_s - 5 and lp_iter < max_lp_iters:
        lp_iter += 1

        # current maximal self‑convolution
        conv_self = np.convolve(cur, cur)
        M = conv_self.max()

        # Build linear constraints   (pattern @ diag(cur)) * b  ≤  M
        D = diags(cur, format="csr")
        A_ub = pattern @ D                     # shape (2n‑1, n)
        b_ub = np.full(2 * n - 1, M, dtype=float)

        # maximise sum(b)  → minimise -sum(b)
        c = -np.ones(n, dtype=float)

        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method="highs",
                      options={"presolve": True})

        if not res.success:
            print(f"[LP] iteration {lp_iter} failed: {res.message}")
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 0:
            print("[LP] zero sum(b), aborting LP loop")
            break

        # Normalise to a probability vector
        g = b / sum_b

        # Compute the peak of its self‑convolution
        conv_g = np.convolve(g, g)
        M_g = conv_g.max()
        new_score = 2.0 * n * M_g

        if new_score + 1e-12 < cur_score:
            # Strict improvement – accept g outright
            cur = g
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_seq = g.tolist()
                best_score = new_score
                print(f"[LP] iter={lp_iter} improvement → {best_score:.6f}")
            continue

        # No direct improvement – try a modest convex combination
        improved = False
        for t in (0.5, 0.3, 0.1):
            cand = (1.0 - t) * cur + t * g
            cand = np.maximum(cand, 0.0)
            cand /= cand.sum()
            cand_score = 2.0 * n * np.convolve(cand, cand).max()
            if cand_score + 1e-12 < cur_score:
                cur = cand
                cur_score = cand_score
                if cand_score < best_score - 1e-12:
                    best_seq = cand.tolist()
                    best_score = cand_score
                    print(f"[LP] iter={lp_iter} mix t={t:.2f} → {best_score:.6f}")
                improved = True
                break
        if not improved:
            print(f"[LP] iter={lp_iter} no improvement – terminating LP phase")
            break

    # ------------------------------------------------------------------
    # 3. Targeted hill‑climbing polishing
    # ------------------------------------------------------------------
    cur_conv = np.convolve(cur, cur)
    max_hc_iters = 50000
    hc_it = 0
    while time.time() - start < budget_s - 1 and hc_it < max_hc_iters:
        hc_it += 1
        max_idx = int(np.argmax(cur_conv))

        # contribution of each index to the maximal convolution entry
        i_arr = np.arange(n)
        rev = max_idx - i_arr
        mask = (rev >= 0) & (rev < n)
        contrib = np.zeros_like(cur)
        contrib[mask] = cur[i_arr[mask]] * cur[rev[mask]]

        src = int(np.argmax(contrib))
        tgt = int(np.argmin(contrib))
        if src == tgt or cur[src] <= 0:
            src, tgt = rng.choice(n, size=2, replace=False)

        # step size shrinks slowly
        frac = 0.05 * max(0.0, 1.0 - hc_it / max_hc_iters) + 0.01
        step = cur[src] * rng.random() * frac
        if step <= 0:
            continue

        new = cur.copy()
        new[src] -= step
        new[tgt] += step
        new /= new.sum()                     # maintain normalisation

        new_conv = np.convolve(new, new)
        new_score = 2.0 * n * new_conv.max()

        if new_score + 1e-12 < cur_score:
            cur = new
            cur_conv = new_conv
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_seq = new.tolist()
                best_score = new_score
                print(f"[refine] it={hc_it} new best={best_score:.6f}")

        # occasional re‑projection to the simplex for numerical safety
        if hc_it % 1000 == 0:
            cur = _project_to_simplex(cur)
            cur_conv = np.convolve(cur, cur)

    print(f"[final] best score found: {best_score:.6f}")
    return best_seq
