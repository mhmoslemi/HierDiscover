        # compute conv
        conv = _full_convolution(seq)
        max_val = conv.max()

        tol = max_val * 0.005  # within 0.5% of max considered
        peak_mask = conv >= max_val - tol
        # compute weight vector via convolution of peak_mask with reversed seq
        mask = peak_mask.astype(float)
        # convolution with reversed seq
        tmp = np.convolve(mask, seq[::-1])
        w = tmp[n-1:n-1+n]

        # define sink set as indices with smallest w
        thresh = np.percentile(w, sink_percent)
        sink_mask = w <= thresh
        sink_idx = np.flatnonzero(sink_mask)
        if sink_idx.size == 0:
            sink_idx = np.array([np.argmin(w)])
            sink_mask = np.zeros_like(w, dtype=bool)
            sink_mask[sink_idx[0]] = True

        # sources: indices where w is large, could be all non-sink
        source_idx = np.flatnonzero(~sink_mask)
        # compute total excess
        w_sink_min = w[sink_idx].min()
        # For each source i, contribution to peaks = w[i]*seq[i]
        excess = seq * (w - w_sink_min)
        total_excess = excess[source_idx].sum()
        if total_excess <= 0:

            lam = target_reduction / total_excess
            lam = max(min(lam, 1.0), 1e-6)

            delta[sink_mask] = 0.0

    m = 1 << ((len(x) + len(y) - 1).bit_length())

"""
Propose a step‑function with a lower evaluation score.

Algorithm (Aggregated‑peak mass transfer)
------------------------------------------------
1. Start from the classic Catalan (“arcsine”) sequence
   a_i = C(2i,i)/4**i .  Its self‑convolution is (almost) constant,
   giving the π/2 ≈ 1.571 bound.

2. In each iteration we look at *all* indices of the convolution that are
   close to the current maximum (within a few per‑cent).  From these
   “active peaks’’ we build a weight vector

        w_i = Σ_{k active} a_{k‑i}

   (zero outside the valid range).  Large w_i means that coefficient a_i
   participates heavily in the current peaks.

3. A small percentage of the positions with the smallest w_i are chosen
   as “sinks’’ – moving mass there hardly increases any peak because the
   corresponding weights are tiny.  All other positions are sources.

4. The excess contribution of a source i is a_i·(w_i‑w_min).  To lower the
   peak by a prescribed relative amount we move a fraction λ of the
   source mass (proportional to this excess) into the sinks, renormalise
   the sequence to keep the total sum unchanged, and clip negatives.

5. The move is accepted only if the true evaluation score improves.
   Occasionally a random jitter or a tiny random re‑allocation is tried
   to escape plateaux.

6. As the optimisation proceeds the sink set becomes tighter (percentile
   goes from ~5 % down to ~0.5 %) and the target peak‑reduction is
   gradually reduced, leading to fine‑grained adjustments.

The whole loop runs until the time budget (≈1000 s) is exhausted.  The
best sequence seen is returned.  The implementation uses FFT‑based
convolutions for O(n log n) work per iteration and therefore can afford
hundreds of thousands of iterations within the given time limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast evaluation – identical to the supplied routine but without the
# security checks.  Returns the score (lower is better).
# ----------------------------------------------------------------------
def _fast_evaluation(seq: np.ndarray) -> float:
    n = seq.size
    if n == 0:
        return float("inf")
    # FFT size = next power of two ≥ 2*n-1
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Catalan (arcsine) coefficients a_i = binom(2i,i)/4**i
# ----------------------------------------------------------------------
def _catalan_sequence(length: int) -> np.ndarray:
    seq = np.empty(length, dtype=float)
    a = 1.0
    seq[0] = a
    for i in range(1, length):
        a *= (2 * i - 1) / (2.0 * i)      # a_i = a_{i-1} * (2i‑1)/(2i)
        seq[i] = a
    return seq


# ----------------------------------------------------------------------
# Generic FFT convolution, returns the full linear convolution.
# ----------------------------------------------------------------------
def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    size = x.size + y.size - 1
    m = 1 << (size - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, m)
    fy = np.fft.rfft(y, m)
    conv = np.fft.irfft(fx * fy, m)[:size]
    return conv


# ----------------------------------------------------------------------
# Full self‑convolution a*a (length 2*n‑1)
# ----------------------------------------------------------------------
def _full_convolution(seq: np.ndarray) -> np.ndarray:
    return _fft_convolve(seq, seq)


# ----------------------------------------------------------------------
# Main optimisation routine – aggregates the contribution of all
# near‑maximal convolution entries.
# ----------------------------------------------------------------------
def _optimise_aggregated(seq: np.ndarray, deadline: float) -> np.ndarray:
    n = seq.size
    # keep the sum constant (the arcsine sequence already has the
    # “right’’ normalisation for the score)
    total_sum = seq.sum()

    best_seq = seq.copy()
    best_score = _fast_evaluation(best_seq)

    # parameters that change during the run
    sink_start = 5.0          # start percentile for sink set
    sink_end = 0.5            # final percentile
    rel_target_reduce = 0.012   # aim to cut the peak by ~1.2 % each step
    min_lambda = 1e-7
    iteration = 0
    start_time = time.time()

    while time.time() < deadline:
        # ---- current self‑convolution ------------------------------------
        conv = _full_convolution(seq)
        max_val = conv.max()

        # ---- identify all peaks that are “close’’ to the maximum ----------
        # we treat any entry within 0.5 % of the max as active
        tol = max_val * 0.005
        peak_mask = (conv >= max_val - tol).astype(float)   # 0/1 mask

        # ---- weight vector w_i = Σ_{active k} a_{k‑i} --------------------
        # this is a convolution of the mask with the reversed sequence
        w_full = _fft_convolve(peak_mask, seq[::-1])
        # slice the part that corresponds to i = 0…n‑1
        w = w_full[n - 1 : n - 1 + n]

        # ---- dynamic sink percentile ------------------------------------
        elapsed = time.time() - start_time
        total_allowed = deadline - start_time
        sink_percent = sink_start - (sink_start - sink_end) * (elapsed / total_allowed)

        thresh = np.percentile(w, sink_percent)
        sink_mask = w <= thresh
        sink_idx = np.flatnonzero(sink_mask)

        if sink_idx.size == 0:                               # fallback
            sink_idx = np.array([int(np.argmin(w))])
            sink_mask = np.zeros_like(w, dtype=bool)
            sink_mask[sink_idx[0]] = True

        # ---- total “excess’’ contribution that can be removed -------------
        w_min = w[sink_idx].min()
        excess = seq * (w - w_min)                           # excess per position
        total_excess = excess[~sink_mask].sum()              # only non‑sink positions

        if total_excess <= 0:
            # tiny random jitter – helps if we are stuck
            i = np.random.randint(n)
            jitter = (np.random.rand() - 0.5) * seq[i] * 1e-4
            new_seq = seq.copy()
            new_seq[i] = max(0.0, new_seq[i] + jitter)
        else:
            # ---- compute the fraction λ needed to achieve the target reduction
            target_reduction = rel_target_reduce * max_val
            lam = target_reduction / total_excess
            lam = max(min(lam, 1.0), min_lambda)

            # amount taken away from each non‑sink coordinate
            delta = lam * seq
            delta[sink_mask] = 0.0
            moved_mass = delta.sum()
            add_per_sink = moved_mass / sink_idx.size

            new_seq = seq - delta
            new_seq[sink_idx] += add_per_sink

        # ---- enforce feasibility (non‑negative, correct sum) --------------
        new_seq = np.maximum(new_seq, 0.0)
        new_sum = new_seq.sum()
        if new_sum > 0:
            new_seq = new_seq * (total_sum / new_sum)   # renormalise
        else:
            new_seq = seq.copy()                         # extremely unlikely

        new_score = _fast_evaluation(new_seq)

        # ---- accept if we have made progress -------------------------------
        if new_score < best_score - 1e-12:
            seq = new_seq
            best_seq = new_seq.copy()
            best_score = new_score
            print(f"Iter {iteration:6d}: improved score = {best_score:.8f}, "
                  f"max = {max_val:.8f}")
        else:
            # occasional random shake‑up to jump out of flat regions
            if np.random.rand() < 0.02:
                i, j = np.random.choice(n, 2, replace=False)
                if seq[i] > 0:
                    eps = min(seq[i], seq[i] * np.random.rand() * 0.1)
                    tmp = seq.copy()
                    tmp[i] -= eps
                    tmp[j] += eps
                    tmp = tmp * (total_sum / tmp.sum())
                    tmp_score = _fast_evaluation(tmp)
                    if tmp_score < best_score:
                        seq = tmp
                        best_seq = tmp.copy()
                        best_score = tmp_score
                        print(f"Iter {iteration:6d}: random shake improved "
                              f"score = {best_score:.8f}")

        iteration += 1

    return best_seq


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Return a non‑negative step function whose evaluation score is
    (hopefully) well below the classic π/2 bound.
    """
    np.random.seed(seed)
    # keep a small safety margin for the harness (avoid timeout)
    deadline = time.time() + budget_s - 5.0

    # a few thousand entries give a good trade‑off between quality and speed
    n = 5000

    # 1. start from the Catalan / arcsine seed
    seq = _catalan_sequence(n)

    # 2. aggressive yet stable reduction of the maximal convolution entry
    seq = _optimise_aggregated(seq, deadline)

    # 3. return the best sequence discovered (plain Python list)
    return seq.tolist()
