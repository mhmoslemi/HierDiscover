        if s == 0:

        seq = idx.astype(np.float64) ** (-p)

    dir_init = np.random.dirichlet(np.ones(n))
    # Choose whichever yields smaller cheap score

        fft_len = 1 << ((2*n - 1).bit_length())
        Fa = np.fft.fft(a, fft_len)
        conv = np.fft.ifft(Fa * Fa).real[:2*n-1]

    best_a = seq0
    best_score = cheap_score(best_a)
    rnd_score = cheap_score(dir_init)
    if rnd_score < best_score:
        best_a = dir_init
        best_score = rnd_score

    # Set up optimizer params
    fft_len = 1 << ((3*n - 2).bit_length())  # big enough for gradient conv
    len_conv = 2*n - 1

    a = best_a.copy()
    m = np.zeros_like(a)

    beta2 = 0.999
    eps = 1e-12
    lr = 0.2   # base step size
    lam0 = 5.0
    lam_max = 200.0

    best_iter_a = a.copy()
    best_iter_score = best_score
    iteration = 0

    while time.time() < deadline:
        iteration += 1

        # compute convolution a*a
        Fa = np.fft.fft(a, fft_len)
        conv_full = np.fft.ifft(Fa * Fa).real[:len_conv]
        max_conv = conv_full.max()

        # cheap objective
        score = 2 * n * max_conv
        if score < best_iter_score:

        print(f"Final exact score = {final_score:.6f}")

#!/usr/bin/env python
"""
Propose a non‑negative step‑function that minimises the evaluation metric

    2·n·max_k (a*a)[k] / (sum a)^2

using a smooth‑max (log‑sum‑exp) surrogate and Adam optimisation on the
simplex.  The algorithm

* normalises the candidate to sum = 1 (the objective is scale‑invariant);
* repeatedly evaluates the self‑convolution via FFT;
* builds a soft‑max weight vector over the convolution entries;
* obtains an exact gradient of the smooth surrogate by another FFT‑based
  convolution (the gradient is 2·∑_k w_k·a_{k‑i});
* uses Adam with a diminishing learning‑rate and projects onto the
  probability simplex after each step;
* slowly raises the soft‑max temperature λ so that the surrogate approaches
  the true max;
* keeps the best candidate seen (cheap score) and finally evaluates it
  with the provided ``evaluate_sequence`` function.

The routine respects the given time budget (default 1000 s) and prints
progress messages.  It works with an optional global ``height_sequence_1``
(starting point from a previous run) or falls back to a normalised
power‑law initialisation.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project vector v onto the probability simplex {x ≥ 0, sum x = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:          # all entries are non‑positive
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


def _normalise(a: np.ndarray) -> np.ndarray:
    """Scale a non‑negative vector to sum = 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        # fallback to uniform distribution
        return np.full_like(a, 1.0 / a.shape[0])
    return a / s


def _cheap_score(a: np.ndarray, fft_len: int, conv_len: int) -> float:
    """
    Compute 2·n·max_k (a*a)[k] for a vector a that already satisfies sum=1.
    Uses FFT for speed.
    """
    Fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(Fa * Fa).real[:conv_len]
    return 2.0 * a.shape[0] * conv.max()

# Main entry point required by the platform
# ----------------------------------------------------------------------

    Return a list of heights (non‑negative, sum > 0) that (hopefully) beats the
    target bound 1.503.  The routine respects the supplied time budget.

    start_time = time.time()

    except Exception:
        n = 12000                  # default if no previous sequence is present

    # ------------------------------------------------------------------
    def power_law(p: float) -> np.ndarray:
        idx = np.arange(1, n + 1, dtype=np.float64)
        seq = idx ** (-p)
        return _normalise(seq)

    candidates.append(power_law(0.60))

        a_candidate = a - step
        a = _simplex_projection(a_candidate)

        # ---- (e) occasional random perturbation to escape plateaus ------
        if it % 5000 == 0:
            if np.random.rand() < 0.15:
                perturb = 0.01 * np.random.randn(n)
                a = _simplex_projection(a + perturb)

    # ------------------------------------------------------------------
    # 4. Final exact evaluation (uses the provided evaluate_sequence)
    # ------------------------------------------------------------------
    final_seq = best_iter_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:  # defensive – should never happen
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # ------------------------------------------------------------------
    # 5. Return the best sequence found (as a plain Python list)
    # ------------------------------------------------------------------
    return final_seq
```
