import time
import numpy as np

def _fft_convolve(a):
    n = a.shape[0]
    size = 1 << (2 * n - 1).bit_length()
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[: 2 * n - 1]

def _catalan_normalised(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * (i - 1) + 1) / (2.0 * i)
    a /= a.sum()
    return a

def _exp_gradient_step(p, beta, lr):
    """
    Perform one exponentiated‑gradient step that attempts to
    reduce the smooth max of the self‑convolution of ``p``.
    ``p`` must already be normalised to sum = 1.
    Returns the updated probability vector.
    """
    n = p.shape[0]

    # 1. convolution (linear) of p with itself
    conv = _fft_convolve(p)          # length 2n‑1

    # 2. soft‑max weights that concentrate on the large entries
    scaled = conv * beta
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum                    # w is a probability vector over the 2n‑1 sums

    # 3. gradient of the smooth max:
    #    ∂/∂p_i  (∑_k w_k (p∗p)_k) = 2 * Σ_k w_k p_{k‑i}
    #    which is exactly a correlation of p with w.
    #    Compute it with FFT (linear, no wrap‑around).

    # reverse p for correlation
    p_rev = p[::-1]

    # length needed for linear correlation of length‑n and length‑(2n‑1) vectors
    corr_len = n + (2 * n - 1) - 1          # = 3n‑2
    size = 1 << (corr_len).bit_length()    # next power of two

    Fa = np.fft.rfft(p_rev, size)
    Fw = np.fft.rfft(w, size)
    corr_full = np.fft.irfft(Fa * np.conj(Fw), size)      # linear correlation (real)

    # extract the part that corresponds to indices 0 … n‑1
    #   corr[k] = Σ_j p_j w_{j+k},  k = 0 … n‑1
    start = n - 1
    grad = 2.0 * corr_full[start : start + n]

    # 4. exponentiated‑gradient update on the simplex
    p_new = p * np.exp(-lr * grad)
    # Numerical safety – eliminate negative artefacts
    p_new = np.maximum(p_new, 0.0)
    total = p_new.sum()
    if total > 0.0:
        p_new /= total
    else:
        # fallback to a uniform point if everything vanished
        p_new = np.full_like(p_new, 1.0 / n)

    return p_new, conv.max()

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Exponentiated‑gradient optimisation of a step‑function.
    Starts from the classic Catalan‑type distribution (the analytically
    known near‑optimal shape) and performs many cheap FFT‑based updates
    that directly reduce the maximal self‑convolution.  The optimisation
    runs until the time budget expires and returns the best sequence
    discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # small safety margin

    # Choose a length that gives a comfortable speed‑/memory trade‑off.
    n = 8000

    # Initialise with the Catalan‑type law (exactly the same shape used
    # by the reference solution) – it already has a low max‑convolution.
    p = _catalan_normalised(n)

    best_seq = p.copy()
    best_score = 2.0 * n * np.max(_fft_convolve(p))

    # Hyper‑parameters – they are slowly annealed during the run.
    lr = 0.3                      # initial learning rate for the exp‑gradient
    beta = 5.0                    # temperature for the soft‑max (larger → sharper)

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # One optimisation step
        p, cur_max = _exp_gradient_step(p, beta, lr)

        cur_score = 2.0 * n * cur_max   # sum(p) == 1 → metric simplifies
        if cur_score < best_score:
            best_score = cur_score
            best_seq = p.copy()
            print(f"[improve] it {iteration:05d}  score {best_score:.9f}")

        # Simple annealing schedule – a slower step size and a sharper
        # soft‑max as the search proceeds.
        lr *= 0.9995
        beta = min(beta * 1.005, 200.0)

        # Occasionally inject a tiny random perturbation to escape
        # possible plateaus (the probability is tiny and only used when
        # we have plenty of time left).
        if iteration % 250 == 0 and time.time() < deadline - 5.0:
            noise = np.random.rand(n) * 1e-6
            p = p + noise
            p = np.maximum(p, 0.0)
            p /= p.sum()

    # The evaluate_sequence routine expects a plain Python list.
    # The returned vector is already non‑negative and normalised;
    # the harness will apply the full metric (including the L1‑norm)
    # itself.
    final_metric = 2.0 * n * np.max(_fft_convolve(best_seq))
    print(f"[final] length={n}  best score≈{final_metric:.9f}")

    return best_seq.tolist()

    – builds a soft‑max weight vector over the 2·n‑1 convolution entries,
    – obtains the gradient of the smooth max via a correlation (again by FFT),
    – performs an exponentiated‑gradient step on the simplex,
    – slowly anneals the learning‑rate and the soft‑max temperature.
* keeps the best sequence seen before the time budget expires
  and returns it as a plain Python list.

All helper functions are top‑level; no I/O or closures are used.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast linear convolution (real FFT, full linear mode)
# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT.
    Result length = 2*len(a)-1.
    """
    n = a.shape[0]
    size = 1 << (2 * n - 1).bit_length()        # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Catalan‑type coefficients (central binomial) normalised to unit L¹ norm
# ----------------------------------------------------------------------
def _catalan_normalised(n: int) -> np.ndarray:
    """First n coefficients a_i = C(2i,i)/4**i, normalised so that sum = 1.
    The recurrence a_i = a_{i-1} * (2i-1)/(2i) is numerically stable.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * (i - 1) + 1) / (2.0 * i)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# One exponentiated‑gradient step that reduces a smooth approximation of
# the max convolution entry.
# ----------------------------------------------------------------------
def _exp_gradient_step(p: np.ndarray, beta: float, lr: float):
    """
    Perform a single update:
        p ← p · exp(‑lr·grad)   (followed by normalisation)

    The gradient is taken of    L(p)=∑_k w_k (p∗p)_k
    where w_k ∝ exp(beta·(p∗p)_k) is a soft‑max over the convolution.
    This gives a descent direction for the true max when beta→∞.

    Parameters
    ----------
    p : np.ndarray
        Current non‑negative vector, already normalised to sum = 1.
    beta : float
        Temperature of the soft‑max (larger → sharper focus on the true max).
    lr : float
        Learning rate for the exponentiated‑gradient step.

    Returns
    -------
    p_new : np.ndarray
        Updated probability vector (normalised to sum = 1).
    cur_max : float
        The current (non‑smoothed) maximum entry of the convolution.
    """
    n = p.shape[0]

    # 1) convolution of p with itself (full linear mode)
    conv = _fft_convolve(p)                 # length 2n‑1

    # 2) soft‑max weights over the convolution entries
    scaled = conv * beta
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)         # subtract max for numerical stability
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum                          # w is now a probability vector

    # 3) gradient of the smooth objective:
    #    ∂L/∂p_i = 2· Σ_k w_k·p_{k‑i}   (indices out of range are ignored)
    #    This is a correlation of p with w.
    #    Compute it via FFT to stay O(n log n).

    p_rev = p[::-1]                         # reverse for correlation

    # linear correlation length = n + (2n‑1) – 1 = 3n‑2
    corr_len = 3 * n - 2
    fft_len = 1 << corr_len.bit_length()    # next power of two

    Fa = np.fft.rfft(p_rev, fft_len)
    Fw = np.fft.rfft(w, fft_len)

    # correlation = ifft( Fa · conj(Fw) )
    corr_full = np.fft.irfft(Fa * np.conj(Fw), fft_len)

    # extract the part corresponding to indices 0 … n‑1:
    #   corr[k] = Σ_j p_j w_{j+k},   k = 0 … n‑1
    start = n - 1
    grad = 2.0 * corr_full[start:start + n]

    # 4) exponentiated‑gradient update on the simplex
    p_new = p * np.exp(-lr * grad)
    p_new = np.maximum(p_new, 0.0)          # enforce non‑negativity
    total = p_new.sum()
    if total > 0.0:
        p_new /= total
    else:                                   # degenerate – fall back to uniform
        p_new = np.full_like(p_new, 1.0 / n)

    return p_new, conv.max()


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a sequence of non‑negative heights that minimises
    ``evaluate_sequence``.  The routine runs for at most *budget_s* seconds
    and returns the best sequence found.

    The optimisation uses exponentiated‑gradient descent on a smooth
    approximation of the maximum self‑convolution.  It starts from the
    analytically known Catalan‑type distribution and performs many cheap
    FFT‑based iterations until the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # 1) Problem size – a few thousand entries give a good
    #    speed / accuracy trade‑off for the many FFT‑based steps.
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # 2) Initialise with the Catalan‑type law (the state‑of‑the‑art start).
    # ------------------------------------------------------------------
    p = _catalan_normalised(n)

    # Keep the best candidate seen so far.
    best_seq = p.copy()
    best_score = 2.0 * n * np.max(_fft_convolve(p))

    # ------------------------------------------------------------------
    # 3) Hyper‑parameters (they are annealed during the run).
    # ------------------------------------------------------------------
    lr = 0.30          # initial learning rate for the exp‑gradient
    beta = 5.0         # start with a smooth max, sharpen later

    it = 0
    while time.time() < deadline:
        it += 1

        # One optimisation step
        p, cur_max = _exp_gradient_step(p, beta, lr)

        cur_score = 2.0 * n * cur_max    # sum(p) == 1, so metric simplifies
        if cur_score < best_score:
            best_score = cur_score
            best_seq = p.copy()
            print(f"[improve] it {it:05d}  score {best_score:.9f}")

        # Annealing – slower steps and a sharper soft‑max as we progress
        lr *= 0.9995
        beta = min(beta * 1.005, 200.0)

        # Small random jitter every few hundred iterations to avoid
        # stagnation (only applied when we still have time left).
        if it % 250 == 0 and time.time() < deadline - 5.0:
            noise = np.random.rand(n) * 1e-6
            p += noise
            p = np.maximum(p, 0.0)
            p /= p.sum()

    final_metric = 2.0 * n * np.max(_fft_convolve(best_seq))
    print(f"[final] length={n}  best score≈{final_metric:.9f}")

    # The evaluation harness will call ``evaluate_sequence`` on the list.
    return best_seq.tolist()
