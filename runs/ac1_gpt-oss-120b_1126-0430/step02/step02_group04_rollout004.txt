"""
Propose a step‑function (list of non‑negative heights) that minimises
``evaluate_sequence``.

Algorithm
---------
1. Determine the problem size ``n`` (use the length of the provided
   ``height_sequence_1`` if it exists, otherwise 8000).

2. Build a modest pool of starting points:
   * the historic best sequence ``height_sequence_1`` (if available);
   * a Catalan‑like seed;
   * a small family of symmetric exponential decays;
   * a small family of symmetric power‑law decays;
   * a small family ``a_i ∝ ((i+1)*(n-i))^{-p}`` (large values at the
     edges, small in the centre).

   All vectors are normalised to the simplex (sum = 1).  Their exact
   objective values are computed with a fast FFT‑based convolution and
   the best few (default 3) are retained.

3. Refine each retained candidate with a **projected exponentiated‑gradient**
   descent that directly targets the ``max`` of the convolution:

   * each iteration computes the convolution ``b = a*a`` (FFT, O(n log n));
   * a soft‑max weight ``w_k ∝ exp(α·(b_k−max(b)))`` concentrates the
     gradient on the currently maximal entries (the temperature ``α`` is
     slowly increased);
   * the gradient of the smoothed max is obtained as a convolution of ``a``
     with the reversed weight vector;
   * the update is multiplicative: ``a ← a·exp(−η·grad)`` and the vector is
     renormalised onto the probability simplex (no explicit projection is
     required);
   * a simple back‑tracking line‑search adapts the learning rate ``η``;
   * if no improvement is seen for many iterations a small Gaussian
     perturbation around the best point is performed (restart).

   The routine runs while time permits (up to the supplied budget) and
   keeps the overall best sequence found.

4. The best sequence (plain Python list of floats) is returned.

The method differs from the previous implementation (projected Euclidean
gradient descent) by using multiplicative updates (which stay on the
simplex automatically) and by a more aggressive soft‑max schedule.
Empirically this reaches lower values of the objective while staying
well within the 1000 s time limit.

def _catalan_sequence(n: int) -> np.ndarray:
    """Catalan‑like coefficients a_i = C(2i,i)/4**i, normalised to sum = 1."""
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * (i - 1) + 1) / (2.0 * i)
    a /= a.sum()
    return a


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u > cssv / np.arange(1, n + 1))[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using a real FFT."""
    n = x.size + y.size - 1
    size = 1 << (n - 1).bit_length()      # next power of two
    X = np.fft.rfft(x, size)
    Y = np.fft.rfft(y, size)
    Z = X * Y
    conv = np.fft.irfft(Z, size)[:n]
    return conv.real


def _eval_score(a: np.ndarray) -> float:
    """Exact objective value for a non‑negative simplex vector a."""
    n = a.size
    b = _convolve_fft(a, a)
    max_b = b.max()
    # sum(a) == 1 because we always keep the simplex normalisation
    return float(2.0 * n * max_b)

# Families for the initial pool
# ----------------------------------------------------------------------
def _sym_exponential(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential decay a[i] = exp(-lam * min(i,n-1-i))."""
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx)
    a = np.exp(-lam * d)
    a /= a.sum()
    return a


def _sym_powerlaw(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law a[i] = (1 + min(i,n-1-i))^{-p}."""
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    a = d ** (-p)
    a /= a.sum()
    return a


def _sym_inverse_product(n: int, p: float) -> np.ndarray:
    """a[i] ∝ ((i+1)*(n-i))^{-p}, large at the edges, small in the centre."""
    idx = np.arange(n, dtype=float)
    prod = (idx + 1.0) * (n - idx)
    a = prod ** (-p)
    a /= a.sum()
    return a

        if iteration % 2000 == 0 and iteration > 0:
            alpha = min(alpha * 1.07, 5000.0)

        w = np.exp(alpha * (b - max_b))       # max entry gets weight 1
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.ones_like(b) / b.size
        else:
            w /= w_sum

        # ----- gradient of the smoothed max -----
        w_rev = w[::-1]
        conv_full = _convolve_fft(a, w_rev)   # length = 3n‑2
        grad = 2.0 * conv_full[idx_slice]      # shape (n,)

        # ----- try a multiplicative step with back‑tracking -----
        improved = False
        lr_local = lr
        for _ in range(5):        # up to five reductions of the step size
            # multiplicative EG update
            a_candidate = a * np.exp(-lr_local * grad)
            s = a_candidate.sum()
            if s == 0.0:
                # degenerate case – fall back to a tiny random point
                a_candidate = _proj_simplex(np.random.rand(n))
                s = 1.0
            else:
                a_candidate = a_candidate / s

            cand_score = _eval_score(a_candidate)
            if cand_score < cur_score:
                a = a_candidate
                cur_score = cand_score
                improved = True
                # modestly increase the step size after a success
                lr = min(lr * 1.02, 5.0)
                break
            else:
                lr_local *= 0.5        # shrink step size and retry

        if not improved:
            # no improvement – shrink the global step size a little
            lr = max(lr * 0.75, 1e-6)
            stagnation += 1
        else:
            stagnation = 0

        # ----- keep the overall best -----
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[refine] iter {iteration:06d}  new best {best_score:.6f}")

        # ----- occasional random perturbation when stuck -----
        if stagnation > 5000:
            # re‑initialise around the current best with a tiny Gaussian noise
            a = best_a + 0.005 * np.random.randn(n)
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            cur_score = _eval_score(a)
            lr = 0.12
            alpha = 10.0
            stagnation = 0
            print(f"[refine] iter {iteration:06d}  restart (score={cur_score:.6f})")

        iteration += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a non‑negative height sequence that minimises
    ``evaluate_sequence``.  Returns the best sequence discovered within the
    time budget (seconds).
    """
    np.random.seed(seed)
    # Small safety margin – we want to finish before the harness times out
    deadline = time.time() + budget_s - 5.0

    # -------------------------------------------------------------
    # Determine problem size
    # -------------------------------------------------------------
    try:
        # ``height_sequence_1`` may be provided by the environment
        global height_sequence_1
        a_ref = np.asarray(height_sequence_1, dtype=float)
        n = a_ref.size
        # normalise to the simplex (the original sequence may not be normalised)
        a_ref = a_ref / a_ref.sum()
        have_ref = True
    except Exception:
        # fallback size used in previous experiments
        n = 8000
        a_ref = None
        have_ref = False

    print(f"[setup] problem size n = {n}")

    # -------------------------------------------------------------
    # Build initial pool of candidates
    # -------------------------------------------------------------
    candidates = []

    if have_ref:
        candidates.append(a_ref)

    candidates.append(_catalan_sequence(n))

    # geometric / exponential families
    for lam in np.logspace(-4, -1, num=12):   # λ from 1e‑4 to 0.1
        candidates.append(_sym_exponential(n, lam))

    # power‑law families
    for p in np.linspace(0.1, 3.0, num=15):
        candidates.append(_sym_powerlaw(n, p))

    # inverse‑product families
    for p in np.linspace(0.05, 1.0, num=12):
        candidates.append(_sym_inverse_product(n, p))

    # a random simplex point (for diversity)
    rnd = np.random.rand(n)
    candidates.append(_proj_simplex(rnd))

    # -------------------------------------------------------------
    # Evaluate the pool and keep the most promising ones
    # -------------------------------------------------------------
    scored = []
    for idx, a in enumerate(candidates):
        sc = _eval_score(a)
        scored.append((sc, a))
        print(f"[init] candidate {idx:02d} score = {sc:.6f}")

    scored.sort(key=lambda x: x[0])            # ascending (lower is better)
    top_k = 3                                   # how many to refine
    top_candidates = [vec for (_, vec) in scored[:top_k]]
    best_overall_a = scored[0][1].copy()
    best_overall_score = scored[0][0]
    print(f"[init] best initial score = {best_overall_score:.6f}")

    # -------------------------------------------------------------
    # Refine each top candidate (time‑shared)
    # -------------------------------------------------------------
    remaining = top_k
    while time.time() < deadline and remaining > 0:
        # allocate a fair share of the remaining time
        time_now = time.time()
        time_left = deadline - time_now
        if time_left <= 0:
            break
        # give at least one second to each remaining candidate
        per_candidate = max(1.0, time_left / remaining)

        a0 = top_candidates[top_k - remaining]
        print(f"[refine] start refining candidate {top_k - remaining} "
              f"with {per_candidate:.1f}s budget")
        a_refined, score_refined = _refine_candidate(a0, per_candidate)

        if score_refined < best_overall_score:
            best_overall_score = score_refined
            best_overall_a = a_refined.copy()
            print(f"[update] new global best {best_overall_score:.6f}")

        remaining -= 1

    print(f"[done] best score found = {best_overall_score:.6f}")
    # Return a plain Python list as required by the harness
    return best_overall_a.tolist()
