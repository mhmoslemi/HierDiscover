def _loss_and_grad_lp(a, p, N_fft):
    n = a.shape[0]
    # FFT of a
    A = np.fft.fft(a, N_fft)
    # convolution a*a
    H = A * A
    h_full = np.fft.ifft(H).real[:2 * n - 1]
    # Lp norm
    # compute sum of powers
    h_pow = h_full ** p
    sum_hp = np.sum(h_pow)
    # To avoid zero sum (shouldn't), but handle
    if sum_hp == 0:
        loss = 0.0
        grad = np.zeros_like(a)
        return loss, grad
    norm_lp = sum_hp ** (1.0 / p)
    # compute gradient dL/dh
    factor = norm_lp ** (1.0 - p)  # = sum_hp^{(1/p)-1}
    dLdh = factor * (h_full ** (p - 1.0))
    # FFT of dLdh
    G = np.fft.fft(dLdh, N_fft)
    # conv term: a * dLdh truncated to first n
    conv_full = np.fft.ifft(A * G).real
    conv_term = conv_full[:n]  # first n entries
    # cross term: rev_a * dLdh
    rev_a = a[::-1]
    revA = np.fft.fft(rev_a, N_fft)
    cross_full = np.fft.ifft(revA * G).real
    # cross term entries correspond n-1...n-1+n-1
    start = n - 1
    cross_term = cross_full[start:start + n]
    grad = conv_term + cross_term
    # loss = norm_lp (the Lp norm)
    return norm_lp, grad

#!/usr/bin/env python
"""
A fast “smooth‑Lp + Adam” optimiser for the step‑function problem.

Algorithm
---------
1. Choose a working length *n* (≈ 8000) and pre‑compute a power‑of‑two FFT size
   that is large enough for all convolutions (length ≈ 3 n).
2. Build a small pool of promising initial sequences.
   - Power‑law a_i ∝ (i+1)^{-p} optionally multiplied by an exponential cut‑off
     exp(-λ i).  Grid‑search over p∈[0.5,0.8] and λ∈{0,2e‑4,5e‑4,1e‑3,2e‑3}.
   - The previously found sequence (``height_sequence_1``) is also added.
   For each candidate we compute the cheap bound 2 n·max(a⋆a) via FFT and keep
   the best few (default 5).
3. Refine every retained candidate with a projected Adam optimiser.
   • The loss is the L_p‑norm of the full auto‑convolution (p grows from 4 to 12).
   • The gradient of the L_p‑norm is obtained analytically and evaluated by
     FFT (≈ 3 FFTs per iteration).  After each Adam step the vector is projected
     onto the probability simplex.
   • The cheap bound 2 n·max(a⋆a) is monitored; the best iterate is kept.
4. The best refined vector is finally evaluated with the supplied
   ``evaluate_sequence``.  If time permits a tiny random‑perturbation hill‑climb
   is performed to try to squeeze a little more improvement.
5. The function returns the best list of non‑negative heights found within the
   time budget (default 1000 s).
"""

import time
import numpy as np
from math import sqrt

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                       # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _convolution_max(a: np.ndarray, fft_len: int) -> float:
    """Maximum entry of a⋆a computed via FFT (full length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa).real[: 2 * a.size - 1]
    return float(conv.max())


def _loss_and_grad_lp(a: np.ndarray, p: float, fft_len: int):
    """
    Compute the L_p norm of the full auto‑convolution and its gradient w.r.t. ``a``.
    Returns (loss, grad) where loss = (∑ (a⋆a)_k^p)^{1/p}.
    """
    n = a.shape[0]
    # FFT of a
    A = np.fft.fft(a, fft_len)
    # Full auto‑convolution h = a⋆a (length 2n‑1)
    H = A * A
    h = np.fft.ifft(H).real[: 2 * n - 1]

    # L_p norm
    hp = h ** p
    sum_hp = np.sum(hp)
    if sum_hp == 0.0:
        loss = 0.0
        grad = np.zeros_like(a)
        return loss, grad
    norm_lp = sum_hp ** (1.0 / p)

    # dL/dh = (norm_lp)^{1-p} * h^{p-1}
    factor = norm_lp ** (1.0 - p)
    dh = factor * (h ** (p - 1.0))

    # FFT of dL/dh (same padded length)
    G = np.fft.fft(dh, fft_len)

    # First term: (a * dh) truncated to first n entries
    conv_full = np.fft.ifft(A * G).real
    term1 = conv_full[:n]

    # Second term: (rev(a) * dh) evaluated at indices n‑1 … n‑1+n‑1
    rev_a = a[::-1]
    revA = np.fft.fft(rev_a, fft_len)
    cross_full = np.fft.ifft(revA * G).real
    start = n - 1
    term2 = cross_full[start:start + n]

    grad = term1 + term2
    return norm_lp, grad


def _refine_sequence(a_init: np.ndarray, n: int, fft_len: int,
                     deadline: float, max_iters: int = 2000):
    """
    Refine ``a_init`` using Adam on the smooth L_p surrogate.
    Returns the best iterate (according to the cheap bound 2 n·max(a⋆a))
    and its cheap score.
    """
    a = a_init.copy()
    best_a = a.copy()
    best_score = float('inf')
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1

        # cheap bound
        A = np.fft.fft(a, fft_len)
        h_full = np.fft.ifft(A * A).real[: 2 * n - 1]
        maxb = float(h_full.max())
        cheap_score = 2.0 * n * maxb

        if cheap_score < best_score:
            best_score = cheap_score
            best_a = a.copy()
            print(f"    [Iter {t:05d}] new cheap best = {cheap_score:.6f}")

        # smoothly increasing p (4 → 12)
        p = 4.0 + 8.0 * (t / max_iters)

        # gradient of the L_p norm
        loss, grad = _loss_and_grad_lp(a, p, fft_len)

        # Adam parameters
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # step size (fixed learning‑rate factor)
        step = 0.05 * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # projection back onto the simplex
        a = _simplex_projection(a)

        # occasional random jitter to escape plateaus
        if t % 500 == 0 and np.random.rand() < 0.2:
            jitter = 0.01 * np.random.rand(n)
            a = _simplex_projection(a + jitter)

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) beats the target bound.
    The routine respects the supplied time budget.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0  # safety margin

    # --------------------------------------------------------------
    # 0. Choose a working length
    # --------------------------------------------------------------
    try:
        # Use the length of a previously discovered construction if it exists
        n_prev = len(height_sequence_1)  # type: ignore[name-defined]
        n = max(5000, min(n_prev, 12000))
    except Exception:
        n = 8000
    n = max(2000, n)                       # safeguard

    # FFT length sufficient for all convolutions (a⋆a, a⋆g, rev(a)⋆g)
    fft_len = _next_pow2(3 * n)

    # --------------------------------------------------------------
    # 1. Build a small pool of candidates
    # --------------------------------------------------------------
    candidates = []   # list of (cheap_score, sequence_array)
    best_overall = float('inf')

    # include the previously found sequence if available
    try:
        seq0 = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if seq0.shape[0] > n:
            seq0 = seq0[:n]
        elif seq0.shape[0] < n:
            seq0 = np.concatenate([seq0,
                                   np.full(n - seq0.shape[0], 1e-12, dtype=np.float64)])
        seq0 = seq0 / seq0.sum()
        cheap0 = 2.0 * n * _convolution_max(seq0, fft_len)
        candidates.append((cheap0, seq0.copy()))
        best_overall = min(best_overall, cheap0)
        print(f"[Init] Existing sequence cheap score = {cheap0:.6f}")
    except Exception:
        pass

    # grids for the parametric family a_i ∝ (i+1)^{-p}·exp(-λ·i)
    p_vals = np.linspace(0.5, 0.8, 7)                      # 0.5 … 0.8
    lam_vals = [0.0, 2e-4, 5e-4, 1e-3, 2e-3]              # exponential cut‑offs
    idx = np.arange(n, dtype=np.float64)

    for p in p_vals:
        for lam in lam_vals:
            if time.time() >= deadline:
                break
            seq = (idx + 1) ** (-p) * np.exp(-lam * idx)
            total = seq.sum()
            if total <= 0:
                continue
            seq /= total
            cheap = 2.0 * n * _convolution_max(seq, fft_len)
            candidates.append((cheap, seq.copy()))
            if cheap < best_overall:
                best_overall = cheap
        if time.time() >= deadline:
            break

    print(f"[Init] Grid search best cheap score = {best_overall:.6f}")

    # keep only the top few candidates for refinement
    candidates.sort(key=lambda x: x[0])
    top_k = 5
    top_candidates = candidates[:top_k]

    # --------------------------------------------------------------
    # 2. Refine each retained candidate
    # --------------------------------------------------------------
    best_seq = None
    best_seq_score = float('inf')
    remaining = len(top_candidates)

    for idx, (init_score, seq) in enumerate(top_candidates, 1):
        # allocate a tiny slice of the remaining budget per candidate
        cur_deadline = deadline - (remaining - 1) * 1.0
        if time.time() >= cur_deadline:
            break
        print(f"[Refine] Candidate {idx}/{len(top_candidates)} init cheap = {init_score:.6f}")
        refined_seq, refined_score = _refine_sequence(seq, n, fft_len,
                                                      cur_deadline, max_iters=2500)
        if refined_score < best_seq_score:
            best_seq_score = refined_score
            best_seq = refined_seq.copy()
            print(f"    -> New best after refinement: cheap = {refined_score:.6f}")
        remaining -= 1

    # fallback – should never happen
    if best_seq is None:
        best_seq = top_candidates[0][1]
        best_seq_score = top_candidates[0][0]

    # --------------------------------------------------------------
    # 3. Exact evaluation of the best refined candidate
    # --------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)  # provided by the platform
        print(f"[Result] Exact score = {final_score:.6f}")
    except Exception as exc:
        final_score = float('inf')
        print(f"[Error] Evaluation failed: {exc}")

    # --------------------------------------------------------------
    # 4. Optional tiny random hill‑climb if time permits
    # --------------------------------------------------------------
    if time.time() < deadline:
        a = np.array(final_seq, dtype=np.float64)
        best_exact = final_score
        best_exact_seq = a.copy()
        # a few cheap random moves with immediate exact re‑evaluation
        for attempt in range(200):
            if time.time() >= deadline:
                break
            i, j = np.random.randint(n, size=2)
            delta = (np.random.rand() - 0.5) * 0.01   # up to ±0.005 mass shift
            a_new = a.copy()
            a_new[i] = max(0.0, a_new[i] + delta)
            a_new[j] = max(0.0, a_new[j] - delta)
            a_new = _simplex_projection(a_new)
            score_new = evaluate_sequence(a_new.tolist())
            if score_new < best_exact:
                best_exact = score_new
                best_exact_seq = a_new.copy()
                a = a_new
                print(f"    [HillClimb] Improved exact score = {score_new:.6f}")
        final_seq = best_exact_seq.tolist()
        final_score = best_exact

    # --------------------------------------------------------------
    # 5. Return the best sequence found
    # --------------------------------------------------------------
    return final_seq
