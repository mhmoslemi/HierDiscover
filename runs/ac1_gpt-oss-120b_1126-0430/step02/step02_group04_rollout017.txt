"""
Propose a candidate step‑function (list of non‑negative heights) that minimises
the supplied ``evaluate_sequence`` function.

Algorithm
---------
* Problem size is fixed to n = 8000 (fast FFT‑based convolutions).
* A small pool of diverse initial points is created:
    - uniform,
    - the Catalan‑like sequence,
    - several beta‑shaped families (a_i ∝ (i+1)^{α‑1}(n‑i)^{α‑1}),
    - several sine‑power families (a_i ∝ sin(π·(i+½)/n)^p),
    - a few random Dirichlet samples.
  The best of those (exact objective) is kept as the starting point.
* Optimisation is performed with a projected *exponentiated‑gradient* /
  *additive‑gradient* method.
  At each iteration the convolution b = a∗a is computed, the current maximum
  index k_max is located and the exact sub‑gradient of the max‑functional is
  obtained (this is simply a shifted copy of ``a``).  Two candidate updates are
  generated:
        – additive: a ← Π_simplex(a − η·grad)
        – multiplicative: a ← Π_simplex(a·exp(−η·grad))
  (Π_simplex is the Euclidean projection onto the probability simplex.)
  The better one (according to the true objective) is accepted.
* The learning rate η is decreased geometrically; if there is no improvement
  for a while the algorithm is re‑seeded with a small random perturbation.
* The loop runs until the time budget (≈ 1000 s) is exhausted; the best
  sequence found is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, sum=1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    # Fast projection (Duchi et al., 2008)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u > cssv / np.arange(1, n + 1))[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of real vectors x and y using a real FFT."""
    n = x.size + y.size - 1
    size = 1 << (n - 1).bit_length()      # next power of two
    X = np.fft.rfft(x, size)
    Y = np.fft.rfft(y, size)
    Z = X * Y
    conv = np.fft.irfft(Z, size)[:n]
    return conv.real


def _eval_score(a: np.ndarray) -> float:
    """Exact value of evaluate_sequence for a non‑negative simplex vector a."""
    n = a.size
    b = _convolve_fft(a, a)
    max_b = b.max()
    return float(2.0 * n * max_b / (a.sum() ** 2))


# ----------------------------------------------------------------------
# Families of candidate shapes
# ----------------------------------------------------------------------


def _catalan_sequence(n: int) -> np.ndarray:
    """Catalan‑like seed, normalised to sum 1."""
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * (i - 1) + 1) / (2.0 * i)
    a /= a.sum()
    return a


def _beta_shape(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta‑type shape: a_i ∝ (i+1)^{α‑1}(n‑i)^{α‑1}."""
    i = np.arange(n, dtype=float)
    a = (i + 1.0) ** (alpha - 1.0) * (n - i) ** (alpha - 1.0)
    a /= a.sum()
    return a


def _sin_power_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ sin(π·(i+½)/n)^p, normalised."""
    i = np.arange(n, dtype=float)
    a = np.sin(np.pi * (i + 0.5) / n) ** p
    a /= a.sum()
    return a


def _random_dirichlet(n: int, conc: float) -> np.ndarray:
    """Random point on the simplex (Dirichlet with concentration conc)."""
    # np.random.gamma works for conc>0; when conc<1 distribution is spiky.
    g = np.random.gamma(conc, 1.0, size=n)
    a = g / g.sum()
    return a


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a non‑negative height sequence that minimises ``evaluate_sequence``.
    Returns the best sequence found before the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0  # safety margin

    # --------------------------------------------------------------
    # problem size
    # --------------------------------------------------------------
    n = 8000

    # --------------------------------------------------------------
    # build a pool of starting points
    # --------------------------------------------------------------
    candidates = [_catalan_sequence(n)]

    # uniform
    candidates.append(np.full(n, 1.0 / n))

    # beta families (symmetric, alpha in [0.3, 3] – logarithmic sweep)
    for alpha in np.logspace(np.log10(0.3), np.log10(3.0), num=12):
        candidates.append(_beta_shape(n, alpha))

    # sine‑power families (p in [0.2, 4])
    for p in np.linspace(0.2, 4.0, num=12):
        candidates.append(_sin_power_shape(n, p))

    # a few random Dirichlet samples (concentration 0.3 … 3)
    for conc in [0.3, 0.7, 1.0, 2.0, 3.0]:
        for _ in range(3):
            candidates.append(_random_dirichlet(n, conc))

    # --------------------------------------------------------------
    # pick the best initial candidate (exact evaluation)
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    for a in candidates:
        sc = _eval_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
    a = best_seq.copy()
    cur_score = best_score
    print(f"[init] n={n}, starting score={cur_score:.6f}")

    # --------------------------------------------------------------
    # optimisation parameters
    # --------------------------------------------------------------
    eta = 0.6                     # initial step size (learning rate)
    eta_min = 1e-6
    eta_decay = 0.97
    beta1, beta2 = 0.9, 0.999    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t_iter = 0
    no_improve = 0
    max_no_improve = 8000

    # --------------------------------------------------------------
    # main optimisation loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        # ----- current convolution and maximum index -----
        b = _convolve_fft(a, a)
        max_b = b.max()
        # there may be several entries attaining the max – pick one at random
        max_idxs = np.flatnonzero(b >= max_b - 1e-15)
        k_max = np.random.choice(max_idxs)

        # ----- exact sub‑gradient of the max‑functional -----
        # create a one‑hot weight vector for the selected k_max
        w = np.zeros_like(b)
        w[k_max] = 1.0
        w_rev = w[::-1]
        conv_full = _convolve_fft(a, w_rev)
        # slice to obtain gradient of size n
        idx = (len(b) - 1) - np.arange(n)
        grad = 2.0 * conv_full[idx]

        # ----- Adam momentum update -----
        t_iter += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)
        step = eta * m_hat / (np.sqrt(v_hat) + 1e-10)

        # ----- generate two candidates and keep the better -----
        improved = False

        # 1) additive projected gradient
        a_add = _proj_simplex(a - step)
        sc_add = _eval_score(a_add)

        # 2) multiplicative (exponentiated) update
        # avoid overflow/underflow by limiting exponent magnitude
        exp_factor = np.exp(-step)
        a_mul = a * exp_factor
        a_mul = _proj_simplex(a_mul)
        sc_mul = _eval_score(a_mul)

        # pick the best of the two
        if sc_add < sc_mul:
            a_candidate = a_add
            cand_score = sc_add
        else:
            a_candidate = a_mul
            cand_score = sc_mul

        if cand_score < cur_score:
            a = a_candidate
            cur_score = cand_score
            improved = True
            # record global best
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{t_iter:07d}] new best {best_score:.6f} (η={eta:.4g})")
                no_improve = 0
        else:
            no_improve += 1

        # ----- adapt learning rate -----
        if not improved:
            eta = max(eta * eta_decay, eta_min)

        # ----- occasional restart if stuck too long -----
        if no_improve > max_no_improve:
            # small random perturbation around the current best
            a = best_seq + 0.005 * np.random.randn(n)
            a = np.maximum(a, 0.0)
            a = _proj_simplex(a)
            cur_score = _eval_score(a)
            m.fill(0.0)
            v.fill(0.0)
            t_iter = 0
            eta = 0.6
            no_improve = 0
            print(f"[{t_iter:07d}] restart around best (score={cur_score:.6f})")

        # safety check – if learning rate is extremely low, break early
        if eta < 1e-8:
            break

    print(f"[done] best score = {best_score:.6f}")
    # return the best sequence as a plain list of floats
    return best_seq.tolist()
