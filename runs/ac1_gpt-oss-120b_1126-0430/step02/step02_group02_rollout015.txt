#!/usr/bin/env python
"""
Propose a non‑negative step‑function (list of heights) with a low evaluation
score.

Algorithm
---------
1. Choose a working length `n` (default 5000).  The length appears in the
   objective `2·n·max(a*a)`, so we keep it moderate to avoid inflating the
   bound.
2. Initialise a population of candidate shapes using a two‑parameter family

       a_i ∝ (i+1)^(-p) · exp(-λ·i) ,   p∈[0.4,0.8] , λ∈[0,0.002]

   The vector is normalized to sum to 1 and the cheap score
   `2·n·max(conv)` (computed via FFT) is recorded.  The best few candidates
   are kept.
3. Refine the best candidate with projected gradient descent on a *smooth*
   approximation of the max‑convolution:

       f_β(a) = Σ_k w_k·(a*a)_k ,   w_k ∝ exp(β·((a*a)_k – max(a*a)))

   The gradient of `f_β` is the correlation of `w` with `a`.  We use an
   increasing schedule for β (5 → 160) and a diminishing step size
   η_t = η₀ / sqrt(t).  After each gradient step the vector is projected back
   onto the probability simplex.
4. Perform a few short runs from random Dirichlet starts to escape possible
   local minima.
5. Keep the best vector (lowest cheap score) and evaluate it with the
   provided ``evaluate_sequence`` function for the final exact score.
6. Return the best sequence as a plain ``list[float]``.

All heavy work (convolutions) is performed with FFTs, so the routine easily
fits into the 1000 s time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – top level, no closures
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the probability simplex:
        Σ_i x_i = 1 ,  x_i ≥ 0 .
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # all entries non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _cheap_score(a: np.ndarray, n: int, fft_len: int) -> float:
    """
    Fast evaluation of the objective
        2·n·max((a*a)) / (sum(a)^2)

    The convolution is performed by FFT; the function works for any `a`
    (it is *scale‑invariant* – we keep the vector as a simplex, i.e.
    sum(a)=1, but the formula works for any scaling).
    """
    s = a.sum()
    if s <= 0:
        return float("inf")
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa).real[: 2 * n - 1]
    maxb = conv.max()
    return float(2.0 * n * maxb / (s * s))


def _gradient_smooth_max(a: np.ndarray,
                         beta: float,
                         n: int,
                         fft_len: int,
                         fft_len2: int) -> tuple[float, np.ndarray]:
    """
    Compute the smooth approximation of the max‑convolution and its gradient.

    Parameters
    ----------
    a : np.ndarray
        Current coefficient vector (non‑negative, not necessarily normalized).
    beta : float
        Smoothing parameter – larger values bring the surrogate closer to the
        true max.
    n : int
        Length of `a`.
    fft_len : int
        Power‑of‑two size for the self‑convolution (≥ 2·n‑1).
    fft_len2 : int
        Power‑of‑two size for the correlation needed for the gradient
        (≥ 3·n‑2).

    Returns
    -------
    f : float
        The smooth surrogate value `Σ w_k·conv_k`.
    g : np.ndarray
        Gradient of `f` w.r.t. `a` (same shape as `a`).
    """
    # --- self‑convolution (full) ---------------------------------
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa).real[: 2 * n - 1]

    # --- soft‑max weights -----------------------------------------
    m = conv.max()
    # shift by the max to keep the exponent bounded (≤ 0)
    w = np.exp(beta * (conv - m))
    w_sum = w.sum()
    if w_sum == 0:
        w_norm = np.full_like(w, 1.0 / w.size)
    else:
        w_norm = w / w_sum

    # --- gradient = 2 * correlation(w_norm, a) --------------------
    # correlation: (G_i) = 2 * Σ_k w_k·a_{k-i}
    w_rev = w_norm[::-1]                     # reverse for linear convolution
    fw_rev = np.fft.fft(w_rev, fft_len2)
    fa2 = np.fft.fft(a, fft_len2)
    corr = np.fft.ifft(fa2 * fw_rev).real
    # linear convolution length = n + (2n‑1) – 1 = 3n‑2
    corr = corr[: 3 * n - 2]
    # index mapping: G_i = 2 * corr[(2n‑2) – i]
    G = 2.0 * corr[(2 * n - 2) - np.arange(n)]

    # smooth surrogate value (for monitoring)
    f = float(np.dot(w_norm, conv))

    return f, G

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) beats the target
    bound 1.503.  The routine respects the given time budget.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0          # safety margin for final eval

    # ------------------------------------------------------------------
    # 0. Working length -------------------------------------------------
    # The bound grows roughly like 2·n·max (a*a).  A moderate n gives a better
    # chance to beat 1.503 while keeping each iteration cheap.
    # ------------------------------------------------------------------
    n = 5000

    # FFT sizes (next power of two) ------------------------------------
    fft_len = 1 << ((2 * n - 1).bit_length())      # for self‑convolution
    fft_len2 = 1 << ((3 * n - 2).bit_length())     # for correlation (gradient)

    # ------------------------------------------------------------------
    # 1. Coarse grid search over a two‑parameter family
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    # parameter ranges (empirically useful)
    p_grid = np.linspace(0.45, 0.75, 31)           # power‑law exponent
    lam_grid = np.linspace(0.0, 0.002, 21)        # exponential decay rate

    idx = np.arange(n, dtype=np.float64)

    for p in p_grid:
        for lam in lam_grid:
            if time.time() >= deadline:
                break
            seq = (idx + 1) ** (-p) * np.exp(-lam * idx)
            total = seq.sum()
            if total <= 0:
                continue
            seq /= total
            sc = _cheap_score(seq, n, fft_len)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
        if time.time() >= deadline:
            break

    # Also test pure exponential (p = 0)
    for lam in lam_grid[1:]:
        if time.time() >= deadline:
            break
        seq = np.exp(-lam * idx)
        total = seq.sum()
        if total <= 0:
            continue
        seq /= total
        sc = _cheap_score(seq, n, fft_len)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()

    # If a previously discovered construction exists, try it as well
    try:
        if height_sequence_1:                     # global from previous runs
            old = np.array(height_sequence_1, dtype=np.float64)
            # Resize / interpolate to our working length if needed
            if old.size != n:
                old = np.interp(
                    np.linspace(0, 1, n),
                    np.linspace(0, 1, old.size),
                    old,
                )
            old = np.maximum(old, 0.0)
            old /= old.sum()
            sc = _cheap_score(old, n, fft_len)
            if sc < best_score:
                best_score = sc
                best_seq = old.copy()
    except Exception:
        pass

    if best_seq is None:
        # Fallback – uniform distribution (should never happen)
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _cheap_score(best_seq, n, fft_len)

    print(f"Initial grid search: best cheap score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Refine with smoothed‑max gradient descent (projected onto simplex)
    # ------------------------------------------------------------------
    a = best_seq.copy()
    a = a / a.sum()                     # ensure exact simplex
    best_iter_seq = a.copy()
    best_iter_score = best_score

    # schedule of (beta, steps) – beta grows, sharpening the approximation
    beta_schedule = [(5,   2000),
                     (10,  2000),
                     (20,  2000),
                     (40,  2000),
                     (80,  2000),
                     (160, 2000)]

    total_iters = sum(steps for _, steps in beta_schedule)
    iter_counter = 0
    eta0 = 0.5

    for beta, steps in beta_schedule:
        for _ in range(steps):
            if time.time() >= deadline:
                break
            iter_counter += 1

            # compute smooth surrogate and gradient
            _, grad = _gradient_smooth_max(a, beta, n, fft_len, fft_len2)

            # diminishing step size
            eta = eta0 / math.sqrt(iter_counter)

            # gradient step + projection onto simplex
            a_candidate = a - eta * grad
            a = _simplex_projection(a_candidate)
            a = a / a.sum()                         # numerics

            # occasional cheap evaluation
            if iter_counter % 500 == 0:
                sc = _cheap_score(a, n, fft_len)
                if sc < best_iter_score:
                    best_iter_score = sc
                    best_iter_seq = a.copy()
                    print(f"[Iter {iter_counter:06d}] new best cheap = {sc:.6f}")

                # occasional jitter to escape plateaus
                if iter_counter % 2000 == 0 and np.random.rand() < 0.2:
                    jitter = 0.02 * np.random.rand(n)
                    a = _simplex_projection(a + jitter)

        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # 3. A few short runs from random Dirichlet starts (optional)
    # ------------------------------------------------------------------
    rnd_restarts = 4
    rnd_iters = 800
    for r in range(rnd_restarts):
        if time.time() >= deadline:
            break
        # Dirichlet with small concentration → sparse vectors
        rnd = np.random.gamma(0.5, 1.0, size=n)
        a_rand = rnd / rnd.sum()
        for t in range(rnd_iters):
            if time.time() >= deadline:
                break
            # use a modest beta for the short run
            _, grad = _gradient_smooth_max(a_rand, beta=30.0, n=n,
                                          fft_len=fft_len, fft_len2=fft_len2)
            eta = 0.1 / math.sqrt(t + 1)
            a_rand = _simplex_projection(a_rand - eta * grad)
            a_rand = a_rand / a_rand.sum()
        sc = _cheap_score(a_rand, n, fft_len)
        if sc < best_iter_score:
            best_iter_score = sc
            best_iter_seq = a_rand.copy()
            print(f"[Random restart {r}] new best cheap = {sc:.6f}")

    # ------------------------------------------------------------------
    # 4. Final exact evaluation
    # ------------------------------------------------------------------
    final_seq = best_iter_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # Return the best sequence (as a plain Python list)
    return final_seq
