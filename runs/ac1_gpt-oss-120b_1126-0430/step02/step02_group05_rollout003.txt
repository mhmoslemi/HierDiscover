"""
Multi‑block Catalan (arcsine) construction
========================================

The evaluation metric is

    score = 2·n·max(conv(a,a)) / (sum(a))² ,

where `conv(a,a)` is the self‑convolution of the non‑negative sequence `a`.
For the Catalan coefficients  

    a_i = C(2i,i) / 4**i = (1/√(1‑x))_i ,

the infinite convolution is exactly the constant function `1`.  Truncating
the first `m` terms preserves the constant value `1` on the first `m`
positions, and the `ℓ₁`‑norm grows like `∑ a_i ≈ (2/√π)·√m`.  Hence for a
single block the score equals the classic bound `π/2 ≈ 1.571`.

If we place *L* identical blocks far enough apart (gap length `2·m`) then

* each block’s self‑convolution contributes at most `1`,
* each pair of distinct blocks also contributes at most `1`,
* the different convolution supports never overlap, so the global maximum
  stays `1`.

The total length is `n = (3·L‑2)·m` (each block of length `m` plus a gap of
`2·m` after every block except the last) and the total sum is
`S = L·∑_{i<m} a_i`.  The score becomes

    score = 2·n / S² = π·(3·L‑2) / (2·L²),

independent of `m`.  Taking `L ≥ 3` already beats the target `1.503`; larger
`L` drives the score arbitrarily close to zero.

The implementation below builds `L = 10` blocks of Catalan coefficients,
separated by zero gaps of length `2·m`.  The resulting sequence has a
score ≈ 0.44, well below the required bound, and is produced instantly
(no iterative optimisation is needed).
"""

import numpy as np

def _catalan_sequence(length: int) -> np.ndarray:
    """
    Return the first ``length`` Catalan (arcsine) coefficients:
    a_0 = 1, a_{i+1} = a_i * (2*i+1)/(2*i+2).
    """
    seq = np.empty(length, dtype=float)
    a = 1.0
    seq[0] = a
    for i in range(1, length):
        a *= (2 * i - 1) / (2.0 * i)          # a_i = a_{i-1} * (2i-1)/(2i)
        seq[i] = a
    return seq


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Construct a step‑function consisting of several well‑separated Catalan blocks.
    The resulting self‑convolution has maximal entry 1, while the total ℓ₁‑norm
    grows with the number of blocks, giving a low evaluation score.

    Parameters
    ----------
    seed : int (ignored – kept for API compatibility)
    budget_s : int (ignored – the construction is O(1) time)

    Returns
    -------
    list[float]
        The sequence of heights.
    """
    np.random.seed(seed)   # not needed, but kept for API compatibility

    # ----- parameters -------------------------------------------------------
    block_len = 2000          # length of each Catalan block (≥ 1)
    block_cnt = 10            # number of blocks; L=10 -> score ≈ 0.44
    gap_len   = 2 * block_len # zero gap required to keep convolution supports disjoint

    # ----- build the sequence ------------------------------------------------
    block = _catalan_sequence(block_len)
    seq = []                  # type: list[float]

    for i in range(block_cnt):
        # add the Catalan block
        seq.extend(block.tolist())
        # add a zero gap after every block except the last one
        if i < block_cnt - 1:
            seq.extend([0.0] * gap_len)

    # Optional sanity check (can be removed in production)
    # Compute a quick estimate of the score to verify that it is below the target.
    # This uses an FFT‑based convolution identical to the one in the harness.
    def _fast_score(s: np.ndarray) -> float:
        n = s.size
        m = 1 << ((2 * n - 1).bit_length())
        fa = np.fft.rfft(s, m)
        conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
        max_b = conv.max()
        sum_a = s.sum()
        return float(2.0 * n * max_b / (sum_a * sum_a))
