Propose a candidate step‑function (list of non‑negative heights) that minimises the
``evaluate_sequence`` objective.

Algorithm
---------
* Choose a moderate problem size (default 12 000).  The size is kept small enough
  for many FFT‑based iterations yet large enough to give a fine resolution.
* Build a diverse pool of starting points:
  – Catalan‑like coefficients (the classic seed).
  – A family of exponential decays.
  – Random points on the simplex.
  – If the global variable ``height_sequence_1`` exists, use it (truncated /
    padded to the chosen size) as an additional seed.
* Keep the best seed (exact objective computed by ``_eval_score``) as the initial
  iterate.
* Optimise by *projected accelerated gradient descent* on the simplex.
  – The gradient of the smooth maximum of the auto‑convolution is
    ``g = 2·(a ∗ w_rev)`` where ``w`` are soft‑max weights on the convolution
    entries.
  – In the first phase (warm‑up) a soft‑max with a temperature ``α`` is used;
    ``α`` is slowly increased so that the optimisation gradually focuses on the
    true maximum.
  – After a configurable number of warm‑up iterations the weights become a hard
    indicator of the maximal convolution entries (sub‑gradient of the true
    max).  This yields a sharp descent in the final stage.
  – Momentum (Nesterov/Polyak) is added and the iterate is projected back onto
    the probability simplex after each step (Euclidean projection, Duchi et al.).
  – If a step does not improve the objective the step size is halved and the
    momentum is reset; periodic decay of the learning rate is also performed.
  – When the algorithm stagnates for many iterations, a random perturbation
    around the current best point restarts the search.
* The loop runs until the time budget (≈ 1000 s) is exhausted; the best
  sequence seen is returned as a plain Python list.

The implementation uses only NumPy and the provided ``evaluate_sequence``
function; no file or network I/O is performed.

# Helper utilities (top‑level, no closures)

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, sum=1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    # fast O(n log n) projection (Duchi et al. 2008)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u > cssv / np.arange(1, n + 1))[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of real vectors using a real FFT (fast)."""
    n = x.size + y.size - 1
    size = 1 << (n - 1).bit_length()          # next power of two
    X = np.fft.rfft(x, size)
    Y = np.fft.rfft(y, size)
    Z = X * Y
    conv = np.fft.irfft(Z, size)[:n]
    return conv.real


def _eval_score(a: np.ndarray) -> float:
    """
    Exact value of ``evaluate_sequence`` for a non‑negative simplex vector ``a``.
    ``a`` is assumed to sum to 1 (the scale is irrelevant for the objective).
    """
    n = a.size
    b = _convolve_fft(a, a)
    max_b = b.max()
    return float(2.0 * n * max_b)            # because sum(a)==1


def _catalan_sequence(n: int) -> np.ndarray:
    """Catalan‑like coefficients a_i = C(2i,i)/4**i, normalised to sum = 1."""
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * (i - 1) + 1) / (2.0 * i)
    a /= a.sum()
    return a


def _softmax_weights(b: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max (temperature ``alpha``) on a vector ``b``."""
    max_b = b.max()
    w = np.exp(alpha * (b - max_b))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.ones_like(b) / b.size
    else:
        w /= w_sum
    return w


def _indicator_weights(b: np.ndarray, tol_factor: float = 1e-12) -> np.ndarray:
    """Hard indicator of entries that attain the (near) maximum of ``b``."""
    max_b = b.max()
    tol = max_b * tol_factor
    mask = (b >= max_b - tol).astype(float)
    w_sum = mask.sum()
    if w_sum == 0.0:
        mask = np.ones_like(b) / b.size
    else:
        mask /= w_sum
    return mask


def _gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the (smoothed) maximum of ``a∗a`` w.r.t. ``a``.
    Implements  g = 2·(a ∗ w_rev)  restricted to the original length.
    """
    w_rev = w[::-1]
    conv_full = _convolve_fft(a, w_rev)                 # length = 3n‑2
    idx = (len(w) - 1) - np.arange(a.size)              # pick needed slice
    return 2.0 * conv_full[idx]


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a non‑negative height sequence that minimises
    ``evaluate_sequence``.  Returns the best sequence found before the time
    budget expires.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Overall time budget in seconds.

    Returns
    -------
    list[float]
        The best step‑function height sequence discovered.
    """
    np.random.seed(seed)
    # give a small safety margin for the harness
    deadline = time.time() + budget_s - 5.0

    # ------------------------------------------------------------------
    # 1. Problem size ----------------------------------------------------
    # We pick a size that allows many fast FFTs yet is large enough to
    # approach the asymptotic regime.  12 000 points fit in a 2^15 FFT.
    n = 12000

    # ------------------------------------------------------------------
    # 2. Build a pool of starting points ---------------------------------
    candidates = [_catalan_sequence(n)]

    # exponential (geometric) decays with different rates
    for p in np.linspace(0.5, 4.0, 9):          # effective support factor
        eps = p / n
        r = 1.0 - eps
        a = eps * (r ** np.arange(n))
        a /= a.sum()
        candidates.append(a)

    # random simplex point
    rnd = np.random.rand(n)
    candidates.append(_proj_simplex(rnd))

    # optional: use the previously discovered construction if available
    if 'height_sequence_1' in globals():
        try:
            hs = np.asarray(height_sequence_1, dtype=float)
            if hs.size >= n:
                a0 = hs[:n].copy()
            else:
                # pad with a tiny uniform background to keep sum>0
                pad = np.full(n - hs.size, 0.0)
                a0 = np.concatenate([hs, pad])
            a0 = np.maximum(a0, 0.0)
            a0 = _proj_simplex(a0)
            candidates.append(a0)
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 3. Pick the best seed according to the exact objective -------------
    best_seq = None
    best_score = np.inf
    for a in candidates:
        sc = _eval_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
    a = best_seq.copy()
    cur_score = best_score
    print(f"[init] n={n}, initial best score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Optimisation hyper‑parameters -----------------------------------
    lr = 0.22                     # initial step size (before projection)
    momentum = 0.92               # Polyak momentum factor
    v = np.zeros_like(a)          # velocity vector

    alpha = 8.0                   # soft‑max temperature (will be increased)
    warmup_iters = 15000          # number of iterations using the smooth weight
    iter_no_improve = 0
    max_no_improve = 8000         # trigger a restart after this many stale iters

    iteration = 0
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # --------------------------------------------------------------
        #   a) compute convolution and weights
        # --------------------------------------------------------------
        b = _convolve_fft(a, a)
        max_b = b.max()

        if iteration < warmup_iters:
            # soft‑max weighting (gradually sharpened)
            w = _softmax_weights(b, alpha)
            # slowly increase temperature to focus on the true max
            if iteration % 2500 == 0 and iteration > 0:
                alpha = min(alpha * 1.45, 2500.0)
        else:
            # hard indicator – true sub‑gradient of the max
            w = _indicator_weights(b, tol_factor=1e-12)

        # --------------------------------------------------------------
        #   b) gradient and projected momentum step
        # --------------------------------------------------------------
        grad = _gradient(a, w)

        # Polyak momentum update
        v = momentum * v - lr * grad
        a_candidate = _proj_simplex(a + v)

        cand_score = _eval_score(a_candidate)

        # --------------------------------------------------------------
        #   c) acceptance test
        # --------------------------------------------------------------
        if cand_score < cur_score:
            a = a_candidate
            cur_score = cand_score
            # keep momentum (it helped to reach a better point)
            iter_no_improve = 0
            # record global best
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{iteration:06d}] new best {best_score:.6f} "
                      f"(lr={lr:.4g}, α={alpha:.2f})")
        else:
            # reject – roll back momentum and shrink step size
            v = np.zeros_like(a)          # reset velocity
            lr *= 0.73                     # aggressive decay when stuck
            iter_no_improve += 1

        # --------------------------------------------------------------
        #   d) periodic housekeeping
        # --------------------------------------------------------------
        # gentle decay of the learning rate (prevents it from blowing up)
        if iteration % 700 == 0 and iteration > 0:
            lr = max(lr * 0.93, 1e-5)

        # restart if we have been idle for a while
        if iter_no_improve >= max_no_improve:
            # perturb the current best and continue from there
            pert = best_seq + 0.015 * np.random.randn(n)
            pert = np.maximum(pert, 0.0)
            a = _proj_simplex(pert)
            cur_score = _eval_score(a)
            v = np.zeros_like(a)
            lr = 0.22
            alpha = 8.0
            iter_no_improve = 0
            print(f"[{iteration:06d}] restart (score={cur_score:.6f})")

        iteration += 1

    # ------------------------------------------------------------------
    # 5. Final reporting -------------------------------------------------
    print(f"[done] best score discovered = {best_score:.6f}")
    # Convert to plain Python list as required by the harness
    return best_seq.tolist()
