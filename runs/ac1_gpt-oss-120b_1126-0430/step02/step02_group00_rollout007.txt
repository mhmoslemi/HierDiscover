"""
Hybrid optimiser for the step‑function minimisation problem.

We wish to minimise
      O(a) = 2·n·max(conv(a,a))   with   a_i ≥ 0 ,  Σ a_i = 1 .
The objective is scale‑invariant, so the search is performed on the
probability simplex.

Method
------
1. **Adam on a smooth surrogate** – a log‑sum‑exp (soft‑max) approximation of
   the max‑convolution.  The sequence is parametrised by a log‑softmax
   (x → softmax(x)), which automatically satisfies the simplex constraints.
   An Adam optimiser (with bias‑correction) updates the parameters while the
   temperature α of the LSE surrogate is annealed from a small to a large
   value.  The best simplex vector seen is stored.

2. **Focused local refinement** – starting from the best vector, repeated
   random restarts perform cheap “mass‑transfer’’ moves guided by the true
   sub‑gradient of the max‑convolution entry:
        * move a fraction of probability mass from the index with the
          largest sub‑gradient (most responsible for the current peak) to the
          index with the smallest sub‑gradient (ideally zero);
        * accept moves that improve the exact objective, otherwise accept
          with a small Metropolis probability (temperature decays);
        * every few steps a plain sub‑gradient descent step and a brief
          diffusion (averaging with neighbours) are applied.
   The best sequence over all restarts is kept.

The whole routine respects a user‑supplied time budget (default 1000 s) and
returns the best sequence found as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex {x≥0, Σx=1}
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex."""
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        # fallback to uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full self‑convolution (length 2·n‑1)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return the full linear convolution a*a using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Exact objective for a simplex‑normalised vector
def _objective(a: np.ndarray, n: int, L: int) -> float:
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Sub‑gradient of the maximal convolution entry (as in the reference code)
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Derivative of max(conv) w.r.t. a (non‑negative sub‑gradient)."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)          # typically a single index
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    g /= len(ks)      # average if several maximisers
    return g


# ----------------------------------------------------------------------
# Gradient of the soft‑max (log‑sum‑exp) approximation of the max
def _smooth_gradient(a: np.ndarray, conv: np.ndarray,
                     alpha: float, L: int) -> np.ndarray:
    """
    Compute ∇_a ( (2·n/α)·log Σ_k exp(α·conv_k) ).
    The result equals 2·corr(w, a) where w_k ∝ exp(α·conv_k).
    """
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)          # stabilise
    w = exp_scaled / exp_scaled.sum()

    # correlation w ⋆ a   (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)   # length ≥ 3·n‑2
    n = a.size
    grad = 2.0 * corr_rev[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """Search for a non‑negative step‑function minimising the evaluation metric."""
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # Initialise from a previous construction if available
    if 'height_sequence_1' in globals():
        base = np.array(globals()['height_sequence_1'], dtype=float)
        n = base.size
        a0 = base / base.sum()                     # normalise to simplex
        print(f"[init] using existing sequence, n={n}")
    else:
        n = 4000
        a0 = np.full(n, 1.0 / n)                    # uniform start
        print(f"[init] no prior sequence – using uniform, n={n}")

    # FFT length – must be ≥ 3·n‑2 for the correlation used in the smooth gradient
    L = 1 << ((3 * n - 2).bit_length())            # next power of two

    # --------------------------------------------------------------
    # Phase 1 – Adam optimisation on the LSE surrogate
    best_seq = a0.copy()
    best_score = _objective(best_seq, n, L)
    print(f"[phase‑1] start O={best_score:.6f}")

    # Soft‑max temperature schedule
    alpha_start = 5.0
    alpha_end = 2000.0
    max_iter_phase1 = 250_000
    iter1 = 0
    phase1_deadline = start_time + budget_s * 0.55

    # Adam parameters (operating in the log‑softmax domain)
    eps_adam = 1e-12
    lr = 0.05
    beta1, beta2 = 0.9, 0.999
    m = np.zeros_like(a0)
    v = np.zeros_like(a0)

    # Parameterisation a = softmax(x)
    x = np.log(a0 + 1e-12)          # avoid log(0)

    while time.time() < phase1_deadline and iter1 < max_iter_phase1:
        # ----- simplex vector from logits -----
        shift = np.max(x)                     # for numerical stability
        exps = np.exp(x - shift)
        a = exps / exps.sum()

        # ----- convolution & objective -----
        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[{time.time():.1f}] phase‑1 new best O={best_score:.6f} (iter={iter1})")

        # ----- anneal α -----
        frac = min(1.0, iter1 / max_iter_phase1)
        alpha = alpha_start + (alpha_end - alpha_start) * frac

        # ----- smooth gradient (scaled to the true objective) -----
        grad_smooth = _smooth_gradient(a, conv, alpha, L)      # 2·corr
        grad = 2.0 * n * grad_smooth                          # dO/da (≈4n·corr)

        # ----- map to logits (softmax) -----
        avg = np.dot(grad, a)
        grad_x = a * (grad - avg)          # ∂O/∂x_i = a_i (∂O/∂a_i - Σ a_j ∂O/∂a_j)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** (iter1 + 1))
        v_hat = v / (1.0 - beta2 ** (iter1 + 1))

        step = lr / (np.sqrt(v_hat) + eps_adam)

        x -= step * m_hat
        # keep logits in a reasonable range to avoid overflow
        x = np.clip(x, -50.0, 50.0)

        iter1 += 1

    # --------------------------------------------------------------
    # Phase 2 – local hill‑climbing & sub‑gradient refinement
    print(f"[phase‑2] entering local refinement, best O={best_score:.6f}")

    max_iter_per_restart = 150_000
    temp0 = 0.02                     # initial Metropolis temperature
    subgrad_interval = 7
    diffusion_interval = 13
    no_improve_limit = 5_000

    while time.time() < deadline:
        # ------- initialise a new restart -------
        if np.random.rand() < 0.5:
            a = best_seq.copy()
            a += np.random.randn(n) * 0.003
            a = np.clip(a, 0, None)
            a = _project_to_simplex(a)
        else:
            a = np.random.rand(n)
            a = _project_to_simplex(a)

        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        iter_local = 0
        no_improve = 0
        T = temp0

        while time.time() < deadline and iter_local < max_iter_per_restart:
            # ---------------------------------------------------------
            # guided mass‑transfer move based on the sub‑gradient
            subgrad = _subgradient_max_conv(a, conv)          # ≥0
            i = int(np.argmax(subgrad))                       # most “dangerous’’ index
            j = int(np.argmin(subgrad))                       # safest destination

            if a[i] > 1e-12:
                # move up to 15 % of the mass at i (randomly scaled)
                delta = a[i] * np.random.rand() * 0.15
                delta = min(delta, a[i])

                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta
                a_new = np.clip(a_new, 0, None)
                a_new = _project_to_simplex(a_new)

                conv_new = _conv_fft(a_new, L)
                new_obj = 2.0 * n * conv_new.max()

                # Metropolis acceptance
                if new_obj < cur_obj or np.random.rand() < math.exp((cur_obj - new_obj) / max(T, 1e-12)):
                    a = a_new
                    conv = conv_new
                    cur_obj = new_obj
                    no_improve = 0
                    if cur_obj < best_score:
                        best_score = cur_obj
                        best_seq = a.copy()
                        print(f"[{time.time():.1f}] New global best O={best_score:.6f} (restart, iter={iter_local})")
                else:
                    no_improve += 1
            else:
                no_improve += 1

            # ---------------------------------------------------------
            # occasional diffusion (averaging with neighbours)
            if iter_local % diffusion_interval == 0:
                a_smooth = (a +
                            np.roll(a, 1) +
                            np.roll(a, -1)) / 3.0
                a = _project_to_simplex(a_smooth)
                conv = _conv_fft(a, L)
                cur_obj = 2.0 * n * conv.max()
                if cur_obj < best_score:
                    best_score = cur_obj
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] Diffusion new best O={best_score:.6f}")

            # ---------------------------------------------------------
            # sub‑gradient descent step on the current maximal entry
            if iter_local % subgrad_interval == 0:
                g = _subgradient_max_conv(a, conv) * (2.0 * n)   # gradient of O
                eta = 0.07 * (1.0 - iter_local / max_iter_per_restart)  # decreasing step
                a = a - eta * g
                a = _project_to_simplex(a)
                conv = _conv_fft(a, L)
                cur_obj = 2.0 * n * conv.max()
                if cur_obj < best_score:
                    best_score = cur_obj
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] Sub‑grad new best O={best_score:.6f}")

            # ---------------------------------------------------------
            # temperature decay
            T = temp0 * (1.0 - iter_local / max_iter_per_restart)

            # restart if stuck too long
            if no_improve >= no_improve_limit:
                break

            iter_local += 1

        # next restart – loop continues until overall budget expires
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Final evaluation with the provided function (sanity check)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:
        print("[final] evaluate_sequence failed:", e)

    return best_seq.tolist()
