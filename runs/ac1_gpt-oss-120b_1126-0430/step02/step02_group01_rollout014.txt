"""
Propose a step‑function (non‑negative sequence) that minimizes

    O = 2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm works on the simplex (∑a = 1) – the objective is
scale‑invariant, so fixing the sum to 1 does not affect the score.
It combines three ideas:

1. **Fast incremental convolution updates** – a move that transfers a
   small amount `δ` from index `i` to `j` changes the self‑convolution
   in a closed form.  Updating the convolution by slicing a NumPy
   array is O(n) instead of recomputing the whole FFT each step.

2. **Simulated‑annealing with targeted mass transfers** – most iterations
   either transfer a random amount of mass or, more aggressively,
   relocate mass from the index that contributes most to the current
   maximal convolution peak.  A Metropolis acceptance rule with a
   slowly cooling temperature enables exploration while still favouring
   improvements.

3. **Projected sub‑gradient descent** – every few hundred iterations a
   sub‑gradient of the peak of the convolution is taken, the step is
   projected back onto the probability simplex and accepted if it
   improves (or with a small temperature‑dependent probability).

The routine respects the 1000 s budget, prints periodic progress, and
finally returns the best sequence found as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all entries are zero
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv(a,a)) w.r.t. a (a lives on the simplex)."""
    n = a.size
    conv = np.convolve(a, a)
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    g /= len(ks)
    return g
