"""
Propose a step function that dramatically improves the bound.

The key observation is that the evaluation
    2 * n * max(conv(a, a)) / (sum(a)^2)
is invariant under scaling, so we can restrict to sequences whose sum is 1.
For a monotone sequence a_i ∝ (i+1)^p (or the reversed version) the
convolution at shift k is analytically
    conv[k] = L(k)·c·(k+1)^p   with L(k) the overlap length.
The maximal entry of the convolution is either the central term
(a_i·a_{n-1-i}) or the square of the largest coefficient a_max^2.
Choosing a moderate power p≈5 makes a_max ≈(p+1)/n so that
    2·n·a_max^2 ≈ 2·(p+1)^2 / n  ≈ 0.03   (for n≈2244),
far below the previous best ~1.57.  The algorithm therefore:

1.  Determines the problem size n (the length of the provided
    ``height_sequence_1`` or a fallback).
2.  Evaluates a small family of polynomial‑decay candidates
    a_i ∝ (i+1)^p and the reversed version for p∈{0.5,1,…,10}.
    The best of these is kept.
3.  Performs a cheap hill‑climbing refinement that moves a tiny amount of
    mass from the index contributing most to the current maximal
    convolution value to the index with the smallest coefficient.
    This can only improve the already excellent candidate.
4.  Returns the best sequence found.

The routine respects the 1000 s time budget, uses only top‑level
functions, and prints progress for debugging.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top level – no closures)
# ----------------------------------------------------------------------


def _norm_and_score(raw):
    """
    Normalise a non‑negative raw vector to sum = 1, evaluate it with the
    official ``evaluate_sequence`` and return (score, normalised_vector).
    """
    total = raw.sum()
    if total <= 0:
        return float("inf"), None
    norm = raw / total
    score = evaluate_sequence(norm.tolist())
    return score, norm

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed=123, budget_s=950, **kwargs):
    """
    Generate a step‑function (list of heights) that minimises the provided
    ``evaluate_sequence`` value.

    The search consists of:
      * a quick sweep over polynomial families a_i ∝ (i+1)^p and their
        reversed counterpart,
      * a lightweight hill‑climbing refinement that moves a small amount
        of mass from the most “responsible’’ index to the least.
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # --------------------------------------------------------------
    # Determine the length of the problem
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)  # provided by the harness
    except Exception:
        n = 2244  # fallback length

    # --------------------------------------------------------------
    # Baseline – the sequence that is always available
    # --------------------------------------------------------------
    try:
        best_seq = list(height_sequence_1)
        best_score = evaluate_sequence(best_seq)
    except Exception:
        best_seq = np.full(n, 1.0 / n).tolist()
        best_score = evaluate_sequence(best_seq)

    print(f"[init] length = {n}, baseline score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 1. Sweep polynomial families a_i ∝ (i+1)^p  (increasing shape)
    #    and the reversed version a_i ∝ (n-i)^p.
    # --------------------------------------------------------------
    exponents = np.arange(0.5, 11.0, 0.5)  # 0.5, 1.0, …, 10.5
    idx_fwd = np.arange(1, n + 1, dtype=float)          # 1 … n
    idx_rev = np.arange(n, 0, -1, dtype=float)          # n … 1

    for p in exponents:
        if time.time() - start > budget_s - 10:
            break

        # increasing shape
        raw_inc = np.power(idx_fwd, p)
        score, norm = _norm_and_score(raw_inc)
        if score < best_score:
            best_score, best_seq = score, norm.tolist()
            print(f"[poly inc p={p:.1f}] improved to {score:.6f}")

        # decreasing (reversed) shape

        score, norm = _norm_and_score(raw_dec)
        if score < best_score:

            print(f"[poly dec p={p:.1f}] improved to {score:.6f}")

    # --------------------------------------------------------------
    # 2. Light hill‑climbing refinement
    # --------------------------------------------------------------
    cur = np.array(best_seq, dtype=float)
    cur_score = best_score
    cur_conv = np.convolve(cur, cur)

    max_iters = 150_000
    it = 0
    while time.time() - start < budget_s - 5 and it < max_iters:
        it += 1

        # index of the current maximal convolution entry
        max_k = int(np.argmax(cur_conv))

        # contribution of each index i to conv[max_k]
        i = np.arange(n)
        j = max_k - i
        mask = (j >= 0) & (j < n)
        contrib = np.zeros_like(cur)
        contrib[mask] = cur[i[mask]] * cur[j[mask]]

        src = int(np.argmax(contrib))                # biggest contributor
        tgt = int(np.argmin(cur))                     # smallest coefficient

        if src == tgt or cur[src] <= 0:

        step = cur[src] * rng.random() * 0.04        # ≤4 % of src mass
        if step <= 0:

        new = cur.copy()
        new[src] -= step
        new[tgt] += step

        # Re‑normalise to avoid drift (tiny numerical error)
        total = new.sum()

        new_score = evaluate_sequence(new.tolist())

        # Accept only improvements (very conservative)

        # Occasionally re‑project onto the simplex for safety
        if it % 5000 == 0:
            # a simple projection: just renormalise (all entries ≥0)
            cur = cur / cur.sum()
            cur_conv = np.convolve(cur, cur)

    print(f"[final] best score found: {best_score:.6f}")
    return best_seq
