def _lp_step(f, M):
    import scipy.sparse as sp
    from scipy.optimize import linprog

    n = f.shape[0]
    # Build constraint matrix A_ub of shape (2*n-1, n)
    # For each i (0..n-1), entries f[i] along diagonal offset i.
    # Use sp.diags with offsets to create matrix sums?
    # Using sp.diags, we can produce a banded matrix:
    # But sp.diags expects each diag to be length n, with diag offset i, zero-padded for rows beyond length.
    # However, we need shape (2n-1,n). Use sp.diags with shape argument.
    diags = [f] * n   # list of arrays each of length n we will treat as diag
    offsets = np.arange(n)  # diag offsets 0,...,n-1
    A = sp.diags(diags, offsets, shape=(2*n-1, n), format='csr')
    # This creates matrix where column j has f[i] on row i+j.
    b = np.full(2*n-1, M)
    # objective - maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    # bounds for variables
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        return None
    g0 = res.x
    return g0

def _create_blocks(n, B):
    block_sizes = [n // B] * B
    for i in range(n % B):
        block_sizes[i] += 1
    blocks = []
    start = 0
    for size in block_sizes:
        end = start + size
        blocks.append((start, end))
        start = end
    return blocks

def _lp_refine(a, M, blocks, L):
    import scipy.sparse as sp
    from scipy.optimize import linprog

    n = a.shape[0]
    B = len(blocks)

    # compute FFT of a once
    fft_a = np.fft.rfft(a, n=L)

    # allocate A matrix dense (2n-1, B)
    rows = 2 * n - 1
    A = np.empty((rows, B), dtype=np.float64)
    for idx, (st, en) in enumerate(blocks):
        mask = np.zeros(n, dtype=np.float64)
        mask[st:en] = 1.0
        conv = np.fft.irfft(fft_a * np.fft.rfft(mask, n=L), n=L)
        conv = conv[:rows]
        block_len = en - st
        A[:, idx] = conv / block_len

    b = np.full(rows, M, dtype=np.float64)
    c = -np.ones(B, dtype=np.float64)  # maximize sum(y).
    bounds = [(0, None)] * B
    # Solve linear program
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        return None
    y = res.x
    # Build full g from block masses y:
    g = np.empty_like(a)
    for idx, (st, en) in enumerate(blocks):
        block_len = en - st
        if block_len > 0:
            g[st:en] = y[idx] / block_len
        else:
            g[st:en] = 0.0
    sum_g = g.sum()
    if sum_g <= 0:
        return None
    g = g / sum_g
    return g
```

        # so lower bound max(0, k - (en-1)), upper bound min(n-1, k - st)

        lo = np.maximum(lo, 0)
        hi = np.minimum(hi, n - 1)

        sums = np.zeros(rows, dtype=np.float64)

        lo_valid = lo[valid]

    n = a.shape[0]
    rows = 2 * n - 1
    # Build constraint matrix A (rows x B)
    A = np.empty((rows, len(blocks)), dtype=np.float64)
    cum = np.concatenate(([0.0], np.cumsum(a)))
    for idx, (st, en) in enumerate(blocks):
        Lb = en - st
        # compute sums via vectorized approach
        k = np.arange(rows)
        lo = k - (en - 1)
        hi = k - st
        lo = np.maximum(lo, 0)
        hi = np.minimum(hi, n-1)
        # valid entries where lo <= hi
        valid = lo <= hi
        sums = np.zeros(rows, dtype=np.float64)
        # vectorized indexing
        hi_v = hi[valid]
        lo_v = lo[valid]
        sums[valid] = cum[hi_v + 1] - cum[lo_v]
        A[:, idx] = sums / Lb

        Lb = en - st
        g[st:en] = y[idx] / Lb
    sum_g = g.sum()

"""
Propose a non‑negative step‑function sequence that minimises

    O(a) = 2·n·max(conv(a,a))   (a is normalised so that sum(a)=1)

Algorithm
---------
1.  Initialise from a previously found sequence (if any) or from the uniform
    distribution.  The vector is projected onto the probability simplex.

2.  Phase 1 – smooth optimisation
    * Optimise the soft‑max (log‑sum‑exp) surrogate of the max‐convolution
      with projected gradient descent.  The smoothing parameter α is raised
      exponentially from a small value to a large one.
    * The gradient of the surrogate is obtained by a correlation (FFT‑based
      O(N log N) operation).  The step size is scaled so that the maximal
      absolute update stays ≤ 0.3.
    * Every few thousand iterations a block‑wise linear‑programming step
      (see below) is performed to push the distribution towards a flatter
      self‑convolution.

3.  Block‑wise LP refinement
    * The current distribution f is fixed.  We look for a new distribution
      g (in a reduced block representation) that satisfies the linear
      constraints   (f * g)_k ≤ M   for every k, where M = max(conv(f,f)).
      The objective is to maximise the total mass of g; after solving we
      renormalise g to sum to 1.
    * The block representation reduces the LP size dramatically:
      the interval [0,n) is split into B≈200 contiguous blocks.
      For each block the convolution (f * block‑indicator) is computed
      with a simple cumulative‑sum formula, giving a dense matrix
      A∈ℝ^{(2n−1)×B}.  The LP is then
          maximise    1ᵀy
          subject to  A·y ≤ M·1 ,   y≥0
      where y are the block masses.  The solution y is expanded to a full‑size
      vector g (uniform inside each block) and mixed with f:
          f ← (1−t)·f + t·g   (t≈0.15).

4.  Phase 2 – local hill‑climbing + sub‑gradient polishing
    * Random “mass‑transfer’’ moves: move a fraction of the probability
      from one index to another; accept if the true objective improves,
      otherwise accept with a tiny Metropolis probability.
    * Every few iterations a sub‑gradient step for the current maximal
      convolution entry is taken (the same direction used in the reference
      code) with a decreasing step size.
    * The best sequence seen so far is always retained.

5.  The function respects the 1000 s budget, prints occasional status
    messages and finally returns the best sequence as a plain Python list.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x≥0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:                         # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full self‑convolution (length 2·n−1)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Smooth (log‑sum‑exp) surrogate gradient
def _smooth_gradient(a: np.ndarray, conv: np.ndarray, alpha: float, L: int) -> np.ndarray:
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)          # stabilise
    w = exp_scaled / exp_scaled.sum()                 # soft‑max weights

    # correlation w ⋆ a   (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)   # length ≥ 3·n−2
    n = a.size
    grad = 2.0 * corr_rev[n - 1 : n - 1 + n]          # central n entries
    return grad


# ----------------------------------------------------------------------
# Sub‑gradient of the maximal convolution entry (as used in the reference code)
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)   # typically a single index
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    g /= len(ks)          # average if several maximisers
    return g


# ----------------------------------------------------------------------
# Create contiguous blocks for the LP refinement
def _create_blocks(n: int, B: int):
    """Return a list of (start, end) pairs (end exclusive) giving B blocks."""
    base = n // B
    extra = n % B
    blocks = []
    start = 0
    for i in range(B):
        size = base + (1 if i < extra else 0)
        end = start + size
        blocks.append((start, end))
        start = end
    return blocks


# ----------------------------------------------------------------------
# Build the constraint matrix A for the block‑wise LP.
# Each column corresponds to one block; A_{k, b} = (f * indicator_{block b})_k / |block b|
def _build_block_constraint_matrix(f: np.ndarray, blocks):
    n = f.shape[0]
    rows = 2 * n - 1
    B = len(blocks)
    A = np.empty((rows, B), dtype=np.float64)
    # prefix sums of f for O(1) interval sums
    pref = np.concatenate(([0.0], np.cumsum(f)))   # length n+1
    k = np.arange(rows, dtype=np.int64)

    for idx, (st, en) in enumerate(blocks):
        Lb = en - st
        # i must satisfy 0 ≤ i ≤ n‑1 and st ≤ k‑i ≤ en‑1
        lo = k - (en - 1)          # lower bound for i
        hi = k - st                # upper bound for i
        lo = np.maximum(lo, 0)
        hi = np.minimum(hi, n - 1)
        valid = lo <= hi
        sums = np.zeros(rows, dtype=np.float64)
        # using prefix sums: sum_{i=lo..hi} f_i = pref[hi+1] - pref[lo]
        lo_v = lo[valid]
        hi_v = hi[valid]
        sums[valid] = pref[hi_v + 1] - pref[lo_v]
        A[:, idx] = sums / Lb
    return A


# ----------------------------------------------------------------------
# One LP refinement step (block‑wise)
def _lp_refine_block(a: np.ndarray, M: float, blocks):
    """Return a new distribution g (full length) obtained from the LP step.
    If the LP fails, None is returned."""
    n = a.shape[0]
    rows = 2 * n - 1
    A = _build_block_constraint_matrix(a, blocks)   # shape (2n‑1, B)
    b = np.full(rows, M, dtype=np.float64)
    c = -np.ones(len(blocks), dtype=np.float64)      # maximise sum(y)
    bounds = [(0, None)] * len(blocks)

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        return None
    y = res.x
    # expand y to a full‑size vector (uniform inside each block)
    g = np.empty_like(a)
    for idx, (st, en) in enumerate(blocks):
        Lb = en - st
        if Lb > 0:
            g[st:en] = y[idx] / Lb
        else:
            g[st:en] = 0.0
    total = g.sum()
    if total <= 0:
        return None
    return g / total
