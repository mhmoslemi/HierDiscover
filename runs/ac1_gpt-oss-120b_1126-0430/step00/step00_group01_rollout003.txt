def project_simplex(v):
    # ensure non-negative and sum to 1
    # Use algorithm by Duchi et al., 2008.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w

import numpy as np
import time

def compute_score(a):
    n = len(a)
    b = np.convolve(a, a)
    max_b = b.max()
    sum_a = a.sum()
    return 2.0 * n * max_b / (sum_a * sum_a)

def project_simplex(v):
    # Duchi et al.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def gradient_of_max(a):
    b = np.convolve(a, a)
    k = int(np.argmax(b))
    n = len(a)
    idx = k - np.arange(n)
    mask = (idx >= 0) & (idx < n)
    grad = np.zeros_like(a)
    grad[mask] = 2.0 * a[idx[mask]]

import numpy as np
import time

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------
def compute_score(a: np.ndarray) -> float:
    """
    Exact replica of evaluate_sequence (without the safety caps) for speed.
    a must be a 1‑D numpy array of non‑negative numbers.
    """
    n = a.shape[0]
    b = np.convolve(a, a)              # full convolution, length 2·n‑1
    max_b = b.max()
    s = a.sum()
    # the original evaluator caps sum to be ≥0.01; our a always sums to 1.
    return 2.0 * n * max_b / (s * s)


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.  Implements the algorithm of
    Duchi et al., ICML 2008.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]                # descending sort
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    # Find the largest ρ satisfying the KKT condition
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def gradient_of_max(a: np.ndarray):
    """
    Returns the gradient of the currently maximal convolution entry.
    For the index k = argmax (a∗a) the gradient is
        g_i = 2·a_{k−i}  (if 0 ≤ k−i < n) else 0.
    Also returns k and the full convolution vector (for possible debugging).
    """
    b = np.convolve(a, a)
    k = int(np.argmax(b))
    n = a.shape[0]
    idx = k - np.arange(n)                      # potential partner positions
    mask = (idx >= 0) & (idx < n)
    g = np.zeros_like(a)
    g[mask] = 2.0 * a[idx[mask]]
    return g, k, b


def arcsine_distribution(n: int) -> np.ndarray:
    """
    Discrete approximation of the Beta(1/2,1/2) (arcsine) law.
    Its self‑convolution is (asymptotically) uniform, which minimises
    the evaluation function.
    """
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n                         # mid‑point of each cell in [0,1]
    raw = 1.0 / np.sqrt(x * (1.0 - x))        # 1/√(x(1−x))
    a = raw / raw.sum()                       # normalise to Σ a_i = 1
    return a

                best_local_a = a_candidate

        # --------------------------------------------------------------
        # Accept the best local move (always if it improves,
        # otherwise with a small Metropolis probability)
        # --------------------------------------------------------------
        if best_local_score < cur_score or rng.random() < np.exp((cur_score - best_local_score) / max(temp, 1e-12)):
            a = best_local_a
            cur_score = best_local_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[improve] iter {it:6d}, new best = {best_score:.6f}")

        # --------------------------------------------------------------
        # Simple annealing of step size and temperature
        # --------------------------------------------------------------
        if it % 1000 == 0:
            tau = max(tau * 0.9, 1e-4)
            temp = max(temp * 0.9, 1e-4)

    # ------------------------------------------------------------------
    # Return the best sequence found
    # ------------------------------------------------------------------
    print(f"[final] best score = {best_score:.6f}")
    return best_a.tolist()
```
