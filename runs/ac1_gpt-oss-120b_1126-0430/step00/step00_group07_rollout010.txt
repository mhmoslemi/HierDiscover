def solve_convolution_lp(f, M):
    n=len(f)
    c = -np.ones(n) # maximize sum(b)
    A_ub = []
    b_ub = []
    # For each k in 0..2n-2:
    for k in range(2*n-1):
        row = np.zeros(n)
        # Compute coefficients: b_j coefficient = f_{k - j} if 0 <= k-j < n else 0
        # Equivalent: for j in range(n):
        #   i = k - j
        #   if 0<=i<n:
        #        row[j] = f[i]
        # Do via vectorized approach
        # We'll just loop:
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                row[j] = f[i]
        A_ub.append(row)
        b_ub.append(M)
    # Nonnegativity constraints: b >= 0 (already handled by linprog's bounds)
    # Use bounds d for each variable: (0, None)
    bounds = [(0, None) for _ in range(n)]
    
    # Solve
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', options={'time_limit':5})
    if result.success:
        return result.x
    else:
        return None

"""
Propose a better step‑function (height sequence) for the ``evaluate_sequence`` test.

The idea is a simple yet powerful iterative scheme:

1. **Start point** – use the provided ``height_sequence_1`` (if it exists) or
   a U‑shaped Beta distribution (lots of mass near the ends, little in the middle).
   The distribution is normalised to sum = 1, which is safe because the
   evaluation is scale‑invariant.

2. **Linear‑programming direction** – for the current sequence *f* compute
   the maximal self‑convolution value ``M = max(f*f)``.
   Solve a linear program

        maximise   Σ b_j
        subject to  (f * b)_k ≤ M   for all k            (linear in b)
                    b_j ≥ 0

   The LP finds a non‑negative sequence ``g`` that can increase the total mass
   without ever increasing the current worst convolution bound ``M``.
   The solution is always feasible (``b = f``) and usually has a larger sum.

3. **Mixing** – after scaling ``g`` to have the same ℓ₁‑norm as *f* we look for
   the best convex combination ``h(t) = (1‑t)·f + t·g``.
   Because the sum is unchanged, the evaluation reduces exactly when the
   maximal convolution of ``h(t)`` becomes smaller.
   A tiny line‑search over a handful of ``t`` values (including ``t=1``) is
   performed and the best improvement is kept.

4. **Exploration** – if the LP does not give any progress a small random
   perturbation (move a tiny amount of mass between two positions) is applied.
   After a few unsuccessful steps a full random restart (different length or
   different initial shape) is performed.

5. **Termination** – the loop runs until the allotted time (≈ 1000 s) is
   exhausted; the best sequence seen is returned.

The method combines the guarantee of the LP step (the convolution bound never
gets worse) with an aggressive line‑search for the step size, which usually
achieves a much faster reduction of the objective than the original
``t = 0.01`` updates.  Random restarts help to avoid stagnation.
"""

import time
import numpy as np
from scipy.optimize import linprog


# -------------------------------------------------------------
# Helper functions (top‑level – required by the platform)
# -------------------------------------------------------------

def _conv_max(seq: np.ndarray) -> float:
    """Return the maximal value of the linear convolution of seq with itself."""
    return float(np.max(np.convolve(seq, seq)))


def _solve_lp(f: np.ndarray, bound: float):
    """
    Solve the LP:
        maximise   Σ b_j
        subject to  (f * b)_k ≤ bound   for all k
                    b_j ≥ 0
    Parameters
    ----------
    f : np.ndarray   – normalised (sum==1) current sequence
    bound : float    – current maximal convolution value (M)

    Returns
    -------
    np.ndarray | None   – optimal b (unnormalised) or None if LP fails.
    """
    n = f.size
    # maximise sum(b)  → minimise -sum(b)
    c = -np.ones(n)

    # Build the (2n-1) convolution constraints: (f * b)[k] ≤ bound
    rows = []
    rhs = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # coefficient of b_j is f_{k-j} if that index is valid
        # we fill the row by iterating over j
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                row[j] = f[i]
        rows.append(row)
        rhs.append(bound)

    # No‑negativity is handled via bounds
    bounds = [(0, None) for _ in range(n)]

    # Use HiGHS (fast for medium‑size LPs)
    res = linprog(
        c,
        A_ub=rows,
        b_ub=rhs,
        bounds=bounds,
        method='highs',
        options={'time_limit': 5.0, 'presolve': True, 'symm': False},
    )
    if res.success:
        return res.x
    return None


def _line_search_mix(f: np.ndarray, g: np.ndarray, cur_max: float):
    """
    Given two sequences f and g (both sum to 1) and the current maximal convolution
    ``cur_max`` of f, try several mixing coefficients t∈[0,1] and return the best
    new sequence together with its evaluation score.
    """
    best_seq = f
    best_score = None
    # candidates – include the extreme points and a few intermediate values
    cand_ts = [0.0, 0.025, 0.05, 0.10, 0.20, 0.40, 0.60, 0.80, 1.0]
    for t in cand_ts:
        seq = (1.0 - t) * f + t * g
        # small numerical noise may introduce tiny negatives; clip them
        seq = np.clip(seq, 0.0, None)
        # re‑normalise to keep the sum exactly 1 (convex combination already does,
        # but clipping may have disturbed it slightly)
        s = seq.sum()
        if s == 0.0:
            continue
        seq = seq / s
        # evaluate – we use the fast local computation of the max convolution,
        # but eventually we call the official ``evaluate_sequence`` to obtain the
        # exact score (including the final safe‑guards).
        m = _conv_max(seq)
        # quick lower bound on the final score (exact formula, no clipping)
        quick_score = 2.0 * seq.size * m / (seq.sum() ** 2)
        # only compute the heavy ``evaluate_sequence`` when we have an improvement
        if (best_score is None) or (quick_score < best_score):
            # verify with the provided evaluator (the function is already in the
            # global namespace of the competition environment)
            try:
                real_score = evaluate_sequence(seq.tolist())
                if real_score < np.inf and (best_score is None or real_score < best_score):
                    best_score = real_score
                    best_seq = seq.copy()
            except Exception:
                # any unexpected problem – just ignore this candidate
                pass
    return best_seq, best_score


def _random_u_shape(n: int, alpha: float = 0.3):
    """
    Produce a U‑shaped distribution on n points.
    The shape follows a Beta(α,α) density (α<1 gives more weight near the ends).
    """
    x = (np.arange(n) + 0.5) / n  # mid‑point of each bin
    # Beta PDF (unnormalised) – avoid infinities at the exact boundaries
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    raw = (x ** (alpha - 1)) * ((1.0 - x) ** (alpha - 1))
    raw = raw / raw.sum()
    return raw

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence with a low value of ``evaluate_sequence``.
    The function respects the time budget (≈ budget_s seconds) and returns the
    best sequence found so far.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # a little safety margin

    # ---------------------------------------------------------
    # Initialisation – try the known good sequence first
    # ---------------------------------------------------------
    if 'height_sequence_1' in globals():
        init_seq = np.array(height_sequence_1, dtype=float)
        init_seq = np.clip(init_seq, 0.0, None)
        # Normalise to sum = 1 (scale is irrelevant for the objective)
        if init_seq.sum() == 0.0:
            init_seq = np.ones_like(init_seq) / init_seq.size
        else:
            init_seq = init_seq / init_seq.sum()
        best_seq = init_seq.copy()
        best_score = evaluate_sequence(best_seq.tolist())
    else:
        # fall back to a simple U‑shaped start
        n0 = np.random.randint(150, 350)
        best_seq = _random_u_shape(n0)
        best_score = evaluate_sequence(best_seq.tolist())

    print(f"[init] length={best_seq.size}, score={best_score:.6f}")

    # ---------------------------------------------------------
    # Main search loop
    # ---------------------------------------------------------
    no_improve_counter = 0
    while time.time() < deadline:
        n = best_seq.size

        # -----------------------------------------------------------------
        # 1) LP direction (g) – maximise Σb under the current convolution bound
        # -----------------------------------------------------------------
        cur_max = _conv_max(best_seq)
        # Solve LP with the current sequence (which is already normalised to sum=1)
        g_unscaled = _solve_lp(best_seq, cur_max)
        if g_unscaled is not None and g_unscaled.sum() > 0.0:
            # Scale g to have the same ℓ₁‑norm as the current sequence (i.e. 1)
            g = g_unscaled * (1.0 / g_unscaled.sum())
            # -----------------------------------------------------------------
            # 2) Find a good mixing coefficient t (line‑search)
            # -----------------------------------------------------------------
            new_seq, new_score = _line_search_mix(best_seq, g, cur_max)
            if new_score is not None and new_score < best_score:
                best_seq = new_seq
                best_score = new_score
                no_improve_counter = 0
                print(f"[improve] t‑mix → length={best_seq.size}, score={best_score:.6f}")
                continue   # go to the next iteration immediately
        # -----------------------------------------------------------------
        # If we reach here the LP did not give a better direction – random tweak
        # -----------------------------------------------------------------
        no_improve_counter += 1
        if no_improve_counter >= 30:
            # Full random restart – possibly with a different length
            n_new = np.random.choice([200, 300, 400, 500, 600, 800, 1000])
            best_seq = _random_u_shape(n_new, alpha=0.25 + 0.2 * np.random.rand())
            # ensure exact sum = 1
            best_seq = best_seq / best_seq.sum()
            best_score = evaluate_sequence(best_seq.tolist())
            no_improve_counter = 0
            print(f"[restart] length={best_seq.size}, score={best_score:.6f}")
            continue

        # Small random perturbation: move a tiny amount of mass between two cells
        i, j = np.random.choice(n, size=2, replace=False)
        delta = 0.01 * best_seq[i] * np.random.rand()
        # avoid making the donor negative
        delta = min(delta, best_seq[i])
        best_seq[i] -= delta
        best_seq[j] += delta
        # re‑normalise (still sum == 1, but due to rounding we enforce it)
        best_seq = best_seq / best_seq.sum()
        # evaluate the perturbed sequence
        cur_score = evaluate_sequence(best_seq.tolist())
        if cur_score < best_score:
            best_score = cur_score
            no_improve_counter = 0
            print(f"[perturb] length={n}, score={best_score:.6f}")

    # ---------------------------------------------------------
    # Return the best sequence encountered (as a plain Python list)
    # ---------------------------------------------------------
    print(f"[final] best length={best_seq.size}, best score={best_score:.6f}")
    return best_seq.tolist()
