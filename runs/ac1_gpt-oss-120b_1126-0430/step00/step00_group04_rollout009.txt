def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Construct a monotone decreasing polynomial sequence a_i ∝ (n-i)^p,

    # close to the asymptotic regime where the analytical bound holds
    n = 2000

def propose_candidate(seed=42, budget_s=1000, **kwargs):

    """

    # --------------------------------------------------------------
    # 1.  Closed‑form construction
    # --------------------------------------------------------------

    # initial evaluation (for logging)
    best_seq = seq
    best_score = evaluate_sequence(best_seq)
    print(f"[init] n={n}, p={p} → score={best_score:.6f}")

    # 2.  Tiny local search (optional – cheap and safe)
