Generates random Dirichlet vectors of varying length, normalizes sum to

import numpy as np, time, math

def _make_random_seq(n):
    # draw positive numbers from exponential distribution
    raw = np.random.exponential(scale=1.0, size=n)
    total = raw.sum()
    if total == 0:
        # fallback to uniform
        raw = np.ones(n)
        total = n
    scale = math.sqrt(2.0 * n) / total
    return (raw * scale).tolist()

def _perturb(seq, eps=0.05):
    """Transfer a small fraction of mass from a random index to another."""
    a = np.array(seq, copy=False)
    n = a.shape[0]

        return seq

    a[i] -= amount
    a[j] += amount

"""
Randomised search for a low‑value step‑function.

The routine draws many random positive vectors of varying length,
normalises each to have ∑a_i = √(2 n) (the scaling used by
``evaluate_sequence``) and keeps the one with the smallest score.
When a candidate beats the target (≈1.5) the search switches to a very
cheap hill‑climbing phase that repeatedly transfers a tiny amount of mass
between two randomly chosen positions; the move is kept only if it
improves the score.

Both stages respect the global time budget (default 1000 s).  The
function returns a plain Python list of floats ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _random_sequence(n: int) -> list[float]:
    """
    Generate a random non‑negative sequence of length n.
    Entries are drawn from an exponential distribution and then
    scaled so that the sum equals √(2 n).
    """
    # exponential draws give a Dirichlet‑like distribution after normalisation
    raw = np.random.exponential(scale=1.0, size=n)
    total = raw.sum()
    # safety – should never happen, but guard against a zero sum
    if total == 0.0:
        raw = np.ones(n, dtype=float)
        total = n
    scale = math.sqrt(2.0 * n) / total
    return (raw * scale).tolist()


def _transfer_mass(seq: np.ndarray, eps: float = 0.05) -> None:
    """
    Move a small fraction of the total mass from one random index to another.
    The total sum stays unchanged, and all entries stay non‑negative.
    """
    n = seq.shape[0]
    i, j = np.random.randint(n, size=2)
    if i == j:
        return
    # amount is a random share (up to eps) of the source entry
    delta = seq[i] * eps * np.random.rand()
    if delta <= 0.0:
        return
    seq[i] -= delta
    seq[j] += delta


# ----------------------------------------------------------------------
# Main entry point required by the grader
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Return a sequence that (hopefully) gives a low value of ``evaluate_sequence``.
    The algorithm works in two stages:
    1) a fast random search that tries many different lengths;
    2) a tiny hill‑climbing on the best candidate found so far.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5          # a tiny safety margin

    best_seq: list[float] | None = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0) If an existing construction is available, use it as a baseline.
    # ------------------------------------------------------------------
    try:
        # ``height_sequence_1`` is defined globally in the evaluation environment.
        cand = list(height_sequence_1)                 # copy to avoid side‑effects
        sc = evaluate_sequence(cand)
        if sc < best_score:
            best_score, best_seq = sc, cand
            print(f"Baseline from height_sequence_1: score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1) Random sampling phase
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # choose a length that balances statistical quality and speed
        n = np.random.randint(2000, 8000)              # 2 k … 8 k elements
        seq = _random_sequence(n)

        try:
            sc = evaluate_sequence(seq)
        except Exception:
            continue

        if sc < best_score:
            best_score, best_seq = sc, seq
            print(f"New best (random)   n={n:<5d}  score={sc:.6f}")

            # stop the random phase early if we are already comfortably below the target
            if best_score <= 1.45:
                break

    # ------------------------------------------------------------------
    # 2) Simple hill‑climbing on the current best sequence
    # ------------------------------------------------------------------
    if best_seq is not None:
        a = np.array(best_seq, copy=False)             # work in‑place on a NumPy view
        n = a.shape[0]

        # hill‑climbing parameters
        EPS = 0.05                                      # max relative transfer per move
        NO_IMPROVE_LIMIT = 2000                        # stop after many failed trials
        no_improve = 0

        while time.time() < deadline and no_improve < NO_IMPROVE_LIMIT:
            _transfer_mass(a, eps=EPS)

            try:
                sc = evaluate_sequence(a.tolist())
            except Exception:
                # a move that produced an invalid sequence – revert it
                _transfer_mass(a, eps=-EPS)            # undo (same magnitude, opposite sign)
                continue

            if sc < best_score:
                best_score, best_seq = sc, a.tolist()
                no_improve = 0
                print(f"Improved (hill)    n={n:<5d}  score={sc:.6f}")
                # tighten the step size once we are close to the target
                if best_score < 1.3:
                    EPS = max(EPS * 0.9, 0.005)
            else:
                # undo the move
                _transfer_mass(a, eps=-EPS)
                no_improve += 1

    # ------------------------------------------------------------------
    # Fallback – a trivial constant sequence (never needed in practice)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 100
        val = math.sqrt(2.0 * n) / n
        best_seq = [val] * n
        best_score = evaluate_sequence(best_seq)
        print(f"Fallback sequence: n={n} score={best_score:.6f}")

    print(f"Final best score: {best_score:.6f} (length {len(best_seq)})")
    return best_seq
