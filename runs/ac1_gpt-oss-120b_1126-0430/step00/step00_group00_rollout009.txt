import numpy as np
import time

def propose_candidate(seed=42, budget_s=1000):
    """Search for a non‑negative step function minimizing the evaluation metric.
    The method works on a fixed length `n` (the length of the known ``height_sequence_1`` if
    available, otherwise 2000).  It keeps the total sum of the coefficients equal to
    1 (the objective is scale‑invariant) and performs a simulated‑annealing / local‑search
    on the simplex.  At each iteration a small amount of mass is transferred from a
    source index that contributes most to the current maximal convolution value to a
    destination index that contributes least.  The convolution is updated incrementally,
    which allows millions of very cheap moves inside the allotted time budget.
    The best sequence encountered is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # a small guard

    # ----------------------------------------------------------------------
    # Determine the length n and initialise a first candidate sequence.
    # ----------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        init = np.array(height_sequence_1, dtype=np.float64)
        n = init.size
        # normalise to sum = 1 (objective is scale‑invariant)
        a = init / init.sum()
    else:
        n = 2000
        a = np.random.dirichlet(np.ones(n))

    # ----------------------------------------------------------------------
    # Helper to build the shifted array a_{k‑shift}
    # ----------------------------------------------------------------------
    def shifted(a_vec, shift):
        """Return array of length 2*n‑1 where entry k is a_vec[k‑shift] if in [0,n) else 0."""
        # shift is integer between 0 and n‑1
        if shift == 0:
            return np.concatenate([a_vec, np.zeros(n-1, dtype=np.float64)])
        if shift == n-1:
            return np.concatenate([np.zeros(n-1, dtype=np.float64), a_vec])
        return np.concatenate([np.zeros(shift, dtype=np.float64),
                               a_vec,
                               np.zeros(n-1-shift, dtype=np.float64)])

    # initial convolution and objective
    c = np.convolve(a, a)
    M = c.max()
    best_a = a.copy()
    best_M = M

    # ----------------------------------------------------------------------
    # Simulated‑annealing / local‑search parameters
    # ----------------------------------------------------------------------
    step_frac = 0.2          # maximal relative amount moved from a source index
    min_step = 1e-4          # absolute lower bound for the step (to avoid stagnation)
    step_decay = 0.99993     # exponential decay of step size
    temperature = 0.02       # initial temperature for Metropolis acceptance
    min_temp = 1e-6
    temp_decay = 0.99994

    it = 0
    refresh_counter = 0    # after many moves we recompute the full convolution

    # ----------------------------------------------------------------------
    # Main loop
    # ----------------------------------------------------------------------
    while time.time() < deadline:
        # locate the current maximal convolution index
        kmax = int(np.argmax(c))

        # partner contribution array: g_i = a_{kmax-i} (0 if out of range)
        # i runs from max(0, kmax-(n-1)) to min(n-1, kmax)
        i_low = max(0, kmax - (n - 1))
        i_high = min(n - 1, kmax)
        idx_range = np.arange(i_low, i_high + 1, dtype=np.int64)
        partner = a[kmax - idx_range]          # shape (len(idx_range),)

        # source index: choose with probability proportional to partner (large contribution)
        if partner.sum() > 0:
            src = np.random.choice(idx_range, p=partner / partner.sum())
        else:
            src = np.random.randint(n)

        # destination index: bias towards small partner values (few contributions)
        # we build a complementary distribution on the whole index set.
        # For indices outside the partner range the contribution is zero.
        comp = np.ones(n, dtype=np.float64)
        comp[idx_range] = 1.0 - (partner / (partner.max() + 1e-12))
        comp_sum = comp.sum()
        if comp_sum > 0:
            dst = np.random.choice(np.arange(n), p=comp / comp_sum)
        else:
            dst = np.random.randint(n)

        if dst == src:
            # pick a different destination
            dst = (dst + 1) % n

        # amount of mass to move
        delta = a[src] * step_frac * np.random.rand()
        # ensure we move at least a tiny amount and never make a[src] negative
        delta = max(min(delta, a[src]), min_step)

        # ------------------------------------------------------------------
        # incremental update of the convolution (vectorised, O(n))
        # ------------------------------------------------------------------
        shift_src = shifted(a, src)    # length 2*n-1
        shift_dst = shifted(a, dst)

        c_new = c - 2.0 * delta * shift_src + 2.0 * delta * shift_dst

        # the quadratic correction ΔΔ
        c_new[2 * src] += delta * delta
        c_new[2 * dst] += delta * delta
        c_new[src + dst] -= 2.0 * delta * delta

        M_new = c_new.max()

        # --------------------------------------------------------------
        # Acceptance test (Metropolis)
        # --------------------------------------------------------------
        improve = M_new < M
        if improve:
            accept = True
        else:
            # probability to accept a worsening move
            if temperature > 0:
                prob = np.exp(-(M_new - M) / temperature)
                accept = np.random.rand() < prob
            else:
                accept = False

        if accept:
            # apply the move
            a[src] -= delta
            a[dst] += delta
            c = c_new
            M = M_new

            if M < best_M:
                best_M = M
                best_a = a.copy()
                # log progress
                eval_val = 2.0 * n * best_M
                print(f"[{time.time():.1f}] iter={it}  M={best_M:.6e}  eval={eval_val:.6f}")

        # --------------------------------------------------------------
        # schedule updates
        # --------------------------------------------------------------
        step_frac = max(min_step, step_frac * step_decay)
        temperature = max(min_temp, temperature * temp_decay)

        it += 1
        refresh_counter += 1

        # occasionally recompute the full convolution to avoid accumulation errors
        if refresh_counter >= 5000:
            c = np.convolve(a, a)
            M = c.max()
            refresh_counter = 0

        # occasional random restart if we have been stuck for a while
        if it % 200000 == 0 and time.time() < deadline - 10:
            # keep the current best, start a fresh random point for diversification
            a = np.random.dirichlet(np.ones(n))
            c = np.convolve(a, a)
            M = c.max()
            # reset schedules a bit hotter to explore again
            step_frac = 0.2
            temperature = 0.02

    # --------------------------------------------------------------
    # Return best sequence found (convert to plain Python list)
    # --------------------------------------------------------------
    # scale back to a convenient magnitude (optional)
    # we keep the sum exactly 1 – evaluate_sequence works with any scale.
    return best_a.tolist()

"""
Propose a sequence of non‑negative step heights that minimizes the
evaluation function defined in the challenge.

The algorithm works on a fixed length `n` (the length of the known
`height_sequence_1` if it exists, otherwise 2000).  The total sum of the
coefficients is kept equal to 1 – the objective is invariant under
scaling, so this restriction does not affect optimality.

A simulated‑annealing / mass‑transfer local search is performed:
* The current maximal convolution value `M = max(a*a)` is located.
* For that index `k_max` the gradient of the max term is
  `g_i = a_{k_max-i}` (zero when the index is out of range).  Large
  `g_i` means that coefficient `i` contributes heavily to the peak.
* One source index is drawn with probability proportional to `g_i`,
  one destination index is drawn with probability favouring small
  `g_i`.  A small amount of mass `δ` is transferred from source to
  destination.
* The convolution is updated **incrementally** using slice operations,
  which is O(n) and avoids full recomputation.
* Moves that lower `M` are always accepted; worsening moves are accepted
  with a Metropolis probability that slowly cools down.
* The step size and temperature are exponentially decayed; occasional
  full recomputation of the convolution and random restarts keep the
  search numerically stable and help escape local minima.

The best sequence seen during the allowed time budget (default 1000 s)
is returned as a plain Python list.
"""

import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    # ------------------------------------------------------------------
    # Setup & initialisation
    # ------------------------------------------------------------------
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5          # safety margin

    # length of the problem (use the previously found construction if present)
    if "height_sequence_1" in globals():
        a = np.array(height_sequence_1, dtype=np.float64)
        n = a.size
        a = a / a.sum()                              # normalise to sum = 1
    else:
        n = 2000
        a = np.random.dirichlet(np.ones(n, dtype=np.float64))

    # full convolution, current max, and the best record so far
    c = np.convolve(a, a)
    M = c.max()
    best_a = a.copy()
    best_M = M

    # ------------------------------------------------------------------
    # Hyper‑parameters for the annealing / mass‑transfer search
    # ------------------------------------------------------------------
    step_frac   = 0.20       # maximal fraction of the source mass we may move
    min_step    = 1e-6       # lower bound on an absolute transfer amount
    step_decay  = 0.99993    # exponential decay of the step size

    temperature = 0.02       # initial temperature for Metropolis acceptance
    min_temp    = 1e-7
    temp_decay  = 0.99994    # exponential decay of temperature

    it          = 0
    refresh_cnt = 0          # full recompute after a while (numerical safety)

    # ------------------------------------------------------------------
    # Helper: full recomputation of convolution and maximum
    # ------------------------------------------------------------------
    def recompute():
        nonlocal c, M
        c = np.convolve(a, a)
        M = c.max()

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # --------------------------------------------------------------
        # Determine the index where the convolution attains its maximum
        # --------------------------------------------------------------
        k_max = int(np.argmax(c))

        # Build the gradient vector g_i = a_{k_max-i} (zeros outside the domain)
        g = np.zeros(n, dtype=np.float64)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        if i_low <= i_high:
            idx = np.arange(i_low, i_high + 1, dtype=np.int64)
            g[idx] = a[k_max - idx]

        # ----------------------------------------------------------------
        # Choose a source index (large contribution to the peak)
        # ----------------------------------------------------------------
        sum_g = g.sum()
        if sum_g > 0:
            src = np.random.choice(np.arange(n), p=g / sum_g)
        else:
            src = np.random.randint(n)

        # ----------------------------------------------------------------
        # Choose a destination index (small contribution to the peak)
        # ----------------------------------------------------------------
        # We use a simple “inverse” weighting: the larger g[i] the smaller
        # the chance to become a destination.
        max_g = g.max()
        inv_weights = max_g - g + 1e-12          # all non‑negative
        sum_inv = inv_weights.sum()
        if sum_inv > 0:
            dst = np.random.choice(np.arange(n), p=inv_weights / sum_inv)
        else:
            dst = np.random.randint(n)

        if dst == src:
            dst = (dst + 1) % n

        # ----------------------------------------------------------------
        # Amount of mass to transfer
        # ----------------------------------------------------------------
        delta = a[src] * step_frac * np.random.rand()
        # respect the available amount and enforce a tiny absolute move
        delta = min(delta, a[src])
        if delta < min_step:
            # Nothing meaningful can be moved – skip this iteration
            continue

        # ----------------------------------------------------------------
        # Incremental update of the convolution (vectorised slice updates)
        # ----------------------------------------------------------------
        c_new = c.copy()
        # subtract the contribution of the shrinking source
        c_new[src:src + n] -= 2.0 * delta * a
        # add the contribution of the growing destination
        c_new[dst:dst + n] += 2.0 * delta * a
        # quadratic correction ΔΔ (exact)
        c_new[2 * src] += delta * delta
        c_new[2 * dst] += delta * delta
        c_new[src + dst] -= 2.0 * delta * delta

        M_new = c_new.max()

        # --------------------------------------------------------------
        # Acceptance test (Metropolis)
        # --------------------------------------------------------------
        improved = M_new < M
        if improved:
            accept = True
        else:
            if temperature > 0.0:
                prob = np.exp(-(M_new - M) / temperature)
                accept = np.random.rand() < prob
            else:
                accept = False

        if accept:
            # apply the move permanently
            a[src] -= delta
            a[dst] += delta
            c = c_new
            M = M_new

            # keep the best sequence seen so far
            if M < best_M:
                best_M = M
                best_a = a.copy()
                eval_val = 2.0 * n * best_M
                print(f"[{time.time():.1f}] iter={it:>7d}  M={best_M:.6e}  eval={eval_val:.6f}")

        # --------------------------------------------------------------
        # Schedule updates
        # --------------------------------------------------------------
        step_frac = max(min_step, step_frac * step_decay)
        temperature = max(min_temp, temperature * temp_decay)

        it += 1
        refresh_cnt += 1

        # occasional full recomputation to keep numerical errors in check
        if refresh_cnt >= 5000:
            recompute()
            refresh_cnt = 0

        # occasional random restart to diversify the search
        if it % 200_000 == 0 and time.time() < deadline - 5.0:
            a = np.random.dirichlet(np.ones(n, dtype=np.float64))
            recompute()
            # reset the annealing schedule a bit warmer
            step_frac = 0.20
            temperature = 0.02

    # ------------------------------------------------------------------
    # Return the best sequence found (as a plain Python list)
    # ------------------------------------------------------------------
    # The sum is already 1, which is well inside the evaluator’s limits.
    return best_a.tolist()
