def project_onto_simplex(v):
    """Projects v (numpy array) onto the probability simplex (sum = 1, v>=0)."""
    n = len(v)
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def simplex_projection(v):
    # v: 1D numpy array
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]

"""
Optimisation of the step‑function bound via smooth‑max gradient descent.

The objective is   score = 2·n·max(conv(a,a)) / (sum(a)^2).
Because the score is homogeneous we may enforce sum(a) = 1.
We minimise  max_k  c_k(a)   where  c_k(a)=∑_i a_i a_{k-i}.

A smooth approximation of the maximum is
    Φ_γ(a) = (1/γ)·log(∑_k exp(γ·c_k(a))) .
Its gradient w.r.t. a is
    ∇Φ_γ = 2·(a ⋆ w) ,
where w_k = exp(γ·c_k) / ∑_j exp(γ·c_j) and “⋆” denotes the
cross‑correlation (np.correlate with mode='full').

The algorithm iteratively:
    1. computes the self‑convolution,
    2. builds the soft‑max weights w,
    3. takes a gradient step (with a decreasing step size),
    4. projects the iterate back onto the probability simplex
       (non‑negative entries summing to 1).

γ is increased slowly so the smooth‑max becomes tighter to the true
maximum.  Several random restarts are performed within the allotted
time budget; the best sequence found (according to the true
evaluate_sequence) is returned.
"""

import time
import numpy as np
import random

# ----------------------------------------------------------------------
# Helper functions (must be top‑level, no closures)
# ----------------------------------------------------------------------


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x ≥ 0, ∑x = 1}.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    # sort in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho_arr = u - cssv / np.arange(1, n + 1)
    # find the last index where rho_arr > 0
    rho_candidates = np.where(rho_arr > 0)[0]
    if rho_candidates.size == 0:
        theta = 0.0
    else:
        rho = rho_candidates[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax_weights(conv: np.ndarray, gamma: float) -> np.ndarray:
    """
    Return the soft‑max weights w_k = exp(gamma·conv_k) / Σexp(...).
    Subtracts the maximum for numerical stability.
    """
    max_c = conv.max()
    exps = np.exp(gamma * (conv - max_c))
    sum_exps = exps.sum()
    if sum_exps == 0.0:
        # fallback – uniform distribution (should never happen)
        return np.full_like(conv, 1.0 / conv.size)
    return exps / sum_exps


def convolution_gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth‑max wrt a:
        ∇ = 2·(a ⋆ w)
    where ⋆ is the cross‑correlation.
    """
    # np.correlate returns length len(a)+len(w)-1
    r = np.correlate(a, w, mode="full")
    offset = len(w) - 1
    # slice the part corresponding to lags 0 … n‑1
    return 2.0 * r[offset: offset + a.size]


def smooth_max_score(a: np.ndarray, conv: np.ndarray) -> float:
    """
    Exact score (sum(a) is assumed to be 1).
    """
    return 2.0 * a.size * conv.max()


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑score height sequence.
    Uses a smooth‑max gradient descent with multiple random restarts.
    Returns the best sequence found (list of floats).
    """
    np.random.seed(seed)
    random.seed(seed)

    # Determine the length of the sequence.  Use the known SOTA length if
    # the global variable is present, otherwise fall back to 4000.
    try:
        n = len(height_sequence_1)  # pylint: disable=undefined-variable
    except Exception:
        n = 4000

    deadline = time.time() + budget_s - 5.0  # keep a small safety margin

    # Best solution tracking
    best_seq = None
    best_score = np.inf

    # Hyper‑parameters
    gamma_start = 0.5
    gamma_max = 30.0
    gamma_factor = 1.02          # geometric increase per iteration
    eta_start = 0.12
    eta_decay = 0.9995

    # Maximum inner iterations for a single run (adjusted dynamically)
    max_inner = 3000

    # ------------------------------------------------------------------
    # Main optimisation loop (restarts while time permits)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # ----------------------------------------------------------------
        # Initialise a new candidate
        # ----------------------------------------------------------------
        if best_seq is not None and random.random() < 0.3:
            # Perturb around the best known point
            a = best_seq + 0.02 * np.random.randn(n)
            a = np.maximum(a, 0.0)
            a = project_onto_simplex(a)
        else:
            a = np.full(n, 1.0 / n)  # uniform start

        gamma = gamma_start
        eta = eta_start

        # Gradient‑descent steps for this restart
        # ----------------------------------------------------------------
        for it in range(max_inner):
            if time.time() >= deadline:
                break

            conv = np.convolve(a, a)          # full self‑convolution
            cur_score = smooth_max_score(a, conv)

        # ignore any error and return the best sequence we have.
        pass

    return best_seq.tolist()
