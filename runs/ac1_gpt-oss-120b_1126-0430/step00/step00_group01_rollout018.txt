"""
propose_candidate
-----------------

A fast stochastic optimizer that builds a probability distribution (the
step‑function heights) with a very flat self‑convolution.  The objective
is

    2 * n * max_k (a * a)[k] / (sum a)^2

which is scale‑invariant, so we work with sequences normalised to sum = 1.
The search consists of three stages:

1. **Baseline candidates** – uniform and a single exponential (the
   analytically optimal exponential gives ≈ 1.81).

2. **Random mixtures of exponentials** – a short list of components
   `w_i·exp(-t_i·i)` (weights `w_i` drawn from a Dirichlet
   distribution, decay rates `t_i` drawn from a small interval).  By
   mixing several exponentials the shape of the convolution can be made
   much flatter than a single exponential.

3. **Local hill‑climbing** – tiny transfers of mass between two positions
   (preserving the total sum).  This refines the best mixture found so
   far.

A fast FFT based evaluation (`_fast_score`) is used inside the loop
(≈ 50 µs for n = 2000) while the final sequence is returned as a plain
list, ready for the provided `evaluate_sequence` function.

The whole routine stops a few seconds before the time budget expires
(to avoid forced termination) and returns the best sequence discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast evaluation – the same quantity that `evaluate_sequence` returns,
# but without the security checks.  The function normalises the input
# (scale invariance) and computes the linear convolution via FFT.
# ----------------------------------------------------------------------
def _fast_score(seq: np.ndarray) -> float:
    """Return 2 * n * max(conv) for a non‑negative sequence with sum = 1."""
    a = np.asarray(seq, dtype=np.float64)
    a = np.clip(a, 0.0, None)                     # non‑negative
    s = a.sum()
    if s < 1e-12:                                 # avoid division by zero
        return np.inf
    a = a / s                                     # normalise → sum = 1
    n = a.shape[0]
    # length of linear convolution
    L = 2 * n - 1
    # use a power‑of‑two FFT length (fast)
    M = 1 << (L - 1).bit_length()
    fa = np.fft.rfft(a, n=M)
    conv = np.fft.irfft(fa * fa, n=M)[:L]        # linear convolution
    max_b = np.max(conv)
    return float(2.0 * n * max_b)                # sum_a == 1 after normalisation


# ----------------------------------------------------------------------
# Helper: build a monotone exponential vector and normalise it.
# ----------------------------------------------------------------------
def _exp_sequence(n: int, t: float) -> np.ndarray:
    """Return a normalised decreasing exponential a_i = exp(-t*i)."""
    idx = np.arange(n, dtype=np.float64)
    vec = np.exp(-t * idx)
    return vec / vec.sum()


# ----------------------------------------------------------------------
# Helper: random mixture of a few exponentials.
# ----------------------------------------------------------------------
def _random_mixture(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Create a distribution that is a convex combination of 2‑4 exponentials.
    Decay rates are chosen so that the useful parameter x = t·n lies in
    roughly [0.2, 3.0]; this covers the analytically optimal point
    (x ≈ 1.256) and allows additional flexibility.
    """
    # number of components (2‑4)
    k = rng.integers(2, 5)
    # random weights that sum to 1
    weights = rng.dirichlet(np.ones(k))
    # choose x = t·n uniformly in a useful interval, then obtain t
    xs = rng.uniform(0.2, 3.0, size=k)
    ts = xs / n
    idx = np.arange(n, dtype=np.float64)
    seq = np.zeros(n, dtype=np.float64)
    for w, t in zip(weights, ts):
        seq += w * np.exp(-t * idx)
    # safety normalisation (sum should already be 1, but we keep it)
    seq /= seq.sum()
    return seq


# ----------------------------------------------------------------------
# Helper: a tiny local perturbation that keeps the vector on the simplex.
# ----------------------------------------------------------------------
def _local_move(seq: np.ndarray, rng: np.random.Generator, step: float) -> np.ndarray:
    """
    Move a small amount of probability mass from one index to another.
    The step size is a fraction of the source entry (default ≈ 2‑5 %).
    The returned vector is re‑normalised to sum = 1.
    """
    n = seq.shape[0]
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return seq
    # propose a change proportional to the magnitude at i
    delta = step * (rng.random() - 0.5) * seq[i]
    new_seq = seq.copy()
    new_seq[i] = max(0.0, new_seq[i] + delta)
    new_seq[j] = max(0.0, new_seq[j] - delta)
    # re‑normalise – the total may have shifted a little due to clipping
    total = new_seq.sum()
    if total <= 0:
        return seq
    new_seq /= total
    return new_seq


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(budget_s: float = 1000, seed: int = 0, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that
    minimises the supplied `evaluate_sequence` function.

    Parameters
    ----------
    budget_s : float
        Time budget in seconds (default 1000).  The routine stops a few
        seconds early to guarantee a return.
    seed : int
        RNG seed for reproducibility.

    Returns
    -------
    list[float]
        The best sequence found within the time limit.
    """
    rng = np.random.default_rng(seed)
    # a modest length – large enough for a smooth shape, still cheap to eval
    n = 2000

    # --------------------------------------------------------------
    # 1️⃣  Baseline candidates
    # --------------------------------------------------------------
    best_seq = np.full(n, 1.0 / n, dtype=np.float64)         # uniform
    best_score = _fast_score(best_seq)

    # Single exponential (near the analytically optimal point)
    t_opt = 1.256 / n
    cand = _exp_sequence(n, t_opt)
    sc = _fast_score(cand)
    if sc < best_score:
        best_seq, best_score = cand, sc
        print(f'Baseline exponential improvement: {best_score:.6f}')

    # --------------------------------------------------------------
    # 2️⃣  Random mixture search
    # --------------------------------------------------------------
    deadline = time.time() + budget_s - 5.0   # leave a safety margin
    it = 0
    step = 0.05                               # size of local mass transfers

    while time.time() < deadline:
        # ----- mixture of exponentials -----
        cand = _random_mixture(n, rng)
        sc = _fast_score(cand)
        if sc < best_score:
            best_seq, best_score = cand, sc
            print(f'Iter {it:06d} – new best (mixture)  {best_score:.6f}')

        # ----- occasional local hill‑climbing on the current best -----
        # every few hundred iterations we try a handful of small moves
        if it % 500 == 0:
            for _ in range(30):
                cand = _local_move(best_seq, rng, step)
                sc = _fast_score(cand)
                if sc < best_score:
                    best_seq, best_score = cand, sc
                    print(f'Iter {it:06d} – local refinement {best_score:.6f}')
        it += 1

        # early exit if we already beat the target
        if best_score < 1.503:
            break

    # --------------------------------------------------------------
    # 3️⃣  Final sanity check using the official evaluator
    # --------------------------------------------------------------
    # Convert to a plain Python list; the outer harness will call
    # `evaluate_sequence` on it.  Normalisation does not matter for the
    # score, but we keep the sum > 0.01 to avoid the security guard.
    final_seq = best_seq.tolist()
    if sum(final_seq) < 0.01:                     # safeguard
        scale = 0.01 / sum(final_seq)
        final_seq = [x * scale for x in final_seq]

    print(f'Finished search – best score ≈ {best_score:.6f}')
    return final_seq
