def flatten_step(p, alpha, eta):
    conv = np.convolve(p, p)
    w = np.exp(alpha * (conv - np.max(conv)))  # shape (2n-1)
    w = w / w.sum()
    p_rev = p[::-1]
    # compute g = 2 * convolution of w and p_rev, take middle n elements
    full = np.convolve(w, p_rev)
    g = 2 * full[(len(p)-1):(len(p)-1+len(p))]
    # update via multiplicative (exponential) step
    p = p * np.exp(-eta * g)
    p = p / p.sum()
    return p
```

def full_convolve(a, b):
    # returns convolution of a and b using FFT
    n = len(a) + len(b) - 1
    size = 1 << (n-1).bit_length()
    A = np.fft.rfft(a, size)
    B = np.fft.rfft(b, size)
    C = A * B
    conv = np.fft.irfft(C, size)[:n]
    return conv

def gradient_step(p, alpha, eta):
    conv_pp = np.convolve(p, p)  # length 2n-1
    max_c = conv_pp.max()
    # compute softmax weights
    w = np.exp(alpha * (conv_pp - max_c))  # shape 2n-1
    w_sum = w.sum()
    if w_sum == 0:
        w = np.ones_like(w) / len(w)
    else:
        w = w / w_sum
    # compute gradient
    p_rev = p[::-1]
    full = np.convolve(w, p_rev)  # length 3n-2
    # extract middle n values: indices from n-1 to n-1 + n -1 inclusive
    start = len(p) - 1
    g = 2.0 * full[start : start + len(p)]
    # multiplicative update
    p_new = p * np.exp(-eta * g)
    # renormalize to sum 1
    p_new = p_new / p_new.sum()
    return p_new
```
