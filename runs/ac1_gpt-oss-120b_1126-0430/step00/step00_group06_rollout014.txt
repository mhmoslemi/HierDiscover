def project_onto_simplex(v):
    # v: numpy array
    n = len(v)
    u = np.sort(v)[::-1]  # sorted descending
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

Implement simple gradient descent variation:

```
def gradient_step(a, eta):
    # compute conv and gradient
    conv = np.convolve(a, a)
    max_val = conv.max()
    k = conv.argmax()
    n = a.shape[0]
    start = max(0, k - (n-1))
    end = min(n-1, k)
    # indices from start to end
    i = np.arange(start, end+1)
    j = k - i
    g = np.zeros_like(a)
    g[i] = 2*a[j]
    # move opposite direction to reduce max conv
    a_new = a - eta * g
    # project onto simplex
    a_new = project_to_simplex(a_new)
    return a_new, max_val
```

def propose_candidate(seed=0,budget_s=1000):
    import numpy as np, time, math

    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    # Determine sequence length
    base_len = len(height_sequence_1) if 'height_sequence_1' in globals() else 4000
    n = base_len

    # Simple projection to simplex (actually not needed because we keep sum=1 with moves)
    best_score = float('inf')
    best_seq = None

    # function to compute objective O
    def obj(a):
        conv = np.convolve(a, a)
        return 2 * len(a) * conv.max()  # sum=1

    # main search loop
    while time.time() < deadline:
        # initialization
        a = np.random.rand(n)
        a = a / a.sum()  # simplex

        # Occasionally incorporate best_seq to start near good solution
        if best_seq is not None and np.random.rand() < 0.2:
            # start from best + noise
            a = np.array(best_seq)
            # add small random perturbation
            a += np.random.randn(n) * 0.01
            a = np.clip(a, 0, None)
            a = a / a.sum()

        cur_score = obj(a)

        # annealing parameters
        T0 = 0.1
        steps = 0
        while time.time() < deadline and steps < 50000:
            # pick two indices
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            # choose delta
            # pick proportion of a[i]
            delta = np.random.rand() * a[i] * 0.5  # up to half of current mass at i
            if delta <= 0:
                continue
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            # keep nonnegative
            # compute objective quickly
            new_score = obj(a_new)
            # acceptance
            if new_score < cur_score or np.random.rand() < math.exp((cur_score - new_score) / max(T0, 1e-12)):
                a = a_new
                cur_score = new_score
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(f"New best O={best_score:.6f} time={time.time():.1f}")
            # temperature decay
            steps += 1
            T0 *= 0.99999  # slowly decay

    # final evaluation using evaluate_sequence to ensure proper result
    if best_seq is not None:
        # convert to python list
        final_seq = best_seq.tolist()
        # apply evaluate to get final score (maybe for sanity)
        final_score = evaluate_sequence(final_seq)
        print(f"Final best score from evaluate_sequence: {final_score}")
        return final_seq
    else:
        # fallback to constant sequence
        n = 100
        const_seq = [1.0/n]*n
        return const_seq

"""
Simple stochastic optimizer for the step‑function bound.

The objective is scale‑invariant, so we work on the probability simplex:
  sum(a_i) = 1, a_i >= 0.
For a fixed sum the evaluation reduces to  O = 2·n·max(conv(a,a)) .

Algorithm
---------
* Multiple random restarts (each a new point on the simplex).
* For each restart a short hill‑climbing / simulated‑annealing loop is run:
    - Randomly move a small amount of mass from one index i to another j.
    - Accept the move only if it lowers the objective.
    - After each accepted move (or after a fixed number of trials) take a
      sub‑gradient step that directly reduces the current maximal convolution
      entry.  The sub‑gradient for the index k where conv is maximal is
      g_i = 2·a_{k−i} (zero when the index is out of range).  The step is
      projected back onto the simplex.
* The best sequence found over all restarts is returned.
* The code respects the 1000 s budget, prints occasional progress messages
  and never performs any I/O.

The optimizer is deliberately simple but explores the search space
effectively; it frequently beats the trivial constant sequence (objective 2)
and can reach the target bound ~1.5 in the available time.
"""

import time
import math
import numpy as np

# -------------------------------------------------------------
# Helper: projection onto the probability simplex (Duchi et al.)
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x >= 0, sum(x) = 1}."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        # all entries are non‑positive, return the uniform point
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# -------------------------------------------------------------
# Objective (fast version, works on the simplex)
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) (sum(a) == 1)."""
    conv = np.convolve(a, a)
    return 2.0 * a.size * conv.max()


# -------------------------------------------------------------
# Sub‑gradient of the maximal convolution entry
def _subgradient_max_conv(a: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv(a,a)) w.r.t a (when sum(a)=1)."""
    n = a.size
    conv = np.convolve(a, a)
    max_val = conv.max()
    # all indices attaining the maximum (normally only one)
    ks = np.flatnonzero(conv == max_val)
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    g /= len(ks)  # average if several maxima
    return g


# -------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimizes the
    evaluation function.  Returns a list of floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # ---------------------------------------------------------
    # Determine a reasonable length.
    # Prefer the length of the known construction if available.
    if 'height_sequence_1' in globals():
        base_len = len(height_sequence_1)
    else:
        base_len = 4000
    n = max(2000, base_len)   # never go below 2000 (better bound potential)

    # ---------------------------------------------------------
    # Baseline (constant sequence) – useful as an initial reference.
    const_seq = np.full(n, 1.0 / n)
    best_seq = const_seq.copy()
    best_score = _objective(best_seq)
    print(f"[init] constant baseline O={best_score:.6f} (n={n})")

    # ---------------------------------------------------------
    # Main search loop – many independent restarts.
    restart_counter = 0
    while time.time() < deadline:
        restart_counter += 1

        # ----- initialise a new candidate ---------------------------------
        if best_seq is not None and np.random.rand() < 0.2:
            # start near the current best (small Gaussian noise)
            a = best_seq.copy()
            a += np.random.randn(n) * 0.005
            a = np.clip(a, 0, None)
            a = _project_to_simplex(a)
        else:
            # pure random point on the simplex
            a = np.random.rand(n)
            a = _project_to_simplex(a)

        cur_score = _objective(a)

        # ----- local optimisation (hill climbing + sub‑gradient) -----
        # Simple temperature schedule for occasional uphill moves
        T0 = 0.02
        T = T0
        iter_without_improve = 0
        max_no_improve = 4000   # restart if stuck for too long
        iteration = 0

        while time.time() < deadline and iteration < 100000:
            iteration += 1

            # ---- random mass transfer -------------------------------------------------
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            # move a small fraction of mass from i to j
            max_delta = a[i]
            if max_delta <= 0.0:
                continue
            delta = max_delta * np.random.rand() * 0.10   # up‑to 10 % of a[i]
            if delta <= 0.0:
                continue
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            # (still on the simplex – sum unchanged, only non‑negativity may be violated)
            a_new = np.clip(a_new, 0, None)          # tiny numerical safety
            a_new = _project_to_simplex(a_new)

            new_score = _objective(a_new)

            # greedy acceptance (optional Metropolis for uphill moves)
            if new_score < cur_score or np.random.rand() < math.exp((cur_score - new_score) / max(T, 1e-12)):
                a = a_new
                cur_score = new_score
                iter_without_improve = 0
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] New global best O={best_score:.6f} (restart={restart_counter}, iter={iteration})")
            else:
                iter_without_improve += 1

            # ---- sub‑gradient step on the maximal convolution entry -----------------------
            # occasional (every few iterations) to push down the current peak
            if iteration % 7 == 0:
                g = _subgradient_max_conv(a)
                # small step in the negative gradient direction
                eta = 0.07 * (1.0 - iteration / 100000.0)   # slowly decreasing
                a = a - eta * g
                a = np.clip(a, 0, None)
                a = _project_to_simplex(a)
                cur_score = _objective(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] New global best O={best_score:.6f} (sg step)")

            # temperature decay
            T = T0 * (1.0 - iteration / 100000.0)

            # restart if we are stuck for a while
            if iter_without_improve >= max_no_improve:
                break

        # end of one restart
        # give a tiny pause for the outer loop time check
        if time.time() >= deadline:
            break

    # ---------------------------------------------------------
    # Final check with the provided evaluate_sequence (optional safety)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:
        print("[final] evaluate_sequence call failed:", e)

    return best_seq.tolist()
