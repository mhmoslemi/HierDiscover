def gradient_step(seq, eta):
    n = len(seq)
    conv = np.convolve(seq, seq)
    k_max = int(np.argmax(conv))
    # compute gradient vector
    grad = np.zeros(n)
    for i in range(n):
        j = k_max - i
        if 0 <= j < n:
            grad[i] = 2 * seq[j]
    # new seq
    new_seq = np.array(seq) - eta * grad
    new_seq = np.maximum(new_seq, 0.0)
    # renorm to target sum
    if new_seq.sum() > 0:
        new_seq = new_seq * (np.sqrt(2*n) / new_seq.sum())
    else:
        new_seq = np.full(n, np.sqrt(2*n) / n)
    return new_seq.tolist()

def random_move(seq):
    n = len(seq)
    new_seq = seq.copy()
    i, j = np.random.choice(n, size=2, replace=False)
    # transfer some epsilon from i to j
    eps = np.random.rand() * 0.1 * new_seq[i]  # up to 10% of source
    new_seq[i] = max(0.0, new_seq[i] - eps)
    new_seq[j] = new_seq[j] + eps
    # renorm

"""
Propose a candidate step‑function height sequence.

The objective (evaluate_sequence) is
    2 * n * max(conv(a,a)) / (sum(a)**2)

After normalising the sum (sum(a) = √(2n)) the objective equals the
maximum entry of the auto‑convolution.  By the Cauchy–Schwarz inequality
the quantity sum(a_i²) is minimised for a uniform vector, and because
conv(0) = sum(a_i²) the uniform vector attains the global lower bound
of the objective, namely 2.  Therefore the safest strategy is to start
from a uniform sequence and then try to improve it with cheap local
perturbations and occasional linear‑programming directions.  In practice
no improvement below 2 is possible, so the algorithm returns the uniform
sequence (or the best sequence found before the time budget expires).

The implementation:
* initialises several random candidates (including the known
  `height_sequence_1` if available);
* repeatedly applies three kinds of moves:
  – a small mass‑transfer move that tries to flatten the convolution;
  – an LP‑based direction (if SciPy is present);
  – a random reshuffling move;
* keeps the best sequence seen and returns it when the time budget ends.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Optional linear‑programming utilities (used only if SciPy is available)
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog
except Exception:  # pragma: no cover
    linprog = None


def solve_convolution_lp(f_sequence, rhs):
    """
    Solve the LP:
        maximise   sum(b_j)
        subject to conv(f, b) <= rhs  (for all shifts)
                  b_j >= 0
    Returns the optimal b as a NumPy array, or None if the LP cannot be
    solved.
    """
    if linprog is None:
        return None

    n = len(f_sequence)
    # objective: maximise sum(b)  -> minimise -sum(b)
    c = -np.ones(n)

    # Build constraints for each convolution index k = 0 .. 2n-2
    a_rows = []
    b_rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_sequence[i]
        a_rows.append(row)
        b_rows.append(rhs)

    # Non‑negativity constraints: -I * b <= 0
    a_rows.append(-np.eye(n))
    b_rows.append(np.zeros(n))

    a_ub = np.vstack(a_rows)
    b_ub = np.hstack(b_rows)

    # Solve with a modest time limit (10 s) to keep the overall runtime low
    result = linprog(c, A_ub=a_ub, b_ub=b_ub, bounds=(0, None),
                     method='highs',
                     options={'time_limit': 10.0, 'disp': False})
    if result.success:
        return result.x
    return None


def get_good_direction_to_move_into(sequence):
    """
    Compute a direction that keeps the convolution bound unchanged.
    The routine follows the idea from the reference implementation:
      1. Normalise the sequence so that its sum is √(2 n).
      2. Solve the LP that maximises the sum of a new vector g under the
         same convolution bound.
      3. Return a small convex combination (t = 0.01) of the old and the
         new normalised vector.
    If the LP cannot be solved, None is returned.
    """
    n = len(sequence)
    total = sum(sequence)
    if total < 1e-12:
        return None

    # Normalise to the standard sum √(2 n)
    target = math.sqrt(2.0 * n)
    norm_seq = [x * target / total for x in sequence]

    rhs = max(np.convolve(norm_seq, norm_seq))
    g = solve_convolution_lp(norm_seq, rhs)
    if g is None:
        return None

    sum_g = np.sum(g)
    if sum_g < 1e-12:
        return None
    # Normalise g to the same ℓ₁‑norm
    g_norm = [x * target / sum_g for x in g]

    t = 0.01
    new_seq = [(1 - t) * a + t * b for a, b in zip(sequence, g_norm)]
    return new_seq

    except Exception:   # pragma: no cover
        pass

    # Fallback to a uniform vector if nothing else succeeded
    if best_seq is None:
        L = 2244
        target_sum = math.sqrt(2.0 * L)
        val = target_sum / L
        best_seq = [val] * L
        best_score = evaluate_sequence(best_seq)

    # --------------------------------------------------------------
    # 2. Main optimisation loop
    # --------------------------------------------------------------
    cur_seq = best_seq.copy()
    n = len(cur_seq)
    target_sum = math.sqrt(2.0 * n)

    # Pre‑compute the current score once per iteration (used for acceptance)
    cur_score = best_score

    it = 0
    while time.time() < deadline:
        it += 1
        move_type = random.random()

        # ----------------------------------------------------------
        # A. Small mass‑transfer move (aims to flatten the convolution)
        # ----------------------------------------------------------
        if move_type < 0.5:
            conv = np.convolve(cur_seq, cur_seq)
            k_max = int(np.argmax(conv))

            # Find the index whose contribution to this maximal entry is largest
            contributions = np.zeros(n)
            for i in range(n):
                j = k_max - i
                if 0 <= j < n:
                    contributions[i] = cur_seq[i] * cur_seq[j]
                else:
                    contributions[i] = 0.0
            src = int(np.argmax(contributions))

            # Destination indices where the partner for k_max is out‑of‑range
            dest_candidates = [i for i in range(n)
                               if not (0 <= k_max - i < n)]
            if dest_candidates:
                dst = random.choice(dest_candidates)
            else:
                dst = random.randrange(n)

            # Transfer a modest amount of mass
            eps = min(cur_seq[src] * 0.5,
                      target_sum * 0.02 / n + 1e-12)
            if eps > 0:
                new_seq = cur_seq.copy()
                new_seq[src] = max(0.0, new_seq[src] - eps)
                new_seq[dst] = new_seq[dst] + eps
                # Re‑normalise to keep the ℓ₁‑norm fixed
                s = sum(new_seq)
                if s > 0:
                    scale = target_sum / s
                    new_seq = [x * scale for x in new_seq]
                else:
                    new_seq = [target_sum / n] * n
            else:
                new_seq = cur_seq.copy()

        # ----------------------------------------------------------
        # B. LP‑guided direction (if SciPy is available)
        # ----------------------------------------------------------
        elif move_type < 0.8:
            new_seq = get_good_direction_to_move_into(cur_seq)
            if new_seq is None:
                # fallback: tiny random bump
                new_seq = cur_seq.copy()
                idx = random.randrange(n)
                new_seq[idx] = max(0.0,
                                   new_seq[idx] + random.gauss(0, 0.05) * target_sum / n)
                s = sum(new_seq)
                if s > 0:
                    new_seq = [x * target_sum / s for x in new_seq]

        # ----------------------------------------------------------
        # C. Random reshuffling of a small fraction of entries
        # ----------------------------------------------------------
        else:
            new_seq = cur_seq.copy()
            m = max(1, int(0.02 * n))
            idxs = random.sample(range(n), m)
            for idx in idxs:
                new_seq[idx] = max(0.0,
                                   new_seq[idx] + random.gauss(0, 0.2) * target_sum / n)
            s = sum(new_seq)
            if s > 0:
                new_seq = [x * target_sum / s for x in new_seq]

        # ----------------------------------------------------------
        # Evaluate the new candidate
        # ----------------------------------------------------------
        try:
            new_score = evaluate_sequence(new_seq)
        except Exception:
            continue

        # Keep the best ever found
        if new_score < best_score:
            best_score = new_score
            best_seq = new_seq.copy()
            print(f"Iteration {it}: new best score = {best_score:.6f}")

        # Simple hill‑climbing acceptance (always accept improvement,
        # otherwise accept with a small probability to escape plateaus)
        if new_score < cur_score or random.random() < 0.02:
            cur_seq = new_seq
            cur_score = new_score

        # Occasionally re‑initialise the current point to avoid stagnation
        if it % 2000 == 0:
            cur_seq = [random.random() for _ in range(n)]
            s = sum(cur_seq)
            cur_seq = [x * target_sum / s for x in cur_seq]
            cur_score = evaluate_sequence(cur_seq)

    print("Search finished.")
    print(f"Best score discovered: {best_score:.6f} (length = {len(best_seq)})")
    return best_seq
