"""
Optimise a non‑negative step‑function to minimise

    V(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)**2)

where n = len(seq).  The objective is invariant under scaling,
so we keep the total mass equal to 1 during the search.
The algorithm is a fast hill‑climbing / simulated‑annealing loop
based on the following ideas:

* Use the FFT to compute the full self‑convolution in O(n log n).
* Identify the index k where the convolution attains its maximum.
  The contribution of element i to that peak is   seq[i]·seq[k‑i].
* Move a small amount of mass from the index with the largest
  contribution to an index with a small contribution (or a random
  one).  This directly reduces the current peak.
* Accept the move if it improves V; otherwise accept it with a
  temperature‑dependent probability to escape local minima.
* Periodically perform a random “restart” (draw a Dirichlet sample)
  if no improvement has been seen for a while.
* Keep the best sequence ever seen and return it at the end.

The method works for sequences of a few thousand entries and fits
comfortably into the 1000 s time budget.
"""

import time
import numpy as np

# -------------------------------------------------------------
# Helper functions (all top‑level – no closures)
# -------------------------------------------------------------

def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Return the full discrete self‑convolution of a real vector x
    using an FFT of length that is a power of two.
    """
    n = x.shape[0]
    # length needed for linear convolution
    m = 2 * n - 1
    # next power of two for efficient FFT
    nfft = 1 << (m - 1).bit_length()
    # real‑FFT / inverse‑FFT pair
    X = np.fft.rfft(x, n=nfft)
    Y = X * X                     # element‑wise product = convolution in frequency domain
    conv = np.fft.irfft(Y, n=nfft)
    return conv[:m]               # keep only the linear part


def _value_from_seq(x: np.ndarray) -> float:
    """
    Compute the evaluation value V for a sequence with total mass 1.
    Since we keep sum(x) == 1, V = 2·n·max(conv) .
    """
    n = x.shape[0]
    M = _full_convolution(x).max()
    return float(2.0 * n * M)     # sum(x) == 1 → denominator disappears


def _move_mass(seq: np.ndarray, temperature: float) -> np.ndarray:
    """
    Perform a single mass‑transfer move aimed at reducing the current peak.
    The move transfers a fraction of the mass from a “high‑contributing”
    index to a “low‑contributing” index.
    """
    n = seq.shape[0]
    conv = _full_convolution(seq)
    k = int(np.argmax(conv))          # location of the current maximum

    # indices that actually participate in this convolution term
    i_min = max(0, k - (n - 1))
    i_max = min(n - 1, k)
    i = np.arange(i_min, i_max + 1, dtype=int)
    j = k - i

    # contribution of each pair (i,j) to the peak
    contrib = seq[i] * seq[j]

    # choose a source index with a large contribution
    src = int(i[np.argmax(contrib)])

    # choose a target index with a small contribution.
    # Prefer indices that do NOT contribute (contrib == 0); if none,
    # fall back to a random low‑percentile.
    zero_candidates = i[contrib == 0.0]
    if zero_candidates.size:
        tgt = int(np.random.choice(zero_candidates))
    else:
        # pick one of the 20 % lowest contributors
        threshold = np.percentile(contrib, 20)
        low_candidates = i[contrib <= threshold]
        if low_candidates.size:
            tgt = int(np.random.choice(low_candidates))
        else:
            tgt = int(np.random.randint(n))

    # amount of mass to move – a fraction of the source value
    delta = temperature * seq[src]
    # never move more than what is available
    delta = min(delta, seq[src])

    if delta <= 0.0:
        # degenerate case – perform a tiny random jitter instead
        idx = int(np.random.randint(n))
        seq[idx] = max(0.0, seq[idx] + np.random.randn() * 1e-4)
        seq /= seq.sum()
        return seq

    # execute the transfer
    new_seq = seq.copy()
    new_seq[src] -= delta
    new_seq[tgt] += delta

    # protect against negative numbers caused by floating‑point noise
    np.clip(new_seq, 0.0, None, out=new_seq)
    # renormalise to keep total mass exactly 1
    total = new_seq.sum()
    if total == 0.0:                     # very unlikely, but be safe
        new_seq = np.random.rand(n)
        new_seq /= new_seq.sum()
    else:
        new_seq /= total
    return new_seq


def _random_initial(n: int) -> np.ndarray:
    """
    Return a random probability vector of length n (Dirichlet(1,…,1)).
    """
    v = np.random.rand(n)
    v /= v.sum()
    return v

    # -----------------------------------------------------------------
    try:
        # height_sequence_1 is expected to be a global variable when present
        base = np.array(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0:
            raise Exception
        # normalise its total mass to 1
        base /= base.sum()
        use_known = np.random.rand() < 0.6
    except Exception:
        use_known = False

    if use_known:
        seq = base.copy()
    else:
        # length is chosen in the same ball‑park as the current best
        n = np.random.randint(2000, 4000)
        seq = _random_initial(n)

    # keep the best sequence seen so far
    best_seq = seq.copy()
    best_val = _value_from_seq(best_seq)
    # also evaluate with the official function for sanity (cheap, rare)
    try:
        best_score = evaluate_sequence(best_seq.tolist())
        best_val = best_score
    except Exception:
        pass
    print(f'Initial length {len(seq)}  →  score {best_val:.6f}')

    # -----------------------------------------------------------------
    # Main optimisation loop: hill‑climbing with simulated annealing.
    # -----------------------------------------------------------------
    temperature = 0.05           # initial step size as a fraction of source mass
    decay = 0.9997               # exponential cooling
    it = 0
    no_improve = 0
    RESTART_AFTER = 5000         # iterations without improvement → random restart

    while time.time() < deadline:
        # take a guided move
        new_seq = _move_mass(seq, temperature)

        # evaluate the new candidate (fast, own formula)
        new_val = _value_from_seq(new_seq)

        # decide acceptance
        accept = False
        if new_val < best_val:
            accept = True
        else:
            # Metropolis acceptance based on current temperature
            prob = np.exp((best_val - new_val) / max(best_val, 1e-12) / temperature)
            if np.random.rand() < prob:
                accept = True

        if accept:
            seq = new_seq
            if new_val < best_val:
                best_val = new_val
                best_seq = new_seq.copy()
                no_improve = 0
                # occasional full re‑evaluation with the reference function
                try:
                    # evaluate_sequence applies clipping – use it for the final score
                    official = evaluate_sequence(best_seq.tolist())
                    # keep the true (possibly slightly different) value for reporting
                    best_val = official
                except Exception:
                    pass
                print(f'Iter {it:6d}: new best {best_val:.6f}')
            else:
                # accepted a worsening move – keep it as current state
                seq = new_seq
        else:
            # reject – stay at the previous state (seq unchanged)
            no_improve += 1

        # cooling schedule
        temperature *= decay
        if temperature < 1e-6:
            temperature = 1e-6

        # random restart if stagnation is observed
        if no_improve >= RESTART_AFTER:
            n = len(seq)
            seq = _random_initial(n)
            no_improve = 0
            temperature = 0.05

        it += 1

    # -----------------------------------------------------------------
    # Return the best sequence we have found (as a plain list).
    # -----------------------------------------------------------------
    print(f'Finished after {it} iterations – best score {best_val:.6f}')
    return best_seq.tolist()
