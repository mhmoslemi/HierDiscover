def simplex_projection(v, target_sum):
    # Project onto the set {x >=0, sum x = target_sum}
    # Standard algorithm: sort v descending, compute cumulative sum.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def build_sequence_power(n, p, d):
    raw = [ (i + d) ** (-p) for i in range(n) ]
    s = sum(raw)

"""
propose_candidate
=================

Search for a step‑function (non‑negative list of heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

where ``conv`` is the linear convolution (numpy.convolve).  The score is
scale‑invariant, therefore we keep the L1‑norm fixed to the “normalised”
value `sqrt(2*n)`.  The optimiser works in two phases:

1. **Parameter sweep** – generate families of analytically‑parameterised
   sequences (power‑law, shifted power‑law, polynomial decay, simple
   linear decay, random Dirichlet vectors).  Each candidate is scaled to
   the required sum and evaluated.  The best candidate so far is kept.

2. **Local refinement** – starting from the best candidate, perform a
   simple hill‑climbing / simulated‑annealing move: transfer a small
   amount of mass from one index to another while keeping the total sum
   fixed and staying non‑negative.  Accept the move if it improves the
   score (or with a tiny probability that decays with a temperature
   schedule).  Continue until the time budget expires.

The algorithm is deliberately lightweight (no external optimisation
libraries) and respects the 1000 s time limit.  It works for any length
`n` (the provided global `height_sequence_1` is used if available).  The
returned list is ready to be fed to ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_to_simplex(x, target_sum):
    """
    Project the vector x (numpy array) onto the simplex
    {y >= 0, sum(y) = target_sum} using the O(n) algorithm.
    """
    # Shift so that the average is target_sum / len(x)
    # This implementation follows the algorithm by
    # Wang & Carreira‑Perpiñán (2013).
    n = len(x)
    if target_sum <= 0:
        return np.zeros_like(x)

    # Sort descending
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(x - theta, 0.0)
    return w


def _evaluate(seq):
    """Thin wrapper around the provided evaluate_sequence."""
    # The grader supplies evaluate_sequence in the global namespace.
    return evaluate_sequence(seq)


def _make_sequence_from_raw(raw, target_sum):
    """Scale a raw (numpy) non‑negative vector so that its L1‑norm equals target_sum."""
    raw_sum = raw.sum()
    if raw_sum == 0:
        # fallback to uniform
        n = raw.shape[0]
        return [target_sum / n] * n
    scale = target_sum / raw_sum
    seq = (raw * scale).tolist()
    return seq


# ----------------------------------------------------------------------
# Main candidate generator
# ----------------------------------------------------------------------


def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) yields a low
    evaluation score.  The function respects the time budget ``budget_s``.
    """
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Determine the problem size
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
    else:
        # fall back to a moderate size that is still fast to evaluate
        n = 2000

    target_sum = math.sqrt(2.0 * n)          # the normalising L1‑norm
    best_seq = [target_sum / n] * n         # uniform start
    best_score = _evaluate(best_seq)
    print(f"[init] n={n}, uniform score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – parameter sweep
    # ------------------------------------------------------------------
    deadline = time.time() + budget_s - 5.0   # keep a few seconds for refinement
    it = 0
    while time.time() < deadline:
        it += 1
        shape_type = np.random.choice(
            ["power", "shifted_power", "poly", "linear", "rand_dir"]
        )
        # ----------------------------------------------------------------
        # generate raw (pre‑scaled) vector
        # ----------------------------------------------------------------
        if shape_type == "power":
            # a_i = (i+1)^{-p}
            p = np.random.uniform(0.1, 1.0)
            raw = (np.arange(1, n + 1, dtype=float)) ** (-p)

        elif shape_type == "shifted_power":
            # a_i = (i + d)^{-p}, d > 0 to avoid huge first element
            p = np.random.uniform(0.1, 1.0)
            d = np.random.uniform(0.0, n * 0.5) + 1e-6
            raw = (np.arange(n, dtype=float) + d) ** (-p)

        elif shape_type == "poly":
            # a_i = (1 - i/(n-1))^{alpha}
            alpha = np.random.uniform(0.1, 5.0)
            raw = (1.0 - np.arange(n, dtype=float) / (n - 1)) ** alpha

        elif shape_type == "linear":
            # simple linear decay from 1 to 0
            raw = np.linspace(1.0, 0.0, n, dtype=float)

        else:  # "rand_dir"
            # random Dirichlet (Gamma) – thinly spread mass
            raw = np.random.gamma(shape=0.7, scale=1.0, size=n)

        # ensure non‑negative
        raw = np.maximum(raw, 0.0)

        # ----------------------------------------------------------------
        # scale to required L1‑norm and evaluate
        # ----------------------------------------------------------------
        candidate = _make_sequence_from_raw(raw, target_sum)
        score = _evaluate(candidate)

        if score < best_score:
            best_score = score
            best_seq = candidate
            print(
                f"[sweep {it:04d}] new best {best_score:.6f} "
                f"via {shape_type}"
            )

        # occasional random perturbation to escape plateaus
        if it % 250 == 0 and np.random.rand() < 0.3:
            # take the current best, shuffle a tiny fraction of it
            tmp = np.array(best_seq, copy=True)
            idx = np.random.choice(n, size=2, replace=False)
            delta = tmp[idx[0]] * np.random.uniform(0.0, 0.02)
            tmp[idx[0]] -= delta
            tmp[idx[1]] += delta
            tmp = _project_to_simplex(tmp, target_sum)
            pert_score = _evaluate(tmp.tolist())
            if pert_score < best_score:
                best_score = pert_score
                best_seq = tmp.tolist()
                print(
                    f"[perturb {it:04d}] improved to {best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # Phase 2 – local hill climbing / simulated annealing
    # ------------------------------------------------------------------
    current_seq = np.array(best_seq, copy=True)
    current_score = best_score
    temperature = 0.05
    decay = 0.9995

    while time.time() < deadline + 4.0:   # a few extra seconds
        # select two distinct indices
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        # move a small fraction of mass from i to j
        max_move = current_seq[i] * 0.03
        if max_move <= 0:
            continue
        delta = np.random.random() * max_move

        new_seq = current_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        # keep non‑negative and project to the simplex (preserves sum)
        if np.any(new_seq < -1e-12):
            continue
        new_seq = _project_to_simplex(new_seq, target_sum)

        # evaluate
        new_score = _evaluate(new_seq.tolist())

        # accept if better, otherwise maybe accept with tiny probability
        if new_score < current_score:
            current_seq = new_seq
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.tolist()
                print(f"[refine] improved to {best_score:.6f}")
        else:
            # simulated‑annealing style acceptance
            prob = math.exp(-(new_score - current_score) / (temperature + 1e-12))
            if np.random.rand() < prob:
                current_seq = new_seq
                current_score = new_score

        temperature *= decay

    print(f"[final] best score found: {best_score:.6f}")
    return best_seq
