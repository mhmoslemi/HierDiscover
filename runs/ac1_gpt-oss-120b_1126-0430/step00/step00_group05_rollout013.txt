    ----------
    seed : int
        Ignored (kept for compatibility with the evaluation harness).
    budget_s : float
        Ignored – the construction finishes instantly.
    Returns
    -------
    list[float]
        A sequence of non‑negative heights with a very low evaluation score.
    """
    # Length – we use the same order of magnitude as the previously known
    # construction (a few thousand entries).  Larger n only improves the bound.
    n = 5467                     # any integer ≥ 2 works; 5467 matches the given size
    p = 2                        # quadratic decay (p=2 gives ~0.6 score; larger p is fine)

    # Build raw decreasing polynomial values: (n-i)^p, i = 0 … n‑1.
    # Using numpy for speed and precise float handling.
    raw = np.arange(n, 0, -1, dtype=np.float64) ** p   # [n^p, (n-1)^p, … , 1^p]

    # Normalise to make the L1‑norm equal to 1 (the evaluation is scale‑invariant).
    total = raw.sum()
    seq = (raw / total).tolist()

    # One quick evaluation for logging – not required for correctness.
    try:
        score = evaluate_sequence(seq)
        print(f"Generated sequence (n={n}, p={p}) → score = {score:.6f}")
    except Exception as exc:      # pragma: no cover
        # If something unexpected happens we still return the sequence;
        # the harness will then report the proper score.
        print(f"Score evaluation failed: {exc}")

    return seq
