    def obj(seq):
        b = np.convolve(seq, seq)
        return 2 * len(seq) * b.max() / (seq.sum()**2)

    # initialize best from height_sequence_1, maybe also random
    best_seq = np.array(height_sequence_1, dtype=float)
    best_score = obj(best_seq)

    # Could also start with a perturbed version
    curr_seq = best_seq.copy()
    # generic parameters
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # subgradient step
        n = len(curr_seq)
        # compute convolution and max index once
        conv = np.convolve(curr_seq, curr_seq)
        max_b = conv.max()
        k_max = np.argmax(conv)
        # compute gradient
        grad = np.zeros_like(curr_seq)
        idx = k_max - np.arange(n)
        valid = (idx>=0) & (idx<n)
        grad[valid] = 2 * curr_seq[idx[valid]]
        # step size
        # heuristics: set step = step_factor * max_b / (grad.dot(grad) + 1e-12)
        grad_norm2 = (grad*grad).sum()
        if grad_norm2 == 0:
            # no gradient, random perturb
            step = 0
        else:
            step = 0.8 * max_b / grad_norm2   # factor 0.8 to be safe
        # try step, if not improving, reduce
        improved = False
        for factor in [1.0, 0.5, 0.2, 0.1]:

            new_seq = curr_seq - t * grad
            new_seq = np.maximum(new_seq, 0.0)
            s = new_seq.sum()
            if s == 0:
                continue

            new_score = obj(new_seq)
            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.copy()
                print(f'Iter {iteration} new best score {best_score:.6f}')
            if new_score < obj(curr_seq):
                curr_seq = new_seq
                improved = True
                break
        if not improved:
            # random perturbation
            # pick a small fraction of entries to modify by gaussian noise
            num_change = max(1, n // 200)  # ~0.5% of entries

def _objective_exponential(u, n, S):
    # u = delta*n
    delta = u / n
    # compute C
    # use q = exp(-delta)
    # denominator 1 - q**n = 1 - exp(-u)
    denom = 1.0 - math.exp(-u)
    if denom == 0:
        return np.inf
    C = S * delta / denom
    # Determine optimal k for max
    if delta == 0:
        # uniform case
        k = n-1
        max_val = C * C * n
        return max_val
    k_opt = int(np.floor(1.0/delta - 1.0))
    if k_opt >= n:
        k_opt = n-1
    if k_opt < 0:
        k_opt = 0
    # compute b = C^2 * (k+1) * exp(-delta * k)
    max_val = C*C * (k_opt+1) * math.exp(-delta * k_opt)
    return max_val

"""
Propose a candidate step‑function sequence.

The evaluator is scale‑invariant, so we normalise the sum of the sequence to
sqrt(2·n).  For a uniform sequence the score is exactly 2.  By choosing a
geometric (exponential) decay a_i = C·q^i the maximum autocorrelation can be
reduced.  For a fixed length n the score depends only on the product
u = n·(1−q) = n·δ (δ = –log q).  The function

    score(u) = max_k C²·(k+1)·q^k,   C = S·δ / (1−e^{−u}),   S = √(2n)

can be evaluated analytically (the maximising k is ≈⌊1/δ⌋−1).  Scanning u
yields a near‑optimal exponential shape with score ≈1.8 < 2.

We first locate the best u by a cheap analytical scan, build the corresponding
sequence, and then apply a very light hill‑climbing phase (mass moves between
two coordinates) to possibly improve the score a little further.  The whole
procedure runs well within the 1000 s budget and returns a list of non‑negative
floats ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions for the exponential family
# ----------------------------------------------------------------------
def _objective_exponential(u: float, n: int, S: float) -> float:
    """
    Analytical score (max convolution value) for a geometric sequence
    a_i = C·exp(−δ·i) with δ = u/n and C = S·δ/(1−e^{−u}).
    """
    if u <= 0.0:
        return float('inf')
    delta = u / n                       # δ = (1−q) ≈ u/n for small u
    # denominator 1 - q^n = 1 - e^{-u}
    denom = 1.0 - math.exp(-u)
    if denom == 0.0:
        return float('inf')
    C = S * delta / denom                # normalisation constant

    # optimal k satisfies (k+1) ≈ 1/δ ; clipping to the valid range [0,n‑1]
    k_opt = int(math.floor(1.0 / delta - 1.0))
    if k_opt < 0:
        k_opt = 0
    if k_opt >= n:
        k_opt = n - 1

    # max value = C²·(k+1)·q^k = C²·(k+1)·exp(−δ·k)
    max_val = C * C * (k_opt + 1) * math.exp(-delta * k_opt)
    return max_val


def _build_exponential_sequence(u: float, n: int, S: float) -> np.ndarray:
    """Return the geometric sequence a_i = C·exp(‑u·i/n) with normalised sum."""
    delta = u / n
    denom = 1.0 - math.exp(-u)
    C = S * delta / denom
    # a_i = C·exp(‑δ·i)  – use np.exp for stability
    i = np.arange(n, dtype=float)
    seq = C * np.exp(-delta * i)
    # Numerical drift may slightly disturb the sum; re‑normalise exactly.
    seq = np.maximum(seq, 0.0)
    seq *= S / seq.sum()
    return seq
