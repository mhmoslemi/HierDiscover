"""
Propose a step‐function sequence that minimizes the evaluation score.

The algorithm consists of three stages:
1. **Power‑law sweep** – generate monotone sequences a_i ∝ (i+1)^(-β) with a
   small additive offset.  After normalisation (∑a = √(2n)) the score is
   evaluated and the best parameters are kept.
2. **Projected gradient descent** – starting from the best candidate,
   minimise a smooth approximation of the max‑convolution using the
   gradient  g = 2·corr(w,a) where w are soft‑max weights on the
   convolution values.  After each step the vector is projected back onto
   the simplex {a ≥ 0, ∑a = √(2n)}.
3. **Optional random perturbations** – if time permits a few random
   jitter moves are tried to escape possible local minima.

All helper functions are defined at top level (no closures).  The routine
runs until the supplied time budget (default 1000 s) expires and returns
the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the simplex {x >= 0, sum(x) = target}
def project_to_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the non‑negative simplex of given L1‑sum."""
    if target < 0:
        raise ValueError("target sum must be non‑negative")
    # Work on a copy to avoid side effects
    u = np.sort(v)[::-1]                     # descending sort
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)           # 1‑based index
    # Find the largest rho where u_i > cssv_i / i
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries will be zero
        return np.zeros_like(v)
    rho = ind[cond][-1]                      # last true index
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

# ----------------------------------------------------------------------
# Main candidate generator
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a good non‑negative step function.
    Returns a list of floats representing the heights.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # Determine the problem size (use the known construction if available)
    if 'height_sequence_1' in globals() and isinstance(height_sequence_1, (list, np.ndarray)):
        n = len(height_sequence_1)
        init_seq = np.array(height_sequence_1, dtype=float)
    else:
        # a reasonable default size – larger sizes give more freedom but cost more time
        n = 2000
        init_seq = np.random.rand(n)

    target_sum = np.sqrt(2.0 * n)                    # the “standard” normalisation

    # ------------------------------------------------------------------
    # Utility to normalise any non‑negative vector to the required sum
    def normalise(vec: np.ndarray) -> np.ndarray:
        vec = np.maximum(vec, 0.0)                  # enforce non‑negativity
        s = vec.sum()
        if s == 0.0:
            vec = np.ones_like(vec)
            s = vec.sum()
        return vec * (target_sum / s)

    # ------------------------------------------------------------------
    # Keep track of the best sequence seen so far
    best_seq = normalise(init_seq)
    try:
        best_score = evaluate_sequence(best_seq.tolist())
    except Exception:
        best_score = float('inf')
    print(f'Initial score (seed sequence): {best_score:.6f}')

    # ------------------------------------------------------------------
    # 1️⃣  Power‑law sweep (a_i ∝ (i+1)^‑β + μ)
    betas = np.linspace(0.10, 0.90, 17)               # 0.10 … 0.90 step 0.05
    mu_factors = [0.0, 0.05, 0.10, 0.20, 0.40, 0.60]  # relative to max(base)

    indices = np.arange(1, n + 1, dtype=np.float64)   # i+1
    for beta in betas:
        if time.time() > deadline:
            break
        base = indices ** (-beta)                    # monotone decreasing
        max_base = base.max()
        for mu_factor in mu_factors:
            mu = mu_factor * max_base
            raw = base + mu
            candidate = normalise(raw)
            try:
                score = evaluate_sequence(candidate.tolist())
            except Exception:
                continue
            if score < best_score:
                best_score = score
                best_seq = candidate.copy()
                print(f'New best after sweep: β={beta:.3f}, μ={mu_factor:.2f} → {score:.6f}')

    # ------------------------------------------------------------------
    # 2️⃣  Projected gradient descent on a smooth approximation of max(conv)
    # --------------------------------------------------------------
    # Parameters for the descent (they adapt during the run)
    step_size = 0.5
    step_decay = 0.95          # geometric decay each iteration
    beta_soft = 15.0           # soft‑max sharpness (larger → closer to true max)
    max_iters = 400

    a = best_seq.copy()
    for it in range(max_iters):
        if time.time() > deadline:
            break

        # Linear convolution (full length 2n‑1)
        conv = np.convolve(a, a)
        max_c = conv.max()

        # Soft‑max weights that concentrate on the largest entries
        w = np.exp(beta_soft * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            break
        w /= w_sum

        # Gradient of the smooth max: g_i = 2 * Σ_k w_k * a_{k-i}
        # This is exactly 2·correlation(w, a) (valid mode gives length n)
        grad = 2.0 * np.correlate(w, a, mode='valid')

        # Gradient step and projection
        a_new = a - step_size * grad
        a_new = project_to_simplex(a_new, target_sum)

        # Evaluate
        try:
            score = evaluate_sequence(a_new.tolist())
        except Exception:
            # If the sequence becomes invalid, discard the step
            step_size *= step_decay
            continue

        # Accept the step if it improves the score, otherwise keep the old a
        if score < best_score:
            best_score = score
            best_seq = a_new.copy()
            print(f'Iter {it:03d}: improved score → {score:.6f}')
        a = a_new

        # Reduce step size and increase soft‑max sharpness gradually
        step_size *= step_decay
        beta_soft = min(beta_soft * 1.02, 100.0)

    # ------------------------------------------------------------------
    # 3️⃣  A few random jitter moves (optional, cheap)
    jitter_iters = int(0.05 * max_iters)   # ~5 % of gradient steps
    for _ in range(jitter_iters):
        if time.time() > deadline:
            break
        # pick a random index and perturb it a little
        idx = np.random.randint(n)
        delta = np.random.randn() * 0.01 * target_sum / n
        a_jit = best_seq.copy()
        a_jit[idx] = max(0.0, a_jit[idx] + delta)
        a_jit = project_to_simplex(a_jit, target_sum)
        try:
            score = evaluate_sequence(a_jit.tolist())
        except Exception:
            continue
        if score < best_score:
            best_score = score
            best_seq = a_jit.copy()
            print(f'Jitter improvement → {score:.6f}')

    print(f'Final best score: {best_score:.6f}')
    return best_seq.tolist()
