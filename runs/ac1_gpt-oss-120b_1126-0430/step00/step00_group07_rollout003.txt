def project_simplex(v):
    """Projects v onto the probability simplex (sum to 1, nonnegative)."""
    # Implementation using sorting.
    v_sort = np.sort(v)[::-1]
    cssv = np.cumsum(v_sort) - 1
    ind = np.arange(len(v)) + 1
    cond = v_sort - cssv / ind > 0
    if not np.any(cond):
        rho = len(v) - 1
    else:
        rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def full_convolution(x, y):
    n = len(x) + len(y) - 1
    m = 1 << ((n - 1).bit_length())
    X = np.fft.rfft(x, m)
    Y = np.fft.rfft(y, m)

def softmax_weights(conv, beta):
    # stable
    max_c = np.max(conv)
    exp_vals = np.exp(beta * (conv - max_c))
    sum_exp = np.sum(exp_vals)
    if sum_exp == 0:
        return np.ones_like(conv) / conv.size
    return exp_vals / sum_exp

def gradient_step(a, beta=1.0, eta=0.1):
    n = len(a)
    conv = full_convolution(a, a)
    w = softmax_weights(conv, beta)
    # Pad a
    pad_len = 2 * n - 1
    a_padded = np.zeros(pad_len)
    a_padded[n-1:] = a
    # gradient full conv
    grad_full = full_convolution(w, a_padded)
    grad = 2.0 * grad_full[(n-1):(n-1)+n]
    # gradient descent
    a_new = a - eta * grad
    # project to simplex to maintain feasibility
    a_proj = project_simplex(a_new)
    return a_proj

"""
A fast, gradient‑based search for a step‑function that minimizes
evaluate_sequence.

The algorithm works on a probability simplex (sum = 1).  The objective
2·n·max(conv)/sum² is scale‑invariant, therefore we may always
renormalise the sequence to have unit sum and finally scale it to
√(2n) before returning it.

Main ideas
----------
1. **Random initialisations** – several families (uniform, sine,
   exponential, random‑phase) provide diverse starting points.
2. **Soft‑max smooth max** – the non‑smooth maximum of the convolution
   is approximated by a soft‑max with a temperature β that grows
   during the search.
3. **Gradient step** – the gradient of the soft‑max (which is a linear
   combination of shifted copies of the current sequence) is computed
   via FFT‑based convolutions, giving an O(n log n) step.
4. **Projection onto the simplex** – after each update the vector is
   projected back to the simplex (non‑negative, sum = 1) so all
   iterates stay feasible.
5. **Time‑bounded loop** – the loop runs until the given budget is
   exhausted, keeping the best sequence found.

The whole search fits easily into the 1000 s limit and typically
drops the score well below the target 1.503.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
    {x | x_i >= 0, sum_i x_i = 1}
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # numerical safety: renormalise precisely to 1
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _full_convolution(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Compute the full (linear) convolution using FFT.
    Returns an array of length len(x) + len(y) - 1.
    """
    n = len(x) + len(y) - 1
    # next power of two for speed
    m = 1 << ((n - 1).bit_length())
    X = np.fft.rfft(x, m)
    Y = np.fft.rfft(y, m)
    conv = np.fft.irfft(X * Y, m)[:n]
    return conv


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Soft‑max over the convolution values.
    Higher beta => closer to the true maximum.
    """
    c_max = np.max(conv)
    # subtracting max for numerical stability
    exp_vals = np.exp(beta * (conv - c_max))
    sum_exp = np.sum(exp_vals)
    if sum_exp == 0.0:
        # fallback – uniform weights
        return np.full_like(conv, 1.0 / conv.size)
    return exp_vals / sum_exp


def _gradient_step(a: np.ndarray, beta: float, eta: float) -> np.ndarray:
    """
    One gradient‑descent step for the smoothed max objective.
    Returns a new feasible vector (projected onto the simplex).
    """
    n = a.shape[0]

    # 1. convolution a * a
    conv = _full_convolution(a, a)               # length 2n‑1

    # 2. soft‑max weights over the convolution entries
    w = _softmax_weights(conv, beta)             # same length 2n‑1

    # 3. gradient: g_i = 2 * sum_k w_k * a_{k-i}
    #    this is a convolution of w with a zero‑padded version of a
    pad_len = 2 * n - 1
    a_padded = np.zeros(pad_len, dtype=a.dtype)
    a_padded[n - 1:] = a                         # centre the padded vector
    full_grad = _full_convolution(w, a_padded)   # length 4n‑3
    grad = 2.0 * full_grad[(n - 1):(n - 1) + n]   # take the slice we need

    # 4. gradient descent and projection
    a_new = a - eta * grad
    a_proj = _project_simplex(a_new)
    return a_proj


# ----------------------------------------------------------------------
# Candidate generators
# ----------------------------------------------------------------------
def _gen_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _gen_sine(n: int) -> np.ndarray:
    """Positive sine‑like shape, normalized to sum = 1."""
    x = np.arange(1, n + 1)
    a = np.sin(np.pi * x / (n + 1))
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _gen_exponential(n: int) -> np.ndarray:
    a = np.random.exponential(scale=1.0, size=n)
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _gen_random_phase(n: int) -> np.ndarray:
    """
    Create a non‑negative sequence via a random‑phase Fourier transform.
    The resulting time‑domain signal is shifted to be non‑negative and
    projected onto the simplex.
    """
    L = 2 * n - 1                     # length of the zero‑padded FFT domain
    # random real signal, then force unit magnitude in frequency domain
    rand = np.random.randn(L)
    spec = np.fft.rfft(rand)
    spec = np.exp(1j * np.angle(spec))        # magnitude = 1
    seq = np.fft.irfft(spec, n=L)
    a = seq[:n]
    a = a - a.min() + 1e-9                   # make non‑negative
    return _project_simplex(a)

# Main search routine
# ----------------------------------------------------------------------

    """

    deadline = time.time() + budget_s - 0.5   # a small safety margin

    # temperature schedule for the soft‑max (β grows slowly)
    def beta_of(it):
        return min(10.0, 0.2 + 0.015 * it)

    # step‑size schedule (decreases with time)
    def eta_of(it):
        return max(0.02, 0.5 / (1.0 + 0.001 * it))

    print(f"Starting search: n = {n}, budget = {budget_s}s")

    while time.time() < deadline:
        # ---- generate a fresh candidate ------------------------------------------------
        if best_seq is None or np.random.rand() < 0.25:
            # random initialisation

                a = _gen_sine(n)
            elif r < 0.75:

            else:

        else:
            # start from the current best and take a few gradient steps
            a = best_seq.copy()

            steps = min(5, int(5 + 0.001 * (time.time() - deadline + budget_s)))
            for _ in range(steps):
