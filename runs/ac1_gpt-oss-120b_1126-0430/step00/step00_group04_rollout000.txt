Propose a new height sequence for the step‑function problem.

Algorithm
---------
* Work on the probability simplex (non‑negative entries that sum to 1).  
  The evaluation function is scale‑invariant, so fixing the sum to 1 is
  safe and makes the optimisation well‑behaved.
* Use a **projected sub‑gradient descent** on the objective  
      f(a) = max_k (a*a)[k]                     (self‑convolution L∞ norm)
  The sub‑gradient at the index of the maximal entry `k*` is
      g_i = 2·a_{k*‑i}   (if 0 ≤ k*‑i < n, otherwise 0).
* After each gradient step the vector is projected back onto the simplex
  with the O(n log n) algorithm of Duchi et al.
* Several random restarts are performed while time permits.  The best
  sequence seen is kept and returned.
* Occasionally (every 200 iterations) a tiny random perturbation is added
  to help escape flat regions.
* The routine works with any length; if the global ``height_sequence_1``
  exists its length is used (the current best construction has 6396
  entries).  The whole search runs inside the supplied time budget
  (default 1000 s) and never exceeds it.

import time
import numpy as np

def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the probability simplex:
        {x | x_i >= 0, sum x_i = 1}
    Implements the algorithm from
    Duchi, Shalev‑Shwartz, Singer & Chandra (2008).
    """
    # sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    # find the largest rho where the condition holds
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback (should not happen for reasonable v)
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

    C = A * A
    conv = np.fft.irfft(C, n=size_fft)[: 2 * len(a) - 1]
    return conv

    # Evaluate the existing construction if it exists
    if 'height_sequence_1' in globals():
        try:
            init_seq = list(height_sequence_1)
            init_score = evaluate_sequence(init_seq)
            print(f"[Info] Initial known sequence score: {init_score:.6f}")
            best_seq = init_seq
            best_score = init_score
        except Exception as e:
            print(f"[Warning] Could not evaluate height_sequence_1: {e}")

    # ------------------------------------------------------------------
    # Sub‑gradient descent parameters
    # ------------------------------------------------------------------
    beta = 0.9               # momentum coefficient
    init_lr = 5.0            # initial learning rate (will decay)
    min_iter_per_restart = 200   # ensure some progress before checking time

    restart_counter = 0

    # ------------------------------------------------------------------
    # Main optimisation loop – many random restarts until time runs out
    # ------------------------------------------------------------------
    while time.time() < deadline:
        restart_counter += 1
        # ---- Random start (occasionally warm‑start from best) ----
        if best_seq is not None and np.random.rand() < 0.2:
            a = np.array(best_seq, dtype=float)
            a += 0.005 * np.random.randn(n)      # gentle perturbation
            a = np.clip(a, 0.0, None)
        else:
            a = np.random.rand(n)

        # Normalise to the simplex
        a_sum = a.sum()
        if a_sum <= 0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / a_sum

        # Momentum buffer
        v = np.zeros_like(a)

        iter_idx = 0
        # ------------------------------------------------------------------
        # Inner optimisation – projected sub‑gradient descent
        # ------------------------------------------------------------------
        while time.time() < deadline:
            iter_idx += 1

            # Compute self‑convolution and locate the maximal entry
            conv = _conv_self_fft(a, size_fft)
            k_max = int(np.argmax(conv))

            # Build the sub‑gradient of the max‑entry w.r.t. a
            # g_i = 2 * a_{k_max - i}  if the partner index lies inside [0,n)
            grad = np.zeros_like(a)
            idx = np.arange(n)
            partner = k_max - idx
            mask = (partner >= 0) & (partner < n)
            grad[idx[mask]] = 2.0 * a[partner[mask]]

            # Momentum update
            v = beta * v + (1.0 - beta) * grad

            # Decaying learning rate
            lr = init_lr / np.sqrt(iter_idx)

            # Gradient step and projection onto the simplex
            a_candidate = a - lr * v
            a = _simplex_projection(a_candidate)

            # Occasionally add a tiny random jitter to avoid stagnation
            if iter_idx % 200 == 0:
                jitter = 0.001 * np.random.randn(n)
                a += jitter
                a = np.clip(a, 0.0, None)
                a = _simplex_projection(a)

            # Evaluate the current point every few iterations
            if iter_idx % 20 == 0:
                try:
                    score = evaluate_sequence(a.tolist())
                except Exception:
                    # Should not happen, but guard against unexpected errors
                    score = float("inf")
                if score < best_score:
                    best_score = score
                    best_seq = a.tolist()
                    print(f"[Update] Restart {restart_counter}, iter {iter_idx}, "
                          f"new best score = {best_score:.6f}")
                    # Early stop if we reach the target bound
                    if best_score < 1.503:
                        print("[Success] Target bound reached – exiting early.")
                        return best_seq

            # Stop the inner loop if we already performed enough iterations
            # w.r.t. the remaining time (to keep the outer restarts responsive)
            if iter_idx >= min_iter_per_restart and (time.time() + 0.5) > deadline:
                break

        # End of one restart – final evaluation of the obtained point
        try:
            final_score = evaluate_sequence(a.tolist())
        except Exception:
            final_score = float("inf")
        if final_score < best_score:
            best_score = final_score
            best_seq = a.tolist()
            print(f"[Restart End] Restart {restart_counter} gave "
                  f"score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence found (fallback to a single element if needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        # Should never happen, but be defensive
        return [1.0]
    print(f"[Result] Best score after all restarts: {best_score:.6f}")
    return best_seq
