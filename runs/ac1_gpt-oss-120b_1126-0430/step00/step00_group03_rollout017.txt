def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import numpy as np, time, random, math
    rng = random.Random(seed)
    deadline = time.time() + budget_s - 5
    n = len(height_sequence_1) if 'height_sequence_1' in globals() else 2000

    best_seq = None
    iteration = 0
    while time.time() < deadline:
        S = greedy_sidon(n, rng)
        seq = np.zeros(n, dtype=float)
        if len(S) == 0: # improbable
            continue
        seq[list(S)] = 1.0
        score = evaluate_sequence(seq.tolist())
        iteration += 1
        if score < best_score:
            best_score = score
            best_seq = seq.tolist()
            print(f'Iter {iteration}: New best score {best_score:.6f} with m={len(S)}')
        # early exit if below target
        if best_score < 1.5:
            break
    if best_seq is None:
        # fallback uniform
        best_seq = [1.0] * n
    return best_seq

def propose_candidate(seed=123, budget_s=1000, **kwargs):
    import numpy as np, time, random
    from scipy.optimize import linprog
    rng = np.random.RandomState(seed)
    deadline = time.time() + budget_s - 5

    # Use length from height_sequence_1 if exists else 2000
    n = len(height_sequence_1) if 'height_sequence_1' in globals() else 2000

    # Initialize with uniform distribution
    seq = np.full(n, np.sqrt(2.0*n) / n)  # ensure sum sqrt(2n)
    best_seq = seq.copy()
    best_score = evaluate_sequence(seq.tolist())
    print(f'Initial score {best_score:.6f}')

    # Prepare some t values
    t_grid = [0.01, 0.03, 0.07, 0.15, 0.3, 0.5, 0.8]

    iteration = 0
    while time.time() < deadline:
        # Get direction via LP
        # Normalize seq to sqrt(2n) sum
        sum_seq = seq.sum()
        f_norm = seq * np.sqrt(2.0 * n) / sum_seq

        M = np.max(np.convolve(f_norm, f_norm))

        # solve LP
        g = solve_convolution_lp(f_norm, M)
        if g is None:
            # fallback: random perturbation
            idx = rng.randint(n)
            seq[idx] = max(0.0, seq[idx] + rng.randn() * 0.01)
            continue

        # Normalize g
        sum_g = np.sum(g)
        g_norm = g * np.sqrt(2.0 * n) / sum_g

        # line search for best t
        best_local_seq = None
        best_local_score = float('inf')
        for t in t_grid:
            new_seq = (1 - t) * seq + t * g_norm
            # ensure nonnegative
            new_seq = np.maximum(new_seq, 0.0)
            score = evaluate_sequence(new_seq.tolist())
            if score < best_local_score:
                best_local_score = score
                best_local_seq = new_seq

        iteration += 1
        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local_seq.copy()
            print(f'Iter {iteration}: improvement to {best_score:.6f}')
        # accept new seq (even if not better?) we may move to best_local_seq
        seq = best_local_seq

        # early exit if below target
        if best_score < 1.5:
            break

    # Return best_seq as list
    return best_seq.tolist()

import numpy as np
import time
from scipy.optimize import linprog
from scipy.linalg import toeplitz

def solve_convolution_lp(f, rhs):
    n = len(f)
    # Build Toeplitz matrix for convolution f * x (ordered pairs)
    col = np.concatenate([f, np.zeros(n-1)])
    row = np.concatenate([f[:1], np.zeros(n-1)])
    A = toeplitz(col, row)   # shape (2n-1, n)
    # constraints: A @ x <= rhs
    c = -np.ones(n)  # maximize sum(x)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, rhs), bounds=bounds,
                  method='highs', options={'disp': False, 'time_limit': 5.0})
    if res.success:
        return res.x
    return None

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Iterative improvement of a non‑negative step function.
    Starts from the uniform (optimal‑norm) sequence, repeatedly
    solves a linear programme that finds a direction which can
    increase the L1‑mass without increasing the L∞‑norm of the
    self‑convolution, and moves a configurable step size in that
    direction.  A small line‑search over several step sizes is used
    to obtain the largest decrease of the evaluation function.
    The process runs until the budget expires or the target bound
    (≈1.5) is reached.
    """
    import numpy as np, time, random
    # Length of the sequence – use the length of the known construction
    # but cap it to keep the linear programme tractable.
    if 'height_sequence_1' in globals():

    else:
        max_len = 2000

    # Normalisation: we keep sum(seq) = sqrt(2*n) because then the
    # evaluation equals the maximal convolution value.
    target_sum = np.sqrt(2.0 * n)
    seq = np.full(n, target_sum / n)            # uniform start
    best_seq = seq.copy()
    best_score = evaluate_sequence(seq.tolist())
    print(f'Initial score {best_score:.6f} (uniform)')

    t_grid = [0.01, 0.03, 0.07, 0.12, 0.2, 0.35, 0.5]

    while time.time() < deadline:

        cur_sum = np.sum(seq)
        if cur_sum == 0:

            idx = random.randrange(n)
            seq[idx] = max(0.0, seq[idx] + random.gauss(0, 0.005))

        best_local_seq = None

"""
Iterative improvement of a non‑negative step function.

The algorithm works with a sequence whose L¹‑norm is fixed to
√(2·n) (n = length of the sequence).  For such a normalised sequence
the evaluation returned by ``evaluate_sequence`` equals the maximal
value of the self‑convolution, i.e.  eval = max (f∗f).  Starting from
the uniform (optimal‑norm) sequence we repeatedly:

1. Compute the current bound M = max(f∗f).
2. Solve the linear programme

      maximise   Σ b_j
      subject to  (f ∗ b)_k ≤ M   for all k
                  b_j ≥ 0

   The solution g gives a direction that can increase the total mass
   without raising the convolution bound.
3. Normalise g to the same L¹‑mass as f.
4. Perform a line‑search over several mixing coefficients t ∈ (0,1)
   and keep the convex combination  (1‑t)·f + t·g that yields the
   smallest evaluation.  This step strictly reduces the maximal
   convolution value whenever possible.
5. Keep the best sequence seen so far.

The loop runs until the time budget expires (or the target bound
≈1.5 is reached).  The implementation uses ``scipy.optimize.linprog``
with the efficient “highs’’ solver and builds the convolution matrix
once per iteration via a Toeplitz construction.  Random perturbation
is used as a fallback when the LP fails.
"""

import time
import random
import numpy as np
from scipy.optimize import linprog
from scipy.linalg import toeplitz

def solve_convolution_lp(f_seq: np.ndarray, rhs: float):
    """
    Solve the LP:
        maximise   sum(b)
        subject to  conv(f_seq, b) <= rhs  (component‑wise)
                    b >= 0
    Returns the optimal b as a 1‑d NumPy array, or None if the LP
    fails.  ``f_seq`` must be non‑negative.
    """
    n = f_seq.size
    # Toeplitz matrix that maps b to conv(f_seq, b)
    # A[k, j] = f_seq[k - j] if 0 <= k-j < n else 0
    col = np.concatenate([f_seq, np.zeros(n - 1)])
    row = np.concatenate([f_seq[:1], np.zeros(n - 1)])
    A = toeplitz(col, row)          # shape (2*n-1, n)

    # Inequalities: A @ b <= rhs
    c = -np.ones(n)                 # maximise sum(b)  == minimise -sum(b)
    bounds = [(0, None)] * n        # b_j ≥ 0
    b_ub = np.full(2 * n - 1, rhs)

    # Use the fast HiGHS solver; give a modest time limit per call.
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'disp': False, 'time_limit': 5.0})
    if res.success:
        return res.x
    return None


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises
    ``evaluate_sequence``.  The routine respects the 1000‑second budget
    and returns the best list of floats found.
    """
    rnd = random.Random(seed)

    # -----------------------------------------------------------------
    # 1. Choose the problem size.
    #    Use the length of the known construction if available,
    #    but cap it to keep the LP cheap.
    # -----------------------------------------------------------------
    if 'height_sequence_1' in globals():
        max_len = len(height_sequence_1)
    else:
        max_len = 2000
    n = min(max_len, 800)                 # a comfortable size for the LP

    # -----------------------------------------------------------------
    # 2. Normalise to sum = sqrt(2*n) – then the evaluation equals
    #    the maximal convolution entry.
    # -----------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    seq = np.full(n, target_sum / n)      # uniform start (already normalised)

    best_seq = seq.copy()
    best_score = evaluate_sequence(seq.tolist())
    print(f'Initial score (uniform) : {best_score:.6f}')

    # -----------------------------------------------------------------
    # 3. Main optimisation loop.
    # -----------------------------------------------------------------
    deadline = time.time() + budget_s - 5.0   # a small safety margin
    iteration = 0
    # Step‑size candidates for the line search.
    t_grid = [0.01, 0.03, 0.07, 0.12, 0.20, 0.35, 0.50]

    while time.time() < deadline:
        iteration += 1

        # -----------------------------------------------------------------
        # (a) Normalise the current iterate (floating point drift may happen)
        # -----------------------------------------------------------------
        cur_sum = seq.sum()
        if cur_sum == 0.0:
            seq = np.full(n, target_sum / n)
            cur_sum = target_sum
        seq_norm = seq * (target_sum / cur_sum)   # guarantees sum = target_sum

        # -----------------------------------------------------------------
        # (b) Current bound M = max( f ∗ f )
        # -----------------------------------------------------------------
        conv_self = np.convolve(seq_norm, seq_norm)
        M = np.max(conv_self)

        # -----------------------------------------------------------------
        # (c) Solve the LP to obtain a feasible direction g.
        # -----------------------------------------------------------------
        g = solve_convolution_lp(seq_norm, M)
        if g is None:
            # LP failed – random small perturbation to escape possible stalls.
            idx = rnd.randrange(n)
            seq[idx] = max(0.0, seq[idx] + rnd.gauss(0.0, 0.005))
            continue

        # -----------------------------------------------------------------
        # (d) Normalise g to the same L¹‑mass as seq_norm.
        # -----------------------------------------------------------------
        g_sum = g.sum()
        if g_sum == 0.0:
            continue
        g_norm = g * (target_sum / g_sum)

        # -----------------------------------------------------------------
        # (e) Line–search: try several mixing fractions t and keep the best.
        # -----------------------------------------------------------------
        best_local_seq = None
        best_local_score = float('inf')
        for t in t_grid:
            cand = (1.0 - t) * seq_norm + t * g_norm
            # The security checks in ``evaluate_sequence`` clip to [0,1000];
            # we do the clipping ourselves to avoid NaNs.
            cand = np.clip(cand, 0.0, 1000.0)

            score = evaluate_sequence(cand.tolist())
            if score < best_local_score:
                best_local_score = score
                best_local_seq = cand

        # -----------------------------------------------------------------
        # (f) Accept the locally best candidate.
        # -----------------------------------------------------------------
        if best_local_seq is None:
            continue
        seq = best_local_seq
        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = seq.copy()
            print(f'Iter {iteration:4d}: new best = {best_score:.6f}')
            # Early exit if we already beat the target bound.
            if best_score < 1.5:
                break

    # -----------------------------------------------------------------
    # 4. Return the best sequence discovered (as plain Python list).
    # -----------------------------------------------------------------
    return best_seq.tolist()
