def smooth(seq, window=5):
    if window <= 1: return seq
    kernel = np.ones(window)/window
    sm = np.convolve(seq, kernel, mode='same')
    # ensure nonnegative and maintain sum
    sm = np.maximum(sm, 0.0)
    # renorm to original sum
    total = np.sum(seq)
    total_sm = np.sum(sm)
    if total_sm > 0:
        sm *= total / total_sm
    else:
        sm = seq.copy()
    return sm.tolist()
```

def perturb(seq, scale=0.01):
    new = seq.copy()
    for _ in range(5):
        idx = np.random.randint(len(seq))
        new[idx] = max(0.0, new[idx] + np.random.randn()*scale)
    # enforce sum > 0.01: adjust if sum too low
    total = np.sum(new)
    if total < 0.01:
        # rescale with positive shift
        add = (0.01 - total)/len(new)
        new = [x + add for x in new]
    return new
```

    if sum_seq < 0.01:

    target_sum = np.sqrt(2*n)

def propose_candidate(seed=123, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2

    best_seq = None
    best_score = np.inf

    # Phase 1: parameter search - generate many candidate shapes and evaluate
    param_start = time.time()
    while time.time() < deadline and time.time() - param_start < 20: # first 20 seconds for shape search
        n = np.random.randint(200, 2000)  # length
        typ = np.random.choice(['geo','hyper','power','gauss','rand'])
        if typ == 'geo':
            q = np.random.uniform(0.1, 0.9)
            seq = (q ** np.arange(n)).tolist()
        elif typ == 'hyper':
            seq = (1.0/(np.arange(n)+1)).tolist()
        elif typ == 'power':
            p = np.random.uniform(0.2, 2.0)
            seq = (1.0/(np.arange(n)+1)**p).tolist()
        elif typ == 'gauss':
            sigma = np.random.uniform(0.1*n, 0.4*n)
            center = np.random.uniform(0, n-1)
            i = np.arange(n)
            seq = np.exp(-0.5*((i-center)/sigma)**2).tolist()
        elif typ == 'rand':
            seq = np.random.rand(n).tolist()
        # Ensure non-negative and sum>0
        total = sum(seq)
        if total < 0.01:
            continue
        # evaluate
        score = evaluate_sequence(seq)
        if score < best_score:
            best_score = score
            best_seq = seq
            print(f"Initial candidate: type {typ}, n={n}, score={best_score:.6f}")
    # If no candidate, fallback to height_sequence_1 or random
    if best_seq is None:
        best_seq = [float(x) for x in height_sequence_1] if 'height_sequence_1' in globals() else [np.random.rand() for _ in range(500)]
        best_score = evaluate_sequence(best_seq)

    # Phase 2: iterative improvement using LP, smoothing, perturb
    curr_seq = best_seq.copy()
    iter_cnt = 0
    while time.time() < deadline:
        iter_cnt += 1
        # Choose action
        r = np.random.rand()
        if r < 0.4:
            # LP direction (gives candidate)
            cand_seq = get_lp_candidate(curr_seq)
        elif r < 0.7:
            # smoothing
            win = np.random.choice([3,5,7,9])
            cand_seq = smooth(curr_seq, window=win)
        else:
            # perturb
            cand_seq = perturb(curr_seq, scale=np.random.rand()*0.2)
        # ensure positivity and clip
        cand_seq = [max(0.0, min(1000.0, x)) for x in cand_seq]
        # evaluate
        try:

"""
Propose a candidate step‑function (list of non‑negative heights) that tries to
minimise the evaluation function

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

using a hybrid search:

1.   A short “parameter sweep” creates many simple parametric families
     (geometric, hyperbolic, power‑law, Gaussian, random).  The best of those
     seeds the main search.

2.   The main loop repeatedly improves the current candidate by one of three
     stochastic operators:
        • LP‑direction: solve the linear programme that maximises the L1‑norm
          of a new sequence `g` under the constraint `conv(f, g) ≤ M`,
          where `M` is the current maximal self‑convolution of `f`.  The
          resulting `g` is rescaled to the same total mass as `f` and used as
          a new candidate.
        • Smoothing: a short moving‑average filter spreads mass and often
          reduces the peak of the self‑convolution.
        • Small random perturbations of a few entries.

   Improvements are kept if they lower the evaluation score; the current
   sequence is also occasionally replaced by the new one (even if not
   improving) to escape shallow local minima.

The algorithm respects the 1000 s budget, prints occasional progress
information and finally returns the best sequence found.
"""

import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Helper functions (must be top‑level – no closures)
# ----------------------------------------------------------------------


def solve_convolution_lp(f_seq, rhs):
    """
    Solve the LP:
        maximise   sum(b)
        subject to conv(f_seq, b) <= rhs   (for every shift)
                   b >= 0
    Returns the optimal `b` as a NumPy array or None on failure.
    """
    n = len(f_seq)
    # maximise sum(b)  → minimise -1·b
    c = -np.ones(n)

    # Build the convolution constraints: for each shift k we need
    #   sum_i f_i * b_{k-i} <= rhs
    # This is a Toeplitz‑like matrix.
    rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # i runs over indices of f_seq, j = k - i is the index in b
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # Fill the non‑zero entries
        for i in range(i_min, i_max + 1):
            j = k - i
            row[j] = f_seq[i]
        rows.append(row)
    a_ub = np.vstack(rows)

    # Non‑negativity constraints  ( -b <= 0 )
    a_ub = np.vstack([a_ub, -np.eye(n)])
    b_ub = np.hstack([np.full(2 * n - 1, rhs), np.zeros(n)])

    # Use the default 'highs' method which is fast and reliable.
    try:
        res = optimize.linprog(c, A_ub=a_ub, b_ub=b_ub, method='highs')
    except Exception:  # pragma: no cover
        return None

    if res.success:
        return res.x
    return None


def get_lp_candidate(seq):
    """
    Produce a new candidate using the LP direction described in the
    problem statement.  The returned sequence has the same length as `seq`
    and the same total mass (up to the rounding performed by the LP).
    """
    n = len(seq)
    sum_seq = np.sum(seq)
    if sum_seq < 0.01:   # avoid degenerate case
        return seq

    # Normalise to the standard L1‑norm sqrt(2n) used by the LP.
    target_norm = np.sqrt(2.0 * n)
    f_norm = np.array(seq) * (target_norm / sum_seq)

    # Current bound M = max(conv(f_norm, f_norm))
    max_conv = float(np.max(np.convolve(f_norm, f_norm)))

    # Solve the LP
    g = solve_convolution_lp(f_norm, max_conv)
    if g is None:
        return seq

    # Rescale g back so that its total mass matches the original seq.
    sum_g = np.sum(g)
    if sum_g < 1e-12:
        return seq
    g_rescaled = g * (sum_seq / sum_g)
    # Clip to the allowed range (evaluate_sequence later clips anyway)
    g_rescaled = np.clip(g_rescaled, 0.0, 1000.0)
    return g_rescaled.tolist()


def smooth(seq, window=5):
    """
    Simple moving‑average smoothing (same length output).  The total mass
    is restored to the original value.
    """
    if window <= 1:
        return seq
    kernel = np.ones(window) / window
    sm = np.convolve(seq, kernel, mode='same')
    sm = np.maximum(sm, 0.0)
    total_original = np.sum(seq)
    total_sm = np.sum(sm)
    if total_sm > 0:
        sm *= total_original / total_sm
    else:   # extremely unlikely, fall back to original
        sm = np.array(seq)
    return sm.tolist()


def perturb(seq, scale=0.05, n_changes=3):
    """
    Add Gaussian noise to a few random positions.
    """
    new_seq = np.array(seq, dtype=float)
    n = len(seq)
    for _ in range(n_changes):
        idx = np.random.randint(n)
        new_seq[idx] = max(0.0, new_seq[idx] + np.random.randn() * scale)
    # Re‑normalise to avoid the sum drifting to zero
    total = np.sum(new_seq)
    if total < 0.01:
        # force a tiny amount of mass uniformly
        new_seq += (0.01 - total) / n
    return new_seq.tolist()

    """

    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # 5 s safety margin

    # ------------------------------------------------------------------
    # Phase 0 – obtain an initial seed (either the known good one or a
    #           quick parametric sweep)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # If the global `height_sequence_1` exists, start from it.
    if 'height_sequence_1' in globals():
        best_seq = [float(x) for x in height_sequence_1]

    # Quick parametric sweep (≈20 s or until time runs out)
