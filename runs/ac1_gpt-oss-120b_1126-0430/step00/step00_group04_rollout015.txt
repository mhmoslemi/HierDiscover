def propose_candidate(seed=0, budget_s=1000):
    import numpy as np, time
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5

    # Determine length n
    try:
        n = len(height_sequence_1)
    except NameError:
        n = 6396

    # Precompute FFT size
    fft_len = 1 << ((2*n - 1).bit_length())

    def max_conv(a):
        # a is numpy 1D
        Fa = np.fft.rfft(a, fft_len)
        c = np.fft.irfft(Fa * Fa, fft_len)
        return c[:2*n-1].max()

    def objective(a):
        return 2.0 * n * max_conv(a)

    # Helper to evaluate and keep best
    best_a = None
    best_val = np.inf

    # Simulated annealing with multiple restarts
    restart_interval = 5.0  # seconds
    while time.time() < deadline:
        # start a random simplex vector
        # Dirichlet with parameter 1 (uniform random over simplex)
        a = rng.dirichlet(np.ones(n))
        # optional smoothing
        # compute initial objective
        cur_val = objective(a)

        # temperature schedule
        T0 = 0.1
        T = T0
        decay = 0.9995

        # inner loop time for this start
        inner_deadline = time.time() + restart_interval
        while time.time() < inner_deadline and time.time() < deadline:
            # pick source and target
            src = rng.integers(n)
            if a[src] == 0:
                continue
            # pick target biased towards lower values
            # choose among indices where a is minimal
            # compute probabilities
            # Choose dest uniformly among all other indices
            dst = rng.integers(n-1)
            if dst >= src:
                dst += 1
            # transfer amount
            # amount fraction relative to a[src] and temperature
            max_transfer = a[src] * min(0.5, T)  # up to 50% of src or T fraction
            delta = rng.random() * max_transfer

            a_new = a.copy()
            a_new[src] -= delta
            a_new[dst] += delta

            new_val = objective(a_new)
            # accept if better or with probability exp(-(new - cur)/T)
            if new_val < cur_val or rng.random() < np.exp((cur_val - new_val) / (T + 1e-12)):
                a = a_new
                cur_val = new_val
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"New best {best_val:.6f} at time {time.time():.1f}")

            T *= decay
            # occasionally random jitter to avoid stagnation
            if rng.random() < 0.001:
                # random perturb small noise across all elements small sum zero
                noise = rng.normal(scale=0.001, size=n)
                a = a + noise
                a = np.clip(a, 0, None)
                a_sum = a.sum()
                if a_sum == 0:
                    a = rng.dirichlet(np.ones(n))
                else:
                    a /= a_sum
                cur_val = objective(a)

    # after loop, produce final sequence
    if best_a is None:
        best_a = np.full(n, 1.0/n)
        best_val = objective(best_a)
        print("Fallback to uniform sequence")
    # scale to 0-1000
    output_seq = (best_a * 1000.0).tolist()
    print(f"Best value found: {best_val:.6f}")
    return output_seq

def project_simplex(v):
    # Projects onto the simplex {x >=0, sum x = 1}
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a step‑function (non‑negative heights) that minimizes the
evaluation function

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm works on the *shape* of the sequence, i.e. on a probability
vector p with sum(p)=1.  For such a vector the objective becomes
    2 * max_z (p * p)(z)                     (continuous limit),
so scaling the vector to the range [0,1000] does not change the score.
We therefore optimise a probability vector on the simplex.

The optimisation uses a smooth approximation of the max (log‑sum‑exp) and
gradient descent with projection onto the simplex.  The gradient is
computed analytically from the convolution using FFTs, which makes each
iteration fast even for a few‑thousand‑element vectors.  Several random
restarts are performed within the time budget and the best vector found
is returned (scaled to [0,1000]).

The code respects all contest constraints: it defines a top‑level
`propose_candidate` function, makes no file/ network I/O and only uses
NumPy / SciPy which are available in the execution environment.
"""
import time
import numpy as np
from scipy import optimize  # only imported for compatibility; not used

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x >= 0, sum(x)=1}."""
    if v.ndim != 1:
        raise ValueError("simplex projection works on 1‑D arrays")
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a sequence of heights that gives a low value of evaluate_sequence().
    The function runs for roughly `budget_s` seconds and returns the best
    sequence found (scaled to the range [0, 1000]).
    """
    rng = np.random.default_rng(seed)
    t_end = time.time() + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # problem size
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)                # type: ignore[name-defined]
    except Exception:
        n = 6396

    # FFT length for a*a (full convolution length = 2*n-1)
    fft_len = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # helper functions (fast convolution, objective, smooth gradient)
    # ------------------------------------------------------------------
    def _conv_max(a: np.ndarray) -> float:
        """Maximum entry of a * a using FFT."""
        Fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(Fa * Fa, n=fft_len)[: 2 * n - 1]
        return float(conv.max())

    def _objective(a: np.ndarray) -> float:
        """Exact objective for a probability vector (sum = 1)."""
        return 2.0 * n * _conv_max(a)

    # ------------------------------------------------------------------
    # initialise best solution (maybe from the provided construction)
    # ------------------------------------------------------------------
    best_a = None
    best_val = np.inf

    if "height_sequence_1" in globals():
        seq0 = np.array(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        seq0 = np.clip(seq0, 0.0, 1000.0)
        total = seq0.sum()
        if total > 0:
            a0 = seq0 / total
            val0 = _objective(a0)
            best_a = a0.copy()
            best_val = val0
            print(f"[init] baseline from height_sequence_1 -> {best_val:.6f}")

    # ------------------------------------------------------------------
    # optimisation parameters
    # ------------------------------------------------------------------
    restart_interval = 5.0                 # seconds per random restart
    min_beta = 5.0                         # start of the log‑sum‑exp smoothing
    max_beta = 200.0                       # maximum smoothing (approaches true max)
    beta_inc = 1.001                       # multiplicative increase per iteration
    init_lr = 0.5                          # initial learning rate
    lr_decay = 0.9995

    # ------------------------------------------------------------------
    # main optimisation loop (multiple random restarts)
    # ------------------------------------------------------------------
    while time.time() < t_end:
        # ---- random start on the simplex --------------------------------
        a = rng.dirichlet(np.ones(n))      # uniform random probability vector
        cur_val = _objective(a)

        beta = min_beta
        lr = init_lr

        inner_deadline = min(time.time() + restart_interval, t_end)

        # ----------------------------------------------------------------
        # iterative gradient descent with smooth max
        # ----------------------------------------------------------------
        while time.time() < inner_deadline:
            # 1) convolution a*a
            Fa = np.fft.rfft(a, n=fft_len)
            conv_full = np.fft.irfft(Fa * Fa, n=fft_len)[: 2 * n - 1]

            # 2) smooth approximation of the max (log‑sum‑exp)
            #    subtract the global max for numerical stability
            max_c = conv_full.max()
            exp_terms = np.exp(beta * (conv_full - max_c))
            w = exp_terms / exp_terms.sum()          # normalized weights, sum(w)=1

            # 3) gradient = 2 * (w ⋆ a)  (cross‑correlation)
            #    compute w ⋆ a via FFT (linear correlation)
            corr_len = len(w) + n - 1                 # = 3*n-2
            N = 1 << ((corr_len).bit_length())       # power‑of‑two padding
            W = np.fft.rfft(w, n=N)
            A = np.fft.rfft(a, n=N)
            corr_full = np.fft.irfft(W * np.conj(A), n=N)[:corr_len]
            # cross‑correlation g_i = sum_j w[i+j] * a[j]
            g = corr_full[n - 1 : n - 1 + n]          # length n
            grad = 2.0 * g

            # 4) gradient step + projection onto simplex
            a = a - lr * grad
            a = _project_simplex(a)

            # 5) schedule updates
            beta = min(beta * beta_inc, max_beta)
            lr = max(lr * lr_decay, 1e-7)

            # occasional evaluation of the true objective
            if rng.random() < 0.01:
                cur_val = _objective(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[{time.time():.1f}s] new best = {best_val:.6f}")

        # end of inner loop – evaluate once more
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.1f}s] restart improvement -> {best_val:.6f}")

    # ------------------------------------------------------------------
    # fallback in the unlikely case we never found a candidate
    # ------------------------------------------------------------------
    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_val = _objective(best_a)
        print("[fallback] using uniform distribution")

    # ------------------------------------------------------------------
    # scale to the range required by evaluate_sequence (max 1000)
    # ------------------------------------------------------------------
    out_seq = (best_a * 1000.0).tolist()
    print(f"[final] best objective = {best_val:.6f}")
    return out_seq
