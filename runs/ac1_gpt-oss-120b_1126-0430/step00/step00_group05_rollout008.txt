        conv = conv_fft(seq)
        max_idx = np.argmax(conv)
        max_val = conv[max_idx]
        # find pair with max contribution
        i_start = max(0, max_idx - (n-1))
        i_end = min(n-1, max_idx)
        # vector of contributions
        i_vals = np.arange(i_start, i_end+1)
        j_vals = max_idx - i_vals
        contrib = seq[i_vals] * seq[j_vals]
        # choose index of max contribution
        idx = np.argmax(contrib)
        i = i_vals[idx]
        j = j_vals[idx]
        # choose other two indices randomly
        p, q = np.random.choice(n, 2, replace=False)
        # ensure p,q not equal to i or j (optional)
        # compute epsilon limited by current values
        eps = eps_factor * min(seq[i], seq[j])
        if eps < 1e-8:
            # if too small, shrink eps_factor or break
            eps_factor *= 0.5
            continue
        # transfer epsilon from i and j to p and q
        seq[i] -= eps
        seq[j] -= eps
        seq[p] += eps
        seq[q] += eps
        # enforce nonnegativity
        seq = np.maximum(seq, 0)
        # renormalize sum to sqrt(2n)
        total = seq.sum()
        if total <= 0:
            seq = np.random.rand(n)
        seq = seq * (np.sqrt(2*n) / total)
        # evaluate

                seq = best_seq.copy()

def _conv_max(seq: np.ndarray):
    """
    Return the full linear convolution of seq with itself,
    together with the index of the maximal entry and the maximal value.
    Uses an FFT of length that is a power of two for speed.
    """
    n = seq.shape[0]
    size = 1 << ((2 * n - 1).bit_length())        # next power of two
    f = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(f * f, n=size)[: 2 * n - 1]
    k = int(np.argmax(conv))
    return conv, k, float(conv[k])
