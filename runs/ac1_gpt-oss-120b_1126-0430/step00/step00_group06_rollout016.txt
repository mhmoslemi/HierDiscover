    while time.time() < deadline:
        # Compute conv and max
        conv = np.convolve(seq, seq)
        max_val = np.max(conv)

        # attempt gradient move
        # compute products for this max shift
        products = np.empty(n)
        products.fill(0.0)
        for i in range(n):
            j = max_idx - i
            if 0 <= j < n:
                products[i] = seq[i] * seq[j]
        # find i with largest product
        i_high = int(np.argmax(products))
        # find i with smallest product (or zero)
        i_low = int(np.argmin(products))

            new_seq = seq.copy()

            # ensure non-negative and bounded
            # clip to 1000 maybe not needed
            score = evaluate_sequence(new_seq)
            if score < best_score:
                best_score = score
                best_seq = new_seq

                        best_seq = cand[:]

"""
Propose a low‑evaluation step‑function using a hybrid of:
* a greedy “mass‑relocation’’ step that reduces the current maximal
  convolution value,
* occasional linear‑programming direction updates (the published SOTA
  method) with a larger mixing factor,
* random jitter to escape plateaus.

The algorithm keeps the total ℓ₁‑norm constant (the evaluation is
scale‑invariant) and iteratively lowers the peak of the convolution,
which directly reduces the score 2·n·max(b)/(sum(a)²).
"""

import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Helper: linear‑programming direction (identical to the reference code)
# ----------------------------------------------------------------------
def solve_convolution_lp(f_seq, rhs):
    """Linear program: max Σb  s.t.  conv(f_seq, b) ≤ rhs , b ≥ 0."""
    n = len(f_seq)
    c = -np.ones(n)                     # maximise sum(b)
    a_ub = []
    b_ub = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_seq[i]
        a_ub.append(row)
        b_ub.append(rhs)
    # non‑negativity constraints
    a_ub_nonneg = -np.eye(n)
    b_ub_nonneg = np.zeros(n)
    a_ub = np.vstack([a_ub, a_ub_nonneg])
    b_ub = np.hstack([b_ub, b_ub_nonneg])

    # use HiGHS (the default) – fast and reliable
    res = optimize.linprog(c, A_ub=a_ub, b_ub=b_ub,
                           method='highs',
                           options={'time_limit': 5.0})
    return res.x if res.success else None


def get_lp_direction(seq):
    """Return a new sequence in the direction of the LP‑maximiser."""
    n = len(seq)
    s = np.sum(seq)
    if s < 1e-12:
        return None
    # normalise to the standard ℓ₁‑norm sqrt(2n) used by the SOTA method
    target_sum = np.sqrt(2.0 * n)
    f_norm = seq * (target_sum / s)

    rhs = np.max(np.convolve(f_norm, f_norm))
    g_raw = solve_convolution_lp(f_norm, rhs)
    if g_raw is None:
        return None

    # renormalise g to the same ℓ₁‑norm
    s_g = np.sum(g_raw)
    if s_g < 1e-12:
        return None
    g_norm = g_raw * (target_sum / s_g)
    return g_norm.tolist()


# ----------------------------------------------------------------------
# Helper: greedy mass‑relocation step
# ----------------------------------------------------------------------
def greedy_mass_step(arr, step_factor=0.05):
    """
    Move a small amount of mass from the pair of indices that
    contributes most to the current maximal convolution value
    towards the extremes (indices 0 and n‑1).  The ℓ₁‑norm stays
    constant, so the score can only improve if the peak is reduced.
    """
    n = len(arr)
    conv = np.convolve(arr, arr)
    max_val = conv.max()
    k_max = int(conv.argmax())

    # feasible i where both i and j=k_max-i lie inside [0,n-1]
    i_low = max(0, k_max - (n - 1))
    i_high = min(n - 1, k_max)
    idx = np.arange(i_low, i_high + 1)
    jdx = k_max - idx
    products = arr[idx] * arr[jdx]

    if products.size == 0:
        return None

    # pair with largest contribution to the peak
    arg = int(products.argmax())
    i_a = int(idx[arg])
    i_b = int(jdx[arg])

    # choose a destination index that does not participate in the peak
    # (if possible).  For k_max < n‑1 we can use the rightmost index,
    # otherwise the leftmost.
    if k_max < n - 1:
        i_dest = n - 1
    else:
        i_dest = 0
    # avoid overwriting the source indices
    if i_dest in (i_a, i_b):
        i_dest = 0 if i_dest == n - 1 else n - 1

    # amount to move – a fraction of the smaller of the two participants
    delta = step_factor * min(arr[i_a], arr[i_b])
    if delta <= 0.0:
        return None

    new_arr = arr.copy()
    # take mass from the two contributors (if they are different)
    new_arr[i_a] = max(0.0, new_arr[i_a] - delta)
    if i_b != i_a:
        new_arr[i_b] = max(0.0, new_arr[i_b] - delta)
        total_removed = 2.0 * delta
    else:
        total_removed = delta

    # deposit the removed mass at the destination (split evenly if we took
    # from two positions)
    new_arr[i_dest] += total_removed

    # safeguard against the built‑in clipping (values >1000 are reduced)
    np.clip(new_arr, 0.0, 1000.0, out=new_arr)
    return new_arr.tolist()

    end_time = time.time() + budget_s - 5.0   # keep a few seconds margin
    it = 0
    last_print = time.time()

    # ------------------------------------------------------------------
    # main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < end_time:
        it += 1

        # ----------------------------------------------------------------
        # 1) greedy mass‑relocation step (cheap, most frequent)
        # ----------------------------------------------------------------
        new_seq = greedy_mass_step(seq, step_factor=0.07)
        if new_seq is not None:
            cur_score = evaluate_sequence(new_seq)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = np.array(new_seq, dtype=float)
                seq = best_seq.copy()
                # fast success – we keep the improvement and skip the rest of
                # this iteration
                continue

        # ----------------------------------------------------------------
        # 2) occasional LP direction + larger mixing factor
        # ----------------------------------------------------------------
        if it % 30 == 0:
            g = get_lp_direction(seq.tolist())
            if g is not None:
                # try a few mixing coefficients, keep the best that improves
                for t in (0.2, 0.1, 0.05):
                    cand = (1.0 - t) * seq + t * np.array(g, dtype=float)
                    cand = np.clip(cand, 0.0, 1000.0)
                    cur_score = evaluate_sequence(cand.tolist())
                    if cur_score < best_score:
                        best_score = cur_score
                        best_seq = cand.copy()
                        seq = best_seq.copy()
                        break   # accept the first improvement

        # ----------------------------------------------------------------
        # 3) tiny random jitter – helps escaping tiny plateaus
        # ----------------------------------------------------------------
        if np.random.rand() < 0.03:
            idx = np.random.randint(len(seq))
            delta = np.random.randn() * 0.02  # small Gaussian perturbation
            cand = seq.copy()
            cand[idx] = max(0.0, cand[idx] + delta)
            cur_score = evaluate_sequence(cand.tolist())
            if cur_score < best_score:
                best_score = cur_score
                best_seq = cand.copy()
                seq = best_seq.copy()

        # ----------------------------------------------------------------
        # occasional progress report (stdout)
        # ----------------------------------------------------------------
        if time.time() - last_print > 5.0:
            print(f"[{time.time():.1f}] it={it} best={best_score:.6f} "
                  f"len={len(best_seq)}")
            last_print = time.time()

    # final report
    print(f"Finished after {it} iterations – best score = {best_score:.6f}")
    return best_seq.tolist()
