def simplex_projection(v):
    # v is a numpy array; projects onto {x >= 0, sum x = 1}
    n = len(v)
    u = np.sort(v)[::-1]  # descending
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a step‐function (non‑negative sequence) with a low evaluation score.

Algorithm Overview
------------------
1. **Initial candidates** – generate several simple families (pure exponential,
   a mixture of two exponentials, random vectors) and keep the best one.
2. **LP refinement** – for the current sequence `f` compute its self‑convolution,
   find the maximum value `M` and an *active set* of shifts whose convolution
   values are close to `M`.  Solve a small linear program:
       maximise   sum(b)
       subject to conv(f,b)[k] ≤ M  for every active shift `k`
                  b ≥ 0
   The LP gives a direction `g` that can increase the total sum without
   violating the current bound.  Mix `f` and `g` with a step size `t` (several
   candidates for `t` are tried, the one giving the largest improvement is kept).
   Iterate until the time budget is exhausted.
3. **Local hill‑climbing** – once the LP stage stalls, perform cheap
   pair‑wise mass‑transfer moves that explicitly reduce the contribution to the
   most violating shift.  Accept moves that improve the score.

The whole search runs for at most the allotted budget (default 1000 s) and
returns the best sequence encountered.
"""

import time
import math
import numpy as np
from scipy import optimize, sparse

# Helper utilities
# ----------------------------------------------------------------------
def _simplex_projection(v):
    """Project a vector onto the probability simplex (non‑negative, sum = 1)."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _normalize_to_sum(seq, target):
    """Scale `seq` so that its ℓ₁‑norm becomes `target` (preserving sign)."""
    s = np.sum(seq)
    if s == 0:
        return seq
    return seq * (target / s)


def _evaluate(seq):
    """Thin wrapper around the provided evaluate_sequence (cached locally)."""
    # evaluate_sequence is assumed to be already imported in the environment
    return evaluate_sequence(seq.tolist() if isinstance(seq, np.ndarray) else seq)


# ----------------------------------------------------------------------
# LP solver – only the active (binding) convolution constraints are used
# ----------------------------------------------------------------------
def _solve_lp_active(f, M, active_shifts):
    """
    Solve:
        maximise   Σ b_j
        subject to Σ_i f_i * b_{k-i} ≤ M   for each k in `active_shifts`
                  b_j ≥ 0
    Returns the optimal b vector or None on failure.
    """
    n = f.shape[0]
    rows = []
    cols = []
    data = []
    for row_idx, k in enumerate(active_shifts):
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        for i in range(i_min, i_max + 1):
            j = k - i
            rows.append(row_idx)
            cols.append(j)
            data.append(f[i])
    A = sparse.csr_matrix((data, (rows, cols)), shape=(len(active_shifts), n))
    c = -np.ones(n)                     # maximise sum(b)  → minimise -sum(b)
    bounds = [(0, None)] * n            # b_j ≥ 0
    # use the high‑performance HiGHS interior‑point solver
    res = optimize.linprog(c,
                           A_ub=A,
                           b_ub=np.full(len(active_shifts), M),
                           bounds=bounds,
                           method='highs')
    if res.success:
        return res.x
    return None

    return result
