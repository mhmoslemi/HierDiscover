"""
propose_candidate

Search for a non‑negative step‑function (a list of heights) that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

The search is built around a simple family of power‑law sequences

    a_i ∝ (i + shift) ** (-alpha)

which are normalised to have unit sum (the evaluation function is
scale‑invariant).  For a wide range of exponents α the convolution
becomes much flatter than the uniform sequence and the score drops
well below the previous bound of 2.0.  The algorithm proceeds in three
stages:

1. **Grid search** – iterate over a grid of (α, shift) values, create the
   normalised power‑law sequence, evaluate it and keep the best one.
2. **Optional local refinement** – starting from the best sequence,
   repeatedly move a small amount of mass between two random positions
   (preserving the total sum) and accept the move if it improves the
   score.  This simple hill‑climbing can squeeze a little extra gain.
3. **Fallback** – if nothing beats the current best, return the
   previously found construction ``height_sequence_1`` (which is always
   available as a global variable).

The whole routine is time‑boxed (default 950 s) and prints occasional
progress messages.  It uses only NumPy and the already‑imported
``evaluate_sequence`` function.
"""

import time
import numpy as np

def _project_to_simplex(v):
    """Project a non‑negative vector onto the simplex {x >= 0, sum(x)=1}."""
    # Implementation of the algorithm from
    # Wang & Carreira‑Perpiñán, “Projection onto the probability simplex:
    # An efficient algorithm with a simple proof”, 2013.
    n = len(v)
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero – return uniform
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


def propose_candidate(seed=123, budget_s=950, **kwargs):
    """
    Generate a step‑function (list of non‑negative heights) that tries to
    minimise the provided ``evaluate_sequence`` value.

    Parameters
    ----------
    seed : int, optional
        Seed for NumPy's RNG (default=123).
    budget_s : float, optional
        Time budget in seconds.  The function will stop before this limit
        (a few seconds are left for the caller).
    **kwargs
        Ignored – kept for compatibility with the evaluation harness.

    Returns
    -------
    list[float]
        The best sequence found within the time budget.
    """
    np.random.seed(seed)
    start = time.time()
    # -----------------------------------------------------------------
    # 0. Length of the problem (use the length of the provided solution)
    # -----------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # provided by the environment
    except Exception:                     # very unlikely, but be safe
        n = 2244                            # the length mentioned in the description

    best_seq = list(height_sequence_1)      # fallback
    best_score = float('inf')
    try:
        best_score = evaluate_sequence(best_seq)
    except Exception:
        best_score = float('inf')

    print(f"[init] using length n={n}, baseline score={best_score:.6f}")

    # -----------------------------------------------------------------
    # 1. Grid search over power‑law families a_i = (i+shift)^(-alpha)
    # -----------------------------------------------------------------
    # The admissible region for alpha is (0, 1).  Values around 0.5‑0.7
    # are known to give very good scores for large n.
    alphas = np.linspace(0.45, 0.75, 61)           # step ≈0.005
    shifts = np.linspace(1.0, 2.0, 5)             # 1.0, 1.25, 1.5, 1.75, 2.0

    # Pre‑compute the index array once
    idx = np.arange(n, dtype=float)               # i = 0 … n‑1

    # small helper to evaluate a raw (not yet normalised) vector
    def _eval_raw(vec):
        # normalise to have sum = 1 (the score is scale‑invariant)
        if np.all(vec == 0):
            return float('inf')
        norm = vec / np.sum(vec)
        return evaluate_sequence(norm.tolist())

    # Grid loop
    for alpha in alphas:
        # early break if time is over
        if time.time() - start > budget_s - 5:
            break
        for shift in shifts:
            # Build the power‑law vector
            raw = np.power(idx + shift, -alpha)
            # Normalise and evaluate
            score = _eval_raw(raw)
            if score < best_score:
                best_score = score
                best_seq = (raw / np.sum(raw)).tolist()
                print(f"[grid] alpha={alpha:.4f}, shift={shift:.2f}, "
                      f"score={score:.6f}")

    print(f"[grid] best after grid search: score={best_score:.6f}")

    # -----------------------------------------------------------------
    # 2. Local refinement – simple hill climbing on the best vector
    # -----------------------------------------------------------------
    # Convert the best sequence to a NumPy array for fast updates.
    cur = np.array(best_seq, dtype=float)
    cur_score = best_score

    # Parameters for the perturbation
    max_iter = 20000                     # safety cap
    iter_cnt = 0
    while time.time() - start < budget_s - 5 and iter_cnt < max_iter:
        iter_cnt += 1
        # pick two distinct positions
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        # propose moving a small amount of mass from i to j
        # the step size is a fraction of the source coefficient
        step = cur[i] * (np.random.rand() * 0.02)   # up to 2 % of cur[i]
        if step <= 0:
            continue

        if cur[j] - step < 0:          # cannot make j negative
            continue

        new = cur.copy()
        new[i] = cur[i] - step
        new[j] = cur[j] + step

        # re‑project to the simplex to avoid tiny numerical drift
        # (projection is very cheap for a single move)
        new = _project_to_simplex(new)

        # evaluate
        try:
            new_score = evaluate_sequence(new.tolist())
        except Exception:
            continue

        if new_score < cur_score:
            cur = new
            cur_score = new_score
            # keep track of the global best as well
            if new_score < best_score:
                best_score = new_score
                best_seq = new.tolist()
                print(f"[refine] iter={iter_cnt}, improved score={new_score:.6f}")

    print(f"[final] best score found: {best_score:.6f}")
    return best_seq
