def evaluate(seq):

def normalize(seq):
    # sum to sqrt(2*n)
    n = len(seq)
    target = (2 * n) ** 0.5
    s = sum(seq)
    if s == 0: return seq
    factor = target / s
    seq = [max(0.0, x * factor) for x in seq]  # ensure nonneg maybe
    return seq

def compute_conv(seq):
    # compute convolution via FFT for speed
    a = np.array(seq, dtype=np.float64)
    conv = np.fft.ifft(np.fft.fft(a, n*2) * np.fft.fft(a, n*2)).real[:2*n-1]
    # due to rounding errors may have small negative values: clip to zero?
    return conv

def descent_step(seq, step_size):
    n = len(seq)
    conv = compute_conv(seq)
    max_val = conv.max()
    k_max = int(conv.argmax())
    # compute partner array
    # gradient g_i = 2*a_{k-max - i} if index valid else 0
    # we will move mass from i with large gradient to j with small gradient
    grad = np.zeros(n)
    for i in range(n):
        j = k_max - i
        if 0 <= j < n:
            grad[i] = 2 * seq[j]  # subgradient
    # Determine indices sorted by gradient.
    i_high = np.argmax(grad)
    i_low = np.argmin(grad)
    if grad[i_high] <= grad[i_low] + 1e-12:
        # no direction
        return seq
    # move epsilon from high to low
    epsilon = step_size * seq[i_high]  # maybe fraction
    if epsilon <= 0:
        epsilon = step_size
    # ensure not exceed low capacity? low can receive unlimited
    # ensure i_high >= epsilon
    epsilon = min(epsilon, seq[i_high])
    new_seq = seq.copy()
    new_seq[i_high] -= epsilon
    new_seq[i_low] += epsilon
    # renormalize to maintain sum; after move sum unchanged.
    # clip negative
    if new_seq[i_high] < 0:

    return new_seq

import numpy as np
import time

def normalize_to_target(seq):
    n = len(seq)
    target = np.sqrt(2.0 * n)
    s = np.sum(seq)
    if s <= 0:
        # assign uniform
        seq = np.full(n, target / n)
    else:
        seq = np.array(seq, dtype=np.float64) * (target / s)
    # enforce nonneg
    seq[seq < 0] = 0.0
    return seq

def compute_objective(seq):
    n = len(seq)
    s = np.sum(seq)
    if s < 1e-12:
        return np.inf
    # compute convolution via FFT
    # determine size as next power of two for speed
    m = 1
    while m < 2*n:
        m <<= 1
    # double length to avoid circular convolution
    a = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(a * a, n=m)[:2*n-1]
    max_b = np.max(conv)
    return 2.0 * n * max_b / (s * s)

def gradient_direction(seq):
    n = len(seq)
    # compute convolution via FFT
    m = 1
    while m < 2*n:
        m <<= 1
    a = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(a * a, n=m)[:2*n-1]
    k = np.argmax(conv)
    # subgradient g_i = 2*seq[k - i] if 0 <= k-i < n else 0
    g = np.zeros(n)
    for i in range(n):
        j = k - i
        if 0 <= j < n:
            g[i] = 2.0 * seq[j]
    return g, k, conv[k]

"""
Arcsine-convolution heuristic.

We construct a sequence whose entries follow the discrete arcsine distribution,
the continuous convolution square-root of the uniform distribution.
For this distribution the self‑convolution is (almost) constant,
so the evaluation function 2·n·max(b)/sum(a)^2 ≈ max(b) is close to the
theoretical lower bound 2n/(2n−1)≈1.  After an optional short projected‑gradient
refinement we return the best sequence found before the time budget expires.
"""

import time
import numpy as np

def _target_sum(n: int) -> float:
    """Return the L1‑norm that makes the evaluation function scale‑invariant."""
    return np.sqrt(2.0 * n)

def _normalize(seq: np.ndarray) -> np.ndarray:
    """Scale a non‑negative vector so that its L1‑norm equals the target sum."""
    n = seq.shape[0]
    target = _target_sum(n)
    s = seq.sum()
    if s == 0:
        # fall back to uniform distribution
        return np.full(n, target / n, dtype=np.float64)
    scaled = seq * (target / s)
    # Guard against tiny negative values caused by rounding
    scaled[scaled < 0] = 0.0
    return scaled

def _convolution(seq: np.ndarray) -> np.ndarray:
    """Fast (real) convolution a*a using an FFT of power‑of‑two length."""
    n = seq.shape[0]
    # length of linear convolution is 2n‑1; choose next power of two for speed
    m = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    # tiny negative values from round‑off are harmless but we clip for safety
    conv[conv < 0] = 0.0
    return conv

def _objective(seq: np.ndarray) -> float:
    """Exact evaluation used for internal optimisation."""
    n = seq.shape[0]
    total = seq.sum()
    if total < 1e-12:
        return np.inf
    conv = _convolution(seq)
    max_b = conv.max()
    return float(2.0 * n * max_b / (total * total))

def _arcsine_sequence(n: int) -> np.ndarray:
    """Discrete arcsine probabilities on n bins, scaled to the target sum."""
    i = np.arange(n, dtype=np.float64)
    x0 = i / n
    x1 = (i + 1) / n
    # CDF of the arcsine law: 2/π * arcsin(√x)
    cdf0 = (2.0 / np.pi) * np.arcsin(np.sqrt(x0))
    cdf1 = (2.0 / np.pi) * np.arcsin(np.sqrt(x1))
    w = cdf1 - cdf0                     # probabilities, sum≈1
    a = _target_sum(n) * w
    return a

def _gradient_step(seq: np.ndarray, step_frac: float) -> np.ndarray:
    """
    One projected‑gradient move.
    Takes a small amount of mass from the index that most contributes
    to the current maximal convolution entry and puts it to the index
    that contributes the least.
    """
    n = seq.shape[0]
    conv = _convolution(seq)
    k = int(conv.argmax())
    # contribution of each index to the maximal entry
    contrib = np.zeros(n, dtype=np.float64)
    for i in range(n):
        j = k - i
        if 0 <= j < n:
            contrib[i] = seq[i] * seq[j]
    # pick donor (largest contribution) and receiver (smallest)
    i_high = int(np.argmax(contrib))
    i_low = int(np.argmin(contrib))
    if i_high == i_low or seq[i_high] == 0.0:
        # fallback: a tiny random perturbation that respects the sum constraint
        delta = (np.random.rand() - 0.5) * step_frac * seq[i_high]
        new_seq = seq.copy()
        new_seq[i_high] = max(0.0, new_seq[i_high] + delta)
        new_seq = _normalize(new_seq)
        return new_seq
    eps = step_frac * seq[i_high]
    eps = min(eps, seq[i_high])          # never make the donor negative
    new_seq = seq.copy()
    new_seq[i_high] -= eps
    new_seq[i_low] += eps
    # sum is unchanged, but we renormalise to cure round‑off drift
    new_seq = _normalize(new_seq)
    return new_seq

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Generate a step‑function sequence that minimises the provided
    ``evaluate_sequence`` metric.

    The method consists of three phases:
        1. Build a deterministic starting point using the discrete arcsine
           distribution, which theoretically makes the self‑convolution
           nearly flat.
        2. (Optional) compare with the historic solution ``height_sequence_1``
           and keep the better one.
        3. Perform a fast projected‑gradient descent that repeatedly moves a
           small amount of mass from the indices that most increase the
           current maximal convolution entry to indices that have the smallest
           impact, always keeping the L1 norm fixed.
    The loop runs until the supplied time budget expires and the best sequence
    found is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + max(0.0, budget_s - 5.0)  # keep a few seconds margin

    # ------------------------------------------------------------------
    # 1. Determine the problem size (use the historic length when possible)
    # ------------------------------------------------------------------
    try:
        n = int(len(height_sequence_1))
        # Ensure it is a positive integer
        if n <= 0:
            raise Exception
    except Exception:
        n = 2000   # fallback size – 2000 works well within the time budget

    # ------------------------------------------------------------------
    # 2. Build the deterministic arcsine candidate
    # ------------------------------------------------------------------
    arcsine_seq = _arcsine_sequence(n)
    arcsine_seq = _normalize(arcsine_seq)

    # ------------------------------------------------------------------
    # 3. Load the previous best sequence (if available) and pick the better
    # ------------------------------------------------------------------
    best_seq = arcsine_seq
    best_score = _objective(best_seq)

    try:
        previous = np.array(height_sequence_1, dtype=np.float64)
        previous = _normalize(previous)
        prev_score = _objective(previous)
        if prev_score < best_score:
            best_seq = previous
            best_score = prev_score
            print(f"Started from historic sequence: score {best_score:.6f}")
        else:
            print(f"Arcsine start better: score {best_score:.6f}")
    except Exception:
        # No historic data – just use arcsine
        print(f"Arcsine start (n={n}): score {best_score:.6f}")

    cur_seq = best_seq.copy()
    cur_score = best_score

    # ------------------------------------------------------------------
    # 4. Main optimisation loop (projected‑gradient style)
    # ------------------------------------------------------------------
    step_frac = 0.05          # initial fraction of mass moved per step
    no_improve = 0
    iter_cnt = 0

    while time.time() < deadline:
        iter_cnt += 1

        # propose a new candidate
        cand_seq = _gradient_step(cur_seq, step_frac)
        cand_score = _objective(cand_seq)

        # accept if we made progress
        if cand_score < cur_score - 1e-12:
            cur_seq = cand_seq
            cur_score = cand_score
            no_improve = 0
            # keep global best
            if cand_score < best_score - 1e-12:
                best_seq = cand_seq.copy()
                best_score = cand_score
                print(f"Iter {iter_cnt:6d}: new best {best_score:.6f}")
        else:
            no_improve += 1

        # adapt step size and occasional random shake‑up
        if iter_cnt % 2000 == 0:
            step_frac = max(step_frac * 0.9, 0.001)

        if no_improve >= 5000:
            # random restart around the best point to escape possible plateaus
            perturb = np.random.randn(n) * 0.01 * best_seq.mean()
            cur_seq = best_seq + perturb
            cur_seq = np.maximum(cur_seq, 0.0)
            cur_seq = _normalize(cur_seq)
            cur_score = _objective(cur_seq)
            no_improve = 0
            step_frac = 0.05

    print(f"Finished after {iter_cnt} iterations – best objective {best_score:.6f}")

    # Return a plain Python list, as required by the evaluation harness
    return [float(x) for x in best_seq.tolist()]
