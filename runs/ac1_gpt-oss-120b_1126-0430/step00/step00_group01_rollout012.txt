def project_simplex(v):
    # return projection onto simplex {x >= 0, sum x = 1}
    # Using algorithm by Duchi et al.
    n = v.shape[0]
    u = np.sort(v)[::-1]  # descending
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def project_simplex(v):
    # From https://github.com/jaehyunp/stanfordacm/blob/master/code/ProjectOntoSimplex.py
    n = v.shape[0]
    u = np.sort(v)[::-1]

    w = np.maximum(v - theta, 0)

"""
Propose a sequence that minimizes the evaluation function.

The objective is  
    2 * n * max(conv(a, a)) / (sum(a)^2) ,

which is homogeneous: scaling a does not change the score.
Hence we fix sum(a)=1 and minimise  max(conv(a, a)).

Strategy
---------
1. Random initialisation of a non‑negative vector a (sum(a)=1).  A few
   restarts are performed and the best result is kept.
2. Sub‑gradient descent on the function f(a)=max_k (a*a)[k].
   For the active indices K where the maximum is attained the
   sub‑gradient is
        g_i = 2 * Σ_{k∈K} a_{k‑i}   (out‑of‑range indices give 0)
   which can be obtained efficiently with a single convolution.
3. After each gradient step the vector is projected onto the simplex
   {x≥0, Σx=1}.  This keeps the feasibility without extra clipping.
4. To avoid the trivial uniform fixed point (gradient zero) a small
   random jitter is added every few hundred iterations.
5. The current best sequence (lowest score) is stored and returned at
   the end of the allotted time.

The algorithm runs completely in memory, uses only NumPy and the
built‑in `time` module and respects the 1000 s budget.
"""

import numpy as np
import time

def _project_simplex(v):
    """Euclidean projection of v onto the simplex {x>=0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # sort descending
    cssv = np.cumsum(u) - 1                  # cumulative sum minus 1
    rho = np.where(u + cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                         # all entries will be zero
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _score_from_conv(conv, n):
    """Return the evaluation score for a known convolution (sum(a)=1)."""
    return float(2.0 * n * np.max(conv))

        evaluate_sequence).
    """

    best_seq = None
    best_score = float('inf')

    # we work with a relatively large n; the theoretical lower bound

    n = 4000

    # helper to compute current score quickly
    def current_score(a_vec):
        conv = np.convolve(a_vec, a_vec)
        return _score_from_conv(conv, n)

    # ------------------------------------------------------------------
    # main search loop – several independent restarts
    # ------------------------------------------------------------------
    restart = 0
    while time.time() < deadline:
        restart += 1
        # ------------------------------------------------------------------
        # initialise a new candidate
        # ------------------------------------------------------------------
        if np.random.rand() < 0.3 and 'height_sequence_1' in globals():
            # start from the provided SOTA construction if available
            a = np.array(height_sequence_1, dtype=float)
            a = np.maximum(a, 0.0)                 # guard against negatives
        else:

        a = a / np.sum(a)                # enforce sum = 1 (simplex)

        it = 0

        # inner optimisation loop (sub‑gradient descent + jitter)
        # ------------------------------------------------------------------

            it += 1
