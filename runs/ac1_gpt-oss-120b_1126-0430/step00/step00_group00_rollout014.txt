"""
propose_candidate
-----------------
Search for a step‑function (list of non‑negative heights) that gives a low
value of

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

where `conv` is a linear convolution.
The search is based on the observation that an exponentially decaying
profile `a_i = exp(-α·i)` flattens the self‑convolution and can push the
evaluation below the uniform‑sequence value 2.0.

The algorithm:
1. Choose a length `n`.  If the reference sequence `height_sequence_1`
   exists we reuse its length (the current construction uses 2244
   entries), otherwise a default of 2000 is used.
2. Scan a set of decay rates `α` (roughly 0.0002 … 0.002) and build the
   corresponding exponential sequences.
3. Evaluate each candidate with the provided `evaluate_sequence`
   function, keep the best (smallest) score.
4. As a tiny refinement step, perform a short simulated‑annealing /
   hill‑climbing walk starting from the best candidate: repeatedly move
   a small amount of mass from the index contributing most to the
   current maximal convolution entry to a random other index.  Accept the
   move whenever it improves the score.
5. Return the best sequence found.  If, for any reason, no improvement
   is discovered, the function falls back to the known construction
   `height_sequence_1`.

The whole routine respects the 1000‑second time budget and uses only
standard NumPy operations (no external files or network access)."""

import time
import numpy as np

# the evaluation routine is available in the execution environment
# from the statement we can call evaluate_sequence(seq)

def _conv_max(seq):
    """Convenient wrapper returning the maximum convolution entry."""
    b = np.convolve(seq, seq)
    return np.max(b)


def _exponential_sequence(n, alpha):
    """Generate a decreasing exponential sequence a_i = exp(-alpha*i)."""
    idx = np.arange(n, dtype=np.float64)
    seq = np.exp(-alpha * idx)
    return seq


def _refine_sequence(seq, deadline, best_score):
    """
    Simple hill‑climbing refinement:
    - locate the index `k_max` of the maximal convolution value,
    - compute the contribution of every a_i to that entry,
    - move a small fraction of the largest contributor to a random index.
    Accept only improving moves.  Stop when time runs out.
    """
    n = len(seq)
    seq = seq.copy()
    while time.time() < deadline:
        # compute convolution once
        b = np.convolve(seq, seq)
        k_max = int(np.argmax(b))
        max_val = b[k_max]

        # contribution of each index i to b[k_max] (ordered pairs)
        # contribution_i = a_i * a_{k_max - i}  (zero when out of range)
        contrib = np.zeros(n)
        for i in range(n):
            j = k_max - i
            if 0 <= j < n:
                contrib[i] = seq[i] * seq[j]

        # pick the index with the largest contribution
        i_src = int(np.argmax(contrib))
        if contrib[i_src] == 0:
            break    # nothing to improve

        # choose a destination index at random (different from source)
        i_dst = np.random.randint(n - 1)
        if i_dst >= i_src:
            i_dst += 1

        # move a small fraction (up to 10%) of the source value
        delta = 0.1 * seq[i_src]
        if delta <= 0:
            break
        seq[i_src] -= delta
        seq[i_dst] += delta

        # evaluate the new candidate
        cur_score = evaluate_sequence(seq.tolist())
        if cur_score < best_score:
            best_score = cur_score
            # keep the improvement
        else:
            # revert the move
            seq[i_src] += delta
            seq[i_dst] -= delta
    return seq, best_score

    # ------------------------------------------------------------------
    # Choose the length of the sequence.
    # Prefer the length of the known construction if it exists.
    # ------------------------------------------------------------------
    try:

    best_seq = None

    # The range 0.0002 … 0.002 works well for n ≈ 2000‑2500.
    # ------------------------------------------------------------------
    alphas = np.linspace(0.0002, 0.0020, 30)

    for alpha in alphas:
        if time.time() >= deadline:

        seq = _exponential_sequence(n, alpha)
        # The evaluation function is scale‑invariant, but we keep the raw

        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.tolist()
            print(f"  New best (alpha={alpha:.6f}) -> score={score:.6f}")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    if best_seq is None:

            # Finally, a trivial uniform sequence (guaranteed to give 2.0)
