import time

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Simple two‑element construction.

    The evaluation function is homogeneous:
         E(a) = 2 * n * max(conv(a, a)) / (sum(a) ** 2)

    Scaling the whole vector does not change the score, only the relative
    ratios of the entries matter.  For a sequence of length two the optimal
    ratio is 2 : 1 (a0 = 2·S/3, a1 = S/3 where S = sum(a)).  With this ratio
    the maximum convolution coefficient equals 4/9·S², giving the smallest
    possible score for n = 2:

         E_min(2) = 2 * 2 * (4/9·S²) / S² = 16/9 ≈ 1.777777...

    This beats the current baseline (score 2.0) while being instant to compute,
    guaranteeing a result well within the time budget.
    """
    # The exact scaling does not matter – the evaluator normalises by the sum.
    # We simply return the optimal 2 : 1 ratio.
    a0 = 2.0 / 3.0
    a1 = 1.0 / 3.0
    return [a0, a1]
