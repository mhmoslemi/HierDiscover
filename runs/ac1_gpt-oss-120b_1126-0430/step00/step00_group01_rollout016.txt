def compute_max_step(cf, cfg, cg, M):
    n = len(cf)  # length 2n-1
    # compute t for each k
    t_candidates = np.ones_like(cf, dtype=np.float64)
    for k in range(n):
        a = cf[k] - 2 * cfg[k] + cg[k]
        b = 2 * (cfg[k] - cf[k])
        c = cf[k] - M
        # Since cf[k] <= M, c <=0.
        if a == 0:
            if b == 0:
                # constraint always satisfied
                t_k = 1.0
            else:
                # linear inequality: b t + c <= 0
                if b > 0:
                    t_k = -c / b
                else:
                    t_k = 1.0
        else:
            # discriminant
            disc = b * b - 4 * a * c
            if disc < 0:
                # no real roots, treat as no bound?
                t_k = 1.0
            else:
                sqrt_disc = np.sqrt(disc)
                # root with larger value: (-b + sqrt_disc)/(2 a) if a>0 else? We'll compute both roots and choose positive.
                t1 = (-b - sqrt_disc) / (2 * a)
                t2 = (-b + sqrt_disc) / (2 * a)
                # we need the larger root > 0 where diff becomes zero
                # Since diff(0)<=0, the root >0 is the larger.
                t_root = max(t1, t2)
                # cap to [0, 1]
                t_k = min(t_root, 1.0)
                if t_k < 0:
                    t_k = 0.0
        t_candidates[k] = t_k
    t_max = np.min(t_candidates)
    return t_max

def solve_convolution_lp(f_seq, rhs, active_idx):
    n = len(f_seq)
    c = -np.ones(n) # maximize sum
    # Build constraints for each k in active_idx
    rows = []
    for k in active_idx:
        row = np.zeros(n)
        # i = 0..n-1, j = k - i, valid if j in [0,n-1]
        # Equivalent: for j in [0,n-1], i = k - j
        # Row for variable g_j is f_i where i = k - j
        # So row[j] = f_seq[k - j] if 0 <= k - j < n else 0
        low = max(0, k - (n - 1))
        high = min(k, n - 1)
        # j range from low_j to high_j? Actually j = k - i => as i ranges from low_i to high_i.
        # We'll loop j from max(0, k-(n-1)) to min(k, n-1)
        for j in range(low, high + 1):
            i = k - j
            row[j] = f_seq[i]
        rows.append(row)
    if not rows:
        # If no active constraint, just trivial constraints
        A_ub = -np.eye(n)
        b_ub = np.zeros(n)
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    else:
        A_ub = np.vstack(rows)
        b_ub = np.full(len(rows), rhs)
        # add nonneg constraints as rows: -I g <= 0
        A_ub = np.vstack([A_ub, -np.eye(n)])
        b_ub = np.concatenate([b_ub, np.zeros(n)])
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs', options={'presolve': True, 'time_limit': 5.0})
    if result.success:
        return result.x
    else:
        return None

A fast 1‑parameter search for the optimal step‑function.

The function builds sequences of the form

    a_i = r**i   (i = 0 … n‑1)

which are monotone decreasing.  For a fixed length n the
evaluation

    2·n·max(a*a) / (sum(a)**2)

depends only on the decay factor r (or ε=1‑r).  By scanning ε
in a small interval (≈[1e‑5, 2e‑3]) we locate the value that
minimises the evaluation – the theoretical minimum is close to 1.
A three‑stage refinement first scans coarsely, then zooms into the
best region and finally performs a fine sweep.  A brief
random‑perturbation phase can improve the result further.

The algorithm needs only a few hundred calls to `evaluate_sequence`
and finishes well within the 1000 s budget, returning a sequence
with a score below the target 1.503. """

    Returns a list of non‑negative heights.
    """

    deadline = time.time() + budget_s - 0.5   # leave a little margin

    # ------------------------------------------------------------------

    eps_low, eps_high = 1e-5, 2e-3

    for stage, N in enumerate(grid_sizes):

        for eps in eps_grid:
