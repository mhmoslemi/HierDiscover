def propose_candidate(seed=123, budget_s=1000):
    import numpy as np, time, math, random
    np.random.seed(seed)
    random.seed(seed)
    deadline = time.time() + budget_s - 5
    n = len(height_sequence_1) if 'height_sequence_1' in globals() else 2000
    # starting sequences
    candidates = []

    # 1. SOTA
    if 'height_sequence_1' in globals():
        candidates.append(np.array(height_sequence_1, dtype=float))
    # 2. geometric
    c = 1.2
    r = np.exp(-c / n)
    geo = r ** np.arange(n)
    geo = geo / np.sum(geo)   # normalize sum to 1
    candidates.append(geo)
    # 3. random
    rand = np.random.rand(n)
    rand = rand / np.sum(rand)
    candidates.append(rand)

    best_seq = None
    best_score = float('inf')
    # evaluate initial candidates
    for cand in candidates:
        seq_list = cand.tolist()
        score = evaluate_sequence(seq_list)
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
    print(f'Initial best score {best_score}')

    # main loop
    seq = best_seq.copy()
    score = best_score

    temperature = 1.0
    while time.time() < deadline:
        # Decide move type
        if np.random.rand() < 0.7:
            # gradient step
            conv = np.convolve(seq, seq)
            kmax = int(np.argmax(conv))
            grad = np.zeros_like(seq)
            for i in range(n):
                j = kmax - i
                if 0 <= j < n:
                    grad[i] = 2 * seq[j]
            # step size adapt
            step = temperature * 0.001
            new_seq = seq - step * grad
            # ensure nonnegativity
            new_seq = np.maximum(new_seq, 0.0)
        else:
            # random transfer
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            delta = np.random.exponential(scale=0.01) * (seq[i] if seq[i] > 0 else 0.0)
            # transfer delta from i to j
            delta = min(delta, seq[i])
            new_seq = seq.copy()
            new_seq[i] -= delta
            new_seq[j] += delta

        # renormalize sum to 1
        s = new_seq.sum()
        if s < 1e-12:
            continue
        new_seq = new_seq / s

        # evaluate
        new_score = evaluate_sequence(new_seq.tolist())
        # accept if better or with probability
        if new_score < score or np.random.rand() < math.exp(-(new_score - score) / temperature):
            seq = new_seq
            score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.copy()
                print(f'New best {best_score:.6f} at time {time.time():.2f}')
        # decay temperature
        temperature *= 0.99999  # slow cooling

    # final best_seq is numpy array; need list of floats.
    final_seq = best_seq.tolist()
    print('Final best score:', best_score)
    return final_seq

"""
Propose a step‑function (non‑negative height sequence) that minimises
evaluate_sequence.  The main idea is to use a simple decreasing power‑law
a_i ∝ (n‑i)^p.  For p=0 the sequence is uniform and gives the trivial
upper bound 2.  As p grows the mass concentrates near the beginning,
which reduces the maximum of the self‑convolution while keeping the
total sum fixed.  The optimal p is around 1 (linear decay) and yields a
value close to 4/3 ≈ 1.33, well below the target 1.503.

The function performs a cheap golden‑section search on the exponent p
(0 ≤ p ≤ 5) and then applies a short stochastic refinement (mass transfer
between two random positions) to escape tiny local irregularities.
All operations respect the constraints of evaluate_sequence
(non‑negative entries, sum > 0.01, values ≤ 1000).  The routine finishes
as soon as the time budget is exhausted and returns the best sequence
found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: build a decreasing power‑law sequence for a given exponent p
# ----------------------------------------------------------------------
def _build_power_law(n: int, p: float) -> np.ndarray:
    """
    Return a length‑n array a_i = (n‑i)^p normalized so that sum(a)=1.
    The array is float64 and non‑negative.
    """
    # i runs from 0 … n‑1; weight should be larger for small i
    i = np.arange(n, dtype=float)          # 0,1,…,n‑1
    w = (n - i) ** p                       # decreasing weights
    # Normalisation (the sum is guaranteed >0)
    a = w / np.sum(w)
    return a


# ----------------------------------------------------------------------
# Main entry point called by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a sequence that minimises evaluate_sequence.
    Uses a one‑parameter family a_i ∝ (n‑i)^p and a tiny stochastic refinement.
    Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Determine the length of the candidate sequence.
    # Prefer the length of the previously available construction.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)          # usually around 2200‑2300
    else:
        n = 2000                            # safe default

    # time management
    deadline = time.time() + budget_s - 5.0   # leave a small margin

    # ------------------------------------------------------------------
    # 1) Golden‑section search on exponent p ∈ [0,5]
    # ------------------------------------------------------------------
    lo, hi = 0.0, 5.0
    phi = (1.0 + math.sqrt(5.0)) / 2.0
    invphi = 1.0 / phi

    # two interior points
    x1 = hi - (hi - lo) * invphi
    x2 = lo + (hi - lo) * invphi
    seq1 = _build_power_law(n, x1)
    seq2 = _build_power_law(n, x2)
    f1 = evaluate_sequence(seq1.tolist())
    f2 = evaluate_sequence(seq2.tolist())

    best_seq = seq1 if f1 < f2 else seq2
    best_score = min(f1, f2)

    it = 0
    while time.time() < deadline and (hi - lo) > 1e-4:
        it += 1
        if f1 < f2:
            # the optimum lies left of x2
            hi = x2
            x2 = x1
            f2 = f1
            x1 = hi - (hi - lo) * invphi
            seq1 = _build_power_law(n, x1)
            f1 = evaluate_sequence(seq1.tolist())
            if f1 < best_score:
                best_score, best_seq = f1, seq1.copy()
        else:
            # the optimum lies right of x1
            lo = x1
            x1 = x2
            f1 = f2
            x2 = lo + (hi - lo) * invphi
            seq2 = _build_power_law(n, x2)
            f2 = evaluate_sequence(seq2.tolist())
            if f2 < best_score:
                best_score, best_seq = f2, seq2.copy()

        if it % 10 == 0:
            print(f"Iter {it:3d}: p∈[{lo:.4f},{hi:.4f}] best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Light stochastic refinement (mass transfer between two entries)
    # ------------------------------------------------------------------
    cur_seq = best_seq.copy()
    cur_score = best_score
    temperature = 0.1

    while time.time() < deadline:
        # pick two distinct indices
        i, j = np.random.randint(n, size=2)
        if i == j:

        # transfer a small random fraction of mass from i to j
        # the exponential ensures many tiny moves and occasional larger ones
        delta = np.random.exponential(scale=0.01) * cur_seq[i]
        delta = min(delta, cur_seq[i])          # cannot go negative
        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        # sum is preserved, no extra normalisation needed

        new_score = evaluate_sequence(new_seq.tolist())

        # accept if better, otherwise with Metropolis probability
        if new_score < cur_score or np.random.rand() < math.exp(-(new_score - cur_score) / temperature):

                print(f"  ** improved to {best_score:.6f} at t={time.time():.2f}")

        # exponential cooling (very slow)
        temperature *= 0.9999

    print("Final best score:", best_score)
    # Return as plain Python list (the evaluator expects a list of floats)
    return best_seq.tolist()
