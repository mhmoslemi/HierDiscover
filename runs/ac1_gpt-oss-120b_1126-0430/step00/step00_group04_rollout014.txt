def simplex_projection(v):
    n = len(v)
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def improve(seq, iterations, step_initial):
    best_seq = seq.copy()
    best_score = evaluate_sequence(seq)
    step = step_initial
    for t in range(iterations):
        conv = np.convolve(seq, seq)
        max_k = int(np.argmax(conv))
        # compute gradient
        grad = np.zeros_like(seq)
        for i in range(len(seq)):

def propose_candidate(seed=..., budget_s=..., **kwargs):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5

    best_seq = None
    best_score = np.inf

    for n in [500,1000,2000,5000,10000]:
        if time.time()>deadline: break
        for alpha in [0.0,0.5,1.0,1.5,2.0]:
            a = 1.0/np.sqrt(np.arange(n, dtype=float)+alpha+1.0) # ensure >=0
            # Normalize to sum=1
            a = a / np.sum(a)
            score = evaluate_sequence(a.tolist())
            print(f"n={n}, alpha={alpha}, score={score}")
            if score < best_score:
                best_score=score
                best_seq = a.tolist()
    # Optionally run few gradient descent refinements on best_seq.
    return best_seq

def simplex_projection(v):
    # return projection onto probability simplex
    v = np.asarray(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho)==0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w

"""
propose_candidate
-----------------

Search for a non‑negative step‐function that minimizes the provided
`evaluate_sequence` metric.

The metric can be written (after normalising the sum of the coefficients
to 1) as  

    2·n·max_k (a * a)[k] .

For a fixed length `n` the sum of the convolution entries is 1, hence the
average value is 1/(2·n‑1).  The goal is to make the largest entry as close
to this average as possible.  A classic construction that almost attains the
theoretical optimum is the discrete analogue of the function 1/√x :

    a_i ∝ (i+1)^{-p} ,   p ≈ ½ .

With `p = ½` the maximal convolution value tends to π/(4·n) and the whole
objective tends to π/2 ≈ 1.57, a large improvement over the uniform
distribution (value = 2).  The algorithm explores several lengths and
exponents around the optimal `p = 0.5`, keeps the best candidate and then
performs a few projected sub‑gradient steps to reduce the maximal
convolution a little further.

All numbers are kept non‑negative, the vector is projected onto the
probability simplex (sum = 1) after each update – this does not change the
objective because the evaluation function is homogeneous of degree 0.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex (sum = 1, all >= 0)
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x | x_i >= 0, sum x_i = 1}."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    # find rho = max {j | u_j - cssv_j / (j+1) > 0}
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive → return the vertex e_i with i=argmax(v)
        w = np.zeros_like(v)
        w[np.argmax(v)] = 1.0
        return w
    rho = ind[cond][-1] - 1          # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights (floats) that gives a low
    evaluation score.  The routine runs for up to ``budget_s`` seconds,
    trying a family of power‑law sequences a_i ∝ (i+1)^{-p} for several
    lengths and exponents, keeps the best, and finally performs a short
    projected sub‑gradient descent to fine‑tune the result.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # a little safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1) scan a grid of lengths and power exponents
    lengths = [500, 1000, 2000, 4000, 6000, 8000]
    exponents = np.linspace(0.30, 0.80, 11)   # includes p = 0.5

    for n in lengths:
        if time.time() >= deadline:
            break
        idx = np.arange(1, n + 1, dtype=float)   # i+1 to avoid division by zero

        for p in exponents:
            if time.time() >= deadline:
                break

            # raw power‑law, then normalise to sum = 1 (scaling does not matter)
            a = idx ** (-p)
            a /= a.sum()

            # evaluate – fast for moderate n
            score = evaluate_sequence(a.tolist())

            print(f"[grid] n={n:5d}  p={p:.3f}  score={score:.6f}")

            if score < best_score:
                best_score = score
                best_seq = a.copy()

    # ------------------------------------------------------------------
    # 2) optional refinement: a few projected sub‑gradient steps.
    #    The sub‑gradient of the maximal convolution entry (at index k*)
    #    w.r.t. a_i is 2 * a_{k* - i} (if the index is in range).
    if best_seq is not None:
        a = np.array(best_seq, copy=True)
        step = 0.07                         # initial step size
        while time.time() < deadline:
            conv = np.convolve(a, a)
            k_max = int(np.argmax(conv))

            # vectorised gradient of the active convolution entry
            i = np.arange(a.size)
            j = k_max - i
            mask = (j >= 0) & (j < a.size)
            grad = np.zeros_like(a)
            grad[mask] = 2.0 * a[j[mask]]

            # gradient step + projection onto the simplex (sum = 1)
            a_new = _project_simplex(a - step * grad)

            new_score = evaluate_sequence(a_new.tolist())

            # keep improvement
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_seq = a_new.tolist()
                a = a_new
                step *= 0.93                     # slowly shrink step after success
            else:
                step *= 0.95                     # otherwise reduce the step
                if step < 1e-7:                  # step too small → stop early
                    break

    # ------------------------------------------------------------------
    # safety fallback – if nothing was produced, fall back to the uniform vector
    if best_seq is None:
        n = 1000
        best_seq = [1.0 / n] * n

    print(f"[final] best score = {best_score:.6f} with length {len(best_seq)}")
    return best_seq
