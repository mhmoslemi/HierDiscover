def solve_convolution_lp(a_seq, M):

    n = len(a_seq)
    c = -np.ones(n)  # maximize sum

    # build A_ub and b_ub
    rows = []
    b_ub = np.full(2*n - 1, M, dtype=np.float64)
    for k in range(2*n - 1):
        row = np.zeros(n, dtype=np.float64)
        # j range
        start = max(0, k-(n-1))
        end = min(n-1, k)
        # j values
        j = np.arange(start, end+1, dtype=int)
        row[j] = a_seq[k - j]  # a_{k-j}
        rows.append(row)
    A_ub = np.vstack(rows)  # shape (2n-1, n)
    # bounds for variables: non-negative
    bounds = [(0, None)] * n

    # Use high-performance method
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True, 'time_limit': 2.0})
    if res.success:
        return res.x
    else:
        return None

def project_simplex(v):
    """Projects a vector onto the probability simplex (sum = 1, non-negative)."""
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # Find rho
    rho = np.where(u - cssv / ind > 0)[0]
    if len(rho) == 0:
        # All zero or negative, return uniform?
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a candidate sequence of non‑negative heights that improves the
evaluation score 2·n·max(conv(a,a))/sum(a)^2.

The optimizer combines three ideas:

1. **Simplex projection** – keep the vector on the probability simplex
   (non‑negative, sum = 1) so the objective is scale‑invariant.
2. **LP direction** – for the current sequence `a` we solve a linear program

        maximise   1ᵀ b
        subject to conv(a,b) ≤ M   (M = max(conv(a,a)))
                  b ≥ 0

   The LP finds a vector `b` with a larger total mass while respecting the
   same worst‑case convolution value `M`.  After scaling `b` to the same sum
   as `a` we obtain a promising direction `g`.
3. **Line‑search mixing** – we replace `a` by a convex combination
   `(1‑α)·a + α·g` and keep the best α (tested from large to small
   values).  If the LP fails we perform a small random mass transfer as a
   fallback.

The process repeats until the time budget is exhausted.  The algorithm
works well for moderate lengths (≈ 400–800) where the LP is cheap, yet the
resulting score does not depend on the absolute length, only on the
relative shape of the distribution.  Random restarts are performed
occasionally to escape possible stagnation.

The final sequence is returned as a plain list; the supplied
`evaluate_sequence` will clamp and check it automatically.
"""

import time
import numpy as np
from scipy.optimize import linprog


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (sum=1, non‑negative)."""
    # Implementation of the algorithm from Duchi et al. (2008)
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution(a: np.ndarray) -> float:
    """Return max_{k} (a * a)[k] using an FFT (O(n log n))."""
    n = a.shape[0]
    # length for FFT – next power of two ≥ 2n‑1
    L = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    return float(conv.max())


def _score(a: np.ndarray) -> float:
    """Compute the evaluation score (2·n·max(conv)/sum²)."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return 2.0 * a.shape[0] * _max_convolution(a) / (s * s)


def _solve_lp_direction(a: np.ndarray, M: float, time_limit: float = 1.0):
    """
    Solve  maximise   sum(b)  subject to   conv(a,b) ≤ M ,  b ≥ 0 .
    Returns the optimal b (np.ndarray) or None on failure.
    """
    n = a.shape[0]
    # Objective: minimise -sum(b)  →  maximise sum(b)
    c = -np.ones(n)

    # Build the convolution constraints conv(a,b) ≤ M
    # There are (2n‑1) constraints, one for each output index.
    rows = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        # indices j that appear in the k‑th convolution sum
        j_low = max(0, k - (n - 1))
        j_high = min(n - 1, k)
        j = np.arange(j_low, j_high + 1, dtype=int)
        rows[k, j] = a[k - j]   # coefficient a_{k‑j} for variable b_j

    A_ub = rows
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # Bounds: b_j ≥ 0
    bounds = [(0.0, None)] * n

    # Use HiGHS interior‑point method – fast for this size.
    try:
        res = linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"time_limit": time_limit, "presolve": True},
        )
    except Exception:  # pragma: no cover
        return None

    if res.success:
        return np.maximum(res.x, 0.0)  # guard against tiny negatives
    return None


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) achieve a score
    below the current best upper bound of 2.0.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds; the function stops before this deadline.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # small safety margin

    # ------------------------------------------------------------------
    # 1. Choose a working length.  Length does not affect the final score
    #    much; a moderate size keeps the LP cheap.
    # ------------------------------------------------------------------
    n = 600                     # you may tune this (≈ 400–800 works well)
    # initialise with a uniform probability vector (sum = 1)
    a = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = a.copy()
    best_val = _score(best_seq)

    # occasional random restarts to avoid being stuck in a local optimum
    iterations_without_improve = 0
    restart_interval = 200      # after this many iterations we try a fresh start

    # α values evaluated in descending order (large steps first)
    alphas = [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]

    it = 0
    while time.time() < deadline:
        it += 1

        # ------------------------------------------------------------------
        # 2. Compute the current worst‐case convolution value M.
        # ------------------------------------------------------------------
        M = _max_convolution(a)

        # ------------------------------------------------------------------
        # 3. Solve the LP to obtain a direction with larger mass.
        # ------------------------------------------------------------------
        g_raw = _solve_lp_direction(a, M, time_limit=0.8)
        improved = False

        if g_raw is not None:
            # Scale g so that it has the same total sum as a (i.e. 1)
            sum_a = a.sum()
            sum_g = g_raw.sum()
            if sum_g > 0:
                scale = sum_a / sum_g
                g = g_raw * scale
            else:   # degenerate case – fall back to random perturbation
                g = np.copy(a)

            # Try several mixing factors and keep the best improvement.
            for alpha in alphas:
                cand = (1.0 - alpha) * a + alpha * g
                # Projection keeps any tiny numerical drift off the simplex.
                cand = _project_onto_simplex(cand)
                val = _score(cand)
                if val < best_val:
                    best_val = val
                    best_seq = cand.copy()
                    a = cand.copy()
                    improved = True
                    iterations_without_improve = 0
                    # Small break – we already found an improvement for this iter.
                    break
        else:
            # ------------------------------------------------------------------
            # 4. LP failed (rare); perform a tiny random mass‑transfer.
            # ------------------------------------------------------------------
            i, j = np.random.choice(n, size=2, replace=False)
            delta = min(a[i], 0.02 * a[i] + 1e-4)   # small fraction of source
            a[i] -= delta
            a[j] += delta
            a = _project_onto_simplex(a)
            val = _score(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                improved = True
                iterations_without_improve = 0

        if not improved:
            iterations_without_improve += 1

        # ------------------------------------------------------------------
        # 5. Random restart if we have been stagnating for a while.
        # ------------------------------------------------------------------
        if iterations_without_improve >= restart_interval:
            # fresh Dirichlet draw (sum = 1) – this gives a completely new shape.
            a = np.random.exponential(scale=1.0, size=n)
            a = _project_onto_simplex(a)
            # evaluate the new start – keep it only if it improves the global best
            val = _score(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                iterations_without_improve = 0
            else:
                # otherwise revert to the best known sequence for the next round
                a = best_seq.copy()
                iterations_without_improve = 0

        # Optional debug print every few seconds
        if it % 20 == 0:
            elapsed = time.time() + budget_s - deadline - time.time()
            print(
                f"Iter {it:5d} | best score {best_val:.6f} | "
                f"time left {deadline - time.time():.1f}s"
            )

    # Ensure the final candidate respects the evaluator's clamping rules.
    final = np.clip(best_seq, 0.0, 1000.0).tolist()
    print(f"Finished after {it} iterations – final best score ≈ {best_val:.6f}")
    return final
