def compute_score(a):
    n = len(a)
    # ensure nonnegative, sum
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return np.inf
    # compute linear convolution using FFT (real)
    L = 1 << ((2*n-1).bit_length())
    # compute convolution
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa*fa, n=L)[:2*n-1]
    max_b = np.max(conv)
    score = 2.0 * n * max_b / (sum_a**2)
    return float(score)
