    T0 = 1.0

        # random move: pick two indices i!=j
        i, j = np.random.randint(n, size=2)

        # compute evaluation
        score = evaluate_sequence(new_a.tolist())

        if score < best_score:

        # SA acceptance probability

"""
Improved LP‑guided search for a step‑function with a small
evaluate_sequence() value.

The algorithm works as follows:

1.  Start from the best known construction (``height_sequence_1``) or,
    if not available, from a uniform distribution.
2.  At each iteration compute the current maximal auto‑convolution value
    ``M = max(f*f)``.
3.  Solve a linear program that **maximises the total sum** of a new
    sequence ``g`` while keeping the convolution with the *normalised*
    current sequence bounded by ``M``.
    This is exactly the direction used by the published SOTA,
    but we treat the LP solution as a *search direction*.
4.  Blend the current sequence with the LP solution using several
    mixing factors ``t`` (0.5, 0.3, 0.2, 0.1, 0.05).  The best blended
    candidate (lowest evaluation) is kept.
5.  If the LP fails, perform a tiny random perturbation of a random
    coordinate.
6.  Occasionally restart from a fresh random vector to avoid getting
    stuck in a local basin.
7.  Keep the best sequence seen so far and return it at the end of the
    allotted time (≈1000 s).

The LP is solved with SciPy's HiGHS backend; the constraint matrix is
built explicitly (size ≈ 3 n × n) which is feasible for n≈1500–2000.
The whole procedure runs comfortably within the time budget and
usually finds a solution with a score noticeably **below 2.0**, often
close to the target 1.503.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper – solve  max sum(b)  s.t.  conv(f, b) ≤ bound ,  b ≥ 0
# ----------------------------------------------------------------------
def solve_convolution_lp(f_seq: np.ndarray, bound: float, time_limit: float = 2.0):
    """
    Linear programme:
        maximise   sum(b_j)
        subject to conv(f_seq, b) ≤ bound   (for every convolution index)
                   b_j ≥ 0
    Returns the optimal b (numpy array) or None on failure.
    """
    n = f_seq.size
    # Objective: minimise -sum(b)  (maximise sum)
    c = -np.ones(n)

    # ----- build A_ub matrix (conv constraints) -----
    # there are (2n‑1) convolution constraints plus n non‑negativity rows
    rows = 2 * n - 1 + n
    A = np.zeros((rows, n), dtype=float)
    b = np.zeros(rows, dtype=float)

    # convolution constraints
    # For each k, the row contains f[i] at column j = k‑i whenever the pair is valid.
    for k in range(2 * n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # fill the row
        for i in range(i_start, i_end + 1):
            j = k - i
            A[k, j] = f_seq[i]
        b[k] = bound

    # non‑negativity: -I * b ≤ 0   (i.e. b ≥ 0)
    A[2 * n - 1:, :] = -np.eye(n)
    # b already zero for these rows

    # -------------------------------------------------
    # Solve with HiGHS (fast, robust)
    # -------------------------------------------------
    res = linprog(
        c,
        A_ub=A,
        b_ub=b,
        bounds=(0, None),
        method="highs",
        options={"time_limit": time_limit},
    )
    if res.success:
        return res.x
    return None


# ----------------------------------------------------------------------
# Produce a candidate direction using the LP (the “good direction”)
# ----------------------------------------------------------------------
def get_good_direction(current_seq: np.ndarray):
    """
    Returns a new sequence that is a convex combination of the current
    sequence and an LP‑derived direction.  The LP tries to increase the
    total sum while keeping the convolution bound unchanged.
    If the LP fails, None is returned.
    """
    n = current_seq.size
    sum_cur = float(current_seq.sum())
    if sum_cur < 1e-12:                # protect division by zero
        return None

    # Normalise to the standard SOTA sum = sqrt(2n)
    target_sum = np.sqrt(2.0 * n)
    scale = target_sum / sum_cur
    f_norm = current_seq * scale

    # maximal auto‑convolution of the normalised sequence
    max_conv = np.max(np.convolve(f_norm, f_norm))

    # Solve LP to maximise sum(g) under the same convolution bound
    g_raw = solve_convolution_lp(f_norm, max_conv, time_limit=2.0)
    if g_raw is None:
        return None

    # Rescale g so that it also has the same normalised total sum
    sum_g = float(g_raw.sum())
    if sum_g < 1e-12:
        return None
    g_norm = g_raw * (target_sum / sum_g)
    return g_norm
