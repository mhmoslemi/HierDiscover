def propose_candidate(seed=42, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5
    best_seq = None
    best_score = np.inf

    # scans several lengths
    for n in [50, 100, 200, 400, 800]:
        # create random start
        a = np.random.rand(n)
        a = a / a.sum()
        a = a.astype(float).tolist()

        # local hill climbing
        # compute initial score
        score = evaluate_sequence(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()

        step = 0.1  # fraction of sum moved
        # time loop
        while time.time() < deadline:
            # pick random i,j
            i, j = np.random.choice(n, 2, replace=False)
            # propose move delta = step * a[i] * random sign
            # to ensure positivity, propose new_i = max(0, a[i] + delta)
            # resulting change in sum: we also adjust a[j] to keep sum constant? Actually we will ensure sum unchanged by adjusting both i and j accordingly.
            # propose delta = (np.random.rand() - 0.5) * step
            delta = (np.random.rand() - 0.5) * step * a[i]  # relative scale
            # adjust i down, j up
            new_i = a[i] - delta
            new_j = a[j] + delta
            if new_i < 0:
                # clip delta to a[i]
                delta = a[i]
                new_i = 0
                new_j = a[j] + delta
            # clip max large? not needed
            a_new = a.copy()
            a_new[i] = new_i
            a_new[j] = new_j

            # compute score
            new_score = evaluate_sequence(a_new)
            if new_score < score:
                a = a_new
                score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_seq = a_new.copy()
                    print(f"New best {best_score:.6f} n={n} time={time.time():.2f}")
                # optionally increase step
                step *= 1.01
            else:
                # maybe accept with probability decreasing with temperature
                step *= 0.99  # reduce step gradually

            # may occasionally random restart
            if np.random.rand() < 0.001:
                a = np.random.rand(n)
                a = a / a.sum()
                score = evaluate_sequence(a)

    return best_seq

        a = a / a.sum()

        alpha = 0.5

        while time.time() < deadline:

            delta = np.random.rand() * alpha * a[i]

            # annealing schedule
            temperature *= 0.9999
            alpha *= 0.9999
            # restart early if stuck
            if np.random.rand() < 0.001:

def simplex_projection(v):
    """Project v onto simplex {x>=0, sum(x)=1}."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # find rho where u_i - cssv_i / i > 0
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def gradient_descent(a, max_iter, deadline):
    n = len(a)
    best_a = a.copy()
    best_score = fast_score(a)
    step = 0.5  # initial step
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        conv = compute_conv(a)
        max_idx = np.argmax(conv)
        grad = np.zeros_like(a)
        # compute gradient for max_idx
        j = max_idx - np.arange(n)
        mask = (j >= 0) & (j < n)
        grad[mask] = 2 * a[j[mask]]
        # step
        new_a = a - step * grad
        # projection
        new_a = np.maximum(new_a, 0)
        s = new_a.sum()
        if s == 0:
            new_a = np.random.rand(n)
            new_a = new_a / new_a.sum()
        else:
            new_a = new_a / s
        # evaluate
        new_score = fast_score(new_a)
        if new_score < best_score:
            best_score = new_score
            best_a = new_a.copy()
        # acceptance
        if new_score < fast_score(a):
            a = new_a
            # perhaps increase step
            step = min(step * 1.02, 5.0)
        else:
            # revert, reduce step
            step = step * 0.9
        # occasional random perturb to escape
        if np.random.rand() < 0.01:
            # add gaussian noise and renormalize
            noise = np.random.randn(n) * step * 0.1
            a = a + noise
            a = np.maximum(a, 0)
            sum_a = a.sum()
            if sum_a == 0:
                a = np.random.rand(n)
                a = a / a.sum()
            else:
                a = a / sum_a
    return best_a, best_score

def hill_climb(a, max_steps, deadline):
    n = len(a)
    curr_score = fast_score(a)
    best_local_a = a.copy()
    best_local_score = curr_score
    temperature = 0.1  # adjustable
    for step in range(max_steps):
        if time.time() >= deadline: break
        i, j = np.random.choice(n, 2, replace=False)
        # propose delta: fraction of a[i]
        fraction = np.random.rand() * 0.5  # up to 0.5 of a[i]
        delta = fraction * a[i]
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        # ensure negativity not happen
        if new_a[i] < 0:
            continue
        new_score = fast_score(new_a)
        # accept if better or with some probability
        if new_score < curr_score or np.random.rand() < np.exp((curr_score - new_score) / max(temperature, 1e-12)):
            a = new_a
            curr_score = new_score
            if new_score < best_local_score:
                best_local_score = new_score
                best_local_a = new_a.copy()
        # temperature schedule
        temperature *= 0.9995
    return best_local_a, best_local_score

"""
Simple stochastic search for a step‑function with a low value of
evaluate_sequence.  The algorithm repeatedly:

* draws a random length n (20‑300) and a random positive vector a of
  that length (sum(a)=1);
* performs a cheap hill‑climbing / simulated‑annealing loop that moves
  a small amount of mass from one coordinate to another;
* keeps the best sequence seen over the whole run.

Fast evaluation of the objective (2·n·max(conv)/sum²) is done with an
FFT‑based autocorrelation, which is much faster than numpy.convolve for
many iterations.  The search stops before the allocated budget expires
and returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _fast_score(a):
    """Return the evaluation value for a sequence a (sum(a) is assumed > 0)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    # length for FFT – next power of two >= 2*n-1
    L = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    maxb = conv.max()
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return 2.0 * n * maxb / (s * s)


def _conv_and_maxidx(a):
    """Compute the linear convolution a*a and return (conv, argmax)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    idx = int(conv.argmax())
    return conv, idx


def _project_to_simplex(v):
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i >= 0, sum x_i = 1}.  Implements the algorithm of
    Duchi et al. (2008).
    """
    v = np.asarray(v, dtype=np.float64)
    if v.sum() == 1.0 and np.all(v >= 0):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # should not happen for non‑empty vectors, fall back to uniform
        return np.full_like(v, 1.0 / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # numerical safety – renormalise to exactly 1
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(v, 1.0 / len(v))
    else:
        w /= w_sum
    return w


# ----------------------------------------------------------------------
def propose_candidate(seed=42, budget_s=1000):
    """
    Search for a non‑negative step‑function (list of heights) that yields a
    low value of evaluate_sequence.  The routine runs for roughly
    ``budget_s`` seconds and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # leave a small safety margin

    # ------------------------------------------------------------------
    # start from the provided SOTA construction if it exists
    try:
        best_seq = list(height_sequence_1)   # pylint: disable=undefined-variable
        best_score = _fast_score(best_seq)
        print(f"Starting from height_sequence_1 – score {best_score:.6f}")
    except Exception:                         # pragma: no cover
        # very simple fallback
        best_seq = [1.0]
        best_score = _fast_score(best_seq)

    # ------------------------------------------------------------------
    # parameters for the stochastic search
    lengths = list(range(20, 301, 2))           # even lengths from 20 to 300
    attempts_per_length = 500                  # max hill‑climb steps per trial

    while time.time() < deadline:
        # choose a length and a fresh random start
        n = int(np.random.choice(lengths))
        a = np.random.rand(n)
        a /= a.sum()

        cur_score = _fast_score(a)

        # quick acceptance if this random start is already better
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.tolist()
            print(f"New global best (random start) {best_score:.6f}  n={n}")

        # ----------------------------------------------------------------
        # hill‑climbing / simulated‑annealing loop
        temperature = 0.2
        for _ in range(attempts_per_length):
            if time.time() >= deadline:
                break

            # propose moving a fraction of mass from i to j
            i, j = np.random.choice(n, 2, replace=False)
            if a[i] == 0.0:
                continue
            frac = np.random.rand() * 0.5          # up to 50 % of a[i]
            delta = a[i] * frac
            new_a = a.copy()
            new_a[i] -= delta
            new_a[j] += delta

            new_score = _fast_score(new_a)

            # deterministic improvement
            if new_score < cur_score:
                a, cur_score = new_a, new_score
                if new_score < best_score:
                    best_score, best_seq = new_score, new_a.tolist()
                    print(f"New global best {best_score:.6f}  n={n}")
                continue

            # occasional uphill move (simulated annealing)
            prob = math.exp((cur_score - new_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                a, cur_score = new_a, new_score

            # cooling schedule
            temperature *= 0.9995

            # early exit if we beat the target
            if best_score < 1.503:
                break

        # ----------------------------------------------------------------
        # a short gradient‑descent refinement on the current best of this trial
        # (only if we still have time)
        if time.time() < deadline and cur_score < best_score:
            # one tiny gradient descent run
            step = 0.5
            for _ in range(200):
                if time.time() >= deadline:
                    break
                conv, kmax = _conv_and_maxidx(a)
                # gradient of the max‑convolution entry w.r.t. a
                grad = np.zeros_like(a)
                idx = np.arange(n)
                j = kmax - idx
                mask = (j >= 0) & (j < n)
                grad[mask] = 2.0 * a[j[mask]]
                new_a = a - step * grad
                new_a = _project_to_simplex(new_a)
                new_score = _fast_score(new_a)
                if new_score < cur_score:
                    a, cur_score = new_a, new_score
                    if new_score < best_score:
                        best_score, best_seq = new_score, new_a.tolist()
                        print(f"New global best (refine) {best_score:.6f}  n={n}")
                    step = min(step * 1.02, 5.0)
                else:
                    step *= 0.9
                if best_score < 1.503:
                    break

    print(f"Search finished – best score {best_score:.6f}  length={len(best_seq)}")
    return best_seq
