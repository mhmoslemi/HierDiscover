"""
Propose a new height sequence using a smooth‑max gradient descent on the
autocorrelation (self‑convolution).  The algorithm works on a probability
simplex (sum of coefficients = 1, all non‑negative), because the evaluation
score is scale‑invariant.  At each iteration we

  1. compute the convolution a*a (FFT for speed),
  2. build a soft‑max weighting of the diagonal values,
  3. obtain a gradient of the weighted sum w.r.t. a,
  4. take a projected gradient step and renormalise onto the simplex.

The soft‑max temperature β is increased slowly so the optimisation first moves
towards a flat autocorrelation and then focuses on the true maximum.
Occasional random perturbations avoid stagnation.  The best sequence found
within the allotted time is returned.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Helper utilities
# --------------------------------------------------------------
def _fast_score(a: np.ndarray) -> float:
    """Return the evaluation score for a sequence that already sums to 1.

    Because the score is scale‑invariant we can use the simplified form
        score = 2 * n * max(conv(a, a))
    where n = len(a).  All calculations are done with FFT for O(n log n) time.
    """
    n = a.shape[0]
    # length for an efficient FFT (next power of two ≥ 2n‑1)
    nfft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b)


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (non‑negative, sum = 1)."""
    if v.ndim != 1:
        raise ValueError("Simplex projection only defined for 1‑D arrays.")
    # implementation of the algorithm from
    #   Wang & Carreira‑Perpiñán, “Projection onto the probability simplex:
    #   An efficient algorithm with a simple proof, and an application”.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – fallback to uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond].max()
    theta = cssv[cond][rho - 1] / rho
    w = np.maximum(v - theta, 0.0)
    return w
