    import numpy as np, time
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5
    # determine n
    n = len(height_sequence_1) if 'height_sequence_1' in globals() else 2000
    target_sum = np.sqrt(2.0 * n)
    # initial uniform sequence
    a = np.full(n, target_sum / n, dtype=float)
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f'Init score {best_score}')
    # maybe try a few random seeds
    for _ in range(3):
        rand = np.random.dirichlet(np.ones(n))
        a_rand = rand * target_sum
        score = evaluate_sequence(a_rand.tolist())
        if score < best_score:
            best_score = score
            best_seq = a_rand.copy()
            print(f'Random init improved: {best_score}')
    # set current sequence to best
    a = best_seq.copy()
    # main loop
    eta = 0.1  # step size for update
    while time.time() < deadline:
        conv = np.convolve(a, a)
        max_b = conv.max()
        # compute indices of near-maximum
        # tolerance relative: consider values within 1e-12 of max
        tol = max_b * 1e-12 + 1e-14
        max_idx = np.where(conv >= max_b - tol)[0]
        # compute contributions
        contributions = np.zeros(n)
        for k in max_idx:
            start = max(0, k - (n - 1))
            end = min(n - 1, k)
            idx = np.arange(start, end + 1)
            contributions[k - idx] += a[idx]
        avg_contrib = contributions.mean()
        direction = avg_contrib - contributions  # move from high contrib to low
        # apply update
        a_new = a + eta * direction
        # enforce nonnegative
        a_new = np.maximum(a_new, 0.0)
        # renormalize to target sum
        s = a_new.sum()
        if s == 0:
            # fallback random restart
            a_new = np.full(n, target_sum / n)
            s = target_sum
        a_new = a_new * (target_sum / s)
        # evaluate
        score = evaluate_sequence(a_new.tolist())
        if score < best_score:
            best_score = score
            best_seq = a_new.copy()
            print(f'Iter improvement: {best_score:.6f}')
        # accept move if improvement, else maybe accept with probability
        # simple acceptance: keep new if score better else keep old
        if score <= evaluate_sequence(a.tolist()):
            a = a_new
        else:
            # possibly revert and take small random perturbation
            if np.random.rand() < 0.01:
                # random jitter
                idx = np.random.randint(n)
                delta = (np.random.rand() - 0.5) * target_sum * 0.01
                a[idx] = max(0.0, a[idx] + delta)
                a = a * (target_sum / a.sum())
        # occasionally shrink eta
        if np.random.rand() < 0.001:
            eta *= 0.5
            eta = max(eta, 1e-4)
    return best_seq.tolist()

"""
Gradient‑based heuristic for lowering the evaluation score of a step‑function.
The algorithm works with a fixed length (taken from the supplied
`height_sequence_1` if available) and tries to spread the mass of the
sequence so that the self‑convolution becomes as flat as possible.
It repeatedly moves a small amount of weight from indices that
contribute most to the largest convolution entries towards indices that
contribute least, while keeping the total sum fixed (≈ √(2 n)).

The main steps are:
* Initialise a random Dirichlet vector (sum = √(2 n)).
* In each iteration compute the convolution a*a and locate the
  largest few coefficients.
* For those coefficients compute how much each a[i] participates
  (a simple “contribution” count).
* Move mass from high‑contribution indices to low‑contribution indices.
* Clip to non‑negative values and renormalise to the target sum.
* Keep the best sequence found during the allotted time budget.

The routine returns a Python list of floats that can be fed directly to
`evaluate_sequence`.  It respects the 1000 s time limit and prints
intermediate improvements.
"""

import time
import numpy as np

# The evaluation function is assumed to be available in the global scope.
# It can be called directly as `evaluate_sequence(seq)`.

def _random_sequence(n: int, target_sum: float) -> np.ndarray:
    """Generate a random Dirichlet vector and scale it to `target_sum`."""
    # Concentration <1 gives a sparse start, >1 a more uniform start.
    alpha = np.full(n, 0.5)
    vec = np.random.dirichlet(alpha)
    return vec * target_sum


def _top_convolution_indices(conv: np.ndarray, fraction: float = 0.005) -> np.ndarray:
    """Return indices of the largest `fraction` of convolution entries."""
    k_len = conv.size
    k_top = max(1, int(k_len * fraction))
    # `np.argpartition` is O(k_len); we fetch the k_top largest indices.
    return np.argpartition(conv, -k_top)[-k_top:]


def _contributions_from_top(a: np.ndarray, top_idx: np.ndarray) -> np.ndarray:
    """
    For each position i compute the sum of a[j] over all top convolution
    positions k where i + j = k (i.e. a contributes to those large k).
    """
    n = a.size
    contrib = np.zeros(n, dtype=float)
    for k in top_idx:
        start = max(0, k - (n - 1))
        end = min(n - 1, k)
        # indices j where a[j] participates in convolution entry k
        j = np.arange(start, end + 1, dtype=int)
        # i = k - j
        contrib[k - j] += a[j]
    return contrib


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises the
    `evaluate_sequence` score.  The function runs for at most `budget_s`
    seconds and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5  # a small safety margin

    # Length of the problem – use the previously discovered sequence if present.
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
    else:
        n = 2000                     # reasonable default if nothing is provided

    target_sum = np.sqrt(2.0 * n)    # the normalisation used by the reference method

    # Helper to evaluate a numpy array (list conversion is cheap compared to the convolution)
    def _evaluate(arr: np.ndarray) -> float:
        return evaluate_sequence(arr.tolist())

    best_seq = None
    best_score = float('inf')

    # ------------------------------------------------------------------
    # several random initialisations – keep the best one as a starting point
    # ------------------------------------------------------------------
    for init_id in range(5):
        seq = _random_sequence(n, target_sum)
        sc = _evaluate(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[init {init_id}] score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # main optimisation loop
    # ------------------------------------------------------------------
    step_fraction = 0.02          # initial step size as a fraction of total sum
    min_step_fraction = 1e-4

    while time.time() < deadline:
        # start a new trial from the current best (with a tiny jitter occasionally)
        cur = best_seq.copy()
        if np.random.rand() < 0.1:               # occasional random shake
            idx = np.random.choice(n, size=3, replace=False)
            delta = (np.random.rand(3) - 0.5) * target_sum * 0.01
            cur[idx] = np.maximum(0.0, cur[idx] + delta)
            cur *= target_sum / cur.sum()

        cur_score = _evaluate(cur)

        # inner gradient descent – a modest number of steps before a restart
        for inner in range(200):
            # convolution and the set of large coefficients
            conv = np.convolve(cur, cur)
            max_b = conv.max()
            top_idx = _top_convolution_indices(conv, fraction=0.005)

            # contribution of each position to those large convolution entries
            contrib = _contributions_from_top(cur, top_idx)

            # direction: move mass from high‑contribution indices to low‑contribution ones
            avg_c = contrib.mean()
            direction = avg_c - contrib   # positive where we want to add mass

            max_abs = np.max(np.abs(direction))
            if max_abs == 0.0:           # degenerate case – apply a tiny random move
                i1, i2 = np.random.randint(n, size=2)
                eps = target_sum * 0.001
                cur[i1] = max(0.0, cur[i1] - eps)
                cur[i2] += eps
                cur *= target_sum / cur.sum()
                continue

            # step size: keep the absolute change limited (avoid overshooting)
            step_size = target_sum * step_fraction
            delta = step_size * direction / max_abs
            nxt = cur + delta

            # enforce non‑negativity and re‑normalise to the fixed sum
            nxt = np.maximum(nxt, 0.0)
            s = nxt.sum()
            if s <= 0.0:                      # numerical safeguard
                nxt = np.full(n, target_sum / n)
                s = target_sum
            nxt *= target_sum / s

            nxt_score = _evaluate(nxt)

            # keep the globally best sequence
            if nxt_score < best_score:
                best_score = nxt_score
                best_seq = nxt.copy()
                print(f"[{time.time():.1f}s] new best = {best_score:.6f}")

            # accept the move if it improves the current score
            if nxt_score <= cur_score:
                cur = nxt
                cur_score = nxt_score
            else:
                # occasional uphill move (simulated‑annealing style)
                if np.random.rand() < 0.01:
                    cur = nxt
                    cur_score = nxt_score

            # decay the step size slowly; if we are not improving, shrink it
            if inner % 20 == 0 and inner > 0:
                step_fraction = max(min_step_fraction, step_fraction * 0.99)

            # early exit if we already reached a very good bound
            if best_score < 1.55:
                break

        # stop completely once we have a satisfactory bound
        if best_score < 1.51:
            break

    # return the best sequence as a plain Python list
    return best_seq.tolist()
