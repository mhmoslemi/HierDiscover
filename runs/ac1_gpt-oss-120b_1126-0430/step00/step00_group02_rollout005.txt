def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Generate a monotone increasing linear step function.

    The sequence length is taken from the reference variable ``height_sequence_1``

    by ``evaluate_sequence``.  For a linear ramp the convolution maximum is

    """

    # ``height_sequence_1`` is provided globally by the competition harness.
    # ----------------------------------------------------------------------
