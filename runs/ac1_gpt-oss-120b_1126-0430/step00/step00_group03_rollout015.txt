"""
Propose a high‑quality step‑function sequence.

The algorithm works in two complementary phases:

1. **Frank–Wolfe style improvement**  
   The current sequence `f` is normalized so that sum(f) = sqrt(2·n)
   (the evaluation then equals the max of the self‑convolution).
   We solve a linear program that finds a direction `g` with the same
   sum but respecting the bound `conv(f,g) ≤ M` where `M` is the present
   maximum convolution value.  A short line‑search chooses the best
   convex combination `f ← (1‑t)·f + t·g`.  This step reduces the bound
   while keeping the normalization.

2. **Local stochastic refinement**  
   Occasionally a random mass transfer between two coordinates is tried.
   If the move lowers the evaluation it is kept; otherwise it is rejected.
   This helps to escape possible plateaus.

The loop runs until the time budget expires (or the target bound
~1.503 is reached).  The best sequence found is returned.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper: linear program max sum(b) subject to conv(f,b) <= rhs,
#         b_i ≥ 0.  Returns a feasible vector `b` or None.
def solve_convolution_lp(f_seq: np.ndarray, rhs: float) -> np.ndarray | None:
    """Linear program to maximise sum(b) under convolution constraints.

    Parameters
    ----------
    f_seq : np.ndarray, shape (n,)
        Fixed non‑negative sequence (already normalised).
    rhs : float
        Upper bound for each convolution entry conv(f_seq, b).

    Returns
    -------
    np.ndarray or None
        The optimal `b` (length n) if the LP succeeds, otherwise None.
    """
    n = f_seq.size
    # maximise sum(b)  → minimise -sum(b)
    c = -np.ones(n)

    # Build the inequality matrix A_ub * b ≤ b_ub.
    rows = []
    ub = []

    # Convolution constraints: for each shift k we have Σ_i f_i·b_{k‑i} ≤ rhs.
    for k in range(2 * n - 1):
        row = np.zeros(n)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # i runs from i_start … i_end, j = k‑i
        for i in range(i_start, i_end + 1):
            j = k - i
            row[j] = f_seq[i]
        rows.append(row)
        ub.append(rhs)

    # Non‑negativity constraints: -b_i ≤ 0
    rows.append(-np.eye(n))
    ub.append(np.zeros(n))

    A_ub = np.vstack(rows)
    b_ub = np.hstack(ub)

    # Solve with HiGHS (fast and reliable).  A short time limit avoids hangs.
    res = linprog(
        c,
        A_ub=A_ub,
        b_ub=b_ub,
        method="highs",
        options={"presolve": True, "time_limit": 5.0},
    )
    if res.success:
        return res.x
    return None

                    continue  # skip the random move this round

        # --------------------------------------------------------------
        # 2) Small random mass transfer (always executed if LP step didn't improve)
        i, j = np.random.choice(n, 2, replace=False)
        if seq[j] > 0.0:
            # transfer a modest fraction of the donor mass
            delta = np.random.rand() * min(seq[j], 0.05 * seq[j])
            seq[i] += delta
            seq[j] -= delta

        # keep the sum exact (numerical drift may appear)
        cur_sum = seq.sum()
        if abs(cur_sum - target_sum) > 1e-9:
            seq = seq * (target_sum / cur_sum)

        # evaluate
        cur_score = evaluate_sequence(seq.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = seq.copy()
            no_improve = 0
        else:
            no_improve = iteration - 0  # not used further, placeholder

        # occasional logging
        if iteration % 30 == 0:
            remaining = deadline - time.time()
            print(f"[{iteration}] score={best_score:.6f}, time left {remaining:.1f}s")

        # early exit when target is reached
        if best_score <= 1.503:
            break

    # Final clean‑up: enforce non‑negativity and exact normalisation
    best_seq = np.maximum(best_seq, 0.0)
    best_seq = best_seq * (target_sum / best_seq.sum())
    return best_seq.tolist()
