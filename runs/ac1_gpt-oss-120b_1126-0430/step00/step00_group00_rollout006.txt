def solve_convolution_lp(f, M):
    n = len(f)
    # Build A_ub and b_ub
    # Create list of rows
    rows = []
    b = []
    # For each k
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # i indices where both i and j = k - i within range
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # Fill row for j = k - i: coefficient f[i]
        # Equivalent: for each i in [i_min,i_max], set row[k-i] = f[i]
        for i in range(i_min, i_max + 1):
            j = k - i
            row[j] = f[i]  # note that we assign directly; later overwritten maybe but each j appears only once per i because mapping.
        rows.append(row)
        b.append(M)
    # add non-negativity constraints: -I* b <= 0
    rows.append(-np.eye(n))
    b.append(np.zeros(n))
    A_ub = np.vstack(rows)
    b_ub = np.concatenate(b)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
    if res.success:
        return res.x
    return None

"""
propose_candidate
-----------------

A simple yet effective heuristic for the step‑function minimisation problem.
The algorithm alternates between two phases:

1. **Linear‑programming direction** –  
   For the current sequence `f` we solve the LP

       maximise   sum(b)
       s.t.       f * b  ≤  M   (elementwise)
                  b ≥ 0

   where `M = max(f*f)`.  The solution `g` has the same non‑negative
   support as `f` but a (typically) larger ℓ¹‑norm while respecting the
   same convolution bound.  Mixing a fraction of `g` into `f` raises the
   total mass without increasing the peak convolution much, which reduces
   the objective `2·n·max(f*f)/sum(f)²`.

2. **Local random perturbations** –  
   When the LP fails or stagnates we perform cheap random transfers of a
   small amount of mass between two positions.  This helps escape plateaus.

The mixing factor `t` is adapted: it is increased when an improvement is
observed and decreased otherwise, allowing both fast progress and fine
tuning.  The whole search runs until the allotted time (default 1000 s)
is exhausted, after which the best sequence found is returned.

The code uses only `numpy`, `scipy.optimize.linprog` (HiGHS) and the
provided `evaluate_sequence`.  No file or network I/O is performed.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper: solve the LP  maximise sum(b)  s.t.  conv(f,b) ≤ M , b ≥ 0
# ----------------------------------------------------------------------
def solve_convolution_lp(f_seq, M):
    """
    Solve the linear program:
        maximise   sum(b)
        subject to conv(f_seq, b) ≤ M   (for every shift)
                   b ≥ 0
    Returns the optimal b as a NumPy array, or None if the solver fails.
    """
    n = len(f_seq)
    # Objective: minimise -sum(b)  (i.e. maximise sum(b))
    c = -np.ones(n, dtype=float)

    # Build A_ub * b ≤ b_ub  for the convolution constraints.
    # There are (2*n-1) convolution rows plus n non‑negativity rows.
    rows = []
    ub = []

    # Convolution rows
    for k in range(2 * n - 1):
        row = np.zeros(n, dtype=float)
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # For each i in the valid overlap range set column j = k-i
        for i in range(i_min, i_max + 1):
            j = k - i
            row[j] = f_seq[i]
        rows.append(row)
        ub.append(M)

    # Non‑negativity rows: -I * b ≤ 0   ⇔   b ≥ 0
    rows.append(-np.eye(n))
    ub.append(np.zeros(n, dtype=float))

    A_ub = np.vstack(rows)
    b_ub = np.concatenate(ub)

    # Use HiGHS (the default) – it is fast for dense problems of this size.
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
    if res.success:
        return res.x
    return None


# ----------------------------------------------------------------------
# Main entry point required by the runner
# ----------------------------------------------------------------------
def propose_candidate(seed=0, budget_s=1000):
    """
    Search for a sequence of non‑negative heights that minimises
    evaluate_sequence().  The routine runs for at most `budget_s` seconds
    and returns the best sequence discovered.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # a small safety margin

    # ------------------------------------------------------------------
    # 0. Choose an initial sequence
    # ------------------------------------------------------------------
    # Prefer the provided SOTA construction, but occasionally start from
    # a fresh random vector (helps avoid being stuck in a local minimum).
    if 'height_sequence_1' in globals() and np.random.rand() < 0.6:
        f = np.array(height_sequence_1, dtype=float)
    else:
        # a modest length – large enough to give a good bound but cheap to
        # handle inside the LP.
        n = 800
        f = np.random.rand(n).astype(float)

    # Clip to the allowed range and guarantee positivity for the first run.
    f = np.clip(f, 0.0, 1000.0)
    if f.sum() < 0.01:
        f = np.ones_like(f)

    best_f = f.copy()
    best_val = evaluate_sequence(best_f.tolist())
    print(f"[init] n={len(f)}  value={best_val:.6f}")

    # ------------------------------------------------------------------
    # 1. Main optimisation loop
    # ------------------------------------------------------------------
    mix_factor = 0.2          # initial mixing coefficient (1‑t)
    min_mix = 0.01
    max_mix = 0.5
    no_improve = 0

    while time.time() < deadline:
        # Current convolution bound M
        conv_f = np.convolve(f, f)
        M = conv_f.max()

        # --------------------------------------------------------------
        # LP step – try to find a direction with larger ℓ¹‑norm
        # --------------------------------------------------------------
        g = solve_convolution_lp(f.tolist(), float(M))

        if g is not None:
            # Normalise g to have the same total mass as f
            sum_f = f.sum()
            sum_g = g.sum()
            if sum_g > 0:
                g = g * (sum_f / sum_g)

            # Mix a fraction of the new direction into f
            new_f = (1.0 - mix_factor) * f + mix_factor * g
        else:
            # LP failed (unlikely) – fall back to a tiny random transfer
            idx_from, idx_to = np.random.choice(len(f), 2, replace=False)
            delta = np.random.randn() * 0.01 * max(1.0, f[idx_from])
            new_f = f.copy()
            new_f[idx_from] = max(0.0, new_f[idx_from] - delta)
            new_f[idx_to]   = max(0.0, new_f[idx_to]   + delta)

        # ----------------------------------------------------------------
        # Evaluate the new candidate
        # ----------------------------------------------------------------
        val = evaluate_sequence(new_f.tolist())

        if val < best_val:
            best_val = val
            best_f = new_f.copy()
            no_improve = 0
            # increase step size modestly – we are making progress
            mix_factor = min(max_mix, mix_factor * 1.1)
            print(f"[improve] val={best_val:.6f}  mix={mix_factor:.3f}  t={time.time()-start:.1f}s")
        else:
            # no improvement – shrink the mixing factor to fine‑tune
            mix_factor = max(min_mix, mix_factor * 0.85)
            no_improve += 1

        f = new_f

        # ----------------------------------------------------------------
        # Occasional random diversification if we have been stuck too long
        # ----------------------------------------------------------------
        if no_improve >= 150:
            # perturb the best known solution a little
            f = best_f + np.random.randn(len(best_f)) * 0.002
            f = np.clip(f, 0.0, 1000.0)
            no_improve = 0
            mix_factor = 0.2
            print(f"[reset] diversification at t={time.time()-start:.1f}s")

    # ----------------------------------------------------------------------
    # Return the best sequence found (as a plain Python list)
    # ----------------------------------------------------------------------
    final_val = evaluate_sequence(best_f.tolist())
    print(f"[final] best value={final_val:.6f}  length={len(best_f)}")
    return best_f.tolist()
