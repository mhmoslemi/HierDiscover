"""
Propose a candidate step‑function sequence.
The algorithm combines a simple sub‑gradient style mixing with a linear‑programming
direction finder (the same idea as the published “LP‑move” method) and a
final hill‑climbing phase.  It runs for the allotted time budget (≈1000 s)
and returns the best sequence seen.

Key ideas
---------

* The evaluation is scale‑invariant, therefore we keep the total sum of the
  coefficients close to √(2 n) (the normalisation used in the original LP
  scheme) – this keeps the LP well‑scaled.
* From a current sequence `f` we solve a linear programme that finds a
  non‑negative vector `g` with maximal ℓ₁‑norm under the constraint
  `conv(f,g) ≤ max(conv(f,f))`.  Mixing `f` with `g` (using a step size `t`)
  provably does not increase the maximal convolution value.
* After a few LP moves we perform a cheap stochastic hill‑climbing step:
  add small Gaussian noise to a few coordinates, keep the change if the
  evaluation improves.
* The whole process is repeated from several random starts (including the
  supplied `height_sequence_1`) while respecting the time budget.

The implementation is pure NumPy / SciPy and respects all contest rules.
"""

import numpy as np
import time
from scipy import optimize

# --------------------------------------------------------------------------- #
# Linear programming sub‑routine                                            #
# --------------------------------------------------------------------------- #
def solve_convolution_lp(f_seq, rhs, time_limit=5.0):
    """
    Solve the LP:
        maximise    Σ_j b_j
        subject to   (f_seq * b)_k ≤ rhs   for all k
                     b_j ≥ 0

    Parameters
    ----------
    f_seq : np.ndarray, shape (n,)
        Non‑negative coefficients of the current sequence (already normalised).
    rhs : float
        Upper bound for the convolution (max over all offsets of f_seq*f_seq).
    time_limit : float, optional
        Maximum CPU time for the LP solver (seconds).

    Returns
    -------
    np.ndarray or None
        Optimal vector `b` (length n) if the LP succeeds, else ``None``.
    """
    n = f_seq.shape[0]
    # ----- Build A_ub, b_ub for the convolution constraints -----
    # There are (2n‑1) convolution rows.
    rows = []
    rhs_vec = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # i runs over indices where both i and k-i are inside [0, n‑1]
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        # for each such i, coefficient of b_{k-i} is f_i
        idxs = np.arange(i_min, i_max + 1, dtype=int)
        js = k - idxs
        row[js] = f_seq[idxs]
        rows.append(row)
        rhs_vec.append(rhs)

    A_ub = np.vstack(rows)
    b_ub = np.array(rhs_vec, dtype=float)

    # Bounds enforce b_j ≥ 0 (no explicit upper bound)
    bounds = [(0.0, None)] * n

    # Solve with HiGHS (fast interior‑point) and a modest time limit
    try:
        res = optimize.linprog(
            c=-np.ones(n),                      # maximise Σ b_j  <=> minimise -Σ b_j
            A_ub=A_ub,
            b_ub=b_ub,
            bounds=bounds,
            method='highs',
            options={'time_limit': time_limit, 'presolve': True, 'disp': False},
        )
    except Exception:  # linprog may raise if the problem is pathological
        return None

    if res.success:
        return np.maximum(res.x, 0.0)  # numerical safety
    return None


# --------------------------------------------------------------------------- #
# Main candidate generator                                                   #
# --------------------------------------------------------------------------- #
def propose_candidate(budget_s: float = 1000.0, seed: int = 123, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises
    ``evaluate_sequence``.  The routine respects the given time budget and
    returns the best list found.

    Parameters
    ----------
    budget_s : float, optional
        Time budget in seconds (default 1000 s).
    seed : int, optional
        Random seed for reproducibility.

    Returns
    -------
    list[float]
        The best sequence discovered within the time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0               # safety margin

    # ------------------------------------------------------------------- #
    # Parameters controlling the search
    # ------------------------------------------------------------------- #
    n = 250                     # modest size – fast LP solves, still expressive
    sqrt_2n = np.sqrt(2.0 * n)
    t_mix = 0.12                # mixing fraction with the LP direction
    lp_time = 5.0               # per‑LP time limit (seconds)
    max_outer_iters = 30        # how many LP‑mixing cycles at most
    # ------------------------------------------------------------------- #

    # Helper to evaluate a numpy array (list conversion required by the judge)
    def eval_seq(arr):
        return evaluate_sequence(arr.tolist())

    # ------------------------------------------------------------------- #
    # Initialise from a few different seeds (including the provided one)
    # ------------------------------------------------------------------- #
    candidates = []

    # 1. Use the supplied SOTA sequence if available
    try:
        cand = np.array(height_sequence_1, dtype=float)
        if cand.size > 0:
            candidates.append(cand[:n] if cand.size >= n else
                              np.pad(cand, (0, n - cand.size), constant_values=0.0))
    except Exception:
        pass

    # 2. Random starts – a few quick trials
    for _ in range(3):
        a0 = rng.random(n)
        a0 = np.maximum(a0, 0.0)
        candidates.append(a0)

    # Record the best overall sequence
    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------- #
    # Main optimisation loop for each initial candidate
    # ------------------------------------------------------------------- #
    for start_vec in candidates:

            a += 0.01
        # Normalise to a reasonable scale (optional – the objective is scale‑free)
        a = a / a.sum() * sqrt_2n

        try:
            cur_score = eval_seq(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()

        except Exception:
            cur_score = np.inf

            outer += 1

            sum_a = a.sum()
            if sum_a == 0.0:

            if g is not None:

                sum_g = g.sum()
                if sum_g == 0.0:
                    # degenerate – fallback to random perturbation
                    idx = rng.integers(0, n)
                    a[idx] = max(0.0, a[idx] + rng.normal() * 0.02)
                else:
                    g_norm = g * sqrt_2n / sum_g
                    # Convex combination – guarantees non‑increase of the bound
                    a = (1.0 - t_mix) * a + t_mix * g_norm
            else:
                # LP failed (e.g. time out) – small random tweak to escape stagnation
                idx = rng.integers(0, n)
                a[idx] = max(0.0, a[idx] + rng.normal() * 0.02)

            # ----- Evaluate the new candidate -----
            try:
                cur_score = eval_seq(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(f"New best after LP mix: {best_score:.6f} (iter {outer})")
            except Exception:
                # Invalid sequence – ignore
                cur_score = np.inf

        # ------------------------------------------------------------------- #
        # After the LP‑mixing phase, perform a short stochastic hill‑climb
        # ------------------------------------------------------------------- #
        hill_iters = 0
        sigma = 0.02
        while time.time() < deadline and hill_iters < 2000:
            hill_iters += 1
            # Propose a tiny random perturbation
            proposal = best_seq.copy()
            # modify a few random positions
            idxs = rng.choice(n, size=max(1, n // 50), replace=False)
            proposal[idxs] += rng.normal(scale=sigma, size=idxs.shape)
            proposal = np.maximum(proposal, 0.0)          # enforce non‑negativity

            # Re‑normalise to keep scale comparable (optional)
            if proposal.sum() == 0.0:
                proposal = best_seq.copy()
            else:
                proposal = proposal / proposal.sum() * sqrt_2n

            try:
                val = eval_seq(proposal)
                if val < best_score:
                    best_score = val
                    best_seq = proposal.copy()
                    print(f"Improved by hill‑climb: {best_score:.6f} (step {hill_iters})")
                    # shrink step size a bit when we succeed
                    sigma *= 0.9
                else:
                    # slowly decay the perturbation magnitude
                    sigma *= 0.9995
            except Exception:
                continue

        # If time is exhausted, break early
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------- #
    # Return the best sequence (as a plain Python list)
    # ------------------------------------------------------------------- #
    if best_seq is None:
        # As a safety net, fall back to a simple uniform sequence
        best_seq = np.full(n, 1.0 / n, dtype=float)
    return best_seq.tolist()
