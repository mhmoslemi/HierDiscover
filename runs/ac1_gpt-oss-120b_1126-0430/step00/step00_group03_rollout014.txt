"""
Propose a sequence that dramatically lowers the evaluation score.

The evaluation function is
    score = 2 * n * max(conv(a, a)) / (sum(a) ** 2)

For a monotone power‑law shape
    a_i = C * (i / (n-1)) ** p   (0 ≤ i < n)

the maximum of the convolution is either the square of the last element
(= C²) or the value at the centre (≈ C² * n * B(p+1,p+1)),
where B is the beta function.  For p≈5 the centre dominates and the
score becomes

    score = 2 * B(p+1, p+1) * (p+1)² ≈ 0.0259

independent of the length n (as soon as n > 1/B ≈ 2800).  Using a length
of a few thousand keeps the convolution cheap while staying well below
the target 1.503.

The function builds several candidate exponents, evaluates them and
returns the best sequence found.
"""
import numpy as np
import time

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Generate a monotone power‑law step function.
    The sequence is a_i = 1000 * (i/(n-1))**p with p≈5.
    Several exponents are tried; the one giving the smallest score is
    returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # leave a tiny safety margin

    # choose a reasonable length (large enough that the centre of the
    # convolution dominates, but not so large that the O(n²) convolution
    # becomes a bottleneck).  4000 works well on typical hardware.
    n = 4000

    # candidate exponents (including a few non‑integers for fine tuning)
    exponents = [1, 2, 3, 4, 4.5, 5, 5.5, 6, 7]

    best_seq = None
    best_score = np.inf

    # pre‑allocate index array to avoid repeated allocations
    idx = np.arange(n, dtype=float)

    # evaluate all candidates quickly
    for p in exponents:
        # build the raw shape in [0,1]
        raw = (idx / (n - 1)) ** p
        # scale so that the maximum equals 1000 (the clipping limit)
        seq = raw * 1000.0

        # convert to plain Python list (evaluate_sequence expects a list)
        seq_list = seq.tolist()

        # call the provided evaluation function
        try:
            score = evaluate_sequence(seq_list)
        except Exception:  # safety – the verifier may reject weird inputs
            continue

        if score < best_score:
            best_score = score
            best_seq = seq_list

        # stop early if we run out of time
        if time.time() > deadline:
            break

    # If for any reason none of the candidates succeeded (should not happen)
    # fall back to a simple linear ramp.
    if best_seq is None:
        raw = idx / (n - 1)
        best_seq = (raw * 1000.0).tolist()
        best_score = evaluate_sequence(best_seq)

    # Log the chosen exponent and resulting score for debugging.
    # (The print output is captured and shown back to you.)
    chosen_exp = exponents[0] if best_seq is None else None
    print(f"Chosen sequence length: {len(best_seq)}")
    print(f"Best score achieved: {best_score:.6f}")

    return best_seq
