def autocorr(a):
    n = len(a)
    N = 1 << (2*n-1).bit_length()
    fa = np.fft.rfft(a, N)
    conv = np.fft.irfft(fa*fa, N)[:2*n-1]
    return conv

        j = np.random.randint(n-1)
        if j >= i:
            j += 1

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        a_new /= a_new.sum()

        if score_new < best_score or np.random.rand() < np.exp((best_score - score_new) / (T+1e-12)):
            a = a_new
            if score_new < best_score:
                best_score = score_new
                best_a = a_new.copy()

        # occasionally (every 1000 iter) do gradient-based improvement step:
        if iteration % 1000 == 0:

                # sample random other index not in its counterpart pair

def proj_simplex(v):
    """Project onto the probability simplex {x>=0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u - (cssv - 1) / np.arange(1, n+1) > 0)[0][-1]

    w = np.maximum(v - theta, 0)
    return w

def propose_candidate(seed=0, budget_s=1000):
    """Search for a non‑negative step sequence that minimizes the supplied
    ``evaluate_sequence`` objective.  The method uses a projected sub‑gradient
    descent on the maximal self‑convolution value together with occasional
    random restarts.  It works entirely on the probability simplex, i.e.
    the coefficients are kept non‑negative and their sum is fixed to one,
    which is safe because the evaluation function is scale‑invariant.
    """
    import numpy as np, time
    np.random.seed(seed)
    # Determine sequence length
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
    else:
        n = 500   # fallback
    # initialise with a random Dirichlet point
    a = np.random.rand(n)
    a = a / a.sum()
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())
    print(f"Initial random score: {best_score:.6f}")
    deadline = time.time() + budget_s - 0.5
    it = 0
    no_improve_cnt = 0
    
    # parameters
    step0 = 0.5    # initial step size
    step_decay = 0.9995  # per iteration decay
    restart_interval = 5000   # iterations without improvement before restart
    top_k = 3      # number of largest convolution entries used for sub‑gradient
    
    while time.time() < deadline:
        it += 1
        # compute autocorrelation via FFT (full length)
        N_fft = 1 << (2*n - 1).bit_length()
        fa = np.fft.rfft(a, N_fft)
        conv = np.fft.irfft(fa * fa, N_fft)[:2*n-1]
        max_val = conv.max()
        # choose up to top_k indices with largest values
        if top_k == 1:
            ks = [int(conv.argmax())]
        else:
            ks = np.argpartition(conv, -top_k)[-top_k:]
        # build sub‑gradient
        g = np.zeros_like(a)
        for k in ks:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi+1)
            g[i] += 2.0 * a[k - i]
        # step size schedule
        step = step0 / np.sqrt(it)
        a_new = a - step * g
        # project onto simplex
        a_new = proj_simplex(a_new)
        # evaluate
        try:
            score = evaluate_sequence(a_new.tolist())
        except Exception:
            # protect against any numeric mishap
            continue
        # accept if improvement, otherwise keep old
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            no_improve_cnt = 0
            print(f"Iter {it:8d}  new best {best_score:.6f}")
        else:
            # maybe accept worse move with small probability (simulated annealing)
            # simple schedule: prob = exp(-(score - best_score) / (step*0.1 + 1e-12))
            if np.random.rand() < np.exp(-(score - best_score) / (0.01 + 1e-12)):
                a = a_new
            else:
                # otherwise, keep current a unchanged (or still assign a_new to allow wandering)
                a = a_new
        # keep track for possible restart
        if score >= best_score:
            no_improve_cnt += 1
        if no_improve_cnt >= restart_interval:
            # restart with fresh random point
            a = np.random.rand(n)
            a = a / a.sum()
            no_improve_cnt = 0
            print(f"Restart at iteration {it}")
        # optional periodic perturbation to avoid stagnation
        if it % 2000 == 0:
            # add small Gaussian noise then re‑project
            perturb = np.random.randn(n) * 0.01
            a = proj_simplex(a + perturb)
    # return best sequence found
    print(f"Finished after {it} iterations, best score {best_score:.6f}")
    return best_a.tolist()

"""Projected sub‑gradient search for a step sequence minimizing the
``evaluate_sequence`` objective.

The method works on the probability simplex (coefficients are non‑negative
and sum to one).  Because the evaluation function is homogeneous
(its value is invariant under scaling of the whole vector) we can fix the
sum at 1 without loss of generality.  At each iteration we

1. compute the full self‑convolution via a fast FFT,
2. identify the index (or a few indices) where the convolution attains its
   maximum,
3. build a sub‑gradient of the max‑convolution w.r.t. the coefficients,
4. take a small step opposite to that sub‑gradient and project back onto
   the simplex,
5. keep the best vector seen so far.

Occasionally a random restart or a small Gaussian perturbation is applied
to avoid being trapped in a stationary point.  The algorithm runs until
the allocated time budget expires and returns the best sequence found.

The implementation is completely self‑contained; it only uses ``numpy``,
``time`` and the provided ``evaluate_sequence`` function.
"""

import numpy as np
import time

def _project_simplex(v):
    """Projection of a vector onto the probability simplex {x>=0, sum x = 1}."""
    # Duchi et al. (2008) algorithm
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def _full_convolution(a):
    """Full discrete convolution a*a using an FFT of length power‑of‑2."""
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, N)
    conv = np.fft.irfft(fa * fa, N)
    return conv[: 2 * n - 1]

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    """Return a non‑negative sequence that (hopefully) lowers ``evaluate_sequence``.

    seed : int, optional

    **kwargs : dict
        Ignored – present for compatibility with the caller.

    Returns
    -------
    list[float]
        The best sequence discovered within the time budget.
    """

    # Prefer the length of the supplied reference sequence, otherwise fall back.
    # ------------------------------------------------------------------

        n = len(height_sequence_1)
    else:
        n = 500   # a reasonable default

    # ------------------------------------------------------------------
    a = np.random.rand(n)
    a = a / a.sum()

    # Keep the best incumbent.
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Parameters for the sub‑gradient descent + simulated‑annealing style
    # ------------------------------------------------------------------
    deadline = time.time() + budget_s - 0.5   # some safety margin
    it = 0
    no_improve = 0
    restart_after = 5000   # iterations without progress before a full restart
    perturb_every = 2000   # add a tiny random perturbation periodically
    top_k = 3              # how many largest convolution entries to use for the gradient
    step0 = 0.7            # initial step size (will be divided by sqrt(it))

    while time.time() < deadline:
        it += 1

        # --------------------------------------------------------------
        # 1) Full self‑convolution
        # --------------------------------------------------------------
        conv = _full_convolution(a)
        max_val = conv.max()

        # --------------------------------------------------------------
        # 2) Choose the indices that realize (or almost realize) the max.
        # --------------------------------------------------------------
        if top_k == 1:
            ks = [int(conv.argmax())]
        else:
            # argpartition is O(n) and gives the top‑k without full sort
            ks = np.argpartition(conv, -top_k)[-top_k:]

        # --------------------------------------------------------------
        # 3) Build the sub‑gradient of the max‑convolution.
        #    For a fixed k the gradient is g_i = 2 * a_{k-i} on the feasible
        #    index range and zero elsewhere.
        # --------------------------------------------------------------
        g = np.zeros_like(a)
        for k in ks:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            g[i] += 2.0 * a[k - i]

        # --------------------------------------------------------------

        a_candidate = _project_simplex(a_candidate)

        #    accepted to allow escape from shallow plateaus.

            a = np.random.rand(n)
            a = a / a.sum()
            no_improve = 0
            print(f"[{it:7d}] restart due to stagnation")

    # ------------------------------------------------------------------
    # Return the best sequence discovered.
    # ------------------------------------------------------------------
    print(f"finished after {it} iterations, best score={best_score:.6f}")
    return best_a.tolist()
