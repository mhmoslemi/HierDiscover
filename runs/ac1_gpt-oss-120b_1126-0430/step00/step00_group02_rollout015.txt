"""
Propose a sequence of non‑negative heights that improves the current score.

The core idea is to start from the discrete analogue of the arcsine
distribution – the coefficients of the power series of (1‑z)^‑1/2.
For a sequence a_i = C_i = binom(2i,i) / 4**i the self‑convolution
is exactly constant (equal to 1) for the first |seq| entries, giving a
very good value 2·n·max(b)/sum(a)^2 ≈ π/2 ≈ 1.57, much better than the
previous bound of 2.0.

From this analytically good seed we run a cheap greedy refinement:
the index k where the convolution attains its maximum is identified;
the contribution of each a_i to this maximal entry is a_{k‑i}.
Mass is repeatedly moved from the index that contributes most to the
maximum to an index that contributes almost nothing.  This never
increases the objective and typically reduces the maximal convolution
value, thus lowering the evaluation score.

The whole search respects the 1000 s budget and returns the best
sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: fast evaluation (identical to evaluate_sequence but without the
# security checks and using an FFT based convolution).
# ----------------------------------------------------------------------
def _fast_evaluation(seq: np.ndarray) -> float:
    """Return the same metric as evaluate_sequence but much faster."""
    n = seq.size
    if n == 0:
        return np.inf
    # FFT size = next power of two >= 2*n-1
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Generate the Catalan (arcsine) seed.
# a_0 = 1, a_{i+1} = a_i * (2*i+1) / (2*i+2)
# ----------------------------------------------------------------------
def _catalan_sequence(length: int) -> np.ndarray:
    """Return the first ``length`` Catalan coefficients (as a float array)."""
    seq = np.empty(length, dtype=float)
    a = 1.0
    seq[0] = a
    for i in range(1, length):
        # recurrence: a_{i} = a_{i-1} * (2*i-1) / (2*i)
        a *= (2 * i - 1) / (2.0 * i)
        seq[i] = a
    return seq


# ----------------------------------------------------------------------
# Greedy local improvement – move mass away from indices that heavily
# influence the current maximal convolution entry.
# ----------------------------------------------------------------------
def _refine(seq: np.ndarray, deadline: float) -> np.ndarray:
    """Improves ``seq`` in‑place until the time limit is reached."""
    n = seq.size
    best_seq = seq.copy()
    best_score = _fast_evaluation(best_seq)

    # pre‑compute FFT size (does not change)
    fft_len = 1 << ((2 * n - 1).bit_length())

    while time.time() < deadline:
        # current convolution and the index where it is maximal
        fa = np.fft.rfft(seq, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        k = int(conv.argmax())
        max_val = conv[k]

        # contribution weights for each position to that maximal entry
        # w[i] = a_{k-i} if the index exists, otherwise 0
        i_arr = np.arange(n)
        j_arr = k - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        w = np.zeros_like(seq)
        w[mask] = seq[j_arr[mask]]

        # pick a source (largest contribution) and a sink (almost none)
        contrib = seq * w
        src = int(np.argmax(contrib))
        sink = int(np.argmin(w))  # often a location with zero contribution

        if seq[src] < 1e-12:
            break          # nothing left to move

        # move a modest fraction of the mass
        eps = 0.05 * seq[src]          # 5 % of the source amount
        new_seq = seq.copy()
        new_seq[src] -= eps
        new_seq[sink] += eps

        # evaluate the new candidate
        score = _fast_evaluation(new_seq)

        # accept only if we improve (strictly better)
        if score < best_score - 1e-9:
            seq = new_seq
            best_score = score
            best_seq = new_seq.copy()
        else:
            # occasional random jitter to escape tiny plateaus
            if np.random.rand() < 0.01:
                jitter_idx = np.random.randint(n)
                jitter = np.random.randn() * 0.001
                new_seq = seq.copy()
                new_seq[jitter_idx] = max(0.0, new_seq[jitter_idx] + jitter)
                seq = new_seq

    return best_seq


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Produce a list of non‑negative heights aiming at a low evaluation score.

    The function works within ``budget_s`` seconds (default 1000).  It starts
    from the Catalan (arcsine) sequence, performs a fast greedy refinement
    that repeatedly reduces the peak of the self‑convolution, and finally
    returns the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5   # keep a small safety margin

    # length of the candidate – a few thousand gives a good trade‑off
    n = 3000

    # --------------------------------------------------------------
    # 1.  Initial seed: Catalan / arcsine coefficients.
    # --------------------------------------------------------------
    seq = _catalan_sequence(n)

    # --------------------------------------------------------------
    # 2.  Greedy refinement.
    # --------------------------------------------------------------
    seq = _refine(seq, deadline)

    # --------------------------------------------------------------
    # 3.  Return the best sequence found (as a plain Python list).
    # --------------------------------------------------------------
    # The official ``evaluate_sequence`` will clip to [0,1000] and apply
    # the same security checks, so we can safely hand over the raw values.
    return seq.tolist()
