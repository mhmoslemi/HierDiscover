Pseudo-code for projection onto simplex of given sum:

def project_onto_simplex(v, s):
    # Input v: array of real numbers, output w: w >=0, sum w = s, w is Euclidean projection.
    # Algorithm: sort v descending, find rho; compute theta; w = max(v - theta, 0)
    # Implementation reference Duchi et al.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv/(np.arange(1, n+1)) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def gradient_step(a, step_frac):
    n = len(a)
    b = np.convolve(a, a)
    max_b = b.max()
    max_inds = np.where(b >= max_b - 1e-12)[0]
    # compute subgradient g
    g = np.zeros_like(a)
    for k in max_inds:
        start_i = max(0, k - (n - 1))
        end_i = min(k, n - 1)

def project_onto_simplex(v, s):
    """Project v onto simplex {x >=0, sum x = s}."""
    n = len(v)
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n+1)) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

    max_b = conv.max()

"""
Propose a step‑function (non‑negative vector) that minimizes the
evaluation function:

    score = 2·n·max(conv(a,a)) / (sum(a)**2)

The score is invariant to scaling, therefore we keep the L1 norm
fixed to √(2·n).  The algorithm works as follows:

1.  Several initial shapes are generated (uniform, triangular,
    sine‑wave, beta‑distribution, and random Dirichlet).  All are
    projected onto the simplex {a ≥ 0, ∑a = √(2·n)}.

2.  For each start a simple sub‑gradient descent is performed:
    -   Compute the full convolution a∗a and locate the index/indices
        where the maximum is attained.
    -   The sub‑gradient of the max‑convolution is
            g_i = 2·∑_{k∈argmax} a_{k‑i}   (ignoring out‑of‑range terms).
    -   Take a step opposite to the sub‑gradient with a decreasing
        step size and project back onto the simplex.

    Occasionally a random “mass‑transfer’’ move or a small Gaussian
    perturbation is tried; the move is kept only if it does not increase
    the current maximum convolution value.  This provides exploration
    and helps escape shallow plateaus.

3.  The best sequence seen over the whole allotted time (default
    1000 s) is returned.

All helper functions are defined at the top level; no file I/O is used.
"""

import time
import numpy as np
import random

# ----------------------------------------------------------------------
# Helper: projection onto the simplex {x ≥ 0, sum(x) = s}
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of size s."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Return a sub‑gradient of max(conv(a,a)) at a."""
    n = a.shape[0]
    max_val = conv.max()
    # treat ties within a tiny tolerance as active
    active = np.where(conv >= max_val - 1e-12)[0]
    g = np.zeros_like(a)
    for k in active:
        # indices i with 0 ≤ k‑i < n
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1)
        # contribution 2·a_{k‑i}
        g[i_range] += 2.0 * a[k - i_range]
    return g


# ----------------------------------------------------------------------
# Various deterministic initial shapes (all are projected onto the simplex)
def _init_uniform(n: int) -> np.ndarray:
    """Flat distribution."""
    return np.full(n, np.sqrt(2.0 * n) / n)


def _init_triangular(n: int) -> np.ndarray:
    """Symmetric triangle shape."""
    if n % 2 == 0:
        half = np.arange(1, n // 2 + 1)
        a = np.concatenate([half, half[::-1]])
    else:
        half = np.arange(1, (n + 1) // 2 + 1)
        a = np.concatenate([half, half[-2::-1]])
    a = a.astype(float)
    a = a / a.sum() * np.sqrt(2.0 * n)
    return a


def _init_sine(n: int) -> np.ndarray:
    """Half‑sine wave (positive values)."""
    x = np.arange(n) + 0.5
    a = np.sin(np.pi * x / n)
    a = a / a.sum() * np.sqrt(2.0 * n)
    return a


def _init_beta(n: int) -> np.ndarray:
    """Beta‑type distribution with random shape parameters."""
    # draw shape parameters in a reasonable range
    alpha = np.random.uniform(0.2, 2.0)
    beta = np.random.uniform(0.2, 2.0)
    xs = np.linspace(0.0, 1.0, n)
    # avoid infinities at the borders
    eps = 1e-8
    xs = np.clip(xs, eps, 1.0 - eps)
    pdf = (xs ** (alpha - 1)) * ((1.0 - xs) ** (beta - 1))
    pdf = np.maximum(pdf, 0.0)
    a = pdf / pdf.sum() * np.sqrt(2.0 * n)
    return a


def _init_random_dirichlet(n: int) -> np.ndarray:
    """Random Dirichlet (exponential) vector."""
    exp = np.random.exponential(scale=1.0, size=n)
    a = exp / exp.sum() * np.sqrt(2.0 * n)
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """Return a list of non‑negative heights (step function) with a low score.

    The function runs a time‑bounded stochastic search, trying several
    initial shapes and improving them with a simple sub‑gradient descent
    combined with occasional random mass‑transfer moves.
    """
    np.random.seed(seed)
    random.seed(seed)

    # safety margin before the hard deadline
    deadline = time.time() + budget_s - 0.5

    # ------------------------------------------------------------------
    # Determine which lengths to explore
    default_n = len(globals().get("height_sequence_1", []))

        default_n = 1500
    lengths = [

    ]
    # keep everything reasonable
    lengths = [int(l) for l in lengths if 200 <= l <= 4000]
    lengths = list(dict.fromkeys(lengths))  # deduplicate, preserve order

    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    # small helper to evaluate a candidate (list or ndarray)
    def _eval(seq):
        # convert to plain Python list – evaluate_sequence expects a list
        return evaluate_sequence(seq.tolist() if isinstance(seq, np.ndarray) else list(seq))

    # main loop over lengths and initial shapes
    for n in lengths:
        if time.time() >= deadline:
            break

        init_functions = [
            ("uniform", _init_uniform),
            ("triangular", _init_triangular),
            ("sine", _init_sine),
            ("beta", _init_beta),
            ("random", _init_random_dirichlet),

        for name, init_fn in init_functions:

            try:
                cur_score = _eval(a)
            except Exception:
                cur_score = float("inf")
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[init] n={n:4d} shape={name:9s}  score={cur_score:.6f}")

            # ------------------------------------------------------------------
            # Local optimisation: sub‑gradient descent + random moves
            max_iters = 2500
            # step size proportional to a typical entry (≈ √(2n) / n)
            typical = np.sqrt(2.0 * n) / n
            base_step = 3.0 * typical  # a moderate starting step

            for it in range(max_iters):
                if time.time() >= deadline:
                    break

                conv = np.convolve(a, a)
                max_b = conv.max()

                # Keep the globally best vector
                if max_b < best_score:
                    best_score = max_b
                    best_seq = a.copy()
                    print(f"[best] n={n:4d} it={it:4d}  score={best_score:.6f}")

                # Choose an update rule
                r = np.random.rand()
                if r < 0.45:
                    # --- Sub‑gradient descent step ---------------------------
                    g = _subgradient_max_conv(a, conv)
                    norm_g = np.linalg.norm(g)
                    if norm_g > 0:
                        # decreasing step size
                        step = base_step / np.sqrt(it + 1)
                        a_new = a - step * g
                        a = _project_onto_simplex(a_new, np.sqrt(2.0 * n))
                elif r < 0.75:

                    i, j = np.random.randint(0, n, size=2)
                    if i != j and a[i] > 0:
                        eps = np.random.rand() * a[i] * 0.5  # move up to 50 % of a[i]
                        a_candidate = a.copy()
                        a_candidate[i] -= eps
                        a_candidate[j] += eps
                        # quick check – only keep if improves the max‑conv
                        conv_cand = np.convolve(a_candidate, a_candidate)
                        if conv_cand.max() < max_b:
                            a = a_candidate
                else:
                    # --- Small Gaussian perturbation -------------------------
                    noise = np.random.randn(n) * typical * 0.02
                    a_candidate = a + noise
                    a_candidate = np.maximum(a_candidate, 0.0)
                    a_candidate = _project_onto_simplex(a_candidate, np.sqrt(2.0 * n))
                    conv_cand = np.convolve(a_candidate, a_candidate)
                    if conv_cand.max() < max_b:
                        a = a_candidate

            # final evaluation for this start
            try:
                final_score = _eval(a)
                if final_score < best_score:
                    best_score = final_score
                    best_seq = a.copy()
                    print(f"[final] n={n:4d} shape={name:9s}  score={final_score:.6f}")
            except Exception:
                pass

    # ----------------------------------------------------------------------
    # Fallback – should never happen, but keep a safe uniform vector
    if best_seq is None:
        n = 500
        best_seq = _init_uniform(n)

    # Return a plain Python list (the caller will feed it to evaluate_sequence)
    return [float(x) for x in best_seq.tolist()]
