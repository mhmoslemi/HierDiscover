    # gradient descent parameters
    lr = 0.5  # initial step size
    decay = 0.9995
    min_lr = 1e-5

    no_improve = 0

    while time.time() < deadline:
        # compute convolution and related values
        b = np.convolve(a, a)
        max_b = b.max()
        S = a.sum()
        const = 2 * n / (S ** 2)

        # find indices achieving max (within tolerance)
        # tolerance relative
        eps = 1e-12
        max_mask = np.isclose(b, max_b, atol=eps)
        Imax = np.flatnonzero(max_mask)

        # compute gradient g
        g = np.zeros_like(a)
        for k in Imax:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i = np.arange(i_start, i_end + 1)
            j = k - i
            g[i] += 2 * a[j]

        # gradient of objective
        grad = const * (g - 2 * max_b / S)

        # gradient step
        a_new = a - lr * grad
        # enforce non-negative
        a_new = np.maximum(a_new, 0.0)
        # renormalize to keep sum stable (although scaling invariant)
        sum_new = a_new.sum()
        if sum_new == 0:
            # reinitialize if all zero
            a_new = np.random.rand(n)
            a_new = a_new / a_new.sum()
            sum_new = a_new.sum()
        a_new = a_new / sum_new

        # evaluate
        score = evaluate_sequence(a_new.tolist())
        if score < best_score - 1e-9:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
            print(f"new best {best_score:.6f} at {time.time():.2f}")
        else:
            no_improve += 1

        a = a_new

        # decay learning rate
        lr = max(lr * decay, min_lr)

        # occasional random perturbation to escape stagnation

            # random shuffle small subset

            no_improve = 0

        return new_seq

"""
Propose a step‑function (non‑negative sequence) that minimises
    2 * n * max(conv(a,a)) / (sum(a)**2)

The algorithm consists of two stages.

1.  Gradient‑descent on the homogeneous objective.
    The gradient of the objective w.r.t. a_i is obtained from the
    sub‑gradient of the max‑convolution term.  After each step the
    vector is projected onto the simplex (non‑negative, sum = 1).

2.  A few refinement steps using the linear‑programming update that
    was used in the published SOTA method:
        • normalise the current vector so that its sum = sqrt(2n)
        • solve   max sum(g)   s.t.   conv(f,g) ≤ max(conv(f,f)),  g ≥ 0
        • renormalise g and mix it with the current vector.

Both phases run for a fraction of the allotted wall‑time and the
best sequence found is returned.  The routine works without any file
IO and respects the 1000 s time budget.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper: convolution (full) – small n, direct method is fast enough.
def _full_conv(x):
    """Linear (full) convolution using NumPy."""
    return np.convolve(x, x)


# ----------------------------------------------------------------------
# Helper: solve the LP that appears in the SOTA iteration.
def _solve_convolution_lp(f_seq, rhs):
    """
    Solve   max sum(b)   s.t.   conv(f_seq, b) ≤ rhs,   b ≥ 0.
    Returns the optimal b (numpy array) or None on failure.
    """
    n = len(f_seq)
    # objective: maximise sum(b)  → minimise -sum(b)
    c = -np.ones(n)

    # Build the constraints for each convolution index k.
    A = []
    b = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # conv(f,b)[k] = Σ_i f_i * b_{k-i}
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_seq[i]
        A.append(row)
        b.append(rhs)

    # non‑negativity b ≥ 0 is encoded as –b ≤ 0
    A.append(-np.eye(n))
    b.append(np.zeros(n))

    A = np.vstack(A)
    b = np.concatenate(b)

    # Use the robust HiGHS solver (default in recent SciPy).
    res = linprog(c, A_ub=A, b_ub=b, method='highs')
    if res.success:
        return res.x
    return None


# ----------------------------------------------------------------------
# Helper: a single LP‑based improvement step.
def _lp_improve(seq, mix=0.05):
    """
    Perform one LP‑based move:
        1.  Normalise seq to have ℓ₁‑norm = sqrt(2n).
        2.  Solve the LP for a better direction g.
        3.  Renormalise g to the same ℓ₁‑norm.
        4.  Return (1‑mix)·seq + mix·g   (projected onto ℝ₊).
    """
    n = len(seq)
    s = seq.sum()
    if s == 0:
        return seq

    # 1 – normalise to the canonical ℓ₁‑norm.
    f = seq * np.sqrt(2.0 * n) / s

    # 2 – bound that the convolution must not exceed.
    rhs = _full_conv(f).max()

    g = _solve_convolution_lp(f, rhs)
    if g is None:
        return seq

    # 3 – bring g back to the same ℓ₁‑norm as the original sequence.
    sg = g.sum()
    if sg == 0:
        return seq
    g = g * np.sqrt(2.0 * n) / sg

    # 4 – convex combination and projection.
    new_seq = (1.0 - mix) * seq + mix * g
    new_seq = np.maximum(new_seq, 0.0)
    return new_seq

    Search for a low‑value sequence within the given time budget.
    Returns a list of non‑negative heights.
    """

    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # a few seconds safety margin

    # ------------------------------------------------------------------
    # Choose a length that is large enough to have expressive power

    a = np.random.rand(n)
    a /= a.sum()

    best_score = 2.0 * n * _full_conv(a).max() / (a.sum() ** 2)

    # ------------------------------------------------------------------
    # Phase 1 – simple projected gradient descent.
    lr = 0.02                     # initial learning rate
    decay = 0.9999                # geometric decay
    min_lr = 1e-6

    while time.time() < deadline - 30.0:   # reserve ~30 s for the LP phase
        conv = _full_conv(a)
        max_b = conv.max()
        S = a.sum()
        const = 2.0 * n / (S * S)
