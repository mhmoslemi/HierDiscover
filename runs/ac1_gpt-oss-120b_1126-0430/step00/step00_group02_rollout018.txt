import numpy as np
import time

def generate_inverse_sqrt(n):
    """Return sequence a_i = 1/sqrt(i+1) for i=0..n-1."""
    idx = np.arange(1, n+1, dtype=float)
    return (1.0/np.sqrt(idx)).tolist()

def generate_sqrt_diff(n):
    """Return sequence a_i = sqrt(i+1) - sqrt(i). Telescoping sum = sqrt(n)."""
    i = np.arange(n+1, dtype=float)
    # sqrt(i) for i =0..n
    sqrt_i = np.sqrt(i)
    return (sqrt_i[1:] - sqrt_i[:-1]).tolist()

def generate_geometric(n, r):
    """Return geometric decay a_i = r**i with 0<r<1."""
    return (r ** np.arange(n, dtype=float)).tolist()

def generate_power_law(n, p, offset=0.0):
    """Return a_i = (i + offset + 1) ** (-p)."""
    idx = np.arange(n, dtype=float) + offset + 1.0
    return (idx ** (-p)).tolist()

def generate_sine_weighted_inverse_sqrt(n):
    """Return a_i = (1/sqrt(i+1)) * sin(pi*(i+0.5)/n). Positive weighting."""
    base = 1.0 / np.sqrt(np.arange(1, n+1, dtype=float))
    weight = np.sin(np.pi * (np.arange(n, dtype=float) + 0.5) / n)
    return (base * weight).tolist()

def evaluate(seq):
    """Wrapper to call the external evaluate_sequence."""
    return evaluate_sequence(seq)

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Search for a step‑function height sequence that minimizes
    ``evaluate_sequence``.  The search explores several analytical families
    (inverse‑square‑root, telescoping square‑root differences, geometric,
    power‑law, sine‑weighted) with a few parameter sweeps, then performs a
    simple hill‑climbing refinement.  The best sequence found before the
    time budget expires is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + max(0.1, budget_s - 0.5)   # leave a tiny margin

    best_seq = None
    best_score = float('inf')

    # ------------------------------------------------------------------
    # 1. Exhaustive sweep over a few handcrafted families
    # ------------------------------------------------------------------
    n_choices = [256, 512, 1024, 2048, 4096, 5000]
    # power‑law exponents around the theoretically promising 0.5
    p_choices = [0.4, 0.45, 0.5, 0.55, 0.6]
    offset_choices = [0.0, 0.5, 1.0, 2.0]
    r_choices = [0.94, 0.96, 0.98, 0.99, 0.995]

    # Helper to evaluate and possibly update the incumbent
    def consider(seq):
        nonlocal best_seq, best_score
        # evaluate_sequence expects a list of floats
        try:
            val = evaluate(seq)
        except Exception:
            return
        if np.isnan(val) or np.isinf(val):
            return
        if val < best_score:
            best_score = val
            best_seq = seq
            print(f"New best: {val:.6f} (len={len(seq)})")
    
    # 1a. Inverse‑square‑root
    for n in n_choices:
        if time.time() >= deadline:
            break
        consider(generate_inverse_sqrt(n))
    
    # 1b. Square‑root differences (telescoping)
    for n in n_choices:
        if time.time() >= deadline:
            break
        consider(generate_sqrt_diff(n))
    
    # 1c. Geometric families
    for n in n_choices:
        if time.time() >= deadline:
            break
        for r in r_choices:
            if time.time() >= deadline:
                break
            consider(generate_geometric(n, r))
    
    # 1d. Power‑law with offset
    for n in n_choices:
        if time.time() >= deadline:
            break
        for p in p_choices:
            for off in offset_choices:
                if time.time() >= deadline:
                    break
                consider(generate_power_law(n, p, off))
    
    # 1e. Sine‑weighted inverse‑sqrt (adds a smooth envelope)
    for n in n_choices:
        if time.time() >= deadline:
            break
        consider(generate_sine_weighted_inverse_sqrt(n))

    # ------------------------------------------------------------------
    # 2. Randomised perturbations (Monte‑Carlo style) starting from the
    #    best sequence obtained above.
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform sequence of moderate length
        best_seq = [1.0] * 512
        best_score = evaluate(best_seq)

    seq = np.array(best_seq, dtype=float)

    # parameters for the local search
    max_iter = 20000
    iteration = 0
    while time.time() < deadline and iteration < max_iter:
        iteration += 1
        # pick two positions and move a small amount of mass
        i, j = np.random.randint(len(seq), size=2)
        if i == j:
            continue
        # magnitude is a small fraction of the source entry
        delta = np.random.rand() * 0.05 * max(seq[i], 1e-8)
        if delta > seq[i]:
            delta = seq[i]
        new_seq = seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        # enforce non‑negativity (already satisfied) and the hard 0‑1000 cap
        new_seq = np.clip(new_seq, 0.0, 1000.0)
        # quick check – if sum is too tiny skip
        if new_seq.sum() < 0.01:
            continue
        val = evaluate(new_seq.tolist())
        if val < best_score:
            best_score = val
            best_seq = new_seq.tolist()
            seq = new_seq
            print(f"Iter {iteration:05d} – improvement: {val:.6f}")
        # occasional random restart to escape flat regions
        if iteration % 2000 == 0:
            rand_n = np.random.choice(n_choices)
            rand_seq = np.random.exponential(scale=1.0, size=rand_n)
            # keep total mass roughly comparable to current best
            if rand_seq.sum() == 0:
                continue
            # normalise to have the same L1‑norm as current best (scale invariance)
            rand_seq *= (np.sum(seq) / rand_seq.sum())
            rand_seq = np.clip(rand_seq, 0.0, 1000.0)
            val = evaluate(rand_seq.tolist())
            if val < best_score:
                best_score = val
                best_seq = rand_seq.tolist()
                seq = rand_seq
                print(f"Random restart gave new best: {val:.6f}")

    print(f"Finished search – best value {best_score:.6f}")

def generate_inverse_sqrt(n: int):
    """a_i = 1 / sqrt(i+1)  (i = 0 … n‑1)."""
    idx = np.arange(1, n + 1, dtype=float)
    return (1.0 / np.sqrt(idx)).tolist()

    """a_i = sqrt(i+1) - sqrt(i);  sum = sqrt(n)."""
    i = np.arange(n + 1, dtype=float)
    return (np.sqrt(i[1:]) - np.sqrt(i[:-1])).tolist()

    """a_i = r**i   (0 < r < 1)."""
    return (r ** np.arange(n, dtype=float)).tolist()
