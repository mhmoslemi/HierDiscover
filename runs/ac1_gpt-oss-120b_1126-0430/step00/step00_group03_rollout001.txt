def solve_convolution_lp(f_seq, M):
    n = len(f_seq)
    c = -np.ones(n)
    A_ub = []
    b_ub = []
    for k in range(2*n - 1):
        row = np.zeros(n)
        # vectorized: valid indices j where 0 <= k-j < n => max(0, k-n+1) <= j <= min(k, n-1)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        # For j in range(j_start, j_end+1): row[j] = f_seq[k - j]
        row[j_start:j_end+1] = f_seq[k - np.arange(j_start, j_end+1)]
        A_ub.append(row)
        b_ub.append(M)

    bounds = [(0, None)] * n
    # Use high-level solver
    res = linprog(c, A_ub=np.array(A_ub), b_ub=np.array(b_ub), bounds=bounds, method='highs')
    if res.success:
        return res.x
    else:
        return None

    seq = seq * (target_sum / seq.sum())
    best_seq = seq.copy()

    # Evaluate using evaluate_sequence for reference

    n = len(f_seq)
    c = -np.ones(n)
    # build rows
    rows = []
    cols = []
    data = []
    b = []
    for k in range(2 * n - 1):
        # valid j: max(0, k-n+1) <= j <= min(k, n-1)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        # Corresponding i = k - j
        # row has entries f_seq[i] for each column j
        # loop j
        for j in range(j_start, j_end + 1):
            i = k - j
            rows.append(k)
            cols.append(j)
            data.append(f_seq[i])
        b.append(M)
    # Build sparse matrix
    A_ub = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    b_ub = np.array(b)
    # bounds non-negative
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    else:
        return None

"""
Propose a step‐function sequence with a low evaluation score.

The algorithm combines three ideas:
1.  A fast gradient descent on the maximal convolution coefficient.
    The gradient of the max‐convolution (at index k*) w.r.t. a_i is
        ∂max/∂a_i = 2 * a_{k*-i} .
    We move opposite to this direction while keeping the total sum fixed.
2.  Simulated–annealing style acceptance to escape shallow plateaus.
3.  An occasional exact linear‑programming move (the same LP that the
    previous state‑of‑the‑art used) to obtain a global improvement.
The sum of the coefficients is always normalised to sqrt(2·n) so that
the evaluation function reduces to the maximal convolution value.
The routine runs until the allotted time (≈1000 s) expires and returns
the best sequence found.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
import scipy.sparse as sp


def _normalise(seq: np.ndarray) -> np.ndarray:
    """Scale the non‑negative sequence so that its sum equals sqrt(2·n)."""
    n = len(seq)
    target = math.sqrt(2.0 * n)
    s = seq.sum()
    if s <= 0:
        # fall back to a uniform positive vector
        seq = np.ones(n, dtype=float)
        s = n
    return seq * (target / s)


def _conv_max_fft(seq: np.ndarray):
    """Return (max value, argmax index) of the linear convolution seq*seq using FFT."""
    n = len(seq)
    L = 1 << ((2 * n - 1).bit_length())        # next power of two
    fft_a = np.fft.fft(seq, n=L)
    conv = np.fft.ifft(fft_a * fft_a).real[: 2 * n - 1]
    idx = int(conv.argmax())
    return float(conv.max()), idx


def _solve_convolution_lp(f_seq, M):
    """
    Linear program:
        maximise   sum(b)
        subject to conv(f,b)_k <= M   for all k,
                   b_j >= 0.
    Returns the optimal b (as a NumPy array) or None if infeasible.
    """
    n = len(f_seq)
    # objective: maximise sum(b)  -> minimise -sum(b)
    c = -np.ones(n, dtype=float)

    # build the (2n-1) inequality rows, one for each convolution index
    rows = []
    cols = []
    data = []
    b_ub = np.full(2 * n - 1, M, dtype=float)

    for k in range(2 * n - 1):
        # valid columns j satisfy 0 <= k-j < n  <=>  k-n+1 <= j <= k
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start > j_end:
            continue
        # for each such j, the coefficient is f_{k-j}
        js = np.arange(j_start, j_end + 1, dtype=int)
        is_ = k - js
        rows.extend([k] * len(js))
        cols.extend(js.tolist())
        data.extend(f_seq[is_].tolist())

    A_ub = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    # bounds enforce b_j >= 0
    bounds = [(0, None)] * n

    # use the HiGHS interior point solver (fast for large sparse problems)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method="highs")
    if res.success:
        return res.x
    return None

    The function respects the global time budget and returns the best
    sequence discovered (as a list of floats).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2  # small safety margin

    # ------------------------------------------------------------

    # ------------------------------------------------------------
    try:

    if init_seq is not None and np.random.rand() < 0.7:
        seq = np.array(init_seq, dtype=float)
        if len(seq) != n:
            seq = seq[:n]
    else:
        # start from a random exponential distribution (breaks symmetry)
        seq = np.random.exponential(scale=1.0, size=n).astype(float)

    seq = _normalise(seq)

    # best record
    best_seq = seq.copy()
    # we evaluate the objective using the official function for the final score
    best_score = evaluate_sequence(best_seq.tolist())

    # ------------------------------------------------------------
    # 2. main optimisation loop
    # ------------------------------------------------------------
    target_sum = math.sqrt(2.0 * n)
    step = target_sum / n * 0.5          # initial step size (≈ half the average entry)
    temp = 0.1                           # initial temperature for simulated annealing
    T_min = 1e-4
    decay = 0.995
    it = 0

    while time.time() < deadline:
        it += 1

        # ----- compute current max‑convolution and its location -----
        M_cur, k_star = _conv_max_fft(seq)

        # ----- gradient of the current max coefficient -----
        i = np.arange(n, dtype=int)
        j = k_star - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros(n, dtype=float)
        grad[mask] = seq[j[mask]]
        # If the gradient is flat (uniform case) we inject a random shake
        if np.allclose(grad, grad.mean()):
            # small random perturbation to break symmetry
            for _ in range(3):
                idx = np.random.randint(n)
                seq[idx] = max(0.0, seq[idx] + np.random.randn() * seq[idx] * 0.2)
            seq = _normalise(seq)
            continue

        # direction: opposite to gradient with zero sum (so sum stays unchanged)
        d = -grad + grad.mean()

        # ----- propose a new candidate -----
        cand = seq + step * d
        cand = np.maximum(cand, 0.0)               # keep non‑negative
        cand = _normalise(cand)

        # evaluate candidate (fast version, uses the fact that sum is fixed)
        M_cand, _ = _conv_max_fft(cand)
        obj_cand = 2.0 * n * M_cand / (target_sum ** 2)
        obj_cur = 2.0 * n * M_cur / (target_sum ** 2)

        # ----- accept / reject -----
        improved = obj_cand < best_score - 1e-8
        if improved or np.random.rand() < math.exp(-(obj_cand - obj_cur) / max(temp, 1e-12)):
            seq = cand
            if improved:
                best_seq = cand.copy()
                best_score = obj_cand
                # slowly increase step size after a successful move
                step *= 1.05
        else:
            # on rejection shrink the step size
            step *= 0.70

        # ----- temperature cooling -----
        temp = max(T_min, temp * decay)

        # ----- occasional LP “global” move (every 50 iterations) -----
        if it % 50 == 0:
            # solve LP to maximise total mass while respecting the current bound M_cur
            lp_solution = _solve_convolution_lp(seq.tolist(), M_cur)
            if lp_solution is not None:
                g = np.array(lp_solution, dtype=float)
                g = _normalise(g)
                M_g, _ = _conv_max_fft(g)
                obj_g = 2.0 * n * M_g / (target_sum ** 2)
                if obj_g + 1e-10 < best_score:
                    seq = g
                    best_seq = g.copy()
                    best_score = obj_g
                    print(f"[{it:05d}] LP improvement: score={best_score:.6f}")

        # ----- cheap progress report -----
        if it % 200 == 0:
            print(f"[{it:05d}] best={best_score:.6f}  step={step:.5f}  temp={temp:.5f}")

        # early stopping if we already beat the target (safety)
        if best_score < 1.55:
            # we keep searching a bit more, but we could break here if desired
            pass

    # ------------------------------------------------------------
    # 3. finalisation
    # ------------------------------------------------------------
    result = [float(x) for x in best_seq.tolist()]
    final_score = evaluate_sequence(result)
    print(f"Search finished after {it} iterations. Best proxy score={best_score:.6f}, "
          f"final evaluate_sequence={final_score:.6f}")
    return result
