def propose_candidate(seed=0, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5 # allow margin
    best_sequence = None
    best_score = float('inf')
    # pick some n values to explore: maybe start with n = 500, 1000, 1500, 2000
    ns = [256, 512, 1024, 2048]
    # we can loop over them sequentially; in each we perform SA for limited time
    for n in ns:
        # random initial p: Dirichlet (maybe uniform)
        # we can start from uniform plus small random noise.
        p = np.full(n, 1.0/n)
        # optionally we can also start from a 2-point distribution embed:
        # place mass at 0 and at n-1 with ratio 2:1
        # but not necessary; we can also switch after some attempts
        current_score = evaluate_sequence(p.tolist())
        if current_score < best_score:
            best_score = current_score
            best_sequence = p.tolist()
        # SA parameters
        T0 = 0.1  # initial temperature (scale of objective difference)
        # note: objective values are ~2 to 1.7 so diff small; we scale accordingly
        # We'll use temperature schedule that decays to near 0 over time.
        # Determine number of iterations: maybe 5000 per n or until time.
        iter = 0
        while time.time() < deadline:
            iter += 1
            # dynamic temperature decay based on progress or iteration
            # compute fraction of time left per sequence
            frac = (iter) / 20000.0
            T = max(1e-6, T0 * (1 - frac))  # simple linear decay
            # propose move: pick two random indices i,j and shift epsilon
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i == j:
                continue
            eps = np.random.exponential(scale=0.01)  # typical magnitude
            # ensure p_i >= eps
            if p[i] < eps:
                eps = p[i] * 0.5
            # propose new vector p2
            p2 = p.copy()
            p2[i] -= eps
            p2[j] += eps
            # ensure no negative (p2[i]>=0) automatically by eps condition
            # evaluate
            score2 = evaluate_sequence(p2.tolist())
            # accept if better or prob
            delta = score2 - current_score
            if delta < 0 or np.random.rand() < np.exp(-delta/T):
                p = p2
                current_score = score2
                if score2 < best_score:
                    best_score = score2
                    best_sequence = p2.tolist()
                    print(f"New best {best_score:.6f} at n={n}, iter={iter}")
            # optionally break after fixed iteration count per n or if no improvement for long time
            if iter >= 50000:
                break
        # early exit if time close to deadline
        if time.time() >= deadline:
            break
    # final check
    print(f"Finished search; best_score={best_score:.6f}, length={len(best_sequence)}")
    return best_sequence

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 1.0
    best_seq = None
    best_score = float('inf')
    # try multiple lengths
    lengths = [256, 512, 1024, 2048]   # maybe up to 2000
    # prior known good sequence height_sequence_1 may be accessible; evaluate and set as baseline.
    if 'height_sequence_1' in globals():
        try:
            score = evaluate_sequence(height_sequence_1)
            best_seq = list(height_sequence_1)
            best_score = score
            print(f"Start from known seq, score {score}")
        except Exception:
            pass

    # Random Dirichlet sampling
    while time.time() < deadline:
        # pick random length
        n = np.random.choice(lengths)
        # pick concentration parameter
        alpha = np.random.uniform(0.1, 5.0)  # smaller alpha = spikier
        # sample Dirichlet
        x = np.random.gamma(alpha, 1.0, size=n)
        seq = x / x.sum()
        # scale to avoid too small sum? sum is 1 already.
        # evaluate
        try:
            score = evaluate_sequence(seq.tolist())
        except Exception as e:
            continue
        if score < best_score:
            best_score = score
            best_seq = seq.tolist()
            print(f"New best {best_score:.6f} n={n} alpha={alpha:.3f}")
        # optionally refine using local search: small random modifications
        # use hill climbing for a few steps
        for _ in range(5):
            # pick random indices i,j
            i,j = np.random.randint(n, size=2)
            if i==j:
                continue
            # move epsilon
            eps = np.random.rand() * seq[i] * 0.5
            new_seq = seq.copy()
            new_seq[i] -= eps
            new_seq[j] += eps
            # ensure nonneg
            if new_seq[i] < 0:
                continue
            new_score = evaluate_sequence(new_seq.tolist())
            if new_score < score:
                seq = new_seq
                score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_seq = new_seq.tolist()
                    print(f"Improved with hillclimb {best_score:.6f}")
    print(f"Best sequence score {best_score}")
    return best_seq

"""
Propose a high‑quality step‑function sequence.

The evaluation function is scale‑invariant:
    evaluate_sequence(a) = 2·n·max_k (p * p)_k
where p_i = a_i / sum(a) is a probability distribution.
The goal is to minimise the maximal self‑convolution probability.
A classical family of distributions that does this well is the power‑law
p_i ∝ (i+1)^{-β}.  For β≈0.6 the theoretical bound
    2·(1−β)²·B(1−β,1−β) ≈ 1.35
( B is the beta‑function ), which is well below the target 1.503.

The algorithm therefore scans a small set of β values, evaluates the
corresponding sequences (length ≈2000), keeps the best one and finally
performs a quick hill‑climb (mass transfer between two positions) to
potentially improve it further.  The whole routine respects the 1000 s
budget and works with any globally supplied `height_sequence_1`.
"""

import time

def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):

    np.random.seed(seed)

    deadline = start + budget_s - 0.1   # a tiny safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1.  Start from the previously found construction, if it exists.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            score = evaluate_sequence(height_sequence_1)
            best_seq = list(height_sequence_1)
            best_score = score
            print(f"[init] Using height_sequence_1, score = {score:.6f}")

            pass

    # 2.  Scan a family of power‑law sequences p_i = (i+1)^(-beta).
    # ------------------------------------------------------------------
    # Choose a length that is comfortably large but still cheap to convolve.

                continue
            # Transfer a small fraction of mass from i to j.
            eps = seq[i] * np.random.rand() * 0.05   # ≤5% of the source entry
            if eps <= 0:
                continue
            new_seq = seq.copy()
            new_seq[i] -= eps
            new_seq[j] += eps
            # Guard against negativity (should never happen with the formula above)
            if new_seq[i] < 0:
                continue
            try:
                new_score = evaluate_sequence(new_seq.tolist())
            except Exception:
                continue
            if new_score < current_score:
                seq = new_seq
                current_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_seq = new_seq.tolist()
                    print(f"[hill] improvement -> {new_score:.6f}")

    print(f"[final] best score = {best_score:.6f}, length = {len(best_seq)}")
    return best_seq
