def propose_candidate(...):
    import numpy as np, time, random
    deadline = time.time() + budget_s - 0.5 # small reserve
    # We'll define a list of candidate lengths (n) to try.
    best_seq = None
    best_score = float('inf')
    # Try for multiple n values
    # We'll loop while time left
    while time.time() < deadline:
        # Choose n randomly among a set, maybe weighted.
        n = np.random.choice([50, 100, 200, 400, 800])
        # Initialize sequence: maybe exponential shape with parameter epsilon = 2 / n ?
        epsilon = 2.0 / n
        r = 1 - epsilon
        a = np.full(n, 0.0)
        # compute geometric series truncated
        # Let's compute unnormalized a_i = epsilon * r**i
        a = epsilon * (r**np.arange(n))
        # Normalize to sum=1
        a = a / a.sum()
        # Compute convolution
        b = np.convolve(a, a)
        cur_obj = 2 * n * b.max() # sum=1 automatically
        # keep best
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = a.copy()
            print(f'Init n={n}, epsilon={epsilon:.5f}, obj={cur_obj:.6f}')
        # Run local search for this sequence
        # number of iterations limited by remaining time and maybe fixed budget
        iter_limit = 2000  # iteration per sequence
        for it in range(iter_limit):
            # exit if time near deadline
            if time.time() >= deadline:
                break
            # move: pick i,j
            i = np.random.randint(n)
            if a[i] < 1e-12:
                continue
            # choose j different
            j = np.random.randint(n-1)
            if j >= i:
                j += 1
            # step fraction
            frac = np.random.uniform(0.01, 0.2)
            delta = a[i] * frac
            # compute delta_b slices
            # Use numpy slices
            delta_b = np.zeros_like(b)
            start_i = i
            end_i = i + n
            delta_b[start_i:end_i] -= 2 * delta * a
            start_j = j
            end_j = j + n
            delta_b[start_j:end_j] += 2 * delta * a
            b_new = b + delta_b
            new_max = b_new.max()
            new_obj = 2 * n * new_max
            # accept if better
            if new_obj < cur_obj:
                # apply update
                a[i] -= delta
                a[j] += delta
                b = b_new
                cur_obj = new_obj
                if new_obj < best_score:
                    best_score = new_obj
                    best_seq = a.copy()
                    print(f'New best obj={new_obj:.6f} at n={n}, iter={it}')
            else:
                # occasionally accept uphill move with small prob to escape
                # Use temperature decreasing
                # compute temperature schedule
                # simple: accept if random < 0.001
                if np.random.rand() < 0.001:
                    a[i] -= delta
                    a[j] += delta
                    b = b_new
                    cur_obj = new_obj
        # If time left after this, loop to next random n
    # After loop, return best_seq as list
    return best_seq.tolist()

propose_candidate
-----------------
Search for a step‑function (non‑negative sequence) that gives a low value of

    2 * n * max(convolve(a, a)) / (sum(a)**2)

The search is deliberately simple:
* Generate candidates that follow a decreasing power‑law
  a_i ∝ (n – i)^p   (i = 0 … n‑1)  with p ∈ {0.5, 1.0, 2.0}.
* Normalise each candidate to unit sum (the evaluation function is
  scale‑invariant, so any normalisation works).
* Evaluate the candidate with the supplied ``evaluate_sequence``.
* Keep the best candidate seen so far.
* After the systematic sweep, perform a short random‑local‑search
  around the best candidate to squeeze a little more improvement.

The decreasing linear profile (p = 1) corresponds to a triangular density.
In the continuous limit its self‑convolution attains a maximum of 2/3,
so the theoretical limit of the objective is 2·(2/3)=4/3≈1.333.
Even for modest lengths (n ≈ 10 … 200) this construction already beats the

The function obeys the required signature and respects the time budget

import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights (the step function) that minimises
    the provided evaluation function.

    Parameters
    ----------

    budget_s : int, optional

    **kwargs : dict

    Returns

    list[float]
        The best sequence found within the time budget.
    """
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # 1. systematic sweep over lengths and simple power‑law shapes
    # ------------------------------------------------------------------
    deadline = time.time() + budget_s - 0.5   # leave a small safety margin

    best_score = float('inf')
    best_seq   = None

    # lengths to try – a geometric progression gives good coverage while
    # staying fast.  We stop early if time runs out.
    lengths = [10, 20, 30, 40, 50, 60, 80, 100, 150, 200, 300, 400, 500, 700, 1000]
    exponents = [0.5, 1.0, 2.0]   # p‑values for (n‑i)^p

    for n in lengths:

        for p in exponents:

            idx = np.arange(n, 0, -1, dtype=float)   # n, n‑1, …, 1

    if best_seq is None:               # should never happen, but be safe
