def hill_climb(a, deadline):
   n = len(a)
   best_a = a.copy()
   best_score = evaluate_sequence(a)
   step = 0.1  # relative amount (percentage)
   while time.time() < deadline:
       # select two indices i (source), j (dest)
       i = np.random.randint(n)
       j = np.random.randint(n)
       if i == j: continue
       # move delta = step * a[i] (or step * random value)
       delta = step * a[i] * np.random.rand()
       if delta <= 1e-12: continue
       new_a = a.copy()
       new_a[i] -= delta
       new_a[j] += delta
       
       # ensure nonnegative
       if new_a[i] < 0: continue  # skip
       
       # Evaluate
       score = evaluate_sequence(new_a.tolist())
       if score < best_score:
           best_score = score
           best_a = new_a.copy()
           a = new_a
           # optionally increase step
           step = min(step * 1.05, 0.5)
       else:
           # maybe accept with small probability (annealing)
           # degrade step if no improvement
           step = max(step * 0.99, 1e-4)
   
   return best_a, best_score

        b = np.convolve(a, a)
        M = b.max()

        max_idxs = np.where(b >= M - 1e-12)[0]

        grad = np.zeros_like(a)

            # for valid i such that 0 <= i < n and 0 <= k-i < n

            # gradient (subgradient of max)

Function project_onto_simplex:

def simplex_projection(v):
    # v array length n
    # Returns projection onto simplex sum=1, v_i >=0.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    # Find rho
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
propose_candidate
-----------------

Search for a non‑negative step‑function (a list of heights) that makes the
evaluation function

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

as small as possible.

The algorithm is a simple projected (sub‑)gradient descent on the
ℓ∞‑norm of the convolution.  The gradient is taken only on the “active”
shifts – the indices where the convolution attains its current maximum.
Because the problem is highly symmetric, starting from a random
distribution is essential; otherwise the method would stay at the
uniform sequence (which is a stationary point).

For each random restart we

1. generate a random probability vector `a` (sum = 1, a_i ≥ 0);
2. repeat a few thousand gradient steps
   - compute the full discrete convolution `b = conv(a, a)`;
   - build a mask for the active shifts (`b == max(b)`);
   - the sub‑gradient of `max(b)` w.r.t. `a` is
       g_i = 2 * Σ_{k active} a_{k-i} .
     This can be obtained by convolving the mask with the reversed `a`.
   - take a small step opposite to the gradient and project back onto the
     probability simplex (the feasible set);
   - keep the best sequence seen so far.
3. stop when the time budget is exhausted or a score ≤ 1.503 is found.

The projection onto the simplex is performed with the classic O(n log n)
algorithm (sorting, thresholding).  All operations are NumPy‑based and
run in well under a second even for a few thousand coefficients, allowing
many restarts within the 1000 s budget.

The returned sequence is a list of Python floats ready to be passed to
`evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# helper: projection onto the probability simplex (sum == 1, all ≥ 0)
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x | x_i ≥ 0, Σ x_i = 1}."""
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find the rho index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

        cur_score = evaluate_sequence(a.tolist())

        max_iters = 3000
        step0 = 0.5 / np.sqrt(n)          # initial step size, scales with n
        for it in range(max_iters):
            if time.time() >= deadline:
                break

            # convolution (full length 2*n-1)
            b = np.convolve(a, a)
            M = b.max()
            # mask of the active constraints (all k where b_k is maximal)
            mask = (b >= M - 1e-12).astype(np.float64)

            # sub‑gradient:  g = 2 * (mask * a_rev)  (extracted from full conv)
            grad = 2.0 * np.convolve(mask, a[::-1])[n - 1:n - 1 + n]

            step = step0 / (1.0 + it * 1e-3)
