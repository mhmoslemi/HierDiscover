"""
Uniform step sequence optimizer.

The evaluation function is

    Q(f) = 2 * n * max(conv(f,f)) / (sum(f)**2)

For any non‑negative vector `f` the Cauchy–Schwarz inequality gives

    max(conv(f,f)) = sum_i f_i**2  ≥ (sum_i f_i)**2 / n

hence Q(f) ≥ 2 with equality if and only if all entries are equal.
Therefore the globally optimal solution is any uniform (constant) sequence.
The function below simply returns such a sequence.  The value of `n` is
chosen modestly (1000) to keep memory usage low, but any length ≥1 works.
The constant `c = 1.0` stays within the allowed [0,1000] range.
"""

import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Return a uniform sequence that attains the theoretical minimum
    value 2.0 of the evaluation function.

    Parameters
    ----------
    seed : int
        Random seed (unused, kept for API compatibility).
    budget_s : float
        Time budget in seconds (unused, we finish instantly).
    kwargs : dict
        Additional arguments ignored.

    Returns
    -------
    list[float]
        A list of identical non‑negative heights.
    """
    # Parameters of the uniform construction
    n = 1000                # length of the step function
    c = 1.0                 # constant height, safely inside [0,1000]

    # Build the sequence
    sequence = [c] * n

    # Quick sanity check – should give the optimal score 2.0
    try:
        score = evaluate_sequence(sequence)
        print(f"[Uniform optimizer] n={n}, constant={c}, evaluation={score}")
    except Exception as e:
        # In the unlikely event evaluate_sequence is unavailable or fails,
        # we still return the constructed sequence.
        print("[Uniform optimizer] evaluation failed:", e)

    # The budget is not needed; we already have the optimal solution.
    return sequence
