def compute_eval(seq):
    seq = np.array(seq, dtype=np.float64)
    # Clip
    seq = np.clip(seq, 0, 1000)
    n = len(seq)
    # FFT size
    size = 1 << ((2*n - 1).bit_length())
    fft = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft * fft, n=size)
    max_b = np.max(conv[:2*n - 1])
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return np.inf
    return float(2 * n * max_b / (sum_a ** 2))

import numpy as np
import time

def compute_score(seq):
    """Compute evaluate_sequence using fast FFT to avoid overhead."""
    # Clip to allowed range (max 1000, min 0)
    seq = np.clip(np.asarray(seq, dtype=np.float64), 0.0, 1000.0)
    n = len(seq)
    if n == 0:
        return float('inf')
    # Use next power of two for FFT convolution
    size = 1 << ((2 * n - 1).bit_length())
    fft_seq = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft_seq * fft_seq, n=size)
    max_b = np.max(conv[: 2 * n - 1])
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """Generate a decreasing geometric sequence that provably beats the trivial
    uniform bound (value 2.0).  The sequence is a_i = r**i with r = exp(-1/n)
    where n is the chosen length.  Optionally a short hill‑climbing phase
    (random mass transfers that keep the sum constant) is performed to tighten
    the bound further."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # leave a small margin

    # Choose a length – larger than the current construction but still cheap
    n = 2000                     # can be tuned; must be >0
    r = np.exp(-1.0 / n)         # optimal decay for the geometric family
    seq = [r ** i for i in range(n)]

    best_seq = seq
    best_score = compute_score(seq)
    print(f"Initial geometric sequence length {n}, score {best_score:.6f}")

    # Simple hill climbing: transfer a tiny amount of mass from a larger entry
    # to a smaller one while preserving the total sum.
    # Only accept moves that decrease the score.
    cur_seq = np.array(seq, dtype=np.float64)
    cur_score = best_score

    # Parameters for the stochastic search
    eps_factor = 0.02            # relative magnitude of a transfer
    attempts_without_improve = 0
    max_no_improve = 1000

    while time.time() < deadline and attempts_without_improve < max_no_improve:
        # pick two distinct indices
        i, j = np.random.choice(n, size=2, replace=False)
        # ensure we move mass from larger to smaller to keep positivity likely
        if cur_seq[i] < cur_seq[j]:
            i, j = j, i
        if cur_seq[i] <= 0:
            continue
        # amount to move (relative to source value)
        epsilon = cur_seq[i] * eps_factor * np.random.rand()
        if epsilon <= 0:
            continue
        new_seq = cur_seq.copy()
        new_seq[i] -= epsilon
        new_seq[j] += epsilon
        # clip to the allowed interval (the algorithm never pushes outside)
        # evaluate
        new_score = compute_score(new_seq)
        if new_score < cur_score:
            cur_seq = new_seq
            cur_score = new_score
            attempts_without_improve = 0
            # keep global best
            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.tolist()
                print(f"New best score {best_score:.6f} after move")
        else:
            attempts_without_improve += 1

    print(f"Finished search. Best score = {best_score:.6f}")
    return best_seq

"""
Fast geometric construction with a tiny hill‑climbing refinement.

The goal is to beat the trivial uniform bound (value 2.0) of the
evaluation function

    E(a) = 2·n·max(a*a) / (sum(a))² .

For a decreasing geometric profile a_i = r**i with
r = exp(-1/n) one can show analytically that the bound is minimised
inside the whole geometric family and attains approximately

    E ≈ 1.84  (< 2.0).

The routine starts from this analytically optimal shape, then performs a
very cheap stochastic hill‑climbing phase that transfers a small amount
of mass from a larger entry to a smaller one while keeping the total sum
constant.  Only moves that strictly improve the score are kept, which
quickly drives the sequence to a local optimum that is usually at least
as good as the pure geometric construction.

All heavy work (convolution and scoring) is done with an FFT based
implementation, which is O(n log n) and easily fits the 1000 s budget even
for sequences of a few thousand entries.

The function `propose_candidate` follows the required signature and
returns the best sequence found before the time limit expires.
"""

import time
import numpy as np

def _fast_score(seq):
    """
    Compute the same value that `evaluate_sequence` returns, but using FFT.
    The sequence is first clipped to the allowed range [0,1000] (the construction
    never leaves this range, but the clipping guarantees identical behaviour).
    """
    a = np.clip(np.asarray(seq, dtype=np.float64), 0.0, 1000.0)
    n = a.size
    if n == 0:
        return float('inf')
    # size for a convolution via FFT (next power of two ≥ 2n‑1)
    size = 1 << ((2 * n - 1).bit_length())
    fft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft * fft, n=size)
    max_b = np.max(conv[: 2 * n - 1])
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Generate a step‑function (list of non‑negative heights) with a low
    evaluation score.

    The algorithm:
    1. Build a geometric decreasing sequence a_i = r**i with
       r = exp(-1/n) where n is the chosen length (default 2000).
       This analytically gives a score ≈ 1.84, already better than the
       uniform bound of 2.0.
    2. Run a lightweight hill‑climbing loop:
       - pick two distinct indices i (source) and j (target);
       - move a small random amount ε from a_i to a_j (preserving total sum);
       - accept the move only if the score improves.
       The loop stops after a time budget is exhausted or after a number of
       consecutive non‑improving attempts.
    3. Return the best sequence encountered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # keep a small safety margin

    # ------------------------------------------------------------------
    # 1. Geometric starting point
    # ------------------------------------------------------------------
    n = 2000                         # length of the construction
    r = np.exp(-1.0 / n)             # optimal decay for the geometric family
    seq = [r ** i for i in range(n)]

    best_seq = seq
    best_score = _fast_score(seq)
    print(f"[init] length={n}, geometric score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Very cheap stochastic hill‑climbing
    # ------------------------------------------------------------------
    cur = np.array(seq, dtype=np.float64)
    cur_score = best_score

    # parameters controlling the random moves
    eps_factor = 0.02                # up to 2 % of the source entry per move
    no_improve = 0
    max_no_improve = 1500           # stop after this many failed attempts

    while time.time() < deadline and no_improve < max_no_improve:
        # pick two distinct positions
        i, j = np.random.choice(n, size=2, replace=False)
        # always move from the larger to the smaller entry (helps keep positivity)
        if cur[i] < cur[j]:
            i, j = j, i
        if cur[i] <= 0.0:
            continue
        # random transfer amount
        eps = cur[i] * eps_factor * np.random.rand()
        if eps <= 0.0:
            continue

        cand = cur.copy()
        cand[i] -= eps
        cand[j] += eps

        # evaluate candidate
        cand_score = _fast_score(cand)

        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            no_improve = 0
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.tolist()
                print(f"[improve] new best score={best_score:.6f}")
        else:
            no_improve += 1

    print(f"[done] best score achieved: {best_score:.6f}")
    return best_seq
