    """Return a monotone power‑law ramp of length *n* with exponent *p*,

        seq = _make_power_ramp(n, p)

        # Evaluate the candidate (the function is already imported in the runtime).
        try:
            score = evaluate_sequence(seq)
        except Exception:                     # safety net – should never happen
            score = float('inf')

        print(f"p={p:<2d}, length={n}, score={score:.6f}")

        if score < best_score:
            best_score = score
            best_seq = seq

        # Abort if we are about to exceed the allotted budget.
        if time.time() - start > budget_s - 0.2:
            print("Time budget almost exhausted – stopping search.")
            break

    # If, for any reason, no sequence was accepted, fall back to a simple constant.
    if best_seq is None:
        best_seq = [1.0] * 100
        best_score = evaluate_sequence(best_seq)
        print("Fallback sequence used, score =", best_score)

    print("Best score achieved:", best_score)
    return best_seq
