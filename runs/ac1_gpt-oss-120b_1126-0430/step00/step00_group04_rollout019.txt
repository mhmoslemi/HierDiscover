def gradient_Lp(a, p):
    n = len(a)
    b = np.convolve(a, a)
    w = b**(p-1)  # elementwise
    g = np.zeros_like(a)
    # compute g_i = 2 * sum_{j=0}^{n-1} a_j * w_{i + j}

        # w index range i..i+n-1 inclusive (must be within 0..2n-2)

        start = i
        end = i + n  # exclusive
        # slice of w of length n
        # g_i = 2 * sum_j a_j * w[i + j]
        g[i] = 2 * np.dot(a, w[start:end])
    return g

"""
Propose a candidate step‐function sequence.

The algorithm explores several parametric families of non‑negative vectors,
normalises them to unit sum (the evaluation function is scale invariant) and
tries a few random permutations of each vector – the ordering strongly
influences the self‑convolution.  The best candidate found so far is then
refined by a simple hill‑climbing on the permutation: random swaps are kept
iff they improve the evaluation.  The whole search respects the time budget
(≤ 1000 s) and returns the best sequence discovered.

No external files are accessed; the function `evaluate_sequence` is assumed to
be already imported.
"""

import time
import numpy as np


def _normalise(v: np.ndarray) -> np.ndarray:
    """Scale a non‑negative vector to have sum = 1 (avoids the sum≈0 guard)."""
    s = v.sum()
    if s == 0.0:
        # fall back to uniform – never happens with our generators
        return np.ones_like(v) / len(v)
    return v / s


def _linear_seq(n: int, lam: float) -> np.ndarray:
    """Decreasing linear shape: 1 – lam·i/(n‑1)."""
    if n == 1:
        return np.ones(1)
    i = np.arange(n, dtype=float)
    seq = 1.0 - lam * i / (n - 1)
    seq = np.maximum(seq, 0.0)
    return _normalise(seq)


def _power_seq(n: int, theta: float) -> np.ndarray:
    """Power‑law decreasing shape: (i+1)^{-theta}."""
    i = np.arange(1, n + 1, dtype=float)
    seq = i ** (-theta)
    return _normalise(seq)


def _exp_seq(n: int, r: float) -> np.ndarray:
    """Geometric decreasing shape: r**i."""
    i = np.arange(n, dtype=float)
    seq = r ** i
    return _normalise(seq)


def _sine_seq(n: int) -> np.ndarray:
    """Half‑sine shape, positive on (0, π)."""
    i = np.arange(1, n + 1, dtype=float)
    seq = np.sin(np.pi * i / (n + 1))
    return _normalise(seq)

                if s < best_score:
                    best_score, best_seq = s, perm.copy()
                    print(f"[n={n}] exp r={r:.2f}  score={best_score:.6f}")

        # ------------------------------------------------------------------
        # Half‑sine shape
        # ------------------------------------------------------------------
        seq = _sine_seq(n)
        for _ in range(5):
            perm = np.random.permutation(seq)
            s = evaluate_sequence(perm.tolist())
            if s < best_score:
                best_score, best_seq = s, perm.copy()
                print(f"[n={n}] sine  score={best_score:.6f}")

        # ------------------------------------------------------------------
        # Random initial vectors (purely stochastic)
        # ------------------------------------------------------------------
        for _ in range(20):
            seq = np.random.rand(n)
            seq = _normalise(seq)
            s = evaluate_sequence(seq.tolist())
            if s < best_score:
                best_score, best_seq = s, seq.copy()
                print(f"[n={n}] random init  score={best_score:.6f}")

    # ----------------------------------------------------------------------
    # 2.  Local optimisation: hill‑climbing on the permutation of the best vector.
    # ----------------------------------------------------------------------
    if best_seq is not None:
        cur_seq = best_seq.copy()
        cur_score = best_score
        n = len(cur_seq)

        # Simple hill‑climbing: keep swaps that improve the score.
        while time.time() < deadline:
            i, j = np.random.randint(n), np.random.randint(n)
            if i == j:
                continue

            # perform swap
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
            new_score = evaluate_sequence(cur_seq.tolist())

            if new_score < cur_score:
                cur_score = new_score
                if new_score < best_score:
                    best_score, best_seq = new_score, cur_seq.copy()
                    print(f"[n={n}] swap improvement  score={best_score:.6f}")
            else:
                # revert the move
                cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

            # occasional random reset to avoid getting stuck
            if np.random.rand() < 0.001:
                cur_seq = best_seq.copy()
                cur_score = best_score

    # ----------------------------------------------------------------------
    # 3.  Fallback – should never happen, but keep a safe default.
    # ----------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.ones(10) / 10
        best_score = evaluate_sequence(best_seq.tolist())
        print("Fallback to trivial uniform sequence.")

    print(f"Final best score: {best_score:.6f}  (length={len(best_seq)})")
    return best_seq.tolist()
