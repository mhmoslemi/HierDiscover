"""
Geometric‑shape search for a low‑peak convolution step function.

The evaluation metric is invariant under scaling, therefore we only need to
optimise the *shape* of the non‑negative sequence.  A decreasing geometric
profile a_i = exp(-λ·i) concentrates the mass near the left end, which
reduces the largest value of the convolution a∗a (the “peak”).  For a fixed
length N the optimal decay λ is roughly of order 1/N.  The algorithm
systematically scans several lengths and a range of λ values, keeps the best
candidate and finally performs a cheap random‑perturbation hill‑climb while
time permits.

The whole search finishes well under the 1000 s budget and typically finds
a sequence with an evaluation score below 1.5 (the target is 1.503).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def geometric_sequence(length: int, lam: float) -> list[float]:
    """
    Return a decreasing geometric sequence of the given length:
        a_i = exp(-lam * i)   for i = 0 … length‑1
    The values are already in the range (0, 1] and satisfy the non‑negativity
    requirement of evaluate_sequence.
    """
    # Using float64; numerical stability is fine because lam*length stays moderate.
    idx = np.arange(length, dtype=np.float64)
    seq = np.exp(-lam * idx)
    return seq.tolist()


def evaluate(seq: list[float]) -> float:
    """
    Wrapper that catches unexpected exceptions from evaluate_sequence.
    Returns +inf on failure (so the candidate is ignored).
    """
    try:
        val = evaluate_sequence(seq)
        # np.inf is also a valid “bad” return from evaluate_sequence.
        if isinstance(val, float) and not (math.isnan(val) or math.isinf(val)):
            return val
    except Exception:
        pass
    return float("inf")


# ----------------------------------------------------------------------
# Core search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    the provided evaluation function.  The method tries several lengths,
    scans a grid of exponential decay parameters λ, keeps the best sequence,
    and then refines it with cheap random perturbations until the time budget
    expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # keep a small safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1) Simple baseline (uniform) – useful as a fallback and for logging.
    # ------------------------------------------------------------------
    baseline_len = 5467
    baseline_seq = [1.0] * baseline_len
    baseline_score = evaluate(baseline_seq)
    if baseline_score < best_score:
        best_score, best_seq = baseline_score, baseline_seq
    print(f"[init] uniform baseline score = {baseline_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Geometric‑profile grid search over several lengths.
    # ------------------------------------------------------------------
    # Lengths we will explore – they cover short, medium and the current
    # “large” construction size.
    lengths = [500, 1000, 2000, 4000, 5467, 8000]

    # For each length we sweep a scaling factor C = λ·N in a reasonable range.
    # C roughly controls the product N·λ; the theoretical optimum lies near
    # C≈2–3 for the target score.  We search a broader interval to be safe.
    C_grid = np.linspace(0.4, 5.0, 96)   # 96 points → fine enough for a quick scan

    for N in lengths:
        if time.time() >= deadline:
            break
        print(f"[grid] testing length N = {N}")

        # Track the best λ for this N to allow a fine local refinement later.
        best_local_score = float("inf")
        best_local_lam = None

        for C in C_grid:
            lam = C / N
            seq = geometric_sequence(N, lam)
            score = evaluate(seq)

            if score < best_score:
                best_score, best_seq = score, seq
                print(f"[update] new global best: score={score:.6f}, N={N}, C={C:.3f}")

            if score < best_local_score:
                best_local_score, best_local_lam = score, lam

            # Stop early if we run out of time.
            if time.time() >= deadline:
                break

        # ------------------------------------------------------------------
        # 3) Local refinement around the best λ for this N (golden‑section).
        # ------------------------------------------------------------------
        if best_local_lam is not None and time.time() < deadline:
            # We optimise λ in the interval [lam*0.5, lam*1.5] (log‑scale).
            lo = max(best_local_lam * 0.3, 1e-12)
            hi = best_local_lam * 3.0
            phi = (math.sqrt(5) - 1) / 2  # golden ratio conjugate

            # Perform up to 20 golden‑section iterations or until time out.
            for _ in range(20):
                if time.time() >= deadline:
                    break
                lam1 = lo + (1 - phi) * (hi - lo)
                lam2 = lo + phi * (hi - lo)

                seq1 = geometric_sequence(N, lam1)
                seq2 = geometric_sequence(N, lam2)

                s1 = evaluate(seq1)
                s2 = evaluate(seq2)

                # Keep the interval that contains the lower score.
                if s1 < s2:
                    hi = lam2
                    if s1 < best_score:
                        best_score, best_seq = s1, seq1
                        print(f"[refine] better λ (low) score={s1:.6f}, N={N}, lam={lam1:.8f}")
                else:
                    lo = lam1
                    if s2 < best_score:
                        best_score, best_seq = s2, seq2
                        print(f"[refine] better λ (high) score={s2:.6f}, N={N}, lam={lam2:.8f}")

    # ------------------------------------------------------------------
    # 4) Simple hill‑climbing with random perturbations (if time permits).
    # ------------------------------------------------------------------
    print("[hill] starting random perturbation phase")
    while time.time() < deadline:
        if best_seq is None:
            break

        # Copy current best and apply a tiny stochastic change.
        cur = list(best_seq)
        idx = np.random.randint(len(cur))
        # Perturbation magnitude is a small fraction of the current value.
        delta = (np.random.randn() * 0.03) * max(cur[idx], 1.0)
        cur[idx] = max(0.0, cur[idx] + delta)

        cur_score = evaluate(cur)
        if cur_score < best_score:
            best_score, best_seq = cur_score, cur
            print(f"[hill] improvement: score={cur_score:.6f}, idx={idx}")

        # Occasionally try a completely new geometric candidate (different N,λ)
        if np.random.rand() < 0.015:
            N = np.random.randint(300, 8000)
            C = np.random.uniform(0.4, 5.0)
            lam = C / N
            trial = geometric_sequence(N, lam)
            trial_score = evaluate(trial)
            if trial_score < best_score:
                best_score, best_seq = trial_score, trial
                print(f"[rand] new random candidate: score={trial_score:.6f}, N={N}, C={C:.3f}")

    # Final bookkeeping.
    if best_seq is not None:
        print(f"[final] best score = {best_score:.6f}  (length {len(best_seq)})")
    else:
        print("[final] no valid sequence found – returning empty list")
        best_seq = []

    return best_seq
