def project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Simple gradient‑based optimizer for the step‑function bound.

The algorithm works on a normalized vector `a` (sum(a)=1, a_i ≥ 0).
It repeatedly:
  1. Computes the full auto‑convolution `c = a * a`.
  2. Forms a smooth approximation of the max (soft‑max with parameter α).
  3. Uses the corresponding weights to obtain the gradient
     of the smoothed objective via a single FFT‑based correlation.
  4. Takes a projected gradient step onto the probability simplex.
  5. Adapts the step size and the soft‑max temperature α.

Multiple random restarts with different lengths are performed until the
budget is exhausted.  The best sequence seen (as measured by the exact
`evaluate_sequence` formula) is returned.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Helper utilities (top level, no closures)
# ------------------------------------------------------------

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        # all non‑positive -> put all mass on the largest component
        w = np.zeros_like(v)
        w[np.argmax(v)] = 1.0
        return w
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]          # last index where condition holds
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x):
    """Return the next power of two greater than or equal to x."""
    return 1 << (int(x) - 1).bit_length()


def _score_from_vector(a):
    """Exact evaluation of the objective (the same as evaluate_sequence)."""
    n = len(a)
    # auto‑convolution (full linear convolution)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 1e-12:   # protective – should never happen
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


# ------------------------------------------------------------
# Main entry point required by the competition framework
# ------------------------------------------------------------

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    `2 * n * max(conv(a,a)) / (sum(a)**2)`.
    Returns the best sequence found before the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2  # a little safety margin

    best_seq = None
    best_score = float('inf')

    # If a previous construction is available, try it as a starting point.
    if 'height_sequence_1' in globals():
        try:
            init = np.array([float(x) for x in height_sequence_1], dtype=float)
            if init.size > 0:
                init = np.maximum(init, 0.0)
                init = init / init.sum()
                best_seq = init.tolist()
                best_score = _score_from_vector(init)
                print(f"[init] Using height_sequence_1, score = {best_score:.6f}")
        except Exception:
            best_seq = None
            best_score = float('inf')

    # Main outer loop – new random restart whenever time permits
    restart_counter = 0
    while time.time() < deadline:
        # ------------------------------------------------------------------
        # 1. Choose a length for this restart (moderate size, not too big)
        # ------------------------------------------------------------------
        n = np.random.randint(200, 2001)               # 200 … 2000 elements
        # ------------------------------------------------------------------
        # 2. Initialise a random probability vector of length n
        # ------------------------------------------------------------------
        if restart_counter == 0 and best_seq is not None and len(best_seq) == n:
            a = np.array(best_seq, dtype=float)        # try to continue from best
        else:
            a = np.random.rand(n)
            a = a / a.sum()

        # Parameters for the inner optimisation
        step = 0.5                 # base step size for projected gradient
        alpha = 5.0                # soft‑max temperature (controls smoothness)
        max_inner = 800           # safety cap on inner iterations

        # ------------------------------------------------------------------
        # 3. Inner optimisation loop (gradient descent on smoothed max)
        # ------------------------------------------------------------------
        inner_it = 0
        while time.time() < deadline and inner_it < max_inner:
            # ----- auto‑convolution via FFT (single forward FFT of a) -----
            size = _next_pow2(3 * n - 2)               # enough for both conv & corr
            Fa = np.fft.rfft(a, n=size)

            # full linear convolution a*a (autocorrelation)
            conv_full = np.fft.irfft(Fa * Fa, n=size)
            conv = conv_full[: 2 * n - 1]               # linear (full) result

            # ----- exact score (to keep track of the best) -----
            cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.tolist()
                print(f"[{restart_counter}:{inner_it}] New best: {best_score:.6f}  (n={n})")
                if best_score <= 1.503:               # target reached – stop early
                    return best_seq

            # ----- compute soft‑max weights over the convolution entries -----
            max_c = conv.max()
            # avoid overflow/underflow by shifting
            w_exp = np.exp(alpha * (conv - max_c))
            w = w_exp / w_exp.sum()                    # length 2n‑1, sums to 1

            # ----- gradient of the smoothed objective (FFT correlation) -----
            Fw = np.fft.rfft(w, n=size)
            corr_full = np.fft.irfft(np.conj(Fa) * Fw, n=size)
            # cross‑correlation a ⋆ w : first n entries give Σ_i a_i w_{i+j}
            grad = 2.0 * corr_full[:n]

            # ----- backtracking line‑search on the projected step -----
            improved = False
            factor = 1.0
            while factor > 1e-4:
                y = a - step * factor * grad
                y = _project_simplex(y)                # stay on simplex
                # evaluate new candidate (exact objective)
                new_score = _score_from_vector(y)
                if new_score <= cur_score:
                    a = y
                    improved = True
                    step = min(step * 1.05, 1.0)       # gently increase step if successful
                    break
                factor *= 0.5

            if not improved:
                step *= 0.5
                if step < 1e-6:
                    break  # unable to make progress – exit inner loop

            # ----- slowly tighten the soft‑max (increase α) -----
            alpha = min(alpha * 1.02, 200.0)

            inner_it += 1

        restart_counter += 1

        # occasional tiny random perturbation to escape possible plateaus
        if np.random.rand() < 0.1:
            pert = np.random.randn(len(a)) * 0.001
            a = np.maximum(a + pert, 0.0)
            a = _project_simplex(a)

    # ----------------------------------------------------------------------
    # If nothing was found (should not happen), fall back to a trivial vector
    # ----------------------------------------------------------------------
    if best_seq is None:
        best_seq = [0.5, 0.5]   # two‑element uniform sequence (score = 2)
    return best_seq
