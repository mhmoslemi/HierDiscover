"""
Simple stochastic optimizer for the convolution‑based evaluation function.

The optimizer works in three stages:

1. **Smart initialization** – a power‑law sequence a_i ∝ (i+1)⁻¹ᐟ² (the
   empirically good shape whose evaluation tends to π/2 ≈ 1.57) is generated
   for several lengths.

2. **Local search** – repeated random “mass‑transfer’’ moves between two
   indices are tried.  Moves that improve the score are always accepted,
   otherwise they are accepted with a tiny probability (simulated annealing).

3. **LP‑guided refinement** – for modest lengths (n ≤ 200) the linear
   programming direction from the reference implementation is used
   occasionally.  The LP maximises the total mass while keeping the
   convolution ≤ the current maximum, and a convex combination of the
   current sequence and the LP solution lowers the worst‑case convolution.

The procedure runs for the allotted time (default 1000 s) and returns the
best sequence found.  All numbers are non‑negative; the evaluation function
is scale‑invariant, so the code keeps the sum of the sequence equal to 1.
"""

import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------


def generate_power_law_seq(n: int, p: float = 0.5) -> np.ndarray:
    """
    Return a normalized sequence a_i ∝ (i+1)^(-p) of length n.
    The sum is forced to 1 (scaling does not affect the objective).
    """
    idx = np.arange(1, n + 1, dtype=np.float64)          # 1 … n
    seq = idx ** (-p)
    seq_sum = seq.sum()
    if seq_sum == 0:
        seq = np.full(n, 1.0 / n)
    else:
        seq = seq / seq_sum
    return seq


def random_perturb(seq: np.ndarray, step: float) -> np.ndarray:
    """
    Move a fraction `step` of mass from a random source index to a random
    destination index (keeping the total sum unchanged and all entries ≥0).
    """
    n = len(seq)
    i, j = np.random.choice(n, 2, replace=False)
    if seq[i] <= 0:
        return seq
    delta = step * seq[i]
    if delta <= 0:
        return seq
    new_seq = seq.copy()
    new_seq[i] -= delta
    new_seq[j] += delta
    return new_seq


def solve_convolution_lp(f_seq: np.ndarray, rhs: float):
    """
    Solve the LP:
        maximise   Σ b_j
        subject to  conv(f_seq, b) ≤ rhs   (for every shift)
                    b_j ≥ 0
    Returns the optimal b (numpy array) or None if the LP fails.
    This routine is only called for modest n (≤200) to keep the LP size
    manageable.
    """
    n = len(f_seq)
    # objective: maximise sum(b)  → minimise -sum(b)
    c = -np.ones(n)

    # Build inequality matrix for each convolution shift (2n‑1 rows)
    rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # a_i contributes to shift k if 0 ≤ i < n and 0 ≤ k‑i < n
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        # for those i, the coefficient of b_{k-i} is f_seq[i]
        # we write the row in terms of b_j (j = k-i)
        for i in range(i_low, i_high + 1):
            j = k - i
            row[j] = f_seq[i]
        rows.append(row)

    A_ub = np.vstack([rows, -np.eye(n)])          # also b_j ≥ 0 → -b_j ≤ 0
    b_ub = np.hstack([np.full(len(rows), rhs), np.zeros(n)])

    # Solve – give a small time limit to avoid hanging
    res = optimize.linprog(
        c,
        A_ub=A_ub,
        b_ub=b_ub,
        method="highs",
        options={"time_limit": 5.0, "disp": False},
    )
    if res.success:
        return res.x
    return None


def get_good_direction_to_move_into(seq: np.ndarray) -> np.ndarray | None:
    """
    Return a better sequence using the LP direction described in the
    problem statement.  For n > 200 the LP is skipped and None is returned.
    """
    n = len(seq)
    total = seq.sum()
    if total < 1e-12:
        return None
    # Normalise to the scaling used by the reference algorithm
    norm_seq = seq * np.sqrt(2.0 * n) / total
    rhs = np.max(np.convolve(norm_seq, norm_seq))

    # LP only for small problems (keeps runtime reasonable)
    if n <= 200:
        g = solve_convolution_lp(norm_seq, rhs)
        if g is not None:
            g_sum = g.sum()
            if g_sum <= 0:
                return None
            norm_g = g * np.sqrt(2.0 * n) / g_sum
            t = 0.02                     # small mixing factor
            return (1 - t) * seq + t * norm_g
    return None


# ----------------------------------------------------------------------
# Main candidate generator
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑score height sequence.
    The function respects the given time budget (`budget_s` seconds) and
    returns a list of non‑negative floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0     # a small safety margin
    best_seq = None
    best_score = float("inf")

    # a few problem sizes – larger sizes give more potential improvement,
    # but are also slower; we mix several to increase chances.
    lengths = [200, 500, 1000, 2000, 4000]

    for n in lengths:
        if time.time() >= deadline:
            break

        # three different initialisations for each length
        initials = [
            generate_power_law_seq(n, 0.5),            # the analytic shape
            np.full(n, 1.0 / n),                      # plain uniform
            np.random.rand(n),                        # random start
        ]

        for init in initials:
            seq = init / init.sum()                    # normalise sum to 1
            curr_score = evaluate_sequence(seq.tolist())
            if curr_score < best_score:
                best_score = curr_score
                best_seq = seq.copy()

            # parameters governing the local search
            step = 0.08                                 # start with ~8 % moves
            no_improve = 0
            max_iter = int(2e5 / n)                     # roughly 100 k evaluations total
            it = 0

            while time.time() < deadline and it < max_iter:
                it += 1

                # Occasionally ask the LP for a promising direction
                if np.random.rand() < 0.25:
                    new_seq = get_good_direction_to_move_into(seq)
                    if new_seq is None:
                        new_seq = random_perturb(seq, step)
                else:
                    new_seq = random_perturb(seq, step)

                # evaluate the candidate
                try:
                    new_score = evaluate_sequence(new_seq.tolist())
                except Exception:
                    continue

                # Accept improvements always; occasional uphill moves help escape traps
                if new_score < curr_score or np.random.rand() < 0.001:
                    seq = new_seq
                    curr_score = new_score
                    if new_score < best_score:
                        best_score = new_score
                        best_seq = seq.copy()
                        no_improve = 0
                    else:
                        no_improve += 1
                else:
                    no_improve += 1

                # gradual cooling of the step size
                if no_improve > 3000:
                    step *= 0.5
                    no_improve = 0
                    if step < 1e-5:
                        break

    # Fallback – should never happen, but guarantees a valid return value
    if best_seq is None:
        best_seq = np.ones(1, dtype=np.float64)

    return best_seq.tolist()
