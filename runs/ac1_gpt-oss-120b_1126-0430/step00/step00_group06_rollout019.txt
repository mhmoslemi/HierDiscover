def get_convolution(seq):
    # compute linear convolution: length 2n-1
    n=len(seq)
    size = 2*n-1
    # compute via FFT:
    m = 1 << (size-1).bit_length()   # next power of 2
    fft_seq = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fft_seq * fft_seq, n=m)
    return conv[:size]

def propose_candidate(seed=123, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5
    n = 4587  # maybe length from height_sequence_1 if present; else choose.
    # Use length of height_sequence_1 (if exists)
    try: n = len(height_sequence_1)
    except NameError: n=5000
    # generate inverse sqrt sequence
    seq = 1.0 / np.sqrt(np.arange(1, n+1))
    seq = seq / seq.sum()
    best_seq = seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    # Subgradient descent function
    def subgrad_opt(x, max_iters, step0):
        x = x.copy()
        for it in range(max_iters):
            conv = np.convolve(x, x)
            max_val = conv.max()
            # indices close to max (active set)
            eps = max_val * 1e-12 + 1e-12
            active = np.where(conv >= max_val - eps)[0]
            # pick one active index randomly
            k = np.random.choice(active)
            nlen = len(x)
            i = np.arange(nlen)
            j = k - i
            mask = (j >= 0) & (j < nlen)
            grad = np.zeros_like(x)
            # compute gradient contributions
            grad[mask] = x[i[mask]] + x[j[mask]]
            # step size
            eta = step0 / np.sqrt(it + 1)
            y = x - eta * grad
            # project onto simplex
            y = project_simplex(y)
            # evaluate new
            score = evaluate_sequence(y.tolist())
            if score < best_score:
                nonlocal best_seq, best_score
                best_seq = y.copy()
                best_score = score
            # Accept if improvement else keep old?
            if score <= evaluate_sequence(x.tolist()):
                x = y
        return x
    # run from inverse sqrt start
    step0 = 0.5
    max_iters = 2000
    subgrad_opt(best_seq, max_iters, step0)
    # further random restarts
    while time.time() < deadline:
        # random init Dirichlet
        alpha = np.random.uniform(0.5, 2.0)  # concentration
        y = np.random.gamma(alpha, 1.0, size=n)
        y = y / y.sum()

"""
Propose a sequence of non‑negative heights that improves the evaluation score.
The construction is based on the analytical insight that a density proportional
to 1/√x (i.e. a_i ∝ 1/√(i+1)) yields an almost flat self‑convolution on the
interval (0,1).  After normalising the sequence to have sum = 1 the resulting
evaluation is roughly 1.57 for large n (much better than the uniform value 2).
A cheap local stochastic refinement (random perturbations with simplex
projection) is applied for a few seconds to try to push the score a little
further.

The code works for any length; by default it uses the length of the previously
found construction `height_sequence_1` if that variable exists, otherwise a
length of 5000.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures, top level as required)
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]               # sorted descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the uniform vector
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def fast_score(seq: np.ndarray) -> float:
    """Return the evaluation score 2*n*max(conv)/sum^2 (no clipping)."""
    s = seq.sum()
    if s < 0.01:                     # guard against degenerate vectors
        return np.inf
    conv = np.convolve(seq, seq)
    m = conv.max()
    n = seq.shape[0]
    return float(2.0 * n * m / (s * s))


def random_refine(seq: np.ndarray, steps: int, step_size: float) -> np.ndarray:
    """
    Perform a few cheap stochastic refinements:
    - add a small zero‑mean random perturbation,
    - project back onto the simplex,
    - keep the change only if it improves the score.
    """
    best = seq.copy()
    best_score = fast_score(best)

    n = seq.shape[0]
    rng = np.random.default_rng()

    for _ in range(steps):
        # random direction with zero mean (so the sum stays roughly constant)
        delta = rng.normal(scale=step_size, size=n)
        delta -= delta.mean()                # keep sum unchanged in expectation
        cand = best + delta
        # keep non‑negative, then project onto simplex (guarantees sum=1)
        cand = project_simplex(cand)
        sc = fast_score(cand)
        if sc < best_score:
            best = cand
            best_score = sc
    return best


# ----------------------------------------------------------------------
# Main candidate builder
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Build and return a sequence (list of floats) with a low evaluation score.
    The routine runs for at most `budget_s` seconds.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + max(0.0, budget_s - 5.0)   # keep a few seconds spare

    # ------------------------------------------------------------------
    # Choose the length (use the already known construction if possible)
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        n = 5000

    # --------------------------------------------------------------
    # 1. Deterministic inverse‑square‑root construction
    # --------------------------------------------------------------
    idx = np.arange(1, n + 1, dtype=np.float64)       # 1 … n
    seq = 1.0 / np.sqrt(idx)                         # a_i ∝ 1/√i
    seq /= seq.sum()                                 # normalise to sum = 1

    best_seq = seq.copy()
    best_score = fast_score(best_seq)
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2. Light stochastic refinement on the inverse‑sqrt start
    # --------------------------------------------------------------
    # Perform a few hundred cheap random moves; they are very fast.
    refine_steps = 800
    step_scale = 0.02                               # size of random perturbation
    best_seq = random_refine(best_seq, refine_steps, step_scale)
    best_score = fast_score(best_seq)
    print(f"[refine‑1] after random refinement score={best_score:.6f}")

    # --------------------------------------------------------------
    # 3. Sub‑gradient style descent to push the max down
    # --------------------------------------------------------------
    # We use the classic sub‑gradient for the max‑convolution:
    #   grad_i = a_{k-i} + a_i  (k = an index where the max is attained)
    #
    # The step size decays with the iteration number.
    # ------------------------------------------------------------------
    max_iter = 3000
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        conv = np.convolve(best_seq, best_seq)
        max_val = conv.max()
        # all indices that are (almost) maximal – use one at random
        cand_idxs = np.where(conv >= max_val - 1e-12)[0]
        k = np.random.choice(cand_idxs)

        i = np.arange(n, dtype=np.int32)
        j = k - i
        mask = (j >= 0) & (j < n)

        grad = np.zeros_like(best_seq)
        # gradient contribution for the active shift k
        grad[mask] = best_seq[i[mask]] + best_seq[j[mask]]

        # step size schedule (simple 1/√t decay)
        eta = 0.6 / np.sqrt(it + 1)

        cand = best_seq - eta * grad
        cand = project_simplex(cand)

        cand_score = fast_score(cand)

        # accept if we made progress; otherwise stay where we are
        if cand_score < best_score:
            best_seq = cand
            best_score = cand_score
            # occasional logging
            if it % 250 == 0:
                print(f"[sgd] iter={it:4d}, score={best_score:.6f}")

    print(f"[sgd] finished after {it+1} iterations, score={best_score:.6f}")

    # --------------------------------------------------------------
    # 4. A few random restarts with short refinement (optional)
    # --------------------------------------------------------------
    # If time still permits we try a handful of random Dirichlet starts.
    while time.time() < deadline:
        # random Dirichlet (Gamma) vector – concentration between 0.3 and 2
        conc = np.random.uniform(0.3, 2.0)
        cand = np.random.gamma(shape=conc, scale=1.0, size=n)
        cand /= cand.sum()

        # brief random refinement
        cand = random_refine(cand, steps=150, step_size=0.015)
        cand_score = fast_score(cand)

        if cand_score < best_score:
            best_seq = cand
            best_score = cand_score
            print(f"[restart] new best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Return the best sequence found (as a plain Python list)
    # --------------------------------------------------------------
    print(f"[final] best score achieved = {best_score:.6f}")
    return best_seq.tolist()
