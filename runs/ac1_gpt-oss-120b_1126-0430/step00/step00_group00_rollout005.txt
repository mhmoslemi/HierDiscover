    best_score = compute_score(a)

            j = np.random.choice(dest_range)

            j = np.random.randint(n)

import numpy as np
import time

def compute_max_convolution(a, fft_len):
    # compute self-convolution using FFT of length fft_len
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[:2*len(a)-1]
    return np.max(conv)

def evaluate_score(a, n):
    s = a.sum()
    if s < 0.01:
        return float('inf')
    max_b = compute_max_convolution(a, fft_len=1 << ((2*n-1).bit_length()))
    return 2 * n * max_b / (s*s)

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """Simple simulated‑annealing / greedy descent that flattens the
    auto‑convolution of a non‑negative step function.  Starts from a uniform
    distribution, repeatedly moves a small amount of mass from indices that
    contribute heavily to the current maximal convolution value towards random
    positions, accepting moves that improve the bound or with a Metropolis
    probability that anneals over the allocated time.  The best sequence found
    within the time budget is returned."""
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # reserve a little margin

    # length of the sequence – larger gives more degrees of freedom, but
    # too large slows the inner loop.  1000 works well within the time budget.
    n = 1000

    # initialise with a uniform distribution (sum = 1)
    a = np.full(n, 1.0 / n, dtype=np.float64)
    fft_len = 1 << ((2 * n - 1).bit_length())   # next power‑of‑two for FFT

    # current score (sum is 1, therefore score = 2 * n * max_b)
    cur_max = compute_max_convolution(a, fft_len)
    cur_score = 2.0 * n * cur_max
    best_a = a.copy()
    best_score = cur_score

    # parameters for the annealing schedule
    Tmax = 0.5            # initial temperature
    Tmin = 1e-4           # final temperature (effective floor)
    total = budget_s - 0.5

    iteration = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        # linear temperature decay
        T = Tmax * max(0.0, 1.0 - elapsed / total)
        if T < Tmin:
            T = Tmin

        # --- locate the index where the convolution is maximal ---
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[:2 * n - 1]
        max_idx = int(np.argmax(conv))
        max_val = float(conv[max_idx])
        # current score recomputed from conv (avoid extra FFT later)
        cur_score = 2.0 * n * max_val

        # --- choose a source index that participates in the maximal term ---
        i_min = max(0, max_idx - (n - 1))
        i_max = min(n - 1, max_idx)
        i_range = np.arange(i_min, i_max + 1, dtype=np.int64)
        j_range = max_idx - i_range               # partner indices
        # mask entries where both a[i] and a[j] are positive
        mask = (a[i_range] > 0) & (a[j_range] > 0)
        if not np.any(mask):
            # fall back to a completely random move
            src = rng.integers(n)
            dst = rng.integers(n)
            while dst == src:
                dst = rng.integers(n)
            eps = min(a[src], T * a[src])
            if eps <= 0:
                continue
            a_new = a.copy()
            a_new[src] -= eps
            a_new[dst] += eps
        else:
            # choose a source proportionally to its contribution a[i]*a[j]
            contrib = a[i_range[mask]] * a[j_range[mask]]
            prob = contrib / contrib.sum()
            src_idx = rng.choice(i_range[mask], p=prob)
            src = int(src_idx)
            # pick a destination that is different from the source
            dst = rng.integers(n)
            while dst == src:
                dst = rng.integers(n)

            # amount of mass to move – a fraction of the source value
            eps = min(a[src], T * a[src])
            if eps <= 0:
                continue
            a_new = a.copy()
            a_new[src] -= eps
            a_new[dst] += eps

        # compute the new score
        new_max = compute_max_convolution(a_new, fft_len)
        new_score = 2.0 * n * new_max

        # Metropolis acceptance
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start:.1f}s] new best score {best_score:.6f}")

        iteration += 1

    # final verification with the provided evaluate_sequence routine
    final_seq = best_a.tolist()
    try:
        official_score = evaluate_sequence(final_seq)
        print(f"official score = {official_score}")
    except Exception as e:
        print("Error calling evaluate_sequence:", e)
    return final_seq

import numpy as np
import time

def _max_self_convolution(a, fft_len):
    """Return max value of a * a (auto‑convolution) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[:2 * len(a) - 1]
    return float(conv.max())

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Simulated‑annealing / greedy descent that reshapes a non‑negative step
    function so that its auto‑convolution becomes as flat as possible.
    The algorithm starts from a uniform vector, repeatedly moves a small
    amount of mass from indices that participate heavily in the current
    maximal convolution value towards random locations, and keeps the best
    sequence encountered within the allotted time.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    # Length of the candidate sequence (adjustable; 1000 is a good compromise)
    n = 1000

    # Uniform initialization, sum == 1
    a = np.full(n, 1.0 / n, dtype=np.float64)

    # FFT length – next power of two that accommodates the full convolution
    fft_len = 1 << ((2 * n - 1).bit_length())

    # Initial score (sum is 1, so score = 2 * n * max_b)
    cur_max = _max_self_convolution(a, fft_len)
    cur_score = 2.0 * n * cur_max
    best_a = a.copy()
    best_score = cur_score

    # Annealing schedule parameters
    Tmax = 0.5    # initial temperature
    Tmin = 1e-4   # near‑zero temperature at the end
    total = budget_s - 0.5

    iteration = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        # Linear decay of temperature
        T = Tmax * max(0.0, 1.0 - elapsed / total)
        if T < Tmin:
            T = Tmin

        # Compute full self‑convolution (needed to locate the peak)
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[:2 * n - 1]
        max_idx = int(np.argmax(conv))
        max_val = float(conv[max_idx])
        cur_score = 2.0 * n * max_val

        # Determine indices that actually contribute to this maximal entry
        i_low = max(0, max_idx - (n - 1))
        i_high = min(n - 1, max_idx)
        i_candidates = np.arange(i_low, i_high + 1, dtype=np.int64)
        j_candidates = max_idx - i_candidates
        mask = (a[i_candidates] > 0) & (a[j_candidates] > 0)

        if np.any(mask):
            # Choose a source index weighted by its contribution a[i] * a[j]
            contrib = a[i_candidates[mask]] * a[j_candidates[mask]]
            prob = contrib / contrib.sum()
            src = int(rng.choice(i_candidates[mask], p=prob))
        else:
            # No contributing pair (should be rare); pick a random source
            src = rng.integers(n)

        # Destination index – any index different from the source
        dst = rng.integers(n)
        while dst == src:
            dst = rng.integers(n)

        # Transfer a fraction of the mass from src to dst
        eps = min(a[src], T * a[src])
        if eps <= 1e-12:
            continue
        a_new = a.copy()
        a_new[src] -= eps
        a_new[dst] += eps

        # Evaluate the new candidate
        new_max = _max_self_convolution(a_new, fft_len)
        new_score = 2.0 * n * new_max

        # Metropolis acceptance
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start:.1f}s] New best score {best_score:.6f}")

        iteration += 1

    # Return the best sequence discovered; evaluate_sequence will be called by the grader.
    return best_a.tolist()

import time
import numpy as np

def _max_self_convolution(a: np.ndarray, fft_len: int) -> float:
    """
    Compute max_{k} (a * a)[k] using a real FFT.
    The convolution length is 2*len(a)-1; we keep only that part.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * len(a) - 1]
    return float(conv.max())


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Simulated‑annealing / greedy descent that flattens the auto‑convolution
    of a non‑negative step function.

    The algorithm starts from a uniform distribution (sum == 1) and
    repeatedly transfers a small amount of mass from indices that
    participate heavily in the current maximal convolution value toward
    positions near the ends of the sequence.  Moves are accepted if they
    improve the bound or with a Metropolis probability that anneals
    over the allotted time.  The best sequence found before the deadline
    is returned.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # tiny safety margin

    # --------------------------------------------------------------------
    # Settings (feel free to adjust n – 1000 is a good trade‑off)
    # --------------------------------------------------------------------
    n = 1000                                   # length of the step function
    a = np.full(n, 1.0 / n, dtype=np.float64)   # uniform start, sum == 1

    # FFT length – next power of two that can hold the full convolution
    fft_len = 1 << ((2 * n - 1).bit_length())

    # current score (since sum(a)=1, score = 2 * n * max_b)
    cur_max = _max_self_convolution(a, fft_len)
    cur_score = 2.0 * n * cur_max
    best_a = a.copy()
    best_score = cur_score

    # annealing schedule
    Tmax = 0.5          # initial temperature (controls step size)
    Tmin = 1e-4         # final temperature (practically zero)
    total_time = budget_s - 0.5

    # indices that we consider “edges” – we like to push mass there
    edge_len = n // 10
    edge_idxs = np.concatenate([np.arange(edge_len), np.arange(n - edge_len, n)])

    iteration = 0
    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1) temperature (linear decay)
        # --------------------------------------------------------------
        elapsed = time.time() - start
        T = Tmax * max(0.0, 1.0 - elapsed / total_time)
        if T < Tmin:
            T = Tmin

        # --------------------------------------------------------------
        # 2) locate the current peak of the auto‑convolution
        # --------------------------------------------------------------
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_idx = int(np.argmax(conv))
        max_val = float(conv[max_idx])
        cur_score = 2.0 * n * max_val

        # --------------------------------------------------------------
        # 3) choose a *source* index that actually contributes to that peak
        # --------------------------------------------------------------
        i_low = max(0, max_idx - (n - 1))
        i_high = min(n - 1, max_idx)
        i_candidates = np.arange(i_low, i_high + 1, dtype=np.int64)
        j_candidates = max_idx - i_candidates
        mask = (a[i_candidates] > 0) & (a[j_candidates] > 0)

        if np.any(mask):
            # weight sources by the size of their contribution a[i] * a[j]
            contrib = a[i_candidates[mask]] * a[j_candidates[mask]]
            prob = contrib / contrib.sum()
            src = int(rng.choice(i_candidates[mask], p=prob))
        else:
            # fallback – pick a random non‑zero entry
            src = rng.integers(n)
            while a[src] <= 0:
                src = rng.integers(n)

        # --------------------------------------------------------------
        # 4) pick a *destination* index (prefer an edge)
        # --------------------------------------------------------------
        if rng.random() < 0.85:
            dst = int(rng.choice(edge_idxs))
            if dst == src:
                dst = rng.integers(n)
                while dst == src:
                    dst = rng.integers(n)
        else:
            dst = rng.integers(n)
            while dst == src:
                dst = rng.integers(n)

        # --------------------------------------------------------------
        # 5) amount of mass to move (a fraction of the source)
        # --------------------------------------------------------------
        eps = min(a[src], T * a[src])          # move at most a fraction T of src
        if eps <= 1e-12:
            continue

        a_new = a.copy()
        a_new[src] -= eps
        a_new[dst] += eps                      # sum stays equal to 1

        # --------------------------------------------------------------
        # 6) evaluate the new candidate
        # --------------------------------------------------------------
        new_max = _max_self_convolution(a_new, fft_len)
        new_score = 2.0 * n * new_max

        # --------------------------------------------------------------
        # 7) Metropolis acceptance
        # --------------------------------------------------------------
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start:.1f}s] new best score {best_score:.6f}")

        iteration += 1

    # --------------------------------------------------------------------
    # Return the best sequence (as a plain Python list)
    # --------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        # sanity check – official evaluation (the grader will call this anyway)
        official = evaluate_sequence(final_seq)   # evaluate_sequence is provided externally
        print(f"official score = {official}")
    except Exception as exc:   # pragma: no cover
        print("Could not call evaluate_sequence:", exc)

    return final_seq
