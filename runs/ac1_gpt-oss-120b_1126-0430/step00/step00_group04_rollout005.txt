def simplex_projection(v):
    # v: array
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _simplex_projection(v):
    """Project a vector onto the probability simplex {x>=0, sum=1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        # all elements zero? return uniform
        w = np.ones_like(v) / n
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
        w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function (non‑negative list of heights) that minimizes
evaluate_sequence().  The algorithm works on the simplex (sum of the
coefficients = 1) – the score is invariant to scaling – and tries to
flatten the self‑convolution a*a.  It uses projected gradient descent on
the variance of the convolution (which encourages a nearly uniform
distribution) together with a simple simulated‑annealing acceptance rule
and occasional random jitter to escape shallow minima.

The main steps are:
1. Initialise a random probability vector.
2. Repeatedly:
   * compute the convolution c = a*a and the variance‑gradient,
   * take a gradient step,
   * project back onto the probability simplex,
   * accept the new point if it improves the maximal convolution value,
     otherwise accept with a temperature‑dependent probability,
   * occasionally add a small random perturbation if no improvement is
     observed for many iterations.
3. Keep the best point seen so far and return it when the allotted time
   (budget_s) expires.
"""

import time
import math
import numpy as np


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x >= 0, sum(x) = 1}."""
    # Algorithm from Duchi et al. (2008)
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # sorted in descending order
    cssv = np.cumsum(u) - 1.0                # cumulative sum minus 1
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # all entries are non‑positive; return the uniform distribution
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that gives a small
    evaluate_sequence() score.  The method is a projected gradient descent
    on the variance of the self‑convolution, combined with a simple
    simulated‑annealing accept rule and occasional random jitter.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 1.0   # keep a safety margin

    # ------------------------------------------------------------------
    # Choose a problem size.
    # If a previously discovered sequence exists, use its length as an
    # upper bound but work with a moderate size that fits comfortably in the
    # time budget.
    # ------------------------------------------------------------------
    try:
        base_seq = list(height_sequence_1)   # pylint: disable=undefined-variable
        base_len = len(base_seq)
        # Limit the working size to a few thousand elements for speed.
        n = min(max(base_len, 1500), 5000)
        # If the base sequence is longer we simply down‑sample it uniformly.
        if base_len > n:
            idx = np.linspace(0, base_len - 1, n).astype(int)
            a = np.array([base_seq[i] for i in idx], dtype=float)
        else:
            a = np.array(base_seq, dtype=float)
    except Exception:
        # No previous sequence available – start from a fresh random vector.
        n = 2000
        a = rng.random(n)

    # Normalise to a probability vector (sum = 1) – the score does not depend on scale.
    a = _simplex_projection(a)

    # ------------------------------------------------------------------
    # Variables that keep track of the best solution found.
    # ------------------------------------------------------------------
    best_a = a.copy()
    conv = np.convolve(a, a)
    best_max = conv.max()
    best_score = 2.0 * n * best_max   # evaluate_sequence would return this (sum=1)
    best_iter = 0

    # ------------------------------------------------------------------
    # Parameters for the optimisation loop.
    # ------------------------------------------------------------------
    iteration = 0
    no_improve_counter = 0
    # learning rate schedule – decays slowly with the number of iterations
    base_lr = 0.8
    # temperature for probabilistic acceptance (simple annealing)
    base_temp = 0.04

    # ------------------------------------------------------------------
    # Main optimisation loop.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # --- current convolution and score ---------------------------------
        conv = np.convolve(a, a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        # --- keep the best -------------------------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            best_max = cur_max
            best_iter = iteration
            no_improve_counter = 0
            print(
                f"Iter {iteration:6d} – new best score {best_score:.6f} "
                f"(max conv {best_max:.6e})"
            )
        else:
            no_improve_counter += 1

        # --- compute gradient of the variance of the convolution -----------
        # variance objective:  J(a) = sum_k ( (a*a)[k] - avg )^2
        avg = 1.0 / (2 * n - 1)          # because sum(a*a) = 1 when sum(a)=1
        d = conv - avg
        # correlation a ⋆ d  (equivalent to convolution with reversed d)
        corr = np.convolve(a, d[::-1])
        # slice the part that corresponds to the gradient entries
        grad = 2.0 * corr[2 * n - 2: 2 * n - 2 + n]

        # --- gradient step with projection ---------------------------------
        lr = base_lr * (0.99 ** (iteration // 5000))
        a_new = a - lr * grad
        a_new = _simplex_projection(a_new)

        # --- evaluate the trial point ---------------------------------------
        conv_new = np.convolve(a_new, a_new)
        max_new = conv_new.max()
        score_new = 2.0 * n * max_new

        # --- simulated‑annealing acceptance ---------------------------------
        # always accept if the score improves; otherwise accept with a
        # probability that slowly vanishes as the temperature decays.
        temp = base_temp * (0.995 ** (iteration / 1000.0))
        accept = False
        if score_new < cur_score:
            accept = True
        else:
            prob = math.exp(-(score_new - cur_score) / (temp + 1e-14))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_new
            cur_max = max_new
            cur_score = score_new

        # --- occasional random jitter to escape plateaus --------------------
        if no_improve_counter >= 8000:
            # add a small Gaussian perturbation and re‑project
            jitter = 0.01 * rng.standard_normal(n)
            a = a + jitter
            a = np.clip(a, 0.0, None)
            a = _simplex_projection(a)
            no_improve_counter = 0
            # recompute the current convolution after the jump
            conv = np.convolve(a, a)
            cur_max = conv.max()
            cur_score = 2.0 * n * cur_max

        iteration += 1

    # ------------------------------------------------------------------
    # Final sanity check using the provided evaluate_sequence() function.
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"Finished – best evaluation_sequence value: {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Error while evaluating the final sequence:", exc)

    return final_seq
