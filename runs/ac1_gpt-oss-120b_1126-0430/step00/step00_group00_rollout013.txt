def evaluate_sequence(sequence):
    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)

"""
propose_candidate
-----------------

Search for a non‑negative step‑function (a list of heights) that minimizes

    2 * n * max(conv(f,f)) / (sum(f)**2)

where `conv` is the full discrete convolution.
The optimizer works as follows:

* A few deterministic “geometric” families of sequences (a_i = A·r^i) are
  evaluated – they already give scores around 1.8, better than the uniform
  bound 2.0.
* Several random restarts are performed with a simple projected sub‑gradient
  method.  At each iteration the index `k` where the convolution is maximal
  is identified; the sub‑gradient of the max‑convolution w.r.t. the
  coefficients is `g_i = 2·a_{k-i}` (zero when the index is out of range).
  An exponential update `a_i ← a_i·exp(−η·g_i)` keeps the vector non‑negative.
  After each step the coefficients are projected onto the simplex
  (non‑negative, sum = 1).  The step size η decays as 1/√t.
* A small random jitter is injected from time to time to escape plateaus.
* The best sequence seen before the time budget expires is returned.

All operations are vectorised and use FFT‑based convolution, so the method
comfortably fits into the 1000 s budget even for sequences of length ≈2000.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _full_convolve(a: np.ndarray) -> np.ndarray:
    """
    Return the full convolution a*a using an FFT of length a power of two.
    The result length is 2*len(a)-1.
    """
    n = a.shape[0]
    # size is next power of two ≥ 2*n-1 for fast FFT
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Orthogonal projection of v onto the probability simplex {x≥0, Σx=1}.
    Returns a new array of the same shape.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u > (cssv - 1) / (np.arange(1, len(v) + 1)))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w


def _score_sequence(a: np.ndarray) -> float:
    """
    Compute the evaluation score for a (non‑negative, sum may be any positive number).
    This mirrors evaluate_sequence but avoids the internal clamping for speed.
    """
    if a.size == 0:
        return np.inf
    n = a.shape[0]
    conv = _full_convolve(a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a ** 2))


# ----------------------------------------------------------------------
# Core optimisation routine (projected sub‑gradient)
# ----------------------------------------------------------------------


def _subgradient_optimize(
    n: int,
    init_seq: np.ndarray,
    max_iters: int,
    step0: float,
    deadline: float,
    verbose: bool = False,
):
    """
    Perform projected sub‑gradient descent on  f → max(conv(f,f)).
    Returns (best_sequence, best_score) seen during the run.
    """
    a = np.maximum(init_seq, 0.0)
    if a.sum() == 0:
        a = np.ones(n) / n
    else:
        a = a / a.sum()

    best_a = a.copy()
    best_score = _score_sequence(a)
    if verbose:
        print(f"  start score {best_score:.6f}")

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # full convolution and location of the current maximum
        conv = _full_convolve(a)
        k = int(conv.argmax())
        max_val = conv[k]

        # current score
        cur_score = 2.0 * n * max_val / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            if verbose:
                print(f"  iter {it:5d}: new best {best_score:.6f}")

        # sub‑gradient of the max‑convolution (only the active index k)
        idx = k - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # step size schedule
        eta = step0 / np.sqrt(it)

        # exponential update keeps positivity; clip exponent for stability
        update = np.exp(np.clip(-eta * grad, -50.0, 0.0))
        a = a * update

        # occasional jitter to shake the system (every 500 iterations)
        if it % 500 == 0:
            jitter = np.random.randn(n) * 0.001
            a = a + jitter
            a = np.clip(a, 0.0, None)

        # project back onto the simplex (sum = 1)
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights (floats) with a low evaluation score.

    The routine:
        1. tries a few geometric families (a_i ∝ r^i) for quick good candidates,
        2. runs several random restarts of a projected sub‑gradient optimiser,
        3. keeps the best sequence seen before the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # keep a small margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 1) deterministic geometric families (very cheap)
    # ------------------------------------------------------------------
    # Choose a moderate length; larger lengths give more freedom.
    n = 2048  # power of two – convenient for FFT
    geom_cs = [0.8, 1.0, 1.2, 1.5, 2.0]  # try several decay rates
    for c in geom_cs:
        if time.time() >= deadline:
            break
        r = 1.0 - c / n
        # avoid numerical issues at the extreme ends
        r_pow = r ** np.arange(n, dtype=np.float64)
        denom = 1.0 - r_pow[-1]  # 1 - r^n
        A = (c / n) / denom
        seq = A * r_pow
        scr = _score_sequence(seq)
        print(f"[geom] c={c:.2f} → score={scr:.6f}")
        if scr < best_score:
            best_score = scr
            best_seq = seq.copy()

    # ------------------------------------------------------------------
    # 2) stochastic sub‑gradient optimisation with several restarts
    # ------------------------------------------------------------------
    restarts = 6
    max_iters = 20000
    step0 = 1.0
    for r_id in range(restarts):
        if time.time() >= deadline:
            break
        # initialise: either perturb the current best or start from scratch
        if best_seq is not None and np.random.rand() < 0.6:
            init = best_seq + np.random.randn(n) * 0.01
            init = np.clip(init, 0.0, None)
        else:
            init = np.random.rand(n)
        # run optimizer
        opt_seq, opt_score = _subgradient_optimize(
            n=n,
            init_seq=init,
            max_iters=max_iters,
            step0=step0,
            deadline=deadline,
            verbose=True,
        )
        print(f"[restart {r_id}] best score this run: {opt_score:.6f}")
        if opt_score < best_score:
            best_score = opt_score
            best_seq = opt_seq.copy()
            print(f"*** global improvement: {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) final sanity check – fall back to a uniform sequence if needed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.ones(n) / n
        best_score = _score_sequence(best_seq)

    # The evaluator does its own clamping; we just return the raw floats.
    print(f"Final best score: {best_score:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
