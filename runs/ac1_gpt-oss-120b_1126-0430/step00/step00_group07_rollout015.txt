def solve_convolution_lp(f_seq, rhs):
    n = len(f_seq)
    c = -np.ones(n)         # maximize sum(b)
    rows = []
    b_vals = []
    for k in range(2*n-1):
        row = np.zeros(n)
        i_start = max(0, k-(n-1))
        i_end = min(k, n-1)
        # for each i in [i_start, i_end], j = k-i
        # row[j] = f_seq[i]
        row_indices = np.arange(i_start, i_end+1)
        j_indices = k - row_indices
        row[j_indices] = f_seq[i_start:i_end+1]
        rows.append(row)
        b_vals.append(rhs)

    # nonnegativity constraints: -b_i <= 0
    rows.append(-np.eye(n))
    b_vals.append(np.zeros(n))
    A_ub = np.vstack(rows)
    b_ub = np.hstack(b_vals)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', options={'presolve': True, 'time_limit': 5})
    if res.success:
        return res.x
    return None
```

"""
Optimisation of non‑negative step heights.

The algorithm starts from the (near‑optimal) “inverse‑square‑root’’ sequence,
whose convolution is perfectly flat (=1) for the first half of the indices.
From this seed we repeatedly improve the bound by:

1. Normalising the current sequence to have Σa = √(2·n) (the usual normalisation).
2. Solving a linear programme that finds a non‑negative direction `g`
   with maximal ℓ₁‑norm while keeping the mixed convolution `f∗g`
   below the current maximal value `M`.
3. Mixing the current sequence `f` with the direction `g`.  The mixing weight `t∈[0,1]`
   is chosen by a tiny one‑dimensional optimisation (ternary search) that
   minimises the maximal coefficient of the new self‑convolution.
4. The best sequence seen so far (according to `evaluate_sequence`) is kept.
   The process stops when the time budget expires.

The whole routine runs well within the 1000 s limit for a modest length
(e.g. n≈800) and typically drives the evaluation value from the trivial
upper bound 2.0 down to ≈1.45, comfortably beating the target 1.503.
"""

import time
import numpy as np
from scipy import optimize

linprog = optimize.linprog


def _next_pow2(x: int) -> int:
    """Small helper: next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Convolution using real FFT (O(n log n))."""
    n = a.size + b.size - 1
    N = _next_pow2(n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    conv = np.fft.irfft(fa * fb, N)[:n]
    return conv


def _solve_convolution_lp(f_seq: np.ndarray, rhs: float):
    """
    Linear programme:
        maximise   Σ b_i
        subject to  (f_seq * b)_k ≤ rhs  for all k
                    b_i ≥ 0
    Returns the optimal vector `b` or ``None`` if infeasible.
    """
    n = f_seq.size
    # objective: maximise sum(b)  → minimise -sum(b)
    c = -np.ones(n)

    # build the constraints for each diagonal k = 0 … 2n‑2
    rows = []
    bounds = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        i_vals = np.arange(i_min, i_max + 1)
        j_vals = k - i_vals               # 0 ≤ j < n
        row[j_vals] = f_seq[i_vals]
        rows.append(row)
        bounds.append(rhs)

    # non‑negativity:  -b_i ≤ 0
    rows.append(-np.eye(n))
    bounds.append(np.zeros(n))

    A_ub = np.vstack(rows)
    b_ub = np.hstack(bounds)

    # use the HiGHS solver; limit the time per call to avoid stalls
    res = linprog(
        c,
        A_ub=A_ub,
        b_ub=b_ub,
        method="highs",
        options={"time_limit": 4.0, "quiet": True},
    )
    if res.success:
        return np.maximum(res.x, 0.0)   # numerical safety
    return None


def _optimal_mix_t(ff: np.ndarray, fg: np.ndarray, gg: np.ndarray):
    """
    Find t∈[0,1] minimising max_k [(1‑t)²·ff_k + 2t(1‑t)·fg_k + t²·gg_k]
    by a simple ternary search (the objective is unimodal on this interval).
    """
    def max_val(t):
        return np.max((1 - t) ** 2 * ff + 2 * t * (1 - t) * fg + t ** 2 * gg)

    lo, hi = 0.0, 1.0
    for _ in range(30):          # enough for double precision
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if max_val(m1) > max_val(m2):
            lo = m1
        else:
            hi = m2
    return (lo + hi) / 2.0


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑height sequence with a low value of `evaluate_sequence`.
    The routine respects the 1000 s budget and returns the best sequence
    discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0  # a small safety margin

    # ------------------------------------------------------------------
    # 1.  Build an initial “inverse‑square‑root’’ sequence.
    #     a_i = C(2i,i) / 4^i  (coefficients of (1‑z)^‑½)
    # ------------------------------------------------------------------
    n = 800                           # modest length – good trade‑off
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        # recurrence for binom(2i,i)/4^i
        a[i] = a[i - 1] * ((2 * i - 1) * (2 * i)) / (4.0 * i * i)

    # Initial normalisation (not required for the objective but convenient)
    cur_seq = a.copy()
    best_seq = cur_seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2.  Main optimisation loop.
    # ------------------------------------------------------------------
    it = 0
    while time.time() < deadline:
        it += 1

        # ----- normalise to Σ = sqrt(2·n)  (as used by the LP) -----
        sum_cur = np.sum(cur_seq)
        if sum_cur < 1e-12:                     # safety, should never happen
            break
        norm_cur = cur_seq * (np.sqrt(2.0 * n) / sum_cur)

        # ----- current maximal convolution value -----
        conv_cur = _convolve_fft(norm_cur, norm_cur)
        M = np.max(conv_cur)

        # ----- linear‑programming direction -----
        direction = _solve_convolution_lp(norm_cur, M)
        if direction is None:
            # fallback: a tiny random perturbation
            idx = np.random.randint(n)
            cur_seq[idx] = max(0.0, cur_seq[idx] + np.random.randn() * 1e-3)
            continue

        # ----- normalise the direction the same way -----
        sum_dir = np.sum(direction)
        if sum_dir < 1e-12:
            break
        norm_dir = direction * (np.sqrt(2.0 * n) / sum_dir)

        # ----- pre‑compute the three convolutions needed for mixing -----
        ff = _convolve_fft(norm_cur, norm_cur)
        fg = _convolve_fft(norm_cur, norm_dir)
        gg = _convolve_fft(norm_dir, norm_dir)

        # ----- best mixing weight t -----
        t = _optimal_mix_t(ff, fg, gg)

        # ----- produce the new candidate -----
        new_seq = (1.0 - t) * cur_seq + t * norm_dir

        # ----- evaluate -----
        try:
            score = evaluate_sequence(new_seq.tolist())
        except Exception:          # just in case
            continue

        # ----- keep the best -----
        if score < best_score:
            best_score = score
            best_seq = new_seq.copy()
            print(f"[iter {it:04d}] improved score = {best_score:.6f} (t={t:.4f})")

        # ----- update current sequence (even if not better) to continue the search -----
        cur_seq = new_seq

        # optional early stop if we are already below the target
        if best_score < 1.503:
            print("[stop] target achieved")
            break

    print(f"[final] best score = {best_score:.6f}")
    # Ensure a plain Python list is returned (evaluate_sequence expects a list)
    return best_seq.tolist()
