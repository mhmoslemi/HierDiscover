"""
Propose a candidate step‑function sequence that (hopefully) beats the
baseline score of 2.0.

The algorithm is a simple projected‑gradient search that explicitly
reduces the *largest* convolution entry.  For a current sequence `a`
the index `k_max` where (a*a)[k] attains its maximum is found.
The gradient of this maximum with respect to each coefficient is
simply the coefficient that it is paired with in that convolution,
i.e.  ∂max/∂a_i = a_{k_max‑i} (zero when the index is out of range).
We move a little *against* this gradient while keeping the ℓ₁‑norm
(constant sum) unchanged and enforcing non‑negativity.
A small random perturbation is injected every few hundred iterations
to escape shallow plateaus and a fallback LP‑direction (the same one
used in the reference implementation) is occasionally tried.

The search stops after the allocated time budget and the best sequence
found is returned as a plain Python list (as required by `evaluate_sequence`).
"""

import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Helper ----------------------------------------------------------------
# ----------------------------------------------------------------------


def _conv_max_index(seq: np.ndarray) -> int:
    """Return the index where the auto‑convolution of `seq` attains its maximum."""
    b = np.convolve(seq, seq)
    return int(np.argmax(b))  # numpy returns numpy.int64 – cast to int for safety


def _gradient_wrt_max(seq: np.ndarray, k_max: int) -> np.ndarray:
    """
    Gradient of the maximal convolution entry (at index k_max)
    with respect to the sequence entries.
    For each i:  grad[i] = seq[k_max - i]  if the partner index is valid,
    otherwise 0.
    """
    n = seq.shape[0]
    grad = np.zeros_like(seq)

    # i must satisfy 0 <= k_max - i < n  →  k_max - (n-1) <= i <= k_max
    i_start = max(0, k_max - (n - 1))
    i_end = min(n - 1, k_max)

    i_vals = np.arange(i_start, i_end + 1, dtype=int)
    j_vals = k_max - i_vals   # partner indices, guaranteed in range

    grad[i_vals] = seq[j_vals]
    return grad


def _project_onto_sum_constraint(vec: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Return `vec` re‑scaled so that its sum equals `target_sum`,
    preserving non‑negativity (values are first clipped at zero and then
    uniformly scaled).  This mimics the behaviour of the evaluate function
    which is scale‑invariant.
    """
    vec = np.maximum(vec, 0.0)          # enforce non‑negativity
    cur_sum = vec.sum()
    if cur_sum == 0.0:
        # fall back to a uniform tiny vector to avoid division by zero
        vec = np.full_like(vec, 1e-12)
        cur_sum = vec.sum()
    scale = target_sum / cur_sum
    # guard against blowing up any entry past 1000 (the evaluation caps at 1000)
    if scale * vec.max() > 800.0:  # a safety margin
        scale = 800.0 / vec.max()
    return vec * scale


def _lp_direction(seq_norm: np.ndarray, rhs: float):
    """
    Solve the LP from the reference solution: maximise sum(b)
    subject to convolution(f, b) ≤ rhs, b ≥ 0.
    Returns the optimal `b` as a NumPy array, or None on failure.
    """
    n = len(seq_norm)
    # objective: maximise sum(b)  → minimise -1ᵀb
    c = -np.ones(n)

    # Build inequality matrix for each convolution index k
    a_ub = []
    b_ub = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # row[j] = f[i] for i+j = k  (i varies)
        # we loop over i and fill row[j]
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = seq_norm[i]
        a_ub.append(row)
        b_ub.append(rhs)

    # non‑negativity constraints: -I * b ≤ 0
    a_ub = np.vstack([a_ub, -np.eye(n)])
    b_ub = np.hstack([b_ub, np.zeros(n)])

    # Use a modest time limit per LP call
    opt = optimize.linprog(
        c,
        A_ub=a_ub,
        b_ub=b_ub,
        bounds=(0, None),
        method='highs',
        options={'time_limit': 5.0, 'presolve': True},
    )
    if opt.success:
        return opt.x
    return None


def _try_lp_improvement(seq: np.ndarray, target_sum: float):
    """
    Attempt a direction‑move using the LP from the original paper.
    Returns a new sequence (as NumPy array) or None if the LP fails.
    """
    n = len(seq)
    sf = seq * np.sqrt(2 * n) / seq.sum()
    rhs = np.max(np.convolve(sf, sf))

    g = _lp_direction(sf, rhs)
    if g is None:
        return None

    # rescale the LP solution to the original target sum
    g = g * np.sqrt(2 * n) / g.sum()
    # mix a small step (t≈0.05) – larger than 0.01 to accelerate progress
    t = 0.05
    new_seq = (1 - t) * seq + t * g
    return _project_onto_sum_constraint(new_seq, target_sum)

# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **_kwargs):
    """
    Search for a sequence of non‑negative heights with a low
    `evaluate_sequence` value.

    The strategy combines three ingredients:
      1. Start from a near‑uniform sequence perturbed with a small random noise.
      2. Repeatedly apply a projected gradient step that reduces the
         currently largest convolution entry.
      3. Occasionally (every ~200 iterations) try the LP‑based direction
         from the reference implementation and accept it if it improves
         the score.

    The best sequence encountered before the time budget expires is returned.
    """
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # 0. initialise ----------------------------------------------------
    # ------------------------------------------------------------------
    # length of the problem (fallback to 4000 if the global is missing)
    try:
        n = len(height_sequence_1)
    except NameError:
        n = 4000

    # target ℓ₁‑norm – any positive number works because the objective is
    # scale‑invariant.  We use the normalising constant sqrt(2*n) used in the
    # reference code, but the exact value is irrelevant.
    target_sum = np.sqrt(2.0 * n)

    # start from a flat sequence with a tiny random perturbation
    base = np.full(n, target_sum / n, dtype=float)
    base += 0.01 * np.random.randn(n)          # small zero‑mean noise
    cur_seq = _project_onto_sum_constraint(base, target_sum)

    best_seq = cur_seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"Initial score: {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1. main optimisation loop ----------------------------------------
    # ------------------------------------------------------------------
    start = time.time()
    deadline = start + budget_s - 5.0               # keep a few seconds margin
    iteration = 0
    alpha0 = 0.3                                    # initial step size
    lp_interval = 200                               # try LP every N iters

    while time.time() < deadline:
        iteration += 1

        # --------------------------------------------------------------
        # a) gradient step that attacks the current max‑convolution entry
        # --------------------------------------------------------------
        k_max = _conv_max_index(cur_seq)
        grad = _gradient_wrt_max(cur_seq, k_max)

        # Projected gradient: keep sum constant, respect non‑negativity
        # Subtract mean(grad) so the update is zero‑sum.
        step = alpha0 * (1.0 - iteration / 5000.0)  # linear decay
        update = cur_seq - step * (grad - grad.mean())
        cur_seq = _project_onto_sum_constraint(update, target_sum)

        # --------------------------------------------------------------
        # b) occasional random jitter to avoid stagnation
        # --------------------------------------------------------------
        if iteration % 150 == 0:
            jitter = 0.02 * np.random.randn(n)
            cur_seq = _project_onto_sum_constraint(cur_seq + jitter, target_sum)

        # --------------------------------------------------------------
        # c) occasionally invoke the LP‑based move from the paper
        # --------------------------------------------------------------
        if iteration % lp_interval == 0:
            cand_seq = _try_lp_improvement(cur_seq, target_sum)
            if cand_seq is not None:
                cur_seq = cand_seq

        # --------------------------------------------------------------
        # d) evaluate current candidate
        # --------------------------------------------------------------
        cur_score = evaluate_sequence(cur_seq.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = cur_seq.copy()
            print(f"[{iteration:6d}] Improved score: {best_score:.6f}")

        # simple safeguard: if the score ever goes below 1.4 we can stop early
        if best_score < 1.4:
            break

    # ------------------------------------------------------------------
    # 2. return the best found sequence --------------------------------
    # ------------------------------------------------------------------
    print(f"Finished after {iteration} iterations – best score {best_score:.6f}")
    return best_seq.tolist()
