"""
Propose a candidate step‑function sequence.

The algorithm builds several initial candidates (constant, edge‑heavy,
random, and the previously discovered SOTA sequence) and then
improves them with a fast simulated‑annealing / hill‑climbing loop.
Moves consist of transferring a small amount of mass from one position
to another while keeping the total sum = 1 (the metric is scale‑invariant).
The temperature is decreased linearly so that early iterations accept
worse moves, helping to escape local minima, while later iterations
focus on greedy improvement.

The implementation stays within the 1000 s budget, periodically checks
the remaining time and returns the best sequence found.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _normalise(seq):
    """Scale a sequence so that its sum equals 1 (avoid division by zero)."""
    s = sum(seq)
    if s <= 0:
        # fallback to uniform distribution
        n = len(seq)
        return [1.0 / n] * n
    return [x / s for x in seq]


def _initial_constant(n):
    """Uniform sequence (sum = 1)."""
    return [1.0 / n] * n


def _initial_edge_heavy(n):
    """
    Edge‑heavy shape: a_i ∝ 1/(i+1) + 1/(n-i).
    Large values near both ends, small in the middle.
    """
    arr = np.array([1.0 / (i + 1) + 1.0 / (n - i) for i in range(n)], dtype=float)
    return _normalise(arr.tolist())


def _initial_random(n):
    """Random Dirichlet‑like vector."""
    arr = np.random.rand(n)
    return _normalise(arr.tolist())


def _initial_sota(n):
    """
    Use the previously discovered sequence (global height_sequence_1)
    if it exists; cut/pad it to length n and renormalise.
    """
    try:
        seq = list(height_sequence_1)  # noqa: F821   (may be undefined)
    except Exception:
        # fall back to random
        return _initial_random(n)

    if len(seq) < n:
        seq = seq + [0.0] * (n - len(seq))
    else:
        seq = seq[:n]
    return _normalise(seq)


def _score(seq):
    """Thin wrapper around the given evaluate_sequence function."""
    # evaluate_sequence is assumed to be already imported in the environment
    return evaluate_sequence(seq)  # noqa: F821


def _propose_move(seq, max_delta_frac=0.2):
    """
    Produce a neighbouring sequence by moving a small amount of mass
    from index i to index j (i != j).  The amount moved is a fraction
    of the source entry (bounded by max_delta_frac).  The result is
    automatically re‑normalised to keep the total sum equal to 1.
    """
    n = len(seq)
    i = random.randrange(n)
    j = random.randrange(n - 1)
    if j >= i:
        j += 1  # ensure i != j

    src = seq[i]
    if src <= 0:
        # nothing can be taken from i – pick another pair
        return seq

    # choose a fraction in [-max_delta_frac, max_delta_frac] but keep src>=0
    frac = random.uniform(-max_delta_frac, max_delta_frac)
    delta = src * frac
    # enforce non‑negativity
    if delta > src:
        delta = src
    if delta < -seq[j]:
        delta = -seq[j]

    new_seq = seq.copy()
    new_seq[i] = max(0.0, new_seq[i] - delta)
    new_seq[j] = max(0.0, new_seq[j] + delta)

    # re‑normalise (numerical drift may appear)
    return _normalise(new_seq)


def _annealing_loop(start_seq, deadline, start_temp=1.0, end_temp=0.001,
                    max_iter=20000):
    """
    Simulated‑annealing improvement of a single starting sequence.
    Returns the best sequence found and its score.
    """
    n = len(start_seq)
    best_seq = start_seq[:]
    best_sc = _score(best_seq)
    cur_seq = start_seq[:]
    cur_sc = best_sc

    # pre‑compute start time for temperature schedule
    start_time = time.time()
    total_time = deadline - start_time
    it = 0

    while time.time() < deadline and it < max_iter:
        # linear temperature schedule from start_temp to end_temp
        elapsed = time.time() - start_time
        frac = min(1.0, elapsed / total_time)
        temp = start_temp * (1.0 - frac) + end_temp * frac

        cand = _propose_move(cur_seq)
        cand_sc = _score(cand)

        # always accept if better; otherwise accept with Metropolis probability
        accept = False
        if cand_sc < cur_sc:
            accept = True
        else:
            # probability to accept worse move
            prob = np.exp((cur_sc - cand_sc) / max(temp, 1e-12))
            if random.random() < prob:
                accept = True

        if accept:
            cur_seq = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_seq = cand
                # occasional progress report
                print(f"[{time.time():.1f}] New best score: {best_sc:.6f} (iter {it})")

        it += 1

    return best_seq, best_sc

    restarts = 0
    while time.time() < deadline:

        now = time.time()
        remaining = deadline - now
