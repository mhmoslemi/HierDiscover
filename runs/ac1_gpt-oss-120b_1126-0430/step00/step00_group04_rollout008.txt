"""
Hybrid optimizer for the step‑function evaluation problem.

The score to minimise is

    score = 2 * n * max(conv(a, a)) / (sum(a) ** 2)

where `conv` is the (full) discrete convolution.
The algorithm combines three complementary ideas:

1. **Sub‑gradient descent on the peak of the convolution**  
   For the index `k*` where the convolution reaches its maximum we have
   ∂max/∂a_i = 2 * a_{k*‑i} (if the index is inside the array).  
   A simple gradient step `a ← a – η·grad` is performed and the resulting
   vector is projected onto the probability simplex (fixed total sum,
   non‑negative entries).  The projection keeps the score invariant under
   scaling, so we fix the sum to 1.0 for numerical stability.

2. **Periodic LP move (the previous state‑of‑the‑art step)**  
   Every few iterations the LP defined in the reference code is solved:
   maximise the total sum while respecting the current worst‑case
   convolution bound `M`.  The LP solution provides a direction `g`
   that can increase the sum without increasing `M`.  A convex combination
   `(1‑t)·a + t·g` with several candidates for `t` is examined and the best
   blend is kept.

3. **Random perturbations**  
   To escape plateaus a small random jitter is applied to a handful of
   entries from time to time.  After jitter we again project onto the simplex.

The routine runs for the allotted ``budget_s`` seconds (default 1000 s),
printing occasional progress messages, and finally returns the best
sequence found as a plain ``list[float]``.  All numbers are non‑negative,
and the final sequence can be directly fed to the provided ``evaluate_sequence``
function.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper: projection onto the simplex {x >= 0, sum(x) = s}
def project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex.
    Returns a vector w with w >= 0 and sum(w) == s.
    """
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    # find the largest rho where u_i - cssv_i / i > 0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1   # zero‑based index
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Linear program that, for a *fixed* f, finds the best g with maximal sum
# while keeping the convolution bound unchanged.
def solve_convolution_lp(f_sequence, rhs):
    """
    Solve:
        maximise    Σ b_j
        subject to  conv(f, b) <= rhs   (for all shifts)
                    b_j >= 0
    Returns the optimal b as a NumPy array, or None on failure.
    """
    n = len(f_sequence)
    c = -np.ones(n)                         # maximise Σ b_j  <=> minimise -Σ b_j
    a_ub = []
    b_ub = []

    # Build constraints conv(f, b)[k] <= rhs  for k = 0 .. 2n‑2
    for k in range(2 * n - 1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_sequence[i]
        a_ub.append(row)
        b_ub.append(rhs)

    # Non‑negativity constraints: -b_j <= 0
    a_ub_nonneg = -np.eye(n)
    b_ub_nonneg = np.zeros(n)

    a_ub = np.vstack([a_ub, a_ub_nonneg])
    b_ub = np.hstack([b_ub, b_ub_nonneg])

    # Solve with a modest time limit (the outer loop already enforces a global budget)
    res = linprog(c, A_ub=a_ub, b_ub=b_ub, method='highs', options={'time_limit': 5.0})
    if res.success:
        return res.x
    return None

# ----------------------------------------------------------------------
def get_good_direction_to_move_into(sequence):
    """
    Compute a new direction `g` that (approximately) maximises the sum
    while keeping the current convolution bound unchanged.
    Returns a *list* of floats (the blended sequence with t=0.01) or None.
    """
    n = len(sequence)
    sum_seq = np.sum(sequence)
    if sum_seq == 0:
        return None

    # Normalise to the conventional scale (sum = sqrt(2n))
    norm_factor = np.sqrt(2.0 * n) / sum_seq
    f_norm = [x * norm_factor for x in sequence]

    # Current worst convolution value
    rhs = np.max(np.convolve(f_norm, f_norm))

    g = solve_convolution_lp(f_norm, rhs)
    if g is None:
        return None

    # Rescale g back to the original total‑sum scale
    sum_g = np.sum(g)
    if sum_g == 0:
        return None
    g_norm = [x * np.sqrt(2.0 * n) / sum_g for x in g]

    # Small blend (the original code used t=0.01)
    t = 0.01
    blended = [(1.0 - t) * a + t * b for a, b in zip(sequence, g_norm)]
    return blended

# ----------------------------------------------------------------------
