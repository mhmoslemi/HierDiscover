import time, numpy as np, from scipy.optimize import linprog

def convolve_self(a):
    n = a.shape[0]
    size = 1 << ((2*n-1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)
    return conv[:2*n-1]

def eval_seq(a):
    n = a.shape[0]
    m = np.max(convolve_self(a))
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    return 2 * n * m / (s * s)

def solve_lp(f, rhs):
    n = len(f)
    c = -np.ones(n)
    # build rows for each convolution index
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        start = max(0, k-(n-1))
        end = min(n-1, k)
        # row j = f[k-j]
        for i in range(start, end+1):
            j = k - i
            A[k, j] = f[i]
    # add nonnegativity constraints: -I * b <= 0
    A = np.vstack([A, -np.eye(n)])
    b = np.concatenate([np.full(2*n-1, rhs), np.zeros(n)])
    res = linprog(c, A_ub=A, b_ub=b, method='highs')
    if res.success:
        return res.x
    else:
        return None

def convolve_self_fft(a):
    """Return self‑convolution a*a using an FFT of size power‑of‑2."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def fast_eval(a):
    """Fast version of evaluate_sequence for non‑negative, bounded a."""
    n = a.shape[0]
    max_b = np.max(convolve_self_fft(a))
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))

def solve_convolution_lp(f_seq, rhs):
    """
    Solve   max sum(b)   s.t.   (f_seq * b)_k <= rhs   for every k,
    b >= 0.
    Returns b as a dense ndarray or None.
    """
    n = len(f_seq)
    # objective: minimize -sum(b)
    c = -np.ones(n)

    # build linear inequality matrix A_ub * b <= b_ub
    rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # b_j coefficient = f_seq[k-j] whenever 0 <= j < n and 0 <= k-j < n
        start = max(0, k - (n - 1))
        end = min(n - 1, k)
        j_idx = np.arange(start, end + 1)
        row[j_idx] = f_seq[k - j_idx]
        rows.append(row)
    A = np.vstack(rows)
    # add non‑negativity constraints –b <= 0
    A = np.vstack([A, -np.eye(n)])
    b_ub = np.concatenate([np.full(2 * n - 1, rhs), np.zeros(n)])

    # use the HiGHS interior‑point/simplex solver – fast and robust
    res = linprog(c, A_ub=A, b_ub=b_ub, method='highs')
    if res.success:
        return res.x
    return None

def convolve_self_fft(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a using an FFT of power‑of‑2 length."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)

def fast_eval(a: np.ndarray) -> float:
    """Fast version of ``evaluate_sequence`` for non‑negative bounded vectors."""
    n = a.shape[0]
    max_b = np.max(convolve_self_fft(a))
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))


def solve_convolution_lp(f_seq: np.ndarray, rhs: float):
    """
    Solve   max sum(b)   s.t.   (f_seq * b)_k <= rhs   for every k,
    b >= 0.
    Returns the optimal vector b (dense ndarray) or ``None`` on failure.
    """
    n = len(f_seq)
    # maximise sum(b)  <=> minimise -sum(b)
    c = -np.ones(n)

    # build the matrix for the convolution constraints
    rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        start = max(0, k - (n - 1))
        end = min(n - 1, k)
        j_idx = np.arange(start, end + 1, dtype=int)
        row[j_idx] = f_seq[k - j_idx]                 # coefficient = f_{k-j}
        rows.append(row)

    A = np.vstack([A, -np.eye(n)])

    result = linprog(c, A_ub=A, b_ub=b_ub, method="highs")

                        candidate = mixed

                # also accept the pure LP vector if it is better
                if score_g < candidate_score:
                    candidate = g
                    candidate_score = score_g

                a = candidate
                if candidate_score < best_score:
                    best_score = candidate_score
                    best_seq = a.copy()
        else:
            # ------------------------------------------------------
            # cheap sub‑gradient iteration
            a, cur_score = subgradient_update(a, target_sum)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()

        # ----------------------------------------------------------
        # occasional random jitter to escape shallow local minima
        if np.random.rand() < 0.03:
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i != j and a[i] > 0.0:
                delta = a[i] * 0.05
                a[i] -= delta
                a[j] += delta
                a = a * (target_sum / np.sum(a))

        it += 1

    # Return a plain ``list`` as required by the evaluation harness
    return best_seq.tolist()
