def propose_candidate(seed=0, budget_s=1000):
    import numpy as np, time, math
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5

    # choose size
    n = 2000

    # initialize a
    a = np.empty(n)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i-1] * (2*(i-1) + 1) / (2 * i)   # but careful: using i-1 for previous index
    a = a / a.sum()   # normalize sum to 1

    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())
    print("Initial score:", best_score)

    # parameters
    lr = 0.1   # step size
    iteration = 0
    no_improve = 0

    while time.time() < deadline:
        # compute conv
        conv = np.convolve(a, a)
        k_max = np.argmax(conv)
        # compute gradient only for max index: g_i = 2*a_{k_max - i}
        grad = np.zeros_like(a)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1)
        grad[idx] = 2 * a[k_max - idx]

        # step
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        sum_a = a_new.sum()
        if sum_a == 0:
            a_new = np.random.random(n)
            a_new /= a_new.sum()
        else:
            a_new /= sum_a

        a = a_new
        score = evaluate_sequence(a.tolist())
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"Iter {iteration}: new best {best_score:.6f}")
            no_improve = 0
        else:
            no_improve += 1

        # adapt lr
        if iteration % 500 == 0 and iteration>0:
            if no_improve > 1000:
                lr *= 0.5
                no_improve = 0
                print(f"Reducing lr to {lr}")
        iteration += 1

        # occasional random restarts if stuck
        if iteration % 5000 == 0 and best_score > 1.6:
            # restart with random perturb
            a = np.random.random(n)
            a /= a.sum()
            lr = 0.1

    # convert best_a to list and return
    return best_a.tolist()

"""
Propose a candidate step‑function (list of non‑negative heights) that tries to
minimise the supplied ``evaluate_sequence`` function.

The algorithm is based on a known near‑optimal analytic construction.
For an infinite non‑negative sequence a_i = C(2i,i)/4^i (the Catalan
coefficients) the linear convolution a*a is constant.  After truncating to a
finite length n and normalising the sum to 1 the evaluation value equals
π/2 ≈ 1.5708.  This is already well below the uniform bound 2.0.

Starting from this “Catalan” seed we perform a cheap projected sub‑gradient
descent that directly reduces the current maximal convolution entry.
At each iteration we:
  * compute the linear convolution,
  * locate its maximal index k_max,
  * form the sub‑gradient g_i = 2 * a_{k_max‑i} (zero where the index is out
    of range),
  * take a small step a ← a – η·g, clip to non‑negative, and re‑normalise to
    unit sum.

The step size η is slowly reduced; occasional random restarts avoid stagnation.
All operations are O(n) per iteration, allowing many iterations within the
1000 s budget.  The best sequence found is returned as a plain Python list.
"""

import time
import numpy as np

def _catalan_sequence(n: int) -> np.ndarray:
    """Return the first *n* Catalan‑like coefficients a_i = C(2i,i)/4**i,
    normalised so that their sum equals 1."""
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    # recurrence: a_{i+1} = a_i * (2*i+1) / (2*(i+1))
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * (i - 1) + 1) / (2.0 * i)
    a /= a.sum()
    return a


def _subgradient_step(a: np.ndarray, lr: float) -> np.ndarray:
    """Perform one projected sub‑gradient step that reduces the current
    maximal convolution entry.

    Parameters
    ----------
    a : np.ndarray
        Current non‑negative sequence (sum = 1).
    lr : float
        Learning rate (step size).

    Returns
    -------
    np.ndarray
        Updated sequence, still non‑negative and unit‑sum.
    """
    n = a.size
    conv = np.convolve(a, a)
    k_max = int(np.argmax(conv))

    # sub‑gradient: g_i = 2 * a_{k_max - i} for feasible indices
    grad = np.zeros_like(a)
    i_start = max(0, k_max - (n - 1))
    i_end = min(n - 1, k_max)
    idx = np.arange(i_start, i_end + 1, dtype=int)
    grad[idx] = 2.0 * a[k_max - idx]

    # gradient descent step, then projection onto the simplex (non‑negative,
    # unit‑sum).  Projection is just a re‑normalisation because we never
    # create negative mass beyond clipping.
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 0:
        a_new /= s
    else:
        # degenerate case – fall back to a random point on the simplex
        a_new = np.random.random(n)
        a_new /= a_new.sum()
    return a_new


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a sequence of non‑negative heights that minimises
    ``evaluate_sequence``.  The routine runs for at most *budget_s* seconds
    and returns the best sequence found.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility (default: 42).
    budget_s : int, optional
        Time budget in seconds (default: 1000).

    Returns
    -------
    list[float]
        The best sequence discovered (as a plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5  # leave a small safety margin

    # ---- problem size -------------------------------------------------
    # A length of a few thousand gives a good trade‑off between quality
    # and speed.  3000 works comfortably within the time limit.
    n = 3000

    # ---- initialise with the Catalan (almost optimal) construction ----
    a = _catalan_sequence(n)

    # keep the best candidate seen so far
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # learning‑rate schedule
    lr = 0.05
    iteration = 0
    no_improve = 0

    while time.time() < deadline:
        # one optimisation step
        a = _subgradient_step(a, lr)

        # evaluate the current sequence
        cur_score = evaluate_sequence(a.tolist())

        # keep the best one
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{iteration:05d}] new best = {best_score:.6f}")
            no_improve = 0
        else:
            no_improve += 1

        # slowly decay the step size
        if iteration % 400 == 0 and iteration > 0:
            lr *= 0.9
            # occasional random restart if we seem stuck
            if no_improve > 2000:
                a = np.random.random(n)
                a /= a.sum()
                lr = 0.05
                no_improve = 0
                print(f"[{iteration:05d}] restart (random init)")

        iteration += 1

    print(f"[done] best score found = {best_score:.6f}")
    return best_seq.tolist()
