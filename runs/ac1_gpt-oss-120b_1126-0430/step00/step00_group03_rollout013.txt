def redistribute(seq):
    n = len(seq)
    conv = np.convolve(seq, seq)
    k = np.argmax(conv)
    # indices contributing to conv[k] are pairs (i, j = k-i) where both within range
    # compute contributions product p_i = seq[i] * seq[j] (ordered? actually each ordered pair counted)
    # For simplicity treat unordered contributions: product = seq[i] * seq[j] for i <= j.
    # We want to decrease max: find i where seq[i] is large and seq[j] also large.
    # Choose i_max = argmax seq[i] * seq[k-i]
    i_candidates = np.arange(n)
    j = k - i_candidates
    valid = (j >= 0) & (j < n)
    products = np.zeros_like(seq)
    products[valid] = seq[i_candidates[valid]] * seq[j[valid]]
    i_max = np.argmax(products)  # index with largest product
    j_max = k - i_max
    # move small amount from i_max (or j_max) to other index l with small partner for which contributions to conv low.
    # Choose l random, maybe l far from i_max.
    delta = 0.01 * seq[i_max]  # small fraction
    if delta == 0: return seq
    # ensure after moving, seq[i_max] - delta >=0
    seq[i_max] -= delta
    # pick l random
    l = np.random.randint(n)
    seq[l] += delta
    return seq

"""
Propose a good step‑function (non‑negative sequence) that minimizes
evaluate_sequence.  The algorithm works with a fixed length n (default 2000)
and iteratively improves the sequence by:

1. Computing the self‑convolution using an FFT (fast O(n log n)).
2. Finding the index k where the convolution attains its maximum.
3. Taking a sub‑gradient step that reduces the contribution of that
   index (the gradient is a shifted reverse of the current sequence).
4. Adding a tiny random perturbation to avoid getting stuck in symmetry.
5. Re‑normalising the total sum to the canonical value sqrt(2·n)
   (the evaluation function is invariant to a global scale).
6. Keeping the best sequence ever seen.

The routine runs until the supplied time budget expires (default 1000 s)
and returns the best sequence found.  Simple simulated‑annealing
acceptance (rarely accepting a worse move) helps escaping shallow local
minima.  The method does not rely on the previous LP‑based code, but can
still discover sequences with a maximal convolution substantially below
the uniform value 2.0 (target ≈ 1.5).
"""

import numpy as np
import time
import math

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _normalize(seq: np.ndarray) -> np.ndarray:
    """Scale seq so that its sum equals sqrt(2·len(seq))."""
    n = len(seq)
    target = math.sqrt(2.0 * n)
    s = seq.sum()
    if s == 0:
        seq[:] = target / n
    else:
        seq *= target / s
    return seq


def _conv_fft(seq: np.ndarray, fft_len: int) -> np.ndarray:
    """Self‑convolution using real FFT, returned length 2·n‑1."""
    # real FFT is a little faster and uses less memory
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * len(seq) - 1]
    return conv


def _gradient_max(seq: np.ndarray, max_idx: int) -> np.ndarray:
    """
    Sub‑gradient of the maximal convolution entry.
    For a given index k the gradient is 2·rev shifted by (k‑(n‑1)).
    """
    n = len(seq)
    rev = seq[::-1]                     # reversed copy
    offset = max_idx - (n - 1)           # shift amount
    grad = np.zeros_like(seq)

    if offset >= 0:
        # region where both sides overlap
        grad[offset:] = 2.0 * rev[: n - offset]
    else:
        # offset is negative → shift the other way
        grad[: n + offset] = 2.0 * rev[-offset:]
    return grad


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list:
    """Search for a sequence that yields a low evaluation score."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # a little safety margin

    # ------------------------------------------------------------------
    # Parameters – they may be tweaked if more time is available.
    # ------------------------------------------------------------------
    n = 2000                     # length of the sequence (feel free to change)
    fft_len = _next_pow2(2 * n - 1)

    # start from the uniform (known baseline) and add a tiny perturbation
    cur = np.full(n, math.sqrt(2.0 * n) / n, dtype=np.float64)
    cur += np.random.randn(n) * 1e-5
    cur = np.maximum(cur, 0.0)
    cur = _normalize(cur)

    # best ever discovered
    best_seq = cur.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    # ---------------------------------------------------------------
    # Main optimisation loop
    # ---------------------------------------------------------------
    # Hyper‑parameters
    step_factor = 0.02          # controls sub‑gradient step size
    noise_factor = 0.001        # amplitude of Gaussian jitter each iteration
    accept_worse = 0.002        # probability of accepting a non‑improving move

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # --- convolution and maximal index ---------------------------------
        conv = _conv_fft(cur, fft_len)
        max_idx = int(np.argmax(conv))
        max_val = conv[max_idx]
        avg_val = conv.mean()

        # --- sub‑gradient step --------------------------------------------
        grad = _gradient_max(cur, max_idx)

        # Step size that scales with the excess of the peak over the mean.
        step = step_factor * (max_val - avg_val)
        if step <= 0:
            step = step_factor * max_val * 1e-3

        nxt = cur - step * grad
        # add a tiny random perturbation to keep the search stochastic
        nxt += np.random.randn(n) * noise_factor

        # non‑negativity, then renormalise to keep the sum fixed
        nxt = np.maximum(nxt, 0.0)
        nxt = _normalize(nxt)

        # evaluate the new candidate
        cur_score = evaluate_sequence(nxt.tolist())

        # keep the best ever found
        if cur_score < best_score:
            best_score = cur_score
            best_seq = nxt.copy()
            print(f"[{time.time():.1f}] iter {iteration:6d}  new best = {best_score:.6f}")

        # accept if we improved, otherwise occasionally accept to escape plateaus
        if cur_score < evaluate_sequence(cur.tolist()) or np.random.rand() < accept_worse:
            cur = nxt

        # -----------------------------------------------------------------
        # Occasionally perform a targeted “mass‑redistribution” move that
        # explicitly steals a bit of mass from the pair (i, j) that
        # contributes most to the current peak.
        # -----------------------------------------------------------------
        if np.random.rand() < 0.05:
            # Determine the pair (i, j) with the largest product a_i * a_j for k = max_idx
            offset = max_idx - (n - 1)
            rev = cur[::-1]
            if offset >= 0:
                # overlapping region = cur[offset:]  with rev[:n‑offset]
                prod = cur[offset:] * rev[: n - offset]
                i_local = int(np.argmax(prod))
                i = offset + i_local
            else:
                prod = cur[: n + offset] * rev[-offset:]
                i_local = int(np.argmax(prod))
                i = i_local
            j = max_idx - i

            if i != j and cur[i] > 0:
                delta = 0.05 * cur[i]               # move 5 % of the larger entry
                # choose a destination index that is *not* i or j
                k = np.random.randint(n)
                while k == i or k == j:
                    k = np.random.randint(n)
                cur[i] -= delta
                cur[k] += delta
                cur = np.maximum(cur, 0.0)
                cur = _normalize(cur)

    # ------------------------------------------------------------------
    # Return the best sequence we have seen (as a plain Python list)
    # ------------------------------------------------------------------
    return best_seq.tolist()
