"""
propose_candidate
-----------------

A time‑bounded stochastic optimizer for the step‑function evaluation problem.

The objective to minimise is

    2 * n * max(conv(a, a)) / (sum(a)**2)

which is invariant under positive scaling, so we normalise the
candidate to have unit sum.  The optimizer works on the normalised
vectors and uses a smooth approximation of the “max” (soft‑max) to
obtain a gradient that pushes the largest entries of the
auto‑convolution down while keeping the vector non‑negative.
A simple simulated‑annealing acceptance rule allows occasional uphill
steps to escape local minima.

The search proceeds with many short restarts (different random
initialisations and a few deterministic starts from the provided
`height_sequence_1`).  Each restart runs a limited number of gradient‑
descent / annealing steps, constantly updating the best solution seen.
The algorithm is deliberately lightweight: the convolution and the
gradient are computed with NumPy’s `np.convolve`, allowing thousands of
iterations within the 1000‑second budget.

The function returns the best sequence (as a plain Python list) found
before the deadline.
"""
import time
import numpy as np

def _softmax_weights(conv, alpha):
    """Stable soft‑max weights for a convolution vector."""
    max_val = conv.max()
    # shift to avoid overflow
    exp_vals = np.exp(alpha * (conv - max_val))
    return exp_vals / exp_vals.sum()


def _gradient(a, weights):
    """
    Compute the gradient of the soft‑max approximated max‑conv objective.
    For a vector `a` (length n) and a weight vector `weights` (length 2n‑1)
    the gradient is   g_i = 2 * sum_{l=0}^{n-1} a[l] * weights[i+l].
    This is efficiently obtained via a single convolution.
    """
    a_rev = a[::-1]                         # reversed a
    grad_full = np.convolve(a_rev, weights)  # length n + (2n‑1) – 1 = 3n‑2
    n = a.shape[0]
    # relevant slice corresponds to indices n‑1 … n‑1+n‑1
    return 2.0 * grad_full[n - 1 : n - 1 + n]


def _project_nonnegative_normalise(v):
    """Project onto the simplex of non‑negative vectors with sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-12:               # avoid division by zero – fall back to uniform
        n = v.shape[0]
        return np.full(n, 1.0 / n)
    return v / s


def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  Uses a fast gradient + annealing scheme and
    returns the best sequence discovered within the allotted time.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # keep a small safety margin

    best_seq = None
    best_score = float('inf')

    # possible lengths – short enough for fast iterations, long enough
    # to let the algorithm approach the theoretical lower bound.
    lengths = [30, 50, 80, 120, 180]

    # include the supplied SOTA sequence as an extra start if it exists
    try:
        # height_sequence_1 is expected to be a global variable in the environment
        sos_seq = np.array(height_sequence_1, dtype=float)
        sos_seq = sos_seq / sos_seq.sum() if sos_seq.sum() > 0 else np.full_like(sos_seq, 1.0 / sos_seq.size)
        # make a copy for safe handling
        sos_seq = sos_seq.copy()
    except Exception:
        sos_seq = None

    # main search loop – many short restarts
    while time.time() < deadline:
        # ----------- initialise one restart -----------------
        n = int(np.random.choice(lengths))
        if sos_seq is not None and np.random.rand() < 0.15:
            # use a sliced version of the SOTA sequence (or pad / truncate)
            if sos_seq.size >= n:
                a = sos_seq[:n].copy()
            else:
                a = np.concatenate([sos_seq,
                                    np.full(n - sos_seq.size, 0.0)])
            a = _project_nonnegative_normalise(a)
        else:
            # random Dirichlet start – guarantees sum = 1
            a = np.random.dirichlet(np.ones(n))

        # hyper‑parameters for this restart
        lr = 0.12                     # initial learning rate
        lr_min = 5e-5
        alpha = 2.0                  # soft‑max temperature
        alpha_max = 20.0
        temp = 0.3                   # annealing temperature for Metropolis step
        temp_decay = 0.9995

        # number of inner iterations – limited so that many restarts fit
        max_inner = 1200

        # ----------- inner optimisation loop -----------------
        for it in range(max_inner):
            if time.time() >= deadline:
                break

            # compute convolution and current objective (sum == 1)
            conv = np.convolve(a, a)
            max_conv = conv.max()
            cur_score = 2.0 * n * max_conv    # exact objective for sum=1

            # keep the best ever seen
            if cur_score < best_score:

    # If nothing was found (unlikely) fall back to a simple uniform vector.
    if best_seq is None:
        best_seq = np.full(100, 1.0 / 100)

    # Convert to plain Python list as required by the framework
    return best_seq.tolist()
