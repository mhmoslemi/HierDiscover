def simplex_projection(v):
    # shift v to sum to 1, ensure >=0
    # If sum of positive entries is zero, return uniform? not needed.
    n = len(v)
    u = sorted(v, reverse=True)
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    # find rho = max i such that cond is true
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

    # also evaluate given height_sequence_1
    try:
        s0 = evaluate_sequence(height_sequence_1)
        best_seq = list(height_sequence_1)
        best_score = s0
    except Exception as e:
        pass
    # Parameter: list of lengths to try
    lengths = [2000, 4000, 8000, 12000]  # up to maybe 20000

    while time.time() < deadline:
        n = np.random.choice(lengths)
        # initialize a random vector on simplex
        a = np.random.rand(n)
        a = a / np.sum(a)  # sum =1
        # iterative improvement
        iter_limit = 5000
        step_eta = 0.6
        for it in range(iter_limit):
            # compute conv and max
            conv = np.convolve(a, a)
            max_val = np.max(conv)
            # compute ratio
            score = 2 * n * max_val / (np.sum(a) ** 2)
            if score < best_score:
                best_score = score
                best_seq = a.tolist()
                print(f"New best: {best_score:.6f}, n={n}, it={it}")
            # early exit if time running out
            if time.time() > deadline:
                break

            # find index of max
            k_max = np.argmax(conv)
            # compute gradient
            grad = np.zeros_like(a)
            # for i where j = k_max - i in range, set grad[i] = 2*a[j]
            low = max(0, k_max - (n - 1))
            high = min(n - 1, k_max)
            # i from low..high
            # vectorized:
            i_indices = np.arange(low, high + 1)
            j_indices = k_max - i_indices
            grad[i_indices] = 2.0 * a[j_indices]

            # exponentiated update
            a = a * np.exp(-step_eta * grad)
            # project onto simplex
            # algorithm: w = a - theta; where theta solves sum max(a_i - theta, 0) = 1
            # we implement function
            a = simplex_project(a)   # returns array sum 1, nonnegative
            # optionally decrease η over time
            step_eta = max(0.01, step_eta * 0.9995)   # decay
        # after iter loop check maybe we want to sample a new random start
    return best_seq

def project_simplex(v):
    """Projects a vector onto the probability simplex (nonnegative, sum=1)."""
    v = np.asarray(v)
    if np.sum(v) == 1 and np.all(v >= 0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)

def generate_sequence(n):
    m_target = int(np.sqrt(n)//2)  # quarter
    max_index = n//4  # limit region for large entries
    large_indices = []
    sum_set = set()
    for x in range(max_index):
        collision = False
        for y in large_indices:
            if x + y in sum_set:
                collision = True
                break
        if collision:
            continue
        # also check x+x
        if x + x in sum_set:
            continue
        # no collisions
        large_indices.append(x)
        # add sums
        for y in large_indices:
            sum_set.add(x + y)
        # also add x+x (already added in above loop)
        if len(large_indices) >= m_target:
            break
    # Determine B
    B = 1.0 / np.sqrt(n)
    a = np.full(n, B, dtype=float)
    for idx in large_indices:
        a[idx] = 1.0
    # Optionally also assign large values symmetrically at mirrored positions? Not needed.
    return a.tolist()

    # generate Sidon set
    large_indices = []

def is_prime(num):
    if num < 2:
        return False
    if num % 2 == 0:
        return num == 2
    r = int(num**0.5)
    for i in range(3, r + 1, 2):
        if num % i == 0:
            return False
    return True

"""
Propose a good step‐function sequence.

The construction is based on a two–level scheme:

* A small value `B = α/√n` is placed at almost every position.
* A set of about √n “large” entries with value `1` is placed on a
  Sidon (B₂) set of indices.  For such a set every unordered sum
  `i+j` is unique, consequently each offset of the convolution receives
  at most two ordered contributions from the large entries.
* The small value contributes `n·B² = α²` to the central convolution
  coefficient, whereas the large entries contribute at most `2`
  (self‑terms give `1`).  The total maximal coefficient is therefore
  bounded by `2+α²`, while the total sum is ≈ `α√n + √n = (α+1)√n`.  The
  resulting evaluation is roughly

        2·n·(2+α²) / ((α+1)²·n) = 2·(2+α²)/(α+1)²

  which is ≤ 1.5 for a wide range of `α`.  By trying several `n` and
  `α` values we quickly obtain sequences with scores well below the
  current bound 2.0.

The algorithm:
  1. Choose a length `n`.
  2. Find the largest prime `p` with `p² ≤ n`; the quadratic Sidon set
     `{ i·p + (i² mod p) | i = 0…p‑1 }` has size `p` and satisfies the
     required sum‑uniqueness.
  3. Build the sequence: fill with `B = α/√n`, then set the Sidon indices
     to `1.0`.
  4. Evaluate with `evaluate_sequence`; keep the best sequence found
     before the time limit expires.
"""

import numpy as np
import time
import math

# ----------------------------------------------------------------------
def _is_prime(num: int) -> bool:
    """Simple deterministic primality test for small integers."""
    if num < 2:
        return False
    if num % 2 == 0:
        return num == 2
    r = int(num ** 0.5)
    for d in range(3, r + 1, 2):
        if num % d == 0:
            return False
    return True


def _largest_prime_sq_leq(limit: int) -> int:
    """
    Return the largest prime p such that p² ≤ limit.
    If none is found, return 2 (the smallest prime).
    """
    p = int(math.isqrt(limit))
    while p >= 2:
        if _is_prime(p) and p * p <= limit:
            return p
        p -= 1
    return 2


def _sidon_indices(n: int) -> list[int]:
    """
    Produce a Sidon set of indices inside [0, n).
    Uses the classical quadratic construction with a prime p where p² ≤ n.
    """
    p = _largest_prime_sq_leq(n)
    indices = []
    for i in range(p):
        idx = i * p + (i * i) % p
        if idx < n:          # safety – the construction stays below p² ≤ n
            indices.append(idx)
    return indices


def _build_sequence(n: int, alpha: float) -> list[float]:
    """
    Build the two‑level sequence for a given length `n` and scaling factor `alpha`.

    Small value B = alpha / sqrt(n) is placed everywhere;
    a Sidon set of indices receives the large value 1.0.
    """
    sqrt_n = math.sqrt(n)
    B = alpha / sqrt_n                      # small level
    seq = np.full(n, B, dtype=float)        # start with the small value
    large_idx = _sidon_indices(n)           # Sidon positions (≈ √n of them)
    seq[large_idx] = 1.0                    # large level
    return seq.tolist()


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence with a low evaluation score.
    The search explores a few lengths and a handful of `alpha` values,
    using the deterministic Sidon‑set construction described above.
    The best sequence found before the time budget expires is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5          # a safety margin

    best_seq = None
    best_score = float("inf")

    def _try(seq):
        """Evaluate `seq` and keep it if it improves the best score."""
        nonlocal best_seq, best_score
        try:
            score = evaluate_sequence(seq)
        except Exception:
            return
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            elapsed = time.time() - start_time
            print(f"[{elapsed:6.2f}s] New best = {best_score:.6f} (len={len(seq)})")

    # ------------------------------------------------------------------
    #   1) start from the provided baseline if it exists
    try:
        _try(list(height_sequence_1))
    except Exception:
        pass

    #   2) systematic exploration of parameters
    n_candidates = [2000, 5000, 8000, 12000, 20000]
    alphas = [0.8, 1.0, 1.2, 1.5, 2.0]

    # loop until time runs out (few repetitions are enough – the construction is cheap)
    while time.time() < deadline:
        for n in n_candidates:
            for alpha in alphas:
                seq = _build_sequence(n, alpha)
                _try(seq)
                if time.time() >= deadline:
                    break
            if time.time() >= deadline:
                break
        # if there is still time, shuffle the order a little and try again
        np.random.shuffle(n_candidates)

    # ------------------------------------------------------------------
    # fallback (should never happen)
    if best_seq is None:
        best_seq = [1.0] * 100
        best_score = evaluate_sequence(best_seq)
    return best_seq
