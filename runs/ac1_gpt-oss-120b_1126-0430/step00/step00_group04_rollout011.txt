def _make_sequence(length: int, power: float) -> np.ndarray:
    """
    Return a normalized (sum = 1) non‑negative sequence of given length with
    entries proportional to (index+1) ** power.
    """
    # indices 1 … length
    base = np.arange(1, length + 1, dtype=float) ** power
    # avoid division by zero (won’t happen because base>0)
    return base / base.sum()

            break

        seq = _make_sequence(n, p)

        # Occasionally add a tiny random jitter (±1 %) and re‑normalise.
        # This helps escape possible ties caused by exact arithmetic.
        jitter = 0.01 * np.random.randn(n)
        seq = np.maximum(0.0, seq + jitter)
        seq /= seq.sum()

        # Convert to plain Python list for the evaluator (it accepts list or
        # any sequence that is iterable).
        try:
            score = evaluate_sequence(seq.tolist())
        except Exception as e:
            # Should never happen because we respect the input checks.
            print(f"Evaluation failed for p={p}: {e}")
            continue

        print(f"Exponent {p:4.2f} → score {score:.6f}")

        if score < best_score:
            best_score = score
            best_seq = seq.tolist()

    # Safety fallback – if something went wrong return the uniform sequence.
    if best_seq is None:
        print("Fallback to uniform sequence.")
        best_seq = (np.ones(n) / n).tolist()
        best_score = evaluate_sequence(best_seq)

    print(f"Best score found: {best_score:.6f}")
    return best_seq
