def propose_candidate(seed=0, budget_s=1000, **kwargs):
    """
    Simulated‑annealing / hill‑climbing search for a non‑negative step function.
    The algorithm:
      * Choose a random length n (2000–8000) and initialise a vector with
        sum == sqrt(2*n) (the normalisation used by the original benchmark).
      * Repeatedly perturb two coordinates while keeping the total sum fixed,
        accept the move if the evaluation improves or with a probability
        decreasing with a temperature schedule.

      * Keep the best sequence seen so far and return it when the time budget
        expires.
    """

    np.random.seed(seed)

    deadline = time.time() + budget_s - 0.5   # keep a tiny margin

                val0 = evaluate_sequence(seq0.tolist())
                best_seq = seq0

    except Exception:
        pass

    # ------------------------------------------------------------------

    # The length is a trade‑off: larger n makes the theoretical lower bound

    # We start with a moderate size and possibly increase it later.
    n = int(np.random.randint(2000, 8000))
    target_sum = math.sqrt(2.0 * n)

    # Random Dirichlet vector – all components positive.
    a = np.random.rand(n)
    a = a * (target_sum / a.sum())

    curr_val = evaluate_sequence(a.tolist())
    if curr_val < best_val:
        best_val = curr_val
        best_seq = a.copy()

        remaining = deadline - time.time()

        i, j = np.random.choice(n, size=2, replace=False)

        step = T * target_sum * 0.1   # up to 10 % of total sum when T≈1, shrinking later

        s = cand.sum()

    if curr_val < best_val:
        best_val = curr_val
        best_seq = a.copy()

    # ------------------------------------------------------------------
    # Simulated‑annealing loop
    # ------------------------------------------------------------------
    deadline = time.time() + budget_s - 0.5   # small safety margin
    iteration = 0
    total_budget = budget_s

    while time.time() < deadline:
        # Linear cooling: temperature goes from ~1 → ~1e‑4
        remaining = deadline - time.time()
        T = max(1e-4, remaining / total_budget)

        # ----------- propose a small move (two‑coordinate update) ----------
        i, j = np.random.choice(n, size=2, replace=False)
        # Step size proportional to the current temperature and total mass
        step = T * target_sum * 0.1
        delta = np.random.randn() * step

        # Apply delta to i and opposite to j, keep entries non‑negative
        ai = a[i] + delta
        aj = a[j] - delta
        if ai < 0:
            delta = -a[i]          # clamp delta so ai becomes zero
            ai = 0.0
            aj = a[j] - delta
        elif aj < 0:
            delta = a[j]           # clamp delta so aj becomes zero
            aj = 0.0
            ai = a[i] + delta

        cand = a.copy()
        cand[i] = ai
        cand[j] = aj

        # Renormalise to keep the exact target sum (tiny scaling improves stability)
        s = cand.sum()
        if s > 0:
            cand = cand * (target_sum / s)
        else:   # Very unlikely, fall back to fresh random vector
            cand = np.random.rand(n)
            cand = cand * (target_sum / cand.sum())

        # ------------------------------------------------------------------
        # Evaluate the candidate
        # ------------------------------------------------------------------
        try:
            cand_val = evaluate_sequence(cand.tolist())
        except Exception:
            cand_val = float('inf')

        # Acceptance rule (Metropolis)
        improved = cand_val < curr_val
        if improved or np.random.rand() < math.exp(-(cand_val - curr_val) / (T + 1e-12)):
            a = cand
            curr_val = cand_val
            if cand_val < best_val:
                best_val = cand_val
                best_seq = cand.copy()
                print(f"[{iteration:6d}] New best = {best_val:.6f} (n={n})")

        # ------------------------------------------------------------------
        # Occasionally smooth the vector – this reduces sharp spikes that
        # tend to increase the convolution maximum.
        # ------------------------------------------------------------------
        if iteration % 250 == 0 and iteration > 0:
            kernel = np.array([0.25, 0.5, 0.25])
            smooth = np.convolve(a, kernel, mode='same')
            smooth = smooth * (target_sum / smooth.sum())
            try:
                smooth_val = evaluate_sequence(smooth.tolist())
            except Exception:
                smooth_val = float('inf')
            if smooth_val < curr_val:
                a = smooth
                curr_val = smooth_val
                if smooth_val < best_val:
                    best_val = smooth_val
                    best_seq = smooth.copy()
                    print(f"[{iteration:6d}] Smoothing improved best to {best_val:.6f}")

        # ------------------------------------------------------------------
        # Random restart with a tiny probability – helps escape deep local minima.
        # ------------------------------------------------------------------
        if np.random.rand() < 0.001:
            a = np.random.rand(n)
            a = a * (target_sum / a.sum())
            try:
                curr_val = evaluate_sequence(a.tolist())
            except Exception:
                curr_val = float('inf')
            print(f"[{iteration:6d}] Restarted randomly, value = {curr_val:.6f}")

        # ------------------------------------------------------------------
        # Occasionally increase the length (if time permits).  A longer
        # sequence pushes the theoretical lower bound nearer to 1.
        # ------------------------------------------------------------------
        if iteration % 2000 == 0 and iteration > 0:
            if time.time() < deadline - 1.0:
                new_n = min(n + 500, 12000)   # cap to keep evaluations cheap
                if new_n != n:
                    # Linear interpolation to up‑sample the current vector
                    x_old = np.linspace(0, 1, n)
                    x_new = np.linspace(0, 1, new_n)
                    a = np.interp(x_new, x_old, a)
                    n = new_n
                    target_sum = math.sqrt(2.0 * n)
                    a = a * (target_sum / a.sum())
                    try:
                        curr_val = evaluate_sequence(a.tolist())
                    except Exception:
                        curr_val = float('inf')
                    print(f"[{iteration:6d}] Length increased to {n}, current value = {curr_val:.6f}")

        iteration += 1

    # ------------------------------------------------------------------
    # If nothing succeeded (should not happen) fall back to a uniform sequence.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        uniform = np.ones(n) * (math.sqrt(2.0 * n) / n)
        best_seq = uniform
        print("Fallback to uniform sequence.")

    return best_seq.tolist()
