import time
import math
import random
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Generate a step‑function (non‑negative sequence) that minimizes the
    evaluate_sequence() metric.

    Strategy
    --------
    * The evaluation score is invariant under uniform scaling of the whole
      sequence, so only the *shape* of the values matters.
    * For a shape a_i = (i/(n‑1))**p (p ≥ 1) the continuous analogue gives the
      exact value   2·(p+1)² / ((2p+1)·C(2p,p)).
      This rapidly drops below 1 as p grows (e.g. p=3 → 0.23, p=5 → 0.026,
      p=10 → 6·10⁻⁵).
    * We therefore try a modest set of lengths (500 … 5000) and powers
      (1,2,3,4,5,6,8,10,12).  For each pair we build the power‑law sequence,
      evaluate it, and keep the best one found.
    * As a secondary family we also test exponential shapes
      a_i = exp(-λ·i/(n‑1)) with several λ values.
    * If time permits a small random search and tiny perturbations of the
      current best are performed.

    The whole procedure finishes well before the allotted 1000 s and returns
    the sequence with the smallest observed score.
    """
    # --------------------------------------------------------------------
    # initialisation
    # --------------------------------------------------------------------
    random.seed(seed)
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1        # leave a tiny safety margin

    best_seq: list[float] | None = None
    best_score = float('inf')

    # --------------------------------------------------------------------
    # helper to evaluate and possibly update the current best
    # --------------------------------------------------------------------
    def try_sequence(seq: list[float], label: str = "") -> None:
        nonlocal best_seq, best_score
        try:
            score = evaluate_sequence(seq)
        except Exception:
            return
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[{label}] new best score = {score:.6f} (len={len(seq)})")

    # --------------------------------------------------------------------
    # deterministic scan of power‑law and exponential families
    # --------------------------------------------------------------------
    lengths = [500, 1000, 2000, 5000]
    powers  = [1, 2, 3, 4, 5, 6, 8, 10, 12]
    lambdas = [0.5, 1.0, 2.0, 3.0, 5.0, 8.0]

    for n in lengths:
        if time.time() > deadline:
            break
        # power‑law sequences (increasing and its reversed version)
        for p in powers:
            # build a_i = (i/(n‑1))**p   for i = 0 … n‑1
            seq = [(i / (n - 1)) ** p for i in range(n)]
            try_sequence(seq, f"power n={n}, p={p}")
            # reversed – the same shape mirrored
            try_sequence(seq[::-1], f"power_rev n={n}, p={p}")

        # exponential decay sequences (also monotone)
        for lam in lambdas:
            seq = [math.exp(-lam * i / (n - 1)) for i in range(n)]
            try_sequence(seq, f"exp n={n}, λ={lam}")

    # --------------------------------------------------------------------
    # random exploration while time permits
    # --------------------------------------------------------------------
    while time.time() < deadline:
        # pick random parameters
        n = random.choice(lengths)
        p = random.choice(powers)
        seq = [(i / (n - 1)) ** p for i in range(n)]
        if random.random() < 0.5:
            seq = seq[::-1]                     # sometimes test the mirror
        try_sequence(seq, "rand_power")

        # occasional tiny perturbation of the current best
        if best_seq and random.random() < 0.1:
            pert = best_seq.copy()
            idx = random.randrange(len(pert))
            pert[idx] = max(0.0, pert[idx] + random.gauss(0, 0.01))
            try_sequence(pert, "perturb")

        # break if we are close to the deadline
        if time.time() + 0.02 > deadline:
            break

    # --------------------------------------------------------------------
    # final report
    # --------------------------------------------------------------------
    if best_seq is None:
        # Very unlikely: return a trivial safe sequence
        best_seq = [1.0]
        best_score = evaluate_sequence(best_seq)
        print(f"fallback to trivial sequence, score={best_score}")

    print(f"final best score = {best_score:.6f} (length {len(best_seq)})")
    return best_seq
