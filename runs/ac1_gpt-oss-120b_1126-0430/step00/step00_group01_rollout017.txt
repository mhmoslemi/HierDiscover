    n = len(a)
    conv = np.convolve(a, a)
    # compute softmax weights
    # subtract max for numerical stability
    scaled = tau * conv
    max_scaled = np.max(scaled)

        # compute gradient contributions:
        g = np.zeros_like(a_current)
        for k in idxs:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_range = np.arange(i_start, i_end + 1)
            j_range = k - i_range
            g[i_range] += 2 * a_current[j_range]

        step = 0.1 * np.mean(a_current)  # adjust

        i_low = np.argmin(g)
        if i_high == i_low:

"""
Propose a candidate step function sequence that minimizes the evaluation function.

The algorithm consists of three stages:

1. **Parametric warm‑start** –  For several sequence lengths we generate
   simple families (geometric decay and power‑law decay).  For each family a
   grid of shape parameters is scanned and the best (smallest evaluation
   score) sequence is kept.

2. **Local refinement** –  Starting from the best parametric sequence we
   perform a cheap projected sub‑gradient descent that repeatedly moves a
   small amount of mass from indices that most contribute to the current
   maximal convolution value towards indices that contribute the least.
   After each move the sequence is projected back onto the simplex
   (non‑negative, fixed ℓ₁‑norm).  The process is stopped when the time
   budget expires.

3. **Result** –  The best sequence found during the whole search is
   returned as a plain Python list of floats.  The implementation respects
   the 1000 s time limit, uses only NumPy and the provided
   ``evaluate_sequence`` function and contains no file or network I/O.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper: compute the evaluation score directly (avoids the heavy checks of
# evaluate_sequence when used many times during optimisation).
def _score(seq: np.ndarray) -> float:
    """Return the raw objective 2*n*max(conv)/sum^2 for a non‑negative array."""
    n = seq.shape[0]
    s = seq.sum()
    if s < 0.01:
        return float("inf")
    m = np.convolve(seq, seq).max()
    return float(2.0 * n * m / (s * s))


# ----------------------------------------------------------------------
# Parametric families ---------------------------------------------------
def _geometric_sequence(n: int, alpha: float, target_sum: float) -> np.ndarray:
    """Geometric decay a_i = alpha**i, scaled to have the prescribed sum."""
    i = np.arange(n, dtype=np.float64)
    a = np.power(alpha, i)
    if a.sum() == 0:
        a = np.full(n, target_sum / n)
    else:
        a *= target_sum / a.sum()
    return a


def _powerlaw_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Power‑law decay a_i = (i+1)^{-p}, scaled to the target sum."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, -p)
    if a.sum() == 0:
        a = np.full(n, target_sum / n)
    else:
        a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Local refinement (projected sub‑gradient descent) --------------------
def _refine(seq: np.ndarray, target_sum: float, deadline: float,
            max_iters: int = 5000) -> (np.ndarray, float):
    """
    Improve ``seq`` by moving a little mass from the indices that contribute
    most to the current maximal convolution value to those that contribute
    the least.  The sequence is kept non‑negative and renormalised to
    ``target_sum`` after each step.
    """
    n = seq.shape[0]
    best_seq = seq.copy()
    best_val = _score(best_seq)

    cur_seq = seq.copy()
    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1

        conv = np.convolve(cur_seq, cur_seq)
        M = conv.max()
        # indices where the convolution attains its maximum (with tiny tolerance)
        idxs = np.where(conv >= M - 1e-12)[0]

        # Gradient of the max‑convolution (sub‑gradient)
        g = np.zeros_like(cur_seq)
        for k in idxs:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_range = np.arange(i_start, i_end + 1, dtype=np.int32)
            j_range = k - i_range
            g[i_range] += 2.0 * cur_seq[j_range]

        # Choose where to take mass from / give mass to
        i_high = int(np.argmax(g))
        i_low = int(np.argmin(g))
        if i_high == i_low:
            # If all gradients equal (rare), pick random indices
            i_high = random.randrange(n)
            i_low = random.randrange(n)

        # Amount to transfer – a small fraction of the donor value
        delta = min(cur_seq[i_high] * 0.5, cur_seq.mean() * 0.1)
        if delta <= 0.0:
            # Random jitter to escape possible stagnation
            cur_seq[i_high] = max(0.0,
                                 cur_seq[i_high] + np.random.randn() * cur_seq.mean() * 0.01)
            continue

        new_seq = cur_seq.copy()
        new_seq[i_high] -= delta
        new_seq[i_low] += delta
        new_seq = np.maximum(new_seq, 0.0)

        # Renormalise to the fixed ℓ₁‑norm (the scale is irrelevant for the objective)
        s = new_seq.sum()
        if s > 0:
            new_seq = new_seq * (target_sum / s)
        else:
            new_seq = np.full_like(new_seq, target_sum / n)

        new_val = _score(new_seq)

        # Accept only genuine improvements (occasional random acceptance helps exploring)
        if new_val < best_val:
            best_val = new_val
            best_seq = new_seq.copy()
            cur_seq = new_seq
        else:
            if random.random() < 0.01:          # tiny chance to accept a worse step
                cur_seq = new_seq

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point ------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list:
    """
    Search for a step‑function height sequence with a low evaluation score.
    The routine respects the time budget (default 1000 s) and returns the best
    sequence found as a list of non‑negative floats.
    """
    np.random.seed(seed)
    random.seed(seed)

    deadline = time.time() + budget_s - 0.5   # safety margin
    best_overall_seq = None
    best_overall_val = float("inf")

    # Lengths to explore – a mix of moderate and large sizes.
    candidate_lengths = [800, 1200, 1800, 2500, 3691]

    for n in candidate_lengths:
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)   # any scaling works; this choice matches the SOTA normalisation
        print(f"\n=== Length {n} (target sum = {target_sum:.4f}) ===")

        # ------------------------------------------------------------------
        # 1) Geometric family (a_i = alpha^i)
        best_alpha = None
        best_alpha_val = float("inf")
        alphas = np.linspace(0.5, 0.9999, 400)   # dense grid near 1
        for alpha in alphas:
            seq = _geometric_sequence(n, alpha, target_sum)
            val = _score(seq)
            if val < best_alpha_val:
                best_alpha_val = val
                best_alpha = alpha
                best_seq_geom = seq
        print(f"Geometric best  α={best_alpha:.6f}  score={best_alpha_val:.6f}")

        # ------------------------------------------------------------------
        # 2) Power‑law family (a_i = (i+1)^{-p})
        best_p = None
        best_p_val = float("inf")
        ps = np.linspace(0.0, 3.0, 301)
        for p in ps:
            seq = _powerlaw_sequence(n, p, target_sum)
            val = _score(seq)
            if val < best_p_val:
                best_p_val = val
                best_p = p
                best_seq_pow = seq
        print(f"Power‑law   best p={best_p:.3f}  score={best_p_val:.6f}")

        # ------------------------------------------------------------------
        # Choose the better of the two parametric starts
        if best_alpha_val <= best_p_val:
            start_seq = best_seq_geom
            start_val = best_alpha_val
            print(f"=> Selected geometric start (score {start_val:.6f})")
        else:
            start_seq = best_seq_pow
            start_val = best_p_val
            print(f"=> Selected power‑law start (score {start_val:.6f})")

        # ------------------------------------------------------------------
        # 3) Local refinement
        seq_refined, val_refined = _refine(start_seq, target_sum, deadline,
                                          max_iters=5000)
        print(f"Refined score = {val_refined:.6f}")

        # Keep the globally best construction
        if val_refined < best_overall_val:
            best_overall_val = val_refined
            best_overall_seq = seq_refined.copy()
            print(f"*** New global best!  length={n}  score={best_overall_val:.6f}")

        # Occasionally try a completely random restart (helps escaping local minima)
        if time.time() < deadline and random.random() < 0.2:
            rand_len = random.randint(600, 2000)
            target_sum_r = math.sqrt(2.0 * rand_len)
            rand_seq = np.random.exponential(scale=1.0, size=rand_len)
            rand_seq = rand_seq * (target_sum_r / rand_seq.sum())
            rand_seq, rand_val = _refine(rand_seq, target_sum_r, deadline,
                                         max_iters=2000)
            print(f"Random restart length={rand_len} final score={rand_val:.6f}")
            if rand_val < best_overall_val:
                best_overall_val = rand_val
                best_overall_seq = rand_seq.copy()
                print(f"*** New global best from random start!  length={rand_len}")

    # Return the best sequence found (as a plain Python list)
    if best_overall_seq is None:
        # Fallback – a uniform sequence (guaranteed to be feasible)
        n = 1000
        target_sum = math.sqrt(2.0 * n)
        best_overall_seq = np.full(n, target_sum / n)

    # Ensure the output is a list of floats (the evaluator expects a list)
    return best_overall_seq.tolist()
