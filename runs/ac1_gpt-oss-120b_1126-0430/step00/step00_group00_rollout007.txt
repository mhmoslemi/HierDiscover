import numpy as np, time, math

def evaluate(seq): # manual
    n = len(seq)
    conv = np.convolve(seq, seq)
    maxb = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * maxb / (sum_a * sum_a)

def compute_contributions(seq, max_indices):
    n = len(seq)
    contrib = np.zeros(n)
    for k in max_indices:
        low = max(0, k - (n-1))
        high = min(k, n-1)
        i = np.arange(low, high+1, dtype=int)
        j = k - i
        contrib[i] += seq[j]
    return contrib

def local_search(seq, budget_time, start_time):
    # iterative improvements until budget exceeded
    n = len(seq)
    current_seq = seq.copy()
    current_score = evaluate(current_seq)
    best_seq = current_seq.copy()
    best_score = current_score

    # steps
    step_factor = 0.1  # initial delta proportion
    # Temperature schedule maybe
    start_temp = 0.001
    # While time left
    while time.time() < budget_time:
        # compute convolution and max
        conv = np.convolve(current_seq, current_seq)
        maxb = conv.max()
        # compute indices of max within tolerance
        # tolerance relative
        tol = maxb * 1e-12 + 1e-12
        max_idx = np.where(conv >= maxb - tol)[0]

        contrib = compute_contributions(current_seq, max_idx)
        # pick i_max, i_min
        i_max = int(np.argmax(contrib))
        i_min = int(np.argmin(contrib))
        if current_seq[i_max] <= 1e-12:
            break
        delta = min(current_seq[i_max] * step_factor, 0.01)  # ensure small
        if delta <= 0:
            break
        new_seq = current_seq.copy()
        new_seq[i_max] -= delta
        new_seq[i_min] += delta
        # ensure nonneg
        if new_seq[i_max] < 0:
            new_seq[i_min] += new_seq[i_max]  # add whatever left
            new_seq[i_max] = 0.0

        new_score = evaluate(new_seq)
        # accept if better or with probability
        if new_score < current_score:
            current_seq = new_seq
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.copy()
                print(f"New best score {best_score:.6f}")
        else:
            # simulated annealing
            remaining = budget_time - time.time()
            progress = (budget_time - start_time - remaining) / (budget_time - start_time)
            temp = start_temp * (1 - progress)
            prob = math.exp(-(new_score - current_score)/ (temp + 1e-12))
            if np.random.rand() < prob:
                current_seq = new_seq
                current_score = new_score
        # adapt step factor
        step_factor = max(0.01, step_factor * 0.999)  # slowly decrease
    return best_seq, best_score

"""
Propose candidate sequence for the step‑function inequality.

Algorithm
---------
* Work with moderate sized sequences (≈200‑800 elements) – large enough to give
  flexibility but cheap enough to evaluate many iterations.
* Initialise a random Dirichlet (positive) vector and normalise its ℓ1‑norm.
* Repeatedly move a small amount of mass from the index that contributes most
  to the current maximal convolution value to the index that contributes the
  least.  This is a sub‑gradient step for the objective
      max_k (a * a)[k]   (the convolution peak).
* The move preserves the total sum, therefore the evaluation
      2·n·max(conv)/(sum²)  is reduced exactly when max(conv) is reduced.
* Accept improvements unconditionally; otherwise accept with a temperature‑
  dependent probability (simulated annealing) to escape plateaus.
* Perform many independent random restarts within the overall time budget,
  keep the best sequence found.
* The final sequence is clipped to the allowed range – the evaluator will
  re‑apply the same safety checks.
"""

import numpy as np
import time
import math

# ----------------------------------------------------------------------
# Utility functions (top‑level, no closures)
# ----------------------------------------------------------------------


def evaluate_score(seq: np.ndarray) -> float:
    """
    Fast manual evaluation of a sequence – matches evaluate_sequence for
    non‑negative, finite values that obey the clipping limits.
    """
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:                     # safeguard – same rule as the official evaluator
        return float("inf")
    return 2.0 * n * max_b / (sum_a * sum_a)


def max_convolution_indices(seq: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Indices where the convolution attains its maximum (within a tiny tolerance)."""
    max_val = conv.max()
    # tolerance protects against floating‑point round‑off
    tol = max_val * 1e-12 + 1e-12
    return np.where(conv >= max_val - tol)[0]


def contribution_vector(seq: np.ndarray, max_idx: np.ndarray) -> np.ndarray:
    """
    For every position i, compute the total contribution of a_i to all maximal
    convolution entries.  If max_idx contains the indices k where conv[k] is
    maximal, the contribution of a_i to those entries equals
        Σ_{k∈max_idx} a_{k-i}   (with out‑of‑range terms treated as zero).
    """
    n = seq.shape[0]
    contrib = np.zeros(n, dtype=float)

    for k in max_idx:
        # i must satisfy 0 ≤ i ≤ n‑1 and 0 ≤ k‑i ≤ n‑1  →  max(0,k-n+1) ≤ i ≤ min(k,n-1)
        lo = 0 if k < n else k - (n - 1)
        hi = k if k < n else n - 1
        i_range = np.arange(lo, hi + 1, dtype=int)
        j = k - i_range            # j = k‑i  is guaranteed to be in [0,n‑1]
        contrib[i_range] += seq[j]

    return contrib


def local_search(start_seq: np.ndarray, stop_time: float) -> tuple[np.ndarray, float]:
    """
    Perform a sub‑gradient hill‑climbing / simulated‑annealing descent on a single
    sequence until the supplied time limit is reached.
    Returns the best sequence encountered and its score.
    """
    seq = start_seq.copy()
    n = seq.shape[0]

    cur_score = evaluate_score(seq)
    best_seq = seq.copy()
    best_score = cur_score

    # step size as a fraction of the current mass at the source index
    step_fraction = 0.2
    # annealing schedule parameters
    start_temp = 5e-4

    while time.time() < stop_time:
        conv = np.convolve(seq, seq)
        max_idx = max_convolution_indices(seq, conv)

        # contribution of each entry to the current maximal convolution values
        contrib = contribution_vector(seq, max_idx)

        i_max = int(contrib.argmax())
        i_min = int(contrib.argmin())

        if i_max == i_min or seq[i_max] <= 1e-12:
            # nothing useful to move – terminate this run
            break

        # amount to shift from i_max to i_min
        delta = min(seq[i_max] * step_fraction, 0.01)   # keep moves tiny
        if delta <= 0.0:
            break

        new_seq = seq.copy()
        new_seq[i_max] -= delta
        new_seq[i_min] += delta

        # ensure non‑negativity (should already hold)
        new_seq = np.maximum(new_seq, 0.0)

        new_score = evaluate_score(new_seq)

        # accept improvements always
        if new_score < cur_score:
            seq = new_seq
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.copy()
                print(f"[{time.strftime('%H:%M:%S')}]  New local best: {best_score:.6f}")

        else:
            # occasional uphill moves (simulated annealing)
            elapsed = stop_time - time.time()
            progress = (stop_time - elapsed) / (stop_time - start_time_global)
            temp = start_temp * (1.0 - progress)
            prob = math.exp(-(new_score - cur_score) / (temp + 1e-12))
            if np.random.rand() < prob:
                seq = new_seq
                cur_score = new_score

        # slowly decay the step size
        step_fraction = max(0.01, step_fraction * 0.999)

    return best_seq, best_score

    """
    global start_time_global
    start_time_global = time.time()
    np.random.seed(seed)

    deadline = start_time_global + budget_s - 5.0

    except Exception:

    best_seq = init_seq.copy()

    # ------------------------------------------------------------------

    while time.time() < deadline:

                best_score = local_score2
                best_seq = local_best2.copy()
                print(f"[{time.strftime('%H:%M:%S')}]  Global best from perturbation: {best_score:.6f}")

    # Final safety clipping (evaluate_sequence applies the same rules)
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    return final_seq.tolist()
