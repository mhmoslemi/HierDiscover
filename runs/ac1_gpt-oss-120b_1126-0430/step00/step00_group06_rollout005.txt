Implementation of projection onto simplex:

def simplex_projection(v):
    """Projects vector v onto simplex (nonnegative sum=1)."""
    # based on algorithm by Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

import numpy as np
import time

def simplex_projection(v):
    """Project vector v onto the probability simplex (sum=1, v>=0)."""
    # Duchi et al. algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0]
    if len(rho) == 0:
        # all zero, return uniform
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def convolve_fft(x, y):
    """Convolution of two 1‑D real arrays using FFT, returns full length."""
    L = len(x) + len(y) - 1

    fy = np.fft.rfft(y, N)

import time
import numpy as np

# --------------------------------------------------------------
# Helper functions (top‑level as required)
# --------------------------------------------------------------

def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the probability simplex:
        {x | x_i >= 0,  Σ_i x_i = 1 }.
    Implementation of the algorithm by Duchi et al. (ICML 2008).
    """
    if np.all(v == 0):
        # uniform fallback – never happens in practice
        return np.full_like(v, 1.0 / len(v))

    u = np.sort(v)[::-1]                     # descending sort
    cssv = np.cumsum(u) - 1.0                # cumsum minus 1
    ind = np.arange(len(v)) + 1.0
    # Find the last index where the condition holds
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero – return uniform
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full convolution of two 1‑D real arrays using FFT.
    Returns an array of length len(x)+len(y)-1.
    """
    L = len(x) + len(y) - 1
    N = 1 << (L - 1).bit_length()           # next power of two ≥ L
    fx = np.fft.rfft(x, N)
    fy = np.fft.rfft(y, N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# --------------------------------------------------------------

# --------------------------------------------------------------

def propose_candidate(seed: int = 0, budget_s: float = 1000, **_ignored) -> list:
    """
    Generate a non‑negative step‑function (list of heights) that
    minimises the evaluation function:

        2 * n * max(conv(a,a)) / (sum(a)^2)

    The algorithm performs several random restarts of a projected
    gradient descent that smoothes the “max” with a soft‑max
    (log‑sum‑exp) and projects back onto the simplex (sum = 1,
    a_i ≥ 0).  The search stops when the time budget expires and
    returns the best sequence encountered.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # lengths we will try – include the SOTA length if it exists
    # ------------------------------------------------------------------
    lengths = [200, 500, 1000, 2000, 3000]
    try:
        lengths.append(len(height_sequence_1))   # global from the environment
    except Exception:
        pass

    attempt = 0
    while time.time() < deadline:
        attempt += 1
        n = lengths[attempt % len(lengths)]

        # ------------------------------------------------------------------
        # Random initial point on the simplex (sum == 1, a_i >= 0)
        # ------------------------------------------------------------------
        a = np.random.rand(n)
        a = simplex_projection(a)

        # Evaluate the very first point (helps if the uniform sequence is already good)
        cur_score = evaluate_sequence(a.tolist())
        if cur_score < best_score:

            best_seq = a.tolist()

        # ------------------------------------------------------------------
        # Hyper‑parameters (they are annealed during the run)
        # ------------------------------------------------------------------
        eta = 0.2            # learning rate
        gamma = 1.0          # smooth‑max temperature
        max_iters = int(min(5000, (deadline - time.time()) * 200))
        # ------------------------------------------------------------------
        # Main projected gradient loop
        # ------------------------------------------------------------------
        for it in range(max_iters):
            if time.time() > deadline:
                break

            # 1) self‑convolution a*a (full)
            b = convolve_fft(a, a)                 # length 2n‑1

            # 2) soft‑max weights that concentrate on large entries of b
            max_b = np.max(b)
            exp_arg = gamma * (b - max_b)          # shift for numerical stability

    # ------------------------------------------------------------------
