"""
Propose a candidate step‑function sequence that minimizes the evaluation metric.

The algorithm explores several sequence lengths (2‑12) and uses a combination of
global optimisation (Differential Evolution on the log‑space of the coefficients)
and a cheap hill‑climbing refinement.  Each candidate is normalised to unit sum
(the evaluation is scale‑invariant) and the best sequence seen so far is kept.
The routine respects the 1000 s time budget and prints progress information.
"""

import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def _eval_ratio(seq):
    """
    Wrapper around the provided evaluate_sequence that returns the score.
    """
    # The security checks inside evaluate_sequence already enforce validity.
    return evaluate_sequence(seq)


def _de_optimize(length, rng, deadline):
    """
    Global optimisation for a fixed length using Differential Evolution.

    Parameters
    ----------
    length : int
        Length of the sequence to optimise.
    rng : np.random.Generator
        Random generator for reproducibility.
    deadline : float
        Unix time after which the optimisation must stop.

    Returns
    -------
    best_seq : list[float]
        The best normalised sequence found (sum == 1.0).
    best_score : float
        Corresponding evaluation value.
    """
    # Objective works in log‑space to enforce positivity.
    def objective(log_vals):
        a = np.exp(log_vals)               # positive
        s = a.sum()
        if s < 1e-12:                       # guard against degenerate case
            return np.inf
        a /= s                             # normalise to sum == 1
        return _eval_ratio(a.tolist())

    bounds = [(-8.0, 8.0)] * length          # broad enough for any shape

    # Callback that stops DE when we run out of time.
    def stop_callback(xk, convergence):
        return time.time() >= deadline

    # Run the optimiser.  polish=False because we will do our own local search.
    result = optimize.differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=200,
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1.0),
        recombination=0.7,
        seed=int(rng.integers(0, 2**32 - 1)),
        callback=stop_callback,
        polish=False,
        disp=False,
        init='latinhypercube',
    )

    # Decode the best candidate.
    best_log = result.x
    best_a = np.exp(best_log)
    best_a /= best_a.sum()
    best_score = result.fun
    return best_a.tolist(), best_score


def _hill_climb(seq, rng, deadline, max_iters=5000):
    """
    Simple stochastic hill‑climbing that moves a small amount of mass between
    two random positions while preserving the total sum.

    Parameters
    ----------
    seq : list[float]
        Starting sequence (already normalised).
    rng : np.random.Generator
        Random generator.
    deadline : float
        Time after which the climb must stop.
    max_iters : int, optional
        Upper bound on iteration count (soft, the real stop is the deadline).

    Returns
    -------
    best_seq : list[float]
        Improved sequence (may be identical to the input).
    best_score : float
        Its evaluation value.
    """
    arr = np.array(seq, dtype=float)
    best_score = _eval_ratio(arr.tolist())
    n = len(arr)

    for _ in range(max_iters):
        if time.time() >= deadline:
            break
        i, j = rng.choice(n, size=2, replace=False)
        # move a modest fraction of the source entry to the target
        delta = arr[i] * rng.uniform(0.01, 0.05)
        if delta <= 0:
            continue
        new_arr = arr.copy()
        new_arr[i] -= delta
        new_arr[j] += delta
        # ensure non‑negativity after numerical noise
        if new_arr[i] < 0:
            continue
        score = _eval_ratio(new_arr.tolist())
        if score < best_score:
            best_score = score
            arr = new_arr
    return arr.tolist(), best_score
