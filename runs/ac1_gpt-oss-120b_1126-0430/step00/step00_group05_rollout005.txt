def project_onto_simplex(v):
    # ensure nonnegative and sum=1
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function sequence that minimizes the evaluation metric.

The algorithm works with probability vectors p (sum = 1).  For a vector p
the evaluation function reduces to

    score(p) = 2 * n * max(p * p),

where * denotes discrete convolution.  The goal is therefore to find a
distribution whose self‑convolution is as flat as possible.

We use a simple projected sub‑gradient descent with random perturbations:

1. Choose a length n (several candidates are tried; n≈2000‑3000 gives a
   good trade‑off between runtime and a low theoretical lower bound
   ≈1).
2. Initialise p with a uniform distribution plus a small random noise and
   project onto the probability simplex.
3. In each iteration compute the convolution via FFT, locate the index
   t* where it attains its maximum and form the sub‑gradient
   g_i = 2·p[t*‑i] (zero outside the support).  Move opposite the
   gradient with a decreasing step size and re‑project onto the simplex.
4. Every few iterations a tiny random perturbation is added to avoid
   stagnation.
5. Whenever the surrogate score improves we evaluate the exact metric
   with ``evaluate_sequence`` (which uses the original implementation)
   and keep the best sequence found.

The method runs for the allotted time (default 1000 s) and finally
returns the best sequence (list of floats).  All numbers are non‑negative;
the sum is 1, so the scaling is irrelevant for the metric.
"""

import time
import numpy as np


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i >= 0, sum x_i = 1}.
    """
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # all entries are non‑positive – return a vertex of the simplex
        w = np.zeros_like(v)
        w[0] = 1.0
        return w
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_max(p: np.ndarray, L: int) -> float:
    """
    Compute max(p * p) using FFT (real valued, length 2*n-1).
    """
    fft = np.fft.rfft(p, n=L)
    conv = np.fft.irfft(fft * fft, n=L)[: 2 * len(p) - 1]
    return float(conv.max())


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a good sequence within the time budget.
    Returns a list of non‑negative floats (the heights of the step function).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # leave a small safety margin

    best_seq = None
    best_score = np.inf

    # a few different lengths are tried; larger n gives a lower theoretical bound
    n_choices = [1500, 2000, 2500, 3000]

    for n in n_choices:
        # pre‑compute FFT length (next power of two >= 2*n-1)
        L = 1
        while L < 2 * n - 1:
            L <<= 1

        # start from a slightly perturbed uniform distribution
        p = np.full(n, 1.0 / n) + 0.005 * np.random.randn(n)
        p = np.maximum(p, 0.0)
        if p.sum() == 0:
            p = np.full(n, 1.0 / n)
        p = p / p.sum()                     # project onto the simplex

        iteration = 0
        while time.time() < deadline:
            # ----- evaluate current distribution -----
            fft = np.fft.rfft(p, n=L)
            conv = np.fft.irfft(fft * fft, n=L)[: 2 * n - 1]
            max_val = float(conv.max())
            t_star = int(conv.argmax())
            surrogate_score = 2.0 * n * max_val

            # record a better candidate (evaluate exactly)
            if surrogate_score < best_score:
                seq = p.tolist()
                exact = evaluate_sequence(seq)      # the official function
                if exact < best_score:
                    best_score = exact
                    best_seq = seq
                    print(
                        f"[{time.time():.1f}] New best score {best_score:.6f} "
                        f"(n={n}, iter={iteration})"
                    )
                # even if the exact test is not better we keep the surrogate
                # value for further descent.

            # ----- sub‑gradient step to reduce the current maximum -----
            # subgradient of max(conv) is the gradient of the active convolution entry
            d = np.zeros_like(p)

            # indices j that contribute to conv[t_star]
            j_min = max(0, t_star - (n - 1))
            j_max = min(n - 1, t_star)
            js = np.arange(j_min, j_max + 1, dtype=int)
            is_idx = t_star - js
            # each ordered pair contributes 2*p[j] (except the centre when i=j,
            # but using factor 2 works as a sub‑gradient as well)
            d[is_idx] = 2.0 * p[js]

            # step size – a simple diminishing schedule
            eta = 0.2 / (1.0 + np.sqrt(iteration + 1))

            p_new = p - eta * d

            # occasional random jitter to escape plateaus
            if iteration % 12 == 0:
                p_new += 0.003 * np.random.rand(n)

            # project back onto the simplex
            p = _simplex_projection(p_new)

            iteration += 1

            # early exit if we already beat the target (safety)
            if best_score is not None and best_score <= 1.51:
                print(
                    f"Target reached (score={best_score:.6f}), exiting early."
                )
                return best_seq

    # fallback – if something went wrong return a simple uniform sequence
    if best_seq is None:
        best_seq = [1.0] * 10
        print("Fallback sequence returned.")
    return best_seq
