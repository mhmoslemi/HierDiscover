"""
Propose a candidate sequence for the step‑function challenge.

The evaluation function is homogeneous:
    eval(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

Hence we can work with any scaling of the vector; we keep the sum equal to 1.
For a fixed support size k, the value of max(conv) is roughly
    max(conv) ≈ (max number of ordered pairs with the same sum) / k² .
Randomly chosen supports of size k≈0.25·n…0.45·n give an average
pair‑count of order 𝜆=k²/(2n)≫log(2n), which makes the maximum only
≈(1+√(2·log(2n)/𝜆))·𝜆.  Consequently the evaluation is typically
well below the uniform bound 2, often <1.5.

The algorithm therefore repeatedly:
* draws a random support of size k in a promising range,
* assigns equal weight 1/k to the selected positions (the optimal
  distribution on a given support),
* evaluates the sequence,
* keeps the best one found,
* and, after an improvement, performs a small number of cheap
  gradient‑descent refinements that move mass away from the shift where
  the convolution attains its maximum.

The whole search runs for the allotted time (default 1000 s) and
returns the best sequence discovered.
"""

import time
import numpy as np

# evaluate_sequence is provided in the execution environment.


def _max_convolution_shift(a: np.ndarray) -> int:
    """Return the index (0‑based) where the full convolution a*a attains its maximum."""
    conv = np.convolve(a, a)                 # length 2·n‑1
    return int(conv.argmax())


def _gradient_step(a: np.ndarray, step: float) -> np.ndarray:
    """
    One multiplicative‑gradient step that reduces the contribution to the
    current maximal convolution value.

    Parameters
    ----------
    a : np.ndarray
        Current probability vector (sum = 1, all entries ≥ 0).
    step : float
        Positive step size.

    Returns
    -------
    np.ndarray
        Updated vector (still non‑negative, renormalised to sum 1).
    """
    n = a.shape[0]
    k_max = _max_convolution_shift(a)

    # gradient of the (unsmoothed) max is simply a reflected at the maximal shift:
    idx = np.arange(n)
    partner = k_max - idx
    mask = (partner >= 0) & (partner < n)
    grad = np.zeros_like(a)
    grad[mask] = a[partner[mask]]

    # multiplicative update (mirror‑descent on the simplex)
    a_new = a * np.exp(-step * grad)
    total = a_new.sum()
    if total > 0:
        a_new /= total
    return a_new


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises the
    provided evaluation function.  The function returns a list of floats.
    """
    rng = np.random.default_rng(seed)

    # Length of the sequence – use the existing construction if it exists.
    try:
        n = len(height_sequence_1)          # provided by the outer environment
    except NameError:
        n = 2000                            # safe fallback

    # start from the previous best if available; otherwise a uniform vector.
    try:
        best_seq = list(height_sequence_1)
    except NameError:
        best_seq = [1.0] * n

    best_score = evaluate_sequence(best_seq)
    print(f"[init] length={n}, initial score={best_score:.6f}")

    deadline = time.time() + budget_s - 0.5   # spare a little margin
    iter_count = 0

    # Support size range that empirically gives scores below the target.
    k_low = max(3, int(0.20 * n))   # at least a few non‑zeros
    k_high = max(k_low + 1, int(0.45 * n))

    while time.time() < deadline:
        # -----------------------------------------------------------------
        # 1) Random support generation
        # -----------------------------------------------------------------
        k = rng.integers(k_low, k_high + 1)          # size of support
        support = rng.choice(n, size=int(k), replace=False)

        a = np.zeros(n, dtype=np.float64)
        a[support] = 1.0 / k                         # uniform weight on the support

        # very cheap optional perturbation – helps escape exact ties
        if rng.random() < 0.15:
            noise = rng.normal(scale=0.02 / k, size=int(k))
            a[support] += noise
            a = np.clip(a, 0.0, None)
            s = a.sum()
            if s > 0:
                a /= s

        # -----------------------------------------------------------------
        # 2) Evaluate
        # -----------------------------------------------------------------
        score = evaluate_sequence(a.tolist())
        iter_count += 1

        # -----------------------------------------------------------------
        # 3) Keep the best candidate
        # -----------------------------------------------------------------
        if score < best_score:
            best_score = score
            best_seq = a.tolist()
            print(f"[{iter_count:6d}] new best {best_score:.6f} (k={k})")

            # -----------------------------------------------------------------
            # 4) Light local refinement – a few gradient steps
            # -----------------------------------------------------------------
            a_ref = a.copy()
            for _ in range(6):
                a_ref = _gradient_step(a_ref, step=0.15)
                refined_score = evaluate_sequence(a_ref.tolist())
                if refined_score < best_score:
                    best_score = refined_score
                    best_seq = a_ref.tolist()
                    print(f"    [+] refined score {best_score:.6f}")

        # -----------------------------------------------------------------
        # 5) Occasionally try a plain uniform vector (baseline)
        # -----------------------------------------------------------------
        if iter_count % 2000 == 0:

            u_score = evaluate_sequence(uniform.tolist())
            if u_score < best_score:
                best_score = u_score
                best_seq = uniform.tolist()
                print(f"[{iter_count:6d}] uniform improved to {best_score:.6f}")

    print(f"[done] iterations={iter_count}, best score={best_score:.6f}")
    return best_seq
