import numpy as np

def project_to_simplex(v, s=1.0):
    """Projection of v onto simplex {x>=0, sum x = s}."""
    n = len(v)
    if s < 0:
        raise ValueError("s must be non-negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero
        return np.zeros_like(v)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

import numpy as np
import time

def project_to_simplex(v, s=1.0):
    """Projects a real vector onto the simplex {x ≥ 0, sum x = s}."""
    n = len(v)
    if s < 0:
        raise ValueError("s must be non‑negative")
    u = np.sort(v)[::-1]                # sorted in descending order
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will become zero
        return np.zeros_like(v)
    rho = ind[cond][-1]                 # last true index
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


def flatten_conv_descent(n, max_iter=300, time_deadline=None):
    """
    A tiny gradient‑descent routine that tries to flatten the
    self‑convolution of a probability vector.
    Returns the best sequence found and its (approximate) score.
    """
    # start from a random point on the simplex (sum = 1)
    p = np.random.dirichlet(np.ones(n))
    best_seq = p.copy()
    best_score = 2 * n * np.max(np.convolve(p, p))   # sum(p)=1 ⇒ eval = 2·n·max_b

    for it in range(max_iter):
        if time_deadline is not None and time.time() >= time_deadline:
            break

        conv = np.convolve(p, p)
        cur_score = 2 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = p.copy()
            print(f"[GD] n={n}, iter={it}, score={cur_score:.6f}")

        diff = conv - target
        # gradient of L2 loss ½‖conv−target‖² w.r.t. p (valid‑mode convolution)
