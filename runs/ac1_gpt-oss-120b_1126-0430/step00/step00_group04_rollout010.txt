def project_to_simplex(v):
    # returns w >= 0 such that sum w = 1, close to v
    # using algorithm of Wang & Carreira-Perpinan (2013)
    u = sorted(v, reverse=True)
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑height sequence by stochastic optimisation.

The function works with the *probability* interpretation of a step
function:  a_i ≥ 0,   Σ a_i = 1.  For such a vector the score returned
by ``evaluate_sequence`` is

    score = 2·n·max_k (a * a)[k]               (1)

where ``*`` denotes the discrete linear convolution.  The goal is to
minimise the peak of the self‑convolution.  The algorithm performs a
simulated‑annealing / random‑walk on the simplex:

* start from a random Dirichlet distribution (many almost‑zero entries);
* repeatedly move a small amount of mass from a “donor’’ index to a
  “receiver’’ index;
* the donor is chosen preferentially among the positions that contribute
  the most to the current maximal convolution entry;
* moves that lower the score are always accepted, otherwise they are
  accepted with a Metropolis probability that depends on a temperature
  that slowly cools down;
* the best‐ever sequence is remembered and returned at the end of the
  allotted time.

All heavy work (the convolution) is performed with FFTs, so each
iteration costs O(n log n) and many thousands of iterations fit easily
into the 1000 s budget.
"""

import time
import math
import numpy as np

def _fft_convolve_self(p: np.ndarray) -> np.ndarray:
    """Return the linear convolution of p with itself using an FFT."""
    n = p.shape[0]
    size = 1 << ((2 * n - 1).bit_length())        # next power of two
    fft = np.fft.rfft(p, n=size)
    conv = np.fft.irfft(fft * fft, n=size)[:2 * n - 1]
    return conv

        limit to avoid a forced termination.

        The best sequence found (the values sum to roughly 1.0).
    """

    # sequence is unavailable.
    try:
        n = len(height_sequence_1)          # provided by the judging environment
    except Exception:                     # pragma: no cover
        n = 2000

    deadline = time.time() + budget_s - 5.0   # leave a small safety margin

    # ---------- helper functions ----------
    def score(p: np.ndarray) -> float:
        """Score that matches evaluate_sequence for a unit‑sum vector."""
        conv = _fft_convolve_self(p)
        return 2.0 * n * np.max(conv)

    # ---------- initial distribution ----------
    # Dirichlet with a small concentration gives many tiny entries (good
    # for load‑balancing) while still respecting the simplex.
    alpha = 0.5
    p = np.random.dirichlet(np.full(n, alpha))
    curr_score = score(p)

    best_p = p.copy()
    best_score = curr_score

    # ---------- annealing parameters ----------
    T = 0.5            # initial temperature
    T_min = 1e-5
    decay = 0.9995     # geometric cooling
    move_frac = 0.25   # maximal fraction of donor mass that may be moved
    stagnation = 0
    iter_cnt = 0

    print(f"Search started – n={n}, initial score={curr_score:.6f}")

    # ---------- main optimisation loop ----------
    while time.time() < deadline:
        # current convolution and the index where it attains its maximum
        conv = _fft_convolve_self(p)
        max_idx = int(np.argmax(conv))
        max_val = conv[max_idx]
        curr_score = 2.0 * n * max_val

        # ----- choose a donor index -----
        # Contributions of each index to the current worst convolution entry:
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_range = np.arange(i_start, i_end + 1, dtype=int)

        contrib = p[idx_range] * p[max_idx - idx_range]   # a_i * a_{max-i}
        total_contrib = contrib.sum()

        if total_contrib > 0 and np.random.rand() < 0.85:
            # directed choice – pick a heavy contributor with probability ∝ contribution
            probs = contrib / total_contrib
            donor = np.random.choice(idx_range, p=probs)
        else:
            # fallback – pick any index that currently holds a non‑negligible mass
            heavy = np.where(p > 1e-12)[0]
            donor = heavy[np.random.randint(len(heavy))]

        # ----- pick a receiver (different from donor) -----
        receiver = np.random.randint(n - 1)
        if receiver >= donor:
            receiver += 1

        # ----- propose a move -----
        delta = p[donor] * move_frac * np.random.rand()   # mass to move
        if delta <= 0.0:
            # nothing to move – skip this iteration
            continue

        # Apply the move in‑place (will be reverted if not accepted)
        p[donor] -= delta
        p[receiver] += delta

        # ----- evaluate the new configuration -----
        conv_new = _fft_convolve_self(p)
        new_score = 2.0 * n * np.max(conv_new)

        # ----- Metropolis acceptance test -----
        accept = False
        if new_score < curr_score:
            accept = True
        else:
            # uphill move – accept with probability exp(-(Δ)/T)
            if T > 0.0:
                prob = math.exp(-(new_score - curr_score) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            # keep the move
            curr_score = new_score
            stagnation = 0
            if new_score < best_score:
                best_score = new_score
                best_p = p.copy()
                print(f"Iter {iter_cnt:6d}: new best {best_score:.6f}")
        else:
            # reject – revert the change
            p[donor] += delta
            p[receiver] -= delta
            stagnation += 1

        # ----- temperature schedule -----
        T = max(T_min, T * decay)

        # ----- occasional random restart to escape deep plateaus -----
        if stagnation > 5000:
            p = np.random.dirichlet(np.full(n, alpha))
            curr_score = score(p)
            stagnation = 0
            T = max(T_min, T)   # keep the current temperature
            print(f"Iter {iter_cnt:6d}: random restart, score={curr_score:.6f}")

        iter_cnt += 1

    print(f"Search finished – iterations={iter_cnt}, best score={best_score:.6f}")
    # Return the best sequence as a plain Python list
    return best_p.tolist()
