def propose_candidate(seed=0, budget_s=1000, **kwargs):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1
    # Determine length n maybe 5000 (or variable)
    n = 5000  # we can set to 5000 from height_sequence_1 length maybe 4587; we can keep that.
    # Use length of height_sequence_1 if exists as fallback:
    try: n = len(height_sequence_1)
    except NameError: pass

    # sum target
    S = np.sqrt(2 * n)
    # initialize a
    # random proportions from Dirichlet
    a = np.random.rand(n)
    a = a / a.sum() * S

    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    # main loop
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # compute conv via FFT
        size = 1
        while size < 2 * n:
            size <<= 1
        # compute conv with numpy FFT
        fa = np.fft.rfft(a, size)
        conv = np.fft.irfft(fa * fa, size)[:2 * n - 1]
        max_val = conv.max()
        max_idx = np.where(np.isclose(conv, max_val, atol=1e-12))[0]

        # compute gradient for these max indices
        grad = np.zeros_like(a)
        for k in max_idx:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            i_range = np.arange(i_start, i_end + 1)
            partner = k - i_range
            # accumulate 2*a[partner]
            grad[i_range] += 2.0 * a[partner]
        # Project gradient to hyperplane sum=0
        grad_mean = grad.mean()
        grad_proj = grad - grad_mean

        # Compute step size
        # Choose initial eta relative to max_val or based on norm.
        eta = 0.1 * S / (np.linalg.norm(grad_proj) + 1e-12)  # small step
        # Ensure we don't go negative
        # compute max feasible scaling factor
        pos = grad_proj > 0
        if np.any(pos):
            eta_max = np.min(a[pos] / grad_proj[pos])
            eta = min(eta, 0.9 * eta_max)
        # Update a
        a_new = a - eta * grad_proj
        # Clip to nonnegative
        a_new = np.maximum(a_new, 0.0)
        # Renormalize sum to S
        sum_a = a_new.sum()
        if sum_a > 0:
            a_new = a_new * (S / sum_a)
        else:
            # reinitialize random
            a_new = np.random.rand(n)
            a_new = a_new / a_new.sum() * S

        a = a_new

        # Evaluate
        score = evaluate_sequence(a.tolist())
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"Iter {iteration}: improved score {score:.6f}")

        # Periodically add small random perturbation to escape stagnation
        if iteration % 200 == 0:
            # add noise
            noise = np.random.randn(n) * (0.01 * S / n)
            a = a + noise
            a = np.maximum(a, 0.0)
            a = a / a.sum() * S

    return best_seq.tolist()

"""
Propose a sequence of non‑negative heights that minimizes the
`evaluate_sequence` objective.

The idea is analytic:  for a sequence of length n with fixed sum
S = sqrt(2*n) the score equals the maximal value of the auto‑
convolution a*a.  The constant sequence gives a maximum of 2.  A
simple decreasing linear shape

        a_i = A·(n-i) ,   i = 0,…,n‑1

with A = 2·S / (n·(n+1))

has a closed‑form convolution peak
        max(a*a) = 4·(n+2) / (3·(n+1)) ≈ 4/3   (< 1.503 for any n>1)

which already beats the target.  The function generates a few
parameterised families (linear, power‑law, exponential), keeps the
best one found, and performs a tiny random walk to possibly improve
it a little more within the allotted time.

The returned list is the best sequence discovered.
"""

import numpy as np
import time
from typing import List

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs) -> List[float]:
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.1   # a bit of safety margin

    # ------------------------------------------------------------------
    # Determine the length of the sequence.
    # Prefer the length of the known construction if available.
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)               # type: ignore[name-defined]
    except Exception:
        n = 4587                                   # fallback length from the statement

    # Fixed target sum (normalisation used in the original paper)
    S_target = np.sqrt(2.0 * n)

    best_seq: List[float] = []
    best_score = np.inf

    # ------------------------------------------------------------------
    # Helper to evaluate a candidate and keep the best one.
    # ------------------------------------------------------------------
    def try_sequence(seq: np.ndarray) -> None:
        nonlocal best_seq, best_score
        try:
            score = evaluate_sequence(seq.tolist())   # evaluate_sequence is provided by the runner
        except Exception:
            return
        if score < best_score:
            best_score = score
            best_seq = seq.tolist()
            print(f"[{time.time()-start:6.2f}s] New best score: {score:.6f}")

    # ------------------------------------------------------------------
    # 1. Linear decreasing shape (the analytically optimal simple family)
    # ------------------------------------------------------------------
    A = 2.0 * S_target / (n * (n + 1))
    lin_seq = np.arange(n, 0, -1, dtype=np.float64) * A
    try_sequence(lin_seq)

    # ------------------------------------------------------------------
    # 2. Power‑law families: a_i = (n-i)^p  (p>0).  After normalisation they
    #    keep the same total sum.
    # ------------------------------------------------------------------
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        raw = np.arange(n, 0, -1, dtype=np.float64) ** p
        seq = raw / raw.sum() * S_target
        try_sequence(seq)

    # ------------------------------------------------------------------
    # 3. Exponential decay families: a_i = r^i .
    # ------------------------------------------------------------------
    for r in (0.9999, 0.9995, 0.998, 0.995, 0.99, 0.98):
        raw = r ** np.arange(n, dtype=np.float64)
        seq = raw / raw.sum() * S_target
        try_sequence(seq)

    # ------------------------------------------------------------------
    # 4. Small stochastic refinement around the best sequence found so far.
    #    Simple random “mass transfer” moves keep the total sum constant.
    # ------------------------------------------------------------------
    if best_seq:
        cur = np.array(best_seq, dtype=np.float64)
    else:
        cur = lin_seq.copy()

    while time.time() < deadline:
        # pick two distinct indices and transfer a tiny amount
        i, j = np.random.choice(n, size=2, replace=False)
        # magnitude of transfer – proportional to the average value
        eps = np.random.randn() * S_target * 0.0005 / n
        if eps == 0.0:
            continue
        if cur[i] - eps < 0:
            continue
        cur[i] -= eps
        cur[j] += eps

        # re‑normalise to the exact target sum (scaling does not change the score)
        cur_sum = cur.sum()
        if cur_sum <= 0.0:
            continue
        cur *= (S_target / cur_sum)

        try_sequence(cur)

        # occasional random restart of the perturbation to explore new basins
        if np.random.rand() < 0.001:
            cur = lin_seq.copy()
            cur += np.random.randn(n) * S_target * 0.001 / n
            cur = np.maximum(cur, 0.0)
            cur *= (S_target / cur.sum())

    # ------------------------------------------------------------------
    # Fallback – if somehow nothing was accepted, return the uniform vector.
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [S_target / n] * n
        print("Fallback to uniform sequence.")
    else:
        print(f"Finished. Best score = {best_score:.6f}")

    return best_seq
