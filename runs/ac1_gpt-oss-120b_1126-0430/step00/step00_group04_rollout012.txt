The projection onto simplex:

def simplex_projection(v):
    """Project vector v onto probability simplex (sum=1, v>=0)."""
    # shift v? We'll use sorting method.
    n = v.shape[0]
    u = np.sort(v)[::-1]  # descending
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Gradient‐flattening search for a step function with a small convolution bound.

The objective evaluated by ``evaluate_sequence`` is
    2·n·max(conv(a, a)) / (sum(a)**2)

Because the ratio is scale‑invariant we can fix sum(a)=1.
For a fixed sum the average value of the convolution is 1/(2n‑1),
so the lower bound for the objective is ≈2n/(2n‑1)≈1.
If all convolution entries were equal to their average we would achieve
the bound.  The search therefore tries to flatten the convolution:
it minimises the variance of conv(a,a) while keeping a non‑negative
and on the probability simplex (sum=1).  The gradient of the variance
has a simple closed form that can be evaluated with FFTs.
A projected (sub‑)gradient descent with a simple back‑tracking line
search is performed, together with occasional random perturbations
to escape plateaus.  The best sequence found before the time budget
expires is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex (non‑negative, sum = 1)
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex {x >= 0, sum(x)=1}."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
def _compute_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT of length fft_len (>= 2*len(a)-1)."""
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)
    return conv[: 2 * a.shape[0] - 1]

def _compute_correlation(r: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (r ⋆ a)[t] = Σ_i a_i * r_{i+t}
    for t = 0 … len(a)-1, using FFT of length fft_len (>= len(r)+len(a)-1).
    """
    A = np.fft.rfft(a, n=fft_len)
    R = np.fft.rfft(r, n=fft_len)
    corr = np.fft.irfft(np.conj(A) * R, n=fft_len)
    return corr[: a.shape[0]]

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises the
    evaluation function.  The routine runs for ``budget_s`` seconds (leaving
    a small safety margin) and returns the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # a tiny safety margin

    # ------------------------------------------------------------------
    # Initialise a – use the known construction if available, otherwise random
    if 'height_sequence_1' in globals():
        init = np.array(height_sequence_1, dtype=np.float64)
        if init.size == 0:
            init = np.random.rand(2000)
    else:
        init = np.random.rand(2000)

    n = init.shape[0]

    # Normalise to a probability vector (sum = 1)
    init_sum = init.sum()
    if init_sum <= 0:
        init = np.random.rand(n)
        init_sum = init.sum()
    a = init / init_sum

    # Pre‑compute FFT lengths (next powers of two)
    conv_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    corr_len = 1 << int(np.ceil(np.log2(3 * n - 2)))

    # constant average of the convolution for sum(a)=1
    avg_const = 1.0 / (2 * n - 1)

    # keep the best solution seen so far
    def _score(x):
        """Exact objective for sum(x)=1."""
        conv = _compute_convolution(x, conv_len)
        return 2.0 * n * conv.max()

    best_a = a.copy()
    best_score = _score(best_a)
    print(f"Initial score: {best_score:.6f}")

    it = 0
    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1) convolution and variance gradient
        conv = _compute_convolution(a, conv_len)                 # length 2n‑1
        max_b = conv.max()
        # current objective (sum(a) = 1)
        cur_score = 2.0 * n * max_b

        # residual from the target average
        resid = conv - avg_const

        # gradient of variance = 4 * (resid ⋆ a)
        grad = 4.0 * _compute_correlation(resid, a, corr_len)

        # --------------------------------------------------------------
        # 2) line search for a suitable step size
        #   we move opposite to the gradient (gradient descent)
        step = 0.5 / (np.linalg.norm(grad) + 1e-12)   # initial magnitude
        improved = False
        for _ in range(7):
            a_candidate = a - step * grad
            a_candidate = _simplex_projection(a_candidate)

            cand_score = _score(a_candidate)
            if cand_score < cur_score:
                a = a_candidate
                cur_score = cand_score
                improved = True
                break
            step *= 0.5          # backtrack

        # --------------------------------------------------------------
        # 3) bookkeeping
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"Iter {it:5d} – new best: {best_score:.6f}")

        # --------------------------------------------------------------
        # 4) occasional random perturbation to escape flat regions
        if not improved and (it % 200 == 0):

            i, j = np.random.randint(n, size=2)
            delta = (np.random.rand() - 0.5) * 0.02
            a[i] = max(0.0, a[i] + delta)
            a[j] = max(0.0, a[j] - delta)

        # --------------------------------------------------------------
        # 5) occasional restart around the best found point
        if it % 4000 == 0 and it > 0:
            a = best_a + np.random.randn(n) * 0.005
            a = np.maximum(a, 0.0)
            a = _simplex_projection(a)

        it += 1

    # ------------------------------------------------------------------
    # Return the best sequence (as a plain Python list of floats)
    # The values are already non‑negative; no further clipping is needed.
    result = [float(x) for x in best_a.tolist()]
    print(f"Finished after {it} iterations – best score: {best_score:.6f}")
    return result
