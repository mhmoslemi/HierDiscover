```
def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a candidate step‑function sequence using a conditional‑gradient (Frank‑Wolfe)
scheme.

The algorithm works with the evaluation function
    value = 2 * n * max(conv(a,a)) / (sum(a)**2)

which is scale‑invariant.  We therefore keep the L1‑norm of the sequence equal to
sqrt(2*n) (the normalisation used in the reference method).  Starting from a
uniform sequence we repeatedly solve a linear programme

    maximise   sum(b)
    subject to  conv(a,b) <= M          (M = max(conv(a,a)))
                b >= 0

The LP finds a direction `g` that can increase the total mass without
increasing the current worst convolution value `M`.  After normalising `g`
to the same L1‑norm we move a small step towards it, re‑normalise and keep the
best sequence found.  The step size `t` is adapted: it is increased when an
improvement is obtained and decreased otherwise.  If the LP fails we perform a
tiny random perturbation to escape possible stagnation.

The whole search runs for the allotted time budget (default 1000 s) and
returns the best sequence discovered.
"""

from scipy import optimize

# Helper functions (top‑level, no closures)

def _norm_factor(n: int) -> float:
    """Return sqrt(2*n) – the L1 norm we keep for all sequences of length n."""
    return np.sqrt(2.0 * n)


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the probability simplex {x >= 0, sum(x) = 1}.
    Uses the O(n log n) algorithm of Duchi et al.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def solve_convolution_lp(f_seq: list[float], rhs: float) -> np.ndarray | None:
    """
    Linear programme:
        maximise   sum(b)
        subject to  conv(f_seq, b) <= rhs   (for all 0 <= k < 2*len(f_seq)-1)
                    b >= 0
    Returns the optimal b as a NumPy array, or ``None`` if the solver fails.
    """
    n = len(f_seq)
    # objective: maximise sum(b)  -> minimise -sum(b)
    c = -np.ones(n)

    # Build inequality matrix for the convolution constraints.
    # Row k enforces   sum_i f_i * b_{k-i} <= rhs.
    a_ub = []
    b_ub = []

    # create rows for each possible shift k
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # i runs over indices of f_seq; j = k-i must be a valid index of b
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # fill the row at positions j = k - i
        for i in range(i_start, i_end + 1):
            j = k - i
            row[j] = f_seq[i]
        a_ub.append(row)
        b_ub.append(rhs)

    # non‑negativity constraints  -b_i <= 0
    a_ub_nonneg = -np.eye(n)
    b_ub_nonneg = np.zeros(n)

    a_ub = np.vstack([a_ub, a_ub_nonneg])
    b_ub = np.hstack([b_ub, b_ub_nonneg])

    # use the modern HiGHS solver (fast and robust)
    res = optimize.linprog(
        c,
        A_ub=a_ub,
        b_ub=b_ub,
        method="highs",
        options={"presolve": True, "quiet": True},
    )

    if res.success:
        return res.x
    return None


def _max_convolution(seq: np.ndarray) -> float:
    """Return the maximum entry of the full convolution seq * seq."""
    # FFT based convolution is faster for moderate sizes
    n = seq.size
    size = 2 * n - 1
    # next power‑of‑two for speed
    fft_len = 1 << (size - 1).bit_length()
    fft_seq = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.ifft(fft_seq * fft_seq, n=fft_len).real[:size]
    return float(conv.max())

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function sequence that minimises the provided
    ``evaluate_sequence`` metric.

    Parameters
    ----------
    seed: int
        Random seed for reproducibility.
    budget_s: int
        Time budget in seconds (the function must return before this
        deadline; a small safety margin is left for finalisation).
    kwargs: dict
        Ignored – present for compatibility with the surrounding framework.

    Returns
    -------
    list[float]
        The best sequence found (non‑negative, arbitrary length).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # keep a few seconds free

    # ------------------------------------------------------------------
    # Choose a working length.
    # If the global variable `height_sequence_1` exists we reuse its size,
    # otherwise we fall back to a modest size (500) that keeps the LP cheap.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
    else:
        n = 500

    # ------------------------------------------------------------------
    # Initialise the sequence.
    # 50 % chance we start from the current best (if available), otherwise from
    # a random Dirichlet vector.  In both cases we enforce the L1 norm
    # sqrt(2*n), which is the normalisation used by the reference method.
    # ------------------------------------------------------------------
    if np.random.rand() < 0.5 and "height_sequence_1" in globals():
        cur = np.array(height_sequence_1[:n], dtype=float)
    else:
        cur = np.random.dirichlet(np.ones(n))

    cur = cur * (_norm_factor(n) / cur.sum())
    best_seq = cur.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n = {n}, initial score = {best_score:.6f}")

    # parameters for the Frank‑Wolfe style updates
    step = 0.20          # initial convex‑combination weight
    min_step = 0.01
    max_step = 0.5

    # ------------------------------------------------------------------
    # Main optimisation loop – runs until the time budget expires.
    # ------------------------------------------------------------------
    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # current worst convolution value M
        M = _max_convolution(cur)

        # ----------------------------------------------------------------
        # Solve the linear programme that finds a direction `g` with larger
        # total mass while respecting conv(cur, g) <= M.
        # ----------------------------------------------------------------
        g = solve_convolution_lp(cur.tolist(), M)

        if g is None:
            # LP failed (rare).  Apply a tiny random perturbation to escape.
            idx = np.random.randint(n)
            cur[idx] = max(0.0, cur[idx] + np.random.randn() * 0.01)
            cur = cur * (_norm_factor(n) / cur.sum())
            continue

        # Normalise g to the same L1 norm as `cur`.
        sum_g = g.sum()
        if sum_g == 0.0:
            # degenerate solution – skip this iteration
            continue
        g = g * (_norm_factor(n) / sum_g)

        # ----------------------------------------------------------------
        # Try a few step sizes (current, half, quarter) and keep the first
        # one that yields an improvement.
        # ----------------------------------------------------------------
        improved = False
        for factor in (step, step / 2, step / 4):
            cand = (1.0 - factor) * cur + factor * g
            # Numerical safety: enforce non‑negativity and exact normalisation
            cand = np.maximum(cand, 0.0)
            cand = cand * (_norm_factor(n) / cand.sum())

            cand_score = evaluate_sequence(cand.tolist())
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                cur = cand
                improved = True
                # gently increase step size for the next iteration
                step = min(max_step, step * 1.1)
                print(
                    f"[{iteration:04d}] improved score = {best_score:.6f}, "
                    f"step = {factor:.3f}"
                )
                break

        if not improved:
            # No improvement – shrink the step size to be more cautious.
            step = max(min_step, step * 0.7)

        # Occasionally inject a random perturbation even when the LP works,
        # to avoid being trapped in a flat region.
        if iteration % 50 == 0 and np.random.rand() < 0.1:
            idx1, idx2 = np.random.choice(n, 2, replace=False)
            delta = np.random.randn() * 0.01 * cur[idx1]
            delta = np.clip(delta, -cur[idx1], cur[idx2])
            cur[idx1] -= delta
            cur[idx2] += delta
            cur = np.maximum(cur, 0.0)
            cur = cur * (_norm_factor(n) / cur.sum())

    # ------------------------------------------------------------------
    # Return the best sequence discovered (as plain Python list of floats).
    # ------------------------------------------------------------------
    print(f"[final] best score = {best_score:.6f} after {iteration} iterations")
    return best_seq.tolist()
