def simplex_projection(v, s):
    # algorithm from Duchi, Shalev-Shwartz et al. (2008)
    # ensure v >= 0 and sum(v) = s
    # input v: np array.
    # returns projected vector.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    # find rho: largest i s.t. u_i > cssv_i / (i+1)
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

import numpy as np, time, math, sys

def proj_simplex(v, s):
    """Project vector v onto simplex of sum s (nonnegative)."""
    # Using algorithm from Duchi.
    n = len(v)
    if s < 0:
        raise ValueError
    # Sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n+1)
    rho = np.nonzero(rho_arr > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def fft_convolve(x, y):
    n = len(x) + len(y) - 1
    size = 1 << (n-1).bit_length()
    X = np.fft.rfft(x, size)
    Y = np.fft.rfft(y, size)
    res = np.fft.irfft(X * Y, size)
    return res[:n]

def compute_score(a):
    # Evaluate can't be used for speed? Use evaluate_sequence directly after final? But we need for intermediate.
    # We'll compute the same formula for speed: 2 * n * max_b / sum_a**2
    b = fft_convolve(a, a)
    max_b = np.max(b)
    sum_a = np.sum(a)
    if sum_a < 0.01:  # avoid division by zero
        return np.inf
    return 2 * len(a) * max_b / (sum_a**2)

def propose_candidate(seed=42, budget_s=1000):
    """Gradient + simplex projection search for a step‑function minimizing the
    evaluation metric.  Starts from a uniform distribution and performs
    projected sub‑gradient descent on the maximum of the self‑convolution,
    interleaved with occasional random perturbations."""
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # leave a little margin

    # length of the vector is taken from the reference construction
    n = len(height_sequence_1) if 'height_sequence_1' in globals() else 500
    if n < 10:
        n = 500

    # Target sum – keep the same scaling used by the reference algorithm
    S = math.sqrt(2.0 * n)

    # initial point – uniform on the simplex
    a = np.full(n, S / n, dtype=float)

    # keep the best encountered solution
    best_a = a.copy()
    best_score = compute_score(best_a)

    # parameters for the projected sub‑gradient loop
    step0 = 0.5 * S               # initial step size (will be scaled later)
    decay = 1e-4                  # decay of step size
    it = 0

    while time.time() < deadline:
        it += 1

        # 1) compute convolution and the index of the maximum entry
        b = fft_convolve(a, a)
        max_idx = int(np.argmax(b))

        # 2) sub‑gradient of max(b) w.r.t. a
        grad = np.zeros_like(a)
        # contribution only where the partner index lies inside the vector
        low = max(0, max_idx - (n - 1))
        high = min(n - 1, max_idx)
        # for i in [low, high], partner = max_idx - i is also in [0, n-1]
        i_range = np.arange(low, high + 1)
        partner = max_idx - i_range
        grad[i_range] = 2.0 * a[partner]

        # 3) take a projected (sub‑)gradient step
        # step size shrinks slowly with the iteration count
        step = step0 / (1.0 + decay * it)
        a_new = a - step * grad

        # 4) project back onto the simplex (non‑negative, fixed sum S)
        a_new = proj_simplex(a_new, S)

        # 5) evaluate
        cur_score = compute_score(a_new)

        # 6) accept if we made progress, otherwise keep the old vector
        if cur_score <= best_score:
            best_score = cur_score
            best_a = a_new.copy()
        # accept also if the move improves the current point (even if not the best)
        if cur_score < compute_score(a):
            a = a_new
        else:
            # Occasionally allow a jump that does not improve – helps escape plateaus
            if np.random.rand() < 0.001:
                a = a_new

        # 7) random shake‑up every few hundred iterations
        if it % 500 == 0:
            # take a tiny random redistribution that respects the simplex
            eps = 0.02 * S
            i, j = np.random.choice(n, size=2, replace=False)
            delta = min(eps, a[i])
            a[i] -= delta
            a[j] += delta
            # re‑project just in case numerical errors accumulated
            a = proj_simplex(a, S)

        # optional: print progress occasionally
        if it % 1000 == 0:
            print(f"Iter {it:6d} – best score {best_score:.6f}")

        # stop early if we already beat the target by a comfortable margin
        if best_score < 1.4:
            # we have reached a score better than the required 1.503; stop early
            break

    # final sanity – ensure the returned list obeys the evaluator’s clamps
    final = [float(x) for x in best_a]
    return final

"""Projected deconvolution descent to flatten self‑convolution.

The algorithm iteratively reduces the peak of the autocorrelation `a*a`
by solving the linearised equation  
    2 a * δa = target – a*a
in the Fourier domain, taking a small step `δa`, and projecting the
result back onto the simplex (non‑negative entries with fixed ℓ₁
norm).  Random re‑initialisations and occasional perturbations help
escape plateaus.  The best sequence found within the time budget is
returned.
"""

import numpy as np
import time
import math

def _proj_simplex(v, s):
    """Project vector v onto the simplex {x ≥ 0, sum(x) = s}."""
    n = v.shape[0]
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = cssv[-1] / n
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _next_pow2(x):
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def _fft_convolve_full(x, y):
    """Full convolution of real vectors x and y using FFT (real‑valued)."""
    n = len(x) + len(y) - 1
    size = _next_pow2(n)
    X = np.fft.rfft(x, size)
    Y = np.fft.rfft(y, size)
    res = np.fft.irfft(X * Y, size)
    return res[:n]

def _score_from_vector(a):
    """Compute the same metric as evaluate_sequence (fast version)."""
    b = _fft_convolve_full(a, a)
    max_b = np.max(b)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return float('inf')
    return 2.0 * len(a) * max_b / (sum_a * sum_a)

def propose_candidate(seed=42, budget_s=1000):
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    the evaluation function.  The method runs a projected de‑convolution
    descent with occasional random restarts; it stops as soon as the target
    score (≈1.503) is reached or the time budget expires.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0  # keep a tiny safety margin

    # ----------------------------------------------------------------------
    # 1) Problem size & scaling
    # ----------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        n = int(len(height_sequence_1))
    else:
        n = 500                     # fallback size
    # Fixed ℓ₁‑norm (the same normalisation used by the reference code)
    S = math.sqrt(2.0 * n)        # sum of the coefficients during the whole run
    target_scalar = S * S / (2.0 * n - 1.0)   # the ideal constant value of the convolution

    # ----------------------------------------------------------------------
    # 2) Helper that performs one de‑convolution step
    # ----------------------------------------------------------------------
    def _deconv_step(a, step):
        """Return a new vector after a single linearised de‑convolution step."""
        # current self‑convolution
        b = _fft_convolve_full(a, a)                 # length 2n‑1
        delta = b - target_scalar                    # we want this to become zero

        # FFT sizes (same for both transforms)
        size = _next_pow2(2 * n)                     # ≥ 2n‑1 is enough
        # pad a and delta to the chosen size
        a_pad = np.r_[a, np.zeros(size - n)]
        delta_pad = np.r_[delta, np.zeros(size - (2 * n - 1))]

        A_hat = np.fft.rfft(a_pad, size)
        D_hat = np.fft.rfft(delta_pad, size)

        # Solve 2 * A_hat * X_hat = -D_hat   →   X_hat = -D_hat / (2*A_hat)
        eps = 1e-12
        denom = 2.0 * A_hat
        # avoid division by (near) zero
        mask = np.abs(denom) > eps
        X_hat = np.zeros_like(D_hat, dtype=np.complex128)
        X_hat[mask] = -D_hat[mask] / denom[mask]
        # if denom is tiny we simply leave the component at zero

        # back‑transform
        u_full = np.fft.irfft(X_hat, size)
        # take the first n entries – they correspond to the update direction
        u = u_full[:n]
        # update and re‑project onto the simplex
        a_new = a + step * u
        a_new = _proj_simplex(a_new, S)
        return a_new

    # ----------------------------------------------------------------------
    # 3) Initial solution(s)
    # ----------------------------------------------------------------------
    # start from the uniform vector (the known stationary point)
    a = np.full(n, S / n, dtype=float)
    cur_score = _score_from_vector(a)

    # also keep the best solution ever found
    best_a = a.copy()
    best_score = cur_score

    init_step = 0.2 * S        # a moderate step; will be annealed
    step = init_step
    decay = 1e-4               # slow decay of the step size
    min_step = 1e-6
    it = 0
    last_improve = it

    # 4) Main optimisation loop
    # ----------------------------------------------------------------------

        a_candidate = _deconv_step(a, step)

        cand_score = _score_from_vector(a_candidate)

        if cand_score < best_score:

            best_a = a_candidate.copy()
            # if we cross the target we can stop early
            if best_score < 1.503:

        if cand_score < cur_score:
            a = a_candidate
            cur_score = cand_score
            last_improve = it
        else:

            step *= 0.5

                a = best_a.copy()
                # add a tiny random redistribution that respects the simplex
                eps = 0.01 * S
                i, j = np.random.choice(n, size=2, replace=False)
                delta = min(eps, a[i])
                a[i] -= delta
                a[j] += delta
                a = _proj_simplex(a, S)
                cur_score = _score_from_vector(a)
                step = init_step

        # gradual annealing of the step size
        step = init_step / (1.0 + decay * it)

        if it % 4000 == 0:
            a = np.random.rand(n)
            a = _proj_simplex(a, S)

            step = init_step

        # occasional progress report (optional)
        if it % 1000 == 0:
            print(f"Iter {it:6d} – best score {best_score:.6f} – current {cur_score:.6f}")

    # ----------------------------------------------------------------------
    # 5) Return the best sequence found (as a plain Python list)
    # ----------------------------------------------------------------------
    return [float(x) for x in best_a]

        mask = np.abs(denom) > eps
        X_hat = np.zeros_like(D_hat, dtype=np.complex128)
        X_hat[mask] = -D_hat[mask] / denom[mask]
        # components where denom≈0 stay zero

        u_full = np.fft.irfft(X_hat, size)           # real‑valued
        u = u_full[:n]                               # update direction (length n)

        a_new = a + step * u
        a_new = _proj_simplex(a_new, S)
        return a_new

    # --------------------------------------------------------------
    # 3) Initial solution
    # --------------------------------------------------------------
    a = np.full(n, S / n, dtype=float)               # uniform start
    cur_score = _score_from_vector(a)

    best_a = a.copy()
    best_score = cur_score

    # descent parameters
    init_step = 0.2 * S                               # initial step magnitude
    step = init_step
    decay = 1e-4                                      # slow annealing of step
    min_step = 1e-6
    it = 0

    # --------------------------------------------------------------
    # 4) Main optimisation loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # one de‑convolution step
        cand = _deconv_step(a, step)
        cand_score = _score_from_vector(cand)

        # update global best
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            if best_score < 1.503:                  # target already reached
                break

        # accept if we improve the current point
        if cand_score < cur_score:
            a = cand
            cur_score = cand_score
        else:
            # no improvement → shrink step
            step *= 0.5
            if step < min_step:
                # re‑initialise from the current best with a tiny random shake
                a = best_a.copy()
                eps = 0.01 * S
                i, j = np.random.choice(n, size=2, replace=False)
                delta = min(eps, a[i])
                a[i] -= delta
                a[j] += delta
                a = _proj_simplex(a, S)
                cur_score = _score_from_vector(a)
                step = init_step

        # anneal step size gradually
        step = init_step / (1.0 + decay * it)

        # periodic full random restart (helps when stuck)
        if it % 4000 == 0:
            a = np.random.rand(n)
            a = _proj_simplex(a, S)
            cur_score = _score_from_vector(a)
            step = init_step

        # occasional progress report (useful for debugging)
        if it % 1000 == 0:
            print(f"Iter {it:6d} – best score {best_score:.6f} – cur score {cur_score:.6f}")

    # --------------------------------------------------------------
    # 5) Return result as plain Python list
    # --------------------------------------------------------------
    return [float(x) for x in best_a]
```
