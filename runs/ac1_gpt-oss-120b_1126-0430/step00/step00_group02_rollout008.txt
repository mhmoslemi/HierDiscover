"""
Propose a step‑function (non‑negative list of heights) that minimises

        2·n·max(conv(a,a)) / (sum(a)**2)

where `conv` is the ordinary discrete convolution.  

The algorithm starts from an analytically motivated “heavy‑ends”
distribution (a_i ∝ 1/√((i+½)(N‑½‑i))) which makes the antidiagonal
sums of a·aᵀ almost constant, i.e. the objective is already close to the
theoretical lower bound 2·N/(2·N‑1)≈1.0.  From this start a simple
sub‑gradient descent on the maximal convolution value is performed.
At each iteration a step is taken opposite to the sub‑gradient, the vector
is projected back onto the probability simplex (non‑negative, sum‑to‑1)
and the step size is adapted by a tiny line‑search.  When progress stalls
a small random jitter is added to escape possible plateaus.  The best
sequence seen during the run is returned.

The whole routine respects the 1000 s time limit and never uses I/O.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top level – no closures)
# ----------------------------------------------------------------------


def init_heavy_ends(n: int) -> np.ndarray:
    """
    Initialise a symmetric distribution that is large at the two ends.
    a[i] ∝ 1 / sqrt( (i+0.5)*(n-0.5-i) ).
    The resulting vector is normalised to sum to 1.
    """
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    a = a / a.sum()
    return a


def max_convolution_and_subgrad(a: np.ndarray):
    """
    Compute the full convolution of a with itself, the maximal entry
    and a sub‑gradient of the max‑function w.r.t. a.

    Sub‑gradient for a given active index k (where conv[k] is maximal):
        ∂conv[k]/∂a[i] = 2 * a[k-i]   (if 0 ≤ k-i < n, else 0)

    The returned gradient is the sum of the contributions of all active
    indices (ties are handled automatically).
    """
    n = a.shape[0]
    conv = np.convolve(a, a)                 # length 2*n-1
    max_val = conv.max()
    # toleranced active set (handles floating‑point ties)
    eps = 1e-12
    active = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # i runs over all indices where a[i] participates in conv[k]
        i = np.arange(i_start, i_end + 1, dtype=np.int64)
        grad[i] += 2.0 * a[k - i]           # sub‑gradient contribution
    return conv, max_val, grad


def project_to_simplex(v: np.ndarray):
    """
    Project a (possibly negative) vector onto the probability simplex:
    {x | x_i >= 0, sum_i x_i = 1}.
    Implements the O(n) algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):            # all entries are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Main routine required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of step heights with a low evaluation score.
    The function runs for at most `budget_s` seconds and returns the
    best sequence encountered.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # leave a safety margin

    # ------------------------------------------------------------------
    # Choose a problem size.
    # If the previous best construction is available we honour its length,
    # otherwise we use a modest size that comfortably fits the time budget.
    # ------------------------------------------------------------------
    try:
        # height_sequence_1 is provided by the host environment
        base_len = len(height_sequence_1)
    except Exception:
        base_len = 2000

    # Working length: a few thousand elements give a good trade‑off.
    # The algorithm works with any length, but very large N slows down a
    # single iteration (O(N log N) convolution).
    N = min(base_len, 3000)

    # ------------------------------------------------------------------
    # Initial vector – heavy ends + tiny random perturbation to break exact
    # symmetry (otherwise the sub‑gradient would be the zero vector).
    # ------------------------------------------------------------------
    a = init_heavy_ends(N)
    a += 0.001 * np.random.rand(N)                # tiny noise
    a = project_to_simplex(a)

    # Evaluate the initial candidate using the provided black‑box.
    best_seq = a.tolist()
    best_score = evaluate_sequence(best_seq)
    print(f"[init] length={N}, loss={best_score:.6f}")

    # ------------------------------------------------------------------
    # Sub‑gradient descent parameters
    # ------------------------------------------------------------------
    lr = 0.10                # initial learning rate
    min_lr = 1e-7
    no_improve = 0
    max_no_improve = 300    # after this many non‑improvements we jitter
    iteration = 0

    while time.time() < deadline:
        iteration += 1

        # current loss using the cheap analytic formula (sum(a) is 1)
        conv, cur_max, grad = max_convolution_and_subgrad(a)
        cur_loss = 2.0 * N * cur_max               # because sum(a) == 1

        # ------------------------------------------------------------------
        # Line‑search: try decreasing step sizes until the loss improves.
        # ------------------------------------------------------------------
        step_accepted = False
        cur_lr = lr
        for _ in range(12):
            a_candidate = a - cur_lr * grad
            a_candidate = project_to_simplex(a_candidate)

            # cheap loss evaluation for the candidate
            new_max = np.convolve(a_candidate, a_candidate).max()
            new_loss = 2.0 * N * new_max

            if new_loss < cur_loss - 1e-12:
                a = a_candidate
                cur_loss = new_loss
                step_accepted = True
                break
            cur_lr *= 0.5

        # ------------------------------------------------------------------
        # Adapt the learning rate and bookkeeping.
        # ------------------------------------------------------------------
        if step_accepted:
            # slightly increase learning rate after a successful step
            lr = min(lr * 1.05, 1.0)
            no_improve = 0
            if cur_loss < best_score:
                best_score = cur_loss
                best_seq = a.tolist()
                elapsed = time.time() - start_time
                print(f"[{iteration}] new best: {best_score:.6f}  "
                      f"(elapsed {elapsed:.1f}s, lr={lr:.4f})")
        else:
            # step rejected → shrink learning rate
            lr = max(lr * 0.5, min_lr)
            no_improve += 1

        # ------------------------------------------------------------------
        # Occasionally inject random jitter to escape plateaus.
        # ------------------------------------------------------------------
        if no_improve >= max_no_improve:
            # start again from the current best but with a tiny random shake
            a = np.array(best_seq, dtype=np.float64)
            jitter = 0.005 * np.random.randn(N)
            a = a + jitter
            a = np.clip(a, 0.0, None)
            a = project_to_simplex(a)
            no_improve = 0
            lr = 0.10
            print(f"[{iteration}] jitter injected – resetting step size")

    # ------------------------------------------------------------------
    # Final evaluation (uses the official evaluator – slower but exact)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    total_time = time.time() - start_time
    print(f"[done] iterations={iteration}, best_score={best_score:.6f}, "
          f"final_eval={final_score:.6f}, time={total_time:.1f}s")
    return best_seq
