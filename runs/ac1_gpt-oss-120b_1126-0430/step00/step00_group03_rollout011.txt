def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import numpy as np, time
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2

    best_seq = None
    best_score = np.inf

    # optionally start from height_sequence_1
    try:
        init = list(height_sequence_1)
        score = evaluate_sequence(init)
        best_seq = init
        best_score = score
    except Exception:
        pass

    iteration = 0
    while time.time() < deadline:
        # choose random strategy
        choice = np.random.rand()
        if choice < 0.3:
            # generate monotone decreasing via power law
            n = np.random.randint(5, 151)  # up to 150
            p = np.random.uniform(0.0, 5.0)
            offset = np.random.uniform(0.0, 5.0)
            # compute values
            idx = np.arange(n)  # 0..n-1
            vals = 1.0/ ((idx + 1 + offset) ** p)
            # avoid all zeros
            if np.all(vals == 0):
                continue
            vals = vals / np.sum(vals)
            seq = vals.tolist()
        elif choice < 0.6:
            # random Dirichlet vector sorted descending
            n = np.random.randint(5, 151)
            seq = np.random.rand(n)
            seq = np.sort(seq)[::-1]  # descending
            seq = seq / np.sum(seq)
            seq = seq.tolist()
        else:
            # gradient descent perturb from best_seq
            if best_seq is None:
                # fallback random
                n = np.random.randint(5, 151)
                seq = np.random.rand(n)
                seq = seq / np.sum(seq)
                seq = seq.tolist()
            else:
                # copy and perturb
                seq = best_seq.copy()
                n = len(seq)
                # perturb: pick random indices i, j and move mass
                i = np.random.randint(n)
                j = np.random.randint(n)
                if i != j:
                    delta = np.random.uniform(-0.05, 0.05) * seq[i]
                    # ensure non-negative
                    delta = max(-seq[i], min(delta, seq[i]))
                    seq[i] -= delta
                    seq[j] += delta
                # normalize
                total = sum(seq)
                if total <= 0:
                    continue
                seq = [x/total for x in seq]
        # evaluate
        try:
            score = evaluate_sequence(seq)
        except Exception:
            continue
        if score < best_score:
            best_score = score
            best_seq = seq
            print(f"Iter {iteration}: new best score {best_score:.6f} len {len(seq)}")
        iteration += 1

    # ensure final best_seq is a list of floats
    return [float(x) for x in best_seq]

def subgradient_descent(seq, steps=20):
    n = len(seq)
    seq = np.array(seq)
    for t in range(steps):
        conv = np.convolve(seq, seq)
        k = np.argmax(conv)

Define:

def project_simplex(v):
    """Project vector v to the probability simplex (nonnegative, sum=1)."""
    # copy
    v = np.asarray(v)
    if np.sum(v) == 1 and np.all(v >= 0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Simple yet effective search for a step‑function that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The search uses three families of candidates:
  1. Monotone power‑law profiles a_i ∝ (i+1+offset)^{-p}
  2. Random Dirichlet vectors (sorted to be monotone)
  3. Small perturbations of the current best solution.

Each candidate is projected onto the probability simplex
(∑a_i = 1, a_i ≥ 0) so that scaling does not affect the objective.
A lightweight sub‑gradient step is applied to the candidate before
its score is measured – this drives the max‑convolution down while
preserving the simplex constraints.

The routine runs for the allotted budget (default 1000 s) and returns
the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex (∑x=1, x≥0)
def _project_simplex(v):
    """Project a vector onto the probability simplex."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]

    # fast path
    if np.all(v >= 0) and np.isclose(v.sum(), 1):
        return v

    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    # find rho
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive → uniform distribution
        return np.full(n, 1.0 / n)

    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient step that tries to lower the maximal convolution entry.
def _subgradient_step(seq, steps=8):
    """Run a few projected sub‑gradient steps on seq."""
    a = np.array(seq, dtype=float)
    n = a.size
    for t in range(steps):
        conv = np.convolve(a, a)
        k = int(np.argmax(conv))                     # index of the current maximum
        grad = np.zeros_like(a)
        # gradient of c_k = Σ a_i a_{k-i}
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]
        # diminishing step size
        lr = 0.2 / (t + 1)
        a = a - lr * grad
        a = _project_simplex(a)
    return a.tolist()


# ----------------------------------------------------------------------
# Candidate generators
def _gen_power_law(n, p, offset):
    """Monotone decreasing power‑law profile."""
    idx = np.arange(n, dtype=float)
    vals = 1.0 / ((idx + 1.0 + offset) ** p)
    if vals.sum() == 0.0:
        vals = np.ones_like(vals)
    vals = vals / vals.sum()
    return vals.tolist()


def _gen_random_dirichlet(n):
    """Random Dirichlet vector, sorted descending."""
    v = np.random.rand(n)
    v = np.sort(v)[::-1]          # enforce monotonicity (helps the max‑conv.)
    v = v / v.sum()
    return v.tolist()


def _perturb(seq):
    """Small random transfer of mass between two positions."""
    a = np.array(seq, copy=True)
    n = a.size
    if n < 2:
        return seq
    i, j = np.random.choice(n, 2, replace=False)
    # move a fraction of the source mass
    delta = np.random.uniform(0.0, 0.5) * a[i]
    a[i] -= delta
    a[j] += delta
    a = _project_simplex(a)
    return a.tolist()

    """
    Search for a sequence of non‑negative heights with a low evaluation score.
    Returns the best sequence discovered before the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # keep a small safety margin

    # keep the best solution seen so far
    best_seq = None
    best_score = np.inf

    # try the provided construction (if available) as a warm start

    it = 0
    while time.time() < deadline:
        it += 1

        # --------------------------------------------------------------
        # choose a generation strategy
        r = np.random.rand()
        if r < 0.35:

            n = np.random.randint(8, 61)                   # length 8 … 60 (fast to evaluate)
            p = np.random.uniform(0.2, 3.0)                # exponent
