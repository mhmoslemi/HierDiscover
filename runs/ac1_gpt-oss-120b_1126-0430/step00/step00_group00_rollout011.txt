def score_from_alpha(alpha, n):
    idx = np.arange(1, n+1, dtype=np.float64)
    a = idx ** (-alpha)
    a /= a.sum()
    # fast convolution using FFT
    size = 2*n - 1
    N = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, N)
    B = A * A
    b = np.fft.irfft(B, N)[:size]
    max_b = b.max()
    score = 2.0 * n * max_b
    return score, a
```

def convolve_fft(a, b):
    size = len(a) + len(b) - 1
    N = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    C = A * B
    c = np.fft.irfft(C, N)[:size]
    return c
```

def compute_score(a):

    return float(2.0 * n * max_b / (sum_a ** 2))

"""
Propose a low‑score step‑function using a power‑law shaped sequence.

The evaluation function is  2·n·max(conv(a,a)) / (sum(a)^2) .
Because the score is invariant under scaling we may enforce sum(a)=1.
For a probability vector a, the self‑convolution b_k = Σ_i a_i a_{k-i}
is the distribution of the sum of two independent draws from a.
If a_i ∝ (i+δ)^{-α} the convolution becomes almost flat for a suitable
choice of the exponent α (the classic 1/√x law gives a flat interior;
choosing a slightly larger α balances the large values at the
left‑hand side of the convolution).  By scanning α (and a small shift
δ) we quickly find a sequence whose maximal convolution value is far
below the uniform‑sequence bound (score ≈ 2).  After a coarse grid search
we refine around the best parameters and finally polish the result with a
few projected‑gradient steps that minimise the L2‑error to a flat target.
All heavy work uses FFT‑based convolution, so the whole search fits easily
into the 1000 s budget.

The returned list contains only non‑negative floats (well below the
1000‑limit) and the algorithm is deterministic for a given seed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: fast linear convolution via FFT (real input)
def convolve_fft(a, b):
    """Return linear convolution of a and b using a power‑of‑two FFT."""
    size = len(a) + len(b) - 1
    N = 1 << (size - 1).bit_length()          # next power of two
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    C = A * B
    conv = np.fft.irfft(C, N)[:size]
    return conv


# ----------------------------------------------------------------------
# Score for a given (already normalised) sequence.
def fast_score(a):
    """Compute the objective 2·n·max(b)/(sum(a)^2) with FFT convolution."""
    n = len(a)
    b = convolve_fft(a, a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 1e-12:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
def power_law_sequence(alpha, delta, n):
    """
    Generate a non‑negative sequence a_i ∝ (i + delta)^{-alpha},
    i = 1 … n, and normalise it to sum‑one.
    """
    # i starts from 1 to avoid a zero denominator.
    idx = np.arange(1.0, n + 1, dtype=np.float64) + delta
    a = idx ** (-alpha)
    s = a.sum()
    if s > 0.0:
        a /= s
    return a


# ----------------------------------------------------------------------
def gradient_refine(a, time_deadline):
    """
    A few projected‑gradient steps that minimise the L2 error of the
    convolution to its average value (which is 1/(2n‑1) after normalisation).
    """
    n = len(a)
    # target constant for the convolution (average value)
    target_const = (a.sum() ** 2) / (2 * n - 1)

    # step‑size control
    step_factor = 0.05

    best_local = a.copy()
    best_score = fast_score(a)

    while time.time() < time_deadline:
        # current convolution
        b = convolve_fft(a, a)
        # compute error w.r.t. the flat target
        err = b - target_const

        # sub‑gradient of the L2 error w.r.t. a
        grad = 4.0 * np.correlate(err, a, mode='valid')   # length n

        # normalise the step size (avoid too large jumps)
        gn = np.linalg.norm(grad)
        if gn == 0.0:
            break
        eta = step_factor / gn

        a = a - eta * grad
        # projection onto the non‑negative orthant and re‑normalise
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        # keep the best seen
        cur = fast_score(a)
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
            # occasional progress report
            print(f'  gradient step improved score {best_score:.6f}')
            # tighten step factor for later fine tuning
            step_factor = max(step_factor * 0.95, 1e-4)

    return best_local


# ----------------------------------------------------------------------
def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Search for a step‑function with a low evaluation score.
    The algorithm scans a family a_i ∝ (i+δ)^{-α}, refines the best
    candidate with a tiny projected‑gradient optimiser and returns the
    sequence as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # slight safety margin

    # ------------------------------------------------------------------
    # Basic parameters (length can be increased if time permits)
    n = 3000

    # start with a trivial uniform sequence – this gives the reference score 2
    a_uniform = np.full(n, 1.0 / n)
    best_seq = a_uniform.copy()
    best_score = fast_score(best_seq)
    print(f'initial uniform score = {best_score:.6f}')

    # ------------------------------------------------------------------
    # Coarse grid over exponent α and small shift δ
    alphas = np.linspace(0.40, 0.75, 71)              # step ≈0.005
    deltas = [0.0, 0.5, 1.0, 2.0]

    best_alpha = None
    best_delta = None

    for delta in deltas:
        for alpha in alphas:
            if time.time() >= deadline:
                break
            a = power_law_sequence(alpha, delta, n)
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                best_alpha = alpha
                best_delta = delta
                print(f'new best (coarse)  α={alpha:.4f} δ={delta} → score={sc:.6f}')

    # ------------------------------------------------------------------
    # Fine search around the best parameters found so far
    if best_alpha is not None:
        alphas_fine = np.linspace(max(0.30, best_alpha - 0.02),
                                 min(0.90, best_alpha + 0.02), 41)
        deltas_fine = [max(0.0, best_delta - 0.5),
                       best_delta,
                       best_delta + 0.5]

        for delta in deltas_fine:
            for alpha in alphas_fine:
                if time.time() >= deadline:
                    break
                a = power_law_sequence(alpha, delta, n)
                sc = fast_score(a)
                if sc < best_score:
                    best_score = sc
                    best_seq = a.copy()
                    best_alpha = alpha
                    best_delta = delta
                    print(f'new best (fine)   α={alpha:.5f} δ={delta:.2f} → score={sc:.6f}')

    # ------------------------------------------------------------------
    # Optional local optimisation (few gradient steps)
    if time.time() < deadline:
        print('starting gradient refinement...')
        best_seq = gradient_refine(best_seq, deadline)

    # Final evaluation using the provided evaluate_sequence (sanity check)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f'final reported score = {final_score:.6f}')
    except Exception as e:
        # If the caller hasn't imported evaluate_sequence we simply ignore.
        pass

    # Ensure the return type matches the specification (list of floats)
    return best_seq.tolist()
