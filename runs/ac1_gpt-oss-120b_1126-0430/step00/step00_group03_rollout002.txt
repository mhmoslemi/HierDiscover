    v = np.asarray(v)

"""
Propose a sequence of non‑negative heights that lowers the evaluation score.

The algorithm works on a fixed length (by default the length of the
previous best sequence if available, otherwise 2000).  The vector is
kept on the probability simplex (non‑negative and sum‑to‑one).  A smooth
approximation of the ``max`` in the objective is used:

        f(a) = (1/α)·log Σ_k exp(α·c_k) ,   c = a * a   (linear convolution)

Its gradient w.r.t. a is
        ∇f_i = 2· Σ_k w_k·a_{k-i}
where w is the soft‑max weight vector.
A mirror‑descent (exponentiated‑gradient) step is performed:
        a ← a·exp(−η·∇f)  and then renormalised.

α is slowly increased (sharpening the approximation) while η
(step‑size) is slowly decreased.  The method is repeated from many
random initialisations within the allotted time budget, keeping the
best sequence found.

The final sequence is returned as a plain list of floats ready for
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the next power of two greater than or equal to x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """Linear convolution via FFT (real inputs)."""
    n = len(u) + len(v) - 1
    size = _next_pow2(n)
    fu = np.fft.rfft(u, size)
    fv = np.fft.rfft(v, size)
    conv = np.fft.irfft(fu * fv, size)[:n]
    return conv


def _smooth_max_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max weighting of convolution entries."""
    # numerical stability: subtract the max before exponentiation
    cmax = conv.max()
    exp_vals = np.exp(alpha * (conv - cmax))
    return exp_vals / exp_vals.sum()


def _exponentiated_gradient_step(a: np.ndarray,
                                 grad: np.ndarray,
                                 step: float) -> np.ndarray:
    """Mirror‑descent step on the simplex (multiplicative update)."""
    # multiplicative update – keep positivity automatically
    a = a * np.exp(-step * grad)
    # renormalise to the simplex (sum to one)
    a_sum = a.sum()
    if a_sum == 0.0:
        # fallback to uniform distribution if everything collapsed
        a = np.full_like(a, 1.0 / len(a))
    else:
        a = a / a_sum
    return a


# ----------------------------------------------------------------------
# Core optimisation routine
# ----------------------------------------------------------------------


def _optimise_one_restart(n: int,
                          time_limit: float,
                          seed: int = None) -> np.ndarray:
    """
    Run a single optimisation run for at most ``time_limit`` seconds.
    Returns the best sequence (on the simplex) discovered in this run.
    """
    rng = np.random.default_rng(seed)
    # start from a random point on the simplex (Dirichlet(1,…,1) ≈ uniform)
    a = rng.random(n)
    a = a / a.sum()

    # hyper‑parameters (these work well for n ≈ 2000)
    alpha = 5.0          # start with a soft approximation of max
    alpha_max = 80.0     # eventual sharpness
    step = 0.4           # initial learning rate for EG
    step_min = 1e-4

    end_time = time.time() + time_limit
    # keep the best vector seen in this restart
    best_a = a.copy()
    best_score = 2.0 * n * np.max(_fft_convolve(a, a))

    while time.time() < end_time:
        # ---- compute convolution and soft‑max weights ----
        conv = _fft_convolve(a, a)               # length 2n‑1
        w = _smooth_max_weights(conv, alpha)     # soft‑max over the same length

        # ---- gradient of the smooth‑max (mirrored linear‐conv) ----
        # gradient_i = 2 * Σ_k w_k * a_{k-i}
        # This is a correlation of w with a (valid mode)
        # Compute via FFT: correlation = conv(w, reverse(a))
        corr_full = _fft_convolve(w, a[::-1])    # length 3n‑2
        grad = 2.0 * corr_full[n - 1:2 * n - 1]   # keep the valid n entries

        # ---- exponentiated‑gradient update on the simplex ----
        a = _exponentiated_gradient_step(a, grad, step)

        # ---- schedule updates ----
        # slowly increase α (sharpen) and decrease η (step size)
        alpha = min(alpha_max, alpha * 1.001)
        step = max(step_min, step * 0.9995)

        # ---- keep local best (cheap score, exact sum = 1) ----
        cur_score = 2.0 * n * np.max(_fft_convolve(a, a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

    return best_a
