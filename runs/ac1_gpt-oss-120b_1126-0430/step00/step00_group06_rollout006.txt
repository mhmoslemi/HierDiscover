def project_simplex(v):
    # Implementation from Duchi et al.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import numpy as np
    import time

    np.random.seed(seed)
    deadline = time.time() + budget_s - 5  # spare time for final steps

    # Choose length n based on time budget, moderate size.
    # We'll try some candidate lengths: [200, 500, 1000, 2000, 4000]
    lengths = [200, 500, 1000, 2000, 4000]
    best_seq = None
    best_val = float('inf')
    for n in lengths:
        if time.time() >= deadline:
            break
        # initialize sequence
        # try using uniform or random
        a = np.full(n, 1.0 / n)
        # optionally apply small random perturbation
        a += np.random.rand(n) * 1e-3
        a = a / a.sum()
        # store best for this length
        val = evaluate_sequence(a.tolist())
        if val < best_val:
            best_val = val
            best_seq = a.tolist()
        # run gradient descent for this length
        a_curr = a.copy()
        best_local_seq = a_curr.copy()
        best_local_val = val
        # hyperparameters schedule
        alpha = 0.1  # initial step size
        beta = 1.0   # initial softmax temperature inverse
        # loop until time almost out or iterations limit
        it = 0
        while time.time() < deadline and it < 5000:
            # compute convolution
            conv = np.convolve(a_curr, a_curr)
            max_c = conv.max()
            # softmax weights
            # for numerical stability subtract max then exponentiate
            exp_weights = np.exp(beta * (conv - max_c))
            w = exp_weights / exp_weights.sum()
            # compute gradient
            rev_a = a_curr[::-1]
            conv_w_rev = np.convolve(w, rev_a)
            grad = 2.0 * beta * conv_w_rev[n-1 : n-1+n]
            # gradient descent step
            a_next = a_curr - alpha * grad
            # project onto simplex
            a_next = project_simplex(a_next)
            # evaluate
            val_next = evaluate_sequence(a_next.tolist())
            # keep best
            if val_next < best_local_val:
                best_local_val = val_next
                best_local_seq = a_next.copy()
                # optionally increase step (if improvement)
                alpha = min(alpha * 1.05, 0.5)
            else:
                # if not improved, reduce step
                alpha = max(alpha * 0.7, 1e-4)
                # also possibly increase beta to focus more on max
                beta = min(beta * 1.02, 20.0)
            a_curr = a_next
            # early break if very low bound reached
            if best_local_val < 1.51:
                break
            it += 1
        # after loop, compare best_local_val to global best
        if best_local_val < best_val:
            best_val = best_local_val
            best_seq = best_local_seq.tolist()
            print(f'New global best: {best_val:.6f} with n={n}')
    # final result, ensure it's list of floats
    if best_seq is None:
        # fallback: uniform
        best_seq = [1.0] * 1
    print('Final best evaluation:', best_val)
    return best_seq

"""
Propose a new sequence of non‑negative heights that lowers the evaluation function.

The algorithm is a simple projected gradient descent on a smooth approximation of the
maximum of the self‑convolution.  For a vector a with sum(a)=1 the evaluation is

    E(a) = 2·n·max_k (a∗a)[k]          (scale‑invariant)

so we minimise max_k (a∗a)[k].  We replace the hard max by a
log‑sum‑exp (soft‑max) with a temperature β and descend its gradient.
After each step the vector is projected onto the probability simplex
(non‑negative, sum = 1).  The method is run for several problem sizes;
whenever a better value is found it is kept as the global best.
The whole search stops before the allotted time limit and returns the best
sequence discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the simplex {x >= 0, sum(x)=1}
# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Projects a real vector v onto the probability simplex.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    # find the rho index
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Core optimisation routine for a given length
# ----------------------------------------------------------------------
def optimise_length(
    n: int,
    deadline: float,
    init: np.ndarray | None = None,
    max_iters: int = 5000,
) -> tuple[np.ndarray, float]:
    """
    Optimises a sequence of length n.
    Returns the best sequence found (as a numpy array) and its evaluation.
    """
    # initialise ---------------------------------------------------------
    if init is None:
        a = np.full(n, 1.0 / n)                     # uniform start
    else:
        # use provided start, padded / trimmed to length n
        a = np.array(init, copy=True, dtype=float)
        if a.size < n:
            a = np.concatenate([a, np.full(n - a.size, 1.0 / n)])
        elif a.size > n:
            a = a[:n]
    # small random jitter to break symmetry
    a += 1e-5 * np.random.rand(n)
    a = project_simplex(a)

    best_a = a.copy()
    # evaluate using the supplied evaluate_sequence (fast for small n)
    best_val = evaluate_sequence(best_a.tolist())

    # parameters ---------------------------------------------------------
    alpha = 0.25            # initial step size
    beta = 0.8              # temperature for the soft‑max (low = smooth)
    it = 0

    while time.time() < deadline and it < max_iters:
        # full self‑convolution
        conv = np.convolve(a, a)            # length 2n‑1
        max_conv = conv.max()

        # soft‑max weights (focus on the largest entries)
        # shifting by max_conv keeps the exponentials ≤ 1 -> no overflow
        exp_weights = np.exp(beta * (conv - max_conv))
        w = exp_weights / exp_weights.sum()   # shape (2n‑1,)

        # gradient of the log‑sum‑exp objective (see analysis)
        rev_a = a[::-1]
        tmp = np.convolve(w, rev_a)          # length 3n‑2
        grad = 2.0 * beta * tmp[n - 1 : n - 1 + n]   # slice length n

        # descent step + projection
        a_next = a - alpha * grad
        a_next = project_simplex(a_next)

        # evaluate candidate
        cur_val = evaluate_sequence(a_next.tolist())

        # keep the best ever seen
        if cur_val < best_val:
            best_val = cur_val
            best_a = a_next.copy()
            # successful step → keep a slightly larger step, slightly cool β
            alpha = min(alpha * 1.04, 0.5)
            beta = max(beta * 0.96, 0.5)
        else:
            # non‑improving step → shrink step size, warm‑up β a little
            alpha = max(alpha * 0.70, 1e-5)
            beta = min(beta * 1.02, 20.0)

        a = a_next
        it += 1

        # occasional logging
        if it % 200 == 0:
            print(
                f" n={n:4d}  it={it:5d}  best={best_val:.6f}  cur={cur_val:.6f}"
                f"  α={alpha:.4g}  β={beta:.4g}"
            )

        # stop early if we already beat the target
        if best_val < 1.51:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of heights) with a low evaluation value.
    Uses projected gradient descent on several problem sizes and returns the
    best sequence found before the time budget expires.
    """
    np.random.seed(seed)
    # give a small safety margin for the final return
    deadline = time.time() + budget_s - 3.0

    # Try a few different lengths – larger n give a lower theoretical bound,
    # but also cost more per iteration.  We start with a modest size, improve it,
    # then upscale the solution for the next size.
    lengths = [256, 512, 1024, 2048]
    best_seq = None
    best_val = float("inf")
    current_init = None  # will be filled with the best sequence for up‑scaling

    for n in lengths:
        if time.time() >= deadline:
            break

        # If we have a good solution from a smaller size, up‑sample it
        if current_init is not None:
            # simple linear interpolation to new size
            x_old = np.linspace(0, 1, len(current_init), endpoint=False)
            x_new = np.linspace(0, 1, n, endpoint=False)
            init = np.interp(x_new, x_old, current_init)
            init = project_simplex(init)          # ensure feasibility
        else:
            init = None

        seq, val = optimise_length(
            n=n,
            deadline=deadline,
            init=init,
            max_iters=5000,
        )

        print(f"Finished length {n}, best evaluation = {val:.6f}")

        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            # keep this sequence for possible up‑sampling
            current_init = seq.copy()

        # early exit if we already met the target
        if best_val < 1.51:
            break

    # final sanity check – ensure we return a plain Python list
    if best_seq is None:
        # fallback: a tiny uniform sequence (will evaluate to 2.0)
        best_seq = [1.0]

    print(f"Final best evaluation: {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
