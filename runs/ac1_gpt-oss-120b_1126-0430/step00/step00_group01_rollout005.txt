def greedy_sidon(nmax):
    sums=set()
    S=[]
    for x in range(nmax):
        conflict=False
        # check duplicate sums with existing elements
        for y in S:
            if (x+y) in sums:
                conflict=True
                break
        if conflict:
            continue
        # check self sum duplicate
        if (2*x) in sums:
            continue
        # add
        for y in S:
            sums.add(x+y)
        sums.add(2*x)
        S.append(x)
    return S

"""
Propose a step‑function sequence with a very low evaluation score.

The idea:
*   Build a Sidon (B₂) set S of integers using the classic quadratic
    construction S = { i·p + (i² mod p) } for a prime p.
    In a Sidon set every unordered sum i+j is unique, so the
    auto‑convolution of the spike part has a maximum value of 2
    (each unordered pair contributes 2·1·1).

*   Add a tiny constant background ε to *every* coordinate.
    With ε = sqrt(2) / sqrt(n) (n = length of the sequence) the
    convolution of the background with itself attains exactly 2,
    i.e.  n·ε² = 2.

*   For the resulting sequence aₖ = ε + 1_{k∈S}
    the maximal convolution value is max_c = 2,
    while the total sum is
        Σa = n·ε + |S| = √(2n) + |S|.
    The evaluation becomes
        2·n·max_c / Σa² = 4·n / (√(2n) + |S|)²,
    which is well below the target 1.503 for any reasonable prime.

The algorithm samples several primes, builds the corresponding
sequence, evaluates it and keeps the best one found before the
time budget expires.
"""

import math
import random
import time


def _sidon_from_prime(p: int) -> list[int]:
    """
    Construct a Sidon (B₂) set of size p using the quadratic
    construction:  i·p + (i² mod p)  for i = 0,…,p‑1.
    """
    return [i * p + (i * i) % p for i in range(p)]


def _build_sequence(p: int) -> list[float]:
    """
    Build the step‑function for a given prime p.
    1. Create the Sidon set.
    2. Determine the needed length (max index + 1).
    3. Add a constant background ε = sqrt(2)/sqrt(n) to every coordinate.
    4. Add a spike of height 1 at each Sidon position.
    """
    sidon = _sidon_from_prime(p)
    n = max(sidon) + 1                     # length of the sequence
    eps = math.sqrt(2.0) / math.sqrt(n)    # background height
    seq = [eps] * n
    for idx in sidon:
        seq[idx] += 1.0                    # spike height = 1
    return seq


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for the best sequence within the time budget.
    The search simply tries a few different primes (hence different
    Sidon sets) and keeps the sequence with the lowest evaluation
    score.  The evaluation function `evaluate_sequence` is assumed to
    be present in the global namespace.
    """
    random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # leave a tiny safety margin

    # A modest list of primes – all give a Sidon set of size p.
    primes = [
        53, 59, 61, 67, 71, 73, 79, 83, 89, 97,
        101, 103, 107, 109, 113, 127, 131, 137,
        139, 149, 151
    ]

    best_seq = None
    best_score = float("inf")

    while time.time() < deadline:
        p = random.choice(primes)
        seq = _build_sequence(p)

        # Evaluate – any exception means an invalid sequence.
        try:
            score = evaluate_sequence(seq)
        except Exception:
            continue

        if score < best_score:
            best_score = score
            best_seq = seq
            print(
                f"New best: score={best_score:.6f}, prime={p}, "
                f"length={len(seq)}, spikes={p}"
            )

    # Fallback: if, for any reason, nothing was found (should not happen),
    # return a simple uniform sequence.
    if best_seq is None:
        n = 3000
        best_seq = [1.0] * n
        best_score = evaluate_sequence(best_seq)
        print(f"Fallback sequence: score={best_score:.6f}")

    return best_seq
