def propose_candidate(seed=..., budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1

    n = 1024  # Choose power-of-two for FFT speed
    # Initialize a with uniform distribution
    a = np.full(n, 1.0 / n)
    best_a = a.copy()
    best_score = compute_score(a)

    # Simulated annealing parameters
    T_start = 1.0
    T_end = 1e-4
    max_iter = 0
    while time.time() < deadline:
        # decaying temperature
        progress = (deadline - time.time()) / (budget_s - 1)  # Actually compute ratio progressed
        T = T_start * (T_end / T_start) ** (1 - progress)  # Exponential decay

        # propose move: select two indices i != j
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # transfer amount delta from i to j
        # choose delta as a fraction of min(a[i], a[j]) + random;
        eps = np.random.exponential(scale=0.01)  # small random magnitude
        delta = min(eps, a[i])  # ensure we don't go negative
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # Optional: add small random noise to other entries to avoid stagnation
        # compute rating
        score = compute_score(a_new)
        # Acceptance criterion: downhill or occasional uphill
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        # Accept if improves or with probability exp(-(score - cur_score)/T)
        if score < cur_score or np.random.rand() < np.exp(-(score - cur_score)/T):
            a = a_new
            cur_score = score

        # Occasionally renormalize to sum 1 to prevent drift due to numeric.
        if (max_iter % 1000) == 0:
            a = a / np.sum(a)

        max_iter += 1
    return best_a.tolist()

"""
optimiser.py – Simple simulated‑annealing / gradient descent search for a
step‑function that minimises the evaluation metric

    score = 2 * n * max(conv(a, a)) / (sum(a)**2)

The optimiser works in three stages:

1. **Initialisation**
   * A moderately large length (default 2048) is chosen.
   * The coefficients are drawn from a Dirichlet distribution
     (i.e. by normalising Gamma(1,1) samples) – this gives a random
     positive vector with sum = 1.

2. **Iterative improvement**
   * In each iteration the full convolution is obtained with an FFT
     (O(n log n)).
   * The index `k_max` where the convolution attains its maximum is
     identified.
   * The gradient of the *max* term with respect to the coefficients is
     `g[i] = 2 * a[k_max‑i]` for all feasible `i`.  Because the denominator
     `sum(a)**2` is kept constant (we always re‑normalise to sum = 1) the
     gradient of the whole score reduces to `2 * n * g`.
   * We take a small step opposite to the gradient, clip negatives to 0,
     and renormalise the vector.  A tiny amount of random noise is added
     occasionally to avoid stagnation.
   * The current and best scores are tracked; the best vector found so
     far is kept.

3. **Return**
   * When the allotted time (default 1000 s) expires the best vector is
     returned as a plain Python list.  The returned sequence can be fed
     directly to the supplied `evaluate_sequence` function.

The method is deliberately lightweight – it needs no external solvers,
uses only NumPy/FFT, and respects the 1000 s wall‑clock budget.
Even if the theoretical optimum (rating = 2) cannot be beaten, the
procedure constantly searches for improvements and will at least return
a valid sequence (the uniform distribution) if nothing better is found.
"""

import numpy as np
import time


def _conv_max(a: np.ndarray) -> tuple[float, int, np.ndarray]:
    """
    Compute the full convolution a⋆a via FFT, return its maximum value,
    the index where it occurs, and the whole convolution array.
    """
    n = a.shape[0]
    # length of the full convolution
    m = 2 * n - 1
    # FFT of the zero‑padded vector
    fft_a = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fft_a * fft_a, n=m)
    # Numerical noise may produce tiny negative values – clamp
    conv = np.maximum(conv, 0.0)
    k = int(np.argmax(conv))
    return float(conv[k]), k, conv


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    length: int = 2048,
    init_noise: float = 1.0,
    step_scale: float = 0.02,
    noise_every: int = 500,
) -> list[float]:
    """
    Search for a coefficient vector that minimises ``evaluate_sequence``.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (the function must return before this time).
    length : int
        Desired length of the step function (default 2048 – a power of two
        is convenient for FFT).
    init_noise : float
        Scale of the initial Dirichlet draw; the raw draws are Gamma(1,1)
        and normalised to sum = 1.
    step_scale : float
        Base step length for the gradient descent move (relative to the
        current coefficient magnitude).
    noise_every : int
        How many iterations between adding a tiny random perturbation
        (to help escape flat regions).

    Returns
    -------
    list[float]
        The best coefficient sequence found within the time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # leave a safety margin

    n = length
    # ---- initial random vector (Dirichlet) ---------------------------------
    a = np.random.gamma(shape=1.0, scale=1.0, size=n)
    a /= a.sum()                     # normalise to sum = 1
    a = np.clip(a, 0.0, None)        # safety, should already be non‑negative

    # evaluate the current score
    max_c, k_max, _ = _conv_max(a)
    cur_score = 2.0 * n * max_c       # denominator is 1 because sum(a)=1
    best_a = a.copy()
    best_score = cur_score

    iteration = 0
    # simple linear temperature decay for occasional uphill moves
    T0 = 1.0
    T1 = 1e-4

    print(f"[START] length={n}, initial_score={cur_score:.6f}")

    while time.time() < deadline:
        # recompute convolution and gradient
        max_c, k_max, _ = _conv_max(a)
        cur_score = 2.0 * n * max_c

        # keep the best solution
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{iteration:07d}] NEW BEST score={best_score:.6f}")

        # ---- gradient of the max‑convolution term -------------------------
        # g[i] = 2 * a[k_max - i]  (if the index is inside the array)
        idx = np.arange(n)
        rev_idx = k_max - idx
        mask = (rev_idx >= 0) & (rev_idx < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[rev_idx[mask]]

        # temperature for occasional uphill acceptance (simulated annealing)
        t_frac = (deadline - time.time()) / (budget_s - 1.0)
        T = T0 * (T1 / T0) ** (1.0 - t_frac)

        # step size – a small fraction of the gradient magnitude
        step = step_scale * g
        # stochastic acceptance: downhill always, uphill with prob exp(-Δ/T)
        a_new = a - step
        a_new = np.maximum(a_new, 0.0)          # non‑negativity
        a_new /= a_new.sum()                    # renormalise to sum = 1

        # evaluate trial score
        max_c_new, _, _ = _conv_max(a_new)
        new_score = 2.0 * n * max_c_new

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # occasional uphill moves (simulated annealing)
            if np.random.rand() < np.exp(-(new_score - cur_score) / max(T, 1e-12)):
                accept = True

        if accept:
            a = a_new
            cur_score = new_score

        # occasional random jitter to avoid getting stuck in a flat region
        if iteration % noise_every == 0 and iteration > 0:
            jitter = np.random.normal(scale=1e-4, size=n)
            a += jitter
            a = np.clip(a, 0.0, None)
            a /= a.sum()

        iteration += 1

    print(f"[END] iterations={iteration}, best_score={best_score:.6f}")
    # The external evaluator will clamp values to [0,1000] anyway,
    # but we stay comfortably inside that range.
    return best_a.tolist()
