def propose_candidate(seed=123, budget_s=1000):
    import numpy as np, time, random
    from scipy.optimize import linprog
    # Use height_sequence_1 if available else random uniform start
    try:
        seq = list(height_sequence_1)
    except NameError:
        seq = np.random.rand(200)  # random
    # If too long, truncate to max_len=1000 for LP speed
    max_len = 1000
    if len(seq) > max_len:
        seq = seq[:max_len]
    # Ensure positive
    seq = [max(0.0, float(x)) for x in seq]
    best_seq = seq[:]
    best_score = evaluate_sequence(best_seq)
    deadline = time.time() + budget_s - 5
    # Parameters
    t_candidates = [0.05, 0.1, 0.2, 0.4, 0.6, 0.8]
    # function for LP
    def solve_dir(f, M):
        n = len(f)
        # Build constraint matrix A for conv(f, b) <= M
        # Use dense for simplicity given moderate n
        A = np.zeros((2*n-1, n))
        for i in range(n):
            A[i:i+n, i] = f  # fill column i
        A_ub = np.vstack([A, -np.eye(n)])
        b_ub = np.concatenate([np.full(2*n-1, M), np.zeros(n)])
        c = -np.ones(n)  # maximize sum
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', options={'time_limit':5})
        if res.success:
            return res.x
        return None
    # Main loop
    while time.time() < deadline:
        f = np.array(best_seq)
        M = float(np.max(np.convolve(f, f)))
        # Solve direction
        g = solve_dir(f, M)
        if g is None:
            # random perturbation
            idx = np.random.randint(len(best_seq))
            best_seq[idx] = max(0.0, best_seq[idx] + np.random.randn()*0.01)
            continue
        # Normalize g to have same L1 norm as f (optional)
        # Actually we can just mix
        for t in t_candidates:
            new_seq = (1 - t) * f + t * g
            score = evaluate_sequence(new_seq.tolist())
            if score < best_score:
                best_score = score
                best_seq = new_seq.tolist()
                print(f"Improved to {best_score:.6f} with t={t}")
        # Occasionally random restart
        if np.random.rand() < 0.05:
            # perturb
            new_seq = np.maximum(0.0, np.array(best_seq) + np.random.randn(len(best_seq))*0.05)
            # normalize sum to original
            new_sum = np.sum(new_seq)
            if new_sum > 0.01:
                best_seq = (new_seq / new_sum * np.sum(best_seq)).tolist()
    # Return best found
    return best_seq

"""
Propose a candidate step‑function sequence that improves the evaluation score.

The algorithm works as follows:

1.  Initialise from the global `height_sequence_1` if it exists;
    otherwise fall back to a random start.
    To keep the linear program tractable we truncate (or down‑sample) the
    sequence to at most `MAX_LEN` elements (default 1500).

2.  Repeatedly solve a linear programme

        max   sum(b)
        s.t.  conv(a, b) ≤ M
              b ≥ 0

    where `a` is the current sequence and `M = max(conv(a,a))`.
    The LP finds a direction `b` that raises the ℓ¹‑norm (the sum) of the
    sequence without increasing the current maximal convolution value.

3.  After a feasible `b` is obtained we try several mixing weights
    `t ∈ {0.05,0.1,0.2,0.4,0.6,0.8}` and keep the mixture

        a_new = (1‑t)·a + t·b

    if it yields a strictly lower evaluation score.  Mixing with a larger
    `t` often gives a bigger gain because the quadratic term
    `t²·conv(b,b)` is still modest for moderate `t`.

4.  If the LP fails (or does not increase the sum) we apply a tiny random
    perturbation to escape possible stagnation.

5.  The loop runs until the allocated time budget expires (the caller
    passes `budget_s`, default 1000 s) and finally returns the best
    sequence found.

The whole procedure stays within the 1000 s limit, uses only NumPy and
SciPy, and respects the non‑negative and bounded‑value requirements of
`evaluate_sequence`.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper: build the dense constraint matrix for conv(a, b) ≤ M
def _conv_constraint_matrix(a: np.ndarray) -> np.ndarray:
    """
    For a given vector `a` (length n) return a matrix A of shape
    (2*n‑1, n) such that  A @ b == conv(a, b) (full convolution).
    """
    n = a.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # each column i contains the vector `a` starting at row i
    for i in range(n):
        A[i:i + n, i] = a
    return A


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) beats the current
    best score.  The function may be called with a time budget; it will
    stop a little before the limit and return the best sequence found.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Total time budget in seconds.
    kwargs : dict
        Ignored – present for compatibility with the evaluation harness.

    Returns
    -------
    list[float]
        The best sequence discovered within the time budget.
    """
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # 1. Initialise ----------------------------------------------------
    MAX_LEN = 1500                     # size that keeps the LP reasonable
    try:
        # height_sequence_1 should be provided by the harness
        init_seq = list(height_sequence_1)
    except NameError:
        init_seq = None

    if init_seq is None or len(init_seq) == 0:
        # random start – uniform values that already satisfy the sum > 0.01
        n0 = np.random.randint(100, MAX_LEN)
        init_seq = list(np.random.rand(n0) + 0.1)   # avoid zeros
    else:
        # Keep at most MAX_LEN entries (simple truncation)
        if len(init_seq) > MAX_LEN:
            init_seq = init_seq[:MAX_LEN]
        # make sure we have no negative numbers (the harness will clamp,
        # but keeping them non‑negative avoids unnecessary work)
        init_seq = [max(0.0, float(x)) for x in init_seq]

    # Ensure the sequence is not trivial
    if np.sum(init_seq) < 0.02:
        init_seq = [0.1] * max(1, len(init_seq))

    best_seq = init_seq[:]
    best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # 2. Main optimisation loop ----------------------------------------
    deadline = time.time() + budget_s - 5.0   # leave a small safety margin
    t_candidates = [0.05, 0.1, 0.2, 0.4, 0.6, 0.8]

    # small helper: solve the LP for the current sequence
    def _solve_lp(a_vec: np.ndarray, max_conv: float):
        """
        Solve   max sum(b)   s.t.   conv(a, b) ≤ max_conv ,  b ≥ 0 .
        Returns the optimal b (numpy array) or None if the problem is
        infeasible or the solver timed out.
        """
        n = a_vec.size
        # Build convex constraints: conv(a,b) ≤ max_conv
        A = _conv_constraint_matrix(a_vec)            # shape (2n-1, n)
        A_ub = np.vstack([A, -np.eye(n)])             # additional non‑negativity
        b_ub = np.concatenate([np.full(A.shape[0], max_conv),
                               np.zeros(n)])
        # Objective: maximise sum(b)  → minimise -sum(b)
        c = -np.ones(n, dtype=np.float64)

        # HiGHS solver with a modest time limit per LP
        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      method='highs',
                      options={'time_limit': 5.0, 'disp': False})
        if res.success and res.x is not None:
            return res.x
        return None

    iteration = 0
    while time.time() < deadline:
        iteration += 1
        a = np.array(best_seq, dtype=np.float64)
        # current maximal convolution value
        conv_full = np.convolve(a, a)
        M = float(conv_full.max())

        # ----------------- LP direction ---------------------------------
        g = _solve_lp(a, M)
        if g is None:
            # fall back to a tiny random perturbation to avoid being stuck
            idx = np.random.randint(len(best_seq))
            pert = np.random.randn() * 0.01
            best_seq[idx] = max(0.0, best_seq[idx] + pert)
            continue

        sum_a = a.sum()
        sum_g = g.sum()

        # If the LP did not increase the ℓ¹‑norm, try a random shake and
        # continue – otherwise we would loop without progress.
        if sum_g <= sum_a * (1.0 + 1e-6):
            # random shake of a few coordinates
            for _ in range(3):
                i = np.random.randint(len(best_seq))
                best_seq[i] = max(0.0, best_seq[i] + np.random.randn() * 0.02)
            continue

        # ----------------- try different mixing factors ---------------
        improved = False
        for t in t_candidates:
            new_a = (1.0 - t) * a + t * g
            # evaluate – `evaluate_sequence` already clamps negatives,
            # large values and enforces the lower bound on the sum.
            try:
                score = evaluate_sequence(new_a.tolist())
            except Exception:
                continue
            if score < best_score:
                best_score = score
                best_seq = new_a.tolist()
                improved = True
                print(f"[Iter {iteration}] t={t:.2f} → score {best_score:.6f}")
                # stop trying larger t if we already improved a lot
                # (but continue the loop to find even better mixes)
        # ----------------------------------------------------------------
        # occasional pure random restart to explore new basins
        if not improved and np.random.rand() < 0.04:
            # create a fresh random sequence of the same length
            rand_seq = np.maximum(0.0,
                                  a + 0.1 * np.random.randn(a.size))
            # keep the total sum roughly the same (helps the score)
            s = rand_seq.sum()
            if s > 0.01:
                rand_seq = rand_seq * (sum_a / s)
                cand_score = evaluate_sequence(rand_seq.tolist())
                if cand_score < best_score:
                    best_score = cand_score
                    best_seq = rand_seq.tolist()
                    print(f"[Iter {iteration}] random restart improved to {best_score:.6f}")

        # Early exit if we already beat the target bound
        if best_score < 1.503:
            print(f"Target reached early: score {best_score:.6f} at iteration {iteration}")
            break

    # ------------------------------------------------------------------
    # Return the best sequence discovered.
    # The harness expects a plain Python list of floats.
    return best_seq
