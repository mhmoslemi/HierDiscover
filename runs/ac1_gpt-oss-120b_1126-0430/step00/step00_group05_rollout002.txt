"""
Gradient‑based optimizer for the step‑function evaluation.

The objective to minimise is

    F(a) = 2 * n * max(conv(a,a)) / (sum(a)**2)

which is scale‑invariant.  We fix the ℓ1‑norm of the coefficient vector to 1,
so the objective becomes 2·n·max(conv).

Because the max is non‑smooth we replace it by a smooth approximation
using a soft‑max (log‑sum‑exp) with a temperature β that is slowly increased.
The gradient of the smoothed max can be written as a cross‑correlation of the
current vector `a` with the soft‑max probabilities, which we compute efficiently
with FFTs.  An Adam‐style update is performed, after which the vector is
projected onto the simplex (non‑negative and ℓ1‑norm = 1).

The routine runs for the allotted time (default 1000 s) and returns the best
sequence found.
"""

import time
import numpy as np

# -------------------------------- helper functions --------------------------------

def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _logsumexp_beta(beta: float, x: np.ndarray) -> float:
    """
    Compute (1/β) * log(∑ exp(β·x)) in a numerically stable way.
    """
    mx = np.max(x)
    # work with shifted values to avoid overflow
    shifted = beta * (x - mx)
    return mx + np.log(np.sum(np.exp(shifted))) / beta


def _softmax_beta(beta: float, x: np.ndarray, M_beta: float) -> np.ndarray:
    """
    Return the soft‑max probabilities p_i = exp(β·x_i - β·M_β)
    where M_β = (1/β)·log∑exp(β·x_i).
    """
    # x ≤ M_β , so exponent never exceeds 1.
    return np.exp(beta * (x - M_beta))


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·n‑1) using FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return conv


def _cross_corr(a: np.ndarray, p: np.ndarray) -> np.ndarray:
    """
    Linear cross‑correlation (a ⋆ p)[m] = Σ_i a_i·p_{i+m} for m = 0 … n‑1.
    Implemented via FFT.  Returns an array of length n.
    """
    n = a.size
    T = p.size          # = 2·n‑1
    L = _next_pow2(n + T - 1)  # size for linear correlation (3·n‑2)

    # FFTs of zero‑padded sequences
    A = np.fft.rfft(a, n=L)
    P = np.fft.rfft(p, n=L)

    # correlation (circular) → take first n+T‑1 entries for the linear version
    corr_full = np.fft.irfft(np.conj(A) * P, n=L)

    # shift to obtain non‑negative m‑indices:
    start = T - 1            # index that corresponds to m = 0
    return corr_full[start : start + n]


# -------------------------------- main optimisation --------------------------------

def propose_candidate(seed: int = 0, budget_s: float = 1000.0, **kwargs):
    """
    Gradient‑descent optimiser for `evaluate_sequence`.
    Returns the best (list of floats) sequence found before `budget_s` seconds.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # a small safety margin

    # ---- initialise a starting vector -------------------------------------------------
    if 'height_sequence_1' in globals():
        a = np.array(height_sequence_1, dtype=float)
        a = np.maximum(a, 0.0)
        if a.sum() == 0.0:
            a = np.ones_like(a)
    else:
        # fall back to a random vector of size ≈ 5000
        n = 5400
        a = np.random.rand(n)

    # Normalise to ℓ1‑norm = 1 (objective is scale‑invariant)
    a = a / a.sum()

    n = a.size
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] length={n}, score={best_score:.6f}")

    # ---- optimisation hyper‑parameters -------------------------------------------------
    lr = 0.2               # base learning rate for Adam
    beta = 0.5             # initial temperature for soft‑max
    beta_max = 20.0
    beta_inc = 1.025       # increase factor per iteration
    eps = 1e-12

    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    it = 0

    # ---- main loop --------------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # current ℓ1 norm (should be 1 but we recompute for safety)
        S = a.sum()
        if S == 0.0:
            a = np.random.rand(n)
            a = a / a.sum()
            continue

        # ----- convolution and smoothed max -------------------------------------------
        conv = _conv_self(a)                     # length 2n‑1
        M_beta = _logsumexp_beta(beta, conv)     # smooth approximation of max(conv)
        prob = _softmax_beta(beta, conv, M_beta) # probabilities (sum = 1)

        # ----- gradient of the smoothed max -------------------------------------------
        grad_M = 2.0 * _cross_corr(a, prob)      # shape (n,)

        # ----- full gradient of the objective -----------------------------------------
        coef = (2.0 * n) / (S * S)               # overall scalar factor
        bias = (2.0 * M_beta / S) * np.ones_like(a)
        grad = coef * (grad_M - bias)            # ∇F(a)

        # ----- Adam update ------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = -lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a + step

        # ----- projection onto the simplex (non‑negative, ℓ1‑norm = 1) ---------------
        a_new = np.maximum(a_new, 0.0)
        total = a_new.sum()
        if total == 0.0:
            a_new = np.random.rand(n)
            total = a_new.sum()
        a_new = a_new / total

        # ----- evaluate ---------------------------------------------------------------
        try:
            cur_score = evaluate_sequence(a_new.tolist())
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a_new.copy()
                print(f"[iter {it:5d}] new best = {best_score:.6f}  β={beta:.3f}  lr={lr:.5f}")
        except Exception:
            # any unexpected behaviour from evaluate_sequence is ignored
            pass

        a = a_new

        # ----- schedule updates -------------------------------------------------------
        beta = min(beta_max, beta * beta_inc)
        lr *= 0.9995   # gentle decay

        # early stop if we already beat the target comfortably
        if best_score < 1.55:
            print("[stop] reached target region")
            break

    print(f"[done] iterations={it}, best_score={best_score:.6f}")
    return best_seq.tolist()
