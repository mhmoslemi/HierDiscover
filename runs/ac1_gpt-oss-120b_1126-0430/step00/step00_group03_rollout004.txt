def simplex_projection(v, s):
    # ensure v>=0 and sum = s.
    n = len(v)
    if s < 0:
        raise
    # similar to algorithm by Wang & Carreira-Perpinan (2013)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def geometric_sequence(n, S, c_factor):
    # c_factor is alpha such that (1 - q) = alpha / n
    # Ensure alpha not larger than n otherwise q negative.

"""
Geometric + local refinement search.

The evaluation functional is
    2 * n * max(conv(a,a)) / (sum(a)**2)

For a fixed length n we can normalise the sum to sqrt(2*n); then the
score equals the maximal convolution entry.
Analytic considerations show that a geometric progression
    a_i = C * q**i     (0 < q < 1)
with  q ≈ 1 - c/n  (c ≈ 1) makes the maximal convolution ≈
(2/e) * n * (1-q)  ≈ 1.5 for n≈2000.  The code enumerates several
lengths and several values of the parameter c, evaluates the exact
score (using the provided ``evaluate_sequence``) and keeps the best
candidate.  After the coarse grid search a few projected‑gradient
steps are applied to the current best vector to squeeze a little more
out of the maximum.

All helper functions are defined at top level as required.
"""

import numpy as np
import time

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the simplex {x >= 0, sum(x)=s}
def project_onto_simplex(v, s):
    """Project vector v onto the simplex of non‑negative numbers with sum s."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    # Sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    # Find the rho where u_i - cssv_i / i > 0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries will be zero
        return np.zeros_like(v)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Compute a geometric sequence with parameters (n, sum_target, c_factor)
def geometric_sequence(n, sum_target, c_factor):
    """
    Returns a list a_i = C * q**i, i=0..n-1,
    where (1-q) = c_factor / n  (so q = 1 - c_factor/n)
    and the sum of the sequence equals sum_target.
    """
    one_minus_q = c_factor / n
    if one_minus_q <= 0.0 or one_minus_q >= 1.0:
        # Invalid, fall back to uniform
        return [float(sum_target) / n] * n
    q = 1.0 - one_minus_q
    # use logs for stability when q is close to 1
    log_q = np.log(q)
    # denominator of geometric series: 1 - q**n
    pow_q_n = np.exp(log_q * n)          # q**n
    denom = 1.0 - pow_q_n
    if denom == 0.0:
        C = 0.0
    else:
        C = sum_target * one_minus_q / denom
    # generate the sequence
    i = np.arange(n, dtype=float)
    seq = C * np.exp(log_q * i)          # C * q**i
    # ensure non‑negativity (numerical safety)
    seq[seq < 0] = 0.0
    return seq.tolist()

# ----------------------------------------------------------------------
# One projected‑gradient refinement step (keeps the sum fixed)
def gradient_refine(seq, max_iter=30, time_deadline=None):
    """
    Perform a few gradient‑descent steps on the maximal convolution entry.
    The sum of the vector is kept constant by a simplex projection.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]
    # target sum (won’t change during refinement)
    S = np.sum(a)
    step = 0.1 * S / max(1, n)  # initial step magnitude

    for _ in range(max_iter):
        if time_deadline is not None and time.time() > time_deadline:
            break
        conv = np.convolve(a, a)
        k_max = int(np.argmax(conv))
        # gradient of the maximal entry: 2 * a_{k_max-i} where defined
        grad = np.zeros_like(a)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        grad[idx] = 2.0 * a[k_max - idx]

        # try a step, adapt step size if needed
        improved = False
        for _ in range(5):
            cand = a - step * grad
            cand = project_onto_simplex(cand, S)
            # evaluate quickly – use the external function for exact score
            cand_score = evaluate_sequence(cand.tolist())
            cur_score = evaluate_sequence(a.tolist())
            if cand_score < cur_score:
                a = cand
                step *= 1.05
                improved = True
                break
            else:
                step *= 0.5
        if not improved:
            # small random perturbation to escape plateaus
            idx = np.random.randint(n)
            a[idx] = max(0.0, a[idx] + np.random.randn() * step)
            a = project_onto_simplex(a, S)
    return a.tolist()

# ----------------------------------------------------------------------
def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Search for a step‑function (non‑negative sequence) with a low
    evaluation score.  The method scans a few lengths and geometric
    parameters, keeps the best, and finally refines it with a few
    projected‑gradient steps.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 1) Coarse grid: try several lengths and geometric factors
    n_candidates = [500, 800, 1200, 1600, 2000, 2500, 3000, 4000, 5000]
    # The theoretical optimum corresponds to c ≈ 1 (since (1-q)=c/n)
    c_fine = np.linspace(0.5, 2.5, 41)   # 0.5 … 2.5 step 0.05

    for n in n_candidates:
        if time.time() >= deadline:
            break
        sum_target = np.sqrt(2.0 * n)   # normalise for direct comparison
        # Predict a good c around 1.0 – also sweep a broader range
        for c_factor in c_fine:
            if time.time() >= deadline:
                break
            seq = geometric_sequence(n, sum_target, c_factor)
            # Evaluate the candidate (the function handles all checks)
            try:
                score = evaluate_sequence(seq)
            except Exception:
                continue
            if np.isnan(score) or score == np.inf:
                continue
            if score < best_score:
                best_score = float(score)
                best_seq = seq
                print(f"[{time.time():.2f}] New best: score={best_score:.6f} "
                      f"n={n} c={c_factor:.4f}")

    # ------------------------------------------------------------------
    # 2) Local refinement of the best candidate found so far
    if best_seq is not None:
        # give a little more time for refinement if budget permits
        refine_deadline = deadline
        refined = gradient_refine(best_seq, max_iter=40,
                                 time_deadline=refine_deadline)
        # evaluate refined version
        try:
            refined_score = evaluate_sequence(refined)
        except Exception:
            refined_score = np.inf
        if refined_score < best_score:
            best_score = float(refined_score)
            best_seq = refined
            print(f"[{time.time():.2f}] Refined best: score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Fallback – if nothing was produced, use the provided baseline
    if best_seq is None:
        # height_sequence_1 is assumed to be defined globally
        best_seq = list(height_sequence_1)

    # Ensure the returned object is a plain Python list of floats
    return [float(x) for x in best_seq]
