import cvxpy as cp

def solve_lp(f):
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [cp.sum(g) == 1]
    # convolution constraints:
    # For each k, sum_i f_i * g_{k-i} <= M
    # This is (conv of f and g)[k] <= M
    # We can implement using cp.conv? cvxpy has convolve? Not directly. But we can generate constraints via loops.
    for k in range(2*n-1):
        # compute inner product
        s = 0
        start = max(0, k - (n-1))
        end = min(k, n-1)
        # indices i in [start, end], j = k - i
        # sum_i f[i] * g[k-i]
        s = cp.sum(cp.multiply(f[start:end+1], g[k - np.arange(start, end+1)]))
        constraints.append(s <= M)
    objective = cp.Minimize(M)
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True)
    return g.value, M.value

"""
Projective soft‑max gradient descent optimizer for the max‑convolution
objective.

The objective to minimise is

    O(a) = 2·n·max_k (a * a)[k]                (sum(a)=1, a_i ≥ 0)

where “*” denotes the linear convolution.
The optimiser works on the probability simplex:

    • a is always projected onto {x ≥ 0, Σ x = 1}.
    • At each iteration the self‑convolution is obtained with an FFT
      (O(n log n)).
    • A smooth approximation of the ℓ∞‑norm is built with a soft‑max
      (exp(λ·(conv‑max))) whose temperature λ grows from 2 to ≈30.
    • The gradient of the soft‑max is computed analytically; it is a
      second convolution that can also be performed with an FFT.
    • An Adam step is taken and the result is projected back onto the
      simplex.
    • Several random restarts (different initial points and hyper‑
      parameters) are performed while respecting the 1000 s budget.
    • The best sequence discovered (lowest O) is finally evaluated
      with the supplied ``evaluate_sequence`` routine and returned.

Because the optimisation lives on the simplex the algorithm is
scale‑invariant, but the length *n* influences the bound (larger *n*
gives a larger factor 2·n).  The code automatically takes the length
of the previously discovered best sequence (if available) as a
reasonable starting size, otherwise it uses a default of 4000.
"""
import time
import numpy as np

# --------------------------------------------------------------------
# Simple projection onto the probability simplex (Duchi et al.)
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x ≥ 0, sum(x) = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:                     # all non‑positive → uniform point
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------
# Fast convolution via real FFT (returns full linear convolution)
def _conv_fft(x: np.ndarray, y: np.ndarray, nfft: int) -> np.ndarray:
    fx = np.fft.rfft(x, n=nfft)
    fy = np.fft.rfft(y, n=nfft)
    conv_full = np.fft.irfft(fx * fy, n=nfft)
    return conv_full[: x.size + y.size - 1]   # length = len(x)+len(y)-1


# --------------------------------------------------------------------
# Objective used during the search (sum(a) is forced to 1)
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)).  Assumes Σ a = 1."""
    conv = np.convolve(a, a)
    return 2.0 * a.size * conv.max()

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine runs several Adam‑based
    gradient descents on the soft‑max surrogate of the ℓ∞ norm of the
    self‑convolution, always projecting onto the simplex.
    The best sequence found before the time budget expires is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5          # safety margin

    # ----------------------------------------------------------------
    # Choose a sensible length.
    if 'height_sequence_1' in globals():
        base_len = len(height_sequence_1)
    else:
        base_len = 4000
    n = max(2000, base_len)                         # never go below 2 k

    # Pre‑compute FFT length (next power‑of‑two ≥ 2·n‑1)
    nfft = 1 << int(np.ceil(np.log2(2 * n - 1)))

    best_seq = None
    best_score = float('inf')
    restart = 0

    while time.time() < deadline and restart < 15:
        restart += 1

        # ---- initialise a fresh point on the simplex -----------------
        if best_seq is not None and np.random.rand() < 0.3:

            a = best_seq.copy()
            a += np.random.randn(n) * 0.001

        else:

            a = _project_to_simplex(a)

        v = np.zeros_like(a)

        eps = 1e-12
        lr0 = 0.2 * (0.5 + np.random.rand())          # base learning rate ∈ [0.1,0.3]

        # finish aggressive.

            lam = 2.0 + 28.0 * (iter_cnt - 1) / max_iters
            # to avoid overflow we shift by the maximum (always ≤ 0 after shift)
            weights = np.exp(lam * (conv - maxconv))
            wsum = weights.sum()
            if wsum == 0.0:                 # numerical safeguard – fall back to true sub‑gradient
                # sub‑gradient of the actual max (single active index)
                # (identical to the helper used previously)
                ks = np.flatnonzero(conv == maxconv)
                grad = np.zeros_like(a)
                for k in ks:
                    i0 = max(0, k - (n - 1))
                    i1 = min(k, n - 1)
                    idx = np.arange(i0, i1 + 1, dtype=int)
                    grad[idx] += 2.0 * a[k - idx]
                grad /= len(ks)
            else:
                weights /= wsum                                   # normalise

                # ---- gradient of the soft‑max surrogate -------------------------
                # grad_i = 2 * Σ_k weights_k * a_{k‑i}
                # This is a convolution of a with the reversed weight vector.
                w_rev = weights[::-1]
                conv_rev = _conv_fft(a, w_rev, nfft)               # length 3·n‑2
                # needed slice: indices 2·n‑2‑i for i=0..n‑1  →  slice [n‑1 : 2·n‑1] reversed
                grad = 2.0 * conv_rev[n - 1 : 2 * n - 1][::-1]

            # ---- Adam step ----------------------------------------------------
            lr = lr0 * (1.0 / np.sqrt(iter_cnt))                   # simple decay
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** iter_cnt)
            v_hat = v / (1.0 - beta2 ** iter_cnt)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            a = a - step                                          # descent
            a = np.clip(a, 0, None)                               # enforce non‑negativity
            a = _project_to_simplex(a)                            # back onto Σ a = 1

            # ---- occasional random mass‑transfer to escape flat regions -----
            if iter_cnt % 500 == 0 and np.random.rand() < 0.2:
                i, j = np.random.randint(n, size=2)
                if i != j and a[i] > 0.0:
                    delta = a[i] * 0.1 * np.random.rand()
                    a[i] -= delta
                    a[j] += delta
                    a = _project_to_simplex(a)

        # end of this restart ----------------------------------------------------
        # Tiny pause for the outer time check
        if time.time() >= deadline:
            break

    # --------------------------------------------------------------------
    # Final verification with the official evaluator (optional safety)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence call failed:", exc)

    return best_seq.tolist()
