```python
import time
import numpy as np
from scipy.optimize import linprog

def arcsine_sequence(n):
    seq = np.empty(n, dtype=np.float64)
    a = 1.0
    seq[0] = a
    for i in range(1, n):
        a *= (2*i - 1) / (2.0 * i)
        seq[i] = a
    return seq

def fast_score(seq):
    n = seq.size
    # Compute convolution via FFT
    m = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[:2*n-1]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a), max_b

def build_constraint_matrix(f):
    n = f.size
    rows = 2*n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        A[j:j+n, j] = f  # put f in column j shifted down by j
    return A
```

"""
Iterative Frank‑Wolfe optimisation for the step‑function bound.

The algorithm starts from the discrete arcsine distribution (Catalan
coefficients) whose self‑convolution is (almost) constant, giving the
classical bound π/2≈1.571.  In each iteration we solve a linear program

    max   Σ g_j    subject to   f * g ≤ M   ,  g ≥ 0

where M = max(f*f).  The LP finds a direction that can increase the
ℓ₁‑norm without raising the convolution maximum.  We rescale the
solution to keep the ℓ₁‑norm unchanged, form a convex combination with
the current iterate and keep the combination only when the maximum
convolution value decreases.  Repeating this “Frank‑Wolfe” step drives
M down while the ℓ₁‑norm stays constant, thereby lowering the evaluation
score 2·n·M/∥f∥₁².  After the LP phase a cheap greedy refinement
(which moves mass from indices that heavily contribute to the peak to
almost‑inactive indices) is applied to squeeze out the last bits of
improvement.

The routine respects the 1000 s budget, works with 2 000–2 500
coefficients and returns the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper: fast evaluation of the objective (no security checks)
# ----------------------------------------------------------------------
def _fast_score(seq: np.ndarray) -> tuple[float, float]:
    """
    Return (score, max_convolution) for a non‑negative sequence.

    score = 2 * n * max(conv(seq,seq)) / (sum(seq) ** 2)
    """
    n = seq.size
    if n == 0:
        return np.inf, np.inf
    # FFT size: next power of two >= 2*n-1
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(seq.sum())
    if s < 0.01:
        return np.inf, max_b
    score = 2.0 * n * max_b / (s * s)
    return score, max_b


# ----------------------------------------------------------------------
# Generate the arcsine (Catalan) seed: a_i = binom(2i,i)/4**i
# ----------------------------------------------------------------------
def _arcsine_sequence(length: int) -> np.ndarray:
    """Return the first ``length`` arcsine coefficients as a float array."""
    seq = np.empty(length, dtype=np.float64)
    a = 1.0
    seq[0] = a
    for i in range(1, length):
        # recurrence a_i = a_{i-1} * (2*i-1)/(2*i)
        a *= (2 * i - 1) / (2.0 * i)
        seq[i] = a
    return seq


# ----------------------------------------------------------------------
# Build the Toeplitz‑like matrix that encodes the convolution constraints
#   (f * g)[k] = Σ_i f_i * g_{k-i}  ≤  M   for all k.
# ----------------------------------------------------------------------
def _constraint_matrix(f: np.ndarray) -> np.ndarray:
    """
    Return a dense matrix A of shape (2*len(f)-1, len(f)) such that
    A @ g equals the self‑convolution f*g.
    """
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # each column j is a copy of f shifted down by j positions
    for j in range(n):
        A[j : j + n, j] = f
    return A


# ----------------------------------------------------------------------
# Greedy local improvement (identical to the earlier prototype)
# ----------------------------------------------------------------------
def _greedy_refine(seq: np.ndarray, deadline: float) -> np.ndarray:
    """
    Simple mass‑moving refinement that never raises the current maximal
    convolution entry.  Runs until time runs out.
    """
    n = seq.size
    best_seq = seq.copy()
    best_score, _ = _fast_score(best_seq)

    # pre‑compute FFT length once
    fft_len = 1 << ((2 * n - 1).bit_length())

    while time.time() < deadline:
        # current convolution and the index where it is maximal
        fa = np.fft.rfft(seq, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        k = int(conv.argmax())
        # contribution weights for each position to that maximal entry
        i_arr = np.arange(n)
        j_arr = k - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        w = np.zeros_like(seq)
        w[mask] = seq[j_arr[mask]]

        # pick a source (largest contribution) and a sink (almost none)
        contrib = seq * w
        src = int(np.argmax(contrib))
        sink = int(np.argmin(w))   # often a location with zero contribution

        if seq[src] < 1e-12:
            break   # nothing worth moving

        eps = 0.05 * seq[src]      # move 5 % of the source amount
        new_seq = seq.copy()
        new_seq[src] -= eps
        new_seq[sink] += eps

        new_score, _ = _fast_score(new_seq)
        if new_score < best_score - 1e-9:
            seq = new_seq
            best_score = new_score
            best_seq = new_seq.copy()
        else:
            # occasional tiny jitter
            if np.random.rand() < 0.01:
                jitter_idx = np.random.randint(n)
                jitter = np.random.randn() * 0.001
                new_seq = seq.copy()
                new_seq[jitter_idx] = max(0.0, new_seq[jitter_idx] + jitter)
                seq = new_seq
    return best_seq


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Produce a list of non‑negative heights for the step function.

    The routine respects ``budget_s`` seconds (default 1000).  It starts
    from the arcsine sequence, performs an LP‑based Frank‑Wolfe style
    optimisation that reduces the maximal self‑convolution while keeping
    the ℓ₁‑norm fixed, and finally applies a cheap greedy polishing step.
    The best sequence encountered is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0          # a small safety margin

    # ------------------------------------------------------------------
    # 1.  Choose a length that balances quality and LP cost.
    # ------------------------------------------------------------------
    n = 2500                      # 2 500 works comfortably within the budget
    f = _arcsine_sequence(n)
    f = np.clip(f, 0.0, 1000.0)   # just in case
    best_score, M = _fast_score(f)
    best_seq = f.copy()

    print(f"[init] n={n}, initial score={best_score:.6f}, M={M:.6f}")

    iteration = 0
    # ------------------------------------------------------------------
    # 2.  Frank‑Wolfe loop based on the linear program described in the
    #     statement.  We stop when time runs out or when no improvement
    #     is observed for a few successive iterations.
    # ------------------------------------------------------------------
    stagnant = 0
    while time.time() < deadline and iteration < 80:
        iteration += 1

        # --- build LP constraints for the current f --------------------
        A = _constraint_matrix(f)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)                # minimise -sum(g)

        # upper bound 1000 for each variable (the security check clips >1000)
        bounds = [(0.0, 1000.0) for _ in range(n)]

        # --- solve LP --------------------------------------------------
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if not res.success:
            print(f"[lp] iteration {iteration} – LP failed: {res.message}")
            break

        g0 = res.x
        sum_f = f.sum()
        sum_g0 = g0.sum()
        if sum_g0 <= 0:
            print("[lp] zero‑sum solution, aborting.")
            break

        # scale g0 to have the same ℓ₁‑norm as f
        scale = sum_f / sum_g0
        g = g0 * scale

        # --- try several convex‑combination weights --------------------
        improved = False
        # larger steps first; they are cheaper to test
        for t in (1.0, 0.5, 0.2, 0.1, 0.05, 0.02):
            f_candidate = (1.0 - t) * f + t * g
            # clip to the permitted range before scoring
            f_candidate = np.clip(f_candidate, 0.0, 1000.0)
            cand_score, cand_M = _fast_score(f_candidate)

            if cand_score < best_score - 1e-9:
                print(f"[iter {iteration}] t={t:.2f}  score={cand_score:.6f}  M={cand_M:.6f}")
                f = f_candidate
                best_score = cand_score
                best_seq = f_candidate.copy()
                M = cand_M
                improved = True
                stagnant = 0
                break   # accept first improving t

        if not improved:
            stagnant += 1
            # if we have been stagnant for a few rounds, add a tiny random jitter
            if stagnant >= 3:
                jitter = np.random.randn(n) * 1e-3
                f = np.clip(f + jitter, 0.0, 1000.0)
                # recompute score after jitter – it may be slightly better
                best_score, M = _fast_score(f)
                best_seq = f.copy()
                stagnant = 0

        # free the matrix early to keep memory low
        del A

    print(f"[post‑lp] best score after LP phase: {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Final cheap greedy polishing (same idea as the previous
    #     submission).  This typically shaves a few 1e‑4 off the score.
    # ------------------------------------------------------------------
    best_seq = _greedy_refine(best_seq, deadline)

    final_score, _ = _fast_score(best_seq)
    print(f"[final] score after greedy refinement: {final_score:.6f}")

    # Return a plain Python list – the surrounding harness will perform the
    # final security checks and clipping.
    return best_seq.tolist()
