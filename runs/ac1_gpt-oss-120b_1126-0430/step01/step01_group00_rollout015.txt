def improve_a(a):
    n = len(a)
    conv = np.convolve(a, a)
    M = conv.max()
    # solve LP
    c = -np.ones(n)   # maximize sum(b)
    # create Toeplitz matrix A_ub
    first_col = np.concatenate([a, np.zeros(n-1)])
    first_row = np.empty(n)
    first_row[0] = a[0]
    first_row[1:] = 0
    A_ub = scipy.linalg.toeplitz(first_col, first_row)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:

    b = res.x
    sum_b = b.sum()
    if sum_b <= 1 + 1e-12:
        # no improvement possible
        return a, None
    b_scaled = b / sum_b

def lp_max_sum(a, M, max_active_iter=10, slack_tol=1e-12):
    n = len(a)
    # initial active set: indices where conv is within 1% of M
    conv = np.convolve(a, a)
    active = np.where(conv >= M * 0.9999)[0]
    if active.size == 0:
        active = np.array([np.argmax(conv)], dtype=int)
    for _ in range(max_active_iter):
        A = build_constraint_matrix(a, active)
        b_ub = np.full(A.shape[0], M)
        c = -np.ones(n)
        bounds = [(0, None)] * n
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            # fall back to dense solve for robustness
            # build dense matrix
            A_dense = A.toarray()
            res = scipy.optimize.linprog(c, A_ub=A_dense, b_ub=b_ub, bounds=bounds, method='highs')
            if not res.success:
                return None
        b = res.x
        # verify constraints:
        conv_ab = np.convolve(a, b)
        viol = np.where(conv_ab > M + slack_tol)[0]
        if viol.size == 0:
            return b
        # add most violated indices (top few)
        # compute excess:
        excess = conv_ab[viol] - M
        # add up to max_new constraints
        num_new = min(10, viol.size)
        new_idx = viol[np.argsort(excess)[-num_new:]]
        active = np.unique(np.concatenate([active, new_idx]))
    # after loops, return last b (maybe infeasible)
    return b

```
"""Search for step‑function sequence minimizing evaluate_sequence.

Algorithm:
- Initialise with known good sequence (or uniform) on the probability simplex.
- Repeatedly solve a small linear programme: maximise sum(b) subject to the
  convolution of the current sequence a with b staying below the current
  maximal self‑convolution M.  Only the constraints that are active (where
  a*a reaches M) are inserted first; violated constraints are added
  iteratively (cut‑plane method).  The resulting b tends to increase the total
  mass while keeping the peak convolution low.
- Scale b to have the same ℓ1‑norm as a and take a convex combination
  (a←(1−t)a+tb).  The step size t is chosen by a tiny line‑search.
- If the LP provides no improvement, a quick stochastic hill‑climb
  (mass‑transfer moves) is applied.
- The best sequence seen is returned.  The routine stops after the given
  time budget (≈1000 s) and prints intermediate objective values.
"""
```

def _project_to_simplex(v):
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        # uniform
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _lp_max_sum(a, M, max_active_iter=15, slack_tol=1e-12):
    """Return a vector b maximizing sum(b) under  a * b <= M.
    Uses a cut‑plane method that starts from the indices where a*a attains its
    maximum and adds violated constraints until feasibility is reached."""
    n = a.shape[0]
    conv_aa = np.convolve(a, a)
    # start with indices where conv is close to the maximum
    active = np.where(conv_aa >= M * 0.9999)[0]
    if active.size == 0:
        active = np.array([np.argmax(conv_aa)], dtype=int)

    for _ in range(max_active_iter):
        A = _build_constraint_matrix(a, active)
        b_ub = np.full(A.shape[0], M)
        c = -np.ones(n)  # maximise sum(b)
        bounds = [(0, None)] * n
        # linprog accepts sparse A_ub for ‘highs’
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                                    method='highs', options={'presolve': True})
        if not res.success:
            # fallback to dense formulation (rare)
            A_dense = A.toarray()
            res = scipy.optimize.linprog(c, A_ub=A_dense, b_ub=b_ub, bounds=bounds,
                                        method='highs')
            if not res.success:
                return None
        b = res.x
        # check all constraints
        conv_ab = np.convolve(a, b)
        viol = np.where(conv_ab > M + slack_tol)[0]
        if viol.size == 0:
            return b
        # add the most violated constraints (up to a few)
        excess = conv_ab[viol] - M
        # take up to 10 worst ones
        n_new = min(10, viol.size)
        new_idxs = viol[np.argsort(excess)[-n_new:]]
        active = np.unique(np.concatenate([active, new_idxs]))
    # return last b even if still infeasible
    return b

def _local_hill_climb(a, max_steps, deadline):
    """Simple stochastic hill‑climbing that moves a small fraction of mass."""
    n = a.shape[0]
    cur_score = _objective(a)
    improved = False
    for _ in range(max_steps):
        if time.time() >= deadline:
            break
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        if a[i] <= 0:
            continue
        delta = a[i] * np.random.rand() * 0.08  # up to 8% of a[i]
        if delta <= 0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = _project_to_simplex(a_new)
        new_score = _objective(a_new)
        if new_score < cur_score:
            a = a_new
            cur_score = new_score
            improved = True
    return a, improved

"""
Search for a step‑function sequence that minimises the provided
evaluate_sequence() objective.

Algorithm
---------
* Initialise with the best known sequence (height_sequence_1) or a uniform
  distribution, normalised to sum = 1 (the objective is scale‑invariant).
* Repeatedly improve the sequence by solving a small linear programme:
  maximise Σ b_i  subject to  a * b ≤ M (elementwise) and b ≥ 0,
  where M = max(a * a) is the current peak of the self‑convolution.
  Only the constraints that are active (where a*a reaches M) are inserted
  first; any violated constraints are added iteratively (cut‑plane method).
  The LP solution b typically has Σb > 1, i.e. it carries more “mass’’ while
  keeping the convolution peak ≤ M.
* Scale b to the simplex (b̂ = b/Σb) and take a convex combination
  a←(1−t)a+tb̂.  The step size t is chosen from a short list by line‑search;
  the new sequence is kept only if the objective improves.
* If the LP provides no improvement, a quick stochastic hill‑climb (mass
  transfer between coordinates) is applied.
* The best sequence found before the time budget expires (≈ 1000 s) is
  returned.  Intermediate objective values are printed for monitoring.
"""

import time
import math
import numpy as np
import scipy.optimize
import scipy.sparse

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, Σx=1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending sort
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(a: np.ndarray) -> float:
    """Objective 2·n·max(conv(a,a))  (a is already normalised to Σa=1)."""
    conv = np.convolve(a, a)
    return 2.0 * a.size * conv.max()


# ----------------------------------------------------------------------
# Build the (sparse) convolution constraint matrix
# ----------------------------------------------------------------------
def _build_constraint_matrix(a: np.ndarray, indices: np.ndarray) -> scipy.sparse.csr_matrix:
    """
    For each k in `indices` create the row of the matrix that represents
    (a * b)_k ≤ M, i.e. Σ_i a_{k-i} b_i ≤ M.
    Returns a CSR matrix of shape (len(indices), len(a)).
    """
    n = a.shape[0]
    rows = []
    cols = []
    data = []
    for r, k in enumerate(indices):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rows.extend([r] * idx.size)
        cols.extend(idx.tolist())
        data.extend(a[k - idx].tolist())
    if len(rows) == 0:  # empty (should not happen)
        return scipy.sparse.csr_matrix((0, n))
    return scipy.sparse.csr_matrix((data, (rows, cols)), shape=(len(indices), n))


# ----------------------------------------------------------------------
# Linear programme: maximise Σb_i  s.t. a * b ≤ M, b ≥ 0
# ----------------------------------------------------------------------
def _lp_max_sum(a: np.ndarray, M: float, max_active_iter: int = 15,
                slack_tol: float = 1e-12) -> np.ndarray | None:
    """
    Return a vector b that maximises Σb_i under the constraints
    a convolved with b does not exceed M.  A cut‑plane method starts
    from the indices where a*a attains its maximum and adds any
    violated constraints iteratively.
    """
    n = a.shape[0]
    conv_aa = np.convolve(a, a)

    # start with constraints that are (almost) tight for a*a
    active = np.where(conv_aa >= M * 0.9999)[0]
    if active.size == 0:
        active = np.array([np.argmax(conv_aa)], dtype=int)

    for _ in range(max_active_iter):
        A = _build_constraint_matrix(a, active)
        b_ub = np.full(A.shape[0], M)
        c = -np.ones(n)               # maximise Σb_i  ⇒ minimise −Σb_i
        bounds = [(0, None)] * n

        # HiGHS accepts a sparse A_ub
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub,
                                    bounds=bounds, method='highs',
                                    options={'presolve': True})
        if not res.success:
            # fall back to a dense formulation (rarely needed)
            A_dense = A.toarray()
            res = scipy.optimize.linprog(c, A_ub=A_dense, b_ub=b_ub,
                                        bounds=bounds, method='highs')
            if not res.success:
                return None
        b = res.x

        # check all convolution constraints
        conv_ab = np.convolve(a, b)
        viol = np.where(conv_ab > M + slack_tol)[0]
        if viol.size == 0:
            return b

        # add the most violated constraints (up to a handful)
        excess = conv_ab[viol] - M
        n_new = min(10, viol.size)
        new_idx = viol[np.argsort(excess)[-n_new:]]
        active = np.unique(np.concatenate([active, new_idx]))

    # after max_active_iter rounds return the last (possibly infeasible) vector
    return b


# ----------------------------------------------------------------------
# Simple stochastic hill‑climb (used as a fallback / fine‑tuning)
# ----------------------------------------------------------------------
def _local_hill_climb(a: np.ndarray, max_steps: int, deadline: float) -> tuple[np.ndarray, bool]:
    """
    Randomly move a small amount of mass from one coordinate to another.
    Accepts the move only if the objective improves.
    Returns the (possibly) improved vector and a flag indicating whether
    any improvement occurred.
    """
    n = a.shape[0]
    cur_score = _objective(a)
    improved = False

    for _ in range(max_steps):
        if time.time() >= deadline:
            break
        i, j = np.random.randint(n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        delta = a[i] * np.random.rand() * 0.08  # ≤ 8 % of a[i]
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = _project_to_simplex(a_new)       # keep Σa = 1
        new_score = _objective(a_new)
        if new_score < cur_score:
            a = a_new
            cur_score = new_score
            improved = True
    return a, improved

# ----------------------------------------------------------------------
