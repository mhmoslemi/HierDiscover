def simplex_projection(v):
    # ensures v >=0 and sum==1
    # returns projected onto probability simplex.
    # algorithm from Wang & Carreira-Perpinan (2013).
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def convolve_fft(a, b):
    size = len(a) + len(b) - 1
    N = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    C = A * B
    conv = np.fft.irfft(C, N)[:size]
    return conv

def optimize_sequence(a_init, deadline, lam_init=5.0, lam_max=200.0, step_init=0.5):
    a = a_init.copy()
    n = len(a)
    avg_const = 1.0/(2*n-1)  # average target conv value with sum=1
    best_a = a.copy()
    best_score = fast_score(a)
    lam = lam_init
    step = step_init
    while time.time() < deadline:
        # compute convolution
        b = convolve_fft(a, a)   # length 2n-1
        max_b = b.max()
        # compute weights for softmax
        shift = max_b  # to avoid overflow
        # smoothing parameter lam
        w = np.exp(lam * (b - shift))
        w_sum = w.sum()
        p = w / w_sum
        # gradient
        grad = 2.0 * valid_correlation_fft(p, a)  # shape n
        # step size
        g_norm = np.linalg.norm(grad)
        if g_norm == 0:
            break
        eta = step / g_norm
        a_new = a - eta * grad
        # project onto simplex
        a_new = simplex_projection(a_new)
        # evaluate
        cur_score = fast_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
            print(f'new best {cur_score:.6f} lam={lam:.2f} step={step:.4f}')
        a = a_new
        # schedule
        lam = min(lam * 1.03, lam_max)
        step = max(step * 0.995, 1e-4)
    return best_a, best_score

def simplex_projection(v):
    """
    Projects a vector v onto the probability simplex {x >=0, sum x = 1}.
    Returns projected vector.
    """
    # Implementation using sorting.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        # all non-positives
        return np.full(n, 1.0/n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def propose_candidate(seed=0, budget_s=1000):

    # Determine n based on remaining time: use 8000 if enough time else 5000.
    n = 8000

    # Helper functions defined earlier.

    # Initialize candidates list
    candidates = []

    # uniform
    candidates.append(np.full(n, 1.0/n))

    # power law
    candidates.append(power_law_sequence(alpha=0.506, delta=0.0, n=n))

    # geometric r

    geo = (r ** np.arange(n))
    geo = geo / geo.sum()
    candidates.append(geo)

    # symmetric sin shape
    idx = np.arange(1, n+1)
    sin_shape = np.sin(np.pi * idx / (n+1))
    sin_shape = sin_shape / sin_shape.sum()

    # try to use global height_sequence_1 if exists
    try:
        import __main__ as main_mod
        if hasattr(main_mod, 'height_sequence_1'):
            seq = np.array(main_mod.height_sequence_1, dtype=float)
            if seq.size >= n:
                seq = seq[:n]
            else:
                # pad with zeros and renormalize
                seq = np.pad(seq, (0, n - seq.size))
            seq = seq / seq.sum()
            candidates.append(seq)
    except Exception:
        pass

    best_a = None
    best_score = float('inf')
    # evaluate each candidate quickly
    for a in candidates:
        sc = fast_score(a)
        print(f'initial candidate score {sc:.6f}')
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    # allocate remaining time for descent
    # allocate multiple restarts
    while time.time() < deadline:
        # Set run time for this iteration: allocate maybe 30 seconds for each optimization.
        remaining = deadline - time.time()
        if remaining < 30:
            # run shorter last optimization
            sub_deadline = deadline
        else:
            sub_deadline = time.time() + 30
        # Run optimization from current best with small random perturbation
        a_start = best_a.copy()
        # random perturbation: add small noise and renormalize
        noise = np.random.randn(n) * 1e-3
        a_start = np.maximum(a_start + noise, 0)
        a_start = a_start / a_start.sum()
        # Optimize

"""
Optimisation of a non‑negative step function by directly minimising a
smooth approximation of the max‑convolution objective.
The algorithm works as follows:

1. Choose a moderate length `n` (default 8000).  All sequences are
   normalised to sum‑to‑one, because the evaluation function is scaling
   invariant.

2. Initialise several candidate distributions (uniform, a power‑law,
   a geometric decay and a sinusoidal shape).  The best of these
   seeds is taken as the starting point.

3. From the current point run a projected sub‑gradient descent on the
   smooth “soft‑max’’ of the self‑convolution:
      f(a) = (1/λ)·log( Σ_k exp( λ·( (a∗a)_k – μ ) ) )
   where μ = 1/(2n‑1) is the average convolution value for a
   normalised sequence.
   The gradient of f is
      ∇f = 2·corr( p , a )      with   p_k ∝ exp( λ·((a∗a)_k – μ) )
   (`corr` is the valid linear correlation).  After each step the
   iterate is projected onto the probability simplex (non‑negative,
   sum = 1).  The smoothing parameter λ is slowly increased, turning
   the smooth objective into the true max‑objective.

4. The optimisation loop runs until the allocated time budget is
   exhausted, repeatedly restarting from the current best with a tiny
   random perturbation to escape shallow local minima.

5. The best sequence found is returned as a plain Python list.
   A final sanity check with the provided `evaluate_sequence` is printed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear convolution (real inputs).
def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using a power‑of‑two FFT."""
    size = len(a) + len(b) - 1
    N = 1 << (size - 1).bit_length()          # next power of two
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    C = A * B
    conv = np.fft.irfft(C, N)[:size]
    return conv


# ----------------------------------------------------------------------
# Valid correlation (equivalent to np.correlate(..., mode='valid')) via FFT.
def valid_correlation_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Return the valid linear correlation of x with y:
        (x ⋆ y)[i] = Σ_j x[i+j]·y[j]   ,  i = 0 … len(x)-len(y)
    Implemented as convolution of x with the reversed y.
    """
    y_rev = y[::-1]
    conv = convolve_fft(x, y_rev)                     # length = len(x)+len(y)-1
    n = len(y)
    start = n - 1
    end = start + (len(x) - n + 1)                    # = start + len(y)
    return conv[start:end]


# ----------------------------------------------------------------------
# Projection onto the probability simplex {z ≥ 0, Σz = 1}.
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all entries are non‑positive
        return np.full(n, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # Numerical safety – renormalise to exactly 1
    w_sum = w.sum()
    if w_sum == 0:
        w[:] = 1.0 / n
    else:
        w /= w_sum
    return w


# ----------------------------------------------------------------------
# Fast evaluation of the objective for a *already normalised* sequence.
def fast_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a))/(sum(a)²).  Assumes a is a 1‑D np.ndarray."""
    n = a.shape[0]
    b = convolve_fft(a, a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 1e-12:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Parameterised families used for initial guesses.
def power_law_sequence(alpha: float, delta: float, n: int) -> np.ndarray:
    """a_i ∝ (i+δ)^(-α),  i = 1 … n, then normalised to sum‑one."""
    idx = np.arange(1.0, n + 1, dtype=np.float64) + delta
    a = idx ** (-alpha)
    s = a.sum()
    if s > 0:
        a /= s
    return a


def geometric_sequence(r: float, n: int) -> np.ndarray:
    """Geometric decay a_i ∝ r^i, normalised to sum‑one."""
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s > 0:
        a /= s
    return a


def sinusoid_sequence(n: int) -> np.ndarray:
    """a_i ∝ sin(π·(i+1)/(n+1)), normalised."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.sin(np.pi * i / (n + 1))
    s = a.sum()
    if s > 0:
        a /= s
    return a


# ----------------------------------------------------------------------
# Core optimiser: smooth‑max (soft‑max) minimisation with simplex projection.
def optimise_smoothed_max(a0: np.ndarray, deadline: float,
                         lam_init: float = 5.0,
                         lam_max: float = 200.0,
                         step_init: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Perform projected sub‑gradient descent on the smoothed max of the
    self‑convolution.  Returns the best sequence found and its score.
    """
    a = a0.copy()
    n = a.shape[0]
    avg_const = 1.0 / (2 * n - 1)               # average value of the convolution
    best_a = a.copy()
    best_sc = fast_score(best_a)

    lam = lam_init
    step = step_init
    it = 0
    while time.time() < deadline:
        it += 1
        # 1. self‑convolution
        b = convolve_fft(a, a)                  # length 2n‑1
        # 2. soft‑max weights (shifted by the current maximum for stability)
        shift = b.max()
        w = np.exp(lam * (b - shift))
        w_sum = w.sum()
        if w_sum == 0:
            # numerical under‑flow → revert to uniform weights
            p = np.full_like(w, 1.0 / w.shape[0])
        else:
            p = w / w_sum                       # p is a probability vector

        # 3. gradient = 2·corr(p, a) (valid correlation, length n)
        grad = 2.0 * valid_correlation_fft(p, a)

        # 4. step size normalised by the gradient norm
        gnorm = np.linalg.norm(grad)
        if gnorm == 0.0:
            break
        eta = step / gnorm
        a_new = a - eta * grad

        # 5. projection onto the simplex
        a_new = simplex_projection(a_new)

        # 6. keep the best point seen so far
        cur_sc = fast_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()
            print(f'  it={it:5d}  new best {best_sc:.6f}  λ={lam:.2f}  step={step:.4f}')

        a = a_new

        # 7. schedule for smoothing parameter and step size
        lam = min(lam * 1.03, lam_max)
        step = max(step * 0.995, 1e-4)

    return best_a, best_sc


# ----------------------------------------------------------------------

    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    candidates = []

    candidates.append(power_law_sequence(alpha=0.506, delta=0.0, n=n))

    candidates.append(geometric_sequence(r=0.9995, n=n))
