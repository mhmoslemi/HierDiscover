```
def fast_score(a):
    n = a.shape[0]
    size = 2*n - 1
    N = 1 << (size-1).bit_length()
    A = np.fft.rfft(a, N)
    B = A * A
    conv = np.fft.irfft(B, N)[:size]
    max_b = np.max(conv)
    return 2.0 * n * max_b / (np.sum(a)**2)
```

```
Search for low-evaluation step function via:
1. power‑law initialisation,
2. coarse/fine grid over (α,δ) to find good shape,
3. simulated annealing with two‑index mass swaps,
4. final gradient refinement targeting a flat convolution.
The algorithm respects the 1000‑second time budget.

import time, math, numpy as np

def convolve_fft(a):
    size = len(a) * 2 - 1
    N = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, N)
    B = A * A
    conv = np.fft.irfft(B, N)[:size]
    return conv

def fast_score(a):
    n = a.shape[0]
    b = convolve_fft(a)
    max_b = np.max(b)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))

def power_law_sequence(alpha, delta, n):
    idx = np.arange(1.0, n + 1) + delta
    a = idx ** (-alpha)
    s = a.sum()
    if s > 0:
        a /= s
    return a

def gradient_refine(a, deadline):
    """Projected gradient descent on L2 distance to flat target."""
    n = a.shape[0]
    target_const = (a.sum() ** 2) / (2 * n - 1)
    step_factor = 0.05
    best = a.copy()
    best_score = fast_score(a)
    while time.time() < deadline:
        # convolution
        b = convolve_fft(a)
        # error w.r.t flat target
        err = b - target_const
        # gradient of L2 error wrt a (derived analytically)
        grad = 4.0 * np.correlate(err, a, mode='valid')
        norm = np.linalg.norm(grad)
        if norm == 0.0:
            break
        eta = step_factor / norm
        a = a - eta * grad
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:

"""
Search for a low‑score step‑function.

The algorithm proceeds in four stages:

1. **Initialisation** – start from a power‑law sequence (or reuse a
   previously‑found `height_sequence_1` if available) and normalise it
   to a probability vector (sum = 1).

2. **Coarse / fine grid search** – sweep a few values of the power‑law
   exponent α and shift δ for the chosen length n.  The best candidate
   from this fast scan becomes the seed for the next stage.

3. **Simulated annealing** – repeatedly perform a tiny “mass‑swap”
   between two random indices (keep the total sum unchanged).  Moves are
   accepted with a Metropolis rule based on the current temperature.
   The temperature and the step size are exponentially cooled; the
   best sequence seen so far is kept.

4. **Gradient polish** – a few projected‑gradient steps minimise the
   L²‑error of the self‑convolution against its flat target, which
   usually squeezes a few more hundredths out of the score.

All heavy work uses FFT‑based convolution (`np.fft.rfft`), so each
evaluation is O(n log n) and comfortably fits the 1000 s budget.  The
final list of non‑negative floats is returned; the evaluator caps
values at 0 … 1000, but the algorithm never generates large numbers.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast linear convolution via FFT (real input)
def convolve_fft(a: np.ndarray) -> np.ndarray:
    """Return the linear convolution a * a using an FFT of length a power of two."""
    size = a.size * 2 - 1
    N = 1 << (size - 1).bit_length()          # next power‑of‑two
    A = np.fft.rfft(a, N)
    B = A * A
    conv = np.fft.irfft(B, N)[:size]
    return conv


# ----------------------------------------------------------------------
# Objective (the evaluator ignores the scaling, so we normalise to sum = 1)
def fast_score(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) / (sum(a)²) – fast version using FFT."""
    n = a.size
    b = convolve_fft(a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 1e-12:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
def power_law_sequence(alpha: float, delta: float, n: int) -> np.ndarray:
    """Generate a_i ∝ (i+δ)^{-α} for i=1…n and normalise to sum = 1."""
    idx = np.arange(1.0, n + 1, dtype=np.float64) + delta
    a = idx ** (-alpha)
    s = a.sum()
    if s > 0.0:
        a /= s
    return a


# ----------------------------------------------------------------------
def gradient_refine(a: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected gradient descent that minimises the L²‑error of the
    self‑convolution to its flat target (the average value).
    """
    n = a.size
    target_const = (a.sum() ** 2) / (2 * n - 1)

    step_factor = 0.05
    best = a.copy()
    best_score = fast_score(a)

    while time.time() < deadline:
        b = convolve_fft(a)
        err = b - target_const                     # error vector
        # gradient of ½‖err‖² w.r.t. a  (derived analytically)
        grad = 4.0 * np.correlate(err, a, mode='valid')
        gn = np.linalg.norm(grad)
        if gn == 0.0:
            break
        eta = step_factor / gn
        a = a - eta * grad
        a = np.maximum(a, 0.0)                     # projection onto ℝ₊
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= s
        cur = fast_score(a)
        if cur < best_score:
            best_score = cur
            best = a.copy()
            print(f'  gradient step improved score {best_score:.6f}')
            step_factor = max(step_factor * 0.95, 1e-4)

    return best


# ----------------------------------------------------------------------
def simulated_annealing(a0: np.ndarray, deadline: float):
    """
    Simple SA that swaps infinitesimal mass between two random indices.
    The total sum stays exactly 1, non‑negativity is enforced by clipping.
    """
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = fast_score(best)
    cur_score = best_score

    # Acceptance temperature (starts proportional to the current score)
    T = 0.01 * best_score
    # Step‑size factor (relative to the current maximal entry)
    step_factor = 0.5
    max_val = a.max()
    iteration = 0

    while time.time() < deadline:
        iteration += 1
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        # propose a symmetric mass transfer ε (add to i, subtract from j)
        eps = (np.random.rand() * 2.0 - 1.0) * step_factor * max_val
        # keep feasibility
        eps = max(eps, -a[i])
        eps = min(eps, a[j])
        if eps == 0.0:
            continue

        a_new = a.copy()
        a_new[i] += eps
        a_new[j] -= eps            # sum unchanged, non‑negativity guaranteed by clipping

        new_score = fast_score(a_new)
        delta = new_score - cur_score

        # Metropolis acceptance
        if delta < 0.0 or np.random.rand() < math.exp(-delta / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a_new.copy()
                print(f'iteration {iteration}: new best {best_score:.6f}')

        # cooling – every few thousand steps
        if iteration % 2000 == 0:
            T *= 0.95
            step_factor = max(step_factor * 0.98, 1e-4)
            max_val = a.max()

    return best, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000):
    """
    Main entry point expected by the grader.  Returns a Python list of
    non‑negative floats that (hopefully) yields a low score for
    `evaluate_sequence`.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # small safety margin

    # ------------------------------------------------------------------
    # 1️⃣ Length choice – a few thousand entries give a good trade‑off.
    n = 5000

    # ------------------------------------------------------------------
    # 2️⃣ Try to reuse a previously discovered sequence if the variable
    #    `height_sequence_1` is present in the global namespace.
    try:
        seq0 = np.array(height_sequence_1, dtype=float)
        if seq0.size != n:
            if seq0.size > n:
                seq0 = seq0[:n]
            else:
                seq0 = np.concatenate([seq0, np.zeros(n - seq0.size)])
        a0 = seq0 / seq0.sum()
        print("starting from supplied height_sequence_1 (adapted)")
    except Exception:
        # No external sequence – fall back to a power‑law seed.
        a0 = power_law_sequence(alpha=0.506, delta=0.0, n=n)
        print("starting from fresh power‑law seed")

    best_score = fast_score(a0)
    print(f'initial score = {best_score:.6f}')

    # ------------------------------------------------------------------
    # 3️⃣ Coarse grid over (α,δ) – very cheap, improves the seed.
    alphas = np.linspace(0.42, 0.58, 33)          # step ≈0.005
    deltas = [0.0, 0.5, 1.0, 2.0]

    for delta in deltas:
        for alpha in alphas:
            if time.time() >= deadline:
                break
            cand = power_law_sequence(alpha, delta, n)
            sc = fast_score(cand)
            if sc < best_score:
                best_score = sc
                a0 = cand
                print(f'grid improvement α={alpha:.4f} δ={delta:.2f} → {sc:.6f}')

    # ------------------------------------------------------------------
    # 4️⃣ Fine grid around the region that performed best.
    if time.time() < deadline:
        alphas_fine = np.linspace(0.48, 0.54, 41)   # centred around the coarse optimum
        deltas_fine = [0.0, 0.5, 1.0]
        for delta in deltas_fine:
            for alpha in alphas_fine:
                if time.time() >= deadline:
                    break
                cand = power_law_sequence(alpha, delta, n)
                sc = fast_score(cand)
                if sc < best_score:
                    best_score = sc
                    a0 = cand
                    print(f'fine grid α={alpha:.5f} δ={delta:.2f} → {sc:.6f}')

    # ------------------------------------------------------------------
    # 5️⃣ Simulated annealing – the heavy optimisation stage.
    if time.time() < deadline:
        print('starting simulated annealing...')
        a_best, best_score = simulated_annealing(a0, deadline)
        print(f'annealing finished – best score {best_score:.6f}')
    else:
        a_best = a0

    # ------------------------------------------------------------------
    # 6️⃣ Final gradient polishing – cheap but often nudges the score
    #    a little further down.
    if time.time() < deadline:
        print('starting gradient polishing...')
        a_best = gradient_refine(a_best, deadline)
        best_score = fast_score(a_best)
        print(f'gradient polish final score {best_score:.6f}')

    # ------------------------------------------------------------------
    # 7️⃣ Sanity check with the official evaluator (optional).
    try:
        final_score = evaluate_sequence(a_best.tolist())
        print(f'final evaluate_sequence score {final_score:.6f}')
    except Exception:
        pass

    # Return a plain Python list of floats as required.
    return a_best.tolist()
```
