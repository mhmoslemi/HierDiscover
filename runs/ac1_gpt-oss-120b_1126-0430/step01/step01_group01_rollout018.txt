def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import numpy as np, time, math, random
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5

    n = 3000  # length
    a = catalan_sequence(n)   # base

    # Greedy refine with bigger moves
    best_seq = a.copy()
    best_score = fast_evaluation(best_seq)

    # Additional search loops:
    while time.time() < deadline:
        # compute conv and max index
        n_len = a.size
        fft_len = 1 << ((2 * n_len - 1).bit_length())
        Fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(Fa * Fa, fft_len)[:2*n_len-1]
        k = int(conv.argmax())
        M = conv[k]
        S = a.sum()
        w = np.zeros_like(a)
        i = np.arange(n_len)
        j = k - i
        mask = (j>=0) & (j<n_len)
        w[mask] = a[j[mask]]

        # Determine target weight
        target = M / S

        # indices where w > target + epsilon (source)
        eps = 1e-12
        src_candidates = np.where(w > target + eps)[0]
        sink_candidates = np.where(w < target - eps)[0]

        if src_candidates.size == 0 or sink_candidates.size == 0:
            # fallback random move
            src = np.random.randint(n_len)
            sink = np.random.randint(n_len)
            delta = 0.01 * a[src]
        else:
            src = src_candidates[np.argmax(a[src_candidates] * w[src_candidates])]  # high contribution
            sink = sink_candidates[np.argmin(w[sink_candidates])]

            # move a substantial fraction
            delta = min(a[src], max(0.5 * a[src], 0.01 * a[src]))  # up to 50% of source
            # ensure not too large if sink already huge
            delta = min(delta, 1000 - a[sink])

        # propose new sequence
        new_seq = a.copy()
        new_seq[src] -= delta
        new_seq[sink] += delta

        # evaluate
        score = fast_evaluation(new_seq)

        if score < best_score - 1e-9:
            a = new_seq
            best_seq = new_seq.copy()
            best_score = score
            print(f"Iter improved: score {best_score:.6f}, move {src}->{sink} delta {delta:.6f}")
            continue
        else:
            # occasionally try random injection of mass at low w location
            if np.random.rand() < 0.02:
                # add small epsilon to a random low-w index
                idx = np.random.choice(sink_candidates) if sink_candidates.size else np.random.randint(n_len)
                incr = np.random.rand() * 0.5
                new_seq = a.copy()
                new_seq[idx] = min(1000.0, new_seq[idx] + incr)
                score = fast_evaluation(new_seq)
                if score < best_score:
                    a = new_seq
                    best_seq = new_seq.copy()
                    best_score = score
                    print(f"Random bump improved: score {best_score:.6f}")
    return best_seq.tolist()

import time
import numpy as np

def _fast_evaluation(seq: np.ndarray) -> float:
    n = seq.size
    if n == 0:
        return np.inf
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))

def _catalan_sequence(length: int) -> np.ndarray:
    seq = np.empty(length, dtype=float)
    a = 1.0
    seq[0] = a
    for i in range(1, length):
        a *= (2 * i - 1) / (2.0 * i)
        seq[i] = a
    return seq

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """Search for a step‑function that beats the π/2 bound.

    The algorithm starts from the Catalan (discrete arcsine) sequence,
    then repeatedly moves mass from indices that contribute most to the current
    convolution peak to many low‑contributing indices in the “right‑hand” part
    of the array.  The movement respects a dynamic upper bound on individual
    heights (a fraction of sqrt(max_convolution)) so that the peak of the
    self‑convolution can actually be lowered.  The process is a greedy
    multi‑sink redistribution that is repeated until the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5   # a safety margin

    # length of the candidate – 3000 gives a good trade‑off
    n = 3000
    a = _catalan_sequence(n)

    # Pre‑allocate FFT size (does not change)
    fft_len = 1 << ((2 * n - 1).bit_length())

    best_seq = a.copy()
    best_score = _fast_evaluation(best_seq)
    print(f"[init] n={n}, score={best_score:.6f}")

    # Main optimisation loop
    while time.time() < deadline:
        # --- compute convolution and locate the current maximum ---
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        k = int(conv.argmax())
        M = conv[k]
        S = a.sum()
        sqrtM = np.sqrt(M)

        # --- contributions of each index to the current maximal entry ---
        i = np.arange(k + 1)                     # i can be 0..k
        j = k - i
        mask = (j < n)                           # keep only valid j
        i = i[mask]
        j = j[mask]
        contrib = a[i] * a[j]                    # a_i * a_{k-i}
        if contrib.size == 0:
            # nothing to move – exit early
            break

        src = i[np.argmax(contrib)]              # source index (largest contribution)

        # we want to move mass from src to indices to the right of k,
        # keeping each height below a fraction of sqrtM so that the peak can drop.
        # The fraction 0.9 works well in practice, but is clamped to 1.0.
        allowed_factor = 0.9
        max_allowed = sqrtM * allowed_factor

        # viable sinks are positions > k with a value smaller than max_allowed
        sink_candidates = np.where((np.arange(n) > k) & (a < max_allowed))[0]
        if sink_candidates.size == 0:
            # No safe sink on the right – fall back to a tiny random move.
            # This also helps us escape plateaus.
            src = np.random.choice(i)
            sink = np.random.randint(n)
            delta = 0.01 * a[src]
            new_seq = a.copy()
            new_seq[src] = max(0.0, new_seq[src] - delta)
            new_seq[sink] = min(1000.0, new_seq[sink] + delta)
        else:
            # Sort sinks by their current height (fill the smallest first)
            order = sink_candidates[np.argsort(a[sink_candidates])]

            remaining = a[src]                     # amount we would like to move
            new_seq = a.copy()
            new_seq[src] = 0.0                      # we will re‑distribute everything from src

            for sink in order:
                if remaining <= 1e-12:
                    break
                cap = max_allowed - new_seq[sink]  # how much we may still add to this sink
                if cap <= 1e-12:
                    continue
                delta = min(remaining, cap)
                new_seq[sink] += delta
                remaining -= delta

            # If we could not move everything (no capacity left) we put the rest back.
            if remaining > 0:
                new_seq[src] = remaining

        # evaluate the candidate
        score = _fast_evaluation(new_seq)

        # accept if strictly better (allow a tiny tolerance for floating noise)
        if score < best_score - 1e-12:
            a = new_seq
            best_seq = new_seq.copy()
            best_score = score
            print(f"[improve] t={time.time():.1f} score={best_score:.6f} M={M:.6f}")
            continue

        # occasionally try a random jitter (helps escaping local minima)
        if np.random.rand() < 0.02:
            idx = np.random.randint(n)
            jitter = np.random.randn() * 0.001
            new_seq2 = a.copy()
            new_seq2[idx] = max(0.0, new_seq2[idx] + jitter)
            score2 = _fast_evaluation(new_seq2)
            if score2 < best_score:
                a = new_seq2
                best_seq = new_seq2.copy()
                best_score = score2
                print(f"[jitter] t={time.time():.1f} score={best_score:.6f}")
                continue

        # if nothing helped, perform a tiny random redistribution
        if np.random.rand() < 0.01:
            src = np.random.choice(i)
            sink = np.random.randint(n)
            delta = 0.02 * a[src] + 1e-6
            new_seq3 = a.copy()
            new_seq3[src] -= delta
            new_seq3[sink] += delta
            score3 = _fast_evaluation(new_seq3)
            if score3 < best_score:
                a = new_seq3
                best_seq = new_seq3.copy()
                best_score = score3
                print(f"[tiny move] t={time.time():.1f} score={best_score:.6f}")

    print(f"[final] score={best_score:.6f}")
    # The harness will clip to [0,1000] anyway.
    return best_seq.tolist()

"""
Search for a step‑function whose self‑convolution peak is lower than the
classical arcsine (Catalan) bound (≈π/2 ≃ 1.5708).  The method starts from the
Catalan sequence, then repeatedly moves mass from indices that contribute most
to the current convolution maximum to indices far to the right, where the
contribution to that maximum is zero.  The amount moved is chosen aggressively
(large fractions of the source value) but never lets any single height exceed
a configurable fraction of sqrt(max_convolution); this keeps the new peak
under control and allows the overall maximum to shrink.  A tiny random jitter
is added occasionally to escape plateaus.  The whole loop runs until the
time budget (≈1000 s) is exhausted and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast metric – identical to ``evaluate_sequence`` but without the
# safety checks (uses FFT for speed).
# ----------------------------------------------------------------------
def _fast_evaluation(seq: np.ndarray) -> float:
    """Return the same score as ``evaluate_sequence`` (lower is better)."""
    n = seq.size
    if n == 0:
        return np.inf
    m = 1 << ((2 * n - 1).bit_length())          # power‑of‑two FFT size
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]   # self‑convolution
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Catalan (discrete arcsine) seed.
# a_0 = 1, a_{i+1} = a_i * (2*i+1)/(2*i+2)
# ----------------------------------------------------------------------
def _catalan_sequence(length: int) -> np.ndarray:
    """First ``length`` Catalan coefficients (float array)."""
    seq = np.empty(length, dtype=float)
    a = 1.0
    seq[0] = a
    for i in range(1, length):
        a *= (2 * i - 1) / (2.0 * i)      # recurrence for C_i
        seq[i] = a
    return seq


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Produce a list of non‑negative heights with a low evaluation score.

    The algorithm works inside ``budget_s`` seconds (default 1000).  It starts
    from the Catalan (arcsine) sequence, then repeatedly shifts mass from
    indices that heavily affect the current convolution peak to later indices
    that do not.  The shift size is chosen from a set of aggressive fractions,
    and a small random jitter is occasionally tried to escape local minima.
    The best sequence seen so far is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5   # small safety margin

    # -----------------------------------------------------------------------------
    # 1.  Seed – Catalan (arcsine) coefficients.
    # -----------------------------------------------------------------------------
    n = 3000                     # length of the candidate (trade‑off quality / speed)
    a = _catalan_sequence(n)    # start point
    best_seq = a.copy()
    best_score = _fast_evaluation(best_seq)
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # Pre‑compute FFT size – does not change during the search.
    fft_len = 1 << ((2 * n - 1).bit_length())

    # -----------------------------------------------------------------------------
    # 2.  Greedy multi‑sink mass redistribution.
    # -----------------------------------------------------------------------------
    # fractions of the source amount that we will try (large moves first).
    move_fractions = (0.5, 0.3, 0.2, 0.1, 0.05, 0.02)

    while time.time() < deadline:
        # ---- current convolution and its maximum index ----
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        k = int(conv.argmax())          # index where the peak occurs
        M = conv[k]                     # current peak value
        S = a.sum()
        sqrtM = np.sqrt(M)

        # ---- contributions of each position to that peak ----
        i = np.arange(k + 1)            # i can be 0 … k
        j = k - i
        mask = j < n
        i = i[mask]
        j = j[mask]
        contrib = a[i] * a[j]           # a_i * a_{k-i}
        if contrib.size == 0:           # safety – should not happen
            break

        src = i[np.argmax(contrib)]    # source index (largest contribution)

        # ---- sink candidates: indices strictly to the right of k.
        # Their contribution to the current peak is zero.
        sink_candidates = np.where(np.arange(n) > k)[0]
        if sink_candidates.size == 0:
            # nothing to the right – give up this iteration
            continue

        # choose a sink uniformly among the right‑hand side
        sink = int(np.random.choice(sink_candidates))

        src_val = a[src]
        if src_val < 1e-12:                 # nothing left to move
            continue

        # we never let any entry exceed a fraction of sqrt(M); this keeps the
        # new peak from rising too fast.
        allowed_factor = 0.9                 # can be tuned
        max_allowed = sqrtM * allowed_factor
        capacity = max_allowed - a[sink]     # how much we may still add here
        if capacity <= 0:
            # sink already too big – pick a different one
            sink = int(np.random.choice(sink_candidates))
            capacity = max_allowed - a[sink]

        # ----- try several move sizes, keep the best improvement -----
        best_candidate = None
        best_candidate_score = best_score

        for frac in move_fractions:
            delta = min(frac * src_val, capacity)
            if delta <= 0:
                continue
            cand = a.copy()
            cand[src] -= delta
            cand[sink] += delta
            score = _fast_evaluation(cand)

            if score < best_candidate_score - 1e-12:
                best_candidate = cand
                best_candidate_score = score

        # ----- accept the best move (if any) -----
        if best_candidate is not None:
            a = best_candidate
            best_seq = best_candidate.copy()
            best_score = best_candidate_score
            print(f"[improve] time={time.time():.1f} score={best_score:.6f} "
                  f"M={M:.6f} src={src} sink={sink}")
            continue

        # ----- no large move helped – try a tiny random jitter -----
        if np.random.rand() < 0.02:
            idx = np.random.randint(n)
            jitter = np.random.randn() * 0.001
            cand = a.copy()
            cand[idx] = max(0.0, cand[idx] + jitter)
            score = _fast_evaluation(cand)
            if score < best_score:
                a = cand
                best_seq = cand.copy()
                best_score = score
                print(f"[jitter] time={time.time():.1f} score={best_score:.6f}")

    # -----------------------------------------------------------------------------
    # 3.  Return the best sequence discovered.
    # -----------------------------------------------------------------------------
    print(f"[final] best score={best_score:.6f}")
    # ``evaluate_sequence`` will clip to [0,1000] – our values already respect that.
    return best_seq.tolist()
