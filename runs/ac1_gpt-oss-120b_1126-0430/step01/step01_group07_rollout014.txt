def project_simplex(v):
    """Projects vector v onto the probability simplex (non-negative, sum=1)."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    # find the largest i where u_i - cssv_i / i > 0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # if all non-positive, return uniform
        return np.full(n, 1.0/n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a low‑score step‑function using a smoothed‑max gradient descent.

The evaluation function is invariant under scaling, so we keep the
sequence on the probability simplex (non‑negative, sum = 1).  The
objective
    f(a) = max_k (a * a)_k
is approximated by a smooth soft‑max
    f_β(a) = (1/β) log Σ_k exp(β·(a*a)_k) .
Its gradient can be written in closed form using FFTs:
    ∇f_β = conv(w, a) + corr(w, a)
where w_k ∝ exp(β·(a*a)_k) are the soft‑max weights,
conv is a linear convolution and corr the (non‑circular) cross‑correlation.
A projected gradient step (onto the simplex) reduces the objective.
The smoothing parameter β is increased slowly, turning the smooth
objective into the true max.  The method starts from a power‑law
distribution (α≈0.5) and runs until the time budget expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: linear convolution via FFT (real input)
def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using a power‑of‑two FFT."""
    size = a.size + b.size - 1
    N = 1 << (size - 1).bit_length()          # next power of two
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    C = A * B
    conv = np.fft.irfft(C, N)[:size]
    return conv


# ----------------------------------------------------------------------
def fast_score(a: np.ndarray) -> float:
    """Score 2·n·max(conv(a,a)) / (sum(a)^2) – a is assumed non‑negative."""
    n = a.size
    b = convolve_fft(a, a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
def power_law_sequence(alpha: float, delta: float, n: int) -> np.ndarray:
    """a_i ∝ (i+δ)^{-α}, i=1…n, normalised to sum = 1."""
    idx = np.arange(1.0, n + 1, dtype=np.float64) + delta
    a = idx ** (-alpha)
    s = a.sum()
    if s > 0.0:
        a /= s
    return a


# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (non‑negative, sum = 1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        # all entries are zero or negative – return the uniform distribution
        return np.full(n, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a step‑function with a low evaluation score.

    The algorithm:
      1.  Initialise a power‑law distribution (α≈0.5) on a moderately large
          support (n≈8000) and normalise to sum = 1.
      2.  Perform a short grid search around the power‑law to obtain a better
          start point.
      3.  Repeatedly:
           • compute the self‑convolution b = a*a,
           • build soft‑max weights w_k ∝ exp(β·(b_k – max(b))),
           • obtain the gradient ∇ = conv(w,a) + corr(w,a) via FFTs,
           • take a projected gradient step (step size ∝ 1/‖∇‖),
           • increase the smoothing parameter β (so the objective turns into a
             true max) and adapt the step size,
           • keep the best sequence seen so far.
      4.  Stop when the time budget is exhausted and return the best sequence.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Problem size – larger n gives a lower theoretical bound.
    n = 8000

    # ------------------------------------------------------------------
    # 1. Initial power‑law guess (α≈0.5, δ≈0)
    a = power_law_sequence(alpha=0.506, delta=0.0, n=n)

    # ------------------------------------------------------------------
    # 2. Tiny coarse grid around the power‑law to improve the start.
    best_seq = a.copy()
    best_score = fast_score(best_seq)
    print(f"initial score (power‑law) = {best_score:.6f}")

    alphas = np.linspace(0.45, 0.56, 23)   # step ≈0.005
    deltas = [0.0, 0.5, 1.0]

    for delta in deltas:
        for alpha in alphas:
            if time.time() >= deadline:
                break
            cand = power_law_sequence(alpha, delta, n)
            sc = fast_score(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                a = cand.copy()
                print(f"grid improvement α={alpha:.4f} δ={delta:.1f} → score={sc:.6f}")

    # ------------------------------------------------------------------
    # 3. Smoothed‑max projected gradient descent
    a = best_seq.copy()
    # pre‑allocate padded buffers (size ≥ 3n‑2)
    conv_len = 3 * n - 2
    N = 1 << (conv_len - 1).bit_length()   # next power of two
    a_pad = np.zeros(N, dtype=np.float64)
    w_pad = np.zeros(N, dtype=np.float64)

    # hyper‑parameters
    step_factor = 0.5          # controls absolute step = step_factor / ‖∇‖
    beta = 1.0                # smoothing of the max
    beta_max = 300.0
    beta_inc = 1.015          # geometric increase each iteration
    prev_score = fast_score(a)
    iteration = 0
    no_improve = 0

    while time.time() < deadline:
        iteration += 1

        # ----- FFT of a -------------------------------------------------
        a_pad[:n] = a
        A = np.fft.rfft(a_pad)

        # ----- self‑convolution b = a*a ---------------------------------
        C = A * A
        b_full = np.fft.irfft(C, n=N)
        b = b_full[: 2 * n - 1]                 # linear convolution length
        max_b = b.max()

        # ----- soft‑max weights w (vector of length 2n‑1) ---------------
        diff = b - max_b                         # ≤ 0
        w_vec = np.exp(beta * diff)
        w_sum = w_vec.sum()
        if w_sum == 0.0:                         # numerical safety
            w_vec = np.full_like(w_vec, 1.0 / (2 * n - 1))
            w_sum = w_vec.sum()
        w_vec /= w_sum

        # ----- FFT of w -------------------------------------------------
        w_pad[: 2 * n - 1] = w_vec
        W = np.fft.rfft(w_pad)

        # ----- gradient = conv(w,a) + corr(w,a) -------------------------
        # conv term (linear convolution)
        conv_full = np.fft.irfft(W * A, n=N)
        conv_term = conv_full[:n]               # conv(w, a)[0:n]

        # correlation term (cross‑correlation a ⋆ w)
        corr_full = np.fft.irfft(np.conj(W) * A, n=N)
        offset = 2 * n - 2                       # start index for lag ≥ 0
        corr_term = corr_full[offset: offset + n]   # corr(w, a)[0:n]

        grad = conv_term + corr_term
        grad_norm = np.linalg.norm(grad)
        if grad_norm == 0.0:
            break

        # ----- projected gradient step ---------------------------------
        eta = step_factor / grad_norm
        a_candidate = a - eta * grad
        a_candidate = project_simplex(a_candidate)

        # ----- evaluate -------------------------------------------------
        cur_score = fast_score(a_candidate)

        # keep the best ever seen
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_candidate.copy()
            print(f"iter {iteration:5d}: new best score = {cur_score:.6f}")

        # adapt step size based on immediate improvement
        if cur_score < prev_score:
            step_factor = min(step_factor * 1.04, 5.0)
            no_improve = 0
        else:
            step_factor = max(step_factor * 0.92, 1e-5)
            no_improve += 1

        prev_score = cur_score
        a = a_candidate

        # increase smoothing (push towards true max)
        beta = min(beta_max, beta * beta_inc)

        # occasional status report
        if iteration % 200 == 0:
            elapsed = time.time() - (deadline - budget_s)
            print(f"[{elapsed:.1f}s] iter {iteration:5d}: score={cur_score:.6f} "
                  f"beta={beta:.2f} step={step_factor:.4f}")

        # optional early stop if stagnating for a long time
        if no_improve >= 800 and beta >= beta_max:
            # add a tiny random perturbation to escape possible flat region
            perturb = np.random.normal(scale=1e-4, size=n)
            a = project_simplex(a + perturb)
            no_improve = 0

    # ------------------------------------------------------------------
    # Final sanity check using the provided evaluate_sequence
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"final reported score = {final_score:.6f}")
    except Exception:
        # evaluate_sequence might not be available in the testing harness
        pass

    # Return as plain Python list of floats
    return best_seq.tolist()
