"""
Advanced refinement of the Catalan (arcsine) seed.

The algorithm starts from the Catalan coefficients (the discretised
arcsine distribution).  The self‑convolution of this seed is constant
≈1 for the first n entries, giving a score ≈π/2 ≈1.571.
To push the score below 1.6 we repeatedly move a modest amount of
mass from indices that contribute most to the current maximal
convolution entry to indices that contribute almost nothing.
The step size is adaptive (large at the beginning, decreasing like 1/√it)
and every few hundred iterations a bulk redistribution from the early
quarter of the vector to the last quarter is performed – this breaks the
perfect symmetry of the arcsine shape and allows the peak of the
convolution to drop further.
Random jitter is injected occasionally to escape tiny plateaus.
All operations use FFT‑based convolution (≈O(n log n)) so millions of
iterations fit easily into the 1000 s budget.
The best sequence found is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast evaluation (identical to evaluate_sequence but without the security
# checks and using an FFT based convolution).  This is used for the
# optimisation loop; the official ``evaluate_sequence`` will be called by
# the harness on the final answer.
# ----------------------------------------------------------------------
def _fast_evaluation(seq: np.ndarray) -> float:
    n = seq.size
    if n == 0:
        return np.inf
    # FFT size = next power of two ≥ 2*n-1
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Catalan (arcsine) seed:  a_0 = 1, a_{i+1}=a_i*(2*i+1)/(2*i+2)
# ----------------------------------------------------------------------
def _catalan_sequence(length: int) -> np.ndarray:
    seq = np.empty(length, dtype=float)
    a = 1.0
    seq[0] = a
    for i in range(1, length):
        a *= (2 * i - 1) / (2.0 * i)
        seq[i] = a
    return seq


# ----------------------------------------------------------------------
# Core refinement routine – aggressive mass redistribution.
# ----------------------------------------------------------------------
def _refine_advanced(seq: np.ndarray, deadline: float) -> np.ndarray:
    """Return the best sequence found before *deadline*.

    The routine works on a vector that is already normalised to L1=1.
    """
    n = seq.size
    # keep the best candidate seen so far
    best_seq = seq.copy()
    best_score = _fast_evaluation(best_seq)

    # FFT length (does not change)
    fft_len = 1 << ((2 * n - 1).bit_length())

    it = 0
    while time.time() < deadline:
        it += 1

        # --------------------------------------------------------------
        # current convolution and the index where it attains its maximum
        # --------------------------------------------------------------
        fa = np.fft.rfft(seq, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        k = int(conv.argmax())
        max_val = float(conv[k])

        # --------------------------------------------------------------
        # contribution of each position to that maximal entry:
        #            w[i] = a_{k-i}   (zero outside the support)
        #            contrib[i] = a_i * w[i]
        # --------------------------------------------------------------
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        w = np.zeros_like(seq)
        w[mask] = seq[j[mask]]
        contrib = seq * w

        # --------------------------------------------------------------
        # source = index with largest contribution,
        # sink   = index with smallest weight w (almost no influence)
        # --------------------------------------------------------------
        src = int(np.argmax(contrib))
        # there may be several equally minimal entries – pick one at random
        sink_candidates = np.where(w == w.min())[0]
        sink = int(np.random.choice(sink_candidates))

        # --------------------------------------------------------------
        # Adaptive step size – larger at the beginning, shrinking like 1/√it
        # --------------------------------------------------------------
        step_factor = max(0.01, 0.5 / np.sqrt(it))          # 0.5 is a tunable constant
        eps = step_factor * seq[src]                       # move a modest fraction
        if eps <= 1e-12:                                   # nothing left to move
            break

        # --------------------------------------------------------------
        # Perform the move and evaluate
        # --------------------------------------------------------------
        new_seq = seq.copy()
        new_seq[src] -= eps
        new_seq[sink] += eps

        # keep numerics tidy
        new_seq = np.clip(new_seq, 0.0, None)

        score = _fast_evaluation(new_seq)

        # --------------------------------------------------------------
        # Accept only if we get a genuine improvement
        # --------------------------------------------------------------
        if score < best_score - 1e-9:          # a tiny tolerance avoids noise
            seq = new_seq
            best_seq = new_seq.copy()
            best_score = score
            print(f"[{it:6d}] improved score = {best_score:.6f}")
        else:
            # occasional random jitter to jump out of flat plateaus
            if np.random.rand() < 0.02:
                i = np.random.randint(n)
                j = np.random.randint(n)
                if i != j and seq[i] > 1e-12:
                    jitter_eps = 0.01 * seq[i]
                    new_seq = seq.copy()
                    new_seq[i] -= jitter_eps
                    new_seq[j] += jitter_eps
                    new_seq = np.clip(new_seq, 0.0, None)
                    seq = new_seq

            # every few hundred steps perform a bulk early‑to‑late transfer
            if it % 400 == 0:
                q1 = n // 4                 # first quarter
                q3 = 3 * n // 4             # last quarter
                early_mass = seq[:q1].sum()
                if early_mass > 0:
                    # move 10 % of the early mass uniformly to the last quarter
                    move = 0.10 * early_mass
                    # proportionally remove from the early quarter
                    prop = seq[:q1] / early_mass
                    delta = move * prop
                    seq[:q1] -= delta
                    # distribute uniformly over the last quarter
                    seq[q3:] += move / (n - q3)

        # numerical safety – very tiny negative values can appear due to rounding
        seq = np.clip(seq, 0.0, None)

    return best_seq


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights with a low evaluation score.

    The routine runs for at most ``budget_s`` seconds (default 1000).  It
    builds a Catalan (arcsine) seed, normalises it to unit L1 norm,
    and then applies an aggressive greedy/annealing style optimisation
    that repeatedly moves mass away from indices that dominate the
    current maximal self‑convolution entry.
    The best sequence discovered before the deadline is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5   # small safety margin

    # length – a few thousand gives a good trade‑off between quality and speed
    n = 3000

    # --------------------------------------------------------------
    # 1.  Initial seed: Catalan / arcsine coefficients.
    # --------------------------------------------------------------
    seq = _catalan_sequence(n)

    # Normalise to L1 = 1 (the objective is scale‑invariant)
    seq /= seq.sum()

    # --------------------------------------------------------------
    # 2.  Aggressive refinement.
    # --------------------------------------------------------------
    seq = _refine_advanced(seq, deadline)

    # --------------------------------------------------------------
    # 3.  Final sanity check & return.
    # --------------------------------------------------------------
    final_score = _fast_evaluation(seq)
    print(f"Final score (fast estimate) = {final_score:.6f}")

    # ``evaluate_sequence`` will clip to the required [0,1000] range,
    # so we can safely hand over the raw values.
    return seq.tolist()
