def _build_toeplitz_matrix(f):
    n = len(f)
    m = 2*n - 1
    # offsets from 0..n-1 inclusive
    diags = [np.full(n, f[d], dtype=np.float64) for d in range(n)]

    A = sp.diags(diags, offsets, shape=(m, n), format='csr')

The algorithm works in three stages:

   combination  h = (1−t)·f + t·g  never exceeds the peak M,
   while the ℓ₁‑norm (the denominator of the evaluation formula) grows.
   A modest mixing weight `t` therefore reduces the score.
   The LP–mix step is repeated until the time budget is exhausted.

3. **Fine‑grained greedy polishing** – a cheap local move that repeatedly
   shifts a small amount of mass from the indices that contribute most to
   the current maximal convolution entry towards indices that contribute
   almost nothing.  This never increases the objective and usually gives a
   final touch.

The whole search respects the 1000 s budget, stays inside the allowed
value range, and returns the best sequence found.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Fast evaluation (identical to evaluate_sequence but without security checks)
# ----------------------------------------------------------------------
def _fast_evaluation(seq: np.ndarray) -> float:
    """Return the metric 2·n·max(conv)/sum² for a non‑negative sequence."""
    n = seq.size
    if n == 0:
        return np.inf
    m = 1 << ((2 * n - 1).bit_length())
    ft = np.fft.rfft(seq, m)
    conv = np.fft.irfft(ft * ft, m)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Catalan (arc‑sine) seed generator
# ----------------------------------------------------------------------
def _catalan_sequence(length: int) -> np.ndarray:
    """First ``length`` coefficients a_i = binom(2i,i) / 4**i."""
    seq = np.empty(length, dtype=float)
    a = 1.0
    seq[0] = a
    for i in range(1, length):
        a *= (2 * i - 1) / (2.0 * i)
        seq[i] = a
    return seq


# ----------------------------------------------------------------------
# Convolution helpers (FFT based, exact up to floating‑point noise)
# ----------------------------------------------------------------------
def _conv(seq1: np.ndarray, seq2: np.ndarray) -> np.ndarray:
    """Linear convolution of two 1‑D real arrays."""
    n1 = seq1.size
    n2 = seq2.size
    size = 1 << ((n1 + n2 - 1).bit_length())
    ft1 = np.fft.rfft(seq1, size)
    ft2 = np.fft.rfft(seq2, size)
    conv = np.fft.irfft(ft1 * ft2, size)[: n1 + n2 - 1]
    return conv


# ----------------------------------------------------------------------
# Build the Toeplitz matrix that represents (f * x)[k] ≤ M  (linear in x)
# ----------------------------------------------------------------------
def _build_toeplitz_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Return a sparse matrix A of shape (2·n‑1, n) such that
    (A @ x)[k] = (f * x)[k]   for k = 0 … 2n‑2.
    Only non‑negative offsets appear because f_i = 0 for i<0.
    """
    n = f.size
    m = 2 * n - 1
    # each diagonal offset d (0 ≤ d < n) contains the constant value f[d]
    # and has length n (the minimum of m‑d and n, which equals n here).
    diags = [np.full(n, f[d], dtype=np.float64) for d in range(n)]
    offsets = np.arange(n)          # 0,1,2,…,n‑1  (row‑col = offset)
    A = sp.diags(diags, offsets, shape=(m, n), format="csr")
    return A


# ----------------------------------------------------------------------
# LP‑guided refinement – one outer iteration
# ----------------------------------------------------------------------
def _lp_refine(seq: np.ndarray, deadline: float) -> np.ndarray:
    """
    Starting from ``seq`` improve it by alternating:
        * solve a linear program to obtain a vector g with maximal ℓ₁ norm
          under the cross‑convolution bound f∗g ≤ M,
        * scale g so that also g∗g ≤ M,
        * mix f and g with a moderate weight t (keeps the peak ≤ M
          while increasing the total mass).
    The procedure is repeated until the time budget expires.
    """
    n = seq.size
    f = seq.copy()
    best_seq = f.copy()
    best_score = _fast_evaluation(best_seq)

    iteration = 0
    while time.time() < deadline - 5:          # keep a little slack
        iteration += 1

        # (1) current convolution statistics
        conv_f = _conv(f, f)
        M_f = conv_f.max()
        sum_f = f.sum()

        # (2) linear program: maximise 1ᵀg  s.t.  f∗g ≤ M_f , g ≥ 0
        A = _build_toeplitz_matrix(f)
        c = -np.ones(n)                         # linprog minimises; –1 → maximise
        b = np.full(2 * n - 1, M_f * (1.0 + 1e-12))   # tiny tolerance
        bounds = [(0.0, None) for _ in range(n)]

        # a modest per‑iteration time limit for the LP
        lp_options = {"presolve": True, "time_limit": 30}
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method="highs", options=lp_options)

        if not res.success:
            # LP failed (possible numerical issues) – give up the LP loop
            print(f"LP failed at iteration {iteration}: {res.message}")
            break

        g0 = res.x
        sum_g0 = g0.sum()

        # (3) ensure the self‑convolution of g does not exceed M_f
        conv_g0 = _conv(g0, g0)
        M_g0 = conv_g0.max()
        if M_g0 <= 0:
            # degenerate solution – stop
            break
        gamma = min(1.0, np.sqrt(M_f / (M_g0 + 1e-14)))
        g = gamma * g0
        sum_g = g.sum()

        # If the new vector does not bring any extra mass, stop.
        if sum_g <= sum_f * (1.0 + 1e-6):
            # nothing to gain any more
            break

        # (4) cross‑convolution f∗g (guaranteed ≤ M_f because we only scaled down)
        conv_fg = _conv(f, g)

        # (5) try a few mixing weights – larger t gives a bigger increase of the sum
        #     while the peak never exceeds M_f (by construction).
        t_candidates = np.array([0.1, 0.25, 0.4, 0.5, 0.6, 0.8])
        improved = False
        for t in t_candidates:
            new_f = (1.0 - t) * f + t * g
            # compute the new score (fast, exact for the objective)
            score = _fast_evaluation(new_f)
            if score < best_score - 1e-12:
                f = new_f
                best_seq = new_f.copy()
                best_score = score
                improved = True
                print(f"Iter {iteration:3d} – t={t:.2f} – score={best_score:.9f}")
                break

        if not improved:
            # No improvement – inject a tiny random jiggle to escape a plateau.
            jitter = np.random.randn(n) * 1e-4
            f = np.maximum(f + jitter, 0.0)

    return best_seq


# ----------------------------------------------------------------------
# Greedy local polishing (identical to the earlier “refine” routine)
# ----------------------------------------------------------------------
def _greedy_refine(seq: np.ndarray, deadline: float) -> np.ndarray:
    """Simple mass‑move heuristic that never increases the objective."""
    n = seq.size
    best_seq = seq.copy()
    best_score = _fast_evaluation(best_seq)

    fft_len = 1 << ((2 * n - 1).bit_length())

    while time.time() < deadline:
        fa = np.fft.rfft(seq, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        k = int(conv.argmax())

        i_arr = np.arange(n)
        j_arr = k - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        w = np.zeros_like(seq)
        w[mask] = seq[j_arr[mask]]

        contrib = seq * w
        src = int(np.argmax(contrib))
        sink = int(np.argmin(w))

        if seq[src] < 1e-12:
            break

        eps = 0.05 * seq[src]                 # move 5 % of the source mass
        new_seq = seq.copy()
        new_seq[src] -= eps
        new_seq[sink] += eps

        score = _fast_evaluation(new_seq)
        if score < best_score - 1e-12:
            seq = new_seq
            best_score = score
            best_seq = new_seq.copy()
        else:
            # occasional tiny random perturbation to escape flat regions
            if np.random.rand() < 0.01:
                j = np.random.randint(n)
                jitter = (np.random.rand() - 0.5) * 0.001
                seq = seq.copy()
                seq[j] = max(0.0, seq[j] + jitter)

    return best_seq


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights aiming for a low evaluation score.

    The function works inside ``budget_s`` seconds (default 1000).  It
    starts from the classic arc‑sine seed, performs a few LP‑guided
    mixing steps, and finally finishes with a cheap greedy polishing.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5   # a small safety margin

    # Length – a few thousand give a good trade‑off between quality and speed.
    # For the LP stage we use a slightly smaller size to keep the linear
    # programs tractable.
    n_lp = 1200
    n_final = 3000

    # ------------------------------------------------------------------
    # 1.  LP‑guided improvement on a medium‑size problem
    # ------------------------------------------------------------------
    seq_lp = _catalan_sequence(n_lp)
    seq_lp = _lp_refine(seq_lp, deadline)

    # ------------------------------------------------------------------
    # 2.  Upscale to the final length (simple zero‑padding + interpolation)
    # ------------------------------------------------------------------
    # We keep the shape of the improved sequence and embed it into a larger
    # vector by linear interpolation.  Padding with zeros does not hurt the
    # objective because the extra zeros add no mass and only increase the
    # denominator (the sum stays the same) while the peak cannot increase.
    if n_final > n_lp:
        # simple linear interpolation to spread the mass a little more evenly
        x_old = np.linspace(0, 1, n_lp)
        x_new = np.linspace(0, 1, n_final)
        seq_big = np.interp(x_new, x_old, seq_lp)
    else:
        seq_big = seq_lp[:n_final]

    # ------------------------------------------------------------------
    # 3.  Final greedy polishing on the full length
    # ------------------------------------------------------------------
    seq_big = _greedy_refine(seq_big, deadline)

    # ------------------------------------------------------------------
    # 4.  Return as a plain Python list (the official evaluator will clip)
    # ------------------------------------------------------------------
    return seq_big.tolist()
