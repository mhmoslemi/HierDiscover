"""
Propose a candidate step‑function (list of heights) that minimises

    2 * n * max(convolve(a, a)) / (sum(a)**2)

The algorithm consists of three phases:

1. **Initial pool** – a few parametric families (power‑law, random Dirichlet)
   and the provided fallback sequence are evaluated.  The best few are kept.
2. **Projected gradient descent on a smoothed max‑objective** –  
   minimise a smooth approximation of the infinity‑norm of the self‑convolution
   using a log‑sum‑exp (temperature β).  After each gradient step the vector
   is projected onto the probability simplex (non‑negative, sum = 1).  
   β is increased gradually, turning the smooth objective into the true max.
3. **Fine‑grained hill‑climbing** – small random mass transfers between two
   positions are tried; improvements are kept.  This fine‑tunes the result.

All work is time‑boxed (default 950 s) and the best sequence found so far is
returned.  The code uses only NumPy and the already‑imported ``evaluate_sequence``.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the simplex {x ≥ 0, sum(x)=1}."""
    # Implementation of Wang & Carreira‑Perpiñán (2013)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _compute_score(a: np.ndarray) -> float:
    """Exact score used by the challenge (scale‑invariant)."""
    n = a.shape[0]
    c = np.convolve(a, a)
    max_c = np.max(c)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


def _gradient_step(a: np.ndarray, beta: float, lr: float) -> np.ndarray:
    """
    One projected gradient step on the smoothed max‑objective.

    Parameters
    ----------
    a : np.ndarray
        Current probability vector (sum = 1, non‑negative).
    beta : float
        Temperature for the log‑sum‑exp smoothing.
    lr : float
        Learning‑rate (step size).

    Returns
    -------
    np.ndarray
        Updated vector, already projected onto the simplex.
    """
    # convolution a * a
    c = np.convolve(a, a)
    c_max = np.max(c)

    # soft‑max weights for the smoothed max
    # we subtract the max to avoid overflow
    w = np.exp(beta * (c - c_max))
    w_sum = np.sum(w)
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.shape[0])
    else:
        w /= w_sum

    # correlation w ⋆ a (full mode) gives Σ_k w_k a_{k-j}
    # the needed slice (for j = 0 … n‑1) is the first n entries of the reversed tail
    corr = np.correlate(w, a, mode='full')
    # corr length = 2*n-1 + n -1 = 3*n-2
    # for j we need corr[n-1-j]; collect the first n elements in reverse order
    grad_base = corr[len(a) - 1 :: -1][: len(a)]

    # diagonal terms (k = 2j) have coefficient 1 instead of 2
    w_diag = w[2 * np.arange(len(a))]
    grad = 2.0 * grad_base - w_diag * a

    # gradient descent step followed by projection onto the simplex
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    a_new = _project_to_simplex(a_new)
    return a_new

# Main candidate generator
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: float = 950, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises the
    evaluation function.  The routine is time‑boxed; it returns the best
    sequence found before the deadline.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # keep a small safety margin

    # ------------------------------------------------------------------
    # Determine the length of the problem (use the provided fallback if present)
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)                 # type: ignore[name-defined]

    best_seq = baseline_seq.copy()
    best_score = _compute_score(best_seq)
    print(f"[init] length n={n}, baseline score={best_score:.6f}")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    candidates = [best_seq]

    # a) power‑law family a_i ∝ (i+shift)^(-alpha)
    idx = np.arange(n, dtype=float)

    top_candidates = [vec for _, vec in scored[:6]]

            if best_score <= 1.503:
                break
        if best_score <= 1.503:

    # ------------------------------------------------------------------
    # 2) Fine‑grained hill‑climbing (small random mass swaps)
    # ------------------------------------------------------------------
    cur = best_seq.copy()
    cur_score = best_score
    max_hc_iter = 30000
    hc_iter = 0
    while time.time() < deadline and hc_iter < max_hc_iter:
        hc_iter += 1
        i, j = np.random.randint(n, size=2)
        if i == j:

        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # Final reporting
    # ------------------------------------------------------------------
    print(f"[final] best score found: {best_score:.6f}")
    # Return as a plain Python list, as required by the harness
    return best_seq.tolist()
