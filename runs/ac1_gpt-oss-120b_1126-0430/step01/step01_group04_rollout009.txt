"""
Propose a step‑function (non‑negative list) that tries to minimise  

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm works in two phases

1️⃣  **Diverse initialisation** –  a handful of analytically‑parameterised
    families (inverse‑sqrt, generic power‑law, shifted power, polynomial,
    random Dirichlet, the previous best if available).  Each candidate is
    scaled to the required L¹‑norm √(2 n) and the best one is kept.

2️⃣  **Sub‑gradient hill‑climbing / simulated‑annealing** –  
    starting from the best candidate, the index `k*` at which the
    convolution attains its maximum is identified.  The derivative of the
    maximal convolution w.r.t. `a_i` is simply `a_{k*‑i}` (0 when the
    index is out of range).  This gives a “contribution” vector.  Mass is
    transferred from an index with a *large* contribution (large impact on
    the peak) to an index with a *small* contribution while preserving the
    total sum.  The move size is a diminishing fraction of the source
    entry; better moves are always accepted, worse moves are accepted with
    a temperature‑controlled probability.  The temperature (and the
    maximal transfer fraction) decay slowly, allowing a long, fine‑grained
    exploration of the search space.

The routine respects the global time budget (default = 1000 s), prints
intermediate scores and finally returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level – no closures)
# ----------------------------------------------------------------------


def _project_to_simplex(x: np.ndarray, target_sum: float) -> np.ndarray:
    """Project a vector onto the probability simplex { y ≥ 0, Σ y = target_sum }."""
    n = x.shape[0]
    if target_sum <= 0:
        return np.zeros_like(x)
    # algorithm of Wang & Carreira‑Perpiñán (2013)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho_arr = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    rho = rho_arr[-1] if rho_arr.size else 0
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(x - theta, 0.0)
    return w


def _evaluate(seq: list[float]) -> float:
    """Thin wrapper around the supplied ``evaluate_sequence``."""
    # The evaluation function is guaranteed to exist in the global namespace.
    return evaluate_sequence(seq)


def _make_sequence_from_raw(raw: np.ndarray, target_sum: float) -> list[float]:
    """Scale a non‑negative raw vector so that its L¹‑norm equals ``target_sum``."""
    raw_sum = raw.sum()
    if raw_sum == 0:
        # fallback to uniform distribution
        n = raw.shape[0]
        return [target_sum / n] * n
    scale = target_sum / raw_sum
    seq = (raw * scale).tolist()
    return seq

# Core optimisation routine
# ----------------------------------------------------------------------

    target_sum = math.sqrt(2.0 * n)         # normalising L¹‑norm

    # --------------------------------------------------------------
    # Phase 1 – generate a few analytic candidates
    # --------------------------------------------------------------
    best_seq = [target_sum / n] * n          # uniform start

        contrib = np.zeros(n, dtype=float)
        contrib[lo:hi + 1] = current[k_max - i_range]

        src_candidates = np.where(current > 1e-12)[0]
        if src_candidates.size == 0:

        tgt_idx = int(np.argmin(contrib))   # may be the same – handle later

        new_score = _evaluate(new_seq.tolist())

                best_seq = shake.tolist()
                print(f"[shake {iter_cnt:06d}] new best {best_score:.6f}")
                current = shake
                current_score = shake_score
            no_improve = 0
            # reset temperature a little to allow exploration
            temperature = max(temperature, 0.005)

    print(f"[final] best score found: {best_score:.6f}")
