def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) with a low
    evaluation score.  The search consists of three phases:
    1.  A coarse grid‑search over the family a_i = (i+offset+1)^−β for several
        offsets, β values and a fixed length n.
    2.  A refined search around the best parameters (finer β step and a small
        neighbourhood of offsets).
    3.  A lightweight hill‑climb that repeatedly transfers a small amount of
        mass between two positions; moves are accepted only when they lower the
        score.
    All phases respect the global 1000 s time‑budget and the function returns
    the best sequence found.
    """
    import numpy as np
    import time

    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.2

    best_seq = None
    best_score = float('inf')

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            sc = evaluate_sequence(height_sequence_1)
            best_seq = list(height_sequence_1)
            best_score = sc
            print(f"[init] existing sequence score = {sc:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    n = 2000                      # fixed length – large enough for a smooth shape
    offsets = list(range(0, 31, 5))   # 0,5,…,30
    betas = np.linspace(0.45, 0.75, 61)   # step ≈0.005

    for off in offsets:

            if time.time() > deadline:

            if score < best_score:

def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    `evaluate_sequence`.  The search proceeds in three phases:

    1. **Coarse grid search** over the family
       a_i = (i + offset + 1)^{-beta}   (i = 0 … n‑1).
       Offsets in {0,5,…,30} and betas in [0.45,0.75] (step ≈0.005) are tried
       for a fixed length n = 2000.  The best parameters are kept.

    2. **Refined local search** around the best offset and beta found in step 1.
       Offsets are examined in ±5 of the current best and betas in a
       finer window ±0.03 with step ≈0.001.

    3. **Hill‑climbing** starting from the best sequence obtained so far.
       Randomly chosen pairs of positions transfer up to 5 % of the source
       mass; a move is kept only if it strictly lowers the score.

    All phases respect the global time budget (≈1000 s).  The function
    returns the best sequence found (as a list of floats).  The algorithm
    relies only on `numpy` and the provided `evaluate_sequence`.
    """
    import numpy as np
    import time

    # --------------------------------------------------------------
    # initialise RNG and timing
    # --------------------------------------------------------------
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.2          # tiny safety margin

    best_seq = None
    best_score = float('inf')
    best_off = 0
    best_beta = 0.0
    n = 2000                                   # primary length for the family

    # --------------------------------------------------------------
    # 0.  Use a previously discovered construction if it exists
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            sc = evaluate_sequence(height_sequence_1)
            best_seq = list(height_sequence_1)
            best_score = sc
            print(f"[init] reused previous sequence, score = {sc:.6f}")
        except Exception:
            pass

    # --------------------------------------------------------------
    # 1.  Coarse grid search on (offset, beta)
    # --------------------------------------------------------------
    offsets = list(range(0, 31, 5))                # 0,5,…,30
    betas = np.linspace(0.45, 0.75, 61)            # step ≈0.005

    for off in offsets:
        if time.time() > deadline:
            break
        for beta in betas:
            if time.time() > deadline:
                break
            # sequence: a_i = (i + 1 + off)^{-beta}
            idx = np.arange(1, n + 1, dtype=np.float64) + off
            seq = idx ** (-beta)
            try:
                score = evaluate_sequence(seq.tolist())
            except Exception:
                continue
            if score < best_score:
                best_score = score
                best_seq = seq.tolist()
                best_off = off
                best_beta = beta
                print(f"[grid] offset={off:2d}, beta={beta:.4f} → new best = {score:.6f}")

    # --------------------------------------------------------------
    # 2.  Refined search around the current best parameters
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        off_range = range(max(0, best_off - 5), best_off + 6)   # ±5
        beta_fine = np.linspace(
            max(0.25, best_beta - 0.03),
            min(0.95, best_beta + 0.03),
            61)                                                # step ≈0.001
        for off2 in off_range:
            if time.time() > deadline:
                break
            for beta2 in beta_fine:
                if time.time() > deadline:
                    break
                idx2 = np.arange(1, n + 1, dtype=np.float64) + off2
                seq2 = idx2 ** (-beta2)
                try:
                    score2 = evaluate_sequence(seq2.tolist())
                except Exception:
                    continue
                if score2 < best_score:
                    best_score = score2
                    best_seq = seq2.tolist()
                    best_off = off2
                    best_beta = beta2
                    print(f"[refine] offset={off2:2d}, beta={beta2:.4f} → new best = {score2:.6f}")

    # --------------------------------------------------------------
    # 3.  Light hill‑climb starting from the best sequence
    # --------------------------------------------------------------

    print(f"[final] best score = {best_score:.6f}, length = {len(best_seq)}")
