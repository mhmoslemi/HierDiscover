"""
propose_candidate – LP based refinement of a step‑function.

The objective that we want to minimise is

    2 * n * max(conv(a, a)) / (sum(a)**2)

which is invariant under scaling of ``a``.  By normalising every candidate
to have   sum(a) = sqrt(2·n)   the denominator cancels and the score is
exactly the maximal convolution value.  The algorithm proceeds in three
phases:

1. **Exploratory grid search** – a collection of simple parametric families
   (power‑law and exponential) is scanned; the best sequence found is kept
   as a starting point.

2. **LP–based sharpening** – given a current sequence ``f`` with maximal
   convolution ``M = max(conv(f,f))`` we solve the linear programme

        maximise   ∑ g_i
        subject to  (f * g)[k] ≤ M   for all k = 0…2n‑2
                    g_i ≥ 0

   (the convolution with ``g`` is linear in ``g``).  The optimiser drives as
   much mass as possible into indices where ``f`` is small, thereby creating
   a new sequence ``g`` with a larger L1‑norm but whose cross‑convolution with
   ``f`` still stays below ``M``.  After scaling ``g`` to the target L1‑norm
   we replace ``f←(1‑t)·f + t·g`` (``t``≈0.1).  The “mixed’’ sequence never
   increases the maximal self‑convolution, so repeating the step drives the
   score down.

3. **Optional hill‑climbing** – a cheap local search moves a tiny amount of
   mass between two random positions while preserving the L1‑norm; moves that
   improve the score are kept.

The whole routine is time‑boxed (default 950 s) and reports intermediate
scores.  If the LP solver is unavailable the code falls back to the original
grid‑+‑hill‑climb strategy.
"""

def _project_to_simplex(v):
    """Project a vector onto the probability simplex {x≥0, sum(x)=1}."""
    x = np.asarray(v, dtype=float)
    n = x.shape[0]
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(x - theta, 0)
    return w


def _normalise_to_target(seq, target):
    """Scale ``seq`` so that its L1‑norm equals ``target`` (preserves sign)."""
    s = np.sum(seq)
    if s == 0:
        return seq
    return seq * (target / s)


def _conv_max(seq):
    """Return max(conv(seq,seq)).  ``seq`` is a 1‑D numpy array."""
    return np.max(np.convolve(seq, seq))


def _build_conv_matrix(f):
    """
    Build the dense matrix A such that (A @ g) == np.convolve(f, g)
    for any vector g of the same length as f.
    Shape of A is (2*len(f)-1) × len(f).
    """
    n = f.shape[0]
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f
    return A

                best_seq = new_seq.tolist()
                print(f"[LP] iter={lp_iter}, improved score={new_score:.6f}")

            # Decide whether to keep the new sequence for the next iteration
            # (if the maximal convolution did not increase we accept it)
            if new_score <= M * 1.000001:       # a tiny tolerance for noise
                cur = new_seq
            else:
                # If the combination made things worse, fall back to the older
                # sequence (the LP step may be too aggressive)
                print(f"[LP] iter={lp_iter}: slight increase, discarding step")
                # nevertheless we keep the best sequence found so far
                # and continue using the previous ``cur``.
                # No change to ``cur``.
                pass

    # ------------------------------------------------------------------
    # 4. Simple hill‑climbing fine‑tuning (cheap, many cheap moves)
    # ------------------------------------------------------------------
    cur = np.array(best_seq, dtype=float)          # start from the current best
    cur_score = best_score
    n_moves = 0
    max_moves = 30000

    while (time.time() - start_time) < (budget_s - 5) and n_moves < max_moves:
        n_moves += 1
        # pick two distinct positions
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        # propose moving a small chunk (up to 1 % of the source mass)
        step = cur[i] * (np.random.rand() * 0.01)
        if step <= 0:
            continue
        if cur[j] + step > 1000.0:   # guard against the hard caps inside evaluate_sequence
            continue

        candidate = cur.copy()
        candidate[i] -= step
        candidate[j] += step

        # optional projection to the target L1‑norm (keeps numerical errors in check)
        candidate = _normalise_to_target(candidate, target_sum)

        # evaluate
        cand_score = float(_conv_max(candidate))

        if cand_score < cur_score - 1e-9:      # strict improvement
            cur = candidate
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cur.tolist()
                print(f"[refine] move={n_moves}, improved score={cand_score:.6f}")

    print(f"[final] best score found: {best_score:.6f}")
    return best_seq
