"""
Propose a candidate step‑function (list of non‑negative heights) that minimises
the supplied ``evaluate_sequence`` function.

The algorithm improves on the previous sub‑gradient method by directly
optimising a smooth approximation of the max‑convolution value.
For a fixed length `n` the objective (after normalising the sum of the
coefficients) is proportional to

    max_i (a * a)_i

where `*` denotes linear convolution.  The function
    M_λ(a) = (1/λ)·log(∑_i exp(λ·(a*a)_i))

smoothly approximates the max when λ is large.  Its gradient can be written
in closed‑form using only two FFT‑based convolutions, so each optimisation step
costs O(n log n).  We apply an Adam‑style adaptive step‑size, gradually
increase λ (sharpening the approximation) and periodically restart from a
random point if progress stalls.  The best sequence seen during the allotted
time budget is returned.

Running time is well under the 1000 s limit even for n≈4000.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------
def _catalan_sequence(n: int) -> np.ndarray:
    """Return the first *n* Catalan‑like coefficients a_i = C(2i,i)/4**i,
    normalised so that their sum equals 1."""
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    # recurrence: a_{i+1} = a_i * (2*i+1) / (2*(i+1))
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * (i - 1) + 1) / (2.0 * i)
    a /= a.sum()
    return a


def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution using real FFT (fast for lengths ≈ few thousand)."""
    n = len(x) + len(y) - 1
    m = 1 << (n - 1).bit_length()          # next power of two
    X = np.fft.rfft(x, m)
    Y = np.fft.rfft(y, m)
    return np.fft.irfft(X * Y, m)[:n]


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex (non‑negative, sum = 1)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 0.0:
        return v / s
    # degenerate case – fall back to a random point on the simplex
    v = np.random.rand(v.shape[0])
    return v / v.sum()


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a non‑negative height sequence that minimises ``evaluate_sequence``.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : int, optional
        Wall‑clock time budget in seconds (default 1000).

    Returns
    -------
    list[float]
        The best sequence encountered (as a plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5          # small safety margin

    # --------------------------------------------------------------
    # Problem size – a few thousand entries give a good trade‑off.
    # --------------------------------------------------------------
    n = 4000                                        # you may tune this

    # Initialise with the (near‑optimal) Catalan construction.
    a = _catalan_sequence(n)

    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # Adam hyper‑parameters
    lr = 0.08                      # base learning rate
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    t = 0

    # smoothing parameter for the log‑sum‑exp approximation
    lam = 5.0

    iteration = 0
    no_improve = 0

    while time.time() < deadline:
        # ---- convolution and smooth max -------------------------------------------------
        conv = _fft_convolve(a, a)                # length 2n‑1
        max_conv = conv.max()

        # log‑sum‑exp (smooth max) – we only need the weights, not the value itself
        # shift by max_conv to avoid overflow; the max entry gets weight 1.
        weights = np.exp(lam * (conv - max_conv))
        w_sum = weights.sum()

        # weighted sum of shifted a needed for the gradient:
        #   grad_h = 2 * Σ_i w_i * a_{i‑h}
        # This is a convolution of the weight vector with the reversed a.
        a_rev = a[::-1]
        weighted = _fft_convolve(weights, a_rev)      # length 3n‑2
        start = n - 1

        t += 1
        m = beta1 * m + (1 - beta1) * grad

        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
