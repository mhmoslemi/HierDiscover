def get_max_and_grad(a):
    conv = np.convolve(a, a)
    max_idx = np.argmax(conv)
    # g[k] = 2*a[max_idx - k] if 0 <= max_idx - k < n else 0
    # using slicing: g = 2 * reversed part
    # We can compute using zeros and assignment
    n = len(a)
    g = np.zeros(n)
    lo = max(0, max_idx - (n - 1))
    hi = min(n - 1, max_idx)
    # for k in range(lo, hi+1):
    #   g[k] = 2 * a[max_idx - k]
    # Use vectorized:
    ks = np.arange(lo, hi+1)
    g[ks] = 2 * a[max_idx - ks]
    return conv[max_idx], g

The score that has to be minimised is  

\[
\text{score}(a)=\frac{2n\;\max (a*a)}{(\sum_i a_i)^2},
\qquad a_i\ge 0 .
\]

Because the denominator is fixed (the optimiser always rescales the
vector to the normalising sum  
\(\displaystyle S=\sqrt{2n}\) ),
the problem reduces to *flattening* the self‑convolution
\(c=a*a\) – we would like every entry of \(c\) to be as close as possible
to its average value
\( \bar c = S^2/(2n-1)\).

The algorithm below consists of three phases

1. **Broad random search** – a cheap parameter sweep that creates many
   analytically‑parameterised families (power law, shifted power,
   polynomial decay, linear decay, random Dirichlet).  
   The best candidate of this phase is kept as the starting point.

2. **Gradient descent on the maximum convolution entry** –  
   For a given vector `a` we compute the convolution `c = a*a`
   (FFT is used, O(n log n)).  
   Let `i* = argmax c`.  
   The sub‑gradient of the maximal entry w.r.t. `a` is

   \[
   \frac{\partial c_{i*}}{\partial a_k}= 2\,a_{i*-k}\quad
   (\text{if }0\le i*-k < n,\;0\text{ otherwise}),
   \]

   i.e. a shifted copy of `a`.  
   A step *opposite* to that gradient reduces the dominant peak.
   After each step the vector is projected back onto the simplex
   \(\{x\ge 0,\ \sum x=S\}\) (exact L1‑norm is kept).  
   The step size is adapted automatically – it grows when a step
   improves the score and shrinks otherwise.

3. **Fine‑grained local refinement** – a small simulated‑annealing /
   hill‑climbing that moves a tiny amount of mass between two random
   positions.  This phase removes the last remaining irregularities.

All three phases respect the 1000 s time budget and work for any
length `n` (the global variable `height_sequence_1` is used when it
exists).  The code prints progress information and finally returns the
best sequence found.

"""
Propose a step‑function (non‑negative list of heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The optimiser works in three stages:

1. Parameter sweep – generate many analytically‑parameterised candidates
   (power‑law, shifted power‑law, polynomial decay, linear decay,
   random Dirichlet).  Keep the best one.

2. Gradient descent on the maximal convolution entry.
   For the current maximal index i* the sub‑gradient is
       g_k = 2 * a[i* - k]  (if the index is inside the vector, else 0).

   (non‑negative, fixed L1‑norm) and accept the move if it improves the

   automatically.

3. Local refinement – simulated‑annealing / hill‑climbing moves that
   transfer a tiny amount of mass between two random positions.
   This cleans up the remaining small irregularities.

(the global `height_sequence_1` is used when present) and returns the

def _project_to_simplex(x: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Project a vector onto the probability simplex
    { y >= 0, sum(y) = target_sum }.
    Implementation follows Wang & Carreira‑Perpiñán (2013) – O(n log n)
    because of the sorting step.
    """
    if target_sum <= 0:
        return np.zeros_like(x)

    # sort descending
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho_arr = u - cssv / np.arange(1, len(x) + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(x - theta, 0.0)
    return w


def _conv_max_score(a: np.ndarray, target_sum: float) -> float:
    """
    Compute the optimisation score of a vector a.
    a is assumed to be non‑negative and its L1‑norm is (approximately)
    target_sum.  The function uses FFT to obtain the convolution quickly.
    """
    n = a.shape[0]
    # length for FFT: next power of two >= 2*n-1
    L = 1
    while L < 2 * n - 1:
        L <<= 1
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_c = conv.max()
    # the score is exactly 2*n*max_c / (target_sum**2)
    return 2.0 * n * max_c / (target_sum ** 2)


def _gradient_max_entry(a: np.ndarray, max_idx: int) -> np.ndarray:
    """
    Sub‑gradient of the maximal convolution entry (index max_idx)
    w.r.t. a.  For each k with 0 ≤ max_idx‑k < n we have
        g_k = 2 * a[max_idx - k]
    otherwise g_k = 0.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    lo = max(0, max_idx - (n - 1))
    hi = min(n - 1, max_idx)
    ks = np.arange(lo, hi + 1, dtype=int)
    grad[ks] = 2.0 * a[max_idx - ks]
    return grad


def _topk_gradient(a: np.ndarray, conv: np.ndarray, k: int = 1) -> np.ndarray:
    """
    Combine the sub‑gradients of the k largest convolution entries.
    """
    n = a.shape[0]
    if k >= conv.size:
        idxs = np.arange(conv.size)
    else:
        idxs = np.argpartition(-conv, np.arange(k))[:k]

    # optional weighting – we use equal weights here
    grad = np.zeros_like(a)
    for idx in idxs:
        lo = max(0, idx - (n - 1))
        hi = min(n - 1, idx)
        ks = np.arange(lo, hi + 1, dtype=int)
        grad[ks] += 2.0 * a[idx - ks]
    return grad

    else:
        n = 2000                     # a reasonable default
    target_sum = math.sqrt(2.0 * n)   # normalising L1‑norm

    # ------------------------------------------------------------------
    # Helper to evaluate a raw (non‑scaled) vector
    # ------------------------------------------------------------------
    def _make_candidate(raw: np.ndarray) -> np.ndarray:
        raw = np.maximum(raw, 0.0)
        s = raw.sum()
        if s == 0:
            raw = np.full_like(raw, target_sum / n)
            s = target_sum
        scale = target_sum / s
        return raw * scale

    # ------------------------------------------------------------------
    # Phase 1 – cheap random sweep
    # ------------------------------------------------------------------
    start = time.time()
    phase1_deadline = start + budget_s * 0.25   # 25 % of the budget
    best_seq = np.full(n, target_sum / n, dtype=float)
    best_score = _conv_max_score(best_seq, target_sum)

    print(f"[phase‑1] start, n={n}, uniform score={best_score:.6f}")

    sweep_it = 0
    while time.time() < phase1_deadline:
        sweep_it += 1
        shape = np.random.choice(
            ["power", "shifted_power", "poly", "linear", "rand_dir"]
        )
        if shape == "power":
            p = np.random.uniform(0.05, 1.2)
            raw = (np.arange(1, n + 1, dtype=float)) ** (-p)

        elif shape == "shifted_power":
            p = np.random.uniform(0.05, 1.2)
            d = np.random.uniform(0.0, n * 0.5) + 1e-6
            raw = (np.arange(n, dtype=float) + d) ** (-p)

        elif shape == "poly":
            alpha = np.random.uniform(0.1, 6.0)
            raw = (1.0 - np.arange(n, dtype=float) / (n - 1)) ** alpha

        elif shape == "linear":
            raw = np.linspace(1.0, 0.0, n, dtype=float)

        else:   # "rand_dir"
            raw = np.random.gamma(shape=0.7, scale=1.0, size=n)

        cand = _make_candidate(raw)
        score = _conv_max_score(cand, target_sum)
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(
                f"[sweep {sweep_it:05d}] new best {best_score:.6f} via {shape}"
            )

        # occasionally do a tiny random perturbation of the current best
        if sweep_it % 500 == 0 and np.random.rand() < 0.15:
            tmp = best_seq.copy()
            i, j = np.random.choice(n, size=2, replace=False)
            delta = tmp[i] * np.random.uniform(0.0, 0.02)
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_to_simplex(tmp, target_sum)
            pscore = _conv_max_score(tmp, target_sum)
            if pscore < best_score:
                best_score = pscore
                best_seq = tmp
                print(f"[perturb {sweep_it:05d}] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – gradient descent on the maximal convolution entry
    # ------------------------------------------------------------------
    print("[phase‑2] gradient descent start")
    a = best_seq.copy()
    current_score = best_score

    # step size handling
    step = target_sum * 0.5          # initial L2 step length (roughly 0.5·S)
    min_step = target_sum * 1e-6
    max_step = target_sum
    decay_factor = 0.97

    # we use only the single largest entry for the gradient (k=1)
    k_gradient = 1

    phase2_deadline = start + budget_s * 0.85   # up to 85 % of total time
    it = 0
    while time.time() < phase2_deadline:
        it += 1

        # compute convolution and its maximum
        # FFT size (pre‑computed power of two)
        L = 1
        while L < 2 * n - 1:
            L <<= 1
        fa = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
        max_idx = int(conv.argmax())
        max_val = conv[max_idx]
        current_score = 2.0 * n * max_val / (target_sum ** 2)

        # gradient of the maximal entry (or top‑k combined)
        grad = _topk_gradient(a, conv, k=k_gradient)

        # take a step opposite to the gradient
        cand = a - step * grad
        cand = _project_to_simplex(cand, target_sum)

        # evaluate candidate
        cand_score = _conv_max_score(cand, target_sum)

        # accept if improvement, otherwise shrink step
        if cand_score < current_score:
            a = cand
            current_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(
                    f"[gd {it:06d}] improved to {best_score:.6f} (step={step:.4f})"
                )
            # gently increase step size after success
            step = min(step * 1.02, max_step)
        else:
            # reject the move and make the step smaller
            step = max(step * decay_factor, min_step)

        # occasional random “kick” to avoid stagnation
        if it % 3000 == 0:
            idx = np.random.choice(n, size=2, replace=False)
            delta = a[idx[0]] * np.random.uniform(0.0, 0.1)
            a[idx[0]] -= delta
            a[idx[1]] += delta
            a = _project_to_simplex(a, target_sum)

    # ------------------------------------------------------------------
    # Phase 3 – fine‑grained simulated‑annealing refinement
    # ------------------------------------------------------------------
    print("[phase‑3] local refinement start")
    cur = best_seq.copy()
    cur_score = best_score
    temperature = 0.02
    cooling = 0.9998

    phase3_deadline = start + budget_s - 1.5   # leave a couple seconds for exit
    while time.time() < phase3_deadline:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        delta = cur[i] * np.random.rand() * 0.015
        nxt = cur.copy()
        nxt[i] -= delta
        nxt[j] += delta
        nxt = _project_to_simplex(nxt, target_sum)
        nxt_score = _conv_max_score(nxt, target_sum)

        if nxt_score < cur_score:
            cur = nxt
            cur_score = nxt_score
            if nxt_score < best_score:
                best_score = nxt_score
                best_seq = nxt.copy()
                print(f"[refine] new best {best_score:.6f}")
        else:
            # Metropolis acceptance
            prob = math.exp(-(nxt_score - cur_score) / (temperature + 1e-12))
            if np.random.rand() < prob:
                cur = nxt
                cur_score = nxt_score

        temperature *= cooling

    print(f"[final] best score found: {best_score:.6f}")
    # Convert to plain list (the evaluator expects a list of floats)
    return best_seq.tolist()
