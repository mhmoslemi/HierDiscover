```
def arcsine_sequence(gamma, delta, n):
    idx = np.arange(n, dtype=np.float64) + delta   # i from 0..n-1
    # compute product with reversed indices: (n - 1 - i + delta)
    rev = (n - 1 - np.arange(n, dtype=np.float64)) + delta
    a = (idx * rev) ** (-gamma)
    s = a.sum()
    if s > 0:
        a /= s
    return a
```

    N = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    C = A * B
    return np.fft.irfft(C, N)[:size]

def fast_score(a):
    n = len(a)
    b = convolve_fft(a, a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 1e-12:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))

def arcsine_sequence(gamma, delta, n):
    # Build product (i+delta)*(N-i+delta)
    i = np.arange(n, dtype=np.float64) + delta
    rev = (n - 1 - np.arange(n, dtype=np.float64)) + delta
    a = (i * rev) ** (-gamma)
    total = a.sum()
    if total > 0:
        a /= total
    return a

def gradient_refine(a, deadline):
    """Projected gradient on L2 error to flat target."""
    n = len(a)
    target_const = (a.sum() ** 2) / (2 * n - 1)
    step_factor = 0.1
    best_a = a.copy()
    best_score = fast_score(a)
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        b = convolve_fft(a, a)
        err = b - target_const
        # gradient of L2 error
        grad = 4.0 * np.correlate(err, a, mode='valid')
        gn = np.linalg.norm(grad)
        if gn == 0.0:
            break
        eta = step_factor / gn
        a = a - eta * grad
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= s
        cur = fast_score(a)
        if cur < best_score:
            best_score = cur
            best_a = a.copy()
            print(f'  gradient iter {iteration} improved score {cur:.6f}')
            step_factor = max(step_factor * 0.95, 1e-4)
    return best_a

def subgradient_descent(a, deadline):
    """Projected sub‑gradient descent on the max‑convolution objective."""
    n = len(a)
    best_a = a.copy()
    best_score = fast_score(a)
    it = 0
    eta0 = 0.5
    while time.time() < deadline:
        it += 1
        b = convolve_fft(a, a)
        max_val = b.max()
        # pick all indices whose value is within a tiny relative tolerance
        tol = max_val * 1e-12
        active = np.where(b >= max_val - tol)[0]
        # construct gradient
        G = len(active) * a
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            G[idx] += a[k - idx]
        eta = eta0 / np.sqrt(it)
        a = a - eta * G
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= s
        cur = fast_score(a)
        if cur < best_score:
            best_score = cur
            best_a = a.copy()
            print(f'  subgrad iter {it} improved score {cur:.6f}')
        # stop early if step extremely small
        if eta * np.linalg.norm(G) < 1e-12:
            break
    return best_a

"""
Search for a low‑score step‑function.

The algorithm proceeds in three stages:

1. **Parametric family** –  Build a symmetric “arcsine”‑type sequence  
   a_i ∝ ((i+δ)*(N‑i+δ))^{-γ} (N = n‑1).  This discrete analogue of the
   continuous arcsine law has a self‑convolution that is nearly flat,
   i.e. close to the theoretical optimum 2 n · 1/(2n‑1) ≈ 1.

2. **Coarse‑then‑fine grid search** –  Scan γ ∈ [0.35,0.70] and a small
   shift δ ∈ [0.2,2.0] (first coarse, then a finer neighbourhood) and keep
   the best candidate (fast_score evaluates 2·n·max(conv(a,a))/(sum(a)²) ).

3. **Refinement** –  Two cheap projected‑gradient phases are applied:
   * an L2‑gradient step that drives the convolution towards the flat target,
   * a sub‑gradient step directly on the max‑convolution objective.
   Both keep the vector non‑negative and renormalise it after each step.

All work fits comfortably inside the 1000 s budget.  The function returns
the best sequence found as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def convolve_fft(a, b):
    """Linear convolution using a power‑of‑two FFT (real inputs)."""
    size = len(a) + len(b) - 1
    N = 1 << (size - 1).bit_length()          # next power of two
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    C = A * B
    return np.fft.irfft(C, N)[:size]


# ----------------------------------------------------------------------
def fast_score(a):
    """Objective 2·n·max(conv(a,a)) / (sum(a)²).  a is assumed to be a np.ndarray."""
    n = len(a)
    b = convolve_fft(a, a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 1e-12:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
def arcsine_sequence(gamma, delta, n):
    """
    Symmetric beta‑type sequence:
        a_i ∝ ((i+δ)*(N-i+δ))^{-γ},   i = 0 … n‑1,  N = n‑1.
    The sequence is normalised to sum‑one.
    """
    i = np.arange(n, dtype=np.float64) + delta                     # i+δ
    rev = (n - 1 - np.arange(n, dtype=np.float64)) + delta          # N-i+δ
    a = (i * rev) ** (-gamma)
    total = a.sum()
    if total > 0.0:
        a /= total
    return a


# ----------------------------------------------------------------------
def gradient_refine(a, deadline):
    """
    Projected gradient descent on the L2 error of the convolution
    to the flat target C = (sum a)² / (2·n‑1)  (C = 1/(2n‑1) after normalisation).
    """
    n = len(a)
    target_const = (a.sum() ** 2) / (2 * n - 1)

    step_factor = 0.1                # initial step‑size factor
    best_a = a.copy()
    best_score = fast_score(a)
    it = 0

    while time.time() < deadline:
        it += 1
        b = convolve_fft(a, a)
        err = b - target_const
        # gradient = 4·corr(err, a) (see derivation in the analysis)
        grad = 4.0 * np.correlate(err, a, mode='valid')
        gn = np.linalg.norm(grad)
        if gn == 0.0:
            break
        eta = step_factor / gn
        a = a - eta * grad
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= s
        cur = fast_score(a)
        if cur < best_score:
            best_score = cur
            best_a = a.copy()
            print(f'  L2‑grad iter {it} improved score {cur:.6f}')
            step_factor = max(step_factor * 0.95, 1e-4)

    return best_a

def subgradient_descent(a, deadline):
    """
    Projected sub‑gradient descent on the max‑convolution objective.
    The sub‑gradient is the sum of gradients of all (near‑)maximal
    convolution entries.
    """
    n = len(a)
    best_a = a.copy()
    best_score = fast_score(a)
    it = 0
    eta0 = 0.5

    while time.time() < deadline:
        it += 1
        b = convolve_fft(a, a)
        max_val = b.max()
        # indices that are (practically) maximal
        tol = max_val * 1e-12
        active = np.where(b >= max_val - tol)[0]

        # sub‑gradient: for each active k, add a_{k-j} (if defined) and a_j
        G = len(active) * a                     # the a_j part appears |S| times
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            G[idx] += a[k - idx]

        eta = eta0 / np.sqrt(it)
        a = a - eta * G
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        cur = fast_score(a)
        if cur < best_score:
            best_score = cur
            best_a = a.copy()
            print(f'  sub‑grad iter {it} improved score {cur:.6f}')

        # early stop if step becomes negligible
        if eta * np.linalg.norm(G) < 1e-12:
            break
