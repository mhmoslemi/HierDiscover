"""
Propose a candidate step‑function that improves on the Catalan construction.

Algorithm
---------
1. Initialise a length‑n probability vector `a` with the Catalan coefficients
   (they already give a value ≈ π/2 ≈ 1.5708 after the evaluation).

2. Optimise `a` on the probability simplex (non‑negative, sum = 1) with a
   projected gradient method.
   * The objective is a smooth approximation of the quantity
     `max_k (a*a)_k`:
       f(a) = (1/β)·log ∑_k exp(β·(a*a)_k) .
     For β → ∞ this converges to the actual max.
   * Its gradient w.r.t. `a` can be written using only convolutions:
       ∇_a f = 2·(a * w_rev)_{n‑1 … 2n‑2},
     where `w_k = exp(β·(a*a)_k) / Σ_l exp(β·(a*a)_l)` and `w_rev`
     is `w` reversed.
   * Convolutions are evaluated by FFT (O(N log N)) for speed.

3. After each gradient step we project back onto the simplex by the
   standard O(n log n) Euclidean projection.

4. β is increased slowly from a modest start (≈5) up to a larger bound
   (≈50) so that the optimisation first follows a smooth landscape and
   later focuses on the true maximum.

5. The best sequence seen during the run is kept and finally returned.
   The whole routine is limited to the supplied time budget (default
   1000 s).

The method can escape shallow local minima by occasional small random
perturbations of the current point and by restarting from the original
Catalan seed if no improvement is observed for a long time.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top level – required by the problem statement)
# ----------------------------------------------------------------------

def _catalan_sequence(n: int) -> np.ndarray:
    """
    Return the first *n* Catalan‑like coefficients a_i = C(2i,i)/4**i,
    normalised to have sum = 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    # recurrence a_{i+1} = a_i * (2*i+1)/(2*(i+1))
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * (i - 1) + 1.0) / (2.0 * i)
    a /= a.sum()
    return a


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ_i x_i = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Input must be 1‑D")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    # find the rho index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerically all entries are zero – return uniform point
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_convolve_full(a: np.ndarray, b: np.ndarray, size: int) -> np.ndarray:
    """
    Compute the (full) linear convolution of a and b using a real FFT of length `size`.
    `size` must be a power of two and at least len(a)+len(b)-1.
    Returns the first len(a)+len(b)-1 entries.
    """
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The algorithm runs for at most *budget_s* seconds
    and returns the best sequence found.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : int, optional
        Time budget in seconds (default: 1000).

    Returns
    -------
    list[float]
        The best sequence discovered.
    """
    # --------------------------------------------------------------
    # Set up reproducibility and time budget
    # --------------------------------------------------------------
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0  # small safety margin

    # --------------------------------------------------------------
    # Problem dimensions – trade‑off between quality and speed
    # --------------------------------------------------------------
    n = 2000                     # length of the sequence
    # initialise from the Catalan construction (already good)
    a = _catalan_sequence(n)    # guarantees sum(a)=1 and all a>0

    # keep best result seen
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # Pre‑compute FFT size (next power of two for the longest convolution)
    # --------------------------------------------------------------
    # longest convolution needed is a (len n) * w_rev (len 2n‑1) → length 3n‑2
    size = 1 << ((3 * n - 2).bit_length())   # power‑of‑two ≥ 3n‑2

    # --------------------------------------------------------------
    # Optimisation hyper‑parameters
    # --------------------------------------------------------------
    lr = 0.04                     # base step size for projected gradient
    beta_start = 5.0              # start of smoothing parameter
    beta_inc = 0.0015             # increase per iteration
    beta_max = 45.0               # maximal β (approaches true max)
    max_no_improve = 4000        # restart threshold
    restart_counter = 0
    iteration = 0
    no_improve = 0

    # --------------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        # ---- 1. Convolution a*a (full length 2n‑1) -----------------
        conv_aa = _fft_convolve_full(a, a, size)

        # ---- 2. Smooth max via soft‑max weighting -----------------
        beta = min(beta_max, beta_start + beta_inc * iteration)
        # avoid overflow – conv_aa values are tiny (≈0.3·10⁻³) for sum=1,
        # so plain exp is safe
        w = np.exp(beta * conv_aa)
        w /= w.sum()                # w_k ≥ 0, Σ w = 1

        # ---- 3. Gradient of the smoothed objective ---------------
        # w_rev is w reversed (necessary for the convolution formula)
        w_rev = w[::-1]
        conv_aw = _fft_convolve_full(a, w_rev, size)   # length 3n‑2
        # gradient formula: ∇_a f = 2·(a * w_rev)_{n‑1 … 2n‑2}
        grad = 2.0 * conv_aw[n - 1: 2 * n - 1]

        # ---- 4. Projected gradient step ---------------------------
        a_candidate = a - lr * grad
        a = _project_onto_simplex(a_candidate)

        # ---- 5. Evaluate and remember the best --------------------
        cur_score = evaluate_sequence(a.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{iteration:06d}] NEW BEST = {best_score:.6f} (β={beta:.2f})")
            no_improve = 0
        else:
            no_improve += 1

        # ---- 6. Adaptive tricks ------------------------------------
        # decay learning rate slowly
        if iteration % 800 == 0 and iteration > 0:
            lr *= 0.94

        # occasional tiny random perturbation to escape flat regions
        if iteration % 2000 == 0 and iteration > 0:
            a += np.random.normal(scale=0.001, size=n)
            a = _project_onto_simplex(a)

        # restart if stagnated for a long time
        if no_improve >= max_no_improve:
            restart_counter += 1
            print(f"[{iteration:06d}] restart #{restart_counter} (stagnation)")
            a = _catalan_sequence(n)          # fresh start from the known good seed
            lr = 0.04
            no_improve = 0

        iteration += 1

    # --------------------------------------------------------------
    # Final report
    # --------------------------------------------------------------
    print(f"[done] iterations={iteration}, best score={best_score:.6f}")
    return best_seq.tolist()
