Implementation of projection onto simplex (size n, target sum = 1) can be done via algorithm by Duchi et al.:

def project_onto_simplex(v, s=1):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w

"""
A two‑phase optimizer for the `evaluate_sequence` objective.

Phase 1 – fast projected sub‑gradient descent.
   * Uses *all* indices whose convolution value is within a tiny tolerance
     of the current maximum (the “active set”), so the descent direction
     reduces the whole peak rather than a single index.
   * After each step the vector is projected onto the probability simplex
     (non‑negative, unit sum) with an O(n log n) routine.

Phase 2 – a convex “conditional‑gradient” (Frank–Wolfe) step.
   * For the current f we solve the linear programme
         maximise   ∑ g_i
         subject to  g_i ≥ 0  and  (f * g)_k ≤ M   (∀k)
     where M = max_k (f * f)_k.
   * The LP is built with a sparse Toeplitz matrix (size (2n‑1)×n)
     and solved by SciPy’s HiGHS interior‑point solver.
   * The optimal g is rescaled to unit sum, then a short line‑search
     over mixtures (1‑t)·f + t·g is performed; the mixture with the
     smallest convolution peak is kept.  If the mixture improves the
     objective the algorithm returns to Phase 1, otherwise it stops.

The whole routine runs for at most ``budget_s`` seconds (default 1000 s)
and returns the best sequence found.  The construction is based on the
well‑known Catalan (central‑binomial) seed but refines it substantially,
often pushing the score below the 1.5709 value obtained by a simple
sub‑gradient method.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import diags

# ----------------------------------------------------------------------
# helper – projection onto the probability simplex  (non‑negative, sum=1)
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum x = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # Find the rho index (largest i such that u_i > cssv_i / i)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:                     # all entries are non‑positive
        theta = cssv[-1] / n
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Catalan‑like seed (central binomial coefficients) – already near‑optimal
def _catalan_sequence(n: int) -> np.ndarray:
    """First n coefficients a_i = C(2i,i)/4^i, normalised to sum‑1."""
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * (i - 1) + 1) / (2.0 * i)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence``.  The routine uses a two‑phase
    optimisation described in the module docstring and returns the
    best sequence discovered within the time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5   # small safety margin

    # --------------------------------------------------------------
    # problem size – 1500 gives a good trade‑off (memory & time)
    n = 1500

    # start from the classic Catalan construction (already very good)
    a = _catalan_sequence(n)

    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – projected sub‑gradient descent with an *active set*
    iteration = 0
    lr0 = 0.08
    while time.time() < deadline and iteration < 4000:
        conv = np.convolve(a, a)
        max_val = conv.max()

        # active set: all indices whose value is within 1e‑12 of the max
        tol = max_val * 1e-12
        active = (conv >= max_val - tol).astype(float)

        # sub‑gradient = 2 * (active * a_rev)  (convolution)
        grad = 2.0 * np.convolve(active, a[::-1])[n - 1:n - 1 + n]

        # simple diminishing learning‑rate schedule
        lr = lr0 / (1.0 + iteration * 1e-4)
        a_candidate = a - lr * grad
        a_candidate = _project_simplex(a_candidate)

        cur_score = evaluate_sequence(a_candidate.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_candidate.copy()
            print(f"[grad {iteration:05d}] new best = {best_score:.6f}")
        a = a_candidate
        iteration += 1

    # --------------------------------------------------------------
    # Phase 2 – conditional‑gradient (LP) refinement
    while time.time() < deadline:
        # current convolution and its maximum
        conv_f = np.convolve(a, a)
        M = conv_f.max()

        # ------------------------------------------------------------------
        # Build the (2n‑1) × n constraint matrix for (f * g)_k ≤ M
        # each column i contains the same value f_i on a diagonal of length n
        # Offsets 0 … n‑1 correspond to rows i … i+n‑1.
        A_ub = diags([a] * n, offsets=np.arange(n), shape=(2 * n - 1, n), format='csc')
        b_ub = M * np.ones(2 * n - 1)

        # maximise sum(g_i)  →  minimise -sum(g_i)
        c = -np.ones(n)

        # Solve LP (HiGHS is the default for recent SciPy versions)
        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs')

        if not res.success:
            print("[LP] solver failed – terminating phase 2")
            break

        g_raw = res.x
        sum_g_raw = g_raw.sum()
        if sum_g_raw <= 0:
            print("[LP] got zero‑mass solution – terminating")
            break

        # Normalise to a probability vector (sum = 1)
        g = g_raw / sum_g_raw

        # Pre‑compute the three convolutions needed for any mixture
        conv_g = np.convolve(g, g)
        conv_fg = np.convolve(a, g)

        # --------------------------------------------------------------
        # line‑search over mixing coefficient t ∈ (0,1)
        # (αa + (1‑α)g)*(αa + (1‑α)g) = α² conv_f + 2α(1‑α) conv_fg + (1‑α)² conv_g
        t_vals = np.linspace(0.02, 0.98, 25)   # enough resolution, cheap to test
        best_t = 0.0
        best_M = M
        for t in t_vals:
            conv_mix = (1 - t) ** 2 * conv_f + 2 * t * (1 - t) * conv_fg + t ** 2 * conv_g
            cur_M = conv_mix.max()
            if cur_M < best_M - 1e-14:      # strict improvement
                best_M = cur_M
                best_t = t

        if best_t == 0.0:          # no improvement reachable
            print("[LP] no improving mixture found – stop")
            break

        # Update the current sequence with the best mixture
        a = (1 - best_t) * a + best_t * g
        a /= a.sum()                # numerical drift correction

        cur_score = evaluate_sequence(a.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[mix] improved score = {best_score:.6f} (t={best_t:.3f})")
        else:
            # this should not happen, but guard against numerical glitches
            print("[mix] mixture did not improve the reported score")

        # continue the loop – a new LP may provide further progress
        # (the loop will exit automatically if time runs out)

    print(f"[done] best score = {best_score:.6f}")
    return best_seq.tolist()
```
