def gradient_refine(a_init, n, budget_s, cur_best_score):
    """Projected gradient descent with smooth max approximation."""
    start = time.time()
    a = a_init.copy()
    a = a / np.sum(a)
    best_seq = a.copy()
    best_score = cur_best_score
    # hyperparameters
    step = 0.1
    min_step = 1e-6
    beta = 5.0
    max_beta = 200.0
    beta_inc_iter = 3000
    beta_mult = 1.07
    # iterate
    it = 0
    while True:
        if time.time() - start > budget_s:
            break
        # compute convolution and max
        b = np.convolve(a, a)
        max_b = b.max()
        cur_score = 2.0 * n * max_b
        # compute smooth weights
        # shift for stability
        shift = max_b
        exp_vals = np.exp(beta * (b - shift))
        w = exp_vals / exp_vals.sum()
        # gradient via correlation
        corr = np.correlate(w, a, mode='full')
        grad = 2.0 * corr[n-1 : n-1 + n]
        # try a step
        step_try = step
        while step_try >= min_step:
            a_new = _project_to_simplex(a - step_try * grad)
            b_new = np.convolve(a_new, a_new)
            new_score = 2.0 * n * b_new.max()
            if new_score <= cur_score:
                a = a_new
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_seq = a_new.copy()
                    print(f"[grad] it={it}, score={new_score:.6f}, step={step_try:.5f}, beta={beta:.2f}")
                # possibly increase step slightly
                step = step_try * 1.02
                break
            else:
                step_try *= 0.5
        else:
            # could not improve, shrink step
            step = max(step * 0.7, min_step)
        # update beta occasionally
        if (it+1) % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * beta_mult, max_beta)
        it += 1
    return best_seq, best_score

"""
Propose a step‑function (list of non‑negative heights) that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm works in several phases:

1. Initialise from the provided construction (``height_sequence_1``) and
   evaluate a uniform fallback.
2. Exhaustive + random search over the 2‑parameter power‑law family
   a_i ∝ (i+shift)^{-alpha}.  The best member is kept as a warm start.
3. Projected gradient descent on the simplex using a smooth approximation
   of the ``max`` (soft‑max with a temperature β).  The gradient of the
   smoothed objective is cheap to compute via a correlation operation.
   The iterate is projected onto the probability simplex after each step.
   β is increased slowly (continuation) so that the optimisation first
   sees a smooth landscape and then focuses on the true max.
4. A final cheap hill‑climbing phase that moves a small amount of mass
   between two random positions, keeping only improving moves.

All operations respect the non‑negativity and normalisation constraints,
so the score can be evaluated directly as ``2*n*max(conv(a,a))`` without
calling ``evaluate_sequence`` (the wrapper does the same checks).  The
whole routine respects the given time budget (default 950 s) and prints
intermediate progress.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x >= 0, sum(x)=1}
# ----------------------------------------------------------------------
def _project_to_simplex(v):
    """Efficient projection of a vector onto the probability simplex."""
    n = len(v)
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero – return uniform
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


# ----------------------------------------------------------------------
# Gradient‑descent refinement (smooth max of the convolution)
# ----------------------------------------------------------------------
def _gradient_refine(a_init, n, time_budget, cur_best_score):
    """
    Projected gradient descent on the simplex using a smoothed max.
    Returns the best sequence found and its (exact) score.
    """
    start = time.time()
    a = a_init.copy()
    a = a / np.sum(a)                     # ensure normalisation
    best_seq = a.copy()
    best_score = cur_best_score

    # hyper‑parameters
    step = 0.12                # initial step size
    min_step = 1e-7
    beta = 5.0                 # temperature for the soft‑max
    max_beta = 250.0
    beta_inc_iter = 3000       # how often β is increased
    beta_mult = 1.07           # multiplicative increase
    iteration = 0

    while True:
        if time.time() - start > time_budget:
            break

        # ----- current objective -----
        b = np.convolve(a, a)                     # linear convolution
        max_b = b.max()
        cur_score = 2.0 * n * max_b

        # ----- smooth weights (soft‑max) -----
        # shift by the max to keep numbers well‑scaled
        shift = max_b
        exp_vals = np.exp(beta * (b - shift))
        w = exp_vals / exp_vals.sum()             # weights sum to 1

        # ----- gradient of the smoothed objective -----
        # grad_j = 2 * Σ_k w_k * a_{k-j}
        corr = np.correlate(w, a, mode='full')
        grad = 2.0 * corr[n - 1: n - 1 + n]       # length‑n gradient

        # ----- line‑search on the simplex -----
        step_try = step
        while step_try >= min_step:
            a_new = _project_to_simplex(a - step_try * grad)
            b_new = np.convolve(a_new, a_new)
            new_score = 2.0 * n * b_new.max()
            if new_score <= cur_score:            # accept any descent
                a = a_new
                cur_score = new_score
                # enlarge step a little after a successful move
                step = step_try * 1.02
                break
            step_try *= 0.5                        # backtrack
        else:
            # no improvement – shrink the step
            step = max(step * 0.7, min_step)

        # ----- record global best -----
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[grad] it={iteration:06d}, score={best_score:.6f}, "
                  f"step={step_try:.5f}, beta={beta:.2f}")

        # ----- increase β (continuation) -----
        if (iteration + 1) % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * beta_mult, max_beta)

        iteration += 1

    return best_seq, best_score


# ----------------------------------------------------------------------
# Tiny hill‑climbing (mass transfer between two positions)
# ----------------------------------------------------------------------
def _hill_climb(seq_init, n, time_budget, cur_best_score):
    """
    Simple random local search that moves a small fraction of mass from one
    index to another while staying on the simplex.
    """
    start = time.time()
    cur = seq_init.copy()
    cur_score = 2.0 * n * np.max(np.convolve(cur, cur))
    best_seq = cur.copy()
    best_score = cur_best_score

    max_frac = 0.02   # up‑to 2 % of the source entry per move

    while time.time() - start < time_budget:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        step = cur[i] * (np.random.rand() * max_frac)
        if step <= 0:
            continue
        new = cur.copy()
        new[i] -= step
        new[j] += step
        new = _project_to_simplex(new)            # keep exact sum = 1
        new_score = 2.0 * n * np.max(np.convolve(new, new))
        if new_score + 1e-12 < cur_score:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = new.copy()
                print(f"[hill] improved score={best_score:.6f}")

    return best_seq, best_score

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed=123, budget_s=950, **kwargs):
    """

    smooth‑max gradient descent and a final hill‑climbing phase.
    """
    np.random.seed(seed)
    start = time.time()

    # --------------------------------------------------------------
    # Determine the problem size (length of the sequence)
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)   # provided by the competition environment
        baseline = np.array(height_sequence_1, dtype=float)
        baseline = baseline / baseline.sum()
    except Exception:
        # fallback: a moderate length (the description mentions 2244)
        n = 2244
        baseline = np.ones(n, dtype=float) / n

    # --------------------------------------------------------------
    # Initial baseline (uniform) and the supplied construction
    # --------------------------------------------------------------
    uniform = np.ones(n, dtype=float) / n
    best_seq = uniform.copy()
    best_score = 2.0 * n * np.max(np.convolve(best_seq, best_seq))

    # evaluate the supplied construction if it exists
    try:
        score0 = 2.0 * n * np.max(np.convolve(baseline, baseline))
        if score0 < best_score:
            best_seq = baseline.copy()
            best_score = score0
    except Exception:
        pass

    print(f"[init] n={n}, baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # 1. Grid search over power‑law families a_i = (i+shift)^{-alpha}
    # --------------------------------------------------------------
    idx = np.arange(n, dtype=float)

    alphas = np.linspace(0.35, 0.90, 111)   # fine grid
    shifts = np.linspace(0.5, 3.0, 11)     # a few shift values

    for alpha in alphas:
        if time.time() - start > budget_s - 40:
            break
        for shift in shifts:
            raw = np.power(idx + shift, -alpha)
            a = raw / raw.sum()
            score = 2.0 * n * np.max(np.convolve(a, a))
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[grid] alpha={alpha:.4f}, shift={shift:.2f}, "
                      f"score={score:.6f}")

    # --------------------------------------------------------------
    # 2. Randomised power‑law search (to escape the deterministic grid)
    # --------------------------------------------------------------
    max_random = 2000
    for _ in range(max_random):

        raw = np.power(idx + shift, -alpha)

        score = 2.0 * n * np.max(np.convolve(a, a))
        if score < best_score:
