"""
propose_candidate
=================

Search strategy for a low‑value step function (non‑negative list) that
minimises

    2 * n * max(conv(a,a)) / (sum(a)**2)

(the evaluator normalises the L¹‑norm, therefore the score is simply
`max(conv(a,a))`).  

The algorithm consists of two stages:

1. **Broad random sweep** – a large number of analytically‑parameterised
   symmetric families (power‑law, exponential, linear, polynomial,
   two‑level edge/inner blocks, edge‑biased Dirichlet) are sampled.
   Each raw vector is scaled to the required L¹‑norm
   `sqrt(2*n)` and evaluated.  The best candidate is retained.

2. **Guided hill‑climbing** – starting from the best candidate, we iteratively
   reduce the peak of the convolution.  At each step we locate the index
   `k_max` where the convolution attains its maximum.  The contribution of
   each coefficient `a_i` to that peak is `2*a_i*a_{k_max-i}` (zero when the
   partner index is out of range).  We then transfer a small fraction of the
   mass from a coefficient with a large contribution to one with a small
   contribution, always keeping the vector non‑negative and the L¹‑norm
   fixed (by a direct transfer, no projection needed).  The step size
   decays over time and a tiny simulated‑annealing acceptance rule helps
   to escape shallow local minima.

Both phases respect the 1000 s time budget and print progress information.
The final list is returned ready for `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_to_simplex(x: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Project a vector onto the simplex {y >= 0, sum(y)=target_sum}.
    O(n log n) algorithm (Wang & Carreira‑Perpiñán, 2013).
    """
    n = x.size
    if target_sum <= 0:
        return np.zeros_like(x)

    # sort descending
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, n + 1)
    # find the largest rho where u_i > cssv_i / i
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(x - theta, 0.0)
    return w


def _evaluate(seq):
    """Thin wrapper around the supplied ``evaluate_sequence``."""
    return evaluate_sequence(seq)


def _scale_to_sum(raw: np.ndarray, target_sum: float) -> np.ndarray:
    """Scale a non‑negative vector so that its L¹‑norm equals ``target_sum``."""
    raw = np.asarray(raw, dtype=float)
    s = raw.sum()
    if s == 0:
        return np.full_like(raw, target_sum / raw.size)
    return raw * (target_sum / s)


def _symmetric_from_half(half: np.ndarray, n: int) -> np.ndarray:
    """Mirror a half‑vector to a full symmetric vector of length ``n``."""
    if n % 2 == 0:
        # even length – just concatenate the reverse
        return np.concatenate([half, half[::-1]])
    else:
        # odd length – omit the duplicated centre element
        return np.concatenate([half, half[-2::-1]])


# ----------------------------------------------------------------------
# Parameterised families (all generate a non‑negative raw vector)
# ----------------------------------------------------------------------


def sym_power(n: int, p: float) -> np.ndarray:
    """a_i = (i+1)^{-p}, mirrored, decreasing from the edges to the centre."""
    m = n // 2 if n % 2 == 0 else (n + 1) // 2
    half = (np.arange(1, m + 1, dtype=float)) ** (-p)
    return _symmetric_from_half(half, n)


def sym_exp(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam * i), mirrored."""
    m = n // 2 if n % 2 == 0 else (n + 1) // 2
    half = np.exp(-lam * np.arange(m, dtype=float))
    return _symmetric_from_half(half, n)


def sym_linear(n: int) -> np.ndarray:
    """Linear decay from the edges (1) to the centre (0)."""
    m = n // 2 if n % 2 == 0 else (n + 1) // 2
    half = np.linspace(1.0, 0.0, m, dtype=float)
    return _symmetric_from_half(half, n)


def sym_poly(n: int, alpha: float) -> np.ndarray:
    """a_i = (1 - i/(m-1))^{alpha}, mirrored."""
    m = n // 2 if n % 2 == 0 else (n + 1) // 2
    if m == 1:
        half = np.ones(1, dtype=float)
    else:
        half = (1.0 - np.arange(m, dtype=float) / (m - 1)) ** alpha
    return _symmetric_from_half(half, n)


def two_level_shape(n: int, w_frac: float, inner_ratio: float) -> np.ndarray:
    """
    Two‑level shape: wide high‑value edges of width ``w`` and a low‑value interior.
    ``inner_ratio`` is the interior height as a fraction of the edge height.
    """
    w = max(1, int(round(w_frac * n)))
    if 2 * w >= n:               # ensure a non‑empty interior
        w = n // 2 - 1
        if w < 1:
            w = 1
    raw = np.full(n, inner_ratio, dtype=float)
    raw[:w] = 1.0
    raw[-w:] = 1.0
    return raw


def edge_biased_shape(n: int) -> np.ndarray:
    """
    Draw a Dirichlet‑like vector and re‑order it so the largest entries sit at
    the two ends of the sequence (largest distance from the centre).
    """
    vals = np.random.gamma(shape=0.7, scale=1.0, size=n)
    vals_sorted = np.sort(vals)[::-1]          # descending
    dist = np.minimum(np.arange(n), n - 1 - np.arange(n))
    order = np.argsort(-dist)                  # edges first
    raw = np.empty(n, dtype=float)
    raw[order] = vals_sorted
    return raw


def random_dirichlet(n: int) -> np.ndarray:
    """Simple Dirichlet‑like vector (Gamma(0.7,1))."""
    return np.random.gamma(shape=0.7, scale=1.0, size=n)


# ----------------------------------------------------------------------
# Main candidate generator
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list:
    """
    Return a list of non‑negative heights (the step function) that
    hopefully yields a low evaluation score.  The function respects the
    overall time budget ``budget_s`` (seconds).
    """
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
        init_seq = list(height_sequence_1)
        best_score = _evaluate(init_seq)
        best_seq = init_seq
        print(f"[init] using supplied height_sequence_1, n={n}, score={best_score:.6f}")
    else:
        # moderate default size – fast enough for many evaluations
        n = 6000
        target_sum = math.sqrt(2.0 * n)
        best_seq = [target_sum / n] * n
        best_score = _evaluate(best_seq)
        print(f"[init] uniform start, n={n}, score={best_score:.6f}")

    target_sum = math.sqrt(2.0 * n)          # fixed L¹‑norm for all candidates

    start_time = time.time()
    stop_time = start_time + budget_s

    # ------------------------------------------------------------------
    # Phase 1 – broad random sweep over parameterised families
    # ------------------------------------------------------------------
    sweep_deadline = stop_time - 30.0          # keep about 30 s for refinement
    it = 0
    families = [
        "sym_power",
        "sym_exp",
        "sym_linear",
        "sym_poly",
        "two_level",
        "edge_bias",
        "rand_dirichlet",
    ]

    while time.time() < sweep_deadline:
        it += 1
        fam = np.random.choice(families)

        # ----- generate a raw (pre‑scaled) vector --------------------
        if fam == "sym_power":
            p = np.random.uniform(0.05, 8.0)
            raw = sym_power(n, p)

        elif fam == "sym_exp":
            lam = np.random.uniform(0.001, 3.0)
            raw = sym_exp(n, lam)

        elif fam == "sym_linear":
            raw = sym_linear(n)

        elif fam == "sym_poly":
            alpha = np.random.uniform(0.1, 8.0)
            raw = sym_poly(n, alpha)

        elif fam == "two_level":
            w_frac = np.random.uniform(0.05, 0.45)          # edge‑block width proportion
            inner = np.random.uniform(0.0, 1.0)            # interior height ratio
            raw = two_level_shape(n, w_frac, inner)

        elif fam == "edge_bias":
            raw = edge_biased_shape(n)

        else:   # "rand_dirichlet"
            raw = random_dirichlet(n)

        # clip negative values (should not happen) and scale
        raw = np.maximum(raw, 0.0)
        cand = _scale_to_sum(raw, target_sum).tolist()
        score = _evaluate(cand)

        if score < best_score:
            best_score = score
            best_seq = cand
            print(f"[sweep {it:06d}] new best {best_score:.6f} via {fam}")

        # occasional tiny perturbation of the current best
        if it % 500 == 0 and np.random.rand() < 0.2:
            tmp = np.array(best_seq, copy=True)
            i, j = np.random.choice(n, size=2, replace=False)
            delta = tmp[i] * np.random.uniform(0.0, 0.01)
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_to_simplex(tmp, target_sum)
            cand2 = tmp.tolist()
            score2 = _evaluate(cand2)
            if score2 < best_score:
                best_score = score2
                best_seq = cand2
                print(f"[perturb {it}] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – guided hill‑climbing / sub‑gradient descent
    # ------------------------------------------------------------------
    seq = np.array(best_seq, copy=True)
    current_score = best_score
    step_frac = 0.05                     # initial fraction of mass to move
    step_decay = 0.9995
    temperature = 0.001
    it_refine = 0

    while time.time() < stop_time - 1.0:   # leave a tiny safety margin
        # convolution and the index of the current maximum
        conv = np.convolve(seq, seq)
        k_max = int(np.argmax(conv))

        # contribution of each coefficient to that peak:
        #   g_i = 2 * a_i * a_{k_max-i}   (0 when partner out of range)
        idx = np.arange(n)
        partner = k_max - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n, dtype=float)
        contrib[mask] = seq[idx[mask]] * seq[partner[mask]] * 2.0

        i_max = int(np.argmax(contrib))
        i_min = int(np.argmin(contrib))

        # mass transfer from a large‑contribution index to a small one
        delta = seq[i_max] * step_frac
        if delta <= 0:
            # fallback to a small random transfer
            i_max, i_min = np.random.choice(n, 2, replace=False)
            delta = seq[i_max] * step_frac
        new_seq = seq.copy()
        new_seq[i_max] -= delta
        new_seq[i_min] += delta

        if new_seq[i_max] < -1e-12:
            # illegal move – skip
            continue

        new_score = _evaluate(new_seq.tolist())
        it_refine += 1

        if new_score < best_score:
            best_score = new_score
            best_seq = new_seq.tolist()
            seq = new_seq
            current_score = new_score
            print(f"[refine {it_refine}] improved best to {best_score:.6f}")
        elif new_score < current_score:
            # accept as new current point even if not global best
            seq = new_seq
            current_score = new_score
        else:
            # occasional uphill acceptance (simulated annealing)
            prob = math.exp(-(new_score - current_score) / (temperature + 1e-12))
            if np.random.rand() < prob:
                seq = new_seq
                current_score = new_score

        # decay step size and temperature
        step_frac *= step_decay
        temperature *= 0.999

        # occasional random jitter to diversify search
        if it_refine % 1000 == 0:
            i, j = np.random.choice(n, 2, replace=False)
            delta2 = seq[i] * np.random.uniform(0.0, step_frac)
            if delta2 > 0 and seq[i] - delta2 >= 0:
                tmp_seq = seq.copy()
                tmp_seq[i] -= delta2
                tmp_seq[j] += delta2
                tmp_score = _evaluate(tmp_seq.tolist())
                if tmp_score < best_score:
                    best_score = tmp_score
                    best_seq = tmp_seq.tolist()
                    seq = tmp_seq
                    current_score = tmp_score
                    print(f"[jitter {it_refine}] improved best to {best_score:.6f}")

    print(f"[final] best score achieved: {best_score:.6f}")
    return best_seq
