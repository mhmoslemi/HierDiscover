   and evaluated by a tiny wrapper around ``evaluate_sequence`` that computes
   ``2·n·max(conv(a,a))`` directly (much faster than calling the full evaluator).

2. **Projected gradient descent on the L∞‑norm of the autocorrelation** –
   at each iteration the index *k* where the convolution attains its maximum
   is identified, the sub‑gradient of that maximal term is formed
   (g_i = 2·a_{k‑i} whenever the index exists) and a step in the opposite
   direction is taken.  The updated vector is projected back onto the
   probability simplex (non‑negative, unit sum).  The step size decays
   slowly; any improvement updates the global best.

3. **Simple stochastic hill‑climbing** – a classic “move a small amount of
   mass from i to j” move is performed many times (mass is transferred as a
   fraction of the source coefficient).  After each move the vector is again
   projected onto the simplex and kept only if the score improves.

All three stages respect the overall time budget (default ≈ 950 s).  The
function prints occasional progress reports and finally returns the best
sequence found as a plain Python list.
"""

import time
import numpy as np

# -------------------------------------------------
# Helper: projection onto the probability simplex
# -------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto {x >= 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero – return uniform distribution
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# -------------------------------------------------
# Fast evaluation (scale‑invariant, avoids the heavy checks)
# -------------------------------------------------
def _fast_score(a: np.ndarray, n: int) -> float:
    """
    a must already be normalised (sum == 1).  Returns
    2 * n * max(conv(a, a)).
    """
    conv = np.convolve(a, a)
    return float(2.0 * n * np.max(conv))


# -------------------------------------------------
# Family generators (return *unnormalised* vectors)
# -------------------------------------------------
def _gen_power_law(idx: np.ndarray, alpha: float, shift: float) -> np.ndarray:
    return (idx + shift) ** (-alpha)


def _gen_exponential(idx: np.ndarray, beta: float) -> np.ndarray:
    return np.exp(-beta * idx)


def _gen_stretched_exp(idx: np.ndarray, scale: float, gamma: float) -> np.ndarray:
    return np.exp(- (idx / scale) ** gamma)


def _gen_power_exp(idx: np.ndarray, alpha: float, shift: float, beta: float) -> np.ndarray:
    return (idx + shift) ** (-alpha) * np.exp(-beta * idx)


def _gen_two_power(idx: np.ndarray,
                   alpha1: float, shift1: float,
                   alpha2: float, shift2: float,
                   w: float) -> np.ndarray:
    p1 = (idx + shift1) ** (-alpha1)
    p2 = (idx + shift2) ** (-alpha2)
    return w * p1 + (1.0 - w) * p2


# -------------------------------------------------
# Main entry point
# -------------------------------------------------
def propose_candidate(seed: int = 123,
                      budget_s: float = 950,
                      **kwargs) -> list:
    """
    Search for a non‑negative step‑function that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

    The routine respects the time budget and returns the best sequence found.
    """
    np.random.seed(seed)

    start = time.time()
    # -------------------------------------------------
    # Determine the problem size (length of the step function)
    # -------------------------------------------------
    try:
        n = len(height_sequence_1)  # provided by the harness
    except Exception:
        n = 2244  # fallback, the length used in the description

    idx = np.arange(n, dtype=float)

    # -------------------------------------------------
    # Baseline – keep the supplied construction as a fallback
    # -------------------------------------------------
    best_seq = np.array(height_sequence_1, dtype=float)
    best_seq = _project_to_simplex(best_seq)  # make absolutely sure
    best_score = _fast_score(best_seq, n)

    # -------------------------------------------------
    # 1) GRID SEARCH over several low‑dimensional families
    # -------------------------------------------------
    print(f"[init] n={n}, baseline score={best_score:.6f}")

    # ----- Power‑law -------------------------------------------------
    alphas = np.linspace(0.35, 0.95, 61)          # step ≈0.01
    shifts = np.linspace(0.5, 5.0, 10)            # 0.5 … 5.0
    for alpha in alphas:
        if time.time() - start > budget_s - 5:
            break
        for shift in shifts:
            raw = _gen_power_law(idx, alpha, shift)
            s = raw.sum()
            if s == 0:
                continue
            a = raw / s
            score = _fast_score(a, n)
            if score < best_score:
                best_score, best_seq = score, a.copy()
                print(f"[grid-power] α={alpha:.3f}, shift={shift:.2f} → {score:.6f}")

    # ----- Exponential -------------------------------------------------
    betas = np.logspace(np.log10(5e-5), np.log10(1e-2), 15)
    for beta in betas:
        if time.time() - start > budget_s - 5:
            break
        raw = _gen_exponential(idx, beta)
        a = raw / raw.sum()
        score = _fast_score(a, n)
        if score < best_score:
            best_score, best_seq = score, a.copy()
            print(f"[grid-exp] β={beta:.6f} → {score:.6f}")

    # ----- Stretched exponential ----------------------------------------
    scales = np.logspace(np.log10(n / 200), np.log10(n), 8)
    gammas = np.linspace(0.3, 2.0, 10)
    for scale in scales:
        for gamma in gammas:
            if time.time() - start > budget_s - 5:
                break
            raw = _gen_stretched_exp(idx, scale, gamma)
            a = raw / raw.sum()
            score = _fast_score(a, n)
            if score < best_score:
                best_score, best_seq = score, a.copy()
                print(f"[grid-stretch] scale={scale:.0f}, γ={gamma:.2f} → {score:.6f}")

    # ----- Power‑law × Exponential ---------------------------------------
    alphas2 = np.linspace(0.45, 0.80, 8)
    shifts2 = np.linspace(0.5, 3.0, 6)
    betas2 = np.linspace(0.0, 0.003, 6)  # β=0 gives back pure power‑law
    for alpha in alphas2:
        for shift in shifts2:
            for beta in betas2:
                if time.time() - start > budget_s - 5:
                    break
                raw = _gen_power_exp(idx, alpha, shift, beta)
                a = raw / raw.sum()
                score = _fast_score(a, n)
                if score < best_score:
                    best_score, best_seq = score, a.copy()
                    print(f"[grid-powexp] α={alpha:.3f}, shift={shift:.2f}, β={beta:.6f} → {score:.6f}")

    # ----- Two‑component power‑law mix ------------------------------------
    ws = [0.3, 0.5, 0.7]
    alphas1 = [0.4, 0.5]
    alphas2 = [0.6, 0.8]
    shift_common = 1.0
    for w in ws:
        for a1 in alphas1:
            for a2 in alphas2:
                if time.time() - start > budget_s - 5:
                    break
                raw = _gen_two_power(idx, a1, shift_common, a2, shift_common, w)
                a = raw / raw.sum()
                score = _fast_score(a, n)
                if score < best_score:
                    best_score, best_seq = score, a.copy()
                    print(f"[grid-mix] w={w:.2f}, α1={a1:.2f}, α2={a2:.2f} → {score:.6f}")

    print(f"[grid] best after grid search: {best_score:.6f}")

    # -------------------------------------------------
    # 2) PROJECTED GRADIENT DESCENT on the max‑convolution term
    # -------------------------------------------------
    cur = best_seq.copy()
    cur_score = best_score
    iter_gd = 0
    max_gd_iter = 30000
    # simple temperature schedule for occasional uphill moves
    while time.time() - start < budget_s - 5 and iter_gd < max_gd_iter:
        iter_gd += 1
        conv = np.convolve(cur, cur)
        k = int(np.argmax(conv))

        # gradient of the active convolution term
        j = k - np.arange(n, dtype=int)
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(cur)
        grad[mask] = 2.0 * cur[j[mask]]

        # classic sub‑gradient step (move opposite to gradient)
        eta = 0.03 / (1.0 + iter_gd / 2000.0)   # decaying step size
        cand = cur - eta * grad
        cand = _project_to_simplex(cand)

        cand_score = _fast_score(cand, n)
        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            if cand_score < best_score:
                best_score, best_seq = cand_score, cand.copy()
                print(f"[gd] iter={iter_gd}, improved to {cand_score:.6f}")
        else:
            # occasional uphill acceptance (simulated annealing flavour)
            # temperature drops linearly with remaining time
            remaining = budget_s - (time.time() - start)
            temp = max(1e-4, remaining / budget_s) * 0.02
            if np.random.rand() < np.exp((cur_score - cand_score) / temp):
                cur, cur_score = cand, cand_score

    print(f"[gd] finished after {iter_gd} iterations, best score={best_score:.6f}")

    # -------------------------------------------------
    # 3) SIMPLE STOCHASTIC HILL‑CLIMBING (mass transfers)
    # -------------------------------------------------
    cur = best_seq.copy()
    cur_score = best_score
    iter_hc = 0
    max_hc_iter = 25000
    while time.time() - start < budget_s - 5 and iter_hc < max_hc_iter:
        iter_hc += 1
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # move up to 2 % of the source mass
        step = cur[i] * (np.random.rand() * 0.02)
        if step <= 0 or cur[j] + step > 1.0:
            continue
        new = cur.copy()
        new[i] -= step
        new[j] += step
        new = _project_to_simplex(new)
        new_score = _fast_score(new, n)
        if new_score < cur_score:
            cur, cur_score = new, new_score
            if new_score < best_score:
                best_score, best_seq = new_score, new.copy()
                print(f"[hc] iter={iter_hc}, improved to {new_score:.6f}")

    print(f"[final] best score found: {best_score:.6f}")
    # Return as a plain Python list (the harness expects a list)
    return best_seq.tolist()
