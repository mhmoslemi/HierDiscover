    rng = np.random.default_rng(seed)
    start = time.time()

    # Determine length

        n = len(height_sequence_1)

        best_score = evaluate_sequence(best_seq.tolist())

    print(f"[init] n={n}, baseline={best_score:.6f}")

    idx = np.arange(n, dtype=float)  # i

    def evaluate_raw(vec):
        # vec is numpy array, possibly non-normalized
        total = vec.sum()
        if total <= 0:
            return float('inf')
        norm = vec / total
        return evaluate_sequence(norm.tolist()), norm.tolist()

    # 1. Grid search power-law
    alphas = np.linspace(0.30, 1.0, 71)
    shifts = np.linspace(0.01, 3.0, 30)
    for alpha in alphas:
        if time.time() - start > budget_s - 5:
            break
        for shift in shifts:
            raw = np.power(idx + shift, -alpha)
            score, seq = evaluate_raw(raw)
            if score < best_score:
                best_score = score
                best_seq = seq
                print(f"[grid] alpha={alpha:.4f} shift={shift:.4f} score={score:.6f}")

    # 2. Random power-law
    random_iters = 3000
    for _ in range(random_iters):
        if time.time() - start > budget_s - 5:
            break
        alpha = rng.uniform(0.2, 1.2)
        shift = rng.uniform(0.0, 5.0)
        raw = np.power(idx + shift, -alpha)
        score, seq = evaluate_raw(raw)
        if score < best_score:
            best_score = score
            best_seq = seq
            print(f"[rand] alpha={alpha:.4f} shift={shift:.4f} score={score:.6f}")

    # 3. Random exponential
    for _ in range(1000):
        if time.time() - start > budget_s - 5:
            break
        beta = rng.uniform(1e-5, 0.1)
        raw = np.exp(-beta * idx)
        score, seq = evaluate_raw(raw)
        if score < best_score:
            best_score = score
            best_seq = seq
            print(f"[exp] beta={beta:.6f} score={score:.6f}")

    # 4. Random mixture of two power laws
    for _ in range(2000):
        if time.time() - start > budget_s - 5:
            break
        alpha1 = rng.uniform(0.2, 1.2)
        shift1 = rng.uniform(0.0, 5.0)
        alpha2 = rng.uniform(0.2, 1.2)
        shift2 = rng.uniform(0.0, 5.0)
        p = rng.random()
        raw = p * np.power(idx + shift1, -alpha1) + (1-p) * np.power(idx + shift2, -alpha2)
        score, seq = evaluate_raw(raw)
        if score < best_score:

            best_seq = seq
            print(f"[mix] score={score:.6f}")

    print(f"[search] best after exploratory search: score={best_score:.6f}")

    # Convert best_seq to numpy array for refinement
    cur = np.array(best_seq, dtype=float)
    cur_score = best_score

    # 5. Refinement
    max_refine_iters = 80000
    iter_cnt = 0
    # we will precompute n constant
    while time.time() - start < budget_s - 5 and iter_cnt < max_refine_iters:
        iter_cnt += 1
        # Compute convolution
        conv = np.convolve(cur, cur)
        max_idx = int(np.argmax(conv))
        max_val = conv[max_idx]
        # compute contribution product
        # compute reverse index map
        i_arr = np.arange(n)
        rev = max_idx - i_arr
        mask = (rev >= 0) & (rev < n)
        prod = np.zeros(n)
        prod[mask] = cur[i_arr[mask]] * cur[rev[mask]]
        # source is index with largest product, target with smallest
        source = int(np.argmax(prod))
        target = int(np.argmin(prod))
        if source == target or cur[source] <= 0:
            # fallback random
            source, target = rng.choice(n, size=2, replace=False)
        # step fraction schedule
        # Reduce step as iterations increase
        frac = 0.08 * max(0.0, 1.0 - iter_cnt / max_refine_iters) + 0.02  # between 0.02 and 0.10
        step = cur[source] * rng.random() * frac
        if step <= 0:
            continue
        new = cur.copy()
        new[source] -= step
        new[target] += step
        # projection every 1000 iterations
        if iter_cnt % 1000 == 0:
            new = _project_to_simplex(new)
        # Compute new score quickly
        conv_new = np.convolve(new, new)
        max_val_new = np.max(conv_new)
        new_score = 2 * n * max_val_new / (np.sum(new) ** 2)  # sum ~1
        # Accept if improved
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_seq = new.tolist()
            cur = new
            cur_score = new_score
            print(f"[refine] iter={iter_cnt} new best={new_score:.6f}")
        elif new_score < cur_score:
            cur = new
            cur_score = new_score
        else:
            # occasional acceptance
            if rng.random() < 0.001:
                cur = new
                cur_score = new_score

    print(f"[final] best score={best_score:.6f}")
    return best_seq

"""
propose_candidate

Search for a non‑negative step‑function (list of heights) that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

where conv is the discrete convolution of the sequence with itself.
The algorithm consists of three phases:

1. **Exploratory sampling** – generate many candidates from several
   parametric families (power‑law, exponential, mixtures of two
   power‑laws).  Every candidate is normalised to a probability vector
   (sum = 1) and evaluated with the provided ``evaluate_sequence``.
   The best candidate is kept as a starting point.

2. **Targeted hill‑climbing** – repeatedly move a small amount of mass
   from the index that contributes most to the current maximal
   convolution value to the index that contributes the least.
   The move size starts relatively large (≈10 % of the source mass)
   and shrinks during the run.  The convolution of the current vector
   is kept from one iteration to the next, so each step needs only a
   single new convolution.

3. **Return the best vector** – the final best sequence (as a Python
   list of floats) is returned.  All numbers are non‑negative and the
   sum is 1, which makes the objective scale‑invariant.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (must be defined at top level – no closures)
# ----------------------------------------------------------------------

def _project_to_simplex(v):
    """Project a vector onto the probability simplex {x ≥ 0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


def _compute_score(arr):
    """
    Compute the evaluation score for a numpy array (non‑negative, not
    necessarily normalised).  The function works directly on the array,
    avoiding the Python‑list conversion required by ``evaluate_sequence``.
    """
    total = arr.sum()
    if total <= 0:
        return float('inf')
    conv = np.convolve(arr, arr)
    max_val = conv.max()
    n = arr.shape[0]
    return 2.0 * n * max_val / (total * total)


def _norm_and_score(raw):
    """Normalise a raw (non‑negative) vector and return (score, normalised)."""
    total = raw.sum()
    if total <= 0:
        return float('inf'), None
    norm = raw / total
    # use the supplied evaluator for a reliable reference score
    score = evaluate_sequence(norm.tolist())
    return score, norm

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed=123, budget_s=950, **kwargs):
    """
    Generate a step‑function (list of heights) that tries to minimise the
    provided ``evaluate_sequence`` value.

    Parameters
    ----------
    seed : int, optional
        Seed for NumPy's RNG (default 123).
    budget_s : float, optional
        Time budget in seconds.  The function will stop before this limit.
    **kwargs
        Ignored – kept for compatibility with the evaluation harness.

    Returns
    -------
    list[float]
        The best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # Determine the length of the problem (use the provided baseline)
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)   # provided by the harness
    except Exception:
        n = 2244  # default length used in the description

    # ------------------------------------------------------------------
    # Baseline (fallback) – the sequence that is always available
    # ------------------------------------------------------------------
    try:
        best_seq = list(height_sequence_1)
        best_score = evaluate_sequence(best_seq)
    except Exception:
        best_seq = np.full(n, 1.0 / n).tolist()
        best_score = evaluate_sequence(best_seq)

    print(f"[init] length = {n}, baseline score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Helper to evaluate a raw (not yet normalised) vector
    # ------------------------------------------------------------------
    idx = np.arange(n, dtype=float)

    def _try_candidate(raw_vec, label):
        """Normalise ``raw_vec``, evaluate it and update the global best."""
        nonlocal best_seq, best_score
        score, norm_vec = _norm_and_score(raw_vec)
        if norm_vec is None:
            return
        if score < best_score:
            best_score = score
            best_seq = norm_vec.tolist()
            print(f"[{label}] improved to {score:.6f}")

    # ------------------------------------------------------------------
    # 1. Grid search over pure power‑law families a_i = (i+shift)^(-alpha)
    # ------------------------------------------------------------------
    alphas = np.linspace(0.30, 1.00, 71)          # 0.30 … 1.00
    shifts = np.linspace(0.01, 3.00, 30)          # 0.01 … 3.00
    for alpha in alphas:
        if time.time() - start > budget_s - 10:
            break
        for shift in shifts:
            raw = np.power(idx + shift, -alpha)
            _try_candidate(raw, "grid")

    # ------------------------------------------------------------------
    # 2. Randomised power‑law sampling
    # ------------------------------------------------------------------
    for _ in range(3000):
        if time.time() - start > budget_s - 10:
            break
        alpha = rng.uniform(0.20, 1.20)
        shift = rng.uniform(0.001, 5.0)
        raw = np.power(idx + shift, -alpha)
        _try_candidate(raw, "randPL")

    # ------------------------------------------------------------------
    # 3. Random exponential families a_i = exp(-beta * i)
    # ------------------------------------------------------------------
    for _ in range(1000):
        if time.time() - start > budget_s - 10:
            break
        beta = rng.uniform(1e-5, 0.1)
        raw = np.exp(-beta * idx)
        _try_candidate(raw, "exp")

    # ------------------------------------------------------------------
    # 4. Random mixtures of two power‑law components
    # ------------------------------------------------------------------
    for _ in range(2000):
        if time.time() - start > budget_s - 10:
            break
        alpha1 = rng.uniform(0.20, 1.20)
        shift1 = rng.uniform(0.001, 5.0)
        alpha2 = rng.uniform(0.20, 1.20)
        shift2 = rng.uniform(0.001, 5.0)
        p = rng.random()
        raw = (p * np.power(idx + shift1, -alpha1) +
               (1 - p) * np.power(idx + shift2, -alpha2))
        _try_candidate(raw, "mixPL")

    print(f"[search] best score after sampling: {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Targeted hill‑climbing refinement
    # ------------------------------------------------------------------
    cur = np.array(best_seq, dtype=float)          # current working vector
    cur_score = best_score
    cur_conv = np.convolve(cur, cur)                # convolution of the current vector

    max_iters = 150000                               # safety cap
    it = 0
    while time.time() - start < budget_s - 5 and it < max_iters:
        it += 1

        # --- Determine the position of the current maximal convolution value
        max_idx = int(np.argmax(cur_conv))

        # --- Compute the contribution of each index to the peak value
        i_arr = np.arange(n)
        rev = max_idx - i_arr
        mask = (rev >= 0) & (rev < n)
        contrib = np.zeros_like(cur)
        contrib[mask] = cur[i_arr[mask]] * cur[rev[mask]]

        # --- Choose source (largest contribution) and target (smallest)
        src = int(np.argmax(contrib))
        tgt = int(np.argmin(contrib))
        if src == tgt or cur[src] <= 0:
            src, tgt = rng.choice(n, size=2, replace=False)

        # --- Step size (fraction of source mass) – slowly shrinks
        frac = 0.08 * max(0.0, 1.0 - it / max_iters) + 0.02   # between 0.02 and 0.10
        step = cur[src] * rng.random() * frac
        if step <= 0:
            continue

        # --- Propose a new vector by moving ``step`` mass from src to tgt
        new = cur.copy()
        new[src] -= step
        new[tgt] += step

        # Normalise to avoid drift (cheap, O(n))
        total = new.sum()
        if abs(total - 1.0) > 1e-12:
            new /= total

        # --- New convolution and score (only one convolution per iteration)
        new_conv = np.convolve(new, new)
        new_max = new_conv.max()
        new_score = 2.0 * n * new_max / (new.sum() * new.sum())

        # --- Accept if we improve (or occasionally accept worse)
        improved = False
        if new_score < cur_score - 1e-12:
            improved = True
        elif rng.random() < 0.001:  # tiny probability to escape plateaus
            improved = True

        if improved:
            # verify the improvement with the official evaluator (rare)
            real_score = evaluate_sequence(new.tolist())
            if real_score < cur_score - 1e-12:
                cur = new
                cur_conv = new_conv
                cur_score = real_score
                if real_score < best_score - 1e-12:
                    best_score = real_score
                    best_seq = new.tolist()
                    print(f"[refine] iter={it} new best={real_score:.6f}")
            else:
                # fallback to the cheap score if the official one is not better
                cur = new
                cur_conv = new_conv
                cur_score = new_score

        # occasional re‑normalisation to the simplex for numerical safety
        if it % 1000 == 0:
            cur = _project_to_simplex(cur)
            cur_conv = np.convolve(cur, cur)

    print(f"[final] best score found: {best_score:.6f}")
    return best_seq
