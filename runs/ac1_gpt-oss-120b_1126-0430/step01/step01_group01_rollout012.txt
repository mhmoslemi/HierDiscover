def _lp_tail(base, tail_len, slack_factor, deadline):
    import time, scipy.sparse as sp, numpy as np
    if time.time() > deadline:
        return np.zeros(tail_len)
    n = base.shape[0]
    conv_base = np.convolve(base, base)
    M = conv_base.max()
    total_rows = n + tail_len - 1
    # Compute slack per row
    slack = np.empty(total_rows, dtype=float)
    for r in range(total_rows):
        k = n + r
        if k < conv_base.size:
            s = M - conv_base[k]
        else:
            s = M
        slack[r] = s if s > 0 else 0.0
    # Apply factor
    b_ub = slack * slack_factor

    # Build Sparse matrix A
    diag_data = []
    offsets = []
    for i in range(n):
        diag_len = total_rows - i
        if diag_len <= 0:
            break
        diag_data.append(np.full(diag_len, base[i], dtype=float))
        offsets.append(i)
    A = sp.diags(diag_data, offsets, shape=(total_rows, tail_len), format='csc')
    # Solve LP
    from scipy.optimize import linprog
    c = -np.ones(tail_len)
    bounds = [(0, None)] * tail_len
    # Optionally set ; we may add to improve speed.
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        print("LP not solved:", res.message)
        return np.zeros(tail_len)
    t = res.x
    t[t < 0] = 0.0
    return t

"""
Improved step‑function construction.

The algorithm starts from the Catalan (arcsine) seed, for which the
self‑convolution is constant (=1) on the first half of the entries and
has a metric ≈ π/2 ≈ 1.57.  We then solve a linear program that adds a
new “tail’’ of non‑negative coefficients while keeping the cross‑convolution
with the seed below a chosen fraction of the available slack (the distance
to the current maximum M).  This increases the total ℓ¹‑mass without
raising the peak of the self‑convolution much, thereby lowering the
evaluation score towards the theoretical optimum 1.  Finally a cheap
greedy refinement (mass moves from indices that contribute most to the
current maximum towards those that contribute little) is applied.

The whole search respects a 1000 s budget, prints progress information
and returns the best sequence found.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Fast evaluation – identical to evaluate_sequence but without the security checks.
# ----------------------------------------------------------------------
def _fast_evaluation(seq: np.ndarray) -> float:
    """Return the same metric as evaluate_sequence but much faster."""
    n = seq.size
    if n == 0:
        return np.inf
    # FFT size = next power of two >= 2*n-1
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Catalan (arcsine) seed – a_i = binom(2i,i) / 4**i.
# ----------------------------------------------------------------------
def _catalan_sequence(length: int) -> np.ndarray:
    """First ``length`` Catalan coefficients."""
    seq = np.empty(length, dtype=float)
    a = 1.0
    seq[0] = a
    for i in range(1, length):
        a *= (2 * i - 1) / (2.0 * i)   # recurrence a_i = a_{i-1} * (2i-1)/(2i)
        seq[i] = a
    return seq


# ----------------------------------------------------------------------
# Linear‑programming tail: maximise added mass while keeping
# base*t ≤ slack·factor (so that the full self‑convolution stays near the
# original maximum M).
# ----------------------------------------------------------------------
def _lp_tail(base: np.ndarray,
             tail_len: int,
             slack_factor: float,
             deadline: float) -> np.ndarray:
    """Return a tail of length ``tail_len`` solving the LP described above."""
    if time.time() > deadline:
        return np.zeros(tail_len)

    n = base.size
    conv_base = np.convolve(base, base)
    M = conv_base.max()

    # rows correspond to convolution indices r of (base * tail)
    rows = n + tail_len - 1
    slack = np.empty(rows, dtype=float)

    for r in range(rows):
        k = n + r                               # overall index in a*a
        if k < conv_base.size:
            s = M - conv_base[k]
        else:
            s = M
        slack[r] = max(0.0, s) * slack_factor   # apply safety factor

    # Build the sparse matrix A where A[r, j] = base[r-j] (if defined)
    diag_data = []
    offsets = []
    for i in range(n):
        diag_len = rows - i
        if diag_len <= 0:
            break
        diag_data.append(np.full(diag_len, base[i], dtype=float))
        offsets.append(i)
    A = sp.diags(diag_data, offsets, shape=(rows, tail_len), format='csc')

    # Linear program: maximise sum(tail)  <=> minimise -sum(tail)
    c = -np.ones(tail_len)                     # objective coefficients
    bounds = [(0, None)] * tail_len             # non‑negative tail entries

    res = linprog(c, A_ub=A, b_ub=slack, bounds=bounds, method='highs')
    if not res.success:
        print("LP failed:", res.message)
        return np.zeros(tail_len)

    tail = res.x
    tail[tail < 0] = 0.0                         # numerical safety
    print(f"LP tail solved – sum={tail.sum():.4f}, max={tail.max():.4f}")
    return tail


# ----------------------------------------------------------------------
# Greedy refinement – move a modest fraction of mass away from indices
# that most influence the current maximum convolution entry.
# ----------------------------------------------------------------------
def _refine(seq: np.ndarray, deadline: float) -> np.ndarray:
    """Iteratively improve ``seq`` until the deadline is reached."""
    n = seq.size
    best_seq = seq.copy()
    best_score = _fast_evaluation(best_seq)

    # FFT size is fixed throughout the refinement
    fft_len = 1 << ((2 * n - 1).bit_length())

    while time.time() < deadline:
        # current convolution and its maximal index
        fa = np.fft.rfft(seq, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        k = int(conv.argmax())
        max_val = conv[k]

        # contribution weights w[i] = a_{k-i} (0 if out of range)
        i_arr = np.arange(n)
        j_arr = k - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        w = np.zeros_like(seq)
        w[mask] = seq[j_arr[mask]]

        # source = index with largest a_i * w_i (largest contribution)
        contrib = seq * w
        src = int(np.argmax(contrib))
        # sink = index with minimal w (almost no influence on the max)
        sink = int(np.argmin(w))

        if seq[src] < 1e-12:
            break

        # move a small fraction of the source mass to the sink
        eps = 0.05 * seq[src]   # 5 % of the source amount
        new_seq = seq.copy()
        new_seq[src] -= eps
        new_seq[sink] += eps

        score = _fast_evaluation(new_seq)

        # accept strictly better candidates
        if score < best_score - 1e-9:
            seq = new_seq
            best_score = score
            best_seq = new_seq.copy()
        else:
            # occasional jitter to escape plateaus
            if np.random.rand() < 0.01:
                jitter_idx = np.random.randint(n)
                jitter = np.random.randn() * 0.001
                new_seq = seq.copy()
                new_seq[jitter_idx] = max(0.0, new_seq[jitter_idx] + jitter)
                seq = new_seq

    return best_seq


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Produce a list of non‑negative heights with a low evaluation score.

    Strategy:
      1. Catalan (arcsine) seed of length N.
      2. Linear‑programming tail of length N that adds as much mass as
         possible while keeping the cross‑convolution below a safe
         fraction of the slack (so the peak does not increase much).
      3. Concatenate seed and tail, then run a cheap greedy refinement.
      4. Return the best sequence discovered before the deadline.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5   # small safety margin

    # ------------------------------------------------------------------
    # 1.  Catalan seed
    # ------------------------------------------------------------------
    N = 2500                     # length of the seed (tunable)
    base = _catalan_sequence(N)

    best_seq = base
    best_score = _fast_evaluation(base)
    print(f"Initial Catalan score (n={N}): {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2.  LP tail – add extra mass without raising the peak too much
    # ------------------------------------------------------------------
    tail_len = N                 # we add a tail of the same size
    slack_factor = 0.5           # keep 50 % of the slack for safety

    tail = _lp_tail(base, tail_len, slack_factor, deadline)
    extended_seq = np.concatenate([base, tail])

    ext_score = _fast_evaluation(extended_seq)
    print(f"After LP – length={len(extended_seq)}, sum={extended_seq.sum():.4f}, "
          f"score={ext_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Greedy refinement on the extended sequence
    # ------------------------------------------------------------------
    refined_seq = _refine(extended_seq, deadline)
    refined_score = _fast_evaluation(refined_seq)
    print(f"After refinement – length={len(refined_seq)}, sum={refined_seq.sum():.4f}, "
          f"score={refined_score:.6f}")

    if refined_score < best_score:
        best_seq = refined_seq
        best_score = refined_score

    # ------------------------------------------------------------------
    # 4.  Final report
    # ------------------------------------------------------------------
    print(f"Final best score: {best_score:.6f}")
    return best_seq.tolist()
