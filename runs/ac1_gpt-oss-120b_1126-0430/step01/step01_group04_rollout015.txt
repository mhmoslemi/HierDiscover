"""
propose_candidate
=================

A two‑phase optimiser for the “max‐convolution” problem.

Phase 1 – Random families
-------------------------
Generate many analytically‑parameterised candidates
(power‑law, shifted power‑law, polynomial, exponential,
linear decay and Dirichlet draws), scale them to the
required L¹‑norm √(2 n) and keep the best one.  The
parameter ranges are deliberately widened compared with the
previous baseline.

Phase 2 – Gradient descent on the active lag
--------------------------------------------
For the current best sequence a, let k\* be the index where
the convolution a∗a attains its maximum.  The gradient of the
objective max(a∗a) w.r.t. a at this active lag is

    g_i = 2·a_{k*‑i}    for 0 ≤ k*‑i < n,   else 0.

A projected gradient step moves the vector opposite to g,
then projects back onto the simplex
{a ≥ 0, ∑a = √(2 n)}.  Step size decays over time.
Occasional random perturbations (mass transfer between two
indices) are injected to escape plateaus.  The best sequence
found over the whole budget is returned.

All heavy work (convolution, gradient) uses FFT‑based
convolution, which is O(n log n) and fast enough for the
1000 s time limit even with n≈7 000.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (O(n) algorithm by Wang & Carreira‑Perpiñán)
# ----------------------------------------------------------------------
def _project_to_simplex(x: np.ndarray, target_sum: float) -> np.ndarray:
    """Project vector x onto {y ≥ 0, ∑y = target_sum}."""
    n = x.size
    if target_sum <= 0:
        return np.zeros_like(x)

    # sort descending
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(x - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast convolution of a vector with itself (real FFT)
# ----------------------------------------------------------------------
def _self_convolve(a: np.ndarray) -> np.ndarray:
    """Return np.convolve(a, a) using an FFT of size power‑of‑2."""
    n = a.size
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=m)
    conv_full = np.fft.irfft(A * A, n=m)[: 2 * n - 1]
    return conv_full.real


# ----------------------------------------------------------------------
# Gradient of max‑convolution at the active lag
# ----------------------------------------------------------------------
def _max_grad(a: np.ndarray, k_max: int) -> np.ndarray:
    """Gradient of max_k (a∗a)[k] evaluated at the currently maximal lag."""
    n = a.size
    g = np.zeros_like(a)
    i_start = max(0, k_max - (n - 1))
    i_end = min(k_max, n - 1)
    idx = np.arange(i_start, i_end + 1, dtype=int)
    g[idx] = 2.0 * a[k_max - idx]
    return g

# Main optimiser
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights (the step function) that
    minimises the evaluation function.  The routine respects the
    supplied time budget (seconds).
    """
    np.random.seed(seed)
    start_time = time.time()

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
        # normalise the supplied sequence to the exact L¹‑norm
        target_sum = math.sqrt(2.0 * n)
        init = np.asarray(height_sequence_1, dtype=float)
        init = init * (target_sum / init.sum())
        best_seq = init.copy()
    else:
        n = 2000
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    best_score = _self_convolve(best_seq).max()
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    phase1_deadline = start_time + budget_s * 0.40
    iter_sweep = 0
    while time.time() < phase1_deadline:
        iter_sweep += 1
        shape = np.random.choice(
            ["power", "shifted_power", "poly", "linear", "exp", "rand_dir"]
        )

        if shape == "power":
