propose_candidate
=================

A two‑stage optimizer for the “max‑autocorrelation” problem.

Stage 1 – *sub‑gradient descent*  
  The objective (with the L1‑norm fixed to √(2·n)) is simply the
  maximum value of the linear convolution a*a.  For the current peak
  index set K we can compute a sub‑gradient g_i = Σ_{k∈K} a_{k‑i}.
  Moving a little in the direction –g and projecting back onto the
  simplex (non‑negative, fixed sum) reduces the peak.  An adaptive
  step‑size and many random restarts (different analytic families such
  as power‑law, exponential, Dirichlet) explore the search space.

Stage 2 – *local hill climbing / simulated annealing*  
  The best vector from stage 1 is refined with cheap pairwise mass‑
  transfers, accepting improving moves and occasionally a worsened
  one according to a temperature schedule.

Both stages respect the 1000 s time budget and run entirely with

valid step‑function for the supplied ``evaluate_sequence`` routine.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_to_simplex(x: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Project the vector x onto the simplex { y ≥ 0, Σy = target_sum }.
    Implementation of the O(n) algorithm by Wang & Carreira‑Perpiñán (2013).
    """
    if target_sum <= 0:
        return np.zeros_like(x)

    # sort descending
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho = np.nonzero(u - cssv / np.arange(1, len(x) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(x - theta, 0.0)
    return w


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution of a with itself using FFT.
    Returns the full length (2·n‑1) array.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())      # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[: 2 * n - 1]


def _make_sequence_from_raw(raw: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Scale a non‑negative raw vector so that its L1‑norm equals target_sum.
    """
    raw_sum = raw.sum()
    if raw_sum == 0:
        return np.full_like(raw, target_sum / raw.shape[0])
    scale = target_sum / raw_sum
    return raw * scale

            a = np.full(n, target_sum / n, dtype=float)
        elif shape == "power":
            p = np.random.uniform(0.2, 1.5)
            raw = (np.arange(1, n + 1, dtype=float)) ** (-p)
            a = _make_sequence_from_raw(raw, target_sum)
        elif shape == "shifted_power":
            p = np.random.uniform(0.1, 1.2)
            d = np.random.uniform(0.0, n * 0.5) + 1e-6
            raw = (np.arange(n, dtype=float) + d) ** (-p)
            a = _make_sequence_from_raw(raw, target_sum)
        elif shape == "exponential":
            lam = np.random.uniform(0.0004, 0.008)   # slower decay → smoother shape
            raw = np.exp(-lam * np.arange(n, dtype=float))
            a = _make_sequence_from_raw(raw, target_sum)
        else:   # "rand_dir"
            raw = np.random.gamma(shape=0.6, scale=1.0, size=n)
            a = _make_sequence_from_raw(raw, target_sum)

        # Project onto the simplex to guarantee the exact L1‑norm.
        a = _project_to_simplex(a, target_sum)

        # ---- sub‑gradient descent ----
        step = init_step_factor * target_sum
        conv = _conv_fft(a)
        M = conv.max()
        it = 0
        while it < max_iters_per_restart and time.time() < refinement_deadline:
            # indices where the convolution attains (or nearly attains) the max
            K = np.where(conv >= M - tolerance)[0]

            # compute sub‑gradient g_i = Σ_{k∈K} a_{k‑i}
            g = np.zeros_like(a)
            for k in K:
                i_start = max(0, k - (n - 1))
                i_end = min(n - 1, k)
                # vectorised contribution for this k:
                idx = np.arange(i_start, i_end + 1, dtype=int)
                g[idx] += a[k - idx]

            norm_g = np.linalg.norm(g)
            if norm_g == 0.0:
                break

            # move opposite to the gradient and re‑project
            a_new = a - step * g
            a_new = _project_to_simplex(a_new, target_sum)

            conv_new = _conv_fft(a_new)
            M_new = conv_new.max()

            # Accept the step if we do not increase the maximal value.
            if M_new <= M:
                a = a_new
                conv = conv_new
                M = M_new
                step *= step_decay          # slowly shrink the step size
                it += 1
            else:
                # If the move was bad, shrink the step aggressively and try again.
                step *= 0.5
                if step < 1e-12 * target_sum:
                    break

        # ---- evaluate candidate -------------------------------------------------
        cand_score = float(M)   # because Σa = √(2n) → score = max(conv)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = a.copy()
            print(
                f"[restart {restart_counter:03d}] new best {best_score:.6f} "
                f"via {shape} (sub‑grad iters={it})"
            )

        # occasional light random perturbation to jump out of shallow plateaus
        if restart_counter % 7 == 0 and np.random.rand() < 0.3:
            tmp = best_seq.copy()
            i, j = np.random.choice(n, size=2, replace=False)
            delta = tmp[i] * 0.015
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_to_simplex(tmp, target_sum)
            tmp_score = float(_conv_fft(tmp).max())
            if tmp_score < best_score:
                best_score = tmp_score
                best_seq = tmp
                print(f"[perturb] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – local hill climbing / simulated annealing for the final
    # few seconds of the budget.
    # ------------------------------------------------------------------
    remaining_time = time.time() + budget_s - time.time()
    after_refine_deadline = start_time + budget_s - 0.5
    current_seq = best_seq.copy()
    current_score = best_score
    temperature = 0.04
    decay = 0.9994

    while time.time() < after_refine_deadline:
        # pick two distinct indices and transfer a small amount of mass
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        max_move = current_seq[i] * 0.04
        if max_move <= 0:
            continue
        delta = np.random.rand() * max_move
        new_seq = current_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        # keep feasibility
        if np.any(new_seq < -1e-12):
            continue
        new_seq = _project_to_simplex(new_seq, target_sum)

        new_score = float(_conv_fft(new_seq).max())
        if new_score < current_score:
            current_seq = new_seq
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.copy()
                print(f"[refine] improved to {best_score:.6f}")
        else:
            # Metropolis acceptance
            prob = math.exp(-(new_score - current_score) / (temperature + 1e-12))
            if np.random.rand() < prob:
                current_seq = new_seq
                current_score = new_score
        temperature *= decay

    print(f"[final] best score found: {best_score:.6f}")
    # Return a plain Python list as required by the grader.
    return best_seq.tolist()
