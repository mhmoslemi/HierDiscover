    """

    # set random seeds
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.1

            pass

    # Parameter grid
    lengths = [2000, 4000, 8000, 12000]   # can be trimmed if time short

    # We'll operate on numpy array for speed
    seq = np.array(best_seq, dtype=np.float64)
    n = len(seq)
    current_score = best_score

    # Temperature schedule for simulated annealing
    # Start with relatively high temperature to accept uphill moves then cool.
    T0 = 0.01 * current_score   # scale with score
    final_T = 1e-6
    # compute total steps feasible approx
    steps = 0

    max_attempts = 200000

"""
Propose a step‑function sequence with a low evaluation score.

The evaluation is scale‑invariant, so we only need to optimise the *shape*
of a probability distribution p (p_i = a_i / sum(a)).  The code proceeds
in three phases:

1. **Coarse grid search** –  several lengths (2000 … 12000) and a
   coarse grid of power‑law exponents β are evaluated.  The best
   (length, β) pair is remembered.

2. **Fine local search** – around the best β a finer grid is tried.
   This usually brings the score close to the theoretical bound
   (≈1.35 for β≈0.6).

3. **Stochastic annealing** – a cheap mass‑transfer hill‑climb with a
   temperature schedule explores the neighbourhood of the best
   candidate and accepts occasional uphill moves.  The number of
   iterations is adapted to the sequence length so that the whole
   routine stays well within the 1000 s budget.

The algorithm is deliberately lightweight (no external solvers) and
relies only on NumPy and the provided `evaluate_sequence` function.
"""

import time
import numpy as np
import random

def _grid_search(lengths, betas, deadline, best_seq, best_score):
    """
    Evaluate power‑law sequences a_i = (i+1)^(-beta) for the given
    lengths and exponent grid.  Returns the best sequence together with
    its score and the parameters that produced it.
    """
    best_n = None
    best_beta = None
    for n in lengths:
        if time.time() > deadline:
            break
        idx = np.arange(1, n + 1, dtype=np.float64)
        for beta in betas:
            if time.time() > deadline:
                break
            seq = idx ** (-beta)          # monotone decreasing shape
            try:
                score = evaluate_sequence(seq.tolist())
            except Exception:
                continue
            if score < best_score:
                best_score = score
                best_seq = seq.tolist()
                best_n = n
                best_beta = beta
                print(f"[grid] n={n:5d}, beta={beta: .3f} → score={score:.6f}")
    return best_seq, best_score, best_n, best_beta


def _anneal(seq, current_score, deadline):
    """
    Simple simulated‑annealing / hill‑climb on a fixed‑length sequence.
    Mass is transferred between two random positions; worsening moves
    are accepted with probability exp(-Δ / T) where the temperature T
    is linearly cooled.
    The function returns the best sequence and its score found during
    the annealing phase.
    """
    n = len(seq)
    seq = np.array(seq, dtype=np.float64)
    best_seq = seq.copy()
    best_score = current_score

    # schedule parameters – a few thousand attempts are enough for larger n
    max_attempts = 50000 if n <= 4000 else 20000 if n <= 8000 else 8000
    T_start = 0.02 * current_score   # initial temperature (relative)
    T_end = 1e-6

    for step in range(1, max_attempts + 1):
        if time.time() > deadline:
            break
        # linear cooling
        T = T_start * (1 - step / max_attempts) + T_end * (step / max_attempts)

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # Transfer ≤5 % of the source mass
        eps = seq[i] * np.random.rand() * 0.05
        if eps == 0.0:
            continue

        new_seq = seq.copy()
        new_seq[i] -= eps
        new_seq[j] += eps
        if new_seq[i] < 0.0:         # safeguard – should not happen
            continue
        try:
            new_score = evaluate_sequence(new_seq.tolist())
        except Exception:
            continue

        delta = new_score - current_score
        # Accept improvement or a probabilistic uphill move
        if delta < 0.0 or np.random.rand() < np.exp(-delta / max(T, 1e-12)):
            seq = new_seq
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.copy()
                print(f"[anneal] step {step:6d}, score={new_score:.6f}")

    return best_seq.tolist(), best_score


def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights minimising ``evaluate_sequence``.
    The routine combines a coarse power‑law grid search, a fine local
    refinement around the best exponent, and a short simulated‑annealing
    phase.  It respects the time budget and prints progress messages.
    """
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.1    # tiny safety margin

    best_seq = None
    best_score = float("inf")
    best_n = None
    best_beta = None

    # ------------------------------------------------------------------
    # 1.  Warm start from the globally supplied sequence, if any.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            sc = evaluate_sequence(height_sequence_1)
            if sc < best_score:
                best_seq = list(height_sequence_1)
                best_score = sc
                print(f"[init] height_sequence_1 score={sc:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 2.  Coarse grid over several lengths and β‑values.
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000, 12000]                # longer arrays give lower bound
    betas = np.linspace(0.45, 0.75, 31)                # step ≈0.01
    best_seq, best_score, best_n, best_beta = _grid_search(
        lengths, betas, deadline, best_seq, best_score
    )

    # ------------------------------------------------------------------
    # 3.  Fine‑grained search around the best β (only on the best length).
    # ------------------------------------------------------------------
    if best_seq is not None and best_n is not None and time.time() < deadline:
        idx = np.arange(1, best_n + 1, dtype=np.float64)
        beta_fine = np.linspace(best_beta - 0.02, best_beta + 0.02, 21)
        for beta in beta_fine:
            if time.time() > deadline:
                break
            seq = idx ** (-beta)
            try:
                score = evaluate_sequence(seq.tolist())
            except Exception:
                continue
            if score < best_score:
                best_score = score
                best_seq = seq.tolist()
                best_beta = beta
                print(f"[fine] beta={beta:.4f} → score={score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Local stochastic optimisation (annealing).
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        best_seq, best_score = _anneal(best_seq, best_score, deadline)

    elapsed = time.time() - start
    print(f"[final] best score={best_score:.6f}, length={len(best_seq)}, time={elapsed:.1f}s")
    return best_seq
