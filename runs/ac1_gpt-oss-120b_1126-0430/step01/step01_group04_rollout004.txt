"""
propose_candidate
=================

Search for a non‑negative step‑function (list of heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The score is scale‑invariant, therefore we keep the L1‑norm fixed to
`sqrt(2*n)`.  The search consists of three stages:

1. **Rich parameter sweep** – a large family of analytically‑parameterised
   shapes (power‑law, shifted power‑law, exponential, mixed power+uniform,
   Beta‑pdf, linear, polynomial, random Dirichlet, Hann/Blackman windows,
   and random mixtures of two shapes) is sampled.  Every raw vector is
   scaled to the required L1‑norm and evaluated; the best sequence so far
   is retained.

2. **Gradient‑guided local refinement** – starting from the best candidate,
   the index `m` where the convolution attains its maximum is found.
   The gradient of the quadratic form `b_m = (a*a)[m]` is
   `g_i = 2 * a[m-i]` (if the index is in range, otherwise 0).  A
   projected gradient step is taken (with an annealing temperature) and
   the new point is projected onto the simplex `{a_i ≥ 0, sum a_i = √(2n)}`.
   Moves that improve the score are always accepted; worse moves are
   accepted with a probability that decays with the temperature.

3. **Occasional random perturbations** – to escape shallow plateaus a
   tiny random mass‑transfer between two indices is performed every few
   hundred iterations.

The whole routine respects the supplied time budget (`budget_s`,
default 1000 s) and prints progress information.  The returned list is
ready to be fed directly into `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------


def _project_to_simplex(x: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    { y >= 0, sum(y) = target_sum }.
    Implements the O(n) algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    if target_sum <= 0:
        return np.zeros_like(x)

    n = x.shape[0]
    # sort in descending order
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(x - theta, 0.0)
    return w


def _evaluate(seq: list[float]) -> float:
    """Thin wrapper around the grader supplied ``evaluate_sequence``."""
    return evaluate_sequence(seq)


def _make_sequence_from_raw(raw: np.ndarray, target_sum: float) -> list[float]:
    """
    Scale a non‑negative raw vector so that its L1‑norm equals ``target_sum``.
    """
    raw_sum = raw.sum()
    if raw_sum == 0:
        # fallback to uniform distribution
        n = raw.shape[0]
        return [target_sum / n] * n
    scale = target_sum / raw_sum
    seq = (raw * scale).tolist()
    return seq


def _conv_max(seq: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Compute the full linear convolution ``seq*seq`` using a FFT and return
    the maximum value together with the whole convolution array.
    """
    n = seq.shape[0]
    # length of linear convolution
    conv_len = 2 * n - 1
    # next power of two for efficient FFT
    N = 1 << ((conv_len - 1).bit_length())
    fft_seq = np.fft.rfft(seq, n=N)
    conv = np.fft.irfft(fft_seq * fft_seq, n=N)[:conv_len]
    # Numerical noise may produce tiny negative values
    conv = np.real(conv)
    max_val = np.max(conv)
    return max_val, conv


# ----------------------------------------------------------------------
# Shape factories – generate raw (un‑scaled) vectors
# ----------------------------------------------------------------------


def _shape_power(n: int) -> np.ndarray:
    p = np.random.uniform(0.1, 5.0)          # allow stronger decays
    return (np.arange(1, n + 1, dtype=float)) ** (-p)


def _shape_shifted_power(n: int) -> np.ndarray:
    p = np.random.uniform(0.1, 5.0)
    d = np.random.uniform(0.0, n * 0.5) + 1e-9
    return (np.arange(n, dtype=float) + d) ** (-p)


def _shape_exponential(n: int) -> np.ndarray:
    lam = np.random.uniform(0.0005, 0.02)   # slower vs faster decay
    return np.exp(-lam * np.arange(n, dtype=float))


def _shape_power_mix(n: int) -> np.ndarray:
    """Mixture of a power‑law tail and a uniform background."""
    p = np.random.uniform(0.1, 5.0)
    w = np.random.rand()                   # weight of the power component
    power = (np.arange(1, n + 1, dtype=float)) ** (-p)
    uniform = np.ones(n, dtype=float)
    return w * power + (1.0 - w) * uniform


def _shape_exponential_mix(n: int) -> np.ndarray:
    """Mixture of exponential decay and uniform background."""
    lam = np.random.uniform(0.0005, 0.02)
    w = np.random.rand()
    exp = np.exp(-lam * np.arange(n, dtype=float))
    uniform = np.ones(n, dtype=float)
    return w * exp + (1.0 - w) * uniform


def _shape_linear(n: int) -> np.ndarray:
    return np.linspace(1.0, 0.0, n, dtype=float)


def _shape_polynomial(n: int) -> np.ndarray:
    alpha = np.random.uniform(0.1, 5.0)
    return (1.0 - np.arange(n, dtype=float) / (n - 1)) ** alpha


def _shape_random_dirichlet(n: int) -> np.ndarray:
    shape = np.random.uniform(0.3, 1.5)    # a little less spiky than before
    return np.random.gamma(shape=shape, scale=1.0, size=n)


def _shape_beta(n: int) -> np.ndarray:
    """
    Beta‑pdf on [0,1] (symmetric or skewed).  Parameters a,b > 0.
    """
    a = np.random.uniform(0.5, 5.0)
    b = np.random.uniform(0.5, 5.0)
    x = (np.arange(n, dtype=float) + 0.5) / n   # mid‑point sampling
    # beta pdf ∝ x^{a-1} (1-x)^{b-1}
    pdf = np.exp((a - 1) * np.log(x) + (b - 1) * np.log(1 - x))
    return pdf


def _shape_hann(n: int) -> np.ndarray:
    """Standard Hann window (raised cosine)."""
    i = np.arange(n, dtype=float)
    return 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))


def _shape_blackman(n: int) -> np.ndarray:
    """Blackman window – a smoother cosine."""
    i = np.arange(n, dtype=float)
    a0 = 0.42
    a1 = 0.5
    a2 = 0.08
    return (a0
            - a1 * np.cos(2.0 * np.pi * i / (n - 1))
            + a2 * np.cos(4.0 * np.pi * i / (n - 1)))


def _shape_mix_two(n: int) -> np.ndarray:
    """Random convex combination of two distinct primitive shapes."""
    factories = [_shape_power, _shape_shifted_power, _shape_exponential,
                 _shape_power_mix, _shape_exponential_mix,
                 _shape_linear, _shape_polynomial, _shape_random_dirichlet,
                 _shape_beta, _shape_hann, _shape_blackman]
    f1, f2 = np.random.choice(factories, size=2, replace=False)
    w = np.random.rand()
    return w * f1(n) + (1.0 - w) * f2(n)


_SHAPE_GENERATORS = {
    "power": _shape_power,
    "shifted_power": _shape_shifted_power,
    "exponential": _shape_exponential,
    "power_mix": _shape_power_mix,
    "exponential_mix": _shape_exponential_mix,
    "linear": _shape_linear,
    "poly": _shape_polynomial,
    "rand_dir": _shape_random_dirichlet,
    "beta": _shape_beta,
    "hann": _shape_hann,
    "blackman": _shape_blackman,
    "mix_two": _shape_mix_two,
}

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights that (hopefully) yields a low
    evaluation score.  The function respects the time budget ``budget_s``.
    """
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
        # sanity: ensure we have enough room for the search
    else:
        # a moderate size that still fits comfortably in the time budget
        n = 8000

    target_sum = math.sqrt(2.0 * n)          # the normalising L1‑norm
    # start from the provided construction if available, otherwise uniform
    if "height_sequence_1" in globals():
        best_seq = list(height_sequence_1)
        # re‑normalise to the exact target sum (in case the stored one differs)
        best_seq = _make_sequence_from_raw(np.array(best_seq, dtype=float), target_sum)
    else:
        best_seq = [target_sum / n] * n       # uniform start

    best_score = _evaluate(best_seq)
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – parameter sweep
    # ------------------------------------------------------------------
    start_time = time.time()
    # leave a few seconds for the refinement phase
    sweep_deadline = start_time + budget_s - 15.0
    sweep_iter = 0

    shape_keys = list(_SHAPE_GENERATORS.keys())

    while time.time() < sweep_deadline:
        sweep_iter += 1
        key = np.random.choice(shape_keys)
        raw = _SHAPE_GENERATORS[key](n)

        # occasional extra smoothing (kernel length 3‑7) – cheap and adds variety
        if np.random.rand() < 0.15:
            k = np.random.randint(3, 8)
            kernel = np.ones(k, dtype=float) / k
            raw = np.convolve(raw, kernel, mode="same")

        # Ensure non‑negative (some generators may produce tiny negatives due to
        # numerical noise, especially the beta pdf)
        raw = np.maximum(raw, 0.0)

        candidate = _make_sequence_from_raw(raw, target_sum)
        score = _evaluate(candidate)

        if score < best_score:
            best_score = score
            best_seq = candidate
            print(f"[sweep {sweep_iter:05d}] new best {best_score:.6f} via {key}")

        # random tiny perturbation every few hundred iterations to escape plateaus
        if sweep_iter % 300 == 0 and np.random.rand() < 0.35:
            tmp = np.array(best_seq, copy=True)
            i, j = np.random.choice(n, size=2, replace=False)
            delta = tmp[i] * np.random.uniform(0.0, 0.02)
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_to_simplex(tmp, target_sum)
            pert_score = _evaluate(tmp.tolist())
            if pert_score < best_score:
                best_score = pert_score
                best_seq = tmp.tolist()
                print(f"[perturb {sweep_iter:05d}] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – gradient‑guided refinement + simulated annealing
    # ------------------------------------------------------------------
    current_seq = np.array(best_seq, copy=True)
    current_score = best_score

    temperature = 0.12
    decay = 0.9996

    refine_deadline = start_time + budget_s - 3.0  # a few seconds to finish cleanly
    refine_iter = 0

    while time.time() < refine_deadline:
        refine_iter += 1

        # compute convolution and locate the peak
        max_val, conv = _conv_max(current_seq)
        max_idx = int(np.argmax(conv))

        # gradient of the peak quadratic form:
        # g_i = 2 * a_{max_idx - i}  if 0 <= max_idx-i < n else 0
        grad = np.zeros_like(current_seq)
        low = max(0, max_idx - (n - 1))
        high = min(n - 1, max_idx)
        idx_range = np.arange(low, high + 1, dtype=int)
        grad[idx_range] = 2.0 * current_seq[max_idx - idx_range]

        # propose a move opposite to the gradient.
        # The move is first centred (zero‑mean) so that total sum stays unchanged,
        # then scaled to a small fraction of the total mass.
        direction = -grad
        direction -= direction.mean()                     # keep zero sum

        # step size adapts to temperature; also cap the L1‑norm of the step
        if np.linalg.norm(direction, 1) > 0:
            max_step = temperature * target_sum * 0.015   # empirical scale
            direction = direction * min(1.0, max_step / np.linalg.norm(direction, 1))

        new_seq = current_seq + direction
        new_seq = _project_to_simplex(new_seq, target_sum)
        new_score = _evaluate(new_seq.tolist())

        # acceptance rule
        if new_score < current_score:
            accept = True
        else:
            prob = math.exp(-(new_score - current_score) / (temperature + 1e-12))
            accept = np.random.rand() < prob

        if accept:
            current_seq = new_seq
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.tolist()
                print(f"[refine {refine_iter:06d}] improved to {best_score:.6f}")

        # occasional random mass‑swap (helps to re‑explore)
        if refine_iter % 500 == 0 and np.random.rand() < 0.25:
            tmp = current_seq.copy()
            i, j = np.random.choice(n, size=2, replace=False)
            delta = tmp[i] * np.random.uniform(0.0, 0.018)
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_to_simplex(tmp, target_sum)
            tmp_score = _evaluate(tmp.tolist())
            if tmp_score < best_score:
                best_score = tmp_score
                best_seq = tmp.tolist()
                current_seq = tmp
                current_score = tmp_score
                print(f"[swap {refine_iter:06d}] improved to {best_score:.6f}")

        # temperature decay
        temperature *= decay
        if temperature < 1e-5:
            temperature = 1e-5

    print(f"[final] best score achieved: {best_score:.6f}")
    return best_seq
