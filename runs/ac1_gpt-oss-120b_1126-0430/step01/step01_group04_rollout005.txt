"""
Improved optimiser for the max‑convolution objective.

The score is   2 * n * max(conv(a,a)) / (sum(a)**2) .
If we keep the L1‑norm fixed to sqrt(2*n) the factor 2*n/(sum**2) = 1, so
the problem reduces to **minimising the largest entry of the auto‑convolution**
of a non‑negative vector a with fixed L1‑norm.

The algorithm works in three stages:

1. **Fast parametric sweep** – generate many analytically‑parameterised
   sequences (power‑law, exponential, Gaussian, polynomial, random Dirichlet)
   and keep the best one.  Evaluation is performed with a cheap FFT‑based
   convolution.

2. **Smoothed gradient descent** – minimise a smooth approximation of
   ``max(conv)`` (soft‑max with a temperature β).  The gradient of the smooth
   objective can be written as a correlation, which is also computed with FFT.
   After each step we project the vector back onto the simplex
   {x≥0, sum(x)=sqrt(2*n)}.  β is increased gradually so the optimiser first
   sees a smooth landscape and then focuses on the true maximum.

3. **Final local refinement** – a short simulated‑annealing / hill‑climbing
   phase with tiny mass transfers to escape possible tiny plateaus.

All heavy work is done with FFTs, allowing the optimiser to handle
sequences of several‑tens of thousands of entries well within the 1000 s
budget.  The function returns a list ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (O(n log n))
# ----------------------------------------------------------------------
def _project_to_simplex(x: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Project a vector onto the simplex { y >= 0 , sum(y) = target_sum }.
    """
    if target_sum <= 0:
        return np.zeros_like(x)

    n = x.shape[0]
    # shift so that the average equals target_sum / n
    # sort descending
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(x - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast convolution / correlation helpers (FFT based)
# ----------------------------------------------------------------------
def _conv_full(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution of a with itself (full length 2*len(a)-1) using FFT.
    """
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=L)
    B = A * A                                     # element‑wise product
    conv = np.fft.irfft(B, n=L)[: 2 * n - 1]
    return conv


def _correlate_full(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Correlation c[j] = sum_i a[i] * w[i+j]  for j = 0..len(a)-1.
    w must be the weight vector (same length as a*a convolution).
    """
    n = a.shape[0]
    m = w.shape[0]                               # = 2*n-1
    L = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(a, n=L)
    W = np.fft.rfft(w, n=L)
    corr_full = np.fft.irfft(A * np.conj(W), n=L)[: n + m - 1]
    start = m - 1                                # zero shift position
    return corr_full[start : start + n]


# ----------------------------------------------------------------------
# Objective (max of auto‑convolution)
# ----------------------------------------------------------------------
def _max_conv(a: np.ndarray) -> float:
    """Return max(conv(a,a)).  a must be non‑negative."""
    conv = _conv_full(a)
    return float(conv.max())


# ----------------------------------------------------------------------
# Parameterised families (used for the initial sweep)
# ----------------------------------------------------------------------
def _make_candidate(shape_type: str, n: int) -> np.ndarray:
    """Return a raw (non‑normalised) non‑negative vector of length n."""
    rng = np.random.default_rng()
    if shape_type == "power":
        p = rng.uniform(0.1, 1.2)
        raw = (np.arange(1, n + 1, dtype=float)) ** (-p)

    elif shape_type == "shifted_power":
        p = rng.uniform(0.1, 1.2)
        d = rng.uniform(0.0, n * 0.5) + 1e-9
        raw = (np.arange(n, dtype=float) + d) ** (-p)

    elif shape_type == "exp":
        c = rng.uniform(0.0005, 0.020) * n   # scale of decay
        raw = np.exp(-c * np.arange(n, dtype=float))

    elif shape_type == "gauss":
        mu = rng.uniform(0.2, 0.8) * n
        sigma = rng.uniform(0.05, 0.15) * n
        raw = np.exp(-0.5 * ((np.arange(n) - mu) / sigma) ** 2)

    elif shape_type == "poly":
        alpha = rng.uniform(0.2, 6.0)
        raw = (1.0 - np.arange(n, dtype=float) / (n - 1)) ** alpha

    elif shape_type == "linear":
        raw = np.linspace(1.0, 0.0, n, dtype=float)

    else:   # "rand_dir"
        raw = rng.gamma(shape=0.6, scale=1.0, size=n)

    raw = np.maximum(raw, 0.0)
    return raw


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list:
    """
    Return a list of non‑negative heights that (hopefully) minimises the
    evaluation function.  The routine respects the ``budget_s`` time limit.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
    else:
        n = 12000                     # a generous size that still stays fast
    target_sum = math.sqrt(2.0 * n)   # the normalising L1‑norm

    # initialise with a uniform vector (guaranteed feasible)
    best_seq = np.full(n, target_sum / n, dtype=float)
    best_score = _max_conv(best_seq)

    print(f"[init] n={n}, uniform score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – fast parametric sweep
    # --------------------------------------------------------------
    sweep_deadline = start_time + 0.15 * budget_s
    shape_choices = ["power", "shifted_power", "exp", "gauss",
                     "poly", "linear", "rand_dir"]
    it = 0
    while time.time() < sweep_deadline:
        it += 1
        shape = rng.choice(shape_choices)
        raw = _make_candidate(shape, n)
        # normalise to the required L1‑norm
        raw_sum = raw.sum()
        if raw_sum == 0:
            continue
        cand = raw * (target_sum / raw_sum)
        score = _max_conv(cand)

        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(f"[sweep {it:05d}] new best {best_score:.6f} (type={shape})")

        # occasional tiny random jitter to escape plateaus
        if it % 300 == 0 and rng.random() < 0.25:
            jitter = best_seq.copy()
            i, j = rng.choice(n, size=2, replace=False)
            delta = jitter[i] * rng.uniform(0.0, 0.02)
            jitter[i] -= delta
            jitter[j] += delta
            jitter = _project_to_simplex(jitter, target_sum)
            s = _max_conv(jitter)
            if s < best_score:
                best_score = s
                best_seq = jitter.copy()
                print(f"[jitter {it:05d}] improved to {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – smoothed gradient descent (several restarts)
    # --------------------------------------------------------------
    gd_deadline = start_time + 0.80 * budget_s   # keep ~20 % for final tweak
    restarts = 4
    repl = 0
    while time.time() < gd_deadline and repl < restarts:
        repl += 1
        # --- initial point for this restart ---
        if repl == 1 and best_seq is not None:
            a = best_seq.copy()
        else:
            # random Dirichlet start
            raw = rng.gamma(shape=0.5, scale=1.0, size=n)
            a = raw * (target_sum / raw.sum())

        # --- hyper‑parameters ---
        beta = 1.0                     # temperature for soft‑max
        step = target_sum / (n * 15.0) # initial step size (≈ 0.07·average)
        step_decay = 0.9994
        beta_mul = 1.025
        max_beta = 800.0

        it_gd = 0
        while time.time() < gd_deadline:
            it_gd += 1
            # compute convolution and current objective
            conv = _conv_full(a)
            cur_max = float(conv.max())

            # update global best
            if cur_max < best_score:
                best_score = cur_max
                best_seq = a.copy()
                print(f"[gd r{repl} it{it_gd}] new best {best_score:.6f}")

            # ----- gradient of smoothed max -----
            # soft‑max weights over convolution entries
            # shift by the maximal value for numerical stability
            shifted = conv - cur_max
            w = np.exp(beta * shifted)
            w_sum = w.sum()
            if w_sum == 0:                     # extremely unlikely
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # gradient = 2 * correlation(a, w)
            grad = 2.0 * _correlate_full(a, w)

            # projected gradient step
            a_new = a - step * grad
            a_new = _project_to_simplex(a_new, target_sum)

            # (optional) simple Metropolis acceptance – allows occasional uphill
            new_conv = _conv_full(a_new)
            new_max = float(new_conv.max())
            if new_max < cur_max or rng.random() < math.exp(
                -(new_max - cur_max) / (step + 1e-12)
            ):
                a = a_new
                # keep the new (maybe worse) max for the next iteration
                cur_max = new_max

            # schedule updates
            step *= step_decay
            beta = min(beta * beta_mul, max_beta)

            # stop condition for this restart (avoid endless loops)
            if it_gd >= 2500:          # enough iterations per restart
                break

    # --------------------------------------------------------------
    # Phase 3 – final local refinement (tiny moves)
    # --------------------------------------------------------------
    refine_deadline = deadline - 0.5   # leave a safety margin
    a = best_seq.copy()
    cur_score = _max_conv(a)
    temperature = 0.03
    decay_temp = 0.9996

    while time.time() < refine_deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # transfer a small fraction of mass from i to j
        max_move = a[i] * 0.025
        if max_move <= 0:
            continue
        delta = rng.random() * max_move
        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta
        a_candidate = _project_to_simplex(a_candidate, target_sum)

        cand_score = _max_conv(a_candidate)

        if cand_score < cur_score:
            a = a_candidate
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = a_candidate.copy()
                print(f"[refine] improved to {best_score:.6f}")
        else:
            # simulated‑annealing style acceptance
            prob = math.exp(-(cand_score - cur_score) / (temperature + 1e-12))
            if rng.random() < prob:
                a = a_candidate
                cur_score = cand_score

        temperature *= decay_temp

    # --------------------------------------------------------------
    # Return the best sequence found (project once more for safety)
    # --------------------------------------------------------------
    best_seq = _project_to_simplex(best_seq, target_sum)
    print(f"[final] best score = {best_score:.6f}")
    return best_seq.tolist()
