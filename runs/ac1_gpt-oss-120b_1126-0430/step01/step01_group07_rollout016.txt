def project_simplex(v):
    # v is array, returns w >=0 sum=1
    # reference: https://github.com/eric-wieser/fastproj There's simple code:
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def project_simplex(v):
    # returns a projection onto the probability simplex (nonnegative, sum=1)
    n = len(v)
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    # Find rho = max{i | u_i - cssv_i / i > 0}
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero? return uniform
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix

def convolve_fft(a, b):
    """Linear convolution using FFT (real)."""
    size = len(a) + len(b) - 1
    N = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    C = A * B
    conv = np.fft.irfft(C, N)[:size]
    return conv

def power_law_sequence(alpha, delta, n):
    """Return a non‑negative vector a_i ∝ (i+δ)^{‑α}, i=1..n, normalized to sum = 1."""
    idx = np.arange(1.0, n + 1) + delta
    a = idx ** (-alpha)
    s = a.sum()
    if s > 0.0:
        a /= s
    return a

def project_simplex(v):
    """Projection of vector v onto the probability simplex {x≥0, sum x = 1}."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # return uniform distribution if all entries are non‑positive
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def frank_wolfe_step(f, M, t):
    """Given current f (sum=1) and current max value M,
    solve the LP max Σ g_i s.t. g≥0 and f∗g ≤ M.
    Return the convex‑combination (1‑t)·f + t·g (renormalised)."""
    n = f.shape[0]
    rows = 2 * n - 1

    # build sparse matrix A where (A g)_j = (f * g)[j]
    A = lil_matrix((rows, n), dtype=np.float64)
    # each column i contributes f shifted by i
    for i in range(n):
        A[i:i + n, i] = f
    A = A.tocsr()

    # right‑hand side
    b = np.full(rows, M, dtype=np.float64)

    # linear programme: minimise -1·g  (i.e. maximise Σ g_i)
    c = -np.ones(n, dtype=np.float64)

    # bounds g_i ≥ 0
    bounds = [(0.0, None)] * n

    # solve using HiGHS (the default for recent SciPy)
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True, 'time_limit': 30})
    if not res.success:
        # fall back to current f if LP failed
        return f.copy()
    g = res.x
    # normalise g to sum = 1 (LP may give a vector with sum < 1)
    s = g.sum()
    if s <= 0:
        g = np.full(n, 1.0 / n)
    else:
        g = g / s

    # convex combination
    f_new = (1.0 - t) * f + t * g
    # numerical cleanup – project back to the simplex
    f_new = project_simplex(f_new)
    return f_new

"""
Frank‑Wolfe optimiser for the convolution‑max objective.

The objective to minimise is
    2·n·max(conv(a,a)) / (sum(a)^2)

Because the expression is scale‑invariant we keep the sequence on the
probability simplex (non‑negative, sum = 1).  Starting from a 1/√x‑like
power‑law we repeatedly solve the linear programme

    maximise   Σ g_i
    subject to g_i ≥ 0  and  (f ∗ g)_k ≤ M   for all k,

where M = max(f ∗ f) for the current iterate f.  The solution g is the
best feasible partner for f; a conditional‑gradient (Frank‑Wolfe)
update f ← (1‑t)·f + t·g (t slowly decays) reduces the maximal
convolution.  A few projected‑gradient polishing steps are performed
at the end.  The whole procedure stops after the allotted time budget
(~1000 s) and returns the best sequence found as a plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via FFT (real inputs)."""
    size = len(a) + len(b) - 1
    N = 1 << (size - 1).bit_length()          # next power of two
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    C = A * B
    conv = np.fft.irfft(C, N)[:size]
    return conv


def power_law_sequence(alpha: float, delta: float, n: int) -> np.ndarray:
    """
    a_i ∝ (i+δ)^{-α},   i = 1 … n,   normalised to sum = 1.
    """
    idx = np.arange(1.0, n + 1) + delta
    a = idx ** (-alpha)
    s = a.sum()
    if s > 0.0:
        a /= s
    return a


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x≥0, Σx=1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def frank_wolfe_step(f: np.ndarray, M: float, t: float) -> np.ndarray:
    """
    One Frank‑Wolfe conditional‑gradient step.
    Solve   maximise Σ g_i   s.t.   g≥0 and (f∗g) ≤ M.
    Returns the convex combination (1‑t)·f + t·g, projected onto the simplex.
    """
    n = f.size
    rows = 2 * n - 1

    # Build the (rows × n) sparse Toeplitz matrix that realises the convolution
    A = lil_matrix((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f                # column i = f shifted by i
    A = csr_matrix(A)

    # Right‑hand side: M for every convolution entry
    b = np.full(rows, M, dtype=np.float64)

    # Maximise Σ g_i  <=> minimise −Σ g_i
    c = -np.ones(n, dtype=np.float64)

    # Bounds: g_i ≥ 0
    bounds = [(0.0, None)] * n

    # Solve the linear programme (HiGHS is the default for recent SciPy)
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 30})
    if not res.success:
        # If the optimiser failed we fall back to the current point
        return f.copy()

    g = res.x
    s = g.sum()
    if s <= 0.0:
        g = np.full(n, 1.0 / n)
    else:
        g = g / s

    # Convex combination and projection back onto the simplex
    f_new = (1.0 - t) * f + t * g
    f_new = project_simplex(f_new)
    return f_new

    Works with the sub‑gradient of the L∞‑norm (the entries attaining the maximum).
    """
    n = a.size
    best = a.copy()
    best_score = evaluate_sequence(best.tolist())
    step = 0.1

    while time.time() < deadline:
        b = convolve_fft(a, a)
        max_val = b.max()
        # indices of (near) maximal entries – a tiny tolerance catches ties
        max_idx = np.flatnonzero(b >= max_val - 1e-14)

        # sub‑gradient of max(conv) w.r.t a
        g = np.zeros_like(a)
        for m in max_idx:
            lo = max(0, m - (n - 1))
            hi = min(m, n - 1)
            # g[lo:hi+1] += 2 * a[m - np.arange(lo, hi+1)]
            idxs = np.arange(lo, hi + 1)
            g[lo:hi + 1] += 2.0 * a[m - idxs]

        gn = np.linalg.norm(g)
        if gn == 0.0:
            break
        eta = step / gn
        a = a - eta * g
        a = np.maximum(a, 0.0)
        a = project_simplex(a)

        cur_score = evaluate_sequence(a.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f'  gradient polish improved to {cur_score:.6f}')
            step = max(step * 0.9, 5e-4)

    return best


def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Main entry point required by the judge.
    Performs a Frank‑Wolfe optimisation (LP‑based conditional gradient)
    starting from several diverse initial points, followed by a short
    gradient polishing phase.  The best sequence discovered before the
    time budget expires is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # -----------------------------------------------------------
    # problem size – large enough to approach the theoretical optimum,
    # small enough that the linear programme stays cheap.
    n = 1800
    # -----------------------------------------------------------

    # ---- a few diverse starting points ---------------------------------
    starters = [
        power_law_sequence(alpha=0.5, delta=0.0, n=n),   # 1/√x shape
        np.full(n, 1.0 / n),                           # uniform
        np.random.rand(n)                              # random Dirichlet‑like
    ]
    starters[-1] = starters[-1] / starters[-1].sum()

    best_seq = None
    best_score = float('inf')

    # --------------------------------------------------------------------
    for start_id, a0 in enumerate(starters):
        if time.time() >= deadline:
            break
        a = project_simplex(a0.copy())
        cur_score = evaluate_sequence(a.tolist())
        print(f'Start {start_id}: initial score = {cur_score:.6f}')
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()

        # ---- Frank‑Wolfe loop -----------------------------------------
        t = 0.2                     # initial step size
        it = 0
        while time.time() < deadline and it < 200:
            conv_f = convolve_fft(a, a)
            M = conv_f.max()

            # linear programme to find the best partner g
            a = frank_wolfe_step(a, M, t)

            new_score = evaluate_sequence(a.tolist())
            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
                print(f'Start {start_id}, iter {it}: improved = {new_score:.6f}')
                # shrink step size modestly after a successful improvement
                t = max(t * 0.9, 0.01)
            else:
                # otherwise let the step size decay slowly
                t = min(t * 1.02, 0.5)

            it += 1

    # --------------------------------------------------------------------
    # final polishing (only if time permits)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    if time.time() < deadline:
        print('Starting final gradient polish...')
        best_seq = gradient_refine(best_seq, deadline)

    final_score = evaluate_sequence(best_seq.tolist())
    print(f'final reported score = {final_score:.6f}')
    return best_seq.tolist()
