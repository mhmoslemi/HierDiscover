"""
Mirrored-Gradient (Exponential) Descent for the self‑convolution bound.

The search works directly on the probability simplex (∑a_i = 1,
a_i ≥ 0).  For a fixed sum the evaluation function reduces to

    O(a) = 2·n·max(conv(a,a))

which is scale‑invariant.  We minimise a smooth approximation of the
maximum using a soft‑max weight vector

    w_k ∝ exp(α·(conv_k – max(conv))).

The gradient of the weighted sum ∑_k w_k·conv_k w.r.t. a is
g_i = 2·∑_k w_k·a_{k−i}.  Using the convolution theorem the gradient
is obtained by a single FFT‑based convolution of the weight vector with
the reversed a.  An exponential (mirror) update

    a_i ← a_i·exp(−η·g_i)

keeps the variables non‑negative; a final renormalisation restores the
simplex condition.  The step‑size η and the soft‑max temperature α are
decreased / increased linearly during the run.  Several random
restarts are performed for a few different sequence lengths; the best
sequence found (lowest objective) is returned.

The implementation respects the 1000 s budget, prints occasional
progress reports and never performs any I/O other than printing.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based utilities
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_self_fft(a: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Full convolution a * a using the FFT.
    Returns an array of length 2·len(a)‑1.
    """
    fa = np.fft.rfft(a, n=size_fft)
    conv = np.fft.irfft(fa * fa, n=size_fft)
    n = a.shape[0]
    return conv[: 2 * n - 1]


def _grad_weighted_fft(a: np.ndarray, w: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Gradient of Σ_k w_k·conv_k(a,a) where conv_k = (a∗a)_k.
    Formula: g_i = 2· Σ_k w_k·a_{k−i}.
    Computed by a single FFT‑based convolution of w with the reversed a.
    Returns a vector of length len(a).
    """
    n = a.shape[0]
    a_rev = a[::-1]
    fw = np.fft.rfft(w, n=size_fft)
    fa_rev = np.fft.rfft(a_rev, n=size_fft)
    full = np.fft.irfft(fw * fa_rev, n=size_fft)
    start = n - 1
    grad = 2.0 * full[start : start + n]
    return grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence``.  The method uses exponential
    mirror descent on a soft‑max approximation of the maximal
    self‑convolution entry, runs several random restarts for a few
    different sequence lengths and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Choose a set of candidate lengths.  Include the length of any
    # previously discovered construction if available.
    lengths = [2000, 3000, 4000, 5000, 6000, 8000]
    if "height_sequence_1" in globals():
        lengths.append(len(height_sequence_1))
    # make them unique and sorted – smaller lengths are cheaper, larger
    # lengths potentially give better bounds.
    lengths = sorted(set(lengths))

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Main outer loop over different lengths.
    for n in lengths:
        if time.time() >= deadline:
            break

        # Pre‑compute the FFT size (must accommodate 3·n‑2 for the gradient).
        size_fft = _next_pow2(3 * n - 2)

        # Simple per‑length budget: spend at most a fraction of the remaining time.
        # The fraction shrinks as we move to larger n.
        remaining = deadline - time.time()
        # 0.6 of the remaining time for the first length, then less for later.
        budget_this_n = remaining * 0.6 / max(1, len(lengths) - lengths.index(n))

        end_n = time.time() + budget_this_n

        restart = 0
        while time.time() < end_n:
            restart += 1

            # ----- initial point on the simplex ---------------------------------
            if best_seq is not None and len(best_seq) == n and np.random.rand() < 0.3:
                a = np.array(best_seq, copy=True)
                a += np.random.randn(n) * 0.005  # tiny Gaussian jitter
                a = np.clip(a, 0, None)
                a /= a.sum()
            else:
                # Dirichlet(1,…,1) gives a uniform random point on the simplex.
                a = np.random.dirichlet(np.ones(n))

            # ---- hyper‑parameters for this restart ----------------------------
            max_iter = 20000
            eta0 = 0.5                     # initial step size for mirror update
            alpha = 5.0                    # initial soft‑max temperature
            momentum = np.zeros_like(a)   # simple momentum buffer
            beta = 0.9                     # momentum factor

            # ---- iterative optimisation ----------------------------------------
            for it in range(1, max_iter + 1):
                if time.time() >= deadline:
                    break

                # (1) current convolution and objective
                conv = _conv_self_fft(a, size_fft)
                max_conv = conv.max()
                cur_val = 2.0 * n * max_conv
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] New global best O={best_val:.6f} "
                          f"(n={n}, restart={restart}, it={it})")

                # (2) soft‑max weights over convolution entries
                #   w_k ∝ exp(α·(conv_k – max_conv))
                # subtract max_conv for numerical stability.
                exp_arg = alpha * (conv - max_conv)
                # clip to avoid underflow for very negative numbers
                exp_arg = np.clip(exp_arg, -700, 0)   # np.exp(-700) ≈ 5e-305
                w = np.exp(exp_arg)
                w_sum = w.sum()
                if w_sum == 0.0:          # should not happen, fallback to hard max
                    w = np.zeros_like(conv)
                    w[conv == max_conv] = 1.0 / np.sum(conv == max_conv)
                else:
                    w /= w_sum

                # (3) gradient of the weighted sum of convolutions
                grad = _grad_weighted_fft(a, w, size_fft)

                # (4) momentum‑augmented mirror step
                momentum = beta * momentum + (1.0 - beta) * grad
                eta = eta0 * (1.0 - it / max_iter)          # linear decay
                step = -eta * momentum
                a = a * np.exp(step)                       # exponential update
                a = np.clip(a, 1e-15, None)                # avoid exact zeros
                a /= a.sum()                               # project onto simplex

                # (5) schedule adjustments
                if it % 2000 == 0:
                    alpha *= 1.4       # sharpen the focus on the peak
                if it % 5000 == 0:
                    # occasional random perturbation to escape shallow flats
                    noise = np.random.dirichlet(np.ones(n))
                    a = 0.9 * a + 0.1 * noise
                    a /= a.sum()

                # early‑stop a restart if there has been no improvement for long
                # (simple heuristic: every 3000 iterations compare with a saved value)
                if it % 3000 == 0:
                    # compute a cheap bound – if current value is already far
                    # above the best global bound, abort this restart.
                    if cur_val > best_val * 1.05:
                        break

    # ------------------------------------------------------------------
    # Final reporting using the provided evaluate_sequence (safety check)
    if best_seq is None:
        # fallback – constant sequence (should never happen)
        best_seq = np.full(2000, 1.0 / 2000)

    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_seq.tolist()
