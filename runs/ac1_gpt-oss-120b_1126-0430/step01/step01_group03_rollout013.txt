"""
High‑quality step‑function proposal.

We treat the sequence as a probability vector p (∑p = 1).  
The evaluation score equals 2·n·max_k (p∗p)_k, i.e. we must
flatten the auto‑convolution of p.  

Algorithm
---------
1. Initialise the best known candidate (global `height_sequence_1` if any).
2. Generate a small portfolio of seeds:
   * power‑law a_i ∝ (i+1)^{-β}
   * exponential a_i ∝ exp(-λ·i)
   * sinusoidal a_i ∝ sin^γ
   * random Dirichlet draws
   All seeds are normalised to sum = 1 and scored with a fast
   internal evaluator.
   The five best seeds are kept.
3. For each kept seed run a projected gradient descent that minimises a
   smooth approximation of the maximal self‑convolution:
      • use a soft‑max with temperature α (increases during optimisation)
      • gradient = 2·corr(weight, p)  (corr = np.correlate(...,'valid'))
      • take a step, then project back onto the probability simplex.
   The iterate with the smallest true max‑convolution is recorded.
4. Perform a lightweight stochastic hill‑climb on the overall best
   sequence (mass transfers between random indices) until the time budget
   is exhausted.
5. Return the best sequence found.

The whole routine respects the supplied time budget (default 1000 s)
and never produces negative numbers; all values stay far below the
hard limit 1000.
"""

import time
import numpy as np
import heapq


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, sum x = 1}."""
    n = v.size
    # Sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # Find the right rho
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # Numerical safety: return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _score_sequence(seq: np.ndarray) -> float:
    """Fast replica of evaluate_sequence (no safety checks needed)."""
    n = seq.shape[0]
    total = np.sum(seq)
    if total < 0.01:
        return float("inf")
    conv = np.convolve(seq, seq)
    max_conv = np.max(conv)
    return float(2.0 * n * max_conv / (total * total))


def _gradient_descent(p0: np.ndarray, deadline: float,
                     max_iter: int = 10000) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the smoothed max‐convolution objective.
    Returns the best (p, cost) encountered.
    """
    p = p0.copy()
    n = p.shape[0]
    best_p = p.copy()
    best_cost = np.max(np.convolve(p, p))

    # hyper‑parameters (they adapt during the run)
    alpha = 5.0          # soft‑max temperature (the larger → sharper)
    eta = 0.5            # initial step length

    for it in range(max_iter):
        if time.time() > deadline:
            break

        # convolution and soft‑max weights
        conv = np.convolve(p, p)
        max_val = np.max(conv)
        # stability shift before exponentiation
        shifted = conv - max_val
        w_exp = np.exp(alpha * shifted)      # weights concentrate on largest entries
        w = w_exp / np.sum(w_exp)

        # gradient of the smooth objective (see analysis)
        grad = 2.0 * np.correlate(w, p, mode='valid')

        # gradient step and projection onto simplex
        p_new = p - eta * grad
        p_new = _project_simplex(p_new)

        # optional symmetrisation – improves stability for many seeds
        p_new = (p_new + p_new[::-1]) * 0.5
        p_new = _project_simplex(p_new)

        # true cost (max‑convolution) for monitoring
        cur_cost = np.max(np.convolve(p_new, p_new))

        # keep the best encountered point
        if cur_cost < best_cost:
            best_cost = cur_cost
            best_p = p_new.copy()

        # simple adaptive scheme for step‑size and temperature
        if cur_cost > best_cost:
            eta *= 0.7               # back‑off if we got worse
        else:
            eta = min(eta * 1.02, 2.0)   # modest acceleration when improving

        if (it + 1) % 50 == 0:
            alpha = min(alpha * 1.05, 400.0)   # sharpen the soft‑max gradually

        p = p_new

    return best_p, best_cost


def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """Return a list of non‑negative heights with an improved evaluation score."""
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.1     # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0.  Use the previous construction if it exists.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(height_sequence_1, dtype=np.float64)
            score = _score_sequence(cand)
            best_seq = cand.tolist()
            best_score = score
            print(f"[init] height_sequence_1 score = {score:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Generate a pool of seed sequences.
    # ------------------------------------------------------------------
    n = 2000                     # fixed length (balances quality / speed)
    top_k = 5                    # retain the top‑k seeds for refinement
    heap = []                    # max‑heap holding (-score, seq) pairs

    def push_candidate(seq: np.ndarray, score: float):
        """Maintain a fixed‑size heap of the best candidates."""
        if len(heap) < top_k:
            heapq.heappush(heap, (-score, seq))
        else:
            # heap stores negative scores → smallest (worst) at top
            if -heap[0][0] > score:
                heapq.heapreplace(heap, (-score, seq))

    # ----- power‑law family ------------------------------------------------
    betas = np.linspace(0.44, 0.78, 71)   # fairly fine grid around the known optimum
    for beta in betas:
        if time.time() > deadline:
            break
        seq = (np.arange(1, n + 1, dtype=np.float64) ** (-beta))
        seq /= seq.sum()
        score = _score_sequence(seq)
        if score < best_score:
            best_score = score
            best_seq = seq.tolist()
            print(f"[seed‑power] β={beta:.3f} → score={score:.6f}")
        push_candidate(seq, score)

    # ----- exponential decay family ----------------------------------------
    lambdas = np.linspace(0.0008, 0.008, 32)
    for lam in lambdas:
        if time.time() > deadline:
            break
        seq = np.exp(-lam * np.arange(n, dtype=np.float64))
        seq /= seq.sum()
        score = _score_sequence(seq)
        if score < best_score:
            best_score = score
            best_seq = seq.tolist()
            print(f"[seed‑exp] λ={lam:.5f} → score={score:.6f}")
        push_candidate(seq, score)

    # ----- sinusoidal family (raised sine) ---------------------------------
    gammas = [1, 2, 3]
    for gamma in gammas:
        if time.time() > deadline:
            break
        x = np.arange(1, n + 1, dtype=np.float64) / (n + 1)
        seq = np.sin(np.pi * x) ** gamma
        seq /= seq.sum()
        score = _score_sequence(seq)
        if score < best_score:
            best_score = score
            best_seq = seq.tolist()
            print(f"[seed‑sin] γ={gamma} → score={score:.6f}")
        push_candidate(seq, score)

    # ----- random Dirichlet draws ------------------------------------------
    for _ in range(15):
        if time.time() > deadline:
            break
        seq = np.random.dirichlet(np.ones(n))
        score = _score_sequence(seq)
        if score < best_score:
            best_score = score
            best_seq = seq.tolist()
            print(f"[seed‑rand] → score={score:.6f}")
        push_candidate(seq, score)

    # ------------------------------------------------------------------
    # 2.  Refine the top seeds using projected gradient descent.
    # ------------------------------------------------------------------
    # Extract candidates from heap (they are stored with negative scores)
    top_candidates = [seq for (_, seq) in heap]

    for idx, seed_seq in enumerate(top_candidates):
        if time.time() > deadline:
            break
        print(f"[refine] start candidate {idx + 1}/{len(top_candidates)}")
        refined_seq, refined_cost = _gradient_descent(seed_seq, deadline)
        refined_score = 2.0 * n * refined_cost    # full evaluation (sum = 1)
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.tolist()
            print(f"[refine] improvement → score={refined_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Light hill‑climb on the overall best solution.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        seq = np.asarray(best_seq, dtype=np.float64)
        current_score = best_score
        # Simple stochastic mass‑transfer moves (as in the reference code)
        while time.time() < deadline:
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            # transfer up to 4 % of the source entry
            eps = seq[i] * np.random.rand() * 0.04
            if eps <= 0:
                continue
            new_seq = seq.copy()
            new_seq[i] -= eps
            new_seq[j] += eps
            # re‑normalise to maintain sum≈1 (projection onto simplex does that)
            new_seq = _project_simplex(new_seq)
            new_score = _score_sequence(new_seq)
            if new_score < current_score:
                seq = new_seq
                current_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_seq = new_seq.tolist()
                    print(f"[hill] new best → {new_score:.6f}")

    print(f"[final] best score = {best_score:.6f}, length = {len(best_seq)}")
    return best_seq
