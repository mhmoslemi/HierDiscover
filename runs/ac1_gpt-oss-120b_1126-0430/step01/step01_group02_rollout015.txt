"""
Propose a candidate sequence that improves on the Catalan‐based baseline.

The algorithm works in three stages:

1. **Initialisation** – start from the Catalan sequence (a_i = C(2i,i)/4^i
   normalised to sum 1) and break the perfect symmetry by adding a tiny random
   perturbation.  The Catalan construction makes the convolution almost flat,
   which is already close to optimal (≈ π/2).  A small perturbation gives the
   optimiser a direction to move.

2. **Smoothed maximum descent** – the true objective is
   ``max_k (a*a)_k``.  Its sub‑gradient is very sparse (only the indices that
   attain the maximum contribute).  To obtain a more informative gradient we
   minimise a *smooth* approximation of the max:
        g(a) = ( Σ ( (a*a)_k / max(a*a) )^{p-1} )^{1/(p-1)} .
   For a large exponent ``p`` the weights concentrate on the largest convolution
   entries, but for moderate ``p`` they still incorporate nearby values.
   The gradient of this surrogate is
        ∇g_i = 2 Σ_k w_k·a_{k-i},
   where ``w_k ∝ ((a*a)_k / max)^ {p-1}`` and the sum is over all valid
   indices.  The convolution needed for the gradient is computed efficiently
   with ``scipy.signal.fftconvolve``.

   An Adam optimiser (adaptive learning rates) is applied to the gradient,
   followed by a projection onto the probability simplex (clip to non‑negative
   and renormalise to sum 1).  The exponent ``p`` is increased gradually
   (continuation method) so the optimisation first smooths the landscape and
   later focuses on the true L∞‑objective.

3. **Exploration & restarts** – if no improvement is seen for a while,
   the current iterate is perturbed randomly (or re‑initialised with a fresh
   Dirichlet sample) to escape possible local minima.  The best sequence ever
   observed is kept and returned after the time budget expires.

The whole routine respects the 1000 s budget, prints progress, and returns a
plain ``list[float]`` suitable for ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.signal import fftconvolve


def _catalan_sequence(n: int) -> np.ndarray:
    """First n Catalan‑like coefficients a_i = C(2i,i)/4**i, normalised to sum = 1."""
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * (i - 1) + 1) / (2.0 * i)
    a /= a.sum()
    return a


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the simplex {x≥0, Σx=1}."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 0.0:
        return v / s
    # degenerate – fall back to a random point on the simplex
    u = np.random.rand(v.shape[0])
    return u / u.sum()


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The procedure combines a smoothed max‑descent (Adam) with occasional random
    restarts.  It returns the best sequence found before the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # a small safety margin

    # --------------------------------------------------------------
    # Problem size – a few thousand entries give a good trade‑off.
    # --------------------------------------------------------------
    n = 3000                     # length of the sequence
    # initialise from the Catalan construction and add tiny noise
    a = _catalan_sequence(n)
    a *= 1.0 + 0.01 * (np.random.rand(n) - 0.5)   # break perfect symmetry
    a = _project_simplex(a)

    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # Adam optimiser state
    lr = 0.6                     # base learning rate
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # continuation schedule for the smoothing exponent p
    p = 2.0                      # start with a smooth surrogate
    p_target = 16.0
    p_step = 2.0                 # increase every few thousand iterations

    iteration = 0
    no_improve = 0

    while time.time() < deadline:
        # ----------------------------------------------------------
        # 1) convolution of the current sequence
        # ----------------------------------------------------------
        conv = fftconvolve(a, a, mode='full')
        max_c = conv.max()

        # ----------------------------------------------------------
        # 2) compute smooth weights that approximate the max
        # ----------------------------------------------------------
        rel = conv / max_c                     # values in [0,1]
        # avoid division by zero when max_c is extremely small
        if max_c == 0.0:
            rel = np.zeros_like(conv)

        # weight proportional to rel^{p-1}
        w = rel ** (p - 1)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.ones_like(w) / w.size
        else:
            w /= w_sum

        # ----------------------------------------------------------
        # 3) gradient of the smooth surrogate
        #    grad_i = 2 * Σ_k w_k * a_{k-i}
        # ----------------------------------------------------------
        grad_full = fftconvolve(w, a[::-1], mode='full')
        # extract the n relevant entries (indices n-1 … n-1+n-1)
        grad = 2.0 * grad_full[n - 1:n - 1 + n]

        # ----------------------------------------------------------
        # 4) Adam update, then projection onto the simplex
        # ----------------------------------------------------------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a)

        # ----------------------------------------------------------
        # 5) evaluate and keep the best sequence
        # ----------------------------------------------------------
        cur_score = evaluate_sequence(a.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            no_improve = 0
            print(f"[{iteration:06d}] new best = {best_score:.6f}")
        else:
            no_improve += 1

        # ----------------------------------------------------------
        # 6) schedule adjustments
        # ----------------------------------------------------------
        # decay learning rate slowly
        if iteration % 800 == 0:
            lr *= 0.90

        # increase the exponent p towards the true L∞ objective
        if iteration % 2000 == 0 and p < p_target:
            p = min(p + p_step, p_target)
            print(f"[{iteration:06d}] increased p to {p:.1f}")

        # occasional random perturbation if we seem stuck
        if no_improve > 5000:
            # restart from the current best with a fresh jitter
            a = best_seq * (1.0 + 0.05 * (np.random.rand(n) - 0.5))
            a = _project_simplex(a)
            # reset Adam state
            m.fill(0.0)
            v.fill(0.0)
            lr = 0.6
            p = 2.0
            no_improve = 0
            print(f"[{iteration:06d}] restart (jitter from best)")

    print(f"[done] best score found = {best_score:.6f}")
    return best_seq.tolist()
