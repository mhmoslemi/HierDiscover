"""...
Algorithm:
1. Prepare starting sequence via random families or existing best.
2. Perform a fast synthetic evaluation loop that repeatedly:
   - computes the convolution peak (max_conv)
   - picks a move type:
        * targeted pairwise reduction of the dominant product
        * sub‑gradient descent step (move mass opposite to gradient of the peak)
   - projects the resulting vector onto the probability simplex (preserving the L1‑norm sqrt(2n))
   - accepts the move if it reduces max_conv, otherwise discards (or occasionally accepts for exploration)
   - keeps track of the best sequence seen.
3. The loop runs until the time budget expires (or until the target 1.503 is reached).
All computations use only NumPy (no external LPs) so a large number of iterations fit in the 1000 s budget.

import time
import math
import numpy as np

def _project_to_simplex(v, target_sum):
    """Project a non‑negative vector onto the simplex {x>=0, sum(x)=target_sum}."""
    # implementation from Wang & Carreira‑Perpiñán (2013)
    if target_sum <= 0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _max_conv(a):
    """Return the maximum value of the linear convolution a*a."""
    return np.max(np.convolve(a, a))

def _subgradient_max(a, max_idx):
    """Sub‑gradient of max(conv(a,a)) with respect to a.  Uses the unique maximiser."""
    n = a.shape[0]
    g = np.zeros(n, dtype=float)
    i_start = max(0, max_idx - (n - 1))
    i_end = min(n - 1, max_idx)
    idx = np.arange(i_start, i_end + 1, dtype=int)
    # for each k in idx we have g[k] = 2 * a[max_idx - k]
    g[idx] = 2.0 * a[max_idx - idx]
    return g

def _targeted_pair_move(a, target_sum, max_idx, rho=0.3):
    """
    Reduce the dominant product that creates the current peak.
    We pick the pair (i, j) with i+j = max_idx that maximises a_i * a_j,
    take a fraction ``rho`` of both entries and relocate the mass to a
    position that does not take part in the same convolution bin.
    """
    n = a.shape[0]
    i_start = max(0, max_idx - (n - 1))
    i_end = min(n - 1, max_idx)
    idx = np.arange(i_start, i_end + 1, dtype=int)
    counterpart = max_idx - idx
    products = a[idx] * a[counterpart]
    maxpos = np.argmax(products)
    i = int(idx[maxpos])
    j = int(counterpart[maxpos])

    # destination index – prefer an index that is *outside* the active band
    if n > (i_end - i_start + 1):
        # there are indices that never appear in the current maximal band
        outside = np.concatenate([np.arange(0, i_start), np.arange(i_end + 1, n)])
        dest = int(np.random.choice(outside))
    else:
        # fallback – any index different from i and j
        candidates = np.delete(np.arange(n), [i, j])
        dest = int(np.random.choice(candidates))

    # mass to move
    if i == j:
        # only one index contributes (the centre of the band)
        delta = a[i] * rho
        delta_i = delta_j = delta
    else:
        delta_i = a[i] * rho
        delta_j = a[j] * rho

    # make sure we do not create negative entries
    delta_i = min(delta_i, a[i])
    delta_j = min(delta_j, a[j])

    a_new = a.copy()
    a_new[i] -= delta_i
    a_new[j] -= delta_j
    a_new[dest] += delta_i + delta_j

    # keep the vector on the simplex (also fixes tiny numerical negatives)
    a_new = _project_to_simplex(a_new, target_sum)
    return a_new

def _random_family(n, target_sum):
    """Generate a random non‑negative vector using a few analytic families."""
    shape_type = np.random.choice(
        ["power", "shifted_power", "poly", "linear", "rand_gamma", "rand_dir"]
    )
    if shape_type == "power":
        p = np.random.uniform(0.1, 1.0)
        raw = (np.arange(1, n + 1, dtype=float)) ** (-p)
    elif shape_type == "shifted_power":
        p = np.random.uniform(0.1, 1.0)
        d = np.random.uniform(0.0, n * 0.5) + 1e-6
        raw = (np.arange(n, dtype=float) + d) ** (-p)
    elif shape_type == "poly":
        alpha = np.random.uniform(0.1, 5.0)
        raw = (1.0 - np.arange(n, dtype=float) / (n - 1)) ** alpha
    elif shape_type == "linear":
        raw = np.linspace(1.0, 0.0, n, dtype=float)
    elif shape_type == "rand_gamma":
        raw = np.random.gamma(shape=0.7, scale=1.0, size=n)
    else:  # rand_dir
        raw = np.random.gamma(shape=0.7, scale=1.0, size=n)
    raw = np.maximum(raw, 0.0)
    # scale to required L1 norm
    raw_sum = raw.sum()
    if raw_sum == 0:
        raw = np.full(n, 1.0 / n)
        raw_sum = 1.0
    raw = raw * (target_sum / raw_sum)
    return raw.tolist()

"""
Propose a step‑function that minimises the peak of its self‑convolution.
The score used by the grader is

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

and the code protects the L1‑norm to be ≈ sqrt(2 n).  Hence the
objective is essentially `max(conv(a, a))`.  The optimiser combines

* a quick random‑family sweep to obtain a decent start,
* a *sub‑gradient* step that moves the current vector opposite the
  gradient of the active convolution bin,
* a *targeted pairwise* move that directly reduces the dominant product
  (`a_i * a_{max‑i}`) that creates the peak.

Both moves keep the L1‑norm fixed by projecting onto the simplex
{ x ≥ 0, Σx = sqrt(2 n) }.  The loop runs until the time budget expires
or a score ≤ 1.503 is reached.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_to_simplex(v, target_sum):
    """Project a non‑negative vector onto the simplex {x≥0, Σx=target_sum}."""
    if target_sum <= 0:
        return np.zeros_like(v)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_conv(a):
    """Return the maximum value of the linear convolution a*a."""
    return np.max(np.convolve(a, a))


def _subgradient_max(a, max_idx):
    """
    Sub‑gradient of max(conv(a, a)) at the current maximising index.
    For a unique maximiser the gradient is:
        g_k = 2 * a_{max_idx - k}   if 0≤max_idx‑k<n, else 0
    """
    n = a.shape[0]
    g = np.zeros(n, dtype=float)
    i_start = max(0, max_idx - (n - 1))
    i_end = min(n - 1, max_idx)
    idx = np.arange(i_start, i_end + 1, dtype=int)
    g[idx] = 2.0 * a[max_idx - idx]
    return g


def _targeted_pair_move(a, target_sum, max_idx, rho=0.3):
    """
    Reduce the dominant product that creates the current peak.
    Find (i, j) with i+j = max_idx that maximises a_i*a_j,
    take a fraction `rho` of both entries and relocate the mass to a
    position that does not belong to the same convolution bin.
    """
    n = a.shape[0]
    i_start = max(0, max_idx - (n - 1))
    i_end = min(n - 1, max_idx)

    idx = np.arange(i_start, i_end + 1, dtype=int)
    counterpart = max_idx - idx
    products = a[idx] * a[counterpart]
    maxpos = np.argmax(products)
    i = int(idx[maxpos])
    j = int(counterpart[maxpos])

    # destination – preferably outside the active band
    if n > (i_end - i_start + 1):
        outside = np.concatenate([np.arange(0, i_start), np.arange(i_end + 1, n)])
        dest = int(np.random.choice(outside))
    else:
        candidates = np.delete(np.arange(n), [i, j])
        dest = int(np.random.choice(candidates))

    if i == j:
        delta = a[i] * rho
        delta_i = delta_j = delta
    else:
        delta_i = a[i] * rho
        delta_j = a[j] * rho

    delta_i = min(delta_i, a[i])
    delta_j = min(delta_j, a[j])

    a_new = a.copy()
    a_new[i] -= delta_i
    a_new[j] -= delta_j
    a_new[dest] += delta_i + delta_j

    # keep the vector on the simplex (fix tiny negatives)
    a_new = _project_to_simplex(a_new, target_sum)
    return a_new


def _random_family(n, target_sum):
    """Generate a random non‑negative vector from a few analytic families."""
    shape_type = np.random.choice(
        ["power", "shifted_power", "poly", "linear", "rand_gamma", "rand_dir"]
    )
    if shape_type == "power":
        p = np.random.uniform(0.1, 1.0)
        raw = (np.arange(1, n + 1, dtype=float)) ** (-p)

    elif shape_type == "shifted_power":
        p = np.random.uniform(0.1, 1.0)
        d = np.random.uniform(0.0, n * 0.5) + 1e-6
        raw = (np.arange(n, dtype=float) + d) ** (-p)

    elif shape_type == "poly":
        alpha = np.random.uniform(0.1, 5.0)
        raw = (1.0 - np.arange(n, dtype=float) / (n - 1)) ** alpha

    elif shape_type == "linear":
        raw = np.linspace(1.0, 0.0, n, dtype=float)

    elif shape_type == "rand_gamma":
        raw = np.random.gamma(shape=0.7, scale=1.0, size=n)

    else:  # "rand_dir"
        raw = np.random.gamma(shape=0.7, scale=1.0, size=n)

    raw = np.maximum(raw, 0.0)
    s = raw.sum()
    if s == 0:
        raw = np.full(n, 1.0 / n)
        s = 1.0
    raw = raw * (target_sum / s)
    return raw.tolist()
