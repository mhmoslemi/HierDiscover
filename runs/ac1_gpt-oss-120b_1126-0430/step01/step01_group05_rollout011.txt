def simplex_projection(v):
    """Projects a vector onto the probability simplex (sum=1, nonnegative)."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, len(v)+1) > 0)[0]
    if rho.size == 0:
        # all entries are zero or negative, return uniform distribution
        return np.full_like(v, 1.0/len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function (a list of non‑negative heights) with a low
evaluation score.

Algorithm
---------
1. Choose a length n (large enough for the asymptotic regime, e.g. 20000).
2. Initialise the sequence with a power‑law a_i ∝ (i+1)^‑p.
   The exponent p is searched on a fine grid; the best p is kept.
3. Refine the best sequence by a simple projected sub‑gradient descent
   on the convex objective  max_k (a*a)[k]  under the simplex constraint
   Σ a_i = 1, a_i ≥ 0.

     analytically (only O(n) work).

   * Step size η_t = η₀ / sqrt(t) (η₀≈0.5) guarantees convergence.

The whole search respects the 1000 s time budget and prints occasional

import time

def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (sum=1, non‑negative)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # All entries are non‑positive – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float) -> np.ndarray:
    """Return the normalised power‑law vector a_i ∝ (i+1)^‑p  (sum = 1)."""
    seq = np.fromiter(((i + 1) ** (-p) for i in range(n)), dtype=np.float64)
    seq /= seq.sum()
    return seq


def _convolution_max(a: np.ndarray, fft_len: int) -> float:
    """Return max_k (a*a)[k] using FFT (fast for large n)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa).real[: 2 * a.size - 1]
    return float(conv.max())


def _ratio_from_max(a_len: int, max_conv: float) -> float:
    """Exact evaluation ratio when Σa_i = 1 (scale‑invariant)."""
    return 2.0 * a_len * max_conv


def _subgradient_max_convolution(a: np.ndarray, max_idx: int) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)[k] at the active index ``max_idx``.
    For the active offset k we have (∂/∂a_i) (a*a)[k] = 2·a_{k‑i}
    whenever the index is valid.
    """
    n = a.shape[0]
    g = np.zeros_like(a)

    lo = max(0, max_idx - (n - 1))
    hi = min(max_idx, n - 1)
    i = np.arange(lo, hi + 1, dtype=np.int64)
    # a_{k-i} is simply a[max_idx - i]
    g[i] = 2.0 * a[max_idx - i]
    return g
