def _gradient_step(seq, alpha, lr):
    n = len(seq)
    # compute convolution
    m = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[:2*n-1]
    # compute softmax weights
    maxc = conv.max()
    exp_term = np.exp(alpha * (conv - maxc))
    p = exp_term / exp_term.sum()
    # gradient via correlation
    grad = 2.0 * np.correlate(p, seq, mode='valid')
    # update and projection
    new_seq = seq - lr * grad
    # enforce non-negativity
    new_seq[new_seq < 0] = 0.0
    # renormalize to original sum
    sum_orig = seq.sum()
    sum_new = new_seq.sum()
    if sum_new > 0:
        new_seq *= (sum_orig / sum_new)
    else:
        # fallback: keep original
        new_seq = seq
    return new_seq, conv.max()
```

"""
Propose a low‑score step function.

The algorithm combines three ideas:

1.  A solid analytical seed – the discrete arcsine (Catalan) coefficients,
    whose self‑convolution is exactly constant (=1) for the first `n` entries.
2.  A cheap parametric sweep over simple power‑law families
    `a_i = (i+shift+1)^{-p}` to discover shapes that already beat the arcsine
    bound (most of them do, because the constant‑convolution property of the
    arcsine is not optimal for the min‑max objective).
3.  A gradient‑based refinement that directly minimises a smooth
    approximation of the objective `max(a*a)`.  The smooth max is taken as
    `L = (1/α)·log Σ exp(α·(a*a))`; its gradient can be computed by a single
    correlation (`np.correlate`) and a projected gradient step that keeps the
    vector non‑negative and preserves the total L1‑mass (scaling does not
    affect the evaluation metric).

All work is bounded by the supplied time budget (default 1000 s).  The best
candidate found is returned as a plain list of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast evaluation – identical to ``evaluate_sequence`` but without the
# security checks and using FFT based convolution.
# ----------------------------------------------------------------------
def _fast_evaluation(seq: np.ndarray) -> float:
    """Return the same metric as ``evaluate_sequence`` but much faster."""
    n = seq.size
    if n == 0:
        return np.inf
    # FFT size = next power of two >= 2*n-1
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Catalan / arcsine seed (coefficients of (1‑z)^‑½).
# ----------------------------------------------------------------------
def _catalan_sequence(length: int) -> np.ndarray:
    """Return the first ``length`` Catalan‑like coefficients."""
    seq = np.empty(length, dtype=float)
    a = 1.0
    seq[0] = a
    for i in range(1, length):
        # recurrence a_i = a_{i‑1} * (2*i‑1) / (2*i)
        a *= (2 * i - 1) / (2.0 * i)
        seq[i] = a
    return seq


# ----------------------------------------------------------------------
# Simple power‑law family a_i = (i+shift+1)^‑p.
# ----------------------------------------------------------------------
def _power_law_sequence(length: int, p: float, shift: float = 0.0) -> np.ndarray:
    """Generate a decreasing power‑law sequence."""
    i = np.arange(length, dtype=float)
    return (i + shift + 1.0) ** (-p)


# ----------------------------------------------------------------------
# Gradient descent on a smooth approximation of the max‑convolution.
# ----------------------------------------------------------------------
def _gradient_step(seq: np.ndarray, alpha: float, lr: float):
    """
    Perform one projected gradient descent step on the smooth max:
        L(seq) = (1/α)·log Σ exp(α·(seq*seq))

    Returns the updated sequence and the original maximal convolution entry.
    """
    n = seq.size
    # convolution (seq * seq) via FFT
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]

    # soft‑max weights over the convolution entries
    c_max = conv.max()
    # shift by the max for numerical stability
    w = np.exp(alpha * (conv - c_max))
    w /= w.sum()                     # |w|₁ = 1

    # gradient of the smooth max   g_i = 2·Σ_k w_k·a_{k‑i}
    # this is exactly the “valid” correlation of w with a
    grad = 2.0 * np.correlate(w, seq, mode="valid")

    # gradient step, then projection onto the non‑negative orthant
    new_seq = seq - lr * grad
    new_seq[new_seq < 0.0] = 0.0

    # rescale to keep the L1‑norm unchanged (scaling does not affect score)
    tot = seq.sum()
    new_tot = new_seq.sum()
    if new_tot > 0.0:
        new_seq *= (tot / new_tot)

    return new_seq, conv.max()


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Produce a list of non‑negative heights aiming at a low evaluation score.

    The function works within ``budget_s`` seconds (default 1000).  It starts
    from the Catalan (arcsine) sequence, explores simple power‑law families,
    then refines the current best candidate with a projected gradient descent
    on a smooth approximation of the max‑convolution.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # small safety margin
    n = 3000                                   # length of the candidate

    # --------------------------------------------------------------
    # 1.  Baseline seed: Catalan / arcsine coefficients.
    # --------------------------------------------------------------
    seq = _catalan_sequence(n)
    best_seq = seq.copy()
    best_score = _fast_evaluation(best_seq)
    print(f"[init] Catalan seed score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2.  Quick parametric sweep over power‑law families.
    # --------------------------------------------------------------
    for p in np.linspace(0.45, 0.65, 9):          # 9 values for the exponent
        for shift in (0, 1, 5, 10, 20):
            cand = _power_law_sequence(n, p, shift)
            sc = _fast_evaluation(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[grid] p={p:.3f}, shift={shift} → score={sc:.6f}")

    # --------------------------------------------------------------
    # 3.  Gradient‑based refinement.
    # --------------------------------------------------------------
    seq = best_seq.copy()
    alpha = 20.0                     # smooth‑max temperature
    lr0 = 5e-3                       # base learning rate
    start_time = time.time()

    # Main optimisation loop (stops a few seconds before the deadline)
    while time.time() < deadline - 1.0:
        elapsed = time.time() - start_time
        # simple linear decay of the learning rate
        lr = lr0 * max(0.02, 1.0 - elapsed / (budget_s * 0.8))

        new_seq, cur_max = _gradient_step(seq, alpha, lr)
        new_score = _fast_evaluation(new_seq)

        # accept only strict improvements
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_seq = new_seq.copy()
            seq = new_seq
            print(f"[grad] t={elapsed:.1f}s  lr={lr:.5f}  max={cur_max:.6f}  score={new_score:.6f}")
        else:
            # occasional random jitter to escape tiny plateaus
            if np.random.rand() < 0.01:
                jitter = np.random.randn(n) * 1e-4 * seq.mean()
                cand = np.maximum(0.0, seq + jitter)
                cand *= (seq.sum() / cand.sum())
                sc = _fast_evaluation(cand)
                if sc < best_score:
                    seq = cand
                    best_seq = cand.copy()
                    best_score = sc
                    print(f"[jitter] t={elapsed:.1f}s → score={sc:.6f}")

    print(f"[final] Best score discovered = {best_score:.6f}")
    # The official ``evaluate_sequence`` will clip to [0,1000] and apply
    # the same security checks, so we can safely hand over the raw values.
    return best_seq.tolist()
