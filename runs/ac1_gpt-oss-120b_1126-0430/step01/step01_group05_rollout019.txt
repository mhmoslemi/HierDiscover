        # compute using FFT with next power-of-two for speed

def fast_score(seq):
    a = np.asarray(seq, dtype=np.float64)
    n = a.shape[0]
    size = 2 * n - 1
    # For speed use rfft; compute conv
    m = 1 << (size-1).bit_length()
    Fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(Fa * Fa, n=m)[:size]
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 1e-12:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))
```

def propose_candidate(seed=0, budget_s=1000, **kwargs):

        5000, 8000, 12000,
    ]))

    length_candidates = [int(l) for l in length_candidates if l > 0]
    
    best_seq = None
    best_score = float('inf')
    # function to update best result
    def consider(seq):
        nonlocal best_seq, best_score
        sc = fast_score(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"New best fast score {sc:.6f} with length {len(seq)}")
    
    # Parameter grid search over lengths and families
    for n in length_candidates:
        if time.time() >= deadline:
            break
        # family 1: pure power law
        p_vals = np.linspace(0.45, 0.75, 61)   # step 0.005
        # Get approximate scores for each p via fast_score
        # Order p by fast_score to prioritize promising ones
        p_scores = []
        for p in p_vals:
            seq = [(i + 1) ** (-p) for i in range(n)]
            sc = fast_score(seq)
            p_scores.append((sc, p, seq))
        # sort by score
        p_scores.sort(key=lambda x: x[0])
        # take top few candidates to refine
        for sc, p, seq in p_scores[:5]:
            consider(seq)
            # local refinement: small random perturbations on p
            for _ in range(10):
                p2 = p + np.random.normal(scale=0.01)
                if p2 <= 0 or p2 >= 1:
                    continue
                seq2 = [(i + 1) ** (-p2) for i in range(n)]
                consider(seq2)
        # family 2: offset power law, try few offsets
        offsets = [0, 1, 2, 5, 10]
        for offset in offsets:
            for p in [0.55, 0.6, 0.65]:
                seq = [(i + 1 + offset) ** (-p) for i in range(n)]
                consider(seq)
        
        # family 3: exponential
        for lam in np.linspace(0.0001, 0.005, 10):
            seq = [math.exp(-lam * i) for i in range(n)]
            consider(seq)
        
        # family 4: combined
        for p in [0.55, 0.6, 0.65]:
            for lam in [0.0005, 0.001, 0.002]:
                seq = [(i + 1) ** (-p) * math.exp(-lam * i) for i in range(n)]
                consider(seq)
    
    # After initial scanning, perform hill climbing on best seq
    if best_seq is None:
        best_seq = [(i + 1) ** (-0.6) for i in range(base_n)]
        best_score = fast_score(best_seq)
    
    current_seq = best_seq.copy()
    current_score = best_score
    
    # hill climbing parameters
    max_iters = 20000
    iter_cnt = 0
    while time.time() < deadline and iter_cnt < max_iters:
        iter_cnt += 1
        # propose random local move: pick i and adjust multiplicatively
        idx = np.random.randint(len(current_seq))
        factor = 1.0 + 0.1 * (np.random.rand() - 0.5) # +/-5%
        old_val = current_seq[idx]
        new_val = max(0.0, old_val * factor)
        current_seq[idx] = new_val
        new_score = fast_score(current_seq)
        if new_score < current_score:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = current_seq.copy()
                print(f"Iter {iter_cnt:05d}: improved fast score {best_score:.6f}")
        else:
            # revert
            current_seq[idx] = old_val
    
    # final verification using the provided evaluate_sequence
    final_score = evaluate_sequence(best_seq)
    print(f"Final evaluated score: {final_score:.6f}")
    return best_seq

#!/usr/bin/env python
"""
Propose a step‑function that improves the bound 1.503.

Algorithm
---------
1.  Quick parametric sweep:
    * Try several lengths (including the length of the previously
      discovered construction and a few modest sizes).
    * For each length evaluate pure power‑law sequences a[i] = (i+1)**(-p)
      for a dense grid of exponents p∈[0.45,0.75].
    * Keep the best candidate according to a *fast* convolution score.

2.  If time permits, run the “alternating linear programme’’ described in the
    problem statement on a small problem (n≈250–350).  Starting from the best
    power‑law found, we repeatedly:
        – compute M = max(a*a);
        – solve   maximise Σ g_i  subject to  g_i≥0  and (a∗g)_k ≤ M;
        – renormalise g so that Σ g_i = √(2 n);
        – replace a ← (1‑t)·a + t·g  (t≈0.5).
    This convex sub‑problem is solved with CVXPY/ECOS and generally reduces the
    maximal convolution value.

3.  A cheap hill‑climbing phase refines the best sequence: random single‑coordinate
    multiplicative perturbations are accepted when they improve the fast score.

All intermediate scores are printed.  The final candidate is evaluated with the
provided ``evaluate_sequence`` (which contains the security checks) before being
returned.
"""

import time
import math
import numpy as np
import cvxpy as cp

# ----------------------------------------------------------------------
# Fast deterministic evaluation (identical to evaluate_sequence but using FFT)
# ----------------------------------------------------------------------
def _fast_score(seq: list[float]) -> float:
    """Return the value of evaluate_sequence but without the security checks."""
    a = np.asarray(seq, dtype=np.float64)
    n = a.size
    if n == 0:
        return float("inf")
    # convolution via FFT (real‑to‑real)
    size = 2 * n - 1
    m = 1 << (size - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[:size]
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))

# ----------------------------------------------------------------------
# Helpers to build candidate families
# ----------------------------------------------------------------------
def _power_sequence(n: int, p: float, offset: float = 0.0) -> list[float]:
    """a[i] = (i+1+offset) ** (-p).  offset may be non‑integer."""
    return [(i + 1 + offset) ** (-p) for i in range(n)]

def _exp_sequence(n: int, lam: float) -> list[float]:
    """a[i] = exp(-lam * i)."""
    return [math.exp(-lam * i) for i in range(n)]

# ----------------------------------------------------------------------
# Linear‑programming improvement step (the “alternating’’ method)
# ----------------------------------------------------------------------
def _lp_improve(f: list[float], t: float = 0.5, iters: int = 15,
                time_deadline: float = 1e9) -> list[float]:
    """
    Perform a few LP‑based improvement steps starting from sequence f.
    Returns the improved sequence (still a Python list).
    """
    n = len(f)
    target_sum = math.sqrt(2.0 * n)

    f_arr = np.asarray(f, dtype=np.float64)

    best_f = f_arr.copy()
    best_score = _fast_score(best_f.tolist())

    for step in range(iters):
        if time.time() >= time_deadline:
            break

        # M = max(f*f)
        conv_ff = np.convolve(f_arr, f_arr)
        M = float(conv_ff.max())

        # Build the convolution matrix A where (A g)_k = Σ_i f_i * g_{k-i}
        # (out‑of‑range entries are simply omitted).
        A = np.zeros((2 * n - 1, n), dtype=np.float64)
        for k in range(2 * n - 1):
            i_min = max(0, k - (n - 1))
            i_max = min(n - 1, k)
            # for all i in [i_min, i_max] we have contribution f[i] * g[k-i]
            # i is row index, j=k-i is column index
            idx = np.arange(i_min, i_max + 1)
            A[k, k - idx] = f_arr[idx]

        # CVXPY model
        g = cp.Variable(n, nonneg=True)
        constraints = [A @ g <= M + 1e-12]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
        except Exception:          # pragma: no cover – solver failures are rare
            break
        if g.value is None:
            break

        g_val = np.asarray(g.value, dtype=np.float64)
        # Renormalise to the standard sum √(2n)
        sum_g = g_val.sum()
        if sum_g < 1e-12:
            break
        g_norm = g_val * (target_sum / sum_g)

        # Mix the two sequences
        f_arr = (1.0 - t) * f_arr + t * g_norm

        cur_score = _fast_score(f_arr.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_f = f_arr.copy()
            print(f"LP step {step:02d}: improved fast score {best_score:.6f}")

    return best_f.tolist()

    """

    start_time = time.time()
    deadline = start_time + budget_s - 5.0           # a small safety margin

    # ------------------------------------------------------------------
    # 1. Choose a few problem sizes (including the historic one)
    # ------------------------------------------------------------------
    try:
        historic_len = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:

    # Work with a range of lengths; very large n makes the LP too slow,

    lengths = sorted(set([

        # try a few offset variants (offset only reshapes, still cheap)
        for offset in (0.0, 1.0, 2.0, 5.0):

            if sc < best_score:
                best_score = sc
                best_seq = seq
                print(f"[offset] n={n:5d}  off={offset:>3.1f}  fastScore={sc:.6f}")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:

        best_score = _fast_score(best_seq)

    # ------------------------------------------------------------------
    # 4. Final cheap hill‑climbing on the whole vector (single‑coordinate
    #    multiplicative moves).  This works directly on the original list.
    # ------------------------------------------------------------------
    if best_seq is None:                     # fallback – should never happen

    it = 0
