def fast_eval(seq):
    n = len(seq)
    # choose proper length for convolution: maybe next power of 2 >= 2*n
    # Use np.fft.rfft for real data to speed.
    m = 1 << (2*n-1).bit_length()
    fft_seq = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fft_seq*fft_seq, n=m)
    # conv length m, but actual convolution length is 2n-1; slice conv[:2*n-1]
    conv = conv[:2*n-1]
    max_b = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return float('inf')
    return float(2 * n * max_b / (sum_a**2))

def evaluate_fast(seq):
    n=len(seq)
    # compute conv with FFT
    m = 1 << ((2*n-1).bit_length())
    # Use real FFT
    fft_seq = np.fft.rfft(seq, n=m)

#!/usr/bin/env python
"""
Propose a step‑function (list of non‑negative heights) with a low
evaluate_sequence score.

The algorithm works in three stages:

1. **Initial candidates** – try a few lengths (powers of two) and a grid of
   power‑law exponents `p` (a_i = (i+1)^(-p)).  Evaluation is performed with a
   fast FFT‑based convolution (`_fast_score`) which is much quicker than the
   provided `evaluate_sequence`.

2. **Refinement of the exponent** – around the best exponent for each length a
   finer grid is searched.

3. **Local re‑allocation hill‑climb** – starting from the best candidate,
   repeatedly move a small amount of mass from early indices (which dominate
   the maximal convolution value) to later indices.  The move is kept only if
   it improves the fast score.  Because the objective is scale‑invariant only
   the shape matters; the search therefore converges toward sequences whose
   self‑convolution is as flat as possible.

All work is bounded by the supplied time budget (default 1000 s).  The
function returns the best sequence found as a plain Python list, ready to be
passed to `evaluate_sequence`.
"""

import time
import numpy as np
import math

# ----------------------------------------------------------------------
# Helper: fast evaluation of the score using FFT (much faster than
# np.convolve for the many calls we need during the search).
# ----------------------------------------------------------------------
def _fast_score(seq: np.ndarray) -> float:
    """Return the same value that evaluate_sequence would give, using FFT."""
    n = seq.size
    if n == 0:
        return float("inf")
    # Length for linear convolution: at least 2*n-1, choose next power of two.
    m = 1 << ((2 * n - 1).bit_length())
    # Real FFT of the zero‑padded sequence.
    fft_seq = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fft_seq * fft_seq, n=m)
    # Keep only the linear part.
    conv = conv[: 2 * n - 1]
    max_b = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 1e-2:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Helper: generate a power‑law sequence of given length and exponent.
# ----------------------------------------------------------------------
def _power_law(length: int, p: float) -> np.ndarray:
    """Return a (length,) array with a_i = (i+1)^(-p)."""
    # using np.arange is faster than a Python loop.
    idx = np.arange(1, length + 1, dtype=np.float64)
    return np.power(idx, -p)


# ----------------------------------------------------------------------
# Main entry point required by the competition harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative height sequence that minimises
    evaluate_sequence.  The function respects the time budget and returns
    the best list it discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for finalisation and printing
    deadline = start_time + budget_s - 5.0

    best_seq = None          # numpy array
    best_score = float("inf")
    best_len = None
    best_p = None

    # ------------------------------------------------------------------
    # 0.  If a previous construction is available, use it as a starting point.
    # ------------------------------------------------------------------
    try:
        seq0 = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        sc0 = _fast_score(seq0)
        best_seq = seq0.copy()
        best_score = sc0
        best_len = seq0.size
        print(f"Loaded height_sequence_1 (len={best_len}) → fast score {sc0:.6f}")
    except Exception:
        # no previous sequence – nothing to do here
        pass

    # ------------------------------------------------------------------
    # 1.  Lengths to explore.  Powers of two give FFT‑friendly sizes.
    # ------------------------------------------------------------------
    length_candidates = [4096, 8192, 16384, 32768]  # 65536 is still OK but heavy for final eval
    # If we already have a sequence longer than these, keep it as an extra candidate.
    if best_len is not None and best_len not in length_candidates:
        length_candidates.append(best_len)

    # ------------------------------------------------------------------
    # 2.  Scan exponent grid for each length.
    # ------------------------------------------------------------------
    for n in length_candidates:
        if time.time() >= deadline:
            break

        # coarse grid – more points around the expected optimum (0.55‑0.62)
        coarse = np.concatenate((
            np.linspace(0.45, 0.55, 20, endpoint=False),
            np.linspace(0.55, 0.65, 30, endpoint=False),
            np.linspace(0.65, 0.75, 10),
        ))

        # sort by cheap analytic approximation to look at promising exponents early
        def _analytic(p):
            u = 1.0 - p
            return 2.0 * (u ** 2) * (math.gamma(u) ** 2) / math.gamma(2.0 * u)

        coarse = sorted(coarse, key=_analytic)

        for p in coarse:
            if time.time() >= deadline:
                break
            seq = _power_law(n, p)
            score = _fast_score(seq)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                best_len = n
                best_p = p
                print(f"[grid] n={n:5d} p={p:.5f} → score {score:.6f} (new best)")

        # ------------------------------------------------------------------
        # 3.  Fine‑grained refinement around the best p found for this length.
        # ------------------------------------------------------------------
        if best_seq is not None and best_len == n:
            centre = best_p
            radii = [0.02, 0.008, 0.003]
            for rad in radii:
                if time.time() >= deadline:
                    break
                fine_grid = np.linspace(centre - rad, centre + rad, 25)
                for p in fine_grid:
                    if p <= 0 or p >= 1:
                        continue
                    if time.time() >= deadline:
                        break
                    seq = _power_law(n, p)
                    score = _fast_score(seq)
                    if score < best_score:
                        best_score = score
                        best_seq = seq.copy()
                        best_len = n
                        best_p = p
                        print(f"[refine] n={n:5d} p={p:.6f} → score {score:.6f} (new best)")

        # ------------------------------------------------------------------
        # 4.  Local hill‑climb: move mass from early to late positions.
        # ------------------------------------------------------------------
        if best_seq is not None and best_len == n:
            cur_seq = best_seq.copy()
            cur_score = best_score
            # Parameters controlling the move.
            early_frac = 0.12          # early zone (12 % of indices)
            late_start_frac = 0.55     # start of the “receiving” zone
            early_end = max(1, int(early_frac * n))
            late_start = int(late_start_frac * n)

            max_iters = 8000
            it = 0
            while time.time() < deadline and it < max_iters:
                it += 1
                # pick a source from the early region, a target from the late region
                src = np.random.randint(early_end)
                dst = np.random.randint(late_start, n)
                # move a modest fraction of src's mass (1‑10 %)
                frac = 0.01 + 0.09 * np.random.rand()
                delta = cur_seq[src] * frac
                if delta <= 0.0:
                    continue
                cur_seq[src] -= delta
                cur_seq[dst] += delta

                new_score = _fast_score(cur_seq)
                if new_score < cur_score:                      # accept improvement
                    cur_score = new_score
                    if new_score < best_score:
                        best_score = new_score
                        best_seq = cur_seq.copy()
                        best_len = n
                        best_p = None
                        print(f"[climb] n={n:5d} iter={it:4d} → score {new_score:.6f}")
                else:
                    # revert
                    cur_seq[src] += delta
                    cur_seq[dst] -= delta

    # ------------------------------------------------------------------
    # 5.  Fallback in the unlikely event that nothing was found.
    # ------------------------------------------------------------------
    if best_seq is None:
        print("Fallback: using a default power‑law (n=5000, p=0.60).")
        best_seq = _power_law(5000, 0.60)
        best_score = _fast_score(best_seq)

    # Convert to plain Python list before returning.
    final_list = best_seq.tolist()
    print(f"Search completed – best score (fast) {best_score:.6f} with length {len(final_list)}")
    # The actual official score will be computed by evaluate_sequence after we return.
    return final_list
