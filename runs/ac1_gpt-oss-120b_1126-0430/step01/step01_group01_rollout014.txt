def propose_candidate(seed=42, budget_s=1000, **kwargs):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5

    n = 3000
    # Initialize with Catalan
    a = _catalan_sequence(n)  # array of float; sum S
    best_seq = a.copy()
    best_score = _fast_evaluation(a)
    print("Initial score:", best_score)

    # Precompute FFT length
    fft_len = 1 << ((2*n-1).bit_length())
    # Define helpers
    while time.time() < deadline:
        # compute convolution
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2*n-1]
        k = int(conv.argmax())
        max_val = conv[k]

        # contributions and partner weights
        i_arr = np.arange(n)
        j_arr = k - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        w = np.zeros_like(a)
        w[mask] = a[j_arr[mask]]
        contrib = a * w

        src = int(np.argmax(contrib))
        # candidate sinks: indices with low weight w (prefer zero)
        # choose among top few smallest w
        low_indices = np.where(w < np.percentile(w, 10))[0]  # maybe 10% lowest
        if low_indices.size == 0:
            sink = int(np.argmin(w))
        else:
            sink = int(np.random.choice(low_indices))

        if a[src] < 1e-12:
            break

        # random fraction
        max_move = a[src] * 0.5  # up to 50% of source
        eps = max_move * np.random.rand()  # uniform [0,max_move]
        if eps < 1e-12:
            continue

        new_a = a.copy()
        new_a[src] -= eps
        new_a[sink] += eps

        # evaluate
        score = _fast_evaluation(new_a)

        if score < best_score - 1e-12:
            a = new_a
            best_seq = new_a.copy()
            best_score = score
            print(f"Iter {iteration}: improved score {best_score:.6f}, move {eps:.6e} from {src} to {sink}")
            # possibly increase step size for future
        else:
            # with small probability accept worse for exploration
            if np.random.rand() < 0.001:
                a = new_a
                # maybe update best if improvement later
        iteration += 1
        # optionally every N iterations do a full gradient descent step or random jitter
        if iteration % 1000 == 0:
            # random jitter to avoid stagnation
            jitter_idx = np.random.randint(n)
            jitter = (np.random.rand() - 0.5) * 0.01  # small perturb
            a[jitter_idx] = max(0.0, a[jitter_idx] + jitter)

    print("Best score:", best_score)
    return best_seq.tolist()

"""
Improved step‑function search.

The algorithm starts from the Catalan (discrete arcsine) coefficients,
which are known to give a near‑optimal value ≈π/2.  A fast projected
greedy local search then repeatedly moves a substantial fraction of the
mass from the index that contributes most to the current maximal
convolution entry to an index that contributes almost nothing.
Both transfer (mass conservation) and pure addition (increase the total
ℓ₁‑norm) moves are tried; the move that reduces the evaluation score is
kept.  The search runs until the time budget expires and returns the
best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast (unchecked) evaluation – identical to ``evaluate_sequence`` but
# without the security checks and using an FFT based convolution.
# ----------------------------------------------------------------------
def _fast_evaluation(seq: np.ndarray) -> float:
    """Return the same metric as evaluate_sequence but much faster."""
    n = seq.size
    if n == 0:
        return np.inf
    # FFT size = next power of two ≥ 2·n‑1
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Catalan (arcsine) seed: a₀ = 1, a_{i+1} = a_i·(2i+1)/(2i+2)
# ----------------------------------------------------------------------
def _catalan_sequence(length: int) -> np.ndarray:
    """First ``length`` Catalan coefficients (as a float array)."""
    seq = np.empty(length, dtype=float)
    a = 1.0
    seq[0] = a
    for i in range(1, length):
        a *= (2 * i - 1) / (2.0 * i)
        seq[i] = a
    return seq


# ----------------------------------------------------------------------
# Aggressive greedy refinement.
# ----------------------------------------------------------------------
def _aggressive_refine(seq: np.ndarray, deadline: float) -> np.ndarray:
    """
    Repeatedly transfers a *large* fraction of the mass from the
    source index (largest contribution to the current maximal
    convolution entry) to a sink index (almost zero contribution).
    Pure addition moves (increase total sum) are also attempted.
    The best sequence seen so far is kept.
    """
    n = seq.size
    best_seq = seq.copy()
    best_score = _fast_evaluation(best_seq)

    # pre‑compute the FFT length (does not change)
    fft_len = 1 << ((2 * n - 1).bit_length())
    iteration = 0

    while time.time() < deadline:
        # current convolution and the index where it attains its maximum
        fa = np.fft.rfft(seq, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        k = int(conv.argmax())

        # contribution weights for each position to that maximal entry:
        # w[i] = a_{k‑i} if the index exists, else 0
        i_arr = np.arange(n)
        j_arr = k - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        w = np.zeros_like(seq)
        w[mask] = seq[j_arr[mask]]

        # source: the index with the largest product a_i·w_i
        contrib = seq * w
        src = int(np.argmax(contrib))

        # sink: an index with (almost) no contribution (minimal w)
        sink = int(np.argmin(w))

        if seq[src] < 1e-12:
            break   # nothing left to move

        improved = False
        # ------------------------------------------------------------------
        # 1) Transfer a sizable fraction from src to sink (mass conserved)
        # ------------------------------------------------------------------
        for factor in (0.5, 0.25, 0.125, 0.0625, 0.03125):
            eps = factor * seq[src]           # move up to 50 % of src
            if eps <= 1e-12:
                continue
            cand = seq.copy()
            cand[src] -= eps
            cand[sink] += eps
            score = _fast_evaluation(cand)
            if score < best_score - 1e-12:
                seq = cand
                best_seq = cand.copy()
                best_score = score
                improved = True
                print(
                    f"Iter {iteration:5d}: transfer {eps:.4e} "
                    f"src {src:4d}→sink {sink:4d}  score {best_score:.6f}"
                )
                break
        if improved:
            iteration += 1
            continue

        # ------------------------------------------------------------------
        # 2) Pure addition – increase the sink without taking from src.
        #    This raises the total ℓ₁‑norm, which may lower the final score
        #    if the increase in the maximal convolution stays modest.
        # ------------------------------------------------------------------
        for factor in (0.5, 0.25, 0.125):
            eps = factor * seq[src]           # use src as a scale reference
            cand = seq.copy()
            cand[sink] += eps
            score = _fast_evaluation(cand)
            if score < best_score - 1e-12:
                seq = cand
                best_seq = cand.copy()
                best_score = score
                improved = True
                print(
                    f"Iter {iteration:5d}: add {eps:.4e} to sink {sink:4d} "
                    f"score {best_score:.6f}"
                )
                break

        # ------------------------------------------------------------------
        # 3) If no improvement, take a tiny random jitter (helps escape plateaus)
        # ------------------------------------------------------------------
        if not improved:
            if np.random.rand() < 0.01:
                jitter_idx = np.random.randint(n)
                jitter = np.random.randn() * 1e-3
                cand = seq.copy()
                cand[jitter_idx] = max(0.0, cand[jitter_idx] + jitter)
                seq = cand

        iteration += 1

    return best_seq


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights aiming at a low evaluation score.

    The function works within ``budget_s`` seconds (default 1000).  It
    starts from the Catalan (arcsine) sequence, performs an aggressive
    greedy refinement that moves substantial mass from indices that
    heavily influence the maximal self‑convolution entry to indices that
    barely influence it, and occasionally adds mass to increase the
    ℓ₁‑norm.  The best sequence discovered before the deadline is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5  # safety margin

    # Length – a few thousand gives a good trade‑off between quality and speed
    n = 3000

    # ------------------------------------------------------------------
    # 1.  Initial seed – Catalan (arcsine) coefficients.
    # ------------------------------------------------------------------
    seq = _catalan_sequence(n)

    # Very small random perturbation to break perfect symmetry
    seq += np.random.rand(n) * 1e-6

    # ------------------------------------------------------------------
    # 2.  Aggressive greedy refinement.
    # ------------------------------------------------------------------
    seq = _aggressive_refine(seq, deadline)

    # ------------------------------------------------------------------
    # 3.  Return the best sequence found (plain Python list).
    # ------------------------------------------------------------------
    final_score = _fast_evaluation(seq)
    print("Final score achieved:", final_score)
    return seq.tolist()
