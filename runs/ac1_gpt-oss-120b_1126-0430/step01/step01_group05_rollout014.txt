#!/usr/bin/env python
"""
A fast multi‑parameter search for a low‑value step‑function.

The evaluation score is
    E = 2·n·max(conv(a,a)) / (sum(a))²
and is invariant under a global scaling of the sequence.
Hence we can freely experiment with families of shapes.

The algorithm proceeds in three stages:

1. **Coarse grid search** (on a modest length n≈2000) over a three‑parameter
   family
        a_i = (i+1+α)^(‑p) · exp(‑λ·i / n)
   (power‑law with optional offset α and optional exponential damping λ).
   The grid is intentionally dense (≈1000 candidates) but fast because the
   sequences are short.

2. **Local refinement** of the best triple (p,α,λ) by small perturbations.
   The refinement keeps the best discovered parameters.

3. **Final construction** on the target length (taken from the existing
   `height_sequence_1` if available, otherwise 5000).  The best parameters are
   scaled to the new length (α is scaled proportionally).  A short hill‑climb
   (single‑coordinate multiplicative mutations) is performed until the time
   budget expires.

The whole routine respects the 1000 s budget, prints progress information,
and returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all are top‑level (no closures) as required.
# ----------------------------------------------------------------------


def _make_sequence(n: int, p: float, alpha: float, lam: float) -> list[float]:
    """
    Build a sequence of length ``n`` of the form
        a_i = (i + 1 + alpha) ** (-p) * exp(-lam * i / n)
    ``alpha`` and ``lam`` may be zero.
    """
    seq = []
    inv_n = 1.0 / n
    for i in range(n):
        # (i+1) because the original code used (i+1)**(-p)
        base = (i + 1 + alpha) ** (-p)
        if lam != 0.0:
            base *= math.exp(-lam * i * inv_n)
        seq.append(base)
    return seq


def _hill_climb(seq: list[float], deadline: float, max_iter: int = 2000) -> tuple[list[float], float]:
    """
    Simple stochastic hill‑climb that perturbs a single coordinate by a
    multiplicative factor drawn from a small normal distribution.
    The function returns the best sequence seen and its score.
    """
    best_seq = seq.copy()
    best_score = evaluate_sequence(best_seq)
    cur_seq = seq
    cur_score = best_score

    n = len(seq)
    it = 0
    # start with a relatively large step and shrink it over time
    sigma = 0.1

    while time.time() < deadline and it < max_iter:
        it += 1
        idx = np.random.randint(n)
        factor = 1.0 + sigma * (np.random.rand() - 0.5)  # approx +/-5 %
        new_val = max(0.0, cur_seq[idx] * factor)

        old_val = cur_seq[idx]
        cur_seq[idx] = new_val
        new_score = evaluate_sequence(cur_seq)

        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = cur_seq.copy()
                print(f"Hill‑climb iter {it:04d}: new best score {best_score:.6f}")
        else:
            # revert
            cur_seq[idx] = old_val

        # slowly reduce the mutation size
        if it % 200 == 0:
            sigma *= 0.9
            sigma = max(sigma, 0.01)

    return best_seq, best_score


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative heights) with a low
    evaluation score.  The routine is deterministic given ``seed`` and respects
    the time budget.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # 1. Choose a small working length for the coarse search.
    # ------------------------------------------------------------------
    n_work = 2000                     # fast enough for thousands of evaluations
    try:
        n_target = len(height_sequence_1)   # type: ignore[name-defined]
    except Exception:
        n_target = 5000

    print(f"Working length = {n_work}, target length = {n_target}")

    # ------------------------------------------------------------------
    # 2. Grid of parameters (p, alpha, lambda).
    # ------------------------------------------------------------------
    # Power exponent
    p_grid = np.linspace(0.44, 0.71, 28)                # 28 points
    # Additive offset (in index units)
    alpha_grid = np.arange(0, 301, 50)                  # 0 … 300 step 50
    # Exponential damping factor
    lam_grid = np.linspace(0.0, 5.0, 6)                 # 0 … 5 step 1

    best_params = None
    best_score = float("inf")
    best_seq_work = None

    total_evals = 0
    for p in p_grid:
        for alpha in alpha_grid:
            for lam in lam_grid:
                if time.time() >= deadline:
                    break
                seq = _make_sequence(n_work, p, alpha, lam)
                score = evaluate_sequence(seq)
                total_evals += 1
                if score < best_score:
                    best_score = score
                    best_params = (p, alpha, lam)
                    best_seq_work = seq
                    print(f"New coarse best: p={p:.4f}, α={alpha}, λ={lam:.2f} → score={score:.6f}")
    print(f"Coarse grid finished – {total_evals} evaluations, best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Refine the best parameters by random local search.
    # ------------------------------------------------------------------
    if best_params is None:
        # fallback – should never happen
        best_params = (0.58, 0.0, 0.0)

    p0, a0, l0 = best_params
    p_best, a_best, lam_best = p0, a0, l0

    # small random walk around the best point
    for _ in range(150):
        if time.time() >= deadline:
            break
        p_try = max(0.01, p_best + np.random.uniform(-0.02, 0.02))
        a_try = max(0.0, a_best + np.random.uniform(-20, 20))
        lam_try = max(0.0, lam_best + np.random.uniform(-0.5, 0.5))

        seq = _make_sequence(n_work, p_try, a_try, lam_try)
        score = evaluate_sequence(seq)
        if score < best_score:
            best_score = score
            best_params = (p_try, a_try, lam_try)
            best_seq_work = seq
            p_best, a_best, lam_best = best_params
            print(f"Refine improvement: p={p_try:.4f}, α={a_try:.1f}, λ={lam_try:.2f} → {score:.6f}")

    # ------------------------------------------------------------------
    # 4. Build the final (large) sequence using the best parameters.
    #    Alpha is scaled proportionally to the new length.
    # ------------------------------------------------------------------
    p_opt, alpha_opt, lam_opt = best_params
    # Scale alpha to the final length (keep the same relative offset)
    alpha_scaled = alpha_opt * (n_target / n_work)

    final_seq = _make_sequence(n_target, p_opt, alpha_scaled, lam_opt)
    final_score = evaluate_sequence(final_seq)
    print(f"Initial final sequence: p={p_opt:.4f}, α={alpha_scaled:.1f}, λ={lam_opt:.2f} → score={final_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Optional hill‑climb on the final sequence (uses remaining time).
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print("Starting hill‑climb on final sequence…")
        best_seq, best_score = _hill_climb(final_seq, deadline, max_iter=5000)
        print(f"Hill‑climb finished – best final score {best_score:.6f}")
    else:
        best_seq = final_seq
        best_score = final_score

    print(f"Search completed in {time.time() - start:.1f}s – best score {best_score:.6f}")
    return best_seq
