```python
import time
import numpy as np

def proj_simplex(v):
    """Project vector v onto probability simplex (nonnegative sum=1)."""
    # Implementation from Duchi et al. 2008.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback: all zeros
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

Fast convolution: using real FFT `np.fft.rfft` as earlier.

```python
def convolve_fft(a, b):
    size = len(a) + len(b) - 1
    N = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    C = A * B
    conv = np.fft.irfft(C, N)[:size]
    return conv
```

Fast score:

```python
def fast_score(a):

"""
Propose a low‑score step‑function.

The algorithm works in three stages:

1. **Initial shape** – a power‑law (i+δ)^(-α) is scanned on a modest grid
   (length n ≈ 8000) and the best performer is kept as a warm‑start.

2. **Smooth max minimisation** – we minimise a smooth surrogate of the
   objective  max(conv(a,a)) using the log‑sum‑exp (LSE) trick.
   For a given λ the LSE gradient is
        ∇  = 2·corr(w , a)     with
        w_i ∝ exp( λ·(b_i−max(b)) ),   b = a*a .
   The gradient is obtained by a single FFT‑based convolution, then the
   iterate is projected onto the probability simplex (non‑negative, sum=1)
   using the Duchi‑et‑al. O(n) projection.  An Adam‑style step size
   automatically adapts to the geometry.  λ is increased slowly from 5 to
   30 so that the optimisation first smooths the whole spectrum and then
   focuses on the current peak.

3. **Final polishing** – a few fast projected‑gradient steps (the
   original L2‑error optimiser) are applied to squeeze the remaining
   peak.

The whole search stays well inside the 1000 s budget; the best sequence
found is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (non‑negative, sum=1)."""
    n = v.shape[0]
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # Numerical safety – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based linear convolution (real input)
def _convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using a power‑of‑two FFT."""
    size = a.size + b.size - 1
    N = 1 << (size - 1).bit_length()      # next power of two
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    C = A * B
    conv = np.fft.irfft(C, N)[:size]
    return conv


# ----------------------------------------------------------------------
def _fast_score(a: np.ndarray) -> float:
    """Objective 2·n·max(conv(a,a)) for a vector with sum=1."""
    n = a.size
    b = _convolve_fft(a, a)
    return 2.0 * n * np.max(b)


# ----------------------------------------------------------------------
def _power_law(alpha: float, delta: float, n: int) -> np.ndarray:
    """Return a normalised power‑law a_i ∝ (i+δ)^(-α), i = 1…n."""
    idx = np.arange(1.0, n + 1, dtype=np.float64) + delta
    a = idx ** (-alpha)
    s = a.sum()
    if s > 0:
        a /= s
    return a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.3   # small safety margin

    # --------------------------------------------------------------
    # 1) Choose a working length – a few ten‑thousands works fine.
    n = 8000
    print(f"[setup] sequence length = {n}")

    # --------------------------------------------------------------
    # 2) Coarse grid search over a power‑law family (quick warm‑start)
    best_a = np.full(n, 1.0 / n)          # uniform (score = 2)
    best_score = _fast_score(best_a)
    print(f"[init] uniform score = {best_score:.6f}")

    alphas = np.linspace(0.42, 0.60, 37)   # step ≈0.005
    deltas = [0.0, 0.5, 1.0]

    for delta in deltas:
        for alpha in alphas:
            if time.time() >= deadline:
                break
            a = _power_law(alpha, delta, n)
            sc = _fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[grid] new best α={alpha:.4f} δ={delta:.1f} → score={sc:.6f}")

    # --------------------------------------------------------------
    # 3) Smooth‑max optimisation (log‑sum‑exp surrogate) with Adam + simplex proj.
    a = best_a.copy()
    m = np.zeros_like(a)     # first moment
    v = np.zeros_like(a)     # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.02                # base learning rate for Adam
    lam = 5.0                # start λ for LSE
    lam_max = 30.0
    lam_inc = 0.02           # additive increase per iteration

    best_seq = a.copy()
    best_score = _fast_score(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # ----- convolution of a with itself -----
        b = _convolve_fft(a, a)               # length 2n‑1
        max_b = b.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[opt] iter {it:5d} improved score = {cur_score:.6f}")

        # ----- smooth‑max weights -----
        # w_i ∝ exp( λ·(b_i−max_b) )   (numerically stable)
        exp_arg = lam * (b - max_b)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:            # should not happen, but guard
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum               # now a proper probability vector

        # ----- gradient of the LSE surrogate -----
        # grad = 2 * corr(w , a)  (valid part, length n)
        conv_full = _convolve_fft(w, a[::-1])          # length 3n‑2
        grad = 2.0 * conv_full[n - 1 : n - 1 + n]      # slice of length n

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_next = a - step
        a_next = _project_simplex(a_next)    # enforce non‑negativity & sum=1
        a = a_next

        # ----- increase λ slowly (more focus on the peak) -----
        if lam < lam_max:
            lam = min(lam + lam_inc, lam_max)

    # --------------------------------------------------------------
    # 4) Final polishing with a few fast L2‑error gradient steps
    #    (the routine from the reference implementation, but tuned a bit)
    def _polish(a_vec: np.ndarray, time_deadline: float) -> np.ndarray:
        step_factor = 0.04
        best_local = a_vec.copy()
        best_local_score = _fast_score(a_vec)

        while time.time() < time_deadline:
            b_loc = _convolve_fft(a_vec, a_vec)
            target_const = (a_vec.sum() ** 2) / (2 * n - 1)   # constant of a flat conv
            err = b_loc - target_const
            # gradient of the L2 error w.r.t. a (same correlation trick)
            grad_loc = 4.0 * np.correlate(err, a_vec, mode='valid')
            gn = np.linalg.norm(grad_loc)
            if gn == 0.0:
                break
            eta = step_factor / gn
            a_vec = a_vec - eta * grad_loc
            a_vec = np.maximum(a_vec, 0.0)
            s = a_vec.sum()
            if s == 0.0:
                a_vec = np.full(n, 1.0 / n)
            else:
                a_vec /= s

            cur = _fast_score(a_vec)
            if cur < best_local_score:
                best_local_score = cur
                best_local = a_vec.copy()
                print(f"[polish] improved score {cur:.6f}")

        return best_local

    if time.time() < deadline:
        print("[polish] starting final refinement")
        best_seq = _polish(best_seq, deadline)

    # --------------------------------------------------------------
    # 5) Sanity check with the provided evaluate_sequence (if present)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception:
        # In the judging environment evaluate_sequence will be defined;
        # ignoring any import errors here is safe.
        pass

    # Return as a plain Python list of floats
    return best_seq.tolist()
