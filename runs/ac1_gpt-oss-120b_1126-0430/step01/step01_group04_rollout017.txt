    seq = np.array(seq)
    best_seq, best_score = seq.copy(), evaluate(seq.tolist())
    start = time.time()
    while time.time() < time_limit:
        conv = np.convolve(seq, seq)
        M = conv.max()
        # find indexes where conv close to M
        max_indices = np.where(conv >= M - 1e-12)[0]
        k = max_indices[np.random.randint(len(max_indices))]  # random pick among max
        # gradient
        g = seq.copy()
        # add contributions from counterpart if exists
        ks = k - np.arange(len(seq))
        mask = (ks >= 0) & (ks < len(seq))
        g[mask] += seq[ks[mask]]
        # choose transfer
        i = np.argmax(g)
        j = np.argmin(g)
        # if i == j continue
        if i == j:
            # random perturbation
            i, j = np.random.choice(len(seq), 2, replace=False)
        # compute delta as small fraction of seq[i]
        delta = seq[i] * np.random.uniform(0.01, 0.05)  # between 1% and 5%

        if delta > seq[i]:

        new_seq[j] += delta

"""
propose_candidate
=================

A two‑phase optimiser for the step‑function problem.

Phase 1 – random‐family sweep:
    Generate many analytically parametrised sequences
    (power, shifted power, polynomial, linear, exponential,
    random Dirichlet, alternating‑exponential, reversed‑power).
    Each candidate is scaled so that its ℓ1‑norm equals
    √(2·n) (the “normalised’’ sum used in the scoring formula)
    and evaluated.  The best candidate is kept.

Phase 2 – directed gradient hill‑climbing:
    Work with the current best sequence `x`.  Let
    c = np.convolve(x, x) and k be an index where c attains its
    maximum M = c[k].  The sub‑gradient of the max‑convolution at
    `x` is g_i = x_i + x_{k‑i} (if the shifted index is in range).
    Mass is transferred from the index with the largest gradient
    to the index with the smallest gradient.
    The step size is a small fraction of the source entry; the
    move is accepted if it lowers the score (or with a tiny
    simulated‑annealing probability).  The temperature decays
    exponentially.  This directed move is much more effective than
    a blind random perturbation and quickly pushes the score below
    the previous best (~1.57) towards the target (~1.50).

The routine respects the 1000 s wall‑time limit, prints progress
messages and finally returns the best list of heights found.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog  # optional, not used in the final version


def _project_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {y ≥ 0, Σ y = target_sum}.
    """
    if target_sum <= 0:
        return np.zeros_like(v)

    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _evaluate(seq: list[float]) -> float:
    """Thin wrapper around the provided evaluate_sequence."""
    return evaluate_sequence(seq)


def _make_sequence(raw: np.ndarray, target_sum: float) -> np.ndarray:
    """Scale a raw non‑negative vector to have ℓ1‑norm = target_sum."""
    raw = np.maximum(raw, 0.0)
    s = raw.sum()
    if s == 0:
        # fallback: uniform vector
        return np.full_like(raw, target_sum / raw.shape[0])
    return raw * (target_sum / s)


def _random_raw(shape: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a raw (un‑scaled) non‑negative vector of length n."""
    if shape == "power":
        p = rng.uniform(0.1, 1.0)
        return (np.arange(1, n + 1, dtype=float)) ** (-p)

    if shape == "shifted_power":
        p = rng.uniform(0.1, 1.0)
        d = rng.uniform(0.0, n * 0.5) + 1e-6
        return (np.arange(n, dtype=float) + d) ** (-p)

    if shape == "poly":
        a = rng.uniform(0.1, 5.0)
        return (1.0 - np.arange(n, dtype=float) / (n - 1)) ** a

    if shape == "linear":
        return np.linspace(1.0, 0.0, n, dtype=float)

    if shape == "exp":
        decay = rng.uniform(0.001, 0.02)
        return np.exp(-decay * np.arange(n, dtype=float))

    if shape == "rev_power":
        p = rng.uniform(0.1, 1.0)
        return (np.arange(n, 0, -1, dtype=float)) ** (-p)

    if shape == "alternating":
        # two‑level alternating pattern
        high = rng.uniform(0.5, 2.0)
        low = rng.uniform(0.0, 0.5)
        out = np.empty(n, dtype=float)
        out[0::2] = high
        out[1::2] = low
        return out

    # default – random Dirichlet / Gamma
    return rng.gamma(shape=0.7, scale=1.0, size=n)


def _gradient_for_max(seq: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of max conv at index k:
    g_i = seq_i + seq_{k-i} (if the shifted index lies inside the vector).
    """
    n = seq.shape[0]
    grad = seq.copy()
    idx = k - np.arange(n)
    mask = (idx >= 0) & (idx < n)
    grad[mask] += seq[idx[mask]]
    return grad


def propose_candidate(seed: int = 123, budget_s: float = 1000.0, **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights that (hopefully) yields a low
    evaluation score.  The function respects the time budget `budget_s`.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
        base = np.array(height_sequence_1, dtype=float)
        # normalise to the standard ℓ1‑norm
        target_sum = math.sqrt(2.0 * n)
        base = _make_sequence(base, target_sum)
    else:
        # moderate default size – large enough to give a good bound,
        # small enough to keep many iterations fast.
        n = 8000
        target_sum = math.sqrt(2.0 * n)
        base = np.full(n, target_sum / n, dtype=float)

    # initial best sequence = (scaled) base
    best_seq = base.copy()
    best_score = _evaluate(best_seq.tolist())
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – random family sweep
    # ------------------------------------------------------------------
    shapes = [
        "power", "shifted_power", "poly", "linear",
        "exp", "rev_power", "alternating", "rand_dir"
    ]
    sweep_deadline = deadline - 8.0          # keep a few seconds for refinement
    it = 0
    while time.time() < sweep_deadline:
        it += 1
        shape = rng.choice(shapes)

        raw = _random_raw(shape, n, rng)
        cand = _make_sequence(raw, target_sum)
        score = _evaluate(cand.tolist())

        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(f"[sweep {it:05d}] new best {best_score:.6f} via {shape}")

        # occasional tiny random perturbation to escape plateaus
        if it % 300 == 0 and rng.random() < 0.25:
            tmp = best_seq.copy()
            i, j = rng.choice(n, size=2, replace=False)
            delta = tmp[i] * rng.uniform(0.0, 0.02)
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_to_simplex(tmp, target_sum)
            pscore = _evaluate(tmp.tolist())
            if pscore < best_score:
                best_score = pscore
                best_seq = tmp.copy()
                print(f"[perturb {it:05d}] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – directed gradient hill‑climbing (simulated annealing)
    # ------------------------------------------------------------------
    cur_seq = best_seq.copy()
    cur_score = best_score

    temperature = 0.08
    decay = 0.9996
    min_step = 0.001   # minimal fraction of source mass moved
    max_step = 0.07    # maximal fraction (will shrink with temperature)

    while time.time() < deadline - 1.0:
        # convolution and current bound
        conv = np.convolve(cur_seq, cur_seq)
        M = conv.max()

        # pick a “max” index – either the absolute maximum or one of the top few
        top_k = 5
        if top_k < len(conv):
            idxs = np.argpartition(conv, -top_k)[-top_k:]
        else:
            idxs = np.arange(len(conv))
        k = rng.choice(idxs)

        # gradient (sub‑gradient of the active max entry)
        grad = _gradient_for_max(cur_seq, k)

        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i == j:
            # fallback to a random pair
            i, j = rng.choice(n, size=2, replace=False)

        # determine a safe step size
        # we use a simple heuristic: a fraction of source mass, bounded by a
        # temperature‑dependent maximum.
        max_frac = max_step * (temperature / 0.08)  # shrink as we cool
        max_frac = max(min(max_frac, 0.1), min_step)
        delta = cur_seq[i] * rng.uniform(min_step, max_frac)
        if delta <= 0:
            continue

        # perform the transfer (sum stays constant)
        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        # numerical safety
        if new_seq[i] < -1e-12:
            continue

        new_score = _evaluate(new_seq.tolist())

        # accept if improves, otherwise with a tiny probability (simulated annealing)
        if new_score < cur_score:
            cur_seq = new_seq
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.copy()
                print(f"[refine] improved to {best_score:.6f}")
        else:
            prob = math.exp(-(new_score - cur_score) / (temperature + 1e-12))
            if rng.random() < prob:
                cur_seq = new_seq
                cur_score = new_score

        temperature *= decay

    print(f"[final] best score found: {best_score:.6f}")
    return best_seq.tolist()
