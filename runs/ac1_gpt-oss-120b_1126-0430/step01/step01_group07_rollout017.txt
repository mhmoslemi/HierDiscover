    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)

def softmax_gradient_descent(a, deadline):
    n = len(a)
    step_factor = 0.5
    beta = 1.0
    best_a = a.copy()
    best_score = fast_score(a)
    iteration = 0
    while time.time() < deadline and step_factor > 1e-6:
        iteration += 1
        # compute b
        b = convolve_fft(a, a)
        max_b = b.max()
        cur_score = 2 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f'Iteration {iteration} improved score to {cur_score:.6f}')
        # compute weights
        max_val = max_b
        exp_beta = np.exp(beta * (b - max_val))
        w = exp_beta / exp_beta.sum()
        # compute gradient via convolution with reversed a
        conv = np.convolve(w, a[::-1], mode='full')
        grad = 2.0 * conv[n-1::-1]  # length n
        # step size adapt
        eta = step_factor / np.linalg.norm(grad) if np.linalg.norm(grad) != 0 else 0
        # exponentiated gradient update (to keep positivity)
        a = a * np.exp(-eta * grad)
        # renormalize
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full(n, 1.0 / n)
        else:
            a /= a_sum
        # schedule: reduce step_factor gradually
        step_factor *= 0.9995
        # increase beta slowly
        if iteration % 100 == 0:
            beta = min(beta * 1.05, 200)
    return best_a

"""
Propose a low‑score step‑function.

Algorithm
---------
1.  Build a small set of candidate sequences:
    * Uniform (baseline).
    * A power‑law shape (α≈0.506, δ=0) – good starting point from the previous run.
    * Several discretisations of the Cantor distribution.
       For a chosen integer k we set n = 3**k and assign
       a_i = 2**(-k) if the ternary representation of i contains only 0/2,
       otherwise a_i = 0.  This distribution has the remarkable property
       that its self‑convolution tends to the flat constant 1/(2n‑1), i.e.
       the objective value approaches the theoretical lower bound 1.
2.  Evaluate every candidate with a fast FFT‑based convolution
    (`fast_score`) and keep the best one.
3.  Refine the best candidate with two simple projected‑gradient schemes:
    * L2‑error reduction (the same routine used in the reference solution).
    * A soft‑max (log‑sum‑exp) approximation of the L∞‑norm.  The gradient
      of  Mβ = (1/β)·log Σ exp(β·b) w.r.t. a is
          g = 2·(w * a_rev)  (cross‑correlation),
      where w are the soft‑max weights of the convolution b = a*a.
      An exponentiated‑gradient step keeps the vector non‑negative,
      after which we project back onto the probability simplex.
    Both phases respect the constraints (a_i≥0, Σa_i=1) and run until the
    time budget (≈1000 s) is exhausted.
4.  Return the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear convolution (real input)
def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using an FFT of power‑of‑two length."""
    size = a.size + b.size - 1
    N = 1 << (size - 1).bit_length()          # next power of two
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    C = A * B
    conv = np.fft.irfft(C, N)[:size]
    return conv


# ----------------------------------------------------------------------
# Fast evaluation of the objective (sum of a is assumed to be 1)
def fast_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a probability vector a."""
    n = a.size
    b = convolve_fft(a, a)
    max_b = b.max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Simple power‑law family (used as a fallback initial guess)
def power_law_sequence(alpha: float, delta: float, n: int) -> np.ndarray:
    """a_i ∝ (i+1+δ)^{-α}, normalised to sum 1."""
    idx = np.arange(1.0, n + 1) + delta
    a = idx ** (-alpha)
    s = a.sum()
    if s > 0:
        a /= s
    return a


# ----------------------------------------------------------------------
# Cantor‑type distribution.
def generate_cantor_sequence(k: int) -> np.ndarray:
    """
    Build a discrete Cantor measure of length n = 3**k.
    a_i = 2**(-k) iff the ternary expansion of i contains only 0/2,
    otherwise a_i = 0.  The vector is automatically normalised.
    """
    n = 3 ** k
    weight = 2.0 ** (-k)          # each surviving point gets this weight
    a = np.zeros(n, dtype=np.float64)

    for i in range(n):
        x = i
        ok = True
        while x:
            if x % 3 == 1:        # a digit equal to 1 -> point removed
                ok = False
                break
            x //= 3
        if ok:
            a[i] = weight

    # numeric renormalisation (guards against rounding errors)
    s = a.sum()
    if s != 1.0:
        a /= s
    return a


# ----------------------------------------------------------------------
# Projection onto the probability simplex (non‑negative, sum = 1)
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=1}."""
    if np.sum(v) == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# L2‑error gradient step (identical to the reference code)
def gradient_refine_l2(a: np.ndarray, deadline: float) -> np.ndarray:
    """
    A few projected‑gradient steps that minimise the L2 error between
    a*a and the flat target 1/(2n‑1).
    """
    n = a.size
    target_const = (a.sum() ** 2) / (2 * n - 1)   # = 1/(2n‑1) after normalisation
    step_factor = 0.05
    best = a.copy()
    best_score = fast_score(a)

    while time.time() < deadline:
        b = convolve_fft(a, a)
        err = b - target_const
        # gradient of ‖err‖² w.r.t. a
        grad = 4.0 * np.correlate(err, a, mode='valid')   # length n
        gn = np.linalg.norm(grad)
        if gn == 0.0:
            break
        eta = step_factor / gn
        a = a - eta * grad
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= s
        cur = fast_score(a)
        if cur < best_score:
            best_score = cur
            best = a.copy()
            print(f'  L2‑refine improved score {best_score:.6f}')
            step_factor = max(step_factor * 0.95, 1e-4)
    return best

# Soft‑max (log‑sum‑exp) approximation of the L∞‑norm and exponentiated‑gradient
def gradient_refine_softmax(a: np.ndarray, deadline: float) -> np.ndarray:
    """
    Minimises a smooth approximation of max(b) where b = a*a.
    The update uses exponentiated gradient and a projection onto the simplex.
    """
    n = a.size
    beta = 1.0                  # start with a gentle softening
    step = 0.5                  # initial step‑size factor
    best = a.copy()
    best_score = fast_score(a)
    it = 0

    while time.time() < deadline:
        it += 1
        b = convolve_fft(a, a)
        max_b = b.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f'  softmax‑refine iter {it}: new best {best_score:.6f}')

        # soft‑max weights
        max_val = max_b
        exp_beta = np.exp(beta * (b - max_val))          # stability trick
        w = exp_beta / exp_beta.sum()                    # Σ w = 1, w_i ≥ 0

        # gradient = 2 * (w ⋆ a)  (cross‑correlation)
        conv = np.convolve(w, a[::-1], mode='full')
        grad = 2.0 * conv[n - 1 :: -1]                    # length n, reversed slice

        # adaptive step
        gn = np.linalg.norm(grad)
        if gn == 0.0:
            break
        eta = step / gn

        # exponentiated‑gradient update (preserves non‑negativity)
        a = a * np.exp(-eta * grad)

        # re‑normalise to the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        # schedule – slowly tighten step and increase β
        step = max(step * 0.9995, 1e-4)
        if it % 100 == 0:
            beta = min(beta * 1.05, 200.0)

    return best

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function with a low evaluation score.
    Returns the best sequence found as a Python list of floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # 1) generate a few plausible starting points
    candidates = []

    # uniform baseline (n≈3000, score = 2)
    n_unif = 3000
    candidates.append(('uniform', np.full(n_unif, 1.0 / n_unif)))

    # power‑law start – close to the best values obtained previously
    n_pl = 3000
    candidates.append(('powerlaw',
                       power_law_sequence(alpha=0.506, delta=0.0, n=n_pl)))

    # Cantor‑type families (different depths)
    for k in range(7, 12):            # 3⁷=2187 … 3¹¹=177147
        n_c = 3 ** k
        # keep the size reasonable (avoid >200 k elements)
        if n_c > 200_000:
            continue
        cand = generate_cantor_sequence(k)
        candidates.append((f'cantor_k{k}', cand))

    # ------------------------------------------------------------------
    # 2) pick the best raw candidate
    best_seq = None
    best_score = float('inf')
    best_name = None

    for name, seq in candidates:
        if time.time() >= deadline:
            break
        sc = fast_score(seq)
        print(f'Initial candidate {name:12s} | n={len(seq):6d} | score={sc:.6f}')
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            best_name = name

    print(f'Best raw candidate: {best_name} with score {best_score:.6f}')

    # ------------------------------------------------------------------
    # 3) first refinement – L2‑error (quick and cheap)
    if time.time() < deadline:
        print('Running L2‑error refinement...')
        best_seq = gradient_refine_l2(best_seq, deadline)

    # ------------------------------------------------------------------
    # 4) second refinement – soft‑max (focuses on the true L∞ objective)
    if time.time() < deadline:
        print('Running soft‑max (L∞) refinement...')
        best_seq = gradient_refine_softmax(best_seq, deadline)

    # ------------------------------------------------------------------
    # 5) final sanity check with the provided evaluator
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f'Final reported score (evaluate_sequence) = {final_score:.6f}')
    except Exception as exc:          # evaluate_sequence may not be present in some test harnesses
        print(f'Warning: could not call evaluate_sequence – {exc}')

    # ensure we return a plain Python list of floats
    return best_seq.tolist()
