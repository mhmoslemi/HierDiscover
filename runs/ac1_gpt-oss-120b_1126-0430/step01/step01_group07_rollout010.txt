```
def project_onto_simplex(v):
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

Will return vector sum=1, non-negative.

def convolve_fft_real(x, y):
    size = len(x) + len(y) - 1
    N = 1 << (size-1).bit_length()
    X = np.fft.rfft(x, N)
    Y = np.fft.rfft(y, N)
    Z = X * Y
    return np.fft.irfft(Z, N)[:size]

"""
propose_candidate – minimise 2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm starts from the analytically‑optimal inverse‑square‑root
distribution a_i ∝ (i+1)^(-1/2) (the extremiser of the continuous
self‑convolution inequality) and then runs a projected mirror‑descent
optimiser that directly reduces the supremum of the convolution.
A soft‑max approximation of the max‑operator is used so that the
gradient is well‑behaved; the smoothing parameter β is increased during
the run.  After each gradient step the vector is projected onto the
probability simplex (non‑negative and sum = 1).  The best sequence seen
is kept and finally returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast linear convolution (real FFT, next power of two)
def convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    size = x.size + y.size - 1
    N = 1 << (size - 1).bit_length()            # next power of two
    X = np.fft.rfft(x, N)
    Y = np.fft.rfft(y, N)
    Z = X * Y
    conv = np.fft.irfft(Z, N)[:size]
    return conv


# ----------------------------------------------------------------------
# Score used during the search (identical to the official one but
# without the input‑validation overhead).
def fast_score(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) / (sum(a)^2)  –  assumes a is non‑negative."""
    n = a.size
    b = convolve_fft(a, a)
    max_b = b.max()
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex (sum = 1, a_i ≥ 0)
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto the simplex."""
    # implementation of Duchi et al. (2008)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – fallback to uniform
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

    # 6) return a plain Python list (as required by the driver)
    # Clip tiny negative artefacts caused by floating‑point noise.
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return best_seq.tolist()
