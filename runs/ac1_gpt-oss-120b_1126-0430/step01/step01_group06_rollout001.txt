"""
Propose a step‑function (list of non‑negative heights) that tries to
minimise

    2 · n · max(conv(a,a)) / (sum(a)²)

The approach is a three‑stage heuristic:

1. **Broad parametric search** –  several families of simple
   parametric sequences are tried on a handful of problem sizes.
   •  monotone power‑law   a_i ∝ (i+shift)^(-α)
   •  symmetric “V‑shaped’’ power‑law
       a_i ∝ (min(i,n‑1‑i)+shift)^(-α)          (heavy at the ends)
   •  a mixture of the above with a uniform component
       a_i = w·base_i + (1‑w)·1 .
   For each family we sweep α, shift and the mixture weight w
   (all on a coarse grid).  The best sequence found so far is kept.

2. **Targeted hill‑climbing** – starting from the best sequence,
   the algorithm repeatedly moves a small amount of mass from an
   index that contributes strongly to the current maximal convolution
   entry to an index that contributes little.  After each move the
   vector is projected back onto the probability simplex.  Improvements
   are accepted greedily; the loop stops when the time budget runs out.

3. **Fallback** – if nothing beats the provided baseline
   (`height_sequence_1`) the baseline is returned.

The routine respects the 1000 s total time budget (≈950 s are used for
search, leaving a safety margin).  All helper functions are top‑level,
no closures or lambdas are used.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------
# Helper – projection onto the probability simplex {x >= 0, sum(x)=1}
# --------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Efficient Euclidean projection of a vector onto the simplex."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero – return uniform distribution
        return np.full(n, 1.0 / n, dtype=v.dtype)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w

# --------------------------------------------------------------
# Helper – evaluate a raw (unnormalised) vector
# --------------------------------------------------------------
def _evaluate_raw(vec: np.ndarray) -> float:
    """Normalise `vec` to sum 1 and evaluate it with the supplied function."""
    if np.allclose(vec, 0):
        return float('inf')
    norm = vec / np.sum(vec)
    # evaluate_sequence is assumed to be already imported in the environment
    return evaluate_sequence(norm.tolist())

# --------------------------------------------------------------
# Sequence factories (all return an unnormalised numpy array)
# --------------------------------------------------------------
def _make_monotone_powerlaw(n: int, alpha: float, shift: float) -> np.ndarray:
    """a_i ∝ (i + shift)^(-alpha)   (i = 0 … n‑1)."""
    idx = np.arange(n, dtype=float)
    return np.power(idx + shift, -alpha)

def _make_symmetric_v_powerlaw(n: int, alpha: float, shift: float) -> np.ndarray:
    """a_i ∝ (min(i, n‑1‑i) + shift)^(-alpha).  Symmetric, larger at the ends."""
    idx = np.arange(n, dtype=float)
    dist = np.minimum(idx, n - 1 - idx)
    return np.power(dist + shift, -alpha)

def _mix_with_uniform(base: np.ndarray, weight: float) -> np.ndarray:
    """Return w·base + (1‑w)·1 (uniform component)."""
    return weight * base + (1.0 - weight) * np.ones_like(base)

        # Pick a source index with probability proportional to its contribution
        if contrib.sum() > 0:
            src = np.random.choice(n, p=contrib / contrib.sum())
        else:
            src = np.random.randint(n)

        # Pick a target index that currently has a *small* contribution
        # (inverse‑weight sampling)
        inv = 1.0 / (contrib + 1e-12)
        tgt = np.random.choice(n, p=inv / inv.sum())

        if src == tgt:
            continue

        # Amount of mass to transfer from src to tgt
        step = cur[src] * (np.random.rand() * frac)
        if step <= 0.0:
            continue

        # Perform the move and re‑project onto the simplex
        new_vec = cur.copy()
        new_vec[src] -= step
        new_vec[tgt] += step
        new_vec = _project_to_simplex(new_vec)

        # Evaluate
        try:
            new_score = evaluate_sequence(new_vec.tolist())
        except Exception:
            continue

        if new_score < cur_score - 1e-9:          # strict improvement
            cur = new_vec
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = new_vec.tolist()
                print(f"[refine] iter={iter_cnt} improved score={new_score:.6f}")

    print(f"[final] best score found = {best_score:.6f} (length {len(best_seq)})")
    return best_seq
