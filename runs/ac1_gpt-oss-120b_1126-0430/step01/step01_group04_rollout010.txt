def _project_to_simplex(x: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Project a vector onto the simplex {y ≥ 0, sum(y) = target_sum}.
    O(n) algorithm based on the paper by Wang & Carreira‑Perpiñán (2013).
    """
    if target_sum <= 0:
        return np.zeros_like(x)

    # sort descending
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho = np.nonzero(u - cssv / np.arange(1, len(x) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(x - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Compute the linear convolution a * a using a real‑FFT.
    Returns the full convolution of length 2·len(a)‑1.
    """
    n = a.shape[0]
    # next power of two ≥ 2·n‑1 for efficient FFT
    m = 1 << ((2 * n - 1).bit_length())
    # rfft automatically zero‑pads to length m
    A = np.fft.rfft(a, n=m)
    C = A * A
    conv = np.fft.irfft(C, n=m)[: 2 * n - 1]
    return conv

            print(
                f"[sweep {sweep_iter:05d}] new best {real_score:.6f} via {shape}"
            )

    # --------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent (multiple short restarts)
    # --------------------------------------------------------------
    # Remaining budget after sweep
    remaining = budget_s - (time.time() - start_time) - 5.0   # keep a few seconds cushion
    if remaining <= 0:
        remaining = 0.0
    restarts = 3
    # allocate roughly equal time per restart
    time_per_restart = remaining / max(1, restarts)

    for rs in range(restarts):
        restart_start = time.time()
        # initialise either from the current best or a fresh random Dirichlet vector
        if rs == 0:
            a = best_seq.copy()
        else:
            raw = np.random.gamma(shape=0.7, scale=1.0, size=n)
            a = raw * (target_sum / raw.sum())
        it = 0
        step0 = target_sum * 0.5   # starting step size (≈ 50 % of the total mass)

        while True:
            it += 1
            # time guard for this restart
            if time.time() - restart_start > time_per_restart:
                break
            # compute full convolution and its maximum
            conv = _max_convolution_fft(a)
            M = conv.max()
            # keep track of the best overall solution
            if M < best_score:
                best_score = M
                best_seq = a.copy()
                print(f"[restart {rs} iter {it}] improved to {best_score:.6f}")

            # ----- sub‑gradient of the active maximal shift(s) -----
            tol = 1e-12
            active_shifts = np.where(conv >= M - tol)[0]
            g = np.zeros_like(a)
            for k in active_shifts:
                i_start = max(0, k - (n - 1))
                i_end = min(n - 1, k)
                idx = np.arange(i_start, i_end + 1, dtype=int)
                # contribution 2·a_{k‑i} for each i in the overlap interval
                g[i_start : i_end + 1] += 2.0 * a[k - idx]
            if len(active_shifts) > 0:
                g /= len(active_shifts)          # any convex combination is a valid sub‑gradient

            # direction that respects the simplex sum‑constraint
            mean_g = g.mean()
            d = -g + mean_g                       # zero‑sum direction

            # step size decays like 1/√t
            step = step0 / math.sqrt(it + 1.0)
            a_new = a + step * d
            # project back onto the simplex (ensures non‑negativity and Σa = target_sum)
            a = _project_to_simplex(a_new, target_sum)

    # --------------------------------------------------------------
    # Final verification & return
    # --------------------------------------------------------------
    final_score = _evaluate(best_seq.tolist())
    print(f"[final] best score after full search: {final_score:.6f}")
    # return a plain Python list as required by the external grader
    return best_seq.tolist()
