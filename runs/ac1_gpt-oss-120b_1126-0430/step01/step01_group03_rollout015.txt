"""
Propose a step‑function sequence that minimises

    evaluate_sequence(a) = 2·n·max_k (a*a)_k / (sum a)^2 .

The metric is scale‑invariant, so we work with probability vectors
p (sum(p)=1).  The objective becomes 2·n·max(p∗p).

Algorithm
---------
1.  Initialise a few candidate families:
        •  power‑law p_i ∝ (i+offset)^{-β}   (several β, offsets, lengths)
        •  a few random Dirichlet vectors.
    Keep the best candidate seen.

2.  For each candidate run a cheap projected‑gradient‑descent:
        •  Find the index k where the self‑convolution is maximal.
        •  The gradient of (p∗p)_k w.r.t. p_i is 2·p_{k-i} (0 when out of range).
        •  Perform a multiplicative update
                p_i ← p_i·exp(−η·p_{k-i}),
            then renormalise (projection onto the probability simplex).
        •  Accept the update only if the score improves;
            otherwise shrink the step size η.
        The loop stops after a fixed number of iterations or when the
        time budget expires.

3.  After the gradient phase a short stochastic “mass‑transfer’’ hill‑climb
    (the same idea as the reference code but now guided by the gradient)
    is performed to polish the solution.

All heavy work (convolutions and score calculations) is done with NumPy,
so we can afford many thousands of evaluations inside the 1000 s budget.
The routine returns the best sequence found (as a plain list of floats).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def fast_score(p: np.ndarray) -> float:
    """
    Compute 2·n·max(p∗p) for a probability vector p (sum(p)=1).
    This matches evaluate_sequence when the sequence is normalised.
    """
    n = p.shape[0]
    conv = np.convolve(p, p)
    return float(2.0 * n * conv.max())


def raw_score(seq: np.ndarray) -> float:
    """
    Exact wrapper for evaluate_sequence without calling the external function.
    Uses the same formula as evaluate_sequence, i.e.
        2·n·max(conv) / (sum(seq)**2)
    """
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    conv = np.convolve(seq, seq)
    return float(2.0 * n * conv.max() / (s * s))


def power_law_vector(n: int, beta: float, offset: float = 0.0) -> np.ndarray:
    """
    Return a normalised probability vector of length n with entries
        p_i ∝ (i+1+offset)^{-β}.
    """
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    a = idx ** (-beta)
    p = a / a.sum()
    return p


def gradient_optimize(
    p0: np.ndarray,
    deadline: float,
    cur_best_score: float,
    max_iters: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Projected multiplicative gradient descent on the objective
    2·n·max(p∗p).  Stops when time runs out or when no progress can be made.
    Returns the best p found and its score.
    """
    p = p0.copy()
    n = p.shape[0]
    best_local = p.copy()
    best_local_score = fast_score(p)

    # initial step size
    eta = 0.8

    it = 0
    while time.time() < deadline and it < max_iters:
        # convolution and index of the current maximum
        conv = np.convolve(p, p)
        k = int(np.argmax(conv))

        # compute the gradient vector g_i = p_{k-i} (zero when out of range)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(p)
        g[mask] = p[j[mask]]

        # multiplicative update (projected back to the simplex)
        p_next = p * np.exp(-eta * g)
        p_next_sum = p_next.sum()
        if p_next_sum == 0.0:          # numerical safety – re‑initialise randomly
            p_next = np.random.rand(n)
            p_next /= p_next.sum()
        else:
            p_next /= p_next_sum

        new_score = fast_score(p_next)

        # accept only improvements
        if new_score < best_local_score - 1e-12:
            p = p_next
            best_local = p.copy()
            best_local_score = new_score
            # a little larger step after a successful move
            eta = min(eta * 1.05, 1.0)
        else:
            # no improvement – shrink the step size
            eta *= 0.7
            if eta < 1e-6:
                break

        it += 1

    # Return the best vector discovered in this run
    return best_local, best_local_score


def stochastic_hillclimb(
    p0: np.ndarray,
    deadline: float,
    cur_best_score: float,
    max_steps: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Simple mass‑transfer hill‑climb (the same idea as the reference code).
    Works on a probability vector that already sums to 1.
    """
    p = p0.copy()
    n = p.shape[0]
    best_local = p.copy()
    best_local_score = fast_score(p)

    step = 0
    while time.time() < deadline and step < max_steps:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # move up to 5 % of the source mass
        eps = p[i] * np.random.rand() * 0.05
        if eps <= 0.0:
            continue
        p[i] -= eps
        p[j] += eps

        # stay on the simplex automatically
        new_score = fast_score(p)
        if new_score < best_local_score - 1e-12:
            best_local = p.copy()
            best_local_score = new_score
        step += 1

    return best_local, best_local_score


# ----------------------------------------------------------------------
# Main entry point required by the grader
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    evaluate_sequence.  The procedure explores several lengths, power‑law
    exponents and offsets, refines each candidate with a fast projected
    gradient descent and finishes with a short stochastic hill‑climb.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.10  # tiny safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0.  Use the previously supplied construction if it exists.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            sc = evaluate_sequence(height_sequence_1)
            best_seq = list(height_sequence_1)
            best_score = sc
            print(f"[init] height_sequence_1 score = {sc:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Parameter grids
    # ------------------------------------------------------------------
    lengths = [800, 1000, 1200, 1500, 1800, 2000]          # modest sizes
    betas = np.linspace(0.48, 0.72, 13)                    # focus around 0.6
    offsets = [0.0, 0.5, 1.0, 2.0]                        # shift the power‑law

    # ------------------------------------------------------------------
    # 2.  Scan the families
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() > deadline:
            break
        for beta in betas:
            if time.time() > deadline:
                break
            for off in offsets:
                # ----- build a power‑law candidate -----
                p = power_law_vector(n, beta, offset=off)

                # quick evaluation (probability vector → fast_score)
                sc = fast_score(p)
                if sc < best_score:
                    best_score = sc
                    best_seq = p.copy()
                    print(f"[scan] n={n:4d} β={beta:0.3f} off={off:0.1f} → {sc:.6f}")

                # ----- gradient optimisation -----
                p_opt, sc_opt = gradient_optimize(p, deadline, best_score)
                if sc_opt < best_score:
                    best_score = sc_opt
                    best_seq = p_opt.copy()
                    print(f"[grad] n={n:4d} β={beta:0.3f} off={off:0.1f} → {sc_opt:.6f}")

                # ----- final short hill‑climb -----
                if time.time() < deadline:
                    p_hc, sc_hc = stochastic_hillclimb(p_opt, deadline, best_score)
                    if sc_hc < best_score:
                        best_score = sc_hc
                        best_seq = p_hc.copy()
                        print(f"[hill] n={n:4d} β={beta:0.3f} off={off:0.1f} → {sc_hc:.6f}")

    # ------------------------------------------------------------------
    # 3.  Random Dirichlet restarts (if time permits)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        n = np.random.choice([800, 1000, 1200, 1500, 1800, 2000])
        p = np.random.rand(n)
        p /= p.sum()
        sc = fast_score(p)
        if sc < best_score:
            best_score = sc
            best_seq = p.copy()
            print(f"[rand] n={n:4d} → {sc:.6f}")

        p_opt, sc_opt = gradient_optimize(p, deadline, best_score, max_iters=800)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = p_opt.copy()
            print(f"[rand‑grad] n={n:4d} → {sc_opt:.6f}")

        # one tiny hill‑climb pass
        if time.time() < deadline:
            p_hc, sc_hc = stochastic_hillclimb(p_opt, deadline, best_score, max_steps=500)
            if sc_hc < best_score:
                best_score = sc_hc
                best_seq = p_hc.copy()
                print(f"[rand‑hill] n={n:4d} → {sc_hc:.6f}")

        # stop early if we already beat the target
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # 4.  Final verification with the official evaluator (optional)
    # ------------------------------------------------------------------
    if best_seq is not None:
        # The official function is scale‑invariant; using the raw probability
        # vector is fine.  Convert to a plain Python list for the return value.
        final_seq = best_seq.tolist()
        # One final call to the provided evaluator for safety (and to print)
        try:
            final_score = evaluate_sequence(final_seq)
            print(f"[final] verified score = {final_score:.6f}, length = {len(final_seq)}")
        except Exception:
            print("[final] verification failed – using internal score")
            final_score = best_score
        return final_seq

    # Fallback (should never happen)
    raise RuntimeError("No candidate sequence was generated.")
