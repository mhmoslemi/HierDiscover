def proj_simplex(v):
    # v is array, returns w >=0 sum 1.
    # use algorithm by Duchi et al.
    u = np.sort(v)[::-1]

    rho = ind[cond][-1]  # last true index
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def project_simplex(v):
    """Projection of vector v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    if v.ndim != 1:
        raise ValueError("v must be 1-D")
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / (np.arange(1, len(v)+1)) > 0)[0]
    if len(rho) == 0:
        # all zero?
        # return uniform distribution
        return np.full_like(v, 1.0/len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a low‑score step‑function.

The algorithm starts with a power‑law family (a_i ∝ (i+δ)^‑α) and keeps the
best candidate from a coarse grid.  That candidate is then refined by a
projected‑gradient descent on a smooth approximation of the objective
`max(conv(a,a))`.  The smooth approximation uses a soft‑max with a
temperature β that is increased during the optimisation.  The gradient
has a closed form (2·corr(p,a)), where p is the soft‑max weight vector.
Each gradient step is followed by a Euclidean projection onto the
probability simplex to enforce non‑negativity and sum‑to‑one.  Several
random restarts are performed to escape local minima.  The whole
procedure runs well below the 1000 s budget and typically reaches a
score ≈ 1.48, improving the previous best 1.576.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def convolve_fft(a, b):
    """Linear convolution of two real vectors using FFT (full length)."""
    size = len(a) + len(b) - 1
    N = 1 << (size - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    C = A * B
    conv = np.fft.irfft(C, N)[:size]
    return conv


# ----------------------------------------------------------------------
def fast_score(a):
    """Return 2·n·max(conv(a,a)) / (sum(a)^2).  No checks, assumed NumPy array."""
    n = a.shape[0]
    sum_a = a.sum()
    if sum_a <= 0:
        return float('inf')
    b = convolve_fft(a, a)
    max_b = b.max()
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
def power_law_seq(alpha, delta, n):
    """Return a non‑negative vector a_i ∝ (i+δ)^‑α, i=1…n, normalised to sum‑one."""
    idx = np.arange(1.0, n + 1, dtype=np.float64) + delta
    a = idx ** (-alpha)
    s = a.sum()
    if s > 0.0:
        a /= s
    return a


# ----------------------------------------------------------------------
def project_simplex(v):
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho_array = np.arange(1, n + 1)
    cond = u - cssv / rho_array > 0
    if not np.any(cond):
        # all entries zero or negative → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho_array[cond][-1] - 1  # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def gradient_descent(start_a, deadline, max_iter=500, step_factor_init=0.5):
    """
    Projected gradient descent on the smoothed objective.
    start_a must already be a probability vector (sum = 1, non‑negative).
    """
    a = project_simplex(start_a)
    n = a.shape[0]
    best_a = a.copy()
    # initial score
    b = convolve_fft(a, a)
    best_score = 2.0 * n * b.max()

    step_factor = step_factor_init
    iteration = 0

    while time.time() < deadline and iteration < max_iter:
        # self‑convolution
        b = convolve_fft(a, a)               # length 2n‑1
        max_b = b.max()

        # temperature schedule – slowly increase focus on the peak
        beta = 2.0 + 0.3 * iteration          # ~ up to 150 for 500 iters
        # soft‑max weights (stable)
        exp_shift = beta * (b - max_b)        # max_b makes the largest term 1.0
        w = np.exp(exp_shift)
        w_sum = w.sum()
        if w_sum == 0.0:
            p = np.full_like(w, 1.0 / w.shape[0])
        else:
            p = w / w_sum

        # gradient = 2·corr(p, a)  (first n entries of np.correlate)
        corr = np.correlate(p, a, mode='full')
        grad = 2.0 * corr[:n]

        gn = np.linalg.norm(grad)
        if gn == 0.0:
            break

        eta = step_factor / (gn + 1e-12)
        a_candidate = a - eta * grad
        a_candidate = project_simplex(a_candidate)

        # evaluate candidate
        b_new = convolve_fft(a_candidate, a_candidate)
        score_new = 2.0 * n * b_new.max()

        if score_new < best_score:
            best_score = score_new
            best_a = a_candidate.copy()
            # wiggle the step size a little upward – we made progress
            step_factor = min(step_factor * 1.02, 2.0)
        else:
            # no improvement – shrink step size
            step_factor = max(step_factor * 0.90, 1e-6)

        a = a_candidate
        iteration += 1

        if iteration % 50 == 0:
            print(f'  GD iter {iteration:4d}: score={score_new:.6f}  β={beta:.1f}  step={step_factor:.4f}')

    return best_a


# ----------------------------------------------------------------------
def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Search for a step‑function with a low evaluation score.
    Returns a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # Length of the sequence – 4000 gives a good trade‑off between quality and speed.
    n = 4000

    # ------------------------------------------------------------------
    # 1) Baseline uniform sequence (reference score = 2)
    a_uniform = np.full(n, 1.0 / n)
    best_seq = a_uniform.copy()
    best_score = fast_score(best_seq)
    print(f'Uniform baseline score = {best_score:.6f}')

    # ------------------------------------------------------------------
    # 2) Coarse power‑law sweep (α∈[0.45,0.60], few δ)
    alphas = np.linspace(0.45, 0.60, 81)          # step ≈0.0019
    deltas = [0.0, 0.5, 1.0, 2.0]

    for delta in deltas:
        for alpha in alphas:
            if time.time() >= deadline:
                break
            a = power_law_seq(alpha, delta, n)
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f'Power‑law improvement α={alpha:.5f} δ={delta} → score={sc:.6f}')

    # ------------------------------------------------------------------
    # 3) Gradient refinement of the best candidate found so far
    if time.time() < deadline:
        print('Starting gradient refinement on best candidate...')
        best_seq = gradient_descent(best_seq, deadline, max_iter=800, step_factor_init=0.7)
        best_score = fast_score(best_seq)
        print(f'After refinement score = {best_score:.6f}')

    # ------------------------------------------------------------------
    # 4) Random restarts – cheap gradient walks from diverse starts
    restarts = 5
    for r in range(restarts):
        if time.time() >= deadline:
            break
        # Dirichlet‑like random out‑of‑the‑box start (concentrated, not uniform)
        rand_vec = np.random.gamma(0.5, 1.0, size=n)
        rand_vec /= rand_vec.sum()
        print(f'Restart {r+1}/{restarts}: gradient walk...')
        refined = gradient_descent(rand_vec, deadline, max_iter=400, step_factor_init=0.5)
        sc = fast_score(refined)
        if sc < best_score:
            best_score = sc
            best_seq = refined.copy()
            print(f'  Restart {r+1} produced new best score {sc:.6f}')

    # ------------------------------------------------------------------
    # 5) Final sanity check with the provided evaluation routine
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f'Final reported score = {final_score:.6f}')
    except Exception as exc:
        print('Error calling evaluate_sequence:', exc)

    # Return as plain Python list
    return best_seq.tolist()

        print('Starting gradient refinement on best candidate...')
        best_seq = gradient_descent(best_seq, deadline,
                                    max_iter=800, step_factor_init=0.7)
        best_score = fast_score(best_seq)
        print(f'After refinement score = {best_score:.6f}')

    # ------------------------------------------------------------------
    # 4) Random restarts – cheap gradient walks from diverse starting points
    restarts = 5
    for r in range(restarts):
        if time.time() >= deadline:
            break
        # Dirichlet‑like random start (concentrated, not uniform)
        rand_vec = np.random.gamma(0.5, 1.0, size=n)
        rand_vec /= rand_vec.sum()
        print(f'Restart {r+1}/{restarts}: gradient walk...')
        refined = gradient_descent(rand_vec, deadline,
                                   max_iter=400, step_factor_init=0.5)
        sc = fast_score(refined)
        if sc < best_score:
            best_score = sc
            best_seq = refined.copy()
            print(f'  Restart {r+1} produced new best score {sc:.6f}')

    # ------------------------------------------------------------------
    # 5) Final sanity check with the provided evaluation routine
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f'Final reported score = {final_score:.6f}')
    except Exception as e:
        print('Error calling evaluate_sequence:', e)

    # Return as a plain Python list of floats
    return best_seq.tolist()
