        print(f"[debug] n={n}, p={p}, objective≈{obj:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[debug] unable to compute objective:", exc)

    # ---------------------------------------------------------
    # Return the sequence as a plain Python list of floats.
    return raw.tolist()
