    n = len(height_sequence_1) if defined else default

    best_score = evaluate_sequence(best_seq)

    # helper
    def evaluate(v):

        # to avoid floating errors, maybe renorm to sum=1
        tot = np.sum(v)
        if tot <= 0:
            return float('inf')
        # scaling for safety: maybe clip each element to 1000 before calling evaluate
        aux = (v / tot).astype(float).tolist()
        return evaluate_sequence(aux)

    # grid search parameterized families
    param_trials = []
    # using loops for alpha, shift
    alphas = np.linspace(0.35, 0.85, 51)   # step 0.01
    shifts = np.linspace(0.5, 5.0, 10)    # step 0.5
    # expo beta not for initial grid, just treat later
    for alpha in alphas:
        for shift in shifts:
            raw = np.power(np.arange(n)+shift, -alpha)
            score = evaluate(raw)
            if score < best_score:
                best_score = score
                best_seq = (raw/np.sum(raw)).tolist()
                print(f"[grid] alpha={alpha:.3f}, shift={shift:.2f}, score={score:.6f}")

    # random sampling for extra parameters
    while time.time()-start < budget_s*0.5: # use half time for random search
        alpha = np.random.uniform(0.3, 0.9)
        shift = np.random.uniform(0.2, 6.0)
        beta = np.random.uniform(0.0, 0.001)  # exponential decay
        i = np.arange(n)
        raw = (i+shift)**(-alpha) * np.exp(-beta*i)
        score = evaluate(raw)
        if score < best_score:
            best_score = score
            best_seq = (raw/np.sum(raw)).tolist()
            print(f"[rand] alpha={alpha:.3f}, shift={shift:.2f}, beta={beta:.6f} score={score:.6f}")

    # Local optimization with simulated annealing
    cur = np.array(best_seq, dtype=float)
    cur_score = best_score

    # parameters
    max_iter = 200000
    temp0 = 0.05 * cur_score  # initial temperature as fraction of cur_score
    cooling_rate = 0.9995  # geometric cooling
    temp = temp0
    iter = 0
    while time.time() - start < budget_s - 2 and iter < max_iter:
        iter += 1
        # pick two indices distinct
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # random step relative to source mass
        step_frac = np.random.rand() * 0.05   # up to 5%
        step = cur[i] * step_frac
        if step <= 0:
            continue
        # propose moving step from i to j
        new = cur.copy()
        new[i] -= step
        new[j] += step
        # ensure non-negative contributions: new[i]>=0 automatically because step <= cur[i]

def gradient_step(a):
    c = np.convolve(a,a)
    kstar = np.argmax(c)
    # compute contributions w_i for each i: if 0 <= kstar-i < n => a[kstar-i] else 0.
    idx = np.arange(n)
    partner_idx = kstar - idx
    mask = (partner_idx >= 0) & (partner_idx < n)
    w = np.zeros(n)
    w[mask] = a[partner_idx[mask]]
    # now source indices where w is largest
    # target indices where w is small
    # also consider current a_i values to ensure moving from those with high a_i.
    score = w * a  # contribution to c_kstar from each i?
    # Actually each term in c_kstar is a_i * a_{kstar-i} = a_i * w_i.
    contrib = a * w
    # choose source as max contrib, target as min contrib.
    src = np.argmax(contrib)
    tgt = np.argmin(contrib + 1e-12) # ensure it's not same
    # magnitude
    step = 0.1 * a[src]   # up to 10%
    # create new a
    a_new = a.copy()
    a_new[src] -= step
    a_new[tgt] += step
    # project onto simplex
    a_new = _project_to_simplex(a_new)

Hybrid search for low‑max‑autocorrelation step function.

- Stage 1: coarse grid over power‑law families a_i ∝ (i+shift)^−α.
- Stage 2: random sampling of extended three‑parameter families
  a_i ∝ (i+shift)^−α·exp(−β·i).
- Stage 3: deterministic gradient‑based descent on the index where the
  convolution attains its maximum, combined with a simple simulated‑annealing
  acceptance rule.  Moves consist of transferring a fraction of the mass
  from the position that contributes most to the current maximal autocorrelation
  to a position that contributes the least.  The algorithm runs until the
  allocated time budget expires and returns the best sequence found.
"""

import time
import numpy as np

def _project_to_simplex(v):
    """Project a vector onto the probability simplex {x>=0, sum(x)=1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / (np.arange(n) + 1) > 0)[0]
    if rho.size == 0:
        # all zero vector – return uniform distribution
        return np.full(n, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _normalize_to_one(v):
    """Scale vector to have sum equal to 1 (returns a new array)."""
    v = np.asarray(v, dtype=float)
    s = np.sum(v)
    if s <= 0:
        # uniform fallback
        return np.full_like(v, 1.0 / v.size)
    return v / s

def _conv_max(v):
    """Return (max value of linear convolution v*v, full convolution array)."""
    n = v.shape[0]
    m = 2 * n - 1
    size = 1 << (m - 1).bit_length()   # next power‑of‑2 >= m
    fv = np.fft.rfft(v, n=size)
    conv = np.fft.irfft(fv * fv, n=size)[:m]
    max_val = np.max(conv)
    return max_val, conv

    """Given max(conv) and length n, return the evaluation‑function value
    for a sequence whose entries sum to 1."""

"""
Hybrid search for a low‑max‑autocorrelation step function.

Stage 1 – a coarse grid over simple power‑law families 
    a_i ∝ (i+shift)^{‑α}.

Stage 2 – random exploration of a richer three‑parameter family
    a_i ∝ (i+shift)^{‑α}·exp(‑β·i).

Stage 3 – gradient‑based simulated‑annealing refinement.
          In each iteration we locate the index k* where the
          convolution (autocorrelation) attains its maximum.
          The contribution of position i to this maximal term is
          a_i·a_{k*‑i}.  We transfer a small fraction of the mass from
          the position with the largest contribution to one with the
          smallest contribution, re‑project onto the probability simplex
          (to keep the sum equal to 1) and accept the move with a
          Metropolis rule.  Occasionally a large random perturbation is
          inserted to escape stagnation.  The loop runs until the allotted
          time budget expires and the best sequence is returned.
"""

import time
import numpy as np


def _project_to_simplex(v):
    """Project a vector onto the probability simplex {x≥0, sum(x)=1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending sort
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries zero – fall back to uniform
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


def _normalize_to_one(v):
    """Scale a vector so that its entries sum to 1."""
    v = np.asarray(v, dtype=float)
    s = np.sum(v)
    if s <= 0:
        return np.full_like(v, 1.0 / v.size)
    return v / s


def _conv_max(v):
    """
    Compute the linear convolution v*v and return its maximum value
    together with the full convolution array.
    """
    n = v.shape[0]
    m = 2 * n - 1
    size = 1 << (m - 1).bit_length()   # next power of two ≥ m
    fv = np.fft.rfft(v, n=size)
    conv = np.fft.irfft(fv * fv, n=size)[:m]
    max_val = np.max(conv)
    return max_val, conv


def _score_from_max(max_val, n):
    """
    Convert max(conv) into the value returned by evaluate_sequence
    when the sequence sum equals 1:
        evaluate = 2 * n * max(conv)
    """
    return 2.0 * n * max_val

    print(f"[init] n={n}, baseline score (sum=1) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1. Grid search over simple power‑law families
    # ------------------------------------------------------------------
    alphas = np.linspace(0.45, 0.75, 61)   # 61 values, step ≈0.005
