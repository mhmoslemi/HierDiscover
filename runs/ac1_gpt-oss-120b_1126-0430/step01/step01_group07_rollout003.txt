def gradient_softmax(a, beta):
    # compute conv = a*a (len 2n-1)
    conv = convolve_fft(a, a)
    # compute weight softmax
    # to avoid overflow, subtract max conv
    maxc = np.max(beta * conv)
    exp_term = np.exp(beta * conv - maxc)
    w = exp_term / np.sum(exp_term)
    # compute gradient
    N = 1 << ((len(a) + len(w) - 1) - 1).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    grad = 2.0 * np.fft.irfft(A * np.conj(W), N)[:len(a)]
    # objective value
    log_sum_exp = (np.log(np.sum(exp_term)) + maxc) / beta
    obj = 2.0 * len(a) * log_sum_exp / (np.sum(a) ** 2)
    return obj, grad

def compute_smooth_obj_grad(a, beta):
    n = a.shape[0]
    conv = convolve_fft(a, a)  # length 2n-1
    # compute w = softmax(beta*conv)
    # Use stable: maxv = np.max(beta*conv)
    # But beta*conv maybe large (since conv <= 1 maybe) times 200 => up to 200, ok.
    scaled = beta * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / np.sum(exp_scaled)
    # compute gradient via correlation
    # compute FFT size
    size = len(a) + len(w) - 1
    N = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    grad = 4.0 * n * np.fft.irfft(A * np.conj(W), N)[:n]  # gradient of objective.
    # compute smooth objective M_approx = (1/beta) * (log sum exp(scaled) + max_scaled)
    logsumexp = max_scaled + np.log(np.sum(exp_scaled))
    M_approx = logsumexp / beta
    obj = 2.0 * n * M_approx  # denominator =1
    return obj, grad

def project_simplex(v):
    # ensure nonnegative and sum to 1.
    # Input: v (array)
    # Output: w (array) such that w >= 0 and sum w = 1, and minimal L2 distance to v.
    # We use Duchi et al. method.
    n = v.shape[0]
    u = np.sort(v)[::-1]  # descending
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    # find rho = max i where cond true
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w
```

```
def compute_smooth(a, beta):
    n = len(a)
    conv = convolve_fft(a, a)
    # stable softmax
    scaled = beta * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / np.sum(exp_scaled)
    # compute correlation via FFT
    size = n + len(w) - 1
    N = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    corr = np.fft.irfft(A * np.conj(W), N)[:n]
    grad = 4.0 * n * corr
    # smooth objective
    logsumexp = max_scaled + np.log(np.sum(exp_scaled))
    M_approx = logsumexp / beta
    obj = 2.0 * n * M_approx
    return obj, grad
```

def optimize_with_adam(a0, deadline, beta_start=5.0, beta_end=200.0,
                       max_iter=20000):
    a = a0.copy()
    n = len(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta = beta_start
    t = 0
    best_a = a.copy()
    best_score = fast_score(a)
    # schedule: increase beta exponentially
    beta_inc_factor = (beta_end / beta_start) ** (1.0 / max_iter)
    while time.time() < deadline and t < max_iter:
        t += 1
        obj_smooth, grad = compute_smooth(a, beta)
        # Adam parameters
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8
        lr = 0.5 / (t ** 0.5)  # decaying learning rate
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = project_simplex(a)
        # Gradual increase beta
        beta = min(beta_end, beta * beta_inc_factor)
        # periodically compute exact score
        if t % 100 == 0 or time.time() >= deadline:
            cur_score = fast_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f'Iter {t}, exact score {cur_score:.6f}, beta {beta:.2f}')
    return best_a, best_score

"""
Low‑score step‑function generator.

The algorithm works in three stages:

1. **Quick power‑law sweep** – for a chosen length *n* we scan a
   family a_i ∝ (i+δ)^‑α (α,δ on a modest grid) and keep the best
   sequence (exact score computed via FFT convolution).

2. **Smooth‑max Adam optimisation** – the true objective
   2·n·max(a*a)/(∑a)² is non‑smooth.  We replace the max by a
   log‑sum‑exp (soft‑max) with a temperature β and minimise the smooth
   surrogate using Adam.  β is slowly increased so that the surrogate
   converges to the true max.  After each Adam step we project onto the
   probability simplex (non‑negative, sum = 1).

3. **Final sanity check** – the best sequence is evaluated with the
   provided `evaluate_sequence` function and returned as a plain Python
   list.

The whole search respects the 1000 s wall‑clock budget and works with
any `seed` for reproducibility.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution (linear) using a power‑of‑two FFT.
def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of real vectors a and b via FFT."""
    size = a.size + b.size - 1
    N = 1 << (size - 1).bit_length()          # next power of two
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    C = A * B
    conv = np.fft.irfft(C, N)[:size]
    return conv


# ----------------------------------------------------------------------
# Exact score (the metric we ultimately want to minimise).
def fast_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) / (sum(a)²).  a is assumed non‑negative."""
    n = a.size
    b = convolve_fft(a, a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 1e-12:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Simple power‑law family (used for a fast warm start).
def power_law_sequence(alpha: float, delta: float, n: int) -> np.ndarray:
    """
    a_i ∝ (i + delta)^‑α for i = 1 … n, normalised to sum = 1.
    """
    idx = np.arange(1.0, n + 1, dtype=np.float64) + delta
    a = idx ** (-alpha)
    s = a.sum()
    if s > 0.0:
        a /= s
    return a


# ----------------------------------------------------------------------
# Projection onto the probability simplex (non‑negative, sum = 1).
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = 1}."""
    n = v.size
    u = np.sort(v)[::-1]                     # descending sort
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the uniform distribution
        return np.full(n, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smooth‑max surrogate and its gradient.
def compute_smooth_obj_grad(a: np.ndarray, beta: float):
    """
    Compute the smoothed objective
        fβ(a) = 2·n· (1/β)·log Σ_k exp(β·(a*a)_k)
    and its gradient w.r.t. a.
    Returns (fβ, grad) where grad has shape (n,).
    """
    n = a.size
    # self‑convolution
    conv = convolve_fft(a, a)                     # length 2n‑1

    # soft‑max weights w_k = exp(β·conv_k) / Σ exp(β·conv)
    scaled = beta * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)      # shift for stability
    w = exp_scaled / np.sum(exp_scaled)           # length 2n‑1, sums to 1

    # gradient of fβ : 4·n· (a ⋆ w)   where ⋆ is the (linear) cross‑correlation
    size = n + w.size - 1                         # = 3n‑2
    N = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    corr = np.fft.irfft(A * np.conj(W), N)[:n]    # a ⋆ w for lags 0 … n‑1
    grad = 4.0 * n * corr

    # value of the smooth objective
    logsumexp = max_scaled + np.log(np.sum(exp_scaled))
    M_approx = logsumexp / beta
    obj = 2.0 * n * M_approx

    return obj, grad


# ----------------------------------------------------------------------
# Adam optimiser on the simplex with a temperature schedule.
def optimise_adam(
    a0: np.ndarray,
    deadline: float,
    beta_start: float = 5.0,
    beta_end: float = 200.0,
    max_iter: int = 10000,
) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth‑max surrogate using Adam + simplex projection.
    Returns the best (exact) sequence found and its exact score.
    """
    a = a0.copy()
    n = a.size
    best_a = a.copy()
    best_score = fast_score(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # temperature schedule (geometric increase)
    beta = beta_start
    if max_iter > 1:
        beta_factor = (beta_end / beta_start) ** (1.0 / max_iter)
    else:
        beta_factor = 1.0

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        obj_smooth, grad = compute_smooth_obj_grad(a, beta)

        # Adam update (step size decays like 1/√t)
        lr = 0.5 / np.sqrt(t)       # simple decay; can be tuned
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the simplex
        a = project_simplex(a)

        # increase temperature
        beta = min(beta_end, beta * beta_factor)

        # occasional exact evaluation
        if t % 100 == 0 or time.time() + 0.2 >= deadline:
            cur_score = fast_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(
                    f"Iter {t:5d} – exact score {cur_score:.6f} – β {beta:.2f}"
                )

    return best_a, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a low‑score non‑negative step function.
    Returns the best sequence found as a list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Choose a problem size.  5000 works comfortably within the budget.
    n = 5000

    # --------------------------------------------------------------
    # 1) QUICK POWER‑LAW SWEEP (very cheap)
    alphas = np.linspace(0.45, 0.58, 71)      # step ≈ 0.0018
    deltas = [0.0, 0.5, 1.0, 2.0]

    best_seq = np.full(n, 1.0 / n)           # fallback
    best_score = fast_score(best_seq)
    best_alpha = best_delta = None

    for delta in deltas:
        for alpha in alphas:
            if time.time() >= deadline:
                break
            a = power_law_sequence(alpha, delta, n)
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                best_alpha, best_delta = alpha, delta
                print(
                    f"Coarse sweep – α={alpha:.4f} δ={delta} → score={sc:.6f}"
                )
        if time.time() >= deadline:
            break

    # --------------------------------------------------------------
    # 2) LOCAL REFINEMENT WITH ADAM (starting from the best power‑law)
    if time.time() < deadline:
        print(
            f"Starting Adam refinement from α={best_alpha:.4f} δ={best_delta}"
        )
        best_seq, best_score = optimise_adam(
            best_seq,
            deadline,
            beta_start=5.0,
            beta_end=200.0,
            max_iter=8000,
        )
        print(f"After refinement – exact score {best_score:.6f}")

    # --------------------------------------------------------------
    # 3) OPTIONAL SECOND RANDOM RESTART (time permitting)
    remaining = deadline - time.time()
    if remaining > 10.0:                     # give it at least a few seconds
        a_rand = np.random.rand(n).astype(np.float64)
        a_rand /= a_rand.sum()
        print("Random restart – launching brief Adam optimisation")
        seq_rand, score_rand = optimise_adam(
            a_rand,
            deadline,
            beta_start=5.0,
            beta_end=200.0,
            max_iter=4000,
        )
        if score_rand < best_score:
            best_score = score_rand
            best_seq = seq_rand
            print(f"Random restart produced new best: {best_score:.6f}")

    # --------------------------------------------------------------
    # Final sanity check with the provided evaluator
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final reported score = {final_score:.6f}")
    except Exception:
        pass

    return best_seq.tolist()
