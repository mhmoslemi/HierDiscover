"""
propose_candidate
-----------------

A three‑stage optimizer for the step‑function problem.

1. **Fast grid search** –  Scan a rich family of power‑law sequences
   a[i] ∝ (i+shift)^(-α).  Scoring is done with a cheap FFT based
   convolution (exact up to floating point) and the best few candidates
   are evaluated with the official ``evaluate_sequence`` routine.

2. **Gradient descent on the active convolution index** –  For a fixed
   sequence the index *k* where the convolution attains its maximum
   dominates the objective.  The sub‑gradient of ``max(conv(a,a))`` at
   that index is 2·a shifted by *k*.  A small step in the opposite
   direction, followed by a projection onto the probability simplex,
   reduces the maximum.  The step size decays over time; the loop runs
   until the time budget is almost exhausted.

3. **Fine‑grained hill‑climbing** –  A simple random‑mass‑transfer
   perturbation (preserving the total sum) is applied many times;
   moves that improve the true objective are kept.  This phase
   fine‑tunes the solution found by the gradient stage.

The code respects the 1000 s overall limit, prints progress messages,
and always returns a feasible (non‑negative, simplex‑projected) sequence.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Helper functions (must be top‑level – no closures)
# ------------------------------------------------------------

def _project_to_simplex(v):
    """Project a vector onto the probability simplex {x≥0, sum(x)=1}."""
    n = len(v)
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zeros – return uniform distribution
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a):
    """
    Compute full linear convolution a*a using FFT and return
    (max_index, max_value, full_convolution_array).
    The input is assumed to be a 1‑D NumPy array.
    """
    n = a.shape[0]
    # Length of the linear convolution
    m = 2 * n - 1
    # Next power of two for efficient FFT
    size = 1 << ((m - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:m]
    max_idx = int(np.argmax(conv))
    return max_idx, float(conv[max_idx]), conv


def _fast_score(a):
    """
    Fast proxy for evaluate_sequence when the vector already sums to 1.
    Since the score is scale‑invariant we can ignore the denominator.
    """
    n = a.shape[0]
    _, max_val, _ = _max_convolution_fft(a)
    return 2.0 * n * max_val   # denominator = 1 because sum(a)=1


# ------------------------------------------------------------
# Main candidate generator
# ------------------------------------------------------------

def propose_candidate(seed=123, budget_s=950, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) beats the
    current best bound.  The routine runs for at most ``budget_s`` seconds.
    """
    np.random.seed(seed)
    start = time.time()
    # --------------------------------------------------------
    # 0. Length of the test vector (fallback to the mentioned 2244)
    # --------------------------------------------------------
    try:
        n = len(height_sequence_1)           # provided by the harness
    except Exception:
        n = 2244

    # --------------------------------------------------------
    # 1. Baseline / fallback
    # --------------------------------------------------------
    fallback_seq = list(height_sequence_1) if 'height_sequence_1' in globals() \
                                            else [1.0 / n] * n
    try:
        best_score = evaluate_sequence(fallback_seq)
    except Exception:
        best_score = float('inf')
    best_seq = fallback_seq
    print(f"[init] n={n}, baseline score={best_score:.6f}")

    # --------------------------------------------------------
    # 2. Fast grid search over a two‑parameter power‑law family
    # --------------------------------------------------------
    # We generate many candidates, keep the top few (by cheap score),
    # then evaluate them exactly.
    alphas = np.linspace(0.30, 1.20, 91)   # 0.30 … 1.20
    shifts = np.linspace(0.0, 5.0, 11)    # 0,0.5,…,5
    idx = np.arange(n, dtype=float)

    # store (fast_score, raw_vector) for the best 8 candidates
    top_candidates = []

    for alpha in alphas:
        if time.time() - start > budget_s - 30:
            break
        for shift in shifts:
            raw = np.power(idx + shift, -alpha)
            # normalise to sum 1 (the exact objective is scale‑invariant)
            raw_sum = raw.sum()
            if raw_sum == 0:
                continue
            a = raw / raw_sum
            sc = _fast_score(a)
            # keep a sorted list of the best 8 so far
            if len(top_candidates) < 8:
                top_candidates.append((sc, a))
                top_candidates.sort(key=lambda x: x[0])
            else:
                if sc < top_candidates[-1][0]:
                    top_candidates[-1] = (sc, a)
                    top_candidates.sort(key=lambda x: x[0])

    # evaluate the selected candidates with the exact function
    for sc, a in top_candidates:
        try:
            true_sc = evaluate_sequence(a.tolist())
        except Exception:
            true_sc = float('inf')
        if true_sc < best_score:
            best_score = true_sc
            best_seq = a.tolist()
            print(f"[grid] exact eval improved to {best_score:.6f}")

    print(f"[grid] best score after grid search: {best_score:.6f}")

    # --------------------------------------------------------
    # 3. Gradient descent on the active convolution index
    # --------------------------------------------------------
    cur = np.array(best_seq, dtype=float)   # always lies on the simplex
    cur_score = best_score

    # hyper‑parameters
    max_grad_iter = 15000
    grad_iter = 0
    # initial learning rate (as a fraction of the current entries)
    base_eta = 0.5

    while (time.time() - start) < (budget_s - 20) and grad_iter < max_grad_iter:
        grad_iter += 1

        # compute the convolution and the index where it is maximal
        k_max, max_val, _ = _max_convolution_fft(cur)

        # construct the sub‑gradient of the max‑convolution term
        grad = np.zeros_like(cur)
        idx2 = k_max - np.arange(n)
        mask = (idx2 >= 0) & (idx2 < n)
        grad[mask] = 2.0 * cur[idx2[mask]]

        # learning rate decays linearly with time
        elapsed = time.time() - start
        eta = base_eta * (1.0 - elapsed / budget_s)

        # propose a gradient step and re‑project onto the simplex
        proposal = cur - eta * grad
        proposal = _project_to_simplex(proposal)

        # quick check with the fast proxy – only evaluate truly better ones
        fast_cur = _fast_score(cur)
        fast_prop = _fast_score(proposal)

        # accept if fast score is strictly smaller (gap > 1e‑12)
        if fast_prop < fast_cur - 1e-12:
            try:
                true_prop = evaluate_sequence(proposal.tolist())
            except Exception:
                true_prop = float('inf')
            if true_prop < cur_score:
                cur = proposal
                cur_score = true_prop
                if true_prop < best_score:
                    best_score = true_prop
                    best_seq = proposal.tolist()
                    print(f"[grad] iter={grad_iter}, improved score={best_score:.6f}")

    print(f"[grad] finished after {grad_iter} iterations, best score={best_score:.6f}")

    # --------------------------------------------------------
    # 4. Fine‑grained stochastic hill‑climbing
    # --------------------------------------------------------
    cur = np.array(best_seq, dtype=float)
    cur_score = best_score
    max_hc_iter = 40000
    hc_iter = 0

    while (time.time() - start) < (budget_s - 5) and hc_iter < max_hc_iter:
        hc_iter += 1
        # pick two distinct positions
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        # move a small fraction of mass from i to j (up to 5 % of a[i])
        step = cur[i] * (np.random.rand() * 0.05)
        if step <= 0:
            continue
        # ensure we do not create a negative entry at j after subtraction
        if cur[j] - step < 0:
            continue

        new = cur.copy()
        new[i] -= step
        new[j] += step
        # re‑project to guarantee feasibility (sum‑preserving)
        new = _project_to_simplex(new)

        # evaluate only when the fast proxy predicts improvement
        fast_cur = _fast_score(cur)
        fast_new = _fast_score(new)
        if fast_new < fast_cur - 1e-12:
            try:
                true_new = evaluate_sequence(new.tolist())
            except Exception:
                true_new = float('inf')
            if true_new < cur_score:
                cur = new
                cur_score = true_new
                if true_new < best_score:
                    best_score = true_new
                    best_seq = new.tolist()
                    print(f"[hill] iter={hc_iter}, improved score={best_score:.6f}")

    print(f"[final] best score found: {best_score:.6f}")
    return best_seq
