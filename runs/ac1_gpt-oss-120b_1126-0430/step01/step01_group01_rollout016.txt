def propose_candidate(seed=42, budget_s=1000, n=4000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5

    # start with arcsine (catalan) synthetic sequence
    seq = _catalan_sequence(n)   # as before, but generate floats.

    # Normalize? Not needed because score scale-invariant. We'll keep as is.

    best_seq = seq.copy()
    best_score = _fast_evaluation(seq)

    fft_len = 1 << ((2 * n - 1).bit_length())

    # Preallocate arrays maybe.

    # Greedy iteration
    while time.time() < deadline - 30:   # leave 30 sec for annealing
        # compute conv and max index
        fa = np.fft.rfft(seq, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[:2*n-1]
        M = conv.max()
        k = int(conv.argmax())
        # compute contributions
        i_arr = np.arange(n)
        j_arr = k - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        w = np.zeros_like(seq)
        w[mask] = seq[j_arr[mask]]
        contrib = seq * w
        src = int(np.argmax(contrib))
        sink = int(np.argmin(w))   # may be zero
        if seq[src] < 1e-12:
            break
        eps = 0.02 * seq[src]   # move 2% of src
        new_seq = seq.copy()
        new_seq[src] -= eps
        new_seq[sink] += eps
        score = _fast_evaluation(new_seq)
        if score < best_score - 1e-9:
            seq = new_seq
            best_seq = new_seq.copy()
            best_score = score
            # optionally increase epsilon? but keep same
        else:
            # apply small random jitter to avoid stagnation
            if np.random.rand() < 0.002:
                idx = np.random.randint(n)
                delta = np.random.randn() * 0.001
                new_seq = seq.copy()
                new_seq[idx] = max(0.0, new_seq[idx] + delta)
                seq = new_seq   # accept jitter

        # End loop condition: we might also stop after many iterations w/out improvement

    # Simulated annealing stage
    # Use current seq as start
    cur_seq = best_seq.copy()
    cur_score = best_score
    T0 = 1.0
    # Determine total annealing time remaining
    while time.time() < deadline:
        # temperature decays linearly
        remaining = deadline - time.time()
        fraction = remaining / (budget_s - (budget_s-5))
        T = T0 * fraction # or linear schedule
        # random move
        src = np.random.randint(n)
        # ensure src has some mass to move
        if cur_seq[src] < 1e-12:
            continue
        sink = np.random.randint(n)
        if src == sink:
            continue
        eps = np.random.rand() * 0.1 * cur_seq[src]  # up to 10% of src
        new_seq = cur_seq.copy()
        new_seq[src] -= eps
        new_seq[sink] += eps
        score = _fast_evaluation(new_seq)
        delta = score - cur_score
        if delta < 0 or np.exp(-delta / max(T, 1e-12)) > np.random.rand():
            cur_seq = new_seq
            cur_score = score
            if cur_score < best_score:
                best_seq = cur_seq.copy()
                best_score = cur_score

    print(f"Best score: {best_score}")
    return best_seq.tolist()

def _eg_refine(seq, deadline):
    n = seq.size
    best_seq = seq.copy()
    best_score = _fast_evaluation(best_seq)
    fft_len = 1 << ((2 * n - 1).bit_length())
    it = 0
    while time.time() < deadline:
        # compute convolution
        fa = np.fft.rfft(seq, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2*n - 1]
        M = conv.max()
        # active indices
        # Use relative tolerance
        tol = 1e-12
        active = np.where(conv >= M - tol)[0]
        # compute subgradient
        g = np.zeros_like(seq)
        # loop over active
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            i_range = np.arange(lo, hi + 1)
            j_range = k - i_range
            g[i_range] += seq[j_range]
        # step size schedule
        eta = 0.5 / np.sqrt(it + 1)
        # multiplicative update
        seq = seq * np.exp(-eta * g)
        # ensure positivity
        seq = np.maximum(seq, 1e-15)
        # renormalize to keep scale (we can fix sum to 1)
        seq_sum = seq.sum()
        if seq_sum == 0:
            # reinitialize to uniform to avoid zero
            seq = np.full(n, 1.0 / n)
        else:
            seq /= seq_sum
        # evaluate
        score = _fast_evaluation(seq)
        # keep best
        if score < best_score - 1e-12:
            best_seq = seq.copy()
            best_score = score
            # optionally print
            print(f"EG iter {it}: improved score {best_score:.6f}")
        # occasional jitter
        if np.random.rand() < 0.001:

"""
Hybrid greedy + exponentiated‑gradient search for a low‑score step‑function.

The algorithm
----------------
* Start from the discrete arcsine (Catalan) seed – the coefficients of
  (1‑z)^‑1/2 – which already give a score ≈ π/2 ≈ 1.5708.
* A fast greedy local move repeatedly shifts a small fraction of mass from
  the index that contributes the most to the current maximal convolution
  entry to an index that contributes almost nothing.  This never increases
  the objective and quickly lowers the peak of the self‑convolution.
* After the greedy phase the remaining time is spent on an exponentiated
  gradient descent on the simplex.  The sub‑gradient of the maximum
  convolution is the sum of the sequence values that appear in the active
  (maximal) convolution positions.  Multiplicative updates push mass away
  from those positions while the simplex projection keeps the coefficients
  non‑negative and normalised (the evaluation metric is scale‑invariant).

The whole routine respects the 1000 s budget, prints intermediate scores
and finally returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast evaluation (identical to evaluate_sequence but without the security
# checks and using an FFT based convolution).
# ----------------------------------------------------------------------
def _fast_evaluation(seq: np.ndarray) -> float:
    """Return the same metric as evaluate_sequence but much faster."""
    n = seq.size
    if n == 0:
        return np.inf
    # FFT size = next power of two >= 2*n-1
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Catalan (arcsine) seed.
# a_0 = 1, a_{i+1} = a_i * (2*i+1) / (2*i+2)
# ----------------------------------------------------------------------
def _catalan_sequence(length: int) -> np.ndarray:
    """Return the first ``length`` Catalan‑like coefficients."""
    seq = np.empty(length, dtype=float)
    a = 1.0
    seq[0] = a
    for i in range(1, length):
        a *= (2 * i - 1) / (2.0 * i)
        seq[i] = a
    return seq


# ----------------------------------------------------------------------
# Simple greedy refinement – move mass from indices that heavily influence
# the current maximal convolution entry to indices that influence it hardly
# at all.
# ----------------------------------------------------------------------
def _refine(seq: np.ndarray, deadline: float) -> np.ndarray:
    """Greedy local improvement; stops when the deadline is reached."""
    n = seq.size
    best_seq = seq.copy()
    best_score = _fast_evaluation(best_seq)

    # pre‑compute FFT length (does not change)
    fft_len = 1 << ((2 * n - 1).bit_length())

    it = 0
    while time.time() < deadline:
        # convolution and its maximum
        fa = np.fft.rfft(seq, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        k = int(conv.argmax())
        max_val = conv[k]

        # contribution weights for each position to that maximal entry
        i_arr = np.arange(n)
        j_arr = k - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        w = np.zeros_like(seq)
        w[mask] = seq[j_arr[mask]]

        # source = index with largest contribution, sink = index with almost
        # no contribution to this maximal entry
        contrib = seq * w
        src = int(np.argmax(contrib))
        sink = int(np.argmin(w))

        if seq[src] < 1e-12:
            break

        # move a modest fraction of the mass (5 % of the source amount)
        eps = 0.05 * seq[src]
        new_seq = seq.copy()
        new_seq[src] -= eps
        new_seq[sink] += eps

        score = _fast_evaluation(new_seq)

        if score < best_score - 1e-12:
            seq = new_seq
            best_seq = new_seq.copy()
            best_score = score
            # occasional log
            if it % 1000 == 0:
                print(f"[greedy] iter {it:6d}  score {best_score:.6f}")
        else:
            # tiny random jitter to escape flat plateaus
            if np.random.rand() < 0.01:
                jitter_idx = np.random.randint(n)
                jitter = np.random.randn() * 0.001
                new_seq = seq.copy()
                new_seq[jitter_idx] = max(0.0, new_seq[jitter_idx] + jitter)
                seq = new_seq
        it += 1
    return best_seq


# ----------------------------------------------------------------------
# Exponentiated‑gradient refinement on the simplex.
# ----------------------------------------------------------------------
def _eg_refine(seq: np.ndarray, deadline: float) -> np.ndarray:
    """
    Exponentiated‑gradient descent that directly minimises the maximal
    self‑convolution.  ``seq`` is assumed to be non‑negative and normalised
    to have sum = 1 (the metric is scale‑invariant).
    """
    n = seq.size
    best_seq = seq.copy()
    best_score = _fast_evaluation(best_seq)

    fft_len = 1 << ((2 * n - 1).bit_length())
    it = 0
    while time.time() < deadline:
        # convolution and its maximal value
        fa = np.fft.rfft(seq, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        M = conv.max()

        # active indices (those attaining the maximum up to a tiny tolerance)
        tol = 1e-12
        active = np.where(conv >= M - tol)[0]

        # sub‑gradient of the max‑operator:
        # g_i = sum_{k in active} a_{k-i}  (zero where index out of range)
        g = np.zeros_like(seq)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            i_range = np.arange(lo, hi + 1)
            j_range = k - i_range
            g[i_range] += seq[j_range]

        # step‑size schedule (larger at the beginning, decays as 1/√t)
        eta = 0.5 / np.sqrt(it + 1)

        # multiplicative update (exponentiated gradient)
        seq = seq * np.exp(-eta * g)

        # keep positivity, renormalise to the simplex
        seq = np.maximum(seq, 1e-15)
        seq_sum = seq.sum()
        if seq_sum == 0.0:                     # extremely unlikely
            seq = np.full(n, 1.0 / n)
        else:
            seq /= seq_sum

        # evaluate
        score = _fast_evaluation(seq)
        if score < best_score - 1e-12:
            best_seq = seq.copy()
            best_score = score
            if it % 500 == 0:
                print(f"[EG]   iter {it:6d}  score {best_score:.6f}")

        # very occasional random jitter to avoid tiny stagnation
        if np.random.rand() < 0.001:
            idx = np.random.randint(n)
            seq[idx] = max(0.0, seq[idx] + np.random.randn() * 1e-3)
            seq /= seq.sum()

        it += 1
    return best_seq


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights aiming at a low evaluation score.

    The routine works within ``budget_s`` seconds (default 1000).  It
    builds a Catalan (arcsine) seed, performs a greedy mass‑transfer
    refinement and finishes with an exponentiated‑gradient descent on the
    simplex.  The best sequence discovered is returned as a plain Python
    list.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5   # safety margin

    # Length – a few thousand gives a good trade‑off between quality and speed.
    n = 5000

    # 1.  Initial seed (Catalan / arcsine coefficients) and normalisation.
    seq = _catalan_sequence(n).astype(float)
    seq /= seq.sum()                     # scale to sum = 1 (any positive sum is fine)

    print(f"[init] length={n}  initial score={_fast_evaluation(seq):.6f}")

    # 2.  Greedy refinement – use roughly the first 30 % of the budget.
    gre_deadline = start + int(0.3 * (budget_s - 5))
    if gre_deadline > time.time():
        seq = _refine(seq, gre_deadline)
        cur_score = _fast_evaluation(seq)
        print(f"[after greedy] score={cur_score:.6f}")

    # 3.  Exponentiated‑gradient refinement for the remaining time.
    seq = _eg_refine(seq, deadline)
    final_score = _fast_evaluation(seq)
    print(f"[final] best score={final_score:.6f}")

    # Return the best sequence as a plain Python list.
    return seq.tolist()
