def project_simplex(v):
    # v is np array
    # ensure nonneg and sum to 1
    # reference: Duchi et al. 2008.
    # Might contain negative values.
    # Step: sort v descending.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‐function (non‑negative vector) that minimises
``evaluate_sequence``.

The algorithm is a *soft‑max projected gradient descent* on the
objective  

    f(a) = max_k (a∗a)[k]      (sum(a)=1, a≥0)

The max is replaced by a temperature‑controlled soft‑max,
so the gradient is a weighted sum of all convolution entries.
After each gradient step the vector is projected onto the
probability simplex (non‑negative, unit sum).  The temperature and
learning rate are slowly annealed; occasional random restarts help
escape plateaus.

Fast convolutions are performed with the FFT (O(n log n)), which
allows thousands of iterations for n≈3000 within the 1000 s budget.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------


def _catalan_sequence(n: int) -> np.ndarray:
    """
    First n Catalan‑like coefficients a_i = C(2i,i)/4**i,
    normalised to sum 1.
    """
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * (i - 1) + 1) / (2.0 * i)
    a /= a.sum()
    return a


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x : x_i >= 0, sum x_i = 1}.
    """
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive – fall back to uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max() - 1  # zero‑based index
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear (full) convolution using FFT.  Returns array of length
    len(x)+len(y)-1.
    """
    n = len(x) + len(y) - 1
    # next power of two for efficient FFT
    N = 1 << (n - 1).bit_length()
    X = np.fft.rfft(x, N)
    Y = np.fft.rfft(y, N)
    conv = np.fft.irfft(X * Y, N)[:n]
    return conv

# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    The routine runs for at most *budget_s* seconds and returns the
    best sequence found (as a plain Python list).

    The optimisation uses a temperature‑controlled soft‑max gradient
    descent with simplex projection and occasional random restarts.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0  # safety margin

    # --------------------------------------------------------------
    # problem size – a few thousand entries is a good trade‑off
    # --------------------------------------------------------------
    n = 3000

    # --------------------------------------------------------------
    # initialise – start from the near‑optimal Catalan construction
    # --------------------------------------------------------------
    a = _catalan_sequence(n)

    # keep the best vector seen so far
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # hyper‑parameters
    # --------------------------------------------------------------
    lr = 0.12                     # base learning rate
    lr_decay = 0.995              # applied every 500 iterations
    temp = max(a.max() * 0.1, 1e-8)   # initial temperature (≈10 % of max conv)
    temp_decay = 0.9995
    min_temp = 1e-8

    iteration = 0
    no_improve = 0
    restart_counter = 0

    while time.time() < deadline:
        # ---- compute convolution and soft‑max weights ----
        conv = _full_convolve(a, a)          # length 2n‑1
        M = conv.max()
        # anneal temperature (never below min_temp)
        if temp > min_temp:
            temp *= temp_decay

        # soft‑max over convolution entries
        # shift by M for numerical stability
        w = np.exp((conv - M) / temp)
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.ones_like(conv) / conv.size
        else:
            w /= w_sum

        # ---- gradient of the smoothed max ----
        a_rev = a[::-1]                     # reversed a
        grad_full = _full_convolve(a_rev, w)   # length 3n‑2
        grad = 2.0 * grad_full[n - 1:n - 1 + n]   # extract the n‑length gradient

        # ---- take a gradient step and project onto the simplex ----
        a_next = a - lr * grad
        a_next = _project_to_simplex(a_next)

        a = a_next

        # ---- evaluate the current candidate ----
        cur_score = evaluate_sequence(a.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{iteration:06d}] new best = {best_score:.6f} "
                  f"(temp={temp:.2e}, lr={lr:.4f})")
            no_improve = 0
        else:
            no_improve += 1

        # ---- occasional learning‑rate decay ----
        if iteration % 500 == 0 and iteration > 0:
            lr *= lr_decay

        # ---- restart if stagnation persists ----
        if no_improve > 3000:
            restart_counter += 1
            a = np.random.rand(n)
            a = _project_to_simplex(a)
            lr = 0.12
            temp = max(a.max() * 0.1, 1e-8)
            no_improve = 0
            print(f"[{iteration:06d}] restart #{restart_counter}")

        iteration += 1

    print(f"[done] best score = {best_score:.6f} after {iteration} iterations")
    return best_seq.tolist()
