"""Propose a high‑quality step‑function sequence.

The search works in three stages:

1. **Parametric scan** – several families of non‑negative sequences are
   generated (power‑law, exponential, symmetric Beta, and the special
   arcsine case which comes from a Beta(½,½) distribution).  For a few
   lengths (≈500‑2500) we evaluate every candidate with the provided
   `evaluate_sequence` and keep the best one.

2. **Fast local refinement** – starting from the best candidate we perform
   a hill‑climb that moves a small portion of the mass from one index
   to another.  The self‑convolution is updated *incrementally* (only the
   three affected indices 2i, 2j and i+j change), so each move is cheap
   and we can afford many iterations within the time budget.

3. **Return** – the best sequence found (as a plain list of floats) is
   returned.  All numbers are non‑negative and ≤ 1000, satisfying the
   security checks of `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – each is a top‑level function (no closures)
# ----------------------------------------------------------------------


def make_powerlaw(n: int, beta: float) -> np.ndarray:
    """a_i = (i+1)^{-beta}  (i = 0 … n‑1)."""
    return np.arange(1, n + 1, dtype=np.float64) ** (-beta)


def make_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam * i)."""
    return np.exp(-lam * np.arange(n, dtype=np.float64))


def make_beta_dist(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Symmetric Beta(α,β) density sampled at the mid‑points of n bins:
    x_i = (i+0.5)/n,  a_i ∝ x_i^{α‑1} (1‑x_i)^{β‑1}.
    """
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # avoid exact 0/1 – the +0.5 offset already guarantees 0<x<1
    return (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))


def make_arcsine(n: int) -> np.ndarray:
    """Special case α=β=0.5 of the Beta distribution (continuous arcsine)."""
    return make_beta_dist(n, 0.5, 0.5)


def safe_evaluate(seq: np.ndarray) -> float:
    """
    Wrapper around the external `evaluate_sequence`.  The function expects a
    Python list, clips the values to the allowed range and returns the score.
    """
    try:
        return evaluate_sequence(seq.tolist())
    except Exception:
        return float("inf")


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a step‑function (a list of non‑negative heights) with a low
    evaluation score.  The routine respects a ~1000 s time budget,
    performs a systematic scan of several parametric families,
    then refines the best candidate with a fast hill‑climb using incremental
    convolution updates.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.1   # keep a tiny safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0.  Take the previously discovered construction if it exists.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            sc = evaluate_sequence(height_sequence_1)
            best_seq = list(height_sequence_1)
            best_score = sc
            print(f"[init] Using height_sequence_1, score = {sc:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Systematic scan over a few lengths and families.
    # ------------------------------------------------------------------
    lengths = [500, 800, 1000, 1500, 2000, 2500]

    for n in lengths:
        if time.time() > deadline:
            break

        # ---- power‑law -------------------------------------------------
        for beta in np.linspace(0.45, 0.80, 36):  # ~0.01 steps
            if time.time() > deadline:
                break
            seq = make_powerlaw(n, beta)
            seq = np.clip(seq, 0.0, 1000.0)   # safeguard against the 1000‑cap
            sc = safe_evaluate(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.tolist()
                print(f"[scan] n={n}, powerlaw β={beta:.3f} → {sc:.6f}")

        # ---- exponential -----------------------------------------------
        for lam in np.linspace(0.0005, 0.0300, 40):
            if time.time() > deadline:
                break
            seq = make_exponential(n, lam)
            seq = np.clip(seq, 0.0, 1000.0)
            sc = safe_evaluate(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.tolist()
                print(f"[scan] n={n}, exponential λ={lam:.5f} → {sc:.6f}")

        # ---- symmetric Beta (including the arcsine limit) ---------------
        for alpha in np.linspace(0.30, 2.00, 18):   # α=β
            if time.time() > deadline:
                break
            seq = make_beta_dist(n, alpha, alpha)
            seq = np.clip(seq, 0.0, 1000.0)
            sc = safe_evaluate(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.tolist()
                print(f"[scan] n={n}, beta α=β={alpha:.2f} → {sc:.6f}")

        # ---- explicit arcsine (α=β=0.5) ---------------------------------
        seq = make_arcsine(n)
        seq = np.clip(seq, 0.0, 1000.0)
        sc = safe_evaluate(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.tolist()
            print(f"[scan] n={n}, arcsine (β=0.5) → {sc:.6f}")

    # ------------------------------------------------------------------
    # 2.  Fast hill‑climb on the current best sequence.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        seq = np.array(best_seq, dtype=np.float64)
        n = len(seq)
        total_mass = seq.sum()                     # stays constant during moves
        conv = np.convolve(seq, seq)               # full convolution (len = 2n‑1)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (total_mass * total_mass)

        best_local_seq = seq.copy()
        best_local_score = cur_score

        # Parameters for the stochastic mass transfer
        max_fraction = 0.20         # move at most 20 % of the source entry
        iter_cnt = 0

        while time.time() < deadline:
            i = np.random.randint(0, n)
            j = np.random.randint(0, n)
            if i == j:
                continue

            eps = seq[i] * np.random.rand() * max_fraction
            if eps <= 0.0:
                continue

            a_i = seq[i]
            a_j = seq[j]

            # indices that change in the convolution
            idx_i = 2 * i
            idx_j = 2 * j
            idx_k = i + j

            # ----- compute the three delta‑contributions -----------------
            delta_i = -2.0 * a_i * eps + eps * eps                     # at 2i
            delta_j = 2.0 * a_j * eps + eps * eps                      # at 2j
            delta_k = 2.0 * eps * (a_i - a_j) - 2.0 * eps * eps        # at i+j

            # accumulate increments (occasionally indices can coincide,
            # e.g. when i=0 or j=0 – the dict handles that safely)
            inc = {}
            for idx, d in ((idx_i, delta_i), (idx_j, delta_j), (idx_k, delta_k)):
                inc[idx] = inc.get(idx, 0.0) + d

            # store old convolution values for a possible revert
            old_vals = {}
            for idx, d in inc.items():
                old_vals[idx] = conv[idx]
                conv[idx] = conv[idx] + d

            # apply the move to the sequence
            seq[i] = a_i - eps
            seq[j] = a_j + eps

            new_max = conv.max()
            new_score = 2.0 * n * new_max / (total_mass * total_mass)

            if new_score < cur_score:
                # accept the move
                cur_score = new_score
                cur_max = new_max
                if new_score < best_local_score:
                    best_local_score = new_score
                    best_local_seq = seq.copy()
                    print(f"[hill] improved to {new_score:.6f}")
            else:
                # reject – revert both the sequence and the convolution
                seq[i] = a_i
                seq[j] = a_j
                for idx, old in old_vals.items():
                    conv[idx] = old

            iter_cnt += 1

        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local_seq.tolist()
            print(f"[final hill] best score after local search = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Final report
    # ------------------------------------------------------------------
    final_len = len(best_seq) if best_seq is not None else 0
    print(f"[final] best score = {best_score:.6f}, length = {final_len}")
    return best_seq
