def hill_climb(seq, deadline):
    best_seq = seq.copy()
    best_score = fast_eval(seq)
    n = len(seq)
    while time.time() < deadline:
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j: continue
        if best_seq[i] < 1e-12: continue
        eps = 0.01 * best_seq[i] * np.random.rand()  # random fraction up to 1%
        new_seq = best_seq.copy()
        new_seq[i] -= eps
        new_seq[j] += eps
        score = fast_eval(new_seq)
        if score < best_score - 1e-12:
            best_seq = new_seq
            best_score = score
    return best_seq

def _hill_climb(seq, deadline):
    best_seq = seq.copy()
    best_score = _fast_evaluation(best_seq)
    n = len(seq)
    while time.time() < deadline:
        # random indices
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        if best_seq[i] < 1e-12:
            continue
        # Choose fractional move eps relative to source amount
        frac = np.random.rand() * 0.1  # up to 10%
        eps = best_seq[i] * frac
        if eps <= 0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= eps
        new_seq[i] = max(new_seq[i], 0.0)
        new_seq[j] += eps
        score = _fast_evaluation(new_seq)
        if score < best_score:
            best_seq = new_seq
            best_score = score
        # optionally early break if improvement is tiny?
    return best_seq

def apply_move(a, conv, i, j, delta):
    # delta positive moving from i to j
    n = len(a)
    # compute indices where a_i interacts
    # create copy of conv
    new_conv = conv.copy()
    # For k values where i+r in range:
    # Linear term: -2*delta*a[r] for each r where r in 0..n-1 and i+r within 0..2n-2
    r = np.arange(n)
    k_i = i + r
    mask_i = (k_i < 2*n-1)
    new_conv[k_i[mask_i]] -= 2*delta * a[r[mask_i]]
    # For k where j+r
    k_j = j + r
    mask_j = (k_j < 2*n-1)
    new_conv[k_j[mask_j]] += 2*delta * a[r[mask_j]]
    # Add quadratic contributions:

    # conv[i+j] -= 2*delta**2

    new_conv[2*j] += delta**2
    new_conv[i+j] -= 2*delta**2
    # Update a:
    a[i] -= delta
    a[j] += delta
    return a, new_conv

"""
Search for a step‑function (non‑negative sequence) with a lower
evaluation score than the classical arcsine (Catalan) seed.
The idea:

1.  Start from the Catalan / arcsine coefficients a_i = C_i =
    binom(2*i,i) / 4**i   (self‑convolution is exactly 1 for the first
    |a| entries).

2.  The value of the evaluation metric is
        score = 2·n·max(conv) / (sum(a)**2) .
    Because the sum is fixed, reducing the peak of the self‑convolution
    directly improves the score.

3.  The peak of the convolution is 1 and is attained at index 0
    (a_0² = 1).  By moving a small amount of mass from the first
    coefficient to a far‑away index we decrease a_0 while the
    contributions of the far‑away coefficient stay tiny, so the new
    maximum stays below the old one.  Repeating this “mass‑transfer”
    step reduces max(conv) below 1 and therefore pushes the score
    below π/2 ≈ 1.5708.

4.  The move is performed analytically: the convolution after a move
    can be updated in O(n) time using the linearised formula
        Δconv = 2·δ·(a_{k‑j} – a_{k‑i})  (plus tiny quadratic terms).
    This allows thousands of cheap iterations within the 1000 s budget.

5.  The algorithm stops when no move can decrease the maximum any
    further or when the time budget is exhausted.  The best sequence
    found is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast evaluation – identical to the provided evaluate_sequence but without
# the security checks.  Used only for logging/diagnostics.
# ----------------------------------------------------------------------
def _fast_evaluation(seq: np.ndarray) -> float:
    n = seq.size
    if n == 0:
        return np.inf
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Catalan / arcsine seed: a_i = binom(2*i,i) / 4**i.
# ----------------------------------------------------------------------
def _catalan_sequence(length: int) -> np.ndarray:
    seq = np.empty(length, dtype=float)
    a = 1.0
    seq[0] = a
    for i in range(1, length):
        a *= (2 * i - 1) / (2.0 * i)   # recurrence a_i = a_{i-1} * (2i-1)/(2i)
        seq[i] = a
    return seq


# ----------------------------------------------------------------------
# Incremental update of both the sequence and its self‑convolution
# after moving a quantity `delta` from index `src` to index `dst`.
# ----------------------------------------------------------------------
def _apply_move(seq: np.ndarray,
                conv: np.ndarray,
                src: int,
                dst: int,
                delta: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Return a new sequence and the corresponding convolution after moving
    `delta` (>0) mass from `src` to `dst`.  The update is performed in
    O(n) time using the linearised change formula.
    """
    n = seq.size
    # keep a view of the original values for the linear part
    a = seq
    # ----- convolution update -----
    new_conv = conv.copy()

    # linear contribution of the source (subtraction)
    r = np.arange(n)
    k_src = src + r
    mask_src = k_src < 2 * n - 1
    new_conv[k_src[mask_src]] -= 2.0 * delta * a[r[mask_src]]

    # linear contribution of the destination (addition)
    k_dst = dst + r
    mask_dst = k_dst < 2 * n - 1
    new_conv[k_dst[mask_dst]] += 2.0 * delta * a[r[mask_dst]]

    # quadratic terms (very small, kept for correctness)
    new_conv[2 * src] += delta * delta
    new_conv[2 * dst] += delta * delta
    new_conv[src + dst] -= 2.0 * delta * delta

    # ----- sequence update -----
    new_seq = seq.copy()
    new_seq[src] -= delta
    new_seq[dst] += delta
    # Numerical safety
    new_seq[src] = max(new_seq[src], 0.0)

    return new_seq, new_conv


# ----------------------------------------------------------------------
# Main optimisation loop – repeatedly move mass from the most
# contributing index to a (near) zero‑contributing index.
# ----------------------------------------------------------------------
def _optimise(seq: np.ndarray, deadline: float) -> np.ndarray:
    n = seq.size
    # initial convolution (exact, not FFT‑based – cheap for modest n)
    conv = np.convolve(seq, seq)
    best_seq = seq.copy()
    best_max = conv.max()
    best_score = _fast_evaluation(best_seq)

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # --- locate the current maximum of the convolution ---
        k = int(conv.argmax())               # an index where max is attained
        # contribution of each coefficient to that entry
        i = np.arange(n)
        j = k - i
        mask = (j >= 0) & (j < n)
        w = np.zeros_like(seq)
        w[mask] = seq[j[mask]]

        # source = coefficient that contributes most to the max entry
        contrib = seq * w
        src = int(np.argmax(contrib))
        if seq[src] < 1e-12:           # nothing left to move
            break

        # sink = coefficient with (almost) zero contribution to the max entry
        # we deliberately pick a far tail index where w is minimal
        min_w = w.min()
        candidates = np.where(w == min_w)[0]
        if candidates.size == 0:
            dst = np.random.randint(n)
        else:
            dst = int(np.random.choice(candidates))

        if src == dst:
            # a degenerate choice – pick a random destination instead
            dst = np.random.randint(n)
            if src == dst:
                continue

        # move a fraction of the source mass; start with a relatively
        # aggressive step (10 %).  If the move is not accepted we try
        # smaller fractions.
        delta0 = 0.10 * seq[src]
        delta0 = min(delta0, seq[src])   # never make the source negative

        accepted = False
        for factor in (1.0, 0.5, 0.25, 0.125, 0.0625):
            delta = delta0 * factor
            if delta < 1e-12:

            new_seq, new_conv = _apply_move(seq, conv, src, dst, delta)
            new_max = new_conv.max()
            # accept only if the peak strictly decreases

                best_seq = new_seq.copy()
                best_max = new_max

                break
        if not accepted:
            # If no reduction was possible, try a tiny random jitter to
            # escape a possible flat region.
            if np.random.rand() < 0.01:
                idx = np.random.randint(n)
                jitter = (np.random.rand() - 0.5) * 1e-3 * seq[idx]
                tmp_seq = seq.copy()
                tmp_seq[idx] = max(0.0, tmp_seq[idx] + jitter)
                tmp_conv = np.convolve(tmp_seq, tmp_seq)
                seq, conv = tmp_seq, tmp_conv
                best_seq = tmp_seq.copy()
                best_max = conv.max()
                best_score = _fast_evaluation(best_seq)

    print(f"Finished optimisation after {iteration} iterations.")
    print(f"Final peak of convolution = {best_max:.8f}")
    print(f"Final evaluation score      = {best_score:.6f}")
    return best_seq


# ----------------------------------------------------------------------
# Public entry point required by the scoring harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Produce a list of non‑negative heights that aims at a low evaluation
    score.  The routine builds the classic arcsine (Catalan) seed,
    then performs a fast greedy optimisation that moves mass away from
    indices that dominate the current convolution peak.  The process
    runs until the time budget is exhausted and returns the best
    sequence discovered.
    """
    np.random.seed(seed)
    # leave a small safety margin for the harness
    deadline = time.time() + budget_s - 5

    # Length of the step function – a few thousand is a good trade‑off.
    n = 3000

    # 1.  Initial seed (arcsine / Catalan coefficients)
    seq = _catalan_sequence(n)

    # 2.  Optimise by moving mass from the most “dangerous’’ coefficients
    seq = _optimise(seq, deadline)

    # 3.  Return as plain Python list (security checks are performed by the
    #     evaluation harness).
    return seq.tolist()
