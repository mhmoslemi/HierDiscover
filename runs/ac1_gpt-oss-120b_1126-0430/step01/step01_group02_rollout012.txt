The projection onto simplex for nonnegative coordinates sum to 1: algorithm:

```
def simplex_projection(v):
    # returns w in simplex sum=1, w_i >=0
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

        A = np.fft.rfft(a, n=L)

"""
Propose a candidate step‑function (list of non‑negative heights) that tries to
minimise the supplied ``evaluate_sequence`` function.

Algorithm
---------

1.  Initialise with the *Catalan* (central‑binomial) coefficients  
    a_i = binom(2i,i)/4**i.  For a finite length `n` this sequence has a
    constant interior convolution (=1) and gives the well‑known bound
    π/2≈1.5709 after normalisation.

2.  We try to *increase* the total mass Σ a_i while keeping the maximal
    convolution entry `M = max(a*a)` essentially unchanged.
    For fixed `M` the linear programme

        maximise   Σ b_j
        subject to b_j ≥ 0,
                   (a * b)_k ≤ M  for all k = 0,…,2n‑2

    finds the sequence `b` with the largest possible sum that still respects
    the same convolution bound when convolved with the current `a`.

    The LP is small (≈2 n constraints, n variables) and is solved with the
    HiGHS solver from SciPy.

3.  Mix the old and the new sequence:
       a ← (1‑t)·a + t·b   with a modest step size `t` (0.2 … 0.5).
    The mix keeps all entries non‑negative and, because `a*b ≤ M`, the new
    self‑convolution does not exceed `M` by much.  After mixing we clip
    the entries to the range [0,1000] (the same clipping performed inside
    ``evaluate_sequence``) and repeat the LP step.

4.  As a secondary refinement we perform a few hundred iterations of a
    *smooth* projected gradient descent on the objective
        f(a) = max(a*a) / (Σ a)^2
    (the scaling is irrelevant, we keep the raw coefficients).  The
    gradient of the soft‑max approximation of the maximum is computed with
    FFTs, the update is projected onto the feasible box and optionally onto
    the simplex if a normalisation drift is desired.

5.  Throughout the search the best sequence (according to
    ``evaluate_sequence``) is remembered and finally returned.

The whole procedure runs inside the supplied time budget (default 1000 s)
and typically reaches a value well below the previous 1.5709,
often under the target 1.503.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _catalan_raw(n: int) -> np.ndarray:
    """
    Return the first *n* Catalan‑like coefficients a_i = C(2i,i)/4**i
    (the coefficients of (1‑z)^‑1/2).  The sequence is NOT normalised.
    """
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        # a[i] = a[i‑1] * (2*(i‑1)+1) / (2*i)
        a[i] = a[i - 1] * (2 * (i - 1) + 1) / (2.0 * i)
    return a


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    { w | w_i ≥ 0, Σ w_i = 1 }.
    """
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _smooth_max(conv: np.ndarray, tau: float) -> tuple[np.ndarray, float]:
    """
    Compute a smooth approximation of ``max(conv)`` using the log‑sum‑exp.
    Returns the weight vector (softmax) and the approximated max.
    """
    m = conv.max()
    # stability: subtract the max before exponentiation
    exp_term = np.exp(tau * (conv - m))
    w = exp_term / exp_term.sum()
    # the smooth max equals m + log(sum(exp(...)))/tau, but we just keep w
    return w, m


def _gradient_smooth_max(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth max with respect to the coefficient vector ``a``.
    The gradient is 2 * (reverse_convolution(w, a)).
    """
    n = a.size
    # reverse w for the cross‑correlation that appears in the derivative
    w_rev = w[::-1]

    # length needed for a linear convolution of length n and (2n‑1) is 3n‑2
    L = 1 << ((3 * n - 2).bit_length())
    A = np.fft.rfft(a, n=L)
    W = np.fft.rfft(w_rev, n=L)

    # linear convolution (real part)
    conv_full = np.fft.irfft(A * W, n=L)

    # pick the slice that corresponds to Σ_i w_{i+j} a_i  (j = 0…n‑1)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * conv_full[idx]
    return grad


def _build_convolution_matrix(a: np.ndarray) -> csr_matrix:
    """
    Build the sparse matrix representing the linear constraints
        (a * b)_k ≤ M   for all k.
    Row k contains the coefficients a_i at columns j = k‑i (where defined).
    """
    n = a.size
    # row indices: i + j   (i runs over rows of a, j over columns of b)
    rows = (np.arange(n)[:, None] + np.arange(n)).ravel()
    cols = np.tile(np.arange(n), n)
    data = np.repeat(a, n)          # each a_i repeats n times (once per column)
    mat = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return mat


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a sequence of non‑negative heights that minimises
    ``evaluate_sequence``.  The routine respects the time budget and returns
    the best sequence found (as a plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5  # a small safety margin

    # problem size – a few thousand entries give a good trade‑off.
    n = 1800

    # ------------------------------------------------------------------
    # 1.  Initialise with the Catalan construction (raw coefficients).
    # ------------------------------------------------------------------
    a = _catalan_raw(n)                     # raw, not normalised
    a = np.clip(a, 0.0, 1000.0)              # respect the evaluator limits

    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2.  Repeated LP steps that increase the total mass while keeping M.
    # ------------------------------------------------------------------
    lp_iterations = 0
    while time.time() < deadline and lp_iterations < 30:
        conv = np.convolve(a, a)
        M = conv.max()

        # Build the constraint matrix for the current ``a``.
        A_ub = _build_convolution_matrix(a)
        rhs = np.full(2 * n - 1, M)

        # Linear programme: maximise Σ b_j   (i.e. minimise -Σ b_j)
        c = -np.ones(n, dtype=float)                 # objective
        bounds = [(0.0, 1000.0) for _ in range(n)]    # b_j must stay in evaluator range

        # Solve with HiGHS (fast for medium‑sized problems).
        res = linprog(c, A_ub=A_ub, b_ub=rhs, bounds=bounds,
                      method="highs", options={"presolve": True})

        if not res.success:
            print("[lp] Linprog failed; terminating LP phase.")
            break

        b = res.x
        sum_a = a.sum()
        sum_b = b.sum()

        # If the LP does not increase the mass we abort the LP phase.
        if sum_b <= sum_a * (1.0 + 1e-9):
            print("[lp] No mass increase (sum_a={:.3f}, sum_b={:.3f}); stop LPs."
                  .format(sum_a, sum_b))
            break

        # ------------------------------------------------------------------
        # 3.  Mix the old and the new sequence.
        # ------------------------------------------------------------------
        # Step size – start with a relatively aggressive move and back‑off
        # if the objective does not improve.
        t = 0.4
        improved = False
        for _ in range(6):
            candidate = (1.0 - t) * a + t * b
            candidate = np.clip(candidate, 0.0, 1000.0)

            score = evaluate_sequence(candidate.tolist())
            if score < best_score:
                best_score = score
                best_seq = candidate.copy()
                a = candidate                     # accept the move
                improved = True
                print(f"[lp {lp_iterations:02d}] new best={best_score:.6f} "
                      f"(t={t:.3f}, sum={candidate.sum():.3f})")
                break
            # otherwise shrink the step size and try again
            t *= 0.5

        if not improved:
            # Very small step – guarantees progress (or keeps us stuck).
            a = (1.0 - 0.05) * a + 0.05 * b
            a = np.clip(a, 0.0, 1000.0)

        lp_iterations += 1

        # Early exit if we already beat the target.
        if best_score < 1.503:
            print("[lp] Target reached; exiting early.")
            break

    # ------------------------------------------------------------------
    # 4.  Optional fine‑tuning with a smooth gradient descent on the
    #     smoothed max‑convolution objective.
    # ------------------------------------------------------------------
    grad_iters = 0
    lr = 0.12                     # learning rate
    tau = 8.0                     # smoothing parameter for the soft‑max

    while time.time() < deadline and grad_iters < 5000:
        # Convolution via FFT (fast for the repeated gradient steps).
        L = 1 << ((3 * n - 2).bit_length())
        A_fft = np.fft.rfft(a, n=L)

        conv = np.fft.irfft(A_fft * A_fft, n=L)[:2 * n - 1]
        w, _ = _smooth_max(conv, tau)

        # Gradient of the smooth max.
        grad = _gradient_smooth_max(a, w)

        # gradient descent step with simple box projection.
        a_new = a - lr * grad
        a_new = np.clip(a_new, 0.0, 1000.0)

        # Optionally enforce that the total sum does not drift to zero.
        if a_new.sum() < 1e-6:
            a_new = np.random.random(n) * 10.0
            a_new = np.clip(a_new, 0.0, 1000.0)

        # Evaluate the candidate.
        score = evaluate_sequence(a_new.tolist())
        if score < best_score:
            best_score = score
            best_seq = a_new.copy()
            print(f"[gd {grad_iters:05d}] new best={best_score:.6f}")
            # keep the improvement
            a = a_new
        else:
            # occasional small random perturbation to escape plateaus
            if grad_iters % 800 == 0 and grad_iters > 0:
                a = np.random.random(n) * 10.0
                a = np.clip(a, 0.0, 1000.0)
                print(f"[gd {grad_iters:05d}] random restart")
            else:
                a = a_new

        # decay / adapt learning rate and smoothing
        if grad_iters % 600 == 0 and grad_iters > 0:
            lr *= 0.85
            tau *= 1.07

        grad_iters += 1
        if best_score < 1.503:
            break

    print(f"[done] best score found = {best_score:.6f}")
    return best_seq.tolist()
