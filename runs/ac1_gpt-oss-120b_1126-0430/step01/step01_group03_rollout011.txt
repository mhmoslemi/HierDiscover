def compute_score(seq):
    arr = np.array(seq, dtype=np.float64)
    n = len(arr)
    sum_a = arr.sum()
    if sum_a < 1e-12:
        return np.inf
    # compute convolution via FFT
    size = 1 << ((2*n-1).bit_length())  # next power of two >= 2*n-1
    # maybe choose size = 2*n
    size = 1<<((2*n-1).bit_length())

def compute_score(seq):
    arr = np.asarray(seq, dtype=np.float64)
    n = arr.shape[0]
    sum_a = arr.sum()
    if sum_a < 1e-12:
        return np.inf
    # Determine FFT size (next power of two)
    size = 1 << ((2 * n - 1).bit_length())
    # Compute real FFT
    fft_a = np.fft.rfft(arr, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[:2 * n - 1]
    max_b = np.max(conv)
    return 2.0 * n * max_b / (sum_a * sum_a)

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.1

    best_seq = None
    best_score = float('inf')
    # if height_sequence_1 exists

            best_seq = list(height_sequence_1)
            best_score = sc
            print(f"[init] from height_sequence_1 fast score {sc:.6f}")
        except Exception:
            pass

    # Set parameters
    n_candidates = [2000, 4000, 8000, 16000, 32000]
    betas = np.linspace(0.45, 0.78, 35)  # step roughly 0.01

    for n in n_candidates:
        if time.time() > deadline: break
        # precompute i+1 array
        i_arr = np.arange(1, n+1, dtype=np.float64)
        for beta in betas:
            if time.time() > deadline: break
            seq = i_arr ** (-beta)
            score = compute_score(seq)
            if score < best_score:
                best_score = score
                best_seq = seq.tolist()
                print(f"[scan] n={n}, beta={beta:.3f} -> {score:.6f}")

    # Local search/hill climbing
    if best_seq is not None:
        seq = np.array(best_seq, dtype=np.float64)
        n = len(seq)
        current_score = best_score
        # Pre compute array for random chooses
        while time.time() < deadline:
            i, j = rng.integers(0, n, size=2)
            if i == j: continue
            # choose fraction such that mass moved small
            frac = rng.random() * 0.08
            eps = seq[i] * frac
            if eps <= 0: continue
            new_seq = seq.copy()
            new_seq[i] -= eps
            new_seq[j] += eps
            if new_seq[i] < 0: continue
            new_score = compute_score(new_seq)
            if new_score < current_score:
                seq = new_seq
                current_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_seq = new_seq.tolist()
                    print(f"[hill] improved {new_score:.6f}")

    # final evaluation
    if best_seq is not None:
        try:
            final_val = evaluate_sequence(best_seq)
            print(f"[final] evaluate_sequence returns {final_val:.6f}")
        except Exception as e:
            print("[final error]", e)
    else:
        print("[final] no sequence found")

    return best_seq

"""
Propose a high‑quality step‑function sequence.

The evaluation function is scale‑invariant:
    evaluate_sequence(a) = 2·n·max_k (a*a)_k / (sum a)^2 .
The quantity to be minimised is the maximal self‑convolution value.
For a family of power‑law shapes a_i ∝ (i+δ)^{-β} the asymptotic bound
is 2·(1‑β)^2·B(1‑β,1‑β) ≈ 1.35 for β≈0.6.  Finite‑length
truncation hurts the bound, but the loss can be mitigated by
(i) using a larger support n,
(ii) adding a small shift δ>0 to smooth the head of the distribution,
(iii) a cheap stochastic hill‑climb on the best candidate.

The programme works in three stages:
  1.  Quick evaluation of many (n,β,δ) triples using an FFT‑based
      convolution (O(n log n)).
  2.  Selection of the best candidate found so far.
  3.  Light local optimisation (random mass transfers) on that candidate.
Only the final sequence is returned; the expensive `evaluate_sequence`
call is performed once at the end (or omitted – the outer grader will
call it anyway).

The implementation respects the 1000 s budget, prints progress and
contains no closures or lambda‑expressions.
"""

import time
import numpy as np

def _fast_score(seq: np.ndarray) -> float:
    """
    Compute the same value that `evaluate_sequence` would return,
    but using an FFT‑based convolution which is O(n log n) and much
    faster for large n.  The result is a float; NaNs or infinities
    are turned into ``np.inf``.
    """
    if seq.size == 0:
        return np.inf
    # ensure non‑negative (the generation code already guarantees this)
    seq = np.maximum(seq, 0.0)

    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return np.inf

    # length of linear convolution = 2*n‑1; choose next power‑of‑2 size
    size = 1 << ((2 * n - 1).bit_length())
    # real FFT, element‑wise square in frequency domain
    fft_a = np.fft.rfft(seq, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = np.max(conv)
    return 2.0 * n * max_b / (s * s)


def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that
    minimises ``evaluate_sequence``.  The search explores a family of
    power‑law sequences with an optional shift, evaluates them quickly
    with an FFT based routine, and refines the best candidate by a
    simple stochastic hill‑climb.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.1   # tiny safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0.  If a previous construction is present, use it as a start.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            seq0 = np.asarray(height_sequence_1, dtype=np.float64)
            sc0 = _fast_score(seq0)
            best_seq = list(seq0)
            best_score = sc0
            print(f"[init] seed sequence from height_sequence_1 → score {sc0:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Grid search over (n, β, δ)
    # ------------------------------------------------------------------
    # n values – keep them reasonable for the final evaluation step
    n_grid = [2000, 5000, 10000, 20000]
    # β range – theory suggests the optimum around 0.55–0.65
    betas = np.linspace(0.45, 0.78, 35)           # step ≈0.0095
    # small shifts to smooth the head of the distribution
    deltas = np.array([0.0, 0.5, 1.0, 2.0, 3.0, 5.0])

    for n in n_grid:
        if time.time() > deadline:
            break
        # pre‑compute the index vector (starting at 1)
        idx = np.arange(1, n + 1, dtype=np.float64)

        for beta in betas:
            if time.time() > deadline:
                break
            # base power‑law (δ = 0) – compute once and reuse for other deltas
            base = np.power(idx, -beta)

            for delta in deltas:
                if time.time() > deadline:
                    break
                # apply shift only if delta > 0 to avoid the i=0 singularity
                if delta == 0.0:
                    seq = base
                else:
                    seq = np.power(idx + delta, -beta)

                score = _fast_score(seq)
                if score < best_score:
                    best_score = score
                    best_seq = seq.tolist()
                    print(
                        f"[grid] n={n:5d}, β={beta:5.3f}, δ={delta:4.1f} → score {score:.6f}"
                    )

    # ------------------------------------------------------------------
    # 2.  Light stochastic hill‑climb on the best sequence.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        seq = np.array(best_seq, dtype=np.float64)
        n = seq.size
        cur_score = best_score

        # Hill‑climb parameters
        max_iters = 200000          # generous upper bound; loop exits early by time
        max_frac = 0.08             # up‑to 8 % of the source entry may be moved

        for _ in range(max_iters):
            if time.time() > deadline:
                break
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            # transfer a small random fraction of mass from i to j
            eps = seq[i] * max_frac * np.random.rand()
            if eps <= 0.0:
                continue
            new_seq = seq.copy()
            new_seq[i] -= eps
            new_seq[j] += eps
            if new_seq[i] < 0.0:   # safety guard
                continue

            new_score = _fast_score(new_seq)
            if new_score < cur_score:
                seq = new_seq
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_seq = new_seq.tolist()
                    print(f"[hill] improved to {new_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  (Optional) final verification with the official evaluator.
    # ------------------------------------------------------------------
    # The outer grading harness will call `evaluate_sequence` on the
    # returned sequence, so this call is only for logging.
    try:
        final_val = evaluate_sequence(best_seq)
        print(f"[final] evaluate_sequence reports {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence failed:", exc)

    print(f"[summary] best approximate score = {best_score:.6f}, length = {len(best_seq)}")
    return best_seq
