def geometric_sequence(n, c):
    r = 1.0 - c / n
    if r <= 0:
        r = 0.0
    # compute exponentials safely
    i = np.arange(n, dtype=float)
    # compute r^i using np.exp(i * np.log(r)) if r>0 else zeros (except i=0)
    if r == 0.0:
        powers = np.zeros(n)
        powers[0] = 1.0
    else:
        powers = np.exp(i * np.log(r))
    a = (c / n) * powers
    # Normalize sum to 1
    sum_a = a.sum()
    if sum_a == 0:
        a[0] = 1.0
        sum_a = 1.0
    a /= sum_a
    return a

def propose_candidate(seed=123, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5
    best_score = float('inf')
    best_seq = None

    # candidate lengths
    lengths = [800, 1500, 3000, 4500]
    for n in lengths:
        if time.time() > deadline:
            break

        # define function to compute score for given c
        def score_for_c(c):
            if c <= 0 or c >= n:
                return float('inf')
            a = geometric_sequence(n, c)
            # Clip to min 0, max 1000 - not needed.
            return evaluate_sequence(a.tolist()) # assumes evaluate_sequence defined

        # golden-section search c in (0.1, 10) but clamp to [1e-4, n*0.9] maybe
        # define lower bound etc.
        low = 0.1
        high = min(10, n-1)  # ensure c < n
        phi = (1 + np.sqrt(5)) / 2
        # initial points
        c1 = high - (high - low) / phi
        c2 = low + (high - low) / phi
        s1 = score_for_c(c1)
        s2 = score_for_c(c2)
        iter_count = 0
        while high - low > 1e-4 and time.time() < deadline and iter_count < 30:
            if s1 > s2:
                low = c1
                c1 = c2
                s1 = s2
                c2 = low + (high - low) / phi
                s2 = score_for_c(c2)
            else:
                high = c2
                c2 = c1
                s2 = s1
                c1 = high - (high - low) / phi
                s1 = score_for_c(c1)
            iter_count += 1

        # final best c roughly at (low+high)/2
        c_opt = (low + high) / 2.0
        a_opt = geometric_sequence(n, c_opt)
        score_opt = evaluate_sequence(a_opt.tolist())

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[geom n={n} c={c_opt:.4f}] new best score={best_score:.6f}")

        # also attempt slight local refinement via subgradient
        # set learning rate and iterations limited by time
        a = a_opt.copy()
        lr = 0.1
        iteration = 0
        while time.time() < deadline and iteration < 5000:
            # subgradient step like earlier, but maybe use momentum
            # Compute conv and k_max
            conv = np.convolve(a, a)
            k_max = int(np.argmax(conv))
            grad = np.zeros_like(a)
            i_start = max(0, k_max - (n - 1))
            i_end = min(n - 1, k_max)
            idx = np.arange(i_start, i_end + 1)
            grad[idx] = 2.0 * a[k_max - idx]

            # momentum approach
            if iteration == 0:
                velocity = np.zeros_like(a)
            velocity = 0.9 * velocity - lr * grad
            a_new = a + velocity
            # project onto simplex (ensuring nonnegative and sum=1)
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s > 0:
                a_new /= s
            else:
                # fallback uniform
                a_new = np.full(n, 1.0 / n)
            a = a_new

            cur_score = evaluate_sequence(a.tolist())
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[refine n={n} iter={iteration}] new best={best_score:.6f}")

            iteration += 1

    n = v.shape[0]
    u = np.sort(v)[::-1]

    cond = u - cssv / ind > 0

    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)

"""
Propose a step‑function (list of non‑negative heights) that minimises
``evaluate_sequence``.  

The implementation builds on the near‑optimal Catalan construction and
applies a deterministic exponentiated‑gradient descent on the simplex
combined with a small simulated‑annealing style perturbation.  The
gradient is the sub‑gradient of the current maximal convolution entry,
which drives the algorithm towards a flatter self‑convolution.
The update uses multiplicative (exponentiated) steps, which keep
positivity automatically; after each update the vector is projected back
onto the probability simplex (non‑negative, unit sum).  A tiny amount of
random noise is added periodically and accepted with a temperature that
slowly cools, helping the search to escape shallow plateaus.  The method
runs for the allotted time budget (default 1000 s) and returns the best
sequence seen.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex (non‑negative, sum = 1)
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x>=0, sum(x)=1}."""
    # Reference: Duchi et al., “Efficient Projections onto the l1‑Ball”
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    # find the largest rho satisfying the condition
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Catalan‑like seed (exactly the construction used in the reference code)
def _catalan_sequence(n: int) -> np.ndarray:
    """First n Catalan‑like coefficients a_i = C(2i,i)/4**i, normalised to sum 1."""
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    # recurrence a_{i+1} = a_i * (2*i+1) / (2*(i+1))
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * (i - 1) + 1) / (2.0 * i)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine runs for at most *budget_s* seconds and returns the best
    sequence discovered as a plain Python list.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : int, optional
        Time budget in seconds (default 1000).

    Returns
    -------
    list[float]
        The best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # problem size – a few thousand entries give a good trade‑off
    n = 3000

    # start from the Catalan seed (already very good)
    cur = _catalan_sequence(n)

    # keep the best solution seen so far
    best_seq = cur.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # parameters for the exponentiated‑gradient / SA loop
    lr = 0.04                     # learning‑rate for the multiplicative update
    temperature = 0.08           # initial temperature for random moves
    eps_noise = 0.015             # scale of the injected Gaussian noise
    iteration = 0

    # pre‑allocate a buffer for the FFT‑based convolution (faster than np.convolve)
    conv_len = 2 * n - 1
    fft_len = 1 << int(np.ceil(np.log2(conv_len)))   # next power of two

    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1) exact convolution of the current sequence (FFT for speed)
        fft_cur = np.fft.rfft(cur, n=fft_len)
        conv = np.fft.irfft(fft_cur * fft_cur, n=fft_len)[:conv_len]

        # 2) locate the index with maximal convolution value
        k_max = int(np.argmax(conv))

        # 3) sub‑gradient of the max‑entry w.r.t. the sequence
        #    (g_i = 2 * a_{k_max-i} for feasible i)
        grad = np.zeros_like(cur)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        grad[idx] = 2.0 * cur[k_max - idx]

        # 4) exponentiated‑gradient step (multiplicative update)
        #    ensures positivity without explicit clipping.
        cur = cur * np.exp(-lr * grad)

        # 5) project back onto the simplex (unit L1‑norm)
        cur = _project_simplex(cur)

        # 6) evaluate the new candidate
        cur_score = evaluate_sequence(cur.tolist())

        # keep the best ever seen
        if cur_score < best_score:
            best_score = cur_score
            best_seq = cur.copy()
            print(f"[{iteration:06d}] new best = {best_score:.6f}")

        # --------------------------------------------------------------
        # 7) occasional stochastic perturbation (simulated annealing)
        if iteration % 12 == 0:
            noise = np.random.randn(n) * eps_noise
            proposal = cur + noise
            proposal = np.maximum(proposal, 0.0)
            proposal = _project_simplex(proposal)

            prop_score = evaluate_sequence(proposal.tolist())
            # accept if better or with Metropolis probability
            if prop_score < cur_score or np.random.rand() < np.exp(-(prop_score - cur_score) / temperature):
                cur = proposal
                cur_score = prop_score
                if prop_score < best_score:
                    best_score = prop_score
                    best_seq = proposal.copy()
                    print(f"[{iteration:06d}] new best (SA) = {best_score:.6f}")

        # --------------------------------------------------------------
        # 8) simple cooling schedule for both lr and temperature
        if iteration % 500 == 0 and iteration > 0:
            lr *= 0.97
            temperature *= 0.96
            eps_noise *= 0.97

        iteration += 1

    print(f"[done] best score found = {best_score:.6f}")
    return best_seq.tolist()
