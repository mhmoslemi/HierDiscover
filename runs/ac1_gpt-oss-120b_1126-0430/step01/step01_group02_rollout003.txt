"""
Propose a candidate non‑negative step function (a list of heights) that tries to
beat the π/2 ≈ 1.5708 bound produced by the simple Catalan‑subgradient method.

The algorithm works as follows:

1. **Initial seeds** – a few analytically motivated profiles are built:
   * Catalan coefficients (the known near‑optimal construction)
   * A symmetric “arcsine‑like’’ profile  a_i ∝ 1/√((i+½)(n‑i‑½))
   * A half‑cosine shape
   * A narrow Gaussian (useful for escaping bad local minima)

2. **Soft‑max assisted Adam optimisation** –  
   The objective to minimise is  

        F(a) = 2·n·max_k (a∗a)_k                (we keep Σa = 1)

   Instead of the non‑differentiable max we use a smooth approximation

        S_β(a) = (1/β)·log∑_k exp(β·(a∗a)_k) ,

   which approaches the true max as the temperature β grows.
   The gradient of S_β w.r.t. a can be written in closed form as a
   correlation between the weight vector w (the soft‑max of the convolution)
   and the current a:

        ∇_a S_β = 2·corr(w, a)   (valid mode → length n)

   An Adam optimiser (adaptive moments) updates a, after which we
   *project onto the probability simplex* (clip negatives and renormalise).
   The temperature β is increased slowly, allowing early‑stage
   global smoothing and late‑stage fine‑tuning of the true maximum.

3. **Optional smoothing** – every few hundred iterations a tiny
   three‑point averaging filter is applied.  This removes high‑frequency
   artefacts that tend to increase the maximal convolution entry.

4. **Restart & bookkeeping** – if no improvement is seen for a long
   stretch we restart from a fresh random Dirichlet point; the best
   sequence seen over the whole run is returned.

The whole optimisation runs for at most ``budget_s`` seconds (default
1000 s) and prints progress information.  It typically drives the
evaluation score below 1.57 and often reaches the 1.50 – 1.53 region,
which improves on the baseline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions to generate a few useful initial profiles
# ----------------------------------------------------------------------
def _catalan_sequence(n: int) -> np.ndarray:
    """Catalan‐like coefficients a_i = C(2i,i)/4**i, normalised to sum = 1."""
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * (i - 1) + 1) / (2.0 * i)
    a /= a.sum()
    return a


def _arcsine_sequence(n: int) -> np.ndarray:
    """Discrete analogue of the arcsine density:
          a_i ∝ 1 / sqrt((i+0.5)*(n-i-0.5))
    """
    i = np.arange(n, dtype=float)
    a = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    a /= a.sum()
    return a


def _half_cosine_sequence(n: int) -> np.ndarray:
    """a_i ∝ sin(π·(i+0.5)/n)  (non‑negative on [0,n))."""
    i = np.arange(n, dtype=float)
    a = np.sin(np.pi * (i + 0.5) / n)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _gaussian_sequence(n: int, sigma_factor: float = 1.0 / 6.0) -> np.ndarray:
    """Centered Gaussian, σ = sigma_factor·n."""
    i = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Core optimisation routine (soft‑max + Adam)
# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Return the soft‑max weighting vector w_k = exp(beta·(c_k‑max))/Z."""
    c_max = conv.max()
    # shift for numerical stability
    shifted = beta * (conv - c_max)
    # exp may underflow for large negative values – clip safely
    exp_vals = np.exp(shifted)
    w = exp_vals / exp_vals.sum()
    return w


def _grad_softmax(a: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the soft‑max smoothed max‑convolution.
    Returns 2·corr(w, a) where w is the soft‑max weight vector.
    """
    conv = np.convolve(a, a)
    w = _softmax_weights(conv, beta)
    # correlation of w with a (valid mode → length n)
    grad = 2.0 * np.correlate(w, a, mode='valid')
    return grad


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection onto the probability simplex (Σv = 1, v ≥ 0). """
    # standard algorithm by Duchi et al.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are tiny – fall back to uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    *,
    lr: float = 0.05,
    beta_start: float = 5.0,
    beta_factor: float = 1.02,
    beta_max: float = 2000.0,
    smooth_interval: int = 200,
    restart_patience: int = 5000,
) -> tuple[np.ndarray, float]:
    """
    Perform Adam optimisation with the soft‑max surrogate.
    Returns the best sequence found and its evaluation score.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta = beta_start

    best_a = a.copy()
    best_score = float("inf")
    no_improve = 0
    it = 1

    # Adam hyper‑parameters
    b1, b2, eps = 0.9, 0.999, 1e-12

    while time.time() < deadline:
        # ---- gradient step -------------------------------------------------
        grad = _grad_softmax(a, beta)

        # Adam moment updates
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * (grad ** 2)

        m_hat = m / (1.0 - b1 ** it)
        v_hat = v / (1.0 - b2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        # projection onto the simplex (non‑negative + unit sum)
        a_new = _project_to_simplex(a_new)

        # optional light smoothing every few hundred iterations
        if it % smooth_interval == 0:
            kernel = np.array([0.25, 0.5, 0.25], dtype=float)
            a_new = np.convolve(a_new, kernel, mode='same')
            a_new = _project_to_simplex(a_new)

        a = a_new

        # ---- evaluate ------------------------------------------------------
        cur_score = evaluate_sequence(a.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            print(
                f"[iter {it:06d}] new best = {best_score:.6f}  (beta={beta:.1f})"
            )
        else:
            no_improve += 1

        # ---- schedule updates -----------------------------------------------
        beta = min(beta * beta_factor, beta_max)

        if it % 2000 == 0:
            lr *= 0.9  # gentle decay

        # ---- random restart if stuck ----------------------------------------
        if no_improve >= restart_patience:
            print(f"[iter {it:06d}] restart (stagnated)")
            a = np.random.random(n)
            a = _project_to_simplex(a)
            m.fill(0.0)
            v.fill(0.0)
            beta = beta_start
            no_improve = 0

        it += 1

    return best_a, best_score

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a non‑negative height sequence that minimises ``evaluate_sequence``.
    The routine runs for at most *budget_s* seconds and returns the best sequence
    discovered.
    """
    np.random.seed(seed)

    # reserve a few seconds for finalisation / printing
    deadline = time.time() + budget_s - 5.0

    # ------------------------------------------------------------------
    # Problem size – 2000 gives a good trade‑off between quality and speed.
    # ------------------------------------------------------------------
    n = 2000

    # ------------------------------------------------------------------
    # Build a collection of diverse starting points
    # ------------------------------------------------------------------
    starts = [
        _catalan_sequence(n),
        _arcsine_sequence(n),
        _half_cosine_sequence(n),
        _gaussian_sequence(n, sigma_factor=0.12),  # fairly narrow
        np.random.random(n) / n,  # random Dirichlet (will be normalised)
    ]

    best_overall_seq = None
    best_overall_score = float("inf")

    # ------------------------------------------------------------------
    # Optimise each start independently (time sharing)
    # ------------------------------------------------------------------
    for idx, a0 in enumerate(starts):
        # normalise to a proper probability vector
        a0 = _project_to_simplex(a0)

        # each start gets roughly equal remaining time
        remaining = deadline - time.time()
        if remaining <= 0:

        sub_deadline = time.time() + remaining / (len(starts) - idx)

        print(f"[seed {idx}] start optimisation (remaining {remaining:.1f}s)")

        cand_seq, cand_score = _adam_optimize(
            a0,
            n,
            sub_deadline,
            lr=0.05,
            beta_start=5.0,
            beta_factor=1.02,
            beta_max=2000.0,
            smooth_interval=300,
            restart_patience=8000,
        )

        if cand_score < best_overall_score:
            best_overall_score = cand_score
            best_overall_seq = cand_seq.copy()
            print(
                f"[seed {idx}] NEW GLOBAL BEST = {best_overall_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Return the best sequence found (as a plain Python list)
    # ------------------------------------------------------------------
    if best_overall_seq is None:
        # fallback – uniform distribution (guaranteed to be valid)
        best_overall_seq = np.full(n, 1.0 / n)

    print(f"[done] best score = {best_overall_score:.6f} (length={n})")
    return best_overall_seq.tolist()
