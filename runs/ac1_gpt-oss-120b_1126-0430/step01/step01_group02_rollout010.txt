    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        # recurrence a_{i} = a_{i-1} * (2*i-1) / (2*i)
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    a /= a.sum()
    return a


def _linear_convolution_fft(x: np.ndarray, y: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution via FFT (real→real). Returns the full length result."""
    fx = np.fft.rfft(x, n=L)
    fy = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(fx * fy, n=L)
    return conv


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """Return a*a (full linear convolution) using FFT."""
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * a.size - 1]
    return conv


def _softmax_weights(conv: np.ndarray, temperature: float) -> np.ndarray:
    """
    Return a probability vector w_i = softmax(conv_i / temperature).
    The maximum entry gets weight ≈1 when temperature→0.
    """
    if temperature <= 0:
        # exact max – fall back to a hard indicator (for safety)
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w
    # subtract the max for numerical stability
    max_c = np.max(conv)
    exps = np.exp((conv - max_c) / temperature)
    sum_exps = np.sum(exps)
    return exps / sum_exps


def _gradient_from_weights(a: np.ndarray, w: np.ndarray, L: int) -> np.ndarray:
    """
    Gradient of  ∑_k w_k * (a*a)_k  w.r.t a.
    Using the identity  ∂/∂a_i (a*a)_k = 2 a_{k-i},
    the gradient is g_i = 2 * Σ_k w_k * a_{k-i},
    which is a cross‑correlation between w and a.
    Implemented with FFT for speed.
    """
    n = a.size
    # reverse w (required for cross‑correlation)
    w_rev = w[::-1]
    # linear convolution w_rev * a (length = (2n-1)+n-1 = 3n-2)
    conv_full = _linear_convolution_fft(w_rev, a, L)[: 3 * n - 2]
    # slice the part that corresponds to i = 0 .. n-1
    start = n - 1
    end = start + n
    g = 2.0 * conv_full[start:end][::-1]  # reverse to get g_i order
    return g


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine runs for at most *budget_s* seconds
    and returns the best sequence encountered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5  # safety margin

    # problem size – a few thousand entries give a good trade‑off
    n = 4000

    # pre‑compute FFT length (power of two ≥ 3n)
    L = _next_pow2(3 * n)

    # initialise with the Catalan construction (near optimal)
    a = _catalan_sequence(n)

    # Adam hyper‑parameters
    lr = 0.04
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # soft‑max temperature schedule
    temp = 1.0            # start fairly smooth
    min_temp = 5e-3       # final temperature (approaches hard max)
    temp_decay = 0.9995   # exponential decay per iteration

    # bookkeeping for the best solution seen so far
    best_seq = a.copy()
    best_score = float("inf")
    # evaluate the initial candidate (expensive but cheap for a single call)
    try:
        best_score = evaluate_sequence(best_seq.tolist())
    except Exception:  # pragma: no cover – safety net
        best_score = float("inf")
    print(f"[init] length={n}, initial score={best_score:.6f}")

    iteration = 0
    no_improve = 0
    eval_every = 25          # call the official evaluator only every few steps
    restart_threshold = 2000  # iterations without improvement → random restart

    while time.time() < deadline:
        # ------------------------------------------------------------------
        # 1. compute self‑convolution (full) and soft‑max weights
        # ------------------------------------------------------------------
        conv = _conv_self(a, L)               # shape (2n-1,)
        w = _softmax_weights(conv, temp)

        # ------------------------------------------------------------------
        # 2. gradient of the soft‑max surrogate (Adam update)
        # ------------------------------------------------------------------
        g = _gradient_from_weights(a, w, L)

        # Adam moment updates
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        # parameter update
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ------------------------------------------------------------------
        # 3. project onto the simplex (non‑negative, sum‑to‑1)
        # ------------------------------------------------------------------
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 0:
            a_new /= s
        else:
            # degenerate case – fall back to a small random point on the simplex
            a_new = np.random.rand(n)
            a_new /= a_new.sum()

        a = a_new

        # ------------------------------------------------------------------
        # 4. temperature and learning‑rate annealing
        # ------------------------------------------------------------------
        temp = max(min_temp, temp * temp_decay)
        if iteration % 500 == 0:
            lr *= 0.95

        # ------------------------------------------------------------------
        # 5. Periodic evaluation with the official black‑box function
        # ------------------------------------------------------------------
        if iteration % eval_every == 0:
            cur_score = evaluate_sequence(a.tolist())
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{iteration:06d}] new best = {best_score:.6f}")
                no_improve = 0
            else:
                no_improve += 1

            # occasional random restart if we appear stuck
            if no_improve >= restart_threshold:
                a = _catalan_sequence(n)               # back to a good basin
                m.fill(0.0)
                v.fill(0.0)
                lr = 0.04
                temp = 1.0
                no_improve = 0
                print(f"[{iteration:06d}] restart (Catalan re‑init)")

    # ----------------------------------------------------------------------
    # Final report
    # ----------------------------------------------------------------------
    print(f"[done] best score found = {best_score:.6f}")
    return best_seq.tolist()
