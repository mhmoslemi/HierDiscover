def _subgradient_max_conv(a, conv):
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)
    g = np.zeros_like(a)
    for k in ks:
        # indexes i such that 0 <= i < n and 0 <= k-i < n
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end+1)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    g /= len(ks)
    return g

def _smooth_gradient(a, conv, alpha, L):
    # conv shape: 2n-1
    # compute softmax weights w for conv entries
    scaled = alpha * conv
    max_s = np.max(scaled)
    exp_scaled = np.exp(scaled - max_s)
    w = exp_scaled / exp_scaled.sum()
    # compute correlation of w with a_rev
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, L)
    fft_a_rev = np.fft.rfft(a_rev, L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, L)[:len(w)+len(a)-1]  # length 3n-2
    grad = 2.0 * corr_rev[len(a)-1:len(a)-1+len(a)]
    # shape n
    return grad

"""
Propose a non‑negative step‑function sequence that minimizes
    O(a) = 2·n·max(conv(a,a))      (∑a_i = 1)

Algorithm
---------
1. Initialise from the previously found best sequence   (height_sequence_1)
   – normalise it to the probability simplex.
2. **Phase 1 – smooth‑gradient descent**
   * Use a soft‑max (log‑sum‑exp) approximation of the max to obtain a
     smooth objective  f_α(a) = (2·n/α)·log⎡∑_k e^{α·conv_k}⎤.
   * Compute its gradient by correlating the weight vector
     w_k ∝ e^{α·conv_k} with a (FFT‑based O(N log N) convolution).
   * Perform a projected gradient step with a step size that scales
     with the maximum absolute gradient component.  α is increased
     linearly from a small value (≈5) to a large one (≈2000) so that the
     objective gradually focuses on the true maximum.
   * Keep the best sequence seen.

3. **Phase 2 – local refinement (hill‑climbing + sub‑gradient)**
   * Repeatedly propose small “mass‑transfer’’ moves: move a random
     fraction of probability mass from index i to a different index j.
   * Accept the move if it lowers the exact objective; otherwise accept
     with a tiny Metropolis probability.
   * Every few iterations apply a sub‑gradient step for the current
     maximal convolution entry (the same direction used in the original
     code) to push the peak down.
   * After each accepted move/project‑step recompute the convolution
     (FFT) and update the best solution.

The whole search respects the 1000 s budget, prints occasional progress
messages and returns the best sequence found as a plain Python list.
"""

import time
import math
import numpy as np

# -------------------------------------------------------------
# Helper: projection onto the probability simplex {x >= 0, sum(x) = 1}
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        # all non‑positive – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# -------------------------------------------------------------
# FFT‑based full self‑convolution (length 2·n‑1)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return a*a (full convolution) using an FFT of length L."""
    # rfft works with real input and returns the positive‑frequency part.
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# -------------------------------------------------------------
# Exact objective for a simplex‑normalised vector
def _objective_simplex(a: np.ndarray, n: int, L: int) -> float:
    """2·n·max(conv(a,a)) (a is assumed to sum to 1)."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# -------------------------------------------------------------
# Sub‑gradient of the maximal convolution entry (as used in the reference code)
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Return the sub‑gradient of max(conv) w.r.t. a."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)  # typically a single index
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    g /= len(ks)  # average when several maximisers exist
    return g


# -------------------------------------------------------------
# Gradient of the soft‑max (log‑sum‑exp) approximation of the max
def _smooth_gradient(a: np.ndarray, conv: np.ndarray, alpha: float, L: int) -> np.ndarray:
    """
    Compute ∇_a ( (2·n/α)·log∑_k e^{α·conv_k} ) .
    The formula reduces to 2·corr(w, a) where w_k ∝ e^{α·conv_k}.
    """
    # soft‑max weights w_k
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)               # subtract for stability
    w = exp_scaled / exp_scaled.sum()

    # correlation w ⋆ a  (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]                                        # reversed a
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)        # length 3·n‑2
    n = a.size
    # slice the central n entries (indices n‑1 … 2·n‑2)
    grad = 2.0 * corr_rev[n - 1 : n - 1 + n]
    return grad


# -------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence that minimises the evaluation
    function.  Returns the sequence as a list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # -----------------------------------------------------------------
    # Determine problem size
    if 'height_sequence_1' in globals():
        base = np.array(height_sequence_1, dtype=float)
        n = base.size
        a0 = base / base.sum()                      # normalise to the simplex
        print(f"[init] using existing sequence, n={n}")
    else:
        n = 4000
        a0 = np.full(n, 1.0 / n)                     # uniform start
        print(f"[init] no prior sequence – using uniform, n={n}")

    # FFT length – must accommodate both convolution (2n‑1) and correlation (3n‑2)
    L = 1 << ((3 * n - 2).bit_length())               # next power of two

    # -----------------------------------------------------------------
    # Phase 1 – smooth gradient descent on a log‑sum‑exp surrogate
    best_seq = a0.copy()
    best_score = _objective_simplex(best_seq, n, L)
    print(f"[phase‑1] start O={best_score:.6f}")

    # hyper‑parameters
    alpha_start = 5.0
    alpha_end = 2000.0
    max_iter_phase1 = 200_000
    iter1 = 0

    # allocate roughly half of the total budget to phase 1
    phase1_deadline = start_time + budget_s * 0.55

    while time.time() < phase1_deadline and iter1 < max_iter_phase1:
        # compute convolution and objective
        conv = _conv_fft(best_seq, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = best_seq.copy()
            print(f"[{time.time():.1f}] Phase‑1 new best O={best_score:.6f} (iter={iter1})")

        # schedule for α (linear annealing)
        frac = min(1.0, iter1 / max_iter_phase1)
        alpha = alpha_start + (alpha_end - alpha_start) * frac

        # smooth gradient
        grad = _smooth_gradient(best_seq, conv, alpha, L)

        # adaptive step size (keep max|step| ≤ 0.5)
        max_abs = np.max(np.abs(grad)) + 1e-12
        step = 0.5 / max_abs

            # (best_score will be updated at the start of next loop)
        else:

                cand2 = _project_to_simplex(cand2)

    print(f"[phase‑2] entering local refinement, best O={best_score:.6f}")

    no_improve_limit = 5_000

    subgrad_interval = 7

    while time.time() < deadline:

        if np.random.rand() < 0.5:
            a = best_seq.copy()
            a += np.random.randn(n) * 0.003
            a = np.clip(a, 0, None)
            a = _project_to_simplex(a)
        else:
            a = np.random.rand(n)
            a = _project_to_simplex(a)

        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        iter_local = 0
        no_improve = 0
        T = temperature0

        while time.time() < deadline and iter_local < max_iter_per_restart:
            # ----- random mass transfer -------------------------------------------------
            i, j = np.random.randint(n, size=2)
            if i != j and a[i] > 0:

                    conv = conv_new
                    cur_obj = new_obj
                    no_improve = 0
                    if cur_obj < best_score:
                        best_score = cur_obj
                        best_seq = a.copy()
                        print(f"[{time.time():.1f}] New global best O={best_score:.6f} (restart={max_restarts}, iter={iter_local})")
                else:
                    no_improve += 1
            else:
                no_improve += 1

            # ----- sub‑gradient step on the current max entry ----------------------------
            if iter_local % subgrad_interval == 0:

                a = _project_to_simplex(a)
                conv = _conv_fft(a, L)
                cur_obj = 2.0 * n * conv.max()
                if cur_obj < best_score:
                    best_score = cur_obj
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] Sub‑grad new best O={best_score:.6f}")

            # temperature decay
            T = temperature0 * (1.0 - iter_local / max_iter_per_restart)

            # restart if stuck too long
            if no_improve >= no_improve_limit:
                break

            iter_local += 1

        max_restarts += 1
        if time.time() >= deadline:
            break

    # -----------------------------------------------------------------

        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:
        print("[final] evaluate_sequence failed:", e)

    return best_seq.tolist()
