```
    # Determine n: if height_sequence_1 defined, use its length else maybe 12000

    start = time.time()

    def max_conv(seq):
        n = seq.shape[0]
        m = 1 << ((2*n - 1).bit_length())
        fft_seq = np.fft.rfft(seq, n=m)
        conv = np.fft.irfft(fft_seq * fft_seq, n=m)[:2*n-1]
        return float(conv.max())
    # generate candidates
    # Parameter sweep loops for allocated time.
    while time.time() < start + budget_s - 10:

        family = rng.choice(['power', 'shifted_power', 'exp', 'linear', 'mix'])
        if family == 'power':

            raw = (np.arange(1, n+1, dtype=float) + c) ** (-p)
        elif family == 'shifted_power':
            p = rng.uniform(0.05, 1.2)
            d = rng.uniform(0.0, n*0.5)
            raw = (np.arange(n, dtype=float) + d) ** (-p)

            raw = np.exp(-lam * np.arange(n, dtype=float))

            slope = rng.uniform(0.0, a0/(n-1))
            raw = np.maximum(a0 - slope * np.arange(n, dtype=float), 0.0)
        elif family == 'mix':
            # two-phase: flat plateau length L, then exponential
            L_frac = rng.uniform(0.0, 0.5)
```

assistantfinal**Solution Explanation**

The score is  

```
score = 2·n·max(conv(a,a)) / (∑a)²
```

`conv` is the linear convolution.  
For every admissible sequence `a`

```
∑ a = √(2·n)          (the normalisation used by the grader)
```

so the denominator is always `2·n` and the score is simply the
maximum element of the auto‑convolution:

```
score = max(conv(a,a))
```

The optimisation problem is therefore

```
minimise   maxk   ∑i a[i]·a[k-i]
subject to a[i] ≥ 0 ,  ∑ a[i] = √(2·n)
```

The maximum is attained for a “bad” shift `k` where large values of `a`
overlap with large values of the reversed sequence.
A monotone decreasing shape makes the two sequences *oppositely* sorted,
which reduces the overlap (rearrangement inequality).  
The search therefore concentrates on decreasing families (power laws,
exponentials, linear ramps, piece‑wise constructions) and on a
gradient‑guided improvement step which always moves mass from a position
with a large contribution to the current worst shift towards a position
with a small contribution.

The algorithm works in three stages

1. **Parameter sweep** – generate many families of decreasing sequences,
   scale them to the required `L¹`‑norm, evaluate the maximum convolution
   *fast* by FFT and keep the best candidate.

2. **Gradient‑guided refinement** – for the current best sequence  
   * compute the full convolution, locate the shift `k*` that gives the
     maximum,
   * compute the sub‑gradient `g[i] = a[i] + a[k*‑i]` (only defined on the
     overlapping part),
   * move a small fraction of mass from the index with the largest
     gradient to the index with the smallest gradient,
   * keep the move if the score improves (otherwise reject or accept
     with a tiny simulated‑annealing probability).

   The step size decays with time, guaranteeing convergence.

3. **Final projection** – after the time budget the sequence is projected
   onto the simplex `{a≥0, ∑a = √(2·n)}` to satisfy the exact constraints
   and returned.

All heavy work (convolution, max search) uses FFT (`O(n log n)`) so
hundreds of thousands of candidates can be examined well inside the
1000 s budget.  The method is completely self‑contained, uses only
`numpy` and `scipy` (the latter only for a very small helper if it is
available) and respects the required function signature.

```python
python
"""
propose_candidate
=================

A fast search for a non‑negative step function that minimises

    max(conv(a, a))

under the normalisation  sum(a) = sqrt(2·n).

The algorithm consists of

1. a wide parameter sweep over several decreasing families
   (power‑law, shifted power‑law, exponential, linear, piecewise);
   evaluation is performed with an FFT based convolution (O(n log n)).

2. a gradient‑guided simplex refinement.
   At each iteration the shift `k*` that attains the current maximum
   is identified, the sub‑gradient `g[i] = a[i] + a[k*‑i]` is built,
   and a small amount of mass is transferred from the index with
   the largest gradient to the index with the smallest gradient.
   The move is accepted if the score improves (or with a tiny
   simulated‑annealing probability).

3. a final projection onto the simplex `{a ≥ 0, sum(a)=sqrt(2·n)}`.

The whole search runs for the supplied time budget (default 1000 s)
and returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all are top‑level, no closures or lambdas
# ----------------------------------------------------------------------


def _project_to_simplex(x, target_sum):
    """Project a vector onto the simplex {y≥0, sum(y)=target_sum}.
    Implementation of the O(n) algorithm by Wang & Carreira‑Perpiñán.
    """
    n = x.shape[0]
    if target_sum <= 0:
        return np.zeros_like(x)

    # sort descending
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(x - theta, 0.0)
    return w


def _fast_full_conv(seq):
    """Return the full linear convolution of `seq` with itself using FFT."""
    n = seq.shape[0]
    # length of the convolution
    length = 2 * n - 1
    # next power of two for efficient FFT
    m = 1 << ((length - 1).bit_length())
    fft_seq = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fft_seq * fft_seq, n=m)[:length]
    return conv


def _fast_max_conv(seq):
    """Maximum element of the auto‑convolution (FFT based)."""
    return float(_fast_full_conv(seq).max())


def _evaluate(seq):
    """Thin wrapper around the supplied `evaluate_sequence`."""
    # the grader injects `evaluate_sequence` into the global namespace
    return evaluate_sequence(seq)


# ----------------------------------------------------------------------
# Parameter families – each returns a non‑negative raw vector
# ----------------------------------------------------------------------


def _family_power(n, rng):
    """a_i = (i+1)^(-p)  with p∈[0.05,1.5]"""
    p = rng.uniform(0.05, 1.5)
    raw = (np.arange(1, n + 1, dtype=float)) ** (-p)
    return raw, ('power', p)


def _family_shifted_power(n, rng):
    """a_i = (i + d)^(-p)   d>0"""
    p = rng.uniform(0.05, 1.5)
    d = rng.uniform(0.0, n * 0.5) + 1e-6
    raw = (np.arange(n, dtype=float) + d) ** (-p)
    return raw, ('shifted_power', p, d)


def _family_exponential(n, rng):
    """a_i = exp(-λ·i)   λ∈[1e‑4, 3e‑2]"""
    lam = rng.uniform(1e-4, 3e-2)
    raw = np.exp(-lam * np.arange(n, dtype=float))
    return raw, ('exp', lam)


def _family_linear(n, rng):
    """a_i = max(0, a0 - slope·i)"""
    a0 = rng.uniform(0.02, 5.0)
    slope = rng.uniform(0.0, a0 / (n - 1))
    raw = np.maximum(a0 - slope * np.arange(n, dtype=float), 0.0)
    return raw, ('linear', a0, slope)


def _family_piecewise(n, rng):
    """plateau of constant value followed by an exponential decay"""
    plateau_frac = rng.uniform(0.0, 0.5)          # fraction of the length
    plateau_len = int(plateau_frac * n)
    lam = rng.uniform(1e-4, 3e-2)
    plateau_val = rng.uniform(0.02, 5.0)
    if plateau_len == 0:
        raw = np.exp(-lam * np.arange(n, dtype=float))
    else:
        tail = np.exp(-lam * np.arange(n - plateau_len, dtype=float))
        raw = np.concatenate((np.full(plateau_len, plateau_val, dtype=float), tail))
    return raw, ('piecewise', plateau_len, plateau_val, lam)


_FAMILY_FUNCS = [
    _family_power,
    _family_shifted_power,
    _family_exponential,
    _family_linear,
    _family_piecewise,
]

# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) yields a low
    evaluation score.  The function respects the time budget `budget_s`.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – use an existing sequence if present, otherwise a
    # moderately large size that still allows many evaluations.
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
        base_seq = np.array(height_sequence_1, dtype=float)
    else:
        n = 12000                     # a safe default
        base_seq = None

    target_sum = math.sqrt(2.0 * n)   # normalisation required by the grader

    # --------------------------------------------------------------
    # Initialise with a uniform start and (if available) the previous
    # best sequence.
    # --------------------------------------------------------------
    uniform = np.full(n, target_sum / n, dtype=float)
    best_seq = uniform.copy()
    best_score = _fast_max_conv(best_seq)

    if base_seq is not None:
        # re‑scale the provided sequence to the current length / norm
        scaled = _project_to_simplex(base_seq, target_sum)
        sc_score = _fast_max_conv(scaled)
        if sc_score < best_score:
            best_seq = scaled.copy()
            best_score = sc_score
        print(f"[init] starting from previous construction, score={best_score:.6f}")

    else:
        print(f"[init] uniform start, n={n}, score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – broad parameter sweep
    # --------------------------------------------------------------
    deadline = time.time() + budget_s - 30.0   # keep some seconds for refinement
    sweep_iter = 0

    while time.time() < deadline:
        sweep_iter += 1
        # pick a random family and generate a raw vector
        family = rng.choice(_FAMILY_FUNCS)
        raw, fam_info = family(n, rng)

        # scale to the required L1 norm
        raw_sum = raw.sum()
        if raw_sum == 0:
            continue
        raw = raw * (target_sum / raw_sum)

        # evaluate quickly via FFT
        score = _fast_max_conv(raw)

        if score < best_score:
            best_score = score
            best_seq = raw.copy()
            print(f"[sweep {sweep_iter:05d}] new best {best_score:.6f} via {fam_info}")

        # occasional tiny random perturbation of the current best
        if sweep_iter % 500 == 0 and rng.random() < 0.25:
            tmp = best_seq.copy()
            i, j = rng.choice(n, size=2, replace=False)
            delta = tmp[i] * rng.uniform(0.0, 0.02)
            tmp[i] -= delta
            tmp[j] += delta
            # re‑project to keep the exact norm (numerical drift may appear)
            tmp = _project_to_simplex(tmp, target_sum)
            pert_score = _fast_max_conv(tmp)
            if pert_score < best_score:
                best_score = pert_score
                best_seq = tmp.copy()
                print(f"[perturb {sweep_iter:05d}] improved to {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – gradient‑guided simplex refinement
    # --------------------------------------------------------------
    a = best_seq.copy()
    it = 0
    refine_deadline = time.time() + budget_s - 5.0
    while time.time() < refine_deadline:
        it += 1
        # full convolution (FFT) and index of the worst shift
        conv = _fast_full_conv(a)
        k_max = int(np.argmax(conv))

        # build the sub‑gradient for this shift
        grad = np.zeros_like(a)
        start = max(0, k_max - (n - 1))
        end = min(n - 1, k_max)
        idx = np.arange(start, end + 1, dtype=int)
        jdx = k_max - idx
        grad[idx] = a[idx] + a[jdx]

        # choose donor (largest gradient) and receiver (smallest gradient)
        i = int(np.argmax(grad))          # donor
        j = int(np.argmin(grad))          # receiver (often zero gradient)

        if a[i] <= 0:
            continue

        # transfer a small fraction of the donor's mass
        frac = 0.02 * (1.0 - (time.time() - (deadline - budget_s + 30.0)) / budget_s)
        delta = a[i] * max(frac, 5e-4)     # never zero
        a[i] -= delta
        a[j] += delta

        # evaluate the new maximum
        new_score = _fast_max_conv(a)

        # accept if we improved the global best, otherwise revert (tiny SA)
        if new_score < best_score:
            best_score = new_score
            best_seq = a.copy()
            print(f"[refine {it:06d}] improved to {best_score:.6f}")
        else:
            # simulated‑annealing style acceptance (temperature decays)
            T = max(1e-4, 0.1 * (refine_deadline - time.time()) / (budget_s - 5.0))
            if rng.random() < math.exp(-(new_score - best_score) / T):
                # keep the move
                pass
            else:
                # revert
                a[i] += delta
                a[j] -= delta

    # --------------------------------------------------------------
    # Final projection onto the admissible simplex and final check
    # --------------------------------------------------------------
    best_seq = _project_to_simplex(best_seq, target_sum)
    final_score = _evaluate(best_seq.tolist())
    print(f"[final] best score reported by grader: {final_score:.6f}")

    # ensure the returned list is a plain Python list (the grader expects that)
    return best_seq.tolist()
```
