"""
A fast hybrid optimizer for the step‑function evaluation problem.

The algorithm combines three ideas:

1. **Arcsine (Catalan) seed** – the coefficients a_i = C_i = binom(2i,i)/4**i,
   normalized so that Σa = √(2·n).  This already gives a score ≈ π/2 ≈ 1.57.

2. **Aggressive mass relocation** – repeatedly moves (sometimes all) the mass from
   the index that contributes most to the current maximal convolution entry to an
   index that hardly contributes at all.  The step size is adaptive (full, ½,
   ⅓, ⅕ of the source mass) and only accepted when the score improves.

3. **Simulated‑annealing fine‑tuning** – stochastic moves that transfer a random
   fraction of mass from a high‑contribution index to a low‑contribution index.
   Worse moves are occasionally accepted with a temperature schedule, allowing
   the search to escape shallow local minima.

All operations use a fast FFT‑based evaluation (identical to `evaluate_sequence`
but without the safety checks) and respect the 1000 s time budget.  The best
sequence discovered during the run is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast evaluation (same as ``evaluate_sequence`` but without checks).
# ----------------------------------------------------------------------
def _fast_evaluation(seq: np.ndarray) -> float:
    """Return the metric 2·n·max(conv)/sum(seq)^2 for a non‑negative sequence."""
    n = seq.size
    if n == 0:
        return np.inf
    # FFT size = next power of two ≥ 2·n‑1
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Catalan (arcsine) seed – a_i = C_i.
# ----------------------------------------------------------------------
def _catalan_sequence(length: int) -> np.ndarray:
    """First ``length`` Catalan coefficients (as a float array)."""
    seq = np.empty(length, dtype=float)
    a = 1.0
    seq[0] = a
    for i in range(1, length):
        a *= (2.0 * i - 1) / (2.0 * i)
        seq[i] = a
    return seq


# ----------------------------------------------------------------------
# Aggressive refinement – move (large) chunks of mass from heavy to light spots.
# ----------------------------------------------------------------------
def _refine_aggressive(seq: np.ndarray, deadline: float) -> np.ndarray:
    """Perform large‑step mass relocation until ``deadline`` is reached."""
    n = seq.size
    best_seq = seq.copy()
    best_score = _fast_evaluation(best_seq)

    # pre‑compute FFT length (does not change)
    fft_len = 1 << ((2 * n - 1).bit_length())

    while time.time() < deadline:
        # current convolution and its maximal index
        fa = np.fft.rfft(seq, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        k = int(conv.argmax())

        # w[i] = a_{k‑i} when defined, else 0
        i_arr = np.arange(n)
        j_arr = k - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        w = np.zeros_like(seq)
        w[mask] = seq[j_arr[mask]]

        # contribution of each position to the maximal entry
        contrib = seq * w
        src = int(np.argmax(contrib))

        # choose a sink with the smallest w (ideally zero)
        # if many zeros exist, pick the first one that is not src
        zero_idx = np.where(w == 0)[0]
        if zero_idx.size > 0:
            sink = int(zero_idx[0])
            if sink == src:
                sink = int(zero_idx[1] if zero_idx.shape[0] > 1 else np.argmin(w))
        else:
            # fallback to absolute minimum
            sink = int(np.argmin(w))
            if sink == src:
                # pick second‑minimum
                sorted_idx = np.argsort(w)
                sink = int(sorted_idx[1])

        if seq[src] < 1e-12:
            break  # nothing left to move

        # try a series of decreasing fractions of the source mass
        for frac in (1.0, 0.5, 0.33, 0.2, 0.1):
            move = seq[src] * frac
            if move < 1e-12:
                continue
            new_seq = seq.copy()
            new_seq[src] -= move
            new_seq[sink] += move

            score = _fast_evaluation(new_seq)
            if score < best_score - 1e-12:
                seq = new_seq
                best_seq = new_seq.copy()
                best_score = score
                break
        else:
            # random jitter to shake the system when no improvement is possible
            if np.random.rand() < 0.01:
                jitter_idx = np.random.randint(n)
                jitter = np.random.randn() * 0.001 * max(1.0, seq[jitter_idx])
                new_seq = seq.copy()
                new_seq[jitter_idx] = max(0.0, new_seq[jitter_idx] + jitter)
                # re‑normalise the sum (preserves scale invariance)
                new_seq *= seq.sum() / new_seq.sum()
                score = _fast_evaluation(new_seq)
                if score < best_score:
                    seq = new_seq
                    best_seq = new_seq.copy()
                    best_score = score

    return best_seq


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuning.
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray, deadline: float) -> np.ndarray:
    """Stochastic mass transfers with a temperature schedule."""
    n = seq.size
    best_seq = seq.copy()
    best_score = _fast_evaluation(best_seq)

    cur_seq = seq.copy()
    cur_score = best_score

    # FFT length (constant)
    fft_len = 1 << ((2 * n - 1).bit_length())

    # temperature schedule
    T = max(0.01, best_score * 0.02)   # start a little above typical score changes
    decay = 0.9995

    # small constant to avoid division by zero
    eps = 1e-12

    while time.time() < deadline:
        # convolution – only needed to locate the current maximum
        fa = np.fft.rfft(cur_seq, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        k = int(conv.argmax())

        # weighting w[i] = a_{k‑i}
        i_arr = np.arange(n)
        j_arr = k - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        w = np.zeros_like(cur_seq)
        w[mask] = cur_seq[j_arr[mask]]

        # source probabilities proportional to contribution to the peak
        contrib = cur_seq * w
        contrib_sum = contrib.sum()
        if contrib_sum > 0:
            src_prob = contrib / contrib_sum
        else:
            src_prob = cur_seq / cur_seq.sum()

        src = np.random.choice(n, p=src_prob)
        if cur_seq[src] < eps:
            continue

        # sink probabilities – favour tiny w (little influence on the peak)
        inv_w = np.where(w > 0, 1.0 / w, 1.0)  # larger when w is small
        inv_w_sum = inv_w.sum()
        sink_prob = inv_w / (inv_w_sum + eps)

        sink = np.random.choice(n, p=sink_prob)
        if sink == src:
            continue

        # random fraction of source mass to move (up to 50 %)
        frac = np.random.rand() * 0.5
        delta = cur_seq[src] * frac

        new_seq = cur_seq.copy()
        new_seq[src] -= delta
        new_seq[sink] += delta

        # evaluate
        new_score = _fast_evaluation(new_seq)

        if new_score < cur_score:
            cur_seq = new_seq
            cur_score = new_score
            if new_score < best_score:
                best_seq = new_seq.copy()
                best_score = new_score
        else:
            # Metropolis acceptance
            prob = np.exp(-(new_score - cur_score) / (T + eps))
            if np.random.rand() < prob:
                cur_seq = new_seq
                cur_score = new_score

        # temperature decay
        T *= decay

    return best_seq


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Produce a list of non‑negative heights with a low evaluation score.

    The function works within ``budget_s`` seconds (default 1000).  It tries a
    few different problem sizes, runs an aggressive deterministic refinement,
    and finishes with a stochastic annealing phase.  The best sequence found
    during the whole run is returned.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5    # small safety margin

    best_overall_seq = None
    best_overall_score = np.inf

    # try several lengths – longer sequences give a richer search space
    lengths = [2500, 3000, 3500, 4000]

    # allocate roughly equal time to each length
    for n in lengths:
        if time.time() >= deadline:
            break

        # ----- 1. initialise from Catalan (arcsine) distribution -----
        seq = _catalan_sequence(n)
        # normalise sum to √(2·n) (scale‑invariant for the metric)
        target_sum = np.sqrt(2.0 * n)
        seq = seq / seq.sum() * target_sum

        # ----- 2. aggressive deterministic refinement (half of the allocated time) -----
        half_deadline = time.time() + (deadline - time.time()) * 0.5
        seq_refined = _refine_aggressive(seq, half_deadline)

        # ----- 3. stochastic annealing on the refined sequence -----
        anneal_deadline = deadline  # use whatever time is left for this length
        seq_final = _anneal(seq_refined, anneal_deadline)

        # ----- 4. keep the best across all lengths -----
        final_score = _fast_evaluation(seq_final)
        print(f"Length {n:4d} – final score {final_score:.9f}")

        if final_score < best_overall_score:
            best_overall_seq = seq_final.copy()
            best_overall_score = final_score
            print(f"*** New overall best (length {n}) ***")

    # If time remains, polish the current best with a final annealing run
    if best_overall_seq is not None and time.time() < deadline:
        print("Final polishing phase...")
        polished = _anneal(best_overall_seq, deadline)
        polished_score = _fast_evaluation(polished)
        if polished_score < best_overall_score:
            best_overall_seq = polished
            best_overall_score = polished_score
            print(f"Polished score improved to {polished_score:.9f}")

    # Fallback – should never happen, but keep a safe default
    if best_overall_seq is None:
        best_overall_seq = _catalan_sequence(3000)
        best_overall_seq = best_overall_seq / best_overall_seq.sum() * np.sqrt(2.0 * 3000)
        best_overall_score = _fast_evaluation(best_overall_seq)

    print(f"Best score achieved: {best_overall_score:.9f}")
    return best_overall_seq.tolist()
