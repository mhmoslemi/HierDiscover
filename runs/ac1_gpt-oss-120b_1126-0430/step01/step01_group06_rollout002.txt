def fast_score(a):
    a = np.array(a, dtype=float)
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    n = len(a)
    # zero-pad to L >= 2*n
    size = 1 << ((2*n-1).bit_length())
    # Might use next power of two for speed.
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)[:2*n-1]
    max_b = np.max(conv)
    return 2.0 * n * max_b / (sum_a * sum_a)

import time

# ----------------------------------------------------------------------
# Helper: fast evaluation (identical to evaluate_sequence but using FFT)
# ----------------------------------------------------------------------
def _fast_score(a: np.ndarray) -> float:
    """Return the objective value for a non‑negative vector a (sum may be ≠1)."""
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    n = a.size
    # next power of two for efficient FFT
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)[: 2 * n - 1]
    max_b = conv.max()
    return 2.0 * n * max_b / (sum_a * sum_a)


# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex (optional safety net)
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a real vector onto {x ≥ 0, Σx = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

    # 1. Grid search over (α, shift, β) power‑law + exponential family
    # ------------------------------------------------------------------
    alphas = np.linspace(0.35, 0.70, 36)          # 0.01 steps
    shifts = np.logspace(np.log10(0.02), np.log10(5.0), 12)  # 12 points
    betas = np.array([0.0, 0.001, 0.003, 0.006, 0.010, 0.020])

    grid_best = None
    for α in alphas:
        if time.time() - start > budget_s - 30:
            break
        for s in shifts:
            for β in betas:
                raw = (idx + s) ** (-α) * np.exp(-β * idx)
                total = raw.sum()
                if total == 0.0:
                    continue
                vec = raw / total
                sc = _fast_score(vec)
                if sc < best_score:
                    best_score = sc
                    best_seq = vec.tolist()
                    print(f"[grid] α={α:.4f}  s={s:.3f}  β={β:.5f} → {sc:.6f}")

    print(f"[grid] best score after grid search = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Random sampling on the simplex (heavy‑tailed draws)
    # ------------------------------------------------------------------
    random_iters = 2500
    for it in range(random_iters):
        if time.time() - start > budget_s - 30:
            break
        # power‑transform to obtain heavy‑tailed vectors
        p = np.random.rand(n) ** 2.5   # exponent >1 makes many tiny entries
        vec = p / p.sum()
        sc = _fast_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[rand] it={it} → {sc:.6f}")

    print(f"[rand] best score after random sampling = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Guided local refinement (sub‑gradient hill climbing)
    # ------------------------------------------------------------------
    cur = np.array(best_seq, dtype=float)
    cur_score = best_score
    max_iter = 60000
    iter_cnt = 0

    while time.time() - start < budget_s - 5 and iter_cnt < max_iter:
        iter_cnt += 1

        # convolution via FFT
        size = 1 << ((2 * n - 1).bit_length())
        fft_cur = np.fft.rfft(cur, n=size)
        conv = np.fft.irfft(fft_cur * fft_cur, n=size)[: 2 * n - 1]
        max_b = conv.max()

        # indices where the maximum is attained (allow tiny tolerance)
        peak_mask = conv >= max_b - 1e-14
        peak_idxs = np.nonzero(peak_mask)[0]

        # compute sub‑gradient: g_i = 2 * Σ_{k in peaks} a_{k-i}
        g = np.zeros_like(cur)
        for k in peak_idxs:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_range = np.arange(i_start, i_end + 1)
            j_range = k - i_range
            g[i_range] += 2.0 * cur[j_range]

        # pick source (largest gradient) and target (smallest gradient)
        src = np.argmax(g)
        tgt = np.argmin(g)
        if src == tgt:
            src, tgt = np.random.choice(n, 2, replace=False)

        # step size slowly decays from 5 % to 0.1 % of the source amount
        max_frac = 0.05
        min_frac = 0.001
        frac = max_frac - (max_frac - min_frac) * (iter_cnt / max_iter)
        step = cur[src] * frac
        if step <= 0:
            continue

        new = cur.copy()
        new[src] -= step
        new[tgt] += step
        # numerical drift correction (optional – cheap)
        # keep the exact sum to avoid rounding accumulation
        new_sum = new.sum()
        if abs(new_sum - 1.0) > 1e-12:
            new /= new_sum

        new_score = _fast_score(new)
        if new_score < cur_score:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = cur.tolist()
                print(f"[refine] it={iter_cnt}  score={new_score:.6f}")

    # ------------------------------------------------------------------
    # Final verification with the official evaluator (optional safety)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq)
        print(f"[final] verified score = {final_score:.6f}")
    except Exception:
        print("[final] verification failed (using internal score)")

    return best_seq
