```
def _gradient(a, beta, N):
    n = a.shape[0]
    # FFT of a
    A = np.fft.rfft(a, N)
    # convolution a*a
    c = np.fft.irfft(A * A, N)[:2*n-1]
    # softmax weights:
    # for stability subtract max before exponentiation
    max_val = np.max(beta * c)
    exp_beta_c = np.exp(beta * c - max_val)
    w = exp_beta_c / np.sum(exp_beta_c)
    # FFT of w
    W = np.fft.rfft(w, N)
    # FFT of reversed a
    A_rev = np.fft.rfft(a[::-1], N)
    # convolution w * a_rev (full)
    conv_wr = np.fft.irfft(W * A_rev, N)[:(2*n-1)+n-1]  # length 3n-2
    # extract g
    start = n - 1
    g = 2.0 * conv_wr[start:start + n]
```

assistantfinal**Solution Overview**

The optimizer works directly on the probability simplex (`∑a_i = 1, a_i ≥ 0`).  
The true objective is  

\[
O(a)=2\,n\;\max_k (a*a)_k\qquad\bigl((a*a)_k=\sum_i a_i a_{k-i}\bigr)
\]

which is *scale‑invariant*.  
Instead of the non‑smooth max we minimise a smooth surrogate

\[
L_\beta(a)=\frac{1}{\beta}\log\Bigl(\sum_k e^{\beta\,(a*a)_k}\Bigr),
\]

whose gradient can be written in closed form.  
For a given temperature β the algorithm performs projected (onto the simplex) gradient
descent.  β is increased geometrically, turning the smooth surrogate into the true
max‑function.  The whole procedure is wrapped in a time‑budget loop; occasional
random perturbations avoid stagnation.  The best sequence found is finally checked
with the provided `evaluate_sequence`.

All helper functions are top‑level, there are no closures, and the code respects the
1000 s budget.

```python
"""
Projected gradient descent on a soft‑max surrogate of the max‑convolution.
The temperature β is increased gradually, turning the smooth objective into the
true max‑convolution.  The iterate is always projected onto the probability simplex.
Random perturbations are added from time to time to escape local minima.
"""

import time
import math
import numpy as np

# -------------------------------------------------------------
# Simple projection onto the probability simplex {x >= 0, sum(x) = 1}
# (Duchi et al., 2008)
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        # all non‑positive → uniform point
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# -------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small utility: next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# -------------------------------------------------------------
def _conv_fft(a: np.ndarray, N: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2*len(a)-1)."""
    n = a.shape[0]
    A = np.fft.rfft(a, N)
    conv_full = np.fft.irfft(A * A, N)[: 2 * n - 1]
    return conv_full


# -------------------------------------------------------------
def _objective(a: np.ndarray, N: int) -> float:
    """Exact objective 2·n·max(conv(a,a)).  Uses FFT for speed."""
    n = a.shape[0]
    conv = _conv_fft(a, N)
    return 2.0 * n * conv.max()


# -------------------------------------------------------------
def _softmax_weights(c: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max over the convolution values (stable version)."""
    # shift by max(beta*c) for numerical stability
    z = beta * c
    shift = np.max(z)
    ex = np.exp(z - shift)
    return ex / np.sum(ex)


# -------------------------------------------------------------
def _gradient(a: np.ndarray, beta: float, N: int) -> (np.ndarray, np.ndarray):
    """
    Gradient of the smoothed objective L_β at a (and also the raw convolution).
    Returns (grad, conv) where conv = a*a (full length 2n‑1).
    """
    n = a.shape[0]

    # ----- convolution a*a -------------------------------------------------
    A = np.fft.rfft(a, N)
    conv = np.fft.irfft(A * A, N)[: 2 * n - 1]

    # ----- soft‑max weights -------------------------------------------------
    w = _softmax_weights(conv, beta)          # length 2n‑1, sum(w)=1

    # ----- gradient = 2 * (w ⋆ a)   (cross‑correlation) --------------------
    # cross‑correlation w ⋆ a equals convolution of w with reversed a.
    a_rev = a[::-1]
    W = np.fft.rfft(w, N)
    A_rev = np.fft.rfft(a_rev, N)
    # linear convolution length (2n‑1)+(n)-1 = 3n‑2
    conv_wr = np.fft.irfft(W * A_rev, N)[: 3 * n - 2]
    start = n - 1
    grad = 2.0 * conv_wr[start : start + n]   # length n
    return grad, conv


# -------------------------------------------------------------
def propose_candidate(seed: int = 42,
                      budget_s: int = 1000,
                      **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    evaluate_sequence().  The algorithm is a projected gradient descent on a
    soft‑max surrogate of the max‑convolution, with a slowly increasing
    temperature β.  Random perturbations are added from time to time.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ---------- choose the length ------------------------------------------------
    if 'height_sequence_1' in globals():
        base_len = len(height_sequence_1)
    else:
        base_len = 4600
    n = max(3000, base_len)            # never too small; larger n helps the bound

    # ---------- FFT size (must accommodate 3n) ----------------------------------
    N = _next_pow2(3 * n)               # safe size for all convolutions

    # ---------- initial point ----------------------------------------------------
    if 'height_sequence_1' in globals():
        a0 = np.array(height_sequence_1, dtype=float)
        if a0.size > n:
            a0 = a0[:n]
        elif a0.size < n:
            a0 = np.concatenate([a0, np.full(n - a0.size, 1e-12)])
        a = a0 / a0.sum()
    else:
        a = np.random.rand(n)
        a = _project_to_simplex(a)

    # ---------- bookkeeping -------------------------------------------------------
    best_a = a.copy()
    best_score = _objective(best_a, N)
    print(f"[init] n={n}, O={best_score:.6f}")

    # ---------- hyper‑parameters -------------------------------------------------
    beta = 0.5
    beta_max = 128.0
    beta_mul = 1.3                # geometric increase
    iter_no_improve = 0
    max_no_improve = 4000        # after this we inject noise
    iteration = 0
    total_iters = 0

    # ---------- main optimisation loop -------------------------------------------
    while time.time() < deadline:
        iteration += 1
        total_iters += 1

        # ----- gradient step ----------------------------------------------------
        grad, conv = _gradient(a, beta, N)

        # step size inversely proportional to the gradient norm
        grad_norm = np.linalg.norm(grad) + 1e-12
        eta = 0.20 / grad_norm          # a modest learning rate

        a_new = a - eta * grad
        a_new = np.clip(a_new, 0.0, None)
        a_new = _project_to_simplex(a_new)

        # ----- evaluate the true objective --------------------------------------
        cur_score = 2.0 * n * conv.max()
        new_conv = _conv_fft(a_new, N)
        new_score = 2.0 * n * new_conv.max()

        # ----- acceptance --------------------------------------------------------
        # deterministic improvement; a tiny probability for uphill moves
        if new_score < cur_score or np.random.rand() < 1e-4:
            a = a_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] New best O={best_score:.6f} (iter={total_iters}, β={beta:.2f})")
                iter_no_improve = 0
            else:
                iter_no_improve += 1
        else:
            iter_no_improve += 1

        # ----- temperature schedule ------------------------------------------------
        if iteration % 3000 == 0 and beta < beta_max:
            beta = min(beta * beta_mul, beta_max)
            print(f"[{time.time():.1f}] Increase β → {beta:.2f}")

        # ----- occasional random perturbation ------------------------------------
        if iter_no_improve >= max_no_improve:
            # add a tiny Gaussian kick and re‑project
            a = a + np.random.randn(n) * 0.001
            a = np.clip(a, 0.0, None)
            a = _project_to_simplex(a)
            iter_no_improve = 0
            print(f"[{time.time():.1f}] Injected noise to escape plateau")

        # ----- safety check -------------------------------------------------------
        if time.time() >= deadline:
            break

    # ---------- final verification ------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence failed:", exc)

    return best_a.tolist()
```
