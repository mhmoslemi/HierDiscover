"""
propose_candidate
-----------------
Search for a non‑negative step‑function (a probability vector) that
minimises

    2 * n * max(conv(a, a))

where `conv` is the linear self‑convolution.  The objective is scale‑invariant,
so we restrict to vectors whose entries sum to one (the simplex).

Algorithm
~~~~~~~~~
1. **Fast power‑law initialisation** – a dense grid over the two‑parameter
   family a_i ∝ (i+shift)^{-α} is evaluated using an FFT‑based convolution.
   The best candidate from this grid supplies a good starting point.

2. **Exponentiated‑gradient descent on the simplex** – at each iteration the
   convolution is computed, the index of the maximal entry is identified, and
   the gradient of that entry (g_i = 2·a_{k_max‑i}) is obtained.  A
   multiplicative update a_i ← a_i·exp(−η·g_i) followed by normalisation
   implements a mirror‑descent step that naturally stays in the simplex.

   The step size η is adaptively reduced by back‑tracking when a move does not
   improve the objective, and is slowly increased when progress is made.

3. **Occasional random mass transfers** – with a small probability a random
   move (transfer a fraction of mass from a high‑contributing index to a
   random index) is attempted.  This helps escape shallow local minima.

The loop runs until the allotted time budget (default 950 s) is exhausted,
printing occasional progress messages.  The best sequence found is returned
as a plain Python list of floats.  All heavy work uses NumPy and FFTs, so the
method can evaluate tens of thousands of candidate vectors well within the
time limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (all defined at top level – no closures or lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    if x <= 0:
        return 1
    return 1 << (x - 1).bit_length()


def _conv_full(a: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution of a (real vector) using a zero‑padded FFT.
    Returns the full length 2·len(a)−1 array.
    """
    n = a.shape[0]
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    conv = np.fft.irfft(fa * fa, N)[: 2 * n - 1]
    return conv.real


def _objective(a: np.ndarray) -> float:
    """
    Compute the evaluation metric for a probability vector a.
    The vector is assumed to be non‑negative and to sum to 1.
    """
    n = a.shape[0]
    return 2.0 * n * _conv_full(a).max()


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, sum(x) = 1}.  Implementation follows
    Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w
