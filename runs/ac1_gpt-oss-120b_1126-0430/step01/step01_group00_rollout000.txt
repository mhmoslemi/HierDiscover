        (a_rev is a reversed) – a cheap convolution.
      – take a step a ← a – η·∇f and project back onto the simplex.
* Continuation on α: start with a small α (smooth landscape) and increase
  it gradually up to a large value (≈200) so the optimiser focuses on the
  true max.
* Multiple independent restarts (including noisy perturbations of the best
  found sequence) are performed until the time budget (≈1000 s) expires.
* Finally a brief hill‑climbing / simulated‑annealing polishing phase
  refines the best sequence.

The routine returns a list of floats that can be fed directly to
`evaluate_sequence`.  All helper functions are defined at top level, no
closures or I/O are used.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex (Duchi et al., 2008)
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x ≥ 0, sum(x) = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        # All entries non‑positive → return uniform point
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _logsumexp(x: np.ndarray) -> float:
    """Stable log‑sum‑exp."""
    m = np.max(x)
    return m + np.log(np.sum(np.exp(x - m)))


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective: 2·n·max(conv(a,a)) with sum(a)=1."""
    conv = np.convolve(a, a)
    return 2.0 * a.size * np.max(conv)


# ----------------------------------------------------------------------
def _convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a∗a (full length 2·n‑1)."""
    return np.convolve(a, a)


# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the smoothed max f_α(a) = (1/α)·log Σ_i exp(α·conv_i).

    Returns ∇f_α(a) of shape (n,).  Uses the identity:
        ∇f = 2·(w ∗ a_rev)_{slice}
    where w_i = exp(α·conv_i) / Σ_j exp(α·conv_j) and a_rev = a[::-1].
    """
    # Soft‑max weights (sum to 1)
    w = np.exp(alpha * conv - alpha * (_logsumexp(alpha * conv)))
    # Convolution of w with reversed a
    a_rev = a[::-1]
    full = np.convolve(w, a_rev)     # length = (2n-1) + n - 1 = 3n-2
    n = a.shape[0]
    grad = 2.0 * full[n - 1 : n - 1 + n]   # extract the n relevant entries
    return grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  Returns the best sequence found (list of floats).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Choose a problem size.
    # Prefer the length of a known construction if it exists.
    if 'height_sequence_1' in globals():
        base_len = len(height_sequence_1)
    else:
        base_len = 4000

    # A length around 5000 is a good trade‑off between quality and speed.
    n = max(2000, base_len, 5000)

    # --------------------------------------------------------------
    # Initial best: either the provided construction or the uniform one.
    if 'height_sequence_1' in globals():
        best_seq = np.array(height_sequence_1, dtype=float)
        if best_seq.size != n:
            # pad / truncate to the chosen length
            if best_seq.size < n:
                pad = np.zeros(n - best_seq.size)
                best_seq = np.concatenate([best_seq, pad])
            else:
                best_seq = best_seq[:n]
        best_seq = _project_to_simplex(best_seq)   # ensure feasibility
    else:
        best_seq = np.full(n, 1.0 / n)

    best_score = _objective(best_seq)
    print(f"[init] start length n={n}, O={best_score:.6f}")

    # --------------------------------------------------------------
    # Main optimisation loop (multiple restarts)
    restart_id = 0
    while time.time() < deadline:
        restart_id += 1

        # ---- initialise a new starting point ---------------------------------
        if np.random.rand() < 0.25 and best_seq is not None:
            # perturb the current best
            a = best_seq.copy()
            a += np.random.randn(n) * 0.003
            a = np.clip(a, 0.0, None)
            a = _project_to_simplex(a)
        else:
            # random point on the simplex
            a = np.random.rand(n)
            a = _project_to_simplex(a)

        # Parameters for this restart
        max_iters = 4000
        # continuation schedule for α (smooth → sharp)
        alpha_start = 5.0
        alpha_end = 200.0
        # step‑size schedule (simple linear decay)
        step_start = 0.25
        step_end = 0.005

        # Keep local best for possible early termination
        local_best_score = _objective(a)

        for it in range(max_iters):
            # time check
            if time.time() >= deadline:
                break

            # current schedules
            frac = it / max_iters
            alpha = alpha_start + frac * (alpha_end - alpha_start)
            step = step_start + (step_end - step_start) * frac

            # ----- compute convolution and smoothed gradient --------------------
            conv = _convolution(a)
            # exact objective for book‑keeping
            cur_score = 2.0 * n * np.max(conv)

            # occasional logging
            if it % 500 == 0:
                print(f"[{time.time():.1f}] restart={restart_id}, it={it}, "
                      f"O={cur_score:.6f}, α={alpha:.1f}")

            # Gradient of smoothed max
            grad = _gradient_softmax(a, conv, alpha)

            # Gradient step + projection
            a_new = a - step * grad
            a_new = np.clip(a_new, 0.0, None)
            a_new = _project_to_simplex(a_new)

            # Evaluate the exact objective after the step
            conv_new = _convolution(a_new)
            new_score = 2.0 * n * np.max(conv_new)

            # Accept if we improved the exact objective, otherwise keep old.
            # Occasionally (with small prob) accept a worse move to escape traps.
            accept = False
            if new_score < cur_score:
                accept = True
            elif np.random.rand() < 0.001:
                accept = True

            if accept:
                a = a_new
                cur_score = new_score
                # Update global best
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] NEW GLOBAL BEST O={best_score:.6f} "
                          f"(restart={restart_id}, it={it})")
                # Update local best
                if cur_score < local_best_score:
                    local_best_score = cur_score

            # ---- small random perturbations every few iterations (exploration) ---
            if it % 250 == 0 and np.random.rand() < 0.3:
                # move a tiny random mass
                i, j = np.random.randint(n, size=2)
                if i != j and a[i] > 1e-12:
                    delta = a[i] * 0.02 * np.random.rand()
                    a[i] -= delta
                    a[j] += delta
                    a = _project_to_simplex(a)

        # End of restart – go to the next one unless we ran out of time.
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Final polishing phase: short simulated‑annealing hill‑climb.
    # Uses the same mass‑transfer move as the original baseline but starts
    # from the best sequence found so far.
    a = best_seq.copy()
    cur_score = _objective(a)
    T0 = 0.02
    T = T0
    patience = 0
    max_patience = 3000

    while time.time() < deadline and patience < max_patience:
        i, j = np.random.randint(n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        delta = a[i] * np.random.rand() * 0.08
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = _project_to_simplex(a_new)
        new_score = _objective(a_new)

        # Metropolis acceptance
        if new_score < cur_score or np.random.rand() < math.exp((cur_score - new_score) / max(T, 1e-12)):
            a = a_new
            cur_score = new_score
            patience = 0
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] POLISH NEW BEST O={best_score:.6f}")
        else:
            patience += 1

        # simple temperature decay
        T = T0 * (1.0 - patience / max_patience)

    # ------------------------------------------------------------------
    # Final validation with the provided evaluate_sequence (optional)
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_val:.6f}")
    except Exception as e:
        print("[final] evaluate_sequence call failed:", e)

    # Return as plain Python list
    return best_seq.tolist()
