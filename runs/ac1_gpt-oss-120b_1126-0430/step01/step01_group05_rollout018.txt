def max_convolution(seq):
    n = len(seq)
    x = np.fft.rfft(seq, n=next_pow_two(2*n-1))
    conv = np.fft.irfft(x*x, n=next_pow_two(2*n-1))
    # conv length is m; we need first 2n-1 values.
    conv = conv[:2*n-1]
    return conv.max()

#!/usr/bin/env python
"""
Propose a step‑function (list of non‑negative heights) with a low evaluation
score.  

The algorithm has two stages:

1. **Coarse power‑law search** – a family a_i = (i+1)^(-p) is scanned for a
   few values of the length `n` and the exponent `p`.  The fast FFT based
   evaluation is used to keep the search cheap.

2. **Iterative linear‑programming refinement** – starting from the best
   power‑law found, we repeatedly solve a linear program

        maximise   Σ g_i
        subject to  (f * g)[k] ≤ M   for all k,
                    g_i ≥ 0

   where `f` is the current sequence, `M = max(f*f)` and `*` denotes ordinary
   convolution.  The LP tries to add as much mass as possible without raising
   the current maximal convolution value.  The new sequence is formed by a
   convex combination `f ← (1‑t)·f + t·g` with a modest step size `t`.  After
   each update the true score (using `evaluate_sequence`) is computed and the
   best sequence encountered is retained.

The whole procedure respects the 1000 s time budget and the returned list
conforms to the required non‑negative format.
"""

import time
import numpy as np
from math import ceil, log2
from scipy.optimize import linprog

# evaluate_sequence is provided in the execution environment.


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _max_convolution(seq: np.ndarray) -> float:
    """Return max_{k} (seq * seq)[k] using an FFT based convolution."""
    n = seq.shape[0]
    m = _next_pow2(2 * n - 1)
    # real‑FFT is enough because the sequence is real.
    fft = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fft * fft, m)[: 2 * n - 1]
    return float(conv.max())


def _fast_score(seq: np.ndarray) -> float:
    """Fast version of evaluate_sequence (same formula, FFT for convolution)."""
    n = seq.shape[0]
    max_b = _max_convolution(seq)
    sum_a = float(seq.sum())
    return 2.0 * n * max_b / (sum_a ** 2)


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1) × n matrix A where A[k, j] = f_{k‑j} if 0 ≤ k‑j < n else 0.
    Then (A @ g)[k] == (f * g)[k].
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=float)
    # Fill column by column – each column j contributes f[0:n] shifted by +j.
    for j in range(n):
        start = j
        end = j + n
        A[start:end, j] = f
    return A


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) beats the target
    bound 1.503.  The implementation follows the two‑stage approach described
    in the module docstring.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0      # safety margin

    # ------------------------------------------------------------------
    # 1. Coarse power‑law scan (quick, cheap FFT evaluation)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    # try a few lengths – larger lengths give more “mass” for the same max.
    length_candidates = [1500, 3000, 6000, 12000]
    for n in length_candidates:
        if time.time() >= deadline:
            break
        # exponent grid – concentrate around the analytic optimum (≈0.58)
        p_grid = np.linspace(0.45, 0.70, 151)
        for p in p_grid:
            if time.time() >= deadline:
                break
            seq = np.array([(i + 1) ** (-p) for i in range(n)], dtype=float)
            score = _fast_score(seq)
            if score < best_score:
                best_score = score
                best_seq = seq
                best_len = n
                best_p = p
                print(f"[grid] n={n} p={p:.5f} -> score={score:.6f}")

    # ------------------------------------------------------------------
    # 2. Linear‑programming refinement (iterative)
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform sequence of modest size
        n = 2000
        best_seq = np.full(n, 1.0, dtype=float)
        best_score = _fast_score(best_seq)
        best_len = n
        best_p = None

    f = best_seq.copy()                     # current working sequence
    n = f.shape[0]
    S_target = np.sqrt(2.0 * n)              # the “normalisation” used in theory
    # scale the starting point to the target sum – scale does *not* affect
    # the evaluation, but keeps numbers nicely sized.
    f = f * (S_target / f.sum())

    best_f = f.copy()
    best_score = _fast_score(f)
    print(f"[init] length={n}, p={best_p}, initial score={best_score:.6f}")

    # LP parameters
    step = 0.3                               # convex‑combination step size
    max_lp_iters = 25                        # safety cap – each LP is cheap for n≈2000

    for it in range(max_lp_iters):
        if time.time() >= deadline:
            break

        M = _max_convolution(f)               # current max convolution value
        # Build the constraint matrix for f * g ≤ M
        A = _build_conv_matrix(f)
        b = np.full(A.shape[0], M, dtype=float)

        # Linear programme: maximise Σ g_i  (i.e. minimise -Σ g_i)
        c = -np.ones(n, dtype=float)
        bounds = [(0.0, None)] * n

        # Solve – the HiGHS interior‑point method is fast for dense LPs.
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if not res.success:
            print(f"[LP {it}] failed: {res.message}")
            break

        g = res.x
        # Form the next iterate by a convex combination.
        f = (1.0 - step) * f + step * g

        # Re‑evaluate (using the exact evaluate_sequence for a reliable score)
        score = evaluate_sequence(f.tolist())
        print(f"[LP {it}] score={score:.6f}  max_conv={M:.6f}")

        if score < best_score:
            best_score = score
            best_f = f.copy()
            print(f"   ** new best score {best_score:.6f}")

        # slowly shrink the step size – helps stability near convergence.
        step = max(0.05, step * 0.85)

    # ------------------------------------------------------------------
    # 3. Light local hill‑climb on the best sequence found so far.
    # ------------------------------------------------------------------
    current = best_f.copy()
    current_score = best_score
    n = current.shape[0]

    # Few thousand cheap single‑coordinate perturbations.
    max_hc_iter = 3000
    for hc in range(max_hc_iter):
        if time.time() >= deadline:
            break
        idx = np.random.randint(n)
        # multiplicative jitter (±5 %)
        factor = 1.0 + 0.10 * (np.random.rand() - 0.5)
        new_val = max(0.0, current[idx] * factor)
        old_val = current[idx]
        current[idx] = new_val
        new_score = evaluate_sequence(current.tolist())
        if new_score < current_score:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_f = current.copy()
                print(f"[HC {hc}] improved score {best_score:.6f}")
        else:
            # reject – restore original value
            current[idx] = old_val

    print(f"Finished search – best score achieved: {best_score:.6f}")
    return best_f.tolist()
