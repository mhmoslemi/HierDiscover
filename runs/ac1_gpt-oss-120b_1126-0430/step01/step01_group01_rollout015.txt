    seq = seq.copy()

"""
search for a step‐function minimizing 2·n·max(conv)/sum(a)^2

Idea
----
The Catalan (discrete arcsine) sequence gives a constant
self‑convolution equal to 1, yielding the bound π/2 ≈ 1.5708.
If we shrink the first part of that sequence (the part which
produces the large convolution values) and put the removed mass
into the far tail, the early convolution plateau drops roughly as
s² (where *s* is the scaling of the early part) while the tail
convolution grows only like (1‑s)².  By choosing a split point and a
scaling factor we can make the two contributions equal and push the
maximum far below 1.  A tiny grid search over split‑ratios and
scaling factors finds an excellent candidate; a quick greedy
refinement then polishes it.

The algorithm:
1. Build a Catalan (arcsine) seed of length *n*.
2. For several split positions (r·n) and scaling factors *s*:
   * scale the first part by *s*,
   * redistribute the removed mass uniformly over the rest,
   * evaluate the objective with a fast FFT based routine,
   * keep the best sequence.
3. Run a fast greedy optimizer that repeatedly moves a modest
   amount of mass from a coefficient that contributes most to the
   current maximal convolution entry to a coefficient that contributes
   almost nothing.  Moves are sized to try to lower the maximal entry
   directly.
4. Return the best sequence found (as a plain Python list).

The whole routine respects the 1000 s budget, prints progress and
never produces negative numbers – the wrapper `evaluate_sequence`
will clip the values to the allowed range.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast evaluation (identical to evaluate_sequence but without the checks)
# ----------------------------------------------------------------------
def _fast_evaluation(seq: np.ndarray) -> float:
    """Return the same metric as evaluate_sequence but much faster."""
    n = seq.size
    if n == 0:
        return np.inf
    # FFT size = next power of two >= 2*n-1
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Catalan (arcsine) seed generator
# ----------------------------------------------------------------------
def _catalan_sequence(length: int) -> np.ndarray:
    """First ``length`` Catalan coefficients (non‑negative, decreasing)."""
    seq = np.empty(length, dtype=float)
    a = 1.0
    seq[0] = a
    for i in range(1, length):
        a *= (2 * i - 1) / (2.0 * i)          # a_i = a_{i-1} * (2i‑1)/(2i)
        seq[i] = a
    return seq


# ----------------------------------------------------------------------
# Greedy local improvement – move mass away from indices that heavily
# influence the current maximal convolution entry.
# ----------------------------------------------------------------------
def _refine(seq: np.ndarray, deadline: float) -> np.ndarray:
    """Aggressive greedy improvement until the time limit is reached."""
    n = seq.size
    best_seq = seq.copy()
    best_score = _fast_evaluation(best_seq)

    # pre‑compute FFT size (does not change)
    fft_len = 1 << ((2 * n - 1).bit_length())

    while time.time() < deadline:
        # convolution and index of its maximum
        fa = np.fft.rfft(seq, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        k = int(conv.argmax())
        max_val = conv[k]

        # contribution weights w[i] = a_{k‑i} (zero if out of range)
        i_arr = np.arange(n)
        j_arr = k - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        w = np.zeros_like(seq)
        w[mask] = seq[j_arr[mask]]

        # source = index with largest a_i * w_i  (most responsible)
        contrib = seq * w
        src = int(np.argmax(contrib))
        # sink = index that almost never contributes to the max entry
        sink = int(np.argmin(w))

        if seq[src] < 1e-12:
            break          # nothing left to move

        # try to move enough mass to make the current peak drop to the
        # second‑largest peak (linear approximation)
        sec_max = np.max(np.delete(conv, k))
        delta_needed = max_val - sec_max
        denom = 2.0 * (w[src] - w[sink])
        if denom > 0 and delta_needed > 0:
            eps_target = delta_needed / denom
            eps = min(seq[src] * 0.5, eps_target * 0.9)
        else:
            eps = seq[src] * 0.2          # fallback step size

        if eps <= 0:
            break

        new_seq = seq.copy()
        new_seq[src] -= eps
        new_seq[sink] += eps

        score = _fast_evaluation(new_seq)

        # keep improvement
        if score < best_score - 1e-12:
            seq = new_seq
            best_score = score
            best_seq = new_seq.copy()
        else:
            # occasional tiny jitter to leave shallow plateaux
            if np.random.rand() < 0.01:
                jitter_idx = np.random.randint(n)
                jitter = (np.random.rand() - 0.5) * 0.01 * seq[jitter_idx]
                tmp = seq.copy()
                tmp[jitter_idx] = max(0.0, tmp[jitter_idx] + jitter)
                seq = tmp

    return best_seq


# ----------------------------------------------------------------------
# Transform: scale the first part and redistribute the removed mass
# uniformly over the remaining part.
# ----------------------------------------------------------------------
def _scaled_transform(base: np.ndarray, split: int, scale: float) -> np.ndarray:
    """Scale ``base[:split]`` by ``scale`` and spread the removed mass
    uniformly over ``base[split:]``."""
    if split == 0 or split == base.size:
        return base.copy()
    early = base[:split] * scale
    removed = base[:split].sum() - early.sum()
    tail_len = base.size - split
    add = removed / tail_len
    out = np.empty_like(base)
    out[:split] = early
    out[split:] = base[split:] + add
    return out


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights with a low evaluation score.

    The routine stays within the time budget, starts from the Catalan
    (arcsine) distribution, explores a small grid of “scale‑early /
    redistribute‑tail” transformations, keeps the best candidate and
    finally runs a fast greedy refinement.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5  # 5 s safety margin

    # length – a few thousand is a good trade‑off
    n = 3000
    base = _catalan_sequence(n)

    best_seq = base.copy()
    best_score = _fast_evaluation(best_seq)

    # --------------------------------------------------------------
    # 1.  Grid search over split point and scaling factor
    # --------------------------------------------------------------
    split_fractions = np.linspace(0.1, 0.9, 9)      # 0.1 … 0.9
    scale_factors = np.linspace(0.0, 1.0, 21)      # 0.0, 0.05, … , 1.0

    for frac in split_fractions:
        split = int(frac * n)
        if split == 0 or split == n:
            continue
        for s in scale_factors:
            if time.time() > deadline:
                break
            cand = _scaled_transform(base, split, s)
            score = _fast_evaluation(cand)
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                print(f"grid improvement: score={best_score:.6f} "
                      f"split={split} ({frac:.2f}) scale={s:.3f}")

    # --------------------------------------------------------------
    # 2.  Greedy refinement starting from the best grid candidate
    # --------------------------------------------------------------
    best_seq = _refine(best_seq, deadline)

    # --------------------------------------------------------------
    # 3.  Return the best sequence found
    # --------------------------------------------------------------
    return best_seq.tolist()
