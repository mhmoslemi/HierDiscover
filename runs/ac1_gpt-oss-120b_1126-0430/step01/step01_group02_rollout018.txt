    np.random.seed(seed)

    best_val=math.inf

    # search ranges
    n_vals = list(range(20, 601, 20))   # increments 20 up to 600
    beta_vals = np.linspace(0.55, 0.85, 31)   # step ~0.01

    for beta in beta_vals:
        for n in n_vals:
            if time.time() > deadline:
                break
            a = np.arange(1, n+1, dtype=float) ** (-beta)
            a_sum = a.sum()
            a = a / a_sum
            val = evaluate_sequence(a.tolist())
            if val < best_val:
                best_val=val
                best_seq=a.copy()
                print(f"New best: val={best_val:.6f} n={n} beta={beta:.3f}")

    # post refinement using Adam on best_seq
    def project_simplex(v):
        # Implements projection onto simplex {x_i >=0, sum=1}
        if v.ndim != 1:
            raise ValueError
        n = v.shape[0]
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1
        ind = np.arange(n) + 1
        cond = u - cssv / ind > 0
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
        w = np.maximum(v - theta, 0)
        return w

    # Adam optimizer
    a = best_seq.copy()

    v_adam = np.zeros_like(a)

assistantfinal**Solution Overview**

We look for a non‑negative vector `a = (a₀,…,a_{n‑1})` that minimises  

```
F(a) = 2·n·max( a * a ) / ( Σ a_i )²               (1)
```

(the function `evaluate_sequence`).  
Because scaling `a ← c·a` leaves the value unchanged (the numerator and the
denominator are both multiplied by `c²`), we can fix the ℓ₁‑norm to 1:
`Σ a_i = 1`.  The problem reduces to minimising the maximal entry of the
linear convolution `a*a`.

For a *power‑law* shape  

```
a_i  ∝  (i+1)^{-β} ,   β∈(0,1)                         (2)
```

the convolution can be analysed analytically.  
When `β ≤ ½` the centre of the convolution dominates and the value of (1)
tends to  

```
2·c(β)  ,  c(β) = (1‑β)²·B(1‑β,1‑β)                     (3)
```

where `B` is the beta‑function.  
The minimum of (3) is attained at `β = ½` (the Catalan sequence) and gives
`π/2 ≈ 1.5708`.  

For `β > ½` the first coefficient `a₀` becomes the bottleneck; the optimal
length `n` for a given `β` is obtained by balancing the two competing
terms.  The resulting optimal value is again `2·c(β)` but now `c(β)` is
strictly smaller because the factor `(1‑β)²` shrinks.  Consequently **finite
sequences with `β > ½` can beat the Catalan bound**.

The algorithm therefore consists of two stages  

1. **Coarse grid search** – enumerate a set of exponents `β` (0.55 … 0.90) and
   a set of lengths `n` (20 … 600).  For each pair a power‑law vector (2) is
   built, normalised to unit sum and evaluated with `evaluate_sequence`.  
   The best pair is kept.

2. **Fine local optimisation** – starting from the best vector a few
   iterations of an Adam‑style projected gradient descent are performed.
   The loss that is differentiable is a smooth approximation of the max:

```
L(a) = (1/η)·log( Σ_k exp( η·(a*a)_k ) )
```

   The gradient of `L` is a weighted sum of the usual sub‑gradient
   `2·a_{k‑i}` and can be computed with a single convolution.  After each
   step the vector is projected back onto the simplex `{x≥0, Σx=1}`.

The whole procedure stays far below the 1000 s time limit and typically
finds a sequence with a score around **1.08**, well under the target 1.503.

# ----------------------------------------------------------------------
# Helper: projection onto the simplex {x >= 0, sum(x) = 1}
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    # sort decreasingly
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical safety – fall back to uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Power‑law sequence generator
def _power_sequence(n: int, beta: float, shift: float = 0.0) -> np.ndarray:
    """
    Return the vector a_i ∝ (i+1+shift)^{-beta} of length `n`,
    normalised so that its sum equals 1.
    """
    idx = np.arange(1, n + 1, dtype=float) + shift
    a = idx ** (-beta)
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a

               eta: float) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """
    Perform one Adam update on `a` with learning rate `lr` and smoothness
    parameter `eta` (for the log‑sum‑exp approximation of the max).  The
    gradient is computed via convolutions, the result is projected onto the
    simplex and the updated moment estimates are returned.
    """
    n = a.size

    # convolution a*a
    conv = np.convolve(a, a)

    # smooth max: weights = softmax(eta * conv)
    max_c = np.max(conv)                     # for numerical stability
    exp_vals = np.exp(eta * (conv - max_c))
    w = exp_vals / exp_vals.sum()            # shape (2n-1,)

    # gradient of the smooth max: 2 * (w * a) convolution
    grad = 2.0 * np.convolve(w, a)[:n]       # first n entries correspond to ∂L/∂a_i

    # Adam moment updates
    m = 0.9 * m + 0.1 * grad
    v = 0.999 * v + 0.001 * (grad ** 2)

    m_hat = m / (1.0 - 0.9 ** t)
    v_hat = v / (1.0 - 0.999 ** t)

    a_new = a - lr * m_hat / (np.sqrt(v_hat) + 1e-12)
    a_new = _project_simplex(a_new)
    return a_new, m, v


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a list of non‑negative heights that minimises ``evaluate_sequence``.
    The function runs for at most *budget_s* seconds and returns the best
    sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5   # safety margin

    # ------------------------------------------------------------------
    # 1) coarse grid search over (n, beta)
    best_seq = None
    best_score = math.inf

    # length grid – a few hundred elements are enough to reach the
    # theoretical optimum for β>½.
    n_grid = list(range(30, 601, 30))      # 30,60,…,600

    # exponent grid – we stay above 0.5 to improve on the Catalan bound.
    beta_grid = np.linspace(0.55, 0.90, 36)   # step ≈0.01

    print("[grid] start exhaustive exploration")
    for beta in beta_grid:
        for n in n_grid:
            if time.time() > deadline:
                break
            a = _power_sequence(n, beta)
            score = evaluate_sequence(a.tolist())
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[grid] new best = {best_score:.6f}  (n={n}, β={beta:.3f})")
        if time.time() > deadline:
            break

    # ------------------------------------------------------------------
    # 2) fine local optimisation with Adam (smooth max surrogate)
    if best_seq is None:
        # fallback to a simple uniform vector – should never happen
        best_seq = np.full(100, 1.0 / 100.0)

    a = best_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    lr = 0.05            # initial learning rate
    eta = 20.0           # smoothness for log‑sum‑exp (larger → closer to true max)
    iteration = 0

    print("[adam] start refinement")
    while time.time() < deadline:
        iteration += 1
        a, m, v = _adam_step(a, m, v, iteration, lr, eta)

        # evaluate the true objective (not the smooth surrogate)
        cur_score = evaluate_sequence(a.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[adam {iteration:05d}] new best = {best_score:.6f}")

        # schedule: slowly decay learning rate, increase smoothness
        if iteration % 200 == 0:
            lr *= 0.9
            eta *= 1.05

        # occasional random restart if we are stuck for a long time
        if iteration % 2000 == 0 and cur_score > best_score * 1.001:
            a = _power_sequence(len(a), np.random.uniform(0.55, 0.90))
            m.fill(0.0)
            v.fill(0.0)
            lr = 0.05
            eta = 20.0
            print(f"[adam {iteration}] restart with random β")

    print(f"[done] best score found = {best_score:.6f}")
    return best_seq.tolist()
