def lp_refine(a, t=0.2):
    try:
        import cvxpy as cp
        import scipy.sparse as sp
    except ImportError:
        return a  # no SDP, skip

    n = a.shape[0]
    M = max_conv   # compute from conv above
    # Build Toeplitz matrix A using sparse
    rows = np.add.outer(np.arange(n), np.arange(n)).ravel()
    cols = np.repeat(np.arange(n), n)
    data = np.repeat(a, n)  # each a_i repeated n times
    A = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))

    b = cp.Variable(n, nonneg=True)
    constraints = [A @ b <= M]
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    except cp.SolverError:
        try:
            prob.solve(solver=cp.GLPK, warm_start=True)
        except Exception:
            return a

    if b.value is None:
        return a

    b_val = np.maximum(b.value, 0.0)  # ensure nonneg
    s = b_val.sum()
    if s <= 0:
        return a
    g = b_val / s
    a_new = (1 - t) * a + t * g
    a_new = _project_to_simplex(a_new) # ensure sum=1
    return a_new

"""
Propose a candidate sequence that minimizes the evaluation bound.

Algorithm
---------
1. Work on the probability simplex: the sequence `a` is non‑negative and
   `sum(a) = 1`.  For a fixed sum the evaluation reduces to
   `O(a) = 2·n·max(conv(a,a))`.

2. Smooth approximation of the `max`:
   `Mρ(a) = (1/ρ)·log Σ_k exp(ρ·conv_k)`.  The gradient of `Mρ`
   w.r.t. `a` is a convex combination of the sub‑gradients of the
   individual convolution entries.  A large ρ makes `Mρ` close to the true
   maximum.

3. Perform a projected Adam optimisation on `Mρ`:
   * each iteration computes the full convolution with FFT (O(n log n));
   * the gradient is obtained by a second convolution (conv(p_rev, a));
   * after the Adam step the vector is projected back onto the simplex.

4. Every few thousand iterations a short linear‑programming refinement
   (optional, uses `cvxpy` if available) is performed:
   * maximise `Σ b_i` subject to `b_i ≥ 0` and `conv(f, b) ≤ M` where
     `M = max(conv(f,f))` is the current bound.
   * the solution `b` is normalised to the simplex and mixed with the
     current iterate (`f ← (1‑t)·f + t·b`).  This step is the same idea
     as the classic “tight‑set” iteration and often pushes the bound a
     little further.

5. Multiple random restarts are performed while time permits; the best
   sequence encountered is returned.

The whole procedure respects the 1000 s budget, prints progress and
never performs any I/O besides `print`.  All numbers stay non‑negative
and the final sequence is returned as a plain list of floats.
"""

import time
import math
import numpy as np

# -------------------------------------------------------------
# Helper: projection onto the probability simplex (Duchi et al.)
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x >= 0, sum(x) = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# -------------------------------------------------------------
# Fast evaluation of the bound (sum of a is assumed to be 1)
def _objective(a: np.ndarray) -> float:
    """Return the evaluation bound 2·n·max(conv(a,a)) for sum(a)=1."""
    n = a.shape[0]
    # linear convolution via FFT (size >= 2n-1)
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return 2.0 * n * conv.max()


# -------------------------------------------------------------
# Compute full convolution (linear) using a common FFT size
def _conv_fft(a: np.ndarray, size: int) -> np.ndarray:
    """Linear convolution a*a using FFT size `size`."""
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * a.shape[0] - 1]
    return conv


# -------------------------------------------------------------
# Gradient of the smooth‐max objective (using FFT)
def _smooth_max_gradient(a: np.ndarray, conv: np.ndarray, rho: float,
                         size: int) -> np.ndarray:
    """
    Compute ∇ Mρ(a) where Mρ is the smooth max of the convolution.
    Returns a gradient vector of length n.
    """
    n = a.shape[0]
    # softmax weights for each convolution entry
    max_val = conv.max()
    w = np.exp(rho * (conv - max_val))          # subtract max for stability
    p = w / w.sum()                             # probabilities, sum to 1

    # Convolution of p_rev with a gives Σ_k p_k a_{k-i}
    p_rev = p[::-1]
    A = np.fft.rfft(a, n=size)
    P = np.fft.rfft(p_rev, n=size)
    grad_full = np.fft.irfft(P * A, n=size)

    # indices that correspond to Σ_k p_k a_{k-i}
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return grad


# -------------------------------------------------------------
# Optional LP refinement step (original “tight‑set” idea)
def _lp_refine(a: np.ndarray, max_conv: float, t: float = 0.2):
    """
    Solve   maximise Σ b_i   subject to  b≥0  and  conv(a,b) ≤ max_conv.
    The solution is mixed back into `a` with weight `t`.
    Returns the new vector (still on the simplex) or the original `a`
    if the LP cannot be solved.
    """
    try:
        import cvxpy as cp
        import scipy.sparse as sp
    except Exception:
        return a   # cvxpy / scipy not available, skip refinement

    n = a.shape[0]
    # Build the (2n‑1)×n Toeplitz matrix for conv(a,b)
    rows = np.add.outer(np.arange(n), np.arange(n)).ravel()
    cols = np.repeat(np.arange(n), n)
    data = np.repeat(a, n)
    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    b = cp.Variable(n, nonneg=True)
    constraints = [A @ b <= max_conv]
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.GLPK, warm_start=True, max_iters=5000, verbose=False)
        except Exception:
            return a

    if b.value is None:
        return a

    b_val = np.maximum(b.value, 0.0)
    s = b_val.sum()
    if s <= 0:
        return a
    g = b_val / s
    a_new = (1.0 - t) * a + t * g
    a_new = _project_to_simplex(a_new)
    return a_new

    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # -----------------------------------------------------------------

    if 'height_sequence_1' in globals():

    const_seq = np.full(n, 1.0 / n)
    if 'height_sequence_1' in globals():

        if seed_vec.size < n:

        else:

    best_seq = a.copy()

    fft_size = 1 << ((3 * n - 2).bit_length())
