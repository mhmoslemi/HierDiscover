"""
Hybrid optimizer for the step‑function bound.

The objective (on the simplex) is
    O(a) = 2·n·max(conv(a,a)),    sum(a)=1, a_i≥0.

Algorithm
---------
1. Initialise a (length n) as the known best sequence (if available) or the
   uniform distribution, normalised to sum = 1.
2. Main loop (time‑bounded):
   * Random “mass‑transfer” hill‑climbing moves on the simplex.
   * Periodic sub‑gradient steps on the maximal convolution entry.
   * Every `lp_interval` iterations solve a linear programme:
        maximise   Σ g_i
        subject to  g_i ≥ 0
                    conv(a,g) ≤ M   (component‑wise)
     where M = max(conv(a,a)).  The LP gives a direction g that can be mixed
     with a without increasing the current worst convolution value.
   * Replace a ← (1‑t)·a + t·g (t decays from 0.5 to a small floor).  Accept
     the new a only if the objective improves.
3. Keep the best sequence seen and finally return it.

The LP is solved with CVXPY’s `conv` atom (linear because the first argument
is a constant).  The whole routine respects a 1000 s budget and reports
progress.  All returned values are non‑negative and sum to 1, so the
`evaluate_sequence` function can be applied directly.
"""

import time
import math
import numpy as np
import cvxpy as cp

# -------------------------------------------------------------
# Helper: projection onto the probability simplex (Duchi et al.)
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x >= 0, sum(x) = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# -------------------------------------------------------------
# Objective (fast version, works on the simplex)
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) assuming sum(a)=1."""
    conv = np.convolve(a, a)
    return float(2.0 * a.size * conv.max())


# -------------------------------------------------------------
# Sub‑gradient of the maximal convolution entry
def _subgradient_max_conv(a: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv(a,a)) w.r.t. a (sum(a)=1)."""
    n = a.size
    conv = np.convolve(a, a)
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)  # indices attaining the maximum
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    g /= len(ks)  # average if several maxima
    return g


# -------------------------------------------------------------
# Solve the LP: maximise Σ g_i  subject to g≥0 and conv(a,g) ≤ M
def _solve_lp_for_f(f: np.ndarray, M: float) -> np.ndarray:
    """Return a feasible probability vector g (sum=1) from the LP."""
    n = f.shape[0]
    g = cp.Variable(n, nonneg=True)

    # Linear expression conv(f,g); because f is constant this is affine.
    conv_f_g = cp.conv(cp.Constant(f), g)

    # tiny slack to avoid numerical infeasibility
    eps = 1e-12
    constraints = [conv_f_g <= M + eps]

    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    # ECOS is a good default for dense linear programmes
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=20000)

    if g.value is None:
        # fallback: return the current f (should be feasible)
        return f.copy()

    g_opt = np.array(g.value).flatten()
    # Numerical cleanup
    g_opt = np.clip(g_opt, 0, None)
    total = g_opt.sum()
    if total <= 0:
        return np.full(n, 1.0 / n)
    return g_opt / total


# -------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimizes the
    evaluation function.  Returns a list of floats (sum to 1).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ---------------------------------------------------------
    # Sequence length
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
    else:
        n = 4000                     # a comfortable default

    # ---------------------------------------------------------
    # Initialise the current point a
    if 'height_sequence_1' in globals():
        a = np.array(height_sequence_1, dtype=float)
        a = np.clip(a, 0, None)
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / a_sum
        # add a whisper of noise to break possible symmetries
        a += np.random.randn(n) * 0.001
        a = np.clip(a, 0, None)
        a = _project_to_simplex(a)
    else:
        a = np.full(n, 1.0 / n)

    cur_obj = _objective(a)
    best_a = a.copy()
    best_obj = cur_obj
    print(f"[init] n={n}, O={cur_obj:.6f}")

    # ---------------------------------------------------------
    # Hyper‑parameters
    T0 = 0.02                     # temperature for occasional uphill moves
    T = T0
    t_max = 0.5                   # initial mixing weight for LP step
    t_min = 0.05                  # lower bound for mixing weight
    lp_interval = 2000           # how many local iterations between LP solves
    max_no_improve = 4000        # restart threshold for local search
    iter_without_improve = 0
    total_iters = 0

    while time.time() < deadline:
        total_iters += 1
        # -------------------------------------------------
        # 1) Random mass‑transfer move (hill‑climbing)
        i, j = np.random.randint(n, size=2)
        if i != j and a[i] > 0:
            # move up to 8 % of the mass from i to j
            delta = a[i] * np.random.rand() * 0.08
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            a_new = np.clip(a_new, 0, None)
            a_new = _project_to_simplex(a_new)

            new_obj = _objective(a_new)

            # greedy acceptance with a tiny Metropolis component
            if new_obj < cur_obj or np.random.rand() < math.exp((cur_obj - new_obj) / max(T, 1e-12)):
                a = a_new
                cur_obj = new_obj
                iter_without_improve = 0
                if cur_obj < best_obj:
                    best_obj = cur_obj
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] New global best (move) O={best_obj:.6f} iter={total_iters}")
            else:
                iter_without_improve += 1

        # -------------------------------------------------
        # 2) Sub‑gradient step (every few iterations)
        if total_iters % 13 == 0:
            g = _subgradient_max_conv(a)
            # diminishing step size
            eta = 0.07 * max(0.0, 1.0 - total_iters / 50000.0)
            a = a - eta * g
            a = np.clip(a, 0, None)
            a = _project_to_simplex(a)
            cur_obj = _objective(a)
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                print(f"[{time.time():.1f}] New global best (subgrad) O={best_obj:.6f} iter={total_iters}")

        # -------------------------------------------------
        # 3) Linear‑programming refinement (periodic)
        if total_iters % lp_interval == 0:
            # current bound
            conv_aa = np.convolve(a, a)
            M = conv_aa.max()
            try:
                g_lp = _solve_lp_for_f(a, M)

                # try a decreasing schedule for the mixing parameter
                t_now = max(t_min, t_max * (1.0 - total_iters / 200000.0))
                candidate = (1.0 - t_now) * a + t_now * g_lp
                candidate = _project_to_simplex(candidate)
                cand_obj = _objective(candidate)

                if cand_obj < cur_obj:
                    a = candidate
                    cur_obj = cand_obj
                    iter_without_improve = 0
                    if cur_obj < best_obj:
                        best_obj = cur_obj
                        best_a = a.copy()
                        print(f"[{time.time():.1f}] New global best (LP) O={best_obj:.6f} iter={total_iters}")
                else:
                    # if no improvement, keep the old a (but still reset temperature a bit)
                    pass
            except Exception as e:
                print(f"[{time.time():.1f}] LP step failed: {e}")

        # -------------------------------------------------
        # Temperature decay
        T = T0 * (1.0 - total_iters / 200000.0)
        T = max(T, 1e-5)

        # -------------------------------------------------
        # Restart if stuck too long
        if iter_without_improve >= max_no_improve:
            # re‑initialise locally around the current best
            a = best_a.copy()
            a += np.random.randn(n) * 0.003
            a = np.clip(a, 0, None)
            a = _project_to_simplex(a)
            cur_obj = _objective(a)
            iter_without_improve = 0
            print(f"[{time.time():.1f}] Restart triggered at iter={total_iters}")

        # small time guard
        if time.time() >= deadline:
            break

    # ---------------------------------------------------------
    # Final verification with the original evaluate_sequence (optional)
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence call failed:", exc)

    return best_a.tolist()
