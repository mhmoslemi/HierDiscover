def propose_candidate(...):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget - 5.0

    # Choose length n
    try: n = len(height_sequence_1) except: n=8000

    # Search best p in [0.45, 0.75] via coarse grid using analytic approx and evaluate full evaluation via our cheap formula
    best_p = None; best_a = None; best_score = inf
    for p in grid:
        a = np.array([(i+1) ** (-p) for i in range(n)], dtype=float)
        a /= a.sum()
        # compute bound using our own compute, not evaluate_sequence
        b = np.convolve(a, a)
        bound = 2 * n * b.max() / (a.sum()**2)
        if bound < best_score:
            best_score = bound; best_p=p; best_a=a.copy()

    print("Initial best p", best_p, "score", best_score)

    # gradient descent starting from best_a
    a = best_a.copy()
    best_seq = a.copy()
    best_score = best_score

    eta0 = 0.5  # initial step size
    iter = 0
    while time.time() < deadline:
        # compute conv and max idx
        b = np.convolve(a, a)
        k = int(b.argmax())
        max_b = float(b[k])
        # compute gradient
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = a[j[mask]]
        # double self term
        self_mask = mask & (j == idx)
        g[self_mask] += a[idx[self_mask]]

        # step size schedule
        eta = eta0 / np.sqrt(iter+1)
        # update multiplicatively
        a = a * np.exp(-eta * g)
        # renorm
        a_sum = a.sum()
        if a_sum == 0:
            # reinitialize maybe
            a = np.full(n, 1.0/n)

        if bound < best_score:

            best_seq = a.copy()
            print(f"Iter {iter} improved bound {best_score}")

        iter+=1
        # optionally break after many iter

def convolve_full_fft(a):
    n = a.shape[0]
    size = 1 << ((2*n - 1).bit_length())
    # compute FFT of a padded to size
    fft_a = np.fft.rfft(a, size)
    fft_c = fft_a * fft_a
    conv = np.fft.irfft(fft_c, size)
    # return first 2*n-1 entries
    return conv[:2*n - 1]

    if total == 0.0:
        # extremely unlikely, fall back to uniform distribution
        a_new = np.full_like(a, 1.0 / a.shape[0])
    else:
        a_new /= total
    return a_new

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that yields a low
    `evaluate_sequence` score.  The function stops before the time budget
    expires and returns the best sequence it found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin so the runner does not force an abrupt timeout
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # 1. Choose a set of candidate lengths.
    # ------------------------------------------------------------------
    try:
        base_n = len(height_sequence_1)  # type: ignore[name-defined]
    except Exception:
        base_n = 5000

    # Lengths around the reference size; keep them modest for speed.
    candidate_lengths = sorted(
        {
            max(200, int(base_n * r))
            for r in [0.5, 0.75, 1.0, 1.25, 1.5]
        }
    )

    # ------------------------------------------------------------------
    # 2. Coarse grid search over (p, offset) for each length.
    # ------------------------------------------------------------------
    best_a = None
    best_score = float("inf")
    best_params = None

    # Parameter ranges (chosen empirically).
    p_vals = np.linspace(0.55, 0.70, 16)            # exponent
    offset_vals = np.linspace(0.5, 2.5, 16)        # shift of the power law

    # Analytical proxy E(p) (independent of offset) to visit promising p first.
    def analytic_E(p):
        u = 1.0 - p
        return 2.0 * (u ** 2) * (gamma(u) ** 2) / gamma(2.0 * u)

    p_vals = sorted(p_vals, key=analytic_E)

    for n in candidate_lengths:
        if time.time() >= deadline:
            break
        print(f"\n--- Scanning length n = {n} ---")
        for offset in offset_vals:
            for p in p_vals:
                if time.time() >= deadline:
                    break
                a = _initial_power_sequence(n, p, offset=offset)
                score = _bound_from_seq(a)
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                    best_params = (n, p, offset)
                    print(
                        f"  New best: n={n}, p={p:.4f}, offset={offset:.2f} → {best_score:.6f}"
                    )

    # ------------------------------------------------------------------
    # 3. Refine the best candidate with projected gradient descent.
    # ------------------------------------------------------------------
    if best_a is None:
        # Fallback (should never happen).
        best_a = _initial_power_sequence(base_n, 0.60, offset=1.0)
        best_score = _bound_from_seq(best_a)
        best_params = (base_n, 0.60, 1.0)

    a = best_a.copy()
    n = a.shape[0]

    eta0 = 0.5          # initial step size
    decay = 0.0005      # decay of the step size
    max_iter = 20000
    iter_count = 0

    best_seq = a.copy()
    best_seq_score = best_score

    while time.time() < deadline and iter_count < max_iter:
        conv = _conv_full_fft(a)
        k_max = int(conv.argmax())
        grad = _gradient_max_conv(a, k_max)
        eta = eta0 / (1.0 + decay * iter_count)
        a = _exponential_update(a, grad, eta)

        cur_score = _bound_from_seq(a)
        if cur_score < best_seq_score:
            best_seq_score = cur_score
            best_seq = a.copy()
            print(
                f"Iter {iter_count:05d}: improved bound {best_seq_score:.6f}"
            )
        iter_count += 1

    # ------------------------------------------------------------------
    # 4. Short random hill‑climb for final polishing.
    # ------------------------------------------------------------------
    a = best_seq.copy()
    cur_score = best_seq_score
    hill_iters = 5000
    for _ in range(hill_iters):
        if time.time() >= deadline:
            break
        idx = np.random.randint(n)
        factor = 1.0 + 0.02 * (np.random.rand() - 0.5)   # ±1 %
        new_val = max(0.0, a[idx] * factor)
        old_val = a[idx]
        a[idx] = new_val
        total = a.sum()
        if total == 0.0:
            a[idx] = old_val
            continue
        a /= total
        new_score = _bound_from_seq(a)
        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_seq_score:
                best_seq_score = new_score
                best_seq = a.copy()
                print(f"Hill climb: new best {best_seq_score:.6f}")
        else:
            a[idx] = old_val   # revert

    print("\nSearch finished.")
    print(f"Best bound achieved: {best_seq_score:.6f}")
    if best_params:
        print(f"Best parameters (n, p, offset): {best_params}")

    # Return a plain Python list, as required.
    return best_seq.tolist()
