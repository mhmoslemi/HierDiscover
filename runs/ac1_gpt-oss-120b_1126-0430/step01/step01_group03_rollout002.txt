"""
propose_candidate
-----------------

Search for a non‑negative step‑function (a list of heights) that minimizes

    evaluate_sequence(a) = 2·n·max_k (a*a)_k / (sum a)^2

where “*” denotes discrete convolution.
The evaluation is scale‑invariant, so we work with probability vectors
(sum = 1).

Algorithm
~~~~~~~~~
1. **Parameter sweep** –  a few families of simple analytic shapes are
   generated with a small number of parameters.
      * Power‑law:      a_i ∝ (i+1+α)^(-β)
      * Symmetric power‑law (large mass at both ends):
                       a_i ∝ min(i+1, n-i)^(-β)
      * Uniform (fallback)

   Several lengths `n` (2000, 4000, 8000) are examined and the best
   candidate seen by `evaluate_sequence` is kept.

2. **Gradient‑guided simulated annealing** – starting from the best
   sequence, repeatedly move a small amount of mass from an index that
   contributes strongly to the current maximal convolution value to a
   random other index.  Moves that improve the score are always kept;
   worsening moves are accepted with a temperature that decays linearly
   with the remaining time.

All time checks respect the 1000 s budget.  The routine prints progress
messages and finally returns the best sequence found (as a plain Python
list).  No closures, file I/O or network access are used.
"""

import time
import math
import numpy as np

# The evaluation routine is provided in the execution environment.
# from <environment> import evaluate_sequence


def _eval(seq: np.ndarray) -> float:
    """Convenient wrapper that forwards a NumPy array to evaluate_sequence."""
    return evaluate_sequence(seq.tolist())


def _normalize(seq: np.ndarray) -> np.ndarray:
    """Scale a non‑negative array to have sum = 1 (protect against division by zero)."""
    s = seq.sum()
    if s < 1e-12:
        return seq
    return seq / s


def _make_power_law(n: int, beta: float, alpha: float = 0.0) -> np.ndarray:
    """a_i ∝ (i+1+α)^(-β)."""
    idx = np.arange(1, n + 1, dtype=np.float64) + alpha
    seq = idx ** (-beta)
    return _normalize(seq)


def _make_sym_power_law(n: int, beta: float) -> np.ndarray:
    """
    Symmetric power law.
    a_i ∝ distance_to_nearest_edge^(-β),
    where distance = min(i+1, n-i).
    """
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i + 1, n - i)   # 1‑based distance to the nearer end
    seq = dist ** (-beta)
    return _normalize(seq)


def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) yields a low
    value of evaluate_sequence within the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.1  # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0.  Start from the previously found construction, if any.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            sc = evaluate_sequence(height_sequence_1)
            best_seq = list(height_sequence_1)
            best_score = sc
            print(f"[init] height_sequence_1 score = {sc:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Parameter sweep over a few analytic families.
    # ------------------------------------------------------------------
    # Lengths to try – larger lengths approach the asymptotic constant.
    length_candidates = [2000, 4000, 8000]

    # Exponents for power‑law families.
    betas = np.linspace(0.40, 0.80, 41)          # step 0.01
    alphas = [0.0, 1.0, 2.0]                    # shift of the power law

    # Reserve about 40 % of the budget for the exhaustive scan.
    scan_deadline = start_time + budget_s * 0.4

    for n in length_candidates:
        if time.time() > scan_deadline:
            break
        # ---- power‑law (with shift) ---------------------------------
        for beta in betas:
            if time.time() > scan_deadline:
                break
            for alpha in alphas:
                seq = _make_power_law(n, beta, alpha)
                try:
                    sc = _eval(seq)
                except Exception:
                    continue
                if sc < best_score:
                    best_score = sc
                    best_seq = seq.tolist()
                    print(f"[scan] power n={n} β={beta:.3f} α={alpha:.1f} → {sc:.6f}")

        # ---- symmetric power‑law ------------------------------------
        for beta in betas:
            if time.time() > scan_deadline:
                break
            seq = _make_sym_power_law(n, beta)
            try:
                sc = _eval(seq)
            except Exception:
                continue
            if sc < best_score:
                best_score = sc
                best_seq = seq.tolist()
                print(f"[scan] sym‑pow n={n} β={beta:.3f} → {sc:.6f}")

    # ------------------------------------------------------------------
    # 2.  Gradient‑guided simulated annealing (local refinement).
    # ------------------------------------------------------------------
    if best_seq is None:
        # Fallback: uniform distribution of length 1000.
        n = 1000
        best_seq = [1.0 / n] * n
        best_score = _eval(np.array(best_seq, dtype=np.float64))
        print(f"[fallback] uniform n={n} → {best_score:.6f}")

    seq = np.array(best_seq, dtype=np.float64)   # working copy
    n = len(seq)
    current_score = best_score

    # Time budget for local search: remaining time (≈60 % of total)
    while time.time() < deadline:
        # Compute the full convolution and the index of its maximum.
        conv = np.convolve(seq, seq)
        max_val = conv.max()
        k0 = int(conv.argmax())

        # Gradient of the maximal convolution entry w.r.t. each component.
        # ∂b_k0/∂a_i = a_{k0-i}   if the index is inside the support.
        grad = np.zeros_like(seq)
        lo = max(0, k0 - (n - 1))
        hi = min(n - 1, k0)
        if lo <= hi:
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] = seq[k0 - idx]

        # Choose a source index i with probability proportional to its
        # contribution to the maximal entry (larger gradient → higher chance).
        probs_i = grad + 1e-14                # avoid all‑zero case
        probs_i /= probs_i.sum()
        i = np.random.choice(n, p=probs_i)

        # Choose a different destination index uniformly.
        j = i
        while j == i:
            j = np.random.randint(0, n)

        # Transfer a small fraction (≤5 %) of the mass from i to j.
        eps = seq[i] * np.random.rand() * 0.05
        if eps <= 0:
            continue
        new_seq = seq.copy()
        new_seq[i] -= eps
        new_seq[j] += eps
        # Numerical guard – should never be negative, but protect anyway.
        if new_seq[i] < 0:
            continue

        # Evaluate the new candidate.
        try:
            new_score = _eval(new_seq)
        except Exception:
            continue

        # Accept always if we improve.
        if new_score < current_score:
            seq = new_seq
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.tolist()
                print(f"[anneal] iter improvement → {new_score:.6f}")
            continue

        # Otherwise accept with a temperature that decays linearly.
        elapsed = time.time() - start_time
        remaining = max(0.0, deadline - time.time())
        # Linear schedule: T_start → T_end
        T_start, T_end = 0.02, 1e-6
        if deadline > start_time:
            t_frac = remaining / (deadline - start_time)
        else:
            t_frac = 0.0
        T = T_start * t_frac + T_end * (1.0 - t_frac)

        if np.random.rand() < math.exp(-(new_score - current_score) / T):
            seq = new_seq
            current_score = new_score

    print(f"[final] best score = {best_score:.6f} (length = {len(best_seq)})")
    return best_seq
