def _smooth_max_and_grad(a, alpha, L):
    # a: 1D array, sum to 1.
    # Compute FFT of a based on length L.
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)[:2*n-1]
    # compute weights
    # Use max-subtraction for numerical stability
    max_conv = conv.max()
    exp_term = np.exp(alpha * (conv - max_conv))
    sum_exp = exp_term.sum()
    w = exp_term / sum_exp
    # gradient: g_i = 2 * Σ_j a_j w_{i+j}
    # Compute correlation via FFT
    W = np.fft.rfft(w, n=L)
    corr = np.fft.irfft(A * np.conj(W), n=L)[:n]  # length n
    grad = 2 * corr
    smooth_max = (np.log(sum_exp) + alpha * max_conv) / alpha
    return smooth_max, grad, conv

"""
Projected‑gradient optimizer for the step‑function bound.

The objective (with sum(a)=1) is  O(a)=2·n·max(conv(a,a)).
Since the function is scale‑invariant we optimise over the probability
simplex.  A smooth approximation of the max (log‑sum‑exp) is used and
the smoothing parameter α is increased during optimisation (continuation).
Both the convolution and the gradient are evaluated by FFT which makes a
single iteration O(L log L) with L≈3·n.  Several random restarts are run;
the best sequence found (projected to the simplex after every update)
is returned.  The code respects the 1000 s budget, prints progress and
finally validates the result with the provided `evaluate_sequence`.
"""

import time
import math
import numpy as np

# -------------------------------------------------------------
# Helper: projection onto the probability simplex (Duchi et al.)
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# -------------------------------------------------------------
# Fast objective (sum of a is 1)
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a simplex‑normalised a."""
    n = a.size
    # convolution via FFT (fast for large n)
    L = 1 << (int(np.ceil(np.log2(3 * n))))
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
    return 2.0 * n * np.max(conv)


# -------------------------------------------------------------
def _smooth_max_and_grad(
    a: np.ndarray, alpha: float, L: int, n: int
) -> tuple[float, np.ndarray, np.ndarray]:
    """
    Smooth approximation of max(conv) and its gradient.

    Returns (smooth_max, grad, conv_full) where
        smooth_max = (1/α)·log Σ_k exp(α·conv_k)
        grad_i      = 2·Σ_j a_j·w_{i+j}   (correlation a⋆w)
    and conv_full is the full convolution a*a (length 2n−1).
    """
    # FFT of the current vector
    A = np.fft.rfft(a, n=L)

    # full convolution a*a
    conv = np.fft.irfft(A * A, n=L)[: 2 * n - 1]

    # stable log‑sum‑exp
    max_val = conv.max()
    exp_term = np.exp(alpha * (conv - max_val))
    sum_exp = exp_term.sum()
    w = exp_term / sum_exp           # weights, length 2n−1

    # gradient : g = 2·(a ⋆ w)  (cross‑correlation)
    W = np.fft.rfft(w, n=L)
    corr = np.fft.irfft(A * np.conj(W), n=L)[:n]   # first n entries = a⋆w
    grad = 2.0 * corr

    smooth_max = (np.log(sum_exp) + alpha * max_val) / alpha
    return smooth_max, grad, conv


# -------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  Returns the best sequence found within the
    time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # ---- determine a reasonable length ---------------------------------
    if "height_sequence_1" in globals():
        base_len = len(height_sequence_1)
    else:
        base_len = 4000

    # work with a power‑of‑two length (nice for FFT) but not too large
    n = 1 << max(int(math.ceil(math.log2(max(base_len, 4096)))), 0)
    n = min(n, 16384)          # upper bound to keep runtime safe

    # FFT length (>= 3n to avoid circular wrap‑around)
    L = 1 << (int(math.ceil(math.log2(3 * n))))

    # ---- initialise best candidate -------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.zeros(n)
        a0[: base_len] = height_sequence_1
        a0 = _project_to_simplex(a0)
    else:
        a0 = np.full(n, 1.0 / n)

    best_seq = a0.copy()
    best_score = _objective(best_seq)
    print(f"[init] n={n}, initial O={best_score:.6f}")

    restart = 0
    while time.time() < deadline:
        restart += 1

        # ----- start a new restart ---------------------------------------
        if best_seq is not None and np.random.rand() < 0.25:
            a = best_seq + np.random.randn(n) * 0.01
            a = np.clip(a, 0, None)
        else:

        a = _project_to_simplex(a)

        # ----- local optimisation (smooth‑max gradient descent) ---------
        max_iters = 25000

                # track global best
                if new_obj < best_score:
                    best_score = new_obj
                    best_seq = a.copy()
                    print(
                        f"[{time.time():.1f}] Restart {restart} it {it} "

            else:

            # optional early stop if we stagnate for a while

        if time.time() < deadline:
            a = best_seq + np.random.randn(n) * 0.005
            a = np.clip(a, 0, None)
            a = _project_to_simplex(a)

    # --------------------------------------------------------------------
    # Final validation with the provided evaluate_sequence (scale‑invariant)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:
        print("[final] evaluate_sequence call failed:", e)

    return best_seq.tolist()
