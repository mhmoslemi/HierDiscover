"""
Propose a step‑function (non‑negative list of heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

where ``conv`` is ``np.convolve``.  The score is scale‑invariant, therefore we
keep the L1‑norm fixed to ``sqrt(2*n)``.  The algorithm works in two stages:

1. **Diverse random sampling** – many analytic families (power‑law,
   shifted power‑law, polynomial decay, linear, Dirichlet, Gaussian,
   symmetric “bell”, cosine, tent, and symmetric‑power shapes) are sampled.
   Each raw vector is scaled to the required L1‑norm and evaluated; the best
   sequence is retained.

2. **Projected sub‑gradient descent** – starting from the best candidate we
   repeatedly locate the convolution index that attains the current maximum,
   compute the sub‑gradient of that quadratic form, move a small fraction of
   mass opposite the gradient, and project back onto the simplex (non‑negative,
   fixed sum).  A simulated‑annealing acceptance rule lets the method escape
   shallow local minima.  The loop runs until the time budget expires.

Both stages are lightweight and respect a 1000 s time limit.  If a global
variable ``height_sequence_1`` is present it is also evaluated as a possible
starting point.

The returned list can be passed directly to ``evaluate_sequence``.
"""

def _project_to_simplex(x: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Projects a real vector onto the simplex {y >= 0, sum(y) = target_sum}.
    Implements the O(n) algorithm of Wang & Carreira‑Perpiñán (2013).

    Parameters
    ----------
    x : np.ndarray
        Input vector (will not be modified).
    target_sum : float
        Desired L1‑norm after projection (must be non‑negative).

    Returns
    -------
    np.ndarray
        Projected vector with the same shape as ``x``.
    """
    n = x.shape[0]
    if target_sum <= 0:
        return np.zeros_like(x)

    # sort descending
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho_arr = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(x - theta, 0.0)
    return w


def _evaluate(seq):
    """Thin wrapper around the provided ``evaluate_sequence``."""
    return evaluate_sequence(seq)


def _make_sequence_from_raw(raw: np.ndarray, target_sum: float) -> list:
    """Scale a non‑negative raw vector so that its L1‑norm equals ``target_sum``."""
    raw_sum = raw.sum()
    if raw_sum == 0.0:
        # fallback to uniform distribution
        n = raw.shape[0]
        return [target_sum / n] * n
    scale = target_sum / raw_sum
    seq = (raw * scale).tolist()
    return seq


def _generate_raw(shape_type: str, n: int) -> np.ndarray:
    """Return a non‑negative raw (pre‑scaled) vector for the given shape."""
    rng = np.random
    if shape_type == "power":
        p = rng.uniform(0.1, 1.0)
        raw = (np.arange(1, n + 1, dtype=float)) ** (-p)

    elif shape_type == "shifted_power":
        p = rng.uniform(0.1, 1.0)
        d = rng.uniform(0.0, n * 0.5) + 1e-6
        raw = (np.arange(n, dtype=float) + d) ** (-p)

    elif shape_type == "poly":
        alpha = rng.uniform(0.1, 5.0)
        raw = (1.0 - np.arange(n, dtype=float) / (n - 1)) ** alpha

    elif shape_type == "linear":
        raw = np.linspace(1.0, 0.0, n, dtype=float)

    elif shape_type == "rand_dir":
        raw = rng.gamma(shape=0.7, scale=1.0, size=n)

    elif shape_type == "gaussian":
        sigma = rng.uniform(0.05, 0.25) * n
        centre = (n - 1) / 2.0
        raw = np.exp(-((np.arange(n, dtype=float) - centre) ** 2) / (2.0 * sigma ** 2))

    elif shape_type == "bell":
        # product of distances from the two ends, raised to a power
        beta = rng.uniform(0.3, 2.0)
        i = np.arange(n, dtype=float)
        prod = (i + 0.5) * (n - 0.5 - i)      # zero at the exact ends, positive inside
        raw = prod ** beta

    elif shape_type == "cosine":
        # 0 at the ends, 1 in the centre
        theta = np.pi * (np.arange(n, dtype=float) - (n - 1) / 2.0) / (n - 1)
        raw = np.cos(theta)
        raw = np.clip(raw, 0.0, None)       # negative values become 0

    elif shape_type == "tent":
        raw = 1.0 - np.abs(np.arange(n, dtype=float) - (n - 1) / 2.0) / ((n - 1) / 2.0)
        raw = np.clip(raw, 0.0, None)

    elif shape_type == "symmetric_power":
        p = rng.uniform(1.0, 6.0)
        q = rng.uniform(0.5, 3.0)
        dist = np.abs(np.arange(n, dtype=float) - (n - 1) / 2.0) / ((n - 1) / 2.0)
        raw = (1.0 - dist ** p) ** q
        raw = np.clip(raw, 0.0, None)

    else:
        # fallback – uniform
        raw = np.ones(n, dtype=float)

    # guarantee non‑negativity
    raw = np.maximum(raw, 0.0)
    return raw


def _subgradient_conv_max(seq: np.ndarray, p_max: int) -> np.ndarray:
    """
    Sub‑gradient of the convolution entry ``c[p_max] = sum_{i+j=p_max} a_i a_j``
    with respect to ``a``.  For each valid index ``i`` the gradient is
    ``2 * a_{p_max - i}``, otherwise ``0``.
    """
    n = seq.shape[0]
    g = np.zeros_like(seq)

    lo = max(0, p_max - (n - 1))
    hi = min(n - 1, p_max)
    i = np.arange(lo, hi + 1, dtype=int)
    j = p_max - i
    g[i] = 2.0 * seq[j]
    return g

        # a moderate size that is still cheap to evaluate
        n = 3000

    target_sum = math.sqrt(2.0 * n)          # enforced L1‑norm

    # ------------------------------------------------------------------
    # Baseline (uniform) and optional seed from the global variable
    # ------------------------------------------------------------------
    best_seq = [target_sum / n] * n
    best_score = _evaluate(best_seq)
    print(f"[init] n={n}, uniform score={best_score:.6f}")

    if "height_sequence_1" in globals():
        raw0 = np.asarray(height_sequence_1, dtype=float)
        # ensure correct length – if the global has a different length, truncate/pad
        if raw0.shape[0] != n:
            if raw0.shape[0] > n:
                raw0 = raw0[:n]
            else:
                pad = np.zeros(n - raw0.shape[0], dtype=float)
                raw0 = np.concatenate([raw0, pad])
        cand = _make_sequence_from_raw(raw0, target_sum)
        sc = _evaluate(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand
            print(f"[seed] global start score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – diverse random sampling
    # ------------------------------------------------------------------
    shape_catalog = [
        "power", "shifted_power", "poly", "linear", "rand_dir",
        "gaussian", "bell", "cosine", "tent", "symmetric_power"
    ]
    it = 0
    while time.time() < deadline_phase1:
        it += 1
        shape = np.random.choice(shape_catalog)

        raw = _generate_raw(shape, n)

        # occasional mixing of two shapes to increase diversity
        if np.random.rand() < 0.15:
            shape2 = np.random.choice(shape_catalog)
            raw2 = _generate_raw(shape2, n)
            alpha = np.random.rand()
            raw = alpha * raw + (1.0 - alpha) * raw2

        cand = _make_sequence_from_raw(raw, target_sum)
        sc = _evaluate(cand)

        if sc < best_score:
            best_score = sc
            best_seq = cand
            print(f"[sweep {it:05d}] new best {best_score:.6f} via {shape}")

        # small random perturbation every so often to shake out of plateaus
        if it % 300 == 0 and np.random.rand() < 0.25:
            tmp = np.array(best_seq, copy=True)
            i, j = np.random.choice(n, size=2, replace=False)
            delta = tmp[i] * np.random.uniform(0.0, 0.02)
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_to_simplex(tmp, target_sum)
            pert_sc = _evaluate(tmp.tolist())
            if pert_sc < best_score:
                best_score = pert_sc
                best_seq = tmp.tolist()
                print(f"[perturb {it:05d}] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent (with optional SA)
    # ------------------------------------------------------------------
    current = np.array(best_seq, copy=True)
    current_score = best_score

    # learning‑rate schedule
    eps_frac = 0.02          # fraction of the total L1‑norm to move each step
    eps_decay = 0.9999
    temperature = 0.02
    temp_decay = 0.9995

    # time left for refinement
    deadline_refine = start_time + budget_s - 2.0
    iter_refine = 0
    while time.time() < deadline_refine:
        iter_refine += 1

        # 1) locate the worst convolution entry
        conv = np.convolve(current, current)
        p_max = int(conv.argmax())
        max_val = conv[p_max]

        # 2) sub‑gradient at that entry
        g = _subgradient_conv_max(current, p_max)

        # 3) move a small fraction of mass opposite the gradient
        #    – we transfer exactly eps * target_sum mass in L1.
        eps = eps_frac * target_sum
        if np.linalg.norm(g, 1) > 0:
            direction = -g / np.linalg.norm(g, 1)   # unit L1‑norm direction
        else:
            direction = np.zeros_like(g)
        new_seq = current + eps * direction
        new_seq = _project_to_simplex(new_seq, target_sum)

        new_score = _evaluate(new_seq.tolist())

        # 4) acceptance rule
        if new_score < current_score:
            current = new_seq
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.tolist()
                print(f"[refine {iter_refine:06d}] new best {best_score:.6f}")
        else:
            # simulated annealing acceptance
            prob = math.exp(-(new_score - current_score) / (temperature + 1e-12))
            if np.random.rand() < prob:
                current = new_seq
                current_score = new_score

        # schedule updates
        eps_frac *= eps_decay
        temperature *= temp_decay

    # ------------------------------------------------------------------
    # Final tiny hill‑climbing sweep (last few seconds)
    # ------------------------------------------------------------------
    deadline_final = start_time + budget_s - 0.1
    cur = np.array(best_seq, copy=True)
    cur_score = best_score
    while time.time() < deadline_final:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # move a tiny amount (up to 1 % of source)
        max_move = cur[i] * 0.01
        if max_move <= 0:
            continue
        delta = np.random.rand() * max_move
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_to_simplex(cand, target_sum)
        sc = _evaluate(cand.tolist())
        if sc < cur_score:
            cur = cand
            cur_score = sc
            if sc < best_score:
                best_score = sc
                best_seq = cand.tolist()
                print(f"[final hill {iter_refine:06d}] improved to {best_score:.6f}")

    print(f"[final] best score found: {best_score:.6f}")
    return best_seq
