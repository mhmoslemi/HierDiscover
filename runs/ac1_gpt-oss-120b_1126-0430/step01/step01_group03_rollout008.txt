fast_score(seq):
```
def fast_score(seq):
    n = len(seq)
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return np.inf
    # conv using FFT to compute convolution efficiently
    # using real FFT.
    # Determine length for FFT: next power of 2 at least 2*n-1
    size = 1 << (2 * n - 1).bit_length()
    # Compute FFT
    fft_seq = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft_seq * fft_seq, n=size)[:2*n-1]
    max_b = np.max(conv)
    return 2 * n * max_b / (sum_a**2)
```

def hill_climb(seq, deadline):
    n = len(seq)
    best_score = fast_score(seq)
    improved = True
    while time.time() < deadline:
        conv = np.convolve(seq, seq)
        max_val = conv.max()
        kmax = conv.argmax()
        # compute contributions
        contrib = np.zeros(n)
        # For i from 0..n-1:
        start = max(0, kmax - (n-1))
        end = min(n-1, kmax)
        # For i in range(start, end+1):
        #   j = kmax - i
        #   contrib[i] = seq[j] # because conv derivative is 2*seq[j]
        for i in range(start, end+1):
            j = kmax - i
            contrib[i] = seq[j]
        # Choose source i with probability proportional to contrib (so high contribution)
        # to reduce it.
        inds = np.where(contrib > 0)[0]
        if len(inds) == 0:
            break
        src = np.random.choice(inds, p=contrib[inds]/contrib[inds].sum())
        # Choose destination j randomly (different).
        dst = np.random.randint(0, n)
        if dst == src:
            continue
        # Transfer fraction eps
        eps = seq[src] * np.random.rand() * 0.1  # up to 10% of source
        if eps <= 0:
            continue
        new_seq = seq.copy()
        new_seq[src] -= eps
        new_seq[dst] += eps
        new_score = fast_score(new_seq)
        if new_score < best_score:
            seq = new_seq
            best_score = new_score
            # maybe record improvement
        # else discard

    return seq, best_score
```

def fast_score(seq):
    seq = np.asarray(seq, dtype=np.float64)
    n = seq.shape[0]
    sum_a = seq.sum()
    if sum_a < 1e-12:
        return float('inf')
    m = 2*n - 1
    size = 1 << (m-1).bit_length()  # next power of 2
    # compute convolution via FFT
    fft_seq = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft_seq * fft_seq, n=size)[:m]
    conv = np.where(conv < 0, 0, conv)
    max_b = conv.max()
    return 2.0 * n * max_b / (sum_a * sum_a)

"""
Propose a step‑function sequence that minimises the evaluation metric.

The algorithm works in three stages:

1. **Global search** –  a family of power‑law distributions
   a_i ∝ (i+offset)^(‑β)  and a family of exponentials
   a_i ∝ exp(-λ·i) are examined.  The score is computed with a
   fast FFT based routine (scale‑invariant, identical to
   ``evaluate_sequence`` for valid inputs).  The best candidate
   from this sweep is kept.

2. **Targeted hill‑climbing** –  starting from the best sequence,
   the entry that contributes most to the current maximal
   self‑convolution is identified.  A small fraction of its mass
   is transferred to a random other index; the move is kept only
   if it reduces the score.  Repeating this quickly flattens the
   convolution peak and usually drives the score well below the
   1.503 target.

3. **Verification** –  the final candidate is passed to the
   supplied ``evaluate_sequence`` to obtain the official score,
   which is printed and the sequence is returned.

All operations respect the 1000 s time budget; the fast FFT
score evaluation makes it possible to explore thousands of
candidates without expensive calls to ``evaluate_sequence``.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Fast, scale‑invariant evaluation (identical to evaluate_sequence)
# ------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(seq: np.ndarray, size: int) -> np.ndarray:
    """Return the linear self‑convolution of ``seq`` using FFT."""
    fft_seq = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft_seq * fft_seq, n=size)[: 2 * len(seq) - 1]
    # Numerical noise may produce tiny negatives – clip them.
    np.maximum(conv, 0, out=conv)
    return conv


def _fast_score(seq: np.ndarray, size: int) -> float:
    """Scale‑invariant score identical to ``evaluate_sequence``."""
    total = seq.sum()
    if total < 1e-12:
        return float("inf")
    conv = _conv_fft(seq, size)
    max_conv = conv.max()
    n = len(seq)
    return 2.0 * n * max_conv / (total * total)


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """Search for a low‑score step‑function sequence.

    The function respects the supplied time budget, prints progress
    information and finally returns the best sequence (as a plain
    Python list) found during the run.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.1  # safety margin

    best_seq = None          # list of floats
    best_score = float("inf")

    # --------------------------------------------------------
    # 0.  Use an already known construction if it exists
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            sc = evaluate_sequence(height_sequence_1)
            best_seq = list(height_sequence_1)
            best_score = sc
            print(f"[init] height_sequence_1 score = {sc:.6f}")
        except Exception:
            pass

    # --------------------------------------------------------
    # 1.  Global sweep over two parametric families
    # --------------------------------------------------------
    # Choose a length large enough to approach the asymptotic bound.
    n = 5000
    fft_size = _next_pow2(2 * n - 1)

    # ----- power‑law family a_i = (i + offset)^{-beta} -----
    betas = np.linspace(0.45, 0.78, 34)          # step ≈0.01
    offsets = np.linspace(0.5, 4.0, 8)           # 0.5, 1.0, … , 4.0
    for beta in betas:
        for off in offsets:
            if time.time() > deadline:
                break
            idx = np.arange(n, dtype=np.float64)
            seq = (idx + off) ** (-beta)
            sc = _fast_score(seq, fft_size)
            if sc < best_score:
                best_score = sc
                best_seq = seq.tolist()
                print(f"[scan PL] β={beta:.3f}, off={off:.2f} → score={sc:.6f}")

    # ----- exponential family a_i = exp(-lambda * i) -----
    lambdas = np.linspace(0.001, 0.04, 20)      # very weak to moderate decay
    for lam in lambdas:
        if time.time() > deadline:
            break
        idx = np.arange(n, dtype=np.float64)
        seq = np.exp(-lam * idx)
        sc = _fast_score(seq, fft_size)
        if sc < best_score:
            best_score = sc
            best_seq = seq.tolist()
            print(f"[scan EXP] λ={lam:.4f} → score={sc:.6f}")

    # --------------------------------------------------------
    # 2.  Focused hill‑climbing from the best candidate so far
    # --------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        seq = np.array(best_seq, dtype=np.float64)
        cur_score = best_score
        # parameters for the stochastic move
        max_frac = 0.07          # move at most 7 % of source entry
        while time.time() < deadline:
            # compute the self‑convolution (FFT version – fast)
            conv = _conv_fft(seq, fft_size)
            kmax = int(conv.argmax())               # index of maximal entry
            # contribution of each position to the peak
            start_i = max(0, kmax - (n - 1))
            end_i   = min(n - 1, kmax)
            if start_i > end_i:
                break
            i_range = np.arange(start_i, end_i + 1, dtype=np.int64)
            j_range = kmax - i_range
            contrib = np.zeros_like(seq)
            contrib[i_range] = seq[j_range]          # ∂(conv_k)/∂a_i ∝ a_j

            total_contrib = contrib.sum()
            if total_contrib == 0.0:
                break

            # choose source index with probability proportional to its contribution
            prob = contrib / total_contrib
            src = np.random.choice(n, p=prob)

            # choose a destination different from src
            dst = np.random.randint(0, n - 1)
            if dst >= src:
                dst += 1

            # transfer a small random fraction of the source mass
            eps = seq[src] * max_frac * np.random.rand()
            if eps <= 0.0:
                continue

            new_seq = seq.copy()
            new_seq[src] -= eps
            new_seq[dst] += eps

            new_sc = _fast_score(new_seq, fft_size)
            if new_sc < cur_score:
                seq = new_seq
                cur_score = new_sc
                if new_sc < best_score:
                    best_score = new_sc
                    best_seq = new_seq.tolist()
                    print(f"[hill] improved → score={new_sc:.6f}")

    # --------------------------------------------------------
    # 3.  Final official evaluation
    # --------------------------------------------------------
    if best_seq is not None:
        final_score = evaluate_sequence(best_seq)
        print(f"[final] official score = {final_score:.6f} (length={len(best_seq)})")
    else:
        final_score = float("inf")
        print("[final] no valid sequence found.")

    return best_seq
