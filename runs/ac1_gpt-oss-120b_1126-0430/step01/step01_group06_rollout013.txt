def evaluate_sequence(sequence):
    sequence = [float(x) for x in sequence]
    sequence = [max(0, x) for x in sequence] # all non-negative; large numbers capped at 1000, small to zero if negative
    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)
    if sum_a < 0.01: return np.inf
    return float(2 * n * max_b / (sum_a**2))

def _exponentiated_descent(a, time_budget, start_time, best_score, best_seq):
    a = a.copy()
    t = 0
    while time.time() - start_time < time_budget - 5:
        t += 1
        conv = np.convolve(a, a)
        max_val = conv.max()
        k = int(np.argmax(conv))
        # compute gradient
        indices = np.arange(n)
        valid = (k - indices >= 0) & (k - indices < n)
        g = np.zeros_like(a)
        g[valid] = 2 * a[k - indices[valid]]
        # step size
        eta = 0.5 / math.sqrt(t)   # adjust
        # avoid overflow
        a = a * np.exp(-eta * g)
        # renormalize onto simplex
        s = a.sum()
        if s == 0:
            # reinitialize uniform
            a = np.full(n, 1.0 / n)
        else:
            a = a / s
        # evaluate
        score = evaluate_sequence(a.tolist())
        if score < best_score:
            best_score = score
            best_seq = a.tolist()
            print(f"[descent] iter={t}, score={score:.6f}")
    return best_seq, best_score

"""
Propose a non‑negative step‑function (list of heights) that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm works in three stages:

1. **Exploratory sampling** – draw many (α, shift) pairs and build the
   normalised power‑law sequence a_i ∝ (i+shift)^(‑α).  The best of these
   is used as a warm start.

2. **Projected (sub‑)gradient descent** – the objective is the maximum
   entry of the self‑convolution, which is a piecewise‑linear function of
   the vector a.  For the active index k* we have
   ∂max(conv)/∂a_i = 2·a_{k*‑i} (if the index exists).  We move in the
   opposite direction, project the result onto the probability simplex
   (non‑negative entries with sum = 1) and accept the move only if the
   objective improves.  A simple back‑tracking line search adapts the
   step size.

3. **Final hill‑climbing** – a cheap random‑swap neighbourhood is used
   to squeeze any remaining improvement.

All stages respect the overall time budget (default ≈ 950 s).  The
function returns the best sequence found; if nothing beats the provided
fallback `height_sequence_1` the fallback is returned unchanged.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the simplex {x ≥ 0, sum(x)=1}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Efficient O(n log n) projection onto the probability simplex."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero – return uniform vector
        return np.full(n, 1.0 / n, dtype=v.dtype)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: generate a normalised power‑law vector
# ----------------------------------------------------------------------
def _powerlaw(alpha: float, shift: float, n: int) -> np.ndarray:
    """Return a normalised vector a_i ∝ (i+shift)^(-alpha)."""
    idx = np.arange(n, dtype=float)
    raw = np.power(idx + shift, -alpha)
    total = raw.sum()
    if total == 0:
        return np.full(n, 1.0 / n, dtype=float)
    return raw / total


# ----------------------------------------------------------------------
# Helper: evaluate the objective without calling the expensive wrapper.
# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Compute 2 * n * max(conv(a,a)) / (sum(a)**2).  a is assumed to be non‑negative."""
    conv = np.convolve(a, a)
    max_conv = conv.max()
    s = a.sum()
    # the wrapper clips negatives/infinities, but we keep the same
    # domain guarantees here.
    if s < 1e-12:
        return float("inf")
    return float(2.0 * a.shape[0] * max_conv / (s * s))


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the simplex
# ----------------------------------------------------------------------
def _gradient_descent(
    a0: np.ndarray,
    n: int,
    time_budget: float,
    start_time: float,
    best_score: float,
    best_seq: list,
) -> tuple[list, float]:
    """
    Perform a simple projected sub‑gradient descent on a starting vector.
    Returns the best sequence (as a list) found during this run and its
    score.
    """
    a = a0.copy()
    cur_score = _score(a)

    # initial step size – fairly aggressive; it will be adapted.
    step = 0.5
    min_step = 1e-7
    it = 0
    # tolerance for detecting “active” constraints
    eps_active = 1e-12

    while True:
        it += 1
        # time check
        if time.time() - start_time > time_budget - 2.0:
            break

        # compute convolution and the active set (indices attaining the max)
        conv = np.convolve(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - eps_active)[0]

        # build the sub‑gradient (average over all active constraints)
        grad = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            valid = (k - idx >= 0) & (k - idx < n)
            grad[valid] += 2.0 * a[k - idx[valid]]
        if len(active) > 1:
            grad /= len(active)

        # trial step
        trial = a - step * grad
        trial = _project_to_simplex(trial)

        trial_score = _score(trial)

        if trial_score < cur_score:
            # improvement – keep the move and slightly increase the step
            a = trial
            cur_score = trial_score
            if trial_score < best_score:
                best_score = trial_score
                best_seq = trial.tolist()
                print(
                    f"[gd] iter={it:5d}, step={step:.5f}, score={trial_score:.6f}"
                )
            step = min(step * 1.1, 1.0)
        else:
            # no improvement – shrink step
            step = max(step * 0.5, min_step)
            if step <= min_step:
                # step becomes too small → stop this descent run
                break

    return best_seq, best_score


# ----------------------------------------------------------------------
# Simple random hill‑climbing used as a final polishing step
# ----------------------------------------------------------------------
def _hill_climb(
    a0: np.ndarray,
    n: int,
    time_budget: float,
    start_time: float,
    best_score: float,
    best_seq: list,
) -> tuple[list, float]:
    """Move small amounts of mass between two random positions."""
    a = a0.copy()
    cur_score = _score(a)
    max_iter = 20000
    i_iter = 0
    while True:
        i_iter += 1
        if time.time() - start_time > time_budget - 2.0 or i_iter > max_iter:
            break

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        # move up to 1 % of mass from i to j
        step = a[i] * (np.random.rand() * 0.01)
        if step <= 0.0:
            continue
        if a[j] + step > 1.0:  # safety – never needed but cheap to check
            continue

        new = a.copy()
        new[i] -= step
        new[j] += step
        new = _project_to_simplex(new)  # eliminate any rounding drift

        new_score = _score(new)
        if new_score < cur_score:
            a = new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = new.tolist()
                print(
                    f"[hill] iter={i_iter:5d}, score={new_score:.6f}"
                )
    return best_seq, best_score
