def simplex_projection(v):
    # v is 1-D np.ndarray
    n = len(v)
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1] - 1  # zero-index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

    # We'll try lengths sequentially; if time left.

    for n in lengths:

        # Select a set of betas to initialize
        betas = np.linspace(0.45,0.75,7)  # 7 values
        for beta in betas:
            if time.time() > deadline: break

            max_b = conv.max()
            score = 2 * n * max_b
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"[init] beta {beta:.3f}, score {score:.6f}")

            # Gradient descent
            max_iters = 3000
            step0 = 1.0   # initial step size in terms of gradient amplitude scaling; we can tune
            # We'll adapt step size based on iteration
            a = seq.copy()
            for it in range(max_iters):
                if time.time() > deadline: break

                # compute convolution via FFT
                fft_a = np.fft.fft(a, 2*n)
                conv_full = np.fft.ifft(fft_a * fft_a).real
                # compute max idx
                k = int(np.argmax(conv_full))
                max_b = conv_full[k]
                # compute gradient
                j = k - np.arange(n)
                mask = (j >= 0) & (j < n)
                grad = np.zeros_like(a)
                grad[mask] = 2 * a[j[mask]]
                # step schedule
                step = step0 / math.sqrt(it + 1)
                a -= step * grad
                # enforce non-negative
                a = np.maximum(a, 0)
                # renormalize
                total = a.sum()
                if total == 0:
                    a = np.full(n, 1.0/n)
                    total = 1.0
                a /= total

                # compute score occasionally

                    if score < best_score:

                        print(f"[gd] n={n} beta={beta:.3f} iter={it} score={score:.6f}")

"""
Propose a high‑quality step‑function sequence.

Strategy
--------
1.  Use several lengths (up to ~20000 elements).  A longer sequence allows the
    self‑convolution peak to shrink proportionally to 1/n, which makes the
    evaluation 2·n·max(b) approach the theoretical limit (≈1.35).

2.  Initialise each length with a simple power‑law distribution
    a_i ∝ (i+1)^‑β   (β around 0.55‑0.65 works well).

3.  Run a projected‑gradient descent that directly minimises the
    maximal convolution value.
    * The convolution is computed with a real FFT (fast O(n log n)).
    * The gradient of the maximal entry b_k is
          ∂b_k/∂a_i = 2·a_{k‑i}   (zero when the index is out of range).
    * After a gradient step we clamp to ≥0 and re‑normalise to sum‑one,
      i.e. we stay on the probability simplex.

4.  Periodically evaluate the true score  (2·n·max(b)) and keep the best
    sequence seen so far.

5.  A very light stochastic “hill‑climb’’ (tiny mass transfers) is applied
    after the gradient phase to polish the result.

The whole routine respects the 1000 s budget, prints progress, and finally
returns the best list of heights it discovered.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Return a power‑of‑two length suitable for an FFT of size 2·n‑1."""
    m = 2 * n - 1
    # next power of two
    return 1 << (m - 1).bit_length()


def _conv_max_and_index(a: np.ndarray, fft_len: int) -> tuple[float, int]:
    """
    Compute the full convolution a*a (using a real FFT) and return
    the maximal value together with its index (restricted to the
    meaningful part 0 … 2·n‑2).
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * len(a) - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx


def _gradient_of_max_convolution(a: np.ndarray, max_idx: int) -> np.ndarray:
    """
    Return the sub‑gradient of the maximal convolution entry.
    For a given index k (max_idx) the gradient is
        g_i = 2 * a_{k-i}   (zero when out of range).
    """
    n = a.shape[0]
    i = np.arange(n)
    j = max_idx - i
    mask = (j >= 0) & (j < n)
    g = np.zeros_like(a)
    g[mask] = 2.0 * a[j[mask]]
    return g


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, sum x = 1}.  Used only as a safety net; the main loop
    normalises explicitly.
    """
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries zero – return uniform distribution
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function sequence that minimises
    `evaluate_sequence`.  The function runs for at most ``budget_s`` seconds
    and returns the best sequence found (as a plain Python list).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2  # tiny safety margin

    best_seq = None          # will hold a NumPy array
    best_score = float("inf")

    # --------------------------------------------------------------
    # 0.  If a previous construction exists, keep it as a starting point.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            sc = evaluate_sequence(height_sequence_1)
            best_seq = np.asarray(height_sequence_1, dtype=np.float64)
            best_score = float(sc)
            print(f"[init] previous best score = {best_score:.6f}")
        except Exception:
            pass

    # --------------------------------------------------------------
    # 1.  Lengths to explore (larger n → lower asymptotic bound).
    # --------------------------------------------------------------
    lengths = [2000, 4000, 8000, 12000, 20000]  # stop early if out of time
    # ------------------------------------------------------------------
    # 2.  Run the optimisation for each length.
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n[run] trying length n = {n}")

        fft_len = _fft_len_for_n(n)

        # a few different power‑law seeds
        betas = np.linspace(0.55, 0.65, 5)   # 0.55,0.575,…,0.65
        for beta in betas:
            if time.time() > deadline:
                break

            idx = np.arange(1, n + 1, dtype=np.float64)
            a = idx ** (-beta)                         # unnormalised
            a /= a.sum()                               # now a is a probability vector

            # ------------------------------------------------------
            # Quick initial score
            # ------------------------------------------------------
            max_b, _ = _conv_max_and_index(a, fft_len)
            score = 2.0 * n * max_b
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[seed] beta={beta:.3f}  init_score={score:.6f}")

            # ------------------------------------------------------
            # Gradient descent on the simplex
            # ------------------------------------------------------
            max_iters = 3000
            step0 = 0.5                                 # initial step size
            for it in range(max_iters):
                if time.time() > deadline:
                    break

                # convolution and index of the current maximum
                max_b, k = _conv_max_and_index(a, fft_len)

                # gradient of that entry
                grad = _gradient_of_max_convolution(a, k)

                # decreasing step size (simple schedule)
                step = step0 / math.sqrt(it + 1.0)

                # projected gradient step
                a_new = a - step * grad
                a_new = np.maximum(a_new, 0.0)
                total = a_new.sum()
                if total == 0.0:
                    a_new = np.full(n, 1.0 / n)
                else:
                    a_new /= total                     # re‑normalise to sum = 1

                a = a_new

                # periodic evaluation (every 50 iterations)
                if (it % 50) == 0:
                    max_b, _ = _conv_max_and_index(a, fft_len)
                    score = 2.0 * n * max_b
                    if score < best_score:
                        best_score = score
                        best_seq = a.copy()
                        print(f"[gd] n={n} β={beta:.3f} it={it:04d} score={score:.6f}")

            # ------------------------------------------------------
            # Light random “hill‑climb’’ fine‑tuning
            # ------------------------------------------------------
            while time.time() < deadline:
                # pick two distinct positions
                i, j = np.random.randint(0, n, size=2)
                if i == j:
                    continue
                # move a tiny fraction of mass from i to j
                eps = a[i] * np.random.rand() * 0.02  # ≤2 % of a[i]
                if eps <= 0.0:
                    continue
                a2 = a.copy()
                a2[i] -= eps
                a2[j] += eps
                a2 = np.maximum(a2, 0.0)
                a2 /= a2.sum()

                max_b, _ = _conv_max_and_index(a2, fft_len)
                score2 = 2.0 * n * max_b
                if score2 < best_score:
                    best_score = score2
                    best_seq = a2.copy()
                    a = a2
                    print(f"[hill] n={n} β={beta:.3f} new_best={score2:.6f}")

                # stop the hill‑climb after a few thousand tries
                if np.random.rand() < 0.001:
                    break

    # ------------------------------------------------------------------
    # 3.  Return the best sequence (as a plain Python list).
    # ------------------------------------------------------------------
    if best_seq is None:
        # should never happen, but return a trivial uniform sequence
        best_seq = np.full(1000, 1.0 / 1000)

    print(f"\n[final] best_score = {best_score:.6f}, length = {len(best_seq)}")
    return best_seq.tolist()
