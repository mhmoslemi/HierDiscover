"""
Hybrid search for a low‑value step‑function.

The score (with the sum of the coefficients normalised to 1) is  

    S(a) = 2·n·max(conv(a,a))

where conv(a,a) is the full discrete convolution.
The algorithm consists of three stages

1)  A fast global search over a family of power‑law / mixture families.
    Candidates are generated, normalised to sum‑1 and scored with a cheap
    FFT‑based convolution.  The best candidate is kept.

2)  A projected sub‑gradient descent on the raw vector.
    For a vector a the sub‑gradient of  max(conv(a,a)) is
    g_i = Σ_{k∈K} 2·a_{k-i}   (K = indices where the maximum is attained).
    After each sub‑gradient step the vector is projected back onto the
    probability simplex (non‑negative, sum = 1) with the standard O(n) simplex
    projection.  The step size decreases as η_t = η₀/√t.

3)  A short stochastic hill‑climbing phase (tiny mass transfers) to
    polish the solution.

All heavy‑weight convolution work is performed with NumPy's FFT
(rfft/irfft) which is O(n log n) and therefore allows many thousands of
iterations within the 1000 s time budget.  The score is evaluated
internally with the fast formula (the official `evaluate_sequence`
is only called at the very end for the final answer).

The function returns the best sequence found, ready to be handed to the
competition harness.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x≥0, sum(x)=1}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    # sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero vector – return the uniform distribution
        return np.full(n, 1.0 / n, dtype=v.dtype)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast full convolution using FFT (real input)
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2*len(a)-1) via FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())   # next power of two
    # real FFT of the same size for both copies
    A = np.fft.rfft(a, n=size)
    C = A * A
    conv = np.fft.irfft(C, n=size)[: 2 * n - 1]
    # tiny negative values caused by round‑off are set to zero
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Fast score for a sequence that already sums to 1
# ----------------------------------------------------------------------
def _score_from_vec(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) assuming sum(a)=1."""
    conv = _full_convolution_fft(a)
    return 2.0 * a.shape[0] * np.max(conv)


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution functional
# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Compute a sub‑gradient of max(conv(a,a)) at a.
    `conv` must be the full convolution a*a.
    """
    n = a.shape[0]
    M = np.max(conv)
    # indices where the maximum is attained (within a tiny tolerance)
    idx_max = np.where(np.isclose(conv, M, atol=1e-14, rtol=0.0))[0]
    grad = np.zeros_like(a)

    for k in idx_max:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i = np.arange(i_start, i_end + 1)
        j = k - i
        grad[i] += 2.0 * a[j]          # derivative of (a*a)_k w.r.t. a_i
    return grad

def propose_candidate(seed: int = 123, budget_s: float = 950, **kwargs):
    """
    Produce a non‑negative step‑function (list of heights) that minimises
    the provided ``evaluate_sequence`` value.

    The method mixes a cheap global search over parametrised families
    with a long projected sub‑gradient descent on the raw vector.
    """
    np.random.seed(seed)
    start = time.time()

    # --------------------------------------------------------------
    # Determine the problem size (length of the fallback solution)
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # provided by the environment
    except Exception:
        n = 2244                            # fallback length from the description

    # --------------------------------------------------------------
    # Baseline (fallback) – the global variable is always present
    # --------------------------------------------------------------
    try:
        best_seq = list(height_sequence_1)
        best_score = evaluate_sequence(best_seq)
    except Exception:
        best_seq = [1.0 / n] * n
        best_score = float("inf")

    print(f"[init] n={n}, baseline score={best_score:.6f}")

    idx = np.arange(n, dtype=float)          # 0 … n‑1 as float

    # --------------------------------------------------------------
    # 1) Global search over simple analytic families
    # --------------------------------------------------------------
    # ---- power‑law family ------------------------------------------------
    alphas = np.linspace(0.45, 0.75, 61)      # 0.45 … 0.75, step ≈0.005
    shifts = np.linspace(1.0, 2.0, 5)        # 1.0, 1.25, …, 2.0

    for alpha in alphas:
        if time.time() - start > budget_s * 0.20:
            break
        for shift in shifts:
            raw = np.power(idx + shift, -alpha)
            a = raw / raw.sum()
            sc = _score_from_vec(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.tolist()
                print(f"[grid] α={alpha:.4f}, shift={shift:.2f}, score={sc:.6f}")

    # ---- random mixtures of two power‑law components --------------------
    # Use a modest number of random samples; each evaluation is cheap.
    mix_trials = 1200
    for _ in range(mix_trials):
        if time.time() - start > budget_s * 0.30:
            break
        w = np.random.rand()
        alpha1 = np.random.uniform(0.4, 0.8)
        alpha2 = np.random.uniform(0.4, 0.8)
        shift1 = np.random.uniform(0.5, 3.0)
        shift2 = np.random.uniform(0.5, 3.0)

        raw = w * np.power(idx + shift1, -alpha1) + (1.0 - w) * np.power(idx + shift2, -alpha2)
        # a small jitter helps to escape degenerate ties
        raw += np.random.rand(n) * 1e-6
        a = raw / raw.sum()
        sc = _score_from_vec(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.tolist()
            print(f"[mix] w={w:.3f}, α1={alpha1:.3f}, α2={alpha2:.3f}, "
                  f"s1={shift1:.3f}, s2={shift2:.3f}, score={sc:.6f}")

    print(f"[global] best after parametrised search: score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Projected sub‑gradient descent on the full vector
    # --------------------------------------------------------------
    cur = np.array(best_seq, dtype=float)   # start point for the descent
    cur_score = best_score
    max_iter = 10 ** 7                       # safety bound, real stop is time
    eta0 = 0.25                               # initial step size multiplier
    it = 0

    while time.time() - start < budget_s * 0.90 and it < max_iter:
        it += 1

        # (a) compute full convolution and its maximum
        conv = _full_convolution_fft(cur)
        M = np.max(conv)
        cur_score = 2.0 * n * M

        # (b) sub‑gradient of the max‑convolution
        grad = _subgradient_max_conv(cur, conv)

        # (c) decreasing step size
        eta = eta0 / np.sqrt(it)

        # (d) gradient step + projection onto the simplex
        cand = cur - eta * grad
        cand = _project_to_simplex(cand)

        # (e) evaluate the candidate (fast score, no call to evaluate_sequence)
        conv_cand = _full_convolution_fft(cand)
        cand_score = 2.0 * n * np.max(conv_cand)

        # (f) accept only if we improve
        if cand_score < cur_score - 1e-12:          # strict improvement
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.tolist()
                print(f"[sgd] iter={it}, score={cand_score:.6f}")

        # occasional tiny random perturbation to escape plateaus
        if it % 5000 == 0:
            # pick two distinct indices and move a tiny share of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = cur[i] * (np.random.rand() * 0.005)   # ≤0.5 % of a_i
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_to_simplex(tmp)
                # quick score
                tmp_score = _score_from_vec(tmp)
                if tmp_score < cur_score:
                    cur = tmp
                    cur_score = tmp_score
                    if tmp_score < best_score:
                        best_score = tmp_score
                        best_seq = tmp.tolist()
                        print(f"[sgd‑perturb] iter={it}, score={tmp_score:.6f}")

    print(f"[sgd] finished after {it} iterations, best score={best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Final short stochastic hill‑climbing (tiny moves)
    # --------------------------------------------------------------
    cur = np.array(best_seq, dtype=float)
    cur_score = best_score
    hill_iters = 30000
    for _ in range(hill_iters):
        if time.time() - start > budget_s - 5:
            break
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # move up to 2 % of the source mass
        step = cur[i] * (np.random.rand() * 0.02)
        if step <= 0 or cur[j] - step < 0:
            continue
        cand = cur.copy()
        cand[i] -= step
        cand[j] += step
        cand = _project_to_simplex(cand)
        cand_score = _score_from_vec(cand)
        if cand_score < cur_score - 1e-13:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.tolist()
                print(f"[refine] score={cand_score:.6f}")

    # --------------------------------------------------------------
    # Final verification with the official evaluator (slow but harmless)
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq)
        print(f"[final] official evaluation score={final_score:.6f}")
    except Exception:
        final_score = float("inf")
        print("[final] official evaluation failed – returning fallback")

    return best_seq
