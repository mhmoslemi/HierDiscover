   i = np.arange(n)
   weights = ( (i+1)*(n-i) )**(-beta)
   # maybe add epsilon to avoid zero.
   a = weights
   a /= a.sum()

   i = np.arange(n)
   weights = (i+1)**(-beta) + (n-i)**(-beta)
   a = weights
   a /= a.sum()

def hill_climb(initial, budget):
   # start with a copy.
   a = initial.copy()
   best_a = a.copy()
   best_obj = compute_objective(a)
   n = len(a)
   start = time.time()
   # temperature schedule
   T0 = 0.05
   iter = 0
   while time.time() < budget:
       iter += 1
       # compute conv and max index
       conv = np.convolve(a,a)
       max_val = conv.max()
       ks = np.flatnonzero(conv == max_val)
       k = np.random.choice(ks)
       # donor set indices where both a_i>0 and partner inside range
       donor_indices = np.arange(max(0, k-(n-1)), min(n-1, k)+1)

       donor_candidates = donor_indices[a[donor_indices] > 1e-12]
       if donor_candidates.size == 0:

       i = np.random.choice(donor_candidates)

       while True:

           if j != i:
               break
       # transfer delta
       max_delta = a[i]
       # random fraction up to 0.3
       delta = max_delta * np.random.rand() * 0.3
       a_new = a.copy()
       a_new[i] -= delta
       a_new[j] += delta
       # ensure non-neg and project onto simplex
       a_new = np.clip(a_new, 0, None)

           g = np.zeros_like(a)

def solve_lp_direction(a, M):
    n = len(a)
    rows = 2*n - 1
    # Create data arrays for CSR
    data = []
    rows_idx = []
    col_idx = []
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(n-1, k)
        # for i in [i_start,i_end], col index = i, value = a[k-i]
        # weight: a_{k-i}
        for i in range(i_start, i_end+1):
            rows_idx.append(k)
            col_idx.append(i)
            data.append(a[k - i])
    A = sp.csr_matrix((data, (rows_idx, col_idx)), shape=(rows, n))
    # objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    # bounds: g_i >=0
    bounds = [(0, None)] * n
    # constraints: A @ g <= M
    res = opt.linprog(c, A_ub=A, b_ub=np.full(rows, M), bounds=bounds, method='highs', options={'presolve': True})
    if not res.success:
        return None
    return res.x

"""
Propose a step‑function (non‑negative sequence) with a low value of
evaluate_sequence().  The algorithm works in three stages:

1️⃣  **Smart initialization** – generate a few hundred candidates from
    simple parametric families (U‑shaped “edge‑heavy” distributions)
    and keep the best one.

2️⃣  **Targeted local search** – hill‑climbing with moves that explicitly
    remove mass from indices that take part in the current maximal
    convolution entry and add it elsewhere.  A small sub‑gradient step
    (negative gradient of the peak) is performed regularly.

3️⃣  **Linear‑programming refinement** – solve a small LP
    (maximise Σg  subject to a·g ≤ current peak) to obtain a
    complementary distribution g, blend it with the current best
    and continue the local search.

All operations respect the simplex constraint Σa = 1, a ≥ 0.
The routine respects the 1000 s budget, prints progress and finally
returns the best sequence found.
"""

import time
import math
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# -------------------------------------------------------------
# Helper: projection onto the probability simplex (Duchi et al.)
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x >= 0, sum(x) = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# -------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Fast 2·n·max(conv(a,a)) for a simplex‑normalised vector."""
    conv = np.convolve(a, a)
    return 2.0 * a.size * conv.max()


# -------------------------------------------------------------
def _max_conv_info(a: np.ndarray):
    """Return (peak value, list of indices where the peak occurs)."""
    conv = np.convolve(a, a)
    m = conv.max()
    idx = np.flatnonzero(conv == m)
    return m, idx, conv


# -------------------------------------------------------------
# Parametric families – “U‑shaped” edge‑heavy distributions.
def _shape_u(n: int, beta: float) -> np.ndarray:
    """a_i ∝ ((i+1)*(n-i))^{-beta}  (symmetrised, edges heavy)."""
    i = np.arange(n, dtype=float)
    w = ((i + 1) * (n - i)) ** (-beta)
    w = np.maximum(w, 1e-30)       # avoid exact zeros
    a = w / w.sum()
    return a


def _shape_u2(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-beta} + (n-i)^{-beta}."""
    i = np.arange(n, dtype=float)
    w = (i + 1) ** (-beta) + (n - i) ** (-beta)
    w = np.maximum(w, 1e-30)
    a = w / w.sum()
    return a


# -------------------------------------------------------------
def _random_parametric_candidate(n: int) -> np.ndarray:
    """Draw a random candidate from one of the two families."""
    if np.random.rand() < 0.5:
        beta = np.random.uniform(0.05, 3.0)
        a = _shape_u(n, beta)
    else:
        beta = np.random.uniform(0.05, 3.0)
        a = _shape_u2(n, beta)
    # add a tiny random perturbation and re‑project – helps escape plateaus
    if np.random.rand() < 0.1:
        a += np.random.randn(n) * 1e-3
        a = np.clip(a, 0, None)
        a = _project_to_simplex(a)
    return a


# -------------------------------------------------------------
def _targeted_hill_climb(a0: np.ndarray,
                         budget_end: float,
                         max_iters: int = 15000) -> tuple[np.ndarray, float]:
    """
    Hill‑climbing that preferentially moves mass away from the
    indices contributing to the current peak of the convolution.
    Returns the best sequence seen and its objective.
    """
    a = a0.copy()
    n = a.size
    best_a = a.copy()
    best_score = _objective(a)

    iter_no_imp = 0
    T0 = 0.03                     # temperature for occasional uphill moves

    for it in range(1, max_iters + 1):
        if time.time() >= budget_end:
            break

        # -------------------------------------------------
        # 1. locate the peak(s) of the convolution
        peak_val, peak_idxs, conv = _max_conv_info(a)
        k = np.random.choice(peak_idxs)               # pick one peak index

        # donor region = all positions that actually participate in this peak
        donor_start = max(0, k - (n - 1))
        donor_end = min(n - 1, k)
        donor_candidates = np.arange(donor_start, donor_end + 1)
        donor_candidates = donor_candidates[a[donor_candidates] > 1e-12]

        if donor_candidates.size == 0:
            # fall back to a completely random move
            i = np.random.randint(n)
            j = np.random.randint(n)
            while j == i:
                j = np.random.randint(n)
        else:
            i = np.random.choice(donor_candidates)
            # choose a recipient preferably *outside* the donor region
            while True:
                j = np.random.randint(n)
                if j != i and not (donor_start <= j <= donor_end):
                    break

        # -----------------------------------------------------------------
        # transfer a modest fraction of the donor's mass
        max_delta = a[i]
        if max_delta <= 0:
            continue
        delta = max_delta * np.random.rand() * 0.25      # up‑to 25 %
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.clip(a_new, 0, None)
        a_new = _project_to_simplex(a_new)

        new_score = _objective(a_new)

        # simple Metropolis acceptance
        T = T0 * (1.0 - it / max_iters)
        if new_score < best_score or np.random.rand() < math.exp((best_score - new_score) / max(T, 1e-12)):
            a = a_new
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
                iter_no_imp = 0
                print(f"[{time.time():.1f}] New global best O={best_score:.6f}  (iter={it})")
            else:
                iter_no_imp += 1
        else:
            iter_no_imp += 1

        # -----------------------------------------------------------------
        # occasional sub‑gradient step on the *active* peak(s)
        if it % 11 == 0:
            g = np.zeros_like(a)
            for kk in peak_idxs:
                start = max(0, kk - (n - 1))
                end = min(n - 1, kk)
                idx = np.arange(start, end + 1, dtype=int)
                partner = kk - idx
                g[idx] += 2.0 * a[partner]
            g /= len(peak_idxs)          # average if several peaks
            eta = 0.02 * (1.0 - it / max_iters)   # decreasing step size
            a = a - eta * g
            a = np.clip(a, 0, None)
            a = _project_to_simplex(a)
            cur_score = _objective(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] Sub‑grad new best O={best_score:.6f}  (iter={it})")

        # -----------------------------------------------------------------
        # restart if we are stuck for a long time
        if iter_no_imp >= 4000:
            break

    return best_a, best_score


# -------------------------------------------------------------
def _lp_refinement(a: np.ndarray, max_peeks: int = 3) -> np.ndarray:
    """
    Perform a small number of LP‑based refinements.
    For a given sequence a (simplex‑normalised) we solve
        maximise   Σ g_j
        subject to g ≥ 0  and  (a * g)[k] ≤ M  (for all k)
    where M is the current maximal convolution entry.
    The resulting g is scaled back to sum‑1 and blended with a.
    """
    n = a.size
    rows = 2 * n - 1

    # Build the constraint matrix A (rows × n)  :  A[k, j] = a_{k-j}  when valid
    data = []
    rows_idx = []
    cols_idx = []
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows_idx.append(k)
            cols_idx.append(i)
            data.append(a[k - i])
    A = sp.csr_matrix((data, (rows_idx, cols_idx)), shape=(rows, n))

    best_a = a.copy()
    best_obj = _objective(a)

    for peek in range(max_peeks):
        # current peak level
        M = np.convolve(best_a, best_a).max()

        # linear programme: minimise -Σg  ⇔ maximise Σg
        c = -np.ones(n)
        b_ub = np.full(rows, M)
        bounds = [(0, None)] * n

        res = opt.linprog(c, A_ub=A, b_ub=b_ub,
                          bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 60})
        if not res.success:
            # LP failed – give up further refinements
            break
        g = res.x
        if g.sum() == 0:
            break
        # scale g back to a simplex point
        g = g / g.sum()

        # blend: use a modest mixing factor, then re‑project
        t = 0.4
        new_seq = (1 - t) * best_a + t * g
        new_seq = _project_to_simplex(new_seq)

        new_obj = _objective(new_seq)
        if new_obj < best_obj:
            best_obj = new_obj
            best_a = new_seq
            print(f"[{time.time():.1f}] LP‑refine improved O={best_obj:.6f} (round={peek+1})")
        else:
            # if no improvement, shrink the mixing factor and try again
            t = 0.2
            new_seq = (1 - t) * best_a + t * g
            new_seq = _project_to_simplex(new_seq)
            new_obj = _objective(new_seq)
            if new_obj < best_obj:
                best_obj = new_obj
                best_a = new_seq
                print(f"[{time.time():.1f}] LP‑refine (small step) O={best_obj:.6f}")

    return best_a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises the
    supplied ``evaluate_sequence`` function.  The routine runs for up to
    ``budget_s`` seconds and returns the best list of floats found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # 0.5 s safety margin

    # -------------------------------------------------
    # Choose a length.  Prefer the length of the previously discovered
    # sequence (if present), otherwise fall back to a healthy size.
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
    else:
        n = 4000
    # make sure we have a reasonable lower bound
    n = max(2000, n)

    # -------------------------------------------------
    # Baseline (uniform) – useful for reference.
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _objective(best_seq)
    print(f"[init] uniform baseline O={best_score:.6f} (n={n})")

    # -------------------------------------------------
    # If a previous construction exists, start from it (normalised).
    if 'height_sequence_1' in globals():
        seq = np.array(height_sequence_1, dtype=float)
        seq = np.clip(seq, 0, None)
        if seq.sum() == 0:
            seq = uniform.copy()
        else:
            seq = seq / seq.sum()
        cand_score = _objective(seq)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = seq.copy()
            print(f"[init] loaded previous seq O={best_score:.6f}")

    # -------------------------------------------------
    # Phase 1 – random parametric candidates (≈ 20 % of the budget)
    phase1_deadline = start_time + 0.2 * budget_s
    while time.time() < phase1_deadline:
        a = _random_parametric_candidate(n)
        score = _objective(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[phase1] new best O={best_score:.6f}")

    # -------------------------------------------------
    # Phase 2 – targeted hill‑climbing (≈ 65 % of the budget)
    phase2_deadline = start_time + 0.85 * budget_s
    restart = 0
    while time.time() < phase2_deadline:
        restart += 1
        # start either from the global best (with a tiny perturbation) or from a fresh parametric draw
        if np.random.rand() < 0.3:
            a_start = best_seq + np.random.randn(n) * 5e-4
            a_start = np.clip(a_start, 0, None)
            a_start = _project_to_simplex(a_start)
        else:
            a_start = _random_parametric_candidate(n)

        a_opt, score_opt = _targeted_hill_climb(a_start, phase2_deadline, max_iters=12000)
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[phase2] restart {restart} yielded new global best O={best_score:.6f}")

    # -------------------------------------------------
    # Phase 3 – LP‑based refinement (remaining time)
    if time.time() < deadline:
        best_seq = _lp_refinement(best_seq, max_peeks=4)
        best_score = _objective(best_seq)
        print(f"[phase3] after LP refinement O={best_score:.6f}")

    # -------------------------------------------------
    # Final safety check with the official evaluator
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:
        print("[final] evaluate_sequence call failed:", e)

    return best_seq.tolist()
