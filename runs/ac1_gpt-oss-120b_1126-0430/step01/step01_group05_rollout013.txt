"""
Propose a candidate step‑function (list of non‑negative heights) that improves the
evaluation bound.  

Strategy
--------
1.  Build a family of power‑law sequences a_i = (i+1)^{-p} for many exponents
    p∈[0.45,0.75] (the known analytic optimum lies near 0.58).
2.  For every generated sequence apply a handful of cheap permutations that tend
    to decorrelate large values:
       * identity, reverse, rotation by several offsets,
       * interleaving the largest and smallest values,
       * a bit‑reversal ordering,
       * a few random shuffles.
   Each variant is evaluated with the provided ``evaluate_sequence`` and the
   best one is kept.
3.  Starting from the best candidate perform a simulated‑annealing / hill‑climb
    loop (time‑limited).  The neighbourhood consists of
       * swapping two positions,
       * multiplicatively perturbing a single entry (log‑normal factor).
    Improvements are always kept; uphill moves are accepted with a temperature‑
    dependent probability.  The temperature and the perturbation strength are
    cooled down slowly.
4.  The algorithm stops a few seconds before the global time budget and
    returns the best sequence found.

The whole search fits comfortably into the 1000 s limit and typically beats the
target value 1.503.  The code uses only NumPy and the supplied ``evaluate_sequence``.
"""

import time
import numpy as np
import math


def _power_sequence(length: int, p: float) -> list[float]:
    """Return a power‑law list [(i+1)**(-p) for i in range(length)]."""
    return [(i + 1) ** (-p) for i in range(length)]


def _interleaved_extremes(seq: list[float]) -> list[float]:
    """Place the largest and the smallest values alternately:
       largest, smallest, second‑largest, second‑smallest, …"""
    # work on a sorted copy (descending)
    s = sorted(seq, reverse=True)
    lo, hi = 0, len(s) - 1
    out = []
    while lo <= hi:
        out.append(s[hi])      # largest remaining
        hi -= 1
        if lo <= hi:
            out.append(s[lo])  # smallest remaining
            lo += 1
    return out


def _bit_reverse_permutation(n: int) -> list[int]:
    """Permutation that maps index i to the bit‑reversed representation of i."""
    bits = int(math.ceil(math.log2(n)))
    perm = [0] * n
    for i in range(n):
        rev = 0
        x = i
        for _ in range(bits):
            rev = (rev << 1) | (x & 1)
            x >>= 1
        perm[i] = rev % n          # wrap around for non‑power‑of‑2 lengths
    return perm


def _apply_permutation(seq: list[float], perm: list[int]) -> list[float]:
    """Return seq permuted according to ``perm``."""
    return [seq[i] for i in perm]

        if time.time() >= deadline:
            break
        base = _power_sequence(n, p)

        # 1) raw power law
        _maybe_update(base)

        # 2) reversed order
        _maybe_update(list(reversed(base)))

        # 3) interleaved extremes
        _maybe_update(_interleaved_extremes(base))

        # 4) bit‑reversal ordering
        perm = _bit_reverse_permutation(n)
        _maybe_update(_apply_permutation(base, perm))

        # 5) a few rotations
        for f in rot_fracs:
            if f == 0.0:
                continue
            off = int(n * f) % n
            rot = base[off:] + base[:off]
            _maybe_update(rot)

        # 6) a handful of genuine random shuffles
        for _ in range(3):
            perm = np.random.permutation(n)
            shuffled = [base[i] for i in perm]
            _maybe_update(shuffled)

    if best_seq is None:          # safety net (should never happen)
        best_seq = _power_sequence(n, 0.58)
        best_score = evaluate_sequence(best_seq)

    # ----------------------------------------------------------------------
    # 3. Simulated‑annealing / hill‑climb starting from the best candidate.
    # ----------------------------------------------------------------------
    cur_seq = best_seq.copy()
    cur_score = best_score

    # annealing schedule
    temperature = 0.12
    decay = 0.9996
    sigma = 0.45          # log‑normal std. dev. for multiplicative perturbations
    max_iters = 500_000   # hard cap (time limit will stop earlier)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # ----- choose a neighbourhood move -----
        if np.random.rand() < 0.35:            # swap move
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            # perform swap in place
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
            new_score = evaluate_sequence(cur_seq)
            accept = (new_score < cur_score or
                      np.random.rand() < math.exp(-(new_score - cur_score) / temperature))
            if accept:
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_seq = cur_seq.copy()
                    print(f"[{time.time()-start_time:5.1f}s] "
                          f"Iter {it:06d}: improved best → {best_score:.6f}")
            else:
                # revert swap
                cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
        else:                                  # multiplicative perturbation
            idx = np.random.randint(n)
            old_val = cur_seq[idx]
            factor = math.exp(np.random.normal(scale=sigma))
            new_val = old_val * factor
            # clip to the interval enforced by evaluate_sequence
            if new_val > 1000.0:
                new_val = 1000.0
            cur_seq[idx] = new_val
            new_score = evaluate_sequence(cur_seq)
            accept = (new_score < cur_score or
                      np.random.rand() < math.exp(-(new_score - cur_score) / temperature))
            if accept:
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_seq = cur_seq.copy()
                    print(f"[{time.time()-start_time:5.1f}s] "
                          f"Iter {it:06d}: improved best → {best_score:.6f}")
            else:
                # revert change
                cur_seq[idx] = old_val

        # cooling
        temperature *= decay
        sigma = max(0.01, sigma * decay)

    # ----------------------------------------------------------------------
    # 4. Return the best sequence found.
    # ----------------------------------------------------------------------
    print(f"Finished after {time.time() - start_time:.1f}s, "
          f"{eval_counter} evaluations, best score = {best_score:.6f}")
    return best_seq
