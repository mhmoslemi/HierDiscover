"""
Propose a step‑function (non‑negative heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm works in three phases:

1. **Broad family search** – generate many candidate shapes
   (power‑law, symmetric power‑law, Beta‑distribution,
   exponential decay and random Dirichlet).  All candidates are
   normalised to unit sum (the objective is scale‑invariant) and the
   best one is kept.

2. **Projected gradient descent** – starting from the best candidate,
   repeatedly reduce the current maximal convolution value.
   The sub‑gradient of the active (max) shift is `2 * a` reversed at the
   active index; a simple line‑search on the learning rate guarantees a
   monotone improvement.  Each step is projected onto the probability
   simplex to keep the variables non‑negative and summing to one.

3. **Targeted hill‑climbing** – a light random‑mass‑transfer local
   optimiser that preferentially moves mass from indices that
   participate in the current maximal shift to arbitrary destinations.
   This often squeezes a few extra digits.

All heavy work is done with NumPy; the helper ``project_to_simplex`` is
an O(n log n) implementation of the algorithm by Wang & Carreira‑Perpiñán.
The total running time is kept below the supplied budget (default 950 s)
by checking the elapsed time in every outer loop.
The returned list is a valid sequence for the external ``evaluate_sequence``
function (non‑negative, sum ≥ 0.01).  The code contains only top‑level
functions – no closures – as required by the evaluation harness.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level – no closures)
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x >= 0, sum(x)=1}."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero – return uniform distribution
        return np.full(n, 1.0 / n, dtype=v.dtype)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Compute the evaluation score for a (already normalised to sum = 1).

    Equivalent to ``evaluate_sequence(a.tolist())`` but without the
    safety checks – much faster for the inner loops.
    """
    conv = np.convolve(a, a)
    max_val = float(np.max(conv))
    return 2.0 * a.shape[0] * max_val


def _normalise(arr: np.ndarray) -> np.ndarray:
    """Return a non‑negative copy of ``arr`` whose sum equals 1."""
    a = np.maximum(arr, 0.0)
    s = a.sum()
    if s <= 0.0:
        # fallback to uniform distribution
        return np.full_like(a, 1.0 / a.shape[0])
    return a / s


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: float = 950, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises the
    provided evaluation function.  The function respects the time budget
    and returns the best list of floats found.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds (a few seconds are kept as a safety margin).
    **kwargs
        Ignored – kept for harness compatibility.

    Returns
    -------
    list[float]
        The best sequence discovered.
    """
    np.random.seed(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # Determine problem size and initialise with the supplied fallback
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # provided by the environment
        best_arr = _normalise(np.asarray(height_sequence_1, dtype=float))
    except Exception:
        n = 2244
        best_arr = np.full(n, 1.0 / n, dtype=float)

    best_score = compute_score(best_arr)
    print(f"[init] n={n}, baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1. Broad family search
    # ------------------------------------------------------------------
    idx = np.arange(n, dtype=float)

    # ----- (a) power‑law ------------------------------------------------
    alphas = np.linspace(0.30, 0.90, 61)          # step ≈0.01
    shifts = np.linspace(0.5, 3.0, 11)            # 0.5 … 3.0
    for alpha in alphas:
        if time.time() - start > budget_s - 5:
            break
        for shift in shifts:
            raw = np.power(idx + shift, -alpha)
            cand = _normalise(raw)
            sc = compute_score(cand)
            if sc < best_score:
                best_score = sc
                best_arr = cand
                print(f"[pow]   α={alpha:.4f} shift={shift:.2f} → {sc:.6f}")

    # ----- (b) symmetric power‑law ------------------------------------
    half_n = (n + 1) // 2
    idx_half = np.arange(half_n, dtype=float)
    for alpha in alphas:
        if time.time() - start > budget_s - 5:
            break
        for shift in shifts:
            raw_half = np.power(idx_half + shift, -alpha)
            raw_half = _normalise(raw_half)
            if n % 2 == 0:
                cand = np.concatenate([raw_half[::-1], raw_half])
            else:
                cand = np.concatenate([raw_half[:-1][::-1], raw_half])
            cand = _normalise(cand)
            sc = compute_score(cand)
            if sc < best_score:
                best_score = sc
                best_arr = cand
                print(f"[sym]   α={alpha:.4f} shift={shift:.2f} → {sc:.6f}")

    # ----- (c) Beta‑distribution ----------------------------------------
    p_vals = np.arange(0.5, 5.5, 0.5)
    q_vals = np.arange(0.5, 5.5, 0.5)
    # use mid‑points to avoid exact 0 or 1
    x = (np.arange(n) + 0.5) / n
    for p in p_vals:
        if time.time() - start > budget_s - 5:
            break
        for q in q_vals:
            raw = np.power(x, p - 1.0) * np.power(1.0 - x, q - 1.0) + 1e-12
            cand = _normalise(raw)
            sc = compute_score(cand)
            if sc < best_score:
                best_score = sc
                best_arr = cand
                print(f"[beta]  p={p:.1f} q={q:.1f} → {sc:.6f}")

    # ----- (d) exponential decay -----------------------------------------
    r_vals = np.linspace(0.85, 0.99, 15)
    for r in r_vals:
        if time.time() - start > budget_s - 5:
            break
        raw = np.power(r, idx)
        cand = _normalise(raw)
        sc = compute_score(cand)
        if sc < best_score:
            best_score = sc
            best_arr = cand
            print(f"[exp]   r={r:.3f} → {sc:.6f}")

    # ----- (e) random Dirichlet ------------------------------------------
    concentrations = [0.1, 0.5, 1.0, 2.0, 5.0]
    for conc in concentrations:
        for _ in range(20):
            if time.time() - start > budget_s - 5:
                break
            raw = np.random.dirichlet([conc] * n)
            cand = _normalise(raw)           # already sums to 1, but safe
            sc = compute_score(cand)
            if sc < best_score:
                best_score = sc
                best_arr = cand
                print(f"[dir]   conc={conc:.1f} → {sc:.6f}")

    # ------------------------------------------------------------------
    # 2. Projected gradient descent on the active shift
    # ------------------------------------------------------------------
    cur = best_arr.copy()
    cur_score = best_score
    lr = 0.2                     # initial learning rate
    iter_cnt = 0
    max_iter = 10 ** 6          # hard cap – the time check stops us earlier

    while time.time() - start < budget_s - 5 and iter_cnt < max_iter:
        iter_cnt += 1
        conv = np.convolve(cur, cur)
        max_idx = int(np.argmax(conv))

        # Build the sub‑gradient of the active shift
        grad = np.zeros_like(cur)
        low = max(0, max_idx - (n - 1))
        high = min(n - 1, max_idx)
        idx_range = np.arange(low, high + 1, dtype=int)
        grad[idx_range] = 2.0 * cur[max_idx - idx_range]

        # Candidate update (simple back‑tracking line search)
        cand = cur - lr * grad
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        if cand_score < cur_score:
            # successful step – keep it and allow a slightly larger step next round
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_arr = cand
                print(f"[gd]    iter={iter_cnt} new best {best_score:.6f}")
            lr = min(lr * 1.02, 1.0)
        else:
            # reject – shrink learning rate
            lr *= 0.7
            if lr < 1e-5:
                lr = 0.2   # reset to a safe default

        # occasional progress report
        if iter_cnt % 5000 == 0:
            elapsed = time.time() - start
            print(f"[gd]    iter={iter_cnt} elapsed={elapsed:.1f}s "
                  f"cur={cur_score:.6f} best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Targeted random hill‑climbing (mass transfer)
    # ------------------------------------------------------------------
    cur = best_arr.copy()
    cur_score = best_score
    hill_iters = 20000

    for it in range(hill_iters):
        if time.time() - start > budget_s - 5:
            break

        # current active shift and the set of indices that participate
        conv = np.convolve(cur, cur)
        max_idx = int(np.argmax(conv))
        low = max(0, max_idx - (n - 1))
        high = min(n - 1, max_idx)
        active_set = np.arange(low, high + 1, dtype=int)

        # pick source from the active set, weighted by current mass
        if cur[active_set].sum() == 0.0:
            src = np.random.randint(n)
        else:
            probs = cur[active_set] / cur[active_set].sum()
            src = np.random.choice(active_set, p=probs)

        dst = np.random.randint(n)
        if src == dst:
            continue

        # move a small fraction of the source mass
        delta = cur[src] * (np.random.rand() * 0.02)   # up to 2 %
        if delta <= 0.0:
            continue

        nxt = cur.copy()
        nxt[src] -= delta
        nxt[dst] += delta
        nxt = project_to_simplex(nxt)
        nxt_score = compute_score(nxt)

        if nxt_score < cur_score:
            cur = nxt
            cur_score = nxt_score
            if nxt_score < best_score:
                best_score = nxt_score
                best_arr = nxt
                print(f"[hill]  iter={it} new best {best_score:.6f}")

    print(f"[final] best score found = {best_score:.6f}")
    return best_arr.tolist()
