#!/usr/bin/env python
"""
Propose a candidate step‑function (list of non‑negative heights) that beats the
previous bound 1.503.

Algorithm
---------
* Use a relatively large length (≈20 000) – this reduces the discretisation
  error of the asymptotic power‑law model.
* The shape a_i = (i+offset+1)^{-p} is known to give very good values of the
  evaluation function.  The analytic approximation

      E(p) = 2·(1-p)²·B(1-p,1-p)

  (independent of the length) tells us which exponents are promising.
* Scan a coarse grid of exponents p∈[0.45,0.75] (ordered by the cheap analytic
  bound) and evaluate the true score with ``evaluate_sequence``.
* Refine the best exponent with a finer grid and also try a few integer
  offsets (0,1,2,4,8) to smooth the head of the distribution.
* Keep the best sequence found before the time budget (≈1000 s) runs out.
"""

import time
import numpy as np
from math import gamma

# evaluate_sequence is assumed to be already imported in the execution environment.


def _analytic_score(p: float) -> float:
    """
    Analytic upper bound 2·(1-p)²·B(1-p,1-p) used only for ordering the search.
    """
    u = 1.0 - p
    # Beta(1-p,1-p) = Gamma(1-p)² / Gamma(2-2p) = Gamma(u)² / Gamma(2*u)
    return 2.0 * (u ** 2) * (gamma(u) ** 2) / gamma(2.0 * u)


def _power_sequence(n: int, p: float, offset: int = 0) -> list[float]:
    """
    Return the list  [(i + 1 + offset)^{-p}  for i = 0 … n-1 ].
    The offset is an integer ≥0; offset=0 reproduces the plain power law.
    """
    # Using numpy for speed, then convert to a plain Python list.
    idx = np.arange(1, n + 1, dtype=np.float64) + float(offset)
    seq = (idx ** (-p)).tolist()
    return seq


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function with a low evaluation score.
    The function returns the best sequence found before the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    # Small safety margin so we always finish before the hard timeout.
    deadline = start_time + budget_s - 4.0

    # ----------------------------------------------------------------------
    # 1. Choose a working length.
    # ----------------------------------------------------------------------
    try:
        # If a previous construction exists, start from its length.
        base_len = len(height_sequence_1)  # type: ignore[name-defined]
    except Exception:
        base_len = 0

    # We deliberately use a fairly large length (≈20 k).  If the previous
    # construction is already larger we keep it, otherwise we bump up to 20 000.
    n = max(base_len, 20000)

    # ----------------------------------------------------------------------
    # 2. Coarse grid of exponents (ordered by the cheap analytic bound).
    # ----------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    best_p = None
    best_offset = 0

    # Coarse grid – 101 points, step ≈0.003.
    p_grid = np.linspace(0.45, 0.75, 101)
    # Order the grid by the analytic estimate (smallest first).
    p_grid = sorted(p_grid, key=_analytic_score)

    # Offsets we will test during the coarse phase.
    offsets_coarse = [0]  # keep it simple first; refinements later.

    for p in p_grid:
        if time.time() >= deadline:
            break
        for off in offsets_coarse:
            seq = _power_sequence(n, p, offset=off)
            score = evaluate_sequence(seq)
            if score < best_score:
                best_score = score
                best_seq = seq
                best_p = p
                best_offset = off
                print(f"Coarse: p={p:.5f} off={off}  score={score:.6f}")

    # If nothing was evaluated (should not happen), fall back to a safe default.
    if best_seq is None:
        best_seq = _power_sequence(n, 0.60)
        best_score = evaluate_sequence(best_seq)
        best_p = 0.60
        best_offset = 0
        print("Fallback to default power law (p=0.60).")

    # ----------------------------------------------------------------------
    # 3. Refine around the best exponent – finer p grid and a few offsets.
    # ----------------------------------------------------------------------
    # Define a narrower window around the best exponent found so far.
    p_fine_low = max(best_p - 0.015, 0.45)
    p_fine_high = min(best_p + 0.015, 0.75)
    p_fine_grid = np.linspace(p_fine_low, p_fine_high, 31)

    offsets_fine = [0, 1, 2, 4, 8]

    for p in p_fine_grid:
        if time.time() >= deadline:
            break
        for off in offsets_fine:
            # Small offsets are cheap to compute; we skip large ones if we run out of time.
            seq = _power_sequence(n, p, offset=off)
            score = evaluate_sequence(seq)
            if score < best_score:
                best_score = score
                best_seq = seq
                best_p = p
                best_offset = off
                print(f"Fine  : p={p:.5f} off={off}  score={score:.6f}")

    # ----------------------------------------------------------------------
    # 4. Very small local perturbations (optional, cheap).
    # ----------------------------------------------------------------------
    # If there is still time, we perform a few random single‑coordinate tweaks.
    # Because the sequence is long we only touch a handful of entries.
    if time.time() < deadline:
        cur_seq = best_seq[:]                # copy
        cur_score = best_score
        max_iter = 2000
        iter_cnt = 0
        # Typical relative step size ≈±3 %
        while time.time() < deadline and iter_cnt < max_iter:
            iter_cnt += 1
            idx = np.random.randint(n)
            factor = 1.0 + 0.06 * (np.random.rand() - 0.5)   # ±3 %
            new_val = max(0.0, cur_seq[idx] * factor)
            old_val = cur_seq[idx]
            cur_seq[idx] = new_val
            new_score = evaluate_sequence(cur_seq)
            if new_score < cur_score:
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_seq = cur_seq[:]          # store a copy
                    print(f"Hill  : iter {iter_cnt:04d}  score={best_score:.6f}")
            else:
                # revert the change
                cur_seq[idx] = old_val

    print(f"Finished – best score = {best_score:.6f} with p={best_p:.5f}, offset={best_offset}")
    return best_seq
