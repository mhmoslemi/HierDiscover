The algorithm works in three phases:

1. **Quick grid search** over a 2‑parameter family
   a_i ∝ (i + shift)⁻ᵅ (power‑law).  This provides a decent
   starting point that already beats the uniform construction.

2. **Projected sub‑gradient descent** on the convex objective
   f(a) = 2·n·max(conv(a,a)) / (sum a)².
   The sub‑gradient of the active convolution term
   (the one attaining the maximum) is
       g_i = 2·a_{k‑i}   (0 ≤ i < n, 0 ≤ k‑i < n)
   where k = argmax conv(a,a).  After a step
       a ← a – η·g
   we project back onto the probability simplex
   {a_i ≥ 0, Σ a_i = 1}.  A diminishing step size
   η = η₀ / sqrt(t) guarantees convergence for this convex problem.

3. **Lightweight random hill‑climbing** for a few thousand
   small mass‑transfer moves to polish the result.

All phases respect the 1000 s time budget, stop a few seconds
early, and print occasional progress reports.  The returned list
contains only non‑negative floats; the external evaluator will
compute the official score.
"""

import time
import numpy as np


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex.

    Implements the O(n log n) algorithm of
    Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]                 # sorted descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    # Find the largest ρ satisfying the KKT condition
    cond = u - cssv / ind > 0
    if not np.any(cond):                 # all zeros – fallback to uniform
        return np.full(n, 1.0 / n, dtype=v.dtype)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _compute_score(a: np.ndarray) -> float:
    """
    Directly compute the evaluation function:

        2 * n * max(conv(a,a)) / (sum(a)**2)

    The function is scale‑invariant; a is assumed non‑negative.
    """
    total = a.sum()
    if total < 0.01:                   # mirrors the safeguard in evaluate_sequence
        return float("inf")
    conv = np.convolve(a, a)
    max_conv = conv.max()
    n = a.shape[0]
    return 2.0 * n * max_conv / (total * total)


def propose_candidate(seed: int = 123, budget_s: float = 950, **kwargs) -> list[float]:
    """
    Generate a step‑function (list of non‑negative heights) that tries
    to minimise the provided evaluation metric.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds (a few seconds are left for the caller).
    kwargs : dict
        Ignored – kept for compatibility with the harness.

    Returns
    -------
    list[float]
        The best sequence found within the time budget.
    """
    np.random.seed(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # Determine the length n (use the length of the supplied reference if any)
    # ------------------------------------------------------------------
    seq_ref = globals().get("height_sequence_1")
    if seq_ref is not None:
        n = len(seq_ref)
    else:
        n = 2244               # fallback length mentioned in the description

    # ------------------------------------------------------------------
    # Initialise with a uniform distribution; keep the reference sequence
    # if it is already better.
    # ------------------------------------------------------------------
    best_seq = np.full(n, 1.0 / n, dtype=float)
    best_score = _compute_score(best_seq)

    if seq_ref is not None:
        try:
            cand = np.asarray(seq_ref, dtype=float)
            sc = _compute_score(cand)
            if sc < best_score:
                best_seq = cand.copy()
                best_score = sc
        except Exception:
            pass

    print(f"[init] n={n}, initial best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1. Grid search over the power‑law family a_i ∝ (i+shift)^{-α}
    # ------------------------------------------------------------------
    idx = np.arange(n, dtype=float)

    alphas = np.linspace(0.45, 0.75, 61)          # 0.45 … 0.75 in steps of 0.005
    shifts = np.linspace(1.0, 2.0, 5)            # 1.0, 1.25, …, 2.0

    for alpha in alphas:
        if time.time() - start > budget_s - 20:
            break
        power = -alpha
        for shift in shifts:
            raw = np.power(idx + shift, power)
            if raw.sum() == 0.0:
                continue
            cand = raw / raw.sum()
            sc = _compute_score(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[grid] α={alpha:.4f}, shift={shift:.2f}, score={sc:.6f}")

    print(f"[grid] best after grid search: {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Projected sub‑gradient descent on the convex objective
    # ------------------------------------------------------------------
    cur = best_seq.copy()
    cur_score = best_score
    conv_cur = np.convolve(cur, cur)

    eta0 = 0.5                     # initial step‑size factor
    iteration = 0
    max_iter = 200_000            # safety cap; real limit is the time budget

    while time.time() - start < budget_s - 15 and iteration < max_iter:
        iteration += 1

        # index attaining the current maximum convolution value
        k_max = int(np.argmax(conv_cur))

        # sub‑gradient of the active term: g_i = 2 * cur_{k_max - i} (if inside range)
        i = np.arange(n)
        j = k_max - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(cur)
        grad[mask] = 2.0 * cur[j[mask]]

        # diminishing step size
        eta = eta0 / np.sqrt(iteration)

        # gradient step and projection onto the simplex
        cand = cur - eta * grad
        cand = _project_to_simplex(cand)

        # evaluate the candidate (only one convolution needed per iteration)
        conv_new = np.convolve(cand, cand)
        new_score = 2.0 * n * conv_new.max() / (cand.sum() ** 2)

        if new_score < cur_score - 1e-12:      # accept strict improvement
            cur = cand
            cur_score = new_score
            conv_cur = conv_new
            if new_score < best_score:
                best_score = new_score
                best_seq = cand.copy()
                print(f"[gd] iter={iteration}, improved score={new_score:.6f}")

    print(f"[gd] finished after {iteration} iterations, best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Final random hill‑climbing refinement (small mass transfers)
    # ------------------------------------------------------------------
    cur = best_seq.copy()
    cur_score = best_score
    max_hc_iter = 20_000
    while time.time() - start < budget_s - 5 and max_hc_iter > 0:
        max_hc_iter -= 1
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # move up to 2 % of the mass from i to j
        step = cur[i] * (np.random.rand() * 0.02)
        if step <= 0.0:
            continue
        cand = cur.copy()
        cand[i] -= step
        cand[j] += step
        cand = _project_to_simplex(cand)
        new_score = _compute_score(cand)
        if new_score < cur_score:
            cur = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = cand.copy()
                print(f"[hc] iter={iteration} improved score={new_score:.6f}")

    print(f"[final] best score achieved: {best_score:.6f}")
    return best_seq.tolist()
