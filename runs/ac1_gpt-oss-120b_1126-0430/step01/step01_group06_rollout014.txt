"""
Propose a non‑negative step function (list of heights) that lowers the
evaluation score 2 * n * max(conv(a, a)) / (sum(a)**2).

The algorithm works in three phases:

1. **Rich grid search** – explores three parametric families:
   * power‑law      a_i ∝ (i+shift)^(-α)
   * exponential    a_i ∝ exp(-λ·i)
   * mixed          a_i ∝ (i+shift)^(-α)·exp(-λ·i)

   All candidates are normalised to a probability simplex (sum = 1),
   the score is computed with a fast in‑house evaluator, and the best
   vector from the grid is kept.

2. **Targeted sub‑gradient descent** – starting from the best grid
   candidate we repeatedly identify the convolution index where the
   maximum occurs, compute the gradient of that entry
   (∂b_s/∂a_i = 2·a_{s‑i}) and take a step opposite to the gradient.
   After each step the vector is projected back onto the simplex.
   A decreasing step‑size schedule (η ∝ 1/√t) provides stability.
   Improvements are printed whenever the global best is updated.

3. **Fallback & safety** – if the search finishes without improving the
   provided baseline `height_sequence_1`, that sequence is returned.

The whole routine is time‑boxed (default 950 s).  It uses only NumPy
and the already‑available `evaluate_sequence` for final verification.

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x ≥ 0, sum(x)=1}
# ----------------------------------------------------------------------
def _project_to_simplex(v):
    """Efficient projection onto the simplex (Wang & Carreira‑Perpiñán, 2013)."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero – return uniform distribution
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


# ----------------------------------------------------------------------
# Fast evaluation (identical to evaluate_sequence for non‑negative values)
# ----------------------------------------------------------------------
def _score_and_conv(vec, n):
    """Return (score, full convolution) for a non‑negative vector."""
    s = vec.sum()
    if s <= 1e-12:
        return float('inf'), None
    b = np.convolve(vec, vec)
    max_b = b.max()
    # the original evaluate_sequence returns 2 * n * max_b / (sum^2)
    score = 2.0 * n * max_b / (s * s)
    return float(score), b

def _gen_exponential(idx, lam):
    return np.exp(-lam * idx)

def _gen_mixed(idx, alpha, shift, lam):
    return np.power(idx + shift, -alpha) * np.exp(-lam * idx)

                    if new_score < cur_score:
                        cur = new
                        cur_score = new_score
                        if new_score < best_score:
                            best_score = new_score
                            best_seq = new.tolist()
                            print(f"[rand] iter={it:5d} improved score={new_score:.6f}")

        # time guard
        if it % 1000 == 0 and time.time() - start > budget_s - 5:
            break

    print(f"[final] best score discovered: {best_score:.6f}")
    return best_seq
