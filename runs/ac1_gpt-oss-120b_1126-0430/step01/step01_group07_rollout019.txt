def optimize_max(a, deadline):
    n = len(a)
    best_a = a.copy()
    best_s = fast_score(a)
    beta = 5.0  # start temperature
    step = 0.1   # initial step size
    while time.time() < deadline:
        b = convolve_fft(a, a)
        M = b.max()
        # compute weights
        # weights exponentially emphasize near max
        w = np.exp(beta * (b - M))
        # optionally zero out far away to speed compute? Not needed.
        # Normalize w to sum=1 (optional). Not required for gradient direction, but for scaling.
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(w) / w.size
        else:
            w = w / w_sum
        # gradient
        grad = 2.0 * np.correlate(w, a, mode='valid')
        # adapt step size: try a few step reductions
        eta = step
        improved = False
        for _ in range(10):
            a_new = a - eta * grad
            # project to simplex
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s == 0:
                a_new = np.full_like(a_new, 1.0 / len(a_new))

                a_new /= s

                a = a_new

def optimize_max(a, deadline, max_iters=100000):
    n = len(a)
    best_a = a.copy()
    best_score = fast_score(a)
    # parameters
    beta = 8.0
    step = 0.05
    iter_count = 0
    while time.time() < deadline and iter_count < max_iters:
        b = convolve_fft(a, a)
        M = b.max()
        # compute softmax weights
        w = np.exp(beta * (b - M))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(b) / len(b)
        else:
            w /= w_sum
        # gradient
        grad = 2.0 * np.correlate(w, a, mode='valid')
        # update with backtracking line search
        eta = step
        improved = False
        for _ in range(10):
            a_new = a - eta * grad
            # simple projection onto simplex
            a_new = np.maximum(a_new, 0.0)
            sum_a = a_new.sum()
            if sum_a == 0:
                a_new = np.full(n, 1.0 / n)
            else:
                a_new /= sum_a
            new_score = fast_score(a_new)
            if new_score < best_score - 1e-9:
                best_score = new_score
                best_a = a_new.copy()
                a = a_new
                improved = True
                break
            else:
                eta *= 0.5
        if not improved:
            # reduce step to avoid stagnation
            step *= 0.9
        # slowly increase beta to focus more on max
        beta = min(beta * 1.01, 200.0)
        # occasional random perturb if stuck
        if iter_count % 500 == 0 and not improved:
            # random jitter
            eps = np.random.randn(n) * 1e-4
            a_jitter = a + eps
            a_jitter = np.maximum(a_jitter, 0.0)
            a_jitter /= a_jitter.sum()
            a = a_jitter
        iter_count += 1
    return best_a, best_score
```

def gradient_max_descent(a, deadline, max_iter=5000):
    n = len(a)
    best_a = a.copy()
    best_score = fast_score(a)
    beta = 5.0
    step = 0.05
    it = 0
    while time.time() < deadline and it < max_iter:
        # compute convolution
        b = convolve_fft(a, a)
        M = b.max()
        # compute weights as softmax around max
        # subtract M for stability
        w = np.exp(beta * (b - M))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(b) / len(b)
        else:
            w /= w_sum
        # gradient: 2 * correlation (valid)
        grad = 2.0 * np.correlate(w, a, mode='valid')
        # adaptive step size with backtracking
        eta = step
        improved = False
        for _ in range(10):
            a_new = a - eta * grad
            # projection onto simplex
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s == 0.0:
                a_new = np.full(n, 1.0 / n)
            else:
                a_new /= s
            sc = fast_score(a_new)
            if sc < best_score - 1e-10:
                best_score = sc
                best_a = a_new.copy()
                a = a_new
                improved = True
                break
            else:
                eta *= 0.5
        if not improved:
            # reduce step if not improving
            step *= 0.9
        # increase beta gradually
        beta = min(beta * 1.02, 200.0)
        # occasional jitter to escape flat areas
        if it % 200 == 0 and not improved:
            # add small random noise
            noise = np.random.randn(n) * 1e-4
            a = a + noise
            a = np.maximum(a, 0.0)
            a /= a.sum()
        it += 1
    return best_a, best_score
```

"""
Optimise a non‑negative step‑function (sequence a_i) for the score

    score = 2·n·max(conv(a,a)) / (sum(a)**2)

The known “square‑root” law (a_i ∝ (i+δ)^‑½) gives a flat interior
convolution and approaches the theoretical bound 2·π/4 ≈ 1.5708.
The program below

1. evaluates a few analytical families (power‑law, power‑law + exponential,
   a two‑component power‑law) on a modestly large size (n≈5000) and keeps the
   best candidate,
2. refines that candidate with a projected sub‑gradient descent that directly
   minimises a smooth approximation of the L∞‑norm of the convolution
   (soft‑max weighting of the largest entries),
3. returns the best sequence found before the time‑budget expires.

All operations are vectorised, the convolution is performed with an FFT
(≈ O(n log n)) and the projection step simply clips to the non‑negative orthant
and renormalises to unit sum, which is enough because the evaluation function
is scale‑invariant.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast linear convolution via FFT (real inputs, zero‑padding)
def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a * b using an FFT of power‑of‑two length."""
    size = a.size + b.size - 1
    N = 1 << (size - 1).bit_length()        # next power of two
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    C = A * B
    conv = np.fft.irfft(C, N)[:size]
    return conv


# ----------------------------------------------------------------------
# Fast evaluation of the objective for a *already normalised* sequence
def fast_score(a: np.ndarray) -> float:
    """
    Return 2·n·max(conv(a,a)) / (sum(a)**2).  The input is assumed to be a
    one‑dimensional NumPy array of non‑negative floats.
    """
    n = a.size
    b = convolve_fft(a, a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 1e-12:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Families of candidate sequences ------------------------------------------------

def power_law(alpha: float, delta: float, n: int) -> np.ndarray:
    """a_i ∝ (i+δ)^‑α,   i = 1 … n,  normalised to sum‑one."""
    idx = np.arange(1.0, n + 1) + delta
    a = idx ** (-alpha)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def power_law_exp(alpha: float, delta: float, lam: float, n: int) -> np.ndarray:
    """a_i ∝ (i+δ)^‑α · e^{-λ i}, normalised."""
    idx = np.arange(1.0, n + 1) + delta
    a = (idx ** (-alpha)) * np.exp(-lam * (idx - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def two_power_law(alpha1: float, delta1: float,
                  alpha2: float, delta2: float,
                  weight: float, n: int) -> np.ndarray:
    """
    Convex combination of two power‑law shapes.
    weight ∈ [0,1] multiplies the first component.
    """
    idx = np.arange(1.0, n + 1)
    a1 = (idx + delta1) ** (-alpha1)
    a2 = (idx + delta2) ** (-alpha2)
    a = weight * a1 + (1.0 - weight) * a2
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Projected sub‑gradient descent aimed at the L∞‑norm of the convolution.
def minimise_max_convolution(a: np.ndarray,
                             deadline: float,
                             max_iter: int = 10000) -> (np.ndarray, float):
    """
    Starting from a (already normalised), perform a few hundred
    sub‑gradient steps on a smooth surrogate of max(conv(a,a)).
    The surrogate is a soft‑max with a temperature that is slowly
    increased, concentrating the gradient on the current peak(s).
    The step size is adjusted by a simple back‑tracking line search.
    """
    n = a.size
    best_a = a.copy()
    best_score = fast_score(best_a)

    beta = 5.0          # soft‑max temperature (low → smooth, high → max)
    step = 0.08         # initial step length
    it = 0

    while time.time() < deadline and it < max_iter:
        # current convolution and its maximum
        b = convolve_fft(a, a)
        M = b.max()

        # soft‑max weights that concentrate on the largest entries of b
        w = np.exp(beta * (b - M))          # 0 ≤ w ≤ 1, w_max = 1
        w_sum = w.sum()
        if w_sum == 0.0:                    # numerical safety
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of Σ_k w_k b_k  (see derivation in analysis)
        grad = 2.0 * np.correlate(w, a, mode='valid')

        # simple back‑tracking line search
        eta = step
        improved = False
        for _ in range(12):
            a_new = a - eta * grad
            # projection onto the simplex (non‑negative, unit sum)
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s == 0.0:
                a_new = np.full(n, 1.0 / n)
            else:
                a_new /= s

            sc = fast_score(a_new)
            if sc < best_score - 1e-12:        # accept only genuine improvement
                best_score = sc
                best_a = a_new.copy()
                a = a_new
                improved = True
                break
            eta *= 0.5                         # shrink step if no improvement

        # adapt parameters when we get stuck
        if not improved:
            step *= 0.9          # become more cautious
        else:
            # gradually tighten the soft‑max (focus on the true maximum)
            beta = min(beta * 1.015, 250.0)

        # occasional tiny random perturbation to escape tiny plateaux
        if it % 250 == 0 and not improved:
            noise = np.random.randn(n) * 1e-5
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()

        it += 1

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a sequence with a low evaluation score.
    The routine performs a cheap random search over a few analytic families,
    then refines the best candidate with the sub‑gradient optimiser described
    above.  The total running time is kept below `budget_s` seconds.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # small safety margin

    # --------------------------------------------------------------
    # 1)   Choose a problem size.  5000 gives a nice trade‑off between
    #      accuracy (discretisation error small) and speed.
    n = 5000

    # start from a simple uniform reference
    a_uniform = np.full(n, 1.0 / n)
    best_seq = a_uniform.copy()
    best_score = fast_score(best_seq)
    print(f'initial uniform score = {best_score:.6f}')

    # --------------------------------------------------------------

    for delta in deltas:
        for alpha in alphas:
            if time.time() >= deadline:
                break
            a = power_law(alpha, delta, n)
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f'power‑law improvement α={alpha:.4f} δ={delta:.2f} → {sc:.6f}')
        if time.time() >= deadline:
            break

    # --------------------------------------------------------------
    # 3)   A few randomised families (power‑law + exponential, two‑component)
    #      Each evaluation is cheap; we keep the best 3 candidates.
    top_candidates = [(best_seq, best_score)]
    for _ in range(1800):

        family = np.random.choice(['plain', 'exp', 'two'])

            w = np.random.rand()
            a = two_power_law(alpha1, delta1, alpha2, delta2, w, n)

        sc = fast_score(a)
        # keep a short sorted list of the best candidates seen so far

            top_candidates.append((a, sc))
            top_candidates.sort(key=lambda x: x[1])
            top_candidates = top_candidates[:5]

    # 4)   Refine the leading candidates with the max‑gradient optimiser.
