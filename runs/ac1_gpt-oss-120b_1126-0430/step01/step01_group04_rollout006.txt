def compute_gradient(a, k):
    n = len(a)
    # padded a: length 2n-1 with zeros at both ends.
    pad = np.zeros(2*n-1, dtype=float)
    pad[n-1:n-1+n] = a
    idx1 = (n-1) + (k - np.arange(n))
    idx2 = (n-1) + (np.arange(n) - k)
    grad = pad[idx1] + pad[idx2]
    return grad

"""
Propose a candidate step‑function.

Algorithm:
1. Determine problem size (n) from global `height_sequence_1` if present,
   otherwise use a moderate default (≈2000).  The L1‑norm is fixed to
   sqrt(2·n) because the evaluation function is scale‑invariant.
2. Phase A – Broad sweep of analytically‑parameterised families
   (power‑law, shifted‑power, polynomial, linear, exponential,
   Gaussian, random Dirichlet).  Every generated raw vector is normalised
   to the required sum and evaluated; the best vector is kept.
3. Phase B – Sub‑gradient descent on the objective
   f(a)=max_k (a⋆a)_k .  The index giving the current maximum is used
   to compute the sub‑gradient g_j = 2·a_{k−j} (zero outside the support).
   A decreasing step size is applied, the result is projected onto the
   simplex (non‑negative, fixed L1‑norm) and accepted if it improves the
   score; otherwise a small random transfer is tried.
   This loop runs for the majority of the time budget.
4. Phase C – Light‑weight random re‑allocation moves for the last few
   seconds to possibly escape very shallow local minima.

The routine respects the 1000 s time limit, prints occasional progress
messages and finally returns the best list of heights found.
"""

def _project_to_simplex(v, s):
    """Project vector v onto the simplex {x ≥ 0, sum(x) = s}."""
    n = v.shape[0]
    if s <= 0:
        return np.zeros_like(v)
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _make_sequence_from_raw(raw, target_sum):
    """Scale a non‑negative raw vector to have L1‑norm equal to target_sum."""
    raw = np.maximum(raw, 0.0)
    total = raw.sum()
    if total == 0:
        # uniform fallback
        n = raw.shape[0]
        return np.full(n, target_sum / n)
    scale = target_sum / total
    return (raw * scale).tolist()


def _score_of(seq):
    """Exact evaluation of the score (without the extra security checks)."""
    seq = np.asarray(seq, dtype=float)
    n = len(seq)
    conv = np.convolve(seq, seq)
    max_conv = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_conv / (sum_a ** 2))


def _gradient(seq, k):
    """
    Sub‑gradient of max_conv at index k for a non‑negative sequence `seq`.
    Returns a vector g with g_j = 2 * seq[k-j] when the index is valid,
    otherwise 0.
    """
    n = seq.shape[0]
    grad = np.zeros_like(seq)
    idx = k - np.arange(n)
    mask = (idx >= 0) & (idx < n)
    grad[mask] = 2.0 * seq[idx[mask]]
    return grad


def _random_raw(shape, n):
    """Generate a raw (unscaled) non‑negative vector according to `shape`."""
    if shape == "power":
        p = np.random.uniform(0.1, 2.0)
        raw = (np.arange(1, n + 1, dtype=float)) ** (-p)
    elif shape == "shifted_power":
        p = np.random.uniform(0.1, 2.0)
        d = np.random.uniform(0.0, n * 0.5) + 1e-6
        raw = (np.arange(n, dtype=float) + d) ** (-p)
    elif shape == "poly":
        alpha = np.random.uniform(0.1, 8.0)
        raw = (1.0 - np.arange(n, dtype=float) / (n - 1)) ** alpha
    elif shape == "linear":
        raw = np.linspace(1.0, 0.0, n, dtype=float)
    elif shape == "exp":
        lam = np.random.uniform(0.1, 5.0)
        raw = np.exp(-lam * np.arange(n, dtype=float) / n)
    elif shape == "gauss":
        mu = np.random.uniform(0.3, 0.7) * n
        sigma = np.random.uniform(0.05, 0.25) * n
        raw = np.exp(-0.5 * ((np.arange(n) - mu) / sigma) ** 2)
    else:  # "dirichlet"
        a = np.random.gamma(shape=0.5 + np.random.rand() * 1.5, scale=1.0, size=n)
        raw = a
    return raw

"""
Propose a candidate step‑function.

Algorithm
---------
1. Determine the problem size (`n`) from the global `height_sequence_1`
   when it exists; otherwise fall back to a moderate default.
   The L1‑norm is fixed to √(2 · n) because the evaluation function is
   scale‑invariant.

2. **Phase A – Broad sweep**  
   Generate families of analytically‑parameterised vectors
   (power‑law, shifted‑power, polynomial, linear, exponential,
   Gaussian, random Dirichlet).  Each raw vector is scaled to the required
   sum, projected onto the non‑negative simplex and evaluated.
   The best vector so far is kept and occasional tiny perturbations are
   tried to escape plateaus.

3. **Phase B – Sub‑gradient descent** (dominant part of the time budget)  
   The objective is `f(a)=max_k (a⋆a)_k`.  For the current maximal index
   `k*` the sub‑gradient is `g_j = 2·a_{k*‑j}` for indices where the
   term exists, otherwise 0.  A decreasing step size
   `step = step0 / √t` is applied, the result is projected back onto the
   simplex and accepted if it improves the score.  If not, a small
   random mass‑transfer move is performed and accepted with a tiny
   simulated‑annealing probability.

4. **Phase C – Light random re‑allocation** for the last few seconds.

All helper functions are top‑level, no closures or lambdas are used.
The routine respects the 1000 s budget, prints progress information and
returns the best list of heights found.

def _project_to_simplex(v, s):
    """Project vector `v` onto the simplex {x ≥ 0, sum(x)=s}."""
    n = v.shape[0]
    if s <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]                # descending
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
