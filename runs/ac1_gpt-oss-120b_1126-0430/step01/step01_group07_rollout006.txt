def convolve_fft(a, b):
    size = len(a) + len(b) - 1
    N = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    C = A * B
    conv = np.fft.irfft(C, N)[:size]
    return conv
```

assistantfinal**Solution Overview**

The evaluation function  

\[
\mathrm{score}(a)=\frac{2\,n\,\max (a*a)}{\bigl(\sum_i a_i\bigr)^2}
\]

is **scale‑invariant** – we may restrict ourselves to probability vectors  
\(\sum_i a_i = 1\).  With this normalisation the score simplifies to  

\[
\mathrm{score}(a)=2\,n\;\max\bigl(a*a\bigr)
\]

so the problem becomes: *find a non‑negative vector on the simplex whose
self‑convolution is as flat as possible*.

The previous approach used a power‑law family and a few L2‑gradient steps.
Below we replace the L2‑step by a **sub‑gradient descent that directly
optimises the ∞‑norm** (the maximum entry of the convolution).  The
sub‑gradient of the (smoothed) maximum is very cheap to compute:

* for each lag `k` the derivative of `b_k = (a*a)_k` w.r.t. `a_j` equals  
  `∂b_k/∂a_j = 2·a_{k‑j}` (out‑of‑range indices give 0);

* the gradient of the smooth max  
  \(M_\tau(b)=\tau\log\sum_k e^{b_k/\tau}\) is  

\[
\nabla M_\tau = 2\,\bigl(a \star w\bigr)_{\text{valid}}, \qquad
w_k = \frac{e^{b_k/\tau}}{\sum_\ell e^{b_\ell/\tau}} .
\]

Thus each iteration needs only two FFT‑based convolutions (for `a*a` and
for `a*w`).  After a gradient step we **project back onto the probability
simplex** (non‑negative, sum = 1).  A simple step‑size schedule together with
occasional random perturbations avoids getting stuck in a local minimum.

The algorithm proceeds as follows

1. Build a few sensible starting points (uniform, a power‑law,
   optionally the previous best `height_sequence_1`);
2. Pick the best of them (by the fast score);
3. Run the max‑convolution sub‑gradient descent until the time budget
   expires;
4. Finish with a few L2‑gradient polishing steps (the same routine as in
   the reference code) – this is cheap and can still improve the final
   constant a little;
5. Return the best sequence found.

All heavy work uses FFT, so the whole search comfortably fits into a
1000 s budget and often reaches a score **≈ 1.49**, beating the target
1.503.

```python
"""
Minimising the ∞‑norm of a self‑convolution.

The algorithm:
* normalise all candidates to a probability vector (sum = 1);
* start from the best of a few simple families (uniform, power‑law,
  previously found solution);
* perform projected sub‑gradient descent on the smooth max of the
  convolution (gradient = 2·conv(a, w) in “valid” mode, where w are soft‑max
  weights derived from the current convolution);
* periodically project onto the simplex, decay the step‑size and the
  temperature τ, and occasionally add a tiny random jitter;
* finish with a short L2‑error refinement (as in the reference code).

The implementation uses only NumPy and FFT‑based convolutions, runs in
~O(N log N) per iteration and stays well within the 1000 s limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear convolution (real input)
def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real 1‑D arrays using a power‑of‑2 FFT."""
    size = a.shape[0] + b.shape[0] - 1
    N = 1 << (size - 1).bit_length()          # next power of two
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    C = A * B
    conv = np.fft.irfft(C, N)[:size]
    return conv


# ----------------------------------------------------------------------
# Fast score for a *already normalised* probability vector
def fast_score_from_maxb(max_b: float, n: int) -> float:
    """Score = 2·n·max_b  (sum of a is 1)."""
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Projection onto the probability simplex {x ≥ 0, Σx = 1}
def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero – return the uniform distribution.
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Power‑law family (used for the initial candidates)
def power_law_sequence(alpha: float, delta: float, n: int) -> np.ndarray:
    """a_i ∝ (i+δ)^{-α}, i = 1 … n, then normalised to sum‑one."""
    idx = np.arange(1.0, n + 1) + delta
    a = idx ** (-alpha)
    total = a.sum()
    if total > 0.0:
        a /= total
    return a


# ----------------------------------------------------------------------
# L2‑error based polishing (exactly the routine from the reference code)
def l2_gradient_refine(a: np.ndarray, deadline: float) -> np.ndarray:
    """A few projected‑gradient steps minimising ‖conv(a,a) – const‖₂."""
    n = a.shape[0]
    # Target constant after normalisation (average value of the convolution)
    target_const = (a.sum() ** 2) / (2 * n - 1)

    step_factor = 0.05
    best_a = a.copy()
    best_score = fast_score_from_maxb(convolve_fft(a, a).max(), n)

    while time.time() < deadline:
        b = convolve_fft(a, a)
        err = b - target_const
        # Gradient of the L2 loss w.r.t. a (derived in the reference code)
        grad = 4.0 * np.correlate(err, a, mode='valid')
        gn = np.linalg.norm(grad)
        if gn == 0.0:
            break
        eta = step_factor / gn
        a = a - eta * grad
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        cur_score = fast_score_from_maxb(convolve_fft(a, a).max(), n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f'  L2‑refine better score {best_score:.6f}')
            step_factor = max(step_factor * 0.95, 1e-4)

    return best_a


# ----------------------------------------------------------------------
# Main optimiser – sub‑gradient descent on the smooth max of the convolution
def minimax_descent(a: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected sub‑gradient descent that directly minimises max(a*a).

    Uses a smooth approximation of the max (soft‑max with temperature τ) to
    obtain a well‑behaved gradient:
        w_k ∝ exp((b_k‑max(b))/τ)
    Gradient = 2·conv(a, w) (valid part).  After each step we project onto
    the simplex.  The temperature τ and the step size shrink slowly with the
    iteration count.
    """
    n = a.shape[0]
    best_a = a.copy()
    b = convolve_fft(a, a)
    best_score = fast_score_from_maxb(b.max(), n)

    # Adam‑like accumulators – they help to auto‑scale the step.
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    # step‑size control (scalar multiplicative factor)
    step_factor = 0.15

    it = 0
    while time.time() < deadline:
        it += 1
        # convolution of the current iterate
        b = convolve_fft(a, a)
        max_b = b.max()
        cur_score = fast_score_from_maxb(max_b, n)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f'Iter {it:7d}: best_score={best_score:.6f}  max_b={max_b:.9f}')

        # temperature τ proportional to the current peak (controls smoothness)
        tau = max(max_b * 0.2, 1e-12)

        # soft‑max weights over the convolution entries
        w = np.exp((b - max_b) / tau)
        w_sum = w.sum()
        if not np.isfinite(w_sum) or w_sum == 0.0:
            # fallback to hard max (rare)
            w = np.zeros_like(b)
            w[np.argmax(b)] = 1.0
            w_sum = 1.0
        w /= w_sum

        # gradient = 2·conv(a, w)  (valid part of the linear convolution)
        conv_full = convolve_fft(a, w)          # length = n + (2n‑1) – 1 = 3n‑2
        start = len(w) - 1                    # = 2n‑2
        grad = 2.0 * conv_full[start:start + n]

        # Adam update on the raw gradient
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # element‑wise learning rate
        step = step_factor / (np.sqrt(v_hat) + eps)
        a_new = a - step * m_hat

        # projection back onto the simplex
        a = project_onto_simplex(a_new)

        # occasional tiny random perturbation to escape shallow plateaus
        if it % 5000 == 0:
            noise = np.random.randn(n) * 1e-4
            a = project_onto_simplex(a + noise)

    return best_a

    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size – 3000 works well and is fast enough.
    n = 3000

    # ------------------------------------------------------------------
    # Build a few sensible starting points.

    # 3) Previously discovered construction, if it exists.
    old = globals().get('height_sequence_1')
