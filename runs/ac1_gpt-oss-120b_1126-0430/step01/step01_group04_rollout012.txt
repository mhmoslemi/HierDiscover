"""
propose_candidate
=================

Search for a non‑negative step‑function (list of heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The evaluation function normalises the L1‑norm to ``sqrt(2*n)``,
so the score is simply the peak of the linear convolution of ``a`` with itself.
The algorithm consists of three stages:

1. **Analytic sweep** – generate monotone power‑law sequences
   a_i ∝ (i+offset)^{-p}.  The exponent `p` and the offset are scanned
   (grid + random sampling).  For each candidate the L1‑norm is rescaled
   to the required value and the score is measured.

2. **Fine local search** – starting from the best analytic candidate,
   repeatedly move a small amount of mass from one index to another
   (preserving the total sum) and accept the move if it improves the score
   (simple hill‑climbing with a tiny simulated‑annealing component).

3. **Return** the best sequence found before the time budget expires
   (default 1000 s).  The implementation respects the 2 CPU limit
   by being essentially single‑threaded and cheap per evaluation.

The core idea is that a decreasing power‑law with exponent
p≈0.6 (and a modest offset) gives a peak close to the theoretical lower
bound ``π/2 ≈ 1.571`` and, for the finite lengths used in the competition,
can dip below the target ``1.503``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level – required by the rules)
# ----------------------------------------------------------------------


def _project_to_simplex(x: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {y | y_i ≥ 0, ∑y_i = target_sum}.
    O(n) algorithm (Wang & Carreira‑Perpiñán, 2013).
    """
    if target_sum <= 0:
        return np.zeros_like(x)

    # sort descending
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho_arr = u - cssv / np.arange(1, len(x) + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(x - theta, 0.0)
    return w


def _evaluate(seq):
    """Thin wrapper around the supplied evaluate_sequence."""
    # The judge injects evaluate_sequence into the global namespace.
    return evaluate_sequence(seq)


def _make_sequence_from_raw(raw: np.ndarray, target_sum: float) -> list:
    """Scale a non‑negative raw vector so that its L1‑norm equals target_sum."""
    raw_sum = raw.sum()
    if raw_sum == 0.0:
        # fallback to uniform distribution if something went wrong
        n = raw.shape[0]
        return [target_sum / n] * n
    scale = target_sum / raw_sum
    seq = (raw * scale).tolist()
    return seq


# ----------------------------------------------------------------------
# Main candidate generator
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list:
    """
    Return a list of non‑negative heights that (hopefully) yields a low
    evaluation score.  The function stops before ``budget_s`` seconds.
    """
    np.random.seed(seed)

    # --------------------------------------------------------------
    # Determine problem size (use the existing solution if present)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
    else:
        # a moderate size that is fast to evaluate yet large enough to profit
        n = 3000

    target_sum = math.sqrt(2.0 * n)          # the normalising L1‑norm

    # initialise with a uniform vector – easy baseline
    best_seq = [target_sum / n] * n
    best_score = _evaluate(best_seq)
    print(f"[init] n={n}, uniform score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # reserve a few seconds for refinement

    # --------------------------------------------------------------
    # Phase 1 – analytic sweep (power‑law families)
    # --------------------------------------------------------------
    # grid for exponent p (focus on the interval where theory predicts the
    # optimum, i.e. roughly 0.5‑0.8)
    p_grid = np.linspace(0.51, 0.80, 30)  # 30 values

    # deterministic offsets (including zero) and a few random offsets
    offset_grid = [0.0, n * 0.01, n * 0.05, n * 0.10, n * 0.20]

    it = 0
    while time.time() < deadline:
        # pick a grid point (deterministic loop) – we cycle through them
        p = p_grid[it % len(p_grid)]
        # occasionally test a random offset
        if it % 7 == 0:
            d = np.random.rand() * n * 0.3   # random offset up to 30 % of length
        else:
            d = offset_grid[(it // len(p_grid)) % len(offset_grid)]

        # ----------------------------------------------------------
        # generate raw vector a_i = (i + 1 + d)^{-p}
        # ----------------------------------------------------------
        idx = np.arange(1, n + 1, dtype=float) + d   # start at (1+d) to avoid zero‑power
        raw = idx ** (-p)

        # ----------------------------------------------------------
        # scale to required L1‑norm and evaluate
        # ----------------------------------------------------------
        candidate = _make_sequence_from_raw(raw, target_sum)
        score = _evaluate(candidate)

        if score < best_score:
            best_score = score
            best_seq = candidate
            print(f"[sweep {it:05d}] p={p:.3f} d={d:.1f} → score={score:.6f}")

        it += 1
        # safety break if we have cycled many times without time left
        if it > 50000:
            break

    # --------------------------------------------------------------
    # Phase 2 – local hill‑climbing / simulated annealing
    # --------------------------------------------------------------
    current_seq = np.array(best_seq, copy=True)
    current_score = best_score
    temperature = 0.04
    decay = 0.9996

    # allow a few extra seconds beyond the sweep deadline
    refine_deadline = start_time + budget_s - 0.2

    while time.time() < refine_deadline:
        # select two distinct indices uniformly
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        # propose moving a small fraction of mass from i to j
        max_move = current_seq[i] * 0.04  # up to 4 % of the source entry
        if max_move <= 0.0:
            continue
        delta = np.random.rand() * max_move

        new_seq = current_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        # project back onto the simplex (preserves sum, guarantees non‑negativity)
        new_seq = _project_to_simplex(new_seq, target_sum)

        new_score = _evaluate(new_seq.tolist())

        # accept if we improve, otherwise accept with a tiny probability
        if new_score < current_score:
            current_seq = new_seq
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.tolist()
                print(f"[refine] improved to {best_score:.6f}")
        else:
            prob = math.exp(-(new_score - current_score) / (temperature + 1e-12))
            if np.random.rand() < prob:
                current_seq = new_seq
                current_score = new_score

        temperature *= decay

    print(f"[final] best score found: {best_score:.6f}")
    return best_seq
