"""
Optimise a step‑function (non‑negative heights) for the supplied
``evaluate_sequence``.  The method starts from the Catalan seed (which
gives a value ≈ π/2) and then runs an exponentiated‑gradient descent on a
smoothed version of the objective  max_i (a*a)_i .  The smoothing is a
log‑sum‑exp (soft‑max) with a temperature that is slowly increased, so the
updates initially move the whole convolution profile towards flatness and
later focus on the true maximum.  Adam‑style momentum is used on the
log‑parameters, a simple projection onto the probability simplex (normalise
to sum 1) keeps the variables feasible, and occasional random restarts
avoid getting stuck.  The best sequence seen before the time budget expires
is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all defined at top level, no closures)
# ----------------------------------------------------------------------


def catalan_sequence(n: int) -> np.ndarray:
    """
    First n Catalan‑like coefficients a_i = C(2i,i)/4**i,
    normalised so that their sum equals 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    # recurrence a_{i+1} = a_i * (2*i+1)/(2*(i+1))
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * (i - 1) + 1) / (2.0 * i)
    a /= a.sum()
    return a


def smoothed_max_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Compute the soft‑max weights w_i = exp(alpha*(conv_i - max(conv))) / Σ.
    This is a numerically stable way to obtain a probability distribution
    that concentrates on the largest convolution entries.
    """
    m = np.max(conv)
    # shift for stability – max becomes zero after the shift
    shifted = alpha * (conv - m)
    # exp of non‑positive numbers, never overflows
    exp_vals = np.exp(shifted)
    return exp_vals / exp_vals.sum()


def gradient_of_smoothed_max(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smoothed max (soft‑max) with respect to a.
    For each j:
        g_j = 2 * Σ_i w_i * a_{i-j}      (indices outside 0..n‑1 are ignored)
    This can be computed via a single convolution.
    """
    n = a.size
    a_rev = a[::-1]                     # reversed a for the correlation step
    # full convolution of w (length 2n‑1) with a_rev (length n)
    conv_wr = np.convolve(w, a_rev, mode='full')
    # the relevant slice is positions n‑1 … n‑1+n‑1
    start = n - 1
    grad = 2.0 * conv_wr[start:start + n]
    return grad


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a non‑negative vector onto the probability simplex.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a sequence of non‑negative heights that minimises
    ``evaluate_sequence``.  The routine runs for at most *budget_s* seconds
    and returns the best sequence found (as a plain Python list).

    The algorithm uses a smooth‑max, exponentiated‑gradient optimiser with
    Adam‑style momentum and occasional restarts.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # a small safety margin

    # Problem size – a few thousand entries give a good trade‑off.
    n = 3000

    # Initialise from the Catalan construction (already good).
    a = catalan_sequence(n)

    # Maintain a log‑parameter vector for the exponentiated update.
    eps = 1e-30
    loga = np.log(a + eps)

    # Adam moment estimates
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    beta1_pow, beta2_pow = 1.0, 1.0

    # Hyper‑parameters
    lr = 0.15          # learning rate in the log‑space
    alpha = 5.0        # initial smoothing factor for the soft‑max
    alpha_inc = 1.001  # increase per iteration (slowly tighten the max)

    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, initial score={best_score:.6f}")

    iteration = 0
    no_improve = 0

    while time.time() < deadline:
        # 1) convolution of current a with itself
        conv = np.convolve(a, a)

        # 2) soft‑max weights focusing on the largest convolution entries
        w = smoothed_max_weights(conv, alpha)

        # 3) gradient of the smoothed objective
        g = gradient_of_smoothed_max(a, w)

        # 4) Adam update on the log‑parameters
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        # 5) exponentiated‑gradient step (log‑space)
        loga -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # 6) map back to the simplex
        a = np.exp(loga)
        a_sum = a.sum()
        if a_sum == 0 or not np.isfinite(a_sum):
            # Degenerate case – fall back to a fresh random point
            a = np.random.rand(n)
            a_sum = a.sum()
        a = a / a_sum

        # 7) evaluate the true objective
        cur_score = evaluate_sequence(a.tolist())

        # 8) keep the best sequence seen so far
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{iteration:06d}] new best = {best_score:.6f}")
            no_improve = 0
        else:
            no_improve += 1

        # 9) schedule updates
        alpha = min(alpha * alpha_inc, 5000.0)   # avoid runaway values
        if iteration % 2000 == 0 and iteration > 0:
            lr *= 0.7    # slow down as we get closer to a plateau

        # 10) occasional random restart if we seem stuck
        if no_improve > 5000:
            a = np.random.rand(n)
            a = a / a.sum()
            loga = np.log(a + eps)
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow, beta2_pow = 1.0, 1.0
            lr = 0.15
            alpha = 5.0
            no_improve = 0
            print(f"[{iteration:06d}] restart (random init)")

        iteration += 1

    print(f"[done] best score found = {best_score:.6f}")
    return best_seq.tolist()
