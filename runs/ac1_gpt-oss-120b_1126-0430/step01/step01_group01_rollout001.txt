"""
Greedy + Simulated Annealing optimization of non‑negative step heights.
Starts from the Catalan (arcsine) seed, performs an aggressive
mass‑relocation greedy refinement using a larger transfer fraction,
then switches to a Metropolis‑Monte‑Carlo annealing that moves
random amounts of mass between pairs of coefficients while keeping the
total ℓ¹‑norm fixed.  All convolutions are computed with FFT for speed.
The best sequence found before the deadline is returned.
"""
import time
import numpy as np
from math import sqrt

def _fast_evaluation(seq: np.ndarray, fft_len: int) -> float:
    """Return the evaluation metric 2*n*max(conv)/(sum(seq)**2)."""
    n = seq.size
    fa = np.fft.rfft(seq, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = seq.sum()
    return float(2.0 * n * max_b / (sum_a * sum_a))

def _catalan_sequence(n: int) -> np.ndarray:
    """Catalan (arcsine) coefficients a_0=1, a_{i}=a_{i-1}*(2*i-1)/(2*i)."""
    seq = np.empty(n, dtype=float)
    a = 1.0
    seq[0] = a
    for i in range(1, n):
        a *= (2 * i - 1) / (2.0 * i)
        seq[i] = a
    return seq

def _greedy_refine(seq: np.ndarray, deadline: float, fraction: float = 0.2) -> np.ndarray:
    """Aggressive greedy mass‑relocation.  Moves a fraction of the
    coefficient that contributes most to the current maximal convolution
    entry to the coefficient that contributes almost nothing.
    """
    n = seq.size
    # pre‑compute FFT length (same for the whole run)
    fft_len = 1 << ((2 * n - 1).bit_length())

    best_seq = seq.copy()
    best_score = _fast_evaluation(best_seq, fft_len)

    while time.time() < deadline:
        # compute convolution and index of the maximal entry
        fa = np.fft.rfft(seq, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        k = int(conv.argmax())
        # contribution weights for each position to that maximal entry
        i_arr = np.arange(n)
        j_arr = k - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        w = np.zeros_like(seq)
        w[mask] = seq[j_arr[mask]]
        # source = index with largest product a_i * w_i (largest contribution)
        contrib = seq * w
        src = int(np.argmax(contrib))
        # sink = index with smallest w (almost zero contribution)
        sink = int(np.argmin(w))

        if seq[src] <= 0.0:
            break

        eps = fraction * seq[src]          # move “fraction” of the source amount
        # make sure we do not move more than the whole source
        eps = min(eps, seq[src])
        if eps < 1e-12:
            break

        new_seq = seq.copy()
        new_seq[src] -= eps
        new_seq[sink] += eps

        # keep the ℓ¹‑norm fixed (scale to the original target)
        target_sum = sqrt(2.0 * n)
        new_seq = new_seq * (target_sum / new_seq.sum())

        score = _fast_evaluation(new_seq, fft_len)
        if score < best_score:
            best_score = score
            best_seq = new_seq.copy()
            seq = new_seq
        else:
            # occasionally try a tiny random jitter to escape a plateau
            if np.random.rand() < 0.01:
                jitter_idx = np.random.randint(n)
                jitter = (np.random.rand() - 0.5) * 0.001 * target_sum
                new_seq = seq.copy()
                new_seq[jitter_idx] = max(0.0, new_seq[jitter_idx] + jitter)
                new_seq = new_seq * (target_sum / new_seq.sum())
                seq = new_seq
    return best_seq

def _simulated_annealing(seq: np.ndarray, deadline: float) -> np.ndarray:
    """Metropolis‑Monte‑Carlo search that swaps small portions of mass
    between two random positions while keeping the ℓ¹‑norm fixed.
    """
    n = seq.size
    fft_len = 1 << ((2 * n - 1).bit_length())
    target_sum = sqrt(2.0 * n)

    cur_seq = seq.copy()
    cur_score = _fast_evaluation(cur_seq, fft_len)
    best_seq = cur_seq.copy()
    best_score = cur_score

    temperature = 1.0
    decay = 0.9996  # slowly cool down

    while time.time() < deadline:
        # propose a move: transfer a random fraction of the source mass to the sink
        src, sink = np.random.randint(n, size=2)
        if src == sink:
            continue
        if cur_seq[src] < 1e-12:
            continue
        # choose epsilon up to 10 % of the source entry
        eps = (np.random.rand() * 0.1) * cur_seq[src]
        new_seq = cur_seq.copy()
        new_seq[src] -= eps
        new_seq[sink] += eps
        # renormalise to keep the total sum constant
        new_seq = new_seq * (target_sum / new_seq.sum())
        new_score = _fast_evaluation(new_seq, fft_len)

        delta = new_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / max(temperature, 1e-12)):
            cur_seq = new_seq
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.copy()
        temperature *= decay
        # keep temperature from vanishing completely
        if temperature < 1e-6:
            temperature = 1e-6

    return best_seq

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights with a low evaluation score.

    The routine uses three stages:
    1.  A Catalan (arcsine) seed that already achieves ≈ π/2.
    2.  An aggressive greedy refinement that moves mass away from the
        indices that dominate the current maximal convolution entry.
    3.  A short simulated‑annealing phase that performs random mass swaps
        while keeping the ℓ¹‑norm fixed.
    The best sequence observed during the run is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0  # a safety margin

    # Choose a length that balances quality and speed.
    # 2500 gives a good trade‑off (≈ 77 KB of data, fast FFTs).
    n = 2500

    # ------------------------------------------------------------------
    # 1. Initial Catalan seed, normalised to the standard total √(2 n).
    # ------------------------------------------------------------------
    seq = _catalan_sequence(n)
    seq = seq * (sqrt(2.0 * n) / seq.sum())

    # ------------------------------------------------------------------
    # 2. Aggressive greedy refinement (first half of the budget).
    # ------------------------------------------------------------------
    half_deadline = time.time() + (budget_s / 2.0) - 5.0
    seq = _greedy_refine(seq, half_deadline, fraction=0.2)

    # ------------------------------------------------------------------
    # 3. Simulated annealing on the remaining time.
    # ------------------------------------------------------------------
    seq = _simulated_annealing(seq, deadline)

    # Return plain Python list – the official `evaluate_sequence` will clip.
    return seq.tolist()

"""
Greedy + Simulated‑Annealing optimisation of non‑negative step heights.

The routine starts from the Catalan (arcsine) seed, which already gives
≈ π/2 ≈ 1.571. It then performs an aggressive greedy mass‑relocation
stage (moving a sizable fraction of the coefficient that most contributes
to the current maximal convolution entry to a coefficient that contributes
almost nothing).  After that a short simulated‑annealing phase swaps small
amounts of mass between random positions while keeping the ℓ¹‑norm fixed.
All convolutions are evaluated with an FFT for speed.  The best sequence
found before the deadline is returned.

import numpy as np
from math import sqrt

# Fast evaluation (identical to evaluate_sequence but without the security
# checks and using an FFT‑based convolution).  The FFT length is fixed for
# the whole run for efficiency.
# ----------------------------------------------------------------------
def _fast_evaluation(seq: np.ndarray, fft_len: int) -> float:
    """Return 2·n·max(conv)/(sum(seq)²) for a real sequence."""
    n = seq.size
    fa = np.fft.rfft(seq, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = seq.sum()
    return float(2.0 * n * max_b / (sum_a * sum_a))

# Catalan (arcsine) seed: a₀=1, a_{i}=a_{i‑1}·(2i‑1)/(2i)
# ----------------------------------------------------------------------
def _catalan_sequence(n: int) -> np.ndarray:
    """First `n` Catalan coefficients as a float array."""
    seq = np.empty(n, dtype=float)
    a = 1.0
    seq[0] = a
    for i in range(1, n):
        a *= (2 * i - 1) / (2.0 * i)
        seq[i] = a
    return seq


# ----------------------------------------------------------------------
# Aggressive greedy refinement.
# ----------------------------------------------------------------------
def _greedy_refine(seq: np.ndarray, deadline: float, fraction: float = 0.2) -> np.ndarray:
    """
    Repeatedly move a `fraction` of the coefficient that contributes most
    to the current maximal convolution entry to a coefficient that contributes
    almost nothing.  Stops when the time limit is reached.
    """
    n = seq.size
    fft_len = 1 << ((2 * n - 1).bit_length())
    target_sum = sqrt(2.0 * n)

    best_seq = seq.copy()
    best_score = _fast_evaluation(best_seq, fft_len)

    while time.time() < deadline:
        # current convolution and its maximal index
        fa = np.fft.rfft(seq, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        k = int(conv.argmax())

    """
    Randomly relocate small amounts of mass between two positions while
    keeping the total sum fixed.  Accepts worsening moves with a probability
    that decreases with a geometric temperature schedule.
