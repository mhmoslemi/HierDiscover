def refine_sequence(seq, target_sum, time_deadline):
    a = np.array(seq, dtype=float)
    best = a.copy()
    best_score = _evaluate(best.tolist())
    temp = 0.1
    decay = 0.9999

    while time.time() < time_deadline:
        # compute conv
        # Use FFT for speed: size = next_pow2(2*n-1)
        size = 1 << (2*n-1).bit_length()
        fa = np.fft.rfft(a, n=size)
        conv = np.fft.irfft(fa * fa, n=size)[:2*n-1]
        max_idx = int(np.argmax(conv))
        # compute contributions
        idx = np.arange(n)
        partner = max_idx - idx
        mask = (partner >= 0) & (partner < n)
        # contributions = a[idx] * a[partner] * mask
        contrib = np.zeros_like(a)
        contrib[mask] = a[idx[mask]] * a[partner[mask]]
        # Choose source and target
        src = int(np.argmax(contrib))
        # choose target among indices with minimal contributions but also maybe lower a
        # we can sample from bottom small values
        # use argmin but avoid src
        # add small random to avoid looping
        # For simplicity:
        tgt = int(np.argmin(contrib + (np.arange(n) == src) * 1e9))
        # compute delta
        delta = min(a[src] * 0.1, target_sum * 0.001)
        if delta <= 1e-12:
            # random small perturb
            i, j = np.random.choice(n, 2, replace=False)
            delta = min(a[i] * 0.01, target_sum*0.0005)
            src, tgt = i, j
        # Move
        a_new = a.copy()
        a_new[src] -= delta
        a_new[tgt] += delta
        # ensure nonnegative; project onto simplex
        if a_new[src] < -1e-12:
            continue
        a_new = _project_to_simplex(a_new, target_sum)
        new_score = _evaluate(a_new.tolist())
        # accept if better, else maybe accept with prob
        if new_score < best_score:
            best = a_new
            best_score = new_score
            a = a_new
            # maybe print improvement
            print(f"[refine] improved to {best_score:.6f}")
        else:
            # annealing
            prob = math.exp(-(new_score - best_score) / (temp + 1e-12))
            if np.random.rand() < prob:
                a = a_new
        temp *= decay

    return best.tolist(), best_score

The score returned by ``evaluate_sequence`` is  

\[
\frac{2n\;\max (a*a)}{(\sum_i a_i)^2}
\]

and the optimiser always normalises the L‑1 norm to  
\(\sum_i a_i=\sqrt{2n}\).  
Consequently the score equals the maximal entry of the self‑convolution
(``max(a*a)``).  
The task is therefore *minimise the largest entry of the convolution*.

The optimiser works in two phases:

1. **Diverse parameter sweep** – a large collection of analytically‑parameterised
   sequences is generated (power‑law, shifted power‑law, exponential,
   logistic, arcsine, half‑sine, random Dirichlet, a mixture of two
   exponentials).  
   Every raw vector is scaled to the required L‑1 norm and its score is
   evaluated.  The best few candidates are retained.

2. **Targeted hill‑climbing** – starting from the current best candidate a
   fast FFT based convolution is used to locate the index *k* of the current
   maximum.  For this index the contribution of each coefficient is
   \(\;c_i = a_i\,a_{k-i}\) (zero when the partner index is out of range).
   The algorithm repeatedly moves a small amount of mass from the index with
   the largest contribution to the index with the smallest contribution.
   Moves are accepted if they lower the maximum; otherwise they are accepted
   with a tiny simulated‑annealing probability.  This focussed move strategy
   reduces the peak far more quickly than random pair swaps.

The whole search respects the 1000 s time budget, prints intermediate
information and finally returns the best sequence found.

```python
"""
propose_candidate
=================

A two‑phase optimiser for the L∞‐norm of a self‑convolution.

Phase 1 – a wide random sweep over analytically‑parameterised families
(power, shifted power, exponential, logistic, arcsine, half‑sine,
random Dirichlet, mixture of exponentials).  Each candidate is scaled to the
required L1‑norm √(2 n) and the best few are remembered.

Phase 2 – targeted hill‑climbing.  At each iteration the convolution is
computed by FFT, the index of the current maximum is located and the
product contributions a[i]·a[k‑i] are evaluated.  Mass is transferred from the
index with the largest contribution to the one with the smallest
contribution (a small fraction of the source).  The move is kept if it
lowers the maximum, otherwise it may be accepted with a decreasing
temperature (simulated annealing).  This directly attacks the active
constraint and converges much faster than naive random swaps.

The function respects the supplied time budget (default 1000 s) and returns
a list of non‑negative heights ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_to_simplex(x: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Project a vector onto the simplex {y >= 0, sum(y) = target_sum}.
    O(n log n) algorithm by Wang & Carreira‑Perpiñán (2013).
    """
    if target_sum <= 0:
        return np.zeros_like(x)

    n = x.shape[0]
    # sort descending
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(x - theta, 0.0)
    return w


def _make_sequence_from_raw(raw: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Scale a non‑negative raw vector so that its L1‑norm equals *target_sum*.
    """
    total = raw.sum()
    if total == 0:
        # fallback to uniform distribution
        return np.full_like(raw, target_sum / raw.shape[0])
    scale = target_sum / total
    return raw * scale


def _objective(a: np.ndarray, target_sum: float) -> float:
    """
    Return the evaluation score for a given sequence *a*.
    Since the sum is already normalised, the score equals max(conv(a,a)).
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    size = 1 << ((conv_len - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:conv_len]
    return float(conv.max())


def _convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a via FFT (real input)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    size = 1 << ((conv_len - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Candidate factories
# ----------------------------------------------------------------------


def _factory_power(n, rng):
    p = rng.uniform(0.05, 3.0)
    raw = (np.arange(1, n + 1, dtype=float)) ** (-p)
    return raw


def _factory_shifted_power(n, rng):
    p = rng.uniform(0.05, 3.0)
    d = rng.uniform(0.0, n * 0.5) + 1e-6
    raw = (np.arange(n, dtype=float) + d) ** (-p)
    return raw


def _factory_exponential(n, rng):
    p = rng.uniform(0.001, 0.5)
    raw = np.exp(-p * np.arange(n, dtype=float))
    return raw


def _factory_logistic(n, rng):
    cx = rng.uniform(0.2, 0.8) * (n - 1)
    scale = rng.uniform(0.05, 0.3) * n
    x = np.arange(n, dtype=float)
    raw = 1.0 / (1.0 + np.exp((x - cx) / scale))
    return raw


def _factory_arcsine(n, rng):
    # a_i ∝ 1/√(i·(n+1-i))  (i starts at 1)
    i = np.arange(1, n + 1, dtype=float)
    raw = 1.0 / np.sqrt(i * (n + 1 - i))
    return raw


def _factory_half_sine(n, rng):
    raw = np.sin(np.pi * (np.arange(n, dtype=float) + 0.5) / n)
    raw = np.maximum(raw, 0.0)
    return raw


def _factory_random_dirichlet(n, rng):
    alpha = rng.uniform(0.1, 2.0)
    raw = rng.gamma(shape=alpha, scale=1.0, size=n)
    return raw


def _factory_mix_exponential(n, rng):
    p1 = rng.uniform(0.001, 0.5)
    p2 = rng.uniform(0.001, 0.5)
    w = rng.uniform(0.0, 1.0)
    raw = w * np.exp(-p1 * np.arange(n, dtype=float)) + \
          (1.0 - w) * np.exp(-p2 * np.arange(n, dtype=float))
    return raw


_FACTORY_FUNCS = [
    _factory_power,
    _factory_shifted_power,
    _factory_exponential,
    _factory_logistic,
    _factory_arcsine,
    _factory_half_sine,
    _factory_random_dirichlet,
    _factory_mix_exponential,
]


# ----------------------------------------------------------------------
# Targeted hill‑climbing refinement
# ----------------------------------------------------------------------


def _refine_sequence(
    seq: np.ndarray,
    target_sum: float,
    time_deadline: float,
    init_temp: float = 0.07,
    decay: float = 0.9997,
) -> tuple[np.ndarray, float]:
    """
    Reduce the maximal convolution entry by moving mass from the most
    contributing coordinate to the least contributing one.
    """
    a = seq.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _objective(best, target_sum)
    temp = init_temp

    while time.time() < time_deadline:
        # full convolution (FFT)
        conv = _convolve_full(a)
        k = int(conv.argmax())                       # index of current max
        # contribution of each coefficient to that max entry
        idx = np.arange(n)
        partner = k - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[idx[mask]] * a[partner[mask]]

        # source = index with largest contribution, target = smallest
        src = int(np.argmax(contrib))
        tgt = int(np.argmin(contrib + (np.arange(n) == src) * 1e9))

        # amount to transfer (5 % of source, limited)
        delta = min(a[src] * 0.05, target_sum * 0.001)
        if delta <= 0:
            # fallback to a tiny random move
            i, j = np.random.choice(n, 2, replace=False)
            src, tgt = int(i), int(j)
            delta = min(a[src] * 0.02, target_sum * 0.0005)
        # perform the move
        a_new = a.copy()
        a_new[src] -= delta
        a_new[tgt] += delta
        # safety: keep non‑negative (project only if necessary)
        if np.any(a_new < -1e-12):
            continue
        # note: sum is preserved, so projection is not required here
        new_score = _objective(a_new, target_sum)

        # acceptance rule
        if new_score < best_score:
            best = a_new
            best_score = new_score
            a = a_new
            print(f"[refine] improved to {best_score:.6f}")
        else:
            prob = math.exp(-(new_score - best_score) / (temp + 1e-12))
            if np.random.rand() < prob:
                a = a_new
        temp *= decay

        # occasional random perturbation to avoid stagnation
        if np.random.rand() < 0.001:
            i, j = np.random.choice(n, 2, replace=False)
            eps = target_sum * 0.0003
            a[i] = max(0.0, a[i] + eps)
            a[j] = max(0.0, a[j] - eps)
            a = _project_to_simplex(a, target_sum)

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) yields a low
    evaluation score.  The function respects the time budget ``budget_s``.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # problem size
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
        start_seq = np.array(height_sequence_1, dtype=float)
        # make sure it obeys the L1‑norm requirement
        target_sum = math.sqrt(2.0 * n)
        start_seq = _make_sequence_from_raw(start_seq, target_sum)
    else:
        n = 5000                     # a safe default if nothing is provided
        target_sum = math.sqrt(2.0 * n)
        start_seq = np.full(n, target_sum / n, dtype=float)

    best_seq = start_seq.copy()
    best_score = _objective(best_seq, target_sum)
    print(f"[init] n={n}, start score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – diverse parameter sweep
    # --------------------------------------------------------------
    sweep_deadline = time.time() + budget_s * 0.45   # ~45 % of total time
    while time.time() < sweep_deadline:
        factory = rng.choice(_FACTORY_FUNCS)
        raw = factory(n, rng)
        # enforce non‑negativity (just in case)
        raw = np.maximum(raw, 0.0)
        cand = _make_sequence_from_raw(raw, target_sum)
        score = _objective(cand, target_sum)
        if score < best_score:
            best_score = score
            best_seq = cand
            print(f"[sweep] new best {best_score:.6f} via {factory.__name__}")

    # --------------------------------------------------------------
    # Phase 2 – targeted hill‑climbing / simulated annealing
    # --------------------------------------------------------------
    refine_deadline = time.time() + budget_s * 0.95   # leave a few seconds margin
    best_seq, best_score = _refine_sequence(best_seq, target_sum, refine_deadline)

    print(f"[final] best score found: {best_score:.6f}")
    # convert to plain Python list (as required by the external evaluator)
    return best_seq.tolist()
```
