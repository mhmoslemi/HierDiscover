def proj_simplex(v):
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u > cssv / np.arange(1, n+1))[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def convolve_fft(x, y):

Propose a candidate step‑function (list of non‑negative heights) that tries to
minimise the supplied ``evaluate_sequence`` function.

Algorithm
---------
1. Choose a moderate problem size (default 8 000).  
2. Build a small pool of initial candidates:
   * the Catalan‑like seed,
   * a family of geometric (exponential) decays with different decay rates,
   * a random simplex point.
   The best of these (according to the exact objective 2·n·max(conv)) is kept.
3. Perform a projected “smooth‑max” gradient descent:
   * compute the convolution b = a∗a,
   * form soft‑max weights w_k ∝ exp(α·(b_k−max(b)))  (α controls how
     sharply we focus on the current maximal entry),
   * the gradient of the smooth max is

   * take a step a←a−η·g and project back onto the probability simplex
     (non‑negative, sum = 1) by the Euclidean‑projection algorithm of
     Duchi et al.
   * simple back‑tracking line‑search guarantees monotone improvement.
4. The smoothing parameter α is increased slowly so that the optimiser
   gradually concentrates on the true maximum; the step size η is
   adaptively reduced when no improvement is seen.
5. If the search stalls for many iterations a small random perturbation
   around the best point is performed.
6. The loop runs until the time budget (≈ 1000 s) is exhausted; the best
   sequence found is returned as a plain list of floats.

The objective is scale‑invariant, hence we always keep the sequence on the
simplex (∑a_i=1).  All heavy work (convolutions) is performed with FFTs,

def _catalan_sequence(n: int) -> np.ndarray:
    """First n Catalan‑like coefficients a_i = C(2i,i)/4**i, normalised to sum 1."""
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    # recurrence: a_{i+1} = a_i * (2*i+1) / (2*(i+1))
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * (i - 1) + 1) / (2.0 * i)
    a /= a.sum()
    return a


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, sum=1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    # Fast projection (Duchi et al., 2008)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u > cssv / np.arange(1, n + 1))[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of real vectors x and y using a real FFT."""
    n = x.size + y.size - 1
    size = 1 << (n - 1).bit_length()      # next power of two
    X = np.fft.rfft(x, size)
    Y = np.fft.rfft(y, size)
    Z = X * Y
    conv = np.fft.irfft(Z, size)[:n]
    return conv.real


def _eval_score(a: np.ndarray) -> float:
    """Exact value of evaluate_sequence for a non‑negative simplex vector a."""
    n = a.size
    b = _convolve_fft(a, a)
    max_b = b.max()
    return float(2.0 * n * max_b / (a.sum() ** 2))

            alpha = 10.0
            no_improve = 0
            print(f"[{iteration:06d}] restart around best (score={cur_score:.6f})")

        iteration += 1

    print(f"[done] best score found = {best_score:.6f}")
    # return as plain Python list, as required by the harness
    return best_seq.tolist()
