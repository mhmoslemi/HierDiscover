def evaluate_sequence(sequence):

def gradient_refine_local(a, deadline, max_iters=100):
    n = len(a)
    target_const = (np.sum(a)**2) / (2*n - 1)
    step_factor = 0.05
    best = a.copy()

"""
Propose a low‑score step function.

The algorithm combines three ideas:

1.  Start from a simple parametric family (power‑law) that already gives a
    good score (~1.58 for n≈3000).
2.  Refine the shape with a *smooth‑max* surrogate:
        Φ(a) = (1/γ)·log Σ exp(γ·(a*a)),
    whose gradient is
        ∇Φ(a) = 2·corr(softmax(γ·(a*a)), a) .
    An exponentiated‑gradient step
        a ← a·exp(−η·∇Φ(a))
    keeps the vector non‑negative, and a simple ℓ₁‑renormalisation enforces
    Σa=1.  The smoothing parameter γ is increased gradually (γ≈5…200) so
    that the optimiser first flattens the whole convolution and later
    focuses on the true maximum.
3.  Finish with a few projected‑gradient steps that minimise the L² error
    between the convolution and its constant target
        (Σa)²/(2n−1).
    This sharpens the solution for the final exact score.

The whole optimisation runs inside the supplied time budget (≈1000 s)
and keeps the best sequence seen.  All operations use FFT‑based
convolutions, so even with n≈4000 the loop can perform many thousands of
iterations within the limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution utilities (FFT for speed)
def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using a power‑of‑two FFT."""
    size = len(a) + len(b) - 1
    N = 1 << (size - 1).bit_length()          # next power of two
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    C = A * B
    conv = np.fft.irfft(C, N)[:size]
    return conv


def correlate_valid(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute the “valid” cross‑correlation of w (length 2n‑1) with a (length n):
    result[i] = Σ_j w[i+j]·a[j]   for i=0…n‑1.
    Implemented via a single FFT convolution of w with the reversed a.
    """
    a_rev = a[::-1]
    conv = convolve_fft(w, a_rev)                     # length 3n‑2
    start = len(a) - 1
    return conv[start:start + len(a)]


# ----------------------------------------------------------------------
# Fast objective (exact evaluation without the security checks)
def fast_score(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) / (sum(a)²)  –  the quantity we want to minimise."""
    n = len(a)
    b = convolve_fft(a, a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 1e-12:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Simple parametric families for initial guesses
def power_law_sequence(alpha: float, delta: float, n: int) -> np.ndarray:
    """
    a_i ∝ (i+1+δ)^{-α}, i = 0 … n‑1   (the +1 avoids division by zero).
    The sequence is normalised to sum‑one.
    """
    idx = np.arange(1.0, n + 1, dtype=np.float64) + delta
    a = idx ** (-alpha)
    s = a.sum()
    if s > 0.0:
        a /= s
    return a


# ----------------------------------------------------------------------
# Smooth‑max based refinement (exponentiated gradient)
def smoothmax_update(a: np.ndarray, gamma: float, eta: float) -> np.ndarray:
    """
    One optimisation step that reduces the smooth‑max of the self‑convolution.
    - gamma: smoothing parameter for the soft‑max (larger → closer to true max)
    - eta  : step size for the exponentiated‑gradient update
    """
    # self‑convolution
    b = convolve_fft(a, a)                     # length 2n‑1
    max_b = b.max()
    # stable soft‑max weights (they sum to 1)
    exp_term = np.exp(gamma * (b - max_b))
    w = exp_term / exp_term.sum()
    # gradient of the smooth‑max (valid correlation)
    grad = 2.0 * correlate_valid(w, a)         # length n, non‑negative
    # exponentiated‑gradient update, staying in the positive orthant
    factor = np.exp(-eta * grad)
    a_new = a * factor
    s = a_new.sum()
    if s == 0.0:
        a_new = np.full_like(a, 1.0 / len(a))
    else:
        a_new /= s
    return a_new


# ----------------------------------------------------------------------
# Fine‑grained L2‑projection (same idea as the original author’s gradient_refine)
def l2_projection_refine(a: np.ndarray, deadline: float,
                         max_iters: int = 200) -> np.ndarray:
    """
    Minimise ‖conv(a,a) – const‖₂² by projected gradient steps.
    The constant is the theoretical average of the convolution (Σa)²/(2n‑1).
    """
    n = len(a)
    target_const = (a.sum() ** 2) / (2 * n - 1)
    step_factor = 0.05
    best = a.copy()
    best_sc = fast_score(a)

    for _ in range(max_iters):
        if time.time() >= deadline:
            break
        b = convolve_fft(a, a)
        err = b - target_const
        # gradient of the L2 error: 4·corr(err, a)
        grad = 4.0 * correlate_valid(err, a)
        gn = np.linalg.norm(grad)
        if gn == 0.0:
            break
        eta = step_factor / gn
        a = a - eta * grad
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= s
        cur_sc = fast_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best = a.copy()
            # gradually tighten step size
            step_factor = max(step_factor * 0.95, 1e-4)
    return best


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function with a low evaluation score.
    The routine respects the supplied time budget (≈1000 s) and returns a
    plain Python list of non‑negative floats.  It starts from a power‑law
    candidate, repeatedly applies the smooth‑max optimiser (with an
    annealing schedule for γ and η) and finishes with a short L₂‑projection.
    The best sequence encountered during the run is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Sequence length – a few thousand gives a good trade‑off between
    # expressiveness and speed.
    n = 4000

    # ------------------------------------------------------------------
    # Gather a few starting points (uniform, several power‑laws, Dirichlet)
    starters = []

    # uniform
    starters.append(np.full(n, 1.0 / n))

    # a few hand‑crafted power‑laws (the region α≈0.5 was promising)
    for alpha in (0.48, 0.505, 0.53):
        for delta in (0.0, 0.5):
            starters.append(power_law_sequence(alpha, delta, n))

    # random Dirichlet draws (various concentrations)
    for conc in (0.2, 0.5, 1.0):
        starters.append(np.random.dirichlet(np.full(n, conc)))

    # if a previous solution is available, use it (trim / pad if needed)
    try:
        # height_sequence_1 may live in the global namespace
        hs = globals().get('height_sequence_1')
        if hs is not None:
            hs_arr = np.asarray(hs, dtype=float)
            if hs_arr.size >= n:
                starters.append(hs_arr[:n].copy())
            else:
                pad = np.full(n - hs_arr.size, 1e-8)
                starters.append(np.concatenate([hs_arr, pad]))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Pick the best of the initial candidates
    best_seq = None
    best_sc = float('inf')
    for a in starters:
        sc = fast_score(a)
        if sc < best_sc:
            best_sc = sc
            best_seq = a.copy()
    print(f'initial best score = {best_sc:.6f}')

    # ------------------------------------------------------------------
    # Main optimisation loop – smooth‑max + L₂ polishing
    iteration = 0
    while time.time() < deadline:
        # Occasionally restart from a fresh random Dirichlet to escape a
        # possible local basin; otherwise continue refining the current best.
        if iteration % 8 == 0 and iteration > 0:
            a = np.random.dirichlet(np.full(n, 0.5))
        else:
            a = best_seq.copy()

        # Parameters for the smooth‑max stage.
        gamma = 5.0            # start with a gentle smoothing
        eta = 0.5              # fairly aggressive step size

        # Run a short inner optimisation (the number of steps adapts to the
        # remaining time).  Each step costs only a few FFTs.
        inner_steps = 250
        for _ in range(inner_steps):
            if time.time() >= deadline:
                break
            a = smoothmax_update(a, gamma, eta)
            # gradual annealing of the hyper‑parameters
            gamma = min(gamma * 1.07, 200.0)   # approach the true max
            eta = max(eta * 0.97, 0.01)       # slow down as we near convergence

        # Fine‑tune with the L² projection (few deterministic steps)
        if time.time() < deadline:
            a = l2_projection_refine(a, deadline, max_iters=120)

        # Assess the candidate
        cur_score = fast_score(a)
        if cur_score < best_sc:
            best_sc = cur_score
            best_seq = a.copy()
            print(f'global improvement #{iteration}: score = {best_sc:.6f}')

        iteration += 1

    # ------------------------------------------------------------------
    # Final sanity check with the provided evaluate_sequence (if present)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f'final reported score = {final_score:.6f}')
    except Exception:
        # evaluate_sequence may not be imported in the local context – ignore.
        pass

    # Return a plain Python list as required by the evaluator.
    return best_seq.tolist()
