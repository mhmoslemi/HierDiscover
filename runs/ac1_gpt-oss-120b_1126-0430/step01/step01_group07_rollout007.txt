def convolve_fft(a, b):
    n = len(a) + len(b) - 1
    N = 1 << (n - 1).bit_length()
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    C = A * B
    conv = np.fft.irfft(C, N)[:n]
    return conv

def spectral_sqrt_sequence(n):
    L = 2 * n - 1
    # uniform target convolution values
    u = np.full(L, 1.0 / L, dtype=np.float64)
    N = 1 << (L - 1).bit_length()
    # Zero-padded arrays
    u_padded = np.zeros(N, dtype=np.float64)
    u_padded[:L] = u
    # FFT
    U = np.fft.fft(u_padded)
    sqrt_U = np.sqrt(U)
    a_complex = np.fft.ifft(sqrt_U).real[:n]
    a = np.maximum(a_complex, 0.0)
    total = a.sum()
    if total <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a

def max_subgradient_refine(a, deadline, max_iter=5000):
    n = len(a)
    step_factor = 0.1
    best = a.copy()
    best_score = fast_score(a)
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        b = convolve_fft(a, a)
        max_b = b.max()
        kmax = np.argmax(b)
        # compute gradient
        grad = np.zeros_like(a)
        # iterate i where kmax - i in [0,n-1]
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)  # inclusive
        # Using vectorized: indices = np.arange(i_start, i_end+1); grad[indices] = 2 * a[kmax - indices]
        idx = np.arange(i_start, i_end + 1)
        grad[idx] = 2.0 * a[kmax - idx]
        gn = np.linalg.norm(grad)
        if gn == 0:
            break
        eta = step_factor / gn
        a = a - eta * grad
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0:
            a = np.full(n, 1.0 / n)
        else:
            a /= s
        cur = fast_score(a)
        if cur < best_score:
            best_score = cur
            best = a.copy()
            # reduce step factor for stability
            step_factor = max(step_factor * 0.95, 1e-5)
    return best

"""
Propose a low‑score step‑function.

Algorithm
---------
1.  Choose several problem sizes (n).  For each n we generate a few
    promising start vectors:
        * uniform (baseline)
        * a “spectral‑sqrt” vector – the inverse FFT of the square‑root
          of the DFT of a flat target convolution (this is the
          L²‑optimal solution of a * a ≈ constant).
        * power‑law shapes a_i ∝ (i+δ)^‑α (grid search around the
          previously known good region).
        * simple exponential decays a_i ∝ exp(‑λ·i).
        * a few random Dirichlet draws.

2.  Each start vector is refined by a fast projected‑gradient method
    that minimises the L²‑error of the self‑convolution to the flat
    target (the target value is 1/(2n‑1) after normalising the sum to
    1).  The iteration projects onto the non‑negative orthant and
    renormalises after each step.  The best vector seen during the
    optimisation is kept.

3.  An additional short sub‑gradient phase directly targets the
    L∞‑objective: in every iteration the gradient of the currently
    maximal convolution entry is subtracted (the gradient is simply a
    reversed copy of the current vector).  The same projection/normal
    step is applied.

4.  All candidates are evaluated with a cheap FFT‑based scoring routine
    `fast_score`.  The overall best sequence (minimum score) is returned
    as a plain Python list.

The whole search respects a wall‑clock budget (default 1000 s) and
stops cleanly when the time limit is reached.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast linear convolution using a power‑of‑two FFT (real input).
def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    size = len(a) + len(b) - 1
    N = 1 << (size - 1).bit_length()          # next power of two
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    C = A * B
    conv = np.fft.irfft(C, N)[:size]
    return conv


# ----------------------------------------------------------------------
# Objective used during the search (scaling invariant, sum=1 → score = 2·n·max(b)).
def fast_score(a: np.ndarray) -> float:
    """Return the evaluation score for a normalised non‑negative vector."""
    n = len(a)
    b = convolve_fft(a, a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 1e-12:
        return float('inf')
    # a is intended to be normalised, but keep the formula generic.
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Candidate generators ---------------------------------------------------
def power_law_sequence(alpha: float, delta: float, n: int) -> np.ndarray:
    """a_i ∝ (i+δ)^‑α for i=1…n, normalised to sum‑one."""
    idx = np.arange(1.0, n + 1, dtype=np.float64) + delta
    a = idx ** (-alpha)
    total = a.sum()
    if total > 0.0:
        a /= total
    else:
        a = np.full(n, 1.0 / n)
    return a


def exponential_sequence(lam: float, n: int) -> np.ndarray:
    """a_i ∝ exp(-λ·i) for i=0…n‑1, normalised to sum‑one."""
    idx = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * idx)
    total = a.sum()
    if total > 0.0:
        a /= total
    else:
        a = np.full(n, 1.0 / n)
    return a


def spectral_sqrt_sequence(n: int) -> np.ndarray:
    """
    Construct a vector whose *self‑convolution* approximates a flat target.
    It is obtained by taking the inverse FFT of the square‑root of the
    DFT of a uniform vector of length 2·n‑1 (the desired convolution).
    """
    L = 2 * n - 1
    # uniform target convolution (probability vector)
    u = np.full(L, 1.0 / L, dtype=np.float64)

    # next power of two for efficient FFTs
    N = 1 << (L - 1).bit_length()

    u_pad = np.zeros(N, dtype=np.float64)
    u_pad[:L] = u

    U = np.fft.fft(u_pad)
    sqrt_U = np.sqrt(U)                     # complex square‑root
    a_complex = np.fft.ifft(sqrt_U).real[:n]

    # enforce non‑negativity and renormalise
    a = np.maximum(a_complex, 0.0)
    total = a.sum()
    if total > 0.0:
        a /= total
    else:
        a = np.full(n, 1.0 / n)
    return a


# ----------------------------------------------------------------------
# Projected gradient refinement (L2‑error to flat target)
def gradient_refine(a: np.ndarray,
                    deadline: float,
                    max_iter: int = 2000) -> np.ndarray:
    """
    Perform a few projected‑gradient steps that minimise
    ∑_k ( (a*a)_k – 1/(2n‑1) )².
    The routine stops when time runs out or after max_iter steps.
    """
    n = len(a)
    target_const = 1.0 / (2 * n - 1)        # because sum(a) is kept = 1
    step_factor = 0.1
    best = a.copy()
    best_score = fast_score(a)

    it = 0
    while time.time() < deadline and it < max_iter:
        b = convolve_fft(a, a)
        err = b - target_const

        # gradient of the L2 loss w.r.t. a (see derivation in the comment)
        grad = 4.0 * np.correlate(err, a, mode='valid')
        gn = np.linalg.norm(grad)
        if gn == 0.0:
            break

        eta = step_factor / gn
        a = a - eta * grad

        # projection onto the non‑negative orthant + renormalisation
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        cur = fast_score(a)
        if cur < best_score:
            best_score = cur
            best = a.copy()
            # gradually shrink the step‑size after a successful improvement
            step_factor = max(step_factor * 0.94, 1e-5)
        else:
            # if no improvement, modestly increase the step‑size to escape plateaus
            step_factor = min(step_factor * 1.02, 0.5)

        it += 1
    return best


# ----------------------------------------------------------------------
# Sub‑gradient refinement that directly attacks the L∞‑objective
def max_subgradient_refine(a: np.ndarray,
                           deadline: float,
                           max_iter: int = 1500) -> np.ndarray:
    """
    Reduce max(a*a) by following the sub‑gradient of the maximal
    convolution entry.  In each iteration the gradient is a reversed
    copy of a shifted according to the index where the maximum occurs.
    """
    n = len(a)
    step_factor = 0.1
    best = a.copy()
    best_score = fast_score(a)

    it = 0
    while time.time() < deadline and it < max_iter:
        b = convolve_fft(a, a)
        kmax = int(b.argmax())

        # compute sub‑gradient of the maximal entry
        grad = np.zeros_like(a)
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)                # inclusive
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1, dtype=int)
            grad[idx] = 2.0 * a[kmax - idx]

        gn = np.linalg.norm(grad)
        if gn == 0.0:
            break

        eta = step_factor / gn
        a = a - eta * grad
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        cur = fast_score(a)
        if cur < best_score:
            best_score = cur
            best = a.copy()
            step_factor = max(step_factor * 0.93, 1e-6)
        else:
            step_factor = min(step_factor * 1.03, 0.3)

        it += 1
    return best

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises the

        print(f'\n=== exploring length n = {n} ===')

        # 2. Spectral‑sqrt construction (theoretically L2‑optimal)

        # refine it with L2 gradient steps

                pl = power_law_sequence(alpha, delta, n)

                # occasionally run a light refinement on promising candidates

                    update_best(pl_ref, f'n={n} pow‑law‑ref α={alpha:.3f} δ={delta}')

        # 4. Simple exponential decay family
        lambdas = np.linspace(0.0002, 0.0080, 20)
        for lam in lambdas:

    # Final sanity check with the official evaluator (if it is available)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f'\nFINAL reported score = {final_score:.6f}')
    except Exception:
        # The caller may have omitted the import; we silently ignore.
        pass

    # Ensure the returned type matches the specification.
    return best_seq.tolist()
