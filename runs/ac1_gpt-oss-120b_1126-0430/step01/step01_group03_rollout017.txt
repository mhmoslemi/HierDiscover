def simplex_projection(v):
    # ensures v>=0 and sum=1.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a high‑quality step‑function sequence.

The evaluation is 2·n·max_k(p∗p) where p_i = a_i/∑a_j.
We explore a richer family of power‑law sequences
a_i ∝ (i+1+offset)^−β·exp(−γ·i) and keep the best
candidate found by exhaustive grid search (different lengths,
β, offset, exponential decay).  From the best candidate we
run a simple sub‑gradient descent on the active constraints
(the indices where the self‑convolution attains its maximum).
The update direction is the exact gradient of the maximal
convolution term, projected onto the probability simplex.
Occasional random jitter helps escaping very flat plateaus.
Finally trailing near‑zero entries are trimmed (they only enlarge
the factor n without influencing the maximum convolution) and
the best sequence is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x ≥ 0, sum x = 1}
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex."""
    n = v.shape[0]
    # fast path
    if v.min() >= 0 and abs(v.sum() - 1.0) < 1e-12:
        return v.copy()
    u = np.sort(v)[::-1]                     # sort descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------

        return evaluate_sequence(seq)
    except Exception:
        return float("inf")

    betas = np.linspace(0.40, 0.80, 41)           # step 0.01

    for n in lengths:
