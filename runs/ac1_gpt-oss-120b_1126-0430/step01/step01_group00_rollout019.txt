def exponential_sequence(n, r):
    if r <=0:
        raise...
    idx = np.arange(n)
    log_r = np.log(r)
    # Use exp for each idx: exp(idx*log_r)
    # Compute denominator = 1 - r**n -> use np.exp(n * log_r) and use np.expm1 for stability
    r_pow_n = np.exp(n * log_r)
    denom = 1 - r_pow_n
    if denom == 0:
        # treat as uniform: a = np.full(n, 1.0/n)
        a = np.full(n, 1.0/n)
    else:
        num = 1 - r
        c = num / denom
        a = c * np.exp(log_r * idx)
    return a
```

Search for a step‑function (non‑negative sequence) that minimizes

    O(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

The evaluation is scale‑invariant, so we work on the probability simplex
(sum = 1).  An exponential profile a_i ∝ r**i (0<r<1) gives a smooth
family that already beats the trivial uniform bound (O=2).  By scanning
the decay parameter r (or equivalently ε = 1−r) we locate a very good

All heavy work (convolution) is done with an FFT for speed.

# Projection onto the probability simplex (Duchi et al.)
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x >= 0, sum(x) = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# -------------------------------------------------------------
def _conv_full(a: np.ndarray, L: int) -> np.ndarray:
    """Full convolution a*a using an FFT of length L (power of two)."""
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * a.size - 1]
    return conv


def _objective_fast(a: np.ndarray, L: int) -> float:
    """Fast evaluation of O(a) for a vector with sum ≈ 1."""
    n = a.size
    conv = _conv_full(a, L)
    max_b = conv.max()
    sum_a = a.sum()
    # guard against accidental near‑zero sums (should not happen)
    if sum_a < 1e-12:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))

def _exponential_sequence(n: int, r: float) -> np.ndarray:
    """
    Return a non‑negative vector of length n with elements
        a_i = c * r**i ,  0 < r < 1,
    normalised so that sum(a)=1.
    """

        # uniform vector – fallback (should never be called with r≈1)
        return np.full(n, 1.0 / n)

    log_r = np.log(r)                # negative
    idx = np.arange(n, dtype=np.float64)

    # 1 - r  and 1 - r**n computed with expm1 for stability
    numer = -np.expm1(log_r)                     # = 1 - r

    # protect against degenerate denominator (r≈1 => uniform case)
    if denom == 0.0:
        return np.full(n, 1.0 / n)

    c = numer / denom
    a = c * np.exp(log_r * idx)
    return a

        # Re‑project onto the simplex (maintains sum=1)
        a = _project_to_simplex(a)
        score = _objective_fast(a, L)
        if score < best_score:
            best_score = score
            best_seq = a
            print(f"[{time.time():.1f}] Local improvement O={best_score:.6f}")
        else:
            # No improvement; exit the hill‑climb loop
            break

    # ---------------------------------------------------------
    # Final verification with the provided evaluate_sequence (optional)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as err:
        print("[final] evaluate_sequence call failed:", err)

    return best_seq.tolist()
