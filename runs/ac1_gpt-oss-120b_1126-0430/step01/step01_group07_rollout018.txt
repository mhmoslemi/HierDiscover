def project_onto_simplex(v):
    # v can be any real numbers
    sorted_v = np.sort(v)[::-1]
    cssv = np.cumsum(sorted_v) - 1
    rho = np.where(sorted_v - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def max_refine(a, deadline):
    n = len(a)
    iter = 0
    beta = 5.0
    step0 = 1.0
    best = a.copy()
    best_score = fast_score(a)
    while time.time() < deadline:
        iter += 1
        conv = convolve_fft(a, a)
        max_b = conv.max()
        # dynamic beta: increase slowly to focus more?
        # maybe beta = min(200, 5 + np.log(iter+1)*10)
        beta = min(200.0, beta * 1.02)  # gradually increase
        w = np.exp(beta * (conv - max_b))
        # ensure numeric stability: shift by max_b (already done)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(w) / len(w)
        else:
            w /= w_sum
        # compute gradient via correlation
        # correlation: valid part only
        # reverse w for convolution
        w_rev = w[::-1]
        grad_full = convolve_fft(w_rev, a)
        start = n - 1
        grad = 2.0 * grad_full[start:start + n]
        # step size schedule
        step = step0 / np.sqrt(iter)
        # backtracking line search
        improved = False
        for _ in range(10):
            a_new = a - step * grad
            a_new = project_onto_simplex(a_new)
            new_score = fast_score(a_new)
            if new_score < best_score:
                best_score = new_score
                best = a_new.copy()
                a = a_new
                improved = True
                if iter % 20 == 0:
                    print(f'iter {iter}, beta {beta:.2f}, step {step:.4e}, score {new_score:.6f}')
                break

                step *= 0.5

            # maybe break or continue with smaller step

import time, numpy as np

def convolve_fft(a,b):
    size = len(a) + len(b) - 1
    N = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    C = A * B
    conv = np.fft.irfft(C, N)[:size]
    return conv

def fast_score(a):
    n = len(a)
    b = convolve_fft(a, a)
    return 2.0 * n * b.max()

def project_simplex(v):
    if np.all(v <= 0):
        # if all nonpositive, put uniform distribution
        w = np.full_like(v, 1.0 / len(v))
        return w
    # Implementation of efficient Euclidean projection onto simplex
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def power_law_sequence(alpha, delta, n):
    idx = np.arange(1.0, n+1) + delta
    a = idx ** (-alpha)
    s = a.sum()
    if s != 0:
        a /= s
    return a

def l2_refine(a, deadline, step_factor=0.1):
    # similar to earlier gradient_refine but maybe more steps
    n = len(a)
    target_const = (a.sum() ** 2) / (2 * n - 1)
    best = a.copy()
    best_score = fast_score(a)
    while time.time() < deadline:
        b = convolve_fft(a, a)
        err = b - target_const
        # gradient of L2 error: 4 * correlate(err, a)
        # Using FFT for correlation
        err_rev = err[::-1]
        grad_full = convolve_fft(err_rev, a)
        start = n - 1
        grad = 4.0 * grad_full[start:start + n]
        norm = np.linalg.norm(grad)
        if norm == 0:
            break
        eta = step_factor / norm
        a_new = a - eta * grad
        a_new = project_simplex(a_new)
        cur_score = fast_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best = a_new.copy()
            # optionally reduce step factor
            step_factor *= 0.95
        a = a_new
    return best

def max_refine(a, deadline, step_factor=0.5, beta_start=5.0, beta_growth=1.02):
    n = len(a)
    best = a.copy()
    best_score = fast_score(a)
    iter_num = 0
    beta = beta_start
    while time.time() < deadline:
        iter_num += 1
        conv = convolve_fft(a, a)
        max_b = conv.max()
        # dynamic beta
        beta = min(200.0, beta * beta_growth)
        # compute weights

        w_sum = w.sum()
        if w_sum == 0:

        else:
            w /= w_sum
        # gradient = 2 * correlate(w, a)

        grad_full = convolve_fft(w_rev, a)

The algorithm works in three stages:

1.  Scan a two‑parameter power‑law family
        a_i ∝ (i + δ)^(-α) ,   i = 1 … n
    on a coarse and then a fine grid.  The best candidate from this
    family is used as a warm start.

2.  Perform a projected L2‑flattening refinement that drives the
    self‑convolution towards its average value (the “flat target”).

3.  Finish with a projected gradient descent on a smoothed version of the
    supremum (soft‑max) of the convolution.  This directly reduces the
    max‑entry of the convolution.

All updates keep the vector non‑negative and re‑normalise it onto the
probability simplex, which respects the scaling invariance of the score.
The routine runs entirely inside the given time budget (default 1000 s)
and returns the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast linear convolution using a power‑of‑two FFT (real inputs)
def convolve_fft(a, b):
    """Linear convolution of two real 1‑D arrays using FFT."""
    size = len(a) + len(b) - 1
    N = 1 << (size - 1).bit_length()            # next power of two
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    C = A * B
    conv = np.fft.irfft(C, N)[:size]
    return conv


# ----------------------------------------------------------------------
# Score used throughout the search (the true objective of evaluate_sequence)
def fast_score(a):
    """Return 2·n·max(conv(a,a)) / (sum(a)^2). Assumes a is a NumPy array."""
    n = len(a)
    b = convolve_fft(a, a)
    max_b = b.max()
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex (non‑negative, sum = 1)
def project_onto_simplex(v):
    """Return the Euclidean projection of v onto the simplex."""
    if np.all(v <= 0):
        # if everything is non‑positive, the uniform distribution is the projection
        return np.full_like(v, 1.0 / len(v))

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Power‑law family a_i ∝ (i + δ)^(-α) (i starts at 1)
def power_law_sequence(alpha, delta, n):
    idx = np.arange(1.0, n + 1) + delta          # i = 1 … n
    a = idx ** (-alpha)
    s = a.sum()
    if s != 0.0:
        a /= s
    return a


# ----------------------------------------------------------------------
# L2‑flattening refinement (the same idea as the original gradient_refine)
def l2_refine(a, deadline, step_factor=0.12):
    """Minimise ‖conv(a,a)‑target‖₂ via projected gradient steps."""
    n = len(a)
    target_const = (a.sum() ** 2) / (2 * n - 1)   # average value of the convolution
    best = a.copy()
    best_score = fast_score(a)

    while time.time() < deadline:
        b = convolve_fft(a, a)
        err = b - target_const                      # error vector
        # gradient of ½‖err‖₂² w.r.t. a: 4 · correlate(err, a) (valid mode)
        err_rev = err[::-1]
        grad_full = convolve_fft(err_rev, a)
        start = n - 1
        grad = 4.0 * grad_full[start:start + n]

        gn = np.linalg.norm(grad)
        if gn == 0.0:
            break
        eta = step_factor / gn                     # normalised step size

        a_new = a - eta * grad
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best = a_new.copy()
            # tighten step size a little after a successful improvement
            step_factor = max(step_factor * 0.95, 1e-4)

        a = a_new

    return best


# ----------------------------------------------------------------------
# Smoothed‑max (soft‑max) refinement – directly attacks the supremum
def max_refine(a, deadline,
               step_factor=0.6,
               beta_start=5.0,
               beta_growth=1.02,
               beta_max=200.0):
    """Projected gradient descent on a smoothed version of max(conv(a,a))."""
    n = len(a)
    best = a.copy()
    best_score = fast_score(a)

    iter_num = 0
    beta = beta_start

    while time.time() < deadline:
        iter_num += 1
        conv = convolve_fft(a, a)
        max_b = conv.max()

        # increase the temperature gradually
        beta = min(beta_max, beta * beta_growth)

        # soft‑max weights (the larger the entry, the larger its weight)
        w = np.exp(beta * (conv - max_b))           # max entry gets weight 1
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / len(w))
        else:
            w /= w_sum

        # gradient of the smoothed max = 2 · correlate(w, a) (valid mode)
        w_rev = w[::-1]
        grad_full = convolve_fft(w_rev, a)
        start = n - 1
        grad = 2.0 * grad_full[start:start + n]

        gn = np.linalg.norm(grad)
        if gn == 0.0:
            break

        # step size decays like 1/√iteration and is normalised by the gradient norm
        eta = step_factor / np.sqrt(iter_num) / gn

        improved = False
        # simple back‑tracking line search
        for _ in range(12):
            a_new = a - eta * grad
            a_new = project_onto_simplex(a_new)
            new_score = fast_score(a_new)

            # accept any strict improvement
            if new_score < best_score - 1e-12:
                best_score = new_score
                best = a_new.copy()
                a = a_new
                improved = True
                if iter_num % 30 == 0:
                    print(f' iter {iter_num:5d}  beta={beta:6.2f}  score={new_score:.6f}')
                break
            else:
                eta *= 0.5                              # shrink step and retry

        if not improved:
            # if we are stuck, inject a tiny random perturbation and continue
            if eta < 1e-12:
                a = project_onto_simplex(a + 1e-4 * np.random.rand(n))

    return best


# ----------------------------------------------------------------------
def propose_candidate(seed=0, budget_s=1000):
    """
    Search for a step‑function with a low evaluation score.
    The approach combines a power‑law grid search, an L2 flattening
    refinement and a smoothed‑max gradient descent.  It respects the
    1000‑second time budget and returns a list of non‑negative floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # a small safety margin

    # --------------------------------------------------------------
    # Choose a medium‑size problem (large enough to be expressive, small
    # enough for many FFT‑based iterations inside the time budget).
    n = 5000

    # Uniform baseline – score = 2.0
    a_uniform = np.full(n, 1.0 / n)
    best_seq = a_uniform.copy()
    best_score = fast_score(best_seq)
    print(f'initial uniform score = {best_score:.6f}')

    # --------------------------------------------------------------
    # 1) Coarse power‑law grid
    alphas = np.linspace(0.55, 0.70, 31)          # step ≈0.005
    deltas = np.linspace(0.0, 5.0, 21)           # step = 0.25

    for alpha in alphas:
        for delta in deltas:
            if time.time() >= deadline:
                break
            a = power_law_sequence(alpha, delta, n)
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f'coarse  α={alpha:.4f} δ={delta:4.2f} → score={sc:.6f}')

    # --------------------------------------------------------------
    # 2) Fine grid around the best region (full scan with finer steps)
    alphas_f = np.linspace(0.55, 0.70, 81)       # step ≈0.0025
    deltas_f = np.linspace(0.0, 5.0, 41)        # step = 0.125

    for alpha in alphas_f:
        for delta in deltas_f:
            if time.time() >= deadline:
                break
            a = power_law_sequence(alpha, delta, n)
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f'fine    α={alpha:.5f} δ={delta:5.3f} → score={sc:.6f}')

    # --------------------------------------------------------------
    # 3) L2‑flattening refinement (helps to make the convolution flat)
    if time.time() < deadline:
        print('starting L2 flattening refinement...')
        best_seq = l2_refine(best_seq, deadline)
        best_score = fast_score(best_seq)
        print(f' after L2 refinement score = {best_score:.6f}')

    # --------------------------------------------------------------
    # 4) Smoothed‑max gradient refinement (directly reduces the max)
    if time.time() < deadline:
        print('starting smoothed‑max refinement...')
        best_seq = max_refine(best_seq, deadline)
        best_score = fast_score(best_seq)
        print(f' after max refinement score = {best_score:.6f}')

    # --------------------------------------------------------------
    # Final sanity check with the provided evaluate_sequence (if available)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f'final reported score = {final_score:.6f}')
    except Exception:
        pass

    # Ensure we return a plain Python list of floats
    return best_seq.tolist()
