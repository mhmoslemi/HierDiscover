            conv_candidate[sink:sink+n] += 2.0 * delta * seq

"""
Propose a sequence of non‑negative heights that pushes the evaluation score
below the previous best (≈1.5709).  The algorithm starts from the discrete
arcsine / Catalan seed (which gives the theoretical value π/2) and then
performs an aggressive mass‑redistribution greedy search:

* The self‑convolution is maintained incrementally (O(n) updates) instead of
  recomputing a full FFT each step.
* At every iteration the index k where the convolution is maximal is found.
  The contribution of each coefficient a_i to that maximal entry is
  w_i = a_{k‑i} (zero when the index is out of range).
* The coefficient that contributes the most (src) is chosen as a source
  and a coefficient with (almost) zero contribution (sink) is chosen as a
  destination.
* A fraction of the source mass is moved to the sink.  The fraction is
  reduced by binary search until the maximal convolution value strictly
  decreases.
* After each successful move the best sequence and its score are recorded.
* To avoid drift the full convolution is recomputed from scratch every few
  thousand moves.

The whole procedure runs within the given time budget (default 1000 s) and
returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper – Catalan / arcsine seed (coefficients of (1‑z)^‑½)
# ----------------------------------------------------------------------
def catalan_sequence(length: int) -> np.ndarray:
    """Return the first ``length`` Catalan‑type coefficients a_i = (2i choose i)/4^i."""
    seq = np.empty(length, dtype=np.float64)
    a = 1.0
    seq[0] = a
    for i in range(1, length):
        a *= (2 * i - 1) / (2.0 * i)          # a_i = a_{i‑1} * (2i‑1)/(2i)
        seq[i] = a
    return seq


# ----------------------------------------------------------------------
# Fast score – identical to evaluate_sequence but without the security checks
# ----------------------------------------------------------------------
def fast_score(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) / (sum(seq)²).  n = len(seq)."""
    n = seq.size
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Greedy mass‑redistribution search starting from the Catalan seed.
    The loop runs until the time budget expires or the score drops below
    the target 1.503.  The returned list contains non‑negative heights.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # small safety margin

    # ------------------------------------------------------------------
    # 1.  Initialise with the Catalan / arcsine distribution.
    # ------------------------------------------------------------------
    n = 3000                                  # length – a good trade‑off
    a = catalan_sequence(n)                  # type: np.ndarray
    conv = np.convolve(a, a)                  # full self‑convolution (2n‑1 entries)
    sum_a = a.sum()

    best_seq = a.copy()
    best_score = fast_score(best_seq)
    print(f"[init] score={best_score:.6f}, max_conv={conv.max():.6f}")

    # pre‑allocate a constant index array for fast w‑computation
    idx = np.arange(n, dtype=np.int64)

    it = 0
    recompute_counter = 0
    # ------------------------------------------------------------------
    # 2.  Greedy iterative improvement.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # location of the current maximal convolution entry
        k = int(conv.argmax())
        cur_max = float(conv[k])

        # stop early if we already beat the target bound
        cur_score = 2.0 * n * cur_max / (sum_a * sum_a)
        if cur_score <= 1.503:
            print(f"[target] achieved score={cur_score:.6f} after {it} iterations.")
            break

        # ----- compute contribution weights w_i = a_{k‑i} (zero out‑of‑range) -----
        j = k - idx
        mask = (j >= 0) & (j < n)
        w = np.zeros_like(a)
        w[mask] = a[j[mask]]

        # ----- choose source (largest a_i·w_i) and sink (≈ zero contribution) -----
        contrib = a * w
        src = int(np.argmax(contrib))

        # sink: any index with (almost) zero contribution to the current max
        zero_mask = w < 1e-12
        if zero_mask.any():
            sink = int(np.random.choice(np.flatnonzero(zero_mask)))
        else:
            sink = int(np.argmin(w))

        if src == sink:
            # fallback – pick a random different index
            cand = np.delete(np.arange(n), src)
            sink = int(np.random.choice(cand))

        src_amount = a[src]
        if src_amount < 1e-12:
            # nothing left to move – exit the loop
            break

        # ----- try to move a fraction of src to sink, decreasing the fraction
        #       until the maximal convolution value strictly drops -----
        fraction = 1.0                      # start by moving the whole amount
        moved = False
        while fraction > 1e-4:
            delta = src_amount * fraction   # amount to shift from src to sink
            # ----- incremental update of the convolution -----
            conv_candidate = conv.copy()

            # linear part  –  –2·δ·a shifted by src  and  +2·δ·a shifted by sink
            conv_candidate[src:src + n] -= 2.0 * delta * a
            conv_candidate[sink:sink + n] += 2.0 * delta * a

            # quadratic part  –  adjust the three diagonal entries
            conv_candidate[2 * src] += delta * delta
            conv_candidate[2 * sink] += delta * delta
            conv_candidate[src + sink] -= 2.0 * delta * delta

            new_max = float(conv_candidate.max())
            if new_max < cur_max - 1e-12:
                # ----- accept the move -----
                a[src] -= delta
                a[sink] += delta
                conv = conv_candidate
                cur_max = new_max
                moved = True

                new_score = 2.0 * n * cur_max / (sum_a * sum_a)
                if new_score < best_score - 1e-12:

                break
            else:
                fraction *= 0.5   # try a smaller transfer

        if not moved:

            if np.random.rand() < 0.02:

                jitter = (np.random.rand() - 0.5) * 0.01 * a[rnd_idx]
                delta = jitter
                # apply a cheap approximate update (recompute later)
                conv[rnd_idx:rnd_idx + n] += 2.0 * delta * a
                a[rnd_idx] = max(0.0, a[rnd_idx] + delta)

        it += 1
        recompute_counter += 1
        # Every few thousand moves recompute the full convolution to avoid drift.
        if recompute_counter >= 5000:
            conv = np.convolve(a, a)
            recompute_counter = 0

    # ------------------------------------------------------------------
    # 3.  Return the best sequence discovered.
    # ------------------------------------------------------------------
    print(f"[done] best score={best_score:.6f} after {it} iterations.")
    return best_seq.tolist()
