def fast_evaluate(seq):
    n = len(seq)
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return np.inf
    # compute convolution via FFT:
    size = 1 << (2 * n - 1).bit_length()
    fft_a = np.fft.rfft(seq, size)
    fft_conv = fft_a * fft_a
    conv = np.fft.irfft(fft_conv, size)[:2*n-1]
    max_b = np.max(conv)
    return 2.0 * n * max_b / (sum_a * sum_a)

    seq = np.asarray(seq, dtype=np.float64)

    fft_seq = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fft_seq * fft_seq, n=m)[:2*n-1]
    max_b = conv.max()

    """
    Fast replica of `evaluate_sequence` using FFT.
    `seq` may be a list or a NumPy array of non‑negative numbers.
    Returns the floating‑point score.
    """
    a = np.asarray(seq, dtype=np.float64)
    n = a.shape[0]
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    # Length for an efficient FFT (next power of two)
    m = 1 << ((2 * n - 1).bit_length())
    # Convolution via FFT (real version is enough)
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    max_b = float(conv.max())
    return 2.0 * n * max_b / (sum_a * sum_a)


def hill_climb(initial_seq, init_score, deadline):
    """
    Stochastic mass‑transfer hill‑climb.
    Returns (best_seq, best_score) found before `deadline`.
    """
    seq = np.array(initial_seq, dtype=np.float64)
    n = seq.shape[0]
    best_seq = seq.copy()
    best_score = init_score
    cur_score = init_score
    # pre‑compute FFT length for speed (unchanged during climb)
    fft_len = 1 << ((2 * n - 1).bit_length())

    while time.time() < deadline:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # move up to 5 % of the source entry
        eps = seq[i] * np.random.rand() * 0.05
        if eps <= 0.0:
            continue
        new_seq = seq.copy()
        new_seq[i] -= eps
        new_seq[j] += eps
        # quick evaluate
        new_score = fast_score(new_seq)
        if new_score + 1e-12 < cur_score:          # accept only improvements
            seq = new_seq
            cur_score = new_score
            if new_score < best_score:
                best_seq = new_seq.copy()
                best_score = new_score
                print(f"[hill] improved to {new_score:.6f}")
    return best_seq.tolist(), best_score


def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    the provided `evaluate_sequence` metric.
    The function respects a ~1000 s CPU budget.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.1   # safety margin

    # ------------------------------------------------------------------
    # 0.  Keep the best solution found so far
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    if "height_sequence_1" in globals():
        try:
            sc = evaluate_sequence(height_sequence_1)
            best_seq = list(height_sequence_1)
            best_score = sc
            print(f"[init] using existing height_sequence_1, score = {sc:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Grid search over lengths, exponents and offsets
    # ------------------------------------------------------------------
    # Lengths – powers of two up to ~3·10⁴ (still cheap for FFT)
    length_candidates = [256, 512, 1024, 2048, 4096, 8192, 16384, 32768]

    # Exponents β (the usual range where the bound is <2)
    betas = np.linspace(0.55, 0.80, 51)          # step 0.005

    # Small integer offsets to shift the start of the power law
    offsets = [0, 2, 5, 10]

    for n in length_candidates:
        if time.time() > deadline:
            break
        # Base indices for offset = 0 : i = 0 … n‑1 → (i+1)
        base_idx = np.arange(n, dtype=np.float64) + 1.0
        for off in offsets:
            if time.time() > deadline:
                break
            # (i+1+off) to the power –β
            shifted = base_idx + off
            for beta in betas:
                if time.time() > deadline:
                    break
                seq = shifted ** (-beta)
                sc = fast_score(seq)
                if sc < best_score:
                    best_score = sc
                    best_seq = seq.tolist()
                    print(f"[grid] n={n:5d}, off={off:2d}, β={beta:.3f} → new best = {sc:.6f}")

    # ------------------------------------------------------------------
    # 2.  Local stochastic refinement (hill‑climb)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        refined_seq, refined_score = hill_climb(best_seq, best_score, deadline)
        if refined_score < best_score:
            best_seq, best_score = refined_seq, refined_score

    # ------------------------------------------------------------------
    # 3.  Final verification with the official evaluator
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq)
        print(f"[final] best score = {final_score:.6f}, length = {len(best_seq)}")
    except Exception as exc:
        print("[final] verification failed:", exc)
        final_score = None

    return best_seq
