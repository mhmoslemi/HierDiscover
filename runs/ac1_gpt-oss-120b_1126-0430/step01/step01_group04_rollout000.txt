"""
Propose a low‑score step‑function.

The algorithm works in three stages:

1. **Seed / normalisation** – start from the provided global
   `height_sequence_1` (if any) otherwise from a uniform vector, and
   renormalise it so that the L¹‑norm equals √(2·n) (the scale at which
   the objective reduces exactly to the maximum convolution value).

2. **Broad parameter sweep** – repeatedly generate raw vectors from a
   diversified collection of analytical families (power‑law,
   shifted‑power, exponential, Gaussian, raised‑cosine, polynomial,
   piece‑wise linear, random Dirichlet).  Each raw vector is scaled to
   the required L¹‑norm and evaluated.  The best candidate so far is
   kept.

3. **Local refinement** – a mixture of cheap random mass‑transfer moves
   and a *sub‑gradient* step that actively reduces the current maximal
   convolution entry.  For the latter we:
   * compute the convolution `b = a * a`,
   * locate the index/indices where `b` attains its maximum,
   * build the sub‑gradient `g_i = Σ_{k ∈ argmax} a_{k‑i}` (zero when the
     partner index is out of range),
   * move a small amount opposite to this gradient and project back onto
     the simplex `{x ≥ 0, Σ x = √(2·n)}`.
   The sub‑gradient step is combined with a simulated‑annealing style
   acceptance rule and a slowly decaying temperature.

The whole search respects the supplied time budget (default 1000 s) and
periodically prints progress.  The returned list is ready for
`evaluate_sequence`.
"""

import time
import math
import numpy as np

# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------


def _project_to_simplex(x: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { y >= 0, sum(y) = target_sum }.
    Implements the O(n) algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    if target_sum <= 0:
        return np.zeros_like(x)

    # Sort descending
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u) - target_sum
    # Find rho
    rho_arr = np.where(u - cssv / np.arange(1, len(x) + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(x - theta, 0.0)
    return w


def _evaluate(seq):
    """Thin wrapper around the evaluator supplied by the grader."""
    return evaluate_sequence(seq)


def _make_sequence_from_raw(raw: np.ndarray, target_sum: float) -> list:
    """Scale a non‑negative raw vector so that its L¹‑norm equals target_sum."""
    raw_sum = raw.sum()
    if raw_sum == 0:
        # fallback to uniform distribution
        n = raw.shape[0]
        return [target_sum / n] * n
    scale = target_sum / raw_sum
    seq = (raw * scale).tolist()
    return seq

# Family generators – each returns a raw (non‑negative) np.ndarray of length n

def _gen_power(n):
    p = np.random.uniform(0.05, 1.2)
    return (np.arange(1, n + 1, dtype=float)) ** (-p)


def _gen_shifted_power(n):
    p = np.random.uniform(0.05, 1.2)
    d = np.random.uniform(0.0, n * 0.4) + 1e-6
    return (np.arange(n, dtype=float) + d) ** (-p)


def _gen_exponential(n):
    lam = np.random.uniform(0.0005, 0.02)  # decay rate
    return np.exp(-lam * np.arange(n, dtype=float))


def _gen_gaussian(n):
    mu = np.random.uniform(0.3, 0.7) * (n - 1)
    sigma = np.random.uniform(0.05, 0.25) * n
    xs = np.arange(n, dtype=float)
    return np.exp(-0.5 * ((xs - mu) / sigma) ** 2)


def _gen_raised_cosine(n):
    # w[i] = 0.5 - 0.5*cos(π * i / (n-1))  (Hann window)
    return 0.5 - 0.5 * np.cos(np.pi * np.arange(n, dtype=float) / (n - 1))


def _gen_sine(n):
    # Simple half‑sine, zero at both ends
    return np.sin(np.pi * np.arange(n, dtype=float) / (n - 1))


def _gen_polynomial(n):
    alpha = np.random.uniform(0.1, 6.0)
    return (1.0 - np.arange(n, dtype=float) / (n - 1)) ** alpha


def _gen_random_dirichlet(n):
    # Gamma distribution with shape <1 creates a spiky Dirichlet
    shape = np.random.uniform(0.2, 0.9)
    return np.random.gamma(shape=shape, scale=1.0, size=n)


def _gen_piecewise_linear(n):
    # Randomly choose 3‑5 breakpoints and random heights, then interpolate
    nbreak = np.random.randint(3, 6)
    # Ensure breakpoints include 0 and n‑1
    positions = np.sort(
        np.concatenate(
            (
                np.array([0, n - 1]),
                np.random.choice(np.arange(1, n - 1), size=nbreak - 2, replace=False),
            )
        )
    )
    heights = np.random.uniform(0.1, 1.0, size=positions.shape[0])
    # Linear interpolation
    xs = np.arange(n, dtype=float)
    raw = np.interp(xs, positions, heights)
    return raw

_FAMILY_GENERATORS = {

    yields a low score for ``evaluate_sequence``.

        Random seed for reproducibility.
    budget_s : int (default 1000)
        Maximum wall‑clock time (seconds) allowed for the search.
    **kwargs
        Ignored – present only to keep a compatible signature.

    Returns
    -------
    list[float]
        Candidate sequence.
    """

    # ------------------------------------------------------------------

        init_raw = np.array([float(v) for v in height_sequence_1], dtype=float)

        best_seq = _make_sequence_from_raw(init_raw, target_sum)
    else:
        # Fallback: modest size that evaluates quickly
        n = 2000
        target_sum = math.sqrt(2.0 * n)
        best_seq = [target_sum / n] * n

    best_score = _evaluate(best_seq)
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Timing set‑up
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # reserve a few seconds for finalisation
    iteration = 0

    # ------------------------------------------------------------------
    family_names = list(_FAMILY_GENERATORS.keys())

        fam = np.random.choice(family_names)

        raw = np.maximum(raw, 0.0)

            print(f"[sweep {iteration:05d}] new best {best_score:.6f} via {fam}")

        if iteration % 500 == 0 and np.random.rand() < 0.25:
            tmp = np.array(best_seq, copy=True)
            i, j = np.random.choice(n, size=2, replace=False)
            delta = tmp[i] * np.random.uniform(0.0, 0.025)
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_to_simplex(tmp, target_sum)
            pert_score = _evaluate(tmp.tolist())
            if pert_score < best_score:
                best_score = pert_score
                best_seq = tmp.tolist()
                print(f"[perturb {iteration:05d}] improved to {best_score:.6f}")

    # ------------------------------------------------------------------

    cur_score = best_score

    temperature = 0.12
    decay = 0.9993

    while time.time() < deadline + 4.0:
        # ------------------------------------------------------------------
        # Choose a move type
        # ------------------------------------------------------------------
        move_type = np.random.choice(["transfer", "subgrad"], p=[0.68, 0.32])

        if move_type == "transfer":
            # Simple random mass transfer (as in the previous solution)
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            max_move = cur_seq[i] * 0.025
            if max_move <= 0:
                continue
            delta = np.random.random() * max_move
            new_seq = cur_seq.copy()
            new_seq[i] -= delta
            new_seq[j] += delta

            # project back onto the simplex (keeps sum exact)
            new_seq = _project_to_simplex(new_seq, target_sum)

            new_score = _evaluate(new_seq.tolist())

            # acceptance rule
            if new_score < cur_score:
                cur_seq, cur_score = new_seq, new_score
                if new_score < best_score:
                    best_score, best_seq = new_score, new_seq.tolist()
                    print(f"[transfer] improved to {best_score:.6f}")
            else:
                prob = math.exp(-(new_score - cur_score) / (temperature + 1e-12))
                if np.random.rand() < prob:
                    cur_seq, cur_score = new_seq, new_score

        else:  # "subgrad"
            # Compute convolution and identify the (set of) maximal entries
            a = cur_seq
            conv = np.convolve(a, a)          # length 2n‑1
            max_val = conv.max()
            # tolerance to catch multiple (almost) equal maxima
            tol = max(1e-12, max_val * 1e-12)

            # Build the sub‑gradient g_i = Σ_{k ∈ max_idxs} a_{k‑i}
            g = np.zeros(n, dtype=float)
            for k in max_idxs:
                i_start = max(0, k - (n - 1))
                i_end = min(n - 1, k)
                i_range = np.arange(i_start, i_end + 1, dtype=int)

                g[i_range] += a[partner]

            # Take a step opposite to the gradient
            step_size = temperature * (target_sum / n)   # scale ≈ a few % of average height
            new_seq = a - step_size * g
            # Projection restores feasibility
            new_seq = _project_to_simplex(new_seq, target_sum)

            new_score = _evaluate(new_seq.tolist())

            if new_score < cur_score:
                cur_seq, cur_score = new_seq, new_score
                if new_score < best_score:
                    best_score, best_seq = new_score, new_seq.tolist()
                    print(f"[subgrad] improved to {best_score:.6f}")
            else:
                prob = math.exp(-(new_score - cur_score) / (temperature + 1e-12))
                if np.random.rand() < prob:
                    cur_seq, cur_score = new_seq, new_score

        temperature *= decay

    # ------------------------------------------------------------------
    # Final report
    # ------------------------------------------------------------------
    print(f"[final] best score found: {best_score:.6f}")
    return best_seq
