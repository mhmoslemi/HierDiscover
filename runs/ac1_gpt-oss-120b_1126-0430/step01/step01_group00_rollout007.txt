"""
Propose a step‑function sequence that minimises
    evaluate_sequence(·) = 2·n·max(conv(a,a)) / (sum a)² .
The objective is scale‑invariant, therefore we restrict the search to the
probability simplex (∑a_i = 1, a_i ≥ 0).  The algorithm differs from the
previous random‑mass‑transfer scheme:

* Multiple independent restarts with different lengths and initial shapes.
* On each restart we run a *mirror‑descent* (exponentiated‑gradient) optimiser
  on a smooth approximation of the max‑convolution:
        f_β(a) = (1/β)·log(∑_k exp(β·conv_k(a))).
  The gradient of f_β is 2·corr(p,a) where p is the soft‑max of the
  convolution entries, i.e.  p_k ∝ exp(β·conv_k).
* Convolutions and correlations are computed by FFTs (single‑precision
  rfft/irfft) on a length L that is a power‑of‑two ≥ 3·n‑2, which makes a
  full iteration O(L log L) and fast enough for thousands of iterations.
* A simple schedule raises β (making the smooth approximation sharper) and
  lowers the learning‑rate η during the run.
* After the mirror‑descent phase a short hill‑climbing phase (random mass
  transfers) is performed to fine‑tune the best solution.
* The best sequence over all restarts is returned.
"""

import time
import numpy as np
import math

# ----------------------------------------------------------------------
# Projection onto the simplex (used only for a few initialisations)
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex {x≥0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:                        # fallback – uniform point
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small helper – next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (probability vector) that gives a low
    value of `evaluate_sequence`.  The method uses several random restarts,
    mirror‑descent on a smoothed max‑convolution, and a final short
    hill‑climbing polishing phase.
    Returns the best sequence (list of floats) found before the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Length handling – try a few lengths around the known construction.
    base_len = len(globals().get('height_sequence_1', []))
    if base_len < 2000:
        base_len = 4000
    possible_lengths = [
        max(2000, int(base_len * f))
        for f in [0.8, 1.0, 1.2, 1.5, 2.0]
    ]

    best_seq = None
    best_score = float('inf')
    restart_id = 0

    while time.time() < deadline:
        restart_id += 1

        # --------------------------------------------------------------
        # Choose a length for this restart
        n = int(np.random.choice(possible_lengths))
        # power‑of‑two size for the FFT that covers full linear correlation (3n‑2)
        L = _next_pow2(3 * n - 2)

        # --------------------------------------------------------------
        # Initialise a probability vector – several heuristics are tried.
        init_mode = np.random.choice(['dirichlet', 'u_shape', 'sinus', 'uniform'])
        if init_mode == 'dirichlet':
            # Gamma can generate Dirichlet samples; parameter α controls sparsity.
            alpha = np.random.uniform(0.3, 2.0)
            a = np.random.gamma(alpha, 1.0, size=n)
            a = _project_to_simplex(a)
        elif init_mode == 'u_shape':
            # Edge‑heavy (“U‑shaped”) distribution.
            p = np.random.uniform(-2.0, -0.2)          # negative exponent → big edges
            idx = np.arange(n, dtype=float) + 1.0
            a = ((idx) * (n + 1 - idx)) ** p
            a = np.clip(a, 0, None)
            a = _project_to_simplex(a)
        elif init_mode == 'sinus':
            # Sinusoidal shape, optionally raised to a power.
            pow_ = np.random.uniform(0.5, 3.0)
            a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** pow_
            a = np.clip(a, 0, None)
            a = _project_to_simplex(a)
        else:   # uniform random point
            a = np.random.rand(n)
            a = _project_to_simplex(a)

        # ------------------------------------------------------------------
        # Mirror‑descent parameters
        max_iters = 8000                       # guard against endless loops
        beta_start = 5.0
        beta_end   = 80.0
        eta_start  = 0.6                       # initial learning‑rate for mirror step
        eps_log    = 1e-15                     # to keep logs well‑defined

        # ------------------------------------------------------------------
        # Main optimisation loop (smooth‑max mirror descent)
        for it in range(max_iters):
            if time.time() >= deadline:
                break

            # schedule for β (smoothness) and η (step size)
            prog = it / max_iters
            beta = beta_start + prog * (beta_end - beta_start)
            eta  = eta_start * (1.0 - prog)    # linear decay

            # ----- convolution of a with itself (FFT, length L) -----
            Fa = np.fft.rfft(a, n=L)
            conv = np.fft.irfft(Fa * Fa, n=L)[: 2 * n - 1]

            # ----- soft‑max weights for the smooth maximum -----
            cmax = conv.max()
            # exponentials are safe – values are O(1e‑4) and β≤80
            exp_vals = np.exp(beta * (conv - cmax))
            p = exp_vals / exp_vals.sum()      # p sums to 1, concentrates on large conv entries

            # ----- gradient of the smooth max: g = 2 * corr(p, a) -----
            Fp = np.fft.rfft(p, n=L)
            corr_full = np.fft.irfft(Fp * np.conj(Fa), n=L)
            # correlation index for shift k = -(0 … n‑1) lives at n‑1‑k
            g = 2.0 * corr_full[(n - 1) - np.arange(n)]

            # ----- mirror‑descent (exponentiated‑gradient) update -----
            log_a = np.log(a + eps_log) - eta * g
            # for numerical stability shift the logs
            log_a -= np.max(log_a)
            a = np.exp(log_a)
            a /= a.sum()                       # back to the simplex

            # ----- evaluate current objective (exact max, not the smooth one) -----
            # reuse the earlier convolution (conv) – its max is the true max.
            cur_score = 2.0 * n * cmax
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] New global best O={best_score:.6f} "
                      f"(restart={restart_id}, iter={it}, n={n})")

            # ----- occasional random diffusion to escape plateaus -----
            if it % 1000 == 0 and it > 0:
                # small Gaussian jitter, renormalise afterwards
                a += np.random.randn(n) * 0.001
                a = np.clip(a, 0, None)
                a = _project_to_simplex(a)

        # ------------------------------------------------------------------
        # Short hill‑climbing polishing phase (random mass transfers)
        # fewer iterations – we just try to improve the best found so far.
        if best_seq is None:
            continue
        # start from the best sequence of this restart
        a = best_seq.copy()
        cur_score = best_score
        for it in range(2000):
            if time.time() >= deadline:
                break
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            # move a modest fraction of mass from i to j
            delta = a[i] * np.random.rand() * 0.07
            if delta <= 0:
                continue
            a[i] -= delta
            a[j] += delta
            # project back onto simplex (cheap, ensures non‑negativity)
            a = _project_to_simplex(a)

            # compute new objective
            Fa = np.fft.rfft(a, n=L)
            conv = np.fft.irfft(Fa * Fa, n=L)[: 2 * n - 1]
            new_score = 2.0 * n * conv.max()
            if new_score < cur_score:
                cur_score = new_score
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] New global best O={best_score:.6f} "
                          f"(polish restart={restart_id}, iter={it})")
            else:
                # undo the move (simple revert) – cheap because we have a copy of a
                a[i] += delta
                a[j] -= delta

        # ------------------------------------------------------------------
        # Abort early if time is almost up
        if time.time() >= deadline - 5:
            break

    # ----------------------------------------------------------------------
    # Final safety check with the provided evaluate_sequence (optional)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence call failed:", exc)

    # Return as plain Python list (the external grader expects a list of floats)
    return best_seq.tolist()
