def _project_simplex(v):
    # From Duchi et al., O(n log n)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a candidate step‑function (list of non‑negative heights) that tries to
minimise ``evaluate_sequence``.

The algorithm improves on the simple sub‑gradient descent by:
  * normalising the sum of the heights to 1 (the scale of the objective is
    invariant under positive scaling);
  * minimising a *smoothed* version of the max‑convolution using the
    log‑sum‑exp (soft‑max) with an annealing parameter ``alpha``;
  * applying the Adam optimiser (adaptive moments) on the full‑length sequence;
  * projecting onto the probability simplex after each update;
  * using an FFT based convolution (O(n log n)) so that many iterations can be
    performed for a few thousand coefficients within the 1000 s budget;
  * occasional random perturbations when progress stalls.

The optimisation is performed on a single length ``n`` (default 2500).  The
best sequence seen during the run is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level – no closures)
# ----------------------------------------------------------------------


def _catalan_sequence(n: int) -> np.ndarray:
    """
    First ``n`` Catalan‑like coefficients a_i = C(2i,i)/4**i,
    normalised to have sum 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * (i - 1) + 1) / (2.0 * i)
    a /= a.sum()
    return a


def _fft_convolve(x: np.ndarray, y: np.ndarray, nfft: int) -> np.ndarray:
    """
    Linear convolution of ``x`` and ``y`` using an FFT of length ``nfft``.
    The result has length ``len(x)+len(y)-1``.
    """
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    conv = np.fft.irfft(X * Y, n=nfft)
    return conv[: len(x) + len(y) - 1]


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a non‑negative vector onto the simplex
    {x >= 0, sum(x)=1}.  Simple scaling after clipping works because the
    gradient updates never create large negative components.
    """
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 0.0:
        return v / s
    # degenerate case – fall back to a random point on the simplex
    w = np.random.random(v.shape[0])
    return w / w.sum()


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a non‑negative height sequence that minimises the supplied
    ``evaluate_sequence`` function.  The routine runs for at most ``budget_s``
    seconds and returns the best sequence discovered.
    """
    np.random.seed(seed)

    # leave a small safety margin for the runner infrastructure
    deadline = time.time() + budget_s - 5.0

    # ------------------------------------------------------------------
    # problem size – a few thousand entries give a good trade‑off
    # ------------------------------------------------------------------
    n = 2500  # can be changed; larger n improves the theoretical bound
    # pre‑compute the FFT length needed for all convolutions
    nfft = 1 << ((3 * n - 2).bit_length())  # next power of two ≥ 3n‑2

    # ------------------------------------------------------------------
    # initialise the sequence (Catalan seed)
    # ------------------------------------------------------------------
    a = _catalan_sequence(n)          # sum(a) == 1
    best_seq = a.copy()
    # we can compute the objective analytically (sum == 1)
    conv_full = _fft_convolve(a, a, nfft)
    best_score = 2.0 * n * conv_full.max()
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # Adam optimiser state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # hyper‑parameters
    base_lr = 0.15          # base learning rate for Adam
    alpha = 0.5             # start of the soft‑max temperature
    alpha_max = 12.0        # final temperature (focuses on the true max)
    iter_cnt = 0
    no_improve = 0
    last_improve_iter = 0

    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1) convolution a * a  (full length 2n‑1)
        # --------------------------------------------------------------
        conv_full = _fft_convolve(a, a, nfft)

        # --------------------------------------------------------------
        # 2) soft‑max weights (smooth approximation of the max)
        # --------------------------------------------------------------
        max_conv = conv_full.max()
        # stabilise the exponentiation
        exp_arg = alpha * (conv_full - max_conv)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv_full, 1.0 / conv_full.size)
        else:
            w /= w_sum

        # --------------------------------------------------------------
        # 3) gradient of the smooth max: grad_i = 2 * Σ_j w_{i+j} a_j
        #    (computed via one more convolution)
        # --------------------------------------------------------------
        rev_a = a[::-1]                      # a reversed
        grad_full = _fft_convolve(w, rev_a, nfft)   # length 3n‑2
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]   # slice of length n

        # --------------------------------------------------------------
        # 4) Adam update (first‑order adaptive method)
        # --------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # bias‑corrected moments
        m_hat = m / (1.0 - beta1 ** (iter_cnt + 1))
        v_hat = v / (1.0 - beta2 ** (iter_cnt + 1))

        a_new = a - base_lr * m_hat / (np.sqrt(v_hat) + eps)

        # --------------------------------------------------------------
        # 5) projection onto the simplex (non‑negative, sum = 1)
        # --------------------------------------------------------------
        a = _project_to_simplex(a_new)

        # --------------------------------------------------------------
        # 6) evaluate the current objective (exact value without the
        #    overhead of evaluate_sequence – all constraints are already
        #    satisfied)
        # --------------------------------------------------------------
        cur_max = conv_full.max()          # max of a*a from the previous step
        cur_score = 2.0 * n * cur_max

        # keep the best sequence
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            no_improve = 0
            last_improve_iter = iter_cnt
            print(f"[{iter_cnt:06d}] new best = {best_score:.6f}  alpha={alpha:.3f}")
        else:
            no_improve += 1

        # --------------------------------------------------------------
        # 7) annealing / hyper‑parameter schedule
        # --------------------------------------------------------------
        iter_cnt += 1

        # increase the soft‑max temperature slowly (focuses more on the true max)
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * 1.003)

        # periodic learning‑rate decay
        if iter_cnt % 3000 == 0:
            base_lr *= 0.85

        # if we have not improved for a long while, inject a tiny random
        # perturbation to escape possible plateaus
        if no_improve > 4000 and (iter_cnt - last_improve_iter) > 4000:
            perturb = np.random.normal(scale=1e-4, size=n)
            a = _project_to_simplex(a + perturb)
            m[:] = 0.0
            v[:] = 0.0
            no_improve = 0
            print(f"[{iter_cnt:06d}] perturbation to escape stagnation")

    print(f"[done] best score found = {best_score:.6f}")
    # convert to plain Python list as required by the evaluation harness
    return best_seq.tolist()
