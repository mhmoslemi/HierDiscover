def _fft_convolve(x, y):
    size = 1 << ((len(x) + len(y) - 1).bit_length())
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)
    return conv[:len(x) + len(y) - 1]

Propose a candidate step‑function (list of non‑negative heights) that tries to
minimise ``evaluate_sequence``.

Algorithm
---------
We treat the sequence as a probability distribution (its sum is forced to 1).
For a distribution ``a`` the evaluation value equals

    2·n·max_k (a * a)[k]                (n = length of a)

so we have to minimise the ∞‑norm of the self‑convolution.
The objective is non‑smooth; we replace the max with a smooth
log‑sum‑exp approximation

    F(a) = log Σ_k exp(α·(a*a)[k])      (α≫1)

and perform projected gradient descent on the simplex.
The gradient of F is

    ∂F/∂a_i = 2·Σ_k p_k·a_{k-i},
    p_k = exp(α·c_k) / Σ_j exp(α·c_j)   (c = a*a)

which is a convolution of the reversed ``a`` with the soft‑max weights ``p``.
Both convolutions are computed with FFTs (O(n log n)) so that many
iterations fit comfortably into the 1000 s budget.

We start from the analytically good Catalan construction, run the
gradient descent with a slowly growing smoothing parameter ``α`` and a
decaying learning rate, and keep the best sequence seen.
Occasional random restarts avoid stagnation.
The final best sequence is returned as a plain Python list.
"""

    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)
    return conv[:n]


def _catalan_sequence(n: int) -> np.ndarray:
    """Catalan‑like coefficients a_i = C(2i,i)/4**i, normalised to sum to 1."""
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    # recurrence a_{i+1} = a_i * (2*i+1) / (2*(i+1))
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * (i - 1) + 1) / (2.0 * i)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a sequence of non‑negative heights that minimises
    ``evaluate_sequence``.  The routine runs for at most *budget_s* seconds
    and returns the best sequence found.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : int, optional
        Time budget in seconds (default 1000).

    Returns
    -------
    list[float]
        The best sequence discovered (plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Problem size and initialisation
    # ------------------------------------------------------------------
    # length around a few thousand offers a good trade‑off between quality
    # and speed.  Using a slightly larger size than the previous 3000 gives
    # a little extra freedom without hurting time limits.
    n = 4000

    # start from the known Catalan construction;
    # if a previous solution is available in the global namespace we use it.
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=float)
        if a.shape[0] != n:
            # resize (truncate or pad with zeros) to the chosen length
            if a.shape[0] > n:
                a = a[:n]
            else:
                a = np.pad(a, (0, n - a.shape[0]), constant_values=0.0)
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum > 0:
            a /= a_sum
        else:
            a = _catalan_sequence(n)
    else:
        a = _catalan_sequence(n)

    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Hyper‑parameters
    # ------------------------------------------------------------------
    lr = 0.02               # base learning rate
    momentum = np.zeros_like(a)
    beta = 0.9              # momentum coefficient
    alpha = 8.0             # initial smoothing (log‑sum‑exp) parameter
    alpha_max = 300.0
    alpha_mul = 1.3         # how fast we increase smoothness
    iter_since_alpha = 0

    iteration = 0
    no_improve = 0
    restart_interval = 25000   # try a random restart if stuck

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # ----- convolution of the current iterate (needed for gradient) -----
        conv = _fft_convolve(a, a)        # length 2n‑1
        max_conv = conv.max()

        # ----- smooth approximation of the max (log‑sum‑exp) ---------------
        # shift by max to avoid overflow
        shifted = conv - max_conv
        weights = np.exp(alpha * shifted)
        weights_sum = weights.sum()
        p = weights / weights_sum          # soft‑max over convolution entries

        # ----- gradient of the smoothed objective ------------------------
        rev_a = a[::-1]
        corr = _fft_convolve(rev_a, p)    # length 3n‑2
        grad = 2.0 * corr[n - 1:n - 1 + n]  # extract the n relevant components

        # ----- momentum update + projection onto the simplex -------------
        momentum = beta * momentum - lr * grad
        a_new = a + momentum

        # enforce non‑negativity and renormalise to sum‑1
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 0:
            a_new /= s
        else:
            a_new = np.random.random(n)
            a_new /= a_new.sum()

        a = a_new

        # ----- evaluate the new candidate ---------------------------------
        conv_new = _fft_convolve(a, a)
        cur_score = 2.0 * n * conv_new.max() / (a.sum() ** 2)   # a.sum() is 1
        # (the above is exactly the value returned by evaluate_sequence)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{iteration:07d}] new best = {best_score:.6f}")
            no_improve = 0
        else:
            no_improve += 1

        # ----- learning‑rate schedule ------------------------------------
        if iteration % 2000 == 0 and iteration > 0:
            lr *= 0.93                     # slow decay

        # ----- increase the smoothing parameter gradually ------------------
        iter_since_alpha += 1
        if iter_since_alpha >= 1500 and alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)
            iter_since_alpha = 0
            print(f"[{iteration:07d}] increased alpha → {alpha:.2f}")

        # ----- occasional random restart to escape plateaus ---------------
        if no_improve > restart_interval:
            a = np.random.random(n)
            a /= a.sum()
            momentum.fill(0.0)
            lr = 0.02
            no_improve = 0
            print(f"[{iteration:07d}] restart (random init)")

        iteration += 1

    print(f"[done] best score found = {best_score:.6f}")
    return best_seq.tolist()
