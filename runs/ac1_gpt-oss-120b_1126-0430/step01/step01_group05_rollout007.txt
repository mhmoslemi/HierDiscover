def subgradient_optimize(a, deadline):
    best = a.copy()
    best_score = evaluate_sequence(best)
    n = len(a)
    iter = 0
    while time.time() < deadline and iter < max_iters:
        iter += 1
        conv = np.convolve(a, a)
        max_c = conv.max()
        # identify all ks with value close to max (within relative tolerance)
        active = np.where(conv >= max_c - 1e-12)[0] # absolute tolerance maybe relative
        # compute gradient
        grad = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            # Create arrays for i and j
            i_range = np.arange(i_start, i_end + 1)
            j_range = k - i_range
            grad[i_range] += 2.0 * a[j_range]
        # step size schedule: eta = alpha / (iter**0.5) maybe
        eta = 0.1 / (np.sqrt(iter))
        a_new = a - eta * grad
        # enforce nonnegativity
        a_new = np.maximum(a_new, 0.0)
        # renormalize to sum=1
        sum_a = a_new.sum()
        if sum_a < 1e-12:
            # random restart
            a_new = np.random.rand(n)
            sum_a = a_new.sum()
        a = a_new / sum_a
        # Evaluate
        score = evaluate_sequence(a)
        if score < best_score:
            best_score = score
            best = a.copy()
            print(f"Iter {iter}: score {best_score:.6f}")
    return best, best_score

"""
Gradient‑based optimisation of the L∞ norm of a self‑convolution.

The evaluation function is homogeneous (scaling the whole sequence does not
change the score) so we keep the L¹ norm equal to 1 during the search.
The algorithm works in three stages:

1. **Coarse exponent sweep** – generate power‑law sequences
   a_i = (i+1)^{-p} for a modest length (≈15000) and a set of exponents
   p∈[0.45,0.75].  The best few candidates are kept.

2. **Projected sub‑gradient descent** – starting from the best candidate,
   repeatedly:
       * compute the convolution a*a,
       * locate the shift(s) where the maximum is attained,
       * form the sub‑gradient of max_k (a*a)[k] (which is 2·a_{k-i} on the
         active shifts),
       * take a small step opposite to the gradient,
       * project back onto the simplex (non‑negative, sum = 1).

   The step size decays as 1/√iteration; only improvements are kept.
   This drives the shape toward one whose autocorrelation is as flat as
   possible, lowering the evaluation score.

3. **Finalisation** – the best sequence found (as a Python list of floats)
   is returned.  The whole routine stops well before the supplied time budget
   (default 1000 s) to avoid a timeout.

No external files or network access are used; only NumPy and the supplied
`evaluate_sequence` function are required.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def power_sequence(length: int, p: float) -> np.ndarray:
    """Return a monotone power‑law sequence a_i = (i+1)^{-p} as a NumPy array."""
    idx = np.arange(1, length + 1, dtype=float)
    return idx ** (-p)


def subgradient_step(
    a: np.ndarray,
    active_shifts: np.ndarray,
) -> np.ndarray:
    """
    Compute the projected sub‑gradient for the max‑convolution objective.

    For each active shift k we add 2·a_{k-i} to the i‑th component whenever
    0 ≤ i < n and 0 ≤ k-i < n.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)

    for k in active_shifts:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # vector of i's
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        j_range = k - i_range          # indices of the partner entries
        grad[i_range] += 2.0 * a[j_range]

    return grad


def projected_normalise(v: np.ndarray) -> np.ndarray:
    """Project onto the non‑negative simplex (sum = 1)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-12:                  # avoid division by zero – re‑initialise
        v = np.random.rand(v.shape[0])
        s = v.sum()
    return v / s


def subgradient_descent(
    init_seq: np.ndarray,
    deadline: float,
    eval_interval: int = 5,
) -> Tuple[np.ndarray, float]:
    """
    Perform projected sub‑gradient descent on ``init_seq`` until ``deadline``.

    Parameters
    ----------
    init_seq   – starting sequence (already normalised to sum = 1).
    deadline   – absolute time (seconds since epoch) after which we stop.
    eval_interval – how many iterations to skip between expensive score checks.

    Returns
    -------
    (best_sequence, best_score)
    """
    a = init_seq.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = float("inf")
    current_score = float("inf")

    iteration = 0
    # initial score
    current_score = evaluate_sequence(a.tolist())
    best_score = current_score
    best_a = a.copy()
    # a small safety margin for the first evaluation
    if time.time() >= deadline:
        return best_a, best_score

    while time.time() < deadline:
        iteration += 1

        # 1) convolution and active shift(s)
        conv = np.convolve(a, a)
        max_val = conv.max()
        # active shifts – use a tiny absolute tolerance to catch ties
        active = np.where(conv >= max_val - 1e-12)[0]

        # 2) sub‑gradient of the max‑convolution term
        grad = subgradient_step(a, active)

        # 3) step size (decaying)
        step = 0.05 / np.sqrt(iteration)

        # 4) gradient descent step + projection onto simplex
        a_new = a - step * grad
        a_new = projected_normalise(a_new)

        # 5) occasional scoring (to avoid calling evaluate_sequence every iteration)
        if iteration % eval_interval == 0 or time.time() > deadline - 1.0:
            score = evaluate_sequence(a_new.tolist())
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
                print(f"Iter {iteration:06d}: new best score {best_score:.6f}")

            # only accept the move if it improves the *current* objective,
            # otherwise keep the old iterate.
            if score < current_score:
                a = a_new
                current_score = score
            # else: reject – keep old ``a``

    return best_a, best_score

    # --------------------------------------------------------------

    # Convert to plain Python list of floats for the output format.
