def solve_g(f, M):
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    constraints = []
    for k in range(2*n-1):
        # compute indices i where k-i in [0,n-1]

        i_end = min(k, n-1)

        i_start = max(0, k-(n-1))
        i_end   = min(k, n-1)
        expr = 0
        for i in range(i_start, i_end+1):
            expr += f[i] * g[k-i]
        constraints.append(expr <= M)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    # Normalize to sum to 1
    total = g_val.sum()
    if total <= 0:
        return None
    g_val /= total
    return g_val

"""
Propose a step‑function sequence that minimises the bound

    O(a) = 2·n·max(conv(a,a))  ,   ∑ a_i = 1 , a_i ≥ 0

The optimizer is a two‑phase stochastic/sub‑gradient method:

* Phase 0 – initialise a candidate.  We start either from the current
  best sequence (if one is known) with a tiny Gaussian perturbation or
  from a fresh random point on the probability simplex.

* Phase 1 – aggressive sub‑gradient descent with an adaptive line‑search.
  The sub‑gradient of the maximal convolution entry is exactly
  g_i = 2·a_{k−i} for the index k where the convolution attains its
  maximum.  A step `a ← a – η·g` is taken, projected back onto the
  simplex and kept only if it lowers the objective.  The step size η
  starts rather large (≈0.2) and is decreased geometrically whenever
  a step does not improve the score.

* Phase 2 – occasional random “mass‑transfer’’ moves.  A small amount of
  probability mass is moved from a random source index to a random
  destination; the move is accepted iff it improves the objective.
  This provides the ability to escape flat regions.

The whole search runs many independent restarts until the allotted
time (default ≈ 1000 s) is exhausted.  The best sequence found is
returned as a plain list of floats.  All helper functions are defined
at module level and no closures or I/O are used.
"""

import time
import math
import numpy as np

# -------------------------------------------------------------
# Projection onto the probability simplex (Duchi et al.)
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x ≥ 0, sum(x) = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# -------------------------------------------------------------
# Fast objective (the quantity that evaluate_sequence returns
# when the sum of the coefficients is 1)
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a))  (assumes sum(a) == 1)."""
    conv = np.convolve(a, a)
    return 2.0 * a.size * conv.max()


# -------------------------------------------------------------
# Sub‑gradient of the maximal convolution entry
def _subgradient_max_conv(a: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv(a,a)) w.r.t. a (sum(a)=1)."""
    n = a.size
    conv = np.convolve(a, a)
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)          # indices attaining the maximum
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    g /= len(ks)                                   # average if several maxima
    return g


# -------------------------------------------------------------
def _improve_sequence(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Perform a local optimisation consisting of aggressive sub‑gradient steps
    together with occasional random mass‑transfer moves.
    Returns the improved sequence and its objective value.
    """
    n = a.shape[0]
    cur_score = _objective(a)
    best_local = a.copy()
    best_local_score = cur_score

    # initial step size for the sub‑gradient descent
    eta = 0.20
    min_eta = 1e-6
    # decay factor applied when a step fails to improve
    decay = 0.70

    # main inner loop – stops when time runs out or eta becomes tiny
    while time.time() < deadline and eta > min_eta:
        # ---- sub‑gradient step with back‑tracking line search -----------------
        g = _subgradient_max_conv(a)

        # try the current step size; if it does not improve, shrink η
        improved = False
        for _ in range(5):                         # at most 5 reductions per iteration
            a_candidate = a - eta * g
            a_candidate = np.clip(a_candidate, 0, None)
            a_candidate = _project_to_simplex(a_candidate)

            cand_score = _objective(a_candidate)
            if cand_score < cur_score:
                a = a_candidate
                cur_score = cand_score
                improved = True
                if cand_score < best_local_score:
                    best_local_score = cand_score
                    best_local = a.copy()
                break
            else:
                eta *= decay                      # shrink step size and retry

        if not improved:
            # if none of the tried reductions helped, keep the smaller η
            continue

        # ---- occasional random mass‑transfer move (exploration) ---------------
        if np.random.rand() < 0.07:                # 7 % of the iterations
            i, j = np.random.randint(n, size=2)
            if i != j and a[i] > 0.0:
                # move up to 30 % of the source mass
                delta = a[i] * np.random.rand() * 0.30
                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta
                a_new = np.clip(a_new, 0, None)
                a_new = _project_to_simplex(a_new)

                new_score = _objective(a_new)
                if new_score < cur_score:
                    a = a_new
                    cur_score = new_score
                    if new_score < best_local_score:
                        best_local_score = new_score
                        best_local = a.copy()

    return best_local, best_local_score


# -------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                      budget_s: int = 1000,
                      **kwargs) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises
    ``evaluate_sequence``.  The function respects the time budget and
    returns the best sequence found (as a plain list of floats).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # a tiny safety margin

    # ---------------------------------------------------------
    # Determine a reasonable length for the step function.
    # Use the length of a previously discovered construction if present,
    # otherwise fall back to a modest size.
    if 'height_sequence_1' in globals():
        base_len = len(height_sequence_1)
    else:
        base_len = 2500
    n = max(2000, base_len)                 # never go below 2000

    # ---------------------------------------------------------
    # Baseline: the uniform distribution (always feasible).
    const_seq = np.full(n, 1.0 / n)
    best_seq = const_seq.copy()
    best_score = _objective(best_seq)
    print(f"[init] uniform baseline O={best_score:.6f} (n={n})")

    restart_counter = 0
    while time.time() < deadline:
        restart_counter += 1

        # ----- Initialise a new candidate ---------------------------------
        if np.random.rand() < 0.30 and best_seq is not None:
            # Start near the current best with a small Gaussian perturbation
            a = best_seq.copy()
            a += np.random.randn(n) * 0.008
            a = np.clip(a, 0, None)
            a = _project_to_simplex(a)
        else:
            # Pure random point on the simplex
            a = np.random.rand(n)
            a = _project_to_simplex(a)

        # ----- Run the local optimiser ------------------------------------
        a_opt, score_opt = _improve_sequence(a, deadline)

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] New global best O={best_score:.6f} "
                  f"(restart={restart_counter})")

        # Early exit if we have already reached the target bound
        if best_score <= 1.503:
            print("[early] target bound reached, terminating search.")
            break

    # ---------------------------------------------------------
    # Final verification using the provided evaluate_sequence (optional)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:
        print("[final] evaluate_sequence call failed:", e)

    return best_seq.tolist()
