    try:

    best_score = evaluate_sequence(best_seq)

    # Helper: compute score etc
    def compute_score(seq):
        # seq is np array (nonnegative)
        s = evaluate_sequence(seq.tolist())
        return s

    # Helper: normalized simplex projection maybe not needed if we always maintain sum=1.
    def normalize(v):
        total = np.sum(v)
        if total <= 0:
            v = np.ones_like(v) / len(v)
        else:
            v = v / total
        return v

    # Starting candidates list
    candidates = []

    # Add baseline.
    candidates.append(normalize(np.array(best_seq)))

    # Add power law with best alpha/shift from earlier approach: we could reuse same method or compute again inside loop.

    # Add random Dirichlet (with concentration parameter maybe 0.2 to create spiky)
    for i in range(5):
        # random Dirichlet
        a = np.random.dirichlet(alpha=np.full(n, 0.5))  # yields sum=1
        candidates.append(a)

    # Add exponential decay: a_i ~ exp(-beta * i)
    betas = np.linspace(0.0005, 0.005, 5)
    idx = np.arange(n)
    for beta in betas:
        a = np.exp(-beta * idx)
        candidates.append(normalize(a))

    # Determine max iterations: while time left.
    best_seq_array = np.array(best_seq, dtype=float) # from baseline

    # For each candidate, run alternating LP + optional hill climb
    for init_seq in candidates:
        if time.time() - start > budget_s - 5:
            break
        a = init_seq.copy()
        a = normalize(a)  # ensure sum 1
        # Evaluate initial
        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_seq_array = a.copy()
            print(f"[init] new best from init candidate: score={score:.6f}")
        # Run iterative refinement
        max_iter = 10  # number of LP steps
        for it in range(max_iter):
            # time check
            if time.time() - start > budget_s - 5:
                break

            conv_a = np.convolve(a, a)
            M = np.max(conv_a)
            # Build constraint matrix
            # Sparse maybe
            # We'll use dense for simplicity.
            a_pad = np.concatenate([a, np.zeros(n-1)])
            rows = []
            for k in range(2*n-1):
                row = a_pad[k:k+n][::-1]   # reversed slice
                rows.append(row)
            A = np.vstack(rows)  # shape (2n-1,n)
            # Linear program
            c = -np.ones(n)  # maximize sum(b)
            # Solve with scipy.optimize.linprog
            # we set bounds >=0
            res = None
            try:
                res = scipy.optimize.linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
            except Exception as e:
                print(f"[lp] failed: {e}")
                break
            if not res.success:
                print(f"[lp] not successful: {res.message}")
                break
            b_opt = res.x
            sum_b = np.sum(b_opt)
            if sum_b <= 1e-12:
                break
            # Normalize b to sum=1 (scale)
            b_norm = b_opt / sum_b

            if new_score < score:

                score = new_score
                if new_score < best_score:

        cur = a.copy()
        cur_score = score

            i, j = np.random.randint(n, size=2)
            if i == j: continue

            step = cur[i] * step_frac * np.random.rand()

            new[i] -= step

"""
Algorithm:
  1. Initialise a set of candidate sequences (power‑law grid, random Dirichlet, exponential).
  2. For each candidate run a short alternating linear‑program / mixing loop.
     – Given a distribution a (sum = 1) compute its self‑convolution peak M.
     – Solve the LP   maximise   Σ b_j
          subject to   a * b ≤ M   (component‑wise) ,   b ≥ 0 .
     – Normalise the optimal b and replace a by a convex mixture
          a ← (1‑t)·a + t·b .
     – The mixture never violates the convolution bound, therefore the peak
       value M can only stay the same or decrease.
  3. After a few LP steps a local hill‑climbing phase (small mass swaps)
     is applied to polish the result.
  4. The best sequence found over all restarts is returned.
"""

import time
import numpy as np

try:
    from scipy.optimize import linprog
    SCIPY_AVAILABLE = True
except Exception:
    SCIPY_AVAILABLE = False

def _normalize(v):
    """Return v scaled to have L1‑norm equal to 1."""
    total = np.sum(v)
    if total <= 0:
        return np.full_like(v, 1.0 / len(v))
    return v / total

def _max_convolution(v):
    """Maximum coefficient of the (linear) convolution v*v."""
    return np.max(np.convolve(v, v))

def _conv_matrix(v):
    """Return the (2n‑1)×n matrix A with A_{k,j}=v_{k‑j} (zero if out of range)."""
    n = len(v)
    pad = np.concatenate([v, np.zeros(n - 1, dtype=v.dtype)])
    stride = pad.strides[0]
    shape = (2 * n - 1, n)
    strides = (stride, stride)
    windows = np.lib.stride_tricks.as_strided(pad, shape=shape, strides=strides)
    # each row i is pad[i:i+n] (forward order); we need reversed order for the convolution
    return windows[:, ::-1].copy()

def _lp_mix_step(a, mix=0.5, eps=1e-12):
    """
    Perform one LP–mix step.
    Returns a new vector a_next and the new peak M_next.
    """
    n = len(a)
    M = _max_convolution(a)
    # Build the constraint matrix
    A = _conv_matrix(a)
    # Linear programme: maximise sum(b)  <=> minimise -sum(b)
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0.0, None)] * n
    # Solve
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    except Exception as exc:
        # Solver not available or failed – return the original vector
        return a, M
    if not res.success:
        return a, M
    b_opt = res.x
    sum_b = np.sum(b_opt)
    if sum_b <= 0:
        return a, M
    b_norm = b_opt / sum_b
    # Convex combination guarantees that (a * a_next) ≤ M, so peak cannot increase
    a_next = (1.0 - mix) * a + mix * b_norm
    # Numerical drift – keep the vector on the simplex and away from exact zeros
    a_next = np.maximum(a_next, eps)
    a_next = _normalize(a_next)
    M_next = _max_convolution(a_next)
    return a_next, M_next

def _hill_climb(a, score, budget_s, start_time, max_iters=5000):
    """
    Simple stochastic hill‑climbing that swaps a tiny amount of mass between two
    coordinates while preserving the total L1 norm.
    """
    n = len(a)
    cur = a.copy()
    cur_score = score
    eps = 1e-12
    for it in range(max_iters):
        if time.time() - start_time > budget_s - 5:
            break
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # move a fraction of the source to the target
        step = cur[i] * 0.01 * np.random.rand()
        if step <= 0 or cur[j] + step > 1.0:
            continue
        new = cur.copy()
        new[i] -= step
        new[j] += step
        # keep on simplex (rounding errors)
        new = _normalize(new)
        try:
            new_score = evaluate_sequence(new.tolist())
        except Exception:
            continue
        if new_score < cur_score:
            cur = new
            cur_score = new_score
            if new_score < score:
                # propagate to caller via return
                return cur, cur_score
    # no improvement found
    return cur, cur_score

def _grid_search(n, budget_s, start_time):
    """
    Very cheap grid search over power‑law families (α, shift). Returns the best
    sequence found and its score.
    """
    best_seq = None
    best_score = float('inf')
    alphas = np.linspace(0.45, 0.75, 61)   # same grid as before
    shifts = np.linspace(1.0, 2.0, 5)
    idx = np.arange(n, dtype=np.float64)

        if time.time() - start_time > budget_s - 5:

        for shift in shifts:
            raw = (idx + shift) ** (-alpha)
            seq = _normalize(raw)
            try:
                sc = evaluate_sequence(seq.tolist())

            if sc < best_score:
                best_score = sc
                best_seq = seq
                print(f"[grid] α={alpha:.4f}, shift={shift:.2f}, score={sc:.6f}")
    return best_seq, best_score

"""
Algorithm
---------

1️⃣ **Quick grid search**  
   Scan a small grid of power‑law families  
   `a_i ∝ (i+shift)^{-α}` (the same family that gave 1.576) and keep the
   best sequence.  This provides a solid deterministic start.

2️⃣ **Linear‑program / mixing refinement** (the core improvement)  
   For a candidate vector `a` (non‑negative, Σa = 1) let  

        M = max(conv(a, a))

   and solve the linear program  

        maximise   Σ b_j
        subject to   a * b ≤ M   (component‑wise)
                     b_j ≥ 0 .

   The constraints are linear because `(a * b)_k = Σ_i a_i b_{k-i}`.
   The LP is solved with *SciPy*'s HiGHS interior‑point solver.
   The optimal `b` is scaled to a probability vector
   `b̂ = b / Σb`.  Any convex combination  

        a←(1‑t)·a + t·b̂      (0<t≤1)

   never violates the bound `a*a ≤ M`, therefore the peak `M`
   can only stay the same or decrease.  Repeating the LP‑mix step a
   few times drives the distribution toward a flatter sum‑distribution
   and reduces the evaluated score.

3️⃣ **Local hill‑climbing polish**  
   After the LP phase a cheap stochastic hill‑climber moves a tiny
   amount of mass between two random positions while keeping the
   total sum = 1.  Moves that improve the score are kept.

4️⃣ **Multiple restarts**  
   The above pipeline is executed from several diverse starts
   (the best grid result, a few Dirichlet draws, simple exponentials).
   The globally best sequence is returned.

The whole routine is time‑boxed (default 950 s) and falls back to the
original hill‑climbing method if *SciPy* is unavailable.  All helper
functions are top‑level, use no closures, and respect the required
interface.
"""

import time
import numpy as np

# SciPy is optional – the code works without it (it will just use the
# original hill‑climbing fallback).
try:
    from scipy.optimize import linprog
    _SCIPY_AVAILABLE = True
except Exception:                         # pragma: no cover
    _SCIPY_AVAILABLE = False


def _normalize(v: np.ndarray) -> np.ndarray:
    """Scale a non‑negative vector to have L¹‑norm equal to 1."""
    total = v.sum()
    if total <= 0.0:
        return np.full_like(v, 1.0 / v.size)
    return v / total


def _max_convolution(v: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution of v."""
    return float(np.max(np.convolve(v, v)))


def _conv_matrix(v: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1)×n matrix A with A[k, j] = v[k‑j] (zero outside the
    index range).  This matrix satisfies `A @ b == np.convolve(v, b)`.
    """
    n = v.size
    pad = np.concatenate([v, np.zeros(n - 1, dtype=v.dtype)])
    stride = pad.strides[0]
    shape = (2 * n - 1, n)
    strides = (stride, stride)
    windows = np.lib.stride_tricks.as_strided(pad, shape=shape, strides=strides)
    # rows are forward slices; we need them reversed for convolution
    return windows[:, ::-1].copy()

                 eps: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    One LP‑mix iteration.
    Returns a new vector `a_next` (still on the simplex) and its peak
    `M_next = max(conv(a_next, a_next))`.
    """
    n = a.size
    M = _max_convolution(a)

    if not _SCIPY_AVAILABLE:
        # No LP solver – just return the input (the caller will skip the step)
        return a, M

    # Build the constraint matrix A (2n‑1)×n  with rows = reversed slices of a.
    A = _conv_matrix(a)

    # LP: maximise Σb   ⇔   minimise −Σb
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0.0, None)] * n

    try:
        # HiGHS is fast for medium‑sized dense LPs.
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    except Exception:                     # pragma: no cover
        return a, M

    if not res.success:
        # In the unlikely event the solver fails, keep the current vector.
        return a, M

    b_opt = res.x
    sum_b = b_opt.sum()
    if sum_b <= 0.0:
        return a, M

    b_norm = b_opt / sum_b
    # Convex combination guarantees the convolution bound stays ≤ M.
    a_next = (1.0 - mix) * a + mix * b_norm
    a_next = np.maximum(a_next, eps)
    a_next = _normalize(a_next)
    M_next = _max_convolution(a_next)
    return a_next, M_next


def _hill_climb(a: np.ndarray, cur_score: float,
                budget_s: float, start_time: float,
                max_iters: int = 4000) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic hill‑climber that swaps a tiny amount of mass
    between two coordinates while staying on the simplex.
    """
    n = a.size
    cur = a.copy()
    for it in range(max_iters):
        if time.time() - start_time > budget_s - 5:
            break
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # move up to 2 % of the source mass to the destination
        step = cur[i] * (np.random.rand() * 0.02)
        if step <= 0.0:
            continue
        new = cur.copy()
        new[i] -= step
        new[j] += step
        new = _normalize(new)
        try:
            new_score = evaluate_sequence(new.tolist())
        except Exception:
            continue
        if new_score < cur_score:
            cur = new
            cur_score = new_score
            # Return early if we find a strict improvement (caller can keep going)
            # but we also keep climbing to possibly find even better points.
    return cur, cur_score


def _grid_search(n: int, budget_s: float, start_time: float):
    """
    Exhaustive search over a modest grid of (α, shift) for the power‑law family.
    Returns the best sequence found on the simplex and its score.
    """
    best_seq = None
    best_score = float('inf')
    alphas = np.linspace(0.45, 0.75, 61)        # 0.005 steps
    shifts = np.linspace(1.0, 2.0, 5)          # 1.0 … 2.0
    idx = np.arange(n, dtype=np.float64)

    for alpha in alphas:
        if time.time() - start_time > budget_s - 5:
            break
        for shift in shifts:
            raw = (idx + shift) ** (-alpha)
            seq = _normalize(raw)
            try:
                sc = evaluate_sequence(seq.tolist())
            except Exception:
                continue
            if sc < best_score:
                best_score = sc
                best_seq = seq
                print(f"[grid] α={alpha:.4f}, shift={shift:.2f}, score={sc:.6f}")
    return best_seq, best_score


def propose_candidate(seed: int = 123, budget_s: float = 950, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) minimise

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

    The routine follows the three‑phase strategy described in the module
    docstring.
    """
    np.random.seed(seed)
    start = time.time()

    # -----------------------------------------------------------------
    # Problem size (fallback length given in the statement)
    # -----------------------------------------------------------------
    try:
        n = len(height_sequence_1)   # provided by the harness
    except Exception:               # pragma: no cover
        n = 2244

    # -----------------------------------------------------------------
    # Baseline – use the sequence that is always present.
    # -----------------------------------------------------------------
    try:
        base = _normalize(np.array(height_sequence_1, dtype=np.float64))
        best_score = evaluate_sequence(base.tolist())
        best_seq = base.copy()
    except Exception:
        best_score = float('inf')
        best_seq = np.full(n, 1.0 / n)

    print(f"[init] n={n}, baseline score={best_score:.6f}")

    # -----------------------------------------------------------------
    # 1) Grid search over the power‑law family (reproduces the 1.576 record)
    # -----------------------------------------------------------------
    grid_seq, grid_score = _grid_search(n, budget_s, start)
    if grid_seq is not None and grid_score < best_score:
        best_score = grid_score
        best_seq = grid_seq.copy()
        print(f"[grid‑best] improved baseline: score={grid_score:.6f}")

    # -----------------------------------------------------------------
    # 2) Assemble a few diverse starting points.
    # -----------------------------------------------------------------
    starts = [best_seq]                      # always keep the current best
    # Dirichlet draws – concentration <1 makes them spiky
    for _ in range(4):
        d = np.random.dirichlet(alpha=np.full(n, 0.5))
        starts.append(_normalize(d))
    # Simple exponential decays
    idx = np.arange(n, dtype=np.float64)
    for beta in np.linspace(0.001, 0.004, 3):
        exp_seq = np.exp(-beta * idx)
        starts.append(_normalize(exp_seq))

    eps = 1e-12          # prevents exact zeros (helps the LP)

    # -----------------------------------------------------------------
    # 3) Main optimisation loop: LP‑mix → hill‑climb for each restart.
    # -----------------------------------------------------------------
    for start_vec in starts:
        if time.time() - start > budget_s - 5:
            break
        a = np.maximum(start_vec, eps)
        a = _normalize(a)

        try:
            cur_score = evaluate_sequence(a.tolist())
        except Exception:
            continue
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[restart] initial improvement, score={cur_score:.6f}")

        # ----- LP‑mix phase (a handful of iterations) -----
        for it in range(12):                     # 12 is a safe upper bound
            if time.time() - start > budget_s - 5:
                break
            a, _ = _lp_mix_step(a, mix=0.5, eps=eps)
            new_score = evaluate_sequence(a.tolist())
            if new_score < cur_score - 1e-12:
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_seq = a.copy()
                    print(f"[lp] it={it}, new best={new_score:.6f}")
            else:
                # No progress – abort the LP phase for this restart.
                break

        # ----- Final local polishing with hill climbing -----
        a, cur_score = _hill_climb(a, cur_score, budget_s, start,
                                   max_iters=4000)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[hill‑final] improved to {cur_score:.6f}")

    print(f"[final] best score achieved: {best_score:.6f}")
    return best_seq.tolist()
