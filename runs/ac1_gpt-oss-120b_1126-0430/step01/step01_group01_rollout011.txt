"""
Propose a step‑function that beats the Catalan (arcsine) construction.

The algorithm works in three phases:

1.  Start from the Catalan (arcsine) coefficients – they give the
    classic bound  π/2 ≈ 1.5709.
2.  Solve a linear program that maximises the total L1‑mass of a new
    sequence `g` while keeping the *mixed* convolution `f * g` bounded
    by the current maximal self‑convolution value `M`.  Adding this
    extra mass (and mixing it with the old sequence) increases the
    total sum without raising the peak too much, thereby lowering the
    evaluation metric.
3.  A cheap “gradient‑free” greedy refinement moves a small fraction
    of mass from entries that contribute heavily to the peak towards
    entries that contribute almost nothing.  This finishes the search
    with a final polish step.

All heavy work is done with FFT‑based convolutions (≈ O(n log n)) and
a dense LP solved by SciPy’s HiGHS backend.  The whole routine respects
the 1000 s budget; intermediate scores are printed for monitoring.
"""
import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Fast convolution based metric (identical to evaluate_sequence but
# without the safety checks).
# ----------------------------------------------------------------------
def _fast_metrics(seq: np.ndarray):
    """Return (max_convolution, sum(seq)) for a real‑valued sequence."""
    n = seq.size
    if n == 0:
        return np.inf, 0.0
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    return float(conv.max()), float(seq.sum())


def _fast_evaluation(seq: np.ndarray) -> float:
    """Return the exact score used by ``evaluate_sequence``."""
    n = seq.size
    max_b, sum_a = _fast_metrics(seq)
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------
def _catalan_sequence(length: int) -> np.ndarray:
    """Catalan / arcsine coefficients a_i = binom(2i,i)/4**i."""
    seq = np.empty(length, dtype=float)
    a = 1.0
    seq[0] = a
    for i in range(1, length):
        a *= (2 * i - 1) / (2.0 * i)
        seq[i] = a
    return seq


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1) × n matrix A for the linear constraints
        (f * g)_k = Σ_i f_i * g_{k-i} ≤ M .
    Row k corresponds to one convolution entry, column j to g_j.
    """
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # column j (g_j) contributes f_i at row i+j
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_step(f: np.ndarray, M: float, mix: float = 0.2):
    """
    One linear‑programming improvement step.

    Parameters
    ----------
    f   : current sequence (numpy array, non‑negative)
    M   : current maximal self‑convolution value (float)
    mix : convex‑mixing coefficient (0 < mix ≤ 1)

    Returns
    -------
    new_f : updated sequence (numpy array) or None if no improvement.
    """
    n = f.size
    A = _build_conv_matrix(f)                # (2n‑1) × n
    b = np.full(2 * n - 1, M, dtype=np.float64)

    # maximise Σ g_j  ⇔ minimise -Σ g_j
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        return None

    g = res.x
    sum_f = float(f.sum())
    sum_g = float(g.sum())

    # If the LP could not increase the L1‑mass we stop.
    if sum_g <= sum_f * (1.0 + 1e-9):
        return None

    # Rescale g so that the total L1‑mass stays unchanged.
    scale = sum_f / sum_g
    g_scaled = g * scale

    # Convex combination: f ← (1‑mix)·f + mix·g_scaled
    new_f = (1.0 - mix) * f + mix * g_scaled
    return new_f


# ----------------------------------------------------------------------
# Greedy refinement (identical in spirit to the original script but
# slightly more aggressive – it uses the full contribution vector
# ``a_i * a_{k-i}`` as a proxy for the gradient).
# ----------------------------------------------------------------------
def _refine(seq: np.ndarray, deadline: float) -> np.ndarray:
    """Local gradient‑free hill climbing until time runs out."""
    n = seq.size
    best_seq = seq.copy()
    best_score = _fast_evaluation(best_seq)

    # FFT size does not change during the refinement.
    fft_len = 1 << ((2 * n - 1).bit_length())

    while time.time() < deadline:
        # current convolution and the index of its maximum
        fa = np.fft.rfft(seq, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        k = int(conv.argmax())
        # contribution of every entry to that maximal location
        i_arr = np.arange(n)
        j_arr = k - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        w = np.zeros_like(seq)
        w[mask] = seq[j_arr[mask]]          # w[i] = a_{k‑i}
        # source = entry with largest a_i * w_i (most responsible for the peak)
        src = int(np.argmax(seq * w))
        # sink  = entry that hardly participates
        sink = int(np.argmin(w + 1e-12))    # add epsilon to avoid all‑zero case

        if seq[src] < 1e-12:
            break

        # move a modest fraction of the mass
        eps = 0.07 * seq[src]               # 7 % of the source amount
        cand = seq.copy()
        cand[src] -= eps
        cand[sink] += eps

        cand_score = _fast_evaluation(cand)
        if cand_score < best_score - 1e-9:
            seq = cand
            best_score = cand_score
            best_seq = cand.copy()
        else:
            # occasional tiny random jitter to escape plateaus
            if np.random.rand() < 0.02:
                jitter_idx = np.random.randint(n)
                jitter = np.random.randn() * 0.001
                cand = seq.copy()
                cand[jitter_idx] = max(0.0, cand[jitter_idx] + jitter)
                seq = cand

    return best_seq


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Generate a non‑negative step‑function whose score is (hopefully)
    below the target 1.503.

    The routine:
        • starts from the Catalan (arcsine) seed,
        • repeatedly solves a linear program that adds as much L1‑mass
          as possible without raising the current convolution peak,
        • mixes the new mass into the current sequence,
        • finally finishes with a fast greedy refinement.

    The whole search respects the supplied time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # tiny safety margin

    # ----- 1.  seed -----
    n = 2500                                 # a few thousand entries give a good trade‑off
    seq = _catalan_sequence(n)

    # Normalise to the classical √(2n) just for nicer numbers (optional)
    target_sum = np.sqrt(2.0 * n)
    seq = seq * (target_sum / seq.sum())

    # ----- 2.  LP driven improvement -----
    mix = 0.25                                # start with a fairly aggressive mixing weight
    while time.time() < deadline:
        max_b, _ = _fast_metrics(seq)
        new_seq = _lp_step(seq, max_b, mix=mix)
        if new_seq is None:
            # No further LP improvement – switch to greedy mode
            break
        seq = new_seq
        # gradually reduce the mixing factor so later steps become finer
        mix = max(0.05, mix * 0.97)

        # keep an eye on the current score
        cur_score = _fast_evaluation(seq)
        print(f"[LP] n={n}, mix={mix:.3f}, max_b={max_b:.6f}, score={cur_score:.6f}")

    # ----- 3.  greedy polish -----
    seq = _refine(seq, deadline)

    # Final sanity check – clip to the limits enforced by ``evaluate_sequence``
    seq = np.clip(seq, 0.0, 1000.0)

    final_score = _fast_evaluation(seq)
    print(f"Final score: {final_score:.6f}  (length={len(seq)})")
    return seq.tolist()
