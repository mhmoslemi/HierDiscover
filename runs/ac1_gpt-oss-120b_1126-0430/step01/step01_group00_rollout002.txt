def convolve_fft(a):
    n = len(a)
    fft_len = 1 << (2*n-1).bit_length()  # next power of two
    FA = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(FA*FA, fft_len)[:2*n-1]
    return conv

"""
Hybrid sub‑gradient + simulated‑annealing optimizer.

* Works on the probability simplex (∑a_i = 1, a_i ≥ 0).
* Convolution is computed with an FFT (O(n log n) instead of O(n²)).
* In each iteration we locate the index k where the auto‑convolution is maximal,
  build the exact sub‑gradient of that entry,
  take a projected step of size limited by positivity,
  and accept the move greedily (with a tiny Metropolis‑type probability for
  uphill moves).  This aggressively pushes down the current peak.
* Occasionally we perform a random “mass‑transfer” move to escape local minima.
* The step‑size slowly decays, allowing a final fine‑tuning phase.
* The routine respects the 1000 s time budget, prints progress, and finally
  checks the result with the provided evaluate_sequence().
"""

import time
import math
import numpy as np

# -------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# -------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using real FFT (length 2*len(a)-1)."""
    n = a.shape[0]
    size = 1 << (2 * n - 1).bit_length()      # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


# -------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Objective on the simplex: 2·n·max(conv(a,a))."""
    n = a.shape[0]
    conv = _conv_fft(a)
    return 2.0 * n * conv.max()


# -------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """
    Exact sub‑gradient of the maximal convolution entry at index k.
    For k = argmax(conv) the derivative w.r.t. a_i is:
        2·a_{k-i}          if i ≠ k-i
        a_i                if i = k-i (i.e. 2·i = k)
    Indices outside the valid range contribute 0.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    idx = np.arange(i_start, i_end + 1, dtype=int)
    jdx = k - idx
    # generic case
    g[idx] = 2.0 * a[jdx]
    # correct the diagonal terms (i == j)
    diag_mask = idx == jdx
    if np.any(diag_mask):
        g[idx[diag_mask]] = a[idx[diag_mask]]
    return g


# -------------------------------------------------------------
