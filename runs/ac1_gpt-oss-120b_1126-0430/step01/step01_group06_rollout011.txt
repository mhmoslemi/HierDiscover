"""
Hybrid search for a low‑peak step‑function.

The algorithm combines three complementary stages:

1. **Grid search** over the two‑parameter family
      a_i ∝ (i + shift)^{‑α},
   with a fine grid for α and shift.  All candidates are normalised
   to the simplex (sum = 1) because the objective
        2·n·max(conv(a,a)) / (sum a)²
   is scale‑invariant.

2. **Random sampling** – Dirichlet draws with a wide range of
   concentration parameters and noisy power‑law perturbations.
   This quickly discovers diverse promising shapes.

3. **Projected gradient descent (PGD)** on the best candidates.
   The objective is  f(a)=2·n·max(conv(a,a))  with the constraint
   a≥0, sum(a)=1.  The sub‑gradient of the max‑convolution is
   computed analytically (it is a simple slice of the vector),
   a step is taken opposite to the gradient and the result is
   projected back onto the simplex.  A simple adaptive step‑size
   scheme and occasional random “shakes” help escape local minima.

All work is time‑boxed: the routine stops a few seconds before the
provided ``budget_s`` and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (must be top‑level – no closures)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, sum x_i = 1}
    Implementation follows Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero – return uniform
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_max(seq: np.ndarray) -> float:
    """Return max(conv(seq, seq)).  ``seq`` is assumed to sum to 1."""
    conv = np.convolve(seq, seq)
    return float(conv.max())


def _score(seq: np.ndarray) -> float:
    """Exact value of the evaluation function for a normalised sequence."""
    n = seq.shape[0]
    return 2.0 * n * _conv_max(seq)


def _subgradient_max_conv(seq: np.ndarray, max_idxs: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of  max_i conv_i(seq)  at the current point.
    If several indices attain the maximum the gradients are averaged.
    """
    n = seq.shape[0]
    grad = np.zeros_like(seq)
    for m in max_idxs:
        low = max(0, m - (n - 1))
        high = min(n - 1, m)
        i_vals = np.arange(low, high + 1)
        j_vals = m - i_vals
        eq = i_vals == j_vals
        # when i == j the term a_i * a_j appears once
        grad[i_vals[eq]] += seq[i_vals[eq]]
        # otherwise the product appears twice (j,i) and (i,j)
        grad[i_vals[~eq]] += 2.0 * seq[j_vals[~eq]]
    if max_idxs.size > 0:
        grad /= float(max_idxs.size)
    return grad


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: float = 950, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the time budget
    (default 950 s) and returns the best list of heights it discovered.
    """
    np.random.seed(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # Determine problem size (use the length of the provided baseline)
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)  # provided by the evaluation harness
    except Exception:
        n = 2244                     # fallback length mentioned in the description

    # ------------------------------------------------------------------
    # Initialise with the supplied construction (fallback)
    # ------------------------------------------------------------------
    fallback = np.array(height_sequence_1, dtype=float)
    fallback = np.clip(fallback, 0.0, 1000.0)          # mimic evaluate_sequence’s clipping
    if fallback.sum() < 0.01:                         # avoid degenerate case
        fallback = np.full(n, 1.0 / n)
    fallback = _project_to_simplex(fallback)          # enforce sum = 1
    best_seq = fallback.copy()
    best_score = _score(best_seq)
    print(f"[init] n={n}, baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1. Grid search over power‑law families a_i = (i+shift)^{-α}
    # ------------------------------------------------------------------
    idx = np.arange(n, dtype=float)                  # i = 0 … n‑1
    alphas = np.linspace(0.45, 0.80, 71)            # step ≈0.005
    shifts = np.linspace(0.5, 3.0, 16)               # 0.5 … 3.0
    for alpha in alphas:
        if time.time() - start > budget_s - 20:
            break
        for shift in shifts:
            raw = np.power(idx + shift, -alpha)
            seq = raw / raw.sum()
            score = _score(seq)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"[grid] α={alpha:.4f}, shift={shift:.3f}, score={score:.6f}")

    # ------------------------------------------------------------------
    # 2. Random sampling (Dirichlet & noisy power‑law)
    # ------------------------------------------------------------------
    random_iter = 0
    while time.time() - start < budget_s - 15 and random_iter < 20000:
        random_iter += 1
        if np.random.rand() < 0.5:
            # Dirichlet draw with random concentration
            conc = 10 ** np.random.uniform(-1.0, 1.0)   # 0.1 … 10
            vec = np.random.gamma(conc, 1.0, size=n)
            seq = vec / vec.sum()
        else:
            # Power‑law with tiny random noise
            alpha = np.random.uniform(0.40, 0.90)
            shift = np.random.uniform(0.5, 3.0)
            raw = np.power(idx + shift, -alpha)
            noise = np.random.rand(n) * np.random.uniform(0.0, 0.01)  # ≤ 1 %
            raw = raw * (1.0 + noise)
            seq = raw / raw.sum()
        score = _score(seq)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[rand] iter={random_iter}, score={score:.6f}")

    # ------------------------------------------------------------------
    # 3. Projected Gradient Descent on a few promising seeds
    # ------------------------------------------------------------------
    # collect a small pool of starting points (the global best + a couple of
    # random Dirichlet vectors)
    candidates = [best_seq.copy()]
    for _ in range(2):
        conc = 10 ** np.random.uniform(-0.5, 0.5)
        vec = np.random.gamma(conc, 1.0, size=n)
        candidates.append(vec / vec.sum())

    # PGD parameters
    max_total_iters = 300_000
    iter_done = 0
    lr = 0.20                     # initial step size (in the simplex)
    min_lr = 1e-7

    while time.time() - start < budget_s - 5 and iter_done < max_total_iters:
        for cid, a in enumerate(candidates):
            if time.time() - start >= budget_s - 5:
                break

            # current objective value
            cur_score = _score(a)

            # convolution and indices where the maximum is attained
            conv = np.convolve(a, a)
            max_val = conv.max()
            max_idxs = np.where(conv == max_val)[0]

            # sub‑gradient of the max‑convolution
            grad = _subgradient_max_conv(a, max_idxs)

            # try a gradient step
            step = lr * grad
            a_new = _project_to_simplex(a - step)
            new_score = _score(a_new)

            iter_done += 1

            if new_score < cur_score - 1e-12:
                # accept move
                candidates[cid] = a_new
                if new_score < best_score:
                    best_score = new_score
                    best_seq = a_new.copy()
                    print(f"[pgd] iter={iter_done}, cand={cid}, score={new_score:.6f}")
                # gently increase step size
                lr = min(lr * 1.02, 1.0)
            else:
                # reject – shrink step size
                lr = max(lr * 0.70, min_lr)

            # occasional random “shake” to escape shallow plateaus
            if iter_done % 5_000 == 0:
                i, j = np.random.randint(n, size=2)
                if i != j:
                    delta = a[i] * np.random.rand() * 0.05   # up to 5 % of source mass
                    shaken = a.copy()
                    shaken[i] -= delta
                    shaken[j] += delta
                    shaken = _project_to_simplex(shaken)
                    shake_score = _score(shaken)
                    if shake_score < cur_score:
                        candidates[cid] = shaken
                        if shake_score < best_score:
                            best_score = shake_score
                            best_seq = shaken.copy()
                            print(f"[shake] iter={iter_done}, cand={cid}, score={shake_score:.6f}")

    print(f"[final] best score found: {best_score:.6f}")
    # convert to plain Python list as required by the harness
    return best_seq.tolist()
