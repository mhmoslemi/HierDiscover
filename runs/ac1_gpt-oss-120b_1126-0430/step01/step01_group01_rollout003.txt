def simplex_projection(v):
    # v is np.ndarray
    # returns projection onto simplex sum=1, non-negative
    if np.sum(v) == 1 and np.all(v >= 0):
        return v
    n = len(v)
    u = np.sort(v)[::-1]  # descending
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # find rho: max i s.t. u_i - cssv_i / i > 0
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0].max()
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def propose_candidate(seed=42, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5
    n = 4000
    # initial arcsine
    seq = _catalan_sequence(n)
    # normalize sum to 1
    seq = seq / seq.sum()
    best_seq = seq.copy()
    best_score = _fast_evaluation(seq)
    # gradient descent parameters
    step0 = 0.4  # initial step size coefficient
    iteration = 0
    while time.time() < deadline:
        # compute conv and max index
        fft_len = 1 << ((2 * n - 1).bit_length())
        fa = np.fft.rfft(seq, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[:2 * n - 1]
        k_max = int(conv.argmax())
        max_val = conv[k_max]
        # gradient of max entry
        g = np.zeros_like(seq)
        # contributions from positions i where j=k_max - i in range
        i_arr = np.arange(n)
        j_arr = k_max - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        g[mask] = 2.0 * seq[j_arr[mask]]
        # step size schedule
        step = step0 / np.sqrt(iteration + 1)
        # candidate updates with backtracking line search
        improved = False
        for factor in (1.0, 0.5, 0.25, 0.125):
            delta = - step * factor * g
            cand = seq + delta
            # project onto simplex
            cand = simplex_projection(cand)
            score = _fast_evaluation(cand)
            if score < best_score - 1e-12:
                seq = cand
                best_score = score
                best_seq = cand.copy()
                improved = True
                break
        if not improved:

            if np.random.rand() < 0.01:
                # random small Gaussian jitter

"""
Propose a low‑score height sequence.

The algorithm works in three stages:

1.  **Seed** – start from the Catalan (discrete arcsine) coefficients,
    which give the classic bound ≈ π/2 ≈ 1.57.
2.  **Projected sub‑gradient descent** – the objective for a
    normalised sequence (sum = 1) is simply 2·n·max(conv(a,a)).
    At each iteration we compute the index k where the convolution attains
    its maximum, form the sub‑gradient g_i = 2·a_{k‑i} (zero when out of
    range), take a small step in the opposite direction, clip to
    non‑negative values and re‑normalise (projection onto the simplex).
    A diminishing step size together with a cheap back‑tracking line‑search
    quickly drives the maximal convolution value down.
3.  **Greedy refinement** – a fast “mass‑transfer’’ heuristic (the one
    used in the reference code) is applied for the remaining time to
    polish the solution.

Both stages use an FFT‑based convolution for speed, allowing thousands of
iterations within the 1000 s budget.  The final sequence is returned as a
plain list of non‑negative floats; the evaluation harness will clip the
values to the required range automatically.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast evaluation (identical to evaluate_sequence but without the checks)
# ----------------------------------------------------------------------
def _fast_evaluation(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    n = seq.size
    if n == 0:
        return np.inf
    # next power of two for the convolution
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Catalan (discrete arcsine) seed
# ----------------------------------------------------------------------
def _catalan_sequence(length: int) -> np.ndarray:
    """First `length` Catalan‑type coefficients a_i = C(2i,i)/4**i."""
    seq = np.empty(length, dtype=float)
    a = 1.0
    seq[0] = a
    for i in range(1, length):
        a *= (2 * i - 1) / (2.0 * i)      # a_i = a_{i-1} * (2i‑1)/(2i)
        seq[i] = a
    return seq


# ----------------------------------------------------------------------
# Projection onto the probability simplex (sum = 1, non‑negative)
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex {x≥0, sum x = 1}."""
    # based on the algorithm by Duchi et al. (2008)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0].max()
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Projected sub‑gradient optimisation of max(convolution)
# ----------------------------------------------------------------------
def _sgd_optimize(seq: np.ndarray, deadline: float) -> np.ndarray:
    """Run a projected sub‑gradient descent on `seq` until `deadline`."""
    n = seq.size
    best_seq = seq.copy()
    best_score = _fast_evaluation(best_seq)

    # pre‑compute FFT length (doesn't change)
    fft_len = 1 << ((2 * n - 1).bit_length())
    iteration = 0
    stagnation = 0

    while time.time() < deadline:
        # current convolution and location of the maximum
        fa = np.fft.rfft(seq, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        k_max = int(conv.argmax())

        # sub‑gradient of the max entry (2·a_{k‑i})
        i_arr = np.arange(n)
        j_arr = k_max - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        g = np.zeros_like(seq)
        g[mask] = 2.0 * seq[j_arr[mask]]

        # diminishing step size
        step = 0.6 / np.sqrt(iteration + 1.0)

        improved = False
        # try a small back‑tracking line search
        for factor in (1.0, 0.5, 0.25):
            cand = seq - step * factor * g
            # enforce non‑negativity and re‑normalise (simple simplex projection)
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s == 0.0:
                continue
            cand = cand / s
            score = _fast_evaluation(cand)
            if score < best_score - 1e-12:
                seq = cand
                best_seq = cand.copy()
                best_score = score
                improved = True
                stagnation = 0
                break

        if not improved:
            stagnation += 1
            # occasional random jitter to escape flat plateaus
            if np.random.rand() < 0.01:
                idx = np.random.randint(n)
                jitter = np.random.randn() * 0.001
                cand = seq.copy()
                cand[idx] = max(0.0, cand[idx] + jitter)
                cand = cand / cand.sum()
                score = _fast_evaluation(cand)
                if score < best_score:
                    seq = cand
                    best_seq = cand.copy()
                    best_score = score
                    stagnation = 0

            # stop if we have been stuck for a long time
            if stagnation > 5000:
                break

        iteration += 1
        # occasional progress report
        if iteration % 2000 == 0:
            print(f"[SGD] iter {iteration:,d}, best_score={best_score:.6f}")

    return best_seq


# ----------------------------------------------------------------------
# Greedy refinement (the method from the reference solution)
# ----------------------------------------------------------------------
def _refine(seq: np.ndarray, deadline: float) -> np.ndarray:
    """Local greedy improvement of `seq` (same as the reference `_refine`)."""
    n = seq.size
    best_seq = seq.copy()
    best_score = _fast_evaluation(best_seq)

    fft_len = 1 << ((2 * n - 1).bit_length())

    while time.time() < deadline:
        fa = np.fft.rfft(seq, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        k = int(conv.argmax())
        # contribution of each position to this maximal entry
        i_arr = np.arange(n)
        j_arr = k - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        w = np.zeros_like(seq)
        w[mask] = seq[j_arr[mask]]

        contrib = seq * w
        src = int(np.argmax(contrib))
        sink = int(np.argmin(w))          # usually a location with almost zero influence

        if seq[src] < 1e-12:
            break

        eps = 0.04 * seq[src]             # move a modest fraction (4 %)
        new_seq = seq.copy()
        new_seq[src] -= eps
        new_seq[sink] += eps

        score = _fast_evaluation(new_seq)
        if score < best_score - 1e-12:
            seq = new_seq
            best_seq = new_seq.copy()
            best_score = score
        else:
            # tiny random jitter to avoid tiny local plateaus
            if np.random.rand() < 0.01:
                jitter_idx = np.random.randint(n)
                jitter = np.random.randn() * 0.001
                new_seq = seq.copy()
                new_seq[jitter_idx] = max(0.0, new_seq[jitter_idx] + jitter)
                seq = new_seq

    return best_seq

# ----------------------------------------------------------------------

    """
    Returns a list of non‑negative heights with a low evaluation score.

    The routine runs for at most ``budget_s`` seconds, using a Catalan
    seed, a projected sub‑gradient phase, and a final greedy refinement.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5   # a small safety margin

    # length – a few thousand gives a good balance between quality and speed
    n = 4000

    # --------------------------------------------------------------
    # 1.  Initial seed – Catalan (discrete arcsine) coefficients,
    #     normalised to a probability vector (sum = 1).
    # --------------------------------------------------------------
    seq = _catalan_sequence(n).astype(float)
    seq = seq / seq.sum()          # project onto the simplex

    # --------------------------------------------------------------
    # 2.  Projected sub‑gradient optimisation.
    # --------------------------------------------------------------
    seq = _sgd_optimize(seq, deadline)

    # --------------------------------------------------------------
    # 3.  Greedy refinement on the remaining time.
    # --------------------------------------------------------------
    if time.time() < deadline:
        seq = _refine(seq, deadline)

    # --------------------------------------------------------------
    # 4.  Return the best sequence as a plain Python list.
    # --------------------------------------------------------------
    return seq.tolist()
