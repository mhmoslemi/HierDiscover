"""
Propose a step‑function sequence that minimizes the bound
    O = 2·n·max(conv(a,a))   (sum(a)=1, a_i ≥ 0).

The algorithm combines three ideas:

1. **LP improvement** – for the current sequence `a` we compute the
   maximal self‑convolution `M`.  Then we solve the linear program

        max   sum(g)
        s.t.  conv(a,g) ≤ M        (element‑wise)
              g ≥ 0

   The solution `g` can carry strictly more total mass than `a` while
   respecting the same convolution bound.  After normalisation
   (`g ← g / sum(g)`) we mix `a` and `g` (a small convex combination)
   which provably does not increase `M`.  Repeating this step drives the
   objective down.

2. **Mixed convex combination** – instead of a fixed mixing coefficient
   we try a few values (0.05 … 0.5) and keep the best result.

3. **Fine‑grained stochastic hill‑climbing** – once the LP phase stalls,
   a fast Monte‑Carlo/Simulated‑Annealing loop with sub‑gradient steps
   refines the solution further.

The whole routine respects the 1000 s time budget, prints progress,
and finally returns the best sequence found (as a plain list of floats).

# Projection onto the probability simplex (Duchi et al.)
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# -------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return the evaluation objective 2·n·max(conv(a,a)) for sum(a)=1."""
    conv = np.convolve(a, a)
    return 2.0 * a.size * conv.max()


# -------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv(a,a)) w.r.t. a on the simplex."""
    n = a.size
    conv = np.convolve(a, a)
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    g /= len(ks)
    return g


# -------------------------------------------------------------
def _solve_lp_max_sum(a: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the LP:
        max sum(g)
        s.t. conv(a,g) ≤ M   (element‑wise)
             g ≥ 0
    Returns the optimal g (may have sum > 1) or None on failure.
    """
    if not _CVX_AVAILABLE:                     # pragma: no cover
        return None

    n = a.shape[0]
    a_param = cp.Parameter(n)
    a_param.value = a
    g = cp.Variable(n, nonneg=True)
    conv = cp.conv(a_param, g)                # length 2n‑1
    constraints = [conv <= M]                # vector inequality
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    # ECOS is a good LP/second‑order cone solver for this size
    try:
        prob.solve(solver=cp.ECOS, warm_start=True,
                    max_iters=4000, abstol=1e-9, reltol=1e-9,
                    feastol=1e-9, verbose=False)
    except Exception:                         # pragma: no cover
        return None

    if prob.status in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
        g_val = np.asarray(g.value, dtype=float)
        # Numerical safety – enforce non‑negativity and simplex projection later
        g_val = np.maximum(g_val, 0.0)
        return g_val
    return None


# -------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the provided
    evaluate_sequence() bound.  Returns the sequence as a list of floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # -----------------------------------------------------------------
    # Choose a starting length.
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=float)
        a = a / a.sum()
        n = a.size
    else:
        n = 3000
        a = np.random.rand(n)
        a = _project_to_simplex(a)

    best_seq = a.copy()
    best_score = _objective(best_seq)
    print(f"[init] n={n}, objective={best_score:.6f}")

    # -----------------------------------------------------------------
    # Phase 1 – LP driven improvement (few iterations, relatively cheap)
    lp_iter = 0
    mix_t_vals = np.linspace(0.05, 0.5, 10)   # candidates for mixing weight
    while time.time() < deadline and lp_iter < 30:
        lp_iter += 1
        conv_self = np.convolve(a, a)
        M = conv_self.max()

        g0 = _solve_lp_max_sum(a, M)
        if g0 is None:
            print(f"[LP {lp_iter}] LP failed or infeasible – exiting LP phase.")
            break

        sum_g0 = g0.sum()
        print(f"[LP {lp_iter}] sum(g)={sum_g0:.6f}")

        # No improvement possible if the LP cannot increase the total mass.
        if sum_g0 <= 1.0 + 1e-8:
            print(f"[LP {lp_iter}] sum(g)≈1, stop LP iterations.")
            break

        # Normalise the candidate to the simplex.
        g = g0 / sum_g0

        # Try several convex combinations a←(1‑α)a+αg and keep the best.
        best_local_a = a
        best_local_score = _objective(a)
        for alpha in mix_t_vals:
            a_c = (1.0 - alpha) * a + alpha * g
            a_c = a_c / a_c.sum()          # numerical re‑normalisation
            sc = _objective(a_c)
            if sc < best_local_score:
                best_local_score = sc
                best_local_a = a_c

        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local_a.copy()
            print(f"[LP {lp_iter}] New global best O={best_score:.6f}")

        a = best_local_a

    # -----------------------------------------------------------------
    # Phase 2 – Stochastic hill‑climbing + occasional sub‑gradient steps
    cur_score = best_score
    a = best_seq.copy()
    T0 = 0.02
    T = T0
    iter_no_improve = 0
    max_no_improve = 5000
    iter_cnt = 0

    while time.time() < deadline and iter_cnt < 300_000:
        iter_cnt += 1

        # ---- random mass transfer -------------------------------------------------
        i, j = np.random.randint(n, size=2)
        if i != j and a[i] > 0.0:
            delta = a[i] * np.random.rand() * 0.12   # move up to 12 % of a[i]
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            a_new = _project_to_simplex(a_new)

            new_score = _objective(a_new)

            # Metropolis acceptance (temperature T)
            if new_score < cur_score or np.random.rand() < math.exp((cur_score - new_score) / max(T, 1e-12)):
                a = a_new
                cur_score = new_score
                iter_no_improve = 0
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(f"[HC {iter_cnt}] New global best O={best_score:.6f}")
            else:
                iter_no_improve += 1

        # ---- occasional sub‑gradient descent on the peak -------------------------
        if iter_cnt % 13 == 0:
            g_sub = _subgradient_max_conv(a)
            eta = 0.07 * (1.0 - iter_cnt / 300_000.0)
            a = a - eta * g_sub
            a = np.clip(a, 0.0, None)
            a = _project_to_simplex(a)
            cur_score = _objective(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[SG {iter_cnt}] New global best O={best_score:.6f}")

        # ---- temperature decay ---------------------------------------------------
        T = T0 * (1.0 - iter_cnt / 300_000.0)

        # ---- restart if stuck ----------------------------------------------------
        if iter_no_improve >= max_no_improve:
            # small random perturbation to escape a flat region
            a = best_seq.copy()
            a += np.random.randn(n) * 0.003
            a = np.clip(a, 0.0, None)
            a = _project_to_simplex(a)
            cur_score = _objective(a)
            iter_no_improve = 0
            print(f"[restart] after {iter_cnt} iter, re‑initialised from best.")

    # -----------------------------------------------------------------
    # Final check with the provided evaluate_sequence (exact scoring)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:                     # pragma: no cover
        print("[final] evaluate_sequence call failed:", e)

    return best_seq.tolist()
