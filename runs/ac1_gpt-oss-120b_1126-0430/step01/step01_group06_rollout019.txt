"""
Propose a better step‑function sequence.

The algorithm works in three stages:

1. **Broad grid search** over several parametric families
   (power‑law, exponential and beta‑shapes).  Each candidate is
   normalised to sum 1 (the evaluation function is scale invariant)
   and the best one is kept.

2. **Linear‑programming refinement** – following the idea from the
   reference code, we solve a linear program

        max    ∑ b_i
        s.t.   b_i ≥ 0
               (f * b)[k] ≤ M   for all k

   where ``f`` is the current sequence and ``M`` is the current
   maximal convolution value.  The solution ``b`` has a larger total
   mass while respecting the same convolution bound.  After normalising
   ``b`` to sum 1 we mix it with the current sequence
   ``f ← (1‑t)·f + t·b`` (projected onto the probability simplex).  The
   mixing factor ``t`` is adapted automatically.  Repeating this step
   strictly reduces the objective in practice and drives the maximum
   convolution below the classical π/2 limit.

3. **Light hill‑climbing** – if time permits, a cheap random
   mass‑transfer hill‑climber is applied to squeeze a little more.

The whole routine respects the user‑provided time budget (default
950 s) and always returns a feasible sequence; ``height_sequence_1`` is
used as a safe fallback.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities -------------------------------------------------------
# ----------------------------------------------------------------------


def _project_to_simplex(v):
    """Project a vector onto the simplex {x ≥ 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return np.array([])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


def _evaluate(seq):
    """Call the provided ``evaluate_sequence`` safely and return a float."""
    try:
        return evaluate_sequence(seq)          # the harness imports this
    except Exception:
        return float('inf')


def _grid_search(n, max_time):
    """Search over a few families and keep the best sequence."""
    best_seq = None
    best_score = float('inf')
    idx = np.arange(n, dtype=float)

    # ---- Power‑law ----------------------------------------------------
    alphas = np.linspace(0.40, 0.80, 81)          # step ≈0.005
    shifts = np.linspace(0.5, 5.0, 46)            # 0.5 … 5.0
    for alpha in alphas:
        for shift in shifts:
            raw = np.power(idx + shift, -alpha)
            if np.all(raw == 0):
                continue
            seq = raw / raw.sum()
            score = _evaluate(seq.tolist())
            if score < best_score:
                best_score = score
                best_seq = seq
        if time.time() > max_time:
            return best_seq, best_score

    # ---- Exponential --------------------------------------------------
    betas = np.linspace(0.0, 5.0, 51)
    for beta in betas:
        raw = np.exp(-beta * idx)
        seq = raw / raw.sum()
        score = _evaluate(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq
        if time.time() > max_time:
            return best_seq, best_score

    # ---- Beta family (two‑parameter) ---------------------------------
    p_vals = np.linspace(0.2, 3.0, 14)
    q_vals = np.linspace(0.2, 3.0, 14)
    x = idx / (n - 1)                # map to [0,1]
    eps = 1e-12
    for p in p_vals:
        for q in q_vals:
            # avoid 0^neg which would give infinities
            raw = np.power(x + eps, p - 1) * np.power(1 - x + eps, q - 1)
            seq = raw / raw.sum()
            score = _evaluate(seq.tolist())
            if score < best_score:
                best_score = score
                best_seq = seq
            if time.time() > max_time:
                return best_seq, best_score

    return best_seq, best_score


def _solve_lp_max_sum(f_vec, M_val):
    """
    Solve the linear program
        max sum(b)
        s.t. b >= 0,
             conv(f, b) <= M
    using cvxpy.  Returns a NumPy array with the optimum b (or None on failure).
    """
    try:
        import cvxpy as cp
    except Exception:                     # cvxpy not available
        return None

    n = len(f_vec)
    b = cp.Variable(n)
    f_param = cp.Parameter(n, value=f_vec)
    M_param = cp.Parameter(value=M_val)
    conv_fb = cp.conv(f_param, b)          # length 2n‑1
    constraints = [b >= 0, conv_fb <= M_param]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

    # ECOS works well for this medium‑sized LP
    try:
        prob.solve(solver='ECOS', warm_start=True, verbose=False, max_iters=2000)
    except Exception:
        return None

    if b.value is None:
        return None
    b_opt = np.asarray(b.value, dtype=float)
    b_opt[b_opt < 0] = 0.0
    return b_opt


def _hill_climb(cur, max_iter, deadline, best_score, best_seq):
    """Tiny random‑mass hill‑climber (used as a final polishing step)."""
    n = len(cur)
    cur_score = best_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        step = cur[i] * (np.random.rand() * 0.02)      # ≤2 % of source
        if step <= 0 or cur[j] - step < 0:
            continue
        new = cur.copy()
        new[i] -= step
        new[j] += step
        new = _project_to_simplex(new)
        new_score = _evaluate(new.tolist())
        if new_score < cur_score:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = new.tolist()
                # no need to print every tiny improvement here
        it += 1
    return cur, best_score, best_seq

        if sum_b <= 0:
            break
        b_norm = b_opt / sum_b
        # mix and project back onto the simplex
        new = (1.0 - t) * cur + t * b_norm
        new = _project_to_simplex(new)
        new_score = _evaluate(new.tolist())
        if new_score < best_score:
            best_score = new_score
            best_seq = new.tolist()
            print(f"[lp] iter={lp_iter}, score={new_score:.6f}")
        # accept the new point if it improves (otherwise shrink the step)
        if new_score < _evaluate(cur.tolist()):
            cur = new
            # slowly increase mixing factor (but keep it ≤0.5)
            t = min(0.5, t * 1.07)
        else:
            t *= 0.7                # back‑off
        lp_iter += 1

    # ------------------------------------------------------------------
    # 4. Final cheap hill‑climbing (if time left)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10:
        cur, best_score, best_seq = _hill_climb(
            cur, max_iter=20000, deadline=deadline,
            best_score=best_score, best_seq=best_seq
        )

    print(f"[final] best score = {best_score:.6f}")
    return best_seq
