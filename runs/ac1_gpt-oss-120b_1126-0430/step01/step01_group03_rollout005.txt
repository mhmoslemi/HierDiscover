def compute_score(seq):
    # seq: numpy array, non-negative floats
    n = len(seq)
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return float('inf')
    # FFT length
    L = 1 << ((2*n - 1).bit_length())
    # Use real FFT
    fft_seq = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fft_seq * fft_seq, n=L)[:2*n-1]
    max_b = np.max(conv)
    return 2.0 * n * max_b / (sum_a * sum_a)

"""
Propose a step‑function sequence that minimises
    2·n·max_k (p*p)_k
where p_i = a_i / sum(a).  The algorithm is:

1.  If a previous construction is available via the global
    ``height_sequence_1`` it is taken as the initial best candidate.

2.  Scan a family of truncated power‑law sequences
        a_i = (i+1)^(-β) ,  i = 0,…,n‑1
    for several lengths n (2000, 5000, 10000, 20000) and for a dense
    range of exponents β around the empirically good value 0.60.
    The score is computed with a fast FFT‑based convolution
    (identical to ``evaluate_sequence`` but much quicker).

3.  Around the best (n,β) found in step 2 a finer grid search for β
    and a small grid of neighbouring lengths is performed.

4.  A light local optimiser (hill‑climb) tweaks the best sequence:
    repeatedly move a tiny amount of mass from an index that contributes
    most to the current maximal convolution entry to a random other
    index.  Only improving moves are accepted.

The whole routine respects the given time budget (default 1000 s) and
returns the best sequence found as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level – no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def compute_score(seq: np.ndarray) -> float:
    """
    Fast evaluation of the objective for a given sequence.
    Equivalent to ``evaluate_sequence`` but uses an FFT based convolution.
    """
    n = seq.shape[0]
    total = float(seq.sum())
    if total < 0.01:                 # same protection as in evaluate_sequence
        return float("inf")
    # length for zero‑padded FFT (power of two, at least 2·n-1)
    L = _next_pow2(2 * n - 1)
    # real FFT: rfft → pointwise product → irfft
    fft_seq = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fft_seq * fft_seq, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    return 2.0 * n * max_b / (total * total)


def power_law_sequence(n: int, beta: float) -> np.ndarray:
    """Return a truncated power‑law a_i = (i+1)^(-beta)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    return i ** (-beta)


def hill_climb(
    seq: np.ndarray,
    deadline: float,
    max_move_frac: float = 0.05,
    max_iters: int = 10_000,
) -> tuple[np.ndarray, float]:
    """
    Simple stochastic local optimisation.
    Moves a tiny fraction of mass from the index contributing most to the
    current maximal convolution entry to a random other index.
    Only moves that improve the score are kept.
    """
    n = seq.shape[0]
    # Pre‑compute a power‑of‑two length for repeated FFTs
    L = _next_pow2(2 * n - 1)

    cur_score = compute_score(seq)
    best_seq = seq.copy()
    best_score = cur_score

    it = 0
    while time.time() < deadline and it < max_iters:
        # FFT of current sequence (reuse in the loop)
        fft_seq = np.fft.rfft(seq, n=L)

        # convolution (only need the location of the maximum)
        conv = np.fft.irfft(fft_seq * fft_seq, n=L)[: 2 * n - 1]
        k_max = int(conv.argmax())

        # indices that actually appear in this maximal entry
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        idx = np.arange(lo, hi + 1)
        # contribution of each i to conv[k_max] is a_i * a_{k_max-i}
        contrib = seq[idx] * seq[k_max - idx]
        # pick the index with the largest contribution
        i = int(idx[np.argmax(contrib)])

        # pick a random different destination index
        j = np.random.randint(0, n - 1)
        if j >= i:
            j += 1

        # transfer a small amount of mass from i to j
        eps = seq[i] * np.random.rand() * max_move_frac
        if eps <= 0.0:
            it += 1
            continue
        new_seq = seq.copy()
        new_seq[i] -= eps
        new_seq[j] += eps

        # evaluate the new sequence
        new_score = compute_score(new_seq)

        if new_score < cur_score:
            seq = new_seq
            cur_score = new_score
            if new_score < best_score:
                best_seq = new_seq.copy()
                best_score = new_score
                print(
                    f"[hill] iter {it:5d} → improved score {new_score:.6f} "
                    f"(k_max={k_max}, i={i}, j={j})"
                )
        it += 1

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the grading harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) with a low
    value of ``evaluate_sequence``.  The function follows the plan described
    in the module docstring and respects the given time budget.
    """
    np.random.seed(seed)
    start = time.time()
    # leave a small safety margin for the caller
    deadline = start + budget_s - 0.5

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0.  Initialise from a previously found construction, if any.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(height_sequence_1, dtype=np.float64)
            score = compute_score(cand)
            best_seq = cand.copy()
            best_score = score
            print(f"[init] Using height_sequence_1, score = {score:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Scan power‑law families for several lengths.
    # ------------------------------------------------------------------
    # Lengths to try – modestly large but still cheap for FFT.
    lengths = [2000, 5000, 10000, 20000]
    # Rough beta grid (covers the region known to be good)
    beta_grid_coarse = np.linspace(0.40, 0.80, 41)   # step 0.01

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"[scan] Exploring length n={n}")
        for beta in beta_grid_coarse:
            if time.time() > deadline:
                break
            seq = power_law_sequence(n, beta)
            score = compute_score(seq)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"[scan] n={n:5d}, β={beta:0.3f} → new best = {score:.6f}")

    # ------------------------------------------------------------------
    # 2.  Refine around the best (n,β) found so far.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        # extract the length and an approximate beta from the current best
        best_n = best_seq.shape[0]

        # fine beta grid centred on the best‑beta (±0.02, step 0.001)
        # we obtain the beta that produced the best sequence by a small search:
        # (re‑run the coarse grid for the current best_n to locate its β)
        fine_candidates = []
        for beta in np.linspace(0.40, 0.80, 81):   # step 0.005
            seq = power_law_sequence(best_n, beta)
            fine_candidates.append((beta, compute_score(seq)))
        beta_best = min(fine_candidates, key=lambda x: x[1])[0]

        # now really fine‑grained search
        for beta in np.arange(beta_best - 0.02, beta_best + 0.021, 0.001):
            if time.time() > deadline:
                break
            seq = power_law_sequence(best_n, beta)
            score = compute_score(seq)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(
                    f"[refine] n={best_n:5d}, β={beta:0.4f} → new best = {score:.6f}"
                )

        # also try lengths a little larger/smaller than best_n
        for n in (best_n - 2000, best_n + 2000):
            if n <= 0 or time.time() > deadline:
                continue
            for beta in np.arange(beta_best - 0.01, beta_best + 0.011, 0.001):
                if time.time() > deadline:
                    break
                seq = power_law_sequence(n, beta)
                score = compute_score(seq)
                if score < best_score:
                    best_score = score
                    best_seq = seq.copy()
                    print(
                        f"[len‑adj] n={n:5d}, β={beta:0.4f} → new best = {score:.6f}"
                    )

    # ------------------------------------------------------------------
    # 3.  Light stochastic hill‑climb on the current best sequence.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        # allocate roughly the last 10 % of the budget to local optimisation
        hc_deadline = deadline
        print("[hill] Starting local optimisation")
        best_seq, best_score = hill_climb(best_seq, hc_deadline)

    # ------------------------------------------------------------------
    # 4.  Return the best candidate found.
    # ------------------------------------------------------------------
    if best_seq is None:
        # This should never happen, but guard against an empty result.
        raise RuntimeError("No valid candidate was generated.")

    print(f"[final] best score = {best_score:.6f}, length = {len(best_seq)}")
    # Convert to plain Python list of floats as required by the grader.
    return best_seq.tolist()
