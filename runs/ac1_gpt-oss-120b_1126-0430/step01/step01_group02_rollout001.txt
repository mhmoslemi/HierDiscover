"""
Propose a candidate step‑function (list of non‑negative heights) that tries to
minimise ``evaluate_sequence``.

The algorithm starts from the truncated Catalan construction (which already
gives ≈ π/2 ≈ 1.5708).  It then performs a *projected* optimisation on the
simplex using the Adam optimiser.  Instead of the non‑smooth max‑operator we
minimise a smooth surrogate – the log‑sum‑exp of the convolution values.
For a given temperature α the surrogate is

    J(a) = (1/α)·log Σ_k exp(α·(a*a)_k)   ,

its gradient w.r.t. the vector a can be written in closed form using a single
convolution:

    ∂J/∂a_i = 2·Σ_k p_k·a_{k‑i}    with   p_k = exp(α·c_k)/Σ_j exp(α·c_j).

We parametrize a with a soft‑max (a = softmax(x)) so that the non‑negativity
and unit‑sum constraints are satisfied automatically.  Adam updates are applied
to x, the temperature α is slowly increased (making the surrogate approach the
true max) and occasional random perturbations restart the search when it stalls.
Finally a small 1‑D line‑search mixing the best found vector with the original
Catalan seed is performed.

The whole loop stops after the allotted time budget (default 1000 s) and the
best sequence discovered is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------
def _catalan_sequence(n: int) -> np.ndarray:
    """
    Return the first *n* Catalan‑like coefficients a_i = C(2i,i)/4**i,
    normalised so that they sum to 1.
    """
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    # recurrence a_{i+1} = a_i * (2*i+1) / (2*(i+1))
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * (i - 1) + 1.0) / (2.0 * i)
    a /= a.sum()
    return a


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector that sums to 1)."""
    m = np.max(x)
    e = np.exp(x - m)
    s = e.sum()
    return e / s if s > 0 else np.full_like(x, 1.0 / x.size)


def _log_sum_exp_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Given the convolution vector `conv` and temperature `alpha`,
    return the probability vector p_k = exp(alpha*conv_k) / Σ exp(...).
    """
    scaled = alpha * conv
    # subtract the maximal value for numerical stability
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    total = exp_scaled.sum()
    return exp_scaled / total if total > 0 else np.full_like(conv, 1.0 / conv.size)


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a sequence of non‑negative heights that minimises
    ``evaluate_sequence``.  The routine runs for at most *budget_s* seconds
    and returns the best sequence found.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility (default: 42).
    budget_s : int, optional
        Time budget in seconds (default: 1000).

    Returns
    -------
    list[float]
        The best sequence discovered (as a plain Python list).
    """
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Problem size – a few thousand entries give a good trade‑off.
    # ------------------------------------------------------------------
    n = 2500                                   # length of the sequence
    a = _catalan_sequence(n)                  # start from the Catalan seed

    # Parameterise a with a soft‑max to enforce simplex constraints.
    eps = 1e-30
    x = np.log(a + eps)                        # initial logits

    # Adam optimiser state
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    adam_eps = 1e-8
    lr = 0.05                                   # base learning rate

    # Temperature for the log‑sum‑exp surrogate (starts smooth, ends sharp)
    alpha = 1.0
    alpha_max = 50.0

    # bookkeeping
    best_seq = a.copy()
    # because sum(a)=1, the evaluation value equals 2*n*max(conv)
    best_score = 2.0 * n * np.max(np.convolve(a, a))
    print(f"[init] n={n}, initial score={best_score:.6f}")

    deadline = time.time() + budget_s - 5      # safety margin
    iteration = 0
    no_improve = 0

    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1) current convolution
        # --------------------------------------------------------------
        conv = np.convolve(a, a)                # length 2n‑1

        # --------------------------------------------------------------
        # 2) smooth max weights (p_k)
        # --------------------------------------------------------------
        p = _log_sum_exp_weights(conv, alpha)   # size 2n‑1, sums to 1

        # --------------------------------------------------------------
        # 3) gradient of the surrogate w.r.t. a
        #    g_a_i = 2 * Σ_k p_k * a_{k-i}
        # --------------------------------------------------------------
        a_rev = a[::-1]                         # reversed a
        grad_temp = np.convolve(p, a_rev)       # length 3n‑2
        g_a = 2.0 * grad_temp[n - 1 : n - 1 + n]   # slice of length n

        # --------------------------------------------------------------
        # 4) chain rule to get gradient w.r.t. logits x (soft‑max)
        #    ∂J/∂x = a * (g_a - <g_a,a>)
        # --------------------------------------------------------------
        dot = np.dot(g_a, a)
        grad_x = a * (g_a - dot)

        # --------------------------------------------------------------
        # 5) Adam update on x
        # --------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        t = iteration + 1
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        x -= lr * m_hat / (np.sqrt(v_hat) + adam_eps)

        # --------------------------------------------------------------
        # 6) go back to the probability simplex
        # --------------------------------------------------------------
        a = _softmax(x)                         # automatically non‑negative & sum=1

        # --------------------------------------------------------------
        # 7) evaluate current candidate (fast – we already have conv)
        # --------------------------------------------------------------
        cur_score = 2.0 * n * np.max(conv)      # because Σ a = 1
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{iteration:06d}] new best = {best_score:.6f}")
            no_improve = 0
        else:
            no_improve += 1

        # --------------------------------------------------------------
        # 8) schedules & occasional perturbations
        # --------------------------------------------------------------
        if iteration % 3000 == 0 and iteration > 0:
            lr *= 0.9               # slowly decay the step size
            alpha = min(alpha_max, alpha * 1.15)   # sharpen the surrogate

        if no_improve > 12000:      # stagnation → random restart
            print(f"[{iteration}] restart (stagnation)")
            a = np.random.rand(n)
            a /= a.sum()
            x = np.log(a + eps)
            m.fill(0.0)
            v.fill(0.0)
            lr = 0.05
            alpha = 1.0
            no_improve = 0

        if iteration % 25000 == 0 and iteration > 0:
            # small random jitter to escape shallow local minima
            a += 0.005 * np.random.randn(n)
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            x = np.log(a + eps)
            m.fill(0.0)
            v.fill(0.0)
            print(f"[{iteration}] jitter applied")

        iteration += 1

    # ------------------------------------------------------------------
    # 9) Post‑processing: line‑search mixing with the original Catalan seed
    #    (convex combinations stay on the simplex).
    # ------------------------------------------------------------------
    catalan = _catalan_sequence(n)
    for lam in np.linspace(0.0, 1.0, 41):
        mixed = lam * best_seq + (1.0 - lam) * catalan
        mixed /= mixed.sum()
        cur = 2.0 * n * np.max(np.convolve(mixed, mixed))
        if cur < best_score:
            best_score = cur
            best_seq = mixed.copy()
            print(f"[mix] λ={lam:.3f} → new best {best_score:.6f}")

    print(f"[done] best score = {best_score:.6f}")
    return best_seq.tolist()
