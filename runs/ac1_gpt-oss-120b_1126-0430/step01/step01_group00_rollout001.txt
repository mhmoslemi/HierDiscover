def _lp_improve(a, M=None, eps=1e-9):
    n = a.size
    conv = np.convolve(a, a)
    if M is None:
        M = conv.max()
    ks = np.where(conv >= M - eps)[0]
    # Build sparse constraint matrix
    rows = []
    cols = []
    data = []
    for row_idx, k in enumerate(ks):
        # iterate over j where 0 <= j < n and 0 <= i = k-j < n
        j_start = max(0, k-(n-1))
        j_end = min(k, n-1)
        # For each j in this range, i = k-j
        # coefficient f_i = a[i]
        js = np.arange(j_start, j_end+1, dtype=int)
        is_ = k - js
        rows.extend([row_idx] * js.size)
        cols.extend(js.tolist())
        data.extend(a[is_].tolist())
    A_ub = csr_matrix((data, (rows, cols)), shape=(len(ks), n))
    b_ub = np.full(len(ks), M)
    c = -np.ones(n)  # maximize sum(b)
    # bounds: b_i >= 0
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return a
    b_opt = res.x
    if b_opt.sum() == 0:
        return a
    b_opt = b_opt / b_opt.sum()  # normalize to sum 1
    # Combine
    t = 0.3 # mixing weight
    new_a = (1 - t) * a + t * b_opt
    # Project onto simplex (should be already sum=1)
    new_a = _project_to_simplex(new_a)
    return new_a
```

"""
propose_candidate
=================

Search for a non‑negative step‑function (a probability vector) that minimizes

    O(a) = 2·n·max(conv(a,a)) / (sum(a)^2)   (scale‑invariant)

The algorithm mixes several ideas:

* **Simplex projection** – all candidates are kept on the probability simplex
  (non‑negative, sum = 1) so the scale factor of the original evaluator does
  not matter.
* **Adam‑style sub‑gradient descent** – the sub‑gradient of the maximal
  convolution entry is computed analytically and an Adam optimizer is applied.
* **Periodic random mass‑transfer hill climbing** – occasional stochastic moves
  help to escape flat regions.
* **Explicit sub‑gradient steps** – a small deterministic step every few
  iterations further pushes down the peak.
* **Linear‑programming refinement** – after a few hundred iterations we solve
  the LP described in the statement (maximise ∑b subject to f∗b ≤ M) and mix
  the result with the current point.  Only a handful of LP calls (≤5) are
  performed so the overall run‑time stays well within the 1000 s budget.

The search stops early if the target bound `O < 1.503` is reached.
The best sequence found is returned as a list of floats.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# -------------------------------------------------------------
# Projection onto the probability simplex (Duchi et al. 2008)
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x >= 0, sum(x) = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        # all non‑positive → uniform point
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# -------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Fast objective on the simplex:
        O = 2·n·max(conv(a,a))   (since sum(a)=1)
    """
    conv = np.convolve(a, a)
    return 2.0 * a.size * conv.max()


# -------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) w.r.t. a (sum(a)=1 fixed).
    Returns a vector of the same shape as a.
    """
    n = a.size
    conv = np.convolve(a, a)
    max_val = conv.max()
    # indices where the maximum is attained
    ks = np.flatnonzero(conv == max_val)
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    g /= len(ks)  # average if multiple maxima
    return g


# -------------------------------------------------------------
def _lp_improve(a: np.ndarray, eps: float = 1e-9) -> np.ndarray:
    """
    One LP refinement step as described in the statement.

    Given current vector a (sum = 1), let M = max(conv(a,a)).
    Solve:
        maximise   sum(b)
        subject to conv(a,b) <= M   (only the active constraints)
                    b >= 0
    The solution b is normalised to sum 1 and mixed with a.
    """
    n = a.size
    conv = np.convolve(a, a)
    M = conv.max()

    # active constraints (indices where conv is practically equal to M)
    ks = np.where(conv >= M - eps)[0]
    if ks.size == 0:
        return a

    # Build sparse matrix for constraints conv(a,b) <= M
    rows = []
    cols = []
    data = []
    for row_idx, k in enumerate(ks):
        # For each column j (index in b) the coefficient is a_{k-j}
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        js = np.arange(j_start, j_end + 1, dtype=int)
        is_ = k - js
        rows.extend([row_idx] * js.size)
        cols.extend(js.tolist())
        data.extend(a[is_].tolist())

    A_ub = csr_matrix((data, (rows, cols)), shape=(ks.size, n))
    b_ub = np.full(ks.size, M)

    # maximise sum(b)  →  minimise -sum(b)
    c = -np.ones(n)
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success or res.x.sum() == 0:
        # fallback – return the original point
        return a

    b_opt = res.x
    b_opt = b_opt / b_opt.sum()          # renormalise to sum = 1

    # convex combination with the current point (mixing weight t)
    t = 0.3
    new_a = (1.0 - t) * a + t * b_opt
    new_a = _project_to_simplex(new_a)
    return new_a
