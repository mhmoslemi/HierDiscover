def _project_to_simplex(v):
    """Project a non‑negative vector onto the simplex {x >= 0, sum(x)=1}."""
    # Implementation of the algorithm from
    # Wang & Carreira‑Perpiñán (2013)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero – return uniform distribution
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


def _normalise(v):
    """Return v scaled to have unit sum (if possible)."""
    s = np.sum(v)
    if s <= 0:
        return None
    return v / s


def _evaluate(v):
    """Evaluate a candidate (expects a 1‑D NumPy array)."""
    # ``evaluate_sequence`` works on Python lists and already performs
    # clipping to [0,1000] and protects against tiny sums.
    norm = _normalise(v)

    return evaluate_sequence(norm.tolist())

def _make_power_law(n, alpha, shift):
    """a_i ∝ (i + shift)^{-alpha}."""
    idx = np.arange(n, dtype=float)
    raw = np.power(idx + shift, -alpha)
    return raw


def _make_exponential(n, beta, shift):
    """a_i ∝ exp(-beta * (i + shift))."""
    idx = np.arange(n, dtype=float)
    raw = np.exp(-beta * (idx + shift))
    return raw


def _make_arcsine(n, eps):
    """
    Discrete arcsine‑type shape:
        a_i ∝ 1 / sqrt((i + eps) * (n‑1‑i + eps))
    The continuous arcsine distribution convolved with itself is uniform;
    this discrete analogue yields a very flat self‑convolution.
    """
    idx = np.arange(n, dtype=float)
    raw = 1.0 / np.sqrt((idx + eps) * (n - 1 - idx + eps))
    return raw

    """Mixture of power‑law and exponential: (i+shift)^(-alpha) * exp(-beta*i)."""
    idx = np.arange(n, dtype=float)
    raw = np.power(idx + shift, -alpha) * np.exp(-beta * idx)

    # Evaluate each candidate while we have time.
    for spec in families:
        if time_left() < 2.0:
            break
        kind = spec[0]

        if kind == 'power':
            _, alpha, shift = spec
            raw = _make_power_law(n, alpha, shift)
        elif kind == 'exp':
            _, beta, shift = spec
            raw = _make_exponential(n, beta, shift)
        elif kind == 'arcsine':
            _, eps = spec
            raw = _make_arcsine(n, eps)
        elif kind == 'mixed':
            _, alpha, beta, shift = spec
            raw = _make_mixed(n, alpha, beta, shift)
        else:
            continue

        # Normalise and evaluate
        vec = _normalise(raw)
        if vec is None:
            continue
        score = _evaluate(vec)
        if score < best_score:
            best_score = score
            best_seq = vec.tolist()
            print(f"[grid] {kind} params={spec[1:]} -> score={score:.6f}")

    print(f"[grid] finished, best score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2. Simulated‑annealing hill climbing
    # --------------------------------------------------------------
    cur = np.array(best_seq, dtype=float)   # already sums to 1 (within rounding)
    cur_score = best_score

    # Annealing schedule parameters
    T_initial = 0.02 * cur_score           # start with ~2 % of the current score
    T = T_initial
    decay = 0.99999                         # exponential cooling per iteration
    max_iter = 500000                       # hard cap; real limit is time budget
    iter_no_improve = 0
    NO_IMPROVE_LIMIT = 20000                # after this we inject random noise

    iter_cnt = 0
    while time_left() > 1.5 and iter_cnt < max_iter:
        iter_cnt += 1

        # Randomly pick source and destination indices
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        # Propose moving a modest fraction of mass from i to j
        frac = np.random.rand() * 0.05       # up to 5 %
        step = cur[i] * frac
        if step <= 0:
            continue

        new = cur.copy()
        new[i] -= step
        new[j] += step

        # Numerical drift could break the simplex constraint – re‑project.
        new = _project_to_simplex(new)

        # Evaluate the new candidate
        new_score = _evaluate(new)

        # Acceptance rule
        if new_score < cur_score:
            cur = new
            cur_score = new_score
            iter_no_improve = 0
            if new_score < best_score:
                best_score = new_score
                best_seq = new.tolist()
                print(f"[anneal] iter={iter_cnt} improved score={new_score:.6f}")
        else:
            # Probabilistic acceptance
            if np.random.rand() < np.exp(-(new_score - cur_score) / max(T, 1e-12)):
                cur = new
                cur_score = new_score
            iter_no_improve += 1

        # Temperature cooling
        T *= decay
        if T < 1e-7:
            T = 1e-7

        # If we stagnate for a long period, inject a small random perturbation
        if iter_no_improve >= NO_IMPROVE_LIMIT:
            # Draw a Dirichlet noise vector and blend it gently
            noise = np.random.dirichlet(np.ones(n) * 0.5)
            cur = 0.98 * cur + 0.02 * noise
            cur = _project_to_simplex(cur)
            cur_score = _evaluate(cur)
            iter_no_improve = 0
            if cur_score < best_score:
                best_score = cur_score
                best_seq = cur.tolist()
                print(f"[anneal] iter={iter_cnt} after restart score={cur_score:.6f}")

    print(f"[final] best score after annealing: {best_score:.6f}")
    return best_seq
