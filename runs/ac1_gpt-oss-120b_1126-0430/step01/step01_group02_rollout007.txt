def simplex_projection(v):
    # ensures v>=0 and sum v = 1
    # Implementation from paper.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w

def normalize_to_simplex(v):
    # uses Duchi algorithm
    v = np.asarray(v, dtype=float)
    if v.sum() == 0:
        # avoid division by zero
        v = np.ones_like(v) / v.shape[0]
        return v
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    # find rho
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def compute_gradient(a, beta):
    conv = np.convolve(a, a)
    max_c = conv.max()
    # compute softmax weights around max
    exps = np.exp(beta * (conv - max_c))
    w = exps / exps.sum()
    # correlation
    corr_full = np.correlate(a, w, mode='full')
    # slice length n indices
    start = len(w) - 1
    grad = 2.0 * corr_full[start:start + len(a)]
    return grad, conv

"""
Optimise a step‑function (non‑negative vector that sums to 1) for the
``evaluate_sequence`` objective.

The objective for a normalised vector a (∑a_i = 1) reduces to

    score = 2 * n * max_k (a * a)[k]                (1)

where ``*`` is the linear convolution.
The infinite Catalan‐like sequence a_i = C(2i,i)/4^i yields a constant
convolution and attains the bound π/2 ≈ 1.5708.  Small perturbations of
this shape can lower the maximal convolution entry and therefore the
score.  The optimiser works directly on the smooth approximation of the
max‑operator (soft‑max or top‑K averaging) and uses Adam with a
projection onto the probability simplex after every update.

Key ideas
---------
* Initialise from the (truncated) Catalan sequence – a known near‑optimal
  starting point.
* At each iteration compute the convolution, pick the K largest
  entries (K ≈ 5) and form a probability vector w that puts equal mass on
  these indices.  This gives a smooth sub‑gradient of the max‑function.
* The gradient of the soft‑max objective is
      g = 2 * corr(a, w)           (corr = cross‑correlation)
  which is obtained efficiently via ``np.correlate``.
* Use the Adam optimiser (adaptive moments) and after the step project
  the vector onto the simplex (non‑negative, sum = 1) with the exact
  Duchi‑et‑al. algorithm.
* Occasionally inject random noise around the current best solution to
  escape possible plateaus and decay the learning‑rate over time.
* Keep the best vector seen so far and return it when the time budget
  (≈ 1000 s) expires.

The routine respects the required API – it defines a top‑level
``propose_candidate`` function and may be run directly in the evaluation
environment.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures or lambdas)
# ----------------------------------------------------------------------


def _catalan_sequence(n: int) -> np.ndarray:
    """
    First ``n`` Catalan‑like coefficients a_i = C(2i,i)/4**i, normalised
    to sum to 1.
    """
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        # recurrence a_{i} = a_{i-1} * (2*i-1) / (2*i)
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    a /= a.sum()
    return a


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i >= 0, sum x_i = 1}.  Implements the algorithm of
    Duchi et al. (2008).
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays.")
    n = v.shape[0]
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0    # cumulative sum minus 1
    ind = np.arange(1, n + 1)
    # Find the largest rho with u[rho] > cssv[rho] / rho
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # Numerical safety: renormalise to exactly 1
    w_sum = w.sum()
    if w_sum == 0:
        # fallback to uniform distribution
        w[:] = 1.0 / n
    else:
        w /= w_sum
    return w


def _topk_weights(conv: np.ndarray, k: int) -> np.ndarray:
    """
    Build a probability vector that places equal mass on the ``k``
    largest entries of ``conv``.
    """
    n = conv.size
    w = np.zeros_like(conv)
    if k >= n:
        w[:] = 1.0 / n
        return w
    # ``np.argpartition`` is O(n) and returns unsorted top‑k indices
    idx = np.argpartition(conv, -k)[-k:]
    w[idx] = 1.0 / k
    return w


def _gradient_topk(a: np.ndarray, k: int) -> (np.ndarray, np.ndarray):
    """
    Compute the gradient of the ``max`` (approximated by top‑k averaging)
    together with the current convolution.

    Returns
    -------
    grad : np.ndarray
        Gradient vector of shape (n,).
    conv : np.ndarray
        Full linear convolution (length 2*n-1).
    """
    n = a.size
    conv = np.convolve(a, a)                # length 2*n-1
    w = _topk_weights(conv, k)              # same length as conv
    # cross‑correlation a ⋆ w ; mode='full' gives length 3*n-2
    corr_full = np.correlate(a, w, mode='full')
    start = len(w) - 1                      # corresponds to offset = 0
    grad = 2.0 * corr_full[start:start + n]  # slice to get n entries
    return grad, conv

# Main optimisation routine
# ----------------------------------------------------------------------
