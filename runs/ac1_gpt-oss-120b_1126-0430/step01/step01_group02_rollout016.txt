def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    n = 3000  # maybe 4000
    # initial a via Catalan
    a0 = _catalan_sequence(n)
    # log-parameters b = log a0
    b = np.log(a0 + 1e-32)   # small epsilon
    # Adam hyperparameters
    lr = 0.1
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(b)
    v = np.zeros_like(b)
    best_score = np.inf
    best_a = None
    start = time.time()
    deadline = start + budget_s - 5

    iteration = 0
    beta_soft = 10.0   # starting softness
    while time.time() < deadline:
        # compute probability vector a from b
        # to prevent overflow: subtract max.
        b_shift = b - np.max(b)
        expb = np.exp(b_shift)
        a = expb / np.sum(expb)

        # compute convolution via FFT maybe for speed:
        conv = np.convolve(a, a)  # shape 2n-1
        max_conv = np.max(conv)

        # evaluate score
        score = evaluate_sequence(a.tolist())
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[{iteration}] new best {best_score:.6f}")

        # compute softmax weights
        # use exponent with parameter beta_soft
        # shift conv for numerical stability
        c_shift = conv - np.max(conv)
        w = np.exp(beta_soft * c_shift)
        w_sum = w.sum()
        w /= w_sum   # softmax

        # compute gradient wrt a
        # w_rev = w[::-1]
        # full = np.convolve(a, w_rev)
        # grad = 2 * full[(2*n - 2) - np.arange(n)]

        w_rev = w[::-1]
        full = np.convolve(a, w_rev)
        # Index offset: we need to pick indices from L-1 - i to get length n
        # Actually L-1 = 2*n-2
        offset = (L - 1)
        grad = 2 * full[offset - np.arange(n)]

        # compute gradient wrt b
        # grad_b = a * (grad - np.dot(a, grad))
        dot = np.dot(a, grad)
        grad_b = a * (grad - dot)

        # Adam update
        iteration += 1
        m = beta1 * m + (1 - beta1) * grad_b
        v = beta2 * v + (1 - beta2) * (grad_b ** 2)
        m_hat = m / (1 - beta1 ** iteration)
        v_hat = v / (1 - beta2 ** iteration)
        b -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasionally increase beta_soft (make approximation tighter) as iteration increases
        if iteration % 200 == 0:
            beta_soft = min(beta_soft * 1.2, 200.0)

        # occasional random perturbation if no improvement
        # maybe track no_improve count
    # after loop
    return best_a.tolist()

Define:

def full_convolve(x, y):
    size = len(x) + len(y) - 1
    nfft = 1 << (size-1).bit_length()
    fx = np.fft.rfft(x, nfft)
    fy = np.fft.rfft(y, nfft)
    f = fx * fy
    conv = np.fft.irfft(f, nfft)
    return conv[:size]
```

import time, numpy as np

def full_convolve(x, y):
    size = len(x) + len(y) - 1
    nfft = 1 << (size - 1).bit_length()
    fx = np.fft.rfft(x, nfft)
    fy = np.fft.rfft(y, nfft)
    conv = np.fft.irfft(fx * fy, nfft)
    return conv[:size]

def _catalan_sequence(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i-1] * (2*i - 1) / (2.0 * i)  # recurrence
    a /= a.sum()
    return a

def propose_candidate(seed=123, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5

    n = 3000  # can increase if time permits
    # starting point: Catalan
    a0 = _catalan_sequence(n)
    # store best
    best_a = a0.copy()
    best_score = evaluate_sequence(best_a.tolist())
    print(f"[init] n={n}, score={best_score:.6f}")

    # parameters for optimization
    lr = 0.05
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    # log-parameters b such that a = softmax(b)
    b = np.log(a0 + 1e-32)

    m = np.zeros_like(b)
    v = np.zeros_like(b)

    # hyper-parameters for the loss
    beta_conv = 5.0   # start with low temperature
    var_coeff = 0.0   # start without variance term

    it = 0
    no_improve = 0

    while time.time() < deadline:
        # compute a from b
        b_shift = b - np.max(b)  # for numerical stability
        expb = np.exp(b_shift)
        a = expb / expb.sum()

        # compute convolution
        conv = full_convolve(a, a)   # length 2n-1
        # compute mean (useful for possible variance term)
        conv_mean = conv.mean()

        # weight vector approximating gradient of max (smooth)
        # softmax over conv
        c_shift = conv - np.max(conv)          # stability
        # increase temperature gradually
        w_soft = np.exp(beta_conv * c_shift)
        w_soft_sum = w_soft.sum()
        w_soft = w_soft / w_soft_sum

        # optional variance term to flatten profile
        w = w_soft + var_coeff * (conv - conv_mean)

        # compute gradient w.r.t. a
        w_rev = w[::-1]
        grad_full = full_convolve(a, w_rev)    # length 3n-2
        offset = 2 * n - 2
        grad_a = 2.0 * grad_full[offset - np.arange(n)]   # shape (n,)

        # gradient w.r.t. b (mirror descent on simplex)
        dot = np.dot(a, grad_a)
        grad_b = a * (grad_a - dot)

        # Adam update
        it += 1
        m = beta1 * m + (1 - beta1) * grad_b
        v = beta2 * v + (1 - beta2) * (grad_b ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        b -= step

        # evaluate current a
        cur_score = evaluate_sequence(a.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{it}] new best {best_score:.6f}")

"""propose_candidate
A fast, gradient‑based optimizer for the “minimise max self‑convolution”
problem.

The objective to minimise is  

    J(a) = 2·n·max_k (a*a)[k] / (∑ a_i)² .

Because the ratio is scale‑invariant we can fix  ∑ a_i = 1  and
simply minimise  max_k (a*a)[k] .  The function is non‑convex, but the
gradient of a smooth approximation of the max (log‑sum‑exp) is easy to
compute – it reduces to convolutions of the current sequence with a
weight vector.  We work on the simplex by exponentiated (mirror) updates:
the parameters b are unconstrained and a = softmax(b).  An Adam optimiser
provides adaptive step sizes and momentum.  The temperature of the
log‑sum‑exp (β) is increased gradually so that the optimisation first
moves in a smooth landscape and finally focuses on the true maximum.
A small variance term is added to the weight vector to encourage a flat
convolution profile.

The whole routine runs for at most *budget_s* seconds, keeps the best
sequence ever seen (according to the supplied ``evaluate_sequence``) and
returns it as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: fast full linear convolution using FFT (real -> real)
def _full_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of two real 1‑D arrays."""
    size = x.size + y.size - 1
    nfft = 1 << (size - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, n=nfft)
    fy = np.fft.rfft(y, n=nfft)
    conv = np.fft.irfft(fx * fy, n=nfft)
    return conv[:size]


# ----------------------------------------------------------------------
# Catalan‑like seed (the known analytic construction)
def _catalan_sequence(n: int) -> np.ndarray:
    """First *n* Catalan‑like numbers a_i = C(2i,i)/4**i, normalised to sum = 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    # recurrence a_{i+1} = a_i * (2*i+1) / (2*(i+1))
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a list of non‑negative heights that minimises
    ``evaluate_sequence``.  The routine respects the 1000 s time budget,
    prints occasional progress information and returns the best sequence
    found (as a plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5          # safety margin

    # problem size – a few thousand entries give a good trade‑off
    n = 3000

    # ------------------------------------------------------------------
    # initialise with the classic Catalan construction
    a0 = _catalan_sequence(n)

    # we keep the best sequence observed so far
    best_a = a0.copy()
    best_score = evaluate_sequence(best_a.tolist())
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # optimisation lives on the probability simplex via a softmax
    b = np.log(a0 + 1e-32)          # log‑probabilities, unconstrained
    m = np.zeros_like(b)           # first moment (Adam)
    v = np.zeros_like(b)           # second moment (Adam)

    # Adam parameters
    lr = 0.05
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    # log‑sum‑exp temperature (starts low, grows)
    beta_conv = 5.0
    # tiny variance‑penalty coefficient – helps to flatten the convolution
    var_coeff = 0.0

    iteration = 0
    no_improve = 0

    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1) current probability vector on the simplex
        b_shift = b - np.max(b)            # for numerical stability
        expb = np.exp(b_shift)
        a = expb / expb.sum()              # a >= 0, sum(a)=1

        # 2) convolution a*a (full length = 2n‑1)
        conv = _full_convolve(a, a)

        # 3) smooth approximation of the max (log‑sum‑exp)
        #    w = softmax(beta * conv)  +  var_coeff * (conv - mean)
        conv_shift = conv - np.max(conv)   # stabilize the exponentials
        w_soft = np.exp(beta_conv * conv_shift)
        w_soft /= w_soft.sum()             # softmax weights, sum = 1

        # optional variance term (encourages flatness)
        if var_coeff > 0.0:
            conv_mean = conv.mean()
            w = w_soft + var_coeff * (conv - conv_mean)
        else:
            w = w_soft

        # 4) gradient of the loss w.r.t. a
        #    ∂L/∂a = 2 * ( w_rev  * a )   (where * is linear convolution)
        w_rev = w[::-1]
        grad_full = _full_convolve(a, w_rev)      # length = 3n‑2
        offset = 2 * n - 2                         # = len(conv)‑1
        grad_a = 2.0 * grad_full[offset - np.arange(n)]

        # 5) transform gradient to the log‑space (mirror descent on simplex)
        dot = np.dot(a, grad_a)
        grad_b = a * (grad_a - dot)                # ∂L/∂b

        # 6) Adam update on the unconstrained parameters b
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad_b
        v = beta2 * v + (1.0 - beta2) * (grad_b ** 2)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        b -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # --------------------------------------------------------------
        # evaluate the current candidate (expensive but cheap enough)
        cur_score = evaluate_sequence(a.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{iteration:06d}] new best = {best_score:.6f}")
            no_improve = 0
        else:
            no_improve += 1

        # schedule: make the approximation tighter, and slowly turn on the
        # variance penalty to push the convolution towards a flat profile
        if iteration % 200 == 0:
            beta_conv = min(beta_conv * 1.2, 200.0)   # temperature ↑
            var_coeff = min(var_coeff + 5e-4, 0.01)   # variance weight ↑

        # occasional restart if we appear stuck for a long time
        if no_improve > 4000:
            # re‑initialise around the current best (adds stochasticity)
            b = np.log(best_a + 1e-32) + 0.2 * np.random.randn(n)
            m.fill(0.0)
            v.fill(0.0)
            lr = 0.05
            no_improve = 0
            print(f"[{iteration:06d}] restart (noise around best)")

    # ------------------------------------------------------------------
    print(f"[done] best score found = {best_score:.6f}")
    return best_a.tolist()
