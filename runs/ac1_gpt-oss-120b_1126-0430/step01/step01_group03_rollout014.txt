def propose_candidate(seed=0, budget_s=1000, **kwargs):
    import time, numpy as np, random, math
    try:
        import cvxpy as cp, scipy.sparse as sp
        have_cvx = True
    except Exception:
        have_cvx = False

    start = time.time()
    deadline = start + budget_s - 0.1

    best_seq = None
    best_score = float('inf')
    # if height_sequence_1 exists, evaluate
    if 'height_sequence_1' in globals():
        try:
            score = evaluate_sequence(list(height_sequence_1))
            best_seq = list(height_sequence_1)
            best_score = score
            print("[init] existing height_sequence_1 score:", score)
        except Exception:
            pass

    # scanning parameters
    n_values = [1500, 2000, 2500, 3000, 4000, 6000, 8000, 10000]
    # ensure n not exceed some max for runtime; maybe restrict last two if time.
    beta_range = np.linspace(0.45, 0.75, 31)  # step 0.01
    offset_range = [0, 1, 2, 5]   # integer offsets
    # If n>5000, reduce beta steps and offset set to keep time low.

    for n in n_values:
        if time.time() > deadline:
            break
        betas = beta_range
        # if n larger than 5000, reduce steps
        if n > 5000:
            betas = np.linspace(0.5, 0.7, 21)   # fewer betas
            cur_offset_range = [0, 2]
        else:
            cur_offset_range = offset_range
        for offset in cur_offset_range:
            for beta in betas:
                if time.time() > deadline:
                    break
                idx = np.arange(1, n+1, dtype=np.float64) + offset
                seq = idx ** (-beta)
                # optionally renormalize to avoid large numbers? Not needed.
                score = evaluate_sequence(seq.tolist())
                if score < best_score:
                    best_score = score
                    best_seq = seq.tolist()
                    print(f"[scan] n={n} offset={offset} beta={beta:.4f} => new best {score:.6f}")
    # local hill climbing
    if best_seq is not None and time.time() < deadline:
        seq = np.array(best_seq, dtype=np.float64)
        n = len(seq)
        current_score = best_score
        # Use decreasing epsilon schedule
        while time.time() < deadline:
            # choose random i,j
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            # pick fraction up to 10% of source mass
            eps = seq[i] * np.random.rand() * 0.10
            if eps <= 0:
                continue
            new_seq = seq.copy()
            new_seq[i] -= eps
            new_seq[j] += eps
            # ensure non-negativity
            if new_seq[i] < 0:
                continue
            new_score = evaluate_sequence(new_seq.tolist())
            if new_score < current_score:
                seq = new_seq
                current_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_seq = new_seq.tolist()
                    print(f"[hill] improved to {new_score:.6f}")
    # optional LP refinement if available and time left
    if have_cvx and best_seq is not None and time.time() < deadline:
        seq = np.array(best_seq, dtype=np.float64)
        n = len(seq)
        target_sum = np.sum(seq)
        # Perform a few iterations of LP refinement
        t = 0.2
        # number of iterations limited by time
        iter_cnt = 0
        while time.time() < deadline and iter_cnt < 12:
            # compute current max conv bound
            conv = np.convolve(seq, seq)
            M = conv.max()
            # build convolution matrix A
            rows = np.add.outer(np.arange(n), np.arange(n)).ravel()
            cols = np.tile(np.arange(n), n)
            data = np.repeat(seq, n)
            A = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
            # define variable
            g = cp.Variable(n, nonneg=True)

            g_scaled = g0 * (target_sum / sum_g0)
            # convex combination update
            seq = (1 - t) * seq + t * g_scaled
            # recompute score
            new_score = evaluate_sequence(seq.tolist())

            iter_cnt += 1

"""
propose_candidate

Searches for a non‑negative step‑function (list of heights) that minimises
the evaluation function

    evaluate_sequence(a) = 2·n·max_k (a*a)_k / (sum a)^2 .

The objective is scale‑invariant, so only the *shape* of the sequence matters.
The algorithm proceeds in three stages:

1. **Coarse global search** –  explores families
   a_i = (i+1+offset)^{‑β} for several lengths (n), offsets and β‑values.
   Each candidate is evaluated directly with ``evaluate_sequence`` and the
   best one is retained.

2. **Local stochastic hill‑climbing** – starting from the best shape,
   repeatedly moves a small random amount of mass from one index to another.
   Only moves that improve the score are kept.

3. **Optional LP refinement** – if `cvxpy` and `scipy.sparse` are available,
   performs a few iterations of the classic “multiply by a feasible direction”
   step: for the current shape f, solves the linear program
        maximise   Σ g_j   subject to   f*g ≤ M   and   g ≥ 0
   where M = max_k (f*f)_k.  The solution g is rescaled to the same total mass
   as f and mixed with f (f←(1‑t)f+t·g).  Each LP iteration strictly reduces
   the maximal self‑convolution and therefore the score.

The whole routine respects the 1000 s time budget; it prints progress
messages and finally returns the best sequence found (as a plain Python list
of floats).  All heights are non‑negative and safely below the internal
clipping limits of ``evaluate_sequence``.
"""

import time
import random
import numpy as np

# cvxpy / scipy are optional – they are only used for the LP refinement step.
try:
    import cvxpy as cp  # type: ignore
    import scipy.sparse as sp  # type: ignore
    _HAVE_CVX = True
except Exception:  # pragma: no cover
    _HAVE_CVX = False


def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """Return a step‑function sequence with a low evaluation score.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : float, optional
        Time budget in seconds (default 1000).  The function stops a few
        hundredths of a second before the deadline to avoid a forced kill.
    **kwargs
        Ignored – part of the required signature.

    Returns
    -------
    list[float]
        The best sequence discovered within the time limit.
    """
    random.seed(seed)
    np.random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 0.1   # safety margin

    best_seq = None          # type: list[float] | None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0)  If a previous construction exists, use it as a starting point.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = list(height_sequence_1)
            sc = evaluate_sequence(cand)
            best_seq = cand
            best_score = sc
            print(f"[init] Using height_sequence_1 → score {sc:.6f}")
        except Exception:  # pragma: no cover
            pass

    # ------------------------------------------------------------------
    # 1)  Global grid search over a family of power‑law shapes.
    # ------------------------------------------------------------------
    # Lengths we are willing to try (larger lengths give a better
    # asymptotic constant but cost a bit more time per evaluation).
    n_candidates = [1500, 2000, 2500, 3000, 4000, 6000, 8000, 10000]

    # Offsets added to the index before raising to -β – they smooth the
    # singularity at the first element.
    offset_list_small = [0, 1, 2, 5]

    # Base β‑grid – we concentrate around the region known to be good.
    beta_grid_fine = np.linspace(0.45, 0.75, 31)   # step ≈0.01
    beta_grid_coarse = np.linspace(0.50, 0.70, 21)  # used for larger n

    for n in n_candidates:
        if time.time() > deadline:
            break

        # reduce the search density for the very large n values
        if n > 5000:
            betas = beta_grid_coarse
            offsets = [0, 2]               # fewer offsets for speed
        else:
            betas = beta_grid_fine
            offsets = offset_list_small

        for offset in offsets:
            for beta in betas:
                if time.time() > deadline:
                    break
                # power‑law shape a_i = (i+1+offset)^{-β}
                idx = np.arange(1, n + 1, dtype=np.float64) + offset
                seq = idx ** (-beta)                     # raw (un‑normalised) heights
                # evaluate – the function internally normalises the vector
                try:
                    score = evaluate_sequence(seq.tolist())
                except Exception:                       # pragma: no cover
                    continue
                if score < best_score:
                    best_score = score
                    best_seq = seq.tolist()
                    print(f"[grid] n={n:5d} offset={offset:2d} β={beta:.4f} → "
                          f"new best {score:.6f}")

    # ------------------------------------------------------------------
    # 2)  Light stochastic hill‑climbing on the current best.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        seq = np.array(best_seq, dtype=np.float64)
        n = len(seq)
        cur_score = best_score

        # hill‑climbing parameters
        max_frac = 0.10          # up to 10 % of the source entry may be moved
        while time.time() < deadline:
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            eps = seq[i] * np.random.rand() * max_frac
            if eps <= 0.0:
                continue
            new_seq = seq.copy()
            new_seq[i] -= eps
            new_seq[j] += eps
            # sanity: never let an entry become negative because of rounding
            if new_seq[i] < 0.0:
                continue
            try:
                new_score = evaluate_sequence(new_seq.tolist())
            except Exception:   # pragma: no cover
                continue
            if new_score < cur_score:
                seq = new_seq
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_seq = new_seq.tolist()
                    print(f"[hill] improved → {new_score:.6f}")

    # ------------------------------------------------------------------
    # 3)  Optional LP‑based refinement (if cvxpy is available).
    # ------------------------------------------------------------------
    if _HAVE_CVX and best_seq is not None and time.time() < deadline:
        seq = np.array(best_seq, dtype=np.float64)
        n = len(seq)
        total_mass = seq.sum()          # always = 1 after normalisation, but keep generic
        t_mix = 0.20                    # initial mixing weight
        lp_iter = 0

        while time.time() < deadline and lp_iter < 12:
            # current bound M = max_k (f * f)_k
            conv_ff = np.convolve(seq, seq)
            M = conv_ff.max()

            # Build the convolution matrix A (size (2n‑1) × n) such that
            #   (A @ g)[k] = Σ_i seq[i] * g[k‑i]   (zero outside the range)
            rows = np.add.outer(np.arange(n), np.arange(n)).ravel()
            cols = np.tile(np.arange(n), n)
            data = np.repeat(seq, n)
            A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

            # Linear program: maximise Σ g_j   subject to   A @ g ≤ M , g ≥ 0
            g = cp.Variable(n, nonneg=True)
            constraints = [A @ g <= M]
            prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

            try:
                prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000, verbose=False)
            except Exception as e:      # pragma: no cover
                print("[lp] solver exception:", e)
                break

            if g.value is None:
                # solver failed to produce a solution
                break

            g0 = np.array(g.value).ravel()
            sum_g0 = g0.sum()
            if sum_g0 <= 0.0:
                break

            # Scale g0 back to the original total mass so that the mixture stays normalised.
            g_scaled = g0 * (total_mass / sum_g0)

            # Convex combination step
            seq = (1.0 - t_mix) * seq + t_mix * g_scaled

            # Re‑evaluate
            try:
                new_score = evaluate_sequence(seq.tolist())
            except Exception:   # pragma: no cover
                break

            if new_score < best_score:
                best_score = new_score
                best_seq = seq.tolist()
                print(f"[lp] iter {lp_iter:02d} → improved score {new_score:.6f}")

            # gradually reduce the mixing weight to stabilise
            t_mix = max(0.01, t_mix * 0.8)
            lp_iter += 1

    print(f"[final] best score = {best_score:.6f}, length = {len(best_seq) if best_seq else 0}")
    return best_seq
