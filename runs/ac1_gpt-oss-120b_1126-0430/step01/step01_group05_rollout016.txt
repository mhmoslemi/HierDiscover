def propose_candidate(seed=42, budget_s=1000, **kwargs):

    start = time.time()
    deadline = start + budget_s - 5

    # Choose length n
    try:
        n = len(height_sequence_1)
    except NameError:
        n = 3000   # moderate

    # Initialize candidates
    best_seq = None
    best_score = np.inf

    # Helper to normalize sum to 1
    def normalize(seq):
        total = np.sum(seq)
        if total == 0:
            return seq
        return seq / total

    # 1. Try simple families
    # power law
    for p in np.linspace(0.45, 0.75, 31):
        seq = np.array([(i+1) ** (-p) for i in range(n)], dtype=float)
        seq = normalize(seq)
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
    # exponential (c parameter)
    for c in np.linspace(0.5, 3.0, 26):
        q = np.exp(-c / n)
        seq = q ** np.arange(n, dtype=float)
        seq = normalize(seq)
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
    # linear descending
    seq = np.linspace(1, 0, n, dtype=float)
    seq = normalize(seq)
    score = evaluate_sequence(seq.tolist())
    if score < best_score:
        best_score = score
        best_seq = seq.copy()
    # start from previous if exists
    if 'height_sequence_1' in globals():
        seq = np.array(height_sequence_1, dtype=float)
        seq = normalize(seq[:n])
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.copy()

    print(f"Initial best score after simple families: {best_score:.6f}")

    seq = best_seq.copy()

#!/usr/bin/env python
"""
Propose a step‑function (list of non‑negative heights) with a low
evaluation score.

Algorithm
---------
1. Choose a working length `n`.  If a previous construction
   (`height_sequence_1`) exists we keep its length, otherwise we fall
   back to 3000 – a size that gives a good trade‑off between quality
   and speed.

2. **Exploratory grid search** over several simple parametric families:
   * Power‑law   a_i = (i+Δ)^(‑p)       (grid on p and Δ)
   * Exponential a_i = exp(‑c·i / n)   (grid on c)
   * Linear      a_i decreasing linearly from 1 to 0.

   All candidates are normalised to have L1‑norm 1 (scale invariant for
   the evaluation function) and the best score seen so far is kept.

3. **Local refinement** – a cheap stochastic hill‑climb that repeatedly
   perturbs a random contiguous block of the current best sequence.
   The perturbation is multiplicative with a factor drawn from a
   moderate range; moves that improve the score are kept.  The
   neighbourhood size and perturbation magnitude are gradually
   reduced, allowing both coarse and fine adjustments.

4. The search stops a few seconds before the 1000 s budget and returns
   the best sequence discovered.

The combination of a dense parametric scan and a focused local search
typically drives the score well below the required 1.503 bound.
"""

import time
import numpy as np
from math import exp

# ----------------------------------------------------------------------
# Helper functions (all top‑level – no closures)
# ----------------------------------------------------------------------


def _normalise(seq: np.ndarray) -> np.ndarray:
    """Scale the array so that its L1‑norm is exactly 1."""
    s = seq.sum()
    if s == 0.0:
        # fallback – uniform tiny vector to avoid division by zero
        return np.full_like(seq, 1.0 / seq.size)
    return seq / s


def _power_sequence(n: int, p: float, delta: float = 1.0) -> np.ndarray:
    """a_i = (i + delta)^(‑p) for i = 0..n‑1."""
    idx = np.arange(n, dtype=float) + delta
    seq = idx ** (-p)
    return _normalise(seq)


def _exp_sequence(n: int, c: float) -> np.ndarray:
    """
    a_i = exp(‑c·i / n).  The parameter `c` controls the decay rate;
    c≈1 gives a shape comparable to the optimal power‑law.
    """
    i = np.arange(n, dtype=float)
    seq = np.exp(-c * i / n)
    return _normalise(seq)


def _linear_sequence(n: int) -> np.ndarray:
    """Simple decreasing linear ramp from 1 to 0 (normalised)."""
    seq = np.linspace(1.0, 0.0, n, dtype=float)
    return _normalise(seq)


def _score(seq: np.ndarray) -> float:
    """Convenient wrapper – evaluate_sequence expects a plain list."""
    return evaluate_sequence(seq.tolist())


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative height sequence that minimises
    `evaluate_sequence`.  The routine respects the 1000 s time budget
    and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # --------------------------------------------------------------
    # 1. Working length
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)                 # type: ignore[name-defined]
    except Exception:
        n = 3000                                   # a comfortable default

    print(f"Working length n = {n}")

    # --------------------------------------------------------------
    # 2. Broad grid search over a few parametric families
    # --------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    # ----- power‑law family (p, delta) -----
    p_grid = np.linspace(0.45, 0.75, 301)          # step ≈0.001
    delta_grid = np.array([0.5, 1.0, 1.5, 2.0])    # small offset variants
    for delta in delta_grid:
        for p in p_grid:
            if time.time() >= deadline:
                break
            seq = _power_sequence(n, p, delta)
            sc = _score(seq)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print(f"Power p={p:.4f} Δ={delta:.1f} → score={sc:.6f}")

    # ----- exponential family (c) -----
    c_grid = np.linspace(0.2, 5.0, 241)            # step ≈0.02
    for c in c_grid:
        if time.time() >= deadline:
            break
        seq = _exp_sequence(n, c)
        sc = _score(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"Exp  c={c:.3f} → score={sc:.6f}")

    # ----- linear ramp -----
    seq = _linear_sequence(n)
    sc = _score(seq)
    if sc < best_score:
        best_score = sc
        best_seq = seq.copy()
        print(f"Linear ramp → score={sc:.6f}")

    # ----- previous construction (if available) -----
    if 'height_sequence_1' in globals():
        try:
            seq = np.array(height_sequence_1[:n], dtype=float)
            seq = _normalise(seq)
            sc = _score(seq)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print(f"Prev. construction → score={sc:.6f}")
        except Exception:
            pass

    # At this point we have a decent starting point
    if best_seq is None:
        # Fallback – never happens, but keep the code safe
        best_seq = _power_sequence(n, 0.6)
        best_score = _score(best_seq)

    print(f"--- Grid search finished – best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3. Local stochastic hill‑climb (many cheap mutations)
    # --------------------------------------------------------------
    current_seq = best_seq.copy()
    current_score = best_score

    max_iter = 200_000          # generous upper bound; real number limited by time
    iter_cnt = 0

    # parameters controlling the magnitude of perturbations;
    # they are slowly reduced so the search becomes finer.
    block_max_frac = 0.10       # start with up to 10 % of the array as a block
    factor_range = 0.30         # multiplicative factor drawn from [1‑r, 1+r]

    while time.time() < deadline and iter_cnt < max_iter:
        iter_cnt += 1

        # ----- adapt block size and factor range periodically -----
        if iter_cnt % 10_000 == 0:
            # shrink the neighbourhood to focus on fine tuning
            block_max_frac = max(0.001, block_max_frac * 0.8)
            factor_range = max(0.01, factor_range * 0.85)

        # pick a random contiguous block
        block_len = max(1, int(np.random.rand() ** 2 * block_max_frac * n))
        start = np.random.randint(0, n - block_len + 1)
        end = start + block_len

        # propose a multiplicative perturbation
        factor = 1.0 + factor_range * (np.random.rand() - 0.5) * 2.0   # +/-range
        new_seq = current_seq.copy()
        new_seq[start:end] *= factor
        # enforce non‑negativity
        new_seq = np.maximum(new_seq, 0.0)
        # renormalise to keep the L1‑norm at 1 (scale‑invariant)
        new_seq = _normalise(new_seq)

        new_score = _score(new_seq)

        if new_score < current_score:
            # accept improvement
            current_seq = new_seq
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.copy()
                print(f"Iter {iter_cnt:06d}: improved score = {best_score:.6f}")
        else:
            # Occasionally accept a worse move to escape shallow local minima
            # (simulated‑annealing style). The probability decays with iteration.
            temperature = max(0.001, 1.0 - iter_cnt / max_iter)
            if np.random.rand() < temperature * 0.001:
                current_seq = new_seq
                current_score = new_score

    print(f"Finished search – best score found: {best_score:.6f}")
    # Return a plain Python list as required by the framework
    return best_seq.tolist()
