"""
search for a low‑score step function.

The evaluation score is   E = 2·n·max(a*a) / (sum(a))².
It is invariant under a global scaling of the sequence,
so we keep the ℓ¹‑norm fixed to sqrt(2·n) during the search.
The algorithm consists of two stages:

1.  A small‑dimensional optimisation using a linear programme.
    For a current sequence f we compute M = max(f*f).
    Keeping M fixed we solve
        maximise   Σ g_j
        subject to g_j ≥ 0  and  (f * g)(k) ≤ M   for every shift k .
    The programme is linear because (f*g)(k) is linear in the unknown g.
    The solution g₀ is then rescaled to the target ℓ¹‑norm and mixed
    with f:   f ← (1‑t)·f + t·g.  Repeating this step strictly reduces M
    (and therefore the score) until the improvement stalls.

2.  A cheap random hill‑climb that fine‑tunes the best sequence found
    by the LP phase.

Both stages are time‑limited: they stop a few seconds before the
overall budget expires.  The final sequence is returned as a plain
list of non‑negative floats.
"""

import time
import numpy as np
from math import sqrt
import scipy.optimize
import scipy.sparse

# the evaluation routine is supplied in the execution environment
# from it we only import the name (the function itself is already present)
# from evaluate_sequence import evaluate_sequence   # not needed – already in globals


def _max_convolution(seq: np.ndarray) -> float:
    """Return max_{k} (seq * seq)[k] using numpy.convolve (exact)."""
    conv = np.convolve(seq, seq)
    return float(conv.max())


def _score_from_seq(seq: np.ndarray, target_sum: float) -> float:
    """Score for a sequence that already has ℓ¹‑norm = target_sum."""
    max_b = _max_convolution(seq)
    n = seq.size
    return 2.0 * n * max_b / (target_sum * target_sum)


def _build_constraint_matrix(f: np.ndarray) -> scipy.sparse.csr_matrix:
    """
    Build the matrix A for the linear programme:
        (f * g)[k] = Σ_j f_{k-j} g_j   ≤  M    for all k.
    Row k corresponds to shift k, column j to variable g_j.
    """
    n = f.size
    rows = 2 * n - 1
    # dense construction followed by conversion to CSR – fast enough for n≈2000
    A = np.zeros((rows, n), dtype=float)
    # column j contains f shifted down by j positions
    for j in range(n):
        A[j : j + n, j] = f
    return scipy.sparse.csr_matrix(A)


def _lp_step(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve the LP described in the docstring and return the optimal g
    (scaled later to the target ℓ¹‑norm).  Returns None if the LP fails.
    """
    n = f.size
    A = _build_constraint_matrix(f)
    b = np.full(2 * n - 1, M, dtype=float)

    # maximise Σ g  →  minimise -Σ g
    c = -np.ones(n, dtype=float)

    bounds = [(0.0, None)] * n
    # use HiGHS (default) – works well for moderate sizes
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                                 method='highs', options={'presolve': True})
    if not res.success:
        return None
    return res.x


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Main entry point.  Returns a list of non‑negative heights that
    minimises ``evaluate_sequence`` as far as the 1000 s budget allows.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # --------------------------------------------------------------
    # 1. Choose a working length.
    #    Larger n gives a finer asymptotic bound but makes the LP slower.
    # --------------------------------------------------------------
    n = 1800                     # a compromise between quality and speed
    target_sum = sqrt(2.0 * n)   # ℓ¹‑norm we keep fixed throughout the search

    # --------------------------------------------------------------
    # 2. Try several power‑law seeds and run a few LP iterations for each.
    # --------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    # candidate exponents around the known optimum
    seed_ps = [0.55, 0.58, 0.60, 0.62, 0.65]

    for p in seed_ps:
        if time.time() >= deadline:
            break

        # initialise with a pure power law a_i = (i+1)^(-p)
        seq = np.array([(i + 1) ** (-p) for i in range(n)], dtype=float)
        seq *= target_sum / seq.sum()      # normalise ℓ¹‑norm

        cur_score = _score_from_seq(seq, target_sum)
        cur_M = _max_convolution(seq)

        print(f"--- start p={p:.3f}  score={cur_score:.6f}  maxconv={cur_M:.6f}")

        # LP‑iteration parameters
        max_lp_iter = 30
        t = 0.35                     # mixing weight (1‑t)*f + t*g

        for it in range(max_lp_iter):
            if time.time() >= deadline:
                break

            g0 = _lp_step(seq, cur_M)
            if g0 is None:
                print(f"LP failed at it={it} for p={p:.3f}")
                break

            # rescale g0 to the target ℓ¹‑norm
            sum_g0 = g0.sum()
            if sum_g0 <= 0:
                break
            g = g0 * (target_sum / sum_g0)

            # convex combination
            new_seq = (1.0 - t) * seq + t * g

            # evaluate the new candidate
            new_score = _score_from_seq(new_seq, target_sum)
            new_M = _max_convolution(new_seq)

            # accept if we made progress
            if new_score < cur_score:
                seq = new_seq
                cur_score = new_score
                cur_M = new_M
                print(f"  LP iter {it:02d} improved to {cur_score:.6f}")
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = seq.copy()
            else:
                # no improvement – reduce the mixing weight to stay close to
                # the current point; eventually we give up on this seed.
                t *= 0.5
                if t < 1e-4:
                    break

        # keep the best sequence seen so far for this seed
        if cur_score < best_score:
            best_score = cur_score
            best_seq = seq.copy()

    # --------------------------------------------------------------
    # 3. Final random hill‑climb (single‑coordinate perturbations).
    # --------------------------------------------------------------
    if best_seq is None:          # safety fallback
        best_seq = np.ones(n) * (target_sum / n)

    cur_seq = best_seq.copy()
    cur_score = best_score
    max_hc_iter = 4000
    hc_step = 0.03                # ±3 % multiplicative changes

    for it in range(max_hc_iter):
        if time.time() >= deadline:
            break
        idx = np.random.randint(n)
        factor = 1.0 + hc_step * (np.random.rand() - 0.5)   # -1.5 % … +1.5 %
        old_val = cur_seq[idx]
        cur_seq[idx] = max(0.0, old_val * factor)

        new_score = _score_from_seq(cur_seq, target_sum)

        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = cur_seq.copy()
                print(f"HC iter {it:04d}: improved score {best_score:.6f}")
        else:
            # revert the change
            cur_seq[idx] = old_val

    # --------------------------------------------------------------
    # 4. Return the best sequence (as plain Python list).
    # --------------------------------------------------------------
    print(f"Search finished – best score = {best_score:.6f}")
    # Convert to list of Python floats (evaluate_sequence expects a list)
    return best_seq.tolist()
