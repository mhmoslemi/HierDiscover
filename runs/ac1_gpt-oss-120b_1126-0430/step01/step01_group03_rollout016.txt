```python
def propose_candidate(seed=0, budget_s=1000):
    import numpy as np, time, math, random

    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.1

    best_seq = None
    best_score = float('inf')

    # Use initial if exists
    if 'height_sequence_1' in globals():
        try:
            s = evaluate_sequence(height_sequence_1)
            best_seq = list(height_sequence_1)
            best_score = s
            print(f"[init] Existing seq score {s:.6f}")
        except Exception:
            pass

    # Parameter grid
    lengths = [800, 1200, 1600, 2000, 2500, 3000, 4000]
    beta_vals = np.linspace(0.45, 0.75, 31)  # step 0.01
    offsets = [0, 2, 5, 10, 20, 50]

    for n in lengths:
        if time.time() > deadline: break
        for offset in offsets:
            if offset >= n: continue
            for beta in beta_vals:
                if time.time() > deadline: break
                seq = np.zeros(n, dtype=np.float64)
                # fill from offset
                idx = np.arange(offset, n) + 1   # i+1
                seq[offset:] = idx.astype(np.float64) ** (-beta)
```

assistantfinal**Solution Overview**

The evaluation

```
score = 2·n·max_k (a*a)_k / (sum a)^2
```

is *scale‑invariant*: multiplying the whole vector by any constant does not change the score.
Hence we only have to search for a good *shape* of the vector.

The best known families are power‑law distributions  

```
a_i ∝ (i+1)^(-β)      (β≈0.6)
```

but the simple scan used before does not explore two useful extensions:

* **offset** – set the first `offset` entries to zero.  
  This removes the huge contribution of the pair `(0,0)` to the convolution and lets the
  maximum move to a region where many small products are summed, often yielding a lower
  overall score.

* **different lengths** – the factor `2·n` in the score means that the optimal
  length is not obvious; trying several lengths is cheap because the convolution
  is performed with FFT.

The algorithm consists of three stages:

1. **Initialisation** – keep the previously found sequence (`height_sequence_1`) if it
   exists.
2. **Coarse grid search** – enumerate
   * sequence length `n` (≈ 800 … 4000),
   * offset `off` (0, 2, 5, 10, 20, 50),
   * exponent `β` (0.45 … 0.75, step 0.01).

   For every triple we build the vector  

   ```
   a_i = 0                        for i < off
   a_i = (i+1)^(-β)               otherwise
   ```

   and evaluate it. The best vector is stored.

3. **Local optimisation (simulated annealing)** – starting from the best vector,
   repeatedly perform cheap random moves that preserve non‑negativity:

   * **mass transfer** – move a random fraction (≤ 30 %) from one index to another,
   * **block scaling** – multiply a short random block by a log‑normal factor,
   * **swap** – exchange two entries.

   A simple exponential temperature schedule (`T←T·decay`) decides whether a
   non‑improving move is accepted.  All moves are cheap, the convolution is the only
   expensive operation and stays well inside the 1000 s budget.

The routine prints occasional progress messages and finally returns the best
sequence found.

```python
"""
Propose a step‑function sequence with a low evaluation score.

Strategy
--------
1.  If a previous solution (`height_sequence_1`) exists, keep it as a starting
    point.
2.  Perform a grid search over:
        - length n            (800 … 4000)
        - offset (zeros prefix)   (0,2,5,10,20,50)
        - power‑law exponent β (0.45 … 0.75, step 0.01)
   The candidate is a_i = 0 for i < offset, otherwise a_i = (i+1)^(-β).
3.  Starting from the best candidate, run a short simulated‑annealing / hill‑climb
    phase that applies random mass‑transfer, block‑scaling and swap moves.
    The temperature decays exponentially; moves that improve the score are always
    accepted, worsening moves are accepted with probability exp(−Δ/T).

The code respects the 1000 s wall‑clock limit, prints progress information and
returns the best sequence as a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper to build a power‑law vector with a zero prefix (offset)
# ----------------------------------------------------------------------
def build_powerlaw_vector(n: int, beta: float, offset: int) -> np.ndarray:
    """Return a length‑n vector:
       a[i] = 0               for i < offset
       a[i] = (i+1)**(-beta)  otherwise
    """
    seq = np.zeros(n, dtype=np.float64)
    if offset < n:
        idx = np.arange(offset, n, dtype=np.float64) + 1.0   # i+1
        seq[offset:] = np.power(idx, -beta)
    return seq

# ----------------------------------------------------------------------
# Simulated‑annealing move generators
# ----------------------------------------------------------------------
def mass_transfer_move(seq: np.ndarray, max_frac: float = 0.30) -> np.ndarray:
    """Move a fraction (≤max_frac) of mass from i to j."""
    n = seq.shape[0]
    i, j = np.random.randint(0, n, size=2)
    if i == j or seq[i] == 0.0:
        return None
    eps = seq[i] * np.random.rand() * max_frac
    if eps <= 0.0:
        return None
    new_seq = seq.copy()
    new_seq[i] -= eps
    new_seq[j] += eps
    # guard against tiny negatives caused by floating point noise
    if new_seq[i] < 0.0:
        new_seq[i] = 0.0
    return new_seq

def block_scale_move(seq: np.ndarray, max_factor: float = 1.5) -> np.ndarray:
    """Multiply a short random block by a log‑normal factor (≈1±20%)."""
    n = seq.shape[0]
    start = np.random.randint(0, n)
    # block length ≈ 1–5 % of n, at least 1
    length = np.random.randint(1, max(2, n // 20) + 1)
    # log‑normal factor centred at 1, sigma≈0.2
    factor = math.exp(np.random.randn() * 0.2)
    factor = max(0.1, min(max_factor, factor))   # keep it reasonable
    new_seq = seq.copy()
    new_seq[start:start + length] *= factor
    # clip to non‑negative (should already be true)
    new_seq = np.maximum(new_seq, 0.0)
    return new_seq

def swap_move(seq: np.ndarray) -> np.ndarray:
    """Exchange two entries."""
    n = seq.shape[0]
    i, j = np.random.randint(0, n, size=2)
    if i == j:
        return None
    new_seq = seq.copy()
    new_seq[i], new_seq[j] = new_seq[j], new_seq[i]
    return new_seq

# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """Return a list of non‑negative heights that (hopefully) minimise
       evaluate_sequence().  The routine respects the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # small safety margin

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 0️⃣  Use the previous solution if it exists
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            s = evaluate_sequence(height_sequence_1)
            best_seq = list(height_sequence_1)
            best_score = s
            print(f"[init] Existing sequence score = {s:.6f}")
        except Exception:
            pass

    # --------------------------------------------------------------
    # 1️⃣  Grid search over length / offset / exponent
    # --------------------------------------------------------------
    lengths = [800, 1200, 1600, 2000, 2500, 3000, 4000]
    offsets = [0, 2, 5, 10, 20, 50]
    betas = np.linspace(0.45, 0.75, 31)   # step ≈0.01

    for n in lengths:
        if time.time() > deadline:
            break
        for off in offsets:
            if off >= n:
                continue
            for beta in betas:
                if time.time() > deadline:
                    break
                seq = build_powerlaw_vector(n, beta, off)
                # evaluate – convert to list for the black‑box function
                try:
                    score = evaluate_sequence(seq.tolist())
                except Exception:
                    continue
                if score < best_score:
                    best_score = score
                    best_seq = seq.tolist()
                    print(f"[grid] n={n:4d} off={off:2d} β={beta:5.3f} → score={score:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Local optimisation – simulated annealing / hill‑climb
    # --------------------------------------------------------------
    if best_seq is not None:
        seq = np.array(best_seq, dtype=np.float64)
        n = seq.shape[0]
        current_score = best_score

        # temperature schedule
        T = 1.0            # initial temperature
        Tmin = 1e-4
        decay = 0.9995     # per iteration

        iteration = 0
        while time.time() < deadline:
            iteration += 1
            move_type = np.random.rand()
            if move_type < 0.70:
                new_seq = mass_transfer_move(seq, max_frac=0.30)
                if new_seq is None:
                    continue
            elif move_type < 0.90:
                new_seq = block_scale_move(seq, max_factor=3.0)
            else:
                new_seq = swap_move(seq)
                if new_seq is None:
                    continue

            try:
                new_score = evaluate_sequence(new_seq.tolist())
            except Exception:
                continue

            delta = new_score - current_score
            accept = False
            if delta < 0:
                accept = True
            else:
                # Metropolis criterion
                prob = math.exp(-delta / max(T, 1e-12))
                if np.random.rand() < prob:
                    accept = True

            if accept:
                seq = new_seq
                current_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_seq = new_seq.tolist()
                    print(f"[sa] iter {iteration:7d} → improvement {new_score:.6f}")

            # temperature cooling
            T = max(Tmin, T * decay)

    # --------------------------------------------------------------
    # 3️⃣  Return the best sequence found
    # --------------------------------------------------------------
    if best_seq is None:
        # should never happen (the problem definition guarantees a non‑empty answer)
        raise RuntimeError("No valid sequence was generated.")
    print(f"[final] best score = {best_score:.6f}  length = {len(best_seq)}")
    return best_seq
```
