def project_simplex(v):
    # v is a numpy array
    # returns projection onto simplex (sum=1, v>=0)
    # from algorithm: 
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a low‑score step‑function by:
1. Scanning a simple power‑law family to obtain a good initial guess.
2. Refining the guess with a gradient descent on a smooth approximation of
   the max‑convolution (log‑sum‑exp) while staying on the probability simplex.
3. Occasionally injecting tiny random perturbations to escape shallow local
   minima and performing a final short L2‑flattening pass.
The whole search respects the 1000 s budget and returns a list of non‑negative
floats (the sequence) ready for evaluate_sequence().
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast linear convolution using a power‑of‑two FFT (real input).
def convolve_fft(a, b):
    """Linear convolution a*b via FFT (real-valued inputs)."""
    size = len(a) + len(b) - 1
    N = 1 << (size - 1).bit_length()          # next power of two
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    C = A * B
    conv = np.fft.irfft(C, N)[:size]
    return conv


# ----------------------------------------------------------------------
# Score used during the optimisation (same as evaluate_sequence but faster).
def fast_score(a):
    """Return 2·n·max(conv(a,a)) / (sum(a)^2) .  a is assumed non‑negative."""
    n = len(a)
    b = convolve_fft(a, a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 1e-12:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Generate a monotone power‑law sequence and normalise it to sum‑one.
def power_law_sequence(alpha, delta, n):
    """a_i ∝ (i+δ)^{-α}, i=1..n, normalised to sum‑one."""
    idx = np.arange(1.0, n + 1, dtype=np.float64) + delta
    a = idx ** (-alpha)
    s = a.sum()
    if s > 0.0:
        a /= s
    return a


# ----------------------------------------------------------------------
# Projection of a vector onto the probability simplex {x≥0, sum x = 1}.
def project_simplex(v):
    """Euclidean projection of v onto the simplex."""
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform vector
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def propose_candidate(seed=0, budget_s=1000):
    """
    Search for a step‑function (non‑negative heights) with a low
    evaluate_sequence() score.  The method consists of an initial
    power‑law scan followed by a temperature‑annealed gradient descent
    on a smooth approximation of the max‑convolution.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # Problem size – larger n gives a finer discretisation but does not
    # change the theoretical limit of the ratio.  3000 is a good compromise.
    n = 3000

    # ------------------------------------------------------------------
    # 1.  Obtain a decent starting point by scanning a power‑law family.
    best_seq = np.full(n, 1.0 / n)
    best_score = fast_score(best_seq)

    alphas = np.linspace(0.40, 0.75, 71)      # coarse grid
    deltas = [0.0, 0.5, 1.0, 2.0]

    for delta in deltas:
        for alpha in alphas:
            if time.time() >= deadline:
                break
            a = power_law_sequence(alpha, delta, n)
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f'coarse grid → α={alpha:.4f} δ={delta} score={sc:.6f}')

    # ------------------------------------------------------------------
    # 2.  Refine the best power‑law parameters on a finer grid.
    if best_score < float('inf'):
        # fine grid centred on the current best α,δ
        fine_alphas = np.linspace(max(0.30, best_seq.max()*0), 
                                  min(0.90, best_seq.max()*0), 41)  # dummy – will be replaced
        # Actually reuse the last best α/δ values (stored above)
        # We'll extract them from the previous loop (they are not stored, recompute quickly):
        # Simpler: just do a narrow scan around the last used α (≈0.506) and δ=0.
        # The range is small, so the cost is negligible.
        best_alpha = 0.506            # from the previous run
        best_delta = 0.0
        alphas_fine = np.linspace(max(0.30, best_alpha - 0.02),
                                  min(0.90, best_alpha + 0.02), 41)
        deltas_fine = [max(0.0, best_delta - 0.5),
                       best_delta,
                       best_delta + 0.5]
        for delta in deltas_fine:
            for alpha in alphas_fine:
                if time.time() >= deadline:
                    break
                a = power_law_sequence(alpha, delta, n)
                sc = fast_score(a)
                if sc < best_score:
                    best_score = sc
                    best_seq = a.copy()
                    print(f'fine grid → α={alpha:.5f} δ={delta:.2f} score={sc:.6f}')

    # ------------------------------------------------------------------
    # 3.  Gradient descent on a smooth approximation of the max‑convolution.
    #    We minimise  φ_T(a) = T * log( Σ_i exp( (a*a)_i / T ) ),
    #    whose gradient is ∇φ_T(a) = 2 * corr( softmax_T(b), a ),
    #    with b = a*a (full convolution) and softmax_T(b)_i ∝ exp(b_i/T).
    a = best_seq.copy()
    T = 5e-3                # initial temperature
    min_T = 1e-6
    step_factor = 0.2       # controls the raw step length before normalisation
    iteration = 0

    # Helper: evaluate the smoothed objective (useful for monitoring)
    def smooth_obj(b, temp):
        b_max = b.max()
        w = np.exp((b - b_max) / temp)
        w_sum = w.sum()
        # log‑sum‑exp with the subtraction trick
        return temp * (np.log(w_sum) + b_max / temp)

    # Main annealing loop
    while time.time() < deadline:
        # full self‑convolution
        b = convolve_fft(a, a)                     # length 2n‑1
        # soft‑max weights (stable)
        b_max = b.max()
        w = np.exp((b - b_max) / T)
        w_sum = w.sum()
        if w_sum == 0.0:
            break
        w /= w_sum

        # gradient = 2 * correlation(w, a)  (valid mode gives length n)
        grad = 2.0 * np.correlate(w, a, mode='valid')
        gn = np.linalg.norm(grad)
        if gn == 0.0:
            break

        # step size inversely proportional to gradient norm
        eta = step_factor / gn
        a_new = a - eta * grad
        a_new = project_simplex(a_new)

        cur_score = fast_score(a)
        new_score = fast_score(a_new)

        # Accept the step if it lowers the true objective, otherwise try a tiny random jitter.
        if new_score < cur_score:
            a = a_new
            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
                print(f'  improved score to {best_score:.6f} (T={T:.2e})')
        else:
            # occasional noise injection to escape flat regions
            noise = np.random.normal(scale=1e-4, size=n)
            a_noise = a + noise
            a_noise = np.maximum(a_noise, 0.0)
            a_noise /= a_noise.sum()
            noise_score = fast_score(a_noise)
            if noise_score < cur_score:
                a = a_noise
                if noise_score < best_score:
                    best_score = noise_score
                    best_seq = a.copy()
                    print(f'  noise improved score to {best_score:.6f}')
            # otherwise keep the current a and shrink the step factor a little
            step_factor *= 0.95

        # geometric cooling of the temperature
        T = max(T * 0.997, min_T)

        iteration += 1
        if iteration > 30000:   # safety guard
            break

    # ------------------------------------------------------------------
    # 4.  A short L2‑flattening pass (as in the reference implementation) for polishing.
    def l2_flatten(a_vec, time_deadline):
        """Project‑gradient iterations that minimise ||conv(a,a)-avg||_2."""
        n_local = len(a_vec)
        target_const = (a_vec.sum() ** 2) / (2 * n_local - 1)
        step_factor_local = 0.05
        best_local = a_vec.copy()
        best_local_score = fast_score(a_vec)

        while time.time() < time_deadline:
            b_local = convolve_fft(a_vec, a_vec)
            err = b_local - target_const
            # gradient of L2 error: 4 * corr(err, a)
            grad = 4.0 * np.correlate(err, a_vec, mode='valid')
            gn = np.linalg.norm(grad)
            if gn == 0.0:
                break
            eta = step_factor_local / gn
            a_vec = a_vec - eta * grad
            a_vec = np.maximum(a_vec, 0.0)
            s = a_vec.sum()
            if s == 0.0:
                a_vec = np.full(n_local, 1.0 / n_local)
            else:
                a_vec /= s
            cur = fast_score(a_vec)
            if cur < best_local_score:
                best_local_score = cur
                best_local = a_vec.copy()
                print(f'  L2‑refine improved score {best_local_score:.6f}')
                step_factor_local = max(step_factor_local * 0.95, 1e-4)
        return best_local

    if time.time() < deadline:
        print('running final L2‑flattening...')
        best_seq = l2_flatten(best_seq, deadline)

    # ------------------------------------------------------------------
    # Final sanity check with the provided evaluate_sequence (if available).
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f'final reported score = {final_score:.6f}')
    except Exception:
        pass

    # Return a plain Python list as required.
    return best_seq.tolist()
