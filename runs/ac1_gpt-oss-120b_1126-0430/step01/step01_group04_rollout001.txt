"""
Projected gradient descent on a smooth approximation of the
max‑convolution objective.

The score returned by ``evaluate_sequence`` (with the sum of the
coefficients normalised to sqrt(2·n)) equals the maximum entry of the
linear convolution of the sequence with itself.  Minimising this
maximum is a non‑convex ℓ∞‑problem.  We approximate the max by a
smooth log‑sum‑exp (soft‑max) with a temperature τ and perform gradient
steps on the resulting differentiable surrogate.  The gradient of the
soft‑max can be written analytically as a correlation of the soft‑max
weights with the reversed sequence, which we compute efficiently with
FFT‑based convolutions.  After each step the vector is projected onto
the simplex {x≥0, Σx = √(2·n)} so that the scale‑invariance of the
score is respected.

The algorithm adapts the step size by a simple back‑tracking line
search, cools the temperature τ, and occasionally injects small random
perturbations to escape shallow plateaus.  It returns the best sequence
found within the given time budget (default 1000 s).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_to_simplex(x: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Project vector ``x`` onto the simplex
    { y ≥ 0, Σ y = target_sum } using the O(n) algorithm of
    Wang & Carreira‑Perpiñán (2013).  Works for any ``target_sum > 0``.
    """
    if target_sum <= 0:
        return np.zeros_like(x)

    # Sort descending
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho = np.nonzero(u - cssv / np.arange(1, len(u) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(x - theta, 0.0)
    return w


def _evaluate(seq) -> float:
    """
    Thin wrapper around the provided ``evaluate_sequence``.
    The grader supplies ``evaluate_sequence`` in the global namespace.
    """
    return evaluate_sequence(seq)


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Convolution of ``a`` with itself (linear convolution) using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = len(a)
    size = 1 << ((2 * n - 1).bit_length())   # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    # Numerical noise may produce tiny negative values – clip them
    np.maximum(conv, 0.0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution of ``x`` and ``y`` using FFT.
    Returns an array of length len(x)+len(y)-1.
    """
    m = len(x) + len(y) - 1
    size = 1 << (m.bit_length())
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:m]
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Main optimiser
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Return a non‑negative step‑function (list of heights) that strives
    to minimise ``evaluate_sequence``.  The routine runs for at most
    ``budget_s`` seconds and prints progress information.
    """
    np.random.seed(seed)

    # --------------------------------------------------------------
    # Problem size and normalisation
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
        raw_start = np.array(height_sequence_1, dtype=float)
    else:
        # a moderately sized vector that evaluates fast
        n = 2000
        raw_start = np.random.rand(n)

    target_sum = math.sqrt(2.0 * n)          # Normalising ℓ₁‑norm

    # Normalise the start vector onto the simplex
    a = _project_to_simplex(raw_start, target_sum)

    # --------------------------------------------------------------
    # Initial evaluation
    # --------------------------------------------------------------
    best_seq = a.copy()
    best_score = _evaluate(best_seq.tolist())
    current_score = best_score
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # Hyper‑parameters
    # --------------------------------------------------------------
    step_size = target_sum * 0.5          # initial step magnitude
    tau = 0.5                             # temperature for soft‑max smoothing
    min_tau = 1e-3
    tau_cool = 0.995                      # geometric cooling per iteration
    max_backtrack = 10                    # attempts in the line search
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # keep a tiny margin

    iteration = 0
    # --------------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # 1) convolution of the current candidate
        conv = _conv_self_fft(a)               # length 2·n‑1
        max_conv = conv.max()

        # 2) soft‑max weights (smooth approximation of the max)
        #    exp((c - max_c)/tau) ≤ 1, so the exponent never overflows.
        exp_arg = (conv - max_conv) / tau
        # Clip to avoid under‑flow for very negative arguments
        np.clip(exp_arg, -50, None, out=exp_arg)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:                      # extremely unlikely, guard anyway
            w = np.zeros_like(w)
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum                         # now a probability vector

        # 3) gradient of the smooth surrogate
        #    g = 2 * (w ⋆ a_rev)   where ⋆ is the linear correlation.
        a_rev = a[::-1]
        g_full = _conv_fft(w, a_rev)           # length 3·n‑2
        g = 2.0 * g_full[n - 1 : n - 1 + n]     # extract the n‑length slice

        # 4) back‑tracking line search
        accepted = False
        factor = 1.0
        for _ in range(max_backtrack):
            a_candidate = a - step_size * factor * g
            # Project back onto the simplex (preserves the required sum)
            a_candidate = _project_to_simplex(a_candidate, target_sum)

            cand_score = _evaluate(a_candidate.tolist())
            if cand_score < current_score:
                accepted = True
                break
            factor *= 0.5                       # shrink the step

        if accepted:
            a = a_candidate
            current_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = a_candidate.copy()
                print(f"[iter {iteration:06d}] new best {best_score:.6f}")
            # slightly enlarge step after a successful move
            step_size *= 1.03
        else:
            # no improvement – shrink the step and try a tiny random jiggle
            step_size *= 0.7
            if np.random.rand() < 0.05:
                i, j = np.random.choice(n, size=2, replace=False)
                delta = a[i] * np.random.uniform(0.0, 0.02)
                a[i] -= delta
                a[j] += delta
                a = _project_to_simplex(a, target_sum)

        # 5) temperature cooling (focuses more on the true max as we
        #    progress)
        tau = max(min_tau, tau * tau_cool)

        # 6) occasional global perturbation to broaden the search
        if iteration % 3000 == 0:
            mask = np.random.rand(n) < 0.015
            nmask = mask.sum()
            if nmask > 0:
                a[mask] = np.random.rand(nmask)
                a = _project_to_simplex(a, target_sum)

    print(f"[final] best score {best_score:.6f} after {iteration} iterations")
    return best_seq.tolist()
