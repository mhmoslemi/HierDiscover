def project_simplex(v):
    # ensures nonneg, sum = 1
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

Propose a low‑score step‑function using softmax‑gradient descent on the
maximum of the self‑convolution.  The method starts from a
power‑law distribution, then repeatedly takes a projected gradient
step that minimizes a smooth approximation of max(conv(a,a)).
A tiny simulated‑annealing perturbation is added occasionally to
escape shallow local minima.  The algorithm runs for the allotted time
and returns the best sequence seen.
"""

import time, math, numpy as np

def convolve_fft(a, b):
    size = len(a) + len(b) - 1
    N = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    C = A * B
    conv = np.fft.irfft(C, N)[:size]
    return conv

def fast_score(a):
    n = len(a)
    b = convolve_fft(a, a)
    maxb = b.max()
    return float(2.0 * n * maxb)

def project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries zero or negative, return uniform
        return np.full_like(v, 1.0/n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def grad_max_softmax(a, beta):
    """Gradient of the smooth max (log‑sum‑exp) of a*a."""
    n = len(a)
    b = convolve_fft(a, a)
    maxb = b.max()
    # weights for each convolution index
    w = np.exp(beta * (b - maxb))
    ws = w.sum()
    if ws == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / ws
    # convolution of a with reversed weights
    conv_rev = convolve_fft(a, w[::-1])
    start = n - 1
    slice_part = conv_rev[start:start + n]
    grad = 2.0 * slice_part[::-1]   # size n
    return grad

    idx = np.arange(1.0, n + 1, dtype=np.float64) + delta
    a = idx ** (-alpha)
    a_sum = a.sum()
    if a_sum > 0:
        a = a / a_sum
    return a

        print(f'final reported score = {final_score:.6f}')

"""
Propose a low‑score step‑function using projected gradient descent on a
smooth approximation of the ∞‑norm of the self‑convolution.

The objective of evaluate_sequence is
    2·n·max(conv(a,a)) / (sum(a)**2)

Because the value is scale‑invariant we keep the vector on the
probability simplex (sum = 1).  The algorithm works as follows:

1.  Initialise a with a power‑law distribution (the shape that gave the
    previous best score ≈1.5765 for n≈3000).

2.  Repeatedly minimise a smooth surrogate of max(conv(a,a)):
    fβ(a) = (1/β)·log∑_k exp(β·b_k)   where   b = a∗a .
    Its gradient is
        ∇fβ = 2·conv_rev(a , w_rev)   (see code)
    with weights w_k = softmaxβ(b_k).

    A projected gradient step (step size ∝1/√t) moves a in the
    negative gradient direction, then a is projected onto the simplex
    (non‑negative, sum = 1).

3.  The smoothing parameter β is increased periodically (β←1.5·β)
    so that the optimisation first sees a smooth landscape and then
    focuses on the true maximum.

4.  Every few hundred iterations a tiny random perturbation is added
    (simulated‑annealing move) to help escape shallow plateaus.

5.  The best vector seen (lowest score) is kept.  At the end a final
    sanity call to ``evaluate_sequence`` is performed and the vector is
    returned as a plain Python list.

The whole routine respects the 1000 s budget, uses only NumPy
(and the supplied ``evaluate_sequence``) and runs fully deterministically
for a given ``seed``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a power‑of‑two FFT (real inputs)."""
    size = a.shape[0] + b.shape[0] - 1
    N = 1 << (size - 1).bit_length()          # next power of two
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    C = A * B
    conv = np.fft.irfft(C, N)[:size]
    return conv


# ----------------------------------------------------------------------
def fast_score(a: np.ndarray) -> float:
    """Score 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    b = convolve_fft(a, a)
    max_b = b.max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def grad_max_softmax(a: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smooth max (log‑sum‑exp) of the self‑convolution.
    Returns ∇fβ where fβ = (1/β)·log∑_k exp(β·b_k), b = a∗a.
    """
    n = a.shape[0]
    b = convolve_fft(a, a)
    maxb = b.max()
    # soft‑max weights (numerically stable)
    w = np.exp(beta * (b - maxb))
    ws = w.sum()
    if ws == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= ws

    # gradient = 2 * reverse‑convolution of a with w
    conv_rev = convolve_fft(a, w[::-1])
    start = n - 1
    slice_part = conv_rev[start:start + n]      # length n
    grad = 2.0 * slice_part[::-1]
    return grad


# ----------------------------------------------------------------------
def power_law_sequence(alpha: float, delta: float, n: int) -> np.ndarray:
    """Generate a_i ∝ (i + δ)^{-α} (i = 1…n) and normalise to sum‑one."""
    idx = np.arange(1.0, n + 1, dtype=np.float64) + delta
    a = idx ** (-alpha)
    s = a.sum()
    if s > 0.0:
        a /= s
    return a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function with a low evaluation score.
    Returns a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ---- 1. problem size and initial point --------------------------------
    n = 3000                         # moderate size, fast FFTs
    # initialise from the previously good power‑law shape
    a = power_law_sequence(alpha=0.505, delta=0.0, n=n)
    a = project_simplex(a)

    best_seq = a.copy()
    best_score = fast_score(best_seq)
    print(f'initial score = {best_score:.6f}')

    # ---- 2. projected gradient descent on smooth max -----------------------
    beta = 1.0           # start with a very smooth approximation
    beta_max = 200.0
    step0 = 0.3          # base step size (will be scaled by 1/√t)
    iteration = 0
    momentum = np.zeros_like(a)   # simple momentum term

    while time.time() < deadline:
        iteration += 1

        # gradient of the smooth max
        grad = grad_max_softmax(a, beta)

        # use a small amount of momentum to stabilise descent
        momentum = 0.9 * momentum + grad

        # step size schedule: η_t = step0 / sqrt(t)
        eta = step0 / math.sqrt(iteration)

        a_next = a - eta * momentum
        a_next = np.maximum(a_next, 0.0)       # non‑negative clip
        a_next = project_simplex(a_next)       # back to the simplex

        cur_score = fast_score(a_next)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_next.copy()
            print(f'iter {iteration:5d}: improved score {best_score:.6f} (beta={beta:.2f})')

        a = a_next

        # increase the hardness of the approximation gradually
        if iteration % 200 == 0 and beta < beta_max:
            beta = min(beta * 1.5, beta_max)

        # occasional random bump (simulated‑annealing style)
        if iteration % 500 == 0:
            eps = 0.01 / math.sqrt(iteration)   # shrinking amplitude
            a_pert = a + eps * (np.random.rand(n) - 0.5)
            a_pert = np.maximum(a_pert, 0.0)
            a_pert = project_simplex(a_pert)
            pert_score = fast_score(a_pert)
            if pert_score < best_score:
                best_score = pert_score
                best_seq = a_pert.copy()
                print(f'iter {iteration:5d}: random perturb improved score {best_score:.6f}')
                a = a_pert   # jump to the better perturbed point

    # ---- 3. final sanity check ---------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f'final reported score = {final_score:.6f}')
    except Exception:
        # ``evaluate_sequence`` may not be imported in the environment – ignore.
        pass

    # return as a plain Python list of floats
    return best_seq.tolist()
