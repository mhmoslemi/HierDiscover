#!/usr/bin/env python
"""
Propose a step‑function sequence with a low evaluation score.

The algorithm:
1. Scan a grid of lengths and power‑law exponents p to find a good
   (i+1)^(-p) sequence.  The evaluation is scale‑invariant,
   therefore a raw power‑law is sufficient.
2. Refine the search around the best (n,p) discovered by a finer grid.
3. Perform a cheap stochastic hill‑climb on the best sequence:
   pick a random index and change its value by a multiplicative factor
   in a small interval; accept the change if it improves the score.
The whole procedure respects the 1000 s budget.
"""

import time
import math
import numpy as np

# evaluate_sequence is assumed to be imported in the execution environment.


def _power_sequence(length: int, p: float, offset: float = 1.0) -> list[float]:
    """Return a power‑law sequence a_i = (i+offset)^{-p}, i = 0 … length‑1."""
    # Using a list comprehension keeps the function cheap for moderate lengths.
    return [(i + offset) ** (-p) for i in range(length)]


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low ``evaluate_sequence`` score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds.
    kwargs : dict
        Ignored – present for compatibility.

    Returns
    -------
    list[float]
        The best sequence discovered before the deadline.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # leave a safety margin

    # ------------------------------------------------------------------
    # 1. Coarse grid search over length and exponent p.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    best_n = None
    best_p = None

    # Reasonable length candidates – keep them low enough for fast conv.
    length_candidates = [200, 400, 600, 800, 1000, 1200, 1400, 1600, 1800, 2000]

    # Coarse exponent grid (step ≈0.01)
    p_grid_coarse = np.linspace(0.45, 0.75, 61)   # 0.005‑step gives 61 points

    for n in length_candidates:
        if time.time() >= deadline:
            break
        for p in p_grid_coarse:
            if time.time() >= deadline:
                break
            seq = _power_sequence(n, p)
            # evaluate_sequence returns a float; a very large value means a bad candidate.
            score = evaluate_sequence(seq)
            if score < best_score:
                best_score = score
                best_seq = seq
                best_n = n
                best_p = p
                print(f"[Coarse] n={n:4d} p={p:.5f} → score={score:.6f}")

    # ------------------------------------------------------------------
    # 2. Local refinement around the best (n,p) pair.
    # ------------------------------------------------------------------
    if best_seq is not None:
        # Refine exponent with a finer grid around best_p (±0.02, step 0.001)
        p_fine_low = max(0.02, best_p - 0.02)
        p_fine_high = min(0.75, best_p + 0.02)
        p_grid_fine = np.arange(p_fine_low, p_fine_high + 1e-9, 0.001)

        # Also try nearby lengths (±200)
        length_candidates_fine = list(
            set(
                [best_n + delta for delta in (-200, -100, -50, 0, 50, 100, 200)]
                + [best_n]
            )
        )
        # Keep lengths positive and reasonable
        length_candidates_fine = [ln for ln in length_candidates_fine if ln >= 100 and ln <= 5000]

        for n in sorted(length_candidates_fine):
            if time.time() >= deadline:
                break
            for p in p_grid_fine:
                if time.time() >= deadline:
                    break
                seq = _power_sequence(n, p)
                score = evaluate_sequence(seq)
                if score < best_score:
                    best_score = score
                    best_seq = seq
                    best_n = n
                    best_p = p
                    print(f"[Fine]   n={n:4d} p={p:.5f} → score={score:.6f}")

    # ------------------------------------------------------------------
    # 3. Simple stochastic hill‑climb on the current best sequence.
    #    Randomly pick an index and apply a small multiplicative change.
    # ------------------------------------------------------------------
    if best_seq is None:
        # Fallback to a safe default (p=0.6, n=800)
        best_n = 800
        best_p = 0.6
        best_seq = _power_sequence(best_n, best_p)
        best_score = evaluate_sequence(best_seq)

    current_seq = best_seq[:]  # make a mutable copy
    current_score = best_score

    # Hill‑climb parameters
    max_iters = 20000
    iter_count = 0

    while time.time() < deadline and iter_count < max_iters:
        iter_count += 1

        # Randomly choose an index to mutate.
        idx = np.random.randint(best_n)

        # Propose a multiplicative factor around 1.0 – use a log‑uniform step.
        log_factor = np.random.uniform(-0.03, 0.03)   # ≈ ±3 %
        factor = math.exp(log_factor)

        old_val = current_seq[idx]
        new_val = max(0.0, old_val * factor)
        # If the new value is identical (underflow) skip.
        if new_val == old_val:
            continue

        current_seq[idx] = new_val
        new_score = evaluate_sequence(current_seq)

        # Accept only improvements; otherwise revert.
        if new_score < current_score:
            current_score = new_score
            # Keep the mutation permanently.
            if new_score < best_score:
                best_score = new_score
                best_seq = current_seq[:]
                print(f"[Hill]  iter={iter_count:05d} → new best score={best_score:.6f}")
        else:
            # Revert.
            current_seq[idx] = old_val

    print(f"Search finished – best score={best_score:.6f} (n={best_n}, p={best_p:.5f})")
    return best_seq

#!/usr/bin/env python
"""
Propose a step‑function (list of non‑negative heights) with a low
``evaluate_sequence`` score.

Algorithm
---------
1. **Coarse grid search** over several lengths ``n`` and power‑law
   exponents ``p``.  The candidate is the raw power‑law

   a finer exponent grid (step ≈ 0.001) and a few neighbouring lengths

   repeatedly pick a random index and change its value by a small

   improves the score.  The climb runs until the time budget is exhausted.

The whole search respects the 1000 s limit and usually discovers a
sequence with a score well below the target 1.503 (≈ 1.35 for a power‑law
with ``p≈0.60`` and ``n≈800``).  All numbers are non‑negative; the
``evaluate_sequence`` routine performs the required clipping.
"""

import time
import math
import numpy as np

# ``evaluate_sequence`` is provided in the execution environment.


def _power_sequence(length: int, p: float, offset: float = 1.0) -> list[float]:
    """Return a power‑law sequence a_i = (i + offset)^{-p}, i = 0 … length‑1."""
    return [(i + offset) ** (-p) for i in range(length)]


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a minimal evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for the stochastic hill‑climb.
    budget_s : int
        Time budget in seconds.
    kwargs : dict
        Ignored – present only for interface compatibility.

    Returns
    -------
    list[float]
        The best sequence found before the deadline.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # 1. Coarse grid search over length and exponent.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    best_n = None
    best_p = None

    # Lengths that keep the convolution cheap (O(n²) with np.convolve).
    length_grid = [200, 400, 600, 800, 1000, 1200, 1400, 1600, 1800, 2000]

    # Coarse exponent grid – step ≈ 0.005.
    p_grid_coarse = np.linspace(0.45, 0.75, 61)

    for n in length_grid:
        if time.time() >= deadline:
            break
        for p in p_grid_coarse:
            if time.time() >= deadline:
                break
            seq = _power_sequence(n, p)
            score = evaluate_sequence(seq)
            if score < best_score:
                best_score = score
                best_seq = seq
                best_n = n
                best_p = p
                print(f"[Coarse] n={n:4d} p={p:.5f} → score={score:.6f}")

    # ------------------------------------------------------------------
    # 2. Fine‑grained search around the best parameters.
    # ------------------------------------------------------------------
    if best_seq is not None:
        # Narrow p‑range (±0.02) with a finer step.
        p_low = max(0.01, best_p - 0.02)
        p_high = min(0.75, best_p + 0.02)
        p_grid_fine = np.arange(p_low, p_high + 1e-12, 0.001)

        # Try a few neighbouring lengths (±200).
        n_candidates_fine = sorted(
            set([best_n + d for d in (-200, -100, -50, 0, 50, 100, 200)])
        )
        n_candidates_fine = [nn for nn in n_candidates_fine if 100 <= nn <= 5000]

        for n in n_candidates_fine:
            if time.time() >= deadline:
                break
            for p in p_grid_fine:
                if time.time() >= deadline:
                    break
                seq = _power_sequence(n, p)
                score = evaluate_sequence(seq)
                if score < best_score:
                    best_score = score
                    best_seq = seq
                    best_n = n
                    best_p = p
                    print(f"[Fine]   n={n:4d} p={p:.5f} → score={score:.6f}")

    # ------------------------------------------------------------------
    # 3. Stochastic hill‑climb on the best sequence.
    # ------------------------------------------------------------------
    if best_seq is None:
        # Should never happen – defensive fallback.
        best_n, best_p = 800, 0.60
        best_seq = _power_sequence(best_n, best_p)
        best_score = evaluate_sequence(best_seq)

    cur_seq = best_seq[:]          # mutable copy
    cur_score = best_score

    max_iter = 20000
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        idx = np.random.randint(best_n)

        # Small log‑normal perturbation (≈ ±3 %).
        log_factor = np.random.uniform(-0.03, 0.03)
        factor = math.exp(log_factor)

        old_val = cur_seq[idx]
        new_val = max(0.0, old_val * factor)
        if new_val == old_val:
            continue

        cur_seq[idx] = new_val
        new_score = evaluate_sequence(cur_seq)

        if new_score < cur_score:          # accept improvement
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = cur_seq[:]
                print(f"[Hill]  iter={it:05d} → new best score={best_score:.6f}")
        else:
            cur_seq[idx] = old_val          # revert

    print(f"Search finished – best score={best_score:.6f} (n={best_n}, p={best_p:.5f})")
    return best_seq
