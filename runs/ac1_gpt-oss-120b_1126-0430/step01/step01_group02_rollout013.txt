"""
Propose a candidate step‑function (list of non‑negative heights) that
minimises the supplied ``evaluate_sequence`` function.

Approach
--------
The objective for a normalised sequence (∑a_i = 1) is 2·n·max(a*a).
Hence we have to make the maximal entry of the self‑convolution as
small as possible.  The exact optimal shape (the Catalan / arcsine
distribution) makes the interior convolution constant, but the
sub‑gradient method used previously gets stuck at this fixed point.
To move past it we optimise a *smooth* approximation of the max:
    S_γ(a) = (1/γ)·log∑_k exp(γ·(a*a)_k) .
∇S_γ can be written as a convolution, so each iteration costs O(n²)
and fits easily into the 1000 s budget.

We start from the Catalan seed (or the previously best global
sequence if available) and run an Adam optimiser on S_γ.  The smoothness
parameter γ is increased slowly so that the optimiser first explores
the landscape and then concentrates on the true maximum.  After each
step the vector is projected onto the probability simplex (non‑negative,
unit‑sum).  The best sequence seen is returned.

The algorithm can be restarted if no improvement is observed for a
long time; this helps to escape possible shallow local minima.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper to generate the Catalan‑like seed (a_i = C(2i,i)/4^i)
def _catalan_sequence(n: int) -> np.ndarray:
    """First n Catalan‑type coefficients, normalised to sum = 1."""
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    # recurrence a_{i+1} = a_i * (2*i+1) / (2*(i+1))
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * (i - 1) + 1) / (2.0 * i)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a sequence of non‑negative heights that minimises
    ``evaluate_sequence``.  Runs for at most *budget_s* seconds and
    returns the best sequence found.

    Parameters

    gamma_max = 200.0
    gamma_inc = 1.001        # slowly increase γ → focus on the true max

    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, initial score={best_score:.6f}")

    iteration = 0
    no_improve = 0

    # ------------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # --------------------------------------------------------------
        # 1. Compute the full linear convolution a*a
        conv = np.convolve(a, a)                 # length 2n‑1

        max_c = conv.max()
