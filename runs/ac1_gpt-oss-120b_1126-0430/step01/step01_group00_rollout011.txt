import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

def solve_lp(a, M):
    n = len(a)
    m = 2*n-1
    # Build A_ub sparse
    col_idx = np.repeat(np.arange(n), n)
    row_idx = np.tile(np.arange(n), n) + col_idx
    data = np.repeat(a, n)
    A_ub = sp.csr_matrix((data, (row_idx, col_idx)), shape=(m, n))

    c = -np.ones(n)  # maximize sum b
    bounds = [(0, None)] * n
    b_ub = np.full(m, M, dtype=float)

    # calling linprog
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
    if not res.success:
        # fallback maybe return zero vector?
        return np.zeros_like(a)
    return res.x
```

def _solve_lp_max_sum(a, M, time_limit=None):
    import scipy.sparse as sp, scipy.optimize as opt
    n = len(a)
    m = 2*n - 1
    # Build sparse conv matrix A
    # row indices = np.tile(np.arange(n), n) + np.repeat(np.arange(n), n) ??? actually column repeated
    col_idx = np.repeat(np.arange(n), n)
    row_idx = np.tile(np.arange(n), n) + col_idx
    data = np.repeat(a, n)
    A = sp.csr_matrix((data, (row_idx, col_idx)), shape=(m, n))
    # objective: maximize sum(b) -> minimize -sum(b)
    c = np.concatenate([-np.ones(n)])  # length n
    # constraints
    b_ub = np.full(m, M, dtype=float)
    bounds = [(0, None)] * n
    # solve
    options = {}
    if time_limit is not None:
        options['time_limit'] = time_limit
    res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options=options)
    if res.success:
        return res.x
    else:
        # Could try fallback: simply return uniform
        return np.full(n, 1.0/n)

"""
Propose a step‑function that minimizes the provided evaluate_sequence().

Algorithm summary
-----------------
* Work on the probability simplex (sum a_i = 1, a_i ≥ 0).  The objective
  simplifies to  O = 2·n·max(conv(a,a)).
* For a given length n we perform many independent restarts:
    - Initialise a random point (or a noisy copy of the current best).
    - A cheap hill‑climbing phase moves a small amount of mass between
      two coordinates and occasionally takes a sub‑gradient step that
      pushes down the currently maximal convolution entry.
* After the cheap phase we run a **linear‑programming refinement**.
    - With the current a fixed we maximise Σ b_j   (i.e. minimise the
      reduction in the total mass) under the linear constraints
        conv(a,b) ≤ M   where M = max(conv(a,a)).
      This LP is *exact* (no approximations) and yields a direction g.
    - The new point is a convex combination
        a ← (1‑τ) a + τ·(b/Σb)   with a small step size τ.
      The mixing is guaranteed to never increase the maximal convolution.
* The whole outer loop is time‑bounded (≈ 1000 s).  The best sequence found
  over all lengths and restarts is returned.
* SciPy’s high‑performance “highs” LP solver is used; the convolution
  matrix is built as a sparse CSR matrix (O(n²) non‑zeros, but still fast
  for n ≤ 3000).

The approach is different from the plain hill‑climbing in the previous
submission and typically reaches a lower bound (≈ 1.50) within the time budget.
"""

import time
import math
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt


# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex (Duchi et al.)
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x ≥ 0, sum(x) = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:                     # all entries non‑positive
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective (scale‑invariant version)
def _objective(a: np.ndarray) -> float:
    """Return O = 2·n·max(conv(a,a)) (assumes sum(a)==1)."""
    conv = np.convolve(a, a)
    return 2.0 * a.size * conv.max()


# ----------------------------------------------------------------------
# Sub‑gradient of the maximal convolution entry
def _subgradient_max_conv(a: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv(a,a)) w.r.t. a (sum(a)=1)."""
    n = a.size
    conv = np.convolve(a, a)
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)          # indices attaining the max
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    g /= len(ks)                                   # average over all maxima
    return g


# ----------------------------------------------------------------------
# Build the convolution matrix A such that (A @ b) = conv(a,b)
def _build_conv_matrix(a: np.ndarray) -> sp.csr_matrix:
    """Return a (2·n‑1)×n sparse matrix A with (A·b)=conv(a,b)."""
    n = a.shape[0]
    m = 2 * n - 1
    # column i contributes a[i] on rows i .. i+n‑1
    col_idx = np.repeat(np.arange(n), n)          # length n²
    row_idx = np.tile(np.arange(n), n) + col_idx
    data = np.repeat(a, n)
    A = sp.csr_matrix((data, (row_idx, col_idx)), shape=(m, n))
    return A


# ----------------------------------------------------------------------
# Solve the LP: maximise Σ b_j  subject to conv(a,b) ≤ M ,  b ≥ 0
def _solve_lp_max_sum(a: np.ndarray, M: float) -> np.ndarray | None:
    """
    Returns the optimal b (numpy array) for the LP.
    If the solver fails, returns None.
    """
    n = a.shape[0]
    m = 2 * n - 1

    # Build A once; each row i is (conv_i(a, b))
    A = _build_conv_matrix(a)

    # Objective: minimise -Σb  (i.e. maximise Σb)
    c = -np.ones(n, dtype=float)

    # Upper‐bound constraints: A @ b ≤ M
    ub = np.full(m, M, dtype=float)

    # Bounds on variables: b_j ≥ 0
    bounds = [(0.0, None)] * n

    res = opt.linprog(c,
                      A_ub=A,
                      b_ub=ub,
                      bounds=bounds,
                      method='highs',
                      options={'presolve': True})

    if res.success:
        return res.x
    # on failure fall back to a uniform vector (still feasible)
    return np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Cheap stochastic hill‑climbing / sub‑gradient phase
def _hill_climb(a: np.ndarray,
                deadline: float,
                max_iters: int = 4000,
                temp0: float = 0.02) -> np.ndarray:
    """Perform a short hill‑climbing search starting from a."""
    n = a.shape[0]
    cur_score = _objective(a)
    T = temp0
    it_without_improve = 0
    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # ---- move a small amount of mass i → j --------------------------------
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        if a[i] <= 0.0:
            continue
        delta = a[i] * np.random.rand() * 0.12            # ≤12 % of a[i]
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.clip(a_new, 0, None)
        a_new /= a_new.sum()

        new_score = _objective(a_new)

        # Metropolis acceptance (allows occasional uphill moves)
        if new_score < cur_score or np.random.rand() < math.exp((cur_score - new_score) / max(T, 1e-12)):
            a = a_new
            cur_score = new_score
            it_without_improve = 0
            T = max(T * 0.9995, 1e-5)                     # cool down slowly
        else:
            it_without_improve += 1

        # ---- sub‑gradient step (every few iterations) -------------------------
        if it % 7 == 0:
            g = _subgradient_max_conv(a)
            eta = 0.07 * (1.0 - it / max_iters)          # decreasing step size
            a = a - eta * g
            a = np.clip(a, 0, None)
            a /= a.sum()
            cur_score = _objective(a)

        if it_without_improve >= 2000:
            # stuck → abort early
            break
    return a


# ----------------------------------------------------------------------
# LP‑based refinement phase
def _lp_refine(a: np.ndarray,
               deadline: float,
               max_iter: int = 6,
               mix: float = 0.2) -> np.ndarray:
    """
    Repeatedly solve the LP “maximise Σb” with the current a,
    then mix a ← (1‑τ)a + τ·(b/Σb).  The mixing factor τ is fixed
    (default 0.2) – a small value keeps the monotonic decrease of the
    maximal convolution guaranteed by the theory.
    """
    n = a.shape[0]
    best_score = _objective(a)

    for it in range(max_iter):
        if time.time() >= deadline:
            break
        M = np.convolve(a, a).max()
        b = _solve_lp_max_sum(a, M)
        if b is None:
            break
        sum_b = b.sum()
        if sum_b <= 0:
            break
        b_norm = b / sum_b
        # convex combination
        a = (1.0 - mix) * a + mix * b_norm
        a = np.clip(a, 0, None)
        a /= a.sum()
        cur_score = _objective(a)
        if cur_score < best_score - 1e-9:
            best_score = cur_score
            # print progress of the refinement
            print(f"[LP-refine] it={it+1} O={best_score:.6f}")
        else:
            # no improvement → stop early
            break
    return a

        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:
        print("[final] evaluate_sequence call failed:", e)

    return best_seq.tolist()
