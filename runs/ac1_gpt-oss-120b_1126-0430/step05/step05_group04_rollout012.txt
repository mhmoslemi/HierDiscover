```python
"""
Exponentiated‑gradient optimiser for the step‑function attack.

The routine builds several candidate support sizes, initialises each
candidate from a random simplex point (or from a known construction), and
optimises the simplex coordinates using an exponentiated‑gradient/soft‑max
approximation of the L∞‑norm of the auto‑convolution.  The temperature
β of the smooth max grows from 0.5 to 2·10⁴, while the learning‑rate η
decays from ≈0.3 to 0.001.  After the gradient phase a short targeted
simulated‑annealing refinement is applied.  All work respects the
overall time budget (default 1000 s) and returns a plain list of non‑negative
floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full auto‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return conv_full[: 2 * n - 1]


def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation g_i = Σ_t w[i+t]·a[t] for i=0,…,n‑1.
    Implemented via FFT; ``w`` must have length 2·n‑1.
    """
    n = a.shape[0]
    m = w.shape[0]  # = 2n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


def _objective_exact(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.size
    conv = _conv_self_fft(a)
    return float(2.0 * n * conv.max())


def _run_eg(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
) -> tuple[list[float], float]:
    """
    Exponentiated‑gradient optimiser on the probability simplex.
    Returns the best sequence (as a list) and its exact objective value.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # ----- initialise a on the simplex ---------------------------------
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
        a = a / a.sum()
    else:
        a = rng.random(n)
        a = a / a.sum()

    best_a = a.copy()
    best_obj = _objective_exact(best_a)

    it = 0
    while time.time() - start < time_limit:
        it += 1

        # ----- evaluate current objective --------------------------------
        conv = _conv_self_fft(a)
        M = conv.max()
        obj = float(2.0 * n * M)
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a.copy()
            print(f"[EG n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        # ----- schedule --------------------------------------------------
        t = min((time.time() - start) / time_limit, 1.0)
        beta_start, beta_end = 0.5, 20000.0
        eta_start, eta_end = 0.30, 0.001
        beta = beta_start * np.exp(np.log(beta_end / beta_start) * t)
        eta = eta_start * np.exp(np.log(eta_end / eta_start) * t)

        # ----- smooth max weights ----------------------------------------
        if beta > 5000.0:
            w = np.zeros_like(conv)
            w[int(np.argmax(conv))] = 1.0
        else:
            shifted = conv - M                 # ≤ 0
            w = np.exp(beta * shifted)
            s = w.sum()
            if s == 0.0:
                w = np.ones_like(conv) / conv.size
            else:
                w /= s

        # ----- gradient of the smooth max (exact) -----------------------
        g = 2.0 * _correlation_fft(w, a)

        # ----- exponentiated‑gradient update -----------------------------
        upd = -eta * g
        # shift to avoid overflow in the exponent
        upd -= np.max(upd)
        factor = np.exp(upd)
        a = a * factor
        a_sum = a.sum()
        if a_sum <= 0.0:
            # fallback to a fresh random point
            a = rng.random(n)
            a = a / a.sum()
        else:
            a = a / a_sum

        # ----- occasional jitter / restart to escape plateaus ----------
        if it % 5000 == 0:
            a = np.clip(a + rng.normal(scale=0.001, size=n), 0.0, None)
            a = a / a.sum()

        if it % 80000 == 0:
            a = best_a.copy()
            a = np.clip(a + rng.normal(scale=0.005, size=n), 0.0, None)
            a = a / a.sum()

    # --------------------------------------------------------------------
    # post‑processing: prune negligible entries
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = _objective_exact(best_a)
        print(f"[compress] reduced length → {best_a.size}, O={best_obj:.6f}")

    # --------------------------------------------------------------------
    # short targeted local search (mass‑transfer moves) if time permits
    remaining = time_limit - (time.time() - start)
    if remaining > 0.5:
        loc_start = time.time()
        a_loc = best_a.copy()
        obj_loc = best_obj
        n_loc = a_loc.size
        while time.time() - loc_start < remaining * 0.2:
            i, j = rng.choice(n_loc, size=2, replace=False)
            delta = a_loc[i] * rng.random() * 0.02
            if delta <= 0.0:
                continue
            a_new = a_loc.copy()
            a_new[i] -= delta
            a_new[j] += delta
            conv_new = _conv_self_fft(a_new)
            obj_new = float(2.0 * n_loc * conv_new.max())
            if obj_new < obj_loc:
                a_loc, obj_loc = a_new, obj_new
                if obj_new < best_obj:
                    best_a, best_obj = a_new.copy(), obj_new
        print(f"[local] final local O={best_obj:.6f}")

    return best_a.tolist(), best_obj


def _run_sa(
    init_seq: list[float],
    time_limit: float,
    seed: int = 0,
) -> tuple[list[float], float]:
    """
    Targeted simulated‑annealing refinement.
    """
    start = time.time()
    rng = np.random.default_rng(seed)
    a = np.asarray(init_seq, dtype=float)
    if a.sum() < 1e-12:
        a = np.ones_like(a) / a.size
    else:
        a = a / a.sum()
    n = a.shape[0]

    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())
    best_seq = a.copy()

    # temperature schedule
    T = best_obj * 0.1 + 1e-6
    T_min = best_obj * 1e-6 + 1e-9
    decay = 0.995
    it = 0

    while time.time() - start < time_limit:
        it += 1
        k_max = int(np.argmax(conv))

        # contributions to the maximal entry
        i_vals = np.arange(n)
        j_vals = k_max - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i_vals[mask]] * a[j_vals[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            probs = contrib / total_contrib
            i = rng.choice(n, p=probs)
        else:
            i = rng.integers(n)

        # pick a different destination uniformly
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # transfer a modest fraction of mass
        delta = a[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.clip(a_new, 0.0, None)
        a_new /= a_new.sum()

        conv_new = _conv_self_fft(a_new)
        obj_new = float(2.0 * n * conv_new.max())

        # Metropolis acceptance
        if obj_new < best_obj - 1e-12:
            a, conv, best_obj, best_seq = a_new, conv_new, obj_new, a_new.copy()
            print(f"[SA it={it}] new best O={best_obj:.6f}")
        else:
            prob = np.exp(-(obj_new - best_obj) / T) if T > 0 else 0.0
            if rng.random() < prob:
                a, conv = a_new, conv_new

        # temperature decay
        if it % 2000 == 0:
            T = max(T * decay, T_min)

        # occasional global jitter
        if it % 10000 == 0:
            noise = rng.normal(scale=0.01, size=n)
            a = np.clip(a + noise, 0.0, None)
            a /= a.sum()

"""
Exponentiated‑gradient optimiser for the step‑function attack.

The routine builds several candidate support sizes, initialises each
candidate from a random simplex point (or from a known construction), and
optimises the simplex coordinates using an exponentiated‑gradient/soft‑max
approximation of the L∞‑norm of the auto‑convolution.  The temperature β
of the smooth max grows from 0.5 to 2·10⁴, while the learning‑rate η
decays from ≈0.3 to 0.001.  After the gradient phase a short targeted
simulated‑annealing refinement is applied.  All work respects the
overall time budget (default 1000 s) and returns a plain list of
non‑negative floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full auto‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return conv_full[: 2 * n - 1]


def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation g_i = Σ_t w[i+t]·a[t] for i=0,…,n‑1.
    Implemented via FFT; ``w`` must have length 2·n‑1.
    """
    n = a.shape[0]
    m = w.shape[0]               # = 2n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


def _objective_exact(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.size
    conv = _conv_self_fft(a)
    return float(2.0 * n * conv.max())


def _run_eg(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
) -> tuple[list[float], float]:
    """
    Exponentiated‑gradient optimiser on the probability simplex.
    Returns the best sequence (as a list) and its exact objective value.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # ----- initialise a on the simplex ---------------------------------
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
        a = a / a.sum()
    else:
        a = rng.random(n)
        a = a / a.sum()

    best_a = a.copy()
    best_obj = _objective_exact(best_a)

    it = 0
    while time.time() - start < time_limit:
        it += 1

        # ----- evaluate current objective --------------------------------
        conv = _conv_self_fft(a)
        M = conv.max()
        obj = float(2.0 * n * M)
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a.copy()
            print(f"[EG n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        # ----- schedule --------------------------------------------------
        t = min((time.time() - start) / time_limit, 1.0)
        beta_start, beta_end = 0.5, 20000.0
        eta_start, eta_end = 0.30, 0.001
        beta = beta_start * np.exp(np.log(beta_end / beta_start) * t)
        eta = eta_start * np.exp(np.log(eta_end / eta_start) * t)

        # ----- smooth‑max weights ----------------------------------------
        if beta > 5000.0:
            w = np.zeros_like(conv)
            w[int(np.argmax(conv))] = 1.0
        else:
            shifted = conv - M                 # ≤ 0
            w = np.exp(beta * shifted)
            s = w.sum()
            if s == 0.0:
                w = np.ones_like(conv) / conv.size
            else:
                w /= s

        # ----- gradient of the smooth max (exact) -----------------------
        g = 2.0 * _correlation_fft(w, a)

        # ----- exponentiated‑gradient update -----------------------------
        upd = -eta * g
        upd -= np.max(upd)               # shift to avoid overflow
        factor = np.exp(upd)
        a = a * factor
        a_sum = a.sum()
        if a_sum <= 0.0:
            a = rng.random(n)
            a = a / a.sum()
        else:
            a = a / a_sum

        # ----- occasional jitter / restart -------------------------------
        if it % 5000 == 0:
            a = np.clip(a + rng.normal(scale=0.001, size=n), 0.0, None)
            a = a / a.sum()
        if it % 80000 == 0:
            a = best_a.copy()
            a = np.clip(a + rng.normal(scale=0.005, size=n), 0.0, None)
            a = a / a.sum()

    # --------------------------------------------------------------------
    # post‑processing: prune negligible entries
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = _objective_exact(best_a)
        print(f"[compress] reduced length → {best_a.size}, O={best_obj:.6f}")

    # --------------------------------------------------------------------
    # short targeted local search (mass‑transfer moves) if time permits
    remaining = time_limit - (time.time() - start)
    if remaining > 0.5:
        loc_start = time.time()
        a_loc = best_a.copy()
        obj_loc = best_obj
        n_loc = a_loc.size
        while time.time() - loc_start < remaining * 0.2:
            i, j = rng.choice(n_loc, size=2, replace=False)
            delta = a_loc[i] * rng.random() * 0.02
            if delta <= 0.0:
                continue
            a_new = a_loc.copy()
            a_new[i] -= delta
            a_new[j] += delta
            conv_new = _conv_self_fft(a_new)
            obj_new = float(2.0 * n_loc * conv_new.max())
            if obj_new < obj_loc:
                a_loc, obj_loc = a_new, obj_new
                if obj_new < best_obj:
                    best_a, best_obj = a_new.copy(), obj_new
        print(f"[local] final local O={best_obj:.6f}")

    return best_a.tolist(), best_obj


def _run_sa(
    init_seq: list[float],
    time_limit: float,
    seed: int = 0,
) -> tuple[list[float], float]:
    """
    Targeted simulated‑annealing refinement.
    """
    start = time.time()
    rng = np.random.default_rng(seed)
    a = np.asarray(init_seq, dtype=float)
    if a.sum() < 1e-12:
        a = np.ones_like(a) / a.size
    else:
        a = a / a.sum()
    n = a.shape[0]

    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())
    best_seq = a.copy()

    # temperature schedule
    T = best_obj * 0.1 + 1e-6
    T_min = best_obj * 1e-6 + 1e-9
    decay = 0.995
    it = 0

    while time.time() - start < time_limit:
        it += 1
        k_max = int(np.argmax(conv))

        # contributions to the maximal entry
        i_vals = np.arange(n)
        j_vals = k_max - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i_vals[mask]] * a[j_vals[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            probs = contrib / total_contrib
            i = rng.choice(n, p=probs)
        else:
            i = rng.integers(n)

        # pick a different destination uniformly
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # transfer a modest fraction of mass
        delta = a[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.clip(a_new, 0.0, None)
        a_new /= a_new.sum()

        conv_new = _conv_self_fft(a_new)
        obj_new = float(2.0 * n * conv_new.max())

        # Metropolis acceptance
        if obj_new < best_obj - 1e-12:
            a, conv, best_obj, best_seq = a_new, conv_new, obj_new, a_new.copy()
            print(f"[SA it={it}] new best O={best_obj:.6f}")
        else:
            prob = np.exp(-(obj_new - best_obj) / T) if T > 0 else 0.0
            if rng.random() < prob:
                a, conv = a_new, conv_new

        # temperature decay
        if it % 2000 == 0:
            T = max(T * decay, T_min)

        # occasional global jitter
        if it % 10000 == 0:
            noise = rng.normal(scale=0.01, size=n)
            a = np.clip(a + noise, 0.0, None)
            a /= a.sum()
            conv = _conv_self_fft(a)

    # final compression of tiny entries
    thresh = 1e-12
    mask = best_seq > thresh
    if mask.sum() < best_seq.size:
        comp = best_seq[mask]
        comp = comp / comp.sum()
        comp_obj = float(2.0 * comp.size * _conv_self_fft(comp).max())
        if comp_obj < best_obj:
            best_seq, best_obj = comp, comp_obj
            print(f"[SA compress] reduced length → {comp.size}, O={best_obj:.6f}")

    return best_seq.tolist(), best_obj
```
