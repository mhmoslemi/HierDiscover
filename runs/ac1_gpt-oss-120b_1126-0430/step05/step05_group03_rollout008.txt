"""
Multi‑scale soft‑max Adam optimizer + simulated‑annealing finetune.

The search proceeds in stages:
1.  A few short Adam runs on modest lengths (256 → 4096) to obtain a
    coarse shape of the step‑function.
2.  The shape is up‑sampled to a larger length (≈ 8000) and a longer Adam
    optimisation is performed with a smoother annealing schedule and
    periodic jitter / restarts.
3.  The best candidate is refined by a simple simulated‑annealing
    “mass‑transfer” local search that can escape shallow plateaus.
All steps respect the security checks of ``evaluate_sequence`` and the
returned list is guaranteed non‑empty.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two greater than or equal to x."""
    return 1 << (x - 1).bit_length()


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv


def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation of weight vector w (length 2·n‑1) with a (length n):
    (w ⋆ a)[i] = Σ_t w[i+t]·a[t]   for 0 ≤ i < n.
    Implemented via FFT.
    """
    n = a.shape[0]
    m = w.shape[0]                 # = 2·n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


def _objective(a: np.ndarray) -> float:
    """Exact evaluation when ∑a = 1  →  2·n·max(conv(a,a))."""
    n = a.size
    conv = _conv_self_fft(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _run_opt(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
) -> tuple[list[float], float]:
    """
    Adam optimisation of a length‑`n` distribution for at most `time_limit`
    seconds.  The implementation follows the previous solution but with
    a smoother annealing schedule, more frequent jitter and restarts,
    and a longer default iteration budget.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # ---- initialisation -------------------------------------------------
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
        a = a / a.sum()
    else:
        a = rng.random(n)
        a = a / a.sum()

    # work in log‑it space (automatically respects the simplex)
    x = np.log(a + 1e-16)

    m = np.zeros_like(x)   # Adam first moment
    v = np.zeros_like(x)   # Adam second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # schedules
    beta_start = 1.0            # start with a fairly smooth max
    beta_end   = 15000.0        # end with a very sharp max
    lr_start   = 0.20
    lr_end     = 0.001
    target_iters = 1_200_000    # large theoretical budget; we stop by time

    best_a = a.copy()
    best_obj = _objective(best_a)

    it = 0
    while time.time() - start < time_limit:
        it += 1

        # probabilities from logits
        a = _softmax(x)

        # exact objective
        conv = _conv_self_fft(a)
        M = conv.max()
        obj = 2.0 * n * M
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        # ---- schedules -------------------------------------------------
        t_frac = min(it / target_iters, 1.0)
        beta = beta_start * np.exp(np.log(beta_end / beta_start) * t_frac)
        lr   = lr_start   * np.exp(np.log(lr_end   / lr_start)   * t_frac)

        # ---- smooth‑max weights ----------------------------------------
        if beta > 12000.0:                      # sharp enough → exact arg‑max
            kmax = int(np.argmax(conv))
            w = np.zeros_like(conv)
            w[kmax] = 1.0
        else:
            shifted = conv - M                # ≤ 0
            w = np.exp(beta * shifted)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(conv, 1.0 / conv.size)
            else:
                w = w / w_sum

        # gradient of the (smooth) max
        g_a = 2.0 * _correlation_fft(w, a)   # shape (n,)

        # natural gradient on the simplex (projected)
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---- jitter ----------------------------------------------------
        if it % 5000 == 0:
            jitter_scale = 0.03 * (1.0 - t_frac) + 0.005   # decay
            jitter = rng.normal(scale=jitter_scale, size=n)
            x += jitter

        # ---- periodic restart -----------------------------------------
        if it % 50000 == 0:
            x = np.log(best_a + 1e-16)
            jitter = rng.normal(scale=0.007, size=n)
            x += jitter
            m.fill(0.0)
            v.fill(0.0)
            print(f"[n={n}] restart at iter {it}")

    # ---- compress negligible entries ------------------------------------
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = _objective(best_a)
        print(f"[compress] reduced length {n} → {best_a.size}, O={best_obj:.6f}")

    # ---- short stochastic local search ---------------------------------
    remaining = time_limit - (time.time() - start)
    if remaining > 0.5:
        a_loc = best_a.copy()
        obj_loc = best_obj
        n_loc = a_loc.size
        local_deadline = time.time() + remaining * 0.35
        while time.time() < local_deadline:
            i, j = rng.choice(n_loc, size=2, replace=False)
            # transfer a random 0–3 % of mass from i to j
            delta = a_loc[i] * rng.random() * 0.03
            if delta <= 0.0:
                continue
            a_new = a_loc.copy()
            a_new[i] -= delta
            a_new[j] += delta
            conv_new = _conv_self_fft(a_new)
            obj_new = 2.0 * n_loc * conv_new.max()
            if obj_new < obj_loc:
                a_loc = a_new
                obj_loc = obj_new
                if obj_new < best_obj:
                    best_a = a_new
                    best_obj = obj_new
        print(f"[local] final local O={best_obj:.6f}")

    return best_a.tolist(), best_obj

    """

        seq0 = np.clip(seq0, 0.0, None)

    else:

        (256, 0.05),
        (512, 0.07),
        (1024, 0.10),

        (4096, 0.15),
        (8192, 0.50),   # bulk of the time on the largest scale
    ]

    best_obj = float("inf")

            x_old = np.linspace(0.0, 1.0, old_n)
            x_new = np.linspace(0.0, 1.0, n)
            init_a = np.interp(x_new, x_old, cur_seq, left=0.0, right=0.0)
            init_a = np.maximum(0.0, init_a)
            if init_a.sum() == 0.0:
                init_a = np.ones(n)
            init_a = init_a / init_a.sum()
        elif seq0 is not None and seq0.size == n:
            init_a = seq0.copy()
        else:
            # random start for the very first stage if nothing else matches
            rng = np.random.default_rng(seed + idx * 23)
            init_a = rng.random(n)
            init_a = init_a / init_a.sum()

        # ----- run Adam optimiser -----------------------------------------
        seq, val = _run_opt(
            n=n,
            time_limit=run_time,
            init_a=init_a,
            seed=seed + idx * 31,
        )
        if val < best_obj:
            best_obj = val
            best_seq = seq
            print(f"[global] new overall best O={best_obj:.6f} (n={n})")

        cur_seq = np.array(best_seq, dtype=float)

    # --------------------------------------------------------------
    # Final simulated‑annealing refinement on the best candidate.
    if best_seq is not None:
        remaining = deadline - time.time()
        if remaining > 1.0:
            a = np.array(best_seq, dtype=float)
            n = a.size
            best_obj = _objective(a)
            rng = np.random.default_rng(seed + 9999)
            local_end = time.time() + remaining * 0.35
            # Linear temperature schedule
            T0 = best_obj * 0.08 + 1e-6
            while time.time() < local_end:
                i, j = rng.choice(n, size=2, replace=False)
                if a[i] <= 0.0:
                    continue
                # transfer a random fraction (up to 8 %) of the source mass
                frac = rng.random() * 0.08
                delta = a[i] * frac
                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta
                obj_new = _objective(a_new)
                if obj_new < best_obj:
                    a = a_new
                    best_obj = obj_new
                    best_seq = a_new.tolist()
                    continue
                # probabilistic acceptance
                elapsed = time.time() - (deadline - remaining)
                progress = elapsed / (local_end - (deadline - remaining) + 1e-12)
                T = T0 * (1.0 - progress)
                if T < 1e-12:
                    T = 1e-12
                if rng.random() < np.exp(-(obj_new - best_obj) / T):
                    a = a_new
            print(f"[final SA] O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check via the provided evaluation routine.
    try:
        final_score = evaluate_sequence(best_seq)
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    # Guarantee a non‑empty return
    if not best_seq:
        best_seq = [1.0]

    return best_seq
