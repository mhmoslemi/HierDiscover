#!/usr/bin/env python
"""
Arc‑sine CDF‐based construction.

We build a step‑function whose heights are the exact bin–integrals of the
continuous arcsine density  

    f(x) = 1/(π·sqrt(x·(1‑x))) ,   x∈(0,1)

The CDF of this density is  
    F(x) = (2/π)·arcsin(√x) .
Hence the i‑th coefficient for a support of size n is  

    a_i = F((i+1)/n) – F(i/n) .

This discretisation makes the discrete self‑convolution a*a almost constant,
so the evaluation metric 2·n·max(a*a)/(∑a)² tends to the theoretical optimum 1.
We try several lengths (the length of any previously found sequence is also
examined) and keep the best candidate according to a cheap FFT‑based estimate.
The function respects the 1000 s time budget and returns a plain Python list
of non‑negative heights.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Helper utilities (no closures)
# --------------------------------------------------------------

def _fft_len_for_corr(n: int) -> int:
    """
    Smallest power‑of‑two ≥ 3·n‑2.
    Sufficient for a linear convolution of length (2·n‑1).
    """
    return 1 << ((3 * n - 2).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """
    Compute max entry of the linear self‑convolution a*a using FFT.
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


# --------------------------------------------------------------
# Main entry point required by the platform
# --------------------------------------------------------------

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that minimises the given
    evaluation metric.

    Strategy:
      1. Determine a set of candidate lengths (use the existing sequence
         length if available, plus several larger powers of two).
      2. For each length n, build the exact bin‑integral arcsine vector:
            a_i = (2/π)·[arcsin(√((i+1)/n)) – arcsin(√(i/n))]
         This guarantees non‑negativity, sum≈1 and entries ≤ 1000.
      3. Estimate the objective using a fast FFT convolution and keep the
         best candidate seen so far.
      4. Return the best sequence as a plain Python list.
    """
    np.random.seed(seed)
    start = time.time()
    # leave a small safety margin for the platform
    deadline = start + budget_s - 2.0

    # --------------------------------------------------------------
    # 0. Candidate support sizes
    # --------------------------------------------------------------
    candidate_lengths = [20000, 50000, 100000, 200000]  # default ladder
    try:
        # If a previous sequence exists, start from its size and explore larger ones
        base_n = len(height_sequence_1)  # type: ignore[name-defined]
        print(f"Found existing sequence of length {base_n}.")
        # add multiples of the existing length (2×,4×…) up to a reasonable bound
        mult = 1
        while mult <= 8:
            candidate_lengths.insert(0, base_n * mult)
            mult <<= 1
    except Exception:
        print("No external sequence found – using default length ladder.")

    # Ensure uniqueness and sort (small → large)
    candidate_lengths = sorted(set(candidate_lengths))

    best_score = float("inf")
    best_seq = None

    # --------------------------------------------------------------
    # 1. Scan candidates while time permits
    # --------------------------------------------------------------
    for n in candidate_lengths:
        if time.time() > deadline:
            print("Time budget exhausted – stopping candidate scan.")
            break

        # --- Build exact arcsine bin‑integrals --------------------
        i = np.arange(n, dtype=np.float64)
        left = i / n
        right = (i + 1) / n
        # CDF of arcsine: (2/π)·arcsin(√x)
        a = (2.0 / np.pi) * (np.arcsin(np.sqrt(right)) - np.arcsin(np.sqrt(left)))
        # Numerical safety – clip tiny negatives that can appear from round‑off
        a = np.clip(a, 0.0, None)

        # --- Cheap objective estimate (FFT) -----------------------
        # This is fast enough for the modest number of candidates.

                continue
            score = 2.0 * n * max_conv / (sum_a ** 2)
            print(f"n = {n:6d} | approx. score = {score:.6f}")
        except Exception as exc:
            # In the unlikely event FFT fails (e.g. memory), skip this n
            print(f"FFT error for n={n}: {exc}")
            continue

        # Keep the best candidate seen so far
        if score < best_score:
            best_score = score

            print(f"  --> New best (score={best_score:.6f})")

    # --------------------------------------------------------------

    # --------------------------------------------------------------

    final_seq = best_seq.tolist()
    # Log the final choice
