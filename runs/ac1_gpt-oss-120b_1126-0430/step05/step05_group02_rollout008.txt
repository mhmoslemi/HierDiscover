#!/usr/bin/env python
"""
Propose a non‑negative step function (list of heights) that minimises

    2·n·max_k (a*a)[k] / (∑ a_i)²

The algorithm works in three stages:

1️⃣  A cheap parametric sweep (power‑law / exponential / beta families)
    to obtain a good deterministic starting point.

2️⃣  A smooth‑max mirror‑descent optimisation.
    * The true objective  max(conv) is approximated by a soft‑max
      (log‑sum‑exp) with a temperature that is gradually lowered.
    * The gradient of the smooth objective is obtained by a single FFT‑based
      convolution, which gives the exact sub‑gradient of the max when the
      temperature → ∞.
    * Updates are performed with an exponentiated‑gradient (KL‑mirror)
      step and a projection onto the probability simplex.

3️⃣  Final exact evaluation with the supplied ``evaluate_sequence``.
    The best sequence seen during the whole run is returned.

The routine respects a user supplied time budget (default 1000 s) and
always returns a feasible sequence before the deadline.
"""

# ----------------------------------------------------------------------
# Imports
# ----------------------------------------------------------------------
import time
import math
import numpy as np

# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, Σ x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                      # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float) -> np.ndarray:
    """Return a normalised power‑law a_i ∝ (i+1)⁻ᵖ (∑a_i = 1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = i ** (-p)
    seq /= seq.sum()
    return seq


def _exponential_sequence(n: int, lam: float) -> np.ndarray:
    """Return a normalised exponential a_i ∝ exp(-lam·i) (∑a_i = 1)."""
    i = np.arange(n, dtype=np.float64)
    seq = np.exp(-lam * i)
    seq /= seq.sum()
    return seq


def _beta_family_sequence(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑shaped distribution a_i ∝ x_i^α (1‑x_i)^β, x_i = (i+1)/n."""
    x = (np.arange(1, n + 1, dtype=np.float64) / n)
    seq = (x ** alpha) * ((1.0 - x) ** beta)
    s = seq.sum()
    if s == 0.0:
        return None
    return seq / s


def _cheap_score(a: np.ndarray, conv: np.ndarray, n: int) -> float:
    """Cheap objective when ∑a = 1 : 2·n·max(conv)."""
    return 2.0 * n * conv.max()

        # ---- occasional restart if stagnation --------------------------------
        if iter_cnt - last_improve > max_no_improve:
            perturb = 0.01 * np.random.rand(n)
            a = _simplex_projection(best_iter_a + perturb)
            last_improve = iter_cnt
            print(f"[Iter {iter_cnt:07d}] Restart after stagnation.")

        # ---- temperature schedule ------------------------------------------------
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

    # ----------------------------------------------------------------------
    # 2. Final exact evaluation (uses the provided ``evaluate_sequence``)
    # ----------------------------------------------------------------------
    final_seq = best_iter_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[Result] Final exact score = {final_score:.6f}")
    except Exception as exc:      # defensive – should never happen
        final_score = float("inf")
        print(f"[Error] final evaluation failed: {exc}")

    # ----------------------------------------------------------------------
    # 3. Return the best sequence found (the one with the lowest exact score)
    # ----------------------------------------------------------------------
    # In practice the exact score equals the cheap score because the simplex
    # normalisation enforces ∑a = 1.
    return final_seq
```
