assistantfinal**Solution Overview**

The function `propose_candidate` searches for a non‑negative step‑function (a list of heights) that minimises  

\[
O(a)=\frac{2\,n\;\max(a\star a)}{(\sum a)^2}.
\]

`O` is invariant under scaling, therefore we may fix the sum of the
sequence to `1`.  The problem then reduces to minimising the maximal entry
of the convolution `c = a★a`.  

The algorithm consists of three layers:

1. **Multi‑start projected Adam.**  
   Several sensible initial shapes (uniform, triangular, cosine‑shaped,
   random, and the previously known best sequence) are created.
   For each start we run two Adam phases:  
   *variance flattening* (minimise the variance of `c`) and a
   *soft‑max* phase (minimise a smooth approximation of `max(c)` using a
   log‑sum‑exp surrogate).  After each iteration the point is projected
   onto the probability simplex (`∑a=1, a_i≥0`).

2. **Sub‑gradient descent on the true max.**  
   Once the surrogate has been removed we directly minimise the true
   objective `max(c)` using its sub‑gradient (the set of indices that
   realise the current maximum).  A simple decreasing step‑size together
   with simplex projection is used.

3. **Guided stochastic refinement.**  
   Finally a simulated‑annealing style local search is performed.  Mass
   is transferred from indices with a large sub‑gradient to indices with a
   small sub‑gradient; moves are accepted with a Metropolis rule that
   slowly cools down.  This phase aggressively removes the last remaining
   peaks.

All phases share the same time budget (≈ 1000 s).  The best sequence found
by any phase is returned.

The code below is completely self‑contained; it only relies on `numpy`,
`math`, and `time`.  The helper functions are defined at top level so
that the judge can import them without closure‑related restrictions.

```python
"""
Multi‑start projected Adam + sub‑gradient + simulated‑annealing search
for a non‑negative step‑function that minimises
    O(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm works on the probability simplex (∑a = 1) because O is
scale‑invariant.  It proceeds in three layers:
  1. several random / deterministic initialisations,
     each optimised by Adam on a variance flattening objective,
     then on a smooth log‑sum‑exp approximation of max(conv).
  2. a pure sub‑gradient descent on the exact max(conv).
  3. a guided stochastic mass‑transfer refinement (simulated annealing).

All operations use FFT‑based convolution (O(n log n)) and simplex
projection.  The best sequence (as a plain list) is returned.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# 1.  Simplex projection  (Euclidean projection onto {x ≥ 0, ∑x = 1})
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        # all entries non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# 2.  FFT‑based convolution (full linear convolution a★a)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full convolution of a with itself using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# 3.  Objective on the simplex (sum(a) == 1)
def _objective_simplex(a: np.ndarray, n: int, L: int) -> float:
    """Exact O(a) value when ∑a = 1  →  2·n·max(conv)."""
    conv = _conv_fft(a, L)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# 4.  Gradient of the variance‑of‑convolution objective
def _grad_variance(a: np.ndarray, conv: np.ndarray, n: int, L: int) -> np.ndarray:
    """
    Gradient of  f(a)=∑_k (conv_k-μ)^2,
    where μ = 1/(2·n‑1) for a simplex‑normalised a.
    Closed form:  g = 4·corr(conv‑μ , a).
    """
    mu = 1.0 / (2 * n - 1)
    diff = conv - mu
    a_rev = a[::-1]
    fft_diff = np.fft.rfft(diff, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_diff * fft_a_rev, n=L)
    start = n - 1
    grad = 4.0 * corr_rev[start:start + n]
    return grad


# ----------------------------------------------------------------------
# 5.  Gradient of the smooth (log‑sum‑exp) surrogate
def _smooth_gradient(a: np.ndarray, conv: np.ndarray, alpha: float, L: int,
                    lam: float = 0.0) -> np.ndarray:
    """
    ∇_a  ( (2·n/α)·log Σ_k exp(α·conv_k) + λ·∑ a_i log a_i )
    The first term yields 2·corr(w,a) with w_i ∝ exp(α·conv_i).
    """
    # soft‑max weights for the surrogate
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)          # shift for stability
    w = exp_scaled / exp_scaled.sum()                # shape (2n‑1,)

    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)

    n_len = a.shape[0]
    grad_corr = 2.0 * corr_rev[n_len - 1:n_len - 1 + n_len]

    if lam > 0.0:
        eps = 1e-20
        grad_ent = lam * (np.log(a + eps) + 1.0)
        return grad_corr + grad_ent
    else:
        return grad_corr


# ----------------------------------------------------------------------
# 6.  Sub‑gradient of the exact max‑convolution objective
def _subgradient_max(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of  max(conv)  w.r.t. a.
    For each k attaining the maximum, all pairs (i,j) with i+j=k contribute
    2·a_j to the gradient w.r.t. a_i.
    The result is averaged over all maximising k.
    """
    n = a.shape[0]
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)
    if ks.size == 0:
        return np.zeros_like(a)
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    g /= ks.size
    return g

# 7.  Core optimisation routine for a single initialisation
def _optimise_one_start(a_init: np.ndarray, n: int,
                        L: int, deadline: float,
                        seed: int) -> tuple[np.ndarray, float]:
    """
    Runs the three optimisation layers on a single starting point.
    Returns the best sequence and its O‑value.
    """
    rng = np.random.default_rng(seed)
    a = a_init.copy()
    a = _project_to_simplex(a)                     # ensure simplex

    best_seq = a.copy()
    best_score = _objective_simplex(best_seq, n, L)

    # ------------------------------------------------------------------
    # Phase A – variance flattening (Adam)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    lr0 = 0.7
    lam_entropy = 1e-4
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    max_iter_a = 150_000
    while time.time() < deadline and it < max_iter_a:
        conv = _conv_fft(a, L)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()

        grad = _grad_variance(a, conv, n, L)
        # tiny entropy term keeps strict positivity and improves numerics
        grad += lam_entropy * (np.log(a + 1e-20) + 1.0)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        lr = lr0 / math.sqrt(it + 1)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_to_simplex(a_new)

        it += 1
        if it % 30_000 == 0:
            # occasional status line
            print(f"[phase‑A] it={it:,}  O={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Phase B – smooth max (log‑sum‑exp) – Adam with annealing α
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.5
    lam = 2e-4                     # tiny entropy regulariser
    alpha_start, alpha_end = 8.0, 20000.0
    max_iter_b = 150_000
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    while time.time() < deadline and it < max_iter_b:
        conv = _conv_fft(a, L)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()

        frac = it / max_iter_b
        alpha = alpha_start + (alpha_end - alpha_start) * frac

        grad = _smooth_gradient(a, conv, alpha, L, lam)

        # Adam step
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        lr = lr0 / math.sqrt(it + 1)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape plateaus
        if rng.random() < 0.001:
            a_new += rng.standard_normal(n) * 1e-4

        a = _project_to_simplex(a_new)

        it += 1
        if it % 30_000 == 0:
            print(f"[phase‑B] it={it:,} α={alpha:.1f}  O={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Phase C – pure sub‑gradient descent on max(conv)
    eta0 = 0.08
    max_iter_c = 120_000
    it = 0
    while time.time() < deadline and it < max_iter_c:
        conv = _conv_fft(a, L)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()

        g = _subgradient_max(a, conv)

        # Decaying step size
        eta = eta0 * (1.0 - it / max_iter_c)
        a_new = a - eta * g
        a = _project_to_simplex(a_new)

        it += 1
        if it % 30_000 == 0:
            print(f"[phase‑C] it={it:,}  O={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Phase D – guided stochastic refinement (simulated annealing)
    T0 = 0.06
    T = T0
    it_loc = 0
    max_iter_d = 200_000
    while time.time() < deadline and it_loc < max_iter_d:
        # compute current state
        conv = _conv_fft(a, L)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()

        # sub‑gradient for guidance
        g = _subgradient_max(a, conv)

        # pick a source (large g) and a sink (small g)
        src_idx = np.argmax(g)
        sink_idx = np.argmin(g)
        if src_idx != sink_idx and a[src_idx] > 1e-14:
            delta = a[src_idx] * rng.random() * 0.12   # up to 12 % of source mass
            a_new = a.copy()
            a_new[src_idx] -= delta
            a_new[sink_idx] += delta
            a_new = _project_to_simplex(a_new)
            conv_new = _conv_fft(a_new, L)
            new_score = 2.0 * n * conv_new.max()
            # Metropolis acceptance
            if new_score < cur_score or rng.random() < math.exp((cur_score - new_score) / max(T, 1e-12)):
                a = a_new
                cur_score = new_score

        # occasional pure random “shake”
        if it_loc % 7_000 == 0:
            jitter = rng.standard_normal(n) * 1e-5
            a = _project_to_simplex(a + jitter)

        # temperature decay
        T *= 0.9999
        it_loc += 1

    return best_seq, best_score

    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence failed:", exc)

    # Return plain Python list as required
    return best_global_seq.tolist()
```
