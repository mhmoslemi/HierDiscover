def propose_candidate(seed=42, budget_s=1000, **_ignore):
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5
    # Determine max_n = 12000 (or maybe 20000)
    max_n = 12000
    lengths = [2000, 4000, 8000, max_n]
    best_seq, best_obj = None, float('inf')
    # Use existing height_sequence_1 for init
    init_seq = None
    if 'height_sequence_1' in globals():
        seq = np.array(height_sequence_1, dtype=float)
        seq = np.clip(seq, 0, None)
        if seq.sum() < 0.01: seq = np.ones_like(seq)
        seq = seq / seq.sum()
        init_seq = seq

    for i,n in enumerate(lengths):
        now = time.time()
        remaining = deadline - now
        if remaining <= 2:
            break
        # allocate time: if first run, give some budget; else remainder split equally
        if i == 0:
            t_alloc = min(remaining * 0.2, 30) # ensure at least short warmup
        else:
            t_alloc = remaining / (len(lengths)-i)
        # prepare init:
        if i == 0:
            init_a = init_seq
        else:
            # upscale best_seq to length n via interpolation
            if best_seq is not None:
                init_a = np.interp(np.linspace(0, best_seq.size-1, n),
                                   np.arange(best_seq.size),
                                   best_seq)
                init_a = init_a / init_a.sum()
            else:
                init_a = None

        seq, val = _run_opt(n=n, time_limit=t_alloc, init_a=init_a,
                            seed=seed + i*101)
        if val < best_obj:
            best_obj, best_seq = val, np.asarray(seq, dtype=float)
            print(f"[global] new best O={best_obj:.6f} length {n}")
        # if time low break
        if time.time() >= deadline-0.5:
            break

    remaining = deadline - time.time()
    if best_seq is not None and remaining > 5:
        seq_refined, val_ref = _run_sa(best_seq.tolist(), time_limit=remaining,
                                      seed=seed+999)

"""
step‑function optimiser – hierarchical up‑scaling + Adam on a soft‑max simplex.

The optimiser works in three stages:

1.  A short warm‑up on a modest support size (≈2 k).  This quickly
    discovers a useful shape of the distribution.

2.  The distribution is linearly interpolated to larger supports
    (4 k → 8 k → 16 k).  Each size receives the remaining time budget and
    is refined with a soft‑max Adam optimiser that uses a smooth
    approximation of the max‑convolution (exponential‑beta annealing).
    The optimiser works in log‑it space (mirrored natural gradient) and
    adds a tiny ℓ₂‑penalty on the logits (weight‑decay) to keep the
    vector spread out.

3.  The best candidate is finally polished by a targeted simulated‑annealing
    local search that moves a small fraction of mass away from the most
    contributing indices of the current maximal convolution entry.

The routine respects the overall 1000 s budget, automatically adapts the
time allocation to the remaining budget and returns a plain list of
non‑negative floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return conv_full[: 2 * n - 1]


def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation g_i = Σ_t w[i+t]·a[t]   for i=0,…,n‑1.
    Implemented via FFT.
    """
    n = a.shape[0]
    m = w.shape[0]                     # = 2n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


def _interpolate_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    # map both old and new indices to [0,1] and interpolate
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(new_x, old_x, seq)
    # numerical safety
    new_seq = np.maximum(new_seq, 0.0)
    if new_seq.sum() == 0.0:
        new_seq = np.ones_like(new_seq) / new_seq.size
    else:
        new_seq = new_seq / new_seq.sum()
    return new_seq


def _run_opt(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
) -> tuple[list[float], float]:
    """
    Adam optimisation on the log‑it simplex for a fixed support size ``n``.
    Returns the best sequence (as a list) and its exact objective value.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # ----- initialise a on the simplex -----
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
        a = a / a.sum()
    else:
        a = rng.random(n)
        a = a / a.sum()

    # work in log‑it space (mirrored natural gradient)
    x = np.log(a + 1e-16)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # annealing / learning‑rate schedule (time‑based)
    beta_start = 0.5
    beta_end = 20000.0          # sharp smoothing at the end
    lr_start = 0.30
    lr_end = 0.0005
    weight_decay = 1e-3         # ℓ₂‑penalty on logits

    best_a = a.copy()
    best_obj = float('inf')
    it = 0

    while time.time() - start < time_limit:
        it += 1
        a = _softmax(x)                     # stays on the simplex

        conv = _conv_self_fft(a)
        M = conv.max()
        obj = float(2.0 * n * M)            # sum(a)=1 ⇒ denominator = 1
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        # ----- schedule -----
        t = min((time.time() - start) / time_limit, 1.0)
        beta = beta_start * np.exp(np.log(beta_end / beta_start) * t)
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t)

        # ----- smooth max weights -----
        if beta > 5000.0:
            w = np.zeros_like(conv)
            w[int(np.argmax(conv))] = 1.0
        else:
            shifted = conv - M                # ≤ 0
            w = np.exp(beta * shifted)
            s = w.sum()
            if s == 0.0:
                w = np.ones_like(conv) / conv.size
            else:
                w /= s

        # ----- gradient w.r.t. a (exact) -----
        g_a = 2.0 * _correlation_fft(w, a)

        # ----- natural (mirrored) gradient in log‑it space -----
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)

        # weight decay on logits (keeps distribution spread)
        g_x += weight_decay * x

        # ----- Adam update -----

        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)

        v_hat = v / (1.0 - beta2 ** it)

        # occasional jitter to escape plateaus
        if it % 2000 == 0:
            jitter = rng.normal(scale=0.02, size=n)
            x += jitter

        # periodic restart from current best (helps avoid stagnation)
        if it % 60000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.005, size=n)
            m.fill(0.0)
            v.fill(0.0)

    # -----------------------------------------------------------------
    # compress negligible entries (helps the final O)
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]

        best_obj = float(2.0 * best_a.size * _conv_self_fft(best_a).max())
        print(f"[compress] reduced length → {best_a.size}, O={best_obj:.6f}")

    # -----------------------------------------------------------------
    # very short greedy local search (mass‑transfer moves) if time permits
    remaining = time_limit - (time.time() - start)

            if delta <= 0.0:

            a_new[i] -= delta
            a_new[j] += delta
            conv_new = _conv_self_fft(a_new)
            obj_new = float(2.0 * n_loc * conv_new.max())
            if obj_new < obj_loc:

                    best_a, best_obj = a_new, obj_new
        print(f"[local] final O={best_obj:.6f}")

    return best_a.tolist(), best_obj


def _run_sa(
    init_seq: list[float],
    time_limit: float,
    seed: int = 0,
) -> tuple[list[float], float]:
    """
    Targeted simulated‑annealing refinement.
    Small mass transfers are guided by the current maximal convolution entry.
    """
    start = time.time()
    rng = np.random.default_rng(seed)
    a = np.asarray(init_seq, dtype=float)
    if a.sum() < 1e-12:
        a = np.ones_like(a) / a.size
    else:
        a = a / a.sum()
    n = a.shape[0]

    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())
    best_seq = a.copy()

    # temperature schedule (Metropolis)
    T = best_obj * 0.1 + 1e-6
    T_min = best_obj * 1e-6 + 1e-9
    decay = 0.995
    it = 0

    while time.time() - start < time_limit:
        it += 1
        k_max = int(np.argmax(conv))

        # contributions to the maximal entry
        i_vals = np.arange(n)
        j_vals = k_max - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i_vals[mask]] * a[j_vals[mask]]
        total = contrib.sum()
        if total > 0.0:
            probs = contrib / total
            i = rng.choice(n, p=probs)
        else:
            i = rng.integers(n)

        # pick a distinct destination uniformly
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # transfer up to 5 % of the donor mass
        delta = a[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.clip(a_new, 0.0, None)
        a_new /= a_new.sum()

        conv_new = _conv_self_fft(a_new)
        obj_new = float(2.0 * n * conv_new.max())

        # accept if improvement, otherwise Metropolis
        if obj_new < best_obj - 1e-12:
            a, conv, best_obj, best_seq = a_new, conv_new, obj_new, a_new.copy()
            print(f"[SA it={it}] new best O={best_obj:.6f}")
        else:
            prob = np.exp(-(obj_new - best_obj) / T) if T > 0 else 0.0
            if rng.random() < prob:
                a, conv = a_new, conv_new

        # temperature decay
        if it % 2000 == 0:
            T = max(T * decay, T_min)

        # occasional global jitter
        if it % 10000 == 0:
            noise = rng.normal(scale=0.01, size=n)
            a = np.clip(a + noise, 0.0, None)
            a /= a.sum()
            conv = _conv_self_fft(a)

    # final compression of tiny entries
    thresh = 1e-12
    mask = best_seq > thresh
    if mask.sum() < best_seq.size:
        comp = best_seq[mask]
        comp = comp / comp.sum()
        comp_obj = float(2.0 * comp.size * _conv_self_fft(comp).max())
        if comp_obj < best_obj:
            best_seq, best_obj = comp, comp_obj
            print(f"[SA compress] reduced length → {comp.size}, O={best_obj:.6f}")

    return best_seq.tolist(), best_obj


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **_ignore,
) -> list[float]:
    """
    Entry point for the competition harness.

    The routine builds a sequence of non‑negative heights that (hopefully)
    improves upon the previous best bound.  It uses a hierarchical
    up‑scaling scheme (2 k → 4 k → 8 k → 16 k) together with an Adam optimiser
    on a smooth approximation of the max‑convolution, followed by a targeted
    simulated‑annealing polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # -----------------------------------------------------------------
    # Initialise from a previously found construction if available.
    init_seq = None
    if 'height_sequence_1' in globals():
        seq0 = np.asarray(height_sequence_1, dtype=float)
        seq0 = np.clip(seq0, 0.0, None)
        if seq0.sum() < 0.01:
            seq0 = np.ones_like(seq0)
        seq0 = seq0 / seq0.sum()
        init_seq = seq0

    # -----------------------------------------------------------------
    # Support sizes – we start modest and finish with a large one.
    # The last size is limited to 16 000 to keep per‑iteration cost reasonable.
    lengths = [2000, 4000, 8000, 16000]
    best_seq = None
    best_obj = float('inf')

    for idx, n in enumerate(lengths):
        now = time.time()
        remaining = deadline - now
        if remaining <= 2.0:
            break

        # allocate time: first stage gets a small fixed share,
        # the remaining budget is split equally among the later stages.
        if idx == 0:
            run_time = min(remaining * 0.1, 30.0)
        else:
            # split the rest uniformly
            run_time = (remaining * 0.9) / (len(lengths) - idx)

        # warm‑start: either the global init or an interpolated version of the
        # best sequence obtained so far.
        if idx == 0 and init_seq is not None and init_seq.size == n:
            init_a = init_seq
        elif best_seq is not None:
            init_a = _interpolate_sequence(best_seq, n)
        else:
            init_a = None

        seq, val = _run_opt(
            n=n,
            time_limit=run_time,
            init_a=init_a,
            seed=seed + idx * 12345,
        )
        if val < best_obj:
            best_obj, best_seq = val, np.asarray(seq, dtype=float)
            print(f"[global] new overall best O={best_obj:.6f} (n={n})")

        if time.time() >= deadline - 0.5:
            break

    # -----------------------------------------------------------------
    # Final refinement with simulated annealing if time permits.
    remaining = deadline - time.time()
    if best_seq is not None and remaining > 5.0:
        seq_refined, val_refined = _run_sa(
            best_seq.tolist(),
            time_limit=remaining,
            seed=seed + 9999,
        )
        if val_refined < best_obj:
            best_obj, best_seq = val_refined, np.asarray(seq_refined, dtype=float)
            print(f"[final] after SA new best O={best_obj:.6f}")

    # -----------------------------------------------------------------
    # Fallback – should never happen but guards against an empty result.
    if best_seq is None or best_seq.size == 0:
        best_seq = np.ones(100, dtype=float) / 100.0

    # -----------------------------------------------------------------
    # Final sanity check / reporting.
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_seq.tolist()
