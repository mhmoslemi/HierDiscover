def propose_candidate(seed=42, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0

    # Determine n
    try:
        prev = height_sequence_1
        n0 = len(prev)
        n = min(max(n0, 8000), 20000)
    except Exception:
        n = 12000

    # define FFT length
    L = 1 << ((3 * n - 2).bit_length())

    # cheap_score function using single fft of length L (since conv length smaller)
    def cheap_score(vec):
        fa = np.fft.fft(vec, L)
        conv = np.fft.ifft(fa * fa).real[:2*n-1]
        return 2.0 * n * conv.max()

    # scanning families
    best_seq = None
    best_val = float('inf')
    # uniform
    uniform = np.full(n, 1.0/n)
    val = cheap_score(uniform)
    best_seq, best_val = uniform, val

    # previous if same n (if length diff, maybe resample to n)
    try:
        prev_arr = np.asarray(prev, dtype=np.float64)
        if prev_arr.sum()>0:
            # resample (maybe linear interpolation) to length n
            # Use np.interp
            x_old = np.linspace(0,1,prev_arr.size)
            x_new = np.linspace(0,1,n)
            prev_resampled = np.interp(x_new, x_old, prev_arr)
            prev_resampled = np.maximum(prev_resampled,0)
            total = prev_resampled.sum()
            if total>0:
                prev_resampled /= total
                val = cheap_score(prev_resampled)
                if val < best_val:
                    best_seq, best_val = prev_resampled, val
    except Exception:
        pass

    # power-law
    for p in np.linspace(0.3, 1.5, 25):
        cand = _power_law_sequence(n, p)
        val = cheap_score(cand)
        if val < best_val:
            best_seq,best_val = cand,val

    # exponential
    for lam in np.linspace(0.1, 4.0, 30):
        idx = np.arange(n, dtype=np.float64)
        cand = np.exp(-lam*idx)
        cand /= cand.sum()
        val = cheap_score(cand)
        if val < best_val:
            best_seq,best_val = cand,val

    for p in np.linspace(0.4, 1.2, 11):

            cand = (idx+1)**(-p) * np.exp(-lam*idx)
            cand_sum = cand.sum()
            if cand_sum>0:
                cand /= cand_sum
                val = cheap_score(cand)
                if val < best_val:
                    best_seq,best_val = cand,val

    # beta shape (positive)

#!/usr/bin/env python
"""
Propose a non‑negative step function (list of heights) that minimises

        2·n·max_k (a*a)[k] / (Σ a_i)²

The method consists of three phases:

1️⃣  A quick scan of a handful of parametric families (uniform,
    power‑law, exponential, power‑law × exponential, beta‑shaped,
    “negative‑beta” and a discrete arcsine shape) to obtain a good
    deterministic start.

2️⃣  A stochastic mirror‑descent (exponentiated‑gradient) refinement.
    The sub‑gradient is obtained from a **soft‑max** over all
    autocorrelation entries – this uses the full information rather
    than the top‑k heuristic of the earlier solution.
    The temperature (α) of the soft‑max is slowly increased so the
    algorithm first follows a smooth direction and later focuses on the
    true maximum.

3️⃣  A final exact call to ``evaluate_sequence`` and return the best
    sequence seen.

The routine respects the supplied time budget (default 1000 s) and
always returns a feasible (non‑negative, normalised) sequence.  It
operates on a single CPU but can be run on up to two cores without
modification.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{‑p}  (Σ a_i = 1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = i ** (-p)
    seq /= seq.sum()
    return seq


def _exponential_sequence(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(‑lam·i)  (Σ a_i = 1)."""
    idx = np.arange(n, dtype=np.float64)
    seq = np.exp(-lam * idx)
    seq /= seq.sum()
    return seq


def _beta_sequence(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ x_i^{α} (1‑x_i)^{β}  where x_i = (i+1)/n  (Σ a_i = 1)."""
    x = (np.arange(1, n + 1, dtype=np.float64) / n)
    seq = (x ** alpha) * ((1.0 - x) ** beta)
    s = seq.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    seq /= s
    return seq


def _neg_beta_sequence(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    a_i ∝ (i+1)^{α}·(n‑i)^{β},  with α,β possibly negative.
    (Σ a_i = 1).  Small epsilon avoids division by zero for extreme exponents.
    """
    i = np.arange(n, dtype=np.float64)
    seq = (i + 1.0) ** alpha * (n - i) ** beta
    seq = np.maximum(seq, 0.0)
    s = seq.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    seq /= s
    return seq


def _arcsine_sequence(n: int) -> np.ndarray:
    """Discrete version of the arcsine law: a_i ∝ 1/√(x·(1‑x))."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    seq = 1.0 / np.sqrt(x * (1.0 - x))
    seq /= seq.sum()
    return seq


def _conv_full_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution a*a using FFT of length L (returns length 2·n‑1)."""
    fa = np.fft.fft(a, L)
    conv = np.fft.ifft(fa * fa).real[: 2 * a.size - 1]
    return conv


def _grad_softmax(a: np.ndarray, w: np.ndarray, L: int) -> np.ndarray:
    """
    Compute the gradient of  max_k (a*a)[k] approximated by a soft‑max
    with weights w (Σ w = 1).  The result is the vector g where

        g_i = 2·∑_k w_k·a_{k‑i}   (out‑of‑range a’s are treated as 0).

    The computation uses FFTs of length L ≥ 3·n‑2.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, L)
    fw = np.fft.fft(w, L)
    grad_full = np.fft.ifft(fa_rev * fw).real
    # slice that corresponds to indices i = 0 … n‑1
    g = 2.0 * grad_full[n - 1:n - 1 + n]
    return g

# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that
    minimizes ``evaluate_sequence``.  The routine adapts its schedule to
    the available time budget and always returns a feasible sequence.
    """
    np.random.seed(seed)
    start_time = time.time()
    # reserve a few seconds for the final exact evaluation
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # Choose a working length.
    # --------------------------------------------------------------
    try:
        prev = height_sequence_1          # provided globally, may not exist
        n0 = len(prev)
        n = max(n0, 8000)                 # not too small, not too huge
    except Exception:
        n = 12000
    n = min(max(n, 5000), 20000)          # 5 k ≤ n ≤ 20 k

    L = 1 << ((3 * n - 2).bit_length())

    # --------------------------------------------------------------
    # Cheap score (assumes Σa_i = 1): 2·n·max(conv)
    # --------------------------------------------------------------
    def cheap_score(vec: np.ndarray) -> float:
        conv = _conv_full_fft(vec, L)
        return 2.0 * n * conv.max()

    # --------------------------------------------------------------
    # 0. Initialise with cheap families
    # --------------------------------------------------------------
    best_a = None
    best_val = float("inf")

    val = cheap_score(cand)
    if val < best_val:

        val = cheap_score(cand)

    # hyper‑parameters
    eta0 = 0.4               # base step size for exponentiated‑gradient
    alpha = 0.5              # start temperature for soft‑max

        conv = _conv_full_fft(a, L)                     # length 2·n‑1
        maxb = conv.max()
        cheap = 2.0 * n * maxb

    # --------------------------------------------------------------
    return final_seq
