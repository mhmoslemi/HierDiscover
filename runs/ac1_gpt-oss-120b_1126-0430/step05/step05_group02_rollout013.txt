#!/usr/bin/env python
"""
Propose a non‑negative step function (list of heights) that minimises
``evaluate_sequence``.

Algorithm
---------
1. Choose a working length `n` (typically 8‑20 k).  
2. Scan a few parametric families (exponential, power‑law×exponential,
   beta‑shaped) with a coarse grid, then perform a quick random refinement
   around the best parameters. The cheapest candidate (cheap score =
   ``2·n·max(a*a)``) is kept.
3. Refine the best candidate with **Adam**‑style mirror‑descent
   (updates are performed in the log‑domain, i.e. exponentiated‑gradient)
   using the sub‑gradient of the active convolution entry
   (only the index attaining the maximum). The iterate is always
   re‑normalised to the probability simplex.
4. If progress stalls, re‑initialise the iterate with a small random
   perturbation of the best point seen so far.
5. Finally evaluate the best sequence with the supplied
   ``evaluate_sequence`` and return it.

The routine respects the supplied time budget (default 1000 s) and always
returns a feasible sequence.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project `v` onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa).real[: 2 * a.size - 1]
    return conv


def _exponential_sequence(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i), normalised to sum 1."""
    idx = np.arange(n, dtype=np.float64)
    seq = np.exp(-lam * idx)
    s = seq.sum()
    return seq / s if s > 0 else None


def _powexp_sequence(n: int, p: float, lam: float) -> np.ndarray:
    """a_i ∝ (i+1)^‑p · exp(-lam·i), normalised."""
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = (i ** (-p)) * np.exp(-lam * (i - 1))
    s = seq.sum()
    return seq / s if s > 0 else None


def _beta_sequence(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discretised beta(α,β) on {1,…,n}, normalised."""
    x = np.arange(1, n + 1, dtype=np.float64) / n
    # avoid overflow for very small α,β by using log‑space when needed
    seq = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    s = seq.sum()
    return seq / s if s > 0 else None


def _sorted_dirichlet(n: int, conc: float) -> np.ndarray:
    """Draw a Dirichlet vector, then sort it descending."""
    g = np.random.gamma(conc, 1.0, size=n)
    seq = np.sort(g)[::-1]
    return seq / seq.sum()

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that
    minimises ``evaluate_sequence``.  The routine adapts its schedule to
    the available time budget and always returns a feasible sequence.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a safety margin for the final exact evaluation
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # Choose working length.
    # --------------------------------------------------------------
    try:
        prev = height_sequence_1                # may be defined globally
        prev_len = len(prev)
    except Exception:
        prev = None
        prev_len = 0

    # aim for a somewhat larger problem than the previous run
    n = max(prev_len, 8000)                     # at least 8 k entries
    n = min(n, 20000)                           # cap to keep FFT cheap

    # FFT length – next power of two ≥ 2·n‑1
    fft_len = 1 << ((2 * n - 1).bit_length())

    # --------------------------------------------------------------
    # Cheap score (assumes Σa_i = 1): 2·n·max(conv)
    # --------------------------------------------------------------
    def cheap_score(vec: np.ndarray) -> float:
        conv = _conv_full_fft(vec, fft_len)
        return 2.0 * n * conv.max()

    # --------------------------------------------------------------
    # 0.   Initialise by scanning several parametric families
    # --------------------------------------------------------------
    best_a = np.full(n, 1.0 / n, dtype=np.float64)          # uniform baseline
    best_val = cheap_score(best_a)

    # previous construction, if it matches the length
    if prev is not None and len(prev) == n:
        cand = np.asarray(prev, dtype=np.float64)
        cand = cand / cand.sum()
        val = cheap_score(cand)
        if val < best_val:
            best_a, best_val = cand.copy(), val

    # ---------- exponential ----------
    best_lam = None
    for lam in np.linspace(0.0, 3.0, 121):          # coarse sweep
        if time.time() >= deadline:
            break
        cand = _exponential_sequence(n, lam)
        if cand is None:
            continue
        val = cheap_score(cand)
        if val < best_val:
            best_a, best_val, best_lam = cand.copy(), val, lam

    # local refinement around best_lam (if any)
    if best_lam is not None:
        for _ in range(300):
            if time.time() >= deadline:
                break
            # propose a multiplicative perturbation of magnitude ~10%
            lam = best_lam * (1.0 + 0.2 * (np.random.rand() - 0.5))
            lam = max(lam, 0.0)
            cand = _exponential_sequence(n, lam)
            if cand is None:
                continue
            val = cheap_score(cand)
            if val < best_val:
                best_a, best_val, best_lam = cand.copy(), val, lam

    # ---------- power‑law × exponential ----------
    best_p, best_lam_pe = None, None
    for p in np.linspace(0.4, 1.2, 41):
        for lam in np.linspace(0.0, 2.0, 41):
            if time.time() >= deadline:
                break
            cand = _powexp_sequence(n, p, lam)
            if cand is None:
                continue
            val = cheap_score(cand)
            if val < best_val:
                best_a, best_val = cand.copy(), val
                best_p, best_lam_pe = p, lam

    # random local refinement around the best (p,lam) pair
    if best_p is not None:
        for _ in range(500):
            if time.time() >= deadline:
                break
            p = best_p * (1.0 + 0.3 * (np.random.rand() - 0.5))
            lam = best_lam_pe * (1.0 + 0.3 * (np.random.rand() - 0.5))
            p = max(p, 0.05)
            lam = max(lam, 0.0)
            cand = _powexp_sequence(n, p, lam)
            if cand is None:
                continue
            val = cheap_score(cand)
            if val < best_val:
                best_a, best_val = cand.copy(), val
                best_p, best_lam_pe = p, lam

    # ---------- beta‑shaped ----------
    x = (np.arange(1, n + 1, dtype=np.float64) / n)
    for alpha in np.linspace(0.05, 5.0, 31):
        for beta in np.linspace(0.05, 5.0, 31):
            if time.time() >= deadline:
                break
            seq = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
            s = seq.sum()
            if s <= 0.0:
                continue
            cand = seq / s
            val = cheap_score(cand)
            if val < best_val:
                best_a, best_val = cand.copy(), val

    # ---------- random Dirichlet (sorted) ----------
    for conc in [0.001, 0.01, 0.05, 0.1, 0.5, 1.0]:
        if time.time() >= deadline:
            break
        cand = _sorted_dirichlet(n, conc)
        val = cheap_score(cand)
        if val < best_val:
            best_a, best_val = cand.copy(), val

    print(f"[Init] best cheap score = {best_val:.6f}")

    # --------------------------------------------------------------
    # 1. Refine the best candidate with Adam‑mirror‑descent
    # --------------------------------------------------------------
    a = best_a.copy()
    log_a = np.log(a + 1e-30)           # work in log‑domain (exponentiated‑gradient)
    m = np.zeros_like(a)               # first moment
    v = np.zeros_like(a)               # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.8
    max_iters = 800_000
    iter_cnt = 0
    last_improve = 0
    best_iter_a = a.copy()
    best_iter_val = best_val

    while time.time() < deadline - 2 and iter_cnt < max_iters:
        iter_cnt += 1

        # ---- re‑normalise a from log‑variables ----
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0 or not np.isfinite(s):
            # fall back to last good iterate
            a = best_iter_a.copy()
            log_a = np.log(a + 1e-30)
            continue
        a /= s

        # ---- convolution & cheap score ----
        conv = _conv_full_fft(a, fft_len)
        maxb = conv.max()
        cheap = 2.0 * n * maxb

        # ---- keep the best ever seen ----
        if cheap < best_iter_val - 1e-12:
            best_iter_val = cheap
            best_iter_a = a.copy()
            last_improve = iter_cnt
            print(f"[Iter {iter_cnt:07d}] NEW BEST cheap = {cheap:.6f}")

        # ---- sub‑gradient from the (single) maximal entry ----
        idx = int(conv.argmax())
        lo = max(0, idx - (n - 1))
        hi = min(idx, n - 1)
        rng = np.arange(lo, hi + 1, dtype=np.int64)
        g = np.zeros_like(a)
        g[rng] = 2.0 * a[idx - rng]      # derivative of max entry w.r.t. a

        # ---- Adam update in log‑space ----
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** iter_cnt)
        v_hat = v / (1 - beta2 ** iter_cnt)
        lr = base_lr / (math.sqrt(iter_cnt) + 1e-8)
        log_a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---- occasional restart if stuck ----
        if iter_cnt - last_improve > 150_000:
            # small Gaussian perturbation around the best iterate so far
            pert = 0.02 * np.random.randn(n)
            log_a = np.log(best_iter_a + 1e-30) + pert
            last_improve = iter_cnt
            print(f"[Iter {iter_cnt:07d}] Restart after stagnation.")

    # --------------------------------------------------------------
    # 2. Final exact evaluation (uses the provided ``evaluate_sequence``)
    # --------------------------------------------------------------
    final_seq = best_iter_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:      # defensive – should never happen
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # --------------------------------------------------------------
    # Return the best sequence found.
    # --------------------------------------------------------------
    return final_seq
