"""
Propose a non‑negative step‑function (list of heights) that tries to minimize
the provided ``evaluate_sequence`` objective.

The algorithm works in three stages:

1. **Exploratory scan** – a quick search over several analytic families
   (power‑law, exponential, logistic) on a modest support size (≈8000)
   to obtain a good starting shape.

2. **Mirror‑descent on the simplex** – Adam‑based mirror descent on the
   logits of the probability vector (softmax).  The objective is the
   smoothed maximum of the self‑convolution, providing a usable gradient.
   A back‑tracking line‑search guarantees monotone descent.  The smoothing
   factor ε is slowly reduced.

3. **Multi‑scale fine‑tuning** – the best distribution is linearly
   interpolated to a larger support (≈25000) and a short second
   mirror‑descent phase refines it.  Finally a lightweight simulated‑annealing
   loop performs random mass transfers between two coordinates to escape any
   remaining plateaus.

All operations respect the non‑negativity and sum‑invariance of the
sequence, so the true objective
`2·n·max(a*a) / (sum a)^2` is unchanged by scaling.  The function returns
the best sequence found before the time budget (default 1000 s) expires.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions – no closures, no lambdas
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max, returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    s = e.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / z.size)
    return e / s


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Inverse soft‑max (log‑it) that reproduces ``a`` after a soft‑max."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """Smallest power‑of‑two ≥ 3·n‑2 (enough for the linear self‑convolution)."""
    return 1 << ((3 * n - 2).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a * a."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Returns (max_conv, grad_a) where

        grad_a = ∇_a [ 2·n·smooth_max(a*a) ]

    ``eps_factor`` multiplies the smoothing constant ε = eps_factor·max_conv.
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    # FFT of a (zero‑padded)
    a_fft = np.fft.rfft(a, L)

    # Linear self‑convolution a*a
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # Smoothed soft‑max weights
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv                     # ≤ 0 for numerical stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # Correlation a ⋆ w (gradient of the smoothed max)
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # Gradient (already multiplied by 4·n, see derivation in the original code)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    s = new_a.sum()
    if s == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= s
    return new_a

# Analytic families (all return a probability vector that sums to 1)
# ----------------------------------------------------------------------


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{‑p}, normalised."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _exp_distribution(n: int, lam: float) -> np.ndarray:
    """Pure left‑hand exponential a_i ∝ exp(-lam·i), normalised."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _logistic_distribution(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ 1/(1+e^{α·(i‑mid)}), normalised; heavier on the left."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = 1.0 / (1.0 + np.exp(alpha * (i - mid)))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _random_dirichlet_distribution(n: int, concentration: float) -> np.ndarray:
    """Sample a Dirichlet distribution with identical concentration parameters."""
    # For concentration << 1 the draw is sparse, for >1 it is smoother.
    samples = np.random.gamma(concentration, 1.0, size=n)
    total = samples.sum()
    if total == 0.0:
        samples[:] = 1.0 / n
    else:
        samples /= total
    return samples


def _cheap_score(a: np.ndarray) -> float:
    """Convenient wrapper for the objective when `sum(a)=1`."""
    n = a.shape[0]
    return 2.0 * n * _conv_max(a)

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):

    simulated‑annealing polishing step.

    deadline = start_time + budget_s - 2.0

    # 0. Quick scan over analytic families (support size ~8 k)
    # ------------------------------------------------------------------
    n_small = 8000
    best_a = None
    best_score = np.inf

    for p in np.linspace(0.40, 1.50, 30):
        if time.time() >= deadline:
            break
        cand = _power_law_distribution(n_small, p)

            best_a = cand

            best_score = sc
            best_a = cand

    # Logistic shapes (asymmetric)
    for alpha in np.linspace(0.01, 8.0, 30):
        if time.time() >= deadline:
            break
        cand = _logistic_distribution(n_small, alpha)
        sc = _cheap_score(cand)

            best_a = cand

            sc = _cheap_score(cand)
            if sc < best_score:
                best_score = sc
                best_a = cand

    # Fallback – uniform distribution
    if best_a is None:
        best_a = np.full(n_small, 1.0 / n_small)
        best_score = _cheap_score(best_a)

    print(f"[Init] Scan complete, best cheap score (small n) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1. Mirror‑descent (Adam) on logits for the small support.
    # ------------------------------------------------------------------
    n = n_small
    a0 = best_a
    # Initialise logits with a small random jitter.
    z = _logits_from_distribution(a0) + 0.01 * np.random.randn(n)

    # Adam state
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    # Learning‑rate and smoothing schedules
    base_alpha = 0.30
    eps_factor = 0.10          # ε = eps_factor·max_conv
    eps_factor_min = 5e-5
    eps_decay = 0.9993

    # Tracking best ever seen
    best_seq = _softmax(z)
    best_score = _cheap_score(best_seq)
    last_improve_iter = 0
    iteration = 0

    # Allocate ~80 % of the budget to this phase
    grad_deadline = start_time + 0.78 * budget_s

    while time.time() < grad_deadline and iteration < 1_500_000:
        iteration += 1

        # Current point on the simplex
        a = _softmax(z)

        # Smoothed max‑convolution and its gradient
        max_conv, grad_a = _conv_max_and_grad(a, eps_factor)
        cheap_score = 2.0 * n * max_conv

        # Keep global best
        if cheap_score < best_score - 1e-12:
            best_score = cheap_score
            best_seq = a.copy()
            last_improve_iter = iteration
            if iteration % 10000 == 0:
                print(f"[Iter {iteration:07d}] NEW BEST cheap = {cheap_score:.6f}")

        # Gradient on logits (mirror‑descent on the simplex)
        mu = np.dot(grad_a, a)                     # <grad_a , a>
        grad_z = a * (grad_a - mu)                  # ∇_z (smooth max)

        # Adam moment updates (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        # Inverse‑sqrt step‑size schedule
        alpha_t = base_alpha / (math.sqrt(iteration) + 1.0)
        step = - m_hat / (np.sqrt(v_hat) + eps_adam)

        accepted = False
        trial_alpha = alpha_t
        for _ in range(4):
            z_new = z + trial_alpha * step
            # Clip to keep exp(z) numerically stable
            z_new = np.clip(z_new, -50.0, 50.0)
            a_new = _softmax(z_new)
            new_score = 2.0 * n * _conv_max(a_new)
            if new_score <= cheap_score:
                z = z_new
                accepted = True
                break
            trial_alpha *= 0.5

        if not accepted:
            # No improvement – still move a modest amount.
            z = z + alpha_t * step
            z = np.clip(z, -50.0, 50.0)

        # Occasional random jitter to escape plateaux
        if iteration % 4000 == 0 and np.random.rand() < 0.20:
            z += 0.015 * np.random.randn(n)
            m *= 0.6
            v *= 0.6

        # Reduce smoothing factor gradually
        eps_factor = max(eps_factor_min, eps_factor * eps_decay)

        # Restart if stagnation persists for a long time
        if iteration - last_improve_iter > 300_000:
            print(f"[Iter {iteration}] Stagnation – applying a restart jitter.")
            z = _logits_from_distribution(best_seq) + 0.02 * np.random.randn(n)
            m.fill(0.0)
            v.fill(0.0)
            eps_factor = 0.10
            last_improve_iter = iteration

    print(f"[Phase 1] Mirror‑descent finished after {iteration} iterations. "
          f"Best cheap score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Upsample to a larger support and do a short second refinement.
    # ------------------------------------------------------------------
    n_large = 25000
    if n_large > n:
        best_seq = _interpolate_distribution(best_seq, n_large)
        n = n_large
        z = _logits_from_distribution(best_seq) + 0.005 * np.random.randn(n)
        m = np.zeros_like(z)
        v = np.zeros_like(z)
        eps_factor = 0.08
        iteration = 0
        # Give the larger problem a modest amount of optimisation time.
        upsample_deadline = start_time + 0.92 * budget_s
        print(f"[Upsample] Interpolated to length {n} – entering second refinement.")
        while time.time() < upsample_deadline and iteration < 120_000:
            iteration += 1
            a = _softmax(z)
            max_conv, grad_a = _conv_max_and_grad(a, eps_factor)
            cheap_score = 2.0 * n * max_conv
            if cheap_score < best_score - 1e-12:
                best_score = cheap_score
                best_seq = a.copy()
                last_improve_iter = iteration
                if iteration % 5000 == 0:
                    print(f"[Upsample iter {iteration:06d}] NEW BEST cheap = {cheap_score:.6f}")

            mu = np.dot(grad_a, a)
            grad_z = a * (grad_a - mu)

            # Adam updates (reuse previous hyper‑parameters)
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
            m_hat = m / (1.0 - beta1 ** iteration)
            v_hat = v / (1.0 - beta2 ** iteration)

            alpha_t = base_alpha / (math.sqrt(iteration) + 1.0)
            step = - m_hat / (np.sqrt(v_hat) + eps_adam)

            # Simple line‑search (max 3 halvings)
            trial_alpha = alpha_t
            for _ in range(3):
                z_new = z + trial_alpha * step
                z_new = np.clip(z_new, -50.0, 50.0)
                a_new = _softmax(z_new)
                new_score = 2.0 * n * _conv_max(a_new)
                if new_score <= cheap_score:
                    z = z_new
                    break
                trial_alpha *= 0.5

            # Brief random perturbations
            if iteration % 5000 == 0 and np.random.rand() < 0.15:
                z += 0.01 * np.random.randn(n)

            eps_factor = max(eps_factor_min, eps_factor * eps_decay)

    # ------------------------------------------------------------------
    # 3. Light simulated‑annealing polishing (remaining time)
    # ------------------------------------------------------------------
    remaining_time = deadline - time.time()
    if remaining_time > 5.0:
        a = best_seq.copy()
        cheap = _cheap_score(a)
        # Linear temperature schedule from a small fraction of the current score down to near‑zero.
        T_start = max(1e-3 * cheap, 1e-4)
        T_end = max(1e-7 * cheap, 1e-7)
        anneal_start = time.time()
        while time.time() < deadline:
            # Choose a donor index with non‑negligible mass.
            i = np.random.randint(n)
            if a[i] < 1e-12:
                continue
            # Transfer a small amount of mass to a random recipient.
            delta = min(a[i] * 0.05, np.random.exponential(scale=0.001))
            if delta <= 0.0:
                continue
            j = np.random.randint(n)
            if i == j:
                continue
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta

            new_cheap = _cheap_score(a_new)

            # Progress‑dependent temperature (linear cooling)
            elapsed = time.time() - anneal_start
            total = deadline - anneal_start
            T = T_start * (1.0 - elapsed / total) + T_end * (elapsed / total)

            # Metropolis acceptance
            if new_cheap < cheap or np.random.rand() < math.exp(-(new_cheap - cheap) / max(T, 1e-12)):
                a = a_new
                cheap = new_cheap
                if cheap < best_score:
                    best_score = cheap
                    best_seq = a.copy()
        print(f"[Anneal] Polishing finished – final cheap score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Final exact evaluation with the provided evaluate_sequence()
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # Return the best sequence we have discovered.
    return final_seq
