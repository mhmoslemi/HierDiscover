Projection function:

def _project_simplex(v):
    # returns projection onto simplex sum=1, v>=0
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # find rho: largest i where u_i - cssv_i / i > 0
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _lp_improve(a, M):
    n = a.size
    rows, cols = 2 * n - 1, n
    A = np.zeros((rows, cols), dtype=float)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # i runs from i_start to i_end
        # For each i, j = k - i
        # coefficient at column j is a[i]
        # So assign row[k, j] = a[i]
        # Could vectorize:
        i_vals = np.arange(i_start, i_end + 1)
        j_vals = k - i_vals
        A[k, j_vals] = a[i_vals]
    # b = np.full(rows, M)
    # objective = -np.ones(cols)
    # bounds = [(0, None)] * cols
    # use linprog
    res = linprog(-np.ones(cols), A_ub=A, b_ub=np.full(rows, M), bounds=[(0, None)]*cols, method='highs')
    if not res.success:
        return None
    g = res.x
    # Ensure not all zeros
    if g.sum() <= 0:
        return None
    # Normalise to sum 1
    g = g / g.sum()
    return g

def _project_simplex(y):
    # y: 1D array
    # Duchi's algorithm
    u = np.sort(y)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(y)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all negative; return uniform distribution
        return np.full_like(y, 1.0 / len(y))
    rho = np.where(cond)[0][-1]

    w = np.maximum(y - theta, 0)
    return w

"""
Hybrid simplex‑gradient/LP optimiser for the step‑function problem.

The algorithm works on a probability simplex (sum = 1, a_i ≥ 0).
For a given support size *n* it alternates between:

* a projected (sub‑)gradient step on the exact max‑convolution
  (the gradient is the correlation of the active lag(s) with the sequence);
* an occasional linear‑programming step that finds a feasible direction
  with larger total mass while keeping the convolution bounded by the
  current worst value.

Both updates are projected back onto the simplex.  After the main phase
a cheap random‑mass‑transfer local search is performed.  Several lengths
are tried (small ones for cheap LP steps, larger ones for long gradient
runs) and the best sequence seen overall is returned.

The code respects the security checks of ``evaluate_sequence`` and never
produces negative or NaN entries.
"""
import time
import numpy as np

# -------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# -------------------------------------------------------------
def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using FFT (real‑valued)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv


# -------------------------------------------------------------
def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation of w with a (g_i = Σ_t w[i+t]·a[t]), 0 ≤ i < len(a).
    Implemented via FFT.
    """
    n = a.shape[0]
    m = w.shape[0]                 # = 2·n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


# -------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)), assumes sum(a)=1."""
    n = a.size
    conv = _conv_self_fft(a)
    return 2.0 * n * conv.max()


# -------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("simplex projection only works on 1‑D arrays")
    n = v.shape[0]
    # sort decreasing
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    # find rho
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are <=0: return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# -------------------------------------------------------------
# LP helper – optional, may be unavailable if SciPy is missing
try:
    from scipy.optimize import linprog
    _SCIPY_AVAILABLE = True
except Exception:                         # pragma: no cover
    _SCIPY_AVAILABLE = False


def _lp_improve(a: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the linear programme
        maximise   sum(g_i)
        subject to conv(a,g)[k] ≤ M   for all k
                    g_i ≥ 0
    The returned vector is normalised to sum = 1.
    Returns ``None`` if the LP fails or SciPy is unavailable.
    """
    if not _SCIPY_AVAILABLE:               # pragma: no cover
        return None

    n = a.size
    rows = 2 * n - 1

    # Build the constraint matrix A_ub * g ≤ M.
    # Row k corresponds to convolution index k.
    A = np.zeros((rows, n), dtype=float)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_vals = np.arange(i_start, i_end + 1)
        j_vals = k - i_vals
        A[k, j_vals] = a[i_vals]

    b = np.full(rows, M, dtype=float)
    c = -np.ones(n, dtype=float)           # maximise sum(g)  → minimise -sum(g)
    bounds = [(0, None)] * n

    # high‑level solver (interior‑point) is fast for these modest sizes
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        return None
    g = res.x
    if g.sum() <= 0.0:
        return None
    # normalise to a probability vector
    g = g / g.sum()
    return g


# -------------------------------------------------------------
def _run_custom_opt(
    n: int,
    time_limit: float,
    seed: int = 0,
) -> tuple[list[float], float]:
    """
    Hybrid optimiser for a single support size.
    Returns the best sequence (as a plain list) and its objective.
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # ---- initialise -------------------------------------------------
    a = np.full(n, 1.0 / n)                 # start uniform
    a += 0.02 * rng.random(n)               # small random perturbation
    a = _project_simplex(a)

    best_a = a.copy()
    best_obj = _objective(a)

    it = 0
    # simple schedules
    eta0 = 0.25
    min_eta = 0.001

    while time.time() - start < time_limit:
        it += 1

        # ----- evaluate current state --------------------------------
        conv = _conv_self_fft(a)
        M = conv.max()
        obj = 2.0 * n * M                     # sum(a)=1 -> objective = 2n·M
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        # ----- sub‑gradient of the exact max ------------------------
        eps = 1e-12
        active = np.where(conv >= M - eps)[0]
        w = np.zeros_like(conv)
        if active.size:
            w[active] = 1.0 / active.size     # uniform weighting over active lags
        g = 2.0 * _correlation_fft(w, a)      # (n,)

        # ----- step size (simple decay) -----------------------------
        eta = eta0 / (1.0 + it / 15000.0)
        eta = max(eta, min_eta)

        # ----- descent step on the simplex ---------------------------
        a_new = a - eta * g
        a = _project_simplex(a_new)

        # ----- occasional jitter – helps escape plateaus -------------
        if it % 6000 == 0:
            jitter = rng.normal(scale=0.02, size=n)
            a = _project_simplex(a + jitter)

        # ----- occasional LP improvement (only for modest sizes) -----
        if n <= 800 and _SCIPY_AVAILABLE and it % 20000 == 0:
            g_lp = _lp_improve(a, M)
            if g_lp is not None:
                # blend the LP solution with the current point
                t = 0.5
                a = _project_simplex((1 - t) * a + t * g_lp)

    # ---- compress negligible entries ---------------------------------
    # (removing near‑zero entries can occasionally improve the objective)
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = _objective(best_a)
        print(f"[compress] reduced length {n} → {best_a.size}, O={best_obj:.6f}")

    # ---- short stochastic local search --------------------------------
    remaining = time_limit - (time.time() - start)
    if remaining > 0.5:
        a_loc = best_a.copy()
        obj_loc = best_obj
        n_loc = a_loc.size
        rng_local = np.random.default_rng(seed + 9999)
        local_deadline = time.time() + remaining * 0.35
        while time.time() < local_deadline:
            i, j = rng_local.choice(n_loc, size=2, replace=False)
            delta = a_loc[i] * rng_local.random() * 0.02
            if delta <= 0.0:
                continue
            a_new = a_loc.copy()
            a_new[i] -= delta
            a_new[j] += delta
            a_new = _project_simplex(a_new)
            obj_new = _objective(a_new)
            if obj_new < obj_loc:
                a_loc = a_new
                obj_loc = obj_new
                if obj_new < best_obj:
                    best_a = a_new
                    best_obj = obj_new
        print(f"[local] final local O={best_obj:.6f}")

    return best_a.tolist(), best_obj

    """

    refinement.

                else:
                    pad = np.zeros(n - old.size)

    if best_seq is not None:

                a_new[i] -= delta
                a_new[j] += delta
                a_new = _project_simplex(a_new)
                obj_new = _objective(a_new)
                if obj_new < best_obj:
                    a = a_new
                    best_obj = obj_new
                    best_seq = a.tolist()
            print(f"[final local] O={best_obj:.6f}")

    if not best_seq:
        best_seq = [1.0]

    return best_seq
