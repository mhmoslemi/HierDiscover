```python
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    new_seq = np.interp(new_x, old_x, seq)
    # ensure nonnegative (interpolation of nonneg values yields nonneg)
    new_seq = np.maximum(new_seq, 0.0)
    total = new_seq.sum()
    if total <= 1e-12:
        new_seq = np.ones(new_n) / new_n
    else:
        new_seq = new_seq / total
    return new_seq
```

"""
Step‑function optimiser for evaluate_sequence.

The objective is 2·n·max(conv(a,a)) (the value returned by
evaluate_sequence for a normalised sequence).  The optimiser works in three
stages, all respecting a total time budget (default 1000 s).

1.  Hierarchical Adam optimisation.
    Several support sizes are tried (e.g. 2000 → 4000 → 8000 → 12000 → 16000).
    After each run the best sequence is linearly interpolated to the next
    size, providing a warm start with the same “shape’’ at higher
    resolution.  The Adam optimiser works on the soft‑max (log‑it)
    parametrisation of the simplex and uses an annealed soft‑max
    approximation of the max‑convolution.

2.  Deterministic peak‑reduction.
    A fast greedy loop repeatedly identifies the convolution index k
    where the current maximum occurs and moves a small fraction of mass
    away from the index i that contributes most to this peak.  Only moves
    that strictly lower the max‑convolution are kept, quickly flattening
    the convolution profile.

3.  Targeted simulated‑annealing refinement.
    Small random mass‑transfer moves are performed, accepting occasional
    uphill moves with a Metropolis probability that cools slowly.
    This helps escape any remaining local minima.

The final sequence (list of non‑negative floats) is returned and can be
passed directly to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return conv_full[: 2 * n - 1]


def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation  g_i = Σ_t w[i+t]·a[t]   (i = 0,…,n‑1) implemented via FFT.
    """
    n = a.shape[0]
    m = w.shape[0]                     # = 2n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) assuming sum(a) = 1."""
    n = a.size
    conv = _conv_self_fft(a)
    return float(2.0 * n * conv.max())


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a normalised sequence to a larger length."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(new_x, old_x, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s <= 1e-12:
        new_seq = np.ones(new_n) / new_n
    else:
        new_seq /= s
    return new_seq


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (soft‑max parametrisation)
# ----------------------------------------------------------------------


def _run_opt(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
) -> tuple[list[float], float]:
    """
    Adam optimisation for a given support size ``n``.
    Returns the best sequence (as a list) and its exact objective.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # initialise a on the simplex
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = _upsample(a, n)
        elif a.sum() < 1e-12:
            a = rng.random(n)
        a = np.clip(a, 0.0, None)
    else:
        a = rng.random(n)

    a = a / a.sum()
    # logits for the soft‑max
    x = np.log(a + 1e-16)
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # schedules – a bit more aggressive than the reference implementation
    beta_start, beta_end = 0.5, 2e5      # smooth‑max sharpness
    lr_start, lr_end = 0.25, 5e-4       # learning‑rate decay

    best_a = a.copy()
    best_obj = _objective(best_a)

    it = 0
    while time.time() - start < time_limit:
        it += 1
        a = _softmax(x)                     # stay on the simplex

        conv = _conv_self_fft(a)
        M = conv.max()
        obj = float(2.0 * n * M)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        # time‑based schedules
        t = min((time.time() - start) / time_limit, 1.0)
        beta = beta_start * np.exp(np.log(beta_end / beta_start) * t)
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t)

        # smooth‑max weights for the convolution entries
        if beta > 1e4:
            k = int(np.argmax(conv))
            w = np.zeros_like(conv)
            w[k] = 1.0
        else:
            shifted = conv - M                # ≤ 0
            w = np.exp(beta * shifted)
            s = w.sum()
            if s == 0.0:
                w = np.ones_like(conv) / conv.size
            else:
                w /= s

        # gradient of (smooth) max w.r.t. a
        g_a = 2.0 * _correlation_fft(w, a)

        # natural gradient in log‑it space
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape plateaus
        if it % 5000 == 0:
            jitter = rng.normal(scale=0.015, size=n)
            x += jitter

        # restart from current best every so often
        if it % 80000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.004, size=n)
            m.fill(0.0)
            v.fill(0.0)

    # --------------------------------------------------------------
    # prune negligible entries (helps later stages)
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = _objective(best_a)
        print(f"[compress] reduced length → {best_a.size}, O={best_obj:.6f}")

    # --------------------------------------------------------------
    # very short greedy local search (optional)
    remaining = time_limit - (time.time() - start)
    if remaining > 0.5:
        a_loc = best_a.copy()
        obj_loc = best_obj
        n_loc = a_loc.size
        local_start = time.time()
        while time.time() - local_start < remaining * 0.3:
            i, j = rng.choice(n_loc, size=2, replace=False)
            delta = a_loc[i] * rng.random() * 0.02
            if delta <= 0.0:
                continue
            a_new = a_loc.copy()
            a_new[i] -= delta
            a_new[j] += delta
            conv_new = _conv_self_fft(a_new)
            obj_new = float(2.0 * n_loc * conv_new.max())
            if obj_new < obj_loc:
                a_loc, obj_loc = a_new, obj_new
                if obj_new < best_obj:
                    best_a, best_obj = a_new, obj_new
        print(f"[local] final O={best_obj:.6f}")

    return best_a.tolist(), best_obj


# ----------------------------------------------------------------------
# Deterministic peak‑reduction loop
# ----------------------------------------------------------------------


def _reduce_peak(
    a_seq: np.ndarray,
    time_limit: float,
    seed: int = 0,
) -> tuple[np.ndarray, float]:
    """
    Greedy loop that moves a small fraction of mass from the index that
    contributes most to the current maximal convolution entry to a random
    other index.  Only strictly improving moves are kept.
    """
    start = time.time()
    rng = np.random.default_rng(seed)
    a = a_seq.copy()
    n = a.shape[0]

    best_obj = float(2.0 * n * _conv_self_fft(a).max())

    # step size (fraction of the selected mass) – slowly decays with time
    while time.time() - start < time_limit:
        conv = _conv_self_fft(a)
        max_val = conv.max()
        k_max = int(np.argmax(conv))

        # contributions for this k_max
        idx = np.arange(n)
        partner = k_max - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[idx[mask]] * a[partner[mask]]

        i_max = int(np.argmax(contrib))
        # pick a destination index far from the offending pair
        j = rng.integers(n - 1)
        if j >= i_max:
            j += 1

        # move a modest fraction of the mass at i_max
        delta = a[i_max] * 0.02  # 2 % of the current entry
        if delta <= 0.0:
            break

        a_new = a.copy()
        a_new[i_max] -= delta
        a_new[j] += delta
        a_new = np.clip(a_new, 0.0, None)
        a_new /= a_new.sum()

        obj_new = float(2.0 * n * _conv_self_fft(a_new).max())
        if obj_new < best_obj:
            a, best_obj = a_new, obj_new
            # occasionally break early if improvement becomes tiny
            if max_val - obj_new < 1e-12:
                continue
        # else discard the move (strict greedy)

    return a, best_obj

# Targeted simulated‑annealing refinement (same as the reference version)
# ----------------------------------------------------------------------


def _run_sa(
    init_seq: list[float],
    time_limit: float,
    seed: int = 0,
) -> tuple[list[float], float]:
    """Small‑scale simulated‑annealing that concentrates on the current peak."""
    start = time.time()
    rng = np.random.default_rng(seed)
    a = np.asarray(init_seq, dtype=float)
    if a.sum() < 1e-12:
        a = np.ones_like(a) / a.size
    else:
        a = a / a.sum()
    n = a.shape[0]

    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())
    best_seq = a.copy()

    # temperature schedule
    T = best_obj * 0.1 + 1e-6
    T_min = best_obj * 1e-6 + 1e-9
    decay = 0.995
    it = 0

    while time.time() - start < time_limit:
        it += 1
        k_max = int(np.argmax(conv))

        # contributions for the maximal index
        i_vals = np.arange(n)
        j_vals = k_max - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i_vals[mask]] * a[j_vals[mask]]
        total = contrib.sum()
        if total > 0.0:
            probs = contrib / total
            i = rng.choice(n, p=probs)
        else:
            i = rng.integers(n)

        # destination – any other index
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.05   # up to 5 %
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.clip(a_new, 0.0, None)
        a_new /= a_new.sum()

        conv_new = _conv_self_fft(a_new)
        obj_new = float(2.0 * n * conv_new.max())

        if obj_new < best_obj - 1e-12:
            a, conv, best_obj, best_seq = a_new, conv_new, obj_new, a_new.copy()
            print(f"[SA it={it}] new best O={best_obj:.6f}")
        else:
            prob = np.exp(-(obj_new - best_obj) / T) if T > 0 else 0.0
            if rng.random() < prob:
                a, conv = a_new, conv_new

        if it % 2000 == 0:
            T = max(T * decay, T_min)

        if it % 10000 == 0:
            noise = rng.normal(scale=0.008, size=n)
            a = np.clip(a + noise, 0.0, None)
            a /= a.sum()
            conv = _conv_self_fft(a)

    # final compression of tiny entries
    thresh = 1e-12
    mask = best_seq > thresh
    if mask.sum() < best_seq.size:
        comp = best_seq[mask]
        comp /= comp.sum()
        comp_obj = float(2.0 * comp.size * _conv_self_fft(comp).max())
        if comp_obj < best_obj:
            best_seq, best_obj = comp, comp_obj
            print(f"[SA compress] reduced length → {comp.size}, O={best_obj:.6f}")

    return best_seq.tolist(), best_obj

def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **_ignore,
) -> list[float]:
    """
    Produce a step‑function (list of non‑negative heights) that minimises
    ``evaluate_sequence``.  The routine respects the overall time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Load a previous construction if it exists.
    init_seq = None
    if "height_sequence_1" in globals():
        seq0 = np.asarray(height_sequence_1, dtype=float)
        seq0 = np.clip(seq0, 0.0, None)
        if seq0.sum() < 0.01:
            seq0 = np.ones_like(seq0)
        seq0 /= seq0.sum()
        init_seq = seq0

    # ------------------------------------------------------------------
    # Sequence lengths to explore (increasing resolution)
    lengths = [2000, 4000, 8000, 12000, 16000]

    # ensure the initial sequence (if any) is considered
    if init_seq is not None and init_seq.size not in lengths:
        lengths.insert(0, init_seq.size)

    best_seq = None
    best_obj = float("inf")

    # ------------------------------------------------------------------
    # Hierarchical Adam optimisation
    for idx, n in enumerate(lengths):
        now = time.time()
        remaining = deadline - now
        if remaining <= 2.0:
            break

        # allocate time: 15 % of the remaining budget for each of the
        # intermediate sizes, the rest for the final (largest) size
        if idx < len(lengths) - 1:
            run_time = remaining * 0.15
        else:
            run_time = remaining * 0.45   # most of the budget for the final size

        # initialise – upsample the best sequence found so far (if any)
        init = None
        if best_seq is not None:
            init = _upsample(best_seq, n)

        # also allow the explicit previous construction to be used directly
        if init_seq is not None and init_seq.size == n:
            init = init_seq

        seq, val = _run_opt(
            n=n,
            time_limit=run_time,
            init_a=init,
            seed=seed + idx * 997,
        )
        if val < best_obj:
            best_obj, best_seq = val, np.asarray(seq, dtype=float)
            print(f"[global] new overall best O={best_obj:.6f} (n={n})")

        # if time is almost exhausted break early
        if time.time() >= deadline - 0.5:
            break

    # ------------------------------------------------------------------
    # Deterministic peak‑reduction (aggressive greedy phase)
    remaining = deadline - time.time()
    if best_seq is not None and remaining > 5.0:
        reduce_time = remaining * 0.25
        a_red, obj_red = _reduce_peak(best_seq, reduce_time, seed=seed + 12345)
        if obj_red < best_obj:
            best_obj, best_seq = obj_red, a_red
            print(f"[reduce] improved O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Final simulated‑annealing refinement
    remaining = deadline - time.time()
    if best_seq is not None and remaining > 5.0:
        seq_refined, val_refined = _run_sa(
            best_seq.tolist(),
            time_limit=remaining,
            seed=seed + 777,
        )
        if val_refined < best_obj:
            best_obj, best_seq = val_refined, np.asarray(seq_refined, dtype=float)
            print(f"[final] after SA new best O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never happen
    if best_seq is None or best_seq.size == 0:
        best_seq = np.ones(100, dtype=float) / 100.0

    # final sanity check (the harness also does this)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_seq.tolist()
