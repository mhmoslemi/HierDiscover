#!/usr/bin/env python
"""
Non‑negative step‑function optimiser (improved).

The algorithm proceeds in three stages:

1. **Wide parametric sweep** – a few classic families (power‑law,
   symmetric product‑power‑law, symmetric exponential, logistic,
   arcsine) are scanned on a modest support size (≈20000).  The
   best‑scoring distribution is kept as a warm start.

2. **Entropic mirror‑descent** – the distribution is represented by
   logits `z` (so that `a = softmax(z)`).  A smoothed version of the
   max‑convolution is differentiated (using FFT) and Adam updates are
   performed on the logits.  The smoothing factor `eps_factor`
   (≈0.2→1e‑6) is annealed; a back‑tracking line‑search guarantees
   monotone improvement of the cheap score `2·n·max(a*a)`.  Periodic
   jitter and a stagnation‑restart keep the search from getting stuck.

3. **Optional up‑sampling** – when time permits, the current best
   distribution is linearly interpolated to a larger support (up to
   50000) and the mirror‑descent continues, allowing a finer
   discretisation of the (almost) optimal continuous shape.

The cheap score is exactly the value returned by `evaluate_sequence`
when the sequence is normalised to sum = 1, so the final call to
`evaluate_sequence` yields the true objective value.

The routine respects the supplied time budget (~1000 s) and returns the
best sequence found within that time.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    s = e.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / z.size)
    return e / s


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it that reproduces `a` after a soft‑max."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """Smallest power‑of‑two ≥ 3·n‑2 (enough for full convolution)."""
    return 1 << ((3 * n - 2).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Returns (max_conv, grad_a) where

        grad_a = ∇_a[ 2·n·smooth_max(a*a) ]

    `eps_factor` multiplies the smoothing constant ε = eps_factor·max_conv.
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv                # ≤ 0 for stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    grad_a = 4.0 * n * corr
    return max_conv, grad_a


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    s = new_a.sum()
    if s == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= s
    return new_a


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}, normalised."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _sym_product_power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ ((i+1)*(n‑i))^{-p}, symmetric, normalised."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) * (n - i)) ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _exp_decay_distribution(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)+exp(-lam·(n‑1‑i)), symmetric exponential."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _logistic_distribution(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ 1/(1+exp(α·(i‑mid))), left‑heavy logistic."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = 1.0 / (1.0 + np.exp(alpha * (i - mid)))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _arcsine_distribution(n: int) -> np.ndarray:
    """a_i ∝ 1/√((i+½)*(n‑½‑i)), a discrete arcsine‑law tail."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 0.5) * (n - 0.5 - i)) ** (-0.5)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) improves the
    `evaluate_sequence` score.  The routine respects the supplied time
    budget (~1000 s) and uses a gradient‑based mirror‑descent on the
    probability simplex, preceded by an extensive parametric sweep.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # 0. Choose a support size
    # ------------------------------------------------------------------
    n = 20000                     # start with a moderate size
    print(f"[Init] Support size = {n}")

    # ------------------------------------------------------------------
    # 1. Wide parametric sweep (fast, cheap scoring)
    # ------------------------------------------------------------------
    best_a = None
    best_score = float("inf")

    # If a global variable `height_sequence_1` exists, use it as a candidate
    try:
        global height_sequence_1
        init_from_global = np.array(height_sequence_1, dtype=np.float64)
        if init_from_global.ndim == 1 and init_from_global.size >= 2:
            a = np.maximum(init_from_global, 0.0)
            a_sum = a.sum()
            if a_sum > 0.0:
                a /= a_sum
            else:
                a = np.full_like(a, 1.0 / a.size)
            # possibly up‑sample / down‑sample to the current n
            if a.size != n:
                a = _interpolate_distribution(a, n)
            sc = 2.0 * n * _conv_max(a)
            print(f"[Init] Global candidate score = {sc:.6f}")
            best_a, best_score = a, sc
    except Exception:
        pass  # no global variable – continue with systematic sweep

    # Helper to update the incumbent
    def _maybe_update(candidate, label):
        nonlocal best_a, best_score
        sc = 2.0 * n * _conv_max(candidate)
        if sc < best_score:
            best_score, best_a = sc, candidate.copy()
            print(f"[Sweep] New best ({label}) score = {sc:.6f}")

    # Power‑law family
    for p in np.linspace(0.05, 5.0, 120):
        if time.time() >= deadline:
            break
        cand = _power_law_distribution(n, p)
        _maybe_update(cand, f"pow_{p:.3f}")

    # Symmetric product‑power‑law family
    for p in np.linspace(0.05, 5.0, 120):
        if time.time() >= deadline:
            break
        cand = _sym_product_power_law_distribution(n, p)
        _maybe_update(cand, f"sym_prod_{p:.3f}")

    # Symmetric exponential decay
    for loglam in np.linspace(-12, -2, 120):
        if time.time() >= deadline:
            break
        lam = 10.0 ** loglam
        cand = _exp_decay_distribution(n, lam)
        _maybe_update(cand, f"exp_{lam:.2e}")

    # Logistic family
    for alpha in np.linspace(0.01, 10.0, 120):
        if time.time() >= deadline:
            break
        cand = _logistic_distribution(n, alpha)
        _maybe_update(cand, f"log_{alpha:.3f}")

    # Arcsine‑type law (single candidate)
    cand = _arcsine_distribution(n)
    _maybe_update(cand, "arcsine")

    # Random Dirichlet draws (few)
    for _ in range(10):
        if time.time() >= deadline:
            break
        cand = np.random.rand(n)
        cand /= cand.sum()
        _maybe_update(cand, "rand")

    if best_a is None:
        # fallback – uniform distribution
        best_a = np.full(n, 1.0 / n)
        best_score = 2.0 * n * _conv_max(best_a)
        print("[Init] No candidate found – using uniform distribution.")

    # Slight jitter before optimisation
    z = _logits_from_distribution(best_a)
    z += 0.01 * np.random.randn(n)

    # ------------------------------------------------------------------
    # 2. Entropic mirror‑descent (Adam on logits)
    # ------------------------------------------------------------------
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    base_alpha = 0.5          # larger base step size
    eps_factor = 0.2          # start smoothing factor
    eps_factor_min = 1e-7
    eps_decay = 0.9995

    best_seq = _softmax(z)               # best distribution ever seen
    best_score = 2.0 * n * _conv_max(best_seq)
    last_improve_iter = 0
    it = 0
    upsampled = False

    # Main optimisation loop
    while time.time() < deadline and it < 2_000_000:
        it += 1

        # ----- current point on the simplex -----
        a = _softmax(z)

        # ----- cheap evaluation + smoothed gradient -----
        max_conv, grad_a = _conv_max_and_grad(a, eps_factor)
        cheap_score = 2.0 * n * max_conv

        # ----- keep best ever seen -----
        if cheap_score < best_score - 1e-13:
            best_score = cheap_score
            best_seq = a.copy()
            last_improve_iter = it
            if it % 5000 == 0:
                print(f"[Iter {it:08d}] NEW BEST cheap = {cheap_score:.6f}")

        # ----- Adam on logits (mirror descent) -----
        mu = np.dot(grad_a, a)
        grad_z = a * (grad_a - mu)

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step_dir = - m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- back‑tracking line search (max 4 halvings) -----
        accepted = False
        trial_alpha = alpha_t
        for _ in range(4):
            z_new = z + trial_alpha * step_dir
            z_new = np.clip(z_new, -50.0, 50.0)
            a_new = _softmax(z_new)
            new_score = 2.0 * n * _conv_max(a_new)
            if new_score <= cheap_score:
                z = z_new
                accepted = True
                break
            trial_alpha *= 0.5

        if not accepted:
            # modest step anyway (no guarantee of improvement)
            z += alpha_t * step_dir
            z = np.clip(z, -50.0, 50.0)

        # ----- occasional random jitter to escape plateaux -----
        if it % 4000 == 0 and np.random.rand() < 0.15:
            z += 0.015 * np.random.randn(n)
            m *= 0.6
            v *= 0.6

        # ----- anneal smoothing factor -----
        eps_factor = max(eps_factor_min, eps_factor * eps_decay)

        # ----- stagnation restart -----
        if it - last_improve_iter > 250_000:
            print(f"[Iter {it}] Stagnation – applying jitter restart.")
            z = _logits_from_distribution(best_seq) + 0.02 * np.random.randn(n)
            m.fill(0.0)
            v.fill(0.0)
            eps_factor = 0.2
            last_improve_iter = it

        # ----- optional up‑sampling (once) if time permits -----
        if (not upsampled) and (it % 100_000 == 0) and (n < 50000):
            # require a safety margin for the remainder of the run
            if time.time() + 30.0 < deadline:
                new_n = min(50000, int(n * 1.5))
                print(f"[Iter {it}] Upsampling support: {n} → {new_n}")
                best_seq = _interpolate_distribution(best_seq, new_n)
                z = _logits_from_distribution(best_seq) + 0.005 * np.random.randn(new_n)
                m = np.zeros_like(z)
                v = np.zeros_like(z)
                n = new_n
                best_score = 2.0 * n * _conv_max(best_seq)
                it = 0
                last_improve_iter = 0
                eps_factor = 0.2
                upsampled = True
                # continue with the larger problem size
                continue

    # ------------------------------------------------------------------
    # 3. Final exact evaluation with the provided evaluate_sequence()
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # ------------------------------------------------------------------
    # 4. Return the best sequence discovered.
    # ------------------------------------------------------------------
    return final_seq
