import cvxpy as cp

def solve_lp(f, M):
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    conv = cp.conv(cp.Constant(f), g)  # size 2n-1
    constraints = [conv <= M]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True)  # maybe also solver=cp.SCS

    if prob.status not in ["optimal", "optimal_inaccurate"]:
        return None
    return g.value

"""
Propose a non‑negative step‑function sequence that (approximately) minimises

    O(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The optimiser consists of four stages:

1️⃣  **Long Adam on a smooth log‑sum‑exp surrogate** – we minimise
    (2·n/α)·log Σ exp(α·conv) + λ·Σ a_i·log a_i
    while projecting onto the probability simplex (∑a_i=1, a_i≥0).  The
    smoothing α is annealed from 2 to 5·10⁴, the entropy weight λ from 0.3 to 0,
    and the Adam step size decays ∝1/√t.

2️⃣  **Aggressive simulated‑annealing refinement** – repeated “mass‑transfer’’
    moves (move a random fraction of mass from i to j) are accepted if they
    improve the exact objective, otherwise with a Metropolis probability.
    An exact sub‑gradient of the max‑convolution is applied every iteration.

3️⃣  **Hybrid swap‑move stage** – we occasionally exchange the heights of two
    indices (swap) which can instantly reduce the peak of the convolution
    without changing the one‑norm.

4️⃣  **Final high‑α Adam polishing** – a short Adam run with a very large
    smoothing (α≈1e⁵) and zero entropy to fine‑tune the remaining slack.

All three phases respect the global 1000 s budget.  The best sequence found
is returned as a plain ``list`` of ``float``.

# ----------------------------------------------------------------------
# Simplex projection (Euclidean).  Guarantees a ≥ 0 and Σa = 1.
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the probability simplex."""
    n = v.shape[0]
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution via FFT (length 2·n‑1).  L is a power‑of‑two ≥ 2·n‑1.
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Exact objective for a probability vector (∑a=1).
def _objective(a: np.ndarray, n: int, L: int) -> float:
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Gradient of the smooth surrogate (log‑sum‑exp) plus entropy term.
def _smooth_grad(a: np.ndarray, conv: np.ndarray,
                 alpha: float, L: int, lam: float) -> np.ndarray:
    # soft‑max weights w_k ∝ exp(α·conv_k)
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / exp_scaled.sum()                       # shape (2n‑1,)

    # correlation w ⋆ a (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n_len = a.size
    grad_corr = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]

    # entropy part λ·(log a_i + 1)
    eps = 1e-20
    grad_ent = lam * (np.log(a + eps) + 1.0)

    return grad_corr + grad_ent

    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)            # indices where the max is attained
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    the provided evaluation function.  Returns the best sequence found.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.2           # small safety margin

    # --------------------------------------------------------------
    # Problem size ---------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.array(height_sequence_1, dtype=float)
        n = base.size
        a = np.maximum(base, 0.0)
        if a.sum() < 1e-12:
            a = np.full(n, 1.0 / n)
        else:
            a = a / a.sum()
        print(f"[init] start from supplied sequence, n={n}")
    else:
        # a moderately large n gives the optimizer more freedom
        n = 5000
        a = np.full(n, 1.0 / n)
        print(f"[init] start from uniform, n={n}")

    # FFT length – next power of two ≥ 2·n‑1
    L = 1
    while L < 2 * n - 1:
        L <<= 1

    # --------------------------------------------------------------
    # Phase‑1 : Annealed Adam on a smooth surrogate -------------------
    best_seq = a.copy()
    best_score = _objective(best_seq, n, L)
    print(f"[phase‑1] start O={best_score:.6f}")

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # schedule parameters
    max_iter_phase1 = 800_000
    alpha_start, alpha_end = 2.0, 50_000.0
    lambda_start, lambda_end = 0.3, 0.0
    lr0 = 0.9

    it = 0
    phase1_deadline = start_time + budget_s * 0.45   # ~45 % of total budget

    while time.time() < phase1_deadline and it < max_iter_phase1:
        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[phase‑1] iter {it:,} new best O={best_score:.6f}")

        # ----- annealing schedule -----
        frac = it / max_iter_phase1
        alpha = alpha_start + (alpha_end - alpha_start) * frac
        lam = max(0.0, lambda_start * (1.0 - frac))

        # ----- smooth gradient -----
        grad = _smooth_grad(a, conv, alpha, L, lam)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        lr = lr0 / math.sqrt(it + 1)                # 1/√t decay
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional tiny jitter to escape plateaus
        if np.random.rand() < 0.001:
            a_new += np.random.randn(n) * 1e-5

        # Euclidean projection onto simplex
        a = _project_to_simplex(a_new)
        it += 1

    # --------------------------------------------------------------
    # Phase‑2 : Simulated‑annealing intensive local refinement ---------
    print(f"[phase‑2] entering local refinement, best O={best_score:.6f}")

    temperature0 = 0.12
    temp_decay = 0.99990
    max_iter_local = 400_000
    no_improve_limit = 3000

    while time.time() < deadline:
        # initialise a restart (mostly from current best, occasionally random)
        if np.random.rand() < 0.7:
            a = best_seq.copy()
            a += np.random.randn(n) * 0.0008
            a = np.clip(a, 0, None)
            a = _project_to_simplex(a)
        else:
            a = np.random.rand(n)
            a = _project_to_simplex(a)

        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        T = temperature0
        it_loc = 0
        no_improve = 0

        while time.time() < deadline and it_loc < max_iter_local:
            # ---- mass‑transfer move ----
            i, j = np.random.randint(n, size=2)
            if i != j and a[i] > 1e-14:
                max_frac = 0.4 * (1.0 - it_loc / max_iter_local) + 0.02
                frac = np.random.rand() * max_frac
                delta = a[i] * frac
                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta
                a_new = _project_to_simplex(a_new)

                conv_new = _conv_fft(a_new, L)
                new_obj = 2.0 * n * conv_new.max()

                if new_obj < cur_obj or np.random.rand() < math.exp((cur_obj - new_obj) / max(T, 1e-16)):
                    a, conv, cur_obj = a_new, conv_new, new_obj
                    no_improve = 0
                    if cur_obj < best_score:
                        best_score = cur_obj
                        best_seq = a.copy()
                        print(f"[{time.time():.1f}] new global best O={best_score:.6f} (iter={it_loc})")
                else:
                    no_improve += 1
            else:
                no_improve += 1

            # ---- swap‑move (exchange two entries) ----
            if it_loc % 2000 == 0:
                i, j = np.random.randint(n, size=2)
                if i != j:
                    a_swapped = a.copy()
                    a_swapped[i], a_swapped[j] = a_swapped[j], a_swapped[i]
                    conv_swapped = _conv_fft(a_swapped, L)
                    obj_swapped = 2.0 * n * conv_swapped.max()
                    if obj_swapped < cur_obj or np.random.rand() < math.exp((cur_obj - obj_swapped) / max(T, 1e-16)):
                        a, conv, cur_obj = a_swapped, conv_swapped, obj_swapped
                        no_improve = 0
                        if cur_obj < best_score:
                            best_score = cur_obj
                            best_seq = a.copy()
                            print(f"[{time.time():.1f}] swap improved O={best_score:.6f}")

            # ---- sub‑gradient step (aggressive descent) ----
            g = _subgradient_max_conv(a, conv)
            eta = 0.12 * (1.0 - it_loc / max_iter_local)   # linearly decreasing step
            a = a - eta * g
            a = _project_to_simplex(a)
            conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_score:
                best_score = cur_obj
                best_seq = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved O={best_score:.6f}")

            # ---- occasional smoothing of the profile ----
            if it_loc % 5000 == 0 and it_loc > 0:
                kernel = np.array([0.125, 0.25, 0.25, 0.25, 0.125])
                a_s = np.convolve(a, kernel, mode="same")
                a = _project_to_simplex(0.85 * a + 0.15 * a_s)

            # temperature update & iteration counter
            T *= temp_decay
            it_loc += 1

            # restart if stuck
            if no_improve >= no_improve_limit:
                break

        if time.time() >= deadline:
            break

    # --------------------------------------------------------------
    # Phase‑3 : Hybrid swap‑only search (helps escape flat plateaus) ----
    print(f"[phase‑3] hybrid swap stage, best O={best_score:.6f}")
    max_swaps = 150_000
    T = 0.08
    decay = 0.99992

    a = best_seq.copy()
    conv = _conv_fft(a, L)
    cur_obj = 2.0 * n * conv.max()
    it_sw = 0
    while time.time() < deadline and it_sw < max_swaps:
        i, j = np.random.randint(n, size=2)
        a_swapped = a.copy()
        a_swapped[i], a_swapped[j] = a_swapped[j], a_swapped[i]
        conv_swapped = _conv_fft(a_swapped, L)
        obj_swapped = 2.0 * n * conv_swapped.max()
        if obj_swapped < cur_obj or np.random.rand() < math.exp((cur_obj - obj_swapped) / max(T, 1e-16)):
            a, conv, cur_obj = a_swapped, conv_swapped, obj_swapped
            if cur_obj < best_score:
                best_score = cur_obj
                best_seq = a.copy()
                print(f"[{time.time():.1f}] swap‑stage new best O={best_score:.6f}")
        T *= decay
        it_sw += 1

    # --------------------------------------------------------------
    # Phase‑4 : Final high‑α Adam polishing -------------------------------
    print(f"[phase‑4] final polishing, best O={best_score:.6f}")
    a = best_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    lr0 = 0.6
    alpha = 80_000.0
    lam = 0.0
    polish_iters = 30_000
    it = 0
    while time.time() < deadline and it < polish_iters:
        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[phase‑4] iter {it:,} new best O={best_score:.6f}")

        grad = _smooth_grad(a, conv, alpha, L, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        lr = lr0 / math.sqrt(it + 1)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_to_simplex(a_new)
        it += 1

    # --------------------------------------------------------------
    # Final sanity check with the provided evaluator
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:
        print("[final] evaluate_sequence failed:", e)

    return best_seq.tolist()
