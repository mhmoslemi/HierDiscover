def project_simplex(v):
    # ensure sum=1, v>=0
    # Duchi algorithm.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def simulated_annealing(seq, time_limit, seed):
    start = time.time()
    rng = np.random.default_rng(seed)
    a = np.asarray(seq, dtype=float)
    a = a / a.sum()
    n = a.size
    conv = conv_fft(a)
    best_obj = 2 * n * conv.max()
    best_seq = a.copy()
    T = best_obj * 0.5
    T_min = best_obj * 1e-6
    decay = 0.995
    it = 0
    while time.time() - start < time_limit:
        it += 1
        # find max index
        k_max = int(np.argmax(conv))
        # contributions array
        i_idx = np.arange(n)
        j_idx = k_max - i_idx
        mask = (j_idx >= 0) & (j_idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i_idx[mask]] * a[j_idx[mask]]
        total_contrib = contrib.sum()
        # pick source proportional to its contribution
        if total_contrib > 0:
            probs = contrib / total_contrib
            i = rng.choice(n, p=probs)
        else:
            i = rng.integers(n)
        # choose target uniformly among others
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        # size of transfer: adapt based on T
        max_delta = a[i] * 0.2  # up to 20%
        delta = max_delta * rng.random() * (T / best_obj)  # scale with temperature
        if delta < 1e-12:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.clip(a_new, 0, None)
        a_new /= a_new.sum()
        conv_new = conv_fft(a_new)
        obj_new = 2 * n * conv_new.max()
        if obj_new < best_obj:
            a, conv, best_obj, best_seq = a_new, conv_new, obj_new, a_new.copy()
            # possibly reduce temperature faster
        else:
            prob = np.exp(-(obj_new - best_obj) / T)
            if rng.random() < prob:
                a, conv = a_new, conv_new
        if it % 2000 == 0:
            T = max(T * decay, T_min)
    return best_seq.tolist(), best_obj
```

def _project_simplex(v):
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero? return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def adam_opt(n, time_limit, init_a, seed):
    start = time.time()
    rng = np.random.default_rng(seed)
    # init a on simplex
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.shape[0] != n:
            a = rng.random(n)
        a = a / a.sum()
    else:
        a = rng.random(n)
        a = a / a.sum()
    x = np.log(a + 1e-16)  # logits
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    # schedule
    beta_start = 0.5
    beta_end = 20000.0
    lr_start = 0.2
    lr_end = 0.0008

    best_a = a.copy()
    best_obj = _objective(best_a)

    it = 0
    while time.time() - start < time_limit:
        it += 1
        # compute a from logits
        a = _softmax(x)

        # compute convolution
        conv = _conv_self_fft(a)
        max_c = conv.max()

        obj = float(2.0 * n * max_c)

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[Adam n={n} it={it}] new best O={best_obj:.6f}")

        # schedule
        elapsed = time.time() - start
        t = min(elapsed / time_limit, 1.0)
        beta = beta_start * np.exp(np.log(beta_end / beta_start) * t)
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t)

        # determine gradient g_a
        # if beta is large use hard max; else smooth weights
        if beta > 5000.0:
            # hard max: gradient is shift of a
            kmax = int(np.argmax(conv))
            # compute shifted gradient
            idx = np.arange(n)
            idx2 = kmax - idx
            mask = (idx2 >= 0) & (idx2 < n)
            g_a = np.zeros_like(a)
            g_a[mask] = a[idx2[mask]]
            g_a = 2.0 * g_a
        else:
            # smooth max weighting
            shifted = conv - max_c
            w = np.exp(beta * shifted)
            s = w.sum()
            if s == 0.0:
                w = np.ones_like(conv) / conv.size
            else:
                w = w / s
            # correlation
            g_a = 2.0 * _correlation_fft(w, a)

        # natural gradient in logit space
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter
        if it % 10000 == 0:
            jitter = rng.normal(scale=0.01, size=n)
            x += jitter

        # restart
        if it % 120000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.005, size=n)
            m.fill(0.0)
            v.fill(0.0)

    # compress negligible entries (optional)
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = _objective(best_a)
        print(f"[compress] n={best_a.size}, O={best_obj:.6f}")

    # optional short local search using gradient descent
    # use remaining time if enough
    remaining = time_limit - (time.time() - start)
    if remaining > 0.5:
        seq, val = subgradient_refine(best_a, remaining * 0.3, seed + 1000)
        if val < best_obj:
            best_a, best_obj = np.asarray(seq, dtype=float), val

    return best_a.tolist(), best_obj

        max_c = conv.max()

def simulated_annealing(seq, time_limit, seed):
    start = time.time()
    rng = np.random.default_rng(seed)
    a = np.asarray(seq, dtype=float)
    a = a / a.sum()
    n = a.size
    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())
    best_seq = a.copy()
    # Initialize temperature
    T = best_obj * 0.2 + 1e-6
    T_min = best_obj * 1e-6 + 1e-9
    decay = 0.998
    it = 0
    while time.time() - start < time_limit:
        it += 1
        # index of max convolution
        k_max = int(np.argmax(conv))
        # contributions to that entry
        i_vals = np.arange(n)
        j_vals = k_max - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i_vals[mask]] * a[j_vals[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            probs = contrib / total_contrib
            i = rng.choice(n, p=probs)
        else:
            i = rng.integers(n)
        # choose destination uniformly (different)
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        # transfer delta
        max_delta = a[i] * 0.2
        delta = max_delta * rng.random() * (T / best_obj)
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.clip(a_new, 0.0, None)
        a_new /= a_new.sum()
        conv_new = _conv_self_fft(a_new)
        obj_new = float(2.0 * n * conv_new.max())
        if obj_new < best_obj:
            a, conv, best_obj, best_seq = a_new, conv_new, obj_new, a_new.copy()
            # optional reduce temperature more aggressively
        else:
            prob = np.exp(-(obj_new - best_obj) / T) if T > 0 else 0.0
            if rng.random() < prob:
                a, conv = a_new, conv_new
        if it % 2000 == 0:
            T = max(T * decay, T_min)
        # occasional global jitter
        if it % 10000 == 0:
            noise = rng.normal(scale=0.01, size=n)
            a = np.clip(a + noise, 0.0, None)
            a /= a.sum()
            conv = _conv_self_fft(a)
    # final compression
    thresh = 1e-12
    mask = best_seq > thresh
    if mask.sum() < best_seq.size:
        comp = best_seq[mask]
        comp = comp / comp.sum()
        comp_obj = _objective(comp)
        if comp_obj < best_obj:
            best_seq, best_obj = comp, comp_obj
    return best_seq.tolist(), best_obj

            seq0 = seq0 / seq0.sum()

            init_seq = None

    # Choose support sizes
    lengths = [6000, 9000, 12000, 15000]
    if init_seq is not None and init_seq.size not in lengths:
        lengths.insert(0, init_seq.size)

    best_seq = None
    best_val = float('inf')

    for idx, n in enumerate(lengths):
        now = time.time()
        remaining = deadline - now
        if remaining < 5.0:
            break
        if idx == 0:
            run_time = remaining * 0.45
        else:
            # allocate remaining time equally among remaining lengths
            remaining_lengths = len(lengths) - idx
            run_time = (remaining * 0.4) / remaining_lengths

        init = None

            old_n = best_seq.size
            x_old = np.arange(old_n)

            init = np.interp(x_new, x_old, best_seq)

        # Run Adam optimizer

"""
Step‑function optimiser.

The algorithm works in three stages while respecting the 1000 s budget:

1. **Adam optimisation on the simplex** – a log‑it (soft‑max) parametrisation
   is used together with an annealed smooth‑max of the self‑convolution.
   When the smoothing parameter β becomes large we switch to the exact
   sub‑gradient of the maximal convolution entry (a cheap O(n) shift), which
   allows us to handle long supports (≈12‑15 k).

2. **Deterministic sub‑gradient descent** – repeatedly moves opposite to the
   gradient of the active convolution entries and projects back onto the
   probability simplex.

3. **Targeted simulated‑annealing** – transfers a small amount of mass from
   indices that contribute most to the current peak to random destinations,
   accepting uphill moves with a temperature schedule.

All intermediate results are kept; the best sequence found is returned as a
plain list of floats suitable for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra / FFT helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a * a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return conv_full[: 2 * n - 1]


def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation g_i = Σ_t w[i+t]·a[t]  for i = 0,…,n‑1.
    Implemented via FFT (real to complex for speed).
    """
    n = a.shape[0]
    m = w.shape[0]                     # = 2n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex {x≥0, Σx=1}.
    Implements the algorithm of Duchi et al. (2008).
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) under the sum‑to‑one normalisation.
    (The denominator sum(a)**2 is 1, so we omit it.)
    """
    n = a.size
    conv = _conv_self_fft(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Stage 1 – Adam optimiser on the simplex
# ----------------------------------------------------------------------


def adam_opt(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None,
    seed: int = 0,
) -> tuple[list[float], float]:
    """Adam optimiser that gradually sharpens a smooth‑max approximation."""
    start = time.time()
    rng = np.random.default_rng(seed)

    # ----- initialise a on the simplex -----
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.shape[0] != n:
            a = rng.random(n)
        a = a / a.sum()
    else:
        a = rng.random(n)
        a = a / a.sum()

    x = np.log(a + 1e-16)               # logits (log‑it parametrisation)
    m = np.zeros_like(x)                # Adam 1‑st moment
    v = np.zeros_like(x)                # Adam 2‑nd moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # schedule parameters
    beta_start = 0.5
    beta_end = 20000.0
    lr_start = 0.2
    lr_end = 0.0008

    best_a = a.copy()
    best_obj = _objective(best_a)

    it = 0
    while time.time() - start < time_limit:
        it += 1

        # ----- current point on the simplex -----
        a = _softmax(x)

        # ----- convolution and objective -----
        conv = _conv_self_fft(a)
        max_c = conv.max()
        obj = float(2.0 * n * max_c)

        # ----- record best -----
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[Adam n={n:5d} it={it:06d}] new best O={best_obj:.6f}")

        # ----- annealing / learning‑rate schedule -----
        t = min((time.time() - start) / time_limit, 1.0)
        beta = beta_start * np.exp(np.log(beta_end / beta_start) * t)
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t)

        # ----- gradient of (smooth) max w.r.t. a -----
        if beta > 5000.0:
            # hard max – use the exact sub‑gradient (cheap O(n) shift)
            kmax = int(np.argmax(conv))
            idx = np.arange(n)
            idx2 = kmax - idx
            mask = (idx2 >= 0) & (idx2 < n)
            g_a = np.zeros_like(a)
            g_a[mask] = a[idx2[mask]]
            g_a = 2.0 * g_a
        else:
            # smooth max – exponential weights
            shifted = conv - max_c               # ≤ 0
            w = np.exp(beta * shifted)
            s = w.sum()
            if s == 0.0:
                w = np.ones_like(conv) / conv.size
            else:
                w = w / s
            g_a = 2.0 * _correlation_fft(w, a)

        # ----- natural gradient in log‑it space (mirrored gradient) -----
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- occasional jitter to escape plateaus -----
        if it % 10000 == 0:
            jitter = rng.normal(scale=0.01, size=n)
            x += jitter

        # ----- occasional restart from the current best -----
        if it % 120000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.005, size=n)
            m.fill(0.0)
            v.fill(0.0)

    # -----------------------------------------------------------------
    # compress negligible entries (helps the final objective)
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = _objective(best_a)
        print(f"[compress] reduced length → {best_a.size}, O={best_obj:.6f}")

    # -----------------------------------------------------------------
    # short deterministic sub‑gradient refinement if time permits
    remaining = time_limit - (time.time() - start)
    if remaining > 0.5:
        seq, val = subgradient_refine(best_a, remaining * 0.3, seed + 777)
        if val < best_obj:
            best_a, best_obj = np.asarray(seq, dtype=float), val

    return best_a.tolist(), best_obj


# ----------------------------------------------------------------------
# Stage 2 – Deterministic sub‑gradient descent on the active set
# ----------------------------------------------------------------------


def subgradient_refine(seq: list[float], time_limit: float, seed: int = 0) -> tuple[
    list[float], float]:
    """Projected sub‑gradient descent that reduces the max‑convolution."""
    start = time.time()
    rng = np.random.default_rng(seed)
    a = np.asarray(seq, dtype=float)
    a = a / a.sum()
    n = a.size

    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() - start < time_limit:
        it += 1
        conv = _conv_self_fft(a)
        max_c = conv.max()

        # active set: entries within 2 % of the maximum
        eps = max_c * 0.02
        mask = conv >= max_c - eps
        w = mask.astype(float)                       # shape (2n‑1,)

        # gradient = 2 * correlation(w, a)
        g = 2.0 * _correlation_fft(w, a)

        # diminishing step size
        eta = 0.4 / np.sqrt(it)

        # exponentiated‑gradient update + simplex projection
        a = a * np.exp(-eta * g)
        a = _project_simplex(a)

        val = _objective(a)
        if val < best_val - 1e-14:
            best_val = val
            best_a = a.copy()
            print(f"[Subgrad it={it}] new best O={best_val:.6f}")

    return best_a.tolist(), best_val


# ----------------------------------------------------------------------
# Stage 3 – Targeted simulated‑annealing refinement
# ----------------------------------------------------------------------


def simulated_annealing(seq: list[float], time_limit: float, seed: int = 0) -> tuple[
    list[float], float]:
    """Fine‑grained SA that moves a small amount of mass from high‑contrib. sites."""
    start = time.time()
    rng = np.random.default_rng(seed)
    a = np.asarray(seq, dtype=float)
    a = a / a.sum()
    n = a.size

    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())
    best_seq = a.copy()

    # temperature schedule
    T = best_obj * 0.2 + 1e-6
    T_min = best_obj * 1e-6 + 1e-9
    decay = 0.998
    it = 0

    while time.time() - start < time_limit:
        it += 1

        # index where convolution attains its maximum
        k_max = int(np.argmax(conv))

        # contributions a[i]·a[k_max‑i]
        i_vals = np.arange(n)
        j_vals = k_max - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i_vals[mask]] * a[j_vals[mask]]
        total_contrib = contrib.sum()

        # choose source index i proportionally to its contribution
        if total_contrib > 0:
            probs = contrib / total_contrib
            i = rng.choice(n, p=probs)
        else:
            i = rng.integers(n)

        # choose a distinct destination index uniformly at random
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # transfer a modest fraction of mass from i to j
        max_delta = a[i] * 0.2
        delta = max_delta * rng.random() * (T / best_obj)
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.clip(a_new, 0.0, None)
        a_new /= a_new.sum()

        conv_new = _conv_self_fft(a_new)
        obj_new = float(2.0 * n * conv_new.max())

        if obj_new < best_obj:
            a, conv, best_obj, best_seq = a_new, conv_new, obj_new, a_new.copy()
        else:
            prob = np.exp(-(obj_new - best_obj) / T) if T > 0 else 0.0
            if rng.random() < prob:
                a, conv = a_new, conv_new

        # temperature decay
        if it % 2000 == 0:
            T = max(T * decay, T_min)

        # occasional global jitter to avoid stagnation
        if it % 10000 == 0:
            noise = rng.normal(scale=0.01, size=n)
            a = np.clip(a + noise, 0.0, None)
            a /= a.sum()
            conv = _conv_self_fft(a)

    # final compression of negligible entries
    thresh = 1e-12
    mask = best_seq > thresh
    if mask.sum() < best_seq.size:
        comp = best_seq[mask]
        comp = comp / comp.sum()
        comp_obj = _objective(comp)
        if comp_obj < best_obj:
            best_seq, best_obj = comp, comp_obj
            print(f"[SA compress] reduced length → {comp.size}, O={best_obj:.6f}")

    return best_seq.tolist(), best_obj
