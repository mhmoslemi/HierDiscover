#!/usr/bin/env python
"""
Propose a non‑negative step function (list of heights) that minimises

        2·n·max_k (a*a)[k] / (Σ a_i)²

The algorithm consists of three stages:

1. **Fast initial sweep** – a few parametric families (uniform,
   power‑law, exponential, power‑law × exponential, beta‑shaped and
   Dirichlet seeds) are evaluated using a cheap FFT based score
   (`2·n·max(conv)`) assuming Σa_i=1. The best candidate is kept.

2. **Refinement** – the candidate is re‑parameterised by logits
   `z_i` with a soft‑max mapping `a_i = exp(z_i)/∑exp(z)`.  A smooth
   approximation of the `max` is used:
        f(z) = (1/λ)·log Σ_k exp(λ·(a*a)[k])
   with a temperature λ that is slowly increased.
   The gradient of f w.r.t. the logits is obtained analytically using
   FFT‑based convolutions, and an Adam optimiser (exponentiated‑gradient
   on the simplex) drives the descent.  Restarts with small random
   perturbations are performed when progress stalls.

3. **Exact evaluation** – the best vector found (still normalised to
   sum = 1) is passed to the supplied ``evaluate_sequence`` for the
   final score, which is returned as a plain Python list.

All helper utilities are defined at top‑level; the routine respects a
user‑supplied time budget (default 1000 s) and always returns a feasible
sequence.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float) -> np.ndarray:
    """Return a normalised power‑law a_i ∝ (i+1)^‑p (Σa_i = 1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = i ** (-p)
    seq /= seq.sum()
    return seq


def _exponential_sequence(n: int, lam: float) -> np.ndarray:
    """Return a normalised exponential a_i ∝ exp(-lam·i) (Σa_i = 1)."""
    i = np.arange(n, dtype=np.float64)
    seq = np.exp(-lam * i)
    seq /= seq.sum()
    return seq


def _beta_sequence(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑shaped a_i ∝ x_i^α (1‑x_i)^β, x_i = i/(n+1)."""
    x = (np.arange(1, n + 1, dtype=np.float64) / (n + 1))
    seq = (x ** alpha) * ((1.0 - x) ** beta)
    s = seq.sum()
    if s == 0.0:
        return None
    seq /= s
    return seq


def _dirichlet_sequence(n: int, conc: float) -> np.ndarray:
    """Sample a Dirichlet vector with a single concentration parameter."""
    g = np.random.gamma(conc, 1.0, size=n)
    if g.sum() == 0.0:
        return None
    return g / g.sum()


def _next_pow2(x: int) -> int:
    """Small helper: next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _cheap_score(vec: np.ndarray, n: int, fft_len: int) -> float:
    """
    Cheap score for a normalised vector ( Σ vec = 1 ):
        2·n·max( a * a )
    where * is the full linear convolution.
    """
    fa = np.fft.rfft(vec, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    maxb = conv.max()
    return 2.0 * n * maxb, conv


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that
    minimises ``evaluate_sequence``.  The routine adapts its schedule to
    the available time budget and always returns a feasible sequence.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a safety margin for the final exact evaluation
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # Choose a working length.
    # --------------------------------------------------------------
    try:
        # ``height_sequence_1`` may be available from a previous run.
        prev = globals().get("height_sequence_1", None)
        if isinstance(prev, list):
            prev_arr = np.asarray(prev, dtype=np.float64)
            n0 = prev_arr.size
        else:
            prev_arr = None
            n0 = None
    except Exception:
        prev_arr = None
        n0 = None

    # Use a modestly larger size than the previous run – up to 20 k.
    n = max(5000, n0 if n0 is not None else 0)
    n = min(max(n, 5000), 20000)            # 5 k ≤ n ≤ 20 k

    # FFT length needed for the longest convolution (a * w).
    fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # 0. Initialise with cheap families
    # --------------------------------------------------------------
    best_a = None
    best_score = float("inf")

    # uniform baseline
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    score, _ = _cheap_score(uniform, n, fft_len)
    best_a, best_score = uniform, score

    # previous construction, if any (and same length)
    if prev_arr is not None and prev_arr.size == n:
        cand = prev_arr.copy()
        cand /= cand.sum()
        score, _ = _cheap_score(cand, n, fft_len)
        if score < best_score:
            best_a, best_score = cand, score

    # ---- power‑law sweep ------------------------------------------------
    for p in np.linspace(0.4, 1.4, 81):
        if time.time() >= deadline:
            break
        cand = _power_law_sequence(n, p)
        score, _ = _cheap_score(cand, n, fft_len)
        if score < best_score:
            best_a, best_score = cand.copy(), score

    # ---- pure exponential sweep -----------------------------------------
    for lam in np.linspace(0.0, 4.0, 71):
        if time.time() >= deadline:
            break
        cand = _exponential_sequence(n, lam)
        score, _ = _cheap_score(cand, n, fft_len)
        if score < best_score:
            best_a, best_score = cand.copy(), score

    # ---- power‑law × exponential (coarse grid) -------------------------
    for p in np.linspace(0.5, 1.0, 11):
        for lam in np.linspace(0.0, 2.0, 11):
            if time.time() >= deadline:
                break
            i = np.arange(n, dtype=np.float64)
            cand = ((i + 1.0) ** (-p)) * np.exp(-lam * i)
            cand /= cand.sum()
            score, _ = _cheap_score(cand, n, fft_len)
            if score < best_score:
                best_a, best_score = cand.copy(), score

    # ---- beta families --------------------------------------------------
    for alpha in np.linspace(0.2, 10.0, 21):
        for beta in np.linspace(0.2, 10.0, 21):
            if time.time() >= deadline:
                break
            cand = _beta_sequence(n, alpha, beta)
            if cand is None:
                continue
            score, _ = _cheap_score(cand, n, fft_len)
            if score < best_score:
                best_a, best_score = cand.copy(), score

    # ---- a few random Dirichlet seeds -----------------------------------
    for conc in [0.001, 0.01, 0.1, 0.5, 1.0, 3.0]:
        if time.time() >= deadline:
            break
        cand = _dirichlet_sequence(n, conc)
        if cand is None:
            continue
        score, _ = _cheap_score(cand, n, fft_len)
        if score < best_score:
            best_a, best_score = cand.copy(), score

    print(f"[Init] best cheap score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 1. Refine the best candidate with Adam‑based mirror descent
    # --------------------------------------------------------------
    # Initialise logits (log‑softmax parameters)
    eps = 1e-30
    logits = np.log(best_a + eps)

    # Adam state
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    alpha_lr = 0.08            # base learning rate for Adam

    # Temperature schedule for the smooth max
    lam_start = 5.0
    lam_end = 180.0
    lam_ramp_iters = 200_000   # number of iterations over which λ grows

    # bookkeeping
    best_iter_a = best_a.copy()
    best_iter_score = best_score
    last_improve = 0
    iter_cnt = 0

    # pre‑compute index slices that will be reused
    conv_len = 2 * n - 1
    rev_start = n - 1
    rev_end = 2 * n - 1   # exclusive

    while time.time() < deadline:
        iter_cnt += 1

        # --- softmax to obtain a (always Σa = 1) --------------------
        max_log = logits.max()
        a = np.exp(logits - max_log)
        a /= a.sum()

        # --- full convolution a*a (via FFT) -------------------------
        FA = np.fft.rfft(a, fft_len)
        conv_full = np.fft.irfft(FA * FA, fft_len)[:conv_len]
        maxb = conv_full.max()
        cheap = 2.0 * n * maxb

        # keep the best ever seen during refinement
        if cheap < best_iter_score - 1e-13:
            best_iter_score = cheap
            best_iter_a = a.copy()
            last_improve = iter_cnt
            print(f"[Iter {iter_cnt:07d}] NEW BEST cheap = {cheap:.6f}")

        # --- temperature λ (smooth max) ------------------------------
        lam = lam_start + (lam_end - lam_start) * min(1.0,
                                                     iter_cnt / lam_ramp_iters)

        # --- softmax weights over convolution entries -----------------
        #   w_k = exp(λ·conv_k) / Σ exp(λ·conv)
        conv_max = conv_full.max()
        exp_weights = np.exp(lam * (conv_full - conv_max))
        w = exp_weights / exp_weights.sum()          # shape (conv_len,)

        # --- gradients of the smoothed objective --------------------
        #   g = ∇_a f = Σ_k w_k ∇_a conv_k
        #   where ∇_a conv_k = a shifted left + a shifted right
        FW = np.fft.rfft(w, fft_len)
        w_rev = w[::-1]
        FW_rev = np.fft.rfft(w_rev, fft_len)

        # a * w  (full convolution) – prefix gives the “right‑shift” part
        conv1_full = np.fft.irfft(FA * FW, fft_len)
        conv1 = conv1_full[:n]                         # length n

        # a * rev(w) – slice gives the “left‑shift” part
        conv_rev_full = np.fft.irfft(FA * FW_rev, fft_len)
        conv_rev_slice = conv_rev_full[rev_start:rev_end]   # length n
        conv_rev_term = conv_rev_slice[::-1]                # reversed

        g = conv1 + conv_rev_term               # gradient w.r.t. a

        # --- gradient wrt logits (mirror‑descent on the simplex) -----
        dot_ag = np.dot(a, g)
        grad_logits = a * (g - dot_ag)          # shape (n,)

        # --- Adam update ---------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
        m_hat = m / (1.0 - beta1 ** iter_cnt)
        v_hat = v / (1.0 - beta2 ** iter_cnt)
        step = alpha_lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # --- occasional restart when stagnating ----------------------
        if iter_cnt - last_improve > 250_000:
            # small random perturbation around the best iterate so far
            perturb = 0.02 * np.random.rand(n)
            logits = np.log(best_iter_a + eps) + perturb
            m.fill(0.0)
            v.fill(0.0)
            last_improve = iter_cnt
            print(f"[Iter {iter_cnt:07d}] Restart after stagnation.")

    # --------------------------------------------------------------
    # 2. Final exact evaluation (uses the provided ``evaluate_sequence``)
    # --------------------------------------------------------------
    final_seq = best_iter_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:      # defensive – should never happen
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # --------------------------------------------------------------
    # 3. Return the best sequence found.
    # --------------------------------------------------------------
    return final_seq
