    n = len(a)
    conv = np.convolve(a, a)
    maxconv = conv.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * maxconv / (s * s)

"""
Propose a non‑negative step‑function that (approximately) minimises

    O(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

Algorithm
---------
The optimiser works in three stages and stays fully on the simplex
by using an exponentiated‑gradient (EG) update (i.e. a ← softmax(loga‑η∇)).
All operations are vectorised and use FFT‑based convolutions.

1️⃣  **Flattening phase** – minimise the variance of the convolution
    vector c = a★a (∑(c‑μ)²) with EG.  This drives the whole
    convolution towards the same value and yields a good starting shape.

2️⃣  **Soft‑max phase** – gradually anneal a smoothed surrogate of the
    max‑operator:
        fα(a) = (2·n/α)·log∑ₖ e^{α·cₖ}
    The gradient of fα is 2·corr(w,a) with w ∝ exp(α·c).  We use EG again,
    increasing α from 5 up to 8000 so that the surrogate converges to the
    true peak.

3️⃣  **Local stochastic refinement** – start from the best vector found,
    perform many Monte‑Carlo “mass‑transfer’’ moves together with explicit
    sub‑gradient steps on the true max‑convolution.  A simple Metropolis
    rule with slowly cooling temperature is used to escape plateaus.

The three phases share the allotted 1000 s time budget (≈45 % / 40 % / 15 %).
The best sequence seen during the whole run is returned.  All numbers are
non‑negative, automatically normalised, and satisfy the constraints of
`evaluate_sequence`.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Simple soft‑max (log‑sum‑exp) utilities
def _softmax(log_vals: np.ndarray) -> np.ndarray:
    """Return soft‑max of log_vals (guarantees sum == 1)."""
    m = np.max(log_vals)
    exp_vals = np.exp(log_vals - m)
    return exp_vals / exp_vals.sum()


# ----------------------------------------------------------------------
# Full linear convolution a★a using FFT of length L (L ≥ 2·n‑1)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution of a with itself, length 2·n‑1, via FFT length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Objective on the simplex (sum(a)=1)
def _objective_maxconv(a: np.ndarray, n: int, L: int) -> float:
    """Exact O(a) value when sum(a)=1."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Gradient of the variance‐of‑convolution objective
def _grad_variance(a: np.ndarray, conv: np.ndarray, n: int, L: int) -> np.ndarray:
    """
    Gradient of   f(a) = Σ_k (conv_k - μ)²   where μ = 1/(2n‑1)  (for sum(a)=1).
    Closed form: g = 4·corr(conv‑μ , a) .
    """
    mu = 1.0 / (2 * n - 1)
    diff = conv - mu
    a_rev = a[::-1]

    fft_diff = np.fft.rfft(diff, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_diff * fft_a_rev, n=L)

    start = n - 1
    grad = 4.0 * corr_rev[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Gradient of smooth max (log‑sum‑exp) surrogate
def _grad_logsumexp(a: np.ndarray, conv: np.ndarray, alpha: float,
                    L: int, lam: float = 0.0) -> np.ndarray:
    """
    ∇_a  ( (2·n/α)·log∑_k e^{α·conv_k}  + λ·∑ a_i log a_i )
    The first term is 2·corr(w,a) with w ∝ exp(α·conv).
    """
    # soft‑max weights over the convolution entries
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / exp_scaled.sum()          # shape (2n‑1,)

    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)

    n_len = a.size
    grad_corr = 2.0 * corr_rev[n_len - 1:n_len - 1 + n_len]

    if lam == 0.0:
        return grad_corr
    # tiny entropy term (helps avoid exact zeros)
    eps = 1e-20
    grad_ent = lam * (np.log(a + eps) + 1.0)
    return grad_corr + grad_ent

                    g /= ks.size
                eta_sub = 0.08 * (1.0 - it_loc / max_local_iters)
                a = a - eta_sub * g
                a = np.clip(a, 0.0, None)
                a = a / a.sum()
                conv = _conv_fft(a, L)
                cur_obj = 2.0 * n * conv.max()
                if cur_obj < best_score:
                    best_score = cur_obj
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] sub‑grad improved O={best_score:.6f}")

            # ---------------------------------------------------------
            # 3) Smoothing move (occasional)
            if it_loc % 6000 == 0 and it_loc > 0:
                kernel = np.array([0.125, 0.25, 0.25, 0.25, 0.125])
                a_smooth = np.convolve(a, kernel, mode="same")
                a = 0.85 * a + 0.15 * a_smooth
                a = np.clip(a, 0.0, None)
                a = a / a.sum()

            # ---------------------------------------------------------
            # 4) Temperature decay
            T *= 0.9997
            it_loc += 1

            # ---------------------------------------------------------
            # Restart if stagnated
            if no_improve >= no_improve_limit:
                break

    # ------------------------------------------------------------------
    # Final sanity check (uses the provided evaluate_sequence)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:
        print("[final] evaluate_sequence failed:", e)

    # Return plain Python list as required
    return best_seq.tolist()
```

This implementation introduces an exponentiated‑gradient optimiser (which respects the simplex without an explicit projection), a longer annealing schedule for the log‑sum‑exp surrogate, and a more aggressive stochastic refinement phase.  
All three phases share the 1000 s budget, and the best sequence encountered is returned.
