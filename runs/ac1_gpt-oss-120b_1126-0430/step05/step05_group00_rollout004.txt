def _softmax(z):
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()

def _logits_from_distribution(a, eps=1e-16):
    return np.log(a + eps)

def _fft_len_for_corr(n):
    return 1 << ((2 * n - 1).bit_length())
    # using 2*n to fit conv length.

def _conv_max(a):
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[:2 * n - 1]
    return float(conv.max())

def _conv_max_and_grad(a, eps_factor):
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[:2 * n - 1]
    max_conv = float(conv.max())
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv
    w = np.exp(shifted / eps)

    if w_sum == 0.0 or np.isnan(w_sum):

    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

#!/usr/bin/env python
"""
Arcsine‑bin construction for the max‑convolution problem.

The continuous arcsine density f(x)=1/(π√(x(1‑x))) satisfies
    (f*f)(y)=½  on [0,2].
If we discretise f on a uniform grid by *exact* bin probabilities
    a_j = F((j+1)/n) – F(j/n)   with  F(x)=2/π·arcsin(√x),
then the self‑convolution of the resulting vector is (almost) uniform,
so the evaluation metric 2·n·max(a*a)/(sum a)² is ≈1.
We compute this vector, optionally polish it with a few Adam steps on the
simplex (mirrored‑gradient on the logits) and return the best sequence
found before the time budget expires.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Return logits that map to the probability vector a via soft‑max."""
    return np.log(a + eps)


def _fft_len_for_conv(n: int) -> int:
    """
    Smallest power‑of‑two ≥ 2·n‑1,
    sufficient for a linear self‑convolution of length (2·n‑1).
    """
    return 1 << ((2 * n - 1).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a (no gradient)."""
    n = a.shape[0]
    L = _fft_len_for_conv(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Compute the smoothed maximum of a*a and the corresponding gradient.

    Parameters
    ----------
    a : np.ndarray
        Probability vector (non‑negative, sum = 1).
    eps_factor : float
        Multiplicative factor for the smoothing constant:
        eps = eps_factor * max(conv).

    Returns
    -------
    max_conv : float
        Maximum entry of the self‑convolution.
    grad_a : np.ndarray
        Gradient of 4·n·smooth_max w.r.t. a (already multiplied by 4·n).
    """
    n = a.shape[0]
    L = _fft_len_for_conv(n)

    a_fft = np.fft.rfft(a, L)

    # linear self‑convolution (length 2·n‑1)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # smoothed soft‑max weights (log‑sum‑exp trick)
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv               # improve stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum

    # cross‑correlation a ⋆ w (linear, length 3·n‑2)
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # gradient (4·n multiplied)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a

# Main entry point required by the platform

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that minimises the evaluation metric.

    1. Build the exact discrete arcsine distribution on a uniform grid.
    2. Run a very short Adam optimisation on the simplex (optional)
       to squeeze a few extra digits of improvement.
    3. Return the best sequence found.
    """
    np.random.seed(seed)
    start = time.time()
    # leave a small safety margin for final logging and return
    deadline = start + budget_s - 2.0

    # ------------------------------------------------------------------
    # 0. Choose the length of the vector.
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
        print(f"Using external length n = {n}")
    except Exception:
        n = 20000
        print(f"No external sequence – using default length n = {n}")

    # ------------------------------------------------------------------
    # 1. Exact discretisation of the arcsine distribution (bin probabilities).
    # ------------------------------------------------------------------
    i = np.arange(n, dtype=np.float64)
    x0 = i / n
    x1 = (i + 1) / n

    # CDF of the arcsine law: F(x) = (2/π)·arcsin(√x)
    sqrt0 = np.sqrt(np.clip(x0, 0.0, 1.0))
    sqrt1 = np.sqrt(np.clip(x1, 0.0, 1.0))
    F0 = (2.0 / np.pi) * np.arcsin(sqrt0)
    F1 = (2.0 / np.pi) * np.arcsin(sqrt1)

    a = F1 - F0                     # probability mass of each bin, sum = 1
    a_sum = a.sum()
    if not np.isfinite(a_sum) or a_sum <= 0.0:
        raise RuntimeError("Failed to generate arcsine bins.")
    a /= a_sum                      # enforce exact normalisation (numerical safety)

    # ------------------------------------------------------------------
    # 2. Initial cheap evaluation.
    # ------------------------------------------------------------------
    best_a = a.copy()
    max_conv = _conv_max(best_a)
    best_score = 2.0 * n * max_conv
    print(f"Initial score (2·n·max_conv) = {best_score:.8f}")

    # ------------------------------------------------------------------
    # 3. (Optional) Very short Adam refinement on the simplex.
    #    This step is cheap (a few thousand iterations) and can shave
    #    a few 1e‑4 off the score.
    # ------------------------------------------------------------------
    # Parameters are deliberately conservative to avoid overshooting.
    z = _logits_from_distribution(best_a)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_alpha = 0.6
    eps_factor = 0.12
    eps_factor_min = 5e-4
    eps_factor_decay = 0.9993

    it = 0
    max_iters = 50000

    while time.time() < deadline and it < max_iters:
        it += 1
        a_cur = _softmax(z)

        max_conv, grad_a = _conv_max_and_grad(a_cur, eps_factor)
        cheap_score = 2.0 * n * max_conv

        if cheap_score < best_score:
            best_score = cheap_score
            best_a = a_cur.copy()
            if it % 1000 == 0:
                print(f"Iter {it:6d}: improved score = {best_score:.8f}")

        # Mirror‑descent gradient on the logits (soft‑max parameterisation)
        mu = np.dot(grad_a, a_cur)
        grad_z = a_cur * (grad_a - mu)

        # Adam bias‑corrected update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # Inverse‑sqrt step‑size schedule
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step = - alpha_t * m_hat / (np.sqrt(v_hat) + eps_adam)
        z = z + step
        # keep logits bounded so that exp(z) stays safe
        z = np.clip(z, -60.0, 60.0)

        # Gradually tighten the smoothing constant
        eps_factor = max(eps_factor_min, eps_factor * eps_factor_decay)

        # Tiny random jitter every few thousand steps (helps escape plateaus)
        if it % 5000 == 0 and np.random.rand() < 0.07:
            z += 0.004 * np.random.randn(n)

        # Early exit if we already have a very good score
        if it % 20000 == 0 and best_score < 1.02:
            print(f"Early stop at iteration {it}, best_score={best_score:.8f}")
            break

    # ------------------------------------------------------------------
    # 4. Return the best sequence (plain Python list).  One final exact
    #    evaluation is logged – it does not affect the returned value.
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        exact_score = evaluate_sequence(final_seq)
        print(f"Exact evaluation (after optimisation): {exact_score:.8f}")
    except Exception as exc:
        print(f"Exact evaluation failed: {exc}")

    return final_seq
