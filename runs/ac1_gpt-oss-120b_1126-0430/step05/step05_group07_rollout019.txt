### Compute O function quickly:

def compute_O(a):
    a = np.maximum(0, a)  # clip negative
    a = np.minimum(1000.0, a)  # clip high
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    conv = np.convolve(a, a)
    max_b = conv.max()
    n = a.size
    return 2.0 * n * max_b / (sum_a*sum_a)
```

def compute_O_fft(a):
    n = a.shape[0]
    L = 1 << ((2*n-1).bit_length())
    fft_a = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft_a*fft_a, n=L)[:2*n-1]

    sum_a = a.sum()

def generate_linear_decreasing(n, p):

def refine_candidate(a_init, remaining_time):
    start = time.time()
    n = a_init.size
    L = 1 << ((2*n - 1).bit_length())
    # initialize v = log(a_init + epsilon)
    eps = 1e-12
    v = np.log(a_init + eps)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    epsadam = 1e-12
    # schedule for alpha
    max_iter = 200000
    it = 0
    best_a = a_init.copy()
    best_score = compute_obj(best_a)
    while time.time() - start < remaining_time * 0.9 and it < max_iter:
        # compute a from v
        max_v = np.max(v)
        exp_v = np.exp(v - max_v)   # shift for stability
        a = exp_v / exp_v.sum()
        # compute conv
        fft_a = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(fft_a * fft_a, n=L)[:2*n-1]
        # compute surrogate gradient
        # compute log-sum-exp gradient as before
        # use high alpha that grows with it
        frac = it / max_iter
        alpha = 5.0 + (20000.0 - 5.0) * frac  # final ~20000
        # compute w = exp(alpha*conv) normalized
        scaled = alpha * conv
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        w = exp_scaled / np.sum(exp_scaled)
        # correlation gradient part: 2 * corr(w, a)
        a_rev = a[::-1]
        fft_w = np.fft.rfft(w, n=L)
        fft_a_rev = np.fft.rfft(a_rev, n=L)
        corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
        grad_corr = 2.0 * corr_rev[n-1:n-1+n]
        # small entropy gradient maybe omitted because we have exponentiated parametrization maintains positivity
        grad_a = grad_corr # ignoring entropy
        # compute expected gradient for v (softmax chain)
        E = np.dot(a, grad_a)
        grad_v = a * (grad_a - E)
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v * grad_v)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v2_hat = v2 / (1.0 - beta2 ** (it + 1))
        # learning rate schedule: maybe 0.1 / sqrt(it+1)
        lr = 0.5 / np.sqrt(it + 1)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + epsadam)
        # evaluate objective (exact O)
        obj = 2.0 * n * conv.max() / (a.sum() * a.sum())
        if obj < best_score:
            best_score = obj
            best_a = a.copy()
        it += 1
    return best_a, best_score

"""Propose a non‑negative step‑function by global shape search + mirror‑descent refinement.

The algorithm proceeds in three stages:
1. A cheap sweep over several parametric families (uniform, power‑law, exponential,
   Beta‑distribution, Gaussian, classic window functions and random sparse patterns).
   The best few candidates are kept.
2. Each top candidate is refined on the simplex using a mirror‑descent (Adam) optimizer
   on a smooth log‑sum‑exp surrogate of the max‑convolution.  The optimization works
   directly on log‑weights (soft‑max) so the non‑negativity and ∑a = 1 constraints
   are satisfied automatically.
3. A short stochastic “mass‑transfer” local search is performed to polish the result.

The whole procedure respects the 1000 s budget and returns the best sequence found.
"""

import time
import math
import numpy as np

def _compute_obj(a: np.ndarray, L: int) -> float:
    """Return the evaluation‑function value O for a (already non‑negative)."""
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    n = a.shape[0]
    fft_a = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft_a * fft_a, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (sum_a * sum_a))

def _generate_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)

def _generate_power_decreasing(n: int, p: float) -> np.ndarray:
    seq = (np.arange(n)[::-1] + 1.0) ** p
    return seq / seq.sum()

def _generate_exponential(n: int, r: float) -> np.ndarray:
    seq = r ** np.arange(n)
    return seq / seq.sum()

def _generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n) + 0.5) / n
    # avoid zero**negative; use np.where for stability
    seq = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    seq = np.nan_to_num(seq, nan=0.0, posinf=0.0, neginf=0.0)
    total = seq.sum()
    if total <= 0:
        return _generate_uniform(n)
    return seq / total

def _generate_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n)
    centre = (n - 1) / 2.0
    seq = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    total = seq.sum()
    if total <= 0:
        return _generate_uniform(n)
    return seq / total

def _generate_hann(n: int) -> np.ndarray:
    i = np.arange(n)
    seq = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return seq / seq.sum()

def _generate_inverse_hann(n: int) -> np.ndarray:
    i = np.arange(n)
    seq = 0.5 + 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return seq / seq.sum()

def _generate_cosine_bell(n: int) -> np.ndarray:
    i = np.arange(n)
    seq = 0.5 * (1.0 + np.cos(np.pi * (i - (n - 1) / 2.0) / ((n - 1) / 2.0)))
    seq = np.clip(seq, 0.0, None)
    return seq / seq.sum()

def _generate_random_sparse(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    seq = np.zeros(n)
    idx = rng.choice(n, size=m, replace=False)
    seq[idx] = 1.0 / m
    return seq

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Search for a step‑function that (approximately) minimises the given O."""
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.3   # safety margin

    # -----------------------------------------------------------------
    # Determine problem size -------------------------------------------------
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
        n = base.size
        print(f"[init] using supplied sequence of length n={n}")
    except NameError:
        n = 4600
        base = np.full(n, 1.0 / n)
        print(f"[init] no supplied sequence – using default n={n}")

    L = 1 << ((2 * n - 1).bit_length())   # FFT length – power of two

    # -----------------------------------------------------------------
    # Helper to keep the current best -------------------------------------------------
    best_seq = base / base.sum()
    best_score = _compute_obj(best_seq, L)
    print(f"[init] initial O={best_score:.6f}")

    # -----------------------------------------------------------------
    # 1️⃣  Global shape sweep (fast, cheap) ------------------------------------
    candidates = []

    # uniform
    candidates.append(_generate_uniform(n))

    # already‑known start (if any)
    if 'height_sequence_1' in globals():
        candidates.append(base / base.sum())

    # power‑law decreasing
    for p in [0.0, 0.5, 1.0, 2.0, 4.0]:
        candidates.append(_generate_power_decreasing(n, p))

    # exponential decay
    for r in [0.99, 0.995, 0.998, 0.999]:
        candidates.append(_generate_exponential(n, r))

    # Beta family (α,β)
    alphas = [0.2, 0.5, 1.0, 2.0, 3.0]
    for a in alphas:
        for b in alphas:
            candidates.append(_generate_beta(n, a, b))

    # Gaussian centred
    for sigma_frac in [0.1, 0.15, 0.2, 0.25, 0.3]:
        sigma = sigma_frac * n
        candidates.append(_generate_gaussian(n, sigma))

    # classic windows
    candidates.append(_generate_hann(n))
    candidates.append(_generate_inverse_hann(n))
    candidates.append(_generate_cosine_bell(n))

    # random sparse patterns
    rng = np.random.default_rng(seed + 12345)
    for m in [5, 10, 20, 50, 100, 200]:
        candidates.append(_generate_random_sparse(n, m, rng))

    # jittered uniform (10 variations)
    for _ in range(10):
        seq = np.full(n, 1.0 / n)
        seq = seq + 0.01 * np.random.randn(n)
        seq = np.clip(seq, 0.0, None)
        if seq.sum() == 0:
            continue
        seq = seq / seq.sum()
        candidates.append(seq)

    print(f"[sweep] evaluating {len(candidates)} candidate families")

    for idx, cand in enumerate(candidates):
        cur = _compute_obj(cand, L)
        if cur < best_score:
            best_score = cur
            best_seq = cand.copy()
            print(f"[sweep] candidate #{idx:03d} improved O={best_score:.6f}")
        # bail out early if time is tight
        if time.time() > deadline - 120:
            break

    # -----------------------------------------------------------------
    # 2️⃣  Mirror‑descent refinement on the simplex (Adam on log‑weights)
    # Allocate most of the remaining budget to this phase.
    remaining = deadline - time.time()
    if remaining <= 0:
        print("[refine] no time left after sweep")
        return best_seq.tolist()

    # Initialise log‑weights (soft‑max parametrisation)
    eps = 1e-14
    v = np.log(best_seq + eps)          # shape (n,)
    m_adam = np.zeros_like(v)
    v_adam = np.zeros_like(v)
    beta1, beta2 = 0.9, 0.999
    max_iter = 250_000
    it = 0
    print(f"[refine] start mirror‑descent, remaining budget ≈{remaining:.1f}s")

    while time.time() < deadline - 30 and it < max_iter:
        # a = softmax(v)
        max_v = np.max(v)
        exp_v = np.exp(v - max_v)
        a = exp_v / exp_v.sum()

        # full convolution via FFT
        fft_a = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(fft_a * fft_a, n=L)[: 2 * n - 1]

        # exact objective for bookkeeping
        cur_obj = 2.0 * n * conv.max() / (a.sum() * a.sum())
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[refine] iter {it:,} new best O={best_score:.6f}")

        # ------- smooth max‑convolution surrogate (log‑sum‑exp) ----------
        frac = it / max_iter
        alpha = 5.0 + (20_000.0 - 5.0) * frac   # anneal to a large value
        scaled = alpha * conv
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        w = exp_scaled / exp_scaled.sum()          # shape (2n‑1,)

        # correlation gradient: 2·corr(w , a)
        a_rev = a[::-1]
        fft_w = np.fft.rfft(w, n=L)
        fft_a_rev = np.fft.rfft(a_rev, n=L)
        corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)

"""
Propose a non‑negative step‑function by a three‑stage search:

1️⃣  Global shape sweep – generate candidates from several parametric families
    (uniform, power‑law, exponential, Beta‑distribution, Gaussian, classic
    windows and random sparse patterns).  The best few are kept.

2️⃣  Mirror‑descent refinement – optimise the candidate on the simplex using
    Adam on log‑weights (soft‑max parametrisation).  The loss is a smooth
    log‑sum‑exp surrogate of the max‑convolution; a large annealing factor
    α drives the surrogate towards the true max.  This respects a≥0 and ∑a=1
    automatically.

3️⃣  Stochastic polishing – a short Monte‑Carlo “mass‑transfer” local search
    (randomly move a fraction of the mass between two positions) with occasional
    jitter, accepting only improvements.  The best sequence found is returned.

All steps respect the 1000 s budget and use FFT‑based convolutions for speed.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _compute_obj(a: np.ndarray, L: int) -> float:
    """Return the evaluation‑function value O for a (non‑negative)."""
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    n = a.shape[0]
    fft_a = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft_a * fft_a, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (sum_a * sum_a))

# Parametric families (all return a normalised distribution, sum = 1)

def _gen_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _gen_power_decreasing(n: int, p: float) -> np.ndarray:
    seq = (np.arange(n)[::-1] + 1.0) ** p
    return seq / seq.sum()


def _gen_exponential(n: int, r: float) -> np.ndarray:
    seq = r ** np.arange(n)
    return seq / seq.sum()


def _gen_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n) + 0.5) / n
    seq = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    seq = np.nan_to_num(seq, nan=0.0, posinf=0.0, neginf=0.0)
    tot = seq.sum()
    if tot <= 0:
        return _gen_uniform(n)
    return seq / tot


def _gen_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n)
    centre = (n - 1) / 2.0
    seq = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    tot = seq.sum()
    if tot <= 0:
        return _gen_uniform(n)
    return seq / tot


def _gen_hann(n: int) -> np.ndarray:
    i = np.arange(n)
    seq = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return seq / seq.sum()


def _gen_inverse_hann(n: int) -> np.ndarray:
    i = np.arange(n)
    seq = 0.5 + 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return seq / seq.sum()


def _gen_cosine_bell(n: int) -> np.ndarray:
    i = np.arange(n)
    seq = 0.5 * (1.0 + np.cos(np.pi * (i - (n - 1) / 2.0) / ((n - 1) / 2.0)))
    seq = np.clip(seq, 0.0, None)
    return seq / seq.sum()


def _gen_random_sparse(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    seq = np.zeros(n)
    idx = rng.choice(n, size=m, replace=False)
    seq[idx] = 1.0 / m
    return seq
