def _pairwise_local_search(a, n, L, budget_deadline):
    best_a = a.copy()
    best_obj = _objective(best_a, n, L)
    cur_a = a.copy()
    cur_obj = best_obj
    while time.time() < budget_deadline:
        # compute subgradient g
        conv = _conv_fft(cur_a, L)
        g = _subgradient_max_conv(cur_a, conv)
        # find source and dest indices
        src = np.argmax(g)
        dst = np.argmin(g)
        if cur_a[src] < 1e-12:
            # find next src?
            src = np.random.choice(np.where(cur_a > 1e-12)[0])
        max_transfer = cur_a[src]
        # try different fractions
        improved = False
        for frac in [0.5, 0.2, 0.1, 0.05, 0.02, 0.01]:
            delta = max_transfer * frac
            if delta < 1e-12:
                continue
            cand = cur_a.copy()
            cand[src] -= delta
            cand[dst] += delta
            # ensure >=0 (it will be)
            obj = _objective(cand, n, L)
            if obj < cur_obj - 1e-9:
                cur_a = cand
                cur_obj = obj
                improved = True
                break
        if not improved:
            # random perturb to escape
            i = np.random.randint(n)
            j = np.random.randint(n)
            while j == i:
                j = np.random.randint(n)
            delta = cur_a[i] * np.random.rand() * 0.05
            cand = cur_a.copy()
            cand[i] -= delta
            cand[j] += delta
            obj = _objective(cand, n, L)
            if obj < cur_obj:
                cur_a = cand
                cur_obj = obj
                improved = True
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = cur_a.copy()
    return best_a, best_obj

"""
Proposes a non‑negative step‑function sequence that minimises  

    O(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The search mixes several complementary optimisation stages that all
respect a global 1000 s wall‑time budget:

0️⃣  Random Dirichlet restarts – a few very cheap Adam runs on a smooth
    log‑sum‑exp surrogate to obtain a diverse set of seeds.

1️⃣  Adam on the smooth surrogate – long optimisation with a schedule for
    the soft‑max temperature α, Euclidean projection onto the probability
    simplex after each step and occasional pruning of negligible entries.

2️⃣  Exact projected sub‑gradient descent – uses the true max‑convolution
    objective with a back‑tracking line‑search for guaranteed descent.

3️⃣  Simulated annealing – mass‑transfer moves that target the current
    maximal convolution entry; Metropolis acceptance with a slowly
    cooling temperature and periodic exact sub‑gradient steps keep the walk
    directed.

4️⃣  Deterministic pair‑wise local search – repeatedly moves a fraction
    of probability mass from the index with the largest sub‑gradient to
    the index with the smallest one; a simple line‑search over a few
    fractions is performed and the move is kept only if the exact objective
    improves.  This stage cleans up the final few percent of the objective.

The best sequence seen in any stage (across a few different problem
sizes) is returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, sum=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # all non‑positive – return the centre of the simplex
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


def _objective(a: np.ndarray, n: int, L: int) -> float:
    """Exact objective 2·n·max(conv(a,a)), assuming sum(a)=1."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. a."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


def _smooth_gradient(a: np.ndarray, conv: np.ndarray,
                     alpha: float, L: int) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate:
        (2·n/α)·log Σ_k exp(α·conv_k).
    It reduces to 2·corr(w, a) where w_k ∝ exp(α·conv_k).
    """
    # soft‑max weights
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)      # numerical stability
    w = exp_scaled / exp_scaled.sum()

    # correlation w ⋆ a  (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n_len = a.size
    grad = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]
    return grad

# ----------------------------------------------------------------------


def _adam_phase(a: np.ndarray, n: int, L: int,
                max_iters: int, deadline: float,
                alpha_start: float = 5.0,
                alpha_end: float = 20000.0,
                lr0: float = 0.6,
                beta1: float = 0.9,
                beta2: float = 0.999,
                eps: float = 1e-12,
                prune_every: int = 5000,
                prune_percentile: float = 0.5) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Adam optimiser on the smooth log‑sum‑exp surrogate.
    Returns (final a, best objective value seen, best a).
    """
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_obj = _objective(a, n, L)

    total_iters = max_iters
    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _conv_fft(a, L)
        cur_obj = _objective(a, n, L)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            if it % 2000 == 0:
                print(f"[adam] iter {it} – new best O={best_obj:.6f}")

        # exponential schedule for α – sharper towards the end
        frac = it / total_iters
        alpha = alpha_start * (alpha_end / alpha_start) ** frac

        grad = _smooth_gradient(a, conv, alpha, L)

        # Adam updates (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        lr = lr0 / math.sqrt(it + 1)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape flat plateaus
        if np.random.rand() < 0.001:
            a_new += np.random.randn(n) * 1e-4

        a_new = _project_to_simplex(a_new)

        # prune tiny entries every few thousand steps (optional)
        if prune_every and it % prune_every == 0 and it > 0:
            thresh = np.percentile(a_new, prune_percentile)
            a_new[a_new < thresh] = 0.0
            a_new = _project_to_simplex(a_new)

        a = a_new

    return a, best_obj, best_a

    """

    for it in range(max_iters):

        cur_obj = _objective(a, n, L)

                print(f"[sub‑grad] iter {it} – new best O={best_obj:.6f}")

        g = _subgradient_max_conv(a, conv)

        # geometric decay of the step size
        eta = eta0 * (0.9999 ** it)

        if line_search:
            step = eta
            for _ in range(5):
                cand = _project_to_simplex(a - step * g)
                cand_obj = _objective(cand, n, L)
                if cand_obj < cur_obj:
                    a = cand
                    break
                step *= 0.5
            else:
                a = _project_to_simplex(a - eta * g)
        else:
            a = _project_to_simplex(a - eta * g)

    return a, best_obj, best_a


def _simulated_annealing_phase(a: np.ndarray, n: int, L: int,
                               max_iters: int, deadline: float,
                               temperature0: float = 0.07,
                               no_improve_limit: int = 8000,
                               verbose: bool = True) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Targeted simulated annealing that moves a small fraction of mass from an
    index participating in the current maximal convolution entry to a random
    destination.  Metropolis acceptance with a slowly cooling temperature.
    """
    best_a = a.copy()
    best_obj = _objective(a, n, L)
    cur_obj = best_obj
    conv = _conv_fft(a, L)

    T = temperature0
    no_improve = 0

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        kmax = int(np.argmax(conv))

        # indices that actually contribute to that peak
        i_candidates = np.arange(max(0, kmax - (n - 1)), min(n - 1, kmax) + 1)
        i_candidates = i_candidates[a[i_candidates] > 1e-14]
        if i_candidates.size == 0:
            i = np.random.randint(n)
        else:
            i = np.random.choice(i_candidates)

        j = np.random.randint(n)
        while j == i:
            j = np.random.randint(n)

        delta = a[i] * np.random.rand() * 0.09
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = _project_to_simplex(a_new)

        conv_new = _conv_fft(a_new, L)
        new_obj = _objective(a_new, n, L)

        # Metropolis acceptance
        if new_obj < cur_obj or np.random.rand() < math.exp((cur_obj - new_obj) / max(T, 1e-12)):
            a = a_new
            conv = conv_new
            cur_obj = new_obj
            no_improve = 0
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                if verbose:
                    print(f"[SA] iter {it} – new global best O={best_obj:.6f}")
        else:
            no_improve += 1

        # occasional exact sub‑gradient step (keeps direction)
        if it % 9 == 0:
            g = _subgradient_max_conv(a, conv)
            eta_loc = 0.07 * (1.0 - it / max_iters)
            a = _project_to_simplex(a - eta_loc * g)
            conv = _conv_fft(a, L)
            cur_obj = _objective(a, n, L)
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                if verbose:
                    print(f"[SA] sub‑grad step – improved O={best_obj:.6f}")

        # temperature decay
        T *= 0.9997

        # restart if stagnating
        if no_improve >= no_improve_limit:
            a = best_a.copy()
            a += np.random.randn(n) * 0.0015
            a = np.clip(a, 0, None)
            a = _project_to_simplex(a)
            conv = _conv_fft(a, L)
            cur_obj = _objective(a, n, L)
            no_improve = 0

    return a, best_obj, best_a


def _pairwise_local_search(a: np.ndarray, n: int, L: int,
                           deadline: float) -> tuple[np.ndarray, float]:
    """
    Deterministic fine‑grained refinement.
    Repeatedly moves a fraction of mass from the index with the largest
    sub‑gradient to the one with the smallest, accepting only moves that
    strictly improve the exact objective.  A few fallback random transfers
    keep the walk from getting stuck.
    """
    best_a = a.copy()
    best_obj = _objective(a, n, L)
    cur_a = a.copy()
    cur_obj = best_obj
    no_improve = 0

    # fractions tried in the inner line‑search
    fractions = (0.4, 0.2, 0.1, 0.05, 0.02, 0.01)

    while time.time() < deadline:
        # compute sub‑gradient of the max‑convolution entry
        conv = _conv_fft(cur_a, L)
        g = _subgradient_max_conv(cur_a, conv)

        src = int(np.argmax(g))
        dst = int(np.argmin(g))

        if cur_a[src] < 1e-12:
            nz = np.where(cur_a > 1e-12)[0]
            if nz.size == 0:
                break
            src = int(np.random.choice(nz))

        improved = False
        for f in fractions:
            delta = cur_a[src] * f
            if delta <= 1e-14:
                continue
            cand = cur_a.copy()
            cand[src] -= delta
            cand[dst] += delta
            obj = _objective(cand, n, L)
            if obj < cur_obj - 1e-12:
                cur_a = cand
                cur_obj = obj
                improved = True
                break

        if not improved:
            # small random perturbation as a last resort
            i = np.random.randint(n)
            j = np.random.randint(n)
            while j == i:
                j = np.random.randint(n)
            delta = cur_a[i] * np.random.rand() * 0.05
            cand = cur_a.copy()
            cand[i] -= delta
            cand[j] += delta
            obj = _objective(cand, n, L)
            if obj < cur_obj:
                cur_a = cand
                cur_obj = obj
                improved = True

        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = cur_a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 5000:
            # tiny random restart to escape flat plateaus
            cur_a = best_a.copy()
            cur_a += np.random.randn(n) * 0.001
            cur_a = np.clip(cur_a, 0, None)
            cur_a = _project_to_simplex(cur_a)
            cur_obj = _objective(cur_a, n, L)
            no_improve = 0

    return best_a, best_obj

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence that minimises the evaluation
    function.  The function respects the 1000 s budget, returns the best
    sequence found as a plain Python list, and prints a short progress log.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # small safety margin

    # ------------------------------------------------------------------
    # Determine an initial problem size (number of steps)
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        a0 = np.array(height_sequence_1, dtype=float)
        a0 = np.maximum(a0, 0.0)
        if a0.sum() == 0.0:
            a0 = np.full_like(a0, 1.0)
        a0 /= a0.sum()
        n0 = a0.size
        print(f"[init] using supplied sequence, n={n0}")
    else:
        n0 = 4600
        a0 = np.full(n0, 1.0 / n0)
        print(f"[init] using uniform start, n={n0}")

    # We will try a few nearby problem sizes – sometimes a slightly
    # different length yields a better optimum.
    n_candidates = [n0]
    for factor in (0.85, 1.15):
        n = int(n0 * factor)
        if 1000 <= n <= 15000:
            n_candidates.append(n)
    # remove duplicates while preserving order
    n_candidates = list(dict.fromkeys(n_candidates))

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Loop over the selected problem sizes
    # ------------------------------------------------------------------
    for n in n_candidates:
        L = 1 << ((3 * n - 2).bit_length())   # FFT length (next power of two)

        # initialise a – keep the supplied sequence when the sizes match,
        # otherwise fall back to a uniform distribution.
        if n == n0:
            a = a0.copy()
        else:
            a = np.full(n, 1.0 / n)

        # --------------------------------------------------------------
        # Phase 0 – cheap random restarts (10 % of the total budget)
        # --------------------------------------------------------------
        phase0_deadline = start_time + budget_s * 0.10
        while time.time() < phase0_deadline:
            conc = np.random.uniform(0.2, 0.8)          # Dirichlet concentration
            a_rand = np.random.gamma(conc, 1.0, size=n)
            a_rand /= a_rand.sum()
            # a very short Adam run (≈2 s)
            sub_dead = time.time() + 2.0
            a_rand, _, best_rand = _adam_phase(
                a_rand, n, L,
                max_iters=2000,
                deadline=sub_dead,
                alpha_start=5.0,
                alpha_end=5000.0,
                lr0=0.8,
                prune_every=0,
                prune_percentile=0.5,
            )
            cur_obj = _objective(best_rand, n, L)
            if cur_obj < best_score:
                best_score = cur_obj
                best_seq = best_rand.copy()
                print(f"[phase0] new best O={best_score:.6f}")

        # --------------------------------------------------------------
        # Phase 1 – extended Adam optimisation (30 % of the budget)
        # --------------------------------------------------------------
        phase1_deadline = start_time + budget_s * 0.30
        a = best_seq.copy() if best_seq is not None else a
        a, obj_adam, best_adam = _adam_phase(
            a, n, L,
            max_iters=400_000,
            deadline=phase1_deadline,
            alpha_start=5.0,
            alpha_end=30_000.0,
            lr0=0.6,
            prune_every=5000,
            prune_percentile=0.1,
        )
        if obj_adam < best_score:
            best_score = obj_adam
            best_seq = best_adam.copy()
            print(f"[phase‑1] improved best O={best_score:.6f}")

        # --------------------------------------------------------------
        # Phase 2 – exact projected sub‑gradient descent (55 % of budget)
        # --------------------------------------------------------------
        phase2_deadline = start_time + budget_s * 0.55
        a = best_seq.copy()
        a, obj_sub, best_sub = _subgradient_phase(
            a, n, L,
            max_iters=250_000,
            deadline=phase2_deadline,
            eta0=0.5,
            line_search=True,
            verbose=False,
        )
        if obj_sub < best_score:
            best_score = obj_sub
            best_seq = best_sub.copy()
            print(f"[phase‑2] improved best O={best_score:.6f}")

        # --------------------------------------------------------------
        # Phase 3 – long simulated annealing walk (remaining time)
        # --------------------------------------------------------------
        phase3_deadline = deadline
        a = best_seq.copy()
        a, obj_sa, best_sa = _simulated_annealing_phase(
            a, n, L,
            max_iters=800_000,
            deadline=phase3_deadline,
            temperature0=0.1,
            no_improve_limit=10000,
            verbose=False,
        )
        if obj_sa < best_score:
            best_score = obj_sa
            best_seq = best_sa.copy()
            print(f"[phase‑3] improved best O={best_score:.6f}")

        # --------------------------------------------------------------
        # Phase 4 – deterministic pair‑wise local refinement (few seconds)
        # --------------------------------------------------------------
        phase4_deadline = deadline - 0.05
        if time.time() < phase4_deadline:
            a = best_seq.copy()
            a, obj_loc = _pairwise_local_search(a, n, L, phase4_deadline)
            if obj_loc < best_score:
                best_score = obj_loc
                best_seq = a.copy()
                print(f"[phase‑4] improved best O={best_score:.6f}")

        # early exit if we have already reached the target
        if best_score <= 1.503:
            print("[info] target reached – exiting early")
            break

    # ------------------------------------------------------------------
    # Final verification via the provided evaluator
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence failed:", exc)

    return best_seq.tolist()
