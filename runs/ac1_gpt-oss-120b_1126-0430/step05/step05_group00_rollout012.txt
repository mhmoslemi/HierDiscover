    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0

    # Determine n: 
    try: n = len(height_sequence_1)
    except: n = 40000

    # Candidate n list maybe [20000, 40000].
    candidate_ns = [n]  # if external variable present, use its length directly; else use default.

    best_overall_a = None
    best_overall_score = float('inf')
    best_overall_n = None

    for n_cur in candidate_ns:
        # Build arcsine cdf distribution
        a_start = arcsine_cdf_distribution(n_cur)  # sum = 1
        # Possibly add small epsilon to avoid zeros: not needed.
        # Convert to logits
        z = _logits_from_distribution(a_start)
        # Adam state
        m = np.zeros_like(z)
        v = np.zeros_like(z)
        beta1, beta2 = 0.9, 0.999
        eps_adam = 1e-8
        base_alpha = 0.4
        eps_factor = 0.1
        eps_factor_min = 5e-4
        eps_factor_decay = 0.9997

        best_a_local = a_start.copy()
        max_conv_best = _conv_max(best_a_local)
        best_score_local = 2.0 * n_cur * max_conv_best

        it = 0
        max_iters = 500_000  # high upper bound

        while time.time() < deadline and it < max_iters:
            it += 1
            a_cur = _softmax(z)

            max_conv, grad_a = _conv_max_and_grad(a_cur, eps_factor)
            cheap_score = 2.0 * n_cur * max_conv

            if cheap_score < best_score_local:
                best_score_local = cheap_score
                best_a_local = a_cur.copy()

            # gradient of objective w.r.t logits
            mu = np.dot(grad_a, a_cur)
            grad_z = a_cur * (grad_a - mu)

            # Adam
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * (grad_z ** 2)
            m_hat = m / (1 - beta1 ** it)
            v_hat = v / (1 - beta2 ** it)
            alpha_t = base_alpha / (np.sqrt(it) + 1.0)
            step = - alpha_t * m_hat / (np.sqrt(v_hat) + eps_adam)
            z = z + step
            # bound logits
            np.clip(z, -60, 60, out=z)

            eps_factor = max(eps_factor_min, eps_factor * eps_factor_decay)

            # random jitter occasionally
            if it % 10000 == 0 and np.random.rand() < 0.15:
                z += 0.01 * np.random.randn(len(z))

            # early stop if we have achieved <1.55 maybe; but continue.

            if it % 20000 == 0:
                # print time/score
                elapsed = time.time() - start
                print(f"n={n_cur}, iter={it}, best_score={best_score_local:.6f}, time={elapsed:.1f}s")

def arcsine_cdf_distribution(n):
    i = np.arange(n, dtype=np.float64)
    # compute CDF at i+1 and i
    # sqrt normalized
    sqrt_i = np.sqrt(i / n)
    sqrt_ip1 = np.sqrt((i+1) / n)
    cdf_i = (2.0 / np.pi) * np.arcsin(sqrt_i)
    cdf_ip1 = (2.0 / np.pi) * np.arcsin(sqrt_ip1)
    a = cdf_ip1 - cdf_i
    # ensure sum = 1 (should be close)
    a_sum = a.sum()
    if a_sum <= 0:
        raise ValueError("Zero sum in arcsine distribution")
    a = a / a_sum
    return a

#!/usr/bin/env python
"""
Arcsine‑CDF + uniform mixing with Adam on the simplex.

We start from the exact discretisation of the arcsine density
    f(x)=1/(π√{x(1‑x)}) ,  x∈(0,1)
by taking the probability mass of each bin
    a_i = CDF((i+1)/n)‑CDF(i/n)  = (2/π)[asin√((i+1)/n)‑asin√(i/n)] .
This distribution makes f∗f (continuous) perfectly flat, and the
exact bin‑integrals remove the large discretisation error of the
mid‑point rule used previously.

A small amount of uniform mass (parameter γ) can be blended in to
soften the inevitable edge spikes.  The resulting vector is turned
into logits, and we run Adam on a smoothed version of the ‑ℓ∞
objective
    J(a)=2·n·max(a∗a) / (∑a)² .
The smoothing uses a soft‑max on the convolution entries; the
gradient is obtained analytically via FFT‑based correlation (see
`_conv_max_and_grad`).  Several γ‑values and two support sizes are
tried; the best sequence (according to the exact `evaluate_sequence`
call) is returned.

The whole routine respects the 1000 s wall‑time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max; returns a probability vector that sums to 1."""
    z_max = np.max(z)
    exp_z = np.exp(z - z_max)
    return exp_z / exp_z.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Logits that reproduce the probability vector `a` after soft‑max."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """
    Smallest power‑of‑two ≥ 3·n‑2.
    Required length for linear convolution (2·n‑1) and correlation (3·n‑2).
    """
    return 1 << ((3 * n - 2).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Smoothed max of a*a and its gradient w.r.t. `a`.

    Parameters
    ----------
    a : np.ndarray
        Probability vector (non‑negative, sum≈1).
    eps_factor : float
        Multiplicative factor for the smoothing constant.
        The actual eps = max(eps_factor·max_conv, 1e‑12).

    Returns
    -------
    max_conv : float
        Ordinary max of the self‑convolution.
    grad_a : np.ndarray
        Gradient of 4·n·smooth_max (already multiplied by 4·n).
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    a_fft = np.fft.rfft(a, L)

    # linear self‑convolution (length 2n‑1)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # smooth‑max weights (soft‑max on the convolution)
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv                     # for numerical stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or np.isnan(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # correlation a ⋆ w (length 3n‑2, we need only the first n entries)
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # gradient of 4·n·smooth_max
    grad_a = 4.0 * n * corr
    return max_conv, grad_a


def _arcsine_cdf_distribution(n: int) -> np.ndarray:
    """
    Exact discrete arcsine distribution:
        a_i = (2/π)[asin(√((i+1)/n)) − asin(√(i/n))] .
    The vector sums to 1 (up to floating‑point noise) and is strictly
    positive for all i.
    """
    i = np.arange(n, dtype=np.float64)
    sqrt_i = np.sqrt(i / n)
    sqrt_ip1 = np.sqrt((i + 1) / n)
    cdf_i = (2.0 / np.pi) * np.arcsin(sqrt_i)
    cdf_ip1 = (2.0 / np.pi) * np.arcsin(sqrt_ip1)
    a = cdf_ip1 - cdf_i
    # Normalise to guard against tiny rounding errors
    a_sum = a.sum()
    if a_sum <= 0.0:
        raise ValueError("Arcsine CDF produced non‑positive total mass.")
    a /= a_sum
    return a


def _run_adam_optimization(
    init_a: np.ndarray,
    n: int,
    deadline: float,
    *,
    base_alpha: float = 0.4,
    eps_factor_start: float = 0.12,
    eps_factor_min: float = 5e-4,
    eps_factor_decay: float = 0.9995,
    max_iters: int = 500_000,
    print_tag: str = "",
):
    """
    Adam optimiser on the simplex (logit parametrisation) for a fixed
    support size `n`.  Returns the best probability vector found and the
    best *cheap* objective value (2·n·max_conv, i.e. the value of J when
    the sum of the vector is 1).
    """
    # initialise logits from the starting distribution
    z = _logits_from_distribution(init_a)
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    eps_factor = eps_factor_start

    best_a = init_a.copy()
    best_score = 2.0 * n * _conv_max(best_a)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        a_cur = _softmax(z)

        # cheap evaluation (smoothed max)
        max_conv, grad_a = _conv_max_and_grad(a_cur, eps_factor)
        cheap_score = 2.0 * n * max_conv

        if cheap_score < best_score:
            best_score = cheap_score
            best_a = a_cur.copy()

        # gradient w.r.t. logits (mirrored gradient on simplex)
        mu = np.dot(grad_a, a_cur)
        grad_z = a_cur * (grad_a - mu)

        # Adam bias‑corrected update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step‑size schedule (inverse‑sqrt decay)
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step = -alpha_t * m_hat / (np.sqrt(v_hat) + eps_adam)
        z = z + step
        # keep the logits in a numerically safe range
        np.clip(z, -60.0, 60.0, out=z)

        # gradually tighten the smoothing constant
        eps_factor = max(eps_factor_min, eps_factor * eps_factor_decay)

        # occasional jitter to escape flat regions
        if it % 10_000 == 0 and np.random.rand() < 0.12:
            z += 0.008 * np.random.randn(n)

        # periodic progress report
        if it % 20_000 == 0:
            elapsed = time.time() - (deadline - (budget_s := deadline - start_time))
            print(
                f"{print_tag} n={n} iter={it:,} best_score={best_score:.6f} "
                f"time={elapsed:.1f}s"
            )
            # early exit if we have already reached a very good bound
            if best_score < 1.2:   # clearly over‑kill, just a safeguard
                break

    return best_a, best_score

    objective.  Several mixing ratios (γ) and two support sizes are

    returned.
    """

    start_time = time.time()

    # ------------------------------------------------------------------
    # Determine candidate support sizes.
    # ------------------------------------------------------------------

    except Exception:

        n_candidates = [20000, 40000]

    # ------------------------------------------------------------------
    # Hyper‑parameter grid (mixing proportion γ).
    # ------------------------------------------------------------------
    gamma_grid = [0.0, 0.08, 0.15]   # 0 = pure arcsine, others add uniform mass

    best_overall_seq = None
    best_overall_score = float("inf")

    # ------------------------------------------------------------------
    # Loop over size / mixing combinations; respect the remaining time.
    # ------------------------------------------------------------------
    for n in n_candidates:
        # Pre‑compute the exact arcsine distribution for this size
        base_arc = _arcsine_cdf_distribution(n)
        uniform = np.full(n, 1.0 / n)

        for gamma in gamma_grid:
            # Ensure we still have enough time for a meaningful run
            if time.time() > deadline - 5.0:
                print("Time budget exhausted – breaking out of the search loop")
                break

            # Blend arcsine and uniform (still a probability vector)
            init = (1.0 - gamma) * base_arc + gamma * uniform
            # Normalise just in case numerical noise accumulated

            if init_sum == 0.0:

            init /= init_sum

                print(f"{tag} evaluation error: {exc}")
                exact_score = None

            # Keep the best overall sequence (according to the exact score)
            if (exact_score is not None) and (exact_score < best_overall_score):
                best_overall_score = exact_score

                    f"from n={n}, γ={gamma:.2f}"

    # ------------------------------------------------------------------
    # Fallback – if, for any reason, no candidate survived, return a

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    except Exception as exc:
