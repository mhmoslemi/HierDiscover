"""
Propose a non‑negative step‑function that (approximately) minimises

    O(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

Algorithm
---------
* **Initialisation** – start from the supplied sequence `height_sequence_1`
  (if present) or from a uniform distribution.  Normalise to sum‑one.

* **Phase 0 – quick variance flattening** – a few hundred Adam steps on the
  variance of the convolution (`Σ (c‑μ)²`).  This pushes the spectrum of
  the autocorrelation towards being flat and gives a good warm‑start.

* **Phase 1 – mirror‑descent on the true max‑objective** – we work directly
  with the sub‑gradient of `max(conv)`.  The update is performed on the
  log‑coordinates (exponentiated‑gradient / mirror descent) which keeps the
  iterate inside the simplex without an explicit projection.  The step‑size
  and a tiny entropy regulariser are annealed so that the iterates become
  increasingly sparse.

* **Phase 2 – stochastic local refinement** – a cheap Metropolis‑Monte‑Carlo
  “mass‑transfer’’ move plus occasional sub‑gradient steps are used to
  escape tiny plateaux and to polish the best solution found so far.

All three phases respect the 1000 s time budget; the best sequence ever
encountered is returned as a plain ``list``.  No file I/O or external
optimisers are used – only ``numpy`` and the supplied ``evaluate_sequence``.
"""

import time
import math

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x ≥ 0, sum(x)=1} using the O(n) algorithm."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution a★a via FFT (length 2·n‑1)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return the full convolution a ★ a (length 2·n‑1) using FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Gradient of the variance‑of‑convolution objective
def _grad_variance(a: np.ndarray, conv: np.ndarray, n: int, L: int) -> np.ndarray:
    """
    Gradient of   f(a) = Σ_k (conv_k - μ)²   where μ = 1/(2n‑1) for sum(a)=1.
    Closed‑form: g = 4·corr(conv‑μ , a).
    """
    mu = 1.0 / (2 * n - 1)
    diff = conv - mu
    a_rev = a[::-1]

    fft_diff = np.fft.rfft(diff, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_diff * fft_a_rev, n=L)

    start = n - 1
    grad = 4.0 * corr_rev[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv)  (used for mirror‑descent)
def _subgrad_maxconv(a: np.ndarray, conv: np.ndarray, n: int) -> np.ndarray:
    """
    Return a sub‑gradient of   g(a) = max(conv(a,a)) .
    For every index k attaining the maximum we add the contribution
    2·a_{k‑i} (if in range) to component i.
    The result is averaged over all maximal indices.
    """
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)
    if ks.size == 0:
        return np.zeros_like(a)

    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    g /= ks.size               # average over all maximal positions
    return g

    # ------------------------------------------------------------------
