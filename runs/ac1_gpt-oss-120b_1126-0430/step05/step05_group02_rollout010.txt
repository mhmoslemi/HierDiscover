def compute_subgradient(a, r):
    n = len(a)
    g = np.zeros_like(a)
    lo = max(0, r - (n - 1))
    hi = min(r, n - 1)
    # loop i from lo to hi inclusive
    rng = np.arange(lo, hi+1)
    j = r - rng
    # where i == j
    eq = (rng == j)
    g[rng[~eq]] = 2 * a[j[~eq]]
    g[rng[eq]] = a[rng[eq]]
    return g

"""
Propose a non‑negative step function (list of heights) that minimises
``evaluate_sequence``.  The algorithm works in three stages:

1. **Broad parametric sweep** – builds many candidate families
   (exponential, symmetric “U‑shaped” exponential, power‑law,
   symmetric power‑law, beta, Dirichlet) and keeps the best.
2. **Exact sub‑gradient mirror descent** – optimises the current best
   vector on the simplex using the true gradient of the maximal
   convolution entry (no top‑k approximation).
3. **Permutation‑local search** – starting from the best mirror‑descent
   iterate, performs a simulated‑annealing style random‑swap search
   on the ordering of the entries to further reduce the max‑convolution.
   Finally the exact ``evaluate_sequence`` is called on the best vector.

All helper utilities are defined at module level; no closures or lambdas
are used.  The routine respects a user‑supplied time budget (default
1000 s) and always returns a feasible sequence.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using a power‑of‑two FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa).real[: 2 * a.size - 1]
    return conv


def _conv_max_fft(a: np.ndarray, fft_len: int) -> tuple[float, int]:
    """Maximum value of the autocorrelation a*a and its index."""
    conv = _conv_full_fft(a, fft_len)
    idx = int(conv.argmax())
    return float(conv[idx]), idx

    best_a, best_val = uniform, val

    # previous construction, if any and same length
    if prev_arr is not None and prev_arr.size == n:
        val = cheap_score(prev_arr)
        if val < best_val:
            best_a, best_val = prev_arr.copy(), val

    # ----- exponential (left) ------------------------------------
    for lam in np.linspace(0.0, 10.0, 41):
        if time.time() >= deadline:
            break
        a = np.exp(-lam * np.arange(n, dtype=np.float64))
        a /= a.sum()
        val = cheap_score(a)
        if val < best_val:
            best_a, best_val = a.copy(), val

    # ----- symmetric exponential (U‑shaped) -----------------------
    for lam in np.linspace(0.0, 10.0, 41):
        if time.time() >= deadline:
            break
        idx = np.arange(n, dtype=np.float64)
        a = np.exp(-lam * np.minimum(idx, n - 1 - idx))
        a /= a.sum()
        val = cheap_score(a)
        if val < best_val:
            best_a, best_val = a.copy(), val

    # ----- power‑law (left) --------------------------------------
    for p in np.linspace(0.01, 3.0, 61):
        if time.time() >= deadline:
            break
        a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
        a /= a.sum()
        val = cheap_score(a)
        if val < best_val:
            best_a, best_val = a.copy(), val

    # ----- symmetric power‑law (U‑shaped) ------------------------
    for p in np.linspace(0.01, 3.0, 61):
        if time.time() >= deadline:
            break
        i = np.arange(n, dtype=np.float64)
        a = np.minimum(i + 1, n - i) ** (-p)
        a /= a.sum()
        val = cheap_score(a)
        if val < best_val:
            best_a, best_val = a.copy(), val

    # ----- beta families (U‑shaped when α,β<1) --------------------
    x = (np.arange(1, n + 1, dtype=np.float64) / n)
    alphas = np.concatenate((
        np.linspace(0.01, 0.2, 5),
        np.linspace(0.2, 1.0, 9),
        np.linspace(1.0, 5.0, 5)
    ))
    betas = alphas.copy()
    for alpha in alphas:
        for beta in betas:
            if time.time() >= deadline:
                break
            a = (x ** alpha) * ((1.0 - x) ** beta)
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = cheap_score(a)
            if val < best_val:
                best_a, best_val = a.copy(), val

    # ----- random Dirichlet seeds --------------------------------
    for conc in [0.001, 0.01, 0.1, 0.5, 1.0, 2.0, 5.0]:
        if time.time() >= deadline:
            break
        g = np.random.gamma(conc, 1.0, size=n)
        a = g / g.sum()
        val = cheap_score(a)
        if val < best_val:
            best_a, best_val = a.copy(), val

    print(f"[Init] best cheap = {best_val:.6f}")

    # --------------------------------------------------------------
    # 1. Exact sub‑gradient mirror descent (exponentiated gradient)
    # --------------------------------------------------------------
    a = best_a.copy()
    best_iter_a = a.copy()
    best_iter_val = best_val
    last_improve = 0
    iter_cnt = 0
    eta0 = 0.7
    max_iters = 2_000_000     # safe upper bound

    while time.time() < deadline and iter_cnt < max_iters:
        iter_cnt += 1

        # full convolution and cheap score (sum is 1)
        conv = _conv_full_fft(a, fft_len)
        maxb = conv.max()
        cheap = 2.0 * n * maxb

        if cheap < best_iter_val - 1e-12:
            best_iter_val = cheap
            best_iter_a = a.copy()
            last_improve = iter_cnt
            print(f"[Iter {iter_cnt:07d}] NEW BEST cheap = {cheap:.6f}")

        # exact sub‑gradient at the maximal entry
        kmax = int(conv.argmax())
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        rng = np.arange(lo, hi + 1, dtype=np.int64)
        j = kmax - rng
        g = np.zeros_like(a)
        eq_mask = (rng == j)
        # off‑diagonal contributions
        g[rng[~eq_mask]] = 2.0 * a[j[~eq_mask]]
        # diagonal contribution (i == j)
        g[rng[eq_mask]] = a[rng[eq_mask]]

        # exponentiated‑gradient update (mirror descent on simplex)
        eta = eta0 / math.sqrt(iter_cnt)
        log_a = np.log(a + eps)
        log_a -= eta * g
        a = np.exp(log_a)
        sum_a = a.sum()
        if sum_a <= 0.0 or not np.isfinite(sum_a):
            a = best_iter_a.copy()
            continue
        a /= sum_a

        # occasional restart if stalled for a long time
        if iter_cnt - last_improve > 200_000:
            perturb = np.random.rand(n) * 0.01
            a = _simplex_projection(best_iter_a + perturb)
            last_improve = iter_cnt
            print(f"[Iter {iter_cnt:07d}] Restart after stagnation.")

    # --------------------------------------------------------------
    # 2. Permutation / swap local search (simulated‑annealing style)
    # --------------------------------------------------------------
    a = best_iter_a.copy()
    best_swap_a = a.copy()
    best_swap_val = best_iter_val
    temperature = 1.0
    cooling = 0.99999

    # number of swap attempts limited by remaining time
    while time.time() < deadline:
        # one random swap
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        a[i], a[j] = a[j], a[i]

        conv = _conv_full_fft(a, fft_len)
        cheap = 2.0 * n * conv.max()
        accept = False
        if cheap < best_swap_val - 1e-12:
            accept = True
        else:
            # annealed acceptance
            delta = cheap - best_swap_val
            prob = math.exp(-delta / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            if cheap < best_swap_val - 1e-12:
                best_swap_val = cheap
                best_swap_a = a.copy()
                print(f"[Swap] NEW BEST cheap = {cheap:.6f}")
        else:
            # revert the swap
            a[i], a[j] = a[j], a[i]

        temperature = max(temperature * cooling, 1e-6)

    # --------------------------------------------------------------
    # 3. Final exact evaluation
    # --------------------------------------------------------------
    final_seq = best_swap_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:  # defensive – should never happen
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    return final_seq
