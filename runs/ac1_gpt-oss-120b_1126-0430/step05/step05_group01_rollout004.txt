    z_max = np.max(z)

    s = np.sum(e)
    if s == 0.0:

    return e / s

    """Log‑it that reproduces `a` after a soft‑max."""
    return np.log(a + eps)

    new_x = np.linspace(0.0, 1.0, new_n)

    else:

    """
    Returns (lp_val, grad_a, max_conv) where

        lp_val = (∑_k (conv_k)^p)^{1/p}
        grad_a = ∇_a lp_val

        return lp_val, grad_a, max_conv

    # Gradient of lp (already contains the factor 2p from derivative of conv)

The evaluation function is  

\[
E(a)=\frac{2\,n\;\max\limits_{k}(a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\qquad a_i\ge 0 .
\]

Because the objective is homogeneous we can restrict the search to the
probability simplex \(\sum_i a_i = 1\).  
If we denote \(c_k=(a*a)_k\) the linear self‑convolution,
the cheap score is simply  

\[
\text{score}=2n\;\max_k c_k .
\]

The direct gradient of the max‑operator is useless (it is non‑smooth
and only looks at a single index).  
Instead we minimise a **smooth \(L_p\) surrogate**

\[
\Phi_p(a)=\bigl(\sum_k c_k^{p}\bigr)^{1/p},
\qquad p\ge 2 .
\]

For \(p\!\rightarrow\!\infty\) the surrogate approaches
\(\max_k c_k\); for small \(p\) it is a smooth average that provides a
useful gradient everywhere.  
The gradient can be computed with three FFTs:

* `A = FFT(a)`.  
* `c = IFFT(A·A)` – the convolution.  
* `w = c^{\,p-1}` (with the same scaling factor that appears in the
  derivative).  
* `grad_a = 2 p · corr(a,w)` – correlation is obtained by the product
  `conj(A)·FFT(w)` and an inverse FFT.

The optimisation is performed on the **log‑it vector** `z`  

\[
a_i = \frac{e^{z_i}}{\sum_j e^{z_j}},
\]

so that the simplex constraint is satisfied automatically.
Mirror‑descent on the simplex is equivalent to a KL‑regularised gradient
step, which we realise with Adam on `z`.

To avoid getting trapped in a narrow basin we start with a small
exponent (`p=4`) and double it every *50 000* iterations up to `p=64`.
Occasional random jitter, a back‑tracking line‑search and a stagnation
restart keep the search robust.  
When time permits we **upsample** the support (e.g. from 25000 to 50000
points) by linear interpolation; the optimisation then continues on the
finer grid.

The whole routine respects the 1000 s budget, prints progress and finally
calls the provided `evaluate_sequence` on the best distribution found.

#!/usr/bin/env python
"""
Non‑negative step‑function optimiser (enhanced).

The algorithm minimises the L∞ self‑convolution objective by a smooth
Lp‑surrogate.  Adam is applied to the log‑it parametrisation (mirror
descent on the simplex).  The surrogate exponent p is increased during the
run (p=4 → 64), and the support size is optionally up‑sampled while the
time budget permits.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures, no lambdas)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max, returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    s = np.sum(e)
    if s == 0.0:
        return np.full_like(z, 1.0 / z.size)
    return e / s


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it that reproduces `a` after a soft‑max."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """Smallest power‑of‑two ≥ 3·n‑2 (enough for all needed convolutions)."""
    return 1 << ((3 * n - 2).bit_length())


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    s = new_a.sum()
    if s == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= s
    return new_a


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _lp_norm_and_grad(a: np.ndarray, p: float):
    """
    Returns (lp_val, grad_a, max_conv) where

        lp_val = (∑_k (conv_k)^p)^{1/p}
        grad_a = ∇_a lp_val
        max_conv = max(conv)

    All operations are O(n log n) via FFTs.
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    # FFT of a (zero‑padded)
    A_fft = np.fft.rfft(a, L)

    # Linear self‑convolution a*a
    conv = np.fft.irfft(A_fft * A_fft, L)[: 2 * n - 1]
    conv = np.maximum(conv, 0.0)          # numerical safety
    max_conv = float(conv.max())

    # p‑norm of the convolution vector
    conv_pow = conv ** p
    sum_pow = conv_pow.sum()
    if sum_pow == 0.0:                     # should never happen
        lp_val = 0.0
        grad_a = np.zeros_like(a)
        return lp_val, grad_a, max_conv

    lp_val = sum_pow ** (1.0 / p)

    # Gradient of the p‑norm:
    #   d(lp)/da = 2·p·corr(a, w) ,  w_k = conv_k^{p‑1}·sum_pow^{1/p‑1}
    factor = sum_pow ** (1.0 / p - 1.0)          # scalar ≥0
    w = conv ** (p - 1.0) * factor               # length 2n‑1
    W_fft = np.fft.rfft(w, L)

    # Correlation a ⋆ w (first n entries)
    corr = np.fft.irfft(np.conj(A_fft) * W_fft, L)[:n]

    grad_a = 2.0 * p * corr
    return lp_val, grad_a, max_conv

    Return a list of non‑negative heights that (hopefully) improves the
    `evaluate_sequence` score.  The routine respects the supplied time
    budget (~1000 s), uses Adam on the simplex and a smooth Lₚ surrogate
    for the L∞ objective, and periodically upsamples the support.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0   # safety margin for final evaluation

    # ------------------------------------------------------------------
    # 0. Initialise the probability vector.
    # ------------------------------------------------------------------
    try:
        # Global variable `height_sequence_1` may be provided by the harness.
        init = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        n0 = init.shape[0]
        a0 = np.maximum(init, 0.0)
        s = a0.sum()
        if s == 0.0:
            a0 = np.full_like(a0, 1.0 / n0)
        else:
            a0 /= s
        print(f"[Init] Loaded global sequence, length = {n0}")
    except Exception:
        # Fallback: a simple decreasing power‑law.
        n0 = 25000
        idx = np.arange(1, n0 + 1, dtype=np.float64)
        a0 = idx ** (-0.8)
        a0 /= a0.sum()
        print(f"[Init] No global sequence, using power‑law, length = {n0}")

    # ------------------------------------------------------------------
    # 1. Adam state on the log‑it representation (mirror descent on simplex)
    # ------------------------------------------------------------------
    n = n0
    z = _logits_from_distribution(a0) + 0.01 * np.random.randn(n)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    # Hyper‑parameters
    base_alpha = 0.25          # base step‑size for logits
    report_every = 5000
    p = 4.0                    # start surrogate exponent
    p_max = 64.0
    p_update_interval = 50000
    upsample_interval = 150000
    max_n = 50000

    # ------------------------------------------------------------------
    # 2. Optimisation loop
    # ------------------------------------------------------------------
    best_seq = _softmax(z)
    best_score = 2.0 * n * _conv_max(best_seq)
    last_improve_iter = 0
    it = 0

    while time.time() < deadline and it < 2_000_000:
        it += 1

        # ----- current point on the simplex -----
        a = _softmax(z)

        # ----- cheap objective (max‑conv) and smooth Lp surrogate -----
        lp_val, grad_a, max_conv = _lp_norm_and_grad(a, p)
        cheap_score = 2.0 * n * max_conv

        # ----- record best ever seen -----
        if cheap_score < best_score - 1e-12:
            best_score = cheap_score
            best_seq = a.copy()
            last_improve_iter = it
            if it % report_every == 0:
                print(f"[Iter {it:07d}] NEW BEST cheap = {cheap_score:.6f}, lp = {lp_val:.6f}, p={p}")

        # ----- gradient on logits (mirror descent) -----
        mu = np.dot(grad_a, a)
        grad_z = a * (grad_a - mu)

        # ----- Adam moment updates (bias‑corrected) -----
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # ----- step‑size schedule (inverse‑sqrt decay) -----
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step_dir = - m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- back‑tracking line‑search (≤ 4 halvings) -----
        accepted = False
        trial_alpha = alpha_t
        for _ in range(4):
            z_new = z + trial_alpha * step_dir
            z_new = np.clip(z_new, -50.0, 50.0)
            a_new = _softmax(z_new)

            L = _fft_len_for_corr(n)
            A_fft = np.fft.rfft(a_new, L)
            conv_new = np.fft.irfft(A_fft * A_fft, L)[: 2 * n - 1]
            new_max = float(conv_new.max())
            new_score = 2.0 * n * new_max

            if new_score <= cheap_score:
                z = z_new
                accepted = True
                break
            trial_alpha *= 0.5

        if not accepted:
            # No improvement – still move a little.
            z = z + alpha_t * step_dir
            z = np.clip(z, -50.0, 50.0)

        # ----- occasional random jitter to escape plateaux -----
        if it % 4000 == 0 and np.random.rand() < 0.20:
            z += 0.015 * np.random.randn(n)
            m *= 0.6
            v *= 0.6

        # ----- increase surrogate exponent p -----
        if it % p_update_interval == 0 and p < p_max:
            p = min(p_max, p * 2.0)
            print(f"[Iter {it}] Increasing surrogate exponent, p = {p}")

        # ----- restart on long stagnation -----
        if it - last_improve_iter > 300_000:
            print(f"[Iter {it}] Stagnation detected – applying jitter restart.")
            z = _logits_from_distribution(best_seq) + 0.02 * np.random.randn(n)
            m.fill(0.0)
            v.fill(0.0)
            last_improve_iter = it

        # ----- optional up‑sampling if time permits -----
        if (it % upsample_interval == 0) and (n < max_n) and (time.time() + 30.0 < deadline):
            new_n = min(max_n, int(n * 1.25))
            print(f"[Iter {it}] Upsampling support: {n} → {new_n}")
            best_seq = _interpolate_distribution(best_seq, new_n)
            z = _logits_from_distribution(best_seq) + 0.005 * np.random.randn(new_n)
            m = np.zeros_like(z)
            v = np.zeros_like(z)
            n = new_n
            # reset surrogate exponent after a change of resolution
            p = 4.0
            last_improve_iter = it

    # ------------------------------------------------------------------
    # 3. Final exact evaluation with the provided evaluate_sequence()
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # ------------------------------------------------------------------
    # 4. Return the best sequence discovered.
    # ------------------------------------------------------------------
    return final_seq
