def construct_sidon_set(n, target_m=None):
    used_sums = set()
    S = []
    candidates = list(range(n))
    random.shuffle(candidates) for better results.
    for i in candidates:
        collision = False
        for j in S:
            s = i + j
            if s in used_sums:
                collision = True
                break
        if not collision:
            # also need to check i+i sum
            if i + i in used_sums:
                continue
            # safe; add
            for j in S:
                used_sums.add(i + j)
            used_sums.add(i + i)
            S.append(i)
            if target_m and len(S) >= target_m:
                break
    return S

"""
Hybrid evolutionary optimisation for the step‑function problem.

The aim is to minimise

    O(a) = 2·n·max(conv(a,a)) / (∑a)²          (∑a is forced to 1)

`conv(a,a)` is the full discrete self‑convolution.
The search uses a small population of candidates that are repeatedly
* mixed (linear interpolation), *mutated with Gaussian noise, and
* refined by a few projected sub‑gradient steps on the exact,
   nonsmooth objective.  The best individual is finally polished by a
   longer sub‑gradient descent with back‑tracking line‑search.

All vectors are kept on the probability simplex, i.e. non‑negative and
summing to one, which guarantees that the evaluation function is
scale‑invariant.  The code respects the 1000 s time budget and prints
intermediate objectives so that progress can be monitored.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, ∑x=1}."""
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑d array")
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # all entries are non‑positive – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full convolution via FFT (real‑valued, uses zero‑padding)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return the full (linear) self‑convolution a*a using FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    # keep the first 2·len(a)‑1 entries (the linear convolution length)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Exact objective (the numerator of O when ∑a = 1)
def _objective(a: np.ndarray, n: int, L: int) -> float:
    """2·n·max(conv(a,a))  (a is assumed to sum to 1)."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Sub‑gradient of the maximal convolution entry
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of  max(conv) w.r.t. a.
    At indices that realise the maximum, the gradient equals the sum of the
    corresponding partner values (multiplied by 2).  When several k attain
    the maximum, the sub‑gradient is averaged over them.
    """
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        # contribution 2·a_j for each i
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
# A few projected sub‑gradient steps (no line‑search)
def _subgrad_steps(a: np.ndarray, n: int, L: int,
                   steps: int, eta0: float) -> np.ndarray:
    """Perform `steps` projected sub‑gradient updates with a decaying stepsize."""
    for it in range(steps):
        conv = _conv_fft(a, L)
        g = _subgradient_max_conv(a, conv)
        eta = eta0 * (0.995 ** it)          # geometric decay, fast enough
        a = _project_to_simplex(a - eta * g)
    return a


# ----------------------------------------------------------------------
# Initialise a random Dirichlet‑type vector.
def _random_dirichlet(n: int, concentration: float) -> np.ndarray:
    """Draw a vector from Dirichlet(alpha) with all alphas = concentration."""
    if concentration <= 0:
        # degenerate: pick a single unit mass at a random index
        a = np.zeros(n)
        a[np.random.randint(n)] = 1.0
        return a
    sample = np.random.gamma(concentration, 1.0, size=n)
    if sample.sum() == 0:
        sample = np.ones_like(sample)
    return sample / sample.sum()


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that minimises the evaluation function.
    Returns the best sequence found as a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2     # a tiny safety margin

    # ------------------------------------------------------------------
    # Problem size ----------------------------------------------------
    if 'height_sequence_1' in globals():
        a0 = np.array(height_sequence_1, dtype=float)
        a0 = np.maximum(a0, 0.0)
        if a0.sum() == 0.0:
            a0 = np.full_like(a0, 1.0)
        a0 = a0 / a0.sum()
        n = a0.size
        print(f"[init] Using supplied sequence, n={n}")
    else:
        n = 4600
        a0 = np.full(n, 1.0 / n)
        print(f"[init] No supplied sequence – using uniform start, n={n}")

    # FFT length – next power of two > 3·n‑2 (to avoid circular wrap‑around)
    L = 1 << ((3 * n - 2).bit_length())

    # ------------------------------------------------------------------
    # Population initialisation ----------------------------------------
    pop_size = 30
    population = []               # list of numpy arrays (simplex‑normalised)
    pop_objs = []                 # corresponding exact objectives

    # Helper to evaluate and store a candidate
    def _store_candidate(vec: np.ndarray):
        vec = _project_to_simplex(vec)           # safety
        obj = _objective(vec, n, L)
        population.append(vec)
        pop_objs.append(obj)

    # 1) the uniform start
    _store_candidate(a0)

    # 2) a few Dirichlet draws with a range of concentrations
    concentrations = [0.05, 0.1, 0.2, 0.4, 0.7, 1.5, 3.0]
    for conc in concentrations:
        a = _random_dirichlet(n, conc)
        # short refinement – 80 sub‑gradient steps (very cheap)
        a = _subgrad_steps(a, n, L, steps=80, eta0=0.6)
        _store_candidate(a)

    # 3) fill the rest of the population with mutated copies of the best so far
    while len(population) < pop_size:
        # pick a random existing member and jiggle it
        base = population[np.random.randint(len(population))]
        child = base + np.random.randn(n) * 0.005
        child = np.maximum(child, 0.0)
        child = _project_to_simplex(child)
        child = _subgrad_steps(child, n, L, steps=60, eta0=0.5)
        _store_candidate(child)

    # keep the population sorted (ascending objective)
    pop = list(zip(pop_objs, population))
    pop.sort(key=lambda x: x[0])
    pop_objs, population = zip(*pop)
    pop_objs = list(pop_objs)
    population = list(population)

    best_obj = pop_objs[0]
    best_seq = population[0].copy()
    print(f"[init] Best of initial pop: O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Evolutionary loop ------------------------------------------------
    # split the remaining time into a flexible loop; we stop when <2 s left
    while time.time() < deadline - 2.0:
        # --- selection -------------------------------------------------
        # tournament‑style: pick two parents from the top half
        top_k = max(5, pop_size // 2)
        p1 = population[np.random.randint(top_k)]
        p2 = population[np.random.randint(top_k)]

        # --- recombination -----------------------------------------------
        w = np.random.rand()
        child = w * p1 + (1.0 - w) * p2

        # --- mutation ----------------------------------------------------
        # add a small amount of Gaussian noise, keep non‑negative
        noise_scale = 0.01 * np.mean(child)   # relative to average mass
        child += np.random.randn(n) * noise_scale
        child = np.maximum(child, 0.0)
        child = _project_to_simplex(child)

        # --- local refinement ---------------------------------------------
        # a handful of cheap sub‑gradient steps; the number is
        # chosen adaptively so that we stay within the budget
        n_steps = 30
        child = _subgrad_steps(child, n, L, steps=n_steps, eta0=0.4)

        # --- evaluation --------------------------------------------------
        obj = _objective(child, n, L)

        # replace the worst individual if the child is better
        if obj < pop_objs[-1]:
            population[-1] = child
            pop_objs[-1] = obj
            # re‑sort the population (tiny list – cheap)
            pop = list(zip(pop_objs, population))
            pop.sort(key=lambda x: x[0])
            pop_objs, population = zip(*pop)
            pop_objs = list(pop_objs)
            population = list(population)

            # keep track of the global best
            if obj < best_obj:
                best_obj = obj
                best_seq = child.copy()
                print(f"[{time.time():.1f}] new global best O={best_obj:.6f}")

        # occasional aggressive shake‑up of the current best (helps escape plateaus)
        if np.random.rand() < 0.005:
            perturbed = best_seq + np.random.randn(n) * 0.02
            perturbed = np.maximum(perturbed, 0.0)
            perturbed = _project_to_simplex(perturbed)
            perturbed = _subgrad_steps(perturbed, n, L, steps=50, eta0=0.5)
            pert_obj = _objective(perturbed, n, L)
            if pert_obj < best_obj:
                best_obj = pert_obj
                best_seq = perturbed.copy()
                print(f"[{time.time():.1f}] shake‑up improved best O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Final polishing – a longer exact sub‑gradient descent with line‑search
    final_deadline = deadline
    # give the final phase at most the remaining time (but at least a few seconds)
    remaining = final_deadline - time.time()
    if remaining > 1.0:
        # use the same routine as in the original code but with a higher iteration budget
        a = best_seq.copy()
        best_local = a.copy()
        best_local_obj = _objective(a, n, L)

        it = 0
        while time.time() < final_deadline and it < 300_000:
            conv = _conv_fft(a, n + (n - 1))  # small helper, same as _conv_fft
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_local_obj:
                best_local_obj = cur_obj
                best_local = a.copy()
            g = _subgradient_max_conv(a, conv)

            # simple back‑tracking line‑search (max 5 reductions)
            step = 0.3 * (0.9999 ** it)
            for _ in range(5):
                cand = _project_to_simplex(a - step * g)
                cand_obj = _objective(cand, n, L)
                if cand_obj < cur_obj:
                    a = cand
                    break
                step *= 0.5
            else:
                a = _project_to_simplex(a - 0.3 * (0.9999 ** it) * g)

            it += 1

        # keep the best from the polishing phase
        if best_local_obj < best_obj:
            best_obj = best_local_obj
            best_seq = best_local.copy()
            print(f"[final‑polish] improved best O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Final verification using the provided evaluator (safety wrapper)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence failed:", exc)

    return best_seq.tolist()
