#!/usr/bin/env python
"""
Improved non‑negative step‑function optimiser.

The optimiser runs a few short “global” scans over simple parametric families,
then launches several independent mirror‑descent runs (Adam on the soft‑max
logits).  After an initial phase on a modest size (≈12 k) the best distribution
is up‑sampled and refined on a larger support (≈25 k).  Symmetry of the shape is
enforced at every iteration – the optimum for the bound is known to be
symmetric – and a slowly decaying smoothing parameter ε turns the ‖·‖∞
objective into a smooth one that admits a cheap gradient.  The whole search
honours the 1000 s budget; the best sequence found is finally evaluated with
the provided ``evaluate_sequence`` function and returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (no closures, no lambdas)
# ----------------------------------------------------------------------
def _fft_len_for_corr(n: int) -> int:
    """Smallest power of two ≥ 3·n‑2 – enough for a linear self‑convolution."""
    return 1 << ((3 * n - 2).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Returns (max_conv, grad_a) where

        grad_a ≈ ∇_a [ 2·n·smooth_max(a*a) ].

    The smooth max uses a log‑sum‑exp with ε = eps_factor·max_conv.
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv                     # ≤ 0
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # gradient of 2·n·smooth_max is 4·n·corr
    grad_a = 4.0 * n * corr
    return max_conv, grad_a


# ----------------------------------------------------------------------
# Simple parametric families (used for the cheap global scan)
# ----------------------------------------------------------------------
def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max, returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    s = e.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / z.size)
    return e / s


def _logits_from_distribution(a: np.ndarray) -> np.ndarray:
    """Log‑it that reproduces `a` after a soft‑max."""
    eps = 1e-16
    return np.log(a + eps)


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    s = new_a.sum()
    if s == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= s
    return new_a


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    a /= a.sum()
    return a


def _sym_product_power_law_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) * (n - i)) ** (-p)
    a /= a.sum()
    return a


def _exp_decay_distribution(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a /= a.sum()
    return a


def _logistic_distribution(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = 1.0 / (1.0 + np.exp(alpha * (i - mid)))
    a /= a.sum()
    return a


def _random_dirichlet_distribution(n: int, concentration: float = 1.0) -> np.ndarray:
    """Draw a random probability vector from Dirichlet(α,…,α)."""
    a = np.random.gamma(concentration, 1.0, size=n)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Mirror‑descent optimiser on the simplex
# ----------------------------------------------------------------------
def _run_mirror_descent(
    init_a: np.ndarray,
    max_iter: int,
    deadline: float,
    base_alpha: float = 0.5,
    eps_factor_start: float = 0.15,
    eps_factor_min: float = 5e-5,
    eps_decay: float = 0.9992,
    report_interval: int = 10000,
):
    """
    Perform Adam‑based mirror descent on the probability simplex.
    Returns (best_seq, best_score) where the score is the cheap value:
        2·n·max(a*a)
    """
    n = init_a.shape[0]
    # initialise logits (numerically stable)
    z = _logits_from_distribution(init_a) + 0.005 * np.random.randn(n)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    eps_factor = eps_factor_start
    best_seq = _softmax(z)
    best_score = 2.0 * n * _conv_max(best_seq)
    last_improve = 0

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # current point
        a = _softmax(z)
        # enforce symmetry – the optimal shape is known to be symmetric
        a = (a + a[::-1]) / 2.0

        max_conv, grad_a = _conv_max_and_grad(a, eps_factor)
        cheap_score = 2.0 * n * max_conv

        if cheap_score < best_score - 1e-12:
            best_score = cheap_score
            best_seq = a.copy()
            last_improve = it
            if it % report_interval == 0:
                print(f"[{n}] it {it:07d} NEW BEST cheap = {cheap_score:.6f}")

        # mirror‑descent gradient (on logits)
        mu = np.dot(grad_a, a)
        grad_z = a * (grad_a - mu)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step‑size schedule (inverse‑sqrt decay)
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step_dir = - m_hat / (np.sqrt(v_hat) + eps_adam)

        # simple back‑tracking line search (max 5 halvings)
        accepted = False
        trial_alpha = alpha_t
        for _ in range(5):
            z_new = z + trial_alpha * step_dir
            # enforce symmetry on logits
            z_new = (z_new + z_new[::-1]) / 2.0
            z_new = np.clip(z_new, -60.0, 60.0)
            a_new = _softmax(z_new)
            a_new = (a_new + a_new[::-1]) / 2.0
            new_score = 2.0 * n * _conv_max(a_new)
            if new_score <= cheap_score:
                z = z_new
                accepted = True
                break
            trial_alpha *= 0.5
        if not accepted:
            # take a modest step anyway (helps escape plateaux)
            z += alpha_t * step_dir
            z = np.clip(z, -60.0, 60.0)
            z = (z + z[::-1]) / 2.0

        # occasional random jitter to jump out of flat regions
        if it % 4000 == 0 and np.random.rand() < 0.15:
            z += 0.02 * np.random.randn(n)
            m *= 0.5
            v *= 0.5

        # smoothness decay
        eps_factor = max(eps_factor_min, eps_factor * eps_decay)

        # restart if stagnation lasts too long
        if it - last_improve > 200_000:
            # re‑initialise from the current best but add jitter
            print(f"[{n}] it {it} – stagnation, jitter restart")
            z = _logits_from_distribution(best_seq) + 0.03 * np.random.randn(n)
            m.fill(0.0)
            v.fill(0.0)
            eps_factor = eps_factor_start
            last_improve = it

    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) improves the
    ``evaluate_sequence`` score.
    The routine respects a ~1000 s time budget, performs a cheap
    global scan, several mirror‑descent runs on a medium‑size support,
    and a final refinement on a larger support.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0      # a small safety margin

    # ------------------------------------------------------------------
    # 0. Global scan over a few families – cheap score, choose the best
    # ------------------------------------------------------------------
    print("[Stage 0] Global parametric scan")
    scan_n = 12000                     # cheap enough for a quick scan
    candidates = []

    # power‑law
    for p in np.linspace(0.3, 1.2, 20):
        candidates.append(_power_law_distribution(scan_n, p))
    # symmetric product‑power‑law
    for p in np.linspace(0.3, 1.2, 20):
        candidates.append(_sym_product_power_law_distribution(scan_n, p))
    # exponential decay (log‑space)
    for loglam in np.linspace(-12, -4, 20):
        lam = 10.0 ** loglam
        candidates.append(_exp_decay_distribution(scan_n, lam))
    # logistic
    for alpha in np.linspace(0.0, 6.0, 25):
        candidates.append(_logistic_distribution(scan_n, alpha))
    # random Dirichlet (few draws)
    for _ in range(5):
        candidates.append(_random_dirichlet_distribution(scan_n, concentration=0.8))

    # include a copy of the global variable if it exists
    try:
        glob = np.array(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if glob.shape[0] >= scan_n:
            # take the first ``scan_n`` entries and normalise
            glob = glob[:scan_n]
            glob = np.maximum(glob, 0.0)
            s = glob.sum()
            if s == 0.0:
                glob[:] = 1.0 / scan_n
            else:
                glob /= s
            candidates.append(glob)
    except Exception:
        pass

    best_a = None
    best_score = float("inf")
    for a in candidates:
        if time.time() >= deadline:
            break
        cheap = 2.0 * scan_n * _conv_max(a)
        if cheap < best_score:
            best_score = cheap
            best_a = a.copy()
    if best_a is None:
        best_a = np.full(scan_n, 1.0 / scan_n)

    print(f"[Stage 0] Chosen initial length = {scan_n}, cheap score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1. Mirror‑descent runs on the medium size support
    # ------------------------------------------------------------------
    medium_n = scan_n
    a_medium = best_a
    best_overall_seq = a_medium.copy()
    best_overall_score = 2.0 * medium_n * _conv_max(a_medium)

    # number of independent runs – we fit as many as the time budget allows
    run_id = 0
    while time.time() < deadline and run_id < 8:
        # give each run a modest budget (≈ 30 s or the remaining time)
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining / (8 - run_id))
        run_deadline = time.time() + run_budget

        # slightly perturb the start point for diversity
        jitter = 0.02 * np.random.randn(medium_n)
        start_a = a_medium + jitter
        start_a = np.maximum(start_a, 0.0)
        s = start_a.sum()
        if s == 0.0:
            start_a[:] = 1.0 / medium_n
        else:
            start_a /= s

        seq, score = _run_mirror_descent(
            init_a=start_a,
            max_iter=400_000,
            deadline=run_deadline,
            base_alpha=0.45,
            eps_factor_start=0.2,
            eps_decay=0.9994,
            report_interval=50000,
        )
        print(f"[Run {run_id}] finished with cheap score = {score:.6f}")
        if score < best_overall_score:
            best_overall_score = score
            best_overall_seq = seq.copy()
        run_id += 1

    # ------------------------------------------------------------------
    # 2. Upsample to a larger support and continue refinement
    # ------------------------------------------------------------------
    final_n = 25000
    if best_overall_seq.shape[0] != final_n:
        print(f"[Stage 1] Upsampling {best_overall_seq.shape[0]} → {final_n}")
        best_overall_seq = _interpolate_distribution(best_overall_seq, final_n)

    # final refinement (more aggressive learning‑rate, very small ε)
    print("[Stage 1] Final refinement on the large support")
    final_seq, final_score = _run_mirror_descent(
        init_a=best_overall_seq,
        max_iter=600_000,
        deadline=deadline,
        base_alpha=0.35,
        eps_factor_start=0.08,
        eps_decay=0.9996,
        report_interval=50000,
    )
    print(f"[Stage 1] refinement cheap score = {final_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Exact evaluation with the provided evaluate_sequence()
    # ------------------------------------------------------------------
    # convert to plain Python list (evaluate_sequence expects a list)
    result = final_seq.tolist()
    try:
        exact = evaluate_sequence(result)
        print(f"[Result] exact evaluate_sequence score = {exact:.6f}")
    except Exception as exc:
        exact = float("inf")
        print("[Result] evaluation failed:", exc)

    # Return the best sequence we have (the refined one)
    return result
