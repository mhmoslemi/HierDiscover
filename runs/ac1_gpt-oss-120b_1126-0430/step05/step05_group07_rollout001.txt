def upsample(seq, target_len):
    seq = np.array(seq, dtype=float)
    current_len = seq.size
    if current_len == target_len:
        return seq
    # use np.interp: produce indices 0..target_len-1 mapping to original
    xs = np.arange(current_len)
    xt = np.linspace(0, current_len - 1, target_len)
    return np.interp(xt, xs, seq)

"""
Propose a non‑negative step function that (approximately) minimises

    O(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm consists of three stages:

1️⃣  **Variance flattening** – Adam optimiser on the
    variance of the self‑convolution (∑(cₖ‑μ)²).  This pushes the
    convolution towards a flat shape while keeping the coefficients
    on the probability simplex.

2️⃣  **Smoothed‐max optimisation** – Adam on a log‑sum‑exp surrogate
    of the peak (α is annealed from 5 up to 20000).  The surrogate
    becomes a very tight approximation of the true max, so the
    updates directly reduce the objective O.

3️⃣  **Stochastic refinement** – a cheap Monte‑Carlo mass‑transfer
    scheme combined with projected sub‑gradient steps on the exact
    max‑convolution.  Random moves are accepted with a Metropolis
    rule at a slowly cooling temperature.

All iterates are projected onto the simplex (∑aᵢ=1, aᵢ≥0) after each
update.  The best sequence found before the deadline is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        # all zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution a★a (length 2·len(a)‑1) using FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
def _grad_variance(a: np.ndarray, conv: np.ndarray, n: int, L: int) -> np.ndarray:
    """Gradient of  Σ (conv_k‑μ)²   where μ = 1/(2n‑1)  (∑a=1)."""
    mu = 1.0 / (2 * n - 1)
    diff = conv - mu                     # length 2n‑1
    a_rev = a[::-1]

    fft_diff = np.fft.rfft(diff, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_diff * fft_a_rev, n=L)

    start = n - 1
    grad = 4.0 * corr_rev[start:start + n]
    return grad


# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, conv: np.ndarray, alpha: float,
                     L: int, lam_entropy: float) -> np.ndarray:
    """
    Gradient of  (2·n/α)·log∑exp(α·conv)  +  λ·∑ a_i·log a_i .
    The first term gives 2·corr(w,a) with w ∝ exp(α·conv).
    """
    # soft‑max weights safely
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / exp_scaled.sum()          # shape (2n‑1,)

    # correlation term
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)

    n_len = a.size
    grad_corr = 2.0 * corr_rev[n_len - 1:n_len - 1 + n_len]

    # tiny entropy term (helps keep strict positivity)
    eps = 1e-20
    grad_ent = lam_entropy * (np.log(a + eps) + 1.0)

    return grad_corr + grad_ent


# ----------------------------------------------------------------------
def _compute_objective(a: np.ndarray, n: int, L: int) -> float:
    """Return 2·n·max(conv(a,a))  (assumes sum(a)=1)."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _subgrad_max(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. a (for sum(a)=1)."""
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)
    n = a.size
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g
