def run_opt(n, time_budget, seed):
    rng = np.random.default_rng(seed)
    a = rng.random(n); a /= a.sum()
    x = np.log(a + 1e-16)
    m = np.zeros_like(x); v = np.zeros_like(x)
    beta_start=0.5
    beta_end=1e6
    lr_start=0.2
    lr_end=0.001
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    it = 0
    start = time.time()
    while time.time() - start < time_budget:
        it += 1
        a = softmax(x)   # function defined
        conv = conv_self_fft(a)
        max_c = conv.max()
        obj = 2.0 * n * max_c  # sum=1 invariant
        if obj < best_obj:
            best_obj = obj; best_a = a.copy()
        # schedule
        t = (time.time() - start) / time_budget
        beta = beta_start * np.exp(np.log(beta_end / beta_start) * t)
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t)
        # compute weights
        if beta > 1e5:
            # exact
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            shifted = conv - max_c
            w = np.exp(beta * shifted)
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(conv, 1.0/conv.size)
            else:
                w = w / w_sum
        # gradient
        g_a = 2.0 * corr_fft(w, a)
        # natural gradient
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)
        # Adam
        m = 0.9 * m + 0.1 * g_x
        v = 0.999 * v + 0.001 * (g_x ** 2)
        m_hat = m / (1 - 0.9 ** it)
        v_hat = v / (1 - 0.999 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + 1e-12)
        # occasional jitter
        if it % 2000 == 0:
            x += rng.normal(scale=0.01, size=n)
        # occasional restart at best
        if it % 50000 == 0:
            x = np.log(best_a + 1e-16) + rng.normal(scale=0.005, size=n)
            m.fill(0)
            v.fill(0)

    # compress small entries
    mask = best_a > 1e-12
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a /= best_a.sum()
        best_obj = 2.0 * best_a.size * conv_self_fft(best_a).max()
    return best_a.tolist(), best_obj

def local_refine(a, time_budget, seed):
    rng = np.random.default_rng(seed)
    a = np.asarray(a, dtype=float)
    a = np.clip(a, 0, None)
    a /= a.sum()
    n = a.size
    conv = conv_self_fft(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    start = time.time()
    while time.time() - start < time_budget:
        # compute max index
        k = int(np.argmax(conv))
        # contributions from each i
        i = np.arange(n)
        j = k - i
        mask = (j >= 0) & (j < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i[mask]] * a[j[mask]]
        total = contrib.sum()
        if total == 0:
            # random pick
            src = rng.integers(n)
        else:
            # choose source index weighted by contribution
            probs = contrib / total
            src = rng.choice(n, p=probs)

        # choose target maybe with minimal a
        # choose randomly weighted by (1 - a) to move to low values
        # Probabilities low for high a.
        inv_a = 1.0 - a / a.max()  # scaled 0..1
        inv_a = np.clip(inv_a, 0, None)
        sum_inv = inv_a.sum()
        if sum_inv > 0:
            probs_t = inv_a / sum_inv
            tgt = rng.choice(n, p=probs_t)
        else:
            tgt = rng.integers(n)
        if tgt == src:
            continue
        delta = a[src] * rng.random() * 0.03  # maybe up to 3%
        if delta <= 0:
            continue
        a_new = a.copy()
        a_new[src] -= delta
        a_new[tgt] += delta
        a_new = np.clip(a_new, 0, None)
        a_new /= a_new.sum()
        conv_new = conv_self_fft(a_new)
        obj_new = 2.0 * n * conv_new.max()
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            conv = conv_new
            a = a_new
        else:
            # maybe accept small worsening with some probability decreasing
            # but for deterministic, just reject
            pass
    return best_a.tolist(), best_obj

"""
Non‑negative step‑function optimiser (improved).

The optimiser works in three stages inside the global time budget:

1.  A fast Adam optimiser on the soft‑max (log‑it) parametrisation of the
    simplex.  The objective is the smoothed max‑convolution using a
    log‑sum‑exp with an exponentially increasing temperature β.
    Larger support sizes receive a larger share of the budget, and the
    best solution from a shorter run is up‑sampled to initialise the next
    (larger) run.

2.  A deterministic “mass‑transfer” local refinement.  At each iteration
    the index k where the convolution attains its maximum is located.
    Contributions a[i]·a[k‑i] are used as a probability to pick a source
    index; mass is moved to a low‑density destination index and the
    move is kept if it reduces the objective.

3.  Final sanity checks, compression of tiny entries and a call to
    ``evaluate_sequence`` for reporting.

All heavy work is done with FFT‑based convolutions for O(n log n)
complexity.  No external libraries other than NumPy are required.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full self‑convolution a*a using real FFT.
    Returns a vector of length 2·len(a)‑1.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return conv_full[: 2 * n - 1]


def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation g_i = Σ_t w[i+t]·a[t]   (0 ≤ i < n)
    Implemented via FFT (real → complex multiplication).
    """
    n = a.shape[0]
    m = w.shape[0]                     # = 2·n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) assuming sum(a)=1
    (the denominator sum(a)² cancels because we keep a normalised).
    """
    n = a.size
    conv = _conv_self_fft(a)
    return float(2.0 * n * conv.max())


def _upsample(seq: np.ndarray, target_n: int) -> np.ndarray:
    """
    Linear interpolation of a normalised sequence to a new length.
    The result is re‑normalised to sum to 1.
    """
    seq = np.asarray(seq, dtype=float)
    if seq.size == target_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, seq.size)
    new_x = np.linspace(0.0, 1.0, target_n)
    up = np.interp(new_x, old_x, seq)
    up = np.clip(up, 0.0, None)
    s = up.sum()
    if s < 1e-12:
        up = np.ones(target_n, dtype=float) / target_n
    else:
        up /= s
    return up


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed max‑convolution
# ----------------------------------------------------------------------


def _run_opt(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
) -> tuple[list[float], float]:
    """
    Adam optimisation on the log‑it simplex for a given support size ``n``.
    Returns the best sequence (as a list) and its exact objective value.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # initialise a on the simplex
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
        a = np.clip(a, 0.0, None)
        if a.sum() < 1e-12:
            a = np.ones(n, dtype=float) / n
        else:
            a = a / a.sum()
    else:
        a = rng.random(n)
        a = a / a.sum()

    # logits (log‑it) representation
    x = np.log(a + 1e-16)
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    # schedule parameters (exponential annealing)
    beta_start = 0.5
    beta_end = 5e5               # final temperature – sharper than before
    lr_start = 0.3
    lr_end = 0.001

    best_a = a.copy()
    best_obj = _objective(best_a)
    it = 0

    while time.time() - start < time_limit:
        it += 1
        # stay on the simplex
        a = _softmax(x)

        # convolution & objective
        conv = _conv_self_fft(a)
        max_c = conv.max()
        obj = 2.0 * n * max_c
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        # schedule (time‑based)
        t = min((time.time() - start) / time_limit, 1.0)
        beta = beta_start * np.exp(np.log(beta_end / beta_start) * t)
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t)

        # smooth‑max weights
        if beta > 2e5:
            # hard max (sub‑gradient)
            w = np.zeros_like(conv)
            w[int(np.argmax(conv))] = 1.0
        else:
            shifted = conv - max_c
            w = np.exp(beta * shifted)
            s = w.sum()
            if s == 0.0:
                w = np.full_like(conv, 1.0 / conv.size)
            else:
                w /= s

        # gradient of (smooth) max w.r.t. a
        g_a = 2.0 * _correlation_fft(w, a)

        # natural gradient in log‑it space (mirrored gradient)
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)

        # Adam update
        m = 0.9 * m + 0.1 * g_x
        v = 0.999 * v + 0.001 * (g_x ** 2)
        m_hat = m / (1.0 - 0.9 ** it)
        v_hat = v / (1.0 - 0.999 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + 1e-12)

        # occasional jitter to escape plateaus
        if it % 2000 == 0:
            x += rng.normal(scale=0.008, size=n)

        # restart from the current best every now and then
        if it % 60000 == 0:
            x = np.log(best_a + 1e-16) + rng.normal(scale=0.004, size=n)
            m.fill(0.0)
            v.fill(0.0)

    # --------------------------------------------------------------
    # compress negligible entries (helps the final objective)
    # --------------------------------------------------------------
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = _objective(best_a)
        print(f"[compress] reduced length → {best_a.size}, O={best_obj:.6f}")

    return best_a.tolist(), best_obj


# ----------------------------------------------------------------------
# Deterministic local refinement (mass‑transfer hill climbing)
# ----------------------------------------------------------------------


def _run_local(
    init_seq: list[float],
    time_limit: float,
    seed: int = 0,
) -> tuple[list[float], float]:
    """
    Greedy local search that repeatedly moves a small fraction of mass from
    indices contributing most to the current maximal convolution entry to
    indices that are under‑populated.  Only improving moves are kept.
    """
    start = time.time()
    rng = np.random.default_rng(seed)
    a = np.asarray(init_seq, dtype=float)
    a = np.clip(a, 0.0, None)
    if a.sum() < 1e-12:
        a = np.ones_like(a) / a.size
    else:
        a = a / a.sum()
    n = a.shape[0]

    conv = _conv_self_fft(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    while time.time() - start < time_limit:
        # location of the current peak
        k_max = int(np.argmax(conv))

        # contributions a[i]·a[k_max‑i] (zero outside range)
        i = np.arange(n)
        j = k_max - i
        mask = (j >= 0) & (j < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i[mask]] * a[j[mask]]
        total_contrib = contrib.sum()

        # source index: weighted by contribution (focus on the worst)
        if total_contrib > 0.0:
            probs_src = contrib / total_contrib
            src = rng.choice(n, p=probs_src)
        else:
            src = rng.integers(n)

        # destination index: prefer low‑density locations
        # use inverted proportionality to current mass
        inv = 1.0 - a / (a.max() + 1e-16)
        inv_sum = inv.sum()
        if inv_sum > 0.0:
            probs_dst = inv / inv_sum
            dst = rng.choice(n, p=probs_dst)
        else:
            dst = rng.integers(n)

        if src == dst:
            continue

        # transfer a modest fraction (≤3 % of src mass)
        delta = a[src] * rng.random() * 0.03
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta
        a_new = np.clip(a_new, 0.0, None)
        a_new /= a_new.sum()

        conv_new = _conv_self_fft(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            conv = conv_new

    return best_a.tolist(), best_obj

def propose_candidate(

    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    best_seq = None
    if 'height_sequence_1' in globals():
        seq0 = np.asarray(height_sequence_1, dtype=float)
        seq0 = np.clip(seq0, 0.0, None)
        if seq0.sum() >= 0.01:
            best_seq = seq0 / seq0.sum()
            print("[init] loaded height_sequence_1 (len={})".format(best_seq.size))

    # ------------------------------------------------------------------
    # Support sizes (ascending).  Larger sizes receive a larger share of the

    # ------------------------------------------------------------------
    lengths = [4000, 8000, 12000, 16000]

        if val < (float('inf') if best_seq is None else _objective(best_seq)):

    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_seq.tolist()
