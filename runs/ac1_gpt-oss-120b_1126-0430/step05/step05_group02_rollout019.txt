#!/usr/bin/env python
"""
Propose a non‑negative step function (list of heights) that minimises

        2·n·max_k (a*a)[k] / (Σ a_i)²

The algorithm proceeds in three phases:

1. **Broad parametric sweep** – evaluates a variety of simple families
   (uniform, power‑law, exponential, beta, Gaussian, mixtures) on a
   relatively large support (up to 25 000 points).  The best candidate
   from this cheap evaluation (which assumes Σa_i=1) becomes the starting
   point for the refinement.

2. **Smooth‑max Adam refinement** – the non‑smooth objective
   ``max(conv)`` is approximated by a temperature‑controlled soft‑max.
   The gradient of the smoothed objective can be written in closed form
   using convolutions; it is computed efficiently with FFTs.  An Adam
   optimiser in the log‑space of the coefficients (followed by a simplex
   projection) drives the max‑convolution down while keeping all entries
   non‑negative and summing to one.

3. **Final exact evaluation** – the best sequence seen during the
   optimisation is handed to the provided ``evaluate_sequence`` function.

The routine respects a user‑supplied time budget (default 1000 s) and
always returns a feasible sequence before the deadline.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)  (normalised to Σa_i = 1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = i ** (-p)
    seq /= seq.sum()
    return seq


def _beta_sequence(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ x_i^α (1‑x_i)^β  with x_i = i/(n+1)  (normalised)."""
    x = (np.arange(1, n + 1, dtype=np.float64) / (n + 1))
    seq = (x ** alpha) * ((1.0 - x) ** beta)
    s = seq.sum()
    if s == 0.0:
        return None
    seq /= s
    return seq


def _exponential_sequence(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam * i)  (normalised)."""
    idx = np.arange(n, dtype=np.float64)
    seq = np.exp(-lam * idx)
    seq /= seq.sum()
    return seq


def _gaussian_sequence(n: int, sigma: float, centre: float = None) -> np.ndarray:
    """a_i ∝ exp(-0.5*((i-centre)/sigma)^2) (normalised)."""
    if centre is None:
        centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    seq = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    seq = np.clip(seq, 0.0, None)
    s = seq.sum()
    if s == 0.0:
        return None
    seq /= s
    return seq


def _conv_fft(a: np.ndarray, b: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of a and b using FFT (length = len(a)+len(b)-1)."""
    fa = np.fft.fft(a, fft_len)
    fb = np.fft.fft(b, fft_len)
    conv = np.fft.ifft(fa * fb).real[: a.size + b.size - 1]
    return conv


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Convenient wrapper for a*a."""
    return _conv_fft(a, a, fft_len)


def _conv_max_fft(a: np.ndarray, fft_len: int) -> tuple[float, int]:
    """Maximum value and its index of the autocorrelation a*a."""
    conv = _conv_full_fft(a, fft_len)
    idx = int(conv.argmax())
    return float(conv[idx]), idx

# ----------------------------------------------------------------------


def _smooth_max_and_gradient(a: np.ndarray, conv_fft_len: int,
                             grad_fft_len: int, beta: float):
    """
    Returns (max_conv, gradient, full_conv).

    max_conv  – the exact maximal entry of a*a
    gradient  – ∇_a  (softmax_β (a*a))   (same shape as a)
    full_conv – full convolution a*a (length 2n‑1)
    """
    n = a.size
    conv = _conv_full_fft(a, conv_fft_len)          # length 2n‑1
    max_val = conv.max()

    # ----- soft‑max weights (stable) -----
    # w_k ∝ exp(beta * (conv_k - max_val))
    exp_vals = np.exp(beta * (conv - max_val))
    w = exp_vals / exp_vals.sum()                   # shape (2n‑1,)

    # ----- gradient: g_i = 2 * Σ_j a_j * w_{i+j}
    #   Let w_rev[t] = w[2n-2-t]; then g_i = 2 * (a * w_rev)[2n-2-i]
    w_rev = w[::-1]

    # linear convolution a (len n) with w_rev (len 2n‑1) → length 3n‑2
    conv_aw = _conv_fft(a, w_rev, grad_fft_len)[: 3 * n - 2]
    # pick the required slice
    g = 2.0 * conv_aw[2 * n - 2 - np.arange(n)]

    return max_val, g, conv

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that
    minimises ``evaluate_sequence`` within the allotted time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0     # safety margin for final eval

    # ------------------------------------------------------------------
    # Choose a working length (larger n gives a better theoretical bound)
    # ------------------------------------------------------------------
    try:
        prev = globals().get('height_sequence_1', None)
        if isinstance(prev, list) and len(prev) > 0:
            prev_arr = np.asarray(prev, dtype=np.float64)
            prev_arr = prev_arr.clip(0.0)          # ensure non‑negative
            if prev_arr.sum() > 0.0:
                prev_arr = prev_arr / prev_arr.sum()
            else:
                prev_arr = None
        else:
            prev_arr = None
    except Exception:
        prev_arr = None

    # target length – at least 10 k, but not more than 25 k (FFT still cheap)
    TARGET_MIN = 10000
    TARGET_MAX = 25000
    if prev_arr is not None:
        n = max(prev_arr.size, TARGET_MIN)
    else:
        n = TARGET_MIN
    n = min(max(n, TARGET_MIN), TARGET_MAX)

    # FFT lengths (next powers of two)
    conv_fft_len = 1 << ((2 * n - 1).bit_length())          # for a*a
    grad_fft_len = 1 << ((3 * n - 2).bit_length())          # for a * w_rev

    # ------------------------------------------------------------------
    # Cheap score (Σa_i = 1 assumed) → 2·n·max(conv)
    # ------------------------------------------------------------------
    def cheap_score(vec: np.ndarray) -> float:
        maxb, _ = _conv_max_fft(vec, conv_fft_len)
        return 2.0 * n * maxb

    # ------------------------------------------------------------------
    # 0. Initial scan over parametric families
    # ------------------------------------------------------------------
    best_a = None
    best_val = float("inf")

    # uniform baseline
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    val = cheap_score(uniform)
    best_a, best_val = uniform, val

    # previous construction, if any (and matching length)
    if prev_arr is not None and prev_arr.size == n:
        val = cheap_score(prev_arr)
        if val < best_val:
            best_a, best_val = prev_arr.copy(), val

    # ----- power‑law sweep -----
    for p in np.linspace(0.4, 1.2, 41):
        if time.time() >= deadline:
            break
        cand = _power_law_sequence(n, p)
        val = cheap_score(cand)
        if val < best_val:
            best_a, best_val = cand.copy(), val

    # ----- exponential sweep -----
    for lam in np.linspace(0.0, 4.0, 41):
        if time.time() >= deadline:
            break
        cand = _exponential_sequence(n, lam)
        val = cheap_score(cand)
        if val < best_val:
            best_a, best_val = cand.copy(), val

    # ----- beta families (α,β) -----
    for alpha in np.linspace(0.2, 5.0, 17):
        for beta in np.linspace(0.2, 5.0, 17):
            if time.time() >= deadline:
                break
            cand = _beta_sequence(n, alpha, beta)
            if cand is None:
                continue
            val = cheap_score(cand)
            if val < best_val:
                best_a, best_val = cand.copy(), val

    # ----- Gaussian families (vary σ) -----
    for sigma in np.linspace(n / 200.0, n / 8.0, 15):
        if time.time() >= deadline:
            break
        cand = _gaussian_sequence(n, sigma)
        if cand is None:
            continue
        val = cheap_score(cand)
        if val < best_val:
            best_a, best_val = cand.copy(), val

    # ----- random Dirichlet seeds (wide concentration range) -----
    for conc in [0.001, 0.01, 0.1, 0.5, 1.0, 3.0, 10.0]:
        if time.time() >= deadline:
            break
        g = np.random.gamma(conc, 1.0, size=n)
        cand = g / g.sum()
        val = cheap_score(cand)
        if val < best_val:
            best_a, best_val = cand.copy(), val

    print(f"[Init] n={n}, best cheap score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 1. Refine the best candidate with Adam on a smooth max objective
    # ------------------------------------------------------------------
    a = best_a.copy()
    log_a = np.log(np.clip(a, 1e-30, None))          # work in log‑space
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.2
    beta_smooth = 12.0          # start temperature for the soft‑max
    iter_cnt = 0
    max_iters = 200_000
    best_iter_a = a.copy()
    best_iter_val = best_val
    last_improve_iter = 0

    while time.time() < deadline and iter_cnt < max_iters:
        iter_cnt += 1

        # ----- compute exact max (for cheap score) -----
        max_conv, grad, conv_full = _smooth_max_and_gradient(
            a, conv_fft_len, grad_fft_len, beta_smooth)

        cheap = 2.0 * n * max_conv
        if cheap < best_iter_val - 1e-12:
            best_iter_val = cheap
            best_iter_a = a.copy()
            last_improve_iter = iter_cnt
            print(f"[Iter {iter_cnt:07d}] NEW BEST cheap = {cheap:.6f}")

        # ----- Adam update on log‑a -----
        # gradient of the smoothed max w.r.t. a is already in ``grad``
        g = grad
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** iter_cnt)
        v_hat = v / (1.0 - beta2 ** iter_cnt)

        log_a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- back‑to probability simplex (exponentiated‑gradient) -----
        a = np.exp(log_a)
        s = a.sum()
        if s <= 0.0 or not np.isfinite(s):
            # revert to the last good iterate
            a = best_iter_a.copy()
            log_a = np.log(np.clip(a, 1e-30, None))
            continue
        a /= s

        # ----- anneal the temperature slowly (sharpen the max) -----
        if iter_cnt % 4000 == 0 and beta_smooth < 200.0:
            beta_smooth *= 1.07

        # ----- occasional restart if stalled -----
        if iter_cnt - last_improve_iter > 30_000:
            perturb = 0.02 * np.random.rand(n)
            a = _simplex_projection(best_iter_a + perturb)
            log_a = np.log(np.clip(a, 1e-30, None))
            m.fill(0.0)
            v.fill(0.0)
            last_improve_iter = iter_cnt
            print(f"[Iter {iter_cnt:07d}] Restart after stagnation.")

    # ------------------------------------------------------------------
    # 2. Final exact evaluation (uses the supplied evaluate_sequence)
    # ------------------------------------------------------------------
    final_seq = best_iter_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:      # defensive – should never happen
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # ------------------------------------------------------------------
    # Return the best sequence found.
    # ------------------------------------------------------------------
    return final_seq
