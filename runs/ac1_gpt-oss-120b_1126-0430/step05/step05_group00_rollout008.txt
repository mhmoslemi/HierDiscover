def arcsine_weights(n):
    # endpoints: i from 0 to n-1
    # w_i = (2/π)*(arcsin(sqrt((i+1)/n)) - arcsin(sqrt(i/n)))
    i = np.arange(n)
    # compute u = sqrt(i/n)
    # Use dtype float64.
    left = np.sqrt(i / n)
    right = np.sqrt((i + 1) / n)
    # arcsin
    arcs_left = np.arcsin(left)
    arcs_right = np.arcsin(right)
    w = (2.0 / np.pi) * (arcs_right - arcs_left)
    # ensure sum approx 1
    w = w / w.sum()
    return w

def arcsine_weights(n):
    # i/ n as float
    i = np.arange(n, dtype=np.float64)
    left = np.sqrt(i / n)
    right = np.sqrt((i + 1) / n)
    # To avoid floating rounding >1
    right = np.minimum(right, 1.0)
    left = np.clip(left, 0.0, 1.0)
    arcs_left = np.arcsin(left)
    arcs_right = np.arcsin(right)
    w = (2.0 / np.pi) * (arcs_right - arcs_left)
    # Normalization to guarantee sum 1 exactly.
    w_sum = w.sum()
    if w_sum <= 0:
        raise ValueError("Zero sum")
    w = w / w_sum
    return w

"""
Arcsine‑integral construction for the max‑convolution score.

The continuous probability density

    f(x) = 1 / (π·√(x·(1‑x))) ,      x∈(0,1)

satisfies (f * f)(y) = ½ on the whole interval [0,2].  If we
discretise the interval [0,1] into *n* equal bins and assign to
each bin the exact integral of *f* over that bin, the resulting
vector *a* has sum = 1 and its self‑convolution satisfies

    (a * a)[k] = ½·area_of_bin_union_k = ½·(1/n)   (for the central k)

so that the evaluation metric

    J(a) = 2·n·max(a*a) / (sum(a)²) = 2·n·max(a*a) = 1

up to a tiny O(1/n) discretisation error.  This is essentially the
optimal value (the theoretical lower bound is 1).

The implementation below

* determines a support size *n* (re‑uses the length of an existing
  `height_sequence_1` when present, otherwise defaults to 20 000),
* builds the exact arcsine weights by integrating the density over
  each bin,
* optionally performs a very short Adam optimisation on the
  probability simplex if time permits,
* returns the best sequence found.

All helper routines are top‑level (no closures) and the code respects
the 1000‑second time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (pure NumPy, no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max producing a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it representation that yields `a` after a soft‑max."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """
    Smallest power‑of‑two ≥ 3·n‑2, sufficient for a linear convolution of
    length (2·n‑1) and the correlation used in the smoothed gradient.
    """
    return 1 << ((3 * n - 2).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a (FFT, no gradient)."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Compute the maximum entry of a*a and the gradient of a smooth
    approximation of the max‑operator.

    Parameters
    ----------
    a : np.ndarray
        Probability vector (sum = 1, non‑negative).
    eps_factor : float
        Multiplicative factor for the smoothing constant:
        eps = eps_factor * max(conv).

    Returns
    -------
    max_conv : float
        Maximum of the linear self‑convolution.
    grad_a : np.ndarray
        Gradient of   4·n·smooth_max   w.r.t. a (already multiplied).
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    a_fft = np.fft.rfft(a, L)

    # linear self‑convolution (length 2n‑1)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # smoothed soft‑max weights
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv                     # for numerical stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or np.isnan(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # cross‑correlation a ⋆ w (length n)
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # gradient (4·n·corr)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a


def _arcsine_weights(n: int) -> np.ndarray:
    """
    Exact discretisation of the arcsine density on n equal sub‑intervals.
    The i‑th weight is

        w_i = (2/π)·[ arcsin(√((i+1)/n)) − arcsin(√(i/n)) ] ,

    which equals ∫_{i/n}^{(i+1)/n} f(x)dx with f(x)=1/(π√{x(1−x)}).
    The vector is normalised to sum exactly 1 (up to floating‑point
    rounding).
    """
    i = np.arange(n, dtype=np.float64)
    left = np.sqrt(i / n)
    right = np.sqrt((i + 1) / n)

    # Guard against occasional round‑off >1
    right = np.clip(right, 0.0, 1.0)

    arcs_left = np.arcsin(left)
    arcs_right = np.arcsin(right)

    w = (2.0 / np.pi) * (arcs_right - arcs_left)

    # Normalise to unit sum – this also repairs the tiny loss of mass
    total = w.sum()
    if total <= 0.0:
        raise RuntimeError("Arcsine weights sum to non‑positive value")
    w /= total
    return w

        m = np.zeros_like(z)
