"""
Algorithm for minimizing O(a) = 2·n·max(conv(a,a)) / (sum a)^2.

- Multiple diversified initialisations (uniform, exponential, linear,
  power‑law, random Dirichlet) are generated and each is refined
  independently with an annealed Adam optimiser on a smooth
  log‑sum‑exp surrogate of the max‑convolution.
- The best intermediate solution is fed to an intensive simulated‑annealing
  local‑search that repeatedly moves a fraction of probability mass from
  indices that increase the max‑convolution (identified via the sub‑gradient)
  to indices that have little influence, accepting occasional worsening
  steps according to a temperature schedule.
- A final short Adam polishing with a very large smoothing parameter
  fine‑tunes the result.
- At the end we also test a monotone ordering of the final vector and keep
  the ordering with the lower exact objective.
- The whole routine respects the 1000 s budget; intermediate progress is
  printed and the best sequence found is returned as a plain list.
"""

# imports
import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # all zeros – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution a * a using an FFT of length L (power of two)."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    # length of full convolution is 2·len(a)-1
    return conv_full[: 2 * a.size - 1]

# ----------------------------------------------------------------------
def _objective_simplex(a: np.ndarray, n: int, L: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) when Σa = 1."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()

# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of the max‑convolution with respect to a (averaged over maximisers)."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g

# ----------------------------------------------------------------------
def _smooth_gradient_plus_entropy(
    a: np.ndarray,
    conv: np.ndarray,
    alpha: float,
    L: int,
    lam: float,
) -> np.ndarray:
    """
    Gradient of the smooth surrogate:
        (2·n/α)·log Σ_k exp(α·conv_k) + λ·Σ_i a_i·log a_i .
    """
    # soft‑max weights w_k ∝ exp(α·conv_k)
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / exp_scaled.sum()                       # shape (2n‑1,)

    # correlation w ⋆ a (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n_len = a.size
    grad_corr = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]

    # entropy gradient: λ·(log a_i + 1)
    eps = 1e-20
    grad_entropy = lam * (np.log(a + eps) + 1.0)

    return grad_corr + grad_entropy

# ----------------------------------------------------------------------
def _generate_exponential(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ exp(-α·i)   (α is a small positive number, not the smoothing parameter)."""
    idx = np.arange(n, dtype=float)
    w = np.exp(-alpha * idx)
    w /= w.sum()
    return w

# ----------------------------------------------------------------------
def _generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=float)
    w = idx ** (-p)
    w /= w.sum()
    return w

# ----------------------------------------------------------------------
def _generate_linear_decreasing(n: int) -> np.ndarray:
    """a_i decreasing linearly from 1 to 0."""
    idx = np.arange(n, dtype=float)
    w = (n - idx)  # positive values
    w /= w.sum()
    return w

# ----------------------------------------------------------------------
def _generate_random_dirichlet(n: int) -> np.ndarray:
    """Sample from Dirichlet(1,…,1) (Uniform on simplex)."""
    x = np.random.gamma(shape=1.0, scale=1.0, size=n)
    w = x / x.sum()
    return w

# ----------------------------------------------------------------------
def _run_annealed_adam(
    a0: np.ndarray,
    n: int,
    L: int,
    deadline: float,
    max_iter: int = 1_200_000,
) -> tuple[np.ndarray, float]:
    """
    Annealed Adam optimiser on the smooth log‑sum‑exp surrogate.
    Returns (best_vector, best_objective) found before ``deadline``.
    """
    a = a0.copy()
    # initialise Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    # Adam hyper‑parameters
    lr0 = 0.9
    # schedule for smoothing (α) and entropy (λ)
    alpha_start, alpha_end = 2.0, 20000.0
    lam_start, lam_end = 0.2, 0.0

    best_seq = a.copy()
    best_obj = _objective_simplex(best_seq, n, L)

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_seq = a.copy()
            print(f"[AnnealedAdam] iter {it:,} new best O={best_obj:.6f}")

        # annealing fractions
        frac = it / max_iter
        alpha = alpha_start + (alpha_end - alpha_start) * frac
        lam = max(0.0, lam_start * (1.0 - frac))

        # gradient of surrogate
        grad = _smooth_gradient_plus_entropy(a, conv, alpha, L, lam)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        lr = lr0 / math.sqrt(it + 1)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape plateaus
        if random.random() < 0.001:
            a_new += np.random.randn(n) * 5e-5

        a = _project_to_simplex(a_new)

    return best_seq, best_obj

# ----------------------------------------------------------------------
def _run_simulated_annealing(
    a0: np.ndarray,
    n: int,
    L: int,
    deadline: float,
    init_temp: float = 0.12,
    decay: float = 0.99985,
    max_iter_local: int = 400_000,
) -> tuple[np.ndarray, float]:
    """
    Intensive simulated‑annealing refinement starting from ``a0``.
    Returns (best_vector, best_objective) found before ``deadline``.
    """
    a = a0.copy()
    conv = _conv_fft(a, L)
    cur_obj = 2.0 * n * conv.max()
    best_seq = a.copy()
    best_obj = cur_obj

    T = init_temp
    it = 0
    no_improve = 0
    no_improve_limit = 2500

    while time.time() < deadline and it < max_iter_local:
        # ----- mass‑transfer move guided by sub‑gradient -----
        # compute sub‑gradient only occasionally (every few steps) to save time
        if it % 7 == 0:
            g = _subgradient_max_conv(a, conv)
        # pick source with largest gradient, target with smallest
        src = int(np.argmax(g))
        tgt = int(np.argmin(g))
        if src != tgt and a[src] > 1e-15:
            # move a fraction of the source mass to the target
            max_frac = 0.4 * (1.0 - it / max_iter_local) + 0.01
            frac = random.random() * max_frac
            delta = a[src] * frac
            a_new = a.copy()
            a_new[src] -= delta
            a_new[tgt] += delta
            a_new = _project_to_simplex(a_new)

            conv_new = _conv_fft(a_new, L)
            new_obj = 2.0 * n * conv_new.max()

            # Metropolis acceptance
            if new_obj < cur_obj or random.random() < math.exp((cur_obj - new_obj) / max(T, 1e-16)):
                a, conv, cur_obj = a_new, conv_new, new_obj
                no_improve = 0
                if new_obj < best_obj:
                    best_obj = new_obj
                    best_seq = a.copy()
                    print(f"[SA] iter {it:,} new global best O={best_obj:.6f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        # occasional random swap (helps break symmetry)
        if it % 5000 == 0:
            i, j = random.sample(range(n), 2)
            a_swapped = a.copy()
            a_swapped[i], a_swapped[j] = a_swapped[j], a_swapped[i]
            conv_swapped = _conv_fft(a_swapped, L)
            obj_swapped = 2.0 * n * conv_swapped.max()
            if obj_swapped < cur_obj or random.random() < math.exp((cur_obj - obj_swapped) / max(T, 1e-16)):
                a, conv, cur_obj = a_swapped, conv_swapped, obj_swapped
                if obj_swapped < best_obj:
                    best_obj = obj_swapped
                    best_seq = a.copy()
                    print(f"[SA] swap improved O={best_obj:.6f}")

        # temperature decay
        T *= decay
        it += 1

        if no_improve >= no_improve_limit:
            # restart from current best with a small Gaussian jitter
            a = best_seq.copy()
            a += np.random.randn(n) * 0.001
            a = np.clip(a, 0, None)
            a = _project_to_simplex(a)
            conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max()
            no_improve = 0

    return best_seq, best_obj

def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution a⋆a using an FFT of length ``L``."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
def _objective_simplex(a: np.ndarray, n: int, L: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) assuming Σa = 1."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of ``max(conv)`` w.r.t. ``a`` (averaged over all maximisers)."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)          # indices attaining the max
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
def _smooth_gradient_plus_entropy(
    a: np.ndarray,
    conv: np.ndarray,
    alpha: float,
    L: int,
    lam: float,
) -> np.ndarray:
    """
    Gradient of the smooth surrogate

        (2·n/α)·log Σ_k exp(α·conv_k) + λ·Σ_i a_i·log a_i .

    The first term reduces to a correlation with the soft‑max weights,
    the second term contributes λ·(log a_i + 1).
    """
    # soft‑max weights
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)          # numerical stability
    w = exp_scaled / exp_scaled.sum()                 # shape (2n‑1,)

    # correlation w ⋆ a   (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n_len = a.size
    grad_corr = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]

    # entropy part
    eps = 1e-20
    grad_entropy = lam * (np.log(a + eps) + 1.0)

    return grad_corr + grad_entropy


# ----------------------------------------------------------------------
def _generate_exponential(n: int, epsilon: float) -> np.ndarray:
    """a_i ∝ exp(-ε·i)  (ε>0, not the smoothing parameter)."""
    idx = np.arange(n, dtype=float)
    w = np.exp(-epsilon * idx)
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
def _generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=float)
    w = idx ** (-p)
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
def _generate_linear_decreasing(n: int) -> np.ndarray:
    """Linear decrease from a maximum at the leftmost index to 0."""
    idx = np.arange(n, dtype=float)
    w = (n - idx)
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
def _generate_random_dirichlet(n: int) -> np.ndarray:
    """Draw a random point on the probability simplex (Dirichlet(1,…,1))."""
    x = np.random.gamma(shape=1.0, scale=1.0, size=n)
    w = x / x.sum()
    return w


# ----------------------------------------------------------------------
def _run_annealed_adam(
    a0: np.ndarray,
    n: int,
    L: int,
    deadline: float,
    max_iter: int = 1_200_000,
) -> tuple[np.ndarray, float]:
    """
    Annealed Adam optimiser on the smooth log‑sum‑exp surrogate.
    Returns the best vector encountered (and its objective) before ``deadline``.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    lr0 = 0.9
    alpha_start, alpha_end = 2.0, 20000.0
    lam_start, lam_end = 0.2, 0.0

    best_seq = a.copy()
    best_obj = _objective_simplex(best_seq, n, L)

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_seq = a.copy()
            print(f"[Adam] iter {it:,} new best O={best_obj:.6f}")

        # annealing schedules
        frac = it / max_iter
        alpha = alpha_start + (alpha_end - alpha_start) * frac
        lam = max(0.0, lam_start * (1.0 - frac))

        grad = _smooth_gradient_plus_entropy(a, conv, alpha, L, lam)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        lr = lr0 / math.sqrt(it + 1)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # tiny random jitter to escape plateaus
        if random.random() < 0.001:
            a_new += np.random.randn(n) * 5e-5

        a = _project_to_simplex(a_new)

    return best_seq, best_obj


# ----------------------------------------------------------------------
def _run_simulated_annealing(
    a0: np.ndarray,
    n: int,
    L: int,
    deadline: float,
    init_temp: float = 0.12,
    decay: float = 0.99985,
    max_iter_local: int = 400_000,
) -> tuple[np.ndarray, float]:
    """
    Intensive simulated‑annealing refinement starting from ``a0``.
    Returns the best vector encountered (and its objective) before ``deadline``.
    """
    a = a0.copy()
    conv = _conv_fft(a, L)
    cur_obj = 2.0 * n * conv.max()
    best_seq = a.copy()
    best_obj = cur_obj

    # initial sub‑gradient (used repeatedly)
    g = _subgradient_max_conv(a, conv)

    T = init_temp
    it = 0
    no_improve = 0
    no_improve_limit = 2500

    while time.time() < deadline and it < max_iter_local:
        # recompute sub‑gradient periodically (cheap enough)
        if it % 7 == 0:
            g = _subgradient_max_conv(a, conv)

        # pick a source (large gradient) and a target (small gradient)
        src = int(np.argmax(g))
        tgt = int(np.argmin(g))
        if src != tgt and a[src] > 1e-15:
            max_frac = 0.4 * (1.0 - it / max_iter_local) + 0.01
            frac = random.random() * max_frac
            delta = a[src] * frac
            a_new = a.copy()
            a_new[src] -= delta
            a_new[tgt] += delta
            a_new = _project_to_simplex(a_new)

            conv_new = _conv_fft(a_new, L)
            new_obj = 2.0 * n * conv_new.max()

            # Metropolis acceptance
            if new_obj < cur_obj or random.random() < math.exp(
                (cur_obj - new_obj) / max(T, 1e-16)
            ):
                a, conv, cur_obj = a_new, conv_new, new_obj
                no_improve = 0
                if new_obj < best_obj:
                    best_obj = new_obj
                    best_seq = a.copy()
                    print(
                        f"[SA] iter {it:,} new global best O={best_obj:.6f}"
                    )
            else:
                no_improve += 1
        else:
            no_improve += 1

        # occasional random swap – helps escape combinatorial plateaux
        if it % 5000 == 0:
            i, j = random.sample(range(n), 2)
            a_swapped = a.copy()
            a_swapped[i], a_swapped[j] = a_swapped[j], a_swapped[i]
            conv_swapped = _conv_fft(a_swapped, L)
            obj_swapped = 2.0 * n * conv_swapped.max()
            if (
                obj_swapped < cur_obj
                or random.random()
                < math.exp((cur_obj - obj_swapped) / max(T, 1e-16))
            ):
                a, conv, cur_obj = a_swapped, conv_swapped, obj_swapped
                if obj_swapped < best_obj:
                    best_obj = obj_swapped
                    best_seq = a.copy()
                    print(f"[SA] swap improved O={best_obj:.6f}")

        T *= decay
        it += 1

        if no_improve >= no_improve_limit:
            # restart from the current best with a little Gaussian jitter
            a = best_seq.copy()
            a += np.random.randn(n) * 0.001
            a = np.clip(a, 0, None)
            a = _project_to_simplex(a)
            conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max()
            no_improve = 0

    return best_seq, best_obj


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    the provided evaluation function ``evaluate_sequence``.
    The routine respects the ``budget_s`` time limit.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.2        # small safety margin

    # --------------------------------------------------------------
    # Determine problem size
    if "height_sequence_1" in globals():
        base = np.array(height_sequence_1, dtype=float)
        n = base.size
        a0 = np.maximum(base, 0.0)
        tot = a0.sum()
        if tot < 1e-12:
            a0 = np.full(n, 1.0 / n)
        else:
            a0 = a0 / tot
        print(f"[init] using supplied sequence, n={n}")
    else:
        n = 6000                                 # default length
        a0 = np.full(n, 1.0 / n)                 # uniform start
        print(f"[init] using uniform start, n={n}")

    # FFT length – next power of two ≥ 2·n‑1
    L = 1
    while L < 2 * n - 1:
        L <<= 1

    # --------------------------------------------------------------
    # 1️⃣  Build a diverse set of seeds
    seeds = []

    # uniform
    seeds.append(np.full(n, 1.0 / n))

    # exponential shapes – a few decay rates (α = ε·n)
    for alpha_factor in (1.0, 2.0, 3.0, 5.0):
        epsilon = alpha_factor / n
        seeds.append(_generate_exponential(n, epsilon))

    # linear decreasing
    seeds.append(_generate_linear_decreasing(n))

    # power‑law shapes
    for p in (0.5, 1.0, 1.5, 2.0):
        seeds.append(_generate_powerlaw(n, p))

    # random Dirichlet draws
    for _ in range(3):
        seeds.append(_generate_random_dirichlet(n))

    # sorted variants of random draws (enforce monotonicity)
    for _ in range(2):
        rnd = _generate_random_dirichlet(n)
        seeds.append(np.sort(rnd)[::-1])   # descending
        seeds.append(np.sort(rnd))        # ascending

    # normalise everything onto the simplex
    seeds = [s / s.sum() for s in seeds]

    # --------------------------------------------------------------
    # 2️⃣  Annealed Adam optimisation on each seed (≈60 % of time)
    phase1_deadline = start_time + budget_s * 0.60
    best_seq = None
    best_score = np.inf

    for idx, seed_vec in enumerate(seeds):
        if time.time() >= phase1_deadline:
            break
        # allocate a fair share of the remaining time to this seed
        remaining = max(1e-6, phase1_deadline - time.time())
        share = remaining / max(1, len(seeds) - idx)
        seed_deadline = time.time() + share
        seq, score = _run_annealed_adam(seed_vec, n, L, seed_deadline)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[global] after seed {idx} best O={best_score:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Simulated‑annealing refinement on the best vector (≈40 % of time)
    if best_seq is None:                     # very unlikely, but safe
        best_seq = np.full(n, 1.0 / n)

    seq, score = _run_simulated_annealing(best_seq, n, L, deadline)
    if score < best_score:
        best_score = score
        best_seq = seq.copy()
        print(f"[global] after SA best O={best_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Final polishing with a high‑α Adam (few ×10³ steps)
    a = best_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    lr0 = 0.6
    alpha = 50000.0
    lam = 0.0
    it = 0
    max_polish = 30_000
    while time.time() < deadline and it < max_polish:
        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[polish] iter {it:,} new best O={best_score:.6f}")

        grad = _smooth_gradient_plus_entropy(a, conv, alpha, L, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        lr = lr0 / math.sqrt(it + 1)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = _project_to_simplex(a_new)
        a = a_new
        it += 1

    # --------------------------------------------------------------
    # 5️⃣  Test monotone reorderings (ascending / descending) – the objective
    #     is invariant to scaling, only the ordering matters.
    candidates = [best_seq, np.sort(best_seq)[::-1], np.sort(best_seq)]
    for cand in candidates:
        obj = _objective_simplex(cand, n, L)
        if obj < best_score:
            best_score = obj
            best_seq = cand.copy()
            print(f"[reorder] improved O={best_score:.6f}")

    # --------------------------------------------------------------
    # Final sanity check with the external evaluator
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence failed:", exc)

    # Return the sequence as a plain Python list of floats
    return best_seq.tolist()
