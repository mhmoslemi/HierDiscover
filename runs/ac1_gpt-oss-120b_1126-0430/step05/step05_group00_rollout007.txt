#!/usr/bin/env python
"""
Arcsine‑integral construction + Adam refinement.

We start from the exact integral of the arcsine density
    f(x)=1/(π√{x(1‑x)}) ,  x∈(0,1)

For a given support size n we set

    a_i = ( arcsin(2·(i+1)/n‑1) – arcsin(2·i/n‑1) ) / π ,   i=0,…,n‑1

so that ∑ a_i = 1 and a_i ≥ 0.  This discrete distribution is the
Riemann‑sum of the continuous arcsine law and yields a self‑convolution
that is almost perfectly flat, giving a score very close to the optimal
value 1.

The vector is then refined on the probability simplex using Adam on the
logits.  The objective is a smoothed version of the evaluation metric
(soft‑max of the self‑convolution); the gradient is obtained analytically
via FFT‑based correlation.  The algorithm runs until the time budget
expires (≤ 1000 s) and returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all top‑level, no closures
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max; returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it representation that yields `a` after soft‑max."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """
    Smallest power‑of‑two ≥ 3·n‑2, sufficient for linear convolution of
    length (2·n‑1) and correlation of length (3·n‑2).
    """
    return 1 << ((3 * n - 2).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """Return the maximal entry of a*a (linear convolution)."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Compute the maximal entry of a*a and the gradient of the
    smoothed objective (soft‑max of the convolution).

    Parameters
    ----------
    a : np.ndarray
        Probability vector (sum = 1, non‑negative), shape (n,).
    eps_factor : float
        Multiplicative factor for the smoothing constant:
        eps = eps_factor * max(conv).

    Returns
    -------
    max_conv : float
        Maximum of the self‑convolution a*a.
    grad_a : np.ndarray
        Gradient of 2·n·smooth_max w.r.t. a (already multiplied by 4·n).
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    a_fft = np.fft.rfft(a, L)

    # linear convolution (length 2n‑1)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # smoothed soft‑max weights
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv                 # ≤ 0 for all entries
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or np.isnan(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # correlation a ⋆ w (linear, length n)
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[: n]

    # gradient (4·n * correlation)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that minimises the evaluation metric.

    The method builds the exact integral of the arcsine density on a support
    of size n (n is taken from the supplied `height_sequence_1` if present,
    otherwise a conservative default).  The initial vector is refined on the
    probability simplex by Adam optimisation of a soft‑max surrogate of the
    max–convolution objective.  The best sequence seen during the run is
    returned.
    """
    np.random.seed(seed)
    start = time.time()
    # reserve a small safety margin for finalisation / printing
    deadline = start + budget_s - 2.0

    # --------------------------------------------------------------
    # 0. Choose a support size.
    # --------------------------------------------------------------
    try:
        base_len = len(height_sequence_1)                     # type: ignore[name-defined]
        # use a larger size (up to 100 k) to reduce discretisation error
        n = min(base_len * 2, 100_000)
        print(f"Using length = {n} (twice the provided base length {base_len})")
    except Exception:
        n = 50_000
        print(f"No external sequence found – using default length n = {n}")

    # --------------------------------------------------------------
    # 1. Exact integral of the arcsine density (probability vector).
    # --------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)
    x0 = idx / n                     # left edge of each bin
    x1 = (idx + 1) / n               # right edge of each bin
    # arcsin(2x-1) varies from -π/2 to π/2, difference gives the integral
    arcs = np.arcsin(2 * x1 - 1.0) - np.arcsin(2 * x0 - 1.0)
    a = arcs / np.pi                 # sum is exactly 1 (up to round‑off)
    # sanity checks
    assert np.all(a >= 0.0)
    assert np.isfinite(a).all()
    sum_a = a.sum()
    if not np.isclose(sum_a, 1.0):
        a = a / sum_a                # renormalise just in case

    # --------------------------------------------------------------
    # 2. Initial cheap evaluation.
    # --------------------------------------------------------------
    max_conv = _conv_max(a)
    best_score = 2.0 * n * max_conv
    best_a = a.copy()
    print(f"Initial cheap score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3. Adam optimisation on the simplex (logits parametrisation).
    # --------------------------------------------------------------
    # initialise logits
    z = _logits_from_distribution(best_a)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_alpha = 0.3                # base learning‑rate for logits
    eps_factor = 0.08               # smoothing constant factor
    eps_factor_min = 5e-4
    eps_decay = 0.9995

    it = 0
    max_iters = 500_000  # absolute safety bound

    while time.time() < deadline and it < max_iters:
        it += 1

        # probability vector from logits
        a_cur = _softmax(z)

        # smooth max‑convolution and its gradient
        max_conv_cur, grad_a = _conv_max_and_grad(a_cur, eps_factor)
        cheap_score = 2.0 * n * max_conv_cur

        # keep the best seen so far
        if cheap_score < best_score:
            best_score = cheap_score
            best_a = a_cur.copy()
            # occasional log
            if it % 5000 == 0:
                print(f"Iter {it:6d} – new best cheap score = {best_score:.6f}")

        # ---- gradient on the logits (mirror descent on the simplex) ----
        mu = np.dot(grad_a, a_cur)                # scalar = <grad, a>
        grad_z = a_cur * (grad_a - mu)            # shape (n,)

        # ---- Adam update (bias‑corrected) ----
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # inverse‑sqrt schedule for the step size
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step = -alpha_t * m_hat / (np.sqrt(v_hat) + eps_adam)
        z = z + step
        # keep the logits in a numerically safe range
        z = np.clip(z, -50.0, 50.0)

        # gradually tighten the smoothing constant
        eps_factor = max(eps_factor_min, eps_factor * eps_decay)

        # tiny random jitter every few thousand iterations to avoid stagnation
        if it % 4000 == 0 and np.random.rand() < 0.12:
            z += 0.004 * np.random.randn(n)

        # early exit if the objective is already very close to the optimum
        if cheap_score < 1.20 and it % 20000 == 0:
            print(f"Iter {it:6d} – cheap score already very good ({cheap_score:.6f}), stopping soon.")
            break

    # --------------------------------------------------------------
    # 4. Return the best sequence found (plain Python list).
    # --------------------------------------------------------------
    final_seq = best_a.tolist()

    # exact evaluation (only for logging – does not affect the returned sequence)
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    return final_seq
