Multi‑phase optimisation for the step‑function objective

    O(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm is a blend of

* 0️⃣  cheap random Dirichlet restarts + a few Adam steps,
* 1️⃣  long Adam optimisation on a smooth log‑sum‑exp surrogate,
* 2️⃣  exact projected sub‑gradient descent on the true non‑smooth objective,
* 3️⃣  a tiny linear‑programming “active‑set” refinement that solves
   max ∑g_j  s.t.  a∗g ≤ current M  (M = max conv(a,a)), and finally
* 4️⃣  a targeted simulated‑annealing walk to polish the final result.

All phases respect the 1000 s wall‑time budget and keep the best
sequence seen.  The final list of heights is returned; the external
``evaluate_sequence`` routine will be called once for verification.

import time
import math
import numpy as np
import cvxpy as cp

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – projects onto {x≥0, ∑x=1}
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:          # all non‑positive → uniform simplex point
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full convolution via FFT (real‑valued signals)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return the full self‑convolution a*a using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    # keep only the first 2·len(a)-1 entries (the true convolution length)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Objective for a simplex‑normalised vector (∑a = 1)
def _objective_simplex(a: np.ndarray, n: int, L: int) -> float:
    """2·n·max(conv(a,a))  (a is on the simplex)."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) w.r.t. a (used for exact sub‑gradient descent)
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Return a sub‑gradient of the max‑convolution entry."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
# Gradient of smooth surrogate (log‑sum‑exp of the convolution)
def _smooth_gradient(a: np.ndarray, conv: np.ndarray,
                     alpha: float, L: int) -> np.ndarray:
    """
    Compute ∇_a ( (2·n/α)·log∑_k exp(α·conv_k) ).
    The result reduces to 2·corr(w, a) where w_k ∝ exp(α·conv_k).
    """
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)          # stabilise
    w = exp_scaled / exp_scaled.sum()

    # correlation w ⋆ a  (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n_len = a.size
    grad = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the smooth surrogate
def _adam_phase(a: np.ndarray, n: int, L: int,
                max_iters: int, deadline: float,
                alpha_start: float = 5.0,
                alpha_end: float = 20000.0,
                lr0: float = 0.6,
                beta1: float = 0.9,
                beta2: float = 0.999,
                eps: float = 1e-12,
                prune_every: int = 0,
                prune_percentile: float = 1.0) -> tuple[np.ndarray, float, np.ndarray]:
    """Adam optimisation of the smooth log‑sum‑exp surrogate."""
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_obj = _objective_simplex(a, n, L)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()

        # schedule the smoothness parameter α (exponential increase)
        frac = it / max_iters
        alpha = alpha_start * (alpha_end / alpha_start) ** frac

        grad = _smooth_gradient(a, conv, alpha, L)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        lr = lr0 / math.sqrt(it + 1)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # tiny jitter to fight plateaus
        if np.random.rand() < 0.001:
            a_new += np.random.randn(n) * 1e-4

        a_new = _project_to_simplex(a_new)

        # optional pruning of tiny entries
        if prune_every and it % prune_every == 0 and it > 0:
            thresh = np.percentile(a_new, prune_percentile)
            a_new[a_new < thresh] = 0.0
            a_new = _project_to_simplex(a_new)

        a = a_new

    return a, best_obj, best_a


# ----------------------------------------------------------------------
# Exact projected sub‑gradient descent on the true objective
def _subgradient_phase(a: np.ndarray, n: int, L: int,
                       max_iters: int, deadline: float,
                       eta0: float = 0.3,
                       line_search: bool = True,
                       verbose: bool = False) -> tuple[np.ndarray, float, np.ndarray]:
    """Projected sub‑gradient descent with simple back‑tracking line search."""
    best_a = a.copy()
    best_obj = _objective_simplex(a, n, L)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            if verbose and it % 2000 == 0:
                print(f"[sub‑grad] iter {it}  O={best_obj:.6f}")

        g = _subgradient_max_conv(a, conv)

        # geometric decay of the step size
        eta = eta0 * (0.9999 ** it)

        if line_search:
            step = eta
            for _ in range(5):
                cand = _project_to_simplex(a - step * g)
                cand_obj = _objective_simplex(cand, n, L)
                if cand_obj < cur_obj:
                    a = cand
                    break
                step *= 0.5
            else:
                a = _project_to_simplex(a - eta * g)
        else:
            a = _project_to_simplex(a - eta * g)

    return a, best_obj, best_a


# ----------------------------------------------------------------------
# Linear‑programming active‑set refinement
def _lp_refinement(a: np.ndarray, n: int, L: int,
                   max_outer_iters: int, deadline: float,
                   step: float = 0.5,
                   tol_frac: float = 1e-3) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve a tiny LP:
        maximise   1ᵀg
        subject to   a∗g ≤ M   (only on the active constraints)
                     g ≥ 0
    Scale g to match ∑a and take a convex combination.
    """
    best_a = a.copy()
    best_obj = _objective_simplex(a, n, L)

    for outer in range(max_outer_iters):
        if time.time() >= deadline:
            break

        conv = _conv_fft(a, L)
        M = conv.max()
        # active set – indices whose convolution value is within a tiny fraction of the max
        eps = max(1e-12, tol_frac * M)
        act_idx = np.where(conv >= M - eps)[0]
        if act_idx.size == 0:
            break

        # Build the (|act| × n) coefficient matrix C where C_kj = a_{k−j} (or 0)
        C = np.zeros((act_idx.size, n), dtype=float)
        for row, k in enumerate(act_idx):
            j_start = max(0, k - (n - 1))
            j_end = min(n - 1, k)
            j_range = np.arange(j_start, j_end + 1, dtype=int)
            C[row, j_range] = a[k - j_range]

        # Linear programme
        g = cp.Variable(n, nonneg=True)
        constraints = [C @ g <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2500,
                       abstol=1e-9, reltol=1e-9, feastol=1e-9)
        except Exception as exc:
            print("[lp] Solver failed:", exc)
            break

        if g.value is None:
            break
        sum_g = np.sum(g.value)
        if sum_g <= 1e-12:
            break

        # Scale g to keep the total mass unchanged (a is on the simplex → sum=1)
        lam = a.sum() / sum_g
        g_scaled = lam * g.value

        # Convex combination
        a_candidate = (1.0 - step) * a + step * g_scaled
        a_candidate = _project_to_simplex(a_candidate)

        cand_obj = _objective_simplex(a_candidate, n, L)
        if cand_obj < best_obj - 1e-12:
            best_obj = cand_obj
            best_a = a_candidate.copy()
            a = a_candidate
            # become slightly more aggressive when we see improvement
            step = min(0.9, step * 1.05)
        else:
            # no progress – shrink the step size
            step *= 0.5
            if step < 1e-4:
                break

    return best_a, best_obj


# ----------------------------------------------------------------------
# Targeted simulated annealing – fine‑grained polishing
def _simulated_annealing_phase(a: np.ndarray, n: int, L: int,
                               max_iters: int, deadline: float,
                               temperature0: float = 0.07,
                               no_improve_limit: int = 8000,
                               verbose: bool = True) -> tuple[np.ndarray, float, np.ndarray]:
    """Local refinement: move a fraction of mass from a peak contributor to a random site."""
    best_a = a.copy()
    best_obj = _objective_simplex(a, n, L)
    cur_obj = best_obj
    conv = _conv_fft(a, L)

    T = temperature0
    no_improve = 0

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # most violating convolution index
        kmax = int(np.argmax(conv))

        # indices that actually contribute to that entry (and carry some mass)
        i_cand = np.arange(max(0, kmax - (n - 1)), min(n - 1, kmax) + 1)
        i_cand = i_cand[a[i_cand] > 1e-14]
        if i_cand.size == 0:
            i = np.random.randint(n)
        else:
            i = np.random.choice(i_cand)

        # destination index (avoid self‑transfer)
        j = np.random.randint(n)
        while j == i:
            j = np.random.randint(n)

        # move a modest share of the source mass
        delta = a[i] * np.random.rand() * 0.09
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = _project_to_simplex(a_new)

        conv_new = _conv_fft(a_new, L)
        new_obj = 2.0 * n * conv_new.max()

        # Metropolis acceptance
        if new_obj < cur_obj or np.random.rand() < math.exp((cur_obj - new_obj) / max(T, 1e-12)):
            a = a_new
            conv = conv_new
            cur_obj = new_obj
            no_improve = 0
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                if verbose:
                    print(f"[SA] iter {it:5d}  new best O={best_obj:.6f}")
        else:
            no_improve += 1

        # occasional exact sub‑gradient step to keep a moving in a good direction
        if it % 9 == 0:
            g = _subgradient_max_conv(a, conv)
            eta_loc = 0.07 * (1.0 - it / max_iters)
            a = _project_to_simplex(a - eta_loc * g)
            conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                if verbose:
                    print(f"[SA] sub‑grad step O={best_obj:.6f}")

        # temperature decay
        T *= 0.9997

        # restart if stuck for a while
        if no_improve >= no_improve_limit:
            a = best_a.copy()
            a += np.random.randn(n) * 0.0015
            a = np.clip(a, 0, None)
            a = _project_to_simplex(a)
            conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max()
            no_improve = 0

    return a, best_obj, best_a

# Main entry point expected by the evaluation harness
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the provided
    evaluation metric.  Returns the height list (Python floats).

    The method proceeds through four stages:
        0 – quick random Dirichlet restarts + short Adam runs,
        1 – long Adam on a smooth surrogate,
        2 – exact projected sub‑gradient descent,
        3 – LP‑based active‑set refinement,
        4 – targeted simulated annealing polish.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # small safety margin

    # ------------------------------------------------------------------
    # Initialise the problem size and a starting vector
    if 'height_sequence_1' in globals():
        a0 = np.array(height_sequence_1, dtype=float)
        a0 = np.maximum(a0, 0.0)
        if a0.sum() == 0.0:
            a0 = np.full_like(a0, 1.0)
        a0 = a0 / a0.sum()               # normalise to simplex
        n = a0.size
        a = a0.copy()
        print(f"[init] using supplied sequence, n={n}")
    else:
        n = 4600
        a = np.full(n, 1.0 / n)           # uniform start on the simplex
        print(f"[init] using uniform start, n={n}")

    # FFT length – next power of two greater than 3·n‑2 (ensures no circular wrap‑around)
    L = 1 << ((3 * n - 2).bit_length())

    # ------------------------------------------------------------------
    # Phase 0 – cheap random restarts to obtain a good seed
    phase0_deadline = start_time + budget_s * 0.13
    best_seq = a.copy()
    best_score = _objective_simplex(best_seq, n, L)

    while time.time() < phase0_deadline:
        conc = 0.4                         # Dirichlet concentration (<1 ⇒ sparse)
        a_rand = np.random.gamma(conc, 1.0, size=n)
        a_rand = a_rand / a_rand.sum()
        sub_dead = time.time() + 3.5       # give each trial a few seconds
        a_rand, _, best_rand = _adam_phase(
            a_rand, n, L,
            max_iters=1500,
            deadline=sub_dead,
            alpha_start=5.0,
            alpha_end=2000.0,
            lr0=0.8,
            prune_every=0
        )
        cur_obj = _objective_simplex(best_rand, n, L)
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = best_rand.copy()
            print(f"[phase0] new best O={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – long Adam optimisation on the smoothed surrogate
    phase1_deadline = start_time + budget_s * 0.45
    a, obj_after_adam, best_adam = _adam_phase(
        best_seq, n, L,
        max_iters=400_000,
        deadline=phase1_deadline,
        alpha_start=5.0,
        alpha_end=50_000.0,
        lr0=0.6,
        prune_every=5000,
        prune_percentile=0.5
    )
    if obj_after_adam < best_score:
        best_score = obj_after_adam
        best_seq = best_adam.copy()
        print(f"[phase1] improved best O={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – exact sub‑gradient descent with back‑tracking line search
    phase2_deadline = start_time + budget_s * 0.70
    a, obj_sub, best_sub = _subgradient_phase(
        best_seq, n, L,
        max_iters=250_000,
        deadline=phase2_deadline,
        eta0=0.25,
        line_search=True,
        verbose=False
    )
    if obj_sub < best_score:
        best_score = obj_sub
        best_seq = best_sub.copy()
        print(f"[phase2] improved best O={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – linear‑programming active‑set refinement
    phase3_deadline = start_time + budget_s * 0.85
    a, obj_lp = _lp_refinement(
        best_seq, n, L,
        max_outer_iters=12,
        deadline=phase3_deadline,
        step=0.5,
        tol_frac=1e-3
    )
    if obj_lp < best_score:
        best_score = obj_lp
        best_seq = a.copy()
        print(f"[phase3‑LP] improved best O={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – stochastic annealing polish
    phase4_deadline = deadline
    a, obj_sa, best_sa = _simulated_annealing_phase(
        best_seq, n, L,
        max_iters=300_000,
        deadline=phase4_deadline,
        temperature0=0.07,
        no_improve_limit=7000,
        verbose=True
    )
    if obj_sa < best_score:
        best_score = obj_sa
        best_seq = best_sa.copy()
        print(f"[phase4‑SA] final best O={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final verification via the supplied evaluator (just for sanity)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence failed:", exc)

    # The harness expects a plain Python list
    return best_seq.tolist()
