"""
Improved search for a non‑negative step function that minimises

        2·n·max_k (a*a)[k] / (Σ a_i)²

The core of the new method is a *soft‑max* weighted exponentiated‑gradient
update.  At each iteration we
  1. compute the full autocorrelation by FFT,
  2. keep the *top‑k* shifts (k≈10) where the autocorrelation is largest,
  3. turn the values of those shifts into a probability distribution via a
     low‑temperature soft‑max,
  4. build a (weighted) sub‑gradient of the smoothed max‑objective,
  5. apply an exponentiated‑gradient (mirror descent) step with a
     momentum term, and finally
  6. re‑project onto the probability simplex.

Compared with the previous heavy‑ball implementation this variant
  * concentrates the gradient on all “active’’ convolution entries
    (instead of a few un‑weighted ones),
  * uses a temperature parameter to obtain a smoother signal for the
    gradient,
  * adds a simple Adam‑like momentum,
  * keeps the same cheap‑family initialisation and a small‑perturbation
    restart policy.

The routine respects the supplied time‑budget (default 1000 s) and always
returns a feasible sequence.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full linear convolution a*a using FFT (returns length 2·n‑1 array).
    a is assumed to be real and non‑negative.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa).real[: 2 * a.size - 1]
    # tiny negative values can appear due to round‑off – clamp to zero
    conv[conv < 0] = 0.0
    return conv
