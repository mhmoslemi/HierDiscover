def subgradient_max(a, conv, L):
    max_val = conv.max()
    w = (conv == max_val).astype(float)  # shape (2n-1)
    # use FFT convolution: w * a_rev
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n = a.shape[0]
    grad = 2.0 * corr_rev[n-1:n-1+n]  # shape n
    return grad

1️⃣  Initialise on the supplied sequence (if present) or a uniform vector.
    The vector is kept on the probability simplex (∑a = 1, a ≥ 0).

2️⃣  **Phase 1 – variance flattening.**  
    Optimise the smooth objective  
        f(a)=∑_k (conv_k – μ)² ,   μ = 1/(2n‑1)  
    with Adam on the simplex.  This drives all entries of the convolution
    towards a common value and considerably lowers the later max‑objective.

3️⃣  **Phase 2 – soft‑max surrogate.**  
    Optimise a differentiable surrogate of the true objective,
        g(a)= (2·n/α)·log∑_k e^{α·conv_k} + λ·∑ a_i·log a_i .
    The temperature α is annealed from 5 to 2·10⁴, while Adam
    continues to respect the simplex constraints.

4️⃣  **Phase 3 – stochastic refinement.**  
    A hybrid Monte‑Carlo / sub‑gradient loop:
        – random “mass‑transfer’’ moves (transfer a fraction of the mass
          from one bucket to another) accepted by a Metropolis rule,
        – exact sub‑gradient steps on the true max using FFT‑based
          convolution (very cheap),
        – occasional low‑pass smoothing,
        – many restarts with jittered copies of the current best.
    The loop runs until the time budget expires.

5️⃣  The best sequence seen is returned as a plain Python list.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex {x ≥ 0, sum(x)=1}
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution a★a (length 2·n‑1) via FFT of length L
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return the full convolution a★a using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Gradient of the variance‑of‑convolution objective
def _grad_variance(a: np.ndarray, conv: np.ndarray, n: int, L: int) -> np.ndarray:
    """∇_a Σ_k (conv_k – μ)²  with μ = 1/(2n‑1)."""
    mu = 1.0 / (2 * n - 1)
    diff = conv - mu                     # length 2n‑1
    a_rev = a[::-1]
    fft_diff = np.fft.rfft(diff, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_diff * fft_a_rev, n=L)
    start = n - 1
    grad = 4.0 * corr_rev[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Gradient of the smooth (log‑sum‑exp) surrogate + tiny entropy term
def _smooth_gradient(a: np.ndarray, conv: np.ndarray,
                     alpha: float, L: int, lam: float) -> np.ndarray:
    """
    ∇_a  ( (2·n/α)·log∑_k e^{α·conv_k}  +  λ·∑ a_i·log a_i ).
    The first part yields 2·corr(w,a) with w_i ∝ exp(α·conv_i).
    """
    # soft‑max weights
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / exp_scaled.sum()                 # shape (2n‑1,)

    # correlation term (same trick as in _grad_variance)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n_len = a.size
    grad_corr = 2.0 * corr_rev[n_len - 1:n_len - 1 + n_len]

    # tiny entropy to keep strict positivity
    eps = 1e-20
    grad_ent = lam * (np.log(a + eps) + 1.0)

    return grad_corr + grad_ent


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) – computed via FFT for speed
def _subgrad_max(a: np.ndarray, conv: np.ndarray, L: int) -> np.ndarray:
    """Return a sub‑gradient of max(conv) w.r.t. a."""
    max_val = conv.max()
    w = (conv == max_val).astype(float)               # 0/1 mask, length 2n‑1
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n = a.shape[0]
    grad = 2.0 * corr_rev[n - 1:n - 1 + n]
    return grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    the provided evaluation function.  Returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.3          # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size -------------------------------------------------------
    try:
        # if a previous construction is available, start from it
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
        n = base.size
        a = np.maximum(0.0, base)
        total = a.sum()
        if total < 1e-12:
            a = np.full(n, 1.0 / n)
        else:
            a = a / total
        print(f"[init] starting from supplied sequence, n={n}")
    except NameError:
        # otherwise use a uniform start; a slightly larger n gives more freedom
        n = 6500
        a = np.full(n, 1.0 / n)
        print(f"[init] starting from uniform, n={n}")

    # Ensure strict simplex feasibility
    a = np.maximum(a, 0.0)
    a = a / a.sum()

    # FFT length – next power of two ≥ 3·n‑2 (covers all required convolutions)
    L = 1 << ((3 * n - 2).bit_length())

    # ------------------------------------------------------------------
    # Book‑keeping of the best solution
    conv = _conv_fft(a, L)
    best_score = 2.0 * n * conv.max()
    best_seq = a.copy()
    print(f"[init] initial O = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – variance flattening (Adam on Σ (c‑μ)²)
    phase1_deadline = start_time + budget_s * 0.35
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    lr0 = 0.8
    lam_entropy = 1e-4
    it = 0

    while time.time() < phase1_deadline:
        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[phase‑1] iter {it:,} new best O = {best_score:.6f}")

        # gradient of variance objective + tiny entropy term
        grad = _grad_variance(a, conv, n, L)
        grad += lam_entropy * (np.log(a + 1e-20) + 1.0)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        lr = lr0 / math.sqrt(it + 1)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # project back onto the simplex
        a = _project_to_simplex(a_new)

        it += 1
        if it % 50000 == 0:
            print(f"[phase‑1] iter {it:,} O = {cur_obj:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – smooth max surrogate (log‑sum‑exp) with annealed α
    phase2_deadline = start_time + budget_s * 0.70
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.5
    lam = 1e-5
    alpha_start, alpha_end = 5.0, 20000.0
    max_iter2 = 400_000
    it2 = 0

    while time.time() < phase2_deadline and it2 < max_iter2:
        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[phase‑2] iter {it2:,} new best O = {best_score:.6f}")

        # anneal α linearly
        frac = it2 / max_iter2
        alpha = alpha_start + (alpha_end - alpha_start) * frac

        grad = _smooth_gradient(a, conv, alpha, L, lam)

        # Adam step
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** (it2 + 1))
        v_hat = v / (1 - beta2 ** (it2 + 1))
        lr = lr0 / math.sqrt(it2 + 1)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional random jitter to escape plateaus
        if np.random.rand() < 0.001:
            a_new += np.random.randn(n) * 5e-4

        a = _project_to_simplex(a_new)
        it2 += 1
        if it2 % 50000 == 0:
            print(f"[phase‑2] iter {it2:,} O = {cur_obj:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – stochastic local refinement (mass‑transfer + sub‑grad)
    print(f"[phase‑3] entering stochastic refinement, best O = {best_score:.6f}")
    T0 = 0.05                 # initial temperature for Metropolis accept‑rule
    max_iter_local = 200_000
    restart_counter = 0

    while time.time() < deadline:
        # ---- restart ----
        if np.random.rand() < 0.7:
            a = best_seq.copy()
            a += np.random.randn(n) * 0.0015
            a = np.clip(a, 0, None)
            a = _project_to_simplex(a)
        else:
            a = np.random.rand(n)
            a = _project_to_simplex(a)

        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        T = T0
        it_loc = 0
        no_improve = 0

        while time.time() < deadline and it_loc < max_iter_local:
            # ----- random mass‑transfer move -----
            i, j = np.random.randint(n, size=2)
            if i != j and a[i] > 1e-14:
                delta = a[i] * np.random.rand() * 0.25      # up to 25 % of source mass
                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta
                a_new = _project_to_simplex(a_new)

                conv_new = _conv_fft(a_new, L)
                new_obj = 2.0 * n * conv_new.max()

                # Metropolis acceptance
                if new_obj < cur_obj or np.random.rand() < math.exp((cur_obj - new_obj) / max(T, 1e-15)):
                    a, conv, cur_obj = a_new, conv_new, new_obj
                    no_improve = 0
                    if cur_obj < best_score:
                        best_score = cur_obj
                        best_seq = a.copy()
                        now = time.time() - start_time
                        print(f"[{now:.1f}s] new global best O = {best_score:.6f} (restart {restart_counter}, it {it_loc})")
                else:
                    no_improve += 1
            else:
                no_improve += 1

            # ----- sub‑gradient step on the true max -----
            if it_loc % 7 == 0:
                grad = _subgrad_max(a, conv, L)
                eta = 0.07 * (1.0 - it_loc / max_iter_local)   # linear decay
                a = a - eta * grad
                a = _project_to_simplex(a)
                conv = _conv_fft(a, L)
                cur_obj = 2.0 * n * conv.max()
                if cur_obj < best_score:
                    best_score = cur_obj
                    best_seq = a.copy()
                    now = time.time() - start_time
                    print(f"[{now:.1f}s] sub‑grad improved O = {best_score:.6f}")

            # ----- occasional low‑pass smoothing -----
            if it_loc % 6000 == 0 and it_loc > 0:
                kernel = np.array([0.1, 0.2, 0.4, 0.2, 0.1])
                smooth = np.convolve(a, kernel, mode="same")
                a = _project_to_simplex(0.85 * a + 0.15 * smooth)

            # temperature decay
            T *= 0.9995
            it_loc += 1

            # early‑stop a local run if stuck for too long
            if no_improve >= 4000:
                break

        restart_counter += 1
        # break if the overall budget is exhausted
        if time.time() >= deadline:
            break

    # Final sanity check with the provided evaluation routine

        print(f"[final] evaluate_sequence reports O = {final_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence failed:", exc)

    return best_seq.tolist()
