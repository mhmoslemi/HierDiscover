"""
Hybrid optimiser for the step-function anti‑convolution problem.

The algorithm consists of three stages:

1. Fast Adam optimisation on the simplex (soft‑max parametrisation) for a
   relatively large support (default 8000).  Several short random restarts
   are performed; the best result is kept.

2. The best sequence is resampled to a moderate size (≈2000) and an
   alternating linear‑programming refinement is applied.  For a fixed
   sequence `f` we solve a linear programme that maximises the total mass
   of a new sequence `g` under the constraint ‑‑ `f*g ≤ max(f*f)`.  Mixing
   `f` and the normalised `g` strictly reduces the objective; the process
   is repeated until the time budget expires.

3. A short targeted simulated‑annealing / mass‑transfer local search is
   executed to polish the final vector.

All steps respect the global 1000 s budget.  The returned list contains
only non‑negative floats; it is already normalised to the required sum
√(2·n) so that ``evaluate_sequence`` can be called directly.
"""

import time
import numpy as np

# try importing scipy, else fall back to pure‑numpy ADAM only
try:
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False

# ------- helper functions -------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()

def _conv_full(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a."""
    return np.convolve(a, a)

def _build_conv_matrix(f: np.ndarray) -> csr_matrix:
    """Return sparse matrix A such that A @ g = conv(f,g) (full)."""
    n = f.shape[0]
    # rows = i + j for all i,j
    rows = np.add.outer(np.arange(n), np.arange(n)).ravel()
    cols = np.tile(np.arange(n), n)
    data = np.repeat(f, n)
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

def _adam_optimize(n: int, time_limit: float, rng: np.random.Generator,
                   init_a: np.ndarray | None = None, start_lr: float = 0.15,
                   end_lr: float = 0.001, start_beta: float = 0.5,
                   end_beta: float = 20000.0) -> np.ndarray:
    """Adam on the simplex using a smooth‑max approximation."""
    start = time.time()
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
        a = a / a.sum()
    else:
        a = rng.random(n)
        a = a / a.sum()
    # target sum √(2n)
    target_sum = np.sqrt(2.0 * n)
    a = a * target_sum
    # work in log‑it space
    x = np.log(a + 1e-16)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    best_a = a.copy()
    best_obj = float('inf')
    it = 0
    while time.time() - start < time_limit:
        it += 1
        a = _softmax(x) * target_sum    # stay on simplex with required sum
        conv = _conv_full(a)
        M = conv.max()
        obj = float(2.0 * n * M / (target_sum ** 2))  # = M effectively
        if obj < best_obj - 1e-14:
            best_obj, best_a = obj, a.copy()
        # schedule
        elapsed = time.time() - start
        t = min(elapsed / time_limit, 1.0)
        beta = start_beta * np.exp(np.log(end_beta / start_beta) * t)
        lr = start_lr * np.exp(np.log(end_lr / start_lr) * t)
        # smooth max weights
        if beta > 5000.0:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            shifted = conv - M
            w = np.exp(beta * shifted)
            ws = w.sum()
            if ws == 0.0:
                w = np.ones_like(conv) / conv.size
            else:
                w /= ws
        # gradient of smooth max
        # convolution of w with a (correlation)
        # correlation can be computed via FFT for speed, but for moderate n np.convolve loops ok
        # Using FFT for consistency
        n_full = conv.size
        # correlation a * w (with reversed w) = (a * rev(w))[k] = sum_i a_i * w_{k+i}
        # simpler: we can compute using np.fft
        L = 1 << ((n_full + n - 1).bit_length())
        Af = np.fft.rfft(a, n=L)
        Wf = np.fft.rfft(w, n=L)
        corr = np.fft.irfft(Af * np.conj(Wf), n=L)[:n]
        g_a = 2.0 * corr
        # natural gradient on logits
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)
        # Adam update
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)
        if it % 5000 == 0:
            x += rng.normal(scale=0.01, size=n)
        if it % 80000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.005, size=n)
            m.fill(0.0)
            v.fill(0.0)
    return best_a

def _lp_refine(a: np.ndarray, time_limit: float,
               rng: np.random.Generator,
               mix_factor: float = 0.5,
               min_mix: float = 0.05,
               decay: float = 0.95) -> np.ndarray:
    """Alternating linear‑programming refinement (see description)."""
    start = time.time()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)
    a = np.clip(a, 0.0, None)
    a = a / a.sum() * S
    best_a = a.copy()
    best_obj = float('inf')
    it = 0
    while time.time() - start < time_limit:
        it += 1
        conv = _conv_full(a)
        M = conv.max()
        obj = float(2.0 * n * M / (S ** 2))   # = M
        if obj < best_obj:
            best_obj, best_a = obj, a.copy()
        # build constraint matrix for conv(f,g) <= M
        A = _build_conv_matrix(a)
        b = np.full(2 * n - 1, M, dtype=float)
        c = -np.ones(n)   # maximise sum(g)
        # solve LP
        if _SCIPY_AVAILABLE:
            # give a per‑iteration timeout to respect overall budget
            remaining = max(0.0, time_limit - (time.time() - start))
            # ``highs`` respects the ``time_limit`` option from SciPy 1.9 onward.
            try:
                res = linprog(c, A_ub=A, b_ub=b,
                              bounds=(0, None),
                              method='highs',
                              options={'presolve': True,
                                       'time_limit': remaining})
            except Exception:  # pragma: no cover
                # fallback to interior‑point if time_limit not understood
                res = linprog(c, A_ub=A, b_ub=b,
                              bounds=(0, None),
                              method='highs-ds')
        else:
            # No scipy – abort LP refinement
            break
        if not res.success:
            # abort when LP fails (numerical issues)
            print(f"[LP it={it}] linprog failed: {res.message}")
            break
        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 < 1e-12:
            break
        # scale to target sum
        g = g0 * (S / sum_g0)
        # mix
        a = (1 - mix_factor) * a + mix_factor * g
        a = np.clip(a, 0.0, None)
        a = a / a.sum() * S
        # adapt mixing factor
        mix_factor = max(min_mix, mix_factor * decay)
        # early exit if improvement tiny
        if it > 2 and abs(best_obj - obj) < 1e-7:
            break
    return best_a

def _local_search(a: np.ndarray, time_limit: float,
                  rng: np.random.Generator) -> np.ndarray:
    """Simple targeted simulated‑annealing (mass‑transfer) on a moderate size."""
    start = time.time()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)
    a = np.clip(a, 0.0, None)
    a = a / a.sum() * S
    best_a = a.copy()
    conv = _conv_full(a)
    best_obj = float(2.0 * n * conv.max() / (S ** 2))
    T = best_obj * 0.2 + 1e-6
    T_min = best_obj * 1e-6 + 1e-9
    decay = 0.995
    it = 0
    while time.time() - start < time_limit:
        it += 1
        conv = _conv_full(a)
        k_max = int(np.argmax(conv))
        # compute contributions to the max entry
        i_vals = np.arange(n)
        j_vals = k_max - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i_vals[mask]] * a[j_vals[mask]]
        total = contrib.sum()
        if total > 0:
            probs = contrib / total
            i = rng.choice(n, p=probs)
        else:
            i = rng.integers(n)
        # destination index
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.08   # up to 8 % of mass
        if delta <= 0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.clip(a_new, 0.0, None)
        a_new /= a_new.sum() * (S / a_new.sum())  # re‑normalise to S
        conv_new = _conv_full(a_new)
        obj_new = float(2.0 * n * conv_new.max() / (S ** 2))
        if obj_new < best_obj - 1e-12:
            best_obj, best_a = obj_new, a_new.copy()
            a = a_new
        else:
            prob = np.exp(-(obj_new - best_obj) / T) if T > 0 else 0.0
            if rng.random() < prob:
                a = a_new
        if it % 2000 == 0:
            T = max(T * decay, T_min)
    return best_a

def _compress(seq: np.ndarray, min_val: float = 1e-12) -> np.ndarray:
    """Remove near‑zero entries and renormalise to √(2·len)."""
    mask = seq > min_val
    if mask.sum() < seq.size:
        seq = seq[mask]
    S = np.sqrt(2.0 * seq.size)
    return seq / seq.sum() * S

"""
Hybrid optimiser for the step‑function anti‑convolution problem.

The algorithm works in three stages:

1️⃣  Fast Adam optimisation on the simplex (soft‑max parametrisation) for a
    relatively large support (default 8000).  A few short random restarts are
    performed and the best result is kept.

2️⃣  The best vector is interpolated to a moderate size (≈2000) and an
    alternating linear‑programming refinement is applied.  For a fixed
    sequence f we solve a linear programme that maximises the total mass of a
    new sequence g under the linear constraints  f∗g ≤ max(f∗f).  Mixing the
    normalised g with f strictly reduces the objective; the process repeats
    until the remaining time budget expires.

3️⃣  A short targeted simulated‑annealing / mass‑transfer local search
    polishes the final vector.

All steps respect the global 1000 s budget.  The returned list contains only
non‑negative floats; it is already normalised to the required sum √(2·n) so
that ``evaluate_sequence`` can be called directly.
"""

import time
import numpy as np

# --------------------------------------------------------------------------- #
# Optional SciPy imports – used for the LP refinement.
try:
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False

# --------------------------------------------------------------------------- #
# Helper functions
def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


def _conv_full(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a."""
    return np.convolve(a, a)


def _build_conv_matrix(f: np.ndarray) -> csr_matrix:
    """
    Return a sparse matrix A such that A @ g == conv(f, g) (full convolution).

    Row index k (0 … 2·n‑2) contains the coefficients f[i] for every i+j=k.
    """
    n = f.shape[0]
    rows = np.add.outer(np.arange(n), np.arange(n)).ravel()
    cols = np.tile(np.arange(n), n)
    data = np.repeat(f, n)
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def _adam_optimize(
    n: int,
    time_limit: float,
    rng: np.random.Generator,
    init_a: np.ndarray | None = None,
    start_lr: float = 0.15,
    end_lr: float = 0.001,
    start_beta: float = 0.5,
    end_beta: float = 20000.0,
) -> np.ndarray:
    """
    Adam optimiser on the simplex using a smooth‑max approximation.
    The returned vector has sum √(2·n) (the normalisation used in the
    evaluation function).
    """
    start = time.time()
    target_sum = np.sqrt(2.0 * n)

    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
        a = a / a.sum()
    else:
        a = rng.random(n)
        a = a / a.sum()
    a = a * target_sum

    # log‑it (soft‑max) parametrisation
    x = np.log(a + 1e-16)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    best_a = a.copy()
    best_obj = float('inf')
    it = 0

    while time.time() - start < time_limit:
        it += 1
        # stay on the simplex with the required sum
        a = _softmax(x) * target_sum

        conv = _conv_full(a)
        M = conv.max()
        obj = float(2.0 * n * M / (target_sum ** 2))   # = M
        if obj < best_obj - 1e-14:
            best_obj, best_a = obj, a.copy()
            print(f"[ADAM it={it:06d}] new best O={best_obj:.6f}")

        # schedule (time‑based)
        elapsed = time.time() - start
        t = min(elapsed / time_limit, 1.0)
        beta = start_beta * np.exp(np.log(end_beta / start_beta) * t)
        lr = start_lr * np.exp(np.log(end_lr / start_lr) * t)

        # smooth‑max weighting for the max‑convolution
        if beta > 5000.0:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            shifted = conv - M              # ≤ 0
            w = np.exp(beta * shifted)
            s = w.sum()
            if s == 0.0:
                w = np.ones_like(conv) / conv.size
            else:
                w /= s

        # gradient of the (smooth) max w.r.t. a
        # correlation w * a  (implemented via FFT)
        n_full = conv.size
        L = 1 << ((n_full + n - 1).bit_length())
        Af = np.fft.rfft(a, n=L)
        Wf = np.fft.rfft(w, n=L)
        corr = np.fft.irfft(Wf * np.conj(Af), n=L)[:n]   # correlation
        g_a = 2.0 * corr

        # natural gradient in log‑it space
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter / restart
        if it % 5000 == 0:
            x += rng.normal(scale=0.01, size=n)
        if it % 80000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.005, size=n)
            m.fill(0.0)
            v.fill(0.0)

    return best_a


def _lp_refine(
    a: np.ndarray,
    time_limit: float,
    rng: np.random.Generator,
    mix_factor: float = 0.5,
    min_mix: float = 0.05,
    decay: float = 0.93,
) -> np.ndarray:
    """
    Alternating linear‑programming refinement.

    At each iteration we maximise Σg_j subject to conv(a, g) ≤ M,
    where M = max(conv(a, a)).  The optimiser g is then scaled to the target
    sum √(2·n) and mixed with a.
    """
    start = time.time()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    a = np.clip(a, 0.0, None)
    a = a / a.sum() * S

    best_a = a.copy()
    best_obj = float('inf')
    it = 0

    while time.time() - start < time_limit:
        it += 1
        conv = _conv_full(a)
        M = conv.max()
        obj = float(2.0 * n * M / (S ** 2))   # = M
        if obj < best_obj:
            best_obj, best_a = obj, a.copy()
            print(f"[LP it={it}] new best O={best_obj:.6f}")

        # build linear‑program constraints conv(a, g) ≤ M
        if not _SCIPY_AVAILABLE:
            break
        A = _build_conv_matrix(a)
        b = np.full(2 * n - 1, M, dtype=float)
        c = -np.ones(n)          # maximise Σ g  → minimise –Σ g

        remaining = max(0.0, time_limit - (time.time() - start))
        try:
            # SciPy ≥1.9 understands ``time_limit`` for the HiGHS methods.
            res = linprog(
                c,
                A_ub=A,
                b_ub=b,
                bounds=(0, None),
                method='highs',
                options={'presolve': True, 'time_limit': remaining},
            )
        except Exception:  # pragma: no cover
            # Fallback – older SciPy versions.
            res = linprog(
                c,
                A_ub=A,
                b_ub=b,
                bounds=(0, None),
                method='highs-ds',
            )

        if not res.success:
            print(f"[LP it={it}] linprog failed: {res.message}")
            break

        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 < 1e-12:
            break

        # scale to the required sum and mix
        g = g0 * (S / sum_g0)
        a = (1.0 - mix_factor) * a + mix_factor * g
        a = np.clip(a, 0.0, None)
        a = a / a.sum() * S

        # decay the mixing factor (more conservative later)
        mix_factor = max(min_mix, mix_factor * decay)

        # stop early if improvement stalls
        if it > 2 and abs(best_obj - obj) < 1e-7:
            break

    return best_a


def _local_search(
    a: np.ndarray,
    time_limit: float,
    rng: np.random.Generator,
) -> np.ndarray:
    """
    Targeted simulated‑annealing / mass‑transfer polish on a moderate‑size
    vector.  Small amounts of mass are moved from indices that heavily
    contribute to the current maximal convolution entry.
    """
    start = time.time()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    a = np.clip(a, 0.0, None)
    a = a / a.sum() * S

    conv = _conv_full(a)
    best_obj = float(2.0 * n * conv.max() / (S ** 2))
    best_a = a.copy()

    T = best_obj * 0.2 + 1e-6
    T_min = best_obj * 1e-6 + 1e-9
    decay = 0.995
    it = 0

    while time.time() - start < time_limit:
        it += 1
        conv = _conv_full(a)
        k_max = int(np.argmax(conv))

        # contributions to the maximal entry
        i_vals = np.arange(n)
        j_vals = k_max - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i_vals[mask]] * a[j_vals[mask]]
        total = contrib.sum()

        if total > 0.0:
            probs = contrib / total
            i = rng.choice(n, p=probs)
        else:
            i = rng.integers(n)

        # pick a destination index uniformly (different from i)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # move up to ~8 % of the mass at i
        delta = a[i] * rng.random() * 0.08
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.clip(a_new, 0.0, None)
        a_new = a_new / a_new.sum() * S

        conv_new = _conv_full(a_new)
        obj_new = float(2.0 * n * conv_new.max() / (S ** 2))

        if obj_new < best_obj - 1e-12:
            best_obj, best_a = obj_new, a_new.copy()
            a = a_new
        else:
            prob = np.exp(-(obj_new - best_obj) / T) if T > 0 else 0.0
            if rng.random() < prob:
                a = a_new

        if it % 2000 == 0:
            T = max(T * decay, T_min)

    return best_a


def _compress(seq: np.ndarray, min_val: float = 1e-12) -> np.ndarray:
    """
    Remove entries whose value is ≤ min_val and renormalise to √(2·len).
    """
    mask = seq > min_val
    seq = seq[mask] if mask.sum() < seq.size else seq
    S = np.sqrt(2.0 * seq.size)
    return seq / seq.sum() * S


# --------------------------------------------------------------------------- #
def propose_candidate(seed: int = 42, budget_s: int = 1000, **_ignore) -> list[float]:
    """
    Main entry point used by the evaluation harness.

    The routine runs a multi‑stage optimiser (Adam → LP refinement →
    simulated annealing) while obeying the overall time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # 1️⃣  Adam stage – a few short restarts on a large support.
    n_adam = 8000
    adam_budget = (deadline - time.time()) * 0.40
    n_restarts = 3
    best_adam = None
    best_score = float('inf')

    # optional warm‑start from a previous construction
    init_from_prev = None
    if 'height_sequence_1' in globals():
        seq0 = np.asarray(height_sequence_1, dtype=float)
        seq0 = np.clip(seq0, 0.0, None)
        if seq0.sum() >= 0.01:
            # interpolate to the required length
            x_old = np.linspace(0.0, 1.0, seq0.size)
            x_new = np.linspace(0.0, 1.0, n_adam)
            init_from_prev = np.interp(x_new, x_old, seq0)

    for r in range(n_restarts):
        remaining = deadline - time.time()
        if remaining < 5.0:
            break
        run_time = min(adam_budget / n_restarts, remaining * 0.85)

        init_vec = init_from_prev if (r == 0 and init_from_prev is not None) else None
        seq = _adam_optimize(
            n=n_adam,
            time_limit=run_time,
            rng=rng,
            init_a=init_vec,
            start_lr=0.25,
            end_lr=0.0008,
            start_beta=0.7,
            end_beta=25000.0,
        )
        score = evaluate_sequence(seq.tolist())
        print(f"[ADAM restart {r}] O={score:.6f} (n={n_adam})")
        if score < best_score:
            best_score, best_adam = score, seq.copy()

    # fallback if nothing was produced
    if best_adam is None:
        n_adam = 2000
        best_adam = rng.random(n_adam)
        best_adam = best_adam / best_adam.sum() * np.sqrt(2.0 * n_adam)

    # --------------------------------------------------------------
    # 2️⃣  Interpolate to a moderate size for the LP phase.
    n_lp = 2000
    x_old = np.linspace(0.0, 1.0, best_adam.size)
    x_new = np.linspace(0.0, 1.0, n_lp)
    a_lp = np.interp(x_new, x_old, best_adam)
    a_lp = a_lp / a_lp.sum() * np.sqrt(2.0 * n_lp)

    # --------------------------------------------------------------
    # 3️⃣  Linear‑programming refinement (if SciPy is available).
    remaining = max(0.0, deadline - time.time())
    if remaining > 10.0 and _SCIPY_AVAILABLE:
        lp_time = remaining * 0.65
        print(f"[LP] start refinement, budget={lp_time:.1f}s, n={n_lp}")
        a_lp = _lp_refine(
            a_lp,
            time_limit=lp_time,
            rng=rng,
            mix_factor=0.5,
            min_mix=0.05,
            decay=0.93,
        )

    # --------------------------------------------------------------
    # 4️⃣  Final simulated‑annealing polish.
    remaining = max(0.0, deadline - time.time())
    if remaining > 2.0:
        sa_time = remaining * 0.90
        print(f"[SA] polishing, time={sa_time:.1f}s")
        a_lp = _local_search(a_lp, time_limit=sa_time, rng=rng)

    # --------------------------------------------------------------
    # 5️⃣  Remove tiny entries and final sanity check.
    final_seq = _compress(a_lp)

    try:
        final_score = evaluate_sequence(final_seq.tolist())
        print(f"[final] evaluate_sequence → O={final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluation failed:", exc)

    return final_seq.tolist()
