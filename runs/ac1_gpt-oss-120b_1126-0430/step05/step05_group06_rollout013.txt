def _exponentiated_gradient(a, n, L, max_iters, deadline, alpha_start, alpha_end):
    # a is probability vector sum=1
    a = a.copy()
    best_a = a.copy()
    best_obj = _objective(a, n, L)
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        # schedule alpha
        frac = it / max_iters
        alpha = alpha_start * (alpha_end/alpha_start) ** frac
        conv = _conv_fft(a, L)
        # compute smooth gradient w.r.t a for surrogate log-sum-exp
        grad = _smooth_gradient(a, conv, alpha, L)   # returns gradient of surrogate
        # scaling maybe needed: we can use grad directly
        # Step size schedule: eta = eta0 / sqrt(it+1) maybe
        eta = 0.2 / np.sqrt(it+1 + 1)
        a = a * np.exp(-eta * grad)   # elementwise
        a = a / a.sum()
        # optional pruning:
        if it % 500 == 0 and it>0:
            # zero out values below percentile
            thresh = np.percentile(a, 0.5)  # 0.5 percentile
            a[a<thresh] = 0.0
            a = a / a.sum()
        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            print(f"[eg] iter {it} new best O={best_obj:.6f}")
    return a, best_obj, best_a

"""
Proposes a non‑negative step‑function sequence that minimises

    O(a) = 2·n·max(conv(a,a)) / (sum(a)²)

The search is split into four phases, each respecting the global time budget
(≈1000 s).  The main ideas are:

1️⃣  Random restarts – generate many Dirichlet draws and run a short
    exponentiated‑gradient (mirror‑descent) optimisation on a smooth
    log‑sum‑exp surrogate of the max‑operator; keep the best candidate.

2️⃣  Long exponentiated‑gradient descent – a multiplicative update on the
    simplex with an exponentially increasing “sharpness’’ α, which drives the
    surrogate ever closer to the true max.

3️⃣  Exact projected sub‑gradient descent – optimise the true nonsmooth
    objective using a back‑tracking line‑search to guarantee descent.

4️⃣  Targeted simulated annealing – repeatedly move a small fraction of mass
    from indices that contribute to the current maximal convolution entry to a
    random destination; moves are accepted either if they improve the exact
    objective or with a Metropolis probability that cools over time.  Periodic
    sub‑gradient steps keep the walk directed.

All phases constantly update the global best sequence and finally verify it
through the provided ``evaluate_sequence`` function.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, ∑x=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                      # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear self‑convolution via FFT
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return a*a (full convolution) using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Exact objective for a simplex‑normalised vector (∑a = 1)
def _objective(a: np.ndarray, n: int, L: int) -> float:
    """2·n·max(conv(a,a))  (a is assumed to sum to 1)."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Gradient of the log‑sum‑exp (smooth max) surrogate
def _smooth_gradient(a: np.ndarray, conv: np.ndarray,
                     alpha: float, L: int) -> np.ndarray:
    """
    Compute ∇_a ( (2·n/α)·log∑_k e^{α·conv_k} ).
    The formula reduces to 2·corr(w, a) where w_k ∝ e^{α·conv_k}.
    """
    # soft‑max weights (stable)
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)      # stabilise
    w = exp_scaled / exp_scaled.sum()             # ∑ w = 1

    # correlation w ⋆ a   (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)   # length ≥ 3·n‑2
    n_len = a.size
    grad = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]
    return grad


# ----------------------------------------------------------------------
# Sub‑gradient of the maximal convolution entry
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. a."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) phase on the smooth surrogate
def _exponentiated_gradient_phase(a: np.ndarray,
                                  n: int,
                                  L: int,
                                  max_iters: int,
                                  deadline: float,
                                  alpha_start: float = 5.0,
                                  alpha_end: float = 200_000.0,
                                  eta0: float = 0.2,
                                  prune_every: int = 2000,
                                  prune_percentile: float = 0.5) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Mirror‑descent on the log‑sum‑exp surrogate.
    Returns (final a, best objective value seen, best a).
    """
    a = a.copy()
    best_a = a.copy()
    best_obj = _objective(a, n, L)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # schedule for α (smoothness → max)
        frac = it / max_iters
        alpha = alpha_start * (alpha_end / alpha_start) ** frac

        # convolution and surrogate gradient
        conv = _conv_fft(a, L)
        grad = _smooth_gradient(a, conv, alpha, L)

        # step‑size (decreases like 1/√t)
        eta = eta0 / math.sqrt(it + 1)

        # multiplicative (exponentiated) update – stays on simplex after renorm
        a = a * np.exp(-eta * grad)
        a_sum = a.sum()
        if a_sum == 0.0:
            # fall back to uniform if everything vanished
            a = np.full_like(a, 1.0 / a.size)
        else:
            a /= a_sum

        # occasional pruning of tiny entries to keep the support sparse
        if prune_every and (it % prune_every) == 0 and it > 0:
            thresh = np.percentile(a, prune_percentile)
            a[a < thresh] = 0.0
            a_sum = a.sum()
            if a_sum == 0.0:
                a = np.full_like(a, 1.0 / a.size)
            else:
                a /= a_sum

        # evaluate exact objective (fast, uses same conv)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            print(f"[eg] iter {it:,} new best O={best_obj:.6f}")

    return a, best_obj, best_a


# ----------------------------------------------------------------------
# Exact projected sub‑gradient descent with simple back‑tracking line search
def _subgradient_phase(a: np.ndarray,
                       n: int,
                       L: int,
                       max_iters: int,
                       deadline: float,
                       eta0: float = 0.25,
                       line_search: bool = True,
                       verbose: bool = False) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Projected sub‑gradient descent on the true (non‑smooth) objective.
    Returns (final a, best objective, best a).
    """
    best_a = a.copy()
    best_obj = _objective(a, n, L)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            if verbose and (it % 5000) == 0:
                print(f"[sub‑grad] iter {it:,} new best O={best_obj:.6f}")

        g = _subgradient_max_conv(a, conv)

        # geometric decay of step size
        eta = eta0 * (0.9999 ** it)

        if line_search:
            step = eta
            for _ in range(5):
                cand = _project_to_simplex(a - step * g)
                cand_obj = _objective(cand, n, L)
                if cand_obj < cur_obj:
                    a = cand
                    break
                step *= 0.5
            else:
                a = _project_to_simplex(a - eta * g)
        else:
            a = _project_to_simplex(a - eta * g)

    return a, best_obj, best_a


# ----------------------------------------------------------------------
# Simulated annealing with mass‑transfer moves targeting the current peak
def _simulated_annealing_phase(a: np.ndarray,
                               n: int,
                               L: int,
                               max_iters: int,
                               deadline: float,
                               temperature0: float = 0.07,
                               no_improve_limit: int = 7000,
                               verbose: bool = True) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Local refinement: move a fraction of mass from an index that contributes
    to the current maximal convolution entry to a random destination; accept
    with Metropolis rule.  Periodic exact sub‑gradient steps keep the walk
    directed.
    """
    best_a = a.copy()
    best_obj = _objective(a, n, L)
    cur_obj = best_obj
    conv = _conv_fft(a, L)

    T = temperature0
    no_improve = 0

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # index of the current peak of the convolution
        kmax = int(np.argmax(conv))

        # indices that actually contribute to that peak
        i_candidates = np.arange(max(0, kmax - (n - 1)), min(n - 1, kmax) + 1)
        i_candidates = i_candidates[a[i_candidates] > 1e-14]
        if i_candidates.size == 0:
            i = np.random.randint(n)
        else:
            i = np.random.choice(i_candidates)

        # random destination different from i
        j = np.random.randint(n)
        while j == i:
            j = np.random.randint(n)

        # transfer a modest fraction of the source mass (≤ 10 %)
        delta = a[i] * np.random.rand() * 0.09
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = _project_to_simplex(a_new)

        conv_new = _conv_fft(a_new, L)
        new_obj = 2.0 * n * conv_new.max()

        # Metropolis acceptance
        if new_obj < cur_obj or np.random.rand() < math.exp((cur_obj - new_obj) / max(T, 1e-12)):
            a = a_new
            conv = conv_new
            cur_obj = new_obj
            no_improve = 0
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                if verbose:
                    print(f"[SA] iter {it:,} new global best O={best_obj:.6f}")
        else:
            no_improve += 1

        # occasional exact sub‑gradient step (keeps direction)
        if it % 9 == 0:
            g = _subgradient_max_conv(a, conv)
            eta_loc = 0.07 * (1.0 - it / max_iters)
            a = _project_to_simplex(a - eta_loc * g)
            conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                if verbose:
                    print(f"[SA] sub‑grad improve O={best_obj:.6f}")

        # temperature decay
        T *= 0.9997

        # restart if stagnating
        if no_improve >= no_improve_limit:
            a = best_a.copy()
            a += np.random.randn(n) * 0.0015
            a = np.clip(a, 0, None)
            a = _project_to_simplex(a)
            conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max()
            no_improve = 0

    return a, best_obj, best_a

# Main entry point expected by the evaluation harness
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence that minimises the evaluation function.
    Returns the sequence as a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # stop a little before the deadline to be safe
    deadline = start_time + budget_s - 0.2

    # ------------------------------------------------------------------
    # Initialise problem size (use supplied sequence if present)
    if 'height_sequence_1' in globals():
        a0 = np.array(height_sequence_1, dtype=float)
        a0 = np.maximum(a0, 0.0)
        if a0.sum() == 0.0:
            a0 = np.full_like(a0, 1.0)
        a0 = a0 / a0.sum()               # simplex normalisation
        n = a0.size
        a = a0.copy()
        print(f"[init] using supplied sequence, n={n}")
    else:
        n = 4600
        a = np.full(n, 1.0 / n)           # uniform start on the simplex
        print(f"[init] using uniform start, n={n}")

    # FFT length – next power of two ≥ 3·n‑2 (sufficient for all convolutions)
    L = 1 << ((3 * n - 2).bit_length())

    # ------------------------------------------------------------------
    # Phase 0 – cheap random restarts with a short EG optimisation
    phase0_deadline = start_time + budget_s * 0.20
    best_seq = a.copy()
    best_score = _objective(best_seq, n, L)

    while time.time() < phase0_deadline:
        conc = np.random.uniform(0.2, 0.7)      # Dirichlet concentration
        a_rand = np.random.gamma(conc, 1.0, size=n)
        a_rand = a_rand / a_rand.sum()
        # short EG run (a few thousand updates)
        sub_dead = time.time() + min(4.0, phase0_deadline - time.time())
        _, _, best_rand = _exponentiated_gradient_phase(
            a_rand, n, L,
            max_iters=2000,
            deadline=sub_dead,
            alpha_start=5.0,
            alpha_end=5000.0,
            eta0=0.2,
            prune_every=1000,
            prune_percentile=0.5)
        cur_obj = _objective(best_rand, n, L)
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = best_rand.copy()
            print(f"[phase0] new best O={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – long exponentiated‑gradient descent on the smooth surrogate
    phase1_deadline = start_time + budget_s * 0.55
    a, obj_after_eg, best_eg = _exponentiated_gradient_phase(
        best_seq, n, L,
        max_iters=500_000,
        deadline=phase1_deadline,
        alpha_start=5.0,
        alpha_end=200_000.0,
        eta0=0.2,
        prune_every=2000,
        prune_percentile=0.5)

    # Keep the best point seen in phase 1
    if obj_after_eg < best_score:
        best_score = obj_after_eg
        best_seq = best_eg.copy()
        print(f"[phase‑1] improved best O={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – exact projected sub‑gradient descent (true objective)
    phase2_deadline = start_time + budget_s * 0.80
    a, obj_sub, best_sub = _subgradient_phase(
        a, n, L,
        max_iters=300_000,
        deadline=phase2_deadline,
        eta0=0.25,
        line_search=True,
        verbose=False)
    if obj_sub < best_score:
        best_score = obj_sub
        best_seq = best_sub.copy()
        print(f"[phase‑2] improved best O={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – stochastic annealing refinement
    phase3_deadline = deadline
    a, obj_sa, best_sa = _simulated_annealing_phase(
        best_seq, n, L,
        max_iters=300_000,
        deadline=phase3_deadline,
        temperature0=0.07,
        no_improve_limit=7000,
        verbose=True)
    if obj_sa < best_score:
        best_score = obj_sa
        best_seq = best_sa.copy()
        print(f"[phase‑3] final best O={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final verification through the provided evaluator
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence failed:", exc)

    return best_seq.tolist()
