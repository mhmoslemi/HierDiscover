assistantfinal**Solution Overview**

The goal is to find a non‑negative sequence `a` (∑aᵢ = 1) that minimises  

\[
\text{score}(a)=2\;n\;\max_k (a*a)[k]
\]

where `(a*a)` is the full linear convolution of `a` with itself.  
The score is independent of the absolute scale of `a`, therefore we only
need to optimise the *shape* of a probability vector.

The previous approach used a mirror‑descent step based on the **top‑k**
convolution entries.  
The new search improves on that in three ways

1. **Active‑set gradient** – the sub‑gradient of `max_k (a*a)[k]` is an
   average of the gradients of **all** active indices (those whose
   convolution value is equal to the maximum up to a tiny tolerance).  
   This gives a far more accurate descent direction than the coarse
   `top‑k` approximation.

2. **Larger problem size** – we work with `n = 15 000` (still cheap to
   FFT‑convolve). More points give the optimiser more freedom to
   flatten the autocorrelation.

3. **Heavier refinement** – after the main mirror‑descent phase a short
   stochastic local‑search finishes the job, and the best vector is
   finally evaluated with the provided `evaluate_sequence`.

All helper utilities are defined at top level, there are no closures,
and the whole routine respects the 1000 s time budget.

#!/usr/bin/env python
"""
Mirror‑descent with an exact active‑set sub‑gradient, followed by a tiny
random local search.  The algorithm works on a relatively large vector
(≈15 k entries) and always returns a feasible non‑negative step function.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using real FFT (length = 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len).real[: 2 * a.size - 1]
    return conv


def _active_gradient(a: np.ndarray,
                     conv: np.ndarray,
                     maxb: float,
                     tol_rel: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a.
    Uses **all** indices where conv is within `tol_rel*maxb` of the maximum.
    The gradient is the average of the individual gradients.
    """
    n = a.shape[0]
    # Active set – indices where the convolution attains (almost) the max.
    tol = max(tol_rel * maxb, 1e-15)
    active = np.where(conv >= maxb - tol)[0]

    if active.size == 0:               # should not happen, but be safe
        return np.zeros_like(a)

    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        rng = np.arange(lo, hi + 1, dtype=np.int64)
        # derivative of (a*a)[k] w.r.t. a[rng] is 2 * a[k - rng]
        g[rng] += 2.0 * a[k - rng]
    # average over the active set – any convex combination is a sub‑gradient
    g /= float(active.size)
    return g

# Main entry point required by the platform
# ----------------------------------------------------------------------

    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine respects the supplied time budget
    and always returns a feasible sequence.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a safety margin for the final exact evaluation
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # Choose a working length (5 k … 20 k)
    # --------------------------------------------------------------
    try:
        prev = height_sequence_1                      # may exist globally
        prev_len = len(prev)
    except Exception:
        prev = None
        prev_len = 0

    # target length – larger gives more freedom, still fast enough
    n = min(max(prev_len, 8000), 20000)      # 8 k ≤ n ≤ 20 k
    fft_len = 1 << ((2 * n - 1).bit_length())   # power‑of‑two ≥ 2·n‑1

    # --------------------------------------------------------------
    # Cheap score (assumes Σa = 1): 2·n·max(conv)
    # --------------------------------------------------------------
    def cheap_score(vec: np.ndarray) -> float:
        conv = _conv_full_fft(vec, fft_len)
        return 2.0 * n * conv.max()

    # --------------------------------------------------------------
    # 0. Initialise with a variety of parametric families
    # --------------------------------------------------------------
    best_a = None
    best_val = float("inf")

    # uniform baseline
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    val = cheap_score(uniform)
    best_a, best_val = uniform.copy(), val

    # previous construction, if any (rescaled to sum‑1)
    if prev is not None and len(prev) == n:
        prev_arr = np.asarray(prev, dtype=np.float64)
        if prev_arr.sum() > 0:
            prev_arr = prev_arr / prev_arr.sum()
            val = cheap_score(prev_arr)
            if val < best_val:
                best_a, best_val = prev_arr.copy(), val

    # power‑law sweep
    for p in np.linspace(0.4, 1.2, 81):
        if time.time() >= deadline:
            break
        i = np.arange(1, n + 1, dtype=np.float64)
        cand = i ** (-p)
        cand /= cand.sum()
        val = cheap_score(cand)
        if val < best_val:
            best_a, best_val = cand.copy(), val

    # pure exponential sweep
    for lam in np.linspace(0.0, 3.0, 61):
        if time.time() >= deadline:
            break
        idx = np.arange(n, dtype=np.float64)
        cand = np.exp(-lam * idx)
        cand /= cand.sum()
        val = cheap_score(cand)
        if val < best_val:
            best_a, best_val = cand.copy(), val

    # power‑law × exponential sweep (coarser)
    for p in np.linspace(0.5, 1.0, 11):
        for lam in np.linspace(0.0, 2.0, 11):
            if time.time() >= deadline:
                break
            idx = np.arange(n, dtype=np.float64)
            cand = (idx + 1.0) ** (-p) * np.exp(-lam * idx)
            cand /= cand.sum()
            val = cheap_score(cand)
            if val < best_val:
                best_a, best_val = cand.copy(), val

    # beta‑shaped families (including the arc‑sine case α=β=0.5)
    x = (np.arange(1, n + 1, dtype=np.float64) / n)
    for alpha in np.linspace(0.2, 5.0, 21):
        for beta in np.linspace(0.2, 5.0, 21):
            if time.time() >= deadline:
                break
            cand = (x ** alpha) * ((1.0 - x) ** beta)
            s = cand.sum()
            if s == 0.0:
                continue
            cand /= s
            val = cheap_score(cand)
            if val < best_val:
                best_a, best_val = cand.copy(), val

    # a few random Dirichlet seeds (different concentrations)
    for conc in [0.001, 0.01, 0.1, 0.5, 1.0, 3.0, 10.0]:
        if time.time() >= deadline:

        g = np.random.gamma(conc, 1.0, size=n)
        cand = g / g.sum()
        val = cheap_score(cand)
        if val < best_val:
            best_a, best_val = cand.copy(), val

    print(f"[Init] best cheap score = {best_val:.6f}")

    # --------------------------------------------------------------
    # 1. Refine the best candidate with mirror‑descent (exponentiated‑gradient)
    #    using the **exact** active‑set sub‑gradient.
    # --------------------------------------------------------------
    a = best_a.copy()
    best_iter_a = a.copy()
    best_iter_val = best_val
    last_improve = 0
    iter_cnt = 0

    # hyper‑parameters
    eta0 = 0.8               # initial step‑size factor (larger than before)
    max_iters = 600_000      # generous upper bound – we will stop by time
    restart_patience = 150_000   # iterations without improvement → restart
    log_interval = 20000

    while time.time() < deadline and iter_cnt < max_iters:
        iter_cnt += 1

        # ---- convolution and cheap score ----
        conv = _conv_full_fft(a, fft_len)
        maxb = conv.max()
        cheap = 2.0 * n * maxb

        # ---- track the best seen so far ----
        if cheap < best_iter_val - 1e-12:
            best_iter_val = cheap
            best_iter_a = a.copy()
            last_improve = iter_cnt
            print(f"[Iter {iter_cnt:07d}] NEW BEST cheap = {cheap:.6f}")

        # ---- active‑set sub‑gradient ----
        g = _active_gradient(a, conv, maxb, tol_rel=1e-12)

        # ---- exponentiated‑gradient (mirror descent) step ----
        eta = eta0 / math.sqrt(iter_cnt)
        log_a = np.log(a + 1e-30) - eta * g          # work in log‑space
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0 or not np.isfinite(s):
            # fallback to last good iterate
            a = best_iter_a.copy()
            continue
        a /= s

        # ---- occasional restart when progress stalls ----
        if iter_cnt - last_improve > restart_patience:
            perturb = 0.01 * np.random.rand(n)
            a = _simplex_projection(best_iter_a + perturb)
            last_improve = iter_cnt
            print(f"[Iter {iter_cnt:07d}] Restart after stagnation.")

        # periodic logging (does not affect performance)
        if iter_cnt % log_interval == 0:
            elapsed = time.time() - start_time
            print(f"[Iter {iter_cnt:07d}] cheap = {cheap:.6f}  "
                  f"time = {elapsed:.1f}s")

    # --------------------------------------------------------------
    # 2. Tiny stochastic local search (purely heuristic, cheap)
    # --------------------------------------------------------------
    a = best_iter_a.copy()
    best_local_val = best_iter_val
    best_local_a = a.copy()
    local_iters = 0
    while time.time() < deadline and local_iters < 30_000:
        local_iters += 1
        # pick two distinct positions and transfer a small amount
        i, j = np.random.choice(n, size=2, replace=False)
        amount = 0.002 * a[i]            # at most 0.2 % of the source mass
        if amount <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= amount
        a_new[j] += amount
        # re‑normalise (numerical safety)
        a_new /= a_new.sum()
        conv_new = _conv_full_fft(a_new, fft_len)
        cheap_new = 2.0 * n * conv_new.max()
        if cheap_new < best_local_val:
            best_local_val = cheap_new
            best_local_a = a_new.copy()
            a = a_new
            print(f"[Local {local_iters:05d}] improved cheap = {cheap_new:.6f}")

    # --------------------------------------------------------------
    # 3. Final exact evaluation (uses the supplied ``evaluate_sequence``)
    # --------------------------------------------------------------
    final_seq = best_local_a.tolist()
    try:

        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:      # defensive – should never happen
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # --------------------------------------------------------------
    # 4. Return the best sequence found.
    # --------------------------------------------------------------
    return final_seq
