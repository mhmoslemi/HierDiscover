def propose_candidate(...):
    start = time.time()
    deadline = start + budget_s - 5.0

    n = try existing variable length, else default 50000.

    # Build candidate vectors:
    initial_vectors = []
    # arcsine:
    idx = np.arange(n)
    x = (idx + 0.5) / n
    raw_arcs = 1.0 / np.sqrt(x * (1.0 - x))
    a_arcs = raw_arcs / raw_arcs.sum()
    initial_vectors.append(a_arcs)

    # uniform:
    a_uni = np.full(n, 1.0/n)
    initial_vectors.append(a_uni)

    # Beta(0.8,0.8) (exponent <1 but >0) maybe produce near arcsine:
    for alpha in [0.8, 1.2, 2.0]:
        raw = (x ** (alpha - 1)) * ((1.0 - x) ** (alpha - 1))
        a = raw / raw.sum()
        initial_vectors.append(a)

    best_a = None
    best_score = np.inf

    for init_a in initial_vectors:
        # run Adam for given iterations or until time
        # initialize logits from init_a
        z = _logits_from_distribution(init_a)
        # Adam hyperparameters
        m = np.zeros_like(z)
        v = np.zeros_like(z)
        beta1, beta2 = 0.9, 0.999
        eps_adam = 1e-8
        base_alpha = 0.6 # maybe
        eps_factor = 0.1
        eps_factor_min = 1e-4
        eps_factor_decay = 0.9998

        it = 0
        while time.time() < deadline and it < max_iters_per_start:
            it += 1
            a_cur = _softmax(z)
            max_conv, grad_a = _conv_max_and_grad(a_cur, eps_factor)
            cheap_score = 2.0 * n * max_conv
            if cheap_score < best_score:
                best_score = cheap_score
                best_a = a_cur.copy()
                print(f"New best {best_score:.6f} at iter {it} from start {len(initial_vectors)}")

            # compute grad_z
            mu = np.dot(grad_a, a_cur)
            grad_z = a_cur * (grad_a - mu)

            # Adam update
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * (grad_z ** 2)
            m_hat = m / (1 - beta1 ** it)
            v_hat = v / (1 - beta2 ** it)

            alpha_t = base_alpha / (np.sqrt(it) + 1.0)
            step = - alpha_t * m_hat / (np.sqrt(v_hat) + eps_adam)
            z = z + step
            # clip to avoid overflow of exp
            np.clip(z, -50.0, 50.0, out=z)

            eps_factor = max(eps_factor_min, eps_factor * eps_factor_decay)

#!/usr/bin/env python
"""
Adaptive Adam refinement of discrete arcsine (and related) distributions.

* Choose a support size `n` (default 50 000, larger than the previous 20 000).
* Build several candidate probability vectors (arcsine, uniform, beta‑α).
* Re‑parameterise each vector with logits `z` (softmax → non‑negative, sum = 1).
* Run Adam on the logits while gradually tightening a smooth‑max
  approximation of the self‑convolution peak.
* Keep the best sequence (lowest 2·n·max(a*a)/(sum a)²) found within the time
  budget and return it as a plain list.
"""

import time
import math
import numpy as np


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it representation of a probability vector (sum ≈ 1)."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """
    Smallest power‑of‑two ≥ 3·n‑2.
    This length is sufficient for linear convolution of length (2·n‑1)
    and correlation of length (3·n‑2) using the FFT.
    """
    return 1 << ((3 * n - 2).bit_length())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Compute the (smoothed) maximum of the self‑convolution `a*a`
    and a gradient of the smooth‑max w.r.t. `a`.

    Parameters
    ----------
    a : np.ndarray
        Probability vector (non‑negative, sum ≈ 1).
    eps_factor : float
        Multiplicative factor for the smoothing constant
        ``eps = max(max_conv * eps_factor, 1e-12)``.

    Returns
    -------
    max_conv : float
        Exact maximum entry of the linear convolution a*a.
    grad_a : np.ndarray
        Gradient of ``2·n·smooth_max`` w.r.t. ``a``
        (already multiplied by ``4·n``).
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    # FFT of the current vector
    a_fft = np.fft.rfft(a, L)

    # linear self‑convolution (length 2·n‑1)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # smoothed soft‑max weights over the convolution entries
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv            # for numerical stability
    w_raw = np.exp(shifted / eps)        # unnormalised weights
    sum_w = w_raw.sum()
    if sum_w == 0.0 or np.isnan(sum_w):
        w = np.full_like(w_raw, 1.0 / w_raw.size)
    else:
        w = w_raw / sum_w

    # correlation a ⋆ w (linear, length ≤ n)
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # gradient of 2·n·smooth_max (the factor 4·n comes from the chain rule)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a

    start_time = time.time()
    # leave a small safety margin for final evaluation and printing

    # ----------------------------------------------------------------------

    # ----------------------------------------------------------------------
    try:

    except Exception:

        print(f"No external sequence found – using default length n = {n}")

    # ----------------------------------------------------------------------

    # ----------------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)
    x = (idx + 0.5) / n                     # mid‑point rule on (0,1)

    init_vectors = []

    # a) Continuous arcsine distribution (Beta(0.5, 0.5))
