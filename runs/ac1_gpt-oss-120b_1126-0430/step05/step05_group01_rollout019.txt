def propose_candidate(seed=123, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0

    # parameters
    n = 20000

    best_seq = None
    best_score = np.inf

    # 1. scan simple families
    def eval_and_update(a):
        nonlocal best_seq,best_score
        score = 2.0 * n * _conv_max(a)  # cheap score (sum=1)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
    
    # uniform baseline just for reference
    a_uniform = np.full(n, 1.0/n)
    eval_and_update(a_uniform)

    # exponential one-sided
    for lam in np.logspace(-7, -3, 100):   # lam from 1e-7 to 1e-3
        if time.time() >= deadline: break
        a = np.exp(-lam * np.arange(n))
        a /= a.sum()
        eval_and_update(a)

    # power-law
    for p in np.linspace(0.45, 0.9, 40):
        if time.time() >= deadline: break
        idx = np.arange(1, n+1)
        a = idx ** (-p)
        a /= a.sum()
        eval_and_update(a)

    # logistic
    for alpha in np.linspace(0.1, 10.0, 50):
        if time.time() >= deadline: break
        i = np.arange(n)
        mid = (n-1) / 2.0
        a = 1.0 / (1.0 + np.exp(alpha * (i - mid)))
        a /= a.sum()
        eval_and_update(a)

    # mixture exponentials random samples
    for _ in range(30):
        if time.time() >= deadline: break
        lam1 = 10**np.random.uniform(-7, -3)
        lam2 = 10**np.random.uniform(-7, -3)
        w = np.random.rand()
        a = w * np.exp(-lam1 * np.arange(n)) + (1-w) * np.exp(-lam2 * np.arange(n))
        a /= a.sum()
        eval_and_update(a)

    # At this point we have a best_seq found.

    # 2. refine with Adam mirror descent
    if best_seq is None:
        best_seq = np.full(n, 1.0/n)
        best_score = 2.0 * n * _conv_max(best_seq)

    # turn to logits
    z = _logits_from_distribution(best_seq) + 0.005 * np.random.randn(n)

    # Adam parameters
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_alpha = 0.2
    eps_factor = 0.05   # initial smoothing factor
    eps_factor_min = 1e-5
    eps_decay = 0.9995

    it = 0
    last_improve = 0
    while time.time() < deadline and it < 800_000:
        it += 1
        a = _softmax(z)
        max_conv, grad_a = _conv_max_and_grad(a, eps_factor)
        cheap_score = 2.0 * n * max_conv
        if cheap_score < best_score - 1e-12:
            best_score = cheap_score
            best_seq = a.copy()
            last_improve = it
        # mirror gradient
        mu = np.dot(grad_a, a)
        grad_z = a * (grad_a - mu)

        # Adam
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * (grad_z **2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step = - m_hat / (np.sqrt(v_hat) + eps_adam)

        # backtracking line search
        accepted = False
        trial_alpha = alpha_t
        for _ in range(5):
            z_new = z + trial_alpha * step
            z_new = np.clip(z_new, -50, 50)
            a_new = _softmax(z_new)
            new_score = 2.0 * n * _conv_max(a_new)
            if new_score <= cheap_score:
                z = z_new
                accepted = True
                break
            trial_alpha *= 0.5
        if not accepted:
            z = z + alpha_t * step
            z = np.clip(z, -50, 50)

        eps_factor = max(eps_factor_min, eps_factor * eps_decay)

        # occasional random jitter to escape
        if it % 5000 == 0 and np.random.rand() < 0.1:
            z += 0.01 * np.random.randn(n)
            m *= 0.5
            v *= 0.5

    # final evaluation
    final_seq = best_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:
        final_score = float('inf')
        print(f"Eval error: {exc}")

    return final_seq
```

#!/usr/bin/env python
"""
Non‑negative step‑function optimiser – second generation.

The routine first builds a diverse set of seed distributions
(power‑law, exponential, logistic and random mixtures of two exponentials),
picks the best according to the cheap bound 2·n·max(a*a) and then refines it
with a mirror‑descent optimiser on the probability simplex (Adam on logits).
A small smoothed max is used for reliable gradients and a back‑tracking line
search guarantees that the cheap bound never increases.  The whole search
runs for the supplied time budget (≈1000 s) and returns the best sequence.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max; returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    s = e.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / z.size)
    return e / s


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it that reproduces `a` after a soft‑max."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """Smallest power‑of‑two ≥ 3·n‑2 (enough for all needed convolutions)."""
    return 1 << ((3 * n - 2).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Returns (max_conv, grad_a) where

        grad_a = ∇_a [ 2·n·smooth_max(a*a) ]

    `eps_factor` multiplies the smoothing constant ε = eps_factor·max_conv.
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    # FFT of a (zero‑padded)
    a_fft = np.fft.rfft(a, L)

    # Linear self‑convolution a*a
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # Smoothed soft‑max weights
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv                # ≤ 0 for numerical stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # Correlation a ⋆ w
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # Gradient (already multiplied by 4·n)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative height sequence that minimises
    `evaluate_sequence`.  The routine respects the supplied time budget
    (≈1000 s).  It returns the best sequence found (list of floats).
    """
    np.random.seed(seed)
    start = time.time()
    # leave a small safety margin for the final evaluation
    deadline = start + budget_s - 2.0

    # --------------------------------------------------------------
    # 0. Problem size
    # --------------------------------------------------------------
    n = 25000                     # length of the final step‑function

    # --------------------------------------------------------------
    # 1. Generate a handful of seed distributions
    # --------------------------------------------------------------
    candidates = []

    # uniform (baseline)
    candidates.append(np.full(n, 1.0 / n))

    # one‑sided exponential scans (lam = λ/(n‑1) with λ in [0.1, 4])
    for lam in np.logspace(-7, -3, 30):
        a = np.exp(-lam * np.arange(n))
        a /= a.sum()
        candidates.append(a)

    # power‑law family
    for p in np.linspace(0.45, 0.90, 30):
        idx = np.arange(1, n + 1, dtype=np.float64)
        a = idx ** (-p)
        a /= a.sum()
        candidates.append(a)

    # logistic (left‑heavy) family
    for alpha in np.linspace(0.1, 10.0, 30):
        i = np.arange(n, dtype=np.float64)
        mid = (n - 1) / 2.0
        a = 1.0 / (1.0 + np.exp(alpha * (i - mid)))
        a /= a.sum()
        candidates.append(a)

    # random mixtures of two exponentials
    for _ in range(20):
        lam1 = 10 ** np.random.uniform(-7, -3)
        lam2 = 10 ** np.random.uniform(-7, -3)
        w = np.random.rand()
        a = w * np.exp(-lam1 * np.arange(n)) + (1.0 - w) * np.exp(-lam2 * np.arange(n))
        a = np.maximum(a, 0.0)
        a /= a.sum()
        candidates.append(a)

    # --------------------------------------------------------------
    # 2. Pick the best seed according to the cheap bound
    # --------------------------------------------------------------
    best_a = None
    best_score = float("inf")
    for a in candidates:
        if time.time() >= deadline:
            break
        score = 2.0 * n * _conv_max(a)          # cheap upper bound (sum == 1)
        if score < best_score:
            best_score = score
            best_a = a.copy()

    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_score = 2.0 * n * _conv_max(best_a)

    # --------------------------------------------------------------
    # 3. Mirror‑descent (Adam on logits) refinement
    # --------------------------------------------------------------
    # initialise logits with a tiny random jitter
    z = _logits_from_distribution(best_a) + 0.01 * np.random.randn(n)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    base_alpha = 0.2          # base learning‑rate for logits
    eps_factor = 0.06         # ε = eps_factor·max_conv (smoothing)
    eps_factor_min = 5e-5
    eps_decay = 0.9992

    best_seq = _softmax(z)    # best sequence seen so far
    best_score = 2.0 * n * _conv_max(best_seq)
    last_improve = 0
    it = 0

    while time.time() < deadline:
        it += 1

        # ----- Current point on the simplex -----
        a = _softmax(z)

        # ----- cheap evaluation + smoothed gradient -----
        max_conv, grad_a = _conv_max_and_grad(a, eps_factor)
        cheap_score = 2.0 * n * max_conv

        # ----- Keep best ever seen -----
        if cheap_score < best_score - 1e-12:
            best_score = cheap_score
            best_seq = a.copy()
            last_improve = it

        # ----- Gradient w.r.t. logits (mirror descent) -----
        mu = np.dot(grad_a, a)          # ⟨∇f(a), a⟩
        grad_z = a * (grad_a - mu)      # ∇_z f = Jᵀ∇_a f (soft‑max Jacobian)

        # ----- Adam moment updates (bias‑corrected) -----
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # ----- Step‑size schedule (inverse‑sqrt decay) -----
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step_dir = - m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- Back‑tracking line search (up to 4 halvings) -----
        accepted = False
        trial_alpha = alpha_t
        for _ in range(4):
            z_new = z + trial_alpha * step_dir
            # keep the exponentials stable
            z_new = np.clip(z_new, -50.0, 50.0)
            a_new = _softmax(z_new)
            new_score = 2.0 * n * _conv_max(a_new)
            if new_score <= cheap_score:
                z = z_new
                accepted = True
                break
            trial_alpha *= 0.5

        if not accepted:
            # take a modest step anyway
            z = z + alpha_t * step_dir
            z = np.clip(z, -50.0, 50.0)

        # ----- Gradually reduce the smoothing factor -----
        eps_factor = max(eps_factor_min, eps_factor * eps_decay)

        # ----- Occasional random perturbations to escape plateaux -----
        if it % 4000 == 0 and np.random.rand() < 0.1:
            z += 0.02 * np.random.randn(n)
            m *= 0.5
            v *= 0.5

        # ----- Restart if stagnation is observed -----
        if it - last_improve > 200_000:
            print(f"[Iter {it}] Stagnation – re‑initialising from current best.")
            z = _logits_from_distribution(best_seq) + 0.02 * np.random.randn(n)
            m.fill(0.0)
            v.fill(0.0)
            eps_factor = 0.06
            last_improve = it

    # --------------------------------------------------------------
    # 4. Final exact evaluation with the provided evaluate_sequence()
    # --------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # --------------------------------------------------------------
    # 5. Return the best sequence discovered.
    # --------------------------------------------------------------
    return final_seq
