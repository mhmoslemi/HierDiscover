def propose_candidate(seed=42, budget_s=1000, **_ignore):
    import time, numpy as np

    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5

    best_seq = None
    best_obj = float('inf')
    # initial from previous solved sequence
    if 'height_sequence_1' in globals():
        seq0 = np.asarray(height_sequence_1, dtype=float)
        seq0 = np.clip(seq0, 0, None)
        if seq0.sum() >= 0.01:
            best_seq = seq0 / seq0.sum()
            best_obj = _objective(best_seq)
            print(f"[init] using prior sequence length {best_seq.size}, O={best_obj:.6f}")

    # Choose lengths
    lengths = [4096, 6144, 8192]  # choose three lengths
    # If there is a previous sequence with different length, include it at front
    if best_seq is not None and best_seq.size not in lengths:
        lengths.insert(0, best_seq.size)

    # Exponential r values
    r_vals = [0.99, 0.995, 0.998, 0.999, 0.9995]

    # Run Adam optimisations
    for idx, n in enumerate(lengths):
        now = time.time()
        remaining = deadline - now
        if remaining < 30:
            break
        # allocate time for this length
        # allocate proportionally: for earlier lengths maybe less.
        # Let's give 0.5 of remaining for first length, then split rest.
        if idx == 0:
            run_time = remaining * 0.5
        else:
            run_time = (remaining * 0.4) / (len(lengths) - idx)
        # build init candidates
        init_candidates = [None]  # random init
        for r in r_vals:
            a = np.array([r**i for i in range(n)], dtype=float)
            a = a / a.sum()
            init_candidates.append(a)
        # also reverse exponential
        for r in r_vals:
            a = np.array([r**i for i in range(n-1, -1, -1)], dtype=float)
            a = a / a.sum()
            init_candidates.append(a)
        # run Adam for each init, respecting per init time budget
        per_init_time = max(5.0, run_time / len(init_candidates))
        for init_idx, init_a in enumerate(init_candidates):
            # stop if out of time
            now2 = time.time()
            if deadline - now2 < 5.0:
                break
            seq, val = _adam_optimize(
                n=n,
                time_limit=min(per_init_time, deadline - now2 - 0.5),
                init_a=init_a,
                seed=seed + idx*123 + init_idx*17,
                beta_end=50000.0,
                lr_start=0.2,
                lr_end=5e-5,
                jitter_interval=4000,
                restart_interval=50000,
            )
            if val < best_obj:
                best_obj, best_seq = val, np.asarray(seq, dtype=float)
                print(f"[global] new best O={best_obj:.6f} (n={n})")
        if time.time() >= deadline - 0.5:
            break

    # Run Simulated Annealing on best sequence with remaining time
    remaining = deadline - time.time()
    if best_seq is not None and remaining > 10.0:
        seq_ref, val_ref = _simulated_annealing(
            best_seq.tolist(),
            time_limit=remaining,
            seed=seed + 9999,
        )
        if val_ref < best_obj:
            best_obj, best_seq = val_ref, np.asarray(seq_ref, dtype=float)
            print(f"[final] after SA new best O={best_obj:.6f}")

    # Fallback if nothing found
    if best_seq is None or best_seq.size == 0:

"""Step‑function optimiser.

The algorithm combines three ideas:

1. **Deterministic warm‑starts** – a handful of simple shapes (exponential
   decay, its reverse and a random start) are prepared for each support size.

2. **Adam optimisation on the simplex** – the sequence is represented by
   logits `x` (soft‑max → probabilities).  A smooth approximation of the
   ``max`` of the self‑convolution (soft‑max with temperature β) provides a
   differentiable objective.  Adam with an exponential schedule for β and the
   learning‑rate rapidly reduces the objective.

3. **Targeted simulated‑annealing refinement** – after the Adam stage the best
   sequence is refined by moving a small amount of mass from indices that
   contribute most to the current maximal convolution entry to random other
   indices.  The temperature is exponentially cooled; occasional global jitter
   and a final compression of negligible entries improve the result.

All work respects the supplied time budget (`budget_s`) and returns a plain
list of non‑negative floats suitable for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return conv_full[: 2 * n - 1]


def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """Correlation g_i = Σ_t w[i+t]·a[t] for i=0,…,n‑1 via FFT."""
    n = a.shape[0]
    m = w.shape[0]                     # = 2n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


# ----------------------------------------------------------------------
# Basic maths
# ----------------------------------------------------------------------


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) assuming Σa = 1."""
    n = a.size
    conv = _conv_self_fft(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Adam optimiser on the simplex
# ----------------------------------------------------------------------


def _adam_optimize(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
    beta_start: float = 0.5,
    beta_end: float = 50000.0,
    lr_start: float = 0.2,
    lr_end: float = 5e-5,
    jitter_interval: int = 4000,
    restart_interval: int = 50000,
) -> tuple[list[float], float]:
    """
    Adam optimiser on the log‑it simplex.

    Returns the best sequence (as a list) and its exact objective value.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # initialise a
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
        a = a / a.sum()
    else:
        a = rng.random(n)
        a = a / a.sum()

    # logits
    x = np.log(a + 1e-16)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    best_a = a.copy()
    best_obj = _objective(best_a)

    it = 0
    while time.time() - start < time_limit:
        it += 1
        # stay on the simplex
        a = _softmax(x)

        conv = _conv_self_fft(a)
        M = conv.max()
        obj = float(2.0 * n * M)

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[Adam n={n:5d} it={it:06d}] new best O={best_obj:.6f}")

        # schedules (time‑based)
        t = min((time.time() - start) / time_limit, 1.0)
        beta = beta_start * np.exp(np.log(beta_end / beta_start) * t)
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t)

        # smooth max weights
        if beta > 20000.0:
            # essentially a hard max
            w = np.zeros_like(conv)
            w[int(np.argmax(conv))] = 1.0
        else:
            shifted = conv - M                    # ≤ 0
            w = np.exp(beta * shifted)
            s = w.sum()
            if s == 0.0:
                w = np.ones_like(conv) / conv.size
            else:
                w /= s

        # gradient of the (smooth) max w.r.t. a
        g_a = 2.0 * _correlation_fft(w, a)

        # natural gradient in log‑it space (mirrored gradient)
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape plateaus
        if it % jitter_interval == 0:
            jitter = rng.normal(scale=0.01, size=n)
            x += jitter

        # restart from the current best every so often
        if it % restart_interval == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.005, size=n)
            m.fill(0.0)
            v.fill(0.0)

    # ------------------------------------------------------------------
    # prune negligible entries (helps the final O)
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = _objective(best_a)
        print(f"[Adam compress] reduced length → {best_a.size}, O={best_obj:.6f}")

    return best_a.tolist(), best_obj

# Simulated‑annealing refinement
# ----------------------------------------------------------------------


def _simulated_annealing(
    init_seq: list[float],
    time_limit: float,
    seed: int = 0,
) -> tuple[list[float], float]:
    """
    Targeted simulated‑annealing refinement.
    Moves a modest amount of mass from indices that contribute most to the
    current maximal convolution entry to random other indices.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    a = np.asarray(init_seq, dtype=float)
    if a.sum() < 1e-12:
        a = np.ones_like(a) / a.size
    else:
        a = a / a.sum()
    n = a.shape[0]

    conv = _conv_self_fft(a)
    cur_obj = float(2.0 * n * conv.max())
    best_obj = cur_obj
    best_seq = a.copy()

    # exponential temperature schedule
    T_start = cur_obj * 0.5 + 1e-9
    T_end = cur_obj * 1e-6 + 1e-12

    it = 0
    while time.time() - start < time_limit:
        it += 1
        # index where convolution is maximal
        k_max = int(np.argmax(conv))

        # contributions a[i]·a[k_max‑i] (zero outside range)
        i_vals = np.arange(n)
        j_vals = k_max - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i_vals[mask]] * a[j_vals[mask]]
        total_contrib = contrib.sum()

        # choose source index i proportional to its contribution
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = rng.choice(n, p=probs)
        else:
            i = rng.integers(n)

        # choose a different destination index uniformly
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # transfer a modest fraction of mass from i to j
        delta = a[i] * rng.random() * 0.10   # up to 10 % of a[i]
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.clip(a_new, 0.0, None)
        a_new /= a_new.sum()

        conv_new = _conv_self_fft(a_new)
        obj_new = float(2.0 * n * conv_new.max())

        # accept if improvement, otherwise Metropolis
        if obj_new < cur_obj - 1e-14:
            a, conv, cur_obj = a_new, conv_new, obj_new
        else:
            # temperature based on elapsed fraction
            t_frac = (time.time() - start) / time_limit
            T = T_start * (T_end / T_start) ** t_frac
            prob = np.exp(-(obj_new - cur_obj) / (T + 1e-18))
            if rng.random() < prob:
                a, conv, cur_obj = a_new, conv_new, obj_new

        # record a new global best
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_seq = a.copy()
            print(f"[SA it={it:07d}] new best O={best_obj:.6f}")

        # occasional global jitter to escape deep valleys
        if it % 5000 == 0:
            noise = rng.normal(scale=0.01, size=n)
            a = np.clip(a + noise, 0.0, None)
            a /= a.sum()
            conv = _conv_self_fft(a)
            cur_obj = float(2.0 * n * conv.max())

    # ------------------------------------------------------------------
    # final compression of near‑zero entries
    thresh = 1e-12
    mask = best_seq > thresh
    if mask.sum() < best_seq.size:
        comp = best_seq[mask]
        comp /= comp.sum()
        comp_obj = float(2.0 * comp.size * _conv_self_fft(comp).max())
        if comp_obj < best_obj:
            best_seq, best_obj = comp, comp_obj
            print(f"[SA compress] reduced length → {comp.size}, O={best_obj:.6f}")

    return best_seq.tolist(), best_obj

        remaining_adam = adam_deadline - now
        if remaining_adam <= 5.0:
            break

        # time split: first length gets a larger share, the rest share the rest
        if idx == 0:
            run_time = remaining_adam * 0.55
        else:
            run_time = (remaining_adam * 0.35) / (len(lengths) - idx)

        # build warm‑starts for this length
        init_candidates = [None]  # random start
        for r in r_vals:
            a = np.array([r ** i for i in range(n)], dtype=float)
            a = a / a.sum()
            init_candidates.append(a)
        for r in r_vals:
            a = np.array([r ** i for i in range(n - 1, -1, -1)], dtype=float)
            a = a / a.sum()
            init_candidates.append(a)

        # give each initialisation a roughly equal slice of the allocated time
        per_init_time = max(5.0, run_time / len(init_candidates))

        for init_idx, init_a in enumerate(init_candidates):
            now2 = time.time()
            if now2 >= adam_deadline - 2.0:
                break
            # respect the global deadline as well
            time_allowed = min(per_init_time, adam_deadline - now2 - 0.2)
            if time_allowed < 2.0:
                continue

            seq, val = _adam_optimize(
                n=n,
                time_limit=time_allowed,
                init_a=init_a,
                seed=seed + idx * 123 + init_idx * 19,
                beta_start=0.5,
                beta_end=50000.0,
                lr_start=0.2,
                lr_end=5e-5,
                jitter_interval=4000,
                restart_interval=50000,
            )
            if val < best_obj:
                best_obj, best_seq = val, np.asarray(seq, dtype=float)
                print(f"[global] new overall best O={best_obj:.6f} (n={n})")

    # --------------------------------------------------------------
    # Simulated‑annealing refinement with the remaining budget.
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if best_seq is not None and remaining > 10.0:
        seq_ref, val_ref = _simulated_annealing(
            best_seq.tolist(),
            time_limit=remaining,
            seed=seed + 9999,
        )
        if val_ref < best_obj:
            best_obj, best_seq = val_ref, np.asarray(seq_ref, dtype=float)
            print(f"[final] after SA new best O={best_obj:.6f}")

    # --------------------------------------------------------------
    # Fallback in the unlikely event that nothing was found.
    # --------------------------------------------------------------
    if best_seq is None or best_seq.size == 0:
        best_seq = np.ones(256, dtype=float) / 256.0
        best_obj = _objective(best_seq)
        print("[fallback] using uniform small sequence.")

    # --------------------------------------------------------------
    # Final sanity check – report the value seen by the evaluator.
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_seq.tolist()
