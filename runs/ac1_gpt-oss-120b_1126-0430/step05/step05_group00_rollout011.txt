#!/usr/bin/env python
"""
Gradient‑based refinement of a symmetric Beta‑like step distribution.

The algorithm proceeds in three stages:
1. Choose a support size `n` (taken from the provided `height_sequence_1`
   if it exists, otherwise default 20 000).
2. Perform a lightweight multi‑start exploration over a few
   Beta(α,α) density shapes (α≈0.5 is the classic arcsine law).  Each
   start runs a short Adam optimisation on the logits of the soft‑max
   parametrisation while using a smooth approximation of the
   max‑convolution objective.
3. Take the best candidate from the exploration and continue a long
   Adam optimisation with a decaying smoothing factor, occasional
   jitter and Adam‑style adaptive step sizes.  The sequence is always
   kept on the probability simplex, guaranteeing non‑negativity.
   The best sequence encountered (according to the exact objective
   `2·n·max(conv)`) is returned.

The code uses only NumPy and the supplied `evaluate_sequence` function.
"""
from __future__ import annotations

import math
import time
from typing import List

import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, top‑level as required)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max; returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    s = e.sum()
    if s == 0.0 or np.isnan(s):
        # fallback to uniform distribution
        return np.full_like(z, 1.0 / z.size)
    return e / s


def _logits_from_distribution(p: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it vector that reproduces `p` after soft‑max (adds a tiny epsilon)."""
    return np.log(np.maximum(p, eps))


def _fft_len_for_corr(n: int) -> int:
    """
    Smallest power of two ≥ 3·n‑2.
    This length is sufficient for a linear convolution of length (2·n‑1)
    and a linear correlation of length (3·n‑2) using FFT.
    """
    need = 3 * n - 2
    return 1 << ((need - 1).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Return the max entry of a*a and the gradient of the smoothed objective
    (2·n·softmax_max) w.r.t. a.

    Parameters
    ----------
    a : np.ndarray
        Probability vector (sum = 1, non‑negative).
    eps_factor : float
        Multiplicative factor for the smoothing constant:
        eps = eps_factor * max(conv).

    Returns
    -------
    max_conv : float
        Maximum of the self‑convolution a*a.
    grad_a : np.ndarray
        Gradient of 2·n·softmax_max w.r.t. a (already multiplied by 4·n).
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    a_fft = np.fft.rfft(a, L)

    # linear self‑convolution (length 2·n‑1)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # smoothed soft‑max weights
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv          # for numerical stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or np.isnan(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # correlation a ⋆ w (linear, length n)
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # gradient of 2·n·softmax_max (the factor 4·n appears because
    # d(conv)/da = 2·a and the outer factor 2·n)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Return a list of non‑negative heights that (approximately) minimises
    the evaluation metric defined in `evaluate_sequence`.

    The search consists of a short multi‑start phase over Beta(α,α)
    initialisations followed by a long Adam optimisation on the
    logit parametrisation of the probability simplex.
    """
    # ------------------------------------------------------------------
    # 0. General set‑up
    # ------------------------------------------------------------------
    np.random.seed(seed)
    start_time = time.time()
    # a small safety margin so we always finish before the hard timeout
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # 1. Determine the support size (n)
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
        print(f"[Info] Using length from existing variable: n = {n}")
    except Exception:
        n = 20000
        print(f"[Info] No external sequence found – using default length n = {n}")

    # --------------------------------------------------------------
    # 2. Small multi‑start exploration over Beta(α,α) shapes
    # --------------------------------------------------------------
    # Hyper‑parameters for the exploration runs
    explore_alphas = [0.4, 0.5, 0.6, 0.8, 1.0]
    explore_iters = 5000               # iterations per α
    explore_time_fraction = 0.15       # portion of total budget for this phase
    explore_deadline = start_time + explore_time_fraction * (budget_s - 2.0)

    best_score = float("inf")
    best_a = None
    best_z = None

    # helper to build a symmetric Beta(α,α) density on the uniform grid
    def _beta_init(alpha: float) -> np.ndarray:
        idx = np.arange(n, dtype=np.float64)
        x = (idx + 0.5) / n
        # avoid infinities at the boundaries when α < 1
        x = np.clip(x, 1e-12, 1.0 - 1e-12)
        raw = (x ** (alpha - 1.0)) * ((1.0 - x) ** (alpha - 1.0))
        # very small additive constant to keep the log defined
        raw += 1e-16
        a = raw / raw.sum()
        return a

    # Adam hyper‑parameters (kept identical for all short runs)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_alpha_exp = 0.5
    eps_factor_init_exp = 0.2
    eps_factor_min_exp = 5e-5
    eps_factor_decay_exp = 0.999

    for alpha in explore_alphas:
        if time.time() >= explore_deadline:
            break
        a0 = _beta_init(alpha)
        z = _logits_from_distribution(a0)
        m = np.zeros_like(z)
        v = np.zeros_like(z)

        eps_factor = eps_factor_init_exp
        it = 0

        while time.time() < explore_deadline and it < explore_iters:
            it += 1
            a = _softmax(z)

            max_conv, grad_a = _conv_max_and_grad(a, eps_factor)
            # Exact objective for a probability vector (sum = 1)
            score = 2.0 * n * max_conv

            if score < best_score:
                best_score = score
                best_a = a.copy()
                best_z = z.copy()

            # Adam update on logits (mirror descent on the simplex)
            mu = float(np.dot(grad_a, a))
            grad_z = a * (grad_a - mu)

            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            alpha_t = base_alpha_exp / (math.sqrt(it) + 1.0)
            step = -alpha_t * m_hat / (np.sqrt(v_hat) + eps_adam)
            z = z + step
            # keep logits bounded for numerical stability
            z = np.clip(z, -60.0, 60.0)

            # decay smoothing
            eps_factor = max(eps_factor_min_exp, eps_factor * eps_factor_decay_exp)

            # occasional jitter to help escape plateaus
            if it % 3000 == 0 and np.random.rand() < 0.15:
                z += 0.003 * np.random.randn(n)

    # If the exploratory phase found nothing (should not happen), fall back to arcsine
    if best_a is None:
        idx = np.arange(n, dtype=np.float64)
        x = (idx + 0.5) / n
        raw = 1.0 / np.sqrt(x * (1.0 - x))
        best_a = raw / raw.sum()
        best_z = _logits_from_distribution(best_a)

    # --------------------------------------------------------------
    # 3. Main long‑run Adam optimisation (soft‑max + decreasing epsilon)
    # --------------------------------------------------------------
    # Re‑initialise Adam state with the best candidate from the exploration phase
    z = best_z.copy()
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    # Adam / smoothing schedules for the main phase
    base_alpha = 0.3
    eps_factor = 0.08
    eps_factor_min = 5e-5
    eps_factor_decay = 0.9995

    # Track the best sequence seen so far
    a_cur = _softmax(z)
    best_seq = a_cur.copy()
    best_score = 2.0 * n * _conv_max(a_cur)

    it = 0
    max_iters = 10_000_000  # effectively “as many as time allows”

    while time.time() < deadline and it < max_iters:
        it += 1

        a = _softmax(z)

        # --- compute smoothed max‑convolution and its gradient ---
        max_conv, grad_a = _conv_max_and_grad(a, eps_factor)
        cheap_score = 2.0 * n * max_conv

        if cheap_score < best_score:
            best_score = cheap_score
            best_seq = a.copy()

        # --- Adam update on logits (mirror descent) ---
        mu = float(np.dot(grad_a, a))
        grad_z = a * (grad_a - mu)

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step = -alpha_t * m_hat / (np.sqrt(v_hat) + eps_adam)
        z = z + step
        # keep logits in a safe numerical range
        z = np.clip(z, -60.0, 60.0)

        # smoothing schedule – slowly tighten toward the true max
        eps_factor = max(eps_factor_min, eps_factor * eps_factor_decay)

        # occasional stochastic jitter to avoid being stuck in a flat region
        if it % 5000 == 0 and np.random.rand() < 0.12:
            z += 0.005 * np.random.randn(n)

        # periodic progress output (lightweight)
        if it % 20000 == 0:
            elapsed = time.time() - start_time
            print(f"[Progress] it={it:,}  best={best_score:.6f}  elapsed={elapsed:.1f}s")

    # ------------------------------------------------------------------
    # 4. Return the best sequence (plain Python list)
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()

    # Log the exact score using the provided evaluator (mostly for debugging)
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[Result] Final exact score = {final_score:.6f}")
    except Exception as exc:
        print(f"[Error] Evaluation failed: {exc}")

    return final_seq
