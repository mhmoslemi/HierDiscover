"""
Improved step‑function search.

The algorithm is a heavier version of the previous Adam‑optimizer approach.
It explores several support sizes (up to 8192 entries) and for each size spends
as much time as the remaining budget allows.  Each run uses a log‑it
parameterisation (implicit simplex) and a smooth–max approximation of the
convolution infinity‑norm; the smoothness (β) is annealed from a very low
value to a large one, after which a true max‑gradient is used.  A few
restarts (jitter + re‑initialisation) are performed automatically.
After the main optimisation the solution is pruned (tiny entries are
removed) and a cheap stochastic “mass‑transfer” local search is run.

If all attempts fail, a simple uniform distribution is returned as a safe
fallback.  The final candidate is checked with the provided ``evaluate_sequence``
routine and its objective is printed.
"""

import time
import numpy as np

# -------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# -------------------------------------------------------------
def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv


# -------------------------------------------------------------
def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation of weight vector w (length 2·n‑1) with a (length n).
    Implements Σ_t w[i+t]·a[t] for 0 ≤ i < n via FFT.
    """
    n = a.shape[0]
    m = w.shape[0]                 # = 2·n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


# -------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)), assuming Σa = 1."""
    n = a.size
    conv = _conv_self_fft(a)
    return 2.0 * n * conv.max()


# -------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


# -------------------------------------------------------------
def _run_opt(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
) -> tuple[list[float], float]:
    """
    Adam optimisation of a length‑``n`` sequence for at most ``time_limit`` seconds.
    Returns the best sequence (as a plain list) and its exact objective.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # ---- initialise -------------------------------------------------
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
        a = np.maximum(0.0, a)
        if a.sum() < 1e-12:
            a = rng.random(n)
        a = a / a.sum()
    else:
        a = rng.random(n)
        a = a / a.sum()

    # log‑it representation (always stays on the simplex)
    x = np.log(a + 1e-16)

    # Adam moments
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # schedules (smoothness β and learning‑rate decay)
    beta_start = 0.5               # start of smooth‑max (very smooth)
    beta_end   = 20000.0           # end (sharp max)
    lr_start   = 0.12
    lr_end     = 0.001

    # target iteration budget – we stop by wall‑clock
    target_iters = 600_000

    best_a = a.copy()
    best_obj = _objective(best_a)

    it = 0
    while time.time() - start < time_limit:
        it += 1

        # probabilities on the simplex
        a = _softmax(x)

        # exact objective (used for bookkeeping)
        conv = _conv_self_fft(a)
        M = conv.max()
        obj = 2.0 * n * M
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a.copy()
            print(f"[n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        # annealing of smooth‑max and of learning‑rate
        t_frac = min(it / target_iters, 1.0)
        beta = beta_start * np.exp(np.log(beta_end / beta_start) * t_frac)
        lr   = lr_start   * np.exp(np.log(lr_end   / lr_start)   * t_frac)

        # weight vector for the (smooth) max of the convolution
        if beta > 13000.0:                     # sharp enough → true max
            kmax = int(np.argmax(conv))
            w = np.zeros_like(conv)
            w[kmax] = 1.0
        else:
            shifted = conv - M                 # ≤ 0
            w = np.exp(beta * shifted)
            w = w / w.sum()                    # length 2·n‑1, sums to 1

        # smooth‑max gradient (chain rule)
        g_a = 2.0 * _correlation_fft(w, a)    # (n,)

        # natural gradient on log‑it space (projected onto simplex)
        dot = np.dot(a, g_a)                  # scalar ⟨a, g_a⟩
        g_x = a * (g_a - dot)                 # (n,)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to shake out of plateaus
        if it % 5000 == 0:
            jitter = rng.normal(scale=0.02, size=n)
            x += jitter

        # periodic restart from the current best (helps escape long stalls)
        if it % 65000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.004, size=n)
            m.fill(0.0)
            v.fill(0.0)
            print(f"[n={n}] restart at iter {it}")

    # ---- compress negligible entries ------------------------------------
    thresh = 1e-13
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        n_new = best_a.size
        best_obj = _objective(best_a)
        print(f"[compress] length {n} → {n_new}, O={best_obj:.6f}")

    # ---- short stochastic local search ----------------------------------
    remaining = time_limit - (time.time() - start)
    if remaining > 0.7:
        a_loc = best_a.copy()
        obj_loc = best_obj
        n_loc = a_loc.size
        local_deadline = time.time() + remaining * 0.35
        while time.time() < local_deadline:
            i, j = rng.choice(n_loc, size=2, replace=False)
            # move a random 0–1.5 % of mass from i to j
            delta = a_loc[i] * rng.random() * 0.015
            if delta <= 0.0:
                continue
            a_new = a_loc.copy()
            a_new[i] -= delta
            a_new[j] += delta
            conv_new = _conv_self_fft(a_new)
            obj_new = 2.0 * n_loc * conv_new.max()
            if obj_new < obj_loc:
                a_loc = a_new
                obj_loc = obj_new
                if obj_new < best_obj:
                    best_a = a_new
                    best_obj = obj_new
        print(f"[local] final O={best_obj:.6f}")

    return best_a.tolist(), best_obj


# -------------------------------------------------------------
def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **_ignore,
) -> list[float]:
    """
    Entry point for the competition harness.
    Tries several support sizes (up to 8192), runs an Adam optimiser on each,
    keeps the best solution and finishes with a final short stochastic refinement.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.6   # safety margin

    # -----------------------------------------------------------------
    # Build a list of lengths to try.
    # Start with the previously known solution (if any) and then sample a
    # geometric ladder up to 8192.
    lengths = [256, 512, 1024, 2048, 4096, 8192]

    # If a previous solution exists, prepend its size (after normalising).
    if "height_sequence_1" in globals():
        seq0 = np.asarray(height_sequence_1, dtype=float).copy()
        seq0 = np.clip(seq0, 0.0, None)
        if seq0.sum() < 0.01:
            seq0 = np.ones_like(seq0)
        seq0 = seq0 / seq0.sum()
        n0 = seq0.size
        if n0 not in lengths:
            lengths.insert(0, n0)
    else:
        seq0 = None
        n0 = None

    best_seq = None
    best_obj = float("inf")

    # -----------------------------------------------------------------
    # Main optimisation loop over lengths.
    for idx, n in enumerate(lengths):
        remaining = deadline - time.time()
        if remaining <= 1.0:
            break

        # allocate a larger share for the first (largest) length,
        # then split the rest evenly.
        if idx == 0:
            run_time = remaining * 0.65
        else:
            run_time = (remaining - 0.3) / (len(lengths) - idx)

        # initialise from the global best (if any) for the first run
        init = None
        if idx == 0 and seq0 is not None:
            if seq0.size != n:
                if seq0.size > n:
                    init = seq0[:n]
                else:
                    pad = np.zeros(n - seq0.size)
                    init = np.concatenate([seq0, pad])
                init = init / init.sum()
            else:
                init = seq0

        # ----- perform a few independent restarts for each length -----
        restarts = 3 if n >= 4096 else 2
        for r in range(restarts):
            sub_time = run_time / restarts
            seq, val = _run_opt(
                n=n,
                time_limit=sub_time,
                init_a=init if r == 0 else None,   # only first restart uses init
                seed=seed + idx * 31 + r * 13,
            )
            if val < best_obj:
                best_obj = val
                best_seq = seq
                print(f"[global] new overall best O={best_obj:.6f} (n={n})")
            # after the first restart keep the best as the new init for the next
            init = np.array(seq, dtype=float)

    # -----------------------------------------------------------------
    # Final short random local search on the overall best (if time permits)
    if best_seq is not None:
        remaining = deadline - time.time()
        if remaining > 1.0:
            a = np.array(best_seq, dtype=float)
            n = a.size
            best_obj = _objective(a)
            rng = np.random.default_rng(seed + 9999)
            local_end = time.time() + remaining * 0.30
            while time.time() < local_end:
                i, j = rng.choice(n, size=2, replace=False)
                delta = a[i] * rng.random() * 0.015
                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta
                obj_new = _objective(a_new)
                if obj_new < best_obj:
                    a = a_new
                    best_obj = obj_new
                    best_seq = a.tolist()
            print(f"[final local] O={best_obj:.6f}")

    # -----------------------------------------------------------------
    # Final sanity check via the provided evaluate_sequence routine.
    try:
        final_score = evaluate_sequence(best_seq)
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    # guarantee a non‑empty return
    if not best_seq:
        best_seq = [1.0]

    return best_seq
