    """Stable soft‑max, returns a probability vector that sums to 1."""
    z_max = np.max(z)

    s = e.sum()
    if s == 0.0:

    return e / s

    return np.log(a + eps)

    n = a.shape[0]

    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]

    s = new_a.sum()
    if s == 0.0:

    else:

            if it % 5000 == 0:
                print(f"[Iter {it:07d}] NEW BEST cheap = {cheap_score:.6f}")

        # ----- sub‑gradient for the *hard* maximum -----
        max_mask = (conv == max_conv).astype(np.float64)
        total_mask = max_mask.sum()
        if total_mask == 0.0:
            w = np.full_like(max_mask, 1.0 / max_mask.size)
        else:
            w = max_mask / total_mask
        w_fft = np.fft.rfft(w, L)
        # correlation a⋆w (see derivation in analysis)
        corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]
        grad_a = 4.0 * n * corr                 # gradient of 2·n·max_conv

        # ----- mirror‑descent step on logits (KL geometry) -----
        mu = np.dot(grad_a, a)                  # ⟨∇f, a⟩
        grad_z = a * (grad_a - mu)              # ∇_z f = a ⊙ (∇_a f – μ)

        # ----- Adam moment updates (bias‑corrected) -----
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # ----- learning‑rate schedule (inverse‑sqrt decay) -----
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step_dir = - m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- back‑tracking line search (up to 4 halvings) -----
        accepted = False
        trial_alpha = alpha_t
        for _ in range(4):
            z_new = z + trial_alpha * step_dir
            z_new = np.clip(z_new, -50.0, 50.0)      # keep exp() stable
            a_new = _softmax(z_new)
            a_fft_new = np.fft.rfft(a_new, L)
            conv_new = np.fft.irfft(a_fft_new * a_fft_new, L)[: 2 * n - 1]
            new_score = 2.0 * n * conv_new.max()
            if new_score <= cheap_score:
                z = z_new
                accepted = True
                break
            trial_alpha *= 0.5

        if not accepted:
            # take a modest step anyway (no guarantee of decrease)
            z += alpha_t * step_dir
            z = np.clip(z, -50.0, 50.0)

        # ----- occasional random jitter to escape plateaux -----
        if it % 4000 == 0 and np.random.rand() < 0.25:
            z += 0.01 * np.random.randn(n)
            m *= 0.6
            v *= 0.6

        # ----- gradually tighten smoothing (mostly unused, kept for safety) -----
        eps_factor = max(eps_factor_min, eps_factor * eps_factor_decay)

        # ----- restart after long stagnation -----
        if it - last_improve_iter > 250_000:
            print(f"[Iter {it}] Stagnation – restarting from a fresh asymmetric seed.")
            # fresh asymmetric seed (random exponential)
            lam = 10 ** np.random.uniform(-6, -1)
            if np.random.rand() < 0.5:
                a0 = np.exp(-lam * np.arange(n, dtype=np.float64))
            else:
                a0 = np.exp(-lam * np.arange(n - 1, -1, -1, dtype=np.float64))
            a0 /= a0.sum()
            z = _logits_from_distribution(a0, eps) + 0.005 * np.random.randn(n)
            m.fill(0.0)
            v.fill(0.0)
            last_improve_iter = it
            eps_factor = 0.05

        # ----- optional up‑sampling if time permits -----
        if (it % 100_000 == 0) and (n < 50000) and (time.time() + 30.0 < deadline):
            new_n = min(50000, int(n * 1.25))
            print(f"[Iter {it}] Upsampling support: {n} → {new_n}")
            best_a = _interpolate_distribution(best_a, new_n)
            best_logit = _logits_from_distribution(best_a, eps) + 0.005 * np.random.randn(new_n)
            # reset optimiser for the larger problem
            n = new_n
            L = _fft_len_for_corr(n)
            z = best_logit
            m = np.zeros_like(z)
            v = np.zeros_like(z)
            best_score = 2.0 * n * _conv_max(best_a)
            last_improve_iter = it
            eps_factor = 0.05
            # continue optimisation with the enlarged size

    # ------------------------------------------------------------------
    # 4. Final exact evaluation with the provided evaluate_sequence()
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # ------------------------------------------------------------------
    # 5. Return the best sequence discovered.
    # ------------------------------------------------------------------
    return final_seq
