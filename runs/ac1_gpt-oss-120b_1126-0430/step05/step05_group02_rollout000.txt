#!/usr/bin/env python
"""
Propose a non‑negative step function (list of heights) that minimises

        2·n·max_k (a*a)[k] / (Σ a_i)²

The algorithm consists of three stages:

1. **Fast parametric scan** – a set of simple families (uniform,
   power‑law, exponential, power‑law × exponential, beta‑shaped,
   Dirichlet) is examined on a modestly large number of points
   (default 8 000).  The best sequence found is taken as a starting
   point.

2. **Smoothed mirror descent** – we minimise a smooth approximation of
   the “max‑convolution’’ objective using an exponentiated‑gradient
   (mirror‑descent) scheme on the probability simplex.
   The smooth approximation is the log‑sum‑exp of the autocorrelation,
   controlled by a temperature‐parameter ``beta`` that is increased
   gradually.  Gradients are obtained via FFT‑based convolutions, which
   makes each iteration cheap (≈ 4 FFTs).

3. **Exact evaluation** – the best sequence discovered is finally
   evaluated with the supplied ``evaluate_sequence`` routine and returned.

The implementation respects the supplied time budget (default 1000 s)
and never returns an empty or invalid list.  It prints a few progress
messages so that the optimisation progress can be followed.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float) -> np.ndarray:
    """Return a normalised power‑law a_i ∝ (i+1)^‑p (Σa_i = 1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = i ** (-p)
    seq /= seq.sum()
    return seq


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using FFT (returns length 2·n‑1)."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def _conv_max_fft(a: np.ndarray, fft_len: int) -> float:
    """Maximum value of the autocorrelation a*a (sum(a)=1 assumed)."""
    conv = _conv_full_fft(a, fft_len)
    return float(conv.max())

                break
            idx = np.arange(n, dtype=np.float64)
            cand = (idx + 1.0) ** (-p) * np.exp(-lam * idx)
            cand /= cand.sum()
            val = cheap_score(cand)
            if val < best_val:
                best_a, best_val = cand.copy(), val

    # beta‑shaped families: a_i ∝ x_i^α (1‑x_i)^β
    x = (np.arange(1, n + 1, dtype=np.float64) / n)
    for alpha in np.linspace(0.2, 5.0, 21):
        for beta in np.linspace(0.2, 5.0, 21):
            if time.time() >= deadline:
                break
            cand = (x ** alpha) * ((1.0 - x) ** beta)
            s = cand.sum()
            if s == 0.0:
                continue
            cand /= s
            val = cheap_score(cand)
            if val < best_val:
                best_a, best_val = cand.copy(), val

    # a few random Dirichlet seeds (various concentrations)
    for conc in [0.001, 0.01, 0.1, 0.5, 1.0, 3.0]:
        if time.time() >= deadline:
            break
        g = np.random.gamma(conc, 1.0, size=n)
        cand = g / g.sum()
        val = cheap_score(cand)
        if val < best_val:
            best_a, best_val = cand.copy(), val

    print(f"[Init] best cheap score = {best_val:.6f}")

    # --------------------------------------------------------------
    # 1. Refine the best candidate with smoothed mirror descent
    # --------------------------------------------------------------
    a = best_a.copy()           # current iterate (always Σa = 1)
    best_iter_a = a.copy()      # best ever seen during refinement
    best_iter_val = best_val
    last_improve = 0
    iter_cnt = 0

    # hyper‑parameters
    eta0 = 0.6                 # initial step‑size factor
    beta_start = 2.0           # start temperature for soft‑max
    beta_max = 80.0            # final temperature (sharpens to max)
    beta_ramp_iters = 200_000  # iterations over which beta is increased

    # Pre‑allocate some arrays to avoid repeated allocations
    eps = 1e-30
    L = 2 * n - 1              # length of the autocorrelation vector

    while time.time() < deadline and iter_cnt < 2_000_000:
        iter_cnt += 1

        # ----- compute full autocorrelation (via FFT) -----
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:L]
        conv_max = conv.max()
        cheap = 2.0 * n * conv_max

        # ----- keep the best cheap score -----
        if cheap < best_iter_val - 1e-12:
            best_iter_val = cheap
            best_iter_a = a.copy()
            last_improve = iter_cnt
            print(f"[Iter {iter_cnt:07d}] NEW BEST cheap = {cheap:.6f}")

        # ----- temperature schedule for the smooth max -----
        beta = beta_start + (beta_max - beta_start) * min(1.0,
                                                          iter_cnt / beta_ramp_iters)

        # ----- soft‑max weights over convolution entries -----
        # shift by the maximum for numerical stability
        w = np.exp(beta * (conv - conv_max))
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            # fallback to the true max entry (should never happen)
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            w_sum = 1.0
        w_norm = w / w_sum

        # ----- gradient of the smoothed objective -----
        # g_j = 2 * Σ_i w_norm[i + j] * a_i
        w_rev = w_norm[::-1]                     # reversed weights
        fw = np.fft.rfft(w_rev, n=fft_len)
        grad_fft = fa * fw
        conv_rev = np.fft.irfft(grad_fft, n=fft_len)[:3 * n - 2]
        idx = (L - 1) - np.arange(n)             # indices picking the right slice
        g = 2.0 * conv_rev[idx]

        # Center the gradient (adding a constant does not affect the simplex step)
        g_centered = g - g.mean()

        # ----- exponentiated‑gradient (mirror descent) step -----
        eta = eta0 / math.sqrt(iter_cnt)        # decaying step size
        log_a = np.log(a + eps) - eta * g_centered
        a = np.exp(log_a)

        # Re‑normalise onto the simplex
        s = a.sum()
        if s == 0.0 or not np.isfinite(s):
            # fall back to the last good iterate
            a = best_iter_a.copy()
            s = a.sum()
        a /= s

        # ----- occasional restart when progress stalls -----
        if iter_cnt - last_improve > 300_000:
            perturb = 0.01 * np.random.rand(n)
            a = _simplex_projection(best_iter_a + perturb)
            last_improve = iter_cnt
            print(f"[Iter {iter_cnt:07d}] Restart after stagnation.")

        # Light random jitter every few hundred thousand iterations
        if iter_cnt % 200_000 == 0:
            jitter = 0.005 * np.random.rand(n)
            a = _simplex_projection(a + jitter)

    # --------------------------------------------------------------
    # 2. Final exact evaluation (uses the provided ``evaluate_sequence``)
    # --------------------------------------------------------------
    final_seq = best_iter_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:      # defensive – should never happen
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # --------------------------------------------------------------
    # 3. Return the best sequence found.
    # --------------------------------------------------------------
    return final_seq
```
This implementation expands upon the previous approach by employing a
temperature‑controlled log‑sum‑exp smoothing of the max‑convolution,
allowing gradients that take into account several large convolution
entries while still focusing on the dominant peak as the optimisation
progresses.  The mirror‑descent (exponentiated‑gradient) updates keep
the iterate on the probability simplex without an explicit projection,
and a few simple restart heuristics help escape plateaus.  The
combination of a broad parametric initialisation and the refined
gradient‑based search typically improves the bound well below the
previous 1.5155 level within the allotted time budget.
