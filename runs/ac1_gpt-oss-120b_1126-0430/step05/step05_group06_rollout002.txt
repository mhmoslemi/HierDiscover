"""
Propose a step‑function sequence that (approximately) minimises

    O(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm is a multi‑stage heuristic that works entirely on the simplex
(∑a = 1).  It combines:

* **Phase 0 – Diverse warm‑starts** – Random Dirichlet draws with a wide range
  of concentration parameters, each followed by a few cheap exponentiated‑gradient
  steps on a smooth log‑sum‑exp surrogate of the max‑convolution.

* **Phase 1 – Long exponentiated‑gradient optimisation** – On the best warm‑starts
  we run many iterations of exponentiated‑gradient descent.  The surrogate
  uses an increasingly sharp soft‑max (α grows from ≈5 to 5·10⁵) and an
  entropy regulariser (λ·∑a log a) that keeps the mass spread and helps avoid
  poor local minima.

* **Phase 2 – Targeted mass‑transfer (peak‑reduction)** – Repeatedly moves a
  small fraction of probability mass from indices that contribute heavily to
  the current maximal convolution entry to random destinations.  Every 50
  iterations an exact sub‑gradient step is performed to keep the walk directed.

* **Phase 3 – Exact projected sub‑gradient refinement** – A final back‑tracking
  line‑search on the true nonsmooth objective guarantees a descent direction
  and polishes the solution.

All phases respect a total wall‑time budget (default 1000 s) and constantly
store the best sequence seen.  The returned list is verified with the provided
``evaluate_sequence`` function before being handed back to the harness.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, ∑x=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                      # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear self‑convolution via FFT (zero‑padded to length L)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return a*a (full convolution) using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Exact objective for a simplex‑normalised vector (∑a = 1)
def _objective(a: np.ndarray, n: int, L: int) -> float:
    """2·n·max(conv(a,a))  (a is assumed to sum to 1)."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Sub‑gradient of the maximal convolution entry (ordered pairs)
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. a (ordered‑pair definition)."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        # each ordered pair (i,j) contributes a[j] to ∂/∂a[i]
        g[i_idx] += a[j_idx]
        # and symmetrically the other direction (handled when iterating over k)
    # because each ordered pair appears once in the loop above, we need the
    # factor 2 for the symmetric contribution.
    g *= 2.0
    if ks.size:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
# Gradient of the log‑sum‑exp (smooth max) surrogate
def _smooth_gradient(a: np.ndarray, conv: np.ndarray,
                     alpha: float, L: int) -> np.ndarray:
    """
    Compute ∇_a ( (2·n/α)·log∑_k e^{α·conv_k} ).
    The resulting vector equals 2·corr(w, a) where
    w_k ∝ e^{α·conv_k} (soft‑max weights).
    """
    # soft‑max weights
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)      # stabilise
    w = exp_scaled / exp_scaled.sum()

    # correlation w ⋆ a   (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)   # length ≥ 3·n‑2
    n_len = a.shape[0]
    grad = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]
    return grad


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the smooth surrogate + entropy regulariser
def _expgrad_phase(a: np.ndarray,
                   n: int,
                   L: int,
                   max_iters: int,
                   deadline: float,
                   alpha_start: float = 5.0,
                   alpha_end: float = 5e5,
                   eta0: float = 0.12,
                   lambda_entropy: float = 1e-3,
                   verbose: bool = False) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Exponentiated‑gradient on the soft‑max surrogate.
    Returns (final a, best objective value, best a).
    """
    best_a = a.copy()
    best_obj = _objective(a, n, L)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            if verbose and it % 2000 == 0:
                print(f"[exp‑grad] iter {it} new best O={best_obj:.6f}")

        # α schedule – exponential growth
        frac = it / max_iters
        alpha = alpha_start * (alpha_end / alpha_start) ** frac

        grad_smooth = _smooth_gradient(a, conv, alpha, L)

        # entropy gradient  λ·(log a + 1)
        grad_entropy = lambda_entropy * (np.log(a + 1e-30) + 1.0)

        grad = grad_smooth + grad_entropy

        # learning‑rate schedule (slow decay)
        eta = eta0 / (1.0 + it * 1e-4)

        # multiplicative update, renormalise to stay on the simplex
        a = a * np.exp(-eta * grad)
        sum_a = a.sum()
        if sum_a == 0.0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / sum_a

    return a, best_obj, best_a

# Exact projected sub‑gradient descent (back‑tracking line‑search)
def _subgradient_phase(a: np.ndarray,
                       n: int,
                       L: int,
                       max_iters: int,
                       deadline: float,
                       eta0: float = 0.25,
                       line_search: bool = True,
                       verbose: bool = False) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Projected sub‑gradient descent on the true objective.
    Returns (final a, best objective, best a).
    """
    best_a = a.copy()
    best_obj = _objective(a, n, L)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            if verbose and it % 5000 == 0:
                print(f"[sub‑grad] iter {it} new best O={best_obj:.6f}")

        g = _subgradient_max_conv(a, conv)

        # geometric decay for the step size
        eta = eta0 * (0.9999 ** it)

        if line_search:
            step = eta
            for _ in range(5):
                cand = _project_to_simplex(a - step * g)
                cand_obj = _objective(cand, n, L)
                if cand_obj < cur_obj:
                    a = cand
                    break
                step *= 0.5
            else:
                a = _project_to_simplex(a - eta * g)
        else:
            a = _project_to_simplex(a - eta * g)

    return a, best_obj, best_a

# Targeted mass‑transfer (peak‑reduction) phase
def _mass_transfer_phase(a: np.ndarray,
                         n: int,
                         L: int,
                         max_iters: int,
                         deadline: float,
                         verbose: bool = True) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Local refinement: move a small fraction of mass away from indices that
    participate heavily in the current maximal convolution entry.
    Occasionally a true sub‑gradient step is taken to keep the walk directed.
    """
    best_a = a.copy()
    best_obj = _objective(a, n, L)
    cur_obj = best_obj
    conv = _conv_fft(a, L)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        kmax = int(np.argmax(conv))

        # contributors to the peak
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        i_idxs = np.arange(i_start, i_end + 1, dtype=int)
        # choose source with the largest product a[i]*a[kmax-i]
        prod = a[i_idxs] * a[kmax - i_idxs]
        src = i_idxs[np.argmax(prod)]

        # pick a random destination (avoid src)
        dst = np.random.randint(n)
        while dst == src:
            dst = np.random.randint(n)

        # transfer a modest fraction of the source mass
        delta = a[src] * np.random.rand() * 0.07  # ≤7 % of source mass
        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta
        a_new = _project_to_simplex(a_new)

        conv_new = _conv_fft(a_new, L)
        new_obj = 2.0 * n * conv_new.max()

        # deterministic acceptance if we improve
        if new_obj < cur_obj:
            a = a_new
            conv = conv_new
            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_a = a.copy()
                if verbose:
                    print(f"[mass] iter {it} new best O={best_obj:.6f}")

        # occasional sub‑gradient “reset”
        if it % 37 == 0:
            g = _subgradient_max_conv(a, conv)
            eta = 0.015 * (1.0 - it / max_iters)
            a = _project_to_simplex(a - eta * g)
            conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                if verbose:
                    print(f"[mass] sub‑grad step O={best_obj:.6f}")

    return a, best_obj, best_a

# Main entry point expected by the evaluation harness
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the provided
    evaluation function.  The routine respects the wall‑time budget and
    returns the best sequence (as a plain Python list) found so far.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a tiny safety margin
    deadline = start_time + budget_s - 0.15

    # --------------------------------------------------------------
    # Initialise problem size
    if 'height_sequence_1' in globals():
        a0 = np.array(height_sequence_1, dtype=float)
        a0 = np.maximum(a0, 0.0)
        if a0.sum() == 0.0:
            a0 = np.full_like(a0, 1.0)
        a0 = a0 / a0.sum()
        n = a0.size
        a = a0.copy()
        print(f"[init] using supplied sequence, n={n}")
    else:
        n = 4600
        a = np.full(n, 1.0 / n)   # uniform start on the simplex
        print(f"[init] using uniform start, n={n}")

    # FFT length – next power of two greater than 3·n‑2
    L = 1 << ((3 * n - 2).bit_length())

    # --------------------------------------------------------------
    # Phase 0 – cheap random warm‑starts
    phase0_deadline = start_time + budget_s * 0.12
    best_seq = a.copy()
    best_score = _objective(best_seq, n, L)

    conc_vals = [0.03, 0.08, 0.2, 0.5, 1.0, 2.0, 5.0]
    while time.time() < phase0_deadline:
        conc = np.random.choice(conc_vals)
        a_rand = np.random.gamma(conc, 1.0, size=n)
        a_rand = a_rand / a_rand.sum()
        # a few light exponentiated‑gradient steps
        a_tmp, _, best_tmp = _expgrad_phase(
            a_rand,
            n,
            L,
            max_iters=800,
            deadline=time.time() + 2.0,
            alpha_start=5.0,
            alpha_end=2000.0,
            eta0=0.25,
            lambda_entropy=5e-4,
            verbose=False,
        )
        cur_obj = _objective(best_tmp, n, L)
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = best_tmp.copy()
            print(f"[phase0] new best O={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – long exponentiated‑gradient optimisation
    phase1_deadline = start_time + budget_s * 0.55
    # keep the top‑K seeds (K=4) from phase 0 for long optimisation
    seeds = [best_seq]
    # add a few extra random seeds to increase diversity
    while len(seeds) < 4 and time.time() < phase1_deadline:
        a_rand = np.random.gamma(0.2, 1.0, size=n)
        a_rand = a_rand / a_rand.sum()
        seeds.append(a_rand)

    for idx, seed_a in enumerate(seeds):
        if time.time() >= phase1_deadline:
            break
        a_opt, obj_opt, best_opt = _expgrad_phase(
            seed_a,
            n,
            L,
            max_iters=250_000,
            deadline=phase1_deadline,
            alpha_start=5.0,
            alpha_end=5e5,
            eta0=0.12,
            lambda_entropy=1e-3,
            verbose=False,
        )
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = best_opt.copy()
            print(f"[phase1‑{idx}] improved best O={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – targeted mass‑transfer (peak‑reduction)
    phase2_deadline = start_time + budget_s * 0.80
    a_mt, obj_mt, best_mt = _mass_transfer_phase(
        best_seq,
        n,
        L,
        max_iters=200_000,
        deadline=phase2_deadline,
        verbose=True,
    )
    if obj_mt < best_score:
        best_score = obj_mt
        best_seq = best_mt.copy()
        print(f"[phase‑2] improved best O={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – exact projected sub‑gradient refinement
    phase3_deadline = deadline
    a_sg, obj_sg, best_sg = _subgradient_phase(
        best_seq,
        n,
        L,
        max_iters=250_000,
        deadline=phase3_deadline,
        eta0=0.22,
        line_search=True,
        verbose=False,
    )
    if obj_sg < best_score:
        best_score = obj_sg
        best_seq = best_sg.copy()
        print(f"[phase‑3] final best O={best_score:.6f}")

    # --------------------------------------------------------------
    # Final verification through the provided evaluator
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence failed:", exc)

    return best_seq.tolist()
