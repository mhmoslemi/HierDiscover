"""
Propose a non‑negative step‑function (list of heights) that approximately
minimises

        O(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm runs three successive phases within the allocated time budget:

1. **Variance flattening (Adam).**  
   Minimise the variance of the full convolution `c = a★a`.  This pushes all
   entries of the convolution towards their common mean `μ = 1/(2n‑1)`.  The
   gradient of the variance is a simple correlation that can be evaluated by
   FFT.  An Adam optimiser on the probability simplex (∑a=1, a≥0) is used.

2. **Smooth max‑surrogate (exponentiated‑gradient).**  
   Replace `max(c)` by the log‑sum‑exp surrogate  
   `Sα(c) = (1/α)·log∑_k e^{α·c_k}` with a slowly increasing temperature `α`.  
   The gradient of `Sα` is a weighted correlation; a mirror‑descent step in
   the Kullback–Leibler geometry (multiplicative update) is performed,
   followed by projection onto the simplex.

3. **Stochastic local refinement.**  
   A mixture of random “mass‑transfer” moves (Metropolis acceptance) and
   deterministic sub‑gradient steps on the true `max(conv)` is carried out.
   Restarts with jitter are triggered when the search stagnates.

All phases keep track of the best sequence encountered (according to the
exact `evaluate_sequence` objective).  The final best sequence is returned
as a plain Python list.
"""

# ----------------------------------------------------------------------
# Simplex utilities -----------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, sum=1}."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:                     # all entries are non‑positive
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution & correlation -----------------------------------
def next_pow2(x: int) -> int:
    """Small helper: the smallest power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())


def conv_full_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution a★a (length 2·n‑1) using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft_a * fft_a, n=L)
    # keep only the first 2·n‑1 entries (the rest are zero‑padded artefacts)
    return conv[: 2 * a.size - 1]


def corr_fft(x: np.ndarray, a_rev: np.ndarray, L: int) -> np.ndarray:
    """
    Correlation of x (length 2·n‑1) with a_rev (length n) via FFT.
    Returns the full correlation vector of length 3·n‑2; the central `n`
    entries (indices n‑1 … n‑1+n‑1) correspond to the gradient components.
    """
    fft_x = np.fft.rfft(x, n=L)
    fft_rev = np.fft.rfft(a_rev, n=L)
    corr = np.fft.irfft(fft_x * fft_rev, n=L)
    return corr


# ----------------------------------------------------------------------
# Objective helpers -----------------------------------------------------
def objective_maxconv(a: np.ndarray, n: int, L: int) -> float:
    """Exact objective for a probability vector (sum(a)=1): 2·n·max(conv)."""
    conv = conv_full_fft(a, L)
    return 2.0 * n * np.max(conv)


def variance_gradient(a: np.ndarray, conv: np.ndarray, n: int, L: int) -> np.ndarray:
    """
    Gradient of Σ_k (conv_k - μ)²  w.r.t. a,
    where μ = 1/(2·n‑1) (the average of the convolution entries for sum(a)=1).
    Closed form: 4·corr(conv‑μ , a).
    """
    mu = 1.0 / (2 * n - 1)
    diff = conv - mu                     # length 2·n‑1
    a_rev = a[::-1]
    corr = corr_fft(diff, a_rev, L)
    start = n - 1
    return 4.0 * corr[start:start + n]


def smooth_max_gradient(a: np.ndarray, conv: np.ndarray,
                        alpha: float, n: int, L: int) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp surrogate:
        (2·n/α)·log ∑_k e^{α·conv_k}
    plus a tiny entropy term to keep strict positivity.
    """
    # ----- soft‑max weights -----
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)      # avoid overflow
    w = exp_scaled / np.sum(exp_scaled)           # shape (2·n‑1,)

    # ----- correlation term (2·corr(w, a)) -----
    a_rev = a[::-1]
    corr = corr_fft(w, a_rev, L)
    start = n - 1
    grad_corr = 2.0 * corr[start:start + n]

    # ----- tiny entropy (helps avoid exact zeros) -----
    eps = 1e-20
    grad_ent = 1e-6 * (np.log(a + eps) + 1.0)      # λ = 1e‑6

    return grad_corr + grad_ent


def subgradient_max(a: np.ndarray, conv: np.ndarray, n: int) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a.
    For every index k attaining the maximum, we add the contribution
    a_{k-i} (with the factor 2 for i≠k‑i) to the gradient component i.
    """
    max_val = np.max(conv)
    ks = np.flatnonzero(conv == max_val)
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        # contribution 2·a_j for i ≠ j, else a_j
        contrib = 2.0 * a[j_idx]
        # correct the diagonal terms (where i == j)
        diag = i_idx == j_idx
        contrib[diag] = a[j_idx[diag]]
        g[i_idx] += contrib
    if ks.size:
        g /= ks.size
    return g

        grad += 1e-5 * (np.log(a + 1e-20) + 1.0)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        lr = lr0 / math.sqrt(it + 1)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = project_to_simplex(a_new)

        it += 1
        if it % 20000 == 0:
            print(f"[phase‑1] iter {it:,} O = {cur_obj:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – smooth‑max surrogate (log‑sum‑exp) + multiplicative updates
    phase2_deadline = start_time + 0.66 * budget_s
    it = 0
    alpha_start, alpha_end = 4.0, 8000.0
    eta0 = 0.6                     # base step‑size for the mirror descent
    while time.time() < phase2_deadline:
        conv = conv_full_fft(a, L)
        cur_obj = 2.0 * n * np.max(conv)

        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[phase‑2] iter {it:,} new best O = {best_score:.6f}")

        # anneal temperature α linearly in iteration count
        frac = min(1.0, it / 200_000)
        alpha = alpha_start + (alpha_end - alpha_start) * frac

        # gradient of the smooth surrogate
        grad = smooth_max_gradient(a, conv, alpha, n, L)

        # mirror‑descent step (multiplicative update)
        # to keep numbers in a reasonable range we shift the gradient:
        grad_shift = grad - np.max(grad)          # avoid overflow in exp
        step = eta0 / math.sqrt(it + 1)           # simple decay
        a_mul = np.exp(-step * grad_shift)
        a = a * a_mul
        a = project_to_simplex(a)

        # occasional jitter to escape plateaus
        if np.random.rand() < 0.001:
            a += np.random.randn(n) * 5e-5
            a = np.clip(a, 0, None)
            a = project_to_simplex(a)

        it += 1
        if it % 25000 == 0:
            print(f"[phase‑2] iter {it:,} O = {cur_obj:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – stochastic local refinement (mass‑transfer + sub‑grad)
    print(f"[phase‑3] start local refinement, best O = {best_score:.6f}")
    temperature0 = 0.04
    max_local_iters = 250_000
    no_improve_limit = 3000
    restart_counter = 0

    while time.time() < deadline:
        # ---- initialise a restart -------------------------------------------------
        if np.random.rand() < 0.6:
            # jittered copy of the best solution so far
            a = best_seq.copy()
            a += np.random.randn(n) * 1e-3
            a = np.clip(a, 0, None)
            a = project_to_simplex(a)
        else:
            # fresh random start (helps exploring new basins)
            a = np.random.rand(n)
            a = project_to_simplex(a)

        conv = conv_full_fft(a, L)
        cur_obj = 2.0 * n * np.max(conv)

        it_loc = 0
        no_improve = 0
        T = temperature0

        while time.time() < deadline and it_loc < max_local_iters:
            # --------------------------------------------------------------
            # (a) Random mass‑transfer move (Metropolis acceptance)
            i, j = np.random.randint(n, size=2)
            if i != j and a[i] > 1e-14:
                delta = a[i] * np.random.rand() * 0.12   # up to 12 % of source mass
                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta
                a_new = project_to_simplex(a_new)

                conv_new = conv_full_fft(a_new, L)
                new_obj = 2.0 * n * np.max(conv_new)

                if new_obj < cur_obj or np.random.rand() < math.exp((cur_obj - new_obj) / max(T, 1e-14)):
                    a, conv, cur_obj = a_new, conv_new, new_obj
                    no_improve = 0
                    if cur_obj < best_score:
                        best_score = cur_obj
                        best_seq = a.copy()
                        ts = time.time() - start_time
                        print(f"[{ts:.1f}s] new global best O={best_score:.6f} (restart={restart_counter}, iter={it_loc})")
                else:
                    no_improve += 1
            else:
                no_improve += 1

            # --------------------------------------------------------------
            # (b) Sub‑gradient step on the current maximum
            if it_loc % 7 == 0:          # relatively frequent
                g = subgradient_max(a, conv, n)
                # simple diminishing step size
                eta = 0.08 * (1.0 - it_loc / max_local_iters)
                a_tmp = a - eta * g
                a_tmp = project_to_simplex(a_tmp)

                conv_tmp = conv_full_fft(a_tmp, L)
                obj_tmp = 2.0 * n * np.max(conv_tmp)
                if obj_tmp < cur_obj:
                    a, conv, cur_obj = a_tmp, conv_tmp, obj_tmp
                    if cur_obj < best_score:
                        best_score = cur_obj
                        best_seq = a.copy()
                        ts = time.time() - start_time
                        print(f"[{ts:.1f}s] sub‑grad improved O={best_score:.6f}")

            # --------------------------------------------------------------
            # (c) Occasional smoothing of the vector (helps in flat regions)
            if it_loc % 5000 == 0 and it_loc > 0:
                kernel = np.array([0.125, 0.25, 0.25, 0.25, 0.125])
                a_s = np.convolve(a, kernel, mode="same")
                a = project_to_simplex(0.85 * a + 0.15 * a_s)
                conv = conv_full_fft(a, L)
                cur_obj = 2.0 * n * np.max(conv)

            # --------------------------------------------------------------
            # temperature decay & bookkeeping
            T *= 0.9997
            it_loc += 1
            if no_improve >= no_improve_limit:
                break

        restart_counter += 1

    # ----------------------------------------------------------------------
    # Final sanity check with the provided evaluation function
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O = {final_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence failed:", exc)

    # Return the best sequence found (as a plain list of floats)
    return best_seq.tolist()
