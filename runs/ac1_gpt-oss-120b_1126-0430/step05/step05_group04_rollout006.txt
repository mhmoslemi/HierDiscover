"""
Non‑negative step‑function optimiser – improved version.

The optimiser works in three phases:

1.  **Coarse optimisation** – several short Adam runs on a hierarchy of
    support sizes (e.g. 2 k → 6 k → 12 k).  Each Adam run uses a
    log‑softmax parametrisation (simplex) and a continuation scheme where
    the smoothing parameter β is increased exponentially until a hard
    max‑gradient is used.  Early runs provide a good warm‑start for the
    larger problems.

2.  **Targeted simulated‑annealing** – a refined SA that moves a modest
    fraction of mass from indices that contribute most to the current
    maximal convolution entry to random destinations.  The temperature is
    initialised from the current objective and decays slowly.

3.  **Deterministic batch‑local search** – a greedy hill‑climber that,
    repeatedly, picks a high‑contributing index and transfers a small
    amount of its mass to a random other index if the objective improves.
    This stage often squeezes a few more hundredths out of the bound.

All steps respect the total 1000 s budget, use FFTs for fast
convolutions and project onto the probability simplex after every move.
The final sequence is returned as a plain list of floats suitable for
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (fast)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return conv_full[: 2 * n - 1]


def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation (w⋆a)[i] = Σ_t w[i+t]·a[t]  for i=0,…,n‑1.
    Implemented via FFT.
    """
    n = a.shape[0]
    m = w.shape[0]                     # = 2n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) under the assumption sum(a)=1.
    """
    n = a.size
    conv = _conv_self_fft(a)
    return float(2.0 * n * conv.max())


def _hard_gradient(a: np.ndarray, kmax: int) -> np.ndarray:
    """
    Gradient of (exact) max‑convolution w.r.t. a when the maximum occurs
    at index kmax.  g_i = 2·a_{kmax‑i} for indices where this term is
    defined, otherwise 0.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    # i such that 0 ≤ kmax‑i < n  ⇒  kmax‑(n‑1) ≤ i ≤ kmax
    i_start = max(0, kmax - (n - 1))
    i_end = min(kmax, n - 1)
    idx = np.arange(i_start, i_end + 1, dtype=int)
    g[idx] = 2.0 * a[kmax - idx]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implementation follows the algorithm of Wang & Carreira‑Perpiñán
    (2013).  Works for any real vector.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _run_adam(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
    beta_start: float = 0.5,
    beta_end: float = 2e5,
    lr_start: float = 0.2,
    lr_end: float = 0.001,
    hard_beta_thresh: float = 5e3,
) -> tuple[list[float], float]:
    """
    Adam optimiser on the log‑softmax simplex.

    Parameters
    ----------
    n : int
        Length of the sequence.
    time_limit : float
        Seconds allocated to this run.
    init_a : np.ndarray | None
        Optional warm‑start (will be normalised to sum=1).
    seed : int
        RNG seed.
    beta_start, beta_end : float
        Smoothing schedule for the soft‑max approximation of the max.
    lr_start, lr_end : float
        Learning‑rate schedule.
    hard_beta_thresh : float
        When the smoothing factor exceeds this value we switch to the
        exact (hard) max gradient, which is much cheaper.

    Returns
    -------
    best_seq : list[float]
        Normalised sequence achieving the lowest objective seen.
    best_obj : float
        Corresponding objective value (2·n·max(conv)).
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # initialise a on the simplex
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
        a = np.clip(a, 0, None)
        s = a.sum()
        if s < 1e-12:
            a = np.ones(n) / n
        else:
            a = a / s
    else:
        a = rng.random(n)
        a = a / a.sum()

    # logits for the soft‑max parametrisation
    x = np.log(a + 1e-16)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    best_a = a.copy()
    best_obj = _objective(best_a)

    it = 0
    while time.time() - start < time_limit:
        it += 1
        a = _softmax(x)  # stay on simplex

        conv = _conv_self_fft(a)
        M = conv.max()
        obj = float(2.0 * n * M)

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[Adam n={n:5d} it={it:06d}] new best O={best_obj:.6f}")

        # schedule (time‑based)
        t = min((time.time() - start) / time_limit, 1.0)
        beta = beta_start * np.exp(np.log(beta_end / beta_start) * t)
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t)

        # gradient of (smooth) max
        if beta > hard_beta_thresh:
            kmax = int(np.argmax(conv))
            g_a = _hard_gradient(a, kmax)
        else:
            # smooth weighting
            shifted = conv - M               # ≤ 0
            w = np.exp(beta * shifted)
            s = w.sum()
            if s == 0.0:
                w = np.ones_like(conv) / conv.size
            else:
                w /= s
            # correlation gives ∂/∂a of Σ w_k conv_k
            g_a = 2.0 * _correlation_fft(w, a)

        # natural gradient in log‑it space (mirrored gradient)
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape plateaus
        if it % 4000 == 0:
            jitter = rng.normal(scale=0.01, size=n)
            x += jitter

        # periodic restart from current best
        if it % 60000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.005, size=n)
            m.fill(0.0)
            v.fill(0.0)

    # ------------------------------------------------------------------
    # prune negligible entries (helps later phases)
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = _objective(best_a)
        print(f"[Adam compress] length → {best_a.size}, O={best_obj:.6f}")

    return best_a.tolist(), best_obj


def _run_sa(
    init_seq: list[float],
    time_limit: float,
    seed: int = 0,
) -> tuple[list[float], float]:
    """
    Targeted simulated‑annealing refinement.

    The move rule is the same as in the reference implementation but
    a larger maximal transfer (up to 10 % of source mass) and a slower
    temperature decay are used.
    """
    start = time.time()
    rng = np.random.default_rng(seed)
    a = np.asarray(init_seq, dtype=float)
    if a.sum() < 1e-12:
        a = np.ones_like(a) / a.size
    else:
        a = a / a.sum()
    n = a.shape[0]

    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())
    best_seq = a.copy()

    # temperature schedule
    T = best_obj * 0.2 + 1e-6
    T_min = best_obj * 1e-6 + 1e-9
    decay = 0.996
    it = 0

    while time.time() - start < time_limit:
        it += 1
        k_max = int(np.argmax(conv))

        # contributions a[i]·a[k_max‑i] (zero outside the range)
        i_vals = np.arange(n)
        j_vals = k_max - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i_vals[mask]] * a[j_vals[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            probs = contrib / total_contrib
            i = rng.choice(n, p=probs)
        else:
            i = rng.integers(n)

        # destination – random index different from i
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # transfer a modest fraction (up to 10 %)
        delta = a[i] * rng.random() * 0.10
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.clip(a_new, 0.0, None)
        a_new /= a_new.sum()

        conv_new = _conv_self_fft(a_new)
        obj_new = float(2.0 * n * conv_new.max())

        # Metropolis acceptance
        if obj_new < best_obj - 1e-12:
            a, conv, best_obj, best_seq = a_new, conv_new, obj_new, a_new.copy()
            print(f"[SA it={it}] new best O={best_obj:.6f}")
        else:
            prob = np.exp(-(obj_new - best_obj) / T) if T > 0 else 0.0
            if rng.random() < prob:
                a, conv = a_new, conv_new

        # temperature decay
        if it % 2500 == 0:
            T = max(T * decay, T_min)

        # occasional global jitter
        if it % 8000 == 0:
            noise = rng.normal(scale=0.01, size=n)
            a = np.clip(a + noise, 0.0, None)
            a /= a.sum()
            conv = _conv_self_fft(a)

    # final compression of tiny entries
    thresh = 1e-12
    mask = best_seq > thresh
    if mask.sum() < best_seq.size:
        comp = best_seq[mask]
        comp /= comp.sum()
        comp_obj = float(2.0 * comp.size * _conv_self_fft(comp).max())
        if comp_obj < best_obj:
            best_seq, best_obj = comp, comp_obj
            print(f"[SA compress] reduced length → {comp.size}, O={best_obj:.6f}")

    return best_seq.tolist(), best_obj


def _deterministic_local_search(
    init_seq: list[float],
    time_limit: float,
    seed: int = 0,
) -> tuple[list[float], float]:
    """
    Simple greedy hill‑climber.

    In each iteration the index that contributes most to the current
    maximal convolution entry is selected as a donor; a random
    destination receives a small fraction of its mass.  The move is kept
    iff it improves the objective.
    """
    start = time.time()
    rng = np.random.default_rng(seed)
    a = np.asarray(init_seq, dtype=float)
    a = np.clip(a, 0, None)
    if a.sum() < 1e-12:
        a = np.ones_like(a) / a.size
    else:
        a /= a.sum()
    n = a.shape[0]
    best_obj = _objective(a)

    while time.time() - start < time_limit:
        conv = _conv_self_fft(a)
        kmax = int(np.argmax(conv))

        # contribution of each entry to the current maximal convolution term
        ii = np.arange(n)
        jj = kmax - ii
        mask = (jj >= 0) & (jj < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[ii[mask]] * a[jj[mask]]
        total = contrib.sum()
        if total > 0:
            probs = contrib / total
            i = rng.choice(n, p=probs)
        else:
            i = rng.integers(n)

        # pick a different destination uniformly
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # transfer a fraction (1‑15 %)
        frac = rng.random() * 0.14 + 0.01
        delta = a[i] * frac
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.clip(a_new, 0.0, None)
        a_new /= a_new.sum()

        obj_new = _objective(a_new)
        if obj_new < best_obj - 1e-12:
            a, best_obj = a_new, obj_new
            # occasional log
            if rng.random() < 0.001:
                print(f"[Local] improvement to O={best_obj:.6f}")

    # final tiny‑value pruning
    thresh = 1e-12
    mask = a > thresh
    if mask.sum() < a.size:
        a = a[mask]
        a /= a.sum()
        best_obj = _objective(a)

    return a.tolist(), best_obj


def _upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear up‑sampling of a simplex‑normalised sequence.
    """
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    up = np.interp(x_new, x_old, seq)
    up = np.clip(up, 0.0, None)
    s = up.sum()
    if s < 1e-12:
        up = np.ones(new_n) / new_n
    else:
        up = up / s
    return up


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **_ignore,
) -> list[float]:
    """
    Return a sequence of non‑negative heights that (hopefully) improves the
    current upper bound.  The routine follows the three‑phase strategy
    described in the module docstring.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 1.5  # safety margin

    # ------------------------------------------------------------------
    # Load a previous construction if present.
    # ------------------------------------------------------------------
    init_seq = None
    if "height_sequence_1" in globals():
        seq0 = np.asarray(height_sequence_1, dtype=float)
        seq0 = np.clip(seq0, 0.0, None)
        if seq0.sum() < 0.01:
            seq0 = np.ones_like(seq0) / seq0.size
        else:
            seq0 = seq0 / seq0.sum()
        init_seq = seq0
        n0 = seq0.size
    else:
        n0 = None

    # ------------------------------------------------------------------
    # Define a hierarchy of support sizes.
    # Larger n gives more flexibility but costs more per iteration.
    # ------------------------------------------------------------------
    lengths = [2500, 6000, 12000]
    if n0 is not None and n0 not in lengths:
        lengths.insert(0, n0)

    best_seq = None
    best_obj = float("inf")

    # ------------------------------------------------------------------
    # 1) Coarse Adam optimisation on the hierarchy.
    # ------------------------------------------------------------------
    for idx, n in enumerate(lengths):
        now = time.time()
        remaining = deadline - now
        if remaining <= 5.0:
            break

        # allocate more time to later (larger) problems – exponential weights
        weights = [2 ** i for i in range(len(lengths))]
        prop = weights[idx] / sum(weights)
        time_for_this = remaining * prop * 0.9  # keep a small slack

        # warm‑start: up‑sample the best sequence seen so far, or use init_seq
        if idx == 0 and init_seq is not None and init_seq.size == n:
            init_a = init_seq
        elif idx > 0 and best_seq is not None:
            init_a = _upsample_sequence(np.asarray(best_seq, dtype=float), n)
        else:
            # random Dirichlet‑type start (concentration 0.6 gives a mildly sparse init)
            alpha = np.full(n, 0.6)
            init_a = rng.gamma(alpha, 1.0)
            init_a = init_a / init_a.sum()

        seq, val = _run_adam(
            n=n,
            time_limit=time_for_this,
            init_a=init_a,
            seed=seed + idx * 12345,
        )
        if val < best_obj:
            best_obj, best_seq = val, np.asarray(seq, dtype=float)
            print(f"[global] new overall best O={best_obj:.6f} (n={n})")

    # ------------------------------------------------------------------
    # 2) Refinement via simulated annealing (if time permits)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if best_seq is not None and remaining > 20.0:
        sa_time = remaining * 0.6
        seq_ref, val_ref = _run_sa(best_seq.tolist(), time_limit=sa_time, seed=seed + 777)
        if val_ref < best_obj:
            best_obj, best_seq = val_ref, np.asarray(seq_ref, dtype=float)
            print(f"[post‑SA] improved O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # 3) Deterministic greedy local search (final polishing)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if best_seq is not None and remaining > 5.0:
        loc_time = remaining * 0.8
        seq_loc, val_loc = _deterministic_local_search(
            best_seq.tolist(),
            time_limit=loc_time,
            seed=seed + 9999,
        )
        if val_loc < best_obj:
            best_obj, best_seq = val_loc, np.asarray(seq_loc, dtype=float)
            print(f"[final local] O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never happen but guards against empty result.
    # ------------------------------------------------------------------
    if best_seq is None or best_seq.size == 0:
        best_seq = np.ones(200, dtype=float) / 200.0

    # ------------------------------------------------------------------
    # Final sanity check (optional, prints the value that evaluate_sequence would report)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_seq.tolist()
