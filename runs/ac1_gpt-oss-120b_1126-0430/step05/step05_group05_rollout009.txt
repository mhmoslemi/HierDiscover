"""
Mirror‑Descent + Simulated‑Annealing optimizer for the step‑function objective

We minimise
    O(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm works in three phases:

1️⃣  **Mirror‑descent (exponentiated Adam) on a smooth log‑sum‑exp surrogate**.
    a is parametrised as a soft‑max of logits z (a_i = exp(z_i)/∑exp(z)).
    The surrogate is
        F(z) = (2·n/α)·log Σ_k exp(α·conv_k) + λ·∑ a_i·log a_i .
    α is increased exponentially from a small value to ~1e6,
    λ is annealed to 0.
    Adam updates are applied to the logits (the natural mirror‑descent
    step for the simplex).  This yields a fast reduction of the
    max‑convolution while staying on the probability simplex.

2️⃣  **Intensive local refinement** (mass‑transfer simulated‑annealing).
    Starting from the best mirror‑descent solution we repeatedly move a
    random fraction of probability mass from one index to another,
    accepting moves that improve O or with a Metropolis probability.
    Every few hundred steps a sub‑gradient step on the exact max‑convolution
    is performed to accelerate convergence.

3️⃣  **Final polishing** – a short run of the same mirror‑descent with a
    very large α (≈5·10⁵) and zero entropy to fine‑tune the remaining slack.

All phases respect the global 1000 s budget.  The routine returns the best
sequence found as a plain list of floats.
"""
import time
import math
import numpy as np

# --------------------------------------------------------------
# Euclidean projection onto the probability simplex (fallback)
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# --------------------------------------------------------------
# Full linear convolution via FFT (length = 2·n‑1)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]

# --------------------------------------------------------------
# Exact objective for a probability vector (∑a=1)
def _objective_simplex(a: np.ndarray, n: int, L: int) -> float:
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()

# --------------------------------------------------------------
# Sub‑gradient of max(conv) (used in the local refinement)
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g

# --------------------------------------------------------------
# Gradient of the smooth surrogate (log‑sum‑exp + entropy)
def _smooth_gradient_plus_entropy(
    a: np.ndarray,
    conv: np.ndarray,
    alpha: float,
    L: int,
    lam: float,
) -> np.ndarray:
    # soft‑max weights w_k ∝ exp(α·conv_k)
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)      # avoid overflow
    w = exp_scaled / exp_scaled.sum()             # shape (2n‑1,)

    # correlation w ⋆ a (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n_len = a.size
    grad_corr = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]

    # entropy part
    eps = 1e-20
    grad_ent = lam * (np.log(a + eps) + 1.0)

    return grad_corr + grad_ent

        # form probability vector from logits
        z_max = np.max(z)
        a = np.exp(z - z_max)
        a /= a.sum()

        # convolution & objective
        conv = _conv_fft(a, L)
        cur_O = 2.0 * n * conv.max()
        if cur_O < best_O:
            best_O = cur_O
            best_a = a.copy()
            print(f"[phase‑1] iter {it:,} new best O={best_O:.6f}")

        # schedule based on elapsed fraction of phase‑1
        frac = (time.time() - start_time) / (phase1_deadline - start_time)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        lam = lam_start * (1.0 - frac)   # linear decay to zero

        # gradient w.r.t. a of the smooth surrogate
        grad_a = _smooth_gradient_plus_entropy(a, conv, alpha, L, lam)

        # mirror‑descent gradient w.r.t. logits (softmax Jacobian)
        grad_dot = np.dot(a, grad_a)
        grad_z = a * (grad_a - grad_dot)

        # Adam update on logits
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 / math.sqrt(it)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape tiny plateaus
        if it % 1000 == 0 and np.random.rand() < 0.05:
            z += np.random.randn(n) * 1e-3

    # ----------------------------------------------------------
    # 3️⃣  Local refinement – simulated‑annealing + sub‑gradient nudges
    print(f"[phase‑2] local refinement, best O={best_O:.6f}")
    a = best_a.copy()
    conv = _conv_fft(a, L)
    cur_O = 2.0 * n * conv.max()

    # parameters for the stochastic search
    temperature0 = 0.08
    temp_decay = 0.9999
    max_iter_local = 300_000
    no_improve_limit = 2000
    restart_counter = 0

    while time.time() < deadline:
        # occasional restart around the current best
        if np.random.rand() < 0.3:
            a = best_a.copy()
            a += np.random.randn(n) * 0.001
            a = np.clip(a, 0, None)
            a /= a.sum()
            conv = _conv_fft(a, L)
            cur_O = 2.0 * n * conv.max()

        it_loc = 0
        no_improve = 0
        T = temperature0

        while time.time() < deadline and it_loc < max_iter_local:
            # ----- random mass‑transfer move -----
            i, j = np.random.randint(n, size=2)
            if i != j and a[i] > 1e-14:
                max_frac = 0.3
                frac = np.random.rand() * max_frac
                delta = a[i] * frac
                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta
                # (sum stays 1, non‑negativity guaranteed by the test above)
                conv_new = _conv_fft(a_new, L)
                new_O = 2.0 * n * conv_new.max()

                # Metropolis acceptance
                if new_O < cur_O or np.random.rand() < math.exp((cur_O - new_O) / max(T, 1e-12)):
                    a, conv, cur_O = a_new, conv_new, new_O
                    no_improve = 0
                    if cur_O < best_O:
                        best_O = cur_O
                        best_a = a.copy()
                        print(f"[{time.time():.1f}] new global best O={best_O:.6f} (restart={restart_counter}, iter={it_loc})")
                else:
                    no_improve += 1
            else:
                no_improve += 1

            # ----- occasional sub‑gradient descent (aggressive) -----
            if it_loc % 500 == 0:
                g = _subgradient_max_conv(a, conv)
                eta = 0.1 * (1.0 - it_loc / max_iter_local)
                a = a - eta * g
                a = np.clip(a, 0, None)
                a /= a.sum()
                conv = _conv_fft(a, L)
                cur_O = 2.0 * n * conv.max()
                if cur_O < best_O:
                    best_O = cur_O
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] sub‑grad improved O={best_O:.6f}")

            # temperature decay
            T *= temp_decay
            it_loc += 1

            if no_improve >= no_improve_limit:
                break

        restart_counter += 1
        # after a restart we keep a jittered copy of the current best
        a = best_a.copy()
        a += np.random.randn(n) * 5e-4
        a = np.clip(a, 0, None)
        a /= a.sum()
        conv = _conv_fft(a, L)
        cur_O = 2.0 * n * conv.max()

    # ----------------------------------------------------------
    # 4️⃣  Final polishing – very high α, zero entropy
    print(f"[phase‑3] final polishing, best O={best_O:.6f}")
    a = best_a.copy()
    z = np.log(a + eps)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    lr0 = 0.6
    alpha = 5e5            # extremely sharp surrogate → almost true max
    lam = 0.0
    polish_iters = 100_000
    it = 0
    while time.time() < deadline and it < polish_iters:
        # probability from logits
        z_max = np.max(z)
        a = np.exp(z - z_max)
        a /= a.sum()

        conv = _conv_fft(a, L)
        cur_O = 2.0 * n * conv.max()
        if cur_O < best_O:
            best_O = cur_O
            best_a = a.copy()
            print(f"[phase‑3] iter {it:,} new best O={best_O:.6f}")

        grad_a = _smooth_gradient_plus_entropy(a, conv, alpha, L, lam)
        grad_dot = np.dot(a, grad_a)
        grad_z = a * (grad_a - grad_dot)

        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 / math.sqrt(it)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

    # ----------------------------------------------------------
    # 5️⃣  Final sanity check with the provided evaluator
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:
        print("[final] evaluate_sequence failed:", e)

    return best_a.tolist()
