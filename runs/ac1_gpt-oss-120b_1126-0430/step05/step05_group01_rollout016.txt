#!/usr/bin/env python
"""
Non‑negative step‑function optimiser (enhanced).

The optimiser works on a modest support (≈ 5 000 points) so that a
large number of Adam‑style updates fit into the 1000 s budget.
It starts from a few hand‑crafted families (power‑law, symmetric
product‑power‑law, exponential, logistic) and keeps the best
distribution.  Adam with a soft‑max (mirror descent on the simplex)
is then run for many iterations, with a smoothly decreasing
ε‑smoothing of the max‑operator.  Periodic jitter, restarts and a
single up‑sampling stage (to ≈ 20 000 points) give the algorithm a
chance to escape plateaux and to refine the shape on a finer grid.
The best sequence seen is finally evaluated with the supplied
`evaluate_sequence` routine and returned.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures or lambdas
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max, returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    s = e.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / z.size)
    return e / s


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it that reproduces `a` after a soft‑max."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """Smallest power‑of‑two ≥ 3·n‑2 (enough for all needed convolutions)."""
    return 1 << ((3 * n - 2).bit_length())


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    s = new_a.sum()
    if s == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= s
    return new_a


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{‑p}, normalised."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _sym_product_power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ ((i+1)*(n‑i))^{‑p}, symmetric, normalised."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) * (n - i)) ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _exp_decay_distribution(n: int, lam: float) -> np.ndarray:
    """a_i ∝ e^{‑lam·i}+e^{‑lam·(n‑1‑i)}, symmetric exponential."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _logistic_distribution(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ 1/(1+e^{α·(i‑mid)}), normalised; heavy on the left side."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = 1.0 / (1.0 + np.exp(alpha * (i - mid)))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())

    """
    Returns (max_conv, grad_a) where

        grad_a = ∇_a [ 2·n·smooth_max(a*a) ]

    `eps_factor` multiplies the smoothing constant ε = eps_factor·max_conv.
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    # FFT of a (zero‑padded)
    a_fft = np.fft.rfft(a, L)

    # Linear self‑convolution a*a
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # Smoothed soft‑max weights
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv                # ≤ 0 for numerical stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # Correlation a ⋆ w
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # Gradient (already multiplied by 4·n)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) improves the
    `evaluate_sequence` score.  The routine respects the given time budget.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0   # safety margin for final eval

    # --------------------------------------------------------------
    # 0. Determine an initial support size and an initial distribution.
    # --------------------------------------------------------------
    # Target support size – small enough for many iterations, large enough
    # for a decent shape.  5 000 works well in practice.
    n = 5000
    best_a = None
    best_score = float("inf")

    # If a global sequence is available, down‑sample it.
    try:
        # ``height_sequence_1`` is supplied by the platform when available.
        init_global = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        best_a = _interpolate_distribution(init_global, n)
        best_score = 2.0 * n * _conv_max(best_a)
        print("[Init] Warm‑started from global sequence (down‑sampled).")
    except Exception:
        # No global – scan a few parametric families.
        print("[Init] Scanning parametric families.")
        # Power‑law
        for p in np.linspace(0.5, 1.5, 25):
            if time.time() >= deadline:
                break
            cand = _power_law_distribution(n, p)
            score = 2.0 * n * _conv_max(cand)
            if score < best_score:
                best_score, best_a = score, cand

        # Symmetric product‑power‑law
        for p in np.linspace(0.5, 1.5, 25):
            if time.time() >= deadline:
                break
            cand = _sym_product_power_law_distribution(n, p)
            score = 2.0 * n * _conv_max(cand)
            if score < best_score:
                best_score, best_a = score, cand

        # Exponential (symmetric)
        for loglam in np.linspace(-6, -2, 25):
            if time.time() >= deadline:
                break
            lam = 10.0 ** loglam
            cand = _exp_decay_distribution(n, lam)
            score = 2.0 * n * _conv_max(cand)
            if score < best_score:
                best_score, best_a = score, cand

        # Logistic (left‑heavy)
        for alpha in np.linspace(0.1, 5.0, 30):
            if time.time() >= deadline:
                break
            cand = _logistic_distribution(n, alpha)
            score = 2.0 * n * _conv_max(cand)
            if score < best_score:
                best_score, best_a = score, cand

        # Fallback – uniform
        if best_a is None:
            best_a = np.full(n, 1.0 / n)
            best_score = 2.0 * n * _conv_max(best_a)

    # Slight jitter before optimisation
    z = _logits_from_distribution(best_a) + 0.01 * np.random.randn(n)

    # --------------------------------------------------------------
    # 1. Adam state on logits (mirror descent on the simplex)
    # --------------------------------------------------------------
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    base_alpha = 0.45          # base learning‑rate for logits
    eps_factor = 0.15          # ε = eps_factor·max_conv (larger to start)
    eps_factor_min = 5e-5
    eps_factor_decay = 0.9992
    report_every = 10_000

    # --------------------------------------------------------------
    # 2. Optimisation loop
    # --------------------------------------------------------------
    best_seq = _softmax(z)
    best_score = 2.0 * n * _conv_max(best_seq)
    last_improve_iter = 0
    it = 0

    while time.time() < deadline and it < 3_500_000:
        it += 1

        # ----- Current point on the simplex -----
        a = _softmax(z)

        # ----- cheap evaluation + smoothed gradient -----
        max_conv, grad_a = _conv_max_and_grad(a, eps_factor)
        cheap_score = 2.0 * n * max_conv

        # ----- Keep best ever seen -----
        if cheap_score < best_score - 1e-12:
            best_score = cheap_score
            best_seq = a.copy()
            last_improve_iter = it
            if it % report_every == 0:
                print(f"[Iter {it:07d}] NEW BEST cheap = {cheap_score:.6f}")

        # ----- Gradient w.r.t. logits (mirror descent) -----
        mu = np.dot(grad_a, a)               # <grad, a>
        grad_z = a * (grad_a - mu)

        # ----- Adam moment updates (bias‑corrected) -----
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # ----- Step‑size schedule (inverse‑sqrt decay) -----
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step_dir = - m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- Back‑tracking line search (up to 6 halvings) -----
        accepted = False
        trial_alpha = alpha_t
        for _ in range(6):
            z_new = z + trial_alpha * step_dir
            # Clamp logits for numerical stability of exp()
            z_new = np.clip(z_new, -50.0, 50.0)
            a_new = _softmax(z_new)
            new_score = 2.0 * n * _conv_max(a_new)
            if new_score <= cheap_score:
                z = z_new
                accepted = True
                break
            trial_alpha *= 0.5

        if not accepted:
            # No improvement – take a modest step anyway.
            z += alpha_t * step_dir
            z = np.clip(z, -50.0, 50.0)

        # ----- Occasionally inject random jitter to shake plateaux -----
        if (it % 3000 == 0) and (np.random.rand() < 0.15):
            z += 0.02 * np.random.randn(n)
            m *= 0.6
            v *= 0.6

        # ----- Gradually reduce smoothing factor -----
        eps_factor = max(eps_factor_min, eps_factor * eps_factor_decay)

        # ----- Restart if we've been stuck for a very long time -----
        if it - last_improve_iter > 300_000:
            print(f"[Iter {it}] Stagnation detected – performing jitter restart.")
            z = _logits_from_distribution(best_seq) + 0.03 * np.random.randn(n)
            m.fill(0.0)
            v.fill(0.0)
            eps_factor = 0.15
            last_improve_iter = it

        # ----- One optional up‑sampling stage (once) -----
        if (it % 250_000 == 0) and (n < 20000) and (time.time() + 30.0 < deadline):
            # up‑sample to give the optimiser extra degrees of freedom
            new_n = min(20000, int(n * 1.6))
            print(f"[Iter {it}] Upsampling support: {n} → {new_n}")
            best_seq = _interpolate_distribution(best_seq, new_n)
            z = _logits_from_distribution(best_seq) + 0.005 * np.random.randn(new_n)
            m = np.zeros_like(z)
            v = np.zeros_like(z)
            n = new_n
            best_score = 2.0 * n * _conv_max(best_seq)
            # reset stagnation counter after a drastic change
            it = 0
            last_improve_iter = 0
            eps_factor = 0.15
            # continue with the enlarged problem size

    # --------------------------------------------------------------
    # 3. Final exact evaluation with the provided evaluate_sequence()
    # --------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # --------------------------------------------------------------
    # 4. Return the best sequence discovered.
    # --------------------------------------------------------------
    return final_seq
