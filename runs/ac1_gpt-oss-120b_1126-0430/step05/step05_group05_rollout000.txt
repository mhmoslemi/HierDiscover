"""
Propose a non‑negative step‑function sequence that (approximately) minimises

    O(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm consists of four stages:

1️⃣  **Initialisation** – if a previous construction (height_sequence_1) is
    available it is normalised; otherwise a uniform distribution of the
    chosen length `n` is used.  A few deterministic shapes (exponential,
    power‑law, cosine‑square) are examined as cheap baselines.

2️⃣  **Broad random search** – many Dirichlet samples with a
    log‑uniform concentration parameter are generated.  The exact objective
    (with ∑a=1) is evaluated via an FFT‑based convolution and the best
    sequence is kept.

3️⃣  **High‑precision smooth optimisation** – starting from the best
    candidate we run Adam on the smooth log‑sum‑exp surrogate of the
    max‑convolution.  The temperature `α` is annealed from 1 e3 to 2 e5;
    the entropy term is omitted after the random search.  The iterate is
    projected onto the probability simplex after each step.

4️⃣  **Local simulated‑annealing + sub‑gradient polishing** – a few
    thousand Metropolis moves (mass transfer between two indices) are
    performed, interleaved with sub‑gradient steps of the exact objective.
    Finally an exponentiated‑gradient sub‑gradient descent (multiplicative
    updates on the simplex) runs for the remaining time budget.

The routine respects the 1000 s budget, prints progress, and finally
returns the best sequence as a plain list of ``float``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution via FFT (length 2·n‑1)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return the full a⨂a convolution (length 2·n‑1) using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) (used for the exact objective)
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. a (averaged over all maximisers)."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)          # indices achieving the max
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
# Gradient of the smooth surrogate (log‑sum‑exp) plus optional entropy term
def _smooth_gradient_plus_entropy(
    a: np.ndarray,
    conv: np.ndarray,
    alpha: float,
    L: int,
    lam: float,
) -> np.ndarray:
    """
    Returns ∇_a ( (2·n/α)·log∑_k e^{α·conv_k} + λ·∑ a_i·log a_i ).
    The first part reduces to 2·corr(w,a) with w_k ∝ e^{α·conv_k};
    the second part contributes λ·(log a_i + 1).
    """
    # ---- soft‑max weights ----------
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)          # numerical stability
    w = exp_scaled / exp_scaled.sum()                 # shape (2n‑1,)

    # ---- correlation w ⋆ a (g_i = 2·∑_j w_{i+j} a_j) ----
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n_len = a.size
    grad_corr = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]

    # ---- entropy part ----------
    eps = 1e-20
    grad_ent = lam * (np.log(a + eps) + 1.0)

    return grad_corr + grad_ent

def _gen_exponential(n: int, decay: float) -> np.ndarray:

    x = np.arange(n, dtype=float)
    a = np.cos(np.pi * x / (n - 1)) ** 2
    a /= a.sum()
    return a

    # --------------------------------------------------------------
    # Stage 3 – Simulated annealing + occasional sub‑gradient steps
    a = best_seq.copy()
    conv = _conv_fft(a, L)
    cur_obj = 2.0 * n * conv.max()
    best_local_seq = a.copy()
    best_local_score = cur_obj

    T = 0.05                     # initial temperature
    decay = 0.9999
    max_sa_iter = 300_000
    it_sa = 0

    while time.time() < deadline and it_sa < max_sa_iter:
        i, j = np.random.randint(n, size=2)
        if i == j or a[i] < 1e-14:
            it_sa += 1
            continue

        max_frac = 0.5 * (1.0 - it_sa / max_sa_iter) + 0.02
        frac = np.random.rand() * max_frac
        delta = a[i] * frac

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = _project_to_simplex(a_new)

        conv_new = _conv_fft(a_new, L)
        new_obj = 2.0 * n * conv_new.max()

        # Metropolis acceptance
        if new_obj < cur_obj or np.random.rand() < math.exp((cur_obj - new_obj) / max(T, 1e-12)):
            a, conv, cur_obj = a_new, conv_new, new_obj
            if cur_obj < best_local_score:
                best_local_score = cur_obj
                best_local_seq = a.copy()
                print(f"[SA] iter {it_sa:,} new best O={best_local_score:.6f}")

        # occasional sub‑gradient descent (aggressive)
        if it_sa % 2000 == 0:
            g = _subgradient_max_conv(a, conv)
            eta = 0.12 * (1.0 - it_sa / max_sa_iter)
            a = a - eta * g
            a = _project_to_simplex(a)
            conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_local_score:
                best_local_score = cur_obj
                best_local_seq = a.copy()
                print(f"[SA] sub‑grad improved O={best_local_score:.6f}")

        T *= decay
        it_sa += 1

        # keep global best up‑to‑date
        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local_seq.copy()

    # --------------------------------------------------------------
    # Stage 4 – Exponentiated‑gradient sub‑gradient polishing
    a = best_seq.copy()
    remaining = deadline - time.time()
    steps = int(min(30_000, remaining * 800))   # aim for ~800 updates per second
    for t in range(steps):
        conv = _conv_fft(a, L)
        cur = 2.0 * n * conv.max()
        if cur < best_score:
            best_score = cur
            best_seq = a.copy()
            if t % 2000 == 0:
                print(f"[polish] step {t:,} new best O={best_score:.6f}")

        g = _subgradient_max_conv(a, conv)
        eta = 0.25 / math.sqrt(t + 1.0)
        a = a * np.exp(-eta * g)               # multiplicative EG update
        a = _project_to_simplex(a)

    # --------------------------------------------------------------
    # Final sanity check with the provided evaluator
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:
        print("[final] evaluate_sequence failed:", e)

    return best_seq.tolist()
