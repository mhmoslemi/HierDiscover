import time
import numpy as np

def _next_pow2(x):
    return 1 << (x - 1).bit_length()

def _conv_self_fft(a):
    n = a.shape[0]
    L = _next_pow2(2*n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)
    return conv[:2*n - 1]

def _correlation_fft(w, a):
    n = a.shape[0]
    m = w.shape[0]  # = 2n-1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]

def _softmax(x):
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / np.sum(e)

The goal is to find a non‑negative vector `a` (the step heights) that minimises  

\[
\text{score}(a)=\frac{2\,n\;\max\bigl(a*a\bigr)}{\bigl(\sum_i a_i\bigr)^2}
\]

where `*` denotes discrete linear convolution.  
Because the score is homogeneous of degree 0 we may normalise the vector to
`sum(a)=1` and simply minimise  

\[
F(a)=2\,n\;\max\bigl(a*a\bigr)\;.
\]

The approach combines three ideas:

1. **Smooth‑max Adam optimisation** – we parametrize the simplex with logits
   (`x → softmax(x)`) and minimise a smooth approximation of the max.
   The smoothing parameter β is annealed from a gentle value to a very large
   one (≈ 2·10⁴) so that the optimiser eventually follows the exact max‑gradient.

2. **Aggressive schedule & restarts** – learning‑rate and β follow exponential
   schedules, jitter is added every few thousand steps and a restart (with a
   small random perturbation) occurs every 25 000 iterations.  This prevents
   stagnation in plateaus.

3. **Final stochastic local refinement** – after the Adam run a cheap
   random‑mass‑transfer local search is performed; it moves a tiny fraction of
   mass from one index to another whenever the objective improves.

The optimiser is run for several lengths (powers of two, plus the length of the
previous best sequence if it is available).  Most of the time budget is given to
the largest length; the shorter lengths act as quick “explorers”.  The best
candidate found over all runs is returned.

The code respects all security checks of `evaluate_sequence`, prints progress,
and guarantees a non‑empty return.  No external files or networks are used.

"""
Multi‑length Adam optimiser with aggressive annealing and a final random
local‑search step.  The routine works on the simplex via a soft‑max
parameterisation, uses a smoothed max of the self‑convolution, adds periodic
jitter and restarts, and finishes with a cheap stochastic mass‑transfer
refinement.  The best sequence found before the time budget expires is
returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear self‑convolution a*a using a real FFT.
    Returns a vector of length 2·len(a)‑1.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation of w (length 2·n‑1) with a (length n):
    g[i] = Σ_t w[i+t]·a[t],   0 ≤ i < n.
    Implemented via FFT.
    """
    n = a.shape[0]
    m = w.shape[0]                # = 2·n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


# ----------------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a normalised to sum(a)=1."""
    n = a.size
    conv = _conv_self_fft(a)
    return 2.0 * n * np.max(conv)

def _run_opt_single(
    n: int,

    seed: int = 0,

    lr_end: float = 0.001,
    jitter_interval: int = 2000,
    jitter_scale: float = 0.015,
    restart_interval: int = 25000,
) -> tuple[list[float], float]:
    """
    Adam optimiser for a single support size `n`.
    Returns the best sequence (as a plain list) and its exact objective.
    """
    start = time.time()

    # ----- initialisation ------------------------------------------------
    if init_seq is not None:
        a = np.asarray(init_seq, dtype=float)

        a = a / a.sum()
    else:

        a = a / a.sum()

    # work in log‑it space (automatically respects the simplex)
    x = np.log(a + 1e-16)

    m = np.zeros_like(x)           # Adam 1st moment

    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # ----- optimisation loop --------------------------------------------
    best_a = a.copy()

    remaining = time_limit - (time.time() - start)
    if remaining > 0.5:
        a_loc = best_a.copy()
        obj_loc = best_obj
        n_loc = a_loc.size
        local_deadline = time.time() + remaining * 0.35
        while time.time() < local_deadline:
            i, j = rng.choice(n_loc, size=2, replace=False)
            if a_loc[i] <= 0.0:
                continue
            # move a tiny random fraction of the mass from i to j
            delta = a_loc[i] * rng.random() * 0.02
            a_new = a_loc.copy()
            a_new[i] -= delta
            a_new[j] += delta
            conv_new = _conv_self_fft(a_new)
            obj_new = 2.0 * n_loc * conv_new.max()
            if obj_new < obj_loc:
                a_loc = a_new
                obj_loc = obj_new
                if obj_new < best_obj:
                    best_a = a_new
                    best_obj = obj_new
        print(f"[local] final O={best_obj:.6f}")

    return best_a.tolist(), float(best_obj)

            lr_start=0.2,
            lr_end=0.001,
            jitter_interval=2000,
            jitter_scale=0.015,
            restart_interval=25000,
        )
        if val < best_obj:
            best_obj = val
            best_seq = seq
            print(f"[global] new overall best O={best_obj:.6f} (n={n})")

    # ----- final short random local search on the overall best ----------
    if best_seq is not None:
        remaining = deadline - time.time()
        if remaining > 1.0:
            a = np.array(best_seq, dtype=float)
            a = a / a.sum()
            best_obj = _objective(a)
            rng = np.random.default_rng(seed + 999)
            local_end = time.time() + remaining * 0.40
            while time.time() < local_end:
                i, j = rng.choice(a.size, size=2, replace=False)
                if a[i] <= 1e-12:
                    continue
                delta = a[i] * rng.random() * 0.03
                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta
                obj_new = _objective(a_new)
                if obj_new < best_obj:
                    a = a_new
                    best_obj = obj_new
            best_seq = a.tolist()
            print(f"[final local] O={best_obj:.6f}")

    # ----- final sanity check via the provided evaluation routine -------
    try:
        final_score = evaluate_sequence(best_seq)
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    # guarantee a non‑empty return
    if not best_seq:
        best_seq = [1.0]

    return best_seq
