"""
Propose a step‑function sequence that minimizes
    2·n·max(conv(sequence,sequence)) / (sum(sequence)^2)

The algorithm:
  1.  Determine a few support sizes (including the previous best length if
      available) and run several independent Adam optimisations for each size.
      The optimisation works in the log‑softmax (log‑it) parametrisation,
      uses a smoothed log‑sum‑exp approximation of the convolution maximum,
      and periodically adds jitter and restarts from the current best.
  2.  After each Adam run we prune negligible entries, re‑normalise and keep
      the best among all lengths.
  3.  Finally we perform a long, pure hill‑climbing local search that moves a
      small amount of mass from one index to another whenever the exact
      objective improves.  The move magnitude slowly decreases, allowing a
      fine‑grained polishing of the solution.
  4.  The resulting sequence is returned; it is guaranteed to be a non‑empty
      list of non‑negative floats and passes the security checks of
      ``evaluate_sequence``.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv


def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation of w with a (g_i = Σ_t w[i+t]·a[t]), 0 ≤ i < len(a).
    Implemented via FFT.
    """
    n = a.shape[0]
    m = w.shape[0]                 # = 2·n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)), assuming ``a`` is already normalised
    to sum = 1 (the scaling cancels in the true evaluation function).
    """
    n = a.size
    conv = _conv_self_fft(a)
    return 2.0 * n * conv.max()


def _run_opt(n: int,
             time_limit: float,
             init_a: np.ndarray | None = None,
             seed: int = 0) -> tuple[list[float], float]:
    """
    Adam optimisation for a fixed support size ``n``.  Returns the best
    sequence (as a plain list) and its exact objective value.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # ---- initialise -------------------------------------------------
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
        a = a / a.sum()
    else:
        a = rng.random(n)
        a = a / a.sum()

    # work in log‑it space (automatically respects the simplex)
    x = np.log(a + 1e-16)

    m = np.zeros_like(x)          # Adam first moment
    v = np.zeros_like(x)          # Adam second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # ---- schedules --------------------------------------------------
    beta_start = 0.5               # start of smooth‑max temperature
    beta_end   = 4000.0            # final (sharp) temperature
    lr_start   = 0.15
    lr_end     = 0.001
    target_iters = 800_000         # theoretical budget; we stop by time

    best_a = a.copy()
    best_obj = _objective(best_a)

    it = 0
    last_improve = 0
    while time.time() - start < time_limit:
        it += 1

        # probabilities from current logits
        a = _softmax(x)

        # exact objective
        conv = _conv_self_fft(a)
        M = conv.max()
        obj = 2.0 * n * M
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            last_improve = it
            print(f"[n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        # ---- schedules ------------------------------------------------
        t_frac = min(it / target_iters, 1.0)
        beta = beta_start * np.exp(np.log(beta_end / beta_start) * t_frac)
        lr   = lr_start   * np.exp(np.log(lr_end   / lr_start)   * t_frac)

        # ---- smooth‑max weights ---------------------------------------
        if beta > 3000.0:                     # effectively a hard max
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            shifted = conv - M                 # ≤ 0
            w = np.exp(beta * shifted)
            w = w / w.sum()

        # ---- gradient (in a‑space) ------------------------------------
        g_a = 2.0 * _correlation_fft(w, a)   # shape (n,)

        # natural gradient in log‑it space (projected onto simplex)
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)

        # ---- Adam update -----------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---- jitter (occasionally) ------------------------------------
        if it % 6000 == 0:
            jitter = rng.normal(scale=0.02, size=n)
            x += jitter

        # ---- restart if stalled ---------------------------------------
        if it % 70_000 == 0:
            # restart from the current best with a tiny perturbation
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.005, size=n)
            m.fill(0.0)
            v.fill(0.0)
            print(f"[n={n}] restart at iter {it}")

        # ---- early exit if no improvement for a long time ------------
        if it - last_improve > 200_000 and time.time() - start > 0.2 * time_limit:
            # no improvement for a while and we have spent a good chunk of time
            break

    # ---- compress negligible entries ----------------------------------
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        n_new = best_a.size
        best_obj = _objective(best_a)
        print(f"[compress] length {n} → {n_new}, O={best_obj:.6f}")

    # ---- short stochastic local search --------------------------------
    remaining = time_limit - (time.time() - start)
    if remaining > 0.5:
        a_loc = best_a.copy()
        obj_loc = best_obj
        n_loc = a_loc.size
        local_deadline = time.time() + remaining * 0.35
        while time.time() < local_deadline:
            i, j = rng.choice(n_loc, size=2, replace=False)
            # move a random up‑to‑2 % of mass from i to j
            delta = a_loc[i] * rng.random() * 0.02
            if delta <= 0.0:
                continue
            a_new = a_loc.copy()
            a_new[i] -= delta
            a_new[j] += delta
            conv_new = _conv_self_fft(a_new)
            obj_new = 2.0 * n_loc * conv_new.max()
            if obj_new < obj_loc:
                a_loc = a_new
                obj_loc = obj_new
                if obj_new < best_obj:
                    best_a = a_new
                    best_obj = obj_new
        print(f"[local] final local O={best_obj:.6f}")

    return best_a.tolist(), best_obj


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **_ignore) -> list[float]:
    """
    Entry point for the competition harness.
    Runs several Adam optimisations for a ladder of lengths, keeps the best
    sequence and finishes with a thorough hill‑climbing refinement.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # -------------------------------------------------------------
    # 1) starting point (previous best, if any)
    base_seq = None
    if "height_sequence_1" in globals():
        base_seq = np.asarray(height_sequence_1, dtype=float).copy()
        base_seq = np.clip(base_seq, 0.0, None)
        if base_seq.sum() < 0.01:
            base_seq = np.ones_like(base_seq)
        base_seq = base_seq / base_seq.sum()

    # -------------------------------------------------------------
    # 2) decide which support sizes to explore
    lengths = [1024, 2048, 4096, 8192]
    if base_seq is not None:
        n0 = base_seq.size
        if n0 not in lengths:
            lengths.insert(0, n0)
    # keep lengths reasonable
    lengths = [l for l in lengths if l <= 20000]
    # sort ascending – quick tests first
    lengths.sort()

    best_seq = None
    best_obj = float("inf")

    # -------------------------------------------------------------
    # 3) run Adam optimiser for each length
    for idx, n in enumerate(lengths):
        remaining = deadline - time.time()
        if remaining < 5.0:
            break

        # allocate time: give the largest length most of the remaining budget
        if n == max(lengths):
            run_time = remaining * 0.70
        else:
            # split the rest equally among the still‑to‑run sizes
            run_time = (remaining - 0.5) / (len(lengths) - idx)

        # number of independent starts for this size
        starts = 2 if n < max(lengths) else 3

        for s in range(starts):
            init = None
            if idx == 0 and base_seq is not None:
                # initialise from the previous construction, padded / truncated if needed
                if base_seq.size != n:
                    if base_seq.size > n:
                        init = base_seq[:n]
                    else:
                        pad = np.zeros(n - base_seq.size)
                        init = np.concatenate([base_seq, pad])
                    init = init / init.sum()
                else:
                    init = base_seq

            seq, val = _run_opt(n=n,
                                time_limit=run_time / starts,
                                init_a=init,
                                seed=seed + idx * 13 + s * 17)
            if val < best_obj:
                best_obj = val
                best_seq = seq
                print(f"[global] new overall best O={best_obj:.6f} (n={n})")

    # -------------------------------------------------------------
    # 4) final intensive hill‑climbing local search
    if best_seq is not None:
        remaining = deadline - time.time()
        if remaining > 2.0:
            a = np.array(best_seq, dtype=float)
            n = a.size
            best_val = _objective(a)
            rng = np.random.default_rng(seed + 9999)

            # schedule of maximum transfer proportion (coarse → fine)
            transfer_schedule = [(0.20, 0.30), (0.10, 0.30), (0.05, 0.20), (0.02, 0.20)]

            end_time = time.time() + remaining * 0.95
            while time.time() < end_time:
                # pick a transfer size according to elapsed proportion
                frac = (time.time() - (deadline - remaining)) / remaining
                # simple linear selection over the schedule
                if frac < 0.2:
                    max_transfer = 0.20
                elif frac < 0.5:
                    max_transfer = 0.10
                elif frac < 0.8:
                    max_transfer = 0.05
                else:
                    max_transfer = 0.02

                i = rng.choice(n, p=a)        # bias to larger entries
                j = rng.integers(n)
                if i == j:
                    continue
                delta = a[i] * rng.random() * max_transfer
                if delta <= 0.0:
                    continue
                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta
                obj_new = _objective(a_new)
                if obj_new < best_val:
                    a = a_new
                    best_val = obj_new
                    best_seq = a.tolist()
            print(f"[final local] polished O={best_val:.6f}")

    # -------------------------------------------------------------
    # 5) final sanity check
    try:
        final_score = evaluate_sequence(best_seq)
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    # guarantee a non‑empty return
    if not best_seq:
        best_seq = [1.0]

    return best_seq
