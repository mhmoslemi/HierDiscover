"""
Improved step‑function optimizer.

The algorithm works on the simplex (non‑negative entries that sum to 1).

1.  It runs a fast exponentiated‑gradient descent that uses the **exact**
    sub‑gradient of the max‑convolution (no smoothing, only one FFT per
    iteration).  The update is multiplicative, therefore the simplex
    constraints are satisfied automatically.

2.  For moderate support sizes (≤ 1800) an LP‑refinement is performed:
    maximise  Σ b_j  subject to  f∗b ≤ M   (M – current max‑convolution).
    The resulting vector is mixed back into the current solution.

3.  A cheap random “mass‑transfer’’ local search finishes each run.

Several support lengths are tried (the previous best is reused as a warm
start).  The whole search respects the 1000 s budget and returns the
best sequence found.

All helper functions are top‑level and hide no closures; the code only
uses numpy and scipy.optimize.linprog (available in the execution
environment).
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))."""
    n = a.size
    conv = _conv_self_fft(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
def _grad_max_conv(a: np.ndarray):
    """
    Returns the sub‑gradient of the max‑convolution at the current argmax,
    together with the convolution array and the index of the argmax.
    """
    n = a.shape[0]
    conv = _conv_self_fft(a)
    idx = int(np.argmax(conv))                # position in 0 … 2n‑2
    shift = idx - (n - 1)                     # actual displacement (can be negative)

    grad = np.zeros_like(a)
    if shift >= 0:
        # conv[k] = Σ a[i] a[i+shift]; derivative w.r.t a[i] is 2·a[i+shift]
        grad[: n - shift] = 2.0 * a[shift:]
    else:
        # shift is negative → use the symmetric part
        grad[-shift : ] = 2.0 * a[: n + shift]

    return grad, conv, idx


# ----------------------------------------------------------------------
def _exp_grad_descent(a0, time_limit, seed):
    """
    Exponentiated‑gradient descent on the simplex.
    Returns the best vector (as a list) and its exact objective.
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    a = np.asarray(a0, dtype=float)
    a /= a.sum()               # safety normalisation
    n = a.shape[0]

    best_a = a.copy()
    best_obj = _objective(best_a)

    it = 0
    eta0 = 0.35                 # initial step size (tuned empirically)

    while time.time() - start < time_limit:
        it += 1

        # 1 – gradient at the current max‑convolution
        grad, conv, idx = _grad_max_conv(a)

        # 2 – step‑size schedule (sqrt‑decay)
        eta = eta0 / np.sqrt(it)

        # 3 – multiplicative update, then re‑normalise
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s <= 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # 4 – occasional jitter to escape plateaus
        if it % 6000 == 0:
            jitter = rng.normal(scale=0.02, size=n)
            a = a * np.exp(jitter)
            a /= a.sum()

        # 5 – evaluate objective (once per iteration)
        obj = float(2.0 * n * conv.max())
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[expGD n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

    return best_a.tolist(), best_obj


# ----------------------------------------------------------------------
def _lp_refine(a, M):
    """
    Linear‑programming refinement: maximise Σ b_j subject to f∗b ≤ M.
    Returns an improved vector (mixed with the original) or the original
    if the LP fails.
    """
    n = a.shape[0]
    L = 2 * n - 1

    # Build the (L × n) convolution matrix; column j is a shifted by +j
    A = np.zeros((L, n), dtype=float)
    for j in range(n):
        A[j : j + n, j] = a

    c = -np.ones(n)                     # maximise Σ b_j  → minimise -Σ b_j
    bounds = [(0.0, None)] * n
    b_ub = M * np.ones(L)

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs")
    if not res.success:
        return a

    b = res.x
    sb = b.sum()
    if sb <= 1e-12:
        return a

    g = b / sb                         # normalised partner
    mu = 0.20                          # mixing weight (tuned)
    a_new = (1.0 - mu) * a + mu * g
    a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
def _local_search(a0, time_limit, seed):
    """
    Simple stochastic local search: repeatedly move a random fraction
    of mass from one index to another and keep improvements.
    """
    rng = np.random.default_rng(seed)
    a = np.asarray(a0, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_obj = _objective(best_a)
    end = time.time() + time_limit

    while time.time() < end:
        i, j = rng.choice(n, size=2, replace=False)

        # transfer ≤ 2 % of the mass at i
        delta = best_a[i] * rng.random() * 0.02

        a_new = best_a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        obj = _objective(a_new)
        if obj < best_obj:
            best_a = a_new
            best_obj = obj

    return best_a.tolist(), best_obj
