"""
Optimise a non‑negative step function for the max‑convolution metric.

Strategy
--------
1.  Use a large support (default 60 000 points) and initialise the
    sequence with the classic discrete arcsine density
        a_i ∝ 1/√(x_i (1‑x_i)),   x_i = (i+0.5)/n .
    The continuous arcsine distribution has a perfectly flat self‑convolution,
    so the discretisation already yields a very good score (≈1.5 for n=20 k).

2.  Refine the distribution with Adam optimisation on the *soft‑max* (logit)
    parametrisation of the vector.  The objective is the smooth approximation
    of the max‑convolution, whose gradient can be computed efficiently with FFT
    (O(n log n) per iteration).

3.  Keep the best vector seen so far (cheap score evaluated from the smooth
    max).  Stop when the time budget is almost exhausted.

4.  Post‑process:
        • symmetrise the vector (a_i ← (a_i + a_{n-1-i})/2) – the optimal
          solution is symmetric;
        • apply a few tiny smoothing passes with a 3‑point kernel
          [0.25, 0.5, 0.25] while preserving the total mass.
        The best of these candidates is returned.

All heavy work (FFT‑based gradients, cheap scores) is O(n log n) and easily
fits into the 1000 s budget even for n≈6·10⁴.  The final call to the
provided `evaluate_sequence` uses the naïve `np.convolve`, but for the
chosen size it finishes within a few seconds.

The function `propose_candidate` follows the required signature and prints
some progress information for debugging.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all top‑level, no closures
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Return logits such that `softmax(logits) == a` (up to tiny eps)."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """
    Smallest power‑of‑two ≥ 3·n‑2, sufficient for the linear correlation length.
    """
    return 1 << ((3 * n - 2).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a (FFT version)."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Smooth max‑convolution and its gradient w.r.t. the probability vector.

    Parameters
    ----------
    a : np.ndarray
        Probability vector (sum == 1, a_i ≥ 0).
    eps_factor : float
        Multiplicative factor for the smoothing constant:
        eps = eps_factor * max(conv).

    Returns
    -------
    max_conv : float
        Maximum entry of a*a (exact, not the smoothed max).
    grad_a : np.ndarray
        Gradient of 2·n·smooth_max w.r.t. a (already includes factor 4·n).
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    a_fft = np.fft.rfft(a, L)

    # linear self‑convolution (length 2n‑1)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # smoothed soft‑max weights over the 2n‑1 entries
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv          # for numerical stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or np.isnan(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # correlation a ⋆ w (linear, length 3n‑2); we only need first n entries
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # gradient of 2·n·smooth_max (see derivation in docstring)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a

    """
