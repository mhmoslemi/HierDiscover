def _subgrad_max(a, conv, L):
    n = a.size
    max_val = conv.max()
    mask = (conv == max_val).astype(float)
    # Normalize to average weight
    mask /= mask.sum()
    # correlation of mask with a: corr = irfft(rfft(mask) * rfft(a_rev))
    a_rev = a[::-1]
    fft_mask = np.fft.rfft(mask, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr = np.fft.irfft(fft_mask * fft_a_rev, n=L)
    start = n - 1
    grad = 2.0 * corr[start:start+n]
    return grad

"""
Propose a step‑function (non‑negative list) that approximately minimises

    O(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm consists of four stages:

1️⃣  **Flattening the convolution** – Adam optimisation of the variance of the
    full convolution vector.  This pushes the convolution entries towards a
    common value and reduces the peak early on.

2️⃣  **Smooth max‑approximation** – Adam on a log‑sum‑exp surrogate of the
    maximum (the parameter α is annealed from 5 to 20000).  A tiny entropy
    term keeps the iterates strictly inside the simplex.

3️⃣  **Deterministic sub‑gradient descent** on the true maximum.  The
    sub‑gradient of `max(conv)` is obtained by a correlation between the
    mask of the maximal entries and the current sequence.

4️⃣  **Fast simulated‑annealing mass‑transfer** – a specialised
    O(n) update for a convolution when a small amount of mass is moved
    from one coordinate to another.  The source index is chosen uniformly
    but the move size is a random fraction of the source mass.
    A geometric temperature schedule accepts occasional uphill moves.
    The convex‐projection is performed only occasionally to keep the
    sequence on the simplex.

All stages respect a global time budget (default 1000 s).  The best
sequence found at any time is returned.  If a previous construction
`height_sequence_1` is present it is used as a warm start, otherwise a
uniform distribution is used.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions ------------------------------------------------------
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x>=0, sum=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # all non‑positive – return uniform point
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution a★a (length 2·n‑1) using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


def _grad_variance(a: np.ndarray, conv: np.ndarray, n: int, L: int) -> np.ndarray:
    """
    Gradient of   f(a) = Σ_k (conv_k – μ)²   where μ = 1/(2n‑1)
    (derivation: g = 4·corr(conv‑μ , a)).
    """
    mu = 1.0 / (2 * n - 1)
    diff = conv - mu
    a_rev = a[::-1]
    fft_diff = np.fft.rfft(diff, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_diff * fft_a_rev, n=L)
    start = n - 1
    return 4.0 * corr_rev[start:start + n]


def _smooth_gradient_plus_entropy(
    a: np.ndarray,
    conv: np.ndarray,
    alpha: float,
    L: int,
    lam: float,
) -> np.ndarray:
    """
    Gradient of    (2·n/α)·log Σ_k exp(α·conv_k)  +  λ·Σ a_i·log a_i .
    The first term yields 2·corr(w , a) where w ∝ exp(α·conv).
    """
    # soft‑max weights for the surrogate max
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)          # shift for stability
    w = exp_scaled / exp_scaled.sum()                 # shape (2n‑1,)

    # correlation w * a
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n_len = a.size
    grad_corr = 2.0 * corr_rev[n_len - 1:n_len - 1 + n_len]

    # tiny entropy term (helps avoid exact zeros)
    eps = 1e-20
    grad_ent = lam * (np.log(a + eps) + 1.0)

    return grad_corr + grad_ent


def _subgrad_max(a: np.ndarray, conv: np.ndarray, L: int) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a (averaged over all maximal indices).
    Returns a vector of length n.
    """
    max_val = conv.max()
    mask = (conv == max_val).astype(float)
    # average over the possibly multiple maximal entries
    mask /= mask.sum()

    a_rev = a[::-1]
    fft_mask = np.fft.rfft(mask, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_mask * fft_a_rev, n=L)
    start = a.size - 1
    # factor 2 comes from ∂c_k/∂a_i = 2·a_{k‑i}
    return 2.0 * corr_rev[start:start + a.size]

    """

    deadline = start_time + budget_s - 0.2          # a tiny safety margin

    L = _next_pow2(2 * n - 1)

    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] initial O={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – variance flattening (Adam) -----------------------------------
    phase1_end = start_time + budget_s * 0.20
    it = 0
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    lr0 = 0.8
    lam_entropy = 1e-3

    while time.time() < phase1_end and it < 250_000:
        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[phase‑1] iter {it:,} new best O={best_score:.6f}")

        grad = _grad_variance(a, conv, n, L)
        grad += lam_entropy * (np.log(a + 1e-20) + 1.0)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        lr = lr0 / math.sqrt(it + 1)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_to_simplex(a_new)

        it += 1
        if it % 20000 == 0:
            print(f"[phase‑1] iter {it:,} O={cur_obj:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – smooth max surrogate (log‑sum‑exp) -------------------------
    phase2_end = start_time + budget_s * 0.55
    it = 0
    max_iter2 = 500_000
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    lr0 = 0.6
    lam = 2e-3
    alpha_start = 5.0
    alpha_end = 20000.0

    while time.time() < phase2_end and it < max_iter2:
        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[phase‑2] iter {it:,} new best O={best_score:.6f}")

        # anneal the soft‑max temperature
        frac = it / max_iter2
        alpha = alpha_start + (alpha_end - alpha_start) * frac

        grad = _smooth_gradient_plus_entropy(a, conv, alpha, L, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        lr = lr0 / math.sqrt(it + 1)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to shake out of plateaus
        if np.random.rand() < 0.001:

        it += 1

    # ------------------------------------------------------------------

    phase3_end = start_time + budget_s * 0.75

        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[phase‑3] iter {it:,} new best O={best_score:.6f}")

        # sub‑gradient of the max entry (averaged if several are tied)
        g = _subgrad_max(a, conv, L)

        # simple decreasing step size
        eta = 0.12 * (1.0 - it / max_iter3)
        if eta < 1e-6:
            eta = 1e-6

        a_new = a - eta * g
        a = _project_to_simplex(a_new)

        it += 1
        if it % 20000 == 0:
            print(f"[phase‑3] iter {it:,}")

    # ------------------------------------------------------------------
    # Phase 4 – fast simulated‑annealing mass‑transfer -----------------------
    #   (updates the convolution in O(n) using the analytic formula)
    it = 0
    T = 0.1
    decay = 0.99995
    conv = _conv_fft(a, L)                     # keep current conv in sync
    cur_obj = 2.0 * n * conv.max()

    # helper to recompute conv occasionally (avoids drift)
    def recompute_conv(x):
        return _conv_fft(x, L)

    while time.time() < deadline:
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j or a[i] < 1e-14:
            continue

        # move a random fraction (up to 20 %) of the source mass
        delta = a[i] * np.random.rand() * 0.20

        a_old = a.copy()
        conv_old = conv.copy()

        a[i] -= delta
        a[j] += delta

        # Update convolution analytically:
        #   conv ← conv + 2·Δ·(a_before shifted by j – a_before shifted by i)
        # The shift of a_before by an offset p is the vector a_before placed
        # starting at index p in a length‑2n‑1 zero array.
        a_before = a_old
        conv[i:i + n] -= 2.0 * delta * a_before
        conv[j:j + n] += 2.0 * delta * a_before

        if new_obj < cur_obj or np.random.rand() < math.exp((cur_obj - new_obj) / max(T, 1e-14)):

                best_seq = a.copy()
                print(f"[SA] iter {it:,} new global best O={best_score:.6f}")

            a = a_old
            conv = conv_old

            conv = recompute_conv(a)
            cur_obj = 2.0 * n * conv.max()
            print(f"[SA] restart at iter {it:,}, O={cur_obj:.6f}")

        it += 1

    # Final sanity check and return -----------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[final] evaluate_sequence reports O={final_score:.6f}")

    return best_seq.tolist()
