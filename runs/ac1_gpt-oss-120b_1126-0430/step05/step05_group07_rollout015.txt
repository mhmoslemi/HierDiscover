"""
Propose a non‑negative step‑function (list of heights) that minimises

    O(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm consists of four phases:

1️⃣  **Variance flattening** – Adam optimisation of the variance of the
    full convolution `c = a★a`.  This pushes the whole convolution
    towards a constant value.

2️⃣  **Smooth max surrogate** – Adam on a log‑sum‑exp approximation of
    `max(c)` (soft‑max).  The temperature `α` is annealed from a small
    value to a large one, gradually focusing the optimisation on the
    true peak.

3️⃣  **Sub‑gradient descent on the true max** – Adam updates using the
    exact sub‑gradient of `max(c)`.  This directly reduces the
    objective `2·n·max(c)`.

4️⃣  **Guided stochastic mass‑transfer** – A Metropolis‐style Monte‑Carlo
    loop that moves mass from indices that strongly increase the peak
    (large sub‑gradient) to indices that have little influence.
    The temperature is slowly annealed.

The best sequence seen throughout the whole run is returned.
All updates stay on the probability simplex (non‑negative, sum = 1),
so the objective is equivalent to minimising `max(conv)`.
"""

import time

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        # all entries zero → uniform vector
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution a★a (length 2·n‑1) using FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


def _objective_max(a: np.ndarray, n: int, L: int) -> float:
    """Exact objective (sum(a)=1): 2·n·max(conv(a,a))."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


def _grad_variance(a: np.ndarray, conv: np.ndarray, n: int, L: int) -> np.ndarray:
    """
    Gradient of   f(a) = Σ_k (conv_k - μ)²   where μ = 1/(2n‑1) (for sum(a)=1).
    Closed form: g = 4·corr(conv‑μ , a) .
    """
    mu = 1.0 / (2 * n - 1)
    diff = conv - mu                     # length 2n‑1
    a_rev = a[::-1]

    fft_diff = np.fft.rfft(diff, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_diff * fft_a_rev, n=L)

    start = n - 1
    grad = 4.0 * corr_rev[start:start + n]
    return grad


def _smooth_gradient_plus_entropy(
    a: np.ndarray,
    conv: np.ndarray,
    alpha: float,
    L: int,
    lam: float,
) -> np.ndarray:
    """
    Gradient of (2·n/α)·log∑_k e^{α·conv_k}  + λ·∑ a_i·log a_i .
    The first part yields 2·corr(w,a) with w ∝ exp(α·conv).
    """
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / exp_scaled.sum()                 # (2n‑1,)

    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n_len = a.size
    grad_corr = 2.0 * corr_rev[n_len - 1:n_len - 1 + n_len]

    eps = 1e-20
    grad_ent = lam * (np.log(a + eps) + 1.0)

    return grad_corr + grad_ent


def _subgrad_max(a: np.ndarray, conv: np.ndarray, L: int):
    """Sub‑gradient of max(conv) w.r.t. a (factor 2 already included)."""
    max_val = conv.max()
    mask = (conv == max_val).astype(float)            # (2n‑1,)
    a_rev = a[::-1]
    fft_mask = np.fft.rfft(mask, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_mask * fft_a_rev, n=L)
    start = a.size - 1
    grad = 2.0 * corr_rev[start:start + a.size]       # shape (n,)
    return grad, max_val

            best_score = cur_obj
            best_seq = a.copy()
            print(f"[phase‑1] iter {it:,}  new best O={best_score:.6f}")

        grad = _grad_variance(a, conv, n, L)
        grad += lam_entropy * (np.log(a + 1e-20) + 1.0)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        lr = lr0 / math.sqrt(it + 1)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = _project_to_simplex(a_new)
        a = a_new

        it += 1
        if it % 20000 == 0:
            print(f"[phase‑1] iter {it:,}  O={cur_obj:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – smooth‑max (log‑sum‑exp) surrogate (Adam)
    print("[phase‑2] smooth‑max surrogate")
    beta1, beta2 = 0.9, 0.999
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    lr0 = 0.5
    lam = 0.02
    alpha_start = 5.0
    alpha_end = 8000.0
    phase2_deadline = start_time + budget_s * 0.55
    it = 0
    max_iter_phase2 = 400_000   # hard cap

    while time.time() < phase2_deadline and it < max_iter_phase2:
        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[phase‑2] iter {it:,}  new best O={best_score:.6f}")

        frac = it / max_iter_phase2
        alpha = alpha_start + (alpha_end - alpha_start) * frac

        grad = _smooth_gradient_plus_entropy(a, conv, alpha, L, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        lr = lr0 / math.sqrt(it + 1)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter
        if np.random.rand() < 0.001:
            a_new += np.random.randn(n) * 1e-4

        a_new = _project_to_simplex(a_new)
        a = a_new
        it += 1
        if it % 25000 == 0:
            print(f"[phase‑2] iter {it:,}  O={cur_obj:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – sub‑gradient descent on the true max (Adam)
    print("[phase‑3] sub‑gradient descent on max(conv)")
    beta1, beta2 = 0.9, 0.999
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    lr0 = 0.5
    lam_entropy = 1e-5
    phase3_deadline = start_time + budget_s * 0.80
    it = 0
    max_iter_phase3 = 300_000

    while time.time() < phase3_deadline and it < max_iter_phase3:
        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[phase‑3] iter {it:,}  new best O={best_score:.6f}")

        # sub‑gradient of max(conv)
        grad, _ = _subgrad_max(a, conv, L)
        # tiny entropy to keep strict positivity
        grad += lam_entropy * (np.log(a + 1e-20) + 1.0)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        lr = lr0 / math.sqrt(it + 1)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = _project_to_simplex(a_new)
        a = a_new

        it += 1
        if it % 50000 == 0:
            print(f"[phase‑3] iter {it:,}  O={cur_obj:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – guided stochastic mass‑transfer (Metropolis)
    print("[phase‑4] guided stochastic refinement")
    temperature = 0.04
    it_total = 0
    no_improve = 0
    NO_IMPROVE_LIMIT = 5000

    while time.time() < deadline:
        # occasionally re‑jitter the current best to escape shallow basins
        if np.random.rand() < 0.05:
            a = best_seq.copy()
            a += np.random.randn(n) * 0.0010
            a = np.clip(a, 0, None)
            a = _project_to_simplex(a)

        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        grad, _ = _subgrad_max(a, conv, L)

        # probabilities for a source (large grad) and a destination (small grad)
        prob_src = grad / (grad.sum() + 1e-12)                     # favour high‑grad indices
        prob_dst = (grad.max() - grad + 1e-12)
        prob_dst = prob_dst / prob_dst.sum()                     # favour low‑grad indices

        i = np.random.choice(n, p=prob_src)
        j = np.random.choice(n, p=prob_dst)

        if a[i] > 1e-14:
            delta = a[i] * np.random.rand() * 0.15                # up to 15 % of source mass
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            a_new = _project_to_simplex(a_new)

            conv_new = _conv_fft(a_new, L)
            new_obj = 2.0 * n * conv_new.max()

            # Metropolis acceptance
            if new_obj < cur_obj or np.random.rand() < math.exp((cur_obj - new_obj) / max(temperature, 1e-16)):
                a, conv, cur_obj = a_new, conv_new, new_obj
                no_improve = 0
                if cur_obj < best_score:
                    best_score = cur_obj
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] new global best O={best_score:.6f} (iter={it_total})")
            else:
                no_improve += 1
        else:
            no_improve += 1

        # occasional jitter to keep exploring
        if np.random.rand() < 0.001:
            a += np.random.randn(n) * 1e-5
            a = np.clip(a, 0, None)
            a = _project_to_simplex(a)

        temperature *= 0.9998
        it_total += 1

        if no_improve >= NO_IMPROVE_LIMIT:
            # restart from the (jittered) best solution
            a = best_seq.copy()
            a += np.random.randn(n) * 0.0012
            a = np.clip(a, 0, None)
            a = _project_to_simplex(a)
            no_improve = 0

    # ------------------------------------------------------------------
    # Final sanity check (optional – evaluate_sequence will be called by the
    # evaluation framework as well).
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:
        print("[final] evaluate_sequence failed:", e)

    return best_seq.tolist()
