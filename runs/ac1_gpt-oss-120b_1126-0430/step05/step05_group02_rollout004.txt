#!/usr/bin/env python
"""
Propose a non‑negative step function (list of heights) with a low
evaluate_sequence() value.

Algorithm
---------
1.  Choose a working length n (at least 5 000, at most 25 000).  If a
    previous construction is available in the global `height_sequence_1`,
    its length is used as a lower bound.

2.  Initialise a probability vector (Σa_i = 1) by scanning a few
    parametric families (uniform, power‑law, exponential,
    power‑law × exponential, beta, and a few random Dirichlet draws)
    and keep the best candidate according to the cheap score
    `2·n·max_k (a*a)[k]`.

3.  Refine the best candidate with **mirror‑descent** (exponentiated‑gradient)
    that follows the *exact* sub‑gradient of the active convolution
    entries:
       * compute the full autocorrelation `c = a*a` via FFT,
       * locate all indices where `c` attains its maximum,
       * the sub‑gradient is the (averaged) correlation of `a` with the
         mask of maximal indices, which can be obtained with a second FFT
         (no explicit Python loops over O(n²) elements),
       * update `log a ← log a – η·g` and project back onto the simplex.

    The step‑size η decays like `η₀ / sqrt(t)`.  If no improvement is seen
    for a long stretch (`stagnation_iters`) a small random perturbation
    followed by a simplex projection is applied (restart).

4.  When the time budget (default 1000 s) is almost exhausted the
    best sequence encountered so far is evaluated with the supplied
    `evaluate_sequence` and returned.

The use of the exact sub‑gradient (instead of the cheap top‑k heuristic
used previously) together with FFT‑based convolution makes each iteration
fast while providing a more accurate descent direction, which typically
yields a lower final bound.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float) -> np.ndarray:
    """Return a normalised power‑law a_i ∝ (i+1)^‑p (Σa_i = 1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = i ** (-p)
    seq /= seq.sum()
    return seq


def _exponential_sequence(n: int, lam: float) -> np.ndarray:
    """Return a normalised exponential a_i ∝ exp(-lam·i)."""
    idx = np.arange(n, dtype=np.float64)
    seq = np.exp(-lam * idx)
    seq /= seq.sum()
    return seq


def _beta_sequence(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑shaped distribution on {1,…,n}."""
    x = np.arange(1, n + 1, dtype=np.float64) / n
    seq = (x ** alpha) * ((1.0 - x) ** beta)
    s = seq.sum()
    if s == 0.0:
        return None
    seq /= s
    return seq


def _conv_max_fft(a: np.ndarray, fft_len: int) -> tuple[float, int]:
    """Maximum value and its index of the full convolution a*a."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa).real[: 2 * a.size - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx


def _cheap_score(a: np.ndarray, fft_len: int) -> float:
    """2·n·max_k (a*a)[k]   (assumes Σa = 1)."""
    n = a.size
    maxb, _ = _conv_max_fft(a, fft_len)
    return 2.0 * n * maxb


def _full_conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT length L."""
    fa = np.fft.fft(a, L)
    conv = np.fft.ifft(fa * fa).real[: 2 * a.size - 1]
    return conv


def _gradient_of_max(a: np.ndarray, max_idxs: np.ndarray, L: int) -> np.ndarray:
    """
    Exact sub‑gradient of max_k (a*a)[k] w.r.t. a.
    Uses the mask of maximal indices and an FFT‑based correlation.
    Returns a vector of length n (the gradient of max, not multiplied
    by the outer 2·n factor).
    """
    n = a.size
    # mask of the same length as the linear convolution (2·n‑1)
    mask = np.zeros(2 * n - 1, dtype=np.float64)
    if max_idxs.size > 0:
        mask[max_idxs] = 1.0
    # FFT of a and mask (both zero‑padded to length L)
    fa = np.fft.fft(a, L)
    fm = np.fft.fft(mask, L)
    # Linear correlation a ⋆ mask = ifft(FA * conj(FM))
    corr = np.fft.ifft(fa * np.conj(fm)).real[:n]
    # sub‑gradient = 2· (a ⋆ mask)  –  averaged if several maxima
    if max_idxs.size > 1:
        corr *= 2.0 / max_idxs.size
    else:
        corr *= 2.0
    return corr
