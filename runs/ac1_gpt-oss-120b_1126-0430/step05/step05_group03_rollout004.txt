def simulated_annealing(a, time_budget, seed):

    n = a.size
    # compute conv and obj
    conv = conv_self_fft(a)
    obj = 2*n*conv.max()

    # We'll use exponential decay based on elapsed fraction.
    while time.time() - start < time_budget:

        i, j = rng.choice(n, size=2, replace=False)

"""
Efficient search for a low‑value step‑function sequence.

The algorithm performs three stages:

1. **Large‑scale Adam optimisation** –  
   Optimises the soft‑max (log‑it) parametrisation of a probability vector on
   several support sizes (starting from a very large length).  A smooth‑max
   approximation of the convolution maximum is used, with an annealed
   “beta’’ parameter that gradually turns the smooth‑max into a hard arg‑max.
   The Adam optimiser runs until the allocated time budget expires.

2. **Pruning** –  
   Negligible entries (below a tiny threshold) are removed, the vector is
   re‑normalised and the objective value is recomputed.  This reduces the
   dimensionality for the final refinement.

3. **Targeted stochastic local search** –  
   A hill‑climbing / simulated‑annealing phase repeatedly transfers a small
   fraction of mass from indices that contribute heavily to the current
   convolution peak to indices that contribute little.  The move is accepted
   only if it improves the exact objective; occasional random jitter helps
   escape plateaus.

All numbers are kept non‑negative, normalised to sum = 1, and the final
sequence is checked with the provided ``evaluate_sequence`` function.
The routine respects the 1000‑second budget and returns the best sequence
found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a * a using FFT (real‑valued)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation of the weight vector w (length 2·n‑1) with a (length n).

    Returns an array of length n with (w ⋆ a)[i] = Σ_t w[i+t]·a[t].
    Implemented via FFT.
    """
    n = a.shape[0]
    m = w.shape[0]                 # = 2·n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) under the assumption sum(a)=1."""
    n = a.size
    conv = _conv_self_fft(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


# ----------------------------------------------------------------------
def _run_opt(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
) -> tuple[list[float], float]:
    """
    Adam optimiser for a length‑``n`` non‑negative vector (sum=1).

    Returns the best sequence as a plain list and the corresponding exact
    objective value.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # ---- initialisation -------------------------------------------------
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s < 1e-12:
            a = np.ones(n) / n
        else:
            a = a / s
    else:
        a = rng.random(n)
        a = a / a.sum()

    # work in log‑it space (automatically respects the simplex)
    x = np.log(a + 1e-16)

    m = np.zeros_like(x)          # Adam first moment
    v = np.zeros_like(x)          # Adam second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # ---- schedules ------------------------------------------------------
    beta_start = 0.2               # start of the smooth‑max smoothing
    beta_end   = 15000.0           # end (sharp enough)
    lr_start   = 0.20
    lr_end     = 0.002
    target_iters = 1_200_000       # only used to compute a fraction

    best_a = a.copy()
    best_obj = _objective(best_a)

    it = 0
    while time.time() - start < time_limit:
        it += 1

        a = _softmax(x)                     # stays on the simplex
        conv = _conv_self_fft(a)
        M = conv.max()
        obj = 2.0 * n * M
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        # schedule parameters (smooth annealing & lr decay)
        t_frac = min(it / target_iters, 1.0)
        beta = beta_start * np.exp(np.log(beta_end / beta_start) * t_frac)
        lr   = lr_start   * np.exp(np.log(lr_end   / lr_start)   * t_frac)

        # weight vector for the smoothed max
        if beta > 10_000.0:
            # hard arg‑max (single active index)
            kmax = int(np.argmax(conv))
            w = np.zeros_like(conv)
            w[kmax] = 1.0
        else:
            shifted = conv - M                     # ≤ 0
            w = np.exp(beta * shifted)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.ones_like(conv) / conv.size
            else:
                w = w / w_sum

        # gradient w.r.t. a (smooth max)
        g_a = 2.0 * _correlation_fft(w, a)       # (n,)

        # natural gradient in log‑it space (projection onto the simplex)
        dot = np.dot(a, g_a)                     # scalar ⟨a, g_a⟩
        g_x = a * (g_a - dot)                    # (n,)

        # optional gradient clipping to keep updates stable
        g_norm = np.linalg.norm(g_x)
        if g_norm > 5.0:
            g_x = g_x * (5.0 / g_norm)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to shake out of plateaus
        if it % 5_000 == 0:
            x += rng.normal(scale=0.015, size=n)

        # periodic restart from the current best (helps escape long stalls)
        if it % 60_000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.005, size=n)
            m.fill(0.0)
            v.fill(0.0)
            print(f"[n={n}] restart at it {it}")

    # ---- compress negligible entries ------------------------------------
    thresh = 1e-9
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        n_new = best_a.size
        best_obj = _objective(best_a)
        print(f"[compress] reduced length {n} → {n_new}, O={best_obj:.6f}")

    return best_a.tolist(), float(best_obj)


# ----------------------------------------------------------------------
def _hill_climb(
    seq: list[float],
    time_limit: float,
    seed: int = 0,
) -> tuple[list[float], float]:
    """
    Simple stochastic local search (mass‑transfer moves) on a fixed length.
    Returns the best sequence found and its exact objective.
    """
    start = time.time()
    rng = np.random.default_rng(seed)
    a = np.asarray(seq, dtype=float)
    n = a.size
    best_a = a.copy()
    best_obj = _objective(best_a)

    # parameters controlling the size of each move
    max_move_frac = 0.05          # up to 5 % of the source mass

    while time.time() - start < time_limit:
        i, j = rng.choice(n, size=2, replace=False)
        # move a random fraction of source mass i → j
        delta = a[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # recompute exact objective
        obj_new = _objective(a_new)
        if obj_new < best_obj:
            best_a = a_new
            best_obj = obj_new
            a = a_new
        # occasional global jitter to escape shallow plateaus
        if rng.random() < 0.001:
            a = a + rng.normal(scale=0.004, size=n)
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            # keep the current reference point
            obj_new = _objective(a)
            if obj_new < best_obj:
                best_a = a.copy()
                best_obj = obj_new

    return best_a.tolist(), float(best_obj)


# ----------------------------------------------------------------------
def _resize_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Resize ``seq`` to length ``new_n`` by linear interpolation.
    The returned vector is normalised to sum = 1.
    """
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    # Interpolate on the interval [0, 1]
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    resized = np.interp(x_new, x_old, seq)
    resized = np.clip(resized, 0.0, None)
    s = resized.sum()
    if s < 1e-12:
        resized = np.ones(new_n) / new_n
    else:
        resized /= s
    return resized


# ----------------------------------------------------------------------
def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **_ignore,
) -> list[float]:
    """
    Entry point for the competition harness.

    The function runs a sequence of Adam optimiser instances on a ladder of
    support sizes (starting with a very large length), keeps the best
    construction, and finally refines it with a focused hill‑climbing phase.
    The returned list satisfies the security checks of ``evaluate_sequence``.
    """
    start = time.time()
    deadline = start + budget_s - 0.5   # safety margin for the harness

    # ------------------------------------------------------------------
    # Optional warm‑start from a previously discovered construction
    init_seq = None
    if "height_sequence_1" in globals():
        seq0 = np.asarray(height_sequence_1, dtype=float)
        seq0 = np.clip(seq0, 0.0, None)
        if seq0.sum() < 0.01:
            seq0 = np.ones_like(seq0)
        seq0 = seq0 / seq0.sum()
        init_seq = seq0

    # ------------------------------------------------------------------
    # Choose a geometric ladder of lengths.  The first (largest) entry gets
    # the majority of the remaining budget.
    lengths = [8192, 4096, 2048, 1024, 512]
    if init_seq is not None:
        n0 = init_seq.size
        if n0 not in lengths:
            lengths.insert(0, n0)

    best_seq = None
    best_obj = float("inf")

    # ------------------------------------------------------------------
    # Main optimisation loop over the chosen lengths
    for idx, n in enumerate(lengths):
        remaining = deadline - time.time()
        if remaining <= 1.0:
            break

        # allocate time (first length gets most, others share the rest)
        if idx == 0:
            run_time = remaining * 0.65
        else:
            run_time = (remaining - 0.2) / (len(lengths) - idx)

        # initialise from the best known sequence, resized to the current length
        init = None
        if idx == 0 and init_seq is not None:
            init = _resize_sequence(init_seq, n)
        elif best_seq is not None:

                print(f"[final local] improved to O={best_obj:.6f}")

    # Final sanity check using the provided evaluate_sequence routine.
    try:

        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    # Guarantee a non‑empty return
    if not best_seq:
        best_seq = [1.0]

    return best_seq
