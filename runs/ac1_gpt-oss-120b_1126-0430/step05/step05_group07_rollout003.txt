"""
Optimise a non‑negative step function (list of heights) for the
objective

    O(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

The optimisation proceeds in four stages:

1️⃣  Gradient flat‑ting (minimise the variance of the convolution) on the
    probability simplex using Adam.

2️⃣  Soft‑max (log‑sum‑exp) surrogate with Adam – pushes the largest
    convolution entries down.

3️⃣  A *linear* programme that, given the current a, finds a partner
    vector b with maximal total mass while keeping the convolution
    a★b below the current peak.  Mixing a and the normalised b reduces
    the peak (the classic “alternating linear program” step).

4️⃣  Stochastic local refinement (mass‑transfer moves + sub‑gradient
    steps) to polish the solution.

The code respects the 1000 s time budget, prints progress, and returns
the best sequence found (as a plain list of floats).  All numbers are
non‑negative; the result is invariant to scaling, so we keep the sum of
the coefficients = 1 (simplex) during the optimisation.
"""

import time
import math
import random

import numpy as np
import cvxpy as cp  # linear programme in stage 3


# ----------------------------------------------------------------------
# Simplex utilities ------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:                     # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution ---------------------------------------------------
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution a★a using an FFT of length *L*."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Gradient of the *variance* of the convolution -------------------------
def _grad_variance(a: np.ndarray, conv: np.ndarray, n: int, L: int) -> np.ndarray:
    """
    Gradient of  f(a) = Σ_k (conv_k - μ)²,
    where μ = 1/(2n‑1) for a normalised to sum 1.
    Closed form:  g = 4·corr(conv‑μ , a) .
    """
    mu = 1.0 / (2 * n - 1)
    diff = conv - mu                       # length 2n‑1
    a_rev = a[::-1]

    fft_diff = np.fft.rfft(diff, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_diff * fft_a_rev, n=L)

    start = n - 1
    grad = 4.0 * corr_rev[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Gradient of the log‑sum‑exp surrogate + tiny entropy term -------------
def _smooth_gradient_plus_entropy(
    a: np.ndarray,
    conv: np.ndarray,
    alpha: float,
    L: int,
    lam: float,
) -> np.ndarray:
    """∇_a ( (2·n/α)·log∑exp(α·conv) + λ·∑ a_i·log a_i )."""
    # soft‑max weights
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / exp_scaled.sum()                 # shape (2n‑1,)

    # correlation term  (same trick as in _grad_variance)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)

    grad_corr = 2.0 * corr_rev[a.size - 1:a.size - 1 + a.size]

    # tiny entropy term (keeps strict positivity)
    eps = 1e-20
    grad_ent = lam * (np.log(a + eps) + 1.0)
    return grad_corr + grad_ent


# ----------------------------------------------------------------------
# Linear‑program improvement step (stage 3) -------------------------------
def _lp_improve(a: np.ndarray, M: float, mix: float = 0.5,
                max_solve_time: float = 5.0) -> np.ndarray:
    """
    Solve the LP:

        max   Σ b_i
        s.t.  a★b   ≤   M  (elementwise)
              b_i ≥ 0

    The solution b is then normalised to sum 1 and mixed with *a*:

        a ← (1‑mix)·a + mix·b_norm

    Returns the mixed vector (already projected onto the simplex).
    """
    n = a.size
    # CVXPY objects – they are cheap to create (matrix‑free)
    a_param = cp.Parameter(n, nonneg=True)
    M_param = cp.Parameter(nonneg=True)
    b = cp.Variable(n, nonneg=True)

    # full convolution a★b (linear in b because a is a constant parameter)
    conv_expr = cp.conv(a_param, b)          # length 2n‑1
    constraints = [conv_expr <= M_param]

    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

    # set parameter values
    a_param.value = a
    M_param.value = M * (1.0 + 1e-9)         # tiny slack for numerical stability

    # solve – ECOS is a reliable LP/QP solver
    try:
        prob.solve(solver=cp.ECOS, max_iters=5000, warm_start=True,
                   # ECOS does not have an explicit time‑limit, we cap it by iteration count
                   verbose=False)
    except Exception:
        return a  # on any failure just return the unchanged vector

    if b.value is None:
        return a

    b_val = np.maximum(b.value, 0.0)
    s = b_val.sum()
    if s < 1e-12:
        return a

    # normalise to the simplex and mix
    b_norm = b_val / s
    a_new = (1.0 - mix) * a + mix * b_norm
    a_new = _project_to_simplex(a_new)
    return a_new

    print(f"[phase‑2] start, best O={best_score:.6f}")

    v = np.zeros_like(a)

        cur_obj = 2.0 * n * conv.max()

        # occasional jitter to escape plateaus
        if np.random.rand() < 0.001:
            a_new += np.random.randn(n) * 1e-4

        a = _project_to_simplex(a_new)
        it += 1

        if it % 40_000 == 0:
            print(f"[phase‑2] iter {it:,} O={cur_obj:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Phase 3 – LP‑based improvement (alternating linear programme)-
    print(f"[phase‑3] LP improvement start, best O={best_score:.6f}")

    lp_deadline = start_time + budget_s * 0.80
    lp_iter = 0
    while time.time() < lp_deadline and lp_iter < 12:
        # current peak of the convolution
        conv = _conv_fft(a, L)
        M = conv.max()

        # attempt a LP improvement
        a_candidate = _lp_improve(a, M, mix=0.5, max_solve_time=5.0)

        # evaluate the candidate
        conv_cand = _conv_fft(a_candidate, L)
        cand_obj = 2.0 * n * conv_cand.max()

        if cand_obj < best_score - 1e-8:
            best_score = cand_obj
            best_seq = a_candidate.copy()
            a = a_candidate
            print(f"[LP #{lp_iter}] new best O={best_score:.6f}")
        else:
            # if the LP did not improve, still adopt the candidate (it may help later)
            a = a_candidate

        lp_iter += 1

    # ------------------------------------------------------------------
    # 5️⃣  Phase 4 – stochastic local refinement -------------------------
    print(f"[local] entering fine‑grained refinement, best O={best_score:.6f}")

    subgrad_interval = 6
    temperature0 = 0.05
    max_iter_local = 180_000
    no_improve_limit = 3500
    max_restarts = 0

    while time.time() < deadline:
        # -------------------- initialise a (restart) --------------------
        if np.random.rand() < 0.55:
            a = best_seq.copy()
            a += np.random.randn(n) * 0.0012
            a = np.clip(a, 0, None)
            a = _project_to_simplex(a)
        else:
            a = np.random.rand(n)
            a = _project_to_simplex(a)

        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        it_loc = 0
        no_improve = 0
        T = temperature0

        while time.time() < deadline and it_loc < max_iter_local:
            # ----- random mass‑transfer move -----
            i, j = np.random.randint(n, size=2)
            if i != j and a[i] > 1e-14:
                delta = a[i] * np.random.rand() * 0.13   # up to 13 % of source mass
                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta
                a_new = _project_to_simplex(a_new)

                conv_new = _conv_fft(a_new, L)
                new_obj = 2.0 * n * conv_new.max()

                # Metropolis acceptance
                if new_obj < cur_obj or np.random.rand() < math.exp((cur_obj - new_obj) / max(T, 1e-14)):
                    a, conv, cur_obj = a_new, conv_new, new_obj
                    no_improve = 0
                    if cur_obj < best_score:
                        best_score = cur_obj
                        best_seq = a.copy()
                        print(f"[{time.time():.1f}] new global best O={best_score:.6f}"
                              f" (restart={max_restarts}, iter={it_loc})")
                else:
                    no_improve += 1
            else:
                no_improve += 1

            # ----- sub‑gradient step on the current maximal entry -----
            if it_loc % subgrad_interval == 0:
                max_val = conv.max()
                ks = np.flatnonzero(conv == max_val)
                g = np.zeros_like(a)
                for k in ks:
                    i_start = max(0, k - (n - 1))
                    i_end = min(n - 1, k)
                    i_idx = np.arange(i_start, i_end + 1, dtype=int)
                    j_idx = k - i_idx
                    g[i_idx] += 2.0 * a[j_idx]
                if ks.size:
                    g /= ks.size
                eta = 0.07 * (1.0 - it_loc / max_iter_local)
                a = a - eta * g
                a = _project_to_simplex(a)
                conv = _conv_fft(a, L)
                cur_obj = 2.0 * n * conv.max()
                if cur_obj < best_score:
                    best_score = cur_obj
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] sub‑grad improved O={best_score:.6f}")

            # ----- occasional smoothing move -----
            if it_loc % 4200 == 0 and it_loc > 0:
                kernel = np.array([0.125, 0.25, 0.25, 0.25, 0.125])
                a_smooth = np.convolve(a, kernel, mode="same")
                a = _project_to_simplex(0.85 * a + 0.15 * a_smooth)

            # temperature decay
            T *= 0.99985
            it_loc += 1

            # restart if we stagnate
            if no_improve >= no_improve_limit:
                break

        max_restarts += 1
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Final sanity check -------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:
        print("[final] evaluate_sequence failed:", e)

    return best_seq.tolist()
