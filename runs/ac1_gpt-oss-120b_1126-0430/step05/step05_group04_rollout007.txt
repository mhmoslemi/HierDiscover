"""
Non‑negative step‑function optimiser (v2).

The optimiser works in three stages:

1. **Shape pre‑search** – a small set of analytically‑motivated
   candidates (exponential, Gaussian, sparse random, uniform) is
   generated for each support length.  The best one (lowest exact
   objective) is taken as the initial point for the next stage.

2. **Adam on the simplex** – we run an Adam optimiser on the soft‑max
   (log‑it) parametrisation of the probability simplex.  The
   “smooth‑max’’ of the autocorrelation is approximated by a
   temperature‑controlled soft‑max; the temperature grows from 0.5
   to 1e6, slowly turning the smooth approximation into the exact
   max.  Learning‑rate and restarts are scheduled to explore the
   landscape thoroughly.

3. **Targeted refinement** – a fast simulated‑annealing / projected‑
   gradient descent hybrid moves mass from indices that contribute
   most to the current maximal autocorrelation entry to random other
   positions.  The final step is a few projected‑gradient iterations
   with the exact max‑gradient, projected back onto the simplex.

All work respects the global time budget (default 1000 s).  The
function ``propose_candidate`` is the entry point used by the
evaluation harness.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return conv_full[: 2 * n - 1]


def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation of a with weight vector w:
        g[i] = Σ_t w[i+t]·a[t]   for i=0,…,n‑1
    Implemented via FFT (real‑valued).
    """
    n = a.shape[0]
    m = w.shape[0]                     # = 2n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Stable soft‑max."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) assuming Σ a = 1.
    Returns a Python float.
    """
    n = a.size
    conv = _conv_self_fft(a)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of vector v onto the probability simplex
    {x ≥ 0, Σ x = 1}.
    Uses the O(n log n) algorithm of Duchi et al.
    """
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero or negative – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Stage 1 – shape pre‑search
# ----------------------------------------------------------------------


def _shape_candidates(n: int, rng: np.random.Generator):
    """
    Produce a list of candidate initial sequences for support length n.
    Each candidate is a NumPy array normalized to sum‑to‑1.
    """
    cand = []

    # Uniform
    cand.append(np.full(n, 1.0 / n))

    # Exponential decay (forward and reversed)
    r_vals = np.linspace(0.90, 0.9997, 15)
    idx = np.arange(n, dtype=float)
    for r in r_vals:
        a = r ** idx
        a = a / a.sum()
        cand.append(a)
        # reversed version (mass concentrated near the right end)
        cand.append(a[::-1])

    # Gaussian windows centred in the middle
    centre = (n - 1) / 2.0
    sigma_vals = np.linspace(0.10 * n, 0.40 * n, 10)
    for sigma in sigma_vals:
        a = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
        a = a / a.sum()
        cand.append(a)

    # Sparse Dirichlet‑like (few non‑zero entries)
    k = max(2, int(0.03 * n))
    for _ in range(4):
        a = np.zeros(n, dtype=float)
        pos = rng.choice(n, size=k, replace=False)
        vals = rng.random(k)
        a[pos] = vals
        a = a / a.sum()
        cand.append(a)

    # Pure random Dirichlet (α=1)
    a = rng.random(n)
    a = a / a.sum()
    cand.append(a)

    return cand


def _best_initial_sequence(n: int, rng: np.random.Generator, time_limit: float):
    """
    Scan a modest set of analytical shapes and return the one with the
    smallest exact objective.  The scan is aborted if time runs out.
    """
    start = time.time()
    best_a = None
    best_obj = float("inf")
    for a in _shape_candidates(n, rng):
        # Quick cut‑off in case we are out of time
        if time.time() - start > time_limit:
            break
        obj = _objective(a)
        if obj < best_obj:
            best_obj, best_a = obj, a
    # Fallback to a random Dirichlet if nothing was evaluated
    if best_a is None:
        a = rng.random(n)
        best_a = a / a.sum()
        best_obj = _objective(best_a)
    return best_a, best_obj

def _run_opt(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
) -> tuple[list[float], float]:
    """
    Adam optimisation on the log‑it simplex.
    Returns the best sequence (as list) and its exact objective.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # Initialise ---------------------------------------------------------
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
        a = a / a.sum()
    else:
        a = rng.random(n)
        a = a / a.sum()

    # logits (log‑it parametrisation)
    x = np.log(a + 1e-16)
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # Annealing schedule parameters
    beta_start = 0.5
    beta_end = 1e6                # much sharper than before
    lr_start = 0.25               # a tad larger initial LR
    lr_end = 0.0003

    best_a = a.copy()
    best_obj = _objective(best_a)

    it = 0
    while time.time() - start < time_limit:
        it += 1

        # Stay on the simplex
        a = _softmax(x)

        conv = _conv_self_fft(a)
        M = conv.max()
        obj = float(2.0 * n * M)

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        # -------------------------------------------------------------
        # schedule (time‑based)
        t = min((time.time() - start) / time_limit, 1.0)
        beta = beta_start * np.exp(np.log(beta_end / beta_start) * t)
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t)

        # smooth‑max weights
        if beta > 5e5:                     # essentially a hard max
            k_max = int(np.argmax(conv))
            w = np.zeros_like(conv)
            w[k_max] = 1.0
        else:
            shifted = conv - M               # ≤ 0
            w = np.exp(beta * shifted)
            s = w.sum()
            if s == 0.0:
                w = np.ones_like(conv) / conv.size
            else:
                w /= s

        # gradient of (smooth) max w.r.t. a
        g_a = 2.0 * _correlation_fft(w, a)

        # natural gradient in log‑it space (mirrored gradient)
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape plateaus
        if it % 3000 == 0:
            jitter = rng.normal(scale=0.015, size=n)
            x += jitter

        # periodic restart from the current best (helps to escape
        # shallow basins)
        if it % 60000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.006, size=n)
            m.fill(0.0)
            v.fill(0.0)

    # -----------------------------------------------------------------
    # prune near‑zero entries (helps the final objective)
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = _objective(best_a)
        print(f"[compress] length → {best_a.size}, O={best_obj:.6f}")

    # -----------------------------------------------------------------
    # quick local refinement (very cheap mass‑transfer moves)
    remaining = time_limit - (time.time() - start)
    if remaining > 0.5:
        a_loc = best_a.copy()
        obj_loc = best_obj
        n_loc = a_loc.size
        while time.time() - start < time_limit * 0.3:
            i, j = rng.choice(n_loc, size=2, replace=False)
            delta = a_loc[i] * rng.random() * 0.02
            if delta <= 0.0:
                continue
            a_new = a_loc.copy()
            a_new[i] -= delta
            a_new[j] += delta
            conv_new = _conv_self_fft(a_new)
            obj_new = float(2.0 * n_loc * conv_new.max())
            if obj_new < obj_loc:
                a_loc, obj_loc = a_new, obj_new
                if obj_new < best_obj:
                    best_a, best_obj = a_new, obj_new
        print(f"[local] post‑Adam O={best_obj:.6f}")

    return best_a.tolist(), best_obj

def _run_sa(
    init_seq: list[float],
    time_limit: float,
    seed: int = 0,
) -> tuple[list[float], float]:
    """
    Targeted simulated‑annealing refinement followed by a few
    projected‑gradient descent steps (exact max‑gradient).
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    a = np.asarray(init_seq, dtype=float)
    if a.sum() < 1e-12:
        a = np.ones_like(a) / a.size
    else:
        a = a / a.sum()
    n = a.shape[0]

    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())
    best_seq = a.copy()

    # SA schedule
    T = best_obj * 0.2 + 1e-8
    T_min = best_obj * 1e-6 + 1e-12
    decay = 0.997

    it = 0
    while time.time() - start < time_limit * 0.65:
        it += 1
        k_max = int(np.argmax(conv))

        # contributions a[i] * a[k_max-i]
        i_vals = np.arange(n)
        j_vals = k_max - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i_vals[mask]] * a[j_vals[mask]]
        total_contrib = contrib.sum()

        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = rng.choice(n, p=probs)
        else:
            i = rng.integers(n)

        # pick a distinct destination uniformly
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # move a modest fraction of mass
        delta = a[i] * rng.random() * 0.07   # up to 7 % of a[i]
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.clip(a_new, 0.0, None)
        a_new /= a_new.sum()

        conv_new = _conv_self_fft(a_new)
        obj_new = float(2.0 * n * conv_new.max())

        if obj_new < best_obj - 1e-14:
            a, conv, best_obj, best_seq = a_new, conv_new, obj_new, a_new.copy()
            print(f"[SA it={it}] new best O={best_obj:.6f}")
        else:
            # Metropolis acceptance
            if rng.random() < math.exp(-(obj_new - best_obj) / T):
                a, conv = a_new, conv_new

        if it % 2500 == 0:
            T = max(T * decay, T_min)

        if it % 8000 == 0:
            # occasional global jitter to shake the system
            noise = rng.normal(scale=0.008, size=n)
            a = np.clip(a + noise, 0.0, None)
            a /= a.sum()
            conv = _conv_self_fft(a)

    # -----------------------------------------------------------------
    # Projected‑gradient descent with the exact max‑gradient
    remaining = time_limit - (time.time() - start)
    if remaining > 0.3:
        a = best_seq.copy()
        n = a.size
        it_g = 0
        eta0 = 0.25
        while time.time() - start < time_limit:
            it_g += 1
            conv = _conv_self_fft(a)
            k_max = int(np.argmax(conv))
            # exact max‑gradient (one‑hot weight)
            w = np.zeros_like(conv)
            w[k_max] = 1.0
            g = 2.0 * _correlation_fft(w, a)     # gradient of max(conv)
            # simple step size schedule
            eta = eta0 * (0.995 ** it_g)
            a_new = a - eta * g
            a_new = _project_simplex(a_new)
            obj_new = float(2.0 * n * _conv_self_fft(a_new).max())
            if obj_new < best_obj - 1e-14:
                a, best_obj, best_seq = a_new, obj_new, a_new.copy()
                print(f"[PG it={it_g}] improved O={best_obj:.6f}")
            else:
                # if not improving, shrink step size more aggressively
                eta0 *= 0.99
                if eta0 < 1e-6:
                    break

    return best_seq.tolist(), best_obj

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **_ignore,
) -> list[float]:
    """
    Main optimisation routine.
    Searches over several support sizes, uses an analytical shape
    pre‑search, performs Adam optimisation, and finalises with a
    targeted simulated‑annealing + projected‑gradient refinement.
    Returns a plain list of non‑negative floats suitable for
    ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # Choose which support lengths to try – longer lengths give more
    # flexibility but also cost more time.
    lengths = [2000, 4000, 6000, 8000]

    best_seq = None
    best_obj = float("inf")

    # --------------------------------------------------------------
    # Stage‑1 + Stage‑2 for each length
    for idx, n in enumerate(lengths):
        now = time.time()
        remaining = deadline - now
        if remaining <= 2.0:
            break

        # allocate time: first length gets most, others share the rest
        if idx == 0:
            run_time = remaining * 0.55
        else:
            run_time = (remaining * 0.35) / (len(lengths) - idx)

        # ------------------------------------------------------------------
        # Shape pre‑search (quick, bounded by a few seconds)
        shape_time = min(3.0, run_time * 0.1)
        init_a, init_obj = _best_initial_sequence(n, rng, shape_time)
        print(f"[init n={n}] shape‑obj={init_obj:.6f}")

        # ------------------------------------------------------------------
        # Adam optimisation – start from the best shape we just built
        seq, val = _run_opt(
            n=n,
            time_limit=run_time,
            init_a=init_a,
            seed=seed + idx * 12345,
        )
        if val < best_obj:
            best_obj, best_seq = val, np.asarray(seq, dtype=float)
            print(f"[global] new overall best O={best_obj:.6f} (n={n})")

        if time.time() >= deadline - 0.5:
            break

    # --------------------------------------------------------------
    # Stage‑3 – targeted simulated annealing + projected gradient
    remaining = deadline - time.time()
    if best_seq is not None and remaining > 6.0:
        seq_ref, val_ref = _run_sa(
            best_seq.tolist(),
            time_limit=remaining,
            seed=seed + 999999,
        )
        if val_ref < best_obj:
            best_obj, best_seq = val_ref, np.asarray(seq_ref, dtype=float)
            print(f"[final] after SA+PG new best O={best_obj:.6f}")

    # --------------------------------------------------------------
    # Fallback – never should happen
    if best_seq is None or best_seq.size == 0:
        best_seq = np.ones(200, dtype=float) / 200.0

    # --------------------------------------------------------------
    # Final sanity check (for diagnostic purposes only)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    # Ensure the sequence respects the evaluator's caps (0 ≤ x ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    # Avoid a pathological sum‑too‑small case
    if best_seq.sum() < 0.01:
        best_seq = np.ones_like(best_seq) / best_seq.size

    return best_seq.tolist()
