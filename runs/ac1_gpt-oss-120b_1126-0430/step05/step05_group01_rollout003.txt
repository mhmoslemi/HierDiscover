#!/usr/bin/env python
"""
Non‑negative step‑function optimiser (enhanced).

The optimiser proceeds in three phases:

1. **Exploratory scan** – a short sweep over several parametric families
   (power‑law, symmetric product‑law, exponential, logistic, beta,
   sine‑shaped, plus the optional global seed) builds a collection of
   candidate probability vectors.  Each candidate is normalised to sum = 1
   and a cheap bound `score = 2·n·max(a*a)` is evaluated via an FFT‑based
   convolution.  The best candidate becomes the starting point.

2. **Mirror‑descent on the simplex** – we optimise the soft‑max logits of the
   current distribution with an Adam‑style optimiser.  The gradient is the
   smoothed sub‑gradient of the max‑convolution (soft‑max weighting of all
   shifts).  A back‑tracking line‑search guarantees that the cheap bound never
   increases.  Random jitter, learning‑rate decay and occasional restarts
   help escape plateaux.

3. **Optional up‑sampling** – if time permits the support size is
   progressively increased (up to 30 000 entries) by linear interpolation,
   after which the optimiser continues on the larger problem.

When the 1000 s budget is nearly exhausted the best distribution found is
converted to a plain Python list and evaluated with the supplied
`evaluate_sequence` function.  The list of heights is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures, no lambdas
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max, returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    s = e.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / z.size)
    return e / s


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Logits that reproduce `a` after the soft‑max."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """Smallest power‑of‑two ≥ 3·n‑2 (enough for all needed convolutions)."""
    return 1 << ((3 * n - 2).bit_length())


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    s = new_a.sum()
    if s == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= s
    return new_a


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Returns (max_conv, grad_a) where

        grad_a = ∇_a[ 4·n·smooth_max(a*a) ]

    `eps_factor` multiplies the smoothing constant ε = eps_factor·max_conv.
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv                # ≤ 0
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # Gradient already multiplied by 4·n (see docstring)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a


# ----------------------------------------------------------------------
# Parametric families (all return a non‑negative vector summing to 1)
# ----------------------------------------------------------------------


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _sym_product_power_law_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) * (n - i)) ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _exp_decay_distribution(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _logistic_distribution(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = 1.0 / (1.0 + np.exp(alpha * (i - mid)))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _beta_distribution(n: int, a_param: float, b_param: float) -> np.ndarray:
    """Beta(p,q) on (0,1) – points are taken at cell centres to avoid zeros."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n                      # centre of each bin, in (0,1)
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    a = np.power(x, a_param - 1.0) * np.power(1.0 - x, b_param - 1.0)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _sin_distribution(n: int) -> np.ndarray:
    """Positive half‑sine shape, zero at the very edges."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    the provided `evaluate_sequence` score.  The routine respects the time
    budget (≈1000 s) and returns the best sequence found.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 4.0   # safety margin for final evaluation

    # ------------------------------------------------------------------
    # 0. Choose an initial support size and scan parametric families
    # ------------------------------------------------------------------
    # start with a moderate size – we will up‑sample later if time permits
    n = 18000
    best_a = None
    best_score = float("inf")

    # Helper to evaluate a candidate quickly (assumes sum(a)=1)
    def cheap_score(a_vec):
        return 2.0 * a_vec.shape[0] * _conv_max(a_vec)

    # 0a. Use global seed if available
    global_seq = globals().get("height_sequence_1")
    if global_seq is not None:
        try:
            cand = np.array(global_seq, dtype=np.float64)
            if cand.ndim == 1 and cand.size > 1:
                cand = np.maximum(cand, 0.0)
                s = cand.sum()
                if s == 0.0:
                    cand = np.full_like(cand, 1.0 / cand.size)
                else:
                    cand /= s
                # truncate / pad to our working size
                if cand.shape[0] > n:
                    cand = cand[:n]
                elif cand.shape[0] < n:
                    cand = _interpolate_distribution(cand, n)
                score = cheap_score(cand)
                best_a = cand
                best_score = score
                print(f"[Init] Global seed used, score = {score:.6f}")
        except Exception:
            pass

    # Parameter ranges
    power_ps = np.linspace(0.45, 1.10, 17)
    sym_ps = np.linspace(0.45, 1.10, 17)
    loglams = np.linspace(-11, -5, 13)           # λ = 10**loglam
    alphas = np.linspace(0.02, 4.0, 19)
    beta_params = [0.5, 0.7, 0.9, 1.1, 1.3]      # both α and β drawn from this set

    # Scan families – stop early if we run out of time
    families = []

    families.append(("uniform", lambda nn: np.full(nn, 1.0 / nn)))

    for p in power_ps:
        families.append((f"power_{p:.3f}", lambda nn, pp=p: _power_law_distribution(nn, pp)))
    for p in sym_ps:
        families.append((f"symprod_{p:.3f}", lambda nn, pp=p: _sym_product_power_law_distribution(nn, pp)))
    for loglam in loglams:
        lam = 10.0 ** loglam
        families.append((f"exp_{lam:.2e}", lambda nn, ll=lam: _exp_decay_distribution(nn, ll)))
    for a in alphas:
        families.append((f"logistic_{a:.3f}", lambda nn, aa=a: _logistic_distribution(nn, aa)))
    families.append(("sin", lambda nn: _sin_distribution(nn)))
    for a_param in beta_params:
        for b_param in beta_params:
            families.append((f"beta_{a_param:.2f}_{b_param:.2f}",
                             lambda nn, aa=a_param, bb=b_param: _beta_distribution(nn, aa, bb)))

    # Random shuffle to improve coverage
    np.random.shuffle(families)

    for name, factory in families:
        if time.time() >= deadline:
            break
        try:
            cand = factory(n)
            # protect against NaNs / infinities
            if not np.all(np.isfinite(cand)):
                continue
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s == 0.0:
                cand = np.full_like(cand, 1.0 / n)
            else:
                cand /= s
            score = cheap_score(cand)
            if score < best_score:
                best_score = score
                best_a = cand
                print(f"[Init] New best from {name}: score = {score:.6f}")
        except Exception:
            continue

    # Fallback – uniform if nothing succeeded
    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_score = cheap_score(best_a)
        print(f"[Init] Fallback to uniform, score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1. Slight random jitter before the main optimisation
    # ------------------------------------------------------------------
    z = _logits_from_distribution(best_a)
    z += 0.01 * np.random.randn(n)

    # ------------------------------------------------------------------
    # 2. Adam state on logits (mirror descent on the simplex)
    # ------------------------------------------------------------------
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    base_alpha = 0.20          # base learning‑rate for logits
    eps_factor = 0.07          # ε = eps_factor·max_conv (smoothing)
    eps_factor_min = 5e-5
    eps_factor_decay = 0.9994

    report_every = 5000

    best_seq = _softmax(z)                 # best distribution seen so far
    best_score = cheap_score(best_seq)
    last_improve_iter = 0
    it = 0

    # ------------------------------------------------------------------
    # 3. Optimisation loop (runs until budget exhausted)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        a = _softmax(z)                     # current point on the simplex
        max_conv, grad_a = _conv_max_and_grad(a, eps_factor)
        cheap = 2.0 * n * max_conv

        # store best ever seen
        if cheap < best_score - 1e-13:
            best_score = cheap
            best_seq = a.copy()
            last_improve_iter = it
            if it % report_every == 0:
                print(f"[Iter {it:07d}] NEW BEST cheap = {cheap:.6f}")

        # gradient w.r.t. logits (mirror descent)
        mu = np.dot(grad_a, a)
        grad_z = a * (grad_a - mu)

        # Adam moment updates (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step‑size schedule (inverse‑sqrt decay)
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step_dir = - m_hat / (np.sqrt(v_hat) + eps_adam)

        # back‑tracking line search (up to 4 halvings)
        accepted = False
        trial_alpha = alpha_t
        for _ in range(4):
            z_new = z + trial_alpha * step_dir
            # keep exponentials numerically stable
            z_new = np.clip(z_new, -50.0, 50.0)
            a_new = _softmax(z_new)
            new_score = cheap_score(a_new)
            if new_score <= cheap:
                z = z_new
                accepted = True
                break
            trial_alpha *= 0.5

        if not accepted:
            # take a modest step anyway – helps when we are on a flat plateau
            z += alpha_t * step_dir
            z = np.clip(z, -50.0, 50.0)

        # occasional random perturbations to escape plateaux
        if it % 4000 == 0 and np.random.rand() < 0.15:
            z += 0.015 * np.random.randn(n)
            m *= 0.6
            v *= 0.6

        # gradually reduce smoothing factor
        eps_factor = max(eps_factor_min, eps_factor * eps_factor_decay)

        # restart if we have been stuck too long
        if it - last_improve_iter > 250_000:
            print(f"[Iter {it}] Stagnation – jitter restart.")
            z = _logits_from_distribution(best_seq) + 0.02 * np.random.randn(n)
            m.fill(0.0)
            v.fill(0.0)
            eps_factor = 0.07
            last_improve_iter = it

        # optional up‑sampling (only if we have a comfortable time margin)
        if (it % 80000 == 0) and (n < 30000) and (time.time() + 30.0 < deadline):
            new_n = min(30000, int(n * 1.15))
            print(f"[Iter {it}] Upsampling support: {n} → {new_n}")
            best_seq = _interpolate_distribution(best_seq, new_n)
            z = _logits_from_distribution(best_seq) + 0.005 * np.random.randn(new_n)
            m = np.zeros_like(z)
            v = np.zeros_like(z)
            n = new_n
            best_score = cheap_score(best_seq)
            it = 0
            last_improve_iter = 0
            eps_factor = 0.07
            continue   # restart loop with new size

    # ------------------------------------------------------------------
    # 4. Final exact evaluation with the provided evaluate_sequence()
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # Return the best sequence discovered (list of floats)
    return final_seq
