"""
Propose a step‑function (non‑negative list) that minimises

    O(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

The search combines three ideas:

* **Coarse block optimisation** – the vector is first represented by a small
  number of block‑weights (piecewise‑constant on a regular grid).  A smooth
  log‑sum‑exp surrogate of the max‑convolution is minimised with Adam while
  projecting the block‑weights onto the simplex.

* **Fine‑grained Adam + Simulated‑Annealing** – the block‑solution is expanded
  to the full length and refined with a long Adam phase (same surrogate) and
  an intensive simulated‑annealing / sub‑gradient phase that transfers mass
  between entries and accepts worsening moves with a slowly cooling schedule.

* **Final polishing** – a short Adam run with a very large smoothing parameter
  finishes the optimisation.

All phases respect the overall 1000 s budget, report intermediate objectives
and finally return the best sequence found (as a plain ``list`` of ``float``).
"""

import time
import math
import numpy as np

# -------------------------------------------------------------------------
# Simplex projection (Euclidean)
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all entries are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# -------------------------------------------------------------------------
# Full linear convolution via FFT (length 2·n‑1)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return the full a⨂a convolution (length 2·n‑1) using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# -------------------------------------------------------------------------
# Exact objective for a vector already on the simplex (∑a=1)
def _objective_simplex(a: np.ndarray, n: int, L: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for sum(a)=1."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# -------------------------------------------------------------------------
# Sub‑gradient of max(conv) (used in phase 2)
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. a (averaged over all maximisers)."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)          # indices achieving the max
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


# -------------------------------------------------------------------------
# Gradient of the smooth surrogate (log‑sum‑exp) plus entropy term
def _smooth_gradient_plus_entropy(
    a: np.ndarray,
    conv: np.ndarray,
    alpha: float,
    L: int,
    lam: float,
) -> np.ndarray:
    """
    Returns ∇_a ( (2·n/α)·log∑_k e^{α·conv_k} + λ·∑ a_i·log a_i ).
    The first part reduces to 2·corr(w,a) with w_k ∝ e^{α·conv_k};
    the second part contributes λ·(log a_i + 1).
    """
    # ---- soft‑max weights ----------
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)          # numerical stability
    w = exp_scaled / exp_scaled.sum()                 # shape (2n‑1,)

    # ---- correlation w ⋆ a (g_i = 2·∑_j w_{i+j} a_j) ----
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n_len = a.size
    grad_corr = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]

    # ---- entropy part ----------
    eps = 1e-20
    grad_ent = lam * (np.log(a + eps) + 1.0)

    return grad_corr + grad_ent


# -------------------------------------------------------------------------
# Block helpers (piecewise‑constant representation)
def _make_block_sizes(n: int, G: int) -> np.ndarray:
    """Return an array of length G with the (integer) size of each block."""
    base = n // G
    rem = n % G
    sizes = np.full(G, base, dtype=int)
    if rem:
        sizes[:rem] += 1
    return sizes


def _expand_blocks(w: np.ndarray, block_sizes: np.ndarray) -> np.ndarray:
    """Expand block weights w (sum=1) to a full length‑n vector a."""
    n = block_sizes.sum()
    a = np.empty(n, dtype=w.dtype)
    idx = 0
    for j, size in enumerate(block_sizes):
        a[idx : idx + size] = w[j] / size
        idx += size
    return a


def _aggregate_block_grad(grad_a: np.ndarray, block_sizes: np.ndarray) -> np.ndarray:
    """Aggregate a gradient on the full vector back onto the block variables."""
    G = block_sizes.shape[0]
    grad_w = np.empty(G, dtype=grad_a.dtype)
    idx = 0
    for j, size in enumerate(block_sizes):
        grad_w[j] = grad_a[idx : idx + size].sum() / size
        idx += size
    return grad_w

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        if np.random.rand() < 0.001:               # occasional jitter
            a_new += np.random.randn(n) * 5e-5

        a = _project_to_simplex(a_new)
        it += 1

    # ---------------------------------------------------------------------
    # Phase 2 – simulated annealing + sub‑gradient refinement
    print(f"[phase‑2] entering intensive local search, best O={best_score:.6f}")

    max_restarts = 0
    temperature0 = 0.20
    temp_decay = 0.99988
    max_iter_local = 350_000
    no_improve_limit = 2500
    subgrad_interval = 1

    while time.time() < deadline:
        # initialise a restart
        if np.random.rand() < 0.7:
            a = best_seq.copy()
            a += np.random.randn(n) * 0.0008        # small Gaussian jitter
            a = np.clip(a, 0, None)
            a = _project_to_simplex(a)
        else:
            a = np.random.rand(n)
            a = _project_to_simplex(a)

        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        it_loc = 0
        no_improve = 0
        T = temperature0

        while time.time() < deadline and it_loc < max_iter_local:
            # ----- mass‑transfer move -----
            i, j = np.random.randint(n, size=2)
            if i != j and a[i] > 1e-14:
                max_frac = 0.5 * (1.0 - it_loc / max_iter_local) + 0.02
                frac = np.random.rand() * max_frac
                delta = a[i] * frac
                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta
                a_new = _project_to_simplex(a_new)

                conv_new = _conv_fft(a_new, L)
                new_obj = 2.0 * n * conv_new.max()

                # Metropolis acceptance
                if new_obj < cur_obj or np.random.rand() < math.exp((cur_obj - new_obj) / max(T, 1e-16)):
                    a, conv, cur_obj = a_new, conv_new, new_obj
                    no_improve = 0
                    if cur_obj < best_score:
                        best_score = cur_obj
                        best_seq = a.copy()
                        print(f"[{time.time():.1f}] new global best O={best_score:.6f} (restart={max_restarts}, iter={it_loc})")
                else:
                    no_improve += 1
            else:
                no_improve += 1

            # ----- sub‑gradient descent -----
            if it_loc % subgrad_interval == 0:
                g = _subgradient_max_conv(a, conv)
                eta = 0.16 * (1.0 - it_loc / max_iter_local)   # slightly larger step
                a = a - eta * g
                a = _project_to_simplex(a)
                conv = _conv_fft(a, L)
                cur_obj = 2.0 * n * conv.max()
                if cur_obj < best_score:
                    best_score = cur_obj
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] sub‑grad improved O={best_score:.6f}")

            # ----- occasional smoothing (helps avoid spikes) -----
            if it_loc % 6000 == 0 and it_loc > 0:
                kernel = np.array([0.125, 0.25, 0.25, 0.25, 0.125])
                a_smooth = np.convolve(a, kernel, mode="same")
                a = _project_to_simplex(0.86 * a + 0.14 * a_smooth)

            T *= temp_decay
            it_loc += 1

            if no_improve >= no_improve_limit:
                break

        max_restarts += 1
        if time.time() >= deadline:
            break

    # ---------------------------------------------------------------------
    # Phase 3 – final polishing with a huge smoothing parameter
    print(f"[phase‑3] final Adam polishing, best O={best_score:.6f}")
    a = best_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    lr0 = 0.6
    alpha = 50000.0
    lam = 0.0
    polish_iters = 30_000
    it = 0

    while time.time() < deadline and it < polish_iters:
        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[phase‑3] iter {it:,} new best O={best_score:.6f}")

        grad = _smooth_gradient_plus_entropy(a, conv, alpha, L, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        lr = lr0 / math.sqrt(it + 1)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_to_simplex(a_new)
        it += 1

    # ---------------------------------------------------------------------
    # Final sanity check with the provided evaluator
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:
        print("[final] evaluate_sequence failed:", e)

    return best_seq.tolist()
