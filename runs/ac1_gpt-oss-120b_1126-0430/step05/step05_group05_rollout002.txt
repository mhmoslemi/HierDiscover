    decays very slowly; after a stagnation period a fresh restart is
    performed.  Every few iterations a sub‑gradient step of the exact
    max‑convolution is taken (the explicit sub‑gradient is coded) and a
    small smoothing of the vector is occasionally applied.

3️⃣  **Polishing Adam**  
    A short Adam run with an extremely large α (5·10⁵) and zero entropy
    cleans up the remaining slack.

All steps respect the 1000 s budget; the current best vector is finally
checked with the supplied ``evaluate_sequence`` routine and returned as a
plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# 1. Utility functions (no closures, top‑level)

def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Small helper – smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """
    Full linear convolution a⨂a using FFT of length L (L ≥ 2·len(a)−1).
    Returns a vector of length 2·len(a)−1.
    """
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


def _objective(a: np.ndarray, n: int, L: int) -> float:
    """
    Exact objective used by the optimiser: 2·n·max(conv(a,a)) / (sum(a)²).
    The denominator is constant for a simplex‐projected vector (sum=1), so
    we simply return 2·n·max(conv) when sum≈1.
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(a / s, L)      # normalise on the fly
    return 2.0 * n * conv.max()


def _smooth_gradient_plus_entropy(a: np.ndarray,
                                 conv: np.ndarray,
                                 alpha: float,
                                 L: int,
                                 lam: float) -> np.ndarray:
    """
    Gradient of the smooth surrogate:
        (2·n/α)·log Σexp(α·conv)  +  λ·Σ a_i·log a_i .
    The first term reduces to 2·corr(w,a) where w∝exp(α·conv).
    """
    # soft‑max weights
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)      # stability
    w = exp_scaled / exp_scaled.sum()             # shape (2n‑1,)

    # correlation w⋆a (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n_len = a.shape[0]
    grad_corr = 2.0 * corr_rev[n_len - 1: n_len - 1 + n_len]

    # entropy part (λ·(log a_i + 1))
    eps = 1e-20
    grad_ent = lam * (np.log(a + eps) + 1.0)

    return grad_corr + grad_ent


def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a (averaged over all maximisers).
    For each index k attaining the max we add the contribution of the
    corresponding slice of a, then average over all such k.
    """
    n = a.shape[0]
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)   # all maximising shifts
    g = np.zeros_like(a)

    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]

    if ks.size:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
# 2. The main optimisation routine

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    the provided evaluation function.  Returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.3          # safety margin

    # ------------------------------------------------------------------
    # a) Initialise a starting point
    if "height_sequence_1" in globals():
        init = np.array(height_sequence_1, dtype=float)
        init = np.maximum(init, 0.0)
        tot = init.sum()
        if tot < 1e-12:
            n = init.size
            a = np.full(n, 1.0 / n)
        else:
            a = init / tot                          # project to simplex
        n = a.size
        print(f"[init] loaded external sequence, n={n}")
    else:
        n = 5200                                      # a comfortable size
        a = np.full(n, 1.0 / n)                      # start uniform
        print(f"[init] uniform start, n={n}")

    L = _next_pow2(2 * n - 1)                         # FFT length

    # ------------------------------------------------------------------
    # b) Helper to keep the global best
    best_seq = a.copy()
    best_score = _objective(best_seq, n, L)
    print(f"[global] initial O={best_score:.6f}")

    # ------------------------------------------------------------------
    # c) Phase 1 – annealed Adam on the smooth surrogate
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # schedule parameters
    max_iter_phase1 = 700_000
    phase1_deadline = start_time + budget_s * 0.55   # ~55 % of budget

    lr0 = 0.95
    alpha_start = 4.0
    alpha_end = 5e5
    lam_start = 0.25

    it = 0
    while time.time() < phase1_deadline and it < max_iter_phase1:
        conv = _conv_fft(a, L)
        cur_obj = _objective(a, n, L)
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[phase‑1] iter {it:,}  new best O={best_score:.6f}")

        # --- annealing schedules -------------------------------------------------
        frac = it / max_iter_phase1
        alpha = alpha_start + (alpha_end - alpha_start) * frac
        lam = max(0.0, lam_start * (1.0 - frac))

        # --- gradient ------------------------------------------------------------
        grad = _smooth_gradient_plus_entropy(a, conv, alpha, L, lam)

        # --- Adam update ---------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        lr = lr0 / math.sqrt(it + 1)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional tiny jitter to escape plateaus
        if np.random.rand() < 0.001:
            a_new += np.random.randn(n) * 4e-5

        a = _project_to_simplex(a_new)
        it += 1

    # ------------------------------------------------------------------
    # d) Phase 2 – intensive simulated‑annealing + sub‑gradient descent
    print(f"[phase‑2] start local refinement, best O={best_score:.6f}")

    # parameters
    temp0 = 0.12
    temp_decay = 0.99986
    max_iter_local = 350_000
    no_improve_limit = 2500

    # keep a copy of the best sequence for restarts
    base_best = best_seq.copy()

    while time.time() < deadline:
        # ---------- initialise a new walk ----------
        if np.random.rand() < 0.55:
            a = best_seq.copy()
            a += np.random.randn(n) * 0.0012          # jitter
            a = np.clip(a, 0.0, None)
            a = _project_to_simplex(a)
        else:
            a = np.random.rand(n)
            a = _project_to_simplex(a)

        conv = _conv_fft(a, L)
        cur_obj = _objective(a, n, L)

        it_loc = 0
        no_improve = 0
        T = temp0

        while time.time() < deadline and it_loc < max_iter_local:
            # ----------- mass‑transfer move -----------
            i, j = np.random.randint(n, size=2)
            if i != j and a[i] > 1e-15:
                # fraction to move – larger early on, shrinking later
                max_frac = 0.45 * (1.0 - it_loc / max_iter_local) + 0.02
                frac = np.random.rand() * max_frac
                delta = a[i] * frac
                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta
                a_new = _project_to_simplex(a_new)

                new_obj = _objective(a_new, n, L)

                if new_obj < cur_obj or np.random.rand() < math.exp((cur_obj - new_obj) / max(T, 1e-16)):
                    a, conv, cur_obj = a_new, _conv_fft(a_new, L), new_obj
                    no_improve = 0
                    if cur_obj < best_score:
                        best_score = cur_obj
                        best_seq = a.copy()
                        print(f"[{time.time():.1f}] NEW GLOBAL BEST O={best_score:.6f} (iter={it_loc})")
                else:
                    no_improve += 1
            else:
                no_improve += 1

            # ----------- occasional swap (exchange) -----------
            if it_loc % 4000 == 0 and it_loc > 0:
                i, j = np.random.randint(n, size=2)
                a_swapped = a.copy()
                a_swapped[i], a_swapped[j] = a_swapped[j], a_swapped[i]
                # swapping does not change the simplex, only positions
                new_obj = _objective(a_swapped, n, L)
                if new_obj < cur_obj or np.random.rand() < math.exp((cur_obj - new_obj) / max(T, 1e-16)):
                    a, conv, cur_obj = a_swapped, _conv_fft(a_swapped, L), new_obj
                    if new_obj < best_score:
                        best_score = new_obj
                        best_seq = a.copy()
                        print(f"[{time.time():.1f}] swap improved O={best_score:.6f}")

            # ----------- sub‑gradient step -----------
            if it_loc % 1 == 0:
                g = _subgradient_max_conv(a, conv)
                # step size linearly decays, a bit larger than before
                eta = 0.18 * (1.0 - it_loc / max_iter_local)
                a = a - eta * g
                a = _project_to_simplex(a)
                conv = _conv_fft(a, L)
                cur_obj = _objective(a, n, L)
                if cur_obj < best_score:
                    best_score = cur_obj
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] sub‑grad improved O={best_score:.6f}")

            # ----------- occasional smoothing -----------
            if it_loc % 6000 == 0 and it_loc > 0:
                kernel = np.array([0.1, 0.2, 0.4, 0.2, 0.1])
                a_s = np.convolve(a, kernel, mode="same")
                a = _project_to_simplex(0.83 * a + 0.17 * a_s)

            # decay temperature
            T *= temp_decay
            it_loc += 1

            if no_improve >= no_improve_limit:
                # stagnated – break to restart
                break

        # after each walk we keep the best ever seen as the starting point for the next walk
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # e) Phase 3 – final Adam polishing (huge α, no entropy)
    print(f"[phase‑3] polishing, best O={best_score:.6f}")
    a = best_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    lr0 = 0.55
    alpha = 5e5
    lam = 0.0
    max_polish = 35_000
    it = 0
    while time.time() < deadline and it < max_polish:
        conv = _conv_fft(a, L)
        cur_obj = _objective(a, n, L)

        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[phase‑3] iter {it:,}  new best O={best_score:.6f}")

        grad = _smooth_gradient_plus_entropy(a, conv, alpha, L, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        lr = lr0 / math.sqrt(it + 1)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_to_simplex(a_new)

        it += 1

    # ------------------------------------------------------------------
    # f) Final sanity check using the provided evaluator
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence failed:", exc)

    # Return the best-found sequence
    return best_seq.tolist()
