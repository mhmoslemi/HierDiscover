def proj_simplex(v):
    # v is array of floats, may not sum to 1, may have negatives.
    # project onto {x >= 0, sum x = 1}
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

#!/usr/bin/env python
"""
Propose a candidate step‑function sequence.

The algorithm works in three stages:

1. **Choose a good starting shape.**  We test a few Beta(α,α)
   densities (α≈0.5 gives the classic arcsine law) on the chosen
   support size and keep the one with the smallest value of the
   evaluation function.

2. **Smooth optimisation (Adam on the simplex).**  The probability
   vector is represented by soft‑max logits; a smoothed soft‑max of the
   self‑convolution provides a differentiable surrogate for the
   max‑convolution.  Adam updates with a decaying learning‑rate are
   performed until 60 % of the time budget has elapsed.

3. **Fine‑tuning with exact sub‑gradient descent.**  After the smooth
   phase we switch to a projected sub‑gradient descent that uses the
   true gradient of the max‑convolution (the sum of two shifted copies
   of the current vector).  After each step the vector is projected
   back onto the probability simplex.  This phase runs until the time
   limit is reached.

The whole routine respects the 1000 s budget, works with any
previously supplied sequence (`height_sequence_1`) to set the support
size, and finally returns a plain list of non‑negative heights.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (no closures)
# ----------------------------------------------------------------------


def _fft_len_for_corr(n: int) -> int:
    """Smallest power‑of‑two ≥ 3·n‑2, enough for linear convolution."""
    return 1 << ((3 * n - 2).bit_length())


def _conv_full(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·n‑1)."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return conv


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of a*a (no gradient)."""
    return float(_conv_full(a).max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Smoothed max of a*a and its gradient w.r.t. a (used for Adam).

    Parameters
    ----------
    a : np.ndarray
        Probability vector (sum = 1, non‑negative).
    eps_factor : float
        Multiplicative factor for smoothing constant.

    Returns
    -------
    max_conv : float
        Maximum of the self‑convolution.
    grad_a : np.ndarray
        Gradient of `2·n·smooth_max` w.r.t. a (already multiplied by 4·n).
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    a_fft = np.fft.rfft(a, L)

    # linear self‑convolution (length 2n‑1)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # smoothing weight (soft‑max)
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or np.isnan(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # correlation of a with the soft‑max weights
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # gradient (4·n * correlation)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a


# ----------------------------------------------------------------------
# Simplex helpers
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max (returns a probability vector that sums to 1)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it representation that yields `a` after soft‑max."""
    return np.log(a + eps)


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x≥0, ∑x = 1}.  Implements the algorithm of
    Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive – return the uniform vertex
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient of the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_maxconv(a: np.ndarray):
    """
    Compute max entry of a*a and a sub‑gradient of that max w.r.t. `a`.

    Returns
    -------
    max_val : float
        Maximum of the self‑convolution.
    grad : np.ndarray
        Sub‑gradient (sum of gradients of all active constraints).
    """
    n = a.shape[0]
    conv = _conv_full(a)
    max_val = float(conv.max())
    # indices where the maximum is attained (numerical tolerance)
    ks = np.where(conv >= max_val - 1e-15)[0]

    # Sub‑gradient: for each active k, ∂(a*a)_k/∂a_j = a_j + a_{k‑j}
    grad = np.zeros_like(a)
    count = ks.size
    if count == 0:
        return max_val, grad

    # term a_j appears count times
    grad += count * a

    # add the shifted contributions
    for k in ks:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]

    return max_val, grad


# ----------------------------------------------------------------------
# Helper: generate a Beta(α,α) discretisation (mid‑point rule)
# ----------------------------------------------------------------------


def _beta_midpoint(n: int, alpha: float) -> np.ndarray:
    """Return a probability vector proportional to x^{α‑1}(1‑x)^{α‑1}."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n                # midpoint in each sub‑interval
    # avoid log(0) – x∈(0,1) so safe
    log_raw = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    raw = np.exp(log_raw)
    # if alpha<1 the extremes become large but still finite
    a = raw / raw.sum()
    return a


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that minimises the evaluation metric.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0    # tiny safety margin

    # --------------------------------------------------------------
    # Determine the support size.
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
        print(f"Using length from existing variable: n = {n}")
    except Exception:
        n = 20000
        print(f"No external sequence found – using default length n = {n}")

    # --------------------------------------------------------------
    # 0. Pick a good starting distribution (Beta(α,α) family).
    # --------------------------------------------------------------
    alphas = [0.30, 0.35, 0.40, 0.45, 0.50, 0.55, 0.60, 0.70, 0.80]
    best_start = None
    best_score = np.inf

    for a in alphas:
        seq = _beta_midpoint(n, a)
        score = 2.0 * n * _conv_max(seq)      # sum(seq)=1 → objective = 2·n·max
        print(f"Alpha={a:.2f} → raw score={score:.6f}")
        if score < best_score:
            best_score = score
            best_start = seq.copy()

    # fall back to arcsine if something went wrong
    if best_start is None:
        idx = np.arange(n, dtype=np.float64)
        x = (idx + 0.5) / n
        raw = 1.0 / np.sqrt(x * (1.0 - x))
        best_start = raw / raw.sum()
        best_score = 2.0 * n * _conv_max(best_start)

    print(f"Initial best score after shape search: {best_score:.6f}")

    # --------------------------------------------------------------
    # 1. Adam optimisation on the simplex (smooth max‑convolution).
    # --------------------------------------------------------------
    a_cur = best_start.copy()
    z = _logits_from_distribution(a_cur)
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_alpha = 0.45          # a slightly larger base learning‑rate
    eps_factor = 0.08
    eps_factor_min = 5e-4
    eps_factor_decay = 0.9995

    best_a = a_cur.copy()
    best_obj = best_score
    it = 0

    # allocate roughly 60 % of the total budget to the Adam phase
    adam_deadline = start + 0.60 * budget_s

    while time.time() < adam_deadline:
        it += 1
        a_cur = _softmax(z)

        # cheap objective value
        max_conv = _conv_max(a_cur)
        cur_obj = 2.0 * n * max_conv
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a_cur.copy()

        # smoothed max and its gradient
        max_conv_s, grad_a = _conv_max_and_grad(a_cur, eps_factor)

        # mirror‑descent (natural gradient) on the simplex
        mu = np.dot(grad_a, a_cur)
        grad_z = a_cur * (grad_a - mu)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step‑size schedule (≈1/√t)
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step = - alpha_t * m_hat / (np.sqrt(v_hat) + eps_adam)
        z = z + step
        # keep exponentials in a safe range
        z = np.clip(z, -50.0, 50.0)

        # tighten smoothing constant gradually
        eps_factor = max(eps_factor_min, eps_factor * eps_factor_decay)

        # occasional tiny jitter to avoid tiny plateaus
        if it % 5000 == 0 and np.random.rand() < 0.1:
            z += 0.01 * np.random.randn(n)

        # optional early stop – not used here
        # if it % 20000 == 0 and best_obj < 1.05:
        #     break

    print(f"After Adam: best objective = {best_obj:.6f} (iterations={it})")
    a_cur = best_a.copy()   # switch to the best point found so far

    # --------------------------------------------------------------
    # 2. Exact sub‑gradient descent with projection onto the simplex.
    # --------------------------------------------------------------
    sub_it = 0
    eta0 = 0.02            # initial step‑size for sub‑gradient phase

    while time.time() < deadline:
        sub_it += 1
        max_conv, grad = _subgradient_maxconv(a_cur)
        cur_obj = 2.0 * n * max_conv
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a_cur.copy()

        # simple decaying step size (linear decay, very gentle)
        eta = eta0 / (1.0 + sub_it * 0.001)
        step = - eta * grad
        a_next = a_cur + step

        # projection onto probability simplex (keeps sum = 1 and non‑negative)
        a_cur = _proj_simplex(a_next)

        # occasional random jitter to shake out of shallow plateaus
        if sub_it % 3000 == 0 and np.random.rand() < 0.07:
            a_cur += 0.002 * np.random.randn(n)
            a_cur = np.maximum(a_cur, 0.0)
            a_cur = _proj_simplex(a_cur)

        # periodic logging
        if sub_it % 5000 == 0:
            elapsed = time.time() - start
            print(f"[{elapsed:.1f}s] sub‑grad iter {sub_it}, "
                  f"best={best_obj:.6f}, cur={cur_obj:.6f}")

    # --------------------------------------------------------------
    # 3. Return the best sequence found (plain Python list).
    # --------------------------------------------------------------
    final_seq = best_a.tolist()

    # final exact evaluation (for logging only)

        print(f"Final exact evaluation score = {final_score:.6f}")
    except Exception as exc:
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    return final_seq
