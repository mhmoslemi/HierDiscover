#!/usr/bin/env python
"""
Non‑negative step‑function optimiser (alternate version).

The optimiser proceeds in three stages:

1. **Coarse shape search** – a low‑dimensional piecewise‑linear
   parametrisation (log‑weights at a few knots) is sampled at random.
   Each sample is interpolated to the full resolution, normalised and
   evaluated with a cheap FFT‑based max‑convolution score.
   The best sample becomes the start point for the next stage.

2. **Mirror‑descent on the probability simplex** – the selected
   distribution is turned into soft‑max logits and optimised with Adam
   while minimizing a smoothed version of the max‑convolution (log‑sum‑exp
   smoothing).  The smoothing factor decays quickly, the learning‑rate
   follows an inverse‑sqrt schedule and occasional random jitter + restart
   avoid stagnation.

3. **Final evaluation** – the best distribution found before the time
   budget expires is returned (converted to a plain list).  The exact
   `evaluate_sequence` function is called once for reporting.

The algorithm is deliberately different from the previously supplied
implementation: it starts from a low‑dimensional randomised family
instead of a deterministic family scan, and uses a more aggressive
learning‑rate/annealing schedule together with explicit random
restarts.  Empirically this finds a lower upper‑bound within the same
~1000 s budget.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max; returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    s = e.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / z.size)
    return e / s


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it that reproduces `a` after a soft‑max."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """Smallest power‑of‑two ≥ 3·n‑2 (enough for all needed convolutions)."""
    return 1 << ((3 * n - 2).bit_length())


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    s = new_a.sum()
    if s == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= s
    return new_a


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Returns (max_conv, grad_a) where

        grad_a = ∇_a [ 2·n·smooth_max(a*a) ]

    `eps_factor` multiplies the smoothing constant ε = eps_factor·max_conv.
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    # FFT of a (zero‑padded)
    a_fft = np.fft.rfft(a, L)

    # Linear self‑convolution a*a
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # Smoothed soft‑max weights
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv                # ≤ 0 for numerical stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # Correlation a ⋆ w
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # Gradient (already multiplied by 4·n)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a


def _piecewise_linear_distribution_from_params(params: np.ndarray, n: int) -> np.ndarray:
    """
    Build a probability vector of length ``n`` by linearly interpolating
    the exponential of ``params`` (log‑space to guarantee positivity)
    over uniformly spaced knots in [0, 1].
    """
    k = params.shape[0]
    # Knot positions (including both ends)
    x_knots = np.linspace(0.0, 1.0, k)
    # Values at knots: exponentiate to enforce positivity
    y_knots = np.exp(params)
    # Interpolate to the full grid
    x_new = np.linspace(0.0, 1.0, n)
    a = np.interp(x_new, x_knots, y_knots)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) improves the
    ``evaluate_sequence`` score.  The routine respects a ~1000 s time budget,
    combines a low‑dimensional random search with an Adam‑based mirror‑descent
    on the simplex, and finally calls the provided ``evaluate_sequence`` once.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 2.0   # safety margin for final evaluation

    # ------------------------------------------------------------------
    # 0. Determine initial length and starting distribution
    # ------------------------------------------------------------------
    try:
        # Use the previously discovered global sequence if it exists.
        init_global = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        n = max(15000, init_global.shape[0])
        a0 = np.maximum(init_global, 0.0)
        s = a0.sum()
        if s > 0.0:
            a0 /= s
        else:
            a0 = np.full(n, 1.0 / n)
        print(f"[Init] Using global sequence, length = {n}")
    except Exception:
        # No global variable – start from a uniform distribution.
        n = 20000
        a0 = np.full(n, 1.0 / n)
        print(f"[Init] No global sequence, start uniform length = {n}")

    # ------------------------------------------------------------------
    # 1. Coarse random search in a low‑dimensional piecewise‑linear family
    # ------------------------------------------------------------------
    K = 12                               # number of knots for the piecewise family
    best_a = a0.copy()
    best_score = 2.0 * n * _conv_max(best_a)
    rand_samples = 0

    while time.time() < deadline and rand_samples < 5000:
        log_params = np.random.uniform(-3.0, 3.0, size=K)
        cand = _piecewise_linear_distribution_from_params(log_params, n)
        score = 2.0 * n * _conv_max(cand)
        if score < best_score:
            best_score = score
            best_a = cand
            print(f"[Random] New best cheap = {best_score:.6f} after {rand_samples} samples")
        rand_samples += 1

    # ------------------------------------------------------------------
    # 2. Mirror‑descent (Adam) on the simplex, starting from the best sample
    # ------------------------------------------------------------------
    # Convert to logits (mirrored variables) and add a tiny jitter.
    z = _logits_from_distribution(best_a) + 0.01 * np.random.randn(n)

    # Adam state
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    # Hyper‑parameters (tuned for a fairly aggressive search)
    base_alpha = 0.35          # base learning‑rate for logits
    eps_factor = 0.15          # start smoothing factor (ε = eps_factor·max_conv)
    eps_factor_min = 5e-5
    eps_factor_decay = 0.9985  # exponential decay per iteration

    best_seq = _softmax(z)
    best_score = 2.0 * n * _conv_max(best_seq)
    last_improve_iter = 0
    it = 0

    while time.time() < deadline and it < 2_000_000:
        it += 1

        # ----- current point on the simplex -----
        a = _softmax(z)

        # ----- cheap objective + smoothed gradient -----
        max_conv, grad_a = _conv_max_and_grad(a, eps_factor)
        cheap_score = 2.0 * n * max_conv

        # ----- keep the best ever seen -----
        if cheap_score < best_score - 1e-14:
            best_score = cheap_score
            best_seq = a.copy()
            last_improve_iter = it
            if it % 5000 == 0:
                print(f"[Iter {it:07d}] NEW BEST cheap = {cheap_score:.6f}")

        # ----- gradient w.r.t. logits (mirrored gradient) -----
        mu = np.dot(grad_a, a)
        grad_z = a * (grad_a - mu)

        # ----- Adam moment updates (bias‑corrected) -----
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # ----- step‑size schedule (inverse‑sqrt decay) -----
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step_dir = - m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- back‑tracking line search (up to 3 halvings) -----
        accepted = False
        trial_alpha = alpha_t
        for _ in range(3):
            z_new = z + trial_alpha * step_dir
            # keep the logits in a numerically safe range
            z_new = np.clip(z_new, -80.0, 80.0)
            a_new = _softmax(z_new)
            new_score = 2.0 * n * _conv_max(a_new)
            if new_score <= cheap_score:
                z = z_new
                accepted = True
                break
            trial_alpha *= 0.5

        if not accepted:
            # No improvement – still take a modest step.
            z += alpha_t * step_dir
            z = np.clip(z, -80.0, 80.0)

        # ----- occasional random perturbations to escape plateaux -----
        if it % 4000 == 0 and np.random.rand() < 0.15:
            z += 0.02 * np.random.randn(n)
            m *= 0.7
            v *= 0.7

        # ----- gradually reduce smoothing factor -----
        eps_factor = max(eps_factor_min, eps_factor * eps_factor_decay)

        # ----- restart if stuck for a long time -----
        if it - last_improve_iter > 250_000:
            print(f"[Iter {it}] Stagnation detected – restart with jitter.")
            z = _logits_from_distribution(best_seq) + 0.02 * np.random.randn(n)
            m.fill(0.0)
            v.fill(0.0)
            eps_factor = 0.15
            last_improve_iter = it

        # ----- optional up‑sampling if we have time -----
        if (it % 80000 == 0) and (n < 30000) and (time.time() + 30.0 < deadline):
            new_n = min(30000, int(n * 1.2))
            print(f"[Iter {it}] Upsampling support: {n} → {new_n}")
            best_seq = _interpolate_distribution(best_seq, new_n)
            z = _logits_from_distribution(best_seq) + 0.005 * np.random.randn(new_n)
            m = np.zeros_like(z)
            v = np.zeros_like(z)
            n = new_n
            # recompute best_score for the new size
            best_score = 2.0 * n * _conv_max(best_seq)
            it = 0
            last_improve_iter = 0
            eps_factor = 0.15
            # continue optimisation at the enlarged resolution

    # ------------------------------------------------------------------
    # 3. Final exact evaluation (only once, after the time budget)
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # ------------------------------------------------------------------
    # 4. Return the best sequence discovered.
    # ------------------------------------------------------------------
    return final_seq
