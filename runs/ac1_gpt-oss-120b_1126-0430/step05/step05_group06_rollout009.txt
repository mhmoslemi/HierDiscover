"""
search_step_function.py

A refined optimizer for the step‑function evaluation problem

    O(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm is split into four timed phases:

0️⃣  Randomised restarts – generate a diverse pool of seeds:
    Dirichlet draws with various concentrations, plus a few
    parametrised families (Gaussian, cosine, exponential, etc.).
    Each seed receives a few Adam steps on the smooth log‑sum‑exp
    surrogate; the best seed is kept.

1️⃣  Long Adam optimisation – minimise the soft‑max surrogate
    𝑓(α) = (2·n/α)·log∑ₖexp(α·convₖ) while projecting onto the
    probability simplex after every step.  The sharpening parameter
    α grows exponentially from ≈1 to ≈10⁶, allowing the optimiser
    to first explore and then converge to the true max‑operator.

2️⃣  Exact projected sub‑gradient descent – directly minimises the
    true objective 2·n·max(conv) using a back‑tracking line‑search
    that guarantees descent.

3️⃣  Targeted simulated‑annealing – repeatedly transfers a small
    fraction of probability mass from an index that contributes a
    large amount to the current peak of the convolution to a random
    destination that contributes little.  Moves are accepted with
    a Metropolis rule whose temperature decays slowly.  Every few
    iterations a sub‑gradient step is performed to keep the walk
    directed.

All phases respect a total wall‑time budget of ``budget_s`` seconds.
The best sequence found is finally verified with the supplied
``evaluate_sequence`` function and returned as a plain list of floats.
"""

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, ∑x=1}."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                       # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution via FFT (zero‑padded)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return the linear (full) convolution a*a using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    # the result of the linear convolution is the first 2·len(a)‑1 entries
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Exact objective for a simplex‑normalised vector (∑a = 1)
def _objective(a: np.ndarray, n: int, L: int) -> float:
    """2·n·max(conv(a,a))  (a is assumed to sum to 1)."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Sub‑gradient of the maximal convolution entry
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. a (see analysis)."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)
    g = np.zeros_like(a)
    # average the sub‑gradient over all maximising indices
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
# Gradient of the log‑sum‑exp (smooth max) surrogate
def _smooth_gradient(a: np.ndarray, conv: np.ndarray,
                     alpha: float, L: int) -> np.ndarray:
    """
    Compute ∇_a ( (2·n/α)·log∑_k e^{α·conv_k} ).

    The gradient reduces to 2·corr(w,a) where w_k ∝ e^{α·conv_k}.
    """
    # soft‑max weights over the convolution entries
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)   # stabilise
    w = exp_scaled / exp_scaled.sum()          # shape (2n‑1,)

    # correlation w ⋆ a   (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)   # length ≥ 3·n‑2
    n_len = a.size
    grad = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]
    return grad


# ----------------------------------------------------------------------
# Adam optimisation on the smooth surrogate
def _adam_phase(a: np.ndarray, n: int, L: int,
                max_iters: int, deadline: float,
                alpha_start: float = 1.0,
                alpha_end: float = 1e6,
                lr0: float = 0.6,
                beta1: float = 0.9,
                beta2: float = 0.999,
                eps: float = 1e-12,
                prune_every: int = 0,
                prune_percentile: float = 1.0) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Adam on the soft‑max surrogate.
    Returns (final a, best objective value seen, best a).
    """
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_obj = _objective(a, n, L)

    total_iters = max_iters
    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            # occasional progress report
            if (it & 0xFFF) == 0:
                print(f"[adam] iter {it:>6d} new best O={best_obj:.6f}")

        # exponential schedule for α – sharper towards the end
        frac = it / total_iters
        alpha = alpha_start * (alpha_end / alpha_start) ** frac

        grad = _smooth_gradient(a, conv, alpha, L)

        # Adam bias‑corrected update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        lr = lr0 / math.sqrt(it + 1)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional tiny jitter to escape flat plateaus
        if np.random.rand() < 0.001:
            a_new += np.random.randn(n) * 1e-4

        a_new = _project_to_simplex(a_new)

        # optional pruning of tiny entries
        if prune_every and it % prune_every == 0 and it > 0:
            thresh = np.percentile(a_new, prune_percentile)
            a_new[a_new < thresh] = 0.0
            a_new = _project_to_simplex(a_new)

        a = a_new

    return a, best_obj, best_a

# Exact projected sub‑gradient descent with simple back‑tracking line‑search
def _subgradient_phase(a: np.ndarray, n: int, L: int,
                       max_iters: int, deadline: float,
                       eta0: float = 0.3,
                       line_search: bool = True,
                       verbose: bool = False) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Projected sub‑gradient descent on the true objective.
    Returns (final a, best objective, best a).
    """
    best_a = a.copy()
    best_obj = _objective(a, n, L)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            if verbose and (it % 5000 == 0):
                print(f"[sub‑grad] iter {it:>6d} new best O={best_obj:.6f}")

        g = _subgradient_max_conv(a, conv)

        # geometric decay of the step size
        eta = eta0 * (0.9999 ** it)

        if line_search:
            step = eta
            for _ in range(5):
                cand = _project_to_simplex(a - step * g)
                cand_obj = _objective(cand, n, L)
                if cand_obj < cur_obj:
                    a = cand
                    break
                step *= 0.5
            else:
                a = _project_to_simplex(a - eta * g)
        else:
            a = _project_to_simplex(a - eta * g)

    return a, best_obj, best_a

# Targeted simulated annealing
def _simulated_annealing_phase(a: np.ndarray, n: int, L: int,
                               max_iters: int, deadline: float,
                               temperature0: float = 0.08,
                               no_improve_limit: int = 8000,
                               verbose: bool = True) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Local refinement: move a modest fraction of probability mass from an
    index that contributes heavily to the current maximal convolution entry
    to a destination that contributes little.  Accept with Metropolis rule.
    Every few steps a true sub‑gradient step is performed.
    """
    best_a = a.copy()
    best_obj = _objective(a, n, L)
    cur_obj = best_obj
    conv = _conv_fft(a, L)

    T = temperature0
    no_improve = 0
    rng = np.random.default_rng()

    # Pre‑compute list of indices for fast random choice
    indices = np.arange(n)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # index where the current peak occurs
        kmax = int(np.argmax(conv))

        # contributions a[i] * a[kmax-i] for all admissible i
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        i_vals = np.arange(i_start, i_end + 1, dtype=int)
        j_vals = kmax - i_vals

        # contribution magnitude
        contrib = a[i_vals] * a[j_vals]
        if contrib.sum() == 0.0:
            # fallback to uniform random move
            src = rng.integers(n)
            dst = rng.integers(n - 1)
            if dst >= src:
                dst += 1
        else:
            # source: sample proportionally to its contribution to the peak
            prob_src = contrib / contrib.sum()
            src = rng.choice(i_vals, p=prob_src)

            # destination: random index, but prefer low contribution to the peak
            # (i.e. where a[dst] * a[kmax-dst] is small)
            dst_candidates = indices[indices != src]
            # evaluate a[dst] * a[kmax-dst] (outside the range is treated as 0)
            k_minus_dst = kmax - dst_candidates
            mask = (k_minus_dst >= 0) & (k_minus_dst < n)
            dst_contrib = np.zeros_like(dst_candidates, dtype=float)
            dst_contrib[mask] = a[dst_candidates[mask]] * a[k_minus_dst[mask]]
            # invert to obtain low‑contribution preference
            inv = 1.0 / (dst_contrib + 1e-12)
            inv = inv / inv.sum()
            dst = rng.choice(dst_candidates, p=inv)

        # transfer a modest fraction of source mass
        delta = a[src] * rng.random() * 0.07  # up to 7 % of src
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta
        a_new = _project_to_simplex(a_new)

        conv_new = _conv_fft(a_new, L)
        new_obj = 2.0 * n * conv_new.max()

        # Metropolis acceptance
        accept = False
        if new_obj < cur_obj:
            accept = True
        else:
            prob = math.exp((cur_obj - new_obj) / max(T, 1e-14))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_new
            conv = conv_new
            cur_obj = new_obj
            no_improve = 0
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                if verbose:
                    print(f"[SA] iter {it:>6d} new global best O={best_obj:.6f}")
        else:
            no_improve += 1

        # occasional exact sub‑gradient step to steer the walk
        if it % 12 == 0:
            g = _subgradient_max_conv(a, conv)
            eta_loc = 0.07 * (1.0 - it / max_iters)
            a = _project_to_simplex(a - eta_loc * g)
            conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                if verbose:
                    print(f"[SA‑sub] iter {it:>6d} improved O={best_obj:.6f}")

        # temperature decay (slow)
        T *= 0.9999

        # restart if stagnation lasts too long
        if no_improve >= no_improve_limit:
            if verbose:
                print(f"[SA] stagnation – restarting from best (iter {it})")
            a = best_a.copy()
            a += rng.normal(scale=0.0015, size=n)
            a = np.clip(a, 0.0, None)
            a = _project_to_simplex(a)
            conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max()
            no_improve = 0

    return a, best_obj, best_a

# Helper factories for parametric initialisations
def _make_gaussian(n: int, sigma: float) -> np.ndarray:
    """Gaussian‑shaped vector centred at (n‑1)/2."""
    mu = (n - 1) / 2.0
    x = np.arange(n, dtype=float)
    a = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    return a / a.sum()


def _make_cosine(n: int, power: float = 1.0) -> np.ndarray:
    """Raised cosine: (1‑cos(π·i/(n‑1)))^power, zero at the boundaries."""
    i = np.arange(n, dtype=float)
    a = 1.0 - np.cos(np.pi * i / (n - 1))
    a = np.power(a, power)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    return a / a.sum()


def _make_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    """Decaying exponential from the left edge."""
    i = np.arange(n, dtype=float)
    a = np.exp(-lam * i)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    return a / a.sum()


def _make_linear_ramp(n: int) -> np.ndarray:
    """Linearly increasing ramp (0 at left, 1 at right)."""
    i = np.arange(n, dtype=float)
    a = i + 1.0
    a = np.maximum(a, 0.0)
    return a / a.sum()

    deadline = start_time + budget_s - 0.2

    # Determine problem size ------------------------------------------------
    if 'height_sequence_1' in globals():
        a0 = np.array(height_sequence_1, dtype=float)
        a0 = np.maximum(a0, 0.0)

            a0 = np.full_like(a0, 1.0)
        a0 = a0 / a0.sum()          # normalise to sum = 1

        a = a0.copy()
        print(f"[init] using supplied sequence, n={n}")

        n = 8000

    L = 1 << ((2 * n - 1).bit_length())

    best_seq = a.copy()

    # a small helper to evaluate quickly

        obj = _objective(candidate, n, L)
        if obj < best_score:

    phase1_deadline = start_time + budget_s * 0.46   # up to ~46 % of total
    print("[phase‑1] starting long Adam optimisation")
    a, obj_after_adam, best_adam = _adam_phase(best_seq, n, L,
                                               max_iters=300_000,
                                               deadline=phase1_deadline,
                                               alpha_start=1.0,
                                               alpha_end=1e6,
                                               lr0=0.6,
                                               prune_every=0)

    # keep any improvement
    if obj_after_adam < best_score:
        best_score = obj_after_adam
        best_seq = best_adam.copy()
        print(f"[phase‑1] improved best O={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – exact sub‑gradient descent with line‑search -----------------
    phase2_deadline = start_time + budget_s * 0.71   # up to ~71 % of total
    print("[phase‑2] starting exact sub‑gradient descent")
    a, obj_sub, best_sub = _subgradient_phase(best_seq, n, L,
                                              max_iters=150_000,
                                              deadline=phase2_deadline,
                                              eta0=0.25,
                                              line_search=True,
                                              verbose=False)
    if obj_sub < best_score:
        best_score = obj_sub
        best_seq = best_sub.copy()
        print(f"[phase‑2] improved best O={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – targeted simulated annealing --------------------------------
    phase3_deadline = deadline
    print("[phase‑3] starting simulated‑annealing refinement")
    a, obj_sa, best_sa = _simulated_annealing_phase(best_seq, n, L,
                                                    max_iters=500_000,
                                                    deadline=phase3_deadline,
                                                    temperature0=0.08,
                                                    no_improve_limit=8000,
                                                    verbose=True)
    if obj_sa < best_score:
        best_score = obj_sa
        best_seq = best_sa.copy()
        print(f"[phase‑3] final best O={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final verification through the provided evaluator
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence failed:", exc)

    # Return as plain Python list (the harness expects this)
    return best_seq.tolist()
