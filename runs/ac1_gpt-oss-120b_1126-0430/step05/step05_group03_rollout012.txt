def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)

import time
import numpy as np

def _next_pow2(x):
    return 1 << (x - 1).bit_length()

def _conv_self_fft(a):
    n = a.shape[0]
    L = _next_pow2(2*n-1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*n-1]
    return conv

def _objective(a):
    n = a.size
    conv = _conv_self_fft(a)
    return float(2.0 * n * conv.max())

def _softmax(x):
    mx = np.max(x)
    e = np.exp(x - mx)
    s = e.sum()
    return e / s

def _run_adam(n, time_limit, init_a=None, seed=0):

def _anneal(a, time_budget, seed):
    start = time.time()

    n = a.size
    conv = _conv_self_fft(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    obj = best_obj

    # temperature schedule
    T0 = best_obj * 0.1  # start maybe 10% of obj
    Tf = best_obj * 0.0005
    max_iters = 500_000  # placeholder
    it = 0
    while time.time() - start < time_budget and it < max_iters:
        it += 1
        # choose source i proportionally to a_i (higher mass)
        i = rng.choice(n, p=a / a.sum())

        if j >= i:

        # choose delta up to a[i] * max_step
        max_step = 0.05  # up to 5% of source mass

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # compute new obj
        conv_new = _conv_self_fft(a_new)
        obj_new = 2.0 * n * conv_new.max()
        # temperature decay

"""
A multi‑stage optimiser for the step‑function bound.

* Stage 1 – a suite of Adam runs on several support sizes.
  The variables are kept in log‑it space so that the simplex
  constraint (non‑negative, unit sum) is fulfilled automatically.
  A soft‑max (log‑sum‑exp) approximation of the convolution maximum
  is annealed from very smooth to sharp while the learning‑rate
  decays.  Jitter and periodic restarts help to escape plateaux.

* Stage 2 – a cheap but aggressive simulated‑annealing refinement.
  Random mass‑transfer moves are accepted with a Metropolis rule,
  slowly cooling the temperature.  The move set is simple (take a
  small fraction of mass from a source index and give it to a
  target index) but works well because each evaluation of the
  objective uses a fast FFT‑based full self‑convolution.

The routine respects the required interface, prints progress
information and finally returns the best sequence found within the
allocated budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation of w with a:
    out[i] = Σ_t w[i+t]·a[t]   for 0 ≤ i < len(a).
    Implemented via FFT (real‑valued).
    """
    n = a.shape[0]          # length of a
    m = w.shape[0]          # length of w (=2·n‑1)
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a on the simplex (sum=1)."""
    n = a.size
    conv = _conv_self_fft(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces a probability vector)."""
    mx = np.max(x)
    e = np.exp(x - mx)
    s = e.sum()
    if s == 0.0:
        return np.full_like(x, 1.0 / x.size)
    return e / s


# ----------------------------------------------------------------------
def _run_adam(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
) -> tuple[list[float], float]:
    """
    Adam optimisation of a length‑``n`` sequence for at most ``time_limit``
    seconds.  Returns the best sequence (as a plain list) and its exact
    objective value.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # ---- initialise ----------------------------------------------------
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
        a = np.clip(a, 0.0, None)
    else:
        a = rng.random(n)
    if a.sum() == 0.0:
        a = np.ones(n)
    a = a / a.sum()

    # work in log‑it space – guarantees the simplex
    x = np.log(a + 1e-16)

    m = np.zeros_like(x)          # Adam first moment
    v = np.zeros_like(x)          # Adam second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # ---- schedule parameters -------------------------------------------
    beta_start = 0.2               # smoothness start
    beta_end   = 8000.0            # smoothness end (practically a hard max)
    lr_start   = 0.25
    lr_end     = 0.001
    target_iters = 600_000

    best_a = a.copy()
    best_obj = _objective(best_a)

    it = 0
    while time.time() - start < time_limit:
        it += 1

        # probabilities from logits
        a = _softmax(x)

        # exact objective
        conv = _conv_self_fft(a)
        M = conv.max()
        obj = 2.0 * n * M
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        # schedule (exponential annealing)
        t_frac = min(it / target_iters, 1.0)
        beta = beta_start * np.exp(np.log(beta_end / beta_start) * t_frac)
        lr   = lr_start   * np.exp(np.log(lr_end   / lr_start)   * t_frac)

        # -------- smoothed max weight vector ----------------------------
        if beta > 5000.0:                     # sharp enough → true arg‑max
            kmax = int(np.argmax(conv))
            w = np.zeros_like(conv)
            w[kmax] = 1.0
        else:
            shifted = conv - M                 # ≤ 0
            w = np.exp(beta * shifted)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(conv, 1.0 / conv.size)
            else:
                w = w / w_sum

        # -------- gradient w.r.t. a (smooth max) -------------------------
        g_a = 2.0 * _correlation_fft(w, a)          # (n,)

        # natural gradient in log‑it space (projection onto simplex)
        dot = np.dot(a, g_a)                        # ⟨a, g_a⟩
        g_x = a * (g_a - dot)                       # (n,)

        # -------- Adam update -------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---- occasional jitter -----------------------------------------

            jitter = rng.normal(scale=0.02, size=n)
            x += jitter

        # ---- periodic restart -------------------------------------------

            x += rng.normal(scale=0.005, size=n)
            m.fill(0.0)
            v.fill(0.0)
            print(f"[n={n}] restart at iter {it}")

    # ---- prune negligible entries ---------------------------------------
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = _objective(best_a)
        print(f"[compress] reduced length → {best_a.size}, O={best_obj:.6f}")

    return best_a.tolist(), best_obj


# ----------------------------------------------------------------------
def _anneal(
    a: np.ndarray,
    time_budget: float,
    seed: int = 0,
) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing refinement.
    Randomly transfers a small fraction of mass from a source index
    to a target index.  Moves that lower the objective are always kept,
    uphill moves are accepted with a Metropolis probability that cools
    exponentially.
    """
    start = time.time()
    rng = np.random.default_rng(seed)
    n = a.size
    conv = _conv_self_fft(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    cur_obj = best_obj

    # temperature schedule (geometric)
    T0 = best_obj * 0.10   # start at 10 % of the current value
    Tf = best_obj * 5e-4   # finish very cold
    max_iters = 400_000

    it = 0
    while time.time() - start < time_budget and it < max_iters:
        it += 1

        # choose a source proportional to its mass
        src = rng.choice(n, p=a / a.sum())
        # choose a distinct target uniformly
        tgt = rng.integers(n - 1)
        if tgt >= src:
            tgt += 1

        # transfer a random fraction (up to 5 %) of the source mass
        max_frac = 0.05
        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[src] -= delta
        a_new[tgt] += delta

        conv_new = _conv_self_fft(a_new)
        obj_new = 2.0 * n * conv_new.max()

        # temperature at this iteration
        frac = it / max_iters
        T = T0 * (Tf / T0) ** frac

        # Metropolis acceptance
        if obj_new < cur_obj or rng.random() < np.exp(-(obj_new - cur_obj) / max(T, 1e-12)):
            a = a_new
            cur_obj = obj_new
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()

        # renormalise occasionally to fight numerical drift
        if it % 10_000 == 0:
            a = a / a.sum()

    return best_a, best_obj
