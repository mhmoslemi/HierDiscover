        w = np.exp(alpha * (conv - maxb))

"""
Propose a non‑negative step function (list of heights) that minimises
    2·n·max_k (a*a)[k] / (Σ a_i)²

The method improves on the previous implementation by

* using a larger working size (up to 25 000 entries),
* an exhaustive sweep of several parametric families for a good start,
* a smooth “soft‑max” surrogate for the max‑convolution,
* an Adam‑style projected gradient descent on the simplex,
* periodic restarts and adaptive smoothing.

All helper utilities are defined at top‑level, there are no closures or
file/network access, and the routine respects the supplied time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, L: int) -> np.ndarray:
    """
    Full linear convolution a*a using FFT of length L (L >= 2·len(a)-1).
    Returns a real array of length 2·len(a)-1.
    """
    fa = np.fft.rfft(a, L)
    conv = np.fft.irfft(fa * fa, L)[: 2 * a.size - 1]
    return conv


def _conv_cross_rev(a_rev: np.ndarray, p: np.ndarray, L: int) -> np.ndarray:
    """
    Linear convolution a_rev * p (both real) using FFT of length L (L >= len(a_rev)+len(p)-1).
    Returns the full convolution (real) of length len(a_rev)+len(p)-1.
    """
    fa = np.fft.rfft(a_rev, L)
    fp = np.fft.rfft(p, L)
    return np.fft.irfft(fa * fp, L)


def _cheap_score(a: np.ndarray, L_conv: int) -> float:
    """
    Cheap upper bound (assumes Σa = 1):
        2·n·max(a*a)
    where the convolution is computed via FFT of length L_conv.
    """
    conv = _conv_full(a, L_conv)
    maxb = conv.max()
    return float(2.0 * a.size * maxb)
