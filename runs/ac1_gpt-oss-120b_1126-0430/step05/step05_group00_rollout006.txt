def project_onto_simplex(v):
    """Project onto probability simplex (sum =1, v_i >=0)."""
    n = v.shape[0]
    # sort v decreasing
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    # find rho
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _conv_max_and_grad(a, eps_factor):

    n = a.shape[0]
    L = fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[:2*n-1]
    max_conv = float(conv.max())
    eps = max(max_conv*eps_factor, 1e-12)
    shifted = conv - max_conv
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or np.isnan(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]
    grad = 4.0 * n * corr
    return max_conv, grad

def project_onto_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    # find rho
    rho_arr = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho_arr.size == 0:
        # all zeros? then return uniform?
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

The objective to be minimised is
    J(a) = 2·n·max(a * a)        (a is a non‑negative vector, sum(a)=1)

A smooth approximation of the max (log‑sum‑exp) is used so that a gradient
is available.  The gradient of 2·n·max_smooth w.r.t. a is
    ∇J = 4·n·(a ⋆ w)
where w are the soft‑max weights of the convolution entries.
All convolutions are performed by FFT (O(n log n)).

The iterate a is kept feasible by an Euclidean projection on the probability
simplex (non‑negative, sum = 1).  Adam with a decaying step size refines the
vector.  The algorithm starts from the discrete arcsine distribution (which
has an almost uniform self‑convolution) or from a provided global sequence
`height_sequence_1`.  It runs until the time budget expires, keeping the best
vector encountered.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _fft_len_for_corr(n: int) -> int:
    """
    Smallest power‑of‑two ≥ 3·n‑2, sufficient for linear convolution of
    length (2·n‑1) and correlation of length (3·n‑2).
    """
    return 1 << ((3 * n - 2).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, sum_i x_i = 1}
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    # Find the rho index
    rho_arr = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        # All entries are zero – return uniform distribution
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Compute the maximum entry of the self‑convolution a*a and the gradient
    of the smooth objective J = 2·n·smooth_max with respect to a.

    Parameters
    ----------
    a : np.ndarray
        Probability vector (sum = 1, a_i ≥ 0).
    eps_factor : float
        Multiplicative factor for the smoothing constant:
        eps = max(eps_factor * max_conv, 1e-12).

    Returns
    -------
    max_conv : float
        Maximum value of the linear convolution a*a.
    grad : np.ndarray
        Gradient of J (already multiplied by 4·n).
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    # FFT of a (zero‑padded to length L)
    a_fft = np.fft.rfft(a, L)

    # Linear convolution a*a (first 2·n‑1 entries)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # Smoothed soft‑max weights
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv                     # ≤ 0 for all entries
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or np.isnan(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # Correlation a ⋆ w (first n entries)
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # Gradient of J = 2·n·smooth_max is 4·n·corr
    grad = 4.0 * n * corr
    return max_conv, grad


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that minimises the evaluation metric.

    The routine builds a discrete arcsine distribution (or re‑uses a provided
    global sequence), then refines it with projected Adam optimisation.  The
    best vector encountered before the time budget expires is returned.
    """
    np.random.seed(seed)
    start = time.time()
    # keep a small safety margin for final evaluation
    deadline = start + budget_s - 5.0

    # --------------------------------------------------------------
    # 0. Determine the support size and the initial vector.
    # --------------------------------------------------------------
    try:
        # Use an existing sequence if the platform defined it.
        existing = globals()["height_sequence_1"]
        a = np.asarray(existing, dtype=np.float64)
        a = a / a.sum()
        n = a.shape[0]
        print(f"Using external sequence: n = {n}")
    except Exception:
        # Default: a fairly large arcsine distribution.
        n = 40000
        print(f"No external sequence – using default length n = {n}")
        idx = np.arange(n, dtype=np.float64)
        x = (idx + 0.5) / n                      # mid‑point rule on (0,1)
        raw = 1.0 / np.sqrt(x * (1.0 - x))       # arcsine density (up to const.)
        a = raw / raw.sum()                     # normalise to a probability vector

    # --------------------------------------------------------------
    # 1. Optimisation hyper‑parameters.
    # --------------------------------------------------------------
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_lr = 0.20                # base learning rate (will be decayed)
    eps_factor = 0.5              # start with relatively smooth approximation
    eps_factor_min = 5e-5
    eps_factor_decay = 0.9995

    m = np.zeros_like(a)         # first moment (Adam)
    v = np.zeros_like(a)         # second moment (Adam)

    max_iters = 500_000
    it = 0
    best_a = a.copy()
    best_score = 2.0 * n * _conv_max_and_grad(a, eps_factor)[0]

    # --------------------------------------------------------------
    # 2. Main optimisation loop.
    # --------------------------------------------------------------
    while time.time() < deadline and it < max_iters:
        it += 1

        # --- evaluate current vector (cheap) ---------------------------------
        max_conv, grad = _conv_max_and_grad(a, eps_factor)
        cheap_score = 2.0 * n * max_conv

        if cheap_score < best_score:
            best_score = cheap_score
            best_a = a.copy()
            # Logging improvement
            print(f"Iter {it:6d} – new best score {best_score:.6f}")

        # --- Adam update (on the probability simplex) -----------------------
        # first‑ and second‑moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias‑corrected moments
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # decaying learning rate
        lr_t = base_lr / (math.sqrt(it) + 1.0)

        # Adam step (direction of descent)
        step = -lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

        # provisional update and projection onto simplex
        a = _project_onto_simplex(a + step)

        # --- smoothing schedule ------------------------------------------------
        eps_factor = max(eps_factor_min, eps_factor * eps_factor_decay)

        # --- occasional random jitter to escape plateaus -----------------------
        if (it % 5000 == 0) and (np.random.rand() < 0.12):
            jitter = 0.001 * np.random.randn(n)
            a = _project_onto_simplex(a + jitter)

        # --- occasional progress report ----------------------------------------
        if it % 2000 == 0:
            elapsed = time.time() - start
            print(f"Iter {it:6d} – score {cheap_score:.6f} – eps {eps_factor:.2e} – "
                  f"time {elapsed:.1f}s")

        # Early stop if we have already reached an almost‑optimal value
        if best_score < 1.01 and it > 5000:
            print("Score sufficiently close to theoretical optimum – stopping early.")
            break

    # --------------------------------------------------------------
    # 3. Return the best sequence (as a plain Python list).
    # --------------------------------------------------------------

        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:   # defensive – should never happen
        print(f"Error during final evaluation: {exc}")
