def _smooth_obj_and_grad(x, n, L, alpha, eps=1e-20):
    # x shape (n,)
    # softmax to a
    max_x = np.max(x)
    exp_x = np.exp(x - max_x)
    a = exp_x / np.sum(exp_x)
    conv = _conv_fft(a, L)
    # smooth surrogate
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    sum_exp = np.sum(exp_scaled)
    # objective value
    f_val = (2.0 * n / alpha) * (max_scaled + np.log(sum_exp))
    # weight vector
    w = exp_scaled / sum_exp
    # gradient w.r.t a
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    grad_a = 2.0 * corr_rev[n-1:n-1+n]
    # gradient w.r.t x via softmax Jacobian
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)
    return f_val, grad_x

"""
Propose a non‑negative step‑function that (approximately) minimises

    O(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

Algorithm
---------
1️⃣  Parameterise the step‑function on the simplex via a soft‑max:
    a_i = exp(x_i) / Σ_j exp(x_j).  This guarantees positivity
    and a fixed ℓ₁‑norm, so the objective simplifies to
    O(a) = 2·n·max(conv(a,a)).

2️⃣  Optimise a smooth surrogate of the max using a continuation
    on the temperature parameter α:
       f_α(x) = (2·n/α)·log Σ_k exp(α·conv_k(a(x))).

    Its gradient w.r.t. the logits x is obtained analytically
    (correlation of the soft‑max weights with a).  We run an
    Adam optimiser on the logits, increasing α from 5 up to 2·10⁴.
    The Adam state is kept between temperature stages.

3️⃣  A cheap stochastic “mass‑transfer’’ refinement (Metropolis
    acceptance, occasional sub‑gradient steps) finishes the search
    within the remaining time budget.

All heavy work (convolutions, correlations) is performed with
FFT of length the next power‑of‑two ≥ 3 n−2, which gives exact
linear convolution without wrap‑around.  The routine returns the
best sequence it found before the time limit expires.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution a ★ a (length 2·n‑1) using FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / np.sum(e)


# ----------------------------------------------------------------------
def _smooth_obj_and_grad(x: np.ndarray, n: int, L: int, alpha: float):
    """
    Returns the value and gradient of the smooth surrogate

        f_α(x) = (2·n/α)·log Σ_k exp(α·conv_k(softmax(x)))

    with respect to the logits `x`.
    """
    a = _softmax(x)                         # a lives on the simplex
    conv = _conv_fft(a, L)                  # length 2·n‑1

    # log‑sum‑exp of α·conv  (stable)
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    sum_exp = np.sum(exp_scaled)
    f_val = (2.0 * n / alpha) * (max_scaled + np.log(sum_exp))

    # soft‑max weights for the surrogate
    w = exp_scaled / sum_exp                # shape (2·n‑1,)

    # gradient w.r.t. a : 2 · corr(w, a)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    grad_a = 2.0 * corr_rev[n - 1 : n - 1 + n]   # slice of length n

    # chain rule through the soft‑max
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)                 # shape (n,)

    return f_val, grad_x

    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] initial O={best_score:.6f}")

    alphas = [5.0, 20.0, 100.0, 500.0, 2000.0, 8000.0, 20000.0]

    O(a) = 2·n·max(conv(a,a)).

2️⃣  Optimise a smooth surrogate of the max:

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:                     # all entries are zero
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution a ★ a (length 2·n‑1) using FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max; returns a probability vector."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / np.sum(e)


# ----------------------------------------------------------------------
def _smooth_obj_and_grad(x: np.ndarray, n: int, L: int, alpha: float):
    """
    Smooth surrogate and its gradient w.r.t. logits x:

        f_α(x) = (2·n/α)·log Σ_k exp(α·conv_k(softmax(x)))

    Returns (f, ∇_x f).
    """
    a = _softmax(x)                         # a lives on the simplex
    conv = _conv_fft(a, L)                  # length 2·n‑1

    # stable log‑sum‑exp of α·conv
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    sum_exp = np.sum(exp_scaled)
    f = (2.0 * n / alpha) * (max_scaled + np.log(sum_exp))

    # weights for the surrogate
    w = exp_scaled / sum_exp                # shape (2·n‑1,)

    # ∇_a f = 2·corr(w, a)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    grad_a = 2.0 * corr_rev[n - 1 : n - 1 + n]   # slice length n

    # chain rule through the soft‑max
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)                 # shape (n,)

    return f, grad_x
