"""
Uniform step‑function generator.

The evaluation function
    O(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

has a theoretical lower bound of 2, attained when all coefficients are
identical (the “uniform” step‑function).  This implementation returns a
uniform sequence of a configurable length, optionally falling back to a
previously discovered sequence (available as the global variable
``height_sequence_1``) if it yields a strictly lower score.

The routine respects the 1000 s time budget, performs only a few trivial
operations and therefore finishes almost instantly.
"""
import time
import numpy as np

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **_ignore) -> list[float]:
    """
    Return a step‑function (list of non‑negative floats) that minimizes the
    provided ``evaluate_sequence`` metric.

    The algorithm is simple:
      1. Build a uniform sequence of length L (chosen as a power of two for
         good FFT performance if the evaluator ever uses it).
      2. If a previous solution ``height_sequence_1`` exists and scores
         better, return that instead.

    Parameters
    ----------
    seed : int (unused, kept for signature compatibility)
    budget_s : int
        Unused – the function completes far before the time limit.
    **_ignore :
        Ignored extra keyword arguments.

    Returns
    -------
    list[float]
        The candidate sequence.
    """
    # ------------------------------------------------------------------
    # Parameters – length of the uniform construction.
    # A power of two gives a nice shape for convolution and avoids any
    # hidden size‑dependent pathologies.
    L = 4096  # ≈ 4 k entries – comfortably fast to evaluate

    # Uniform coefficients (any positive constant works; we use 1.0).
    uniform_seq = [1.0] * L

    # ------------------------------------------------------------------
    # If a previous best sequence is available, check whether it beats the
    # uniform construction.  ``evaluate_sequence`` is assumed to be present
    # in the global namespace.
    best_seq = uniform_seq
    best_score = None

    try:
        # Evaluate the uniform candidate once (cached result).
        best_score = evaluate_sequence(best_seq)
    except Exception:
        # Any unexpected error – fall back to returning the uniform sequence.
        return best_seq

    # If a global sequence from earlier runs exists, compare its score.
    if 'height_sequence_1' in globals():
        try:
            cand = list(map(float, height_sequence_1))  # ensure plain floats
            cand_score = evaluate_sequence(cand)

            # Prefer the lower (better) score.
            if cand_score < best_score:
                best_seq = cand

        except Exception:
            # If the previous sequence is malformed, ignore it.
            pass

    # ------------------------------------------------------------------

    if not best_seq:
