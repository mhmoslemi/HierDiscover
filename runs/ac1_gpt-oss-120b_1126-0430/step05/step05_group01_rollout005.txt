#!/usr/bin/env python
"""
non‑negative step‑function optimiser (improved).

The optimiser works in three stages:

1️⃣  A quick scan of many low‑dimensional parametric families
    (power‑law, symmetric product‑law, exponential, logistic,
    linear‑ramp, Gaussian, Beta‑shaped) and a handful of random
    Dirichlet samples.  Each candidate is evaluated with the cheap
    bound 2·n·max(a*a) (a is already a probability vector).  The
    best few seeds are kept.

2️⃣  A long Adam‑based mirror‑descent on the probability simplex.
    Gradients are obtained from a smooth approximation of the
    max‑convolution (soft‑max over the whole convolution).  The
    step‑size follows an inverse‑sqrt schedule, a tiny back‑tracking
    line‑search guarantees monotone progress, and occasional jitter
    escapes plateaux.

3️⃣  If time permits the current distribution is up‑sampled (linear
    interpolation) to a larger support (up to 50 000 entries) and
    the descent continues.  The final (best ever seen) distribution
    is returned as a plain ``list[float]`` – it will be fed directly
    to ``evaluate_sequence`` by the outer harness.

The whole routine respects the supplied time budget (≈1000 s) and
always returns a valid sequence, even if optimisation stops early.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max (probability vector, sum = 1)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    s = e.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / z.size)
    return e / s


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-15) -> np.ndarray:
    """Inverse of soft‑max –‐ yields logits that reproduce ``a``."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """Smallest power‑of‑two ≥ 3·n‑2 (enough for linear convolution)."""
    return 1 << ((3 * n - 2).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Returns (max_conv, grad_a) for the smooth max‑convolution.

    The smooth max is a soft‑max over the full convolution with temperature
    ε = eps_factor·max_conv.  The returned gradient already includes the
    factor 4·n (the derivative of 2·n·max_conv w.r.t. a).
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    # FFT of a (zero‑padded)
    a_fft = np.fft.rfft(a, L)

    # linear self‑convolution a*a
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # smooth‑max weights
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv                 # ≤ 0
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # correlation a ⋆ w   (gradient of the smoothed max)
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # gradient (already multiplied by 4·n)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    s = new_a.sum()
    if s == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= s
    return new_a


# ----------------------------------------------------------------------
# Parametric families (all return a probability vector, sum = 1)
# ----------------------------------------------------------------------


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _sym_product_power_law_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) * (n - i)) ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _exp_decay_distribution(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _logistic_distribution(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = 1.0 / (1.0 + np.exp(alpha * (i - mid)))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _linear_decay_distribution(n: int) -> np.ndarray:
    """a_i ∝ (n‑i)  (strictly decreasing, left‑heavy)."""
    i = np.arange(n, dtype=np.float64)
    a = n - i
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _gaussian_distribution(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}   (Dirichlet‑type shape)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _random_dirichlet_distribution(n: int, concentration: float) -> np.ndarray:
    """Sample a Dirichlet(α,…,α) with given concentration (α)."""
    if concentration <= 0.0:
        concentration = 0.1
    g = np.random.gamma(concentration, 1.0, size=n)
    s = g.sum()
    if s == 0.0:
        g[:] = 1.0 / n
    else:
        g /= s
    return g

    # Also add the seed_pool entries that were not part of families
    for arr in seed_pool:
        best_by_score.append((cheap_score(arr), arr))

    # retain top 8 distinct seeds
    best_by_score.sort(key=lambda x: x[0])
    top_seeds = [arr for _, arr in best_by_score[:8]]
    print("[Init] Selected {} seeds for optimisation (n = {})".format(len(top_seeds), n))

    # ------------------------------------------------------------------
    # 1. Adam optimiser on the simplex (mirrored descent)
    # ------------------------------------------------------------------
    def _optimise_on_support(a_start: np.ndarray, max_iter: int, eps_factor0: float):
        """Run Adam optimisation from a_start, return best (a, score)."""
        nonlocal deadline
        a_dim = a_start.shape[0]
        z = _logits_from_distribution(a_start) + 0.005 * np.random.randn(a_dim)

        # Adam state
        m = np.zeros_like(z)
        v = np.zeros_like(z)
        beta1, beta2 = 0.9, 0.999
        eps_adam = 1e-8
        base_alpha = 0.4          # a bit larger than before
        eps_factor = eps_factor0
        eps_factor_min = 5e-5
        eps_decay = 0.9995

        best_local_a = _softmax(z)
        best_local_score = cheap_score(best_local_a)
        last_improve = 0

        it = 0
        while time.time() < deadline and it < max_iter:
            it += 1
            a = _softmax(z)

            max_conv, grad_a = _conv_max_and_grad(a, eps_factor)
            cheap = 2.0 * a_dim * max_conv

            # record improvement
            if cheap < best_local_score - 1e-15:
                best_local_score = cheap
                best_local_a = a.copy()
                last_improve = it

            # gradient w.r.t. logits (mirror descent)
            mu = np.dot(grad_a, a)
            grad_z = a * (grad_a - mu)

            # Adam updates (bias‑corrected)
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            # step size (inverse‑sqrt)
            alpha_t = base_alpha / (math.sqrt(it) + 1.0)
            step_dir = - m_hat / (np.sqrt(v_hat) + eps_adam)

            # back‑tracking line‑search (up to 3 halvings)
            accepted = False
            trial_alpha = alpha_t
            for _ in range(3):
                z_new = z + trial_alpha * step_dir
                z_new = np.clip(z_new, -50.0, 50.0)
                a_new = _softmax(z_new)
                new_score = cheap_score(a_new)
                if new_score <= cheap:
                    z = z_new
                    accepted = True
                    break
                trial_alpha *= 0.5

            if not accepted:
                # take a cautious step anyway
                z = z + alpha_t * step_dir
                z = np.clip(z, -50.0, 50.0)

            # occasional jitter to escape plateaux
            if it % 3000 == 0 and np.random.rand() < 0.15:
                z += 0.01 * np.random.randn(a_dim)
                m *= 0.6
                v *= 0.6

            # decay smoothing factor
            eps_factor = max(eps_factor_min, eps_factor * eps_decay)

            # restart if no improvement for a very long stretch
            if it - last_improve > 150_000:
                # re‑initialise around the current best
                z = _logits_from_distribution(best_local_a) + 0.02 * np.random.randn(a_dim)
                m.fill(0.0)
                v.fill(0.0)
                eps_factor = eps_factor0
                last_improve = it

        return best_local_a, best_local_score

    # --------------------------------------------------------------
    # 2. Run optimisation from each seed (short runs first)
    # --------------------------------------------------------------
    overall_best_a = None
    overall_best_score = math.inf

    # Phase‑1: many short runs on the current support size
    max_iter_phase1 = 120_000
    for idx, seed_arr in enumerate(top_seeds):
        if time.time() >= deadline:
            break
        a_opt, sc_opt = _optimise_on_support(seed_arr, max_iter_phase1, eps_factor0=0.12)
        if sc_opt < overall_best_score:
            overall_best_score = sc_opt
            overall_best_a = a_opt
        print("[Phase1] Seed {} → score {:+.6f}".format(idx, sc_opt))

    # --------------------------------------------------------------
    # 3. Upsample if time allows and continue refinement
    # --------------------------------------------------------------
    while time.time() < deadline and overall_best_a is not None and n < max_n:
        # increase support size by ~30 %
        new_n = min(max_n, int(n * 1.30))
        print("[Upsample] {} → {} (score {:.6f})".format(n, new_n, overall_best_score))
        overall_best_a = _interpolate_distribution(overall_best_a, new_n)
        n = new_n
        # continue optimisation on the enlarged support
        a_opt, sc_opt = _optimise_on_support(overall_best_a,
                                             max_iter=80_000,
                                             eps_factor0=0.10)
        if sc_opt < overall_best_score:
            overall_best_score = sc_opt
            overall_best_a = a_opt
            print("[Upsample] Improved after refinement: {:.6f}".format(overall_best_score))
        else:
            # no improvement – keep the interpolated one and break if time low
            overall_best_a = a_opt
            print("[Upsample] No improvement; keep current best.")

        # safety check – stop if remaining time is tiny
        if time.time() + 15.0 > deadline:
            break

    # ------------------------------------------------------------------
    # 4. Final exact evaluation with the provided evaluate_sequence()
    # ------------------------------------------------------------------
    final_seq = overall_best_a.tolist() if overall_best_a is not None else [1.0]
    try:
        final_score = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print("Final exact score = {:.6f}".format(final_score))
    except Exception as exc:
        final_score = float("inf")
        print("Error during final evaluation:", exc)

    # ------------------------------------------------------------------
    # 5. Return the best sequence discovered.
    # ------------------------------------------------------------------
    return final_seq
