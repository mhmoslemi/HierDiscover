#!/usr/bin/env python
"""
Arcsine‑bin construction.

The continuous arcsine density
    f(x) = 1 / (π·√(x·(1‑x))) ,   x∈(0,1)
has the remarkable property that its self‑convolution is the constant ½.
If we discretise the distribution by assigning to each integer bin the
exact probability mass of f on that interval, the resulting discrete
convolution is almost perfectly flat, giving a score J = 2·n·max(a*a)
very close to 1 (the theoretical optimum).

This implementation:
1. Determines a support size `n` (uses the length of the global
   `height_sequence_1` if it exists, otherwise defaults to 20 000).
2. Computes the exact bin masses using the arcsine CDF
   `F(x) = (2/π)·arcsin(√x)`.
3. If the cheap FFT‑based score is still above the target (≈1.503)
   a short Adam‑style mirror‑descent refinement on the probability simplex
   is performed.
4. Returns the best sequence found as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it representation that yields `a` after a soft‑max."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """
    Smallest power‑of‑two ≥ 3·n‑2,
    sufficient for the linear convolution of length (2·n‑1)
    and the correlation of length (3·n‑2).
    """
    return 1 << ((3 * n - 2).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a (FFT‑based)."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Returns the (smoothed) max of a*a and its gradient w.r.t. a.

    Parameters
    ----------
    a : np.ndarray
        Probability vector (non‑negative, sums to 1).
    eps_factor : float
        Multiplicative factor for the smoothing constant:
        eps = eps_factor * max(conv).

    Returns
    -------
    max_conv : float
        Maximum of the linear self‑convolution.
    grad_a : np.ndarray
        Gradient of 2·n·smooth_max (already multiplied by 4·n).
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    a_fft = np.fft.rfft(a, L)

    # linear self‑convolution (length 2·n‑1)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # soft‑max weights on the convolution entries
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv                      # for numerical stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or np.isnan(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # cross‑correlation a ⋆ w (length 3·n‑2)
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # gradient (4·n multiplied already)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a

# Main entry point required by the platform
# ----------------------------------------------------------------------

            # Mirror‑descent gradient on the logits (mirror of the simplex)
            mu = np.dot(grad_a, a_cur)
            grad_z = a_cur * (grad_a - mu)

            # Adam bias‑corrected update
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            # Step‑size schedule (inverse‑sqrt decay)
            alpha_t = base_alpha / (math.sqrt(it) + 1.0)
            step = -alpha_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            z = z + step
            np.clip(z, -50.0, 50.0, out=z)   # keep exp(z) in safe range

            # Decrease smoothing slowly
            eps_factor = max(eps_factor_min, eps_factor * eps_factor_decay)

            # Occasionally add a tiny jitter to avoid stagnation
            if it % 5000 == 0 and np.random.rand() < 0.1:
                z += 0.005 * np.random.randn(n)

        # after the loop, store the best found candidate
        a = best_a
        cheap_score = best_score
        print(f"Refinement finished – best cheap score: {cheap_score:.6f}")

    # --------------------------------------------------------------
    # 4. Return the best sequence (plain Python list).
    # --------------------------------------------------------------
    final_seq = best_a.tolist()

    # Optional exact evaluation for logging (does not affect the returned value)
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score (evaluate_sequence) = {final_score:.6f}")
    except Exception as exc:
        print(f"Error during final evaluation: {exc}")

    return final_seq
