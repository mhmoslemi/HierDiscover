def simplex_projection(v):
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]  # sort descending
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
enhanced step‑function optimiser

The optimiser works in three stages:

1.  Adam on the simplex (log‑it parametrisation) with an
    exponentially‑annealed soft‑max approximation of the max‑convolution.
    The β‑schedule is extended (up to 5·10⁵) and the learning‑rate decays
    more slowly than the reference implementation.  Multiple short restarts
    are performed for each support size to explore different basins.

2.  A greedy projected‑subgradient pass that directly minimises the exact
    max‑convolution (using a one‑hot weight).  The step‑size decays as
    1/√t and the iterate is projected onto the probability simplex after each
    update – this allows true zero entries and often yields a lower peak.

3.  Targeted simulated‑annealing (the same scheme as the reference code)
    finishes the search, using the best sequence from the previous phases as
    a starting point.

All work respects the overall 1000 s budget; intermediate progress and the
final objective (as reported by ``evaluate_sequence``) are printed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return conv_full[: 2 * n - 1]


def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation g_i = Σ_t w[i+t]·a[t]  for i=0,…,n‑1.
    Implemented via FFT.
    """
    n = a.shape[0]
    m = w.shape[0]  # = 2n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – used by the projected sub‑gradient stage
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    if v.sum() == 0.0:
        return np.full(n, 1.0 / n, dtype=v.dtype)

    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries negative – fall back to uniform
        return np.full(n, 1.0 / n, dtype=v.dtype)

    rho = ind[cond][-1]                      # last true index
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


# ----------------------------------------------------------------------
# Adam optimiser on the log‑it simplex (improved hyper‑parameters)
# ----------------------------------------------------------------------


def _adam_optimize(
    n: int,
    time_limit: float,
    seed: int = 0,
    init_a: np.ndarray | None = None,
) -> tuple[list[float], float]:
    """
    Adam optimisation on the simplex (log‑it parametrisation).

    Returns the best sequence (as a plain Python list) and its objective
    value (2·n·max(conv) – sum is already normalised to 1).
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # initialise a (probability vector)
    if init_a is not None:
        a = np.asarray(init_a, dtype=float).copy()
        if a.size != n:
            a = rng.random(n)
    else:
        a = rng.random(n)

    a = np.clip(a, 0.0, None)
    if a.sum() < 1e-12:
        a = np.ones(n, dtype=float) / n
    else:
        a = a / a.sum()

    # logits (log‑it) representation
    x = np.log(a + 1e-16)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # schedule parameters (more aggressive than the reference)
    beta_start = 0.5
    beta_end = 5e5          # sharper smoothing → closer to the exact max
    lr_start = 0.18
    lr_end = 0.0004

    best_a = a.copy()
    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())

    it = 0
    while time.time() - start < time_limit:
        it += 1

        # current point on the simplex
        a = _softmax(x)

        conv = _conv_self_fft(a)
        M = conv.max()
        obj = float(2.0 * n * M)

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[Adam n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        # -------------------- schedules --------------------
        t = min((time.time() - start) / time_limit, 1.0)
        beta = beta_start * np.exp(np.log(beta_end / beta_start) * t)
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t)

        # -------------------- smooth‑max weights --------------------
        if beta > 1e5:                     # already very sharp → use exact max
            w = np.zeros_like(conv)
            w[int(np.argmax(conv))] = 1.0
        else:
            shifted = conv - M               # ≤ 0
            w = np.exp(beta * shifted)
            s = w.sum()
            if s == 0.0:
                w = np.ones_like(conv) / conv.size
            else:
                w /= s

        # gradient of smooth max w.r.t. a
        g_a = 2.0 * _correlation_fft(w, a)

        # natural gradient in log‑it space (mirrored gradient)
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape plateaus
        if it % 4000 == 0:
            jitter = rng.normal(scale=0.01, size=n)
            x += jitter

        # periodic restart from the current best (helps escape bad basins)
        if it % 60000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.004, size=n)   # small noise
            m.fill(0.0)
            v.fill(0.0)

    # --------------------------------------------------------------
    # compress negligible entries (helps the final objective)
    thresh = 1e-13
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a /= best_a.sum()
        best_obj = float(2.0 * best_a.size * _conv_self_fft(best_a).max())
        print(f"[compress] reduced length → {best_a.size}, O={best_obj:.6f}")

    return best_a.tolist(), best_obj


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact max) – a deterministic refinement
# ----------------------------------------------------------------------


def _proj_subgradient_optimize(
    n: int,
    time_limit: float,
    seed: int = 0,
    init_a: np.ndarray | None = None,
) -> tuple[list[float], float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution.
    Works directly on probability vectors (a ≥ 0, Σa = 1).
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    if init_a is not None:
        a = np.asarray(init_a, dtype=float).copy()
        if a.size != n:
            a = rng.random(n)
    else:
        a = rng.random(n)

    a = np.clip(a, 0.0, None)
    if a.sum() < 1e-12:
        a = np.ones(n, dtype=float) / n
    else:
        a = a / a.sum()

    best_a = a.copy()
    best_obj = float(2.0 * n * _conv_self_fft(a).max())

    it = 0
    while time.time() - start < time_limit:
        it += 1

        conv = _conv_self_fft(a)
        k_max = int(np.argmax(conv))

        # one‑hot weight for the active constraint
        w = np.zeros_like(conv)
        w[k_max] = 1.0

        # sub‑gradient of the max‑convolution (exact)
        g = 2.0 * _correlation_fft(w, a)

        # step‑size schedule: η = η0 / sqrt(it)
        eta0 = 0.7
        eta = eta0 / np.sqrt(it)

        a_new = a - eta * g
        a_new = _project_onto_simplex(a_new)

        obj_new = float(2.0 * n * _conv_self_fft(a_new).max())
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new

        # occasional small random perturbation to avoid stagnation
        if it % 20000 == 0:
            noise = rng.normal(scale=0.005, size=n)
            a = _project_onto_simplex(a + noise)

    return best_a.tolist(), best_obj


# ----------------------------------------------------------------------
# Targeted simulated‑annealing (same spirit as the reference)
# ----------------------------------------------------------------------


def _run_sa(
    init_seq: list[float],
    time_limit: float,
    seed: int = 0,
) -> tuple[list[float], float]:
    """
    Targeted simulated‑annealing refinement.
    Moves a modest mass from indices that contribute most to the current
    maximal convolution entry to random other indices.
    """
    start = time.time()
    rng = np.random.default_rng(seed)
    a = np.asarray(init_seq, dtype=float)
    if a.sum() < 1e-12:
        a = np.ones_like(a) / a.size
    else:
        a = a / a.sum()
    n = a.shape[0]

    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())
    best_seq = a.copy()

    # temperature schedule (very slow cooling)
    T = best_obj * 0.12 + 1e-6
    T_min = best_obj * 1e-6 + 1e-9
    decay = 0.997

    it = 0
    while time.time() - start < time_limit:
        it += 1
        k_max = int(np.argmax(conv))

        # contributions a[i]·a[k_max‑i]
        i_vals = np.arange(n)
        j_vals = k_max - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i_vals[mask]] * a[j_vals[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            probs = contrib / total_contrib
            i = rng.choice(n, p=probs)
        else:
            i = rng.integers(n)

        # destination index (different from i)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # transfer a modest fraction of mass
        delta = a[i] * rng.random() * 0.07   # up to 7 %
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.clip(a_new, 0.0, None)
        a_new /= a_new.sum()

        conv_new = _conv_self_fft(a_new)
        obj_new = float(2.0 * n * conv_new.max())

        # Metropolis acceptance
        if obj_new < best_obj - 1e-13:
            a, conv, best_obj, best_seq = a_new, conv_new, obj_new, a_new.copy()
            print(f"[SA it={it}] new best O={best_obj:.6f}")
        else:
            prob = np.exp(-(obj_new - best_obj) / T) if T > 0 else 0.0
            if rng.random() < prob:
                a, conv = a_new, conv_new

        # temperature decay
        if it % 1500 == 0:
            T = max(T * decay, T_min)

        # global jitter occasionally
        if it % 8000 == 0:
            noise = rng.normal(scale=0.008, size=n)
            a = _project_onto_simplex(a + noise)
            conv = _conv_self_fft(a)

    # final compression of tiny entries
    thresh = 1e-13
    mask = best_seq > thresh
    if mask.sum() < best_seq.size:
        comp = best_seq[mask]
        comp /= comp.sum()
        comp_obj = float(2.0 * comp.size * _conv_self_fft(comp).max())
        if comp_obj < best_obj:
            best_seq, best_obj = comp, comp_obj
            print(f"[SA compress] reduced length → {comp.size}, O={best_obj:.6f}")

    return best_seq.tolist(), best_obj


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **_ignore,
) -> list[float]:
    """
    Orchestrates the optimisation pipeline within the allotted time budget.
    Returns a list of non‑negative heights (the step‑function).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Load a previously discovered construction if available.
    # ------------------------------------------------------------------
    init_seq = None
    if "height_sequence_1" in globals():
        seq0 = np.asarray(height_sequence_1, dtype=float)
        seq0 = np.clip(seq0, 0.0, None)
        if seq0.sum() < 0.01:
            seq0 = np.ones_like(seq0)
        seq0 = seq0 / seq0.sum()
        init_seq = seq0
        n0 = seq0.size
    else:
        n0 = None

    # Candidate lengths – we explore up to 12 000 points.
    lengths = [2000, 4000, 6000, 8000, 10000, 12000]
    if n0 is not None and n0 not in lengths:
        lengths.insert(0, n0)

    best_seq = None
    best_obj = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – Adam optimisation (multiple restarts per size)
    # ------------------------------------------------------------------
    for idx, n in enumerate(lengths):
        now = time.time()
        remaining = deadline - now
        if remaining <= 2.0:
            break

        # allocate a generous slice of the remaining budget
        if idx == 0:
            alloc = remaining * 0.55
        else:
            alloc = remaining * 0.35 / (len(lengths) - idx)

        # split the allocation into several short Adam runs (30 s each) –
        # this gives us a diversity of local minima.
        n_runs = max(1, int(alloc // 30))
        per_run = alloc / n_runs
        for run in range(n_runs):
            sub_seed = seed + idx * 1000 + run * 123
            init = None
            if run == 0 and idx == 0 and init_seq is not None and n == init_seq.size:
                init = init_seq
            seq, val = _adam_optimize(
                n=n,
                time_limit=per_run,
                seed=sub_seed,
                init_a=init,
            )
            if val < best_obj:
                best_obj, best_seq = val, np.asarray(seq, dtype=float)
                print(f"[global] new overall best O={best_obj:.6f} (n={n})")

        if time.time() >= deadline - 0.5:
            break

    # ------------------------------------------------------------------
    # Phase 2 – Deterministic projected sub‑gradient optimisation
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if best_seq is not None and remaining > 4.0:
        seq_ref, val_ref = _proj_subgradient_optimize(
            n=best_seq.size,
            time_limit=remaining * 0.5,
            seed=seed + 777,
            init_a=best_seq,
        )
        if val_ref < best_obj:
            best_obj, best_seq = val_ref, np.asarray(seq_ref, dtype=float)
            print(f"[subgrad] improved to O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Targeted simulated‑annealing fine‑tuning
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if best_seq is not None and remaining > 5.0:
        seq_refined, val_refined = _run_sa(
            best_seq.tolist(),
            time_limit=remaining,
            seed=seed + 999,
        )
        if val_refined < best_obj:
            best_obj, best_seq = val_refined, np.asarray(seq_refined, dtype=float)
            print(f"[final] after SA new best O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never happen but guarantees a non‑empty result.
    # ------------------------------------------------------------------
    if best_seq is None or best_seq.size == 0:
        best_seq = np.ones(200, dtype=float) / 200.0
        best_obj = float("inf")

    # ------------------------------------------------------------------
    # Final sanity check with the provided evaluator.
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_seq.tolist()
