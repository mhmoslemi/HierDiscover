"""
Improved search for a non‑negative step‑function sequence that minimises

    O(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm combines:

* Phase 0 – many short random restarts with a variety of Dirichlet
  concentration parameters and a brief Adam optimisation.
* Phase 1 – a long Adam optimisation on the smooth log‑sum‑exp surrogate.
* Phase 2 – exact projected sub‑gradient descent with a back‑tracking line‑search.
* Phase 3 – a custom simulated‑annealing that uses two move types
  (mass‑transfer and index‑swap) together with occasional sub‑gradient
  corrections.

The code keeps the best solution seen so far and finally checks it with
the provided ``evaluate_sequence`` function.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        # All non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full a*a convolution using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int, L: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)**2)."""
    sum_a = a.sum()
    if sum_a < 1e-12:
        return np.inf
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max() / (sum_a * sum_a)


# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of the maximal convolution entry with respect to *a*."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    # Because denominator (sum_a**2) appears in objective,
    # the total gradient of O = 2n*max/ sum^2 is:
    # dO = (2n / sum^2) * dg - (4n * max / sum^3) * 1
    # However we keep the projected sub‑gradient descent simple
    # and let the line‑search handle the scaling.
    return g


# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, conv: np.ndarray,
                     alpha: float, L: int) -> np.ndarray:
    """Gradient of (2·n/α)·log‑sum‑exp(α·conv) w.r.t. *a*."""
    # compute soft‑max weights for each convolution index
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / exp_scaled.sum()

    # correlation w ⋆ a (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n_len = a.size
    grad = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]
    return grad


# ----------------------------------------------------------------------
def _adam_phase(a: np.ndarray, n: int, L: int,
                max_iters: int, deadline: float,
                alpha_start: float = 5.0,
                alpha_end: float = 20000.0,
                lr0: float = 0.6,
                beta1: float = 0.9,
                beta2: float = 0.999,
                eps: float = 1e-12,
                prune_every: int = 5000,
                prune_percentile: float = 1.0) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Adam optimiser on the smooth (log‑sum‑exp) surrogate.
    Returns (final a, best objective value seen, best a).
    """
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_obj = _objective(a, n, L)

    total_iters = max_iters
    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _conv_fft(a, L)
        # explicit objective (non‑smooth) for book‑keeping
        cur_obj = 2.0 * n * conv.max() / (a.sum() ** 2)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            if it % 1000 == 0:
                print(f"[adam] iter {it} new best O={best_obj:.6f}")

        # schedule for the smoothing parameter α
        frac = it / total_iters
        # exponential schedule, steeper towards the end
        alpha = alpha_start * (alpha_end / alpha_start) ** frac

        grad = _smooth_gradient(a, conv, alpha, L)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        lr = lr0 / math.sqrt(it + 1)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape flat plateaux
        if np.random.rand() < 0.001:
            a_new += np.random.randn(n) * 1e-4

        a_new = _project_to_simplex(a_new)

        # optional pruning of near‑zero entries
        if prune_every and it % prune_every == 0 and it > 0:
            thresh = np.percentile(a_new, prune_percentile)
            a_new[a_new < thresh] = 0.0
            a_new = _project_to_simplex(a_new)

        a = a_new

    return a, best_obj, best_a


# ----------------------------------------------------------------------
def _subgradient_phase(a: np.ndarray, n: int, L: int,
                       max_iters: int, deadline: float,
                       eta0: float = 0.3,
                       line_search: bool = True,
                       verbose: bool = False) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Exact projected sub‑gradient descent on the true objective.
    Returns (final a, best objective, best a).
    """
    best_a = a.copy()
    best_obj = _objective(a, n, L)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max() / (a.sum() ** 2)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            if verbose and it % 5000 == 0:
                print(f"[sub‑grad] iter {it} improve O={best_obj:.6f}")

        g = _subgradient_max_conv(a, conv)

        # geometric decay of the step size
        eta = eta0 * (0.9999 ** it)

        if line_search:
            step = eta
            for _ in range(5):
                cand = _project_to_simplex(a - step * g)
                cand_obj = _objective(cand, n, L)
                if cand_obj < cur_obj:
                    a = cand
                    break
                step *= 0.5
            else:
                a = _project_to_simplex(a - eta * g)
        else:
            a = _project_to_simplex(a - eta * g)

    return a, best_obj, best_a

def _simulated_annealing_phase_custom(a_init: np.ndarray, n: int, L: int,
                                     max_iters: int, deadline: float,
                                     temperature0: float = 0.07,
                                     no_improve_limit: int = 5000,
                                     verbose: bool = True) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Simulated annealing with two move types:
    * transfer a fraction of mass from a source index to a destination;
    * swap the values of two indices.
    The walk is occasionally nudged by an exact sub‑gradient step.
    """
    a = a_init.copy()
    conv = _conv_fft(a, L)
    cur_obj = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()
    best_obj = cur_obj

    T = temperature0
    no_improve = 0

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # identify indices that actually take part in the current maximal conv entry
        kmax = int(np.argmax(conv))
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        participating = np.arange(i_start, i_end + 1, dtype=int)
        # keep only those with non‑negligible mass
        participating = participating[a[participating] > 1e-14]
        if participating.size == 0:
            src = np.random.randint(n)
        else:
            src = np.random.choice(participating)

        # choose move type
        if np.random.rand() < 0.75:
            # ------- mass‑transfer move -------
            dst = np.random.randint(n)
            while dst == src:
                dst = np.random.randint(n)

            # size of the transferred chunk – up to ~10 % of the source mass
            delta = a[src] * np.random.rand() * (0.09 + 0.12 * np.random.rand())
            a_new = a.copy()
            a_new[src] -= delta
            a_new[dst] += delta
            a_new = _project_to_simplex(a_new)
        else:
            # ------- swap move -------
            i = np.random.randint(n)
            j = np.random.randint(n)
            while j == i:
                j = np.random.randint(n)
            a_new = a.copy()
            a_new[i], a_new[j] = a[j], a[i]
            # projection guarantees sum=1 after floating‑point drift
            a_new = _project_to_simplex(a_new)

        conv_new = _conv_fft(a_new, L)
        new_obj = 2.0 * n * conv_new.max() / (a_new.sum() ** 2)

        # Metropolis acceptance
        if new_obj < cur_obj or np.random.rand() < math.exp((cur_obj - new_obj) / max(T, 1e-15)):
            a, conv, cur_obj = a_new, conv_new, new_obj
            no_improve = 0
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                if verbose:
                    print(f"[SA] it {it} new global best O={best_obj:.6f}")
        else:
            no_improve += 1

        # occasional exact sub‑gradient step to keep the walk directed
        if it % 8 == 0:
            g = _subgradient_max_conv(a, conv)
            eta_loc = 0.06 * (1.0 - it / max_iters)
            a = _project_to_simplex(a - eta_loc * g)
            conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max() / (a.sum() ** 2)
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                if verbose:
                    print(f"[SA‑sg] it {it} sub‑grad improved O={best_obj:.6f}")

        # temperature decay (geometric)
        T *= 0.9997

        # restart / reheating if we have been stuck too long
        if no_improve >= no_improve_limit:
            # light random jitter around the current best
            a = best_a.copy()
            a += np.random.randn(n) * 0.0010
            a = np.clip(a, 0, None)
            a = _project_to_simplex(a)
            conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max() / (a.sum() ** 2)
            no_improve = 0
            if verbose:
                print(f"[SA] it {it} restart after stagnation, O={cur_obj:.6f}")

    return a, best_obj, best_a

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        lr = lr0 / math.sqrt(it + 1)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # tiny random jitter to help escape plateaux
        if np.random.rand() < 0.001:
            a_new += np.random.randn(n) * 1e-4

        a_new = _project_to_simplex(a_new)

        # occasional pruning of near‑zero entries
        if prune_every and it % prune_every == 0 and it > 0:
            thresh = np.percentile(a_new, prune_percentile)
            a_new[a_new < thresh] = 0.0
            a_new = _project_to_simplex(a_new)

        a = a_new

    return a, best_obj, best_a


# ----------------------------------------------------------------------
def _subgradient_phase(a: np.ndarray, n: int, L: int,
                       max_iters: int, deadline: float,
                       eta0: float = 0.3,
                       line_search: bool = True,
                       verbose: bool = False) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Exact projected sub‑gradient descent on the true objective.
    Returns (final a, best objective, best a).
    """
    best_a = a.copy()
    best_obj = _objective(a, n, L)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _conv_fft(a, L)
        cur_obj = _objective(a, n, L)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            if verbose and it % 5000 == 0:
                print(f"[sub‑grad] iter {it} improved O={best_obj:.6f}")

        g = _subgradient_max_conv(a, conv)

        # geometrically decaying step size
        eta = eta0 * (0.9999 ** it)

        if line_search:
            step = eta
            for _ in range(5):
                cand = _project_to_simplex(a - step * g)
                cand_obj = _objective(cand, n, L)
                if cand_obj < cur_obj:
                    a = cand
                    break
                step *= 0.5
            else:
                a = _project_to_simplex(a - eta * g)
        else:
            a = _project_to_simplex(a - eta * g)

    return a, best_obj, best_a

def _simulated_annealing_phase_custom(a_init: np.ndarray, n: int, L: int,
                                     max_iters: int, deadline: float,
                                     temperature0: float = 0.07,
                                     no_improve_limit: int = 5000,
                                     verbose: bool = True) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Simulated annealing with two move types:
      1. transfer a fraction of mass from a source index to a destination;
      2. swap the values of two indices.
    Periodically a sub‑gradient step is performed to keep the walk
    directed.
    """
    a = a_init.copy()
    conv = _conv_fft(a, L)
    cur_obj = _objective(a, n, L)

    best_a = a.copy()
    best_obj = cur_obj

    T = temperature0
    no_improve = 0

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # ----- choose a source index that actually participates in the peak -----
        kmax = int(np.argmax(conv))
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        part = np.arange(i_start, i_end + 1, dtype=int)
        part = part[a[part] > 1e-14]
        src = np.random.choice(part) if part.size else np.random.randint(n)

        # ----- decide which move to perform -----
        if np.random.rand() < 0.75:                                    # mass‑transfer
            dst = np.random.randint(n)
            while dst == src:
                dst = np.random.randint(n)
            delta = a[src] * np.random.rand() * (0.09 + 0.12 * np.random.rand())
            a_new = a.copy()
            a_new[src] -= delta
            a_new[dst] += delta
            a_new = _project_to_simplex(a_new)
        else:                                                          # swap
            i = np.random.randint(n)
            j = np.random.randint(n)
            while j == i:
                j = np.random.randint(n)
            a_new = a.copy()
            a_new[i], a_new[j] = a[j], a[i]
            a_new = _project_to_simplex(a_new)

        conv_new = _conv_fft(a_new, L)
        new_obj = _objective(a_new, n, L)

        # Metropolis acceptance
        if new_obj < cur_obj or np.random.rand() < math.exp((cur_obj - new_obj) / max(T, 1e-15)):
            a, conv, cur_obj = a_new, conv_new, new_obj
            no_improve = 0
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                if verbose:
                    print(f"[SA] it {it} new global best O={best_obj:.6f}")
        else:
            no_improve += 1

        # ----- occasional exact sub‑gradient correction -----
        if it % 8 == 0:
            g = _subgradient_max_conv(a, conv)
            eta_loc = 0.06 * (1.0 - it / max_iters)
            a = _project_to_simplex(a - eta_loc * g)
            conv = _conv_fft(a, L)
            cur_obj = _objective(a, n, L)
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                if verbose:
                    print(f"[SA‑SG] it {it} sub‑grad improved O={best_obj:.6f}")

        # temperature decay (geometric)
        T *= 0.9997

        # restart if we have stalled for a while
        if no_improve >= no_improve_limit:
            a = best_a.copy()
            a += np.random.randn(n) * 0.0010
            a = np.clip(a, 0, None)
            a = _project_to_simplex(a)
            conv = _conv_fft(a, L)
            cur_obj = _objective(a, n, L)
            no_improve = 0
            if verbose:
                print(f"[SA] it {it} restart after stagnation, O={cur_obj:.6f}")

    return a, best_obj, best_a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Main entry point required by the evaluation harness.
    Searches for a step‑function that minimises the given objective
    and returns the best sequence found as a list of Python floats.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.2                     # tiny safety margin

    # ------------------------------------------------------------------
    # problem size – we deliberately pick a larger n than the previous
    # implementation (≈ 8200) to give the optimiser more freedom.
    default_n = 8192
    n = kwargs.get("n", default_n)

    if "height_sequence_1" in globals():
        a0 = np.array(height_sequence_1, dtype=float)
        a0 = np.maximum(a0, 0.0)
        if a0.sum() == 0.0:
            a0 = np.full_like(a0, 1.0)
        n0 = a0.size
        if n0 >= default_n:
            n = n0
        a0 = a0 / a0.sum()
        a = a0.copy()
        print(f"[init] using supplied sequence length n={n}")
    else:
        a = np.full(n, 1.0 / n)
        print(f"[init] using uniform start, n={n}")

    # FFT length – next power of two ≥ 3·n‑2 (enough for the convolutions)
    L = 1 << ((3 * n - 2).bit_length())

    # ------------------------------------------------------------------
    # Phase 0 – cheap random restarts with varied Dirichlet concentrations
    phase0_dead = start + budget_s * 0.12
    best_seq = a.copy()
    best_score = _objective(best_seq, n, L)

    conc_options = [0.05, 0.2, 0.4, 0.6, 0.9, 1.2]
    while time.time() < phase0_dead:
        conc = float(np.random.choice(conc_options))
        a_rand = np.random.gamma(conc, 1.0, size=n)
        a_rand /= a_rand.sum()
        sub_dead = time.time() + 2.5                     # a few seconds per trial
        a_tmp, _, a_best = _adam_phase(a_rand, n, L,
                                       max_iters=800,
                                       deadline=sub_dead,
                                       alpha_start=5.0,
                                       alpha_end=3000.0,
                                       lr0=0.9)
        cur_obj = _objective(a_best, n, L)
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = a_best.copy()
            print(f"[phase0] new best O={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – long Adam optimisation on the smooth surrogate
    phase1_dead = start + budget_s * 0.45
    a, obj_adam, best_adam = _adam_phase(best_seq, n, L,
                                         max_iters=400_000,
                                         deadline=phase1_dead,
                                         alpha_start=5.0,
                                         alpha_end=100_000.0,
                                         lr0=0.6)
    if obj_adam < best_score:
        best_score = obj_adam
        best_seq = best_adam.copy()
        print(f"[phase1] improved best O={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – exact projected sub‑gradient descent with line search
    phase2_dead = start + budget_s * 0.75
    a, obj_sub, best_sub = _subgradient_phase(best_seq, n, L,
                                              max_iters=250_000,
                                              deadline=phase2_dead,
                                              eta0=0.25,
                                              line_search=True,
                                              verbose=False)
    if obj_sub < best_score:
        best_score = obj_sub
        best_seq = best_sub.copy()
        print(f"[phase2] improved best O={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – enhanced simulated‑annealing refinement
    phase3_dead = deadline
    a, obj_sa, best_sa = _simulated_annealing_phase_custom(best_seq, n, L,
                                                          max_iters=300_000,
                                                          deadline=phase3_dead,
                                                          temperature0=0.08,
                                                          no_improve_limit=6000,
                                                          verbose=True)
    if obj_sa < best_score:
        best_score = obj_sa
        best_seq = best_sa.copy()
        print(f"[phase3] final best O={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final verification through the provided evaluator
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence failed:", exc)

    return best_seq.tolist()
