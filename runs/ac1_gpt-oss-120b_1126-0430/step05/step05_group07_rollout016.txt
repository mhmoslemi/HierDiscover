def solve_lp(a, M):
    n = a.shape[0]
    # Build sparse matrix A_ub
    data = [np.full(n, a_i) for a_i in a]
    offsets = [-i for i in range(n)]
    A = scipy.sparse.diags(data, offsets, shape=(2*n-1, n), format='csc')
    b = np.full(2*n-1, M)
    c = -np.ones(n)  # maximize sum -> minimize -sum
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=(0, None),
                                 method='highs', options={'presolve': True})
    if not res.success:
        # fallback: return a (no change)
        return a
    g = res.x
    # ensure non-negative (should be)
    g = np.maximum(g, 0)
    # scale to sum=1
    sum_g = g.sum()
    if sum_g == 0:
        # fallback to uniform
        g = np.full(n, 1.0/n)
        sum_g = 1.0
    return g / sum_g

import time, math, numpy as np
import scipy.optimize
import scipy.linalg

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Search for a non‑negative step function (list of heights) that
    minimises  O(a) = 2·n·max(conv(a,a)) / (sum(a)**2).

    The algorithm consists of three stages:
    1. Variance flattening – Adam optimisation of the variance of the
       full auto‑convolution, driving the whole convolution vector
       towards a constant value.
    2. Direct sub‑gradient descent on the true max‑norm of the
       convolution, using a decreasing step size and a projection onto
       the probability simplex.
    3. Stochastic local refinement – cheap “mass‑transfer’’ Monte‑Carlo
       moves with a Metropolis acceptance rule, plus occasional
       smoothing, to escape tiny local plateaux.

    Multiple starts (uniform, triangular, sine‑window, and the supplied
    `height_sequence_1` if present) are explored; the best sequence
    found before the time budget expires is returned.
    """
    # ------------------------------------------------------------------
    import numpy as np, math, time
    from typing import List

    # --- helper: projection onto the simplex (∑x = 1, x≥0) ----------
    def _project_to_simplex(v: np.ndarray) -> np.ndarray:
        """Euclidean projection of v onto the probability simplex."""
        n = v.shape[0]
        # sort v in descending order
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1.0
        ind = np.arange(1, n + 1)
        rho = np.where(u - cssv / ind > 0)[0]
        if rho.size == 0:
            # all zero case – return uniform
            return np.full_like(v, 1.0 / n)
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
        w = np.maximum(v - theta, 0.0)
        return w

    # --- helper: full convolution using FFT (real) --------------------
    def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
        """Return the full auto‑convolution a★a (length 2·n‑1)."""
        fft_a = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(fft_a * fft_a, n=L)
        return conv_full[: 2 * a.size - 1]

    # --- gradient of the variance of the convolution ------------------
    def _grad_variance(a: np.ndarray, conv: np.ndarray, n: int, L: int) -> np.ndarray:
        """∇_a Σ_k (conv_k - μ)^2 , μ = 1/(2n-1) for ∑a=1."""
        mu = 1.0 / (2 * n - 1)
        diff = conv - mu
        a_rev = a[::-1]
        fft_diff = np.fft.rfft(diff, n=L)
        fft_a_rev = np.fft.rfft(a_rev, n=L)
        corr_rev = np.fft.irfft(fft_diff * fft_a_rev, n=L)
        start = n - 1
        grad = 4.0 * corr_rev[start:start + n]
        return grad

    # --- sub‑gradient of max(conv) ------------------------------------
    def _subgrad_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
        """Return a sub‑gradient of max(conv(a,a))."""
        max_val = conv.max()
        ks = np.flatnonzero(conv == max_val)
        n = a.shape[0]
        g = np.zeros_like(a)
        for k in ks:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            # i runs i_start .. i_end inclusive
            i_idx = np.arange(i_start, i_end + 1, dtype=int)
            j_idx = k - i_idx
            # contribution 2*a_j for each i (since derivative of a_i*a_j + a_j*a_i)
            g[i_idx] += 2.0 * a[j_idx]
        if ks.size > 0:
            g /= ks.size
        return g

    # ------------------------------------------------------------------
    # initialise --------------------------------------------------------
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # keep a small safety margin

    # Determine problem size and initial sequence -------------------------
    initial_candidates = []

    # 1) use supplied height_sequence_1 if present
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
        n = base.size
        a0 = base / base.sum()
        initial_candidates.append(a0)
        print(f"[init] start from supplied height_sequence_1, n={n}")
    except NameError:
        pass

    # 2) uniform
    if 'n' not in locals():
        n = 4600
    uniform = np.full(n, 1.0 / n)
    initial_candidates.append(uniform)

    # 3) triangular (linear ramp up then down)
    tri = np.arange(1, n // 2 + 1)
    if n % 2 == 1:
        tri = np.concatenate([tri, tri[-2::-1]])
    else:
        tri = np.concatenate([tri, tri[::-1]])
    tri = tri.astype(float)
    tri = tri / tri.sum()
    initial_candidates.append(tri)

    # 4) sine window (half‑sine, zero at the ends)
    idx = np.arange(n)
    sine = np.sin(np.pi * (idx + 0.5) / n)
    sine = sine / sine.sum()
    initial_candidates.append(sine)

    # keep only unique candidates (in case of duplicates)
    candidates = []
    seen = set()
    for cand in initial_candidates:
        key = tuple(np.round(cand, 8))
        if key not in seen:
            seen.add(key)
            candidates.append(cand)

    # FFT length – next power of two not smaller than 3·n‑2
    L = 1 << ((3 * n - 2).bit_length())

    # keep global best
    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # function to evaluate objective (sum is already 1 → O = 2·n·max)
    def _objective(a: np.ndarray) -> float:
        conv = _conv_fft(a, L)
        return 2.0 * n * conv.max()

    # ------------------------------------------------------------------
    # Core optimisation routine for a single start ---------------------
    def _optimise_start(a_start: np.ndarray) -> np.ndarray:
        a = a_start.copy()
        # make sure we are on the simplex
        a = _project_to_simplex(a)

        # Stage 1 – variance flattening (Adam) -------------------------
        max_iter_stage1 = 200_000
        beta1, beta2 = 0.9, 0.999
        eps = 1e-12
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        lr0 = 0.8                # slightly larger learning rate
        lam_entropy = 1e-5       # tiny entropy to keep strict positivity

        stage1_deadline = start_time + (budget_s * 0.3)
        it = 0
        while time.time() < stage1_deadline and it < max_iter_stage1:
            conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max()
            # record best
            nonlocal_best_update(cur_obj, a)

            # gradient of variance
            grad = _grad_variance(a, conv, n, L)
            # small entropy term (helps avoid exact zeros)
            grad += lam_entropy * (np.log(a + 1e-20) + 1.0)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** (it + 1))
            v_hat = v / (1.0 - beta2 ** (it + 1))
            lr = lr0 / math.sqrt(it + 1)
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = _project_to_simplex(a_new)
            it += 1
            if it % 50_000 == 0:
                print(f"[stage‑1] iter {it:,} O={cur_obj:.6f}")

        # Stage 2 – direct sub‑gradient descent on max(conv) ------------
        max_iter_stage2 = 250_000
        stage2_deadline = start_time + (budget_s * 0.6)
        it2 = 0
        eta0 = 0.12        # initial sub‑gradient step size
        while time.time() < stage2_deadline and it2 < max_iter_stage2:
            conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max()
            nonlocal_best_update(cur_obj, a)

            # sub‑gradient
            g = _subgrad_max_conv(a, conv)

            # step size decays like eta0 / sqrt(t)
            eta = eta0 / math.sqrt(it2 + 1)
            a_new = a - eta * g
            a = _project_to_simplex(a_new)

            # occasional jitter to avoid stagnation
            if np.random.rand() < 0.001:
                a += np.random.randn(n) * 1e-4
                a = _project_to_simplex(a)

            it2 += 1
            if it2 % 50_000 == 0:
                print(f"[stage‑2] subgrad iter {it2:,} O={cur_obj:.6f}")

        # Stage 3 – stochastic local refinement (mass transfer) ---------
        # This is deliberately lightweight – we keep it running until the overall deadline.
        subgrad_interval = 7
        temperature0 = 0.03
        max_iter_local = 200_000
        it_loc = 0
        T = temperature0

        while time.time() < deadline and it_loc < max_iter_local:
            # random mass‑transfer move
            i, j = np.random.randint(n, size=2)
            if i != j and a[i] > 1e-14:
                delta = a[i] * np.random.rand() * 0.12
                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta
                a_new = _project_to_simplex(a_new)
                conv_new = _conv_fft(a_new, L)
                new_obj = 2.0 * n * conv_new.max()
                # Metropolis acceptance
                if new_obj < cur_obj or np.random.rand() < math.exp((cur_obj - new_obj) / max(T, 1e-12)):
                    a = a_new
                    cur_obj = new_obj
                    nonlocal_best_update(cur_obj, a)

            # sub‑gradient step every few iterations (acts as a regulariser)
            if it_loc % subgrad_interval == 0:
                conv = _conv_fft(a, L)
                cur_obj = 2.0 * n * conv.max()
                g = _subgrad_max_conv(a, conv)
                eta = 0.06 * (1.0 - it_loc / max_iter_local)
                a = _project_to_simplex(a - eta * g)

            # occasional smoothing
            if it_loc % 4000 == 0 and it_loc > 0:
                kernel = np.array([0.125, 0.25, 0.25, 0.25, 0.125])
                a_smooth = np.convolve(a, kernel, mode='same')
                a = _project_to_simplex(0.85 * a + 0.15 * a_smooth)

            T *= 0.9995
            it_loc += 1

        return a

    # Helper to update global best (closures not allowed – use mutable list)
    # We'll store best_seq and best_score in mutable container.
    best_container = {"seq": None, "score": float("inf")}
    def nonlocal_best_update(cur_score: float, cur_seq: np.ndarray):
        if cur_score < best_container["score"]:
            best_container["score"] = cur_score
            best_container["seq"] = cur_seq.copy()
            print(f"[global] new best O={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Run optimisation for each start (respecting overall deadline)
    for idx, cand in enumerate(candidates):
        if time.time() >= deadline:
            break
        print(f"[run] start optimisation from candidate {idx+1}/{len(candidates)}")
        a_opt = _optimise_start(cand)
        # final evaluation for this candidate
        final_sc = _objective(a_opt)
        if final_sc < best_container["score"]:
            best_container["score"] = final_sc
            best_container["seq"] = a_opt.copy()
            print(f"[run] candidate {idx+1} finished with better O={final_sc:.6f}")

    # ------------------------------------------------------------------
    # sanity check with the provided evaluate_sequence
    best_seq = best_container["seq"]
    if best_seq is None:
        # fallback: uniform
        best_seq = uniform
        print("[fallback] no sequence found, using uniform")
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:
        print("[final] evaluate_sequence failed:", e)

    return best_seq.tolist()

"""
Optimise a non‑negative step function to minimise

        O(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

The optimiser runs four successive stages:

1. **Variance flattening** – Adam optimisation of the variance of the full
   auto‑convolution (Σ (c_k‑μ)²).  This spreads the mass and reduces the
   eventual peak.

2. **Soft‑max surrogate** – Adam optimisation of a smooth approximation of
   the max (log‑sum‑exp) with an annealed temperature α, together with a tiny
   entropy regulariser.

3. **Direct sub‑gradient descent** on the true max‑norm of the convolution.
   A decreasing step size together with projection onto the simplex drives the
   maximum down further.

4. **Stochastic local refinement** – inexpensive “mass‑transfer’’ Monte‑Carlo
   moves with a Metropolis acceptance rule and occasional smoothing moves.
   This phase cleans up any remaining plateaux.

Multiple simple initialisations are tried (the supplied `height_sequence_1`,
uniform, triangular and half‑sine windows).  The best sequence found before
the time budget expires is returned.  All helper functions are defined at the
top level (no closures) and the code respects the 1000 s time limit.
"""

import time
import math
import numpy as np
import scipy.sparse
import scipy.linalg

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex {x ≥ 0, sum(x)=1}
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]                       # descending sort
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:                           # all zeros → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution via FFT (real‑valued, length 2·n‑1)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return the full auto‑convolution a★a (length 2·n‑1)."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Gradient of the variance of the convolution
def _grad_variance(a: np.ndarray, conv: np.ndarray, n: int, L: int) -> np.ndarray:
    """
    Gradient of   f(a) = Σ_k (conv_k - μ)²   where μ = 1/(2n‑1) (for sum(a)=1).
    Closed form: g = 4·corr(conv‑μ , a) .
    """
    mu = 1.0 / (2 * n - 1)
    diff = conv - mu                                 # length 2n‑1
    a_rev = a[::-1]

    fft_diff = np.fft.rfft(diff, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_diff * fft_a_rev, n=L)

    start = n - 1
    grad = 4.0 * corr_rev[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Gradient of the smooth (log‑sum‑exp) surrogate + tiny entropy term
def _smooth_gradient(a: np.ndarray,
                     conv: np.ndarray,
                     alpha: float,
                     L: int,
                     lam: float) -> np.ndarray:
    """
    ∇_a ( (2·n/α)·log∑_k e^{α·conv_k} + λ·∑ a_i·log a_i ).
    The first part equals 2·corr(w,a) with w ∝ exp(α·conv).
    """
    # soft‑max weights
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / exp_scaled.sum()               # shape (2n‑1,)

    # correlation term (same trick as in _grad_variance)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)

    n_len = a.size
    grad_corr = 2.0 * corr_rev[n_len - 1:n_len - 1 + n_len]

    # tiny entropy (helps avoid exact zeros)
    eps = 1e-20
    grad_ent = lam * (np.log(a + eps) + 1.0)

    return grad_corr + grad_ent


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv)
def _subgrad_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Return a sub‑gradient of max(conv(a,a))."""
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)            # indices where max is attained
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
# Objective used during the optimisation (sum of a is 1 → O = 2·n·max(conv))
def _objective(a: np.ndarray, n: int, L: int) -> float:
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max() / (np.sum(a) ** 2)

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):

    `evaluate_sequence`.  The routine respects the 1000 s budget and
    returns the best sequence it found.
    """

    start_time = time.time()
    deadline = start_time + budget_s - 0.2          # tiny safety margin

    # --------------------------------------------------------------
    # Choose a problem size and an initial point
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore

        print(f"[init] Using supplied height_sequence_1, n={n}")

    else:

    alt_starts.append(tri)

    # half‑sine window (zero at ends, peaked in the middle)

    sine = np.sin(np.pi * (idx + 0.5) / n)
    sine = sine / sine.sum()
    alt_starts.append(sine)

    # --------------------------------------------------------------
    # FFT length – next power of two ≥ 3·n‑2 (covers all convolutions)
    L = 1 << ((3 * n - 2).bit_length())

    # Global best tracking
    best_seq = a0.copy()
    best_score = _objective(best_seq, n, L)
    print(f"[global] start O={best_score:.6f}")

    # ------------------------------------------------------------------
    # Helper to update the global best (no closure – uses mutable dict)
    best_state = {"seq": best_seq.copy(), "score": best_score}

    def _maybe_update(cur_score, cur_seq):
        if cur_score < best_state["score"]:
            best_state["score"] = cur_score
            best_state["seq"] = cur_seq.copy()
            print(f"[global] new best O={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Core optimisation loop – runs the four stages on a given start vector.
    def _run_optimisation(start_vec: np.ndarray):
        a = _project_to_simplex(start_vec.copy())
        cur_obj = _objective(a, n, L)
        _maybe_update(cur_obj, a)

        # --------------------------------------------------------------
        # Stage 1 – variance flattening (Adam)
        phase1_end = start_time + budget_s * 0.25
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-12
        lr0 = 0.8
        lam_entropy = 1e-5
        it = 0
        while time.time() < phase1_end:
            conv = _conv_fft(a, L)
            cur_obj = _objective(a, n, L)
            _maybe_update(cur_obj, a)

            grad = _grad_variance(a, conv, n, L)
            grad += lam_entropy * (np.log(a + 1e-20) + 1.0)

            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)

            m_hat = m / (1.0 - beta1 ** (it + 1))
            v_hat = v / (1.0 - beta2 ** (it + 1))

            lr = lr0 / math.sqrt(it + 1)
            a = _project_to_simplex(a - lr * m_hat / (np.sqrt(v_hat) + eps))

            # tiny random jitter to avoid exact stagnation
            if np.random.rand() < 0.001:
                a += np.random.randn(n) * 1e-4
                a = _project_to_simplex(a)

            it += 1
            if it % 50000 == 0:
                print(f"[stage‑1] iter {it:,} O={cur_obj:.6f}")

        # --------------------------------------------------------------
        # Stage 2 – smooth max‑conv surrogate (log‑sum‑exp) with Adam
        phase2_end = start_time + budget_s * 0.45
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1, beta2 = 0.9, 0.999
        lr0 = 0.5
        lam = 0.02
        alpha_start, alpha_end = 5.0, 4000.0
        it = 0
        while time.time() < phase2_end:
            conv = _conv_fft(a, L)
            cur_obj = _objective(a, n, L)
            _maybe_update(cur_obj, a)

            frac = it / max(1, int((phase2_end - start_time) / (budget_s * 0.20)))  # crude scaling
            alpha = alpha_start + (alpha_end - alpha_start) * frac

            grad = _smooth_gradient(a, conv, alpha, L, lam)

            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)

            m_hat = m / (1.0 - beta1 ** (it + 1))
            v_hat = v / (1.0 - beta2 ** (it + 1))

            lr = lr0 / math.sqrt(it + 1)
            a = _project_to_simplex(a - lr * m_hat / (np.sqrt(v_hat) + eps))

            # occasional extra noise
            if np.random.rand() < 0.001:
                a += np.random.randn(n) * 5e-5
                a = _project_to_simplex(a)

            it += 1
            if it % 40000 == 0:
                print(f"[stage‑2] iter {it:,} O={cur_obj:.6f}")

        # --------------------------------------------------------------
        # Stage 3 – direct sub‑gradient descent on max(conv)
        phase3_end = start_time + budget_s * 0.70
        eta0 = 0.13
        it = 0
        while time.time() < phase3_end:
            conv = _conv_fft(a, L)
            cur_obj = _objective(a, n, L)
            _maybe_update(cur_obj, a)

            g = _subgrad_max_conv(a, conv)
            eta = eta0 / math.sqrt(it + 1)
            a = _project_to_simplex(a - eta * g)

            # occasional jitter
            if np.random.rand() < 0.001:
                a += np.random.randn(n) * 7e-5
                a = _project_to_simplex(a)

            it += 1
            if it % 60000 == 0:
                print(f"[stage‑3] iter {it:,} O={cur_obj:.6f}")

        # --------------------------------------------------------------
        # Stage 4 – stochastic local refinement (mass‑transfer + sub‑grad)
        subgrad_interval = 5
        temperature0 = 0.04
        max_iter_local = 200_000
        it_loc = 0
        T = temperature0
        while time.time() < deadline and it_loc < max_iter_local:
            # ----- random mass‑transfer move -----
            i, j = np.random.randint(n, size=2)
            if i != j and a[i] > 1e-14:
                delta = a[i] * np.random.rand() * 0.14
                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta
                a_new = _project_to_simplex(a_new)

                conv_new = _conv_fft(a_new, L)
                new_obj = _objective(a_new, n, L)

                # Metropolis acceptance
                if (new_obj < cur_obj or
                        np.random.rand() < math.exp((cur_obj - new_obj) / max(T, 1e-12))):
                    a = a_new
                    cur_obj = new_obj
                    _maybe_update(cur_obj, a)

            # ----- sub‑gradient step every few iterations -----
            if it_loc % subgrad_interval == 0:

                cur_obj = _objective(a, n, L)
                g = _subgrad_max_conv(a, conv)
                eta = 0.07 * (1.0 - it_loc / max_iter_local)
                a = _project_to_simplex(a - eta * g)
                cur_obj = _objective(a, n, L)
                _maybe_update(cur_obj, a)

            # ----- occasional smoothing move -----
            if it_loc % 4000 == 0 and it_loc > 0:
                kernel = np.array([0.125, 0.25, 0.25, 0.25, 0.125])
                a_smooth = np.convolve(a, kernel, mode="same")
                a = _project_to_simplex(0.85 * a + 0.15 * a_smooth)

            T *= 0.9998
            it_loc += 1

    # Run optimisation from the primary start and, if time permits, from alternatives
    _run_optimisation(a0)

    for alt in alt_starts:
        if time.time() >= deadline:
            break
        print("[alt] trying an alternative initialisation")
        _run_optimisation(alt)

    # ------------------------------------------------------------------
    # Final sanity check with the supplied `evaluate_sequence`
    final_seq = best_state["seq"]
    try:
        final_score = evaluate_sequence(final_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:                     # pragma: no‑cover
        print("[final] evaluate_sequence failed:", exc)

    return final_seq.tolist()
