import time
import numpy as np

def _next_pow2(x):
    return 1 << ((x - 1).bit_length())

def _conv_self_fft(a):
    n = a.shape[0]
    L = _next_pow2(2*n - 1)
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return conv_full[:2*n-1]

def _correlation_fft(w, a):
    n = a.shape[0]
    m = w.shape[0]
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]

def _softmax(x):
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()

def _objective(a):
    n = a.size
    conv = _conv_self_fft(a)
    return float(2.0 * n * conv.max())

def _smooth_max_weights(conv, beta):
    M = conv.max()
    if beta > 1e4:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w
    # ensure non-positive exponent to avoid overflow
    shift = conv - M    # <=0
    w_raw = np.exp(beta * shift)
    s = w_raw.sum()
    if s == 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w
    return w_raw / s

def _spectral_flatten(a, rng):
    """Flatten the magnitude of the real DFT while keeping the phase."""
    n = a.shape[0]
    af = np.fft.rfft(a)
    mag = np.abs(af)
    avg_mag = mag.mean()
    # Keep original phase
    phase = np.angle(af)
    new_af = avg_mag * np.exp(1j * phase)
    a_new = np.fft.irfft(new_af, n=n)
    a_new = np.clip(a_new, 0.0, None)
    total = a_new.sum()
    if total < 1e-12:
        a_new = np.ones_like(a_new) / n
    else:
        a_new /= total
    return a_new

def _adam_optimize(
    n,
    time_limit,
    init_a=None,
    seed=0,
    beta_start=0.5,
    beta_end=50000.0,
    lr_start=0.2,
    lr_end=0.001,
    lambda_start=0.5,
    lambda_end=0.01,
):
    """Adam optimiser on the simplex with extra L2‑regularisation and occasional spectral flattening."""
    start = time.time()
    rng = np.random.default_rng(seed)
    eps = 1e-16

    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
        a = np.clip(a, 0.0, None)
        if a.sum() < 1e-12:
            a = np.ones(n) / n
        else:
            a = a / a.sum()
    else:
        a = rng.random(n)
        a = a / a.sum()

    # logits for softmax parameterisation
    x = np.log(a + eps)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999

    best_a = a.copy()
    best_obj = _objective(best_a)

    it = 0
    while time.time() - start < time_limit:
        it += 1
        a = _softmax(x)                 # stay on the simplex
        conv = _conv_self_fft(a)
        M = conv.max()
        obj = float(2.0 * n * M)
        # record improvement
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[n={n:5d} it={it:06d}] new best O={best_obj:.6f}")

        # schedule parameters based on elapsed fraction
        t = min((time.time() - start) / time_limit, 1.0)
        beta = beta_start * np.exp(np.log(beta_end / beta_start) * t)
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t)
        lambda_reg = lambda_start * np.exp(np.log(lambda_end / lambda_start) * t)

        # smooth‑max weights
        w = _smooth_max_weights(conv, beta)

        # gradient of smooth‑max term
        grad_max = 2.0 * _correlation_fft(w, a)

        # gradient of L2‑norm of the convolution (variance regulariser)
        grad_l2 = 4.0 * _correlation_fft(conv, a)

        # combined gradient
        g_a = grad_max + lambda_reg * grad_l2

        # natural‑gradient in log‑it space, keeping the simplex
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + 1e-12)

            x += jitter

        if it % 50000 == 0:
            x = np.log(best_a + eps)
            x += rng.normal(scale=0.002, size=n)
            m.fill(0.0)
            v.fill(0.0)

        # spectral flattening as an exploratory perturbation
        if it % 10000 == 0:
            a_flat = _spectral_flatten(a, rng)
            # move back to log‑it space
            x = np.log(a_flat + eps)

    # compress negligible entries after optimisation
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = _objective(best_a)
        print(f"[compress] reduced length → {best_a.size}, O={best_obj:.6f}")

    # very short local greedy mass‑transfer improvement if time permits
    remaining = time_limit - (time.time() - start)
    if remaining > 0.5:
        # simple hill‑climbing using random pairwise transfers
        local_start = time.time()
        a_loc = best_a.copy()
        obj_loc = best_obj
        n_loc = a_loc.size
        while time.time() - local_start < remaining * 0.3:
            i, j = rng.choice(n_loc, size=2, replace=False)
            # move up to 2 % of a_i
            delta = a_loc[i] * rng.random() * 0.02
            if delta <= 0.0:
                continue
            a_new = a_loc.copy()
            a_new[i] -= delta
            a_new[j] += delta
            # ensure positivity (clip tiny negativities)
            a_new = np.clip(a_new, 0.0, None)
            a_new /= a_new.sum()
            conv_new = _conv_self_fft(a_new)
            obj_new = float(2.0 * n_loc * conv_new.max())
            if obj_new < obj_loc - 1e-12:
                a_loc, obj_loc = a_new, obj_new
                if obj_new < best_obj:
                    best_a, best_obj = a_new, obj_new
        print(f"[local] final local O={best_obj:.6f}")

    return best_a.tolist(), best_obj

def _run_sa(init_seq, time_limit, seed=0):

    Targeted simulated‑annealing refinement (same as previous version, but
    with a slightly faster temperature schedule).
    """
    start = time.time()
    rng = np.random.default_rng(seed)
    a = np.asarray(init_seq, dtype=float)
    a = np.clip(a, 0.0, None)
    if a.sum() < 1e-12:
        a = np.ones_like(a) / a.size
    else:
        a = a / a.sum()
    n = a.shape[0]

    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())
    best_seq = a.copy()

    # temperature schedule
    T = best_obj * 0.1 + 1e-6
    T_min = best_obj * 1e-7 + 1e-9
    decay = 0.995
    it = 0

    while time.time() - start < time_limit:
        it += 1
        # index where convolution is maximal
        k_max = int(np.argmax(conv))

        # contributions to that index
        i_vals = np.arange(n)
        j_vals = k_max - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i_vals[mask]] * a[j_vals[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            probs = contrib / total_contrib
            i = rng.choice(n, p=probs)
        else:
            i = rng.integers(n)

        # pick a destination index uniformly different from i
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # transfer a modest fraction of mass
        delta = a[i] * rng.random() * 0.06
        if delta <= 0.0:
            continue
        a_new = a.copy()

"""
Non‑negative step‑function optimiser (enhanced).

The optimiser consists of three stages:

1.  Adam optimisation on the simplex, using a soft‑max (log‑it)
    parametrisation.  The objective is the smooth‑max approximation of
    ``max(conv(a,a))`` plus an L₂‑regulariser on the convolution (which
    encourages a flatter autocorrelation).  The algorithm occasionally
    performs a “spectral‑flattening’’ perturbation (flatten all DFT
    magnitudes) and injects jitter/restarts to escape plateaus.

2.  A very short greedy hill‑climbing phase that transfers a tiny
    amount of mass between random positions and keeps any improvement.

3.  Targeted simulated‑annealing refinement (similar to the previous
    baseline) to polish the best sequence found.

All work respects the supplied time budget (≈1000 s).  The routine
returns a plain list of non‑negative floats suitable for
``evaluate_sequence``.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return conv_full[: 2 * n - 1]


def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation g_i = Σ_t w[i+t]·a[t] for i=0,…,n‑1.
    Implemented via FFT.
    """
    n = a.shape[0]
    m = w.shape[0]                     # = 2n‑1 in our use‑case
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.size
    conv = _conv_self_fft(a)
    return float(2.0 * n * conv.max())


def _smooth_max_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Returns a probability vector w approximating the arg‑max of conv.
    For large beta the distribution collapses onto the maximal entry.
    """
    M = conv.max()
    if beta > 1e4:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w
    shift = conv - M                     # ≤ 0
    w_raw = np.exp(beta * shift)
    s = w_raw.sum()
    if s == 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w
    return w_raw / s


def _spectral_flatten(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Flattens the magnitude of the real DFT of a while preserving the phase.
    The resulting vector stays on the simplex (non‑negative, sum = 1).
    """
    n = a.shape[0]
    af = np.fft.rfft(a)
    mag = np.abs(af)
    avg_mag = mag.mean()
    phase = np.angle(af)
    new_af = avg_mag * np.exp(1j * phase)
    a_new = np.fft.irfft(new_af, n=n)
    a_new = np.clip(a_new, 0.0, None)
    tot = a_new.sum()
    if tot < 1e-12:
        a_new = np.ones_like(a_new) / n
    else:
        a_new /= tot
    return a_new
