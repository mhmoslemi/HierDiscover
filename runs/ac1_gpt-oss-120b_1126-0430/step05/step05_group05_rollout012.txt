def _gradient_flatten(a, conv, L):
    n = a.size
    mean = conv.mean()
    diff = conv - mean
    # correlation w ⋆ a with w = diff
    a_rev = a[::-1]
    fft_w = np.fft.rfft(diff, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)

"""
Propose a non‑negative step‑function sequence that (approximately) minimises

    O(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

Algorithm outline (1000 s budget)

1️⃣  **Mirror‑descent on a smoothed log‑sum‑exp surrogate**  
    Minimise (2·n/α)·log∑exp(α·conv) + λ·∑a·log a  on the probability simplex.
    α is annealed from a small to a very large value, λ is annealed to zero,
    and updates are performed by exponentiated‑gradient (multiplicative) steps
    followed by Euclidean projection onto the simplex.

2️⃣  **Projected sub‑gradient / flattening descent on the exact objective**  
    Repeatedly compute the exact max‑convolution, obtain its (averaged)
    sub‑gradient, and take a Polyak step  η = (f(a)‑f*)/‖g‖²  with an optimistic
    lower bound f*≈1.0.  The update is projected back onto the simplex.
    Every few thousand iterations a tiny Gaussian jitter is added to avoid
    stagnation.

3️⃣  **Intensive simulated‑annealing local refinement**  
    Random “mass‑transfer’’ moves (move a random fraction of probability mass
    from a source index to a destination index) are accepted with a
    Metropolis criterion.  In addition, occasional block‑redistribution moves
    (uniformly spread the total mass of a short contiguous block) are tried.
    Temperature decays slowly (≈0.9999 per move).

4️⃣  **Final Adam polishing** – a short Adam run on the smooth surrogate with
    a huge smoothing parameter α to fine‑tune any remaining slack.

The routine prints progress, keeps the best sequence found and finally
returns it as a plain list of float.  All helper functions are defined at the
module level (no closures) and the implementation respects the 1000 s
time‑budget.

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex (Euclidean)
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Full linear convolution using FFT (returns length 2·n‑1)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution a⨂a (length 2·len(a)‑1) using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]

# ----------------------------------------------------------------------
# Exact objective (assumes sum(a) = 1, i.e. normalised)
def _objective_exact(a: np.ndarray, L: int) -> float:
    """Return O = 2·n·max(conv) for a already normalised (sum=1)."""
    n = a.size
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()

# ----------------------------------------------------------------------
# Gradient of the smooth log‑sum‑exp surrogate plus entropy term
def _smooth_gradient_plus_entropy(
    a: np.ndarray,
    conv: np.ndarray,
    alpha: float,
    L: int,
    lam: float,
) -> np.ndarray:
    """
    Gradient of   (2·n/α)·log∑_k e^{α·conv_k}  +  λ·∑ a_i·log a_i .
    The first term reduces to 2·corr(w, a) where w_k ∝ e^{α·conv_k}.
    """
    # soft‑max weights w_k ∝ e^{α·conv_k}
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)          # stabilise
    w = exp_scaled / exp_scaled.sum()                # shape (2n‑1,)

    # correlation w ⋆ a  (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n_len = a.size
    grad_corr = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]

    # entropy gradient λ·(log a + 1)
    eps = 1e-20
    grad_ent = lam * (np.log(a + eps) + 1.0)

    return grad_corr + grad_ent

# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) (average over all maximisers)
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. a (averaged over all maximisers)."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)          # indices where the max is attained
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g

# ----------------------------------------------------------------------
# Gradient of squared deviation from the mean (flattens the convolution)
def _gradient_flatten(a: np.ndarray, conv: np.ndarray, L: int) -> np.ndarray:
    """
    Gradient of ½·‖conv - mean(conv)‖².
    Returns vector of size n.
    """
    n = a.size
    mean = conv.mean()
    diff = conv - mean
    a_rev = a[::-1]
    fft_w = np.fft.rfft(diff, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    grad = 2.0 * corr_rev[n - 1 : n - 1 + n]
    return grad
