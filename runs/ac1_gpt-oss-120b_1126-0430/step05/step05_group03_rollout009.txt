"""
LP‑based iterative improvement for the step‑function problem.

For a fixed support size n we keep a non‑negative vector a with ∑a=1.
Its quality is measured by

    O(a) = 2·n·max(conv(a,a))                (conv = full linear convolution)

We iteratively improve a by solving a linear programme:

    maximise   ∑ g_j
    subject to conv(a,g) ≤ M  (elementwise) , g ≥ 0

where M = max(conv(a,a)).  The current a is feasible (g=a), therefore a
solution g with larger total mass exists whenever a is non‑uniform.
Normalising g back to a unit simplex gives a new candidate a′.
If O(a′) < O(a) we accept the step and repeat; otherwise we stop.

The LP is built from the Toeplitz matrix that represents the convolution
operator, solved with SciPy’s high‑performance “highs” backend.
After a few LP steps we finish with a cheap random‑mass‑transfer local
search.  Several support sizes are tried within the overall time budget;
the best sequence is returned.

The algorithm is deliberately different from the previous Adam‑based
approach and makes use of the linear‑programming structure of the
max‑convolution bound.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full self‑convolution a*a using a real FFT.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution of a."""
    return _conv_self_fft(a).max()


def objective(a: np.ndarray) -> float:
    """
    Exact objective used by ``evaluate_sequence`` (except for the
    security checks).  a is a 1‑D numpy array.
    """
    a = np.asarray(a, dtype=float)
    s = a.sum()
    if s < 0.01:
        return float("inf")
    n = a.size
    M = max_convolution(a)
    return 2.0 * n * M / (s * s)


# ----------------------------------------------------------------------
def build_constraint_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1)×n matrix A such that (A @ g) = conv(f, g) for a
    given non‑negative f.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # each column i contributes f[i] to rows i … i+n‑1
    for i in range(n):
        A[i:i + n, i] = f[i]
    return A


def solve_lp(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve
        maximise   sum(g)
        subject to conv(f,g) ≤ M, g ≥ 0
    Returns the optimal g (may be None on failure).
    """
    n = f.shape[0]
    A = build_constraint_matrix(f)          # shape (2n‑1, n)
    b = np.full(A.shape[0], M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)       # minimise -sum(g)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(
            c,
            A_ub=A,
            b_ub=b,
            bounds=bounds,
            method="highs",
            options={"presolve": True},
        )
    except Exception as exc:            # pragma: no cover
        print("[LP] solver exception:", exc)
        return None

    if not res.success:
        # Solver occasionally reports infeasibility due to numerical noise;
        # we simply treat it as a failure and fall back to the current f.
        return None
    return res.x


# ----------------------------------------------------------------------
def lp_improve(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
) -> tuple[list[float], float]:
    """
    Perform the LP‑based improvement for a single support size.
    Returns the best sequence (as a plain list) and its exact objective.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # ------- initialise -------------------------------------------------
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            if a.size > n:
                a = a[:n]
            else:
                a = np.concatenate([a, np.zeros(n - a.size)])
        if a.sum() < 0.01:
            a = rng.random(n)
        a = a / a.sum()
    else:
        a = rng.random(n)
        a = a / a.sum()

    best_a = a.copy()
    best_obj = objective(best_a)

    iteration = 0
    # ------------------------------------------------------------------
    # Main LP loop (stop either on time or on lack of improvement)
    while time.time() - start < time_limit - 0.5:  # keep a margin for final local search
        iteration += 1
        M = max_convolution(a)

        g = solve_lp(a, M)
        if g is None:
            print(f"[n={n}] LP failed at iteration {iteration}")
            break

        sum_g = g.sum()
        if sum_g <= 1e-12:

        g_norm = g / sum_g                     # keep the simplex constraint

        if obj_new + 1e-12 < best_obj:
            best_a = g_norm.copy()
            best_obj = obj_new
            a = g_norm                          # continue from the improved point
            print(f"[n={n:5d} it={iteration:03d}] new best O={best_obj:.6f}")
        else:
            # No improvement – stop the LP phase for this length.
            break

    # ------- cheap random local refinement ------------------------------
    remaining = time_limit - (time.time() - start)
    if remaining > 1.0:
        a_loc = best_a.copy()
        obj_loc = best_obj
        n_loc = a_loc.size
        local_deadline = time.time() + remaining * 0.9
        while time.time() < local_deadline:
            i, j = rng.choice(n_loc, size=2, replace=False)
            # transfer at most a few percent of the source mass
            delta = a_loc[i] * rng.random() * 0.03
            if delta <= 0.0:
                continue
            cand = a_loc.copy()
            cand[i] -= delta
            cand[j] += delta
            obj_cand = objective(cand)
            if obj_cand < obj_loc:
                a_loc = cand
                obj_loc = obj_cand
                if obj_cand < best_obj:
                    best_a = cand
                    best_obj = obj_cand
        print(f"[n={n}] final local O={best_obj:.6f}")

    return best_a.tolist(), best_obj


# ----------------------------------------------------------------------
def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **_ignore,
) -> list[float]:
    """
    Entry point used by the competition harness.
    Tries several support sizes, runs the LP‑based optimiser for each,
    and finishes with a short random local search on the overall best.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    # --------------------------------------------------------------
    # Retrieve the previous best construction if it exists.
    if "height_sequence_1" in globals():
        seq0 = np.asarray(height_sequence_1, dtype=float).copy()
        seq0 = np.clip(seq0, 0.0, None)
        if seq0.sum() < 0.01:
            seq0 = np.ones_like(seq0)
        seq0 = seq0 / seq0.sum()
        n0 = seq0.size
    else:
        seq0 = None
        n0 = None

    # --------------------------------------------------------------
    # Choose a ladder of lengths.  Larger n gives a lower theoretical
    # bound but also a more expensive LP.
    lengths = [800, 1200, 1600, 2000, 2400, 2800]
    # make sure we also test the size of the previous best if it differs
    if n0 is not None and n0 not in lengths:
        lengths.insert(0, n0)

    best_seq = None
    best_obj = float("inf")

    # --------------------------------------------------------------
    for idx, n in enumerate(lengths):
        remaining = deadline - time.time()
        if remaining <= 2.0:
            break

        # Allocate most of the remaining time to the first (largest) length.
        if idx == 0:
            run_time = remaining * 0.60
        else:
            # split the rest evenly among the remaining lengths
            run_time = (remaining - 0.5) / (len(lengths) - idx)

        # Initialise from the global best if possible (only for the first length)
        init = None
        if idx == 0 and seq0 is not None:
            if seq0.size != n:
                if seq0.size > n:
                    init = seq0[:n]
                else:
                    pad = np.zeros(n - seq0.size)
                    init = np.concatenate([seq0, pad])
                init = init / init.sum()
            else:
                init = seq0

        seq, val = lp_improve(
            n=n,
            time_limit=run_time,
            init_a=init,
            seed=seed + idx * 13,
        )
        if val < best_obj:
            best_obj = val
            best_seq = seq
            print(f"[global] new overall best O={best_obj:.6f} (n={n})")

    # --------------------------------------------------------------
    # Final short random local search on the overall best (if time permits)
    if best_seq is not None:
        remaining = deadline - time.time()
        if remaining > 2.0:
            a = np.array(best_seq, dtype=float)
            best_obj = objective(a)
            rng = np.random.default_rng(seed + 999)
            local_deadline = time.time() + remaining * 0.8
            n = a.size
            while time.time() < local_deadline:
                i, j = rng.choice(n, size=2, replace=False)
                delta = a[i] * rng.random() * 0.03
                if delta <= 0.0:
                    continue
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                obj_cand = objective(cand)
                if obj_cand < best_obj:
                    a = cand
                    best_obj = obj_cand
                    best_seq = a.tolist()
            print(f"[final local] O={best_obj:.6f}")

    # --------------------------------------------------------------
    # Sanity check with the provided evaluate_sequence routine.
    try:
        final_score = evaluate_sequence(best_seq)
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    # guarantee a non‑empty return
    if not best_seq:
        best_seq = [1.0]

    return best_seq
