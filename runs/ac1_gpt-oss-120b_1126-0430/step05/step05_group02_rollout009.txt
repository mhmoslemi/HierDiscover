"""
Propose a non‑negative step function (list of heights) that minimises

        2·n·max_k (a*a)[k] / (Σ a_i)²

This implementation follows a different philosophy from the previous heavy‑ball
code:

*  A diverse set of parametric families (power‑law, exponential, mixed,
   beta, triangular, sine, inverse‑sqrt, random Dirichlet) is scanned once to
   obtain a solid deterministic start.
*  A fast mirror‑descent / exponentiated‑gradient refinement is performed.
   The sub‑gradient is computed *exactly* from the (few) active convolution
   entries (those attaining the current maximum).  This yields a more accurate
   descent direction than the “top‑k” approximation used before.
*  Simple stagnation detection triggers a restart with a tiny random
   perturbation around the best iterate.
*  The routine respects the supplied time budget (default 1000 s) and always
   returns a feasible sequence (non‑negative, sum > 0.01).

The code uses only top‑level helper functions (no closures) and standard
libraries (numpy, math, time).  It assumes ``evaluate_sequence`` is already
imported in the execution environment.

def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float) -> np.ndarray:
    """Return a normalised power‑law a_i ∝ (i+1)^‑p (Σa_i = 1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = i ** (-p)
    seq /= seq.sum()
    return seq


def _tent_sequence(n: int) -> np.ndarray:
    """Triangular (tent) shape, symmetric, normalised."""
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1, dtype=np.float64)
        right = np.arange(half, 0, -1, dtype=np.float64)
    else:
        left = np.arange(1, half + 2, dtype=np.float64)
        right = np.arange(half, 0, -1, dtype=np.float64)
    seq = np.concatenate([left, right])
    seq /= seq.sum()
    return seq


def _sine_sequence(n: int) -> np.ndarray:
    """a_i ∝ sin(π·(i+0.5)/n), positive on (0,π), normalised."""
    i = np.arange(n, dtype=np.float64) + 0.5
    seq = np.sin(np.pi * i / n)
    seq = np.maximum(seq, 0.0)
    seq /= seq.sum()
    return seq


def _inv_sqrt_sequence(n: int) -> np.ndarray:
    """a_i ∝ 1/√( (i+0.5)*(n-i+0.5) ), normalised."""
    i = np.arange(n, dtype=np.float64) + 0.5
    seq = 1.0 / np.sqrt(i * (n - i + 0.5))
    seq = np.maximum(seq, 0.0)
    seq /= seq.sum()
    return seq


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using FFT (returns length 2·n‑1 array)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa).real[: 2 * a.size - 1]
    return conv


def _conv_max_fft(a: np.ndarray, fft_len: int) -> tuple[float, int]:
    """Maximum value and its index of the autocorrelation a*a."""
    conv = _conv_full_fft(a, fft_len)
    idx = int(conv.argmax())
    return float(conv[idx]), idx


def _subgradient_active(a: np.ndarray, active_k: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of max_k (a*a)[k] at the current active set.
    For each active index k we add the contribution 2·a_{k-i} to g_i.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in active_k:
        if k < n:
            # i = 0 .. k  (inclusive)
            # a_{k-i} runs from a_k down to a_0 -> a[k::-1]
            g[: k + 1] += a[k::-1]
        else:
            # i = k-(n-1) .. n-1
            lo = k - (n - 1)
            # a_{k-i} runs from a_{n-1} down to a_{lo}
            g[lo:] += a[lo:][::-1]
    g *= 2.0
    return g

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that
    minimises ``evaluate_sequence``.  The routine adapts its schedule to
    the available time budget and always returns a feasible sequence.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a safety margin for the final exact evaluation
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # Choose a working length (n) – try to improve on any previous result.
    # --------------------------------------------------------------
    try:
        prev = height_sequence_1                     # may be defined globally
        n_prev = len(prev)
        prev_arr = np.asarray(prev, dtype=np.float64)
        prev_arr = np.maximum(prev_arr, 0.0)
        if prev_arr.sum() > 0.0:
            prev_arr /= prev_arr.sum()
        else:
            prev_arr = None
    except Exception:
        prev = None
        n_prev = None
        prev_arr = None

    # Desired length: at least 5 k, at most 25 k, try to use a longer vector if possible
    n = 12000
    if n_prev is not None:
        n = max(n, n_prev)
    n = int(min(max(n, 5000), 25000))

    # FFT length – next power of two ≥ 2·n‑1
    fft_len = 1 << ((2 * n - 1).bit_length())

    # --------------------------------------------------------------
    # Cheap score (assumes Σa_i = 1): 2·n·max(conv)
    # --------------------------------------------------------------
    def cheap_score(vec: np.ndarray) -> float:
        maxb, _ = _conv_max_fft(vec, fft_len)
        return 2.0 * n * maxb

    # --------------------------------------------------------------
    # 0. Initialise with cheap families
    # --------------------------------------------------------------
    best_a = None
    best_val = float("inf")

    # Uniform baseline
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    val = cheap_score(uniform)
    best_a, best_val = uniform, val

    # Previous construction (if compatible)
    if prev_arr is not None:
        if prev_arr.size == n:
            val = cheap_score(prev_arr)
            if val < best_val:
                best_a, best_val = prev_arr.copy(), val
        else:
            # Resize by truncation / zero‑padding
            if prev_arr.size > n:
                cand = prev_arr[:n]
            else:
                cand = np.concatenate([prev_arr, np.zeros(n - prev_arr.size)])
            cand /= cand.sum()
            val = cheap_score(cand)
            if val < best_val:
                best_a, best_val = cand.copy(), val

    # Power‑law sweep
    for p in np.linspace(0.1, 2.0, 91):
        if time.time() >= deadline:
            break
        cand = _power_law_sequence(n, p)
        val = cheap_score(cand)
        if val < best_val:
            best_a, best_val = cand.copy(), val

    # Pure exponential sweep
    for lam in np.linspace(0.0, 5.0, 101):
        if time.time() >= deadline:
            break
        idx = np.arange(n, dtype=np.float64)
        cand = np.exp(-lam * idx)
        cand /= cand.sum()
        val = cheap_score(cand)
        if val < best_val:
            best_a, best_val = cand.copy(), val

    # Power‑law × exponential sweep (coarse grid)
    for p in np.linspace(0.2, 1.5, 14):
        for lam in np.linspace(0.0, 4.0, 14):
            if time.time() >= deadline:
                break
            idx = np.arange(n, dtype=np.float64)
            cand = ((idx + 1.0) ** (-p)) * np.exp(-lam * idx)
            cand /= cand.sum()
            val = cheap_score(cand)
            if val < best_val:
                best_a, best_val = cand.copy(), val

    # Beta‑shaped families
    x = (np.arange(1, n + 1, dtype=np.float64) / n)
    for alpha in np.linspace(0.2, 5.0, 21):
        for beta in np.linspace(0.2, 5.0, 21):
            if time.time() >= deadline:
                break
            cand = (x ** alpha) * ((1.0 - x) ** beta)
            s = cand.sum()
            if s == 0.0:
                continue
            cand /= s
            val = cheap_score(cand)
            if val < best_val:
                best_a, best_val = cand.copy(), val

    # Triangular (tent) shape
    if time.time() < deadline:
        cand = _tent_sequence(n)
        val = cheap_score(cand)
        if val < best_val:
            best_a, best_val = cand.copy(), val

    # Sine shape
    if time.time() < deadline:
        cand = _sine_sequence(n)
        val = cheap_score(cand)
        if val < best_val:
            best_a, best_val = cand.copy(), val

    # Inverse‑sqrt shape
    if time.time() < deadline:
        cand = _inv_sqrt_sequence(n)
        val = cheap_score(cand)
        if val < best_val:
            best_a, best_val = cand.copy(), val

    # Some random Dirichlet seeds (wide range of concentration)
    for conc in [0.001, 0.01, 0.1, 0.5, 1.0, 3.0, 10.0]:
        if time.time() >= deadline:
            break
        g = np.random.gamma(conc, 1.0, size=n)
        cand = g / g.sum()
        val = cheap_score(cand)
        if val < best_val:
            best_a, best_val = cand.copy(), val

    print(f"[Init] best cheap score = {best_val:.6f}")

    # --------------------------------------------------------------
    # 1. Refine the best candidate with exact active‑set mirror descent
    # --------------------------------------------------------------
    a = best_a.copy()
    best_iter_a = a.copy()
    best_iter_val = best_val
    last_improve_iter = 0
    iter_cnt = 0

    # hyper‑parameters
    eta0 = 0.5                # initial step‑size factor
    max_iters = 2_000_000     # soft cap on iterations

    while time.time() < deadline and iter_cnt < max_iters:
        iter_cnt += 1

        # ---- convolution and cheap score ----
        conv = _conv_full_fft(a, fft_len)
        maxb = conv.max()
        cheap = 2.0 * n * maxb

        # ---- keep the best seen so far ----
        if cheap < best_iter_val - 1e-12:
            best_iter_val = cheap
            best_iter_a = a.copy()
            last_improve_iter = iter_cnt
            print(f"[Iter {iter_cnt:07d}] NEW BEST cheap = {cheap:.6f}")

        # ---- active set (indices attaining the maximum) ----
        tol = 1e-14
        active = np.where(conv >= maxb - tol)[0]

        # ---- exact sub‑gradient for the active set ----
        g = _subgradient_active(a, active)

        # ---- exponentiated‑gradient step ----
        eta = eta0 / math.sqrt(iter_cnt)
        a = a * np.exp(-eta * g)          # multiplicative update (keeps positivity)
        a = np.maximum(a, 1e-30)          # guard against exact zeros
        s = a.sum()
        if s == 0.0 or not np.isfinite(s):
            # fallback to the last good iterate
            a = best_iter_a.copy()
            continue
        a /= s

        # ---- restart if no improvement for a while ----
        if iter_cnt - last_improve_iter > 150_000:
            perturb = 0.005 * np.random.rand(n)
            a = _simplex_projection(best_iter_a + perturb)
            last_improve_iter = iter_cnt
            print(f"[Iter {iter_cnt:07d}] Restart after stagnation.")

    # --------------------------------------------------------------
    # 2. Final exact evaluation (uses the provided ``evaluate_sequence``)
    # --------------------------------------------------------------
    final_seq = best_iter_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:      # defensive – should never happen
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # --------------------------------------------------------------
    # 4. Return the best sequence found.
    # --------------------------------------------------------------
    return final_seq
