def power_law_dist(n, p):
    i = np.arange(1, n+1, dtype=np.float64)
    a = i**(-p)
    return a / a.sum()

def sym_product_power_law_dist(n, p):
    i = np.arange(n, dtype=np.float64)
    a = ((i+1)*(n-i))**(-p)  # exponent negative if p positive (edge heavy)
    # If p negative, exponent positive -> central heavy.
    # But we allow p range [-5,5]
    # Avoid overflow: if p large magnitude, values may become huge. So clip.
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    if s == 0: return np.ones(n)/n
    return a / s

def exp_decay_dist(n, lam):
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam*i) + np.exp(-lam*(n-1-i))
    s = a.sum()
    if s == 0: return np.ones(n)/n
    return a/s

def logistic_dist(n, alpha):
    i = np.arange(n, dtype=np.float64)
    mid = (n-1)/2
    a = 1.0/(1.0+np.exp(alpha*(i-mid)))
    s = a.sum()
    if s == 0: return np.ones(n)/n
    return a/s

def beta_dist(n, alpha, beta):
    i = np.arange(n, dtype=np.float64) + 1  # indices from 1..n
    # for continuous Beta on [0,1], we use normalized positions
    # j = (i-0.5)/n maybe; but simpler
    a = ((i)**(alpha-1))*((n+1-i)**(beta-1))
    # Clip large values to avoid overflow
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    if s == 0: return np.ones(n)/n
    return a / s

def _refine_distribution(a0, deadline):
    n = a0.shape[0]
    # convert to logits
    eps = 1e-16
    z = np.log(a0 + eps) + 0.01 * np.random.randn(n)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_alpha = 0.3
    eps_factor = 0.08
    eps_min = 5e-5
    decay = 0.9995
    best_seq = _softmax(z)
    best_score = 2.0 * n * _conv_max(best_seq)
    it = 0
    last_improve = 0
    while time.time() < deadline - 2.0:
        it += 1
        a = _softmax(z)
        max_conv, grad_a = _conv_max_and_grad(a, eps_factor)
        cheap_score = 2.0 * n * max_conv

        if cheap_score < best_score - 1e-12:
            best_score = cheap_score
            best_seq = a.copy()
            last_improve = it

        # compute gradient on logits
        mu = np.dot(grad_a, a)
        grad_z = a * (grad_a - mu)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step = - m_hat / (np.sqrt(v_hat) + eps_adam)

        # backtracking line search
        accepted = False
        trial_alpha = alpha_t
        for _ in range(4):
            z_new = z + trial_alpha * step
            z_new = np.clip(z_new, -50.0, 50.0)
            a_new = _softmax(z_new)
            new_score = 2.0 * n * _conv_max(a_new)
            if new_score <= cheap_score:
                z = z_new
                accepted = True
                break
            trial_alpha *= 0.5
        if not accepted:
            z += alpha_t * step
            z = np.clip(z, -50.0, 50.0)

        # occasional jitter
        if it % 3000 == 0 and np.random.rand() < 0.1:
            z += 0.015 * np.random.randn(n)
            m *= 0.6
            v *= 0.6

        # reduce smoothing
        eps_factor = max(eps_min, eps_factor * decay)

        # restart if stagnated
        if it - last_improve > 200000:
            print("[refine] stagnation, restart jitter")
            z = _logits_from_distribution(best_seq) + 0.02 * np.random.randn(n)
            m.fill(0.0)
            v.fill(0.0)
            eps_factor = 0.08
            last_improve = it

        # optional scaling if time left and n<... maybe not needed since already at full n.
    return best_seq

"""
Non‑negative step‑function optimiser – hybrid parametric + Adam refinement.

* Phase 1 – fast parametric search (≈ 200 s) on a reduced support size.
  Randomly sample a handful of low‑dimensional families (power‑law,
  symmetric product‑power‑law, exponential, logistic, Beta) and evolve the
  parameters with a tiny steady‑state genetic loop.  The best candidate is
  interpolated to the full support length.

* Phase 2 – mirror‑descent on the probability simplex (≈ remaining time).
  The interpolated distribution is turned into soft‑max logits and optimised
  with Adam using a smooth‑max surrogate for the convolution ∞‑norm.
  Occasional jitter‑restarts, back‑tracking line‑search and a decaying
  smoothing factor keep the iterates moving.

The routine respects the given time budget (default 1000 s) and returns the
best sequence found.  All numbers are non‑negative; the final exact score
is obtained via the provided ``evaluate_sequence`` function.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑dimensional families (all return a normalised probability vector)
# ----------------------------------------------------------------------


def _power_law_dist(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    # use log‑space to avoid overflow
    a = np.exp(-p * np.log(i))
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


def _sym_prod_power_law_dist(n: int, p: float) -> np.ndarray:
    """a_i ∝ ((i+1)*(n‑i))^{-p} (U‑shaped for p>0, peaked centre for p<0)."""
    i = np.arange(n, dtype=np.float64)
    # log term = log(i+1) + log(n-i)
    log_term = np.log(i + 1.0) + np.log(n - i)
    a = np.exp(-p * log_term)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


def _exp_decay_dist(n: int, lam: float) -> np.ndarray:
    """a_i ∝ e^{-lam·i}+e^{-lam·(n‑1‑i)} (symmetric exponential)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


def _logistic_dist(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ 1/(1+exp(alpha·(i‑mid))) – heavy on the left for alpha>0."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = 1.0 / (1.0 + np.exp(alpha * (i - mid)))
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


def _beta_dist(n: int, a: float, b: float) -> np.ndarray:
    """a_i ∝ (i+1)^{a‑1}·(n‑i)^{b‑1} – the discrete Beta family."""
    i = np.arange(n, dtype=np.float64) + 1.0
    log_a = (a - 1.0) * np.log(i)
    log_b = (b - 1.0) * np.log(n + 1.0 - i)
    a_vec = np.exp(log_a + log_b)
    a_vec = np.nan_to_num(a_vec, nan=0.0, posinf=0.0, neginf=0.0)
    s = a_vec.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a_vec / s


# ----------------------------------------------------------------------
# FFT‑based convolution utilities (max‑convolution and smoothed gradient)
# ----------------------------------------------------------------------


def _fft_len_for_corr(n: int) -> int:
    """Smallest power‑of‑two ≥ 3·n‑2 (sufficient for linear self‑correlation)."""
    return 1 << ((3 * n - 2).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a (exact, via FFT)."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Smoothed max of a*a and its gradient w.r.t. a.

    Returns (max_conv, grad_a) where
    grad_a = ∇_a [ 2·n·smooth_max(a*a) ].
    The smoothing parameter is ε = eps_factor·max_conv.
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv                     # ≤ 0
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # factor 4·n comes from derivative of 2·n·max_{smooth}
    grad_a = 4.0 * n * corr
    return max_conv, grad_a


# ----------------------------------------------------------------------
# Simple probabilistic utilities (soft‑max on the simplex)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    s = e.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / z.size)
    return e / s


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it that reproduces `a` after a soft‑max."""
    return np.log(a + eps)


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a normalised probability vector."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    s = new_a.sum()
    if s == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= s
    return new_a


# ----------------------------------------------------------------------
# Parametric family handling
# ----------------------------------------------------------------------


_FAMILIES = [0, 1, 2, 3, 4]   # identifiers for the five families above


def _random_params(rng: np.random.Generator):
    """Draw a random (family_id, p1, p2) triple within sensible bounds."""
    fam = rng.choice(_FAMILIES)
    if fam == 0:          # power‑law
        p = rng.uniform(0.01, 10.0)
        return (fam, p, None)
    if fam == 1:          # symmetric product‑power‑law
        p = rng.uniform(-5.0, 5.0)
        return (fam, p, None)
    if fam == 2:          # exponential decay (lam sampled log‑uniform)
        loglam = rng.uniform(-12.0, -1.0)   # lam = 10^loglam
        lam = 10.0 ** loglam
        return (fam, lam, None)
    if fam == 3:          # logistic
        alpha = rng.uniform(-10.0, 10.0)
        return (fam, alpha, None)
    # fam == 4  ->  Beta
    a = rng.uniform(0.1, 5.0)
    b = rng.uniform(0.1, 5.0)
    return (fam, a, b)


def _mutate_params(params, rng: np.random.Generator, scale: float):
    """Small random mutation of a parameter tuple."""
    fam, p1, p2 = params

    # Occasionally switch to a completely new family
    if rng.random() < 0.08:
        return _random_params(rng)

    if fam == 0:          # power‑law (p in [0.01,10])
        low, high = 0.01, 10.0
        p_new = p1 + scale * (high - low) * rng.standard_normal()
        p_new = float(np.clip(p_new, low, high))
        return (fam, p_new, None)

    if fam == 1:          # sym‑prod‑power‑law (p in [-5,5])
        low, high = -5.0, 5.0
        p_new = p1 + scale * (high - low) * rng.standard_normal()
        p_new = float(np.clip(p_new, low, high))
        return (fam, p_new, None)

    if fam == 2:          # exponential (lam ∈ [1e-12, 1e-1])
        # mutating in log‑space is more natural
        loglam = np.log10(p1)
        loglam_new = loglam + scale * 3.0 * rng.standard_normal()
        loglam_new = np.clip(loglam_new, -12.0, -1.0)
        lam_new = 10.0 ** loglam_new
        return (fam, lam_new, None)

    if fam == 3:          # logistic (alpha ∈ [‑10,10])
        low, high = -10.0, 10.0
        p_new = p1 + scale * (high - low) * rng.standard_normal()
        p_new = float(np.clip(p_new, low, high))
        return (fam, p_new, None)

    # fam == 4  ->  Beta (both params in [0.1,5])
    low, high = 0.1, 5.0
    a_new = p1 + scale * (high - low) * rng.standard_normal()
    b_new = p2 + scale * (high - low) * rng.standard_normal()
    a_new = float(np.clip(a_new, low, high))
    b_new = float(np.clip(b_new, low, high))
    return (fam, a_new, b_new)


def _generate_distribution(params, n: int) -> np.ndarray:
    """Instantiate a normalised probability vector from a param tuple."""
    fam, p1, p2 = params
    if fam == 0:
        return _power_law_dist(n, p1)
    if fam == 1:
        return _sym_prod_power_law_dist(n, p1)
    if fam == 2:
        return _exp_decay_dist(n, p1)
    if fam == 3:
        return _logistic_dist(n, p1)
    if fam == 4:
        return _beta_dist(n, p1, p2)
    # Fallback – uniform
    return np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Adam‑based mirror‑descent refinement on the simplex
# ----------------------------------------------------------------------


def _refine_distribution(a_initial: np.ndarray, deadline: float) -> np.ndarray:
    """
    Mirror‑descent (Adam on logits) for a fixed length `n`.
    Terminates when the time budget expires.
    """
    n = a_initial.shape[0]
    eps = 1e-16
    z = np.log(a_initial + eps) + 0.01 * np.random.randn(n)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    base_alpha = 0.35          # base step‑size for logits
    eps_factor = 0.08          # start smoothing factor ε = eps_factor·max_conv
    eps_min = 5e-5
    eps_decay = 0.9995

    best_seq = _softmax(z)
    best_score = 2.0 * n * _conv_max(best_seq)
    last_improve = 0
    it = 0

    while time.time() < deadline - 2.0:
        it += 1
        a = _softmax(z)

        max_conv, grad_a = _conv_max_and_grad(a, eps_factor)
        cheap_score = 2.0 * n * max_conv

        if cheap_score < best_score - 1e-12:
            best_score = cheap_score
            best_seq = a.copy()
            last_improve = it

        # gradient w.r.t. logits (mirror‑descent on simplex)
        mu = np.dot(grad_a, a)
        grad_z = a * (grad_a - mu)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # inverse‑sqrt step‑size schedule
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step = - m_hat / (np.sqrt(v_hat) + eps_adam)

        # back‑tracking line‑search (≤ 4 halvings)
        accepted = False
        trial_alpha = alpha_t
        for _ in range(4):
            z_new = z + trial_alpha * step
            z_new = np.clip(z_new, -50.0, 50.0)
            a_new = _softmax(z_new)
            new_score = 2.0 * n * _conv_max(a_new)
            if new_score <= cheap_score:
                z = z_new
                accepted = True
                break
            trial_alpha *= 0.5

        if not accepted:
            # take a modest step anyway
            z += alpha_t * step
            z = np.clip(z, -50.0, 50.0)

        # occasional stochastic jitter to escape plateaux
        if it % 3000 == 0 and np.random.rand() < 0.12:
            z += 0.015 * np.random.randn(n)
            m *= 0.6
            v *= 0.6

        # decay smoothing
        eps_factor = max(eps_min, eps_factor * eps_decay)

        # restart if we're stuck for a very long time
        if it - last_improve > 200_000:
            print(f"[Refine] stagnation detected at iter {it}; restarting with jitter.")
            z = _logits_from_distribution(best_seq) + 0.02 * np.random.randn(n)
            m.fill(0.0)
            v.fill(0.0)
            eps_factor = 0.08
            last_improve = it

    return best_seq


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) improves the
    `evaluate_sequence` score.  The routine respects the supplied time budget
    (~1000 s) and uses a two‑phase hybrid optimiser:
      1. fast parametric search on a reduced support size,
      2. Adam‑based mirror‑descent refinement on the full support.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start = time.time()
    # keep a small safety margin for the final evaluation call
    final_deadline = start + budget_s - 3.0

    # ------------------------------------------------------------------
    # 0. Global fallback (if a previously discovered sequence exists)
    # ------------------------------------------------------------------
    try:
        if "height_sequence_1" in globals():
            seq0 = np.array(height_sequence_1, dtype=np.float64)
            seq0 = np.maximum(seq0, 0.0)
            s = seq0.sum()
            if s == 0.0:
                seq0 = np.full_like(seq0, 1.0 / seq0.shape[0])
            else:
                seq0 = seq0 / s
            global_candidate = seq0
        else:
            global_candidate = None
    except Exception:
        global_candidate = None

    # ------------------------------------------------------------------
    # 1. Fast parametric search (small support, many samples)
    # ------------------------------------------------------------------
    n_small = 12000
    n_full = 25000

    # initialise a modest population
    pop_size = 30
    population = []   # list of (params, dist, cheap_score)

    for _ in range(pop_size):
        pars = _random_params(rng)
        dist = _generate_distribution(pars, n_small)
        score = 2.0 * n_small * _conv_max(dist)
        population.append((pars, dist, score))

    # optionally inject the global candidate (if any) into the pool
    if global_candidate is not None:
        cand_small = _interpolate_distribution(global_candidate, n_small)
        cand_score = 2.0 * n_small * _conv_max(cand_small)
        population.append((None, cand_small, cand_score))

    # keep the best ``pop_size`` individuals only
    population.sort(key=lambda x: x[2])
    population = population[:pop_size]

    best_params, best_dist_small, best_score_small = population[0]

    # meta‑parameters for the evolutionary loop
    top_k = 10
    mut_scale = 0.20
    param_deadline = start + 200.0   # around 200 s for the parametric phase

    while time.time() < param_deadline:
        # parent selection (uniform over the current elite)
        parent_idx = rng.integers(0, min(top_k, len(population)))
        parent_pars = population[parent_idx][0]

        child_pars = _mutate_params(parent_pars, rng, mut_scale)
        child_dist = _generate_distribution(child_pars, n_small)
        child_score = 2.0 * n_small * _conv_max(child_dist)

        # replace the worst individual if we improved
        if child_score < population[-1][2]:
            population[-1] = (child_pars, child_dist, child_score)
            population.sort(key=lambda x: x[2])
            if child_score < best_score_small:
                best_score_small = child_score
                best_params = child_pars
                best_dist_small = child_dist
                print(f"[ParamSearch] New best cheap score {best_score_small:.6f}")

        # occasional random newcomer to keep diversity
        if rng.random() < 0.06:
            rnd_pars = _random_params(rng)
            rnd_dist = _generate_distribution(rnd_pars, n_small)
            rnd_score = 2.0 * n_small * _conv_max(rnd_dist)
            if rnd_score < population[-1][2]:
                population[-1] = (rnd_pars, rnd_dist, rnd_score)
                population.sort(key=lambda x: x[2])
                if rnd_score < best_score_small:
                    best_score_small = rnd_score
                    best_params = rnd_pars
                    best_dist_small = rnd_dist
                    print(f"[ParamSearch] Random newcomer improved cheap score to {best_score_small:.6f}")

        # slowly shrink the mutation amplitude
        mut_scale = max(0.02, mut_scale * 0.9999)

    # ------------------------------------------------------------------
    # 2. Upsample the best small‑scale distribution to the full length
    # ------------------------------------------------------------------
    best_dist_full = _interpolate_distribution(best_dist_small, n_full)

    # ------------------------------------------------------------------
    # 3. Mirror‑descent refinement on the full support
    # ------------------------------------------------------------------
    refined_seq = _refine_distribution(best_dist_full, final_deadline)

    final_seq = refined_seq.tolist()

    # ------------------------------------------------------------------
    # 4. Final exact evaluation (the platform will also call evaluate_sequence)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[Result] Final exact score = {final_score:.6f}")
    except Exception as exc:
        final_score = float("inf")
        print(f"[Result] evaluate_sequence raised an exception: {exc}")

    return final_seq
