"""
Non‑negative step‑function optimiser (new version).

The optimiser works in three stages:

1. **Diverse shape sampling** – random families of symmetric shapes
   (exponential, Gaussian, polynomial, raised‑cosine, Beta‑like) are
   generated for several support sizes.  Each candidate is evaluated
   exactly (via FFT) and the best few are kept.

2. **Refinement with Adam on the simplex** – the best candidate from
   stage 1 (or the previous global best, if available) is refined by a
   fast Adam optimiser that works on the log‑softmax (mirror) parametrisation
   of the simplex.  A very aggressive β‑schedule (smooth‑max → exact
   max) together with occasional jitter and restarts helps to escape
   plateaus.

3. **Targeted simulated‑annealing** – a short SA run moves a small
   amount of mass from positions that contribute most to the current
   maximal convolution entry to random other positions, accepting
   non‑improving moves with a Metropolis probability.  Finally tiny
   entries are pruned.

All work respects the overall time budget (default 1000 s).  The
returned list is a plain list of non‑negative floats suitable for
``evaluate_sequence``.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return conv_full[: 2 * n - 1]


def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation g_i = Σ_t w[i+t]·a[t] for i=0,…,n‑1.
    Implemented via FFT.
    """
    n = a.shape[0]
    m = w.shape[0]                     # = 2n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) assuming sum(a)=1.
    """
    n = a.size
    conv = _conv_self_fft(a)
    return float(2.0 * n * conv.max())

# Adam optimiser on the simplex (mirrored‑gradient version)
# ----------------------------------------------------------------------
def _run_opt(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
) -> tuple[list[float], float]:
    """
    Adam optimiser on the log‑it simplex.
    Returns the best sequence (list) and its exact objective.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # initialise a on the simplex
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
        a = a / a.sum()
    else:
        a = rng.random(n)
        a = a / a.sum()

    # logits (log‑softmax parametrisation)
    x = np.log(a + 1e-16)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # schedule parameters
    beta_start = 0.5
    beta_end = 50000.0                # sharper smoothing than the previous run
    lr_start = 0.20
    lr_end = 0.0003

    best_a = a.copy()
    best_obj = _objective(best_a)

    it = 0
    while time.time() - start < time_limit:
        it += 1
        a = _softmax(x)                     # stay on the simplex

        conv = _conv_self_fft(a)
        M = conv.max()
        obj = float(2.0 * n * M)            # sum(a)=1 ⇒ denominator = 1
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        # time‑based schedules
        t = min((time.time() - start) / time_limit, 1.0)
        beta = beta_start * np.exp(np.log(beta_end / beta_start) * t)
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t)

        # smooth‑max weights
        if beta > 10000.0:
            # hard max – only the maximal entry contributes
            kmax = int(np.argmax(conv))
            w = np.zeros_like(conv)
            w[kmax] = 1.0
        else:
            shifted = conv - M                # ≤ 0
            w = np.exp(beta * shifted)
            s = w.sum()
            if s == 0.0:
                w = np.ones_like(conv) / conv.size
            else:
                w /= s

        # gradient of (smooth) max wrt a
        g_a = 2.0 * _correlation_fft(w, a)

        # natural gradient in log‑it space
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to pull out of plateaus
        if it % 3000 == 0:
            jitter = rng.normal(scale=0.012, size=n)
            x += jitter

        # restart from the current best every now and then
        if it % 60000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.006, size=n)
            m.fill(0.0)
            v.fill(0.0)

    # --------------------------------------------------------------
    # compress negligible entries
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = _objective(best_a)
        print(f"[compress] reduced length → {best_a.size}, O={best_obj:.6f}")

    # --------------------------------------------------------------
    # brief targeted local search (mass‑transfer moves)
    remaining = time_limit - (time.time() - start)
    if remaining > 0.7:
        local_start = time.time()
        a_loc = best_a.copy()
        obj_loc = best_obj
        n_loc = a_loc.size
        while time.time() - local_start < remaining * 0.3:
            i, j = rng.choice(n_loc, size=2, replace=False)
            delta = a_loc[i] * rng.random() * 0.025   # ≤2.5 % of a_i
            if delta <= 0.0:
                continue
            a_new = a_loc.copy()
            a_new[i] -= delta
            a_new[j] += delta
            conv_new = _conv_self_fft(a_new)
            obj_new = float(2.0 * n_loc * conv_new.max())
            if obj_new < obj_loc:
                a_loc, obj_loc = a_new, obj_new
                if obj_new < best_obj:
                    best_a, best_obj = a_new, obj_new
        print(f"[local] final O={best_obj:.6f}")

    return best_a.tolist(), best_obj


# ----------------------------------------------------------------------
# Simulated‑annealing refinement
# ----------------------------------------------------------------------
def _run_sa(
    init_seq: list[float],
    time_limit: float,
    seed: int = 0,
) -> tuple[list[float], float]:
    """
    Targeted simulated‑annealing refinement.
    Small mass is moved from positions that contribute most to the
    current maximal convolution entry.
    """
    start = time.time()
    rng = np.random.default_rng(seed)
    a = np.asarray(init_seq, dtype=float)
    if a.sum() < 1e-12:
        a = np.ones_like(a) / a.size
    else:
        a = a / a.sum()
    n = a.shape[0]

    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())
    best_seq = a.copy()

    # temperature schedule
    T = best_obj * 0.12 + 1e-6
    T_min = best_obj * 1e-6 + 1e-9
    decay = 0.994
    it = 0

    while time.time() - start < time_limit:
        it += 1
        k_max = int(np.argmax(conv))

        # contribution of each i to the maximal entry
        i_vals = np.arange(n)
        j_vals = k_max - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i_vals[mask]] * a[j_vals[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            probs = contrib / total_contrib
            i = rng.choice(n, p=probs)
        else:
            i = rng.integers(n)

        # choose a different destination uniformly
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # move a modest fraction of mass
        delta = a[i] * rng.random() * 0.07   # up to 7 % of a[i]
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.clip(a_new, 0.0, None)
        a_new /= a_new.sum()

        conv_new = _conv_self_fft(a_new)
        obj_new = float(2.0 * n * conv_new.max())

        # Metropolis acceptance
        if obj_new < best_obj - 1e-13:
            a, conv, best_obj, best_seq = a_new, conv_new, obj_new, a_new.copy()
            print(f"[SA it={it}] new best O={best_obj:.6f}")
        else:
            prob = np.exp(-(obj_new - best_obj) / T) if T > 0 else 0.0
            if rng.random() < prob:
                a, conv = a_new, conv_new

        # temperature decay
        if it % 2500 == 0:
            T = max(T * decay, T_min)

        # occasional global jitter
        if it % 12000 == 0:
            noise = rng.normal(scale=0.012, size=n)
            a = np.clip(a + noise, 0.0, None)
            a /= a.sum()
            conv = _conv_self_fft(a)

    # final compression of negligible entries
    thresh = 1e-12
    mask = best_seq > thresh
    if mask.sum() < best_seq.size:
        comp = best_seq[mask]
        comp /= comp.sum()
        comp_obj = float(2.0 * comp.size * _conv_self_fft(comp).max())
        if comp_obj < best_obj:
            best_seq, best_obj = comp, comp_obj
            print(f"[SA compress] reduced length → {comp.size}, O={best_obj:.6f}")

    return best_seq.tolist(), best_obj


# ----------------------------------------------------------------------
# Random shape families (used in the exploratory stage)
# ----------------------------------------------------------------------
def _shape_symmetric_exponential(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n)
    centre = (n - 1) / 2.0
    return np.exp(-alpha * np.abs(i - centre))


def _shape_symmetric_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n)
    centre = (n - 1) / 2.0
    return np.exp(-((i - centre) ** 2) / (2.0 * sigma ** 2))


def _shape_polynomial(n: int, p: float) -> np.ndarray:
    """Shape a_i = (1 - ((i-centre)/centre)^2)^p, clipped at 0."""
    i = np.arange(n)
    centre = (n - 1) / 2.0
    d = np.abs(i - centre) / centre
    a = (1.0 - d ** 2) ** p
    a[a < 0] = 0.0
    return a


def _shape_hann_power(n: int, p: float) -> np.ndarray:
    """Base Hann window raised to a power."""
    i = np.arange(n)
    base = np.sin(np.pi * i / (n - 1)) ** 2
    return base ** p


def _shape_beta(n: int, p: float, q: float) -> np.ndarray:
    """Beta‑like shape a_i ∝ (i+1)^p·(n‑i)^q."""
    i = np.arange(n)
    return (i + 1) ** p * (n - i) ** q


def _random_shape_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Generate a random shape from a small collection of families,
    normalise it to sum = 1.
    """
    family = rng.integers(6)
    if family == 0:
        # symmetric exponential
        alpha = rng.uniform(0.0, 5.0)
        a = _shape_symmetric_exponential(n, alpha)
    elif family == 1:
        # symmetric Gaussian
        sigma = rng.uniform(n / 20.0, n / 2.0)
        a = _shape_symmetric_gaussian(n, sigma)
    elif family == 2:
        # polynomial (raised‑cosine‑like)
        p = rng.uniform(0.3, 4.5)
        a = _shape_polynomial(n, p)
    elif family == 3:
        # Hann raised to a power
        p = rng.uniform(0.3, 3.0)
        a = _shape_hann_power(n, p)
    elif family == 4:
        # Beta‑like
        p = rng.uniform(-1.8, 1.8)
        q = rng.uniform(-1.8, 1.8)
        a = _shape_beta(n, p, q)
    else:
        # pure random Dirichlet‑like
        a = rng.random(n)

    # ensure non‑negativity and normalise
    a = np.clip(a, 0.0, None)
    total = a.sum()
    if total < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / total
    return a


def _search_random_shapes(
    n: int,
    time_limit: float,
    rng: np.random.Generator,
) -> tuple[list[float] | None, float]:
    """
    Randomly generate shapes for a fixed support size n,
    keep the best one found within the allocated time.
    """
    start = time.time()
    best_seq = None
    best_obj = float("inf")
    it = 0
    while time.time() - start < time_limit:
        it += 1
        a = _random_shape_candidate(n, rng)
        conv = _conv_self_fft(a)
        obj = float(2.0 * n * conv.max())
        if obj < best_obj:
            best_obj = obj
            best_seq = a.copy()
            print(f"[rand n={n:4d} it={it:06d}] new best O={best_obj:.6f}")
    if best_seq is None:
        return None, float("inf")
    return best_seq.tolist(), best_obj


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **_ignore,
) -> list[float]:
    """
    Entry point used by the evaluation harness.
    Runs a multi‑stage search (random shape sampling → Adam
    refinement → simulated annealing) respecting the time budget.
    Returns a plain list of non‑negative floats.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5        # safety margin

    # ------------------------------------------------------------------
    # Optional warm‑start from a previous construction
    init_seq = None
    if "height_sequence_1" in globals():
        seq0 = np.asarray(height_sequence_1, dtype=float)
        seq0 = np.clip(seq0, 0.0, None)
        if seq0.sum() < 0.01:
            seq0 = np.ones_like(seq0)
        seq0 = seq0 / seq0.sum()
        init_seq = seq0

    # ------------------------------------------------------------------
    # Choose a set of support sizes (larger sizes give more freedom)
    lengths = [2000, 4000, 6000, 8000, 12000]
    if init_seq is not None and init_seq.size not in lengths:
        lengths.insert(0, init_seq.size)

    best_seq = None
    best_obj = float("inf")

    # ------------------------------------------------------------------
    # Stage 1 – random shape exploration
    # Allocate roughly 30 % of the total budget to this stage.
    stage1_budget = (deadline - time.time()) * 0.30
    per_len = stage1_budget / max(len(lengths), 1)
    for n in lengths:
        now = time.time()
        remaining = deadline - now
        if remaining < 3.0:
            break
        # give each length its share (but never exceed the global deadline)
        this_budget = min(per_len, remaining * 0.90)
        seq, val = _search_random_shapes(n, this_budget, rng)
        if seq is not None and val < best_obj:
            best_obj, best_seq = val, np.asarray(seq, dtype=float)
            print(f"[global] after random search best O={best_obj:.6f} (n={n})")
        if time.time() >= deadline - 0.5:
            break

    # ------------------------------------------------------------------
    # Stage 2 – Adam refinement (mirrored‑gradient descent)
    # Allocate most of the remaining time to Adam.
    remaining = deadline - time.time()
    if best_seq is not None and remaining > 6.0:
        # run a longer Adam refinement on the current best length
        n_best = best_seq.size
        # give a bit of breathing room for the final SA stage
        adam_time = remaining * 0.70
        seq_opt, val_opt = _run_opt(
            n=n_best,
            time_limit=adam_time,
            init_a=best_seq,
            seed=seed + 777,
        )
        if val_opt < best_obj:
            best_obj, best_seq = val_opt, np.asarray(seq_opt, dtype=float)
            print(f"[refine] Adam improvement: O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – targeted simulated‑annealing refinement
    remaining = deadline - time.time()
    if best_seq is not None and remaining > 4.0:
        seq_sa, val_sa = _run_sa(best_seq.tolist(),
                                 time_limit=remaining,
                                 seed=seed + 9999)
        if val_sa < best_obj:
            best_obj, best_seq = val_sa, np.asarray(seq_sa, dtype=float)
            print(f"[final] after SA best O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Fallback – ensure we always return a non‑empty sequence
    if best_seq is None or best_seq.size == 0:
        best_seq = np.ones(200, dtype=float) / 200.0
        best_obj = _objective(best_seq)
        print("[fallback] using uniform sequence")

    # ------------------------------------------------------------------
    # Final sanity check (also prints the score as observed by the harness)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_seq.tolist()
