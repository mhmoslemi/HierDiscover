#!/usr/bin/env python
"""
Non‑negative step‑function optimiser (independent implementation).

The algorithm:
1.   Builds a portfolio of cheap parametric families (power‑law,
     symmetric product‑power‑law, exponential, logistic, Gaussian,
     random Dirichlet) and selects the best among them as a warm start.
2.   Optimises the probability vector on the simplex with Adam‑based
     mirror‑descent, using a smoothed “soft‑max” of the self‑convolution.
     A back‑tracking line‑search guarantees the cheap score never increases.
3.   Occasionally injects jitter, reduces the smoothing factor,
     restarts after long stagnation, and upsamples the support if time permits.
4.   Returns the best distribution discovered; a final call to
     ``evaluate_sequence`` gives the exact score.

The routine respects a user‑supplied time budget (default ~1000 s) and
never returns an invalid sequence.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max, returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    s = e.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / z.size)
    return e / s


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it that reproduces `a` after a soft‑max."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """Smallest power‑of‑two ≥ 3·n‑2 (enough for all needed convolutions)."""
    return 1 << ((3 * n - 2).bit_length())


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    s = new_a.sum()
    if s == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= s
    return new_a


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{‑p}, normalised."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _sym_product_power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ ((i+1)*(n‑i))^{‑p}, symmetric, normalised."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) * (n - i)) ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _exp_decay_distribution(n: int, lam: float) -> np.ndarray:
    """a_i ∝ e^{‑lam·i}+e^{‑lam·(n‑1‑i)}, symmetric exponential."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _logistic_distribution(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ 1/(1+e^{α·(i‑mid)}), normalised; heavy on the left side."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = 1.0 / (1.0 + np.exp(alpha * (i - mid)))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _gaussian_distribution(n: int, mu: float, sigma: float) -> np.ndarray:
    """a_i ∝ exp(‑(i‑mu)²/(2σ²)), normalised."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Returns (max_conv, grad_a) where

        grad_a = ∇_a [ 2·n·smooth_max(a*a) ]

    The smoothing constant is ε = eps_factor·max_conv.
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    # FFT of a (zero‑padded)
    a_fft = np.fft.rfft(a, L)

    # Linear self‑convolution a*a
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # Smoothed soft‑max weights (log‑sum‑exp trick)
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv                # ≤ 0 for stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # Correlation a ⋆ w  (the gradient of the smooth max)
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # Gradient (already multiplied by 4·n to match the 2·n factor)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a

        alpha = np.random.rand() * 0.9 + 0.1  # concentration in (0.1,1.0)
        cand = np.random.dirichlet(np.full(n, alpha))
        score = 2.0 * n * _conv_max(cand)
        if score < best_cheap:
            best_a = cand
            best_cheap = score

    # fallback – uniform distribution
    if best_a is None:
        best_a = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # 2. Mirror‑descent on the probability simplex (Adam‑style).
    # ------------------------------------------------------------------
    a0 = best_a.astype(np.float64)
    a0 = np.maximum(a0, 0.0)
    s = a0.sum()
    if s == 0.0:
        a0[:] = 1.0 / n
    else:
        a0 /= s

    # small random jitter in the log‑it space
    z = _logits_from_distribution(a0)
    z += 0.01 * np.random.randn(n)

    # Adam state
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    # hyper‑parameters for the schedule
    base_alpha = 0.30          # initial log‑it step size
    eps_factor = 0.20          # start smoothing (ε = eps_factor·max_conv)
    eps_factor_min = 5e-6
    eps_decay = 0.9995
    report_every = 20000

    best_seq = _softmax(z)
    best_score = 2.0 * n * _conv_max(best_seq)
    last_improve_it = 0
    it = 0

    while time.time() < deadline:
        it += 1

        # ----- current point on the simplex -----
        a = _softmax(z)

        # ----- cheap evaluation + smoothed gradient -----
        max_conv, grad_a = _conv_max_and_grad(a, eps_factor)
        cheap_score = 2.0 * n * max_conv

        # ----- keep the best ever seen -----
        if cheap_score < best_score - 1e-12:
            best_score = cheap_score
            best_seq = a.copy()
            last_improve_it = it
            if it % report_every == 0:
                print(f"[Iter {it:07d}] NEW BEST cheap = {cheap_score:.6f}")

        # ----- gradient w.r.t. logits (mirror descent) -----
        mu = np.dot(grad_a, a)                # ⟨∇f, a⟩
        grad_z = a * (grad_a - mu)            # ∇_z f = diag(a)·(∇_a f – μ·1)

        # ----- Adam moment updates (bias‑corrected) -----
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # ----- step‑size schedule (inverse‑sqrt decay) -----
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step_dir = - m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- back‑tracking line search (up to 5 halvings) -----
        accepted = False
        trial_alpha = alpha_t
        for _ in range(5):
            z_new = z + trial_alpha * step_dir
            z_new = np.clip(z_new, -50.0, 50.0)   # keep exp stable
            a_new = _softmax(z_new)
            new_score = 2.0 * n * _conv_max(a_new)
            if new_score <= cheap_score:
                z = z_new
                accepted = True
                break
            trial_alpha *= 0.5

        if not accepted:
            # No improvement – still move a bit.
            z += alpha_t * step_dir
            z = np.clip(z, -50.0, 50.0)

        # ----- occasional random perturbations (to escape plateaux) -----
        if it % 5000 == 0 and np.random.rand() < 0.15:
            z += 0.02 * np.random.randn(n)
            m *= 0.7
            v *= 0.7

        # ----- gradually reduce smoothing factor -----
        eps_factor = max(eps_factor_min, eps_factor * eps_decay)

        # ----- restart after prolonged stagnation -----
        if it - last_improve_it > 300_000:
            print(f"[Iter {it}] Stagnation – applying jitter restart.")
            z = _logits_from_distribution(best_seq) + 0.03 * np.random.randn(n)
            m.fill(0.0)
            v.fill(0.0)
            eps_factor = 0.20
            last_improve_it = it

        # ----- up‑sample if time permits and we are not already huge -----
        if (it % 100_000 == 0) and (n < 40000) and (time.time() + 30.0 < deadline):
            new_n = min(40000, int(n * 1.30))
            print(f"[Iter {it}] Upsampling support: {n} → {new_n}")
            best_seq = _interpolate_distribution(best_seq, new_n)
            z = _logits_from_distribution(best_seq) + 0.005 * np.random.randn(new_n)
            m = np.zeros_like(z)
            v = np.zeros_like(z)
            n = new_n
            best_score = 2.0 * n * _conv_max(best_seq)
            it = 0
            last_improve_it = 0
            eps_factor = 0.20
            # continue with enlarged problem size

    # ------------------------------------------------------------------
    # 3. Final exact evaluation with the provided evaluate_sequence()
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # ------------------------------------------------------------------
    # 4. Return the best sequence discovered.
    # ------------------------------------------------------------------
    return final_seq
