"""
Proposes a non‑negative step‑function sequence that minimises  

    O(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

Algorithm
---------

1. **Initialisation** – start from the supplied sequence (if present) or a uniform
   distribution, normalise it to sum = 1.

2. **Fast simulated‑annealing** – repeatedly transfer a small amount of mass
   `δ` from a source index `i` to a destination `j`.  The convolution is updated
   in O(n) using the exact linear formula  

        conv' = conv
                - 2·δ·shift_i(a)      (source removal)
                + 2·δ·shift_j(a)      (destination addition)
                + δ²·e_{2i} + δ²·e_{2j} – 2·δ²·e_{i+j}

   where `shift_k(a)` is the vector `a` placed at positions `k … k+n‑1` inside
   the full convolution of length `2·n‑1`.  A Metropolis rule with an
   exponentially decaying temperature accepts uphill moves.

3. **Targeted moves** – with high probability the source index is chosen from
   the set of indices that actually contribute to the current maximal
   convolution entry, biasing the transfer toward reducing that peak.

4. **Periodic exact sub‑gradient step** – every few thousand iterations a true
   sub‑gradient of the max‑convolution is applied and the sequence is projected
   back onto the probability simplex.  This keeps the walk directed.

5. **Best‑ever tracking** – the best sequence (lowest `max(conv)`) seen so far is
   stored and printed.

6. **Final polish** – a short projected sub‑gradient descent (using the full
   FFT‑based convolution) cleans up any remaining slack.

The whole search respects the 1000 s wall‑time budget and returns the best
sequence found as a plain Python list of floats.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, ∑x=1}."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # all non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution (used for full recompute)
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full (non‑circular) convolution a*a."""
    return np.convolve(a, a)


# ----------------------------------------------------------------------
# Sub‑gradient of the maximal convolution entry
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. a."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
# Fast update of the full convolution after moving mass δ from i → j
def _update_conv(conv: np.ndarray, a: np.ndarray, i: int, j: int, delta: float) -> np.ndarray:
    """Return a new convolution after transferring `delta` from i to j."""
    n = a.size
    new_conv = conv.copy()
    # linear part: -2·δ·shift_i + 2·δ·shift_j
    new_conv[i:i + n] -= 2.0 * delta * a
    new_conv[j:j + n] += 2.0 * delta * a
    # quadratic correction: +δ² at 2i and 2j, -2·δ² at i+j
    two_i = 2 * i
    two_j = 2 * j
    ij = i + j
    if two_i < new_conv.size:
        new_conv[two_i] += delta * delta
    if two_j < new_conv.size:
        new_conv[two_j] += delta * delta
    if ij < new_conv.size:
        new_conv[ij] -= 2.0 * delta * delta
    return new_conv

# Fast simulated annealing with targeted mass‑transfer moves
def _fast_simulated_annealing(a0: np.ndarray,
                              time_deadline: float,

                              subgrad_interval: int = 5000) -> tuple[np.ndarray, float]:
    """
    Perform simulated annealing using O(n) convolution updates.
    Returns the best sequence found and its maximal convolution entry.
    """
    a = a0.copy()
    n = a.size
    conv = _full_convolution(a)
    cur_max = conv.max()
    best_a = a.copy()
    best_max = cur_max
    T = temperature0
    it = 0

    # helpers for periodic pruning
    prune_every = 2000
    prune_thresh = 1e-12

    while time.time() < time_deadline:
        # -------- choose move ------------------------------------------------
        if np.random.rand() < target_move_prob:
            # source from indices that actually contribute to the current peak
            kmax = int(conv.argmax())
            i_start = max(0, kmax - (n - 1))
            i_end = min(n - 1, kmax)
            idxs = np.arange(i_start, i_end + 1, dtype=int)
            # weight = a[i] * a[kmax-i]; avoid all‑zero mask
            partner_vals = a[kmax - idxs]
            weights = a[idxs] * partner_vals
            if weights.sum() > 0:
                probs = weights / weights.sum()
                i = np.random.choice(idxs, p=probs)
            else:
                i = np.random.randint(n)
        else:
            i = np.random.randint(n)

        # destination j (different from i)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # --------------------------------------------------------------------
        # amount of mass to move
        # scale the fraction with the current temperature (more aggressive early)
        max_frac = max(0.01, min(0.25, T * 5.0))
        frac = np.random.rand() * max_frac
        delta = a[i] * frac
        if delta <= 0.0:
            it += 1
            continue

        # ----------- compute tentative new convolution -----------------------
        new_conv = _update_conv(conv, a, i, j, delta)
        new_max = new_conv.max()

        # ---------- Metropolis acceptance ------------------------------------
        if new_max < cur_max or np.random.rand() < math.exp((cur_max - new_max) / max(T, 1e-12)):
            # accept move
            a[i] -= delta
            a[j] += delta
            conv = new_conv
            cur_max = new_max

            # keep best ever
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
                print(f"[SA] new best O={2 * n * best_max:.6f} (iter={it})")

        # ---------- occasional pruning of tiny entries ----------------------
        if (it & (prune_every - 1)) == 0:
            tiny = a < prune_thresh
            if tiny.any():
                a[tiny] = 0.0
                s = a.sum()
                if s == 0.0:
                    a = np.full_like(a, 1.0 / n)
                else:
                    a = a / s
                conv = _full_convolution(a)
                cur_max = conv.max()
                if cur_max < best_max:
                    best_max = cur_max
                    best_a = a.copy()
                    print(f"[SA] after pruning new best O={2 * n * best_max:.6f}")

        # ---------- periodic exact sub‑gradient descent step -----------------
        if subgrad_interval and it % subgrad_interval == 0 and it > 0:
            g = _subgradient_max_conv(a, conv)
            eta = 0.07 * max(0.0, 1.0 - it / (1e9))   # slowly decay step size
            a = _project_to_simplex(a - eta * g)
            conv = _full_convolution(a)
            cur_max = conv.max()
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
                print(f"[SA] sub‑grad improved O={2 * n * best_max:.6f} (iter={it})")

        # ---------- temperature decay ----------------------------------------
        T *= 0.9998   # slow geometric cooling
        if T < 1e-6:
            T = 1e-6

        it += 1

    return best_a, best_max

# Light‑weight projected sub‑gradient descent (final polishing)
def _polish_subgradient(a: np.ndarray,
                        max_iters: int,
                        time_deadline: float,
                        eta0: float = 0.25) -> np.ndarray:
    """
    Simple projected sub‑gradient descent on the exact objective.
    Uses full FFT‑based convolution for accurate gradients.
    """
    n = a.size
    # length for FFT (next power of two)
    L = 1 << ((3 * n - 2).bit_length())
    best_a = a.copy()
    best_obj = 2.0 * n * _full_convolution(a).max()

    for it in range(max_iters):
        if time.time() >= time_deadline:
            break
        conv = _full_convolution(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
        g = _subgradient_max_conv(a, conv)
        eta = eta0 * (0.9999 ** it)
        a = _project_to_simplex(a - eta * g)

    return best_a
