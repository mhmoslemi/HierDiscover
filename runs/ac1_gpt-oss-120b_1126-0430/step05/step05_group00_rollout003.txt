#!/usr/bin/env python
"""
Improved arcsine–interval construction with Adam‐mirrored descent.

1. Build a probability vector that is exactly the discretisation of the
   continuous arcsine density:
        a_i = (2/π)·[arcsin(√((i+1)/n)) – arcsin(√(i/n))] , i=0…n‑1 .
   This is the optimal (in the limit) distribution whose self‑convolution
   is uniform, so the objective 2·n·max(a*a) is close to 1.

2. Optimise the vector on the probability simplex using a mirrored‑descent
   Adam optimiser on a smooth approximation of the max‐convolution.
   The optimisation works in the log‑softmax (logit) space, which guarantees
   non‑negativity and unit sum.

3. After the main optimisation a short random‑logit “jitter” phase is run
   to escape possible local minima.

The routine respects the 1000 s budget, prints progress, and finally
returns the best sequence found as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max; returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it representation (inverse soft‑max) of a probability vector."""
    return np.log(np.maximum(a, eps))


def _fft_len_for_corr(n: int) -> int:
    """
    Smallest power‑of‑two ≥ 3·n‑2.
    Sufficient for linear convolution of length (2·n‑1) and correlation of
    length (3·n‑2).
    """
    return 1 << ((3 * n - 2).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a (no gradient)."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Return the (approximate) max of a*a together with the gradient of
    J(a)=2·n·max(a*a) using a smooth soft‑max approximation.

    Parameters
    ----------
    a : np.ndarray
        Probability vector (sum = 1, non‑negative).
    eps_factor : float
        Multiplicative factor for the smoothing constant:
        eps = eps_factor * max(conv).

    Returns
    -------
    max_conv : float
        Maximum value of the self‑convolution.
    grad_a : np.ndarray
        Gradient of J w.r.t. a (already multiplied by 4·n).
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    a_fft = np.fft.rfft(a, L)

    # linear self‑convolution (length 2·n‑1)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # smoothed soft‑max weights
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv                     # for numerical stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or np.isnan(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # cross‑correlation a ⋆ w (linear, length n)
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # gradient of 2·n·smooth_max ( = 4·n·corr )
    grad_a = 4.0 * n * corr
    return max_conv, grad_a


def arcsine_interval_distribution(n: int) -> np.ndarray:
    """
    Exact discretisation of the continuous arcsine density on [0,1].

    The probability mass of the i‑th interval [(i)/n,(i+1)/n) is
        (2/π)·[asin(√((i+1)/n)) – asin(√(i/n))].

    Returns a non‑negative vector of length n that sums to 1 (up to
    floating‑point rounding).
    """
    i = np.arange(n, dtype=np.float64)
    lower = np.sqrt(i / n)
    upper = np.sqrt((i + 1) / n)
    # Clip to avoid tiny numeric overshoots
    lower = np.clip(lower, 0.0, 1.0)
    upper = np.clip(upper, 0.0, 1.0)

    lower_a = np.arcsin(lower)
    upper_a = np.arcsin(upper)

    a = (2.0 / np.pi) * (upper_a - lower_a)
    a = np.maximum(a, 0.0)                     # safety
    total = a.sum()
    if total > 0.0:
        a /= total
    return a

    under the constraints a_i ≥ 0.  The method builds the arcsine‑interval
    distribution, then refines it with Adam on a smoothed max‑convolution
    objective, finally applying a brief random‑logit perturbation phase.
    """
    np.random.seed(seed)
    start = time.time()
    # Slight safety margin so the platform can finish the call cleanly
    deadline = start + budget_s - 1.5

    # ------------------------------------------------------------------
    # 0. Determine support size.
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
        print(f"[INFO] Using external length: n = {n}")
    except Exception:
        n = 20000
        print(f"[INFO] No external length – using default n = {n}")

    # ------------------------------------------------------------------
    # 1. Initialise with the exact arcsine‑interval distribution.
    # ------------------------------------------------------------------
    a = arcsine_interval_distribution(n)
    max_conv = _conv_max(a)
    best_score = 2.0 * n * max_conv
    best_a = a.copy()
    print(f"[INFO] Initial objective (approx) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Adam optimisation in log‑softmax (mirrored‑descent) space.
    # ------------------------------------------------------------------
    # Initialise logits from the probability vector
    z = _logits_from_distribution(best_a)
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_alpha = 0.6                # base learning rate for logits
    eps_factor = 0.2                # start smoothing factor
    eps_factor_min = 5e-5
    eps_factor_decay = 0.9995

    it = 0
    max_iters = 500_000
    last_improve = 0                # iteration of last improvement
    improve_patience = 30_000       # stop if no progress for this many iters

    while time.time() < deadline and it < max_iters:
        it += 1

        # Current probability vector
        a_cur = _softmax(z)

        # Smoothed max‑convolution and its gradient
        max_conv, grad_a = _conv_max_and_grad(a_cur, eps_factor)

        # Cheap objective value (exact max is cheap as we already have it)
        cheap_score = 2.0 * n * max_conv

        # Keep the best seen so far
        if cheap_score < best_score:
            best_score = cheap_score
            best_a = a_cur.copy()
            last_improve = it
            # Early‑exit if we beat the target comfortably
            if best_score <= 1.5:
                print(f"[INFO] Target reached (score={best_score:.6f}) at iter {it}")
                break

        # Mirror‑descent (log‑space) gradient
        mu = np.dot(grad_a, a_cur)                 # scalar <grad_a, a>
        grad_z = a_cur * (grad_a - mu)             # ∇_z J

        # Adam bias‑corrected update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step = -alpha_t * m_hat / (np.sqrt(v_hat) + eps_adam)
        z = z + step
        # Keep logits within a numerically safe range
        np.clip(z, -50.0, 50.0, out=z)

        # Reduce smoothing
        eps_factor = max(eps_factor_min, eps_factor * eps_factor_decay)

        # Small random jitter every few thousand steps (helps escape plateaus)
        if it % 5000 == 0 and np.random.rand() < 0.1:
            z += 0.01 * np.random.randn(n)

        # Logging
        if it % 10000 == 0:
            print(f"[PROGRESS] iter={it}  best={best_score:.6f}  cur={cheap_score:.6f}")

        # Stagnation check
        if (it - last_improve) > improve_patience and best_score < 1.2:
            print(f"[INFO] No improvement for {improve_patience} iters – stopping early.")
            break

    # ------------------------------------------------------------------
    # 3. Short random‑logit perturbation phase (optional).
    # ------------------------------------------------------------------
    # Use the current best logits as a centre and try a few noisy samples.
    z_center = _logits_from_distribution(best_a)
    trials = 0
    while time.time() < deadline and trials < 1500:
        trials += 1
        # Add modest Gaussian noise
        z_trial = z_center + 0.02 * np.random.randn(n)
        a_trial = _softmax(z_trial)
        score_trial = 2.0 * n * _conv_max(a_trial)
        if score_trial < best_score:
            best_score = score_trial
            best_a = a_trial.copy()
            z_center = z_trial.copy()
            print(f"[NOISE] improvement after {trials} noise trials: score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Return best sequence (plain list) and report exact evaluation.
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[RESULT] Exact evaluation = {final_score:.6f}")
    except Exception as exc:
        final_score = float("inf")
        print(f"[ERROR] Evaluation failed: {exc}")

    total_time = time.time() - start
    print(f"[INFO] Finished in {total_time:.1f}s, best approximate score = {best_score:.6f}")
    return final_seq
