def _surrogate_and_grad(u, alpha):
    a = np.exp(u)
    a_sum = a.sum()
    a = a / a_sum
    conv = _conv_fft(a, L)
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    sum_exp = exp_scaled.sum()
    loss = (2.0 * n / alpha) * (max_scaled + np.log(sum_exp))
    grad_corr = _smooth_gradient_plus_entropy(a, conv, alpha, L, lam=0.0) # returns grad_corr

    # transform to gradient w.r.t u via softmax jacobian

    avg = np.dot(grad_a, a)
    grad_u = a * (grad_a - avg)
    return loss, grad_u

def run_lbfgs(a0, alpha, max_iter):
    u0 = np.log(a0 + 1e-15)  # shape n
    res = scipy.optimize.minimize(fun=lambda u: _surrogate_and_grad(u, alpha),
                                  x0=u0,
                                  method='L-BFGS-B',
                                  jac=True,
                                  options={'maxiter': max_iter, 'disp': False})
    # convert to a
    a_opt = np.exp(res.x)
    a_opt /= a_opt.sum()
    return a_opt

"""
Propose a non‑negative step‑function (list of heights) that (approximately)
minimises

    O(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The strategy combines three ideas:

1️⃣  **Continuation with a smooth surrogate** – we minimise the
    log‑sum‑exp approximation of the max‑convolution.
    The variables are represented in the soft‑max (log‑probability) space,
    guaranteeing non‑negativity and the simplex constraint automatically.
    A sequence of increasing smoothing parameters `α` is solved with
    L‑BFGS‑B, each stage warm‑started from the previous one.

2️⃣  **Targeted stochastic refinement** – after the continuation we
    perform a fast simulated‑annealing style hill‑climber that moves a
    random fraction of mass from indices that contribute most to the
    current maximal convolution value to random destinations.
    Only improving moves are kept, which rapidly pushes the exact objective
    down.

3️⃣  **Final polishing** – a few projected‑gradient Adam steps on the
    original (non‑smoothed) objective remove any remaining slack.

All helper routines are top‑level, no closures are used.  The function
returns the best sequence found before the 1000 s budget expires.
"""

import time
import math
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution via FFT (length 2·n‑1)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return the full a⨂a convolution (length 2·n‑1) using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Exact objective (for a vector already normalised to sum 1)
def _objective_simplex(a: np.ndarray, n: int, L: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for sum(a)=1."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Gradient of the smooth surrogate (log‑sum‑exp) plus optional entropy term
def _smooth_gradient_plus_entropy(
    a: np.ndarray,
    conv: np.ndarray,
    alpha: float,
    L: int,
    lam: float,
) -> np.ndarray:
    """
    Returns the gradient of
        (2·n/α)·log∑_k e^{α·conv_k} + λ·∑ a_i·log a_i
    with respect to a (ignoring the constant factor 2·n).

    The first term reduces to 2·corr(w,a) where w_k ∝ e^{α·conv_k};
    the second term contributes λ·(log a_i + 1).
    """
    # ----- soft‑max weights ----------
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)          # stability
    w = exp_scaled / exp_scaled.sum()                 # shape (2n‑1,)

    # ----- correlation w ⋆ a (g_i = 2·∑_j w_{i+j} a_j) ----
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n_len = a.size
    grad_corr = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]

    # ----- entropy part ----------
    eps = 1e-20
    grad_ent = lam * (np.log(a + eps) + 1.0)

    return grad_corr + grad_ent


# ----------------------------------------------------------------------
# Surrogate loss (log‑sum‑exp) together with its gradient w.r.t. the
# unconstrained soft‑max variables u (such that a = softmax(u)).
def _surrogate_and_grad(u: np.ndarray, alpha: float, n: int, L: int):
    """Return (loss, grad_u) for the smooth max‑convolution surrogate."""
    a = np.exp(u)
    a_sum = a.sum()
    a = a / a_sum

    conv = _conv_fft(a, L)

    # loss = (2·n/α)·log∑ e^{α·conv}
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    sum_exp = exp_scaled.sum()
    loss = (2.0 * n / alpha) * (max_scaled + np.log(sum_exp))

    # gradient w.r.t. a (including factor 2·n)
    grad_corr = _smooth_gradient_plus_entropy(a, conv, alpha, L, lam=0.0)
    grad_a = 2.0 * n * grad_corr                     # ∂loss/∂a

    # chain rule for a = softmax(u)
    avg = np.dot(grad_a, a)                         # scalar = <grad_a, a>
    grad_u = a * (grad_a - avg)                     # ∂loss/∂u

    return loss, grad_u


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    the provided evaluation function.  Returns the best sequence found
    before the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # safety margin

    # ------------------- initialise -------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.array(height_sequence_1, dtype=float)
        a0 = np.maximum(base, 0.0)
        s = a0.sum()
        if s < 1e-12:
            a0 = np.full_like(base, 1.0 / base.size)
        else:
            a0 = a0 / s
        n = a0.size
        print(f"[init] start from supplied sequence, n={n}")
    else:
        n = 6000
        a0 = np.full(n, 1.0 / n)   # uniform start
        print(f"[init] start from uniform, n={n}")

    # FFT length – next power of two >= 2·n‑1
    L = 1
    while L < 2 * n - 1:
        L <<= 1

    best_seq = a0.copy()
    best_score = _objective_simplex(best_seq, n, L)
    print(f"[init] exact O={best_score:.6f}")

    # ----------------------------------------------------------------------
    # Phase 1 – continuation with L‑BFGS on the smooth surrogate
    # ----------------------------------------------------------------------

    # we will run a small number of LBFGS iterations for each α value

                best_score = cur_score
                best_seq = a.copy()
                print(f"[LBFGS] cand {cid}, α={alpha:.0f} → new best O={best_score:.6f}")

        # after finishing the α‑schedule, store the result as a possible restart
        if time.time() >= deadline:
            break

    # ----------------------------------------------------------------------
    # Phase 2 – targeted stochastic refinement (hill climbing)
    # ----------------------------------------------------------------------
    a = best_seq.copy()
    cur_obj = best_score
    temperature = 0.05
    temp_decay = 0.9999
    max_iter_hc = 300_000
    it_hc = 0
    print("[phase‑2] start hill‑climbing refinement")

    while time.time() < deadline and it_hc < max_iter_hc:
        # compute convolution and the index of its maximum
        conv = _conv_fft(a, L)
        k_max = int(np.argmax(conv))

        # contributions of each index to this maximal entry
        idx = np.arange(n)
        j = k_max - idx
        valid = (j >= 0) & (j < n)
        contrib = np.zeros_like(a)
        if valid.any():
            # ordered pair contribution (both (i,j) and (j,i))
            contrib[valid] = a[idx[valid]] * a[j[valid]] * 2.0
            # correct the double‑count for i == j (i.e. when 2*i == k_max)
            self_mask = valid & (idx == j)
            contrib[self_mask] = a[idx[self_mask]] ** 2
            # avoid moving from indices that have essentially zero mass
            mask_move = (a > 1e-14) & (contrib > 0)
            if mask_move.any():
                probs = contrib[mask_move]
                probs /= probs.sum()
                src = np.random.choice(np.where(mask_move)[0], p=probs)
                # choose a destination uniformly (different from src)
                dst = np.random.randint(n - 1)
                if dst >= src:
                    dst += 1
            else:
                src, dst = np.random.randint(n, size=2)
        else:
            src, dst = np.random.randint(n, size=2)

        if src != dst and a[src] > 1e-14:
            # move a random fraction of the source mass
            max_frac = 0.10
            frac = np.random.rand() * max_frac
            delta = a[src] * frac
            a_new = a.copy()
            a_new[src] -= delta
            a_new[dst] += delta
            a_new = _project_to_simplex(a_new)

            new_obj = _objective_simplex(a_new, n, L)

            # Metropolis‑type acceptance (temperature slowly decays)
            if new_obj < cur_obj or np.random.rand() < math.exp(
                (cur_obj - new_obj) / max(temperature, 1e-12)
            ):
                a = a_new
                cur_obj = new_obj
                if new_obj < best_score:
                    best_score = new_obj
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] new global best O={best_score:.6f}")

        temperature *= temp_decay
        it_hc += 1

    # ----------------------------------------------------------------------
    # Phase 3 – final Adam polishing (no smoothing, pure objective)
    # ----------------------------------------------------------------------
    print(f"[phase‑3] final Adam polishing, best O={best_score:.6f}")
    a = best_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    lr0 = 0.4
    alpha_big = 200_000.0      # very large → essentially the exact max
    polish_iters = 20_000
    it = 0
    while time.time() < deadline and it < polish_iters:
        conv = _conv_fft(a, L)
        cur_obj = _objective_simplex(a, n, L)
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[polish] iter {it:,} new best O={best_score:.6f}")

        # gradient of the (almost) exact max (log‑sum‑exp with huge α)
        grad_corr = _smooth_gradient_plus_entropy(a, conv, alpha_big, L, lam=0.0)
        grad = 2.0 * n * grad_corr        # include the missing 2·n factor

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        lr = lr0 / math.sqrt(it + 1)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = _project_to_simplex(a_new)
        a = a_new
        it += 1

    # ----------------------------------------------------------------------
    # Final sanity check with the provided evaluator
    # ----------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:
        print("[final] evaluate_sequence failed:", e)

    return best_seq.tolist()
