Need simplex projection function: Implementation per Duchi et al:

```
def _simplex_projection(v):
    # Projects onto simplex {x >=0, sum=1}
    # v is np array.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w
```

    """Stable soft‑max, returns a probability vector that sums to 1."""
    z_max = np.max(z)

    if s == 0.0:

    return e / s

    """Log‑it that reproduces `a` after a soft‑max."""
    return np.log(a + eps)

    """Full linear self‑convolution a*a (length 2n‑1)."""
    n = a.shape[0]

    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]

    return float(_conv_full(a).max())

def _grad_exact_max(a: np.ndarray, max_idx: int) -> np.ndarray:
    """
    Gradient of 2·n·max(a*a) w.r.t. a evaluated at the current max entry.
    """
    n = a.shape[0]
    # partner = max_idx - j
    j = np.arange(n)
    partner = max_idx - j
    valid = (partner >= 0) & (partner < n)
    grad = np.zeros_like(a)
    # derivative of max entry is 2·a[partner]; multiplied by 2·n gives 4·n·a[partner]
    grad[valid] = 4.0 * n * a[partner[valid]]
    return grad

    """Linearly interpolate a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    s = new_a.sum()
    if s == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= s
    return new_a

    i = np.arange(n, dtype=np.float64)

    i = np.arange(n, dtype=np.float64)

    if s == 0.0:

    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = 1.0 / (1.0 + np.exp(alpha * (i - mid)))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a

def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (non‑negative, sum = 1)."""
    # Duchi et al. algorithm, O(n log n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

The algorithm proceeds in four stages:

1.  Scan a handful of parametric families (including a heavy‑edge

    j = np.arange(n)
    partner = max_idx - j
    valid = (partner >= 0) & (partner < n)
    grad = np.zeros_like(a)
    # d/d a_j  (2·n·max) = 4·n·a_{max_idx‑j}  when the partner index is valid.
    grad[valid] = 4.0 * n * a[partner[valid]]
    return grad


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    s = new_a.sum()
    if s == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= s
    return new_a


# ----------------------------------------------------------------------
# Parametric families (used for the initial scan)
# ----------------------------------------------------------------------


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{‑p}, normalised."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _sym_product_power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ ((i+1)*(n‑i))^{‑p}, symmetric, normalised."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) * (n - i)) ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _exp_decay_distribution(n: int, lam: float) -> np.ndarray:
    """a_i ∝ e^{‑lam·i}+e^{‑lam·(n‑1‑i)}, symmetric exponential."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _logistic_distribution(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ 1/(1+e^{α·(i‑mid)}), normalised; heavy on the left side."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = 1.0 / (1.0 + np.exp(alpha * (i - mid)))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) improves the
    ``evaluate_sequence`` score.  The routine respects the supplied time
    budget (≈ 1000 s) and uses a simple Adam optimiser on the probability
    simplex, starting from a carefully scanned parametric family.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0   # safety margin for final evaluation

    # ------------------------------------------------------------------
    # 0. Determine the initial support size and an initial distribution.
    # ------------------------------------------------------------------
    try:
        # If the platform already supplied a decent construction, reuse it.
        init_global = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        n = max(20000, init_global.shape[0])
        a0 = np.maximum(init_global, 0.0)
        s = a0.sum()
        if s == 0.0:
            a0 = np.full_like(a0, 1.0 / a0.shape[0])
        else:
            a0 /= s
        print(f"[Init] Using supplied global sequence, length = {n}")
    except Exception:
        # No global variable – build a candidate by scanning families.
        n = 20000
        best_a = None
        best_score = float("inf")
        print("[Init] Scanning parametric families...")

        def _maybe_update(cand, label):
            nonlocal best_a, best_score
            sc = 2.0 * n * _conv_max(cand)
            if sc < best_score:
                best_score = sc
                best_a = cand.copy()
                print(f"[Init] New best {label}: score = {sc:.6f}")

        # Power‑law
        for p in np.linspace(0.1, 5.0, 50):
            if time.time() >= deadline:
                break
            _maybe_update(_power_law_distribution(n, p), f"power_law p={p:.3f}")

        # Symmetric product‑power‑law (the most promising family)
        for p in np.linspace(0.2, 5.0, 70):
            if time.time() >= deadline:
                break
            _maybe_update(_sym_product_power_law_distribution(n, p),
                          f"sym_prod_pow p={p:.3f}")

        # Exponential decay (symmetric)
        for loglam in np.linspace(-12, -3, 30):
            if time.time() >= deadline:
                break
            lam = 10.0 ** loglam
            _maybe_update(_exp_decay_distribution(n, lam),
                          f"exp_decay λ={lam:.2e}")

        # Logistic (left‑biased)
        for alpha in np.linspace(0.01, 10.0, 50):
            if time.time() >= deadline:
                break
            _maybe_update(_logistic_distribution(n, alpha),
                          f"logistic α={alpha:.3f}")

        # Fallback to uniform if nothing succeeded.
        if best_a is None:
            best_a = np.full(n, 1.0 / n)

        # Try a few convex combinations with the uniform distribution.
        uniform = np.full(n, 1.0 / n)
        for w in np.linspace(0.0, 1.0, 11):
            if time.time() >= deadline:
                break
            cand = w * best_a + (1.0 - w) * uniform
            cand = np.maximum(cand, 0.0)
            ss = cand.sum()
            if ss == 0.0:
                continue
            cand /= ss
            _maybe_update(cand, f"mix w={w:.2f}")

        a0 = best_a
        print(f"[Init] Chosen length = {n}")

    # ------------------------------------------------------------------
    # 1. Initialise Adam on the log‑it (soft‑max) representation.
    # ------------------------------------------------------------------
    # Convert the probability vector into logits, add a little noise.
    z = _logits_from_distribution(a0) + 0.01 * np.random.randn(n)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    base_alpha = 0.25          # base learning‑rate for logits
    report_every = 20000

    # ------------------------------------------------------------------
    # 2. Exact‑max sub‑gradient Adam loop.
    # ------------------------------------------------------------------
    a = _softmax(z)
    cur_max, cur_idx = _conv_max_and_idx(a)
    cur_score = 2.0 * n * cur_max
    best_seq = a.copy()
    best_score = cur_score
    print(f"[Iter 0] initial cheap score = {cur_score:.6f}")

    it = 0
    last_improve_iter = 0

    while time.time() < deadline:
        it += 1

        # ---- sub‑gradient of the current maximal entry -----------------
        grad_a = _grad_exact_max(a, cur_idx)

        # ---- gradient w.r.t. logits (mirror descent on the simplex) --
        mu = np.dot(grad_a, a)
        grad_z = a * (grad_a - mu)

        # ---- Adam moment updates --------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # ---- step size ------------------------------------------------
        step = base_alpha / (math.sqrt(it) + 1.0)
        direction = - m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- tentative update -----------------------------------------
        z_new = z + step * direction
        z_new = np.clip(z_new, -50.0, 50.0)   # keep exp(z) in a safe range
        a_new = _softmax(z_new)

        # ---- evaluate candidate -----------------------------------------
        new_max, new_idx = _conv_max_and_idx(a_new)
        new_score = 2.0 * n * new_max

        # ---- accept the step only if the cheap score does not increase --
        if new_score <= cur_score:
            z = z_new
            a = a_new
            cur_max = new_max
            cur_idx = new_idx
            cur_score = new_score
        # (otherwise we keep the current point – this guarantees monotonicity)

        # ---- record the best ever seen ---------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            last_improve_iter = it
            if it % report_every == 0 or it <= 10:
                print(f"[Iter {it:07d}] NEW BEST cheap = {best_score:.6f}")

        # ---- occasional random jitter to escape plateaus ---------------
        if it % 5000 == 0 and np.random.rand() < 0.15:
            z += 0.015 * np.random.randn(n)
            m *= 0.6
            v *= 0.6
            print(f"[Iter {it}] jitter restart")

        # ---- stagnation detection (re‑initialise around the best) ------
        if it - last_improve_iter > 250_000:
            print(f"[Iter {it}] Stagnation – re‑initialising with jitter")
            z = _logits_from_distribution(best_seq) + 0.02 * np.random.randn(n)
            m.fill(0.0)
            v.fill(0.0)
            last_improve_iter = it

        # ---- up‑sampling if enough budget remains (once) ---------------
        if n < 40000 and (it % 150_000 == 0) and (time.time() + 30.0 < deadline):
            new_n = min(40000, int(n * 1.5))
            print(f"[Iter {it}] Upsampling support: {n} → {new_n}")
            best_seq = _interpolate_distribution(best_seq, new_n)
            n = new_n
            # Reset optimiser at the new resolution.
            a = best_seq
            z = _logits_from_distribution(a) + 0.005 * np.random.randn(n)
            m = np.zeros_like(z)
            v = np.zeros_like(z)
            cur_max, cur_idx = _conv_max_and_idx(a)
            cur_score = 2.0 * n * cur_max
            best_score = cur_score
            it = 0
            last_improve_iter = 0
            continue

    # ------------------------------------------------------------------
    # 3. Final exact evaluation with the provided ``evaluate_sequence``.
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # ------------------------------------------------------------------
    # 4. Return the best sequence discovered.
    # ------------------------------------------------------------------
    return final_seq
