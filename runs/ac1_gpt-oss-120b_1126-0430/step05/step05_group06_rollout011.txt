def solve_max_sum_given_f(f, M):

    n = len(f)
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        # i from max(0, k-(n-1)) to min(k,n-1)
        i_start = max(0, k-(n-1))
        i_end = min(k, n-1)
        # for i in [i_start,i_end] inclusive
        for i in range(i_start, i_end+1):
            rows.append(k)
            cols.append(i)
            data.append(f[i])
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    b_ub = np.full(2*n-1, M)  # Upper bound M for each k.

    # Since we have only <= constraints; also b>=0 lower bound.

    # objective: maximize sum(b) => minimize -sum(b). So c = -np.ones(n).

    res = linprog(c=-np.ones(n), A_ub=A, b_ub=b_ub, bounds=[(0,None)]*n,
                  method='highs', options={'presolve': True, 'time_limit': maybe some secs})   # time limit per iteration.

Return b = res.x if res.success else None.

def polyak_subgradient_opt(a, n, L, deadline, lower_bound):

    best_a = a.copy()
    best_obj = _objective(a, n, L)
    while time.time() < deadline:
        conv = _conv_fft(a, L)
        cur_max = conv.max()
        cur_obj = 2.0 * n * cur_max
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()

        g = _subgradient_max_conv(a, conv)

        # compute step size
        # avoid division by zero if g norm small
        g_norm_sq = np.dot(g, g)
        if g_norm_sq == 0:
            break
        step = max(0.0, (cur_obj - lower_bound) / g_norm_sq)
        # optionally cap step
        step = min(step, 0.5)  # cap

        a_new = _project_to_simplex(a - step * g)

        # maybe add small random perturbation to escape.
        if np.random.rand() < 0.001:
            a_new += np.random.randn(n) * 1e-6
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()

        a = a_new

    return a, best_obj, best_a

"""
Proposes a non‑negative step‑function sequence that minimises  

    O(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The search proceeds in four stages:

1️⃣  Random restarts – Initialise from many Dirichlet draws and run a
    very short Adam optimisation on a smooth log‑sum‑exp surrogate.
2️⃣  Long Adam optimisation – Optimises the smooth surrogate with a
    gradually sharpening temperature schedule.
3️⃣  Polyak sub‑gradient descent – Uses the exact non‑smooth objective
    and a Polyak step‑size (gap / ‖g‖²) to drive the solution quickly
    towards the theoretical lower bound 2 n/(2 n‑1).  The step size is
    capped for stability and a tiny jitter is added occasionally.
4️⃣  Targeted simulated‑annealing – Refines the best vector by moving a
    small fraction of mass away from indices that participate in the
    current maximal convolution entry.  Moves are accepted with a
    Metropolis rule; the temperature decays slowly.

All phases respect the overall 1000 s wall‑time, constantly keep the
best sequence seen and finally verify it with the provided
``evaluate_sequence`` function.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, ∑x=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear self‑convolution via FFT
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return a*a (full convolution) using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Exact objective for a simplex‑normalised vector (∑a = 1)
def _objective(a: np.ndarray, n: int, L: int) -> float:
    """2·n·max(conv(a,a))  (a is assumed to sum to 1)."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Sub‑gradient of the maximal convolution entry (exact, non‑smooth)
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. a."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
# Gradient of the log‑sum‑exp (smooth max) surrogate
def _smooth_gradient(a: np.ndarray, conv: np.ndarray,
                     alpha: float, L: int) -> np.ndarray:
    """
    Compute ∇_a ( (2·n/α)·log∑_k e^{α·conv_k} ).
    The formula reduces to 2·corr(w, a) where w_k ∝ e^{α·conv_k}.
    """
    # soft‑max weights
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)      # stabilise
    w = exp_scaled / exp_scaled.sum()

    # correlation w ⋆ a   (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)   # length ≥ 3·n‑2
    n_len = a.size
    grad = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the smooth surrogate
def _adam_phase(a: np.ndarray, n: int, L: int,
                max_iters: int, deadline: float,
                alpha_start: float = 5.0,
                alpha_end: float = 20000.0,
                lr0: float = 0.6,
                beta1: float = 0.9,
                beta2: float = 0.999,
                eps: float = 1e-12,
                prune_every: int = 0,
                prune_percentile: float = 1.0) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Adam on the log‑sum‑exp surrogate.
    Returns (final a, best objective value seen, best a).
    """
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_obj = _objective(a, n, L)

    total_iters = max_iters
    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[adam] iter {it} new best O={best_obj:.6f}")

        # exponential schedule for α – sharper towards the end
        frac = it / total_iters
        alpha = alpha_start * (alpha_end / alpha_start) ** frac

        grad = _smooth_gradient(a, conv, alpha, L)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        lr = lr0 / math.sqrt(it + 1)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape plateaus
        if np.random.rand() < 0.001:
            a_new += np.random.randn(n) * 1e-4

        a_new = _project_to_simplex(a_new)

        # optional pruning of tiny entries
        if prune_every and it % prune_every == 0 and prune_percentile < 100.0:
            thresh = np.percentile(a_new, prune_percentile)
            a_new[a_new < thresh] = 0.0
            a_new = _project_to_simplex(a_new)

        a = a_new

    return a, best_obj, best_a


# ----------------------------------------------------------------------
# Polyak sub‑gradient descent (exact objective)
def _polyak_subgradient_phase(a: np.ndarray, n: int, L: int,
                              deadline: float,
                              lower_bound: float,
                              verbose: bool = False) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Projected Polyak sub‑gradient descent on the exact objective.
    Uses step = (f(x)-f*) / ||g||² (capped for stability).
    """
    best_a = a.copy()
    best_obj = _objective(a, n, L)
    it = 0

    while time.time() < deadline:
        it += 1

        conv = _conv_fft(a, L)
        cur_max = conv.max()
        cur_obj = 2.0 * n * cur_max

        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            if verbose and it % 5000 == 0:
                print(f"[polyak] iter {it} new best O={best_obj:.6f}")

        # stop if we are already at the theoretical lower bound
        if cur_obj <= lower_bound + 1e-12:
            break

        g = _subgradient_max_conv(a, conv)
        g_norm_sq = np.dot(g, g)
        if g_norm_sq == 0.0:
            break

        # Polyak step, capped to avoid too large jumps
        step = max(0.0, (cur_obj - lower_bound) / g_norm_sq)
        step = min(step, 0.5)      # safety cap

        a_new = a - step * g
        a_new = _project_to_simplex(a_new)

        # tiny random perturbation to help escape flat regions
        if np.random.rand() < 0.001:
            a_new += np.random.randn(n) * 1e-6
            a_new = _project_to_simplex(a_new)

        a = a_new

    return a, best_obj, best_a

# Simulated annealing refinement (same as in the reference code)
def _simulated_annealing_phase(a: np.ndarray, n: int, L: int,
                               max_iters: int, deadline: float,
                               temperature0: float = 0.05,
                               no_improve_limit: int = 8000,
                               verbose: bool = True) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Local refinement: move a modest fraction of mass from an index that
    contributes to the current maximal convolution entry to a random
    destination; accept with Metropolis rule.
    """
    best_a = a.copy()
    best_obj = _objective(a, n, L)
    cur_obj = best_obj
    conv = _conv_fft(a, L)

    T = temperature0
    no_improve = 0

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        kmax = int(np.argmax(conv))
        i_candidates = np.arange(max(0, kmax - (n - 1)), min(n - 1, kmax) + 1)
        i_candidates = i_candidates[a[i_candidates] > 1e-14]
        if i_candidates.size == 0:
            i = np.random.randint(n)
        else:
            i = np.random.choice(i_candidates)

        j = np.random.randint(n)
        while j == i:
            j = np.random.randint(n)

        delta = a[i] * np.random.rand() * 0.09
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = _project_to_simplex(a_new)

        conv_new = _conv_fft(a_new, L)
        new_obj = 2.0 * n * conv_new.max()

        # Metropolis acceptance
        if new_obj < cur_obj or np.random.rand() < math.exp((cur_obj - new_obj) / max(T, 1e-12)):
            a = a_new
            conv = conv_new
            cur_obj = new_obj
            no_improve = 0
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                if verbose:
                    print(f"[sa]  iter {it} new global best O={best_obj:.6f}")
        else:
            no_improve += 1

        # occasional exact sub‑gradient step to keep direction
        if it % 9 == 0:
            g = _subgradient_max_conv(a, conv)
            eta_loc = 0.07 * (1.0 - it / max_iters)
            a = _project_to_simplex(a - eta_loc * g)
            conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                if verbose:
                    print(f"[sa‑sub] iter {it} improved O={best_obj:.6f}")

        T *= 0.9997

        if no_improve >= no_improve_limit:
            a = best_a.copy()
            a += np.random.randn(n) * 0.0015
            a = np.clip(a, 0, None)
            a = _project_to_simplex(a)
            conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max()
            no_improve = 0

    return a, best_obj, best_a

# Main entry point expected by the evaluation harness
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence that minimises the evaluation function.
    Returns the sequence as a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # safety margin

    # ------------------------------------------------------------------
    # Initialise problem size
    if 'height_sequence_1' in globals():
        a0 = np.array(height_sequence_1, dtype=float)
        a0 = np.clip(a0, 0.0, None)
        if a0.sum() == 0.0:
            a0 = np.full_like(a0, 1.0)
        a0 = a0 / a0.sum()
        n = a0.size
        a = a0.copy()
        print(f"[init] using supplied sequence, n={n}")
    else:
        n = 4600
        a = np.full(n, 1.0 / n)
        print(f"[init] using uniform start, n={n}")

    # FFT length – next power of two greater than 3·n‑2
    L = 1 << ((3 * n - 2).bit_length())

    # keep track of the best solution globally
    best_seq = a.copy()
    best_score = _objective(best_seq, n, L)

    # ------------------------------------------------------------------
    # Phase 0 – cheap random restarts (≈10 % of the budget)
    phase0_deadline = start_time + budget_s * 0.10
    while time.time() < phase0_deadline:
        conc = np.random.uniform(0.2, 0.8)          # Dirichlet concentration
        a_rand = np.random.gamma(conc, 1.0, size=n)
        a_rand = a_rand / a_rand.sum()
        sub_dead = time.time() + 3.0                # a couple of seconds per trial
        _, _, best_rand = _adam_phase(a_rand, n, L,
                                      max_iters=2000,
                                      deadline=sub_dead,
                                      alpha_start=5.0,
                                      alpha_end=2000.0,
                                      lr0=0.8)
        cur_obj = _objective(best_rand, n, L)
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = best_rand.copy()
            print(f"[phase0] new best O={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – long Adam optimisation on the smooth surrogate
    phase1_deadline = start_time + budget_s * 0.55
    a, obj_adam, best_adam = _adam_phase(best_seq, n, L,
                                         max_iters=900000,
                                         deadline=phase1_deadline,
                                         alpha_start=5.0,
                                         alpha_end=80000.0,
                                         lr0=0.6)
    if obj_adam < best_score:
        best_score = obj_adam
        best_seq = best_adam.copy()
        print(f"[phase1] improved best O={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Polyak sub‑gradient descent (≈30 % of the budget)
    phase2_deadline = start_time + budget_s * 0.85
    lower_O_bound = 2.0 * n / (2 * n - 1)          # theoretical lower bound on O
    a, obj_sub, best_sub = _polyak_subgradient_phase(best_seq, n, L,
                                                     deadline=phase2_deadline,
                                                     lower_bound=lower_O_bound,
                                                     verbose=True)
    if obj_sub < best_score:
        best_score = obj_sub
        best_seq = best_sub.copy()
        print(f"[phase2] improved best O={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – stochastic annealing refinement for any remaining time
    if time.time() < deadline:
        a, obj_sa, best_sa = _simulated_annealing_phase(best_seq, n, L,
                                                        max_iters=500000,
                                                        deadline=deadline,
                                                        temperature0=0.08,
                                                        no_improve_limit=5000,
                                                        verbose=True)
        if obj_sa < best_score:
            best_score = obj_sa
            best_seq = best_sa.copy()
            print(f"[phase3] final best O={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final verification through the provided evaluator
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence failed:", exc)

    return best_seq.tolist()
