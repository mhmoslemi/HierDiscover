-----------------

def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        # all non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution (a*a) using real FFT, length 2·n‑1."""
    # real FFT is faster for real‑valued data
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len).real[: 2 * a.size - 1]
    return conv


def _conv_max_fft(a: np.ndarray, fft_len: int) -> tuple[float, int]:
    """Maximum coefficient of a*a and its index."""
    conv = _conv_full_fft(a, fft_len)
    idx = int(conv.argmax())
    return float(conv[idx]), idx


# ----------------------------------------------------------------------
# Parametric families (all return a normalised probability vector)
# ----------------------------------------------------------------------


def _uniform_seq(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _power_law_seq(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = i ** (-p)
    seq /= seq.sum()
    return seq


def _exponential_seq(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    seq = np.exp(-lam * i)
    seq /= seq.sum()
    return seq


def _power_law_exp_seq(n: int, p: float, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    seq = (i + 1.0) ** (-p) * np.exp(-lam * i)
    seq /= seq.sum()
    return seq


def _stretched_exponential_seq(n: int, lam: float, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    seq = np.exp(-lam * (i ** gamma))
    seq /= seq.sum()
    return seq


def _beta_seq(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(1, n + 1, dtype=np.float64) / n)
    seq = (x ** alpha) * ((1.0 - x) ** beta)
    s = seq.sum()
    if s == 0.0:
        return None
    seq /= s
    return seq


def _dirichlet_seq(n: int, conc: float) -> np.ndarray:
    g = np.random.gamma(conc, 1.0, size=n)
    seq = g / g.sum()
    return seq

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that
    minimises ``evaluate_sequence``.  The routine adapts its schedule to
    the available time budget and always returns a feasible sequence.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the final exact evaluation
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # Choose a working length.
    # --------------------------------------------------------------
    try:
        prev = globals()["height_sequence_1"]
        n = max(len(prev), 5000)
        prev_arr = np.asarray(prev, dtype=np.float64)
        if prev_arr.sum() > 0.0:
            prev_arr /= prev_arr.sum()
        else:
            prev_arr = None
    except Exception:
        prev_arr = None
        n = 5000
    n = min(max(n, 5000), 25000)               # 5 k ≤ n ≤ 25 k
    fft_len = 1 << ((2 * n - 1).bit_length())  # next power of two ≥ 2·n‑1

    # --------------------------------------------------------------
    # Cheap objective (assumes Σa_i = 1): 2·n·max(conv)
    # --------------------------------------------------------------
    def cheap_score(vec: np.ndarray) -> float:
        maxb, _ = _conv_max_fft(vec, fft_len)
        return 2.0 * n * maxb

    # --------------------------------------------------------------
    # Phase 0 – exhaustive search over parametric families
    # --------------------------------------------------------------
    best_a = _uniform_seq(n)
    best_val = cheap_score(best_a)

    # previous construction, if it matches current length
    if prev_arr is not None and prev_arr.size == n:
        val = cheap_score(prev_arr)
        if val < best_val:
            best_a, best_val = prev_arr.copy(), val

    # Power‑law sweep
    for p in np.linspace(0.2, 2.0, 91):
        if time.time() >= deadline:
            break
        cand = _power_law_seq(n, p)
        val = cheap_score(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # Exponential sweep (λ up to 5)
    for lam in np.linspace(0.0, 5.0, 101):
        if time.time() >= deadline:
            break
        cand = _exponential_seq(n, lam)
        val = cheap_score(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # Mixed power‑law × exponential (coarse grid)
    for p in np.linspace(0.2, 1.5, 14):
        for lam in np.linspace(0.0, 3.0, 16):
            if time.time() >= deadline:
                break
            cand = _power_law_exp_seq(n, p, lam)
            val = cheap_score(cand)
            if val < best_val:
                best_a, best_val = cand, val

    # Stretched exponential (γ ∈ {0.5,1,1.5,2})
    for gamma in [0.5, 0.9, 1.2, 1.5, 2.0]:
        for lam in np.linspace(0.0, 4.0, 41):
            if time.time() >= deadline:
                break
            cand = _stretched_exponential_seq(n, lam, gamma)
            val = cheap_score(cand)
            if val < best_val:
                best_a, best_val = cand, val

    # Beta‑shaped families
    x = (np.arange(1, n + 1, dtype=np.float64) / n)
    for α in np.linspace(0.2, 8.0, 25):
        for β in np.linspace(0.2, 8.0, 25):
            if time.time() >= deadline:
                break
            seq = (x ** α) * ((1.0 - x) ** β)
            s = seq.sum()
            if s == 0.0:
                continue
            cand = seq / s
            val = cheap_score(cand)
            if val < best_val:
                best_a, best_val = cand, val

    # Random Dirichlet seeds with a wide range of concentrations
    for conc in [0.001, 0.005, 0.01, 0.05, 0.1, 0.5, 1.0, 2.0, 5.0]:
        if time.time() >= deadline:
            break
        cand = _dirichlet_seq(n, conc)
        val = cheap_score(cand)
        if val < best_val:
            best_a, best_val = cand, val

    # Additional randomised families (random power‑law + exponential)
    for _ in range(200):
        if time.time() >= deadline:
            break
        p = np.random.uniform(0.1, 2.0)
        lam = np.random.exponential(scale=0.5)
        cand = _power_law_exp_seq(n, p, lam)
        val = cheap_score(cand)
        if val < best_val:
            best_a, best_val = cand, val

    print(f"[Init] best cheap score = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – mirror‑descent (exponentiated‑gradient) refinement
    # --------------------------------------------------------------
    a = best_a.copy()
    best_iter_a = a.copy()
    best_iter_val = best_val
    last_improve = 0
    iter_cnt = 0
    eta0 = 0.7               # initial step‑size factor (slightly larger)
    max_iters = 2_000_000    # generous upper bound
    eps = 1e-30

    while time.time() < deadline - 5.0 and iter_cnt < max_iters:
        iter_cnt += 1

        conv = _conv_full_fft(a, fft_len)
        maxb = conv.max()
        cheap = 2.0 * n * maxb

        # record improvement
        if cheap < best_iter_val - 1e-12:
            best_iter_val = cheap
            best_iter_a = a.copy()
            last_improve = iter_cnt
            print(f"[Iter {iter_cnt:07d}] NEW BEST cheap = {cheap:.6f}")

        # active set: indices whose value is within 1e‑12 relative to max
        # (this typically selects only the true maximisers)
        active = np.where(conv >= maxb * (1.0 - 1e-12))[0]

        # build sub‑gradient from active convolution entries
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            rng = np.arange(lo, hi + 1, dtype=np.int64)
            g[rng] += 2.0 * a[k - rng]
        # average over the active set to avoid too large steps
        g /= float(active.size)

        # exponentiated‑gradient (mirror descent) step
        eta = eta0 / math.sqrt(iter_cnt)
        log_a = np.log(a + eps)
        log_a -= eta * g
        a = np.exp(log_a)

        # re‑normalise onto the simplex
        s = a.sum()
        if s <= 0.0 or not np.isfinite(s):
            a = best_iter_a.copy()
            continue
        a /= s

        # occasional random restart if stuck for many iterations
        if iter_cnt - last_improve > 150_000:
            perturb = 0.02 * np.random.rand(n)
            a = _simplex_projection(best_iter_a + perturb)
            last_improve = iter_cnt
            print(f"[Iter {iter_cnt:07d}] Restart after stagnation.")

    # --------------------------------------------------------------
    # Phase 2 – simple (1+1)‑ES hill‑climbing on the cheap score
    # --------------------------------------------------------------
    sigma = 0.12           # mutation strength (log‑space)
    fail_cnt = 0
    while time.time() < deadline - 2.0:
        # multiplicative Gaussian perturbation in log‑space
        cand = best_iter_a * np.exp(sigma * np.random.randn(n))
        cand = _simplex_projection(cand)
        val = cheap_score(cand)
        if val < best_iter_val:
            best_iter_val = val
            best_iter_a = cand
            sigma = max(sigma * 0.85, 1e-4)   # shrink step on success
            fail_cnt = 0
            print(f"[ES] improved cheap = {val:.6f}  sigma={sigma:.5f}")
        else:
            fail_cnt += 1
            if fail_cnt > 200:
                sigma = min(sigma * 1.30, 0.6)   # enlarge step after many failures
                fail_cnt = 0

        # occasional restart from a fresh Dirichlet when plateaus
        if iter_cnt % 200_000 == 0:
            pert = _dirichlet_seq(n, np.random.uniform(0.001, 2.0))
            a = _simplex_projection(best_iter_a + 0.02 * pert)
            # ignore 'a' – the (1+1)ES only mutates best_iter_a

    # --------------------------------------------------------------
    # Phase 3 – final exact evaluation
    # --------------------------------------------------------------
    final_seq = best_iter_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:      # defensive – should never happen
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # --------------------------------------------------------------
    # Return the best sequence found.
    # --------------------------------------------------------------
    return final_seq
