#!/usr/bin/env python
"""
Non‑negative step‑function optimiser (random‑search + mirror‑descent).

The routine performs three phases inside the allotted time budget:

1. **Broad random exploration** of several low‑dimensional parametric families
   (symmetric/asymmetric product‑power‑law, exponential, and pure power‑law).
   Each candidate is normalised to a probability vector (sum = 1) and its
   cheap objective 2·n·max(a*a) is evaluated via an FFT‑based convolution.
   The best candidate seen so far is retained.

2. **Refinement** of the best candidate by Adam‑based mirror descent on the
   probability simplex.  The gradient is obtained from a smoothed max of the
   self‑convolution (soft‑max with a decaying temperature).  Back‑tracking
   line‑search guarantees that the cheap objective never increases.

3. **Final exact evaluation** with the supplied ``evaluate_sequence`` function.
   The returned list is the best distribution discovered (elements are non‑negative
   and safely far below the 1000‑limit).

The whole procedure respects the 1000‑second budget, prints progress
information and finally returns a list of floats ready for the challenge.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max; returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    s = e.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / z.size)
    return e / s


def _fft_len_for_corr(n: int) -> int:
    """Smallest power‑of‑two ≥ 3·n‑2 – enough for a*a via FFT."""
    return 1 << ((3 * n - 2).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a (FFT implementation)."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Returns (max_conv, grad_a) where

        grad_a = ∇_a [ 2·n·smooth_max(a*a) ]

    ``eps_factor`` scales the smoothing constant ε = eps_factor·max_conv.
    The gradient is computed via the correlation of ``a`` with the soft‑max
    weights on the convolution entries.
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    # FFT of a (zero‑padded)
    a_fft = np.fft.rfft(a, L)

    # Linear self‑convolution a*a
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # Smoothed soft‑max weights
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv                # ≤ 0 for stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # Correlation a ⋆ w
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # Gradient (already multiplied by 4·n)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a


# ----------------------------------------------------------------------
# Parametric families (all return a non‑negative ndarray, not normalised)
# ----------------------------------------------------------------------


def _sym_product_power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ ((i+1)*(n-i))^{‑p}, symmetric, unnormalised."""
    i = np.arange(n, dtype=np.float64)
    # log‑space to avoid overflow for large exponents
    log_a = -p * (np.log(i + 1.0) + np.log(n - i))
    a = np.exp(log_a)
    return a


def _asym_product_power_law_distribution(n: int, p1: float, p2: float) -> np.ndarray:
    """a_i ∝ (i+1)^{‑p1}·(n‑i)^{‑p2}, unnormalised."""
    i = np.arange(n, dtype=np.float64)
    log_a = -p1 * np.log(i + 1.0) - p2 * np.log(n - i)
    a = np.exp(log_a)
    return a


def _exp_decay_distribution(n: int, lam: float) -> np.ndarray:
    """a_i ∝ e^{‑lam·i}+e^{‑lam·(n‑1‑i)} (symmetric exponential)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    return a


def _asym_exp_distribution(n: int, lam: float) -> np.ndarray:
    """a_i ∝ e^{‑lam·i} (mass concentrated on the left side)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    return a


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{‑p}, unnormalised."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    return a
