"""
propose_candidate
-----------------

Searches for a non‑negative step‑function (a probability simplex vector)
that minimizes

    O(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm is split into three time‑controlled phases:

1️⃣  **Smooth surrogate minimisation (Adam)**  
    Minimises a log‑sum‑exp surrogate of the max‑convolution while keeping
    the vector on the simplex.  The smoothing parameter α is annealed from
    a small to a very large value and a tiny entropy term λ is gradually
    removed.  This quickly breaks symmetry and provides a good starting
    point.

2️⃣  **Directed local refinement** – mass‑transfer moves guided by the
    true sub‑gradient of the max‑convolution.  At each step the indices
    with the largest sub‑gradient (most responsible for the peak) donate
    a fraction of their probability mass to indices with the smallest
    sub‑gradient.  After the transfer we project back onto the simplex.
    Interleaved are cheap projected‑gradient steps on the exact sub‑gradient
    and occasional Gaussian smoothing to avoid tiny spikes.

3️⃣  **Final polishing (Adam with huge smoothing)**  
    Starts from the best sequence found so far and runs a few Adam steps
    on a very sharp surrogate (α≈5·10⁴, λ=0) to fine‑tune the remaining
    slack.

The routine respects a total wall‑clock budget (default 1000 s) and prints
intermediate scores.  It returns the best sequence (list of floats) found.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------
# Helper 1 – Euclidean projection onto the probability simplex
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project the vector v onto {x ≥ 0, sum(x) = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                      # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
# Helper 2 – full convolution via FFT (real → real)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return the full discrete convolution a*a (length 2·len(a)‑1)."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# --------------------------------------------------------------
# Helper 3 – exact objective for a simplex‑normalised vector
def _objective_simplex(a: np.ndarray, n: int, L: int) -> float:
    """O(a) with the constraint sum(a)=1."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# --------------------------------------------------------------
# Helper 4 – sub‑gradient of max(conv) (averaged over all maximisers)
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. a (average if several maximisers)."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)          # all active ks
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]               # factor 2 as per convolution
    if ks.size:
        g /= ks.size
    return g


# --------------------------------------------------------------
# Helper 5 – gradient of the smooth log‑sum‑exp surrogate + entropy term
def _smooth_gradient_plus_entropy(
    a: np.ndarray,
    conv: np.ndarray,
    alpha: float,
    L: int,
    lam: float,
) -> np.ndarray:
    """
    Gradient of  (2·n/α)·log Σ_k exp(α·conv_k)  +  λ·Σ_i a_i·log a_i.
    The first part equals 2·corr(w, a) where w_k ∝ exp(α·conv_k).
    """
    # ----- soft‑max weights -----
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)          # stable
    w = exp_scaled / exp_scaled.sum()                 # shape (2n‑1,)

    # ----- correlation w ⋆ a (vector of length n) -----
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n_len = a.size
    grad_corr = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]

    # ----- entropy gradient -----
    eps = 1e-20
    grad_ent = lam * (np.log(a + eps) + 1.0)

    return grad_corr + grad_ent
