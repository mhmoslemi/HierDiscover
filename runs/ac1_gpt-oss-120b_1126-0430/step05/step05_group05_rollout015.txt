    data = np.tile(f[:, None], (1, n))  # shape (n, n)

    return res.x

def _lp_max_sum(f, M):
    n = f.shape[0]
    offsets = np.arange(n)
    # Build data: each diagonal i has constant f[i] over n entries
    # Use scipy.sparse.dia_matrix for efficiency - we can reuse building each call
    data = np.tile(f[:, np.newaxis], (1, n))  # shape (n, n)
    A = scipy.sparse.dia_matrix((data, offsets), shape=(2 * n - 1, n)).tocsr()
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M, dtype=float)
    bounds = [(0, None)] * n
    # Solve with HiGHS
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                                 options={'presolve': True, 'time_limit': 30})
    if res.success:
        return res.x
    else:
        return None

"""
Propose a non‑negative step‑function sequence that (approximately) minimises

    O(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

Algorithm
---------
1️⃣  Annealed Adam on a smooth log‑sum‑exp surrogate (phase‑1).  
    Optimises a probability simplex (∑a=1) and gradually sharpens the
    surrogate (large α) while discarding the entropy regulariser.

2️⃣  Linear‑programming improvement (phase‑2).  
    For the current a we solve:
        maximise   ∑b_j
        subject to conv(a,b) ≤ M   (M = max(conv(a,a))) and b ≥ 0 .
    The solution b has larger ℓ₁‑norm, we normalise it to a probability
    vector g = b/∑b and mix a ← (1−t)·a + t·g (t≈0.2).  Repeating a few
    times quickly lowers the objective.

3️⃣  Intensive simulated‑annealing local search (phase‑3).  
    Random “mass‑transfer’’ moves are accepted according to a Metropolis
    rule, plus occasional sub‑gradient steps on the exact max‑convolution.
    Restarts with small jitter keep the search alive.

4️⃣  Final Adam polishing (phase‑4) with a very large smoothing
    parameter (α≈5·10⁵) and zero entropy.

The routine respects the 1000 s budget, prints progress, and finally
returns the best sequence as a plain ``list`` of ``float``.
"""

import time
import math
import numpy as np
import scipy.optimize
import scipy.sparse


# --------------------------------------------------------------
# Projection onto the probability simplex (Euclidean)
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho_arr[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
# Full self‑convolution via FFT (length 2·n‑1)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return a⨂a (full) using FFT of length L (next power of two)."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# --------------------------------------------------------------
# Sub‑gradient of max(conv) (averaged over all maximisers)
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. a (averaged over all maximising indices)."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


# --------------------------------------------------------------
# Gradient of the smooth surrogate (log‑sum‑exp) plus entropy term
def _smooth_gradient_plus_entropy(
    a: np.ndarray,
    conv: np.ndarray,
    alpha: float,
    L: int,
    lam: float,
) -> np.ndarray:
    """
    Returns ∇_a ( (2·n/α)·log∑_k e^{α·conv_k} + λ·∑ a_i·log a_i ).
    """
    # soft‑max weights
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)          # stability
    w = exp_scaled / exp_scaled.sum()                # shape (2n‑1,)

    # correlation w ⋆ a  (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n_len = a.size
    grad_corr = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]

    # entropy part
    eps = 1e-20
    grad_ent = lam * (np.log(a + eps) + 1.0)

    return grad_corr + grad_ent


# --------------------------------------------------------------
# Linear programme: maximise Σ b_j  subject to conv(a,b) ≤ M, b ≥ 0
def _lp_max_sum(a: np.ndarray, M: float) -> np.ndarray | None:
    """Return the optimal b (may be None on failure)."""
    n = a.size
    # Build Toeplitz constraint matrix A where (A @ b) = conv(a,b)
    offsets = np.arange(n)                     # 0 … n‑1
    # each diagonal i has the constant value a[i]
    data = np.broadcast_to(a[:, None], (n, n))  # shape (n,n) – view, no copy
    A = scipy.sparse.dia_matrix((data, offsets), shape=(2 * n - 1, n)).tocsr()

    c = -np.ones(n)                            # maximise Σb  ↔  minimise −Σb
    b_ub = np.full(2 * n - 1, M, dtype=float)
    bounds = [(0, None)] * n

    res = scipy.optimize.linprog(
        c,
        A_ub=A,
        b_ub=b_ub,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )
    if res.success:
        return res.x
    return None

            print(f"[LP] iter {lp_iter} improved O={best_obj:.6f}")
        else:
            print(f"[LP] iter {lp_iter} O={cur_obj:.6f}")

        t_mix = max(0.05, t_mix * 0.95)    # slowly shrink the step size
        lp_iter += 1

    # ------------------------------------------------------------------
    # Phase‑3 : Simulated‑annealing intensive local search
    print(f"[phase‑3] local refinement, best O={best_obj:.6f}")
    max_iter_local = 250_000
    temperature = 0.12
    temp_decay = 0.9999
    no_improve_limit = 3000
    subgrad_interval = 20

    a = best_seq.copy()
    conv = _conv_fft(a, L)
    cur_obj = 2.0 * n * conv.max()
    it_loc = 0
    no_improve = 0

    while time.time() < deadline and it_loc < max_iter_local:
        # ---- random mass‑transfer move ----
        i, j = np.random.randint(n, size=2)
        if i != j and a[i] > 1e-12:
            max_frac = 0.4 * (1.0 - it_loc / max_iter_local) + 0.01
            frac = np.random.rand() * max_frac
            delta = a[i] * frac
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            a_new = _project_to_simplex(a_new)

            conv_new = _conv_fft(a_new, L)
            new_obj = 2.0 * n * conv_new.max()

            # Metropolis acceptance
            if new_obj < cur_obj or np.random.rand() < math.exp((cur_obj - new_obj) / max(temperature, 1e-16)):
                a, conv, cur_obj = a_new, conv_new, new_obj
                no_improve = 0
                if cur_obj < best_obj:
                    best_obj = cur_obj
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] SA new global best O={best_obj:.6f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        # ---- occasional sub‑gradient descent step ----
        if it_loc % subgrad_interval == 0:
            g = _subgradient_max_conv(a, conv)
            eta = 0.12 * (1.0 - it_loc / max_iter_local)
            a = a - eta * g
            a = _project_to_simplex(a)
            conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_seq = a.copy()
                print(f"[{time.time():.1f}] SA sub‑grad improved O={best_obj:.6f}")

        # decay temperature
        temperature *= temp_decay
        it_loc += 1

        # restart if stagnating
        if no_improve >= no_improve_limit:
            a = best_seq.copy()
            a += np.random.randn(n) * 0.001
            a = np.clip(a, 0, None)
            a = _project_to_simplex(a)
            conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max()
            no_improve = 0
            print(f"[SA] restart after {no_improve_limit} non‑improving steps (iter {it_loc})")

    # ------------------------------------------------------------------
    # Phase‑4 : Final Adam polishing (very high α)
    print(f"[phase‑4] final polishing, best O={best_obj:.6f}")
    a = best_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    lr0 = 0.5
    alpha = 5e5
    lam = 0.0
    polish_iters = 30_000
    it = 0
    while time.time() < deadline and it < polish_iters:
        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_seq = a.copy()
            print(f"[polish] iter {it:,} new best O={best_obj:.6f}")

        grad = _smooth_gradient_plus_entropy(a, conv, alpha, L, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        lr = lr0 / math.sqrt(it + 1)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_to_simplex(a_new)
        it += 1

    # ------------------------------------------------------------------
    # Final sanity check with the provided evaluator
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:
        print("[final] evaluate_sequence failed:", e)

    return best_seq.tolist()
