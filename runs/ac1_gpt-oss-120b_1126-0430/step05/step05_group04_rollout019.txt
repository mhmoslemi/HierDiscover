"""
Propose a step‑function (non‑negative heights) with minimal evaluation score.

Algorithm Overview
-------------------
1.  **Multi‑scale Adam optimisation** – for several support sizes we run a
    fast Adam optimiser on the simplex.  Each run starts from a distinct
    deterministic init (uniform, linear decreasing, exponential decays) so
    that the optimiser explores many basins of attraction.  The learning‑rate
    and smooth‑max (β) schedules are a little more aggressive than the
    previous submission.

2.  **Greedy mass‑transfer refinement** – starting from the best sequence
    found by Adam we repeatedly locate the convolution index `k_max` where the
    self‑convolution attains its maximum.  The contribution of every coefficient
    to this peak is `c_i = a_i * a_{k_max-i}`.  We move a small amount of mass
    from a high‑contributing index to a low‑contributing index (chosen
    probabilistically) and keep the move only if the objective improves.
    The step size slowly decays, giving a deterministic local‑search that
    typically beats the original simulated‑annealing stage.

3.  **Final sanity** – tiny entries are pruned, the vector is re‑normalised,
    and the exact objective is reported via ``evaluate_sequence``.

The whole routine respects the global 1000 s budget (a small safety margin
is kept).  The returned list contains only non‑negative floats and is ready
for the evaluation harness.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# low‑level convolution helpers (FFT based, O(n log n))
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return conv_full[: 2 * n - 1]


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.size
    conv = _conv_self_fft(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (soft‑max parametrisation)
def _run_adam(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
) -> tuple[list[float], float]:
    """
    Fast Adam optimiser.

    Parameters
    ----------
    n : int
        Length of the sequence (after compression).
    time_limit : float
        Seconds allocated to this run.
    init_a : np.ndarray | None
        Optional initial vector (will be normalised to sum=1).
    seed : int
        Random seed for reproducibility.

    Returns
    -------
    seq, obj : (list[float], float)
        Best sequence found (as a plain Python list) and its exact objective.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # initialise a on the simplex
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
        a = a / a.sum()
    else:
        a = rng.random(n)
        a = a / a.sum()

    # log‑it representation (mirrored soft‑max)
    x = np.log(a + 1e-16)

    # Adam moments
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # schedules (more aggressive than the reference implementation)
    beta_start, beta_end = 0.5, 30000.0      # smoothing temperature
    lr_start, lr_end = 0.25, 0.001          # learning‑rate

    best_a = a.copy()
    best_obj = _objective(best_a)

    it = 0
    while time.time() - start < time_limit:
        it += 1

        # stay on the simplex
        a = np.exp(x - np.max(x))
        a = a / a.sum()

        # exact convolution (fast via FFT)
        conv = _conv_self_fft(a)
        M = conv.max()
        obj = float(2.0 * n * M)

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        # ---- annealing -------------------------------------------------
        t = min((time.time() - start) / time_limit, 1.0)
        beta = beta_start * np.exp(np.log(beta_end / beta_start) * t)
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t)

        # smooth‑max weights (soft‑max over the convolution values)
        if beta > 8000.0:
            # after enough annealing switch to the exact sub‑gradient
            kmax = int(np.argmax(conv))
            w = np.zeros_like(conv)
            w[kmax] = 1.0
        else:
            shifted = conv - M                      # ≤ 0
            w = np.exp(beta * shifted)
            s = w.sum()
            if s == 0.0:
                w[:] = 1.0 / conv.size
            else:
                w /= s

        # gradient of smooth‑max wrt a  (correlation with the weight vector)
        #   g_a[i] = 2 * Σ_j w[i+j] * a[j]    (implemented via FFT‑correlation)
        # Correlation a * w (same as convolution with reversed w)
        # Re‑use FFT utilities for speed.
        L = _next_pow2(conv.size + n - 1)
        wf = np.fft.rfft(w, n=L)
        af = np.fft.rfft(a, n=L)
        corr_full = np.fft.irfft(wf * np.conj(af), n=L)
        g_a = 2.0 * corr_full[:n]

        # natural gradient on the simplex (mirrored into log‑it space)
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)

        # Adam update (mirrored‑gradient version)
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape plateaus
        if it % 4000 == 0:
            x += rng.normal(scale=0.008, size=n)

        # restart from the best point now and then
        if it % 70000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.006, size=n)
            m.fill(0.0)
            v.fill(0.0)

    # ------------------------------------------------------------------
    # prune negligible entries (helps the exact objective)
    thresh = 1e-13
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = _objective(best_a)
        print(f"[compress] length→{best_a.size}, O={best_obj:.6f}")

    return best_a.tolist(), best_obj

# Greedy deterministic refinement (mass‑transfer moves)
def _greedy_refine(seq: list[float], time_limit: float, seed: int = 0) -> tuple[list[float], float]:
    """
    Targeted greedy local search that always moves mass away from the
    most offending convolution peak.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    a = np.asarray(seq, dtype=float)
    if a.sum() < 1e-12:
        a = np.ones_like(a) / a.size
    else:
        a = a / a.sum()
    n = a.size

    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())
    best_a = a.copy()

    # step‑size schedule – start relatively aggressive and shrink
    delta_factor = 0.07               # up to 7 % of the source entry each move
    decay = 0.995

    while time.time() - start < time_limit:
        # locate the worst convolution entry
        k = int(np.argmax(conv))

        # contributions a[i] * a[k-i] for all i that participate
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        i_vals = np.arange(i_min, i_max + 1)
        j_vals = k - i_vals
        contrib = a[i_vals] * a[j_vals]
        total = contrib.sum()
        if total > 0:
            probs = contrib / total
            i = rng.choice(i_vals, p=probs)
        else:
            i = rng.integers(n)

        # destination index – choose where its counterpart in the same
        # convolution (k‑j) is *small*.
        # Simple heuristic: pick a random index but bias towards low a[j]
        # *and* low a[k-j] (if inside the range).
        # We build a cheap weight vector.
        j_candidates = np.arange(n)
        # exclude the source index
        j_candidates = j_candidates[j_candidates != i]
        # compute a heuristic “danger” = a[j] * a[k-j] (zero if out of range)
        mask = (k - j_candidates >= 0) & (k - j_candidates < n)
        danger = np.empty_like(j_candidates, dtype=float)
        danger[:] = 0.0
        danger[mask] = a[j_candidates[mask]] * a[k - j_candidates[mask]]
        # invert danger to obtain desirability
        desirability = 1.0 / (danger + 1e-12)
        desirability = desirability / desirability.sum()
        j = rng.choice(j_candidates, p=desirability)

        # amount to transfer
        delta = a[i] * rng.random() * delta_factor
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.clip(a_new, 0.0, None)
        a_new /= a_new.sum()

        conv_new = _conv_self_fft(a_new)
        obj_new = float(2.0 * n * conv_new.max())

        if obj_new < best_obj - 1e-14:
            a, conv, best_obj, best_a = a_new, conv_new, obj_new, a_new.copy()
            print(f"[greedy] new best O={best_obj:.6f}")

        # decay the step size slowly
        if best_obj < best_obj * 0.9999:   # any improvement -> keep step size
            pass
        else:
            delta_factor *= decay
            delta_factor = max(delta_factor, 0.001)

    # final pruning of near‑zero entries (optional but cheap)
    thresh = 1e-13
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        comp = best_a[mask]
        comp /= comp.sum()
        comp_obj = float(2.0 * comp.size * _conv_self_fft(comp).max())
        if comp_obj < best_obj:
            best_a, best_obj = comp, comp_obj
            print(f"[greedy compress] length→{comp.size}, O={best_obj:.6f}")

    return best_a.tolist(), best_obj
