"""
Propose a step‑function sequence that minimises
    O(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The optimizer runs several phases inside the allotted 1000 s:

1️⃣  Random restarts → short Adam runs on a *variance* surrogate
    (aims to flatten the full autocorrelation, i.e. make all convolution
    entries close to the average 1/(2n‑1)).  The best seed is kept.

2️⃣  Long Adam optimisation on the variance surrogate.  This quickly drives
    the convolution towards a flat shape, which is known to lower the
    L∞‑norm.

3️⃣  Adam on a smooth max surrogate (log‑sum‑exp) – the classic “soft‑max”
    approximation of the maximum – to focus on the current peak.

4️⃣  Exact projected sub‑gradient descent with back‑tracking line‑search
    (optimises the true max‑objective).

5️⃣  Targeted simulated‑annealing that repeatedly transfers a small amount
    of mass from indices that participate in the current maximal
    convolution entry to indices that contribute least, accepting moves
    with a Metropolis rule.

All phases respect the wall‑time budget, constantly update the best
sequence found, and finally verify it with the provided ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto {x ≥ 0, Σ x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:  # fallback – uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution via FFT (real → real)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full convolution a*a, using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Objective for a simplex‑normalised vector (Σa = 1)
def _objective(a: np.ndarray, n: int, L: int) -> float:
    """Return the exact evaluation O = 2·n·max(conv(a,a))."""
    conv = _conv_fft(a, L)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Gradient of the log‑sum‑exp (smooth max) surrogate
def _smooth_gradient(a: np.ndarray, conv: np.ndarray,
                     alpha: float, L: int) -> np.ndarray:
    """
    ∇_a  ( (2·n/α)·log Σ_k e^{α·conv_k} ).
    Result = 2·corr(w, a) with w_k ∝ e^{α·conv_k}.
    """
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)          # stabilise
    w = exp_scaled / np.sum(exp_scaled)

    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n_len = a.size
    grad = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]
    return grad


# ----------------------------------------------------------------------
# Gradient of the *variance* surrogate (flattening the whole convolution)
def _variance_gradient(a: np.ndarray, diff: np.ndarray, L: int) -> np.ndarray:
    """
    diff = conv - avg   with avg = 1/(2n‑1).
    The objective Σ diff_k² has gradient 4·(diff ⋆ a)  (⋆ = cross‑correlation).
    """
    a_rev = a[::-1]
    fft_diff = np.fft.rfft(diff, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_diff * fft_a_rev, n=L)
    n_len = a.size
    grad = 4.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser (generic – gradient supplied)
def _adam(a: np.ndarray, n: int, L: int,
          max_iters: int, deadline: float,
          grad_func,               # grad_func(a, *extra) → np.ndarray
          extra_args=(),
          lr0: float = 0.8,
          beta1: float = 0.9,
          beta2: float = 0.999,
          eps: float = 1e-12,
          prune_every: int = 0,
          prune_percentile: float = 1.0,
          verbose: bool = False) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Generic Adam optimiser with projection onto the probability simplex.
    Returns (final a, best O value, best a seen).
    """
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_val = _objective(a, n, L)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # evaluate exact objective for bookkeeping
        cur_val = _objective(a, n, L)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            if verbose and it % 1000 == 0:
                print(f"[adam] iter {it:<6d} new best O={best_val:.6f}")

        # gradient of the supplied surrogate
        grad = grad_func(a, *extra_args)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        # simple learning‑rate schedule (≈ 1/√t)
        lr = lr0 / math.sqrt(it + 1)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional tiny random jitter to escape plateaus
        if np.random.rand() < 0.001:
            a_new += np.random.randn(n) * 1e-4

        a_new = _project_to_simplex(a_new)

        # optional pruning of near‑zero entries
        if prune_every and it % prune_every == 0 and it > 0:
            thresh = np.percentile(a_new, prune_percentile)
            a_new[a_new < thresh] = 0.0
            a_new = _project_to_simplex(a_new)

        a = a_new

    return a, best_val, best_a


# ----------------------------------------------------------------------
# Exact projected sub‑gradient descent on the true max‑objective
def _subgradient_phase(a: np.ndarray, n: int, L: int,
                       max_iters: int, deadline: float,
                       eta0: float = 0.3,
                       line_search: bool = True,
                       verbose: bool = False) -> tuple[np.ndarray, float, np.ndarray]:
    """Projected sub‑gradient descent with simple back‑tracking line search."""
    best_a = a.copy()
    best_val = _objective(a, n, L)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _conv_fft(a, L)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            if verbose and it % 5000 == 0:
                print(f"[sub‑grad] iter {it:<6d} new best O={best_val:.6f}")

        # sub‑gradient of the max‑convolution entry
        max_val = conv.max()
        ks = np.flatnonzero(conv == max_val)
        g = np.zeros_like(a)
        for k in ks:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_idx = np.arange(i_start, i_end + 1, dtype=int)
            j_idx = k - i_idx
            g[i_idx] += 2.0 * a[j_idx]
        if ks.size:
            g /= ks.size

        eta = eta0 * (0.9999 ** it)      # geometric decay

        if line_search:
            step = eta
            # at most five back‑tracking trials
            for _ in range(5):
                cand = _project_to_simplex(a - step * g)
                cand_val = _objective(cand, n, L)
                if cand_val < cur_val:
                    a = cand
                    break
                step *= 0.5
            else:
                a = _project_to_simplex(a - eta * g)
        else:
            a = _project_to_simplex(a - eta * g)

    return a, best_val, best_a

# Targeted simulated annealing refinement
def _simulated_annealing_phase(a: np.ndarray, n: int, L: int,
                               max_iters: int, deadline: float,
                               temperature0: float = 0.07,
                               no_improve_limit: int = 7000,
                               verbose: bool = True) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Local stochastic refinement:
    move a modest fraction of mass from an index involved in the current
    maximal convolution entry to a random destination; accept
    improvements unconditionally and non‑improvements with Metropolis rule.
    """
    best_a = a.copy()
    best_val = _objective(a, n, L)
    cur_val = best_val
    conv = _conv_fft(a, L)

    T = temperature0
    no_improve = 0

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        kmax = int(np.argmax(conv))

        # indices that actually contribute to the current peak
        i_candidates = np.arange(max(0, kmax - (n - 1)), min(n - 1, kmax) + 1)
        i_candidates = i_candidates[a[i_candidates] > 1e-14]
        if i_candidates.size == 0:
            i = np.random.randint(n)
        else:
            i = np.random.choice(i_candidates)

        # random destination different from i
        j = np.random.randint(n)
        while j == i:
            j = np.random.randint(n)

        # transfer a small fraction (up to 9 % of the source mass)
        delta = a[i] * np.random.rand() * 0.09
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = _project_to_simplex(a_new)

        conv_new = _conv_fft(a_new, L)
        new_val = 2.0 * n * conv_new.max()

        # Metropolis acceptance
        if new_val < cur_val or np.random.rand() < math.exp((cur_val - new_val) / max(T, 1e-12)):
            a = a_new
            conv = conv_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                if verbose:
                    print(f"[SA] iter {it:<6d} new global best O={best_val:.6f}")
        else:
            no_improve += 1

        # occasional exact sub‑gradient step to re‑align direction
        if it % 9 == 0:
            g = _subgradient_phase(a, n, L,
                                   max_iters=1,
                                   deadline=time.time() + 0.001,
                                   eta0=0.07,
                                   line_search=False)[0]  # we just need the gradient
            # The previous call returns a tuple (a', best_val, best_a) – we discard it
            # Instead compute sub‑gradient directly:
            conv_cur = _conv_fft(a, L)
            max_val = conv_cur.max()
            ks = np.flatnonzero(conv_cur == max_val)
            g = np.zeros_like(a)
            for k in ks:
                i_start = max(0, k - (n - 1))
                i_end = min(n - 1, k)
                i_idx = np.arange(i_start, i_end + 1, dtype=int)
                j_idx = k - i_idx
                g[i_idx] += 2.0 * a[j_idx]
            if ks.size:
                g /= ks.size
            eta_loc = 0.07 * (1.0 - it / max_iters)
            a = _project_to_simplex(a - eta_loc * g)
            conv = _conv_fft(a, L)
            cur_val = 2.0 * n * conv.max()
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                if verbose:
                    print(f"[SA‑sub] iter {it:<6d} improved O={best_val:.6f}")

        # temperature decay
        T *= 0.9997

        # restart if stagnating
        if no_improve >= no_improve_limit:
            a = best_a.copy()
            a += np.random.randn(n) * 0.0015
            a = np.clip(a, 0, None)
            a = _project_to_simplex(a)
            conv = _conv_fft(a, L)
            cur_val = 2.0 * n * conv.max()
            no_improve = 0

    return a, best_val, best_a

            a0 = np.full_like(a0, 1.0)
        a0 = a0 / a0.sum()
        n = a0.size
        a = a0.copy()
        print(f"[init] using supplied sequence, n={n}")
    else:
        # a reasonably large size (the previous best used 4587)
        n = 4600
        a = np.full(n, 1.0 / n)   # uniform start on the simplex
        print(f"[init] using uniform start, n={n}")

    # FFT length – next power of two ≥ 3·n‑2 (sufficient for all convolutions)
    L = 1 << ((3 * n - 2).bit_length())

    # ------------------------------------------------------------------
    # Phase 0 – cheap random restarts with a *variance* Adam (fast)
    phase0_deadline = start_time + budget_s * 0.15
    best_seq = a.copy()
    best_score = _objective(best_seq, n, L)

    while time.time() < phase0_deadline:
        # sample Dirichlet vectors with varied concentration
        conc = np.random.choice([0.2, 0.4, 0.8, 1.2])
        a_rand = np.random.gamma(conc, 1.0, size=n)
        a_rand = a_rand / a_rand.sum()
        sub_dead = time.time() + 3.0                # very short inner run
        a_rand, _, _ = _adam(
            a_rand, n, L,
            max_iters=800,
            deadline=sub_dead,
            grad_func=_variance_gradient,
            extra_args=(np.zeros(2 * n - 1),),      # placeholder for diff (will be recomputed internally)
            lr0=0.9,
            prune_every=0,
        )
        # after the short run recompute the variance gradient correctly
        conv = _conv_fft(a_rand, L)
        avg = 1.0 / (2 * n - 1)
        diff = conv - avg
        a_rand, _, best_rand = _adam(
            a_rand, n, L,
            max_iters=800,
            deadline=sub_dead,
            grad_func=_variance_gradient,
            extra_args=(diff,),
            lr0=0.9,
            prune_every=0,
        )
        cur_obj = _objective(best_rand, n, L)
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = best_rand.copy()
            print(f"[phase0] new best O={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – long Adam optimisation on the variance surrogate
    phase1_deadline = start_time + budget_s * 0.45
    a, obj_var, best_var = _adam(
        best_seq, n, L,
        max_iters=2_000_000,
        deadline=phase1_deadline,
        grad_func=_variance_gradient,
        extra_args=(np.zeros(2 * n - 1),),   # dummy; will be overwritten each loop
        lr0=0.7,
        prune_every=5000,
        prune_percentile=1.0,
        verbose=False,
    )
    # The variance gradient inside _adam expects the current diff, we therefore
    # recompute it each iteration inside a small wrapper.  To avoid rewriting
    # _adam we simply call it once more with the correct diff argument.
    # (This tiny inefficiency is negligible compared to the overall budget.)
    # Re‑run the variance Adam with proper diff handling:
    def _var_grad_wrapper(a_vec, dummy):
        conv = _conv_fft(a_vec, L)
        diff_now = conv - (1.0 / (2 * n - 1))
        return _variance_gradient(a_vec, diff_now, L)

    a, obj_var, best_var = _adam(
        best_seq, n, L,
        max_iters=800_000,
        deadline=phase1_deadline,
        grad_func=_var_grad_wrapper,
        extra_args=(),
        lr0=0.6,
        prune_every=5000,
        prune_percentile=1.0,
        verbose=False,
    )
    if obj_var < best_score:
        best_score = obj_var
        best_seq = best_var.copy()
        print(f"[phase‑1‑var] improved best O={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam on the smooth‑max surrogate (log‑sum‑exp)
    phase2_deadline = start_time + budget_s * 0.65
    a, obj_smooth, best_smooth = _adam(
        best_seq, n, L,
        max_iters=2_000_000,
        deadline=phase2_deadline,
        grad_func=_smooth_gradient,
        extra_args=(5.0, L),          # placeholder; alpha will be swept inside the loop
        lr0=0.6,
        prune_every=5000,
        prune_percentile=1.0,
        verbose=False,
    )
    # The _smooth_gradient needs the current alpha; we therefore embed a tiny
    # wrapper that updates it on the fly according to a linear schedule.
    def _smooth_grad_wrapper(a_vec, state):
        it, total, alpha_start, alpha_end = state
        frac = it / total
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        conv_local = _conv_fft(a_vec, L)
        return _smooth_gradient(a_vec, conv_local, alpha, L)

    total_iters = int((phase2_deadline - time.time()) * 200)  # generous estimate
    a, obj_smooth, best_smooth = _adam(
        best_seq, n, L,
        max_iters=1_200_000,
        deadline=phase2_deadline,
        grad_func=_smooth_grad_wrapper,
        extra_args=(0, total_iters, 5.0, 200_000.0),
        lr0=0.6,
        prune_every=5000,
        prune_percentile=1.0,
        verbose=False,
    )
    if obj_smooth < best_score:
        best_score = obj_smooth
        best_seq = best_smooth.copy()
        print(f"[phase‑2‑smooth] improved best O={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – exact projected sub‑gradient descent
    phase3_deadline = start_time + budget_s * 0.80
    a, obj_sub, best_sub = _subgradient_phase(
        best_seq, n, L,
        max_iters=500_000,
        deadline=phase3_deadline,
        eta0=0.25,
        line_search=True,
        verbose=False,
    )
    if obj_sub < best_score:
        best_score = obj_sub
        best_seq = best_sub.copy()
        print(f"[phase‑3‑sub] improved best O={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – targeted simulated annealing refinement
    phase4_deadline = deadline
    a, obj_sa, best_sa = _simulated_annealing_phase(
        best_seq, n, L,
        max_iters=3_000_000,
        deadline=phase4_deadline,
        temperature0=0.07,
        no_improve_limit=7000,
        verbose=True,
    )
    if obj_sa < best_score:
        best_score = obj_sa
        best_seq = best_sa.copy()
        print(f"[phase‑4‑SA] final best O={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final verification through the provided evaluator
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence failed:", exc)

    return best_seq.tolist()
