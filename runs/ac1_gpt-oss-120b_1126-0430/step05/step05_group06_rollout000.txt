"""
Proposes a non‑negative step‑function sequence that minimises

    O(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The search consists of five stages

  0️⃣  cheap random restarts → a few Adam steps on a Dirichlet start
  1️⃣  long Adam optimisation on a log‑sum‑exp (smooth‑max) surrogate
  2️⃣  exact projected sub‑gradient descent on the true objective
  3️⃣  linear‑programming refinement:
        given the current a, solve   max   Σ b_i
                                 s.t.   b_i ≥ 0,
                                       (a*b)_k ≤ max(conv(a,a))   ∀k .
        The LP is tiny because a is already sparse; the solution is
        normalised to a probability vector and (optionally) convex‑mixed
        with the current a.
  4️⃣  targeted simulated‑annealing with mass‑transfer moves.

All phases respect a total wall‑time of 1000 s, continuously keep the
best sequence seen and finally verify it with the provided
`evaluate_sequence` function.
"""

import time
import math
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Full linear (non‑circular) convolution via FFT
def conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return the full linear convolution a*a (length 2·len(a)-1)."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]

# ----------------------------------------------------------------------
# Exact objective for a simplex‑normalised vector (Σa=1)
def objective(a: np.ndarray, n: int, L: int) -> float:
    """2·n·max(conv(a,a))  (a is assumed to sum to 1)."""
    conv = conv_fft(a, L)
    return 2.0 * n * conv.max()

# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) w.r.t. a
def subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) at a (using all active indices)."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g

# ----------------------------------------------------------------------
# Gradient of the smooth surrogate (log‑sum‑exp)
def smooth_gradient(a: np.ndarray, conv: np.ndarray,
                    alpha: float, L: int) -> np.ndarray:
    """∇_a ( (2·n/α)·log∑_k e^{α·conv_k} )."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)        # stabilise
    w = exp_scaled / exp_scaled.sum()               # soft‑max weights

    # correlation w ⋆ a  (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n_len = a.size
    grad = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]
    return grad

# ----------------------------------------------------------------------
# Adam optimiser on the smooth surrogate
def adam_phase(a: np.ndarray, n: int, L: int,
               max_iters: int, deadline: float,
               alpha_start: float = 5.0,
               alpha_end: float = 20000.0,
               lr0: float = 0.6,
               beta1: float = 0.9,
               beta2: float = 0.999,
               eps: float = 1e-12,
               prune_every: int = 5000,
               prune_percentile: float = 1.0) -> tuple[np.ndarray, float, np.ndarray]:
    """Adam on the log‑sum‑exp surrogate."""
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_obj = objective(a, n, L)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[adam] iter {it} new best O={best_obj:.6f}")

        # exponential schedule for α
        frac = it / max_iters
        alpha = alpha_start * (alpha_end / alpha_start) ** frac

        grad = smooth_gradient(a, conv, alpha, L)

        # Adam updates (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        lr = lr0 / math.sqrt(it + 1)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter
        if np.random.rand() < 0.001:
            a_new += np.random.randn(n) * 1e-4

        a_new = project_to_simplex(a_new)

        # prune tiny entries occasionally
        if prune_every and it % prune_every == 0 and it > 0:
            thresh = np.percentile(a_new, prune_percentile)
            a_new[a_new < thresh] = 0.0
            a_new = project_to_simplex(a_new)

        a = a_new

    return a, best_obj, best_a

# Exact projected sub‑gradient descent (with optional back‑tracking line‑search)
def subgradient_phase(a: np.ndarray, n: int, L: int,
                      max_iters: int, deadline: float,
                      eta0: float = 0.3,
                      line_search: bool = True,
                      verbose: bool = False) -> tuple[np.ndarray, float, np.ndarray]:
    """Projected sub‑gradient descent on the true objective."""
    best_a = a.copy()
    best_obj = objective(a, n, L)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            if verbose and it % 5000 == 0:
                print(f"[sub‑grad] iter {it} new best O={best_obj:.6f}")

        g = subgradient_max_conv(a, conv)

        # geometric decay for step size
        eta = eta0 * (0.9999 ** it)

        if line_search:
            step = eta
            for _ in range(5):
                cand = project_to_simplex(a - step * g)
                cand_obj = objective(cand, n, L)
                if cand_obj < cur_obj:
                    a = cand
                    break
                step *= 0.5
            else:
                a = project_to_simplex(a - eta * g)
        else:
            a = project_to_simplex(a - eta * g)

    return a, best_obj, best_a

# Linear‑programming refinement
def lp_improvement(a: np.ndarray, n: int, L: int) -> np.ndarray:
    """
    Solve  max Σ b_i   s.t.   b_i≥0,
            (a * b)_k ≤ M   ∀k,  where M = max(conv(a,a)).
    Returns a (possibly) improved probability vector (sum=1).
    """
    eps = 1e-12
    M = conv_fft(a, L).max()
    idx = np.where(a > eps)[0]          # keep only strictly positive positions
    if idx.size == 0:
        return a

    # map original index -> column in reduced LP
    col_of = {j: p for p, j in enumerate(idx)}
    n_active = idx.size
    rows = []
    cols = []
    data = []

    # Build constraint matrix: for each active pair (i,j) add a_i to row (i+j), column j
    for i in idx:
        a_i = a[i]
        for j in idx:
            k = i + j
            rows.append(k)
            cols.append(col_of[j])
            data.append(a_i)

    A = sp.coo_matrix((data, (rows, cols)), shape=(2 * n - 1, n_active), dtype=np.float64).tocsr()
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # maximise Σ b_i  → minimise -Σ b_i
    c = -np.ones(n_active, dtype=np.float64)
    bounds = [(0, None)] * n_active

    # Solve LP
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 30.0})
    if not res.success:
        # LP failed – fall back to current a
        return a

    b = np.zeros(n, dtype=np.float64)
    b[idx] = res.x
    total = b.sum()
    if total <= 0:
        return a

    g = b / total                     # normalise to the simplex

    # Evaluate candidates: pure g, and a few convex blends
    cand_best = a
    best_val = objective(a, n, L)

    for t in (0.0, 0.25, 0.5, 0.75, 1.0):
        h = (1 - t) * a + t * g
        h = project_to_simplex(h)
        val = objective(h, n, L)
        if val < best_val - 1e-9:
            best_val = val
            cand_best = h

    return cand_best

# Simulated annealing with mass‑transfer moves
def simulated_annealing_phase(a: np.ndarray, n: int, L: int,
                              max_iters: int, deadline: float,
                              temperature0: float = 0.07,
                              no_improve_limit: int = 7000,
                              verbose: bool = True) -> tuple[np.ndarray, float, np.ndarray]:
    """Local stochastic refinement."""
    best_a = a.copy()
    best_obj = objective(a, n, L)
    cur_obj = best_obj
    conv = conv_fft(a, L)

    T = temperature0
    no_improve = 0

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        kmax = int(np.argmax(conv))
        i_candidates = np.arange(max(0, kmax - (n - 1)), min(n - 1, kmax) + 1)
        i_candidates = i_candidates[a[i_candidates] > 1e-14]
        if i_candidates.size == 0:
            i = np.random.randint(n)
        else:
            i = np.random.choice(i_candidates)

        j = np.random.randint(n)
        while j == i:
            j = np.random.randint(n)

        delta = a[i] * np.random.rand() * 0.09
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = project_to_simplex(a_new)

        conv_new = conv_fft(a_new, L)
        new_obj = 2.0 * n * conv_new.max()

        # Metropolis rule
        if new_obj < cur_obj or np.random.rand() < math.exp((cur_obj - new_obj) / max(T, 1e-12)):
            a = a_new
            conv = conv_new
            cur_obj = new_obj
            no_improve = 0
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                if verbose:
                    print(f"[{time.time():.1f}] new global best O={best_obj:.6f} (iter={it})")
        else:
            no_improve += 1

        # occasional exact sub‑gradient kick
        if it % 9 == 0:
            g = subgradient_max_conv(a, conv)
            eta_loc = 0.07 * (1.0 - it / max_iters)
            a = project_to_simplex(a - eta_loc * g)
            conv = conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                if verbose:
                    print(f"[{time.time():.1f}] sub‑grad improved O={best_obj:.6f}")

        T *= 0.9997

        if no_improve >= no_improve_limit:
            # restart from current best with tiny jitter
            a = best_a.copy()
            a += np.random.randn(n) * 0.0015
            a = np.clip(a, 0, None)
            a = project_to_simplex(a)
            conv = conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max()
            no_improve = 0

    return a, best_obj, best_a

        best_score = obj_adam
        best_seq = best_adam.copy()
        print(f"[phase‑1] improved best O={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – exact projected sub‑gradient descent (20 % of budget)
    phase2_deadline = start_time + budget_s * 0.60
    a, obj_sub, best_sub = subgradient_phase(a, n, L,
                                             max_iters=250_000,
                                             deadline=phase2_deadline,
                                             eta0=0.25,
                                             line_search=True,
                                             verbose=False)
    if obj_sub < best_score:
        best_score = obj_sub
        best_seq = best_sub.copy()
        print(f"[phase‑2] improved best O={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – linear‑programming refinement (5 % of budget)
    # We call the LP step a few times, each followed by a short sub‑gradient spur.
    phase3_deadline = start_time + budget_s * 0.70
    while time.time() < phase3_deadline:
        a_lp = lp_improvement(best_seq, n, L)
        obj_lp = objective(a_lp, n, L)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"[lp] improved best O={best_score:.6f}")

        # after each LP we run a tiny sub‑gradient burst to polish
        sub_dead = min(time.time() + 2.0, phase3_deadline)
        a_lp, _, _ = subgradient_phase(a_lp, n, L,
                                       max_iters=5000,
                                       deadline=sub_dead,
                                       eta0=0.15,
                                       line_search=False,
                                       verbose=False)
        # update best if needed
        cur_obj = objective(a_lp, n, L)
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = a_lp.copy()
            print(f"[lp‑post] improved best O={best_score:.6f}")
        # break early if we run out of time
        if time.time() >= phase3_deadline:
            break

    # ------------------------------------------------------------------
    # Phase 4 – stochastic annealing refinement (remaining time)
    phase4_deadline = deadline
    a, obj_sa, best_sa = simulated_annealing_phase(best_seq, n, L,
                                                  max_iters=300_000,
                                                  deadline=phase4_deadline,
                                                  temperature0=0.08,
                                                  no_improve_limit=8000,
                                                  verbose=True)
    if obj_sa < best_score:
        best_score = obj_sa
        best_seq = best_sa.copy()
        print(f"[phase‑4] final best O={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final verification through the provided evaluator
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence failed:", exc)

    # Return as plain Python list
    return best_seq.tolist()
