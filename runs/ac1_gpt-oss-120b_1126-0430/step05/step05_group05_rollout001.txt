"""
Propose a non‑negative step‑function (list of heights) that (approximately)
minimises

    O(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm consists of three stages:

1️⃣  **Exponentiated‑gradient descent on the exact objective**  
    We work on the probability simplex (∑a = 1) and use the sub‑gradient of
    max(conv) to perform a mirror‑descent (multiplicative) update
    a_i ← a_i·exp(−η·g_i) with a decreasing step‑size η.
    After each update we renormalise a to lie on the simplex.
    Periodic Gaussian jitter and a light smoothing kernel are applied
    to avoid stagnation.

2️⃣  **Restart‑driven exploration** – if no improvement is seen for a while
    the current iterate is re‑initialised as a noisy copy of the best
    solution found so far.

3️⃣  **Final polishing with Adam on a smooth log‑sum‑exp surrogate**  
    A short high‑precision Adam run with a very large smoothing parameter
    fine‑tunes the remaining slack.

All heavy work (convolution, sub‑gradient) is performed with FFTs.
The routine respects the 1000 s budget, prints progress and finally
returns the best sequence (as a plain ``list`` of ``float``) found.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------
# Helper: convolution via FFT (full length 2·n‑1)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full discrete convolution a⨂a using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# --------------------------------------------------------------
# Sub‑gradient of max(conv) (used in stage 1)
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Average sub‑gradient of max(conv) w.r.t. a (for a ≥ 0, sum(a)=1)."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)   # indices where the max is attained
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


# --------------------------------------------------------------
# Smooth surrogate gradient (log‑sum‑exp + entropy) – used in polishing
def _smooth_gradient_plus_entropy(
    a: np.ndarray,
    conv: np.ndarray,
    alpha: float,
    L: int,
    lam: float,
) -> np.ndarray:
    """∇_a ((2·n/α)·log∑ e^{α·conv} + λ·∑ a·log a)."""
    # soft‑max weights over convolution entries
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / exp_scaled.sum()                     # shape (2n‑1,)

    # correlation w ⋆ a (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n_len = a.size
    grad_corr = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]

    # entropy part λ·(log a + 1)
    eps = 1e-20
    grad_ent = lam * (np.log(a + eps) + 1.0)

    return grad_corr + grad_ent


# --------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the given
    evaluation function.  Returns the best sequence found (list of float).
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin so we return before the hard timeout
    deadline = start_time + budget_s - 0.3

    # --------------------------------------------------------------
    # Problem size ---------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.array(height_sequence_1, dtype=float)
        n = base.size
        a = np.maximum(base, 0.0)
        tot = a.sum()
        if tot < 1e-12:
            a = np.full(n, 1.0 / n)
        else:
            a = a / tot                      # normalise to a probability vector
        print(f"[init] start from supplied sequence, n={n}")
    else:
        # start from a uniform distribution; a slightly larger n than before
        n = 6000
        a = np.full(n, 1.0 / n)
        print(f"[init] start from uniform, n={n}")

    # FFT length (next power of two ≥ 2·n‑1)
    L = 1
    while L < 2 * n - 1:
        L <<= 1

    # --------------------------------------------------------------
    # Stage 1 – exponentiated‑gradient descent on the exact objective --
    best_seq = a.copy()
    conv = _conv_fft(a, L)
    best_score = 2.0 * n * conv.max()
    print(f"[stage‑1] start O={best_score:.6f}")

    # hyper‑parameters for the EG loop
    eta_start = 0.45          # initial step size
    eta_end = 5e-6            # final step size
    max_iters_stage1 = 800_000
    jitter_interval = 4000
    jitter_scale = 4e-4
    smooth_interval = 5000
    kernel = np.array([0.125, 0.25, 0.25, 0.25, 0.125])
    no_improve = 0
    no_improve_limit = 8000
    it = 0

    while time.time() < deadline and it < max_iters_stage1:
        # current objective
        conv = _conv_fft(a, L)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            no_improve = 0
            print(f"[stage‑1] iter {it:,} new best O={best_score:.6f}")
        else:
            no_improve += 1

        # sub‑gradient of max(conv)
        g = _subgradient_max_conv(a, conv)

        # step‑size schedule (linear decay)
        frac = it / max_iters_stage1
        eta = eta_start * (1.0 - frac) + eta_end * frac

        # mirror‑descent (exponentiated) update on the simplex
        log_a = np.log(a + 1e-20) - eta * g
        a = np.exp(log_a)
        a /= a.sum()

        # occasional jitter to escape flat regions
        if (it + 1) % jitter_interval == 0:
            a += np.random.randn(n) * jitter_scale
            a = np.clip(a, 0, None)
            a /= a.sum()

        # light smoothing every few thousand iterations
        if (it + 1) % smooth_interval == 0:
            a_smooth = np.convolve(a, kernel, mode="same")
            a = 0.85 * a + 0.15 * a_smooth
            a /= a.sum()

        # restart if stagnating
        if no_improve >= no_improve_limit:
            # re‑initialise as a noisy copy of the best solution so far
            a = best_seq.copy()
            a += np.random.randn(n) * 0.002
            a = np.clip(a, 0, None)
            a /= a.sum()
            no_improve = 0
            print(f"[stage‑1] restart at iter {it:,}")

        it += 1

    # --------------------------------------------------------------
    # Stage 2 – intensive simulated‑annealing local refinement ----------
    print(f"[stage‑2] entering SA refinement, best O={best_score:.6f}")
    # allocate the remaining time budget to this stage
    temperature0 = 0.12
    temp_decay = 0.9998
    max_iter_local = 300_000
    no_improve_limit_loc = 2500
    jitter_prob = 0.004
    subgrad_interval = 1
    it_start_stage2 = time.time()

    while time.time() < deadline:
        # initialise a fresh start – either a jittered best copy or a fresh random point
        if np.random.rand() < 0.55:
            a = best_seq.copy()
            a += np.random.randn(n) * 0.0015
            a = np.clip(a, 0, None)
            a /= a.sum()
        else:
            a = np.random.rand(n)
            a /= a.sum()

        conv = _conv_fft(a, L)
        cur_score = 2.0 * n * conv.max()
        it_loc = 0
        no_improve_loc = 0
        T = temperature0

        while time.time() < deadline and it_loc < max_iter_local:
            # ---------- random mass‑transfer move ----------
            i, j = np.random.randint(n, size=2)
            if i != j and a[i] > 1e-14:
                # fraction of source mass to move – annealed
                max_frac = 0.45 * (1.0 - it_loc / max_iter_local) + 0.03
                frac = np.random.rand() * max_frac
                delta = a[i] * frac
                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta
                a_new /= a_new.sum()

                conv_new = _conv_fft(a_new, L)
                new_score = 2.0 * n * conv_new.max()

                # Metropolis criterion
                if new_score < cur_score or np.random.rand() < math.exp((cur_score - new_score) / max(T, 1e-16)):
                    a, conv, cur_score = a_new, conv_new, new_score
                    no_improve_loc = 0
                    if cur_score < best_score:
                        best_score = cur_score
                        best_seq = a.copy()
                        print(f"[{time.time():.1f}] SA new global best O={best_score:.6f}")
                else:
                    no_improve_loc += 1
            else:
                no_improve_loc += 1

            # ---------- occasional Gaussian jitter ----------
            if np.random.rand() < jitter_prob:
                a += np.random.randn(n) * 1e-4
                a = np.clip(a, 0, None)
                a /= a.sum()
                conv = _conv_fft(a, L)
                cur_score = 2.0 * n * conv.max()

            # ---------- sub‑gradient descent step ----------
            if it_loc % subgrad_interval == 0:
                g = _subgradient_max_conv(a, conv)
                eta = 0.12 * (1.0 - it_loc / max_iter_local) + 1e-5
                log_a = np.log(a + 1e-20) - eta * g
                a = np.exp(log_a)
                a /= a.sum()
                conv = _conv_fft(a, L)
                cur_score = 2.0 * n * conv.max()
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] SA sub‑grad improved O={best_score:.6f}")

            # ---------- light smoothing ----------
            if (it_loc + 1) % 6000 == 0:
                a_smooth = np.convolve(a, kernel, mode="same")
                a = 0.86 * a + 0.14 * a_smooth
                a /= a.sum()
                conv = _conv_fft(a, L)
                cur_score = 2.0 * n * conv.max()

            # temperature decay
            T *= temp_decay
            it_loc += 1

            if no_improve_loc >= no_improve_limit_loc:
                break

    # --------------------------------------------------------------
    # Stage 3 – final Adam polishing (high α, zero entropy) -----------
    print(f"[stage‑3] final Adam polishing, best O={best_score:.6f}")
    a = best_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    lr0 = 0.55
    alpha = 120_000.0            # very large smoothing -> approximates the true max
    lam = 0.0
    polish_iters = 30_000
    it = 0

    while time.time() < deadline and it < polish_iters:
        conv = _conv_fft(a, L)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[stage‑3] iter {it:,} new best O={best_score:.6f}")

        grad = _smooth_gradient_plus_entropy(a, conv, alpha, L, lam)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        lr = lr0 / math.sqrt(it + 1)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # project back onto the simplex (simple Euclidean projection)
        # --- simplex projection -------------------------------------------------
        u = np.sort(a_new)[::-1]
        cssv = np.cumsum(u) - 1.0
        rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
        if rho.size == 0:
            a = np.full_like(a_new, 1.0 / n)
        else:
            rho = rho[-1]
            theta = cssv[rho] / (rho + 1)
            a = np.maximum(a_new - theta, 0.0)
        # -----------------------------------------------------------------------

        it += 1

    # --------------------------------------------------------------
    # Final sanity check with the provided evaluator
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:
        print("[final] evaluate_sequence failed:", e)

    return best_seq.tolist()
```

This implementation builds on the previous approach but replaces the initial Adam‑log‑sum‑exp phase with a dedicated exponentiated‑gradient (mirror‑descent) optimisation of the exact objective, adds more aggressive jitter / smoothing, and uses a restart‑driven simulated‑annealing refinement before a short high‑precision Adam polishing. All steps respect the 1000 s budget and return the best sequence discovered.
