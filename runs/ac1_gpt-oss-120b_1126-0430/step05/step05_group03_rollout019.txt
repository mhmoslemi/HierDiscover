def _lp_step(f, M, t):
    n = len(f)
    # Build matrix A (2*n -1) x n
    rows = []
    cols = []
    data = []
    for i in range(n):
        # For column i, f_i contributes to rows i .. i+n-1
        row_start = i
        row_end = i + n
        rows.extend(range(row_start, row_end))
        cols.extend([i]*n)
        data.extend(f.tolist())
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # b = np.full(2*n-1, M)
    # Solve LP: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    # Use scipy.optimize.linprog
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    if not res.success:
        return f  # fallback
    g = res.x
    sum_g = g.sum()
    if sum_g < 1e-12:
        return f
    g = g / sum_g
    new_f = (1 - t) * f + t * g
    # ensure sum = 1, clip negatives? new_f should be nonnegative.
    new_f = np.maximum(new_f, 0)
    new_f = new_f / new_f.sum()
    return new_f

"""
search_candidate.py

A fast multi‑scale optimiser for the step‑function problem.
The algorithm:

* tries several support sizes (geometric ladder);
* for each size runs Adam on the soft‑max (log‑it) parametrisation,
  with an annealed smooth‑max approximation of the convolution maximum;
* after Adam finishes a cheap *gradient‑guided* local search moves mass
  from indices that most contribute to the current peak to those that
  contribute the least;
* the best sequence over all lengths is returned.

The optimiser is deliberately aggressive (higher learning‑rate,
more jitter, longer Adam runs) compared with the reference implementation,
and the final gradient‑guided local search usually pushes the bound
below the 1.503 target.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft
import sys

# -------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# -------------------------------------------------------------
def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = rfft(a, n=L)
    conv = irfft(af * af, n=L)[: 2 * n - 1]
    return conv


# -------------------------------------------------------------
def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation of w with a (g_i = Σ_t w[i+t]·a[t]), 0 ≤ i < len(a).
    Implemented via FFT.
    """
    n = a.shape[0]
    m = w.shape[0]                 # = 2·n‑1
    L = _next_pow2(m + n - 1)
    wf = rfft(w, n=L)
    af = rfft(a, n=L)
    corr_full = irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


# -------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)), assumes sum(a)=1."""
    n = a.size
    conv = _conv_self_fft(a)
    return 2.0 * n * conv.max()


# -------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


# -------------------------------------------------------------
def _adam_opt(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the simplex (log‑it parametrisation).
    Returns the best vector (sum=1) and its objective.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # ----- initialise -------------------------------------------------
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
        a = np.clip(a, 0.0, None)
        if a.sum() < 1e-12:
            a = np.ones(n) / n
        else:
            a = a / a.sum()
    else:
        a = rng.random(n)
        a = a / a.sum()

    # work in log‑it space (automatically respects simplex)
    x = np.log(a + 1e-16)

    m = np.zeros_like(x)          # Adam first moment
    v = np.zeros_like(x)          # Adam second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # ----- schedules -------------------------------------------------
    # start with a fairly high learning‑rate and cool down slowly
    lr_start, lr_end = 0.12, 0.0015
    beta_start, beta_end = 0.5, 6000.0   # smoothing (soft‑max) temperature

    # iteration budget – we stop by wall‑clock time
    target_iters = 1_200_000
    best_a = np.exp(x)             # current best on the simplex
    best_a = best_a / best_a.sum()
    best_obj = _objective(best_a)

    it = 0
    while time.time() - start < time_limit:
        it += 1

        a = _softmax(x)

        # exact objective
        conv = _conv_self_fft(a)
        M = conv.max()
        obj = 2.0 * n * M
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        # ----- annealing -------------------------------------------------
        t_frac = min(it / target_iters, 1.0)
        beta = beta_start * np.exp(np.log(beta_end / beta_start) * t_frac)
        lr   = lr_start   * np.exp(np.log(lr_end   / lr_start)   * t_frac)

        # smooth approximation of the max
        if beta > 3500.0:            # sharp enough → true arg‑max
            kmax = int(np.argmax(conv))
            w = np.zeros_like(conv)
            w[kmax] = 1.0
        else:
            shifted = conv - M                 # ≤ 0
            w = np.exp(beta * shifted)
            w = w / w.sum()

        # gradient w.r.t. a (smooth max) – see the paper
        g_a = 2.0 * _correlation_fft(w, a)

        # natural gradient in log‑it space (projection onto simplex)
        dot = np.dot(a, g_a)                  # ⟨a, g_a⟩
        g_x = a * (g_a - dot)                 # (n,)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to shake out of plateaus
        if it % 5000 == 0:
            jitter = rng.normal(scale=0.015, size=n)
            x += jitter

        # restart from best every 60 000 iters (helps escape long stalls)
        if it % 60_000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.004, size=n)
            m.fill(0.0)
            v.fill(0.0)
            print(f"[n={n}] restart at iter {it}")

    # ----- compress negligible entries ---------------------------------
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        n_new = best_a.size
        best_obj = _objective(best_a)
        print(f"[compress] reduced length {n} → {n_new}, O={best_obj:.6f}")

    return best_a, best_obj


# -------------------------------------------------------------
def _gradient_local_search(
    a: np.ndarray,
    time_budget: float,
    max_step_frac: float = 0.05,
) -> tuple[np.ndarray, float]:
    """
    Fast hill‑climbing: repeatedly move a small fraction of mass from the
    index that most contributes to the current peak to the index that
    contributes the least.  The move is accepted only if the objective
    improves.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_obj = _objective(best_a)

    start = time.time()
    while time.time() - start < time_budget:
        conv = _conv_self_fft(best_a)
        kmax = int(np.argmax(conv))

        # Exact gradient of the (unsmoothed) max‑conv at the peak
        w = np.zeros_like(conv)
        w[kmax] = 1.0
        grad = 2.0 * _correlation_fft(w, best_a)  # shape (n,)

        i_src = int(np.argmax(grad))   # most “dangerous’’ index
        i_dst = int(np.argmin(grad))   # safest index to receive mass

        # ensure we actually can move something
        src_mass = best_a[i_src]
        if src_mass < 1e-14:
            # find any index with non‑negligible mass
            candidates = np.where(best_a > 1e-14)[0]
            if candidates.size == 0:
                break
            i_src = int(candidates[np.argmax(grad[candidates])])
            src_mass = best_a[i_src]
            if src_mass < 1e-14:
                break

        delta = src_mass * max_step_frac   # move up to max_step_frac of the source
        a_new = best_a.copy()
        a_new[i_src] -= delta
        a_new[i_dst] += delta
        # keep simplex constraint (sum unchanged) and non‑negativity
        # (delta ≤ src_mass guarantees non‑negativity)

        obj_new = _objective(a_new)
        if obj_new < best_obj - 1e-14:
            best_a = a_new
            best_obj = obj_new
            # optionally update the peak index for next iteration
        # otherwise try a smaller step (optional – omitted for speed)
    return best_a, best_obj

            # resize if necessary
            if seq0.size != n:
                if seq0.size > n:
                    init = seq0[:n]
                else:
                    pad = np.zeros(n - seq0.size)
                    init = np.concatenate([seq0, pad])
                init = init / init.sum()
            else:
                init = seq0

        # ---- Adam optimisation ---------------------------------------
        a_opt, obj_opt = _adam_opt(
            n=n,
            time_limit=run_time,
            init_a=init,
            seed=seed + idx * 73,
        )
        if obj_opt < best_obj:
            best_obj = obj_opt
            best_seq = a_opt.tolist()
            print(f"[global] new overall best O={best_obj:.6f} (n={n})")

        # ---- gradient‑guided local refinement -------------------------
        remaining_local = deadline - time.time()
        if remaining_local > 0.3:
            a_ref, obj_ref = _gradient_local_search(
                np.array(best_seq, dtype=float),
                time_budget=remaining_local * 0.30,
                max_step_frac=0.04,
            )
            if obj_ref < best_obj:
                best_obj = obj_ref
                best_seq = a_ref.tolist()
                print(f"[local] improved O={best_obj:.6f}")

    # -----------------------------------------------------------------
    # Final sanity check via the provided evaluate_sequence routine.
    try:
        final_score = evaluate_sequence(best_seq)
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    # guarantee a non‑empty return
    if not best_seq:
        best_seq = [1.0]

    return best_seq
