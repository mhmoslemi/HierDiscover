```
    start=time.time()
    deadline=start+budget_s-5
    best_seq=None
    best_score=float('inf')
    # try lengths
    lengths = [20000, 40000, 80000, 120000]  # up to a limit
    for n in lengths:

        # Build arcsine distribution
        idx=np.arange(n,dtype=np.float64)
        x=(idx+0.5)/n
        raw = 1.0/np.sqrt(x*(1.0-x))
        a = raw/raw.sum()    # sum=1
```

The evaluation metric is  

\[
J(a)=\frac{2\,n\,\max\limits_k (a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\]

which is *scale‑invariant*: multiplying every coefficient by a constant does not change the score.  
The continuous arcsine density  

\[
f(x)=\frac{1}{\pi\sqrt{x(1-x)}} ,\qquad x\in(0,1)
\]

has the remarkable property that its self‑convolution is constant (≈ ½).  
A discrete midpoint‐sampling of this density therefore gives a very good starting point – the larger the support size `n`, the closer the discrete convolution gets to the constant ideal and the smaller the score.

The algorithm proceeds in three stages

1. **Search over support sizes** – try several values of `n` (20 k, 40 k, 80 k, …).  
   For each `n` we build the raw arcsine vector, normalise it to a probability
   vector (sum = 1) and evaluate a *cheap* score `2·n·max(conv)`.  
   The best length and the corresponding vector are kept.

2. **Adam refinement on the probability simplex** – starting from the best
   vector we run a limited number of Adam steps on the *logits* (the
   parameters of a soft‑max).  
   A smoothed version of the max‑convolution (log‑sum‑exp) provides a
   differentiable surrogate; its gradient is obtained analytically via FFT‑based
   correlation.  The learning‑rate schedule, smoothing‑factor decay and a tiny
   random jitter help to escape small plateaus.

3. **Return the best sequence** – the final vector is converted to a plain
   Python list.  A final exact call to `evaluate_sequence` is performed only for
   logging; the returned list is what the platform will score.

The whole routine respects the 1000 s budget, works with a single thread
(and can safely use two CPUs) and never accesses the filesystem or the network.

```python
#!/usr/bin/env python
"""
Propose a step‑function coefficient vector that minimises the given metric.

The method:
  * Scan a few support sizes (20 k … 200 k) and pick the arcsine
    distribution with the lowest cheap score 2·n·max(conv).
  * Refine that vector with Adam on the simplex, using a smoothed
    max‑convolution (log‑sum‑exp) and FFT‑based gradients.
  * Return the best sequence (list of non‑negative floats).

All helper functions are top‑level; the implementation works under the
1000 s time limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable softmax: maps logits `z` to a probability vector."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(p: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Logits that produce `p` after softmax (adds a tiny epsilon)."""
    return np.log(p + eps)


def _fft_len_for_corr(n: int) -> int:
    """
    Smallest power‑of‑two ≥ 3·n‑2.
    This length is sufficient for a linear convolution of length (2·n‑1)
    and for the correlation used in the gradient.
    """
    return 1 << ((3 * n - 2).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a (FFT, no gradient)."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Smoothed max‑convolution and its gradient w.r.t. `a`.

    Parameters
    ----------
    a : np.ndarray
        Probability vector (non‑negative, sums to 1).
    eps_factor : float
        Multiplicative factor that determines the smoothing constant:
        eps = eps_factor * max(conv).  Smaller values make the surrogate
        approach the true max.

    Returns
    -------
    max_conv : float
        Exact maximum of a*a.
    grad_a : np.ndarray
        Gradient of 2·n·smooth_max (already multiplied by 4·n) with respect
        to the probability vector.
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    # FFT of the vector
    a_fft = np.fft.rfft(a, L)

    # Linear self‑convolution (length 2·n‑1)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # Soft‑max smoothing of the max
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv          # for numerical stability
    w = np.exp(shifted / eps)          # unnormalised weights
    w_sum = w.sum()
    if w_sum == 0.0 or np.isnan(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # Correlation a ⋆ w (length n)
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # Gradient of 2·n·smooth_max (the factor 4·n comes from the chain rule)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that minimises the evaluation metric.

    The routine first selects a good support size, builds the arcsine
    distribution, then applies a few hundred Adam steps on the simplex.
    All work stops before the provided time budget expires.
    """
    np.random.seed(seed)
    start = time.time()
    # Small safety margin so that the platform does not kill us while printing
    deadline = start + budget_s - 2.0

    # ------------------------------------------------------------------
    # 1. Scan several support sizes and keep the best arcsine vector.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    best_n = None

    # Candidate lengths (feel free to adjust; they all fit easily into memory)
    lengths = [20000, 40000, 80000, 120000, 200000]

    for n in lengths:
        if time.time() >= deadline:
            break

        # Build raw arcsine density at mid‑points
        idx = np.arange(n, dtype=np.float64)
        x = (idx + 0.5) / n                     # points in (0,1)
        raw = 1.0 / np.sqrt(x * (1.0 - x))

        # Normalise to a probability vector (sum = 1)
        a = raw / raw.sum()

        # Compute the (exact) max of the self‑convolution via FFT
        max_conv = _conv_max(a)
        cheap_score = 2.0 * n * max_conv

        print(f"[scan] n = {n:<6d}  cheap score = {cheap_score:.6f}")

        if cheap_score < best_score:
            best_score = cheap_score
            best_seq = a.copy()
            best_n = n

        # If we have already beaten the target comfortably we can stop early
        if cheap_score < 1.45:
            break

    # ------------------------------------------------------------------
    # 2. Optional Adam refinement on the simplex (only if time permits).
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n = best_n
        a_cur = best_seq.copy()
        # Initialise logits that correspond to the current distribution
        z = _logits_from_distribution(a_cur)

        # Adam state
        m = np.zeros_like(z)
        v = np.zeros_like(z)
        beta1, beta2 = 0.9, 0.999
        eps_adam = 1e-8
        base_alpha = 0.5          # base learning rate for the logits
        eps_factor = 0.12          # start smoothing factor
        eps_factor_min = 5e-5
        eps_factor_decay = 0.9995

        best_score_cur = best_score
        it = 0
        max_iters = 300_000    # generous upper bound; we stop by time

        while time.time() < deadline - 0.5 and it < max_iters:
            it += 1
            a_cur = _softmax(z)

            # Smoothed max‑convolution and its gradient
            max_conv, grad_a = _conv_max_and_grad(a_cur, eps_factor)
            cheap_score = 2.0 * n * max_conv

            if cheap_score < best_score_cur:
                best_score_cur = cheap_score
                best_seq = a_cur.copy()
                print(f"[adam] iter {it:<6d}  new best = {cheap_score:.6f}")

            # Gradient on the logits (mirror‑descent on the simplex)
            mu = np.dot(grad_a, a_cur)          # ⟨grad, a⟩
            grad_z = a_cur * (grad_a - mu)

            # Adam update (bias‑corrected)
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            # Inverse‑sqrt schedule for the step size
            alpha_t = base_alpha / (math.sqrt(it) + 1.0)
            step = - alpha_t * m_hat / (np.sqrt(v_hat) + eps_adam)
            z = z + step

            # Keep logits in a numerically safe range
            z = np.clip(z, -70.0, 70.0)

            # Decay the smoothing constant
            eps_factor = max(eps_factor_min, eps_factor * eps_factor_decay)

            # Occasional tiny random jitter to escape shallow plateaus
            if it % 5000 == 0 and np.random.rand() < 0.1:
                z += 0.005 * np.random.randn(n)

            # Early exit if we already beat the target comfortably
            if best_score_cur < 1.503:
                print("[adam] target reached, stopping early")
                break

        # Log the exact evaluation of the final candidate (for debugging)
        try:
            exact_score = evaluate_sequence(best_seq.tolist())
            print(f"[final] exact evaluation = {exact_score:.6f}")
        except Exception as exc:
            print("[final] evaluation error:", exc)

    # ------------------------------------------------------------------
    # 3. Return the best sequence (list of plain floats).
    # ------------------------------------------------------------------
    if best_seq is None:
        # Fallback – a trivial two‑element vector (should never happen)
        best_seq = np.array([1.0, 1.0])

    return best_seq.tolist()
```
