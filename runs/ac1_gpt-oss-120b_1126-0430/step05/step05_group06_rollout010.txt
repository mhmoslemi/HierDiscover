    n = a.size

    conv = _conv_fft(a, L)
    cur_max = conv.max()

    best_a = a.copy()
    best_obj = cur_obj

    it = 0
    T = init_T

    # optionally compute average step factor schedule: maybe compute fraction = (time.time() - start)/(deadline-start)
    start = time.time()

        it += 1

        # select source index i weighted by a (bigger mass more likely)
        # to avoid zero probabilities, we can add epsilon
        prob = a / a.sum()
        # if a sum may be 0 due to rounding? Not.
        i = np.random.choice(n, p=prob)

        j = np.random.randint(n-1)
        if j >= i:

        elapsed = time.time() - start
        total = deadline - start
        frac = elapsed / total

        # store conv changes for revert if needed

        conv[j:j+n] += 2.0 * delta * a

        conv[2*i] += d2
        conv[2*j] += d2

        new_max = conv.max()
        new_obj = 2.0 * n * new_max

"""
Propose a step‑function sequence that minimises

    O(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm is a fast local search on the simplex:
* start from a supplied or uniform distribution,
* pre‑compute the full self‑convolution via FFT,
* repeatedly move a small fraction of mass from a source index to a
  destination index,
* update the convolution *incrementally* (O(n) per move) instead of
  recomputing it with an FFT,
* accept downhill moves always and uphill moves with a Metropolis rule,
  with a temperature that cools over time,
* periodically prune tiny entries and recompute the convolution from
  scratch to keep numerical drift low,
* keep the best vector seen and return it.

The incremental update is exact (up to floating‑point noise) because

    Δconv_k = –2·δ·a_{k-i} + 2·δ·a_{k-j} + δ²·[δ_{k,2i}+δ_{k,2j}–2·δ_{k,i+j}]

for a transfer of δ from index i to j.
The search is very cheap (≈ 2·n operations per iteration) and can
perform hundreds of thousands of moves within the 1000 s budget,
typically improving the objective beyond the previous best.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full self‑convolution a*a using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
def _objective_from_conv(conv: np.ndarray, n: int) -> float:
    """Exact objective 2·n·max(conv) (sum(a) is assumed to be 1)."""
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _local_search(a0: np.ndarray, deadline: float,
                  init_T: float = 0.02,
                  T_decay: float = 0.99995,
                  min_T: float = 1e-7,
                  max_step_frac: float = 0.25,
                  min_step_frac: float = 0.002,
                  prune_every: int = 8000) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Fast simulated‑annealing style local search.

    Parameters
    ----------
    a0 : np.ndarray
        Starting probability vector (sum = 1, non‑negative).
    deadline : float
        Absolute time at which the search must stop.
    init_T, T_decay, min_T : float
        Temperature schedule.
    max_step_frac, min_step_frac : float
        Fraction of the source mass that may be moved (linearly decayed).
    prune_every : int
        How often to prune tiny entries and recompute the convolution.

    Returns
    -------
    a : np.ndarray
        Final vector (not necessarily the best).
    cur_obj : float
        Objective value of the final vector.
    best_a : np.ndarray
        Best vector seen during the search.
    """
    n = a0.size
    L = 1 << ((3 * n - 2).bit_length())
    a = a0.copy()
    conv = _conv_fft(a, L)
    cur_obj = _objective_from_conv(conv, n)
    best_a = a.copy()
    best_obj = cur_obj

    start = time.time()
    it = 0
    no_improve = 0
    T = init_T

    # maintain list of indices with non‑negligible mass
    nonzero_idx = np.where(a > 1e-14)[0]

    while time.time() < deadline:
        it += 1

        # re‑sample the list of active indices every few thousand steps
        if it % 2000 == 0:
            nonzero_idx = np.where(a > 1e-14)[0]
            if nonzero_idx.size == 0:            # should not happen, re‑initialise
                a = np.full(n, 1.0 / n)
                conv = _conv_fft(a, L)
                cur_obj = _objective_from_conv(conv, n)
                nonzero_idx = np.arange(n)

        # choose a source index weighted by its mass (more mass → more moves)
        i = np.random.choice(nonzero_idx)

        # choose a destination index uniformly (different from i)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        a_i = a[i]
        if a_i <= 0.0:
            continue

        # fraction of source mass to move – decays with time
        elapsed = time.time() - start
        total = deadline - start
        frac = elapsed / total
        step_frac = max_step_frac * (1.0 - frac) + min_step_frac * frac
        delta = a_i * step_frac * np.random.rand()
        if delta <= 0.0:
            continue

        # ---------- incremental convolution update ----------
        # vector used twice (save one multiplication)
        delta_vec = 2.0 * delta * a

        conv[i : i + n] -= delta_vec
        conv[j : j + n] += delta_vec

        d2 = delta * delta
        conv[2 * i] += d2
        conv[2 * j] += d2
        conv[i + j] -= 2.0 * d2
        # ----------------------------------------------------

        new_max = conv.max()
        new_obj = 2.0 * n * new_max

        # Metropolis acceptance
        accept = False
        if new_obj < cur_obj:
            accept = True
        else:
            if np.random.rand() < math.exp((cur_obj - new_obj) / max(T, 1e-12)):
                accept = True

        if accept:
            # commit the mass transfer
            a[i] = a_i - delta
            a[j] += delta
            cur_obj = new_obj
            no_improve = 0

            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                print(f"[search] iter {it:,d} new best O={best_obj:.6f}")

        else:
            # revert the convolution update (a unchanged)
            conv[i : i + n] += delta_vec
            conv[j : j + n] -= delta_vec
            conv[2 * i] -= d2
            conv[2 * j] -= d2
            conv[i + j] += 2.0 * d2
            no_improve += 1

        # cool down
        T = max(min_T, T * T_decay)

        # occasional pruning of tiny entries to keep sparsity and numerical stability
        if it % prune_every == 0:
            # prune the lowest 1 % of entries (or any that fall below a tiny threshold)
            thresh = np.percentile(a, 1.0)
            tiny = a < thresh
            if tiny.any():
                a[tiny] = 0.0
                total_mass = a.sum()
                if total_mass > 0.0:
                    a /= total_mass
                else:
                    a = np.full(n, 1.0 / n)
                # recompute convolution from scratch after pruning
                conv = _conv_fft(a, L)
                cur_obj = _objective_from_conv(conv, n)
                # reset best if the recomputation changed the objective
                if cur_obj < best_obj:
                    best_obj = cur_obj
                    best_a = a.copy()
                # refresh the non‑zero index list
                nonzero_idx = np.where(a > 1e-14)[0]

        # an occasional "restart" if we have been stuck for a long time
        if no_improve >= 20000:
            # jitter the current vector a bit and re‑normalise
            a += np.random.randn(n) * 0.001 * a
            a = np.clip(a, 0.0, None)
            s = a.sum()
            if s > 0.0:
                a /= s
            else:
                a = np.full(n, 1.0 / n)
            conv = _conv_fft(a, L)
            cur_obj = _objective_from_conv(conv, n)
            no_improve = 0
            nonzero_idx = np.where(a > 1e-14)[0]

    return a, cur_obj, best_a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function sequence that minimises the
    provided evaluation function.  The routine employs a fast
    incremental‐update simulated annealing that can perform many
    hundred‑thousands of moves within the allotted time.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # small safety margin

    # ------------------------------------------------------------------
    # initialise the probability vector
    if 'height_sequence_1' in globals():
        a0 = np.array(height_sequence_1, dtype=float)
        a0 = np.maximum(a0, 0.0)
        if a0.sum() == 0.0:
            a0 = np.full_like(a0, 1.0)
        a0 = a0 / a0.sum()
        n = a0.size
        print(f"[init] using supplied sequence, n={n}")
    else:
        n = 4600
        a0 = np.full(n, 1.0 / n)
        print(f"[init] using uniform start, n={n}")

    # ------------------------------------------------------------------
    # Phase 0 – a few cheap random restarts (optional, improves robustness)
    phase0_deadline = start_time + budget_s * 0.10
    best_a = a0.copy()
    conv0 = _conv_fft(best_a, 1 << ((3 * n - 2).bit_length()))
    best_obj = _objective_from_conv(conv0, n)

    while time.time() < phase0_deadline:
        # generate a very sparse Dirichlet‑type start
        conc = 0.12
        a_rand = np.random.gamma(conc, 1.0, size=n)
        if a_rand.sum() == 0.0:
            continue
        a_rand /= a_rand.sum()
        # run a short burst of local search (≈5 s)
        burst_deadline = time.time() + 5.0
        _, obj, best_local = _local_search(a_rand, burst_deadline,
                                           init_T=0.03,
                                           max_step_frac=0.30,
                                           min_step_frac=0.005)
        if obj < best_obj:
            best_obj = obj
            best_a = best_local.copy()
            print(f"[phase0] new best O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – main intensive local search on the current best vector
    print(f"[phase1] start intensive search, current best O={best_obj:.6f}")
    final_a, final_obj, final_best = _local_search(best_a, deadline,
                                                   init_T=0.02,
                                                   max_step_frac=0.25,
                                                   min_step_frac=0.003)

    if final_obj < best_obj:
        best_obj = final_obj
        best_a = final_best.copy()
        print(f"[phase1] improved best O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # final verification with the provided evaluator
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence failed:", exc)

    return best_a.tolist()
