"""
Propose a step‑function sequence that minimises

    O(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

The implementation follows a multi‑phase search:

1️⃣  **Initialisation & deterministic families** – try a few
    analytically‑motivated shapes (U‑shaped cosine, rational‑type,
    power‑law) and a set of random Dirichlet draws.  Keep the best
    candidate seen.

2️⃣  **Adam on a smooth surrogate** – minimise a log‑sum‑exp
    approximation of the max‑operator while projecting onto the
    probability simplex after each step.  Fast and gives a good basin
    of attraction.

3️⃣  **Exponentiated‑gradient (mirror‑descent) refinement** –\
    use the true sub‑gradient of the max‑convolution and update
    multiplicatively (a_i ← a_i·exp(‑η·g_i)), renormalising each
    iteration.  This respects the geometry of the simplex and often
    pushes the solution a little further.

4️⃣  **Projected sub‑gradient descent** – classic Euclidean sub‑gradient
    with a back‑tracking line‑search to guarantee descent.

5️⃣  **Targeted simulated‑annealing** – move a small fraction of mass
    from indices that actively contribute to the current maximal
    convolution entry to random destinations; accept moves with a
    Metropolis rule.  Periodic exact sub‑gradient steps keep the walk
    directed.

All phases respect the global 1000 s wall‑time budget and constantly
update the best sequence seen.  The final sequence is verified with
the provided ``evaluate_sequence`` function before being returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, ∑x=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # should not happen for non‑negative input, but fallback to uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear (self) convolution via FFT
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return a*a (full convolution) using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Exact objective for a simplex‑normalised vector (∑a = 1)
def _objective(a: np.ndarray, n: int, L: int) -> float:
    """2·n·max(conv(a,a))  (a is assumed to sum to 1)."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Sub‑gradient of the maximal convolution entry (for max(conv))
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a.
    If several indices attain the maximum, the gradient is averaged.
    """
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
# Gradient of the log‑sum‑exp (smooth max) surrogate
def _smooth_gradient(a: np.ndarray, conv: np.ndarray,
                     alpha: float, L: int) -> np.ndarray:
    """
    Compute ∇_a ( (2·n/α)·log∑_k e^{α·conv_k} ).
    The result reduces to 2·corr(w, a) where w_k ∝ e^{α·conv_k}.
    """
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)          # stabilise
    w = exp_scaled / exp_scaled.sum()

    # correlation w ⋆ a   (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n_len = a.size
    grad = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the smooth surrogate
def _adam_phase(a: np.ndarray, n: int, L: int,
                max_iters: int, deadline: float,
                alpha_start: float = 5.0,
                alpha_end: float = 20000.0,
                lr0: float = 0.6,
                beta1: float = 0.9,
                beta2: float = 0.999,
                eps: float = 1e-12,
                prune_every: int = 5000,
                prune_percentile: float = 1.0,
                verbose: bool = False) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Adam optimisation on the smooth log‑sum‑exp surrogate.
    Returns (final a, best objective value, best a encountered).
    """
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_obj = _objective(a, n, L)

    total_iters = max_iters
    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            if verbose and it % 1000 == 0:
                print(f"[adam] iter {it:,} new best O={best_obj:.6f}")

        # exponential schedule for α – sharper towards the end
        frac = it / total_iters
        alpha = alpha_start * (alpha_end / alpha_start) ** frac

        grad = _smooth_gradient(a, conv, alpha, L)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        lr = lr0 / math.sqrt(it + 1)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape flat plateaus
        if np.random.rand() < 0.001:
            a_new += np.random.randn(n) * 1e-4

        a_new = _project_to_simplex(a_new)

        # prune tiny entries every few thousand steps
        if prune_every and it % prune_every == 0 and it > 0:
            thresh = np.percentile(a_new, prune_percentile)
            a_new[a_new < thresh] = 0.0
            a_new = _project_to_simplex(a_new)

        a = a_new

    return a, best_obj, best_a


# ----------------------------------------------------------------------
# Exponentiated (mirror) gradient descent on the true objective
def _exp_gradient_phase(a: np.ndarray, n: int, L: int,
                       max_iters: int, deadline: float,
                       eta0: float = 0.5,
                       verbose: bool = False) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Mirror descent on O = 2·n·max(conv) using the exact sub‑gradient.
    Updates are multiplicative: a_i ← a_i·exp(‑η·g_i), followed by normalisation.
    """
    best_a = a.copy()
    best_obj = _objective(a, n, L)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            if verbose and it % 2000 == 0:
                print(f"[exp‑grad] iter {it:,} new best O={best_obj:.6f}")

        # sub‑gradient of max(conv)
        g = _subgradient_max_conv(a, conv)               # ∂max/∂a

        # step‑size schedule (diminishing)
        eta = eta0 / math.sqrt(it + 1.0)

        # multiplicative update (mirror descent on the simplex)
        # we incorporate the factor 2·n into η to simplify
        log_a = np.log(a + 1e-30) - eta * (2.0 * n) * g
        a = np.exp(log_a)
        a /= a.sum()    # projection onto simplex (exact for multiplicative weights)

    return a, best_obj, best_a

# Exact sub‑gradient descent with simple back‑tracking line search
def _subgradient_phase(a: np.ndarray, n: int, L: int,
                       max_iters: int, deadline: float,
                       eta0: float = 0.3,
                       line_search: bool = True,
                       verbose: bool = False) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Projected sub‑gradient descent on the exact objective.
    Returns (final a, best objective, best a).
    """
    best_a = a.copy()
    best_obj = _objective(a, n, L)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            if verbose and it % 2000 == 0:
                print(f"[sub‑grad] iter {it:,} new best O={best_obj:.6f}")

        g = _subgradient_max_conv(a, conv)

        # geometric decay of the step size
        eta = eta0 * (0.9999 ** it)

        if line_search:
            step = eta
            for _ in range(5):
                cand = _project_to_simplex(a - step * g)
                cand_obj = _objective(cand, n, L)
                if cand_obj < cur_obj:
                    a = cand
                    break
                step *= 0.5
            else:
                a = _project_to_simplex(a - eta * g)
        else:
            a = _project_to_simplex(a - eta * g)

    return a, best_obj, best_a

# Simulated annealing with mass‑transfer moves that target the current
# maximal convolution entry.
def _simulated_annealing_phase(a: np.ndarray, n: int, L: int,
                               max_iters: int, deadline: float,
                               temperature0: float = 0.07,
                               no_improve_limit: int = 7000,
                               verbose: bool = True) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Local refinement: move a modest fraction of probability mass from an
    index that participates in the current maximal convolution entry to a
    random destination; accept with Metropolis rule.  Every few steps a
    deterministic sub‑gradient step is applied.
    """
    best_a = a.copy()
    best_obj = _objective(a, n, L)
    cur_obj = best_obj
    conv = _conv_fft(a, L)

    T = temperature0
    no_improve = 0

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # index of the current peak of the convolution
        kmax = int(np.argmax(conv))

        # indices that actually contribute to that peak
        i_candidates = np.arange(max(0, kmax - (n - 1)), min(n - 1, kmax) + 1)
        i_candidates = i_candidates[a[i_candidates] > 1e-14]
        if i_candidates.size == 0:
            i = np.random.randint(n)
        else:
            i = np.random.choice(i_candidates)

        # random destination
        j = np.random.randint(n)
        while j == i:
            j = np.random.randint(n)

        # transfer a modest fraction of the source mass
        delta = a[i] * np.random.rand() * 0.09
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = _project_to_simplex(a_new)

        conv_new = _conv_fft(a_new, L)
        new_obj = 2.0 * n * conv_new.max()

        # Metropolis acceptance
        if new_obj < cur_obj or np.random.rand() < math.exp((cur_obj - new_obj) / max(T, 1e-12)):
            a = a_new
            conv = conv_new
            cur_obj = new_obj
            no_improve = 0
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                if verbose:
                    print(f"[SA] iter {it:,} new global best O={best_obj:.6f}")
        else:
            no_improve += 1

        # occasional exact sub‑gradient step (keeps direction)
        if it % 9 == 0:
            g = _subgradient_max_conv(a, conv)
            eta_loc = 0.07 * (1.0 - it / max_iters)
            a = _project_to_simplex(a - eta_loc * g)
            conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                if verbose:
                    print(f"[SA‑sub] iter {it:,} improved O={best_obj:.6f}")

        # temperature decay
        T *= 0.9997

        # restart if stagnating for a while
        if no_improve >= no_improve_limit:
            a = best_a.copy()
            a += np.random.randn(n) * 0.0015
            a = np.clip(a, 0, None)
            a = _project_to_simplex(a)
            conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max()
            no_improve = 0

    return a, best_obj, best_a

# Deterministic shape families (U‑shaped cosine, rational, power‑law)
def _shape_cosine(n: int, alpha: float) -> np.ndarray:
    """U‑shaped cosine: a_i ∝ 1 + α·cos(2πi/(n‑1)).  α∈[0,1]."""
    a = 1.0 + alpha * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _shape_rational(n: int, delta: float) -> np.ndarray:
    """
    Rational shape: a_i ∝ 1/((i+δ)*(n‑1‑i+δ)).
   δ>0 shifts the singularities away from the borders.
    """
    idx = np.arange(n, dtype=float)
    a = 1.0 / ((idx + delta) * (n - 1 - idx + delta))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _shape_powerlaw(n: int, p: float) -> np.ndarray:
    """
    Symmetric power‑law: a_i ∝ (i+0.5)^(-p) + (n‑i‑0.5)^(-p).
    p>0 makes the ends heavier.
    """
    idx = np.arange(n, dtype=float)
    left = (idx + 0.5) ** (-p)
    right = (n - idx - 0.5) ** (-p)
    a = left + right
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

# Main entry point expected by the evaluation harness
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    provided evaluation function.  Returns a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2       # safety margin

    # ------------------------------------------------------------------
    # Determine problem size ------------------------------------------------
    if 'height_sequence_1' in globals():
        a0 = np.array(height_sequence_1, dtype=float)
        a0 = np.maximum(a0, 0.0)
        if a0.sum() == 0.0:
            a0 = np.full_like(a0, 1.0)
        a0 = a0 / a0.sum()
        n = a0.size
        a = a0.copy()
        print(f"[init] using supplied sequence, n={n}")
    else:
        # default size – a good compromise between expressivity and speed
        n = 4600
        a = np.full(n, 1.0 / n)   # uniform start on the simplex
        print(f"[init] using uniform start, n={n}")

    # FFT length – next power of two greater than 3·n‑2
    L = 1 << ((3 * n - 2).bit_length())

    # ------------------------------------------------------------------
    # Phase 0 – deterministic shapes + cheap random restarts
    phase0_deadline = start_time + budget_s * 0.13
    best_seq = a.copy()
    best_score = _objective(best_seq, n, L)

    # deterministic families
    print("[phase0] evaluating deterministic shape families")
    # cosine family
    for alpha in np.linspace(0.0, 0.99, 15):
        cand = _shape_cosine(n, alpha)
        sc = _objective(cand, n, L)
        if sc < best_score:
            best_score, best_seq = sc, cand.copy()
            print(f"[phase0] cosine α={alpha:.2f} → O={sc:.6f}")

    # rational family
    for delta in [0.1, 0.3, 0.6, 1.0, 2.0, 5.0]:
        cand = _shape_rational(n, delta)
        sc = _objective(cand, n, L)
        if sc < best_score:
            best_score, best_seq = sc, cand.copy()
            print(f"[phase0] rational δ={delta:.1f} → O={sc:.6f}")

    # power‑law family
    for p in np.linspace(0.2, 3.0, 15):
        cand = _shape_powerlaw(n, p)
        sc = _objective(cand, n, L)
        if sc < best_score:
            best_score, best_seq = sc, cand.copy()
            print(f"[phase0] power‑law p={p:.2f} → O={sc:.6f}")

    # cheap Dirichlet restarts
    while time.time() < phase0_deadline:
        conc = np.random.choice([0.2, 0.4, 0.6, 0.8])   # encourage sparsity / diversity
        a_rand = np.random.gamma(conc, 1.0, size=n)
        a_rand = a_rand / a_rand.sum()
        # a few quick Adam steps
        sub_dead = time.time() + 3.0
        a_rand, _, best_rand = _adam_phase(
            a_rand, n, L,
            max_iters=1200,
            deadline=sub_dead,
            alpha_start=5.0,
            alpha_end=2000.0,
            lr0=0.8,
            verbose=False
        )
        cur_obj = _objective(best_rand, n, L)
        if cur_obj < best_score:
            best_score, best_seq = cur_obj, best_rand.copy()
            print(f"[phase0‑rand] new best O={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – long Adam optimisation on the smooth surrogate
    phase1_deadline = start_time + budget_s * 0.55
    a, obj_after_adam, best_adam = _adam_phase(
        best_seq, n, L,
        max_iters=450_000,
        deadline=phase1_deadline,
        alpha_start=5.0,
        alpha_end=50_000.0,
        lr0=0.6,
        verbose=False
    )
    if obj_after_adam < best_score:
        best_score, best_seq = obj_after_adam, best_adam.copy()
        print(f"[phase‑1] improved best O={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – exponentiated (mirror) gradient refinement
    phase2_deadline = start_time + budget_s * 0.75
    a, obj_exp, best_exp = _exp_gradient_phase(
        best_seq, n, L,
        max_iters=200_000,
        deadline=phase2_deadline,
        eta0=0.4,
        verbose=False
    )
    if obj_exp < best_score:
        best_score, best_seq = obj_exp, best_exp.copy()
        print(f"[phase‑2‑exp] improved best O={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – exact sub‑gradient descent with back‑tracking line search
    phase3_deadline = start_time + budget_s * 0.85
    a, obj_sub, best_sub = _subgradient_phase(
        best_seq, n, L,
        max_iters=250_000,
        deadline=phase3_deadline,
        eta0=0.25,
        line_search=True,
        verbose=False
    )
    if obj_sub < best_score:
        best_score, best_seq = obj_sub, best_sub.copy()
        print(f"[phase‑3‑sub] improved best O={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – targeted simulated annealing refinement (full remaining budget)
    a, obj_sa, best_sa = _simulated_annealing_phase(
        best_seq, n, L,
        max_iters=300_000,
        deadline=deadline,
        temperature0=0.07,
        no_improve_limit=7000,
        verbose=True
    )
    if obj_sa < best_score:
        best_score, best_seq = obj_sa, best_sa.copy()
        print(f"[phase‑4‑SA] final best O={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final verification through the provided evaluator
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence failed:", exc)

    return best_seq.tolist()
