        obj = float(2 * seq.size * np.max(np.convolve(seq, seq)))

            best_seq = seq.copy()
            print(f"[candidate] new best O={best_obj:.6f} n={seq.size}")

    # Phase 0: optionally load existing sequence
    if 'height_sequence_1' in globals():
        try_seq(globals()['height_sequence_1'])

            try_seq(a)

                a = a / a_sum
                try_seq(a)
                if time.time() >= deadline: break

    lengths = [len(best_seq) if best_seq is not None else 2000] + [2000, 4000, 8000]
    lengths = sorted(set(l for l in lengths if l>0))
    # For each length allocate time

"""
Step‑function optimizer for the convolution‑max objective.

The algorithm combines:
  1. Fast parametric sweeps (geometric / beta‑shaped families) to obtain a
     good initial shape.
  2. Adam optimisation on the log‑softmax parametrisation with an
     exponentially annealed smooth‑max (β‑schedule).  Several support sizes
     are tried; the longest run receives most of the time budget.
  3. Targeted simulated‑annealing that moves mass away from the indices
     that contribute most to the current maximal convolution entry.
The whole routine respects the 1000 s budget and returns a plain list of
non‑negative floats suitable for ``evaluate_sequence``.
"""
import time
import numpy as np

# ---------- low‑level helpers ----------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return conv_full[: 2 * n - 1]

def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation g_i = Σ_t w[i+t]·a[t]  for i=0,…,n‑1.
    Implemented via FFT.
    """
    n = a.shape[0]
    m = w.shape[0]                     # = 2n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]

def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()

# ---------- exact objective (uses sum=1) ----------
def _objective_exact(a: np.ndarray) -> float:
    """Exact O = 2 * n * max(conv) for sum(a)=1."""
    n = a.size
    conv = _conv_self_fft(a)
    M = conv.max()
    return float(2.0 * n * M)

# ---------- Adam optimiser ----------
def _run_opt(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
) -> tuple[list[float], float]:
    """
    Adam optimisation on the log‑it simplex.

    Returns best sequence (list) and its exact objective.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # initialise a on the simplex
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
        a = a / max(a.sum(), 1e-12)
    else:
        a = rng.random(n)
        a = a / a.sum()

    x = np.log(a + 1e-16)       # logits
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # schedule parameters (more aggressive than the reference)
    beta_start = 0.7
    beta_end = 5e4               # sharper smoothing
    lr_start = 0.2
    lr_end = 4e-4

    best_a = a.copy()
    best_obj = _objective_exact(best_a)

    it = 0
    while time.time() - start < time_limit:
        it += 1
        a = _softmax(x)                     # stay on the simplex

        conv = _conv_self_fft(a)
        M = conv.max()
        obj = float(2.0 * n * M)            # denominator = 1 by construction
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a.copy()
            print(f"[Adam n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        # annealing / learning‑rate schedule (time‑based)
        t = min((time.time() - start) / time_limit, 1.0)
        beta = beta_start * np.exp(np.log(beta_end / beta_start) * t)
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t)

        # smooth max weights
        if beta > 1e4:
            # hard max
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            shifted = conv - M                # ≤ 0
            w = np.exp(beta * shifted)
            s = w.sum()
            if s == 0.0:
                w = np.ones_like(conv) / conv.size
            else:
                w /= s

        # gradient of (smooth) max w.r.t. a
        g_a = 2.0 * _correlation_fft(w, a)

        # natural gradient in log‑it space (mirrored gradient)
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape plateaus
        if it % 4000 == 0:
            jitter = rng.normal(scale=0.008, size=n)
            x += jitter

        # occasional restart from current best
        if it % 70000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.003, size=n)
            m.fill(0.0)
            v.fill(0.0)

    # compress negligible entries
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = _objective_exact(best_a)
        print(f"[compress] length → {best_a.size}, O={best_obj:.6f}")

    # short local mass‑transfer refinement (if time left)
    remaining = time_limit - (time.time() - start)
    if remaining > 0.5:
        a_loc = best_a.copy()
        obj_loc = best_obj
        n_loc = a_loc.size
        it_loc = 0
        while time.time() - start < start + time_limit * 0.3:
            it_loc += 1
            i, j = rng.choice(n_loc, size=2, replace=False)
            delta = a_loc[i] * rng.random() * 0.02
            if delta <= 0.0:
                continue
            a_new = a_loc.copy()
            a_new[i] -= delta
            a_new[j] += delta
            obj_new = _objective_exact(a_new)
            if obj_new < obj_loc:
                a_loc, obj_loc = a_new, obj_new
                if obj_new < best_obj:
                    best_a, best_obj = a_new, obj_new
        print(f"[local] final O={best_obj:.6f}")

    return best_a.tolist(), best_obj

# ---------- Simulated annealing ----------
def _run_sa(
    init_seq: list[float],
    time_limit: float,
    seed: int = 0,
) -> tuple[list[float], float]:
    """
    Targeted simulated‑annealing refinement.
    Two types of moves are used:
      * single‑mass transfer (as in the reference)
      * symmetric pair transfer that respects the current maximal sum index.
    """
    start = time.time()
    rng = np.random.default_rng(seed)
    a = np.asarray(init_seq, dtype=float)
    if a.sum() < 1e-12:
        a = np.ones_like(a) / a.size
    else:
        a = a / a.sum()
    n = a.size
    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())
    best_seq = a.copy()

    # temperature schedule
    T = best_obj * 0.12 + 1e-6
    T_min = best_obj * 1e-6 + 1e-9
    decay = 0.996
    it = 0

    while time.time() - start < time_limit:
        it += 1
        # locate current maximum convolution index
        k_max = int(np.argmax(conv))

        # compute contribution weights for each i
        i_vals = np.arange(n)
        j_vals = k_max - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i_vals[mask]] * a[j_vals[mask]]
        total = contrib.sum()
        if total > 0:
            probs = contrib / total
            i = rng.choice(n, p=probs)
        else:
            i = rng.integers(n)
        j = k_max - i
        # pick destination indices
        i_new = rng.integers(n)
        j_new = rng.integers(n)
        # avoid trivial moves
        if i_new == i and j_new == j:
            continue
        # decide move type
        if rng.random() < 0.6:          # single‑mass move (60 %)
            delta = a[i] * rng.random() * 0.07   # up to 7 % of a_i
            if delta <= 0:
                continue
            a_new = a.copy()
            a_new[i] -= delta
            a_new[i_new] += delta
        else:                           # symmetric pair move (40 %)
            # move the same amount from i and j (if they differ)
            delta_i = a[i] * rng.random() * 0.04
            delta_j = a[j] * rng.random() * 0.04 if j != i else delta_i
            a_new = a.copy()
            a_new[i] = max(0.0, a_new[i] - delta_i)
            a_new[j] = max(0.0, a_new[j] - delta_j)
            a_new[i_new] += delta_i
            a_new[j_new] += delta_j
        # renormalise
        a_new = np.clip(a_new, 0.0, None)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-12:
            continue
        a_new /= a_new_sum

        conv_new = _conv_self_fft(a_new)
        obj_new = float(2.0 * n * conv_new.max())

        # Metropolis acceptance
        if obj_new < best_obj - 1e-13:
            a, conv, best_obj, best_seq = a_new, conv_new, obj_new, a_new.copy()
            print(f"[SA it={it}] new best O={best_obj:.6f}")
        else:
            prob = np.exp(-(obj_new - best_obj) / T) if T > 0 else 0.0
            if rng.random() < prob:
                a, conv = a_new, conv_new

        # temperature decay
        if it % 1500 == 0:
            T = max(T * decay, T_min)

        # occasional global jitter
        if it % 8000 == 0:
            noise = rng.normal(scale=0.008, size=n)
            a = np.clip(a + noise, 0.0, None)
            a /= a.sum()
            conv = _conv_self_fft(a)

    # final pruning of tiny entries
    thresh = 1e-12
    mask = best_seq > thresh
    if mask.sum() < best_seq.size:
        comp = best_seq[mask]
        comp = comp / comp.sum()
        comp_obj = _objective_exact(comp)
        if comp_obj < best_obj:
            best_seq, best_obj = comp, comp_obj
            print(f"[SA compress] reduced length → {comp.size}, O={best_obj:.6f}")

    return best_seq.tolist(), best_obj

"""
Step‑function optimiser for the `evaluate_sequence` objective.

Strategy
--------
1. Load a previous construction if available.
2. Perform a fast parametric sweep using two families of sequences:
   * geometric decay   a_i ∝ (1‑r)·r^i
   * discrete Beta‑shape a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}
   The sweep tries many lengths and parameter values (including random draws)
   and keeps the best candidate found so far.
3. Refine the best candidate with a longer Adam optimisation on the
   soft‑max (log‑it) parametrisation of the simplex.  The smooth‑max
   parameter β grows exponentially (up to 5·10⁴) while the learning‑rate
   decays.  The optimiser works on several support sizes; the most
   promising length receives the bulk of the remaining time budget.
4. Finish with a targeted simulated‑annealing phase that moves mass away
   from indices contributing most to the current maximal convolution entry.
   Two move types are used: a single‑mass transfer and a symmetric pair
   transfer that respects the current maximal‑sum index.
5. Return the best sequence found (normalised to sum = 1).  The routine
   never exceeds the supplied time budget.

All helper functions are top‑level; no closures or lambda expressions are used.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return conv_full[: 2 * n - 1]


def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation  g_i = Σ_t w[i+t]·a[t]   for i=0,…,n‑1.
    Implemented via FFT (zero‑padded).
    """
    n = a.shape[0]
    m = w.shape[0]                     # = 2n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


def _objective_exact(a: np.ndarray) -> float:
    """Exact objective  O = 2·n·max(conv(a,a)) for a on the simplex."""
    n = a.size
    conv = _conv_self_fft(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Adam optimiser (log‑it simplex)
# ----------------------------------------------------------------------
def _run_opt(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
) -> tuple[list[float], float]:
    """
    Adam optimisation of the smooth‑max surrogate on the simplex.
    Returns the best sequence (as a list) and its exact objective.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # initialise a on the simplex
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
        a = a / max(a.sum(), 1e-12)
    else:
        a = rng.random(n)
        a = a / a.sum()

    # logits (log‑it) representation
    x = np.log(a + 1e-16)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # schedule parameters (more aggressive than the reference)
    beta_start = 0.7
    beta_end = 5e4          # sharper smoothing
    lr_start = 0.2
    lr_end = 4e-4

    best_a = a.copy()
    best_obj = _objective_exact(best_a)

    it = 0
    while time.time() - start < time_limit:
        it += 1
        a = _softmax(x)                     # stay on the simplex

        conv = _conv_self_fft(a)
        M = conv.max()
        obj = float(2.0 * n * M)
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a.copy()
            print(f"[Adam n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        # annealed β and learning‑rate (time‑based)
        t = min((time.time() - start) / time_limit, 1.0)
        beta = beta_start * np.exp(np.log(beta_end / beta_start) * t)
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t)

        # smooth‑max weights
        if beta > 1e4:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            shifted = conv - M                # ≤ 0
            w = np.exp(beta * shifted)
            s = w.sum()
            if s == 0.0:
                w = np.ones_like(conv) / conv.size
            else:
                w /= s

        # gradient of smooth‑max w.r.t. a
        g_a = 2.0 * _correlation_fft(w, a)

        # natural gradient in log‑it space (mirrored gradient)
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape plateaus
        if it % 4000 == 0:
            x += rng.normal(scale=0.008, size=n)

        # restart from current best now and then
        if it % 70000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.003, size=n)
            m.fill(0.0)
            v.fill(0.0)

    # -----------------------------------------------------------------
    # prune negligible entries (helps the final objective)
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = _objective_exact(best_a)
        print(f"[compress] reduced length → {best_a.size}, O={best_obj:.6f}")

    # -----------------------------------------------------------------
    # short local mass‑transfer refinement if time permits
    remaining = time_limit - (time.time() - start)
    if remaining > 0.5:
        a_loc = best_a.copy()
        obj_loc = best_obj
        n_loc = a_loc.size
        it_loc = 0
        while time.time() - start < start + time_limit * 0.3:
            it_loc += 1
            i, j = rng.choice(n_loc, size=2, replace=False)
            delta = a_loc[i] * rng.random() * 0.02
            if delta <= 0.0:
                continue
            a_new = a_loc.copy()
            a_new[i] -= delta
            a_new[j] += delta
            obj_new = _objective_exact(a_new)
            if obj_new < obj_loc:
                a_loc, obj_loc = a_new, obj_new
                if obj_new < best_obj:
                    best_a, best_obj = a_new, obj_new
        print(f"[local] final O={best_obj:.6f}")

    return best_a.tolist(), best_obj


# ----------------------------------------------------------------------
# Simulated annealing refinement
# ----------------------------------------------------------------------
def _run_sa(
    init_seq: list[float],
    time_limit: float,
    seed: int = 0,
) -> tuple[list[float], float]:
    """
    Targeted simulated‑annealing that moves mass away from the
    convolution peak.  Two move types are employed:
      * single‑mass transfer
      * symmetric pair transfer (both sides of the peak).
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    a = np.asarray(init_seq, dtype=float)
    if a.sum() < 1e-12:
        a = np.ones_like(a) / a.size
    else:
        a = a / a.sum()
    n = a.size

    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())
    best_seq = a.copy()

    # temperature schedule
    T = best_obj * 0.12 + 1e-6
    T_min = best_obj * 1e-6 + 1e-9
    decay = 0.996
    it = 0

    while time.time() - start < time_limit:
        it += 1
        k_max = int(np.argmax(conv))

        # contribution weights for each index i
        i_vals = np.arange(n)
        j_vals = k_max - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i_vals[mask]] * a[j_vals[mask]]
        total = contrib.sum()
        if total > 0:
            probs = contrib / total
            i = rng.choice(n, p=probs)
        else:
            i = rng.integers(n)
        j = k_max - i

        # destination indices
        i_new = rng.integers(n)
        j_new = rng.integers(n)

        # decide which move type to apply
        if rng.random() < 0.6:                     # single‑mass move
            delta = a[i] * rng.random() * 0.07
            if delta <= 0.0:
                continue
            a_new = a.copy()
            a_new[i] -= delta
            a_new[i_new] += delta
        else:                                      # symmetric pair move
            delta_i = a[i] * rng.random() * 0.04
            delta_j = a[j] * rng.random() * 0.04 if j != i else delta_i
            a_new = a.copy()
            a_new[i] = max(0.0, a_new[i] - delta_i)
            a_new[j] = max(0.0, a_new[j] - delta_j)
            a_new[i_new] += delta_i
            a_new[j_new] += delta_j

        a_new = np.clip(a_new, 0.0, None)
        s = a_new.sum()
        if s < 1e-12:
            continue
        a_new /= s

        conv_new = _conv_self_fft(a_new)
        obj_new = float(2.0 * n * conv_new.max())

        if obj_new < best_obj - 1e-13:
            a, conv, best_obj, best_seq = a_new, conv_new, obj_new, a_new.copy()
            print(f"[SA it={it}] new best O={best_obj:.6f}")
        else:
            prob = np.exp(-(obj_new - best_obj) / T) if T > 0 else 0.0
            if rng.random() < prob:
                a, conv = a_new, conv_new

        if it % 1500 == 0:
            T = max(T * decay, T_min)

        if it % 8000 == 0:
            noise = rng.normal(scale=0.008, size=n)
            a = np.clip(a + noise, 0.0, None)
            a /= a.sum()
            conv = _conv_self_fft(a)

    # final pruning of tiny entries
    thresh = 1e-12
    mask = best_seq > thresh
    if mask.sum() < best_seq.size:
        comp = best_seq[mask]
        comp = comp / comp.sum()
        comp_obj = _objective_exact(comp)
        if comp_obj < best_obj:
            best_seq, best_obj = comp, comp_obj
            print(f"[SA compress] reduced length → {comp.size}, O={best_obj:.6f}")

    return best_seq.tolist(), best_obj

    else:
        preferred = [1024, 2048, 4096]
    preferred = sorted(set(l for l in preferred if l > 0))

    for idx, n in enumerate(preferred):
        now = time.time()
        remaining = deadline - now
        if remaining <= 2.0:
            break

        # allocate most time to the first (most promising) length
        if idx == 0:
            run_time = remaining * 0.55
        else:
            run_time = (remaining * 0.35) / (len(preferred) - idx)

        init = None
        if best_seq is not None and best_seq.size == n:
            init = best_seq.copy()

        seq, obj = _run_opt(
            n=n,
            time_limit=run_time,
            init_a=init,
            seed=seed + idx * 27491,
        )
        _maybe_update(np.asarray(seq, dtype=float))
        if time.time() >= deadline - 0.5:
            break

    # -----------------------------------------------------------------
    # 3) Simulated‑annealing refinement (if time permits)
    remaining = deadline - time.time()
    if best_seq is not None and remaining > 6.0:
        seq_ref, obj_ref = _run_sa(
            best_seq.tolist(),
            time_limit=remaining,
            seed=seed + 999999,
        )
        _maybe_update(np.asarray(seq_ref, dtype=float))

    # -----------------------------------------------------------------
    # 4) fallback
    if best_seq is None or best_seq.size == 0:
        best_seq = np.ones(100, dtype=float) / 100.0

    # -----------------------------------------------------------------
    # final sanity check (calls the external verifier)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_seq.tolist()
