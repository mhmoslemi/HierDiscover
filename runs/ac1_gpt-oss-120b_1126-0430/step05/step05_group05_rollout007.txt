def _grad_conv_penalty(a, conv, L):
    n = a.shape[0]
    avg = 1.0/(2*n - 1)
    d = conv - avg
    a_rev = a[::-1]
    # compute convolution d * a_rev
    fft_d = np.fft.rfft(d, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    conv_d_rev = np.fft.irfft(fft_d * fft_a_rev, n=L)
    # linear conv length = len(d)+len(a_rev)-1 = 3n-2
    # slice relevant part
    # conv_d_rev of length L (padded) contains linear conv in first (3n-2) entries.
    grad = 4.0 * conv_d_rev[n-1 : n-1+n]   # shape (n,)
    return grad

"""
propose_candidate – multi‑stage optimizer for the step‑function objective

The routine works in three phases under a 1000 s budget.

Phase 1 – Adam on a smooth surrogate:
    * Objective = (2·n/α)·log∑exp(α·conv)               (focuses on the max)
                + λ₁·∑a·log a                         (entropy, helps spread mass)
                + λ₂·∑(conv‑avg)²                     (flattens the whole convolution)
    * α, λ₁, λ₂ are annealed (α grows, λ’s shrink) while Adam updates a.
    * After each step the vector is projected onto the probability simplex
      (non‑negative, sum = 1).

Phase 2 – Simulated‑annealing local refinement:
    * Random “mass‑transfer’’ moves (move a random fraction of the probability
      mass from one index to another) are accepted if they improve the exact
      objective O = 2·n·max(conv) or with a Metropolis probability based on a
      temperature schedule.
    * A sub‑gradient step for the exact max‑convolution is performed every
      iteration, keeping the search directed.
    * Occasional smoothing of a helps avoid spikes.

Phase 3 – Final polishing:
    * Adam with an extremely large α (≈ 10⁶) and zero entropy/penalty, i.e.
      pure minimisation of the max‑convolution.

The best sequence found over all phases is returned as a plain list of floats.
"""

import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex {x≥0, sum x = 1}
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all entries are zero
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear self‑convolution via FFT (returns length 2·n‑1)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return a⋆a (full linear convolution) using FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Gradient of the smooth log‑sum‑exp surrogate plus entropy term
def _smooth_gradient_plus_entropy(
    a: np.ndarray,
    conv: np.ndarray,
    alpha: float,
    L: int,
    lam_entropy: float,
) -> np.ndarray:
    """
    Gradient of
        (2·n/α)·log∑_k e^{α·conv_k} + λ·∑ a_i·log a_i
    with respect to a (a is already normalised to sum = 1).

    The first part reduces to 2·corr(w, a) where
        w_k ∝ e^{α·conv_k}.
    The second part contributes λ·(log a_i + 1).
    """
    # ----- soft‑max weights -----
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)          # stable exp
    w = exp_scaled / exp_scaled.sum()                 # shape (2n‑1,)

    # ----- correlation w ⋆ a (grad_i = 2·∑_j w_{i+j} a_j) -----
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n_len = a.size
    grad_corr = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]

    # ----- entropy part -----
    eps = 1e-20
    grad_entropy = lam_entropy * (np.log(a + eps) + 1.0)

    return grad_corr + grad_entropy


# ----------------------------------------------------------------------
# Gradient of the quadratic flattening penalty  λ·∑(conv‑avg)²
def _grad_conv_penalty(
    a: np.ndarray,
    conv: np.ndarray,
    avg_conv: float,
    L: int,
) -> np.ndarray:
    """
    Gradient of   λ·∑_k (conv_k - avg)²   w.r.t. a.
    The analytical form is 4·Σ_k (conv_k‑avg)·a_{k‑i},
    which is a correlation of (conv‑avg) with a.
    """
    n = a.shape[0]
    d = conv - avg_conv                     # length 2·n‑1
    a_rev = a[::-1]
    fft_d = np.fft.rfft(d, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    conv_d_rev = np.fft.irfft(fft_d * fft_a_rev, n=L)
    # linear convolution length = (2n‑1) + (n)‑1 = 3n‑2
    # slice the central n entries (indices n‑1 … 2n‑2)
    grad = 4.0 * conv_d_rev[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) (used in Phase 2)
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a.
    For each index achieving the maximum the contribution is
        2·a_{k-i}   (with the factor 2 from the product rule).
    The returned gradient is averaged over all maximising indices.
    """
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)          # indices where max is attained
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g

            # ---------- random mass‑transfer move ----------
            i, j = np.random.randint(n, size=2)
            if i != j and a[i] > 1e-15:
                # fraction of source mass to move – shrinks linearly
                max_frac = 0.4 * (1.0 - it_loc / max_iter_local) + 0.05
                frac = np.random.rand() * max_frac
                delta = a[i] * frac
                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta
                a_new = _project_to_simplex(a_new)

                conv_new = _conv_fft(a_new, L)
                new_obj = 2.0 * n * conv_new.max()

                # Metropolis acceptance
                if new_obj < cur_obj or np.random.rand() < math.exp((cur_obj - new_obj) / max(T, 1e-12)):
                    a, conv, cur_obj = a_new, conv_new, new_obj
                    no_improve = 0
                    if cur_obj < best_score:
                        best_score = cur_obj
                        best_seq = a.copy()
                        print(f"[{time.time():.1f}] new global best O={best_score:.6f} (restart={max_restarts}, iter={it_loc})")
                else:
                    no_improve += 1
            else:
                no_improve += 1

            # ---------- sub‑gradient step ----------
            if it_loc % subgrad_interval == 0:
                g = _subgradient_max_conv(a, conv)
                eta = 0.12 * (1.0 - it_loc / max_iter_local)   # linearly decaying step
                a = a - eta * g
                a = _project_to_simplex(a)
                conv = _conv_fft(a, L)
                cur_obj = 2.0 * n * conv.max()
                if cur_obj < best_score:
                    best_score = cur_obj
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] sub‑grad improved O={best_score:.6f}")

            # ---------- occasional smoothing ----------
            if it_loc % 4000 == 0 and it_loc > 0:
                kernel = np.array([0.1, 0.2, 0.4, 0.2, 0.1])
                a_s = np.convolve(a, kernel, mode="same")
                a = _project_to_simplex(0.85 * a + 0.15 * a_s)

            # temperature decay
            T *= temp_decay
            it_loc += 1

            # restart if stagnating
            if no_improve >= no_improve_limit:
                break

        max_restarts += 1
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Phase 3 – Final high‑precision Adam polishing
    print("[phase‑3] final polishing")
    a = best_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    lr0 = 0.6
    alpha = 800_000.0          # huge smoothing → almost exact max
    lam_entropy = 0.0
    max_iter_phase3 = 30_000

    while time.time() < deadline and max_iter_phase3 > 0:
        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[phase‑3] iter {max_iter_phase3:,} new best O={best_score:.6f}")

        grad = _smooth_gradient_plus_entropy(a, conv, alpha, L, lam_entropy)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (max_iter_phase3 + 1))
        v_hat = v / (1.0 - beta2 ** (max_iter_phase3 + 1))
        lr = lr0 / math.sqrt(max_iter_phase3 + 1)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_to_simplex(a_new)

        max_iter_phase3 -= 1

    # ------------------------------------------------------------------
    # Final sanity check with the provided evaluator
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_val:.6f}")
    except Exception as e:
        print("[final] evaluate_sequence failed:", e)

    return best_seq.tolist()
