```
def project_simplex(v):
    # Based on algorithm by Duchi et al.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def _project_simplex(v):
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Simple stochastic construction for the step‑function problem.

We repeatedly generate random sparse sequences of a chosen length.
Each sequence puts a (Dirichlet‑sampled) probability mass on a random
subset of positions.  For such a distribution the linear convolution
(a * a) is close to uniform, which drives the objective

    O(a) = 2·n·max(conv(a,a))

evaluated exactly (via FFT) and the best one is optionally refined

current peak of the self‑convolution.  The search runs until the

found.

The algorithm is deliberately different from the previous Adam‑based
method: it relies on random combinatorial constructions and a simple
sub‑gradient optimiser rather than smooth‑max approximations.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear self‑convolution a*a using a real FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) assuming sum(a)=1.
    The vector a must be non‑negative.
    """
    n = a.size
    conv = _conv_self_fft(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x_i ≥ 0, Σ_i x_i = 1}.
    Returns a new array of the same shape.
    """
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find the rho value
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _random_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random probability vector of length n.
    A random subset of positions receives positive mass; the masses are
    drawn from an exponential distribution and then normalised.
    The subset size is chosen uniformly in [0.15·n, 0.5·n].
    """
    frac = rng.uniform(0.15, 0.5)          # density of the support
    m = max(2, int(frac * n))
    # unique indices for the support
    idx = rng.choice(n, size=m, replace=False)
    # random positive weights (Dirichlet with α=1)
    w = rng.exponential(scale=1.0, size=m)
    w = w / w.sum()
    a = np.zeros(n, dtype=float)
    a[idx] = w
    return a


# ----------------------------------------------------------------------
def _subgradient_refine(a: np.ndarray,
                        steps: int,
                        eta0: float,
                        rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    The step size decays geometrically.
    Returns the best vector seen and its objective value.
    """
    n = a.size
    best_a = a.copy()
    best_obj = _objective(a)
    eta = eta0

    for t in range(steps):
        conv = _conv_self_fft(a)
        k = int(conv.argmax())                     # index of the peak
        # gradient of conv_k wrt a: g_i = 2·a_{k-i} (zero out‑of‑range)
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        g = np.zeros_like(a)
        idx = np.arange(i_min, i_max + 1)
        g[idx] = a[k - idx]                        # a_{k-i}
        g *= 2.0

        # gradient step and projection back onto the simplex
        a_new = a - eta * g
        a_new = _project_simplex(a_new)

        # occasional tiny jitter to escape plateaus
        if (t % 50 == 0) and (rng.random() < 0.3):
            jitter = rng.normal(scale=0.008, size=n)
            a_new = a_new + jitter
            a_new = np.clip(a_new, 0.0, None)
            a_new = a_new / a_new.sum()

        obj = _objective(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        a = a_new
        eta *= 0.9995                               # slow decay

    return best_a, best_obj


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **_ignore) -> list[float]:
    """
    Entry point used by the evaluation harness.
    Generates many random sparse sequences, keeps the best, and
    performs a short sub‑gradient refinement.
    """
    start = time.time()
    deadline = start + budget_s - 0.5                # safety margin
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Lengths to explore.  We start with the length of a previous
    # solution (if present) and then try a list of increasing sizes.
    lengths = [1800, 2500, 3200, 4000, 5000]
    if "height_sequence_1" in globals():
        seq0 = np.asarray(height_sequence_1, dtype=float)
        seq0 = np.clip(seq0, 0.0, None)
        if seq0.sum() < 0.01:
            seq0 = np.ones_like(seq0)
        seq0 = seq0 / seq0.sum()
        n0 = seq0.size
        if n0 not in lengths:
            lengths.insert(0, n0)

    best_a = None
    best_obj = float("inf")

    # ------------------------------------------------------------------
    # Main stochastic search loop.
    while time.time() < deadline:
        # pick a length (more weight to larger sizes)
        n = int(rng.choice(lengths, p=np.array(lengths) / sum(lengths)))
        a = _random_candidate(n, rng)
        obj = _objective(a)

        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            print(f"[n={n:5d}] new best O={best_obj:.6f}")

        # If the candidate looks promising, apply a cheap refinement.
        # (We only spend a few hundred sub‑gradient steps to keep the
        # total budget under control.)
        if obj < best_obj * 0.98:
            a_ref, obj_ref = _subgradient_refine(
                a, steps=250, eta0=0.6, rng=rng
            )
            if obj_ref < best_obj:
                best_obj = obj_ref
                best_a = a_ref.copy()
                print(f"[refined n={n:5d}] improved O={best_obj:.6f}")

        # Occasionally insert a pure random jitter on the current best
        # to explore new regions.
        if best_a is not None and rng.random() < 0.005:
            jitter = rng.normal(scale=0.01, size=best_a.size)
            best_a = best_a + jitter
            best_a = np.clip(best_a, 0.0, None)
            best_a = best_a / best_a.sum()
            best_obj = _objective(best_a)

    # ------------------------------------------------------------------
    # Final optional refinement on the overall best solution.
    if best_a is not None:
        remaining = deadline - time.time()
        if remaining > 1.0:
            steps = int(max(200, remaining * 300))   # a few hundred steps
            best_a, best_obj = _subgradient_refine(
                best_a, steps=steps, eta0=0.4, rng=rng
            )
            print(f"[final refinement] O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Safeguard: return a non‑empty sequence.
    if best_a is None or best_a.size == 0:
        best_a = np.array([1.0])

    # ------------------------------------------------------------------
    # Final sanity check via the provided evaluate_sequence routine.
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_a.tolist()
