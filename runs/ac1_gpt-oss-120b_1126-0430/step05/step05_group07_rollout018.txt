def _smooth_gradient(a, conv, alpha, L):
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / exp_scaled.sum()
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    nlen = a.size
    grad_corr = 2.0 * corr_rev[nlen-1:nlen-1+nlen]
    return grad_corr
```

def variance_phase(a, n, L, max_iter, deadline):
    beta1=0.9; beta2=0.999; eps=1e-12
    lr0=0.6; lam_entropy=1e-4
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    it = 0
    best_seq = a.copy()
    best_obj = _objective_simplex(a, n, L)
    while it < max_iter and time.time() < deadline:
        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()  # sum=1 case O
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_seq = a.copy()
        grad = _grad_variance(a, conv, n, L)
        grad += lam_entropy * (np.log(a + 1e-20) + 1.0)  # entropy term
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        lr = lr0 / math.sqrt(it + 1)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = _project_to_simplex(a_new)
        a = a_new
        it += 1
        if it % 50000 == 0:
            print(f"[variance] iter {it} O={cur_obj:.6f}")
    return best_seq, best_obj, a

"""
A multi‑phase optimizer for the step‑function problem.

The algorithm works on a simplex (∑a_i = 1, a_i ≥ 0) and consists of:

1️⃣  Variance‑flattening – Adam optimisation of the variance of the
    full convolution.  This pushes all convolution entries towards the
    uniform mean.

2️⃣  Smooth max‑optimisation – Adam on a log‑sum‑exp surrogate of the
    max‑convolution (α is annealed up to 2·10⁴).  The gradient is obtained
    by a correlation computed with FFTs.

3️⃣  Stochastic local refinement – targeted mass‑transfer moves, a
    sub‑gradient step for the current maximal entry, occasional smoothing
    and a simple Metropolis acceptance rule.  The move set explicitly
    reduces the dominant autocorrelation lag.

The three phases share the total time budget (≈1000 s).  The best
sequence seen throughout the run is returned.  If a previously found
sequence ``height_sequence_1`` is available it is used as the initial
point; otherwise a uniform start is taken.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection ----------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution via FFT (length 2·n‑1) -----------------------
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return a★a (full convolution, length 2·len(a)‑1) using FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Objective (scale‑invariant) -----------------------------------------
def _objective(a: np.ndarray, n: int, L: int) -> float:
    """Return the exact evaluation O = 2·n·max(conv) / (sum a)²."""
    conv = _conv_fft(a, L)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return 2.0 * n * max_c / (s * s)


# ----------------------------------------------------------------------
# Gradient of variance of the convolution -------------------------------
def _grad_variance(a: np.ndarray, conv: np.ndarray, n: int, L: int) -> np.ndarray:
    """
    Gradient of Σ_k (conv_k – μ)²  where μ = 1/(2n‑1) (a is on the simplex).
    Closed form: g = 4·corr(conv‑μ , a) .
    """
    mu = 1.0 / (2 * n - 1)
    diff = conv - mu                     # length 2n‑1
    a_rev = a[::-1]

    fft_diff = np.fft.rfft(diff, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_diff * fft_a_rev, n=L)

    start = n - 1
    grad = 4.0 * corr_rev[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Gradient of the smooth (log‑sum‑exp) surrogate (no entropy) ----------
def _smooth_gradient(a: np.ndarray, conv: np.ndarray,
                     alpha: float, L: int) -> np.ndarray:
    """
    Gradient of (2·n/α)·log∑_k e^{α·conv_k}  (the factor 2·n is omitted,
    it only rescales the step‑size).  Returns Jᵀw where w is the
    soft‑max weight vector over the convolution entries.
    """
    scaled = alpha * conv
    max_scaled = np.max(scaled)               # numeric stability
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / exp_scaled.sum()         # shape (2n‑1,)

    # correlation w ★ a_rev  (same trick as in the variance term)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)

    start = a.size - 1
    grad_corr = 2.0 * corr_rev[start:start + a.size]
    return grad_corr


# ----------------------------------------------------------------------
# Sub‑gradient of the true max‑convolution -------------------------------
def _subgrad_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Return a sub‑gradient of max(conv) w.r.t. a."""
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)
    n = a.size
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]          # derivative of a_i a_j
    if ks.size:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
# Phase 1 – variance flattening (Adam) ---------------------------------
def variance_phase(a: np.ndarray, n: int, L: int,
                  max_iter: int, deadline: float):
    """Adam optimisation of the variance objective."""
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    lr0 = 0.6
    lam_entropy = 1e-4                     # tiny entropy stabilises positivity

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_seq = a.copy()
    best_score = _objective(a, n, L)
    it = 0

    while it < max_iter and time.time() < deadline:
        conv = _conv_fft(a, L)

        cur_score = 2.0 * n * conv.max()   # sum(a)=1 ⇒ exact O
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()

        grad = _grad_variance(a, conv, n, L)
        grad += lam_entropy * (np.log(a + 1e-20) + 1.0)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        lr = lr0 / math.sqrt(it + 1)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_to_simplex(a_new)

        it += 1
        if it % 50000 == 0:
            print(f"[variance] iter {it:,} O={cur_score:.6f}")

    return best_seq, best_score, a


# ----------------------------------------------------------------------
# Phase 2 – smooth max (Adam with annealed α) ---------------------------
def smooth_phase(a: np.ndarray, n: int, L: int,
                 max_iter: int, deadline: float,
                 alpha_start: float = 5.0, alpha_end: float = 20000.0):
    """Adam on the log‑sum‑exp surrogate of max(conv)."""
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    lr0 = 0.5
    lam = 0.0                     # no entropy – we want the true shape

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_seq = a.copy()
    best_score = _objective(a, n, L)
    it = 0

    while it < max_iter and time.time() < deadline:
        conv = _conv_fft(a, L)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()

        frac = it / max_iter
        alpha = alpha_start + (alpha_end - alpha_start) * frac

        grad = _smooth_gradient(a, conv, alpha, L)      # gradient of surrogate
        # (factor 2·n is omitted – it is absorbed by the learning‑rate)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        lr = lr0 / math.sqrt(it + 1)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape plateaus
        if random.random() < 0.001:
            a_new += np.random.randn(n) * 1e-5

        a = _project_to_simplex(a_new)
        it += 1

        if it % 50000 == 0:
            print(f"[smooth] iter {it:,} O={cur_score:.6f} α={alpha:.1f}")

    return best_seq, best_score, a


# ----------------------------------------------------------------------
# Phase 3 – stochastic local refinement ---------------------------------
def local_refine(a: np.ndarray, n: int, L: int, deadline: float):
    """
    Metropolis‑style mass‑transfer plus sub‑gradient steps.
    Extra “targeted” moves try to reduce the currently maximal
    convolution entry directly.
    """
    best_seq = a.copy()
    best_score = _objective(a, n, L)

    conv = _conv_fft(a, L)
    cur_score = 2.0 * n * conv.max()

    temperature = 0.04
    subgrad_interval = 5
    max_iter = 300_000
    it = 0
    no_improve = 0
    no_improve_limit = 4000

    while time.time() < deadline and it < max_iter:
        # ---------- random mass‑transfer ----------
        i, j = random.randint(0, n - 1), random.randint(0, n - 1)
        if i != j and a[i] > 1e-14:

            a_new[j] += delta

            conv_new = _conv_fft(a_new, L)
            new_score = 2.0 * n * conv_new.max()

            if new_score < cur_score or random.random() < math.exp(
                    (cur_score - new_score) / max(temperature, 1e-14)):
                a, conv, cur_score = a_new, conv_new, new_score
                no_improve = 0
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(f"[local] iter {it:,} new global best O={best_score:.6f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        if it % subgrad_interval == 0:

                    a_tmp = a.copy()
                    a_tmp[src] -= delta
                    a_tmp[dst] += delta
                    a_tmp = _project_to_simplex(a_tmp)

                    conv_tmp = _conv_fft(a_tmp, L)
                    new_score = 2.0 * n * conv_tmp.max()
                    if new_score < cur_score:
                        a, conv, cur_score = a_tmp, conv_tmp, new_score
                        no_improve = 0
                        if cur_score < best_score:
                            best_score = cur_score
                            best_seq = a.copy()
                            print(f"[local] targeted move improved O={best_score:.6f}")

        # ---------- occasional smoothing ----------
        if it % 4000 == 0 and it > 0:
            kernel = np.array([0.125, 0.25, 0.25, 0.25, 0.125])
            a_smooth = np.convolve(a, kernel, mode="same")
            a = _project_to_simplex(0.85 * a + 0.15 * a_smooth)

        # temperature decay
        temperature *= 0.9998
        it += 1

        # restart if stuck
        if no_improve >= no_improve_limit:
            a = best_seq.copy()
            a += np.random.randn(n) * 0.0015
            a = np.clip(a, 0, None)
            a = _project_to_simplex(a)
            conv = _conv_fft(a, L)
            cur_score = 2.0 * n * conv.max()
            no_improve = 0

    return best_seq, best_score

# Main entry point ------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a non‑negative step‑function that (approximately) minimises
    the evaluation function.  Returns the best sequence found within the
    allocated time budget.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.3      # safety margin

    # ------------------------------------------------------------------
    # Determine problem size
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
        n = base.size
        a0 = base / base.sum()
        print(f"[init] start from supplied sequence, n={n}")
    except Exception:
        n = 4600                     # reasonable default
        a0 = np.full(n, 1.0 / n)
        print(f"[init] start from uniform, n={n}")

    # FFT length – next power of two ≥ 3·n‑2 (covers all correlation ops)
    L = 1 << ((3 * n - 2).bit_length())

    # ------------------------------------------------------------------
    # Global best tracking
    best_seq = a0.copy()
    best_score = _objective(best_seq, n, L)
    print(f"[global] initial O = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – variance flattening
    phase1_deadline = start_time + budget_s * 0.30
    seq1, score1, a_mid = variance_phase(best_seq, n, L,
                                         max_iter=300_000,
                                         deadline=phase1_deadline)
    if score1 < best_score:
        best_score, best_seq = score1, seq1.copy()
        print(f"[phase‑1] improved global O = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – smooth max‑convolution surrogate
    phase2_deadline = start_time + budget_s * 0.70
    seq2, score2, a_mid = smooth_phase(a_mid, n, L,
                                       max_iter=400_000,
                                       deadline=phase2_deadline,
                                       alpha_start=5.0,
                                       alpha_end=20000.0)
    if score2 < best_score:
        best_score, best_seq = score2, seq2.copy()
        print(f"[phase‑2] improved global O = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – stochastic local refinement (uses remaining budget)
    seq3, score3 = local_refine(a_mid, n, L, deadline)
    if score3 < best_score:
        best_score, best_seq = score3, seq3.copy()
        print(f"[phase‑3] improved global O = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional quick restarts with random starts if time permits
    while time.time() < deadline - 5.0:
        # generate a fresh random start
        a_rand = np.random.rand(n)
        a_rand = _project_to_simplex(a_rand)

        # tiny short runs (few thousand iterations each) to inject diversity
        seq_tmp, _, a_tmp = variance_phase(a_rand, n, L,
                                           max_iter=30_000,
                                           deadline=time.time() + 10.0)
        seq_tmp, _, a_tmp = smooth_phase(a_tmp, n, L,
                                         max_iter=30_000,
                                         deadline=time.time() + 10.0,
                                         alpha_start=5.0,
                                         alpha_end=5000.0)
        seq_tmp, score_tmp = local_refine(a_tmp, n, L, deadline=time.time() + 15.0)

        if score_tmp < best_score:
            best_score, best_seq = score_tmp, seq_tmp.copy()
            print(f"[restart] new global best O = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check using the provided evaluation function
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O = {final_val:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence failed:", exc)

    return best_seq.tolist()
