Advanced multi‑scale optimizer for the step‑function bound.

The algorithm explores several support sizes.  For each size it runs an
Adam optimiser on the log‑it (soft‑max) parametrisation of the simplex,
using the *exact* gradient of the current maximal convolution entry.
Every few thousand iterations a tiny Gaussian jitter is added and every

After the Adam phase a short simulated‑annealing local‑search

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full self‑convolution a*a using real‑valued FFT.
    Returns a NumPy array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv

    """
    Exact objective 2·n·max(conv(a,a)) / (sum(a)**2) and the maximal
    convolution value.  The input is assumed to be non‑negative.
    """
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return float("inf"), 0.0
    conv = _conv_self_fft(a)
    max_b = conv.max()
    obj = 2.0 * n * max_b / (s * s)
    return float(obj), float(max_b)

            a_new[j] += delta
            obj_new, _ = _objective_value(a_new)

            # accept if better, otherwise with Metropolis probability
            if obj_new < obj_loc or rng.random() < np.exp(-(obj_new - obj_loc) / T):
                a_loc = a_new
                obj_loc = obj_new
                if obj_new < best_obj:
                    best_a = a_new
                    best_obj = obj_new

    return best_a.tolist(), best_obj


# -------------------------------------------------------------
def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **_ignore,
) -> list[float]:
    """
    Entry point used by the competition harness.

    A set of support sizes (powers of two) is explored.  For each size a
    dedicated Adam optimisation (with exact max‑convolution gradient) is
    performed, followed by a brief simulated‑annealing local search.
    The overall best sequence found within the time budget is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------- decide which lengths to try ---------------------------
    lengths = [256, 512, 1024, 2048, 4096, 8192]

    # if a previous solution exists, use it as a warm start
    global_seq = globals().get("height_sequence_1")
    if global_seq is not None:
        seq0 = np.asarray(global_seq, dtype=float)
        seq0 = np.clip(seq0, 0.0, None)
        if seq0.sum() < 1e-12:
            seq0 = np.ones_like(seq0)
        seq0 = seq0 / seq0.sum()
        n0 = seq0.size
        if n0 not in lengths:
            lengths.insert(0, n0)

    best_seq = None
    best_obj = float("inf")

    # ------- run optimisation for each length -----------------------
    for idx, n in enumerate(lengths):
        remaining = deadline - time.time()
        if remaining <= 1.0:
            break

        # allocate most time to the first (largest) run
        if idx == 0:
            run_time = remaining * 0.60
        else:
            run_time = (remaining - 0.5) / (len(lengths) - idx)

        # initialise from the global solution for the first round
        init = None
        if idx == 0 and global_seq is not None:
            if seq0.size != n:
                if seq0.size > n:
                    init = seq0[:n]
                else:
                    init = np.concatenate([seq0, np.zeros(n - seq0.size)])
                init = init / init.sum()
            else:
                init = seq0

        seq, val = _run_opt_adam(
            n=n,
            time_limit=run_time,
            seed=seed + idx * 73,
            init_a=init,
        )

        if val < best_obj:
            best_obj = val
            best_seq = seq
            print(f"[global] new overall best O={best_obj:.6f} (n={n})")

    # ------- final annealing on the overall best --------------------
    if best_seq is not None:
        remaining = deadline - time.time()
        if remaining > 1.0:
            a = np.array(best_seq, dtype=float)
            best_obj, _ = _objective_value(a)
            rng = np.random.default_rng(seed + 9999)
            anneal_start = time.time()
            T0 = 0.20
            T_end = 1e-7
            while time.time() - anneal_start < remaining:
                elapsed = time.time() - anneal_start
                progress = elapsed / remaining
                T = T0 * (T_end / T0) ** progress

                i, j = rng.choice(a.size, size=2, replace=False)
                delta_frac = rng.random() * 0.03
                delta = a[i] * delta_frac
                if delta == 0.0:
                    continue
                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta
                obj_new, _ = _objective_value(a_new)

                if obj_new < best_obj or rng.random() < np.exp(-(obj_new - best_obj) / T):
                    a = a_new
                    best_obj = obj_new
                    best_seq = a.tolist()

    # ------- final sanity check via the provided routine ------------
    try:
        final_score = evaluate_sequence(best_seq)  # type: ignore
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    # guarantee a non‑empty return
    if not best_seq:
        best_seq = [1.0]

    return best_seq
