def _subgradient_descent(a, n, L, deadline, max_iters=10**9):

    best_a = a.copy()
    best_obj = _objective(a, n, L)
    cur_obj = best_obj

    # step schedule param
    eta0 = 0.5  # initial step size
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
        # gradient
        g = _subgradient_max_conv(a, conv)
        # compute step size: decays as eta0 / sqrt(it+1)
        step = eta0 / math.sqrt(it + 1)
        # try backtracking to ensure descent
        a_new = _project_to_simplex(a - step * g)
        new_obj = _objective(a_new, n, L)
        # backtrack
        bt = 0
        while new_obj > cur_obj and bt < 5:
            step *= 0.5
            a_new = _project_to_simplex(a - step * g)
            new_obj = _objective(a_new, n, L)
            bt += 1
        a = a_new
    return a, best_obj, best_a

"""
Algorithm for minimising O = 2*n*max(conv(a,a))/(sum(a)^2).

1. Initialise a probability vector:
   - If a previously discovered sequence is available (height_sequence_1), use it.
   - Otherwise perform a cheap grid search over Beta‑distribution shapes
     a_i ∝ i^{α‑1}(n‑i)^{β‑1} (α,β>0) and keep the best.
   - Additionally sample a few Dirichlet draws for diversity.

2. Refine the best candidate with several Adam optimisations on the
   log‑sum‑exp smooth approximation of the max‑operator.  The smoothing
   parameter α is increased geometrically from a modest value to a very
   large one (≈10⁶) so that the surrogate becomes almost the true max.

3. Apply a long projected sub‑gradient descent on the exact objective.
   Step‑sizes follow a Polyak‑like rule based on the current objective
   gap and are back‑tracked to guarantee descent.

4. Finish with a targeted simulated‑annealing walk that repeatedly moves a
   small fraction of probability mass away from indices that actively
   contribute to the current maximal convolution entry.  The Metropolis
   rule gradually cools; occasional exact sub‑gradient steps keep the walk
   directed.

All phases respect a global 1000 s wall‑time budget and constantly keep
the best sequence seen.  The returned list is verified with the provided
`evaluate_sequence` function before being handed to the outer harness.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # All non‑positive – return the uniform simplex point
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution a * a via FFT of length L (L power of two)."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int, L: int) -> float:
    """Exact objective for a probability vector (∑a = 1): 2·n·max(conv)."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. a (averaged over all active ks)."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, conv: np.ndarray,
                     alpha: float, L: int) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp surrogate:
        (2·n/α)·log∑_k exp(α·conv_k)
    Returns a vector of length n.
    """
    # soft‑max weights w_k ∝ exp(α·conv_k)
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)      # stabilised exponentials
    w = exp_scaled / exp_scaled.sum()

    # correlation w ⋆ a  (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n_len = a.size
    grad = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]
    return grad


# ----------------------------------------------------------------------
def _adam_phase(a: np.ndarray, n: int, L: int,
                max_iters: int, deadline: float,
                alpha_start: float = 5.0,
                alpha_end: float = 5e5,
                lr0: float = 0.8,
                beta1: float = 0.9,
                beta2: float = 0.999,
                eps: float = 1e-12,
                prune_every: int = 0,
                prune_percentile: float = 1.0) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Adam optimiser on the smooth log‑sum‑exp surrogate.
    Returns (final a, best objective value seen, best a).
    """
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_obj = _objective(a, n, L)

    total_iters = max_iters
    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            # occasional progress report
            if it % 5000 == 0:
                print(f"[adam] iter {it} new best O={best_obj:.6f}")

        # exponential schedule for the smoothing parameter α
        frac = it / total_iters
        alpha = alpha_start * (alpha_end / alpha_start) ** frac

        grad = _smooth_gradient(a, conv, alpha, L)

        # Adam bias‑corrected update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        lr = lr0 / math.sqrt(it + 1)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # tiny stochastic jitter
        if np.random.rand() < 0.001:
            a_new += np.random.randn(n) * 1e-4

        a_new = _project_to_simplex(a_new)

        # optional pruning of extremely tiny entries
        if prune_every and it % prune_every == 0 and it > 0:
            thresh = np.percentile(a_new, prune_percentile)
            a_new[a_new < thresh] = 0.0
            a_new = _project_to_simplex(a_new)

        a = a_new

    return a, best_obj, best_a


# ----------------------------------------------------------------------
def _subgradient_phase(a: np.ndarray, n: int, L: int,
                       max_iters: int, deadline: float,
                       eta0: float = 0.4,
                       line_search: bool = True,
                       verbose: bool = False) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Projected sub‑gradient descent on the exact (non‑smooth) objective.
    Returns final a, best objective encountered, and the best a.
    """
    best_a = a.copy()
    best_obj = _objective(a, n, L)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            if verbose and it % 5000 == 0:
                print(f"[sub‑grad] iter {it} new best O={best_obj:.6f}")

        g = _subgradient_max_conv(a, conv)

        # Polyak‑type step size:  η = (cur_obj - best_obj) / ||g||^2 if useful,
        # otherwise fall back to a decaying schedule.
        norm_g_sq = np.dot(g, g)
        if norm_g_sq > 0 and cur_obj > best_obj:
            eta = (cur_obj - best_obj) / norm_g_sq
            # clamp to a reasonable interval
            eta = max(min(eta, eta0), eta0 * 0.01)
        else:
            eta = eta0 / math.sqrt(it + 1)

        if line_search:
            step = eta
            # simple back‑tracking (max 5 trials)
            for _ in range(5):
                cand = _project_to_simplex(a - step * g)
                cand_obj = _objective(cand, n, L)
                if cand_obj < cur_obj:
                    a = cand
                    break
                step *= 0.5
            else:
                a = _project_to_simplex(a - eta * g)
        else:
            a = _project_to_simplex(a - eta * g)

    return a, best_obj, best_a


# ----------------------------------------------------------------------
def _simulated_annealing_phase(a: np.ndarray, n: int, L: int,
                               max_iters: int, deadline: float,
                               temperature0: float = 0.07,
                               no_improve_limit: int = 8000,
                               verbose: bool = True) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Local stochastic refinement.  Moves a modest fraction of mass away
    from indices that participate in the current maximal convolution entry.
    Accepted according to the Metropolis rule with a slowly cooling temperature.
    """
    best_a = a.copy()
    best_obj = _objective(a, n, L)
    cur_obj = best_obj
    conv = _conv_fft(a, L)
    T = temperature0
    no_improve = 0

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # index of a current peak of the convolution
        kmax = int(np.argmax(conv))

        # set of i that actually contribute to this peak
        i_candidates = np.arange(max(0, kmax - (n - 1)), min(n - 1, kmax) + 1)
        i_candidates = i_candidates[a[i_candidates] > 1e-14]
        if i_candidates.size == 0:
            i = np.random.randint(n)
        else:
            i = np.random.choice(i_candidates)

        # random destination index (different from i)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # move a random fraction (up to 9%) of the source mass
        delta = a[i] * np.random.rand() * 0.09
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = _project_to_simplex(a_new)

        conv_new = _conv_fft(a_new, L)
        new_obj = 2.0 * n * conv_new.max()

        # Metropolis acceptance
        if new_obj < cur_obj or np.random.rand() < math.exp((cur_obj - new_obj) / max(T, 1e-12)):
            a = a_new
            conv = conv_new
            cur_obj = new_obj
            no_improve = 0
            if new_obj < best_obj:
                best_obj = new_obj
                best_a = a.copy()
                if verbose:
                    print(f"[SA] iter {it} new global best O={best_obj:.6f}")
        else:
            no_improve += 1

        # periodic exact sub‑gradient step (keeps walk directed)
        if it % 11 == 0:
            g = _subgradient_max_conv(a, conv)
            eta_loc = 0.07 * (1.0 - it / max_iters)
            a = _project_to_simplex(a - eta_loc * g)
            conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                if verbose:
                    print(f"[SA] sub‑grad improvement O={best_obj:.6f}")

        # temperature decay
        T *= 0.9997

        # restart if stagnating
        if no_improve >= no_improve_limit:
            a = best_a.copy()
            a += np.random.randn(n) * 0.0015
            a = np.clip(a, 0.0, None)
            a = _project_to_simplex(a)
            conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max()
            no_improve = 0

    return a, best_obj, best_a


# ----------------------------------------------------------------------
def _beta_initial(n: int, alpha: float, beta: float) -> np.ndarray:
    """Return a normalised Beta‑shape vector on {0,…,n‑1}."""
    # use 1‑based indices to avoid zero‑power issues
    i = np.arange(1, n + 1, dtype=float)
    a = (i ** (alpha - 1.0)) * ((n + 1 - i) ** (beta - 1.0))
    # protect against overflow/underflow
    a = np.where(np.isfinite(a), a, 0.0)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s

"""
Algorithm for minimising   O = 2·n·max(conv(a,a)) / (sum(a)^2)

1️⃣  Initialise a probability vector.
    – If a previous construction (height_sequence_1) is present, use it.
    – Otherwise run a cheap global search:
        * a few Dirichlet draws,
        * a dense grid of symmetric Beta‑shaped vectors,
        * a handful of random asymmetric Beta pairs.
    The best of these seeds is kept.

2️⃣  Refine the seed with Adam optimisation on the smooth
    log‑sum‑exp surrogate of the max‑operator.
    The smoothing parameter α is increased geometrically from 5 to 5·10⁵,
    making the surrogate increasingly sharp.

3️⃣  Perform a long projected sub‑gradient descent on the exact (non‑smooth)
    objective.  A Polyak‑type step size is used when it is informative,
    otherwise a √‑decay is applied.  Back‑tracking guarantees descent.

4️⃣  Finish with a targeted simulated‑annealing walk that repeatedly moves a
    small fraction of probability mass away from indices that actively
    contribute to the current maximal convolution entry.  The Metropolis rule
    cools slowly; occasional exact sub‑gradient steps keep the walk directed.

All phases respect a 1000 s wall‑time budget and continuously keep the
best sequence seen.  The final candidate is verified with the provided
`evaluate_sequence` before being returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution a*a via FFT of length L (L is a power of two)."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int, L: int) -> float:
    """Exact objective for a probability vector (∑a = 1): 2·n·max(conv)."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. a (averaged over all active ks)."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, conv: np.ndarray,
                     alpha: float, L: int) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp surrogate:
        (2·n/α)·log∑_k exp(α·conv_k)
    """
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / exp_scaled.sum()

    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n_len = a.size
    grad = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]
    return grad

def _adam_phase(a: np.ndarray, n: int, L: int,
                max_iters: int, deadline: float,
                alpha_start: float = 5.0,
                alpha_end: float = 5e5,
                lr0: float = 0.8,
                beta1: float = 0.9,
                beta2: float = 0.999,
                eps: float = 1e-12,
                prune_every: int = 0,
                prune_percentile: float = 1.0) -> tuple[np.ndarray, float, np.ndarray]:
    """Adam optimiser on the smooth (log‑sum‑exp) surrogate."""
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_obj = _objective(a, n, L)

    total_iters = max_iters
    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[adam] iter {it} new best O={best_obj:.6f}")

        frac = it / total_iters
        alpha = alpha_start * (alpha_end / alpha_start) ** frac

        grad = _smooth_gradient(a, conv, alpha, L)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        lr = lr0 / math.sqrt(it + 1)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        if np.random.rand() < 0.001:
            a_new += np.random.randn(n) * 1e-4

        a_new = _project_to_simplex(a_new)

        if prune_every and it % prune_every == 0 and it > 0:
            thresh = np.percentile(a_new, prune_percentile)
            a_new[a_new < thresh] = 0.0
            a_new = _project_to_simplex(a_new)

        a = a_new

    return a, best_obj, best_a

def _subgradient_phase(a: np.ndarray, n: int, L: int,
                       max_iters: int, deadline: float,
                       eta0: float = 0.5,
                       line_search: bool = True,

    """Projected sub‑gradient descent on the exact objective."""
    best_a = a.copy()
    best_obj = _objective(a, n, L)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            if verbose and it % 5000 == 0:
                print(f"[sub‑grad] iter {it} new best O={best_obj:.6f}")

        g = _subgradient_max_conv(a, conv)

        norm_g_sq = np.dot(g, g)
        if norm_g_sq > 0 and cur_obj > best_obj:
            eta = (cur_obj - best_obj) / norm_g_sq
            eta = max(min(eta, eta0), eta0 * 0.01)
        else:
            eta = eta0 / math.sqrt(it + 1)

            for _ in range(5):

                cand_obj = _objective(cand, n, L)
                if cand_obj < cur_obj:

                    break
                step *= 0.5
            else:
                a = _project_to_simplex(a - eta * g)
        else:

def _beta_initial(n: int, alpha: float, beta: float) -> np.ndarray:
    """Generate a normalised Beta‑shaped vector on {0,…,n‑1}."""
    i = np.arange(1, n + 1, dtype=float)               # 1‑based to avoid 0⁰
    a = (i ** (alpha - 1.0)) * ((n + 1 - i) ** (beta - 1.0))
    a = np.where(np.isfinite(a), a, 0.0)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s
