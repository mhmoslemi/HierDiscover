analysisThe problem is to produce a list of non-negative heights (real numbers) that minimize an evaluation function. The evaluation function is:

- Input is a list of floats.
- It checks it's a list of numbers, rejects non-float/int or NaN or infinite, nonlist error.
- Then sequence is turned into float. Then negative numbers are clamped to zero and numbers > 1000 are clamped to 1000 (should be safe).
- Let n = len(sequence).
- Compute b_sequence = np.convolve(sequence, sequence) (full linear convolution).
- max_b = max(b_sequence)
- sum_a = np.sum(sequence)
- If sum_a < 0.01 return inf.
- Return 2 * n * max_b / (sum_a**2).

So we want to minimize 2 n max_{k} (sum_i a_i a_{k-i}) / (sum a_i)^2.

We can see as we want the self-convolution to be as flat and low as possible relative to sum. Equivalent to maximizing denominator (sum a) and minimizing maximum of convolution. The scaling is not independent: if we scale the sequence by factor t > 0, sum scales by t, max_b scales by t^2, so numerator scales by t^2, denominator by t^2, ratio constant. So scaling does not matter; the metric is homogeneous: scaling a vector doesn't change it because:

- Let a' = c a. Then sum a' = c sum a.
- max_b' = max(c a * c a) = c^2 max_b.
- So 2 n max_b' / (sum a')^2 = 2 n (c^2 max_b) / (c^2 sum a^2) = same.

Thus scaling invariance: the evaluation metric is independent of scaling, and clamping negative to zero, extra scaling inside evaluate becomes irrelevant if we already have positive numbers.

Pseudo code:

def _project_to_simplex(v):
    # ensure sum=1 and v>=0
    v = np.array(v)
    if v.sum() == 1 and np.all(v>=0): return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def propose_candidate(seed=42, budget_s=1000, **kwargs):
   set random seed
   start = time.time()
   deadline = start + budget_s - 1.0  # safety

   # Determine n
   try:
       n = len(height_sequence_1)
   except NameError:
       n = 20000

   # Build integrated arcsine distribution (baseline)
   idx = np.arange(n, dtype=np.float64)
   left = 2*idx / n - 1.0
   right = 2*(idx+1) / n - 1.0
   raw = np.arcsin(right) - np.arcsin(left)   # shape (n,)
   a = raw / np.pi
   a = a / a.sum()  # ensure sum=1
   # ensures nonneg.

   # Evaluate baseline
   baseline_score = 2.0 * n * _conv_max(a)
   print(f"Baseline integrated arcsine score: {baseline_score:.6f}")

   # Optimization parameters (Adam)
   z = _logits_from_distribution(a + 1e-16)
   m = np.zeros_like(z)
   v = np.zeros_like(z)
   beta1 = 0.9
   beta2 = 0.999
   eps_adam = 1e-8
   base_alpha = 0.6
   eps_factor = 0.15
   eps_factor_min = 5e-5
   eps_factor_decay = 0.9996

   best_a = a.copy()
   best_score = baseline_score
   it = 0

   # Phase 1: Adam smoothing, limited time e.g., 60% of budget
   phase1_deadline = start + 0.6 * budget_s

   while time.time() < phase1_deadline:
       it += 1
       a_cur = _softmax(z)
       # enforce symmetry
       a_cur = (a_cur + a_cur[::-1]) * 0.5

       max_conv, grad_a = _conv_max_and_grad(a_cur, eps_factor)
       score = 2.0 * n * max_conv
       if score < best_score:
           best_score = score
           best_a = a_cur.copy()
           print(f"Iter {it}: new best score {best_score:.6f}")

       # compute gradient on logits (mirror descent)
       mu = np.dot(grad_a, a_cur)
       grad_z = a_cur * (grad_a - mu)

       # Adam updates
       m = beta1 * m + (1.0 - beta1) * grad_z
       v = beta2 * v + (1.0 - beta2) * (grad_z**2)
       m_hat = m / (1.0 - beta1**it)
       v_hat = v / (1.0 - beta2**it)
       alpha_t = base_alpha / (math.sqrt(it) + 1.0)
       step = -alpha_t * m_hat / (np.sqrt(v_hat) + eps_adam)
       z = z + step
       # enforce symmetry on logits
       z = (z + z[::-1]) * 0.5
       # clip for numerical stability
       z = np.clip(z, -50.0, 50.0)

       # decay smoothing
       eps_factor = max(eps_factor_min, eps_factor * eps_factor_decay)

       # occasional jitter
       if it % 5000 == 0 and np.random.rand() < 0.2:
           z += 0.001 * np.random.randn(n)

   # Phase 2: subgradient descent on simplex to fine-tune
   # Convert best_a to p for subgradient stage
   p = best_a.copy()
   # Compute step size for subgradient
   eta = 0.1   # initial step size
   eta_min = 1e-8
   eta_decay = 0.9999

   # track start time for phase2
   while time.time() < deadline:
       # compute convolution via FFT for speed
       # use np.convolve p for max and index
       conv = np.convolve(p, p)
       max_conv = conv.max()
       k_max = conv.argmax()
       score = 2.0 * n * max_conv
       if score < best_score:
           best_score = score
           best_a = p.copy()
           print(f"Subgrad iter: new best {best_score:.6f}")

       # compute gradient of max term: g_i = 2 * p_{k_max - i} where valid
       g = np.zeros_like(p)
       i_start = max(0, k_max - (n - 1))
       i_end = min(k_max, n - 1)

       print(f"Final exact score = {final_score:.6f}")

Step‑function optimizer for the max‑convolution bound.

The evaluation metric is  J(a)=2·n·max(a∗a)  (the sum of a is normalised to 1).
The optimum is attained when the self‑convolution is as flat as possible.
The algorithm works in three stages:

1.   A cheap parametric sweep over a two‑parameter family
     a_i(α,δ) ∝ ((i+δ)(n‑1‑i+δ))^{‑α}      (symmetric, α≈½, δ≈½)
     The best parameters give a good initial bound.

2.   Adam optimisation on the probability simplex (logits + softmax)
     with a smoothed soft‑max approximation of the max‑operator.
     Symmetry is enforced after every update.

3.   A short projected sub‑gradient phase that targets the exact
     max‑convolution entry for fine‑grained improvement.

The function returns the best sequence found before the time budget expires.

def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Logit representation that yields `a` after a soft‑max."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """
    Smallest power‑of‑two ≥ 3·n‑2, sufficient for a linear convolution
    of length (2·n‑1) and a correlation of length (3·n‑2).
    """
    return 1 << ((3 * n - 2).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of a∗a (exact linear convolution via FFT)."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a∗a using the same FFT length as _conv_max."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return conv


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Smoothed max of a∗a and its gradient w.r.t. a (mirrored on the simplex).

    Parameters
    ----------
    a : np.ndarray
        Probability vector (sum = 1, non‑negative).
    eps_factor : float
        Multiplicative factor for the smoothing constant:
        eps = eps_factor * max(conv).

    max_conv : float
        Maximum entry of the self‑convolution a∗a.
    grad_a : np.ndarray
        Gradient of 2·n·smooth_max w.r.t. a (already multiplied by 4·n).
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    a_fft = np.fft.rfft(a, L)

    # linear convolution (length 2n‑1)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # smoothed soft‑max weights
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv          # for stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or np.isnan(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # cross‑correlation a ⋆ w (linear, length 3n‑2)
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # gradient (4·n * correlation)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of vector v onto the probability simplex
    {x | x_i ≥ 0, sum_i x_i = 1}.  Returns a new ndarray.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – fall back to uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


def _make_family_distribution(n: int, alpha: float, delta: float) -> np.ndarray:
    """
    Family a_i(alpha,delta) ∝ ((i+δ)(n‑1‑i+δ))^{‑α}.
    Returned vector is already normalised to sum = 1 and symmetric.
    """
    idx = np.arange(n, dtype=np.float64)
    denom = (idx + delta) * (n - 1 - idx + delta)
    # avoid division by zero – with any delta>0 denom>0
    raw = np.power(denom, -alpha)
    a = raw / raw.sum()
    return a
