def _adam_flatten(a, L, n, max_iter, deadline):
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    lr0 = 0.6
    for t in range(max_iter):
        if time.time() >= deadline:
            break
        conv = _conv_fft(a, L)
        # gradient of variance objective
        grad = _grad_variance(a, conv, n, L)
        # small entropy
        grad += 1e-4 * (np.log(a + 1e-20) + 1.0)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (t + 1))

def _subgrad_descent(a, L, n, best_seq, best_score, deadline):
    # momentum
    v = np.zeros_like(a)
    beta = 0.9
    step0 = 0.5
    it = 0
    no_improve_counter = 0
    max_no_improve = 5000
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, L)
        max_val = conv.max()
        # compute subgradient via correlation with indicator
        indicator = (conv == max_val).astype(float)
        grad = _subgrad_corr(a, indicator, L, n)  # returns vector length n
        # momentum
        v = beta * v + (1.0 - beta) * grad
        # step size schedule
        eta = step0 / math.sqrt(it)
        a_new = a - eta * v
        a_new = _project_to_simplex(a_new)
        # compute new objective
        conv_new = _conv_fft(a_new, L)
        cur_obj = 2.0 * n * conv_new.max()
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = a_new.copy()
            print(f"[subgrad] iter {it:,} new best O={best_score:.6f}")
            no_improve_counter = 0
        else:
            no_improve_counter += 1
            # optionally random jitter after many idle steps
            if no_improve_counter >= max_no_improve:
                # restart from best with jitter
                a_new = best_seq.copy()
                a_new += np.random.randn(n) * 0.001
                a_new = np.clip(a_new, 0, None)
                a_new = _project_to_simplex(a_new)
                no_improve_counter = 0
        a = a_new
        # occasional random jitter
        if it % 5000 == 0:
            a += np.random.randn(n) * 5e-4
            a = _project_to_simplex(a)
    return a, best_seq, best_score

"""
Propose a non‑negative step‑function that (approximately) minimises

    O(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

Algorithm
---------
1️⃣  **Warm‑up (variance flattening)** – a short Adam run on the
    variance of the full convolution (∑(c‑μ)²).  This spreads the mass
    and gives a decent starting point for the later phases.

2️⃣  **Main optimisation (sub‑gradient descent on max(conv))** – at each
    iteration we compute the exact sub‑gradient of the `max` term
    (via a correlation with the indicator of the maximal entries) and
    perform a projected momentum step on the simplex.  The step size
    decays as 1/√t.  When progress stalls we jitter the current best
    solution and continue; this phase is the work‑horse that drives the
    bound down.

3️⃣  **Fine‑tuning (smooth surrogate)** – after the sub‑gradient phase we
    switch to a soft‑max (log‑sum‑exp) surrogate with a very large
    temperature (α up to 2·10⁴) and run a few Adam iterations.  This
    cleans up the remaining tiny peaks.

All phases respect the 1000 s time limit and constantly keep the best
sequence seen so far.  The final sequence is returned as a plain list
of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all entries zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution a★a (length 2·n‑1) using FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.shape[0] - 1]


# ----------------------------------------------------------------------
def _grad_variance(a: np.ndarray, conv: np.ndarray, n: int, L: int) -> np.ndarray:
    """
    Gradient of   f(a) = Σ_k (conv_k − μ)²   where μ = 1/(2n−1) (for sum(a)=1).
    Closed form: g = 4·corr(conv−μ , a) .
    """
    mu = 1.0 / (2 * n - 1)
    diff = conv - mu
    a_rev = a[::-1]

    fft_diff = np.fft.rfft(diff, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_diff * fft_a_rev, n=L)

    start = n - 1
    grad = 4.0 * corr_rev[start:start + n]
    return grad


# ----------------------------------------------------------------------
def _subgrad_corr(a: np.ndarray, w: np.ndarray, L: int, n: int) -> np.ndarray:
    """
    Sub‑gradient of max(conv) :  g = 2·corr(w , a) ,
    where w[i] = 1 if conv_i is maximal, else 0.
    """
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a[::-1], n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    grad = 2.0 * corr_rev[n - 1:n - 1 + n]
    return grad


# ----------------------------------------------------------------------
def _adam_flatten(a: np.ndarray, L: int, n: int,
                  max_iter: int, deadline: float) -> np.ndarray:
    """
    A short Adam optimisation of the variance of the convolution.
    Used as a warm‑up to obtain a reasonably “flat’’ start.
    """
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    lr0 = 0.6
    lam_entropy = 1e-4                     # tiny entropy keeps strict positivity

    for it in range(max_iter):
        if time.time() >= deadline:
            break
        conv = _conv_fft(a, L)
        grad = _grad_variance(a, conv, n, L)
        grad += lam_entropy * (np.log(a + 1e-20) + 1.0)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        lr = lr0 / math.sqrt(it + 1)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_to_simplex(a_new)

    return a

    it = 0
    no_improve = 0
    max_no_improve = 6000    # after this we jitter/re‑start

    while time.time() < deadline:
        it += 1

        conv = _conv_fft(a, L)
        cur_max = conv.max()

        # indicator of all maximal entries (tiny tolerance for numerical noise)
        eps_eq = 1e-12
        w = (np.abs(conv - cur_max) < eps_eq).astype(float)

        # sub‑gradient of the max term
        g = _subgrad_corr(a, w, L, n)

        # momentum update
        vel = beta * vel + (1.0 - beta) * g

        # step‑size schedule
        eta = step0 / math.sqrt(it)

        a_new = a - eta * vel
        a_new = _project_to_simplex(a_new)

        # evaluate new objective (fast version, sum(a)=1 => O = 2·n·max)
        conv_new = _conv_fft(a_new, L)
        new_score = 2.0 * n * conv_new.max()

        if new_score < best_score:
            best_score = new_score
            best_seq = a_new.copy()
            no_improve = 0
            print(f"[subgrad] iter {it:,} new best O={best_score:.6f}")
        else:
            no_improve += 1

        # restart / jitter when stuck
        if no_improve >= max_no_improve:
            # jitter the current best and continue
            a_new = best_seq.copy()
            a_new += np.random.randn(n) * 0.0012
            a_new = np.clip(a_new, 0, None)
            a_new = _project_to_simplex(a_new)
            no_improve = 0
            print(f"[subgrad] restart after {it:,} iterations")

        a = a_new

        # occasional tiny random perturbation to keep the dynamics alive
        if it % 5000 == 0:
            a += np.random.randn(n) * 4e-4
            a = _project_to_simplex(a)

    return a, best_seq, best_score


# ----------------------------------------------------------------------
def _smooth_phase(a: np.ndarray, L: int, n: int,
                  best_seq: np.ndarray, best_score: float,
                  deadline: float) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Final polishing using the smooth log‑sum‑exp surrogate.
    The temperature α is annealed from a modest value up to a large one.
    Adam optimisation is used (with a tiny entropy term for stability).
    """
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    lr0 = 0.5
    lam = 2e-2                     # entropy weight
    alpha_start = 5.0
    alpha_end = 2.0e4              # very sharp approximation of max
    max_iter = 120_000

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[smooth] iter {it:,} new best O={best_score:.6f}")

        # anneal α linearly in the log‑scale
        frac = it / max_iter
        alpha = alpha_start * (alpha_end / alpha_start) ** frac

        # soft‑max weights
        scaled = alpha * conv
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        w = exp_scaled / exp_scaled.sum()               # shape (2n‑1,)

        # gradient of smooth term (2·corr(w , a))
        fft_w = np.fft.rfft(w, n=L)
        fft_a_rev = np.fft.rfft(a[::-1], n=L)
        corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
        grad_corr = 2.0 * corr_rev[n - 1:n - 1 + n]

        # tiny entropy term
        grad_ent = lam * (np.log(a + 1e-20) + 1.0)

        grad = grad_corr + grad_ent

        # Adam step
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        lr = lr0 / math.sqrt(it + 1)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_to_simplex(a_new)

    return a, best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    the evaluation function `evaluate_sequence`.  Returns the best sequence
    found before the time budget expires.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.3          # safety margin for the caller

    # ------------------------------------------------------------------
    # Determine problem size and initial point
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
        n = base.size
        a = base / base.sum()
        print(f"[init] using supplied sequence, n={n}")
    except NameError:
        n = 4600
        a = np.full(n, 1.0 / n)
        print(f"[init] using uniform start, n={n}")

    # FFT length – next power of two ≥ 3·n‑2 (covers all convolutions we need)
    L = 1 << ((3 * n - 2).bit_length())

    # Initial objective (fast version, sum(a)=1 ⇒ O = 2·n·max(conv))
    conv = _conv_fft(a, L)
    best_score = 2.0 * n * conv.max()
    best_seq = a.copy()
    print(f"[init] initial O={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 0 – quick Adam variance flattening (≈10 % of the budget)
    phase0_deadline = start + budget_s * 0.10
    a = _adam_flatten(a, L, n, max_iter=80_000, deadline=phase0_deadline)

    # ------------------------------------------------------------------
    # Phase 1 – heavy sub‑gradient descent on max(conv) (≈70 % of budget)
    phase1_deadline = start + budget_s * 0.80
    a, best_seq, best_score = _subgradient_phase(
        a, L, n, best_seq, best_score, phase1_deadline
    )

    # ------------------------------------------------------------------
    # Phase 2 – smooth surrogate polishing (remaining time)
    a, best_seq, best_score = _smooth_phase(
        a, L, n, best_seq, best_score, deadline
    )

    # ------------------------------------------------------------------
    # Final sanity check via the provided evaluate_sequence()
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence failed:", exc)

    return best_seq.tolist()
