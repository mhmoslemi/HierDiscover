"""
Optimise a non‑negative step‑function for the objective

    O(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

using a multi‑stage approach:

1.  Several Adam runs on the soft‑max (log‑it) parametrisation of the simplex.
    The smooth‑max used for the convolution peak is annealed from a very
    soft β≈0.5 up to β≈5·10⁴, then the exact max‑gradient is used for a few
    final steps.  Different support sizes are tried; the longest run gets the
    largest share of the time budget.

2.  A cheap deterministic refinement that repeatedly moves a small amount of
    mass from the index that contributes most to the current maximal
    convolution entry to an index that contributes the least.  Moves are
    accepted if they improve the objective, otherwise they are accepted with a
    tiny Metropolis probability (simulated‑annealing style).

3.  Very small entries are pruned before returning the final list.

The routine respects the overall time budget (default 1000 s) and returns a
plain list of floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution helpers
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return conv_full[: 2 * n - 1]


def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation g_i = Σ_t w[i+t]·a[t] for i=0,…,n‑1.
    Implemented via FFT.
    """
    n = a.shape[0]
    m = w.shape[0]                     # = 2n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()

# Adam optimiser (log‑it parametrisation of the simplex)
def _run_opt(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
    beta_end: float = 50000.0,
    lr_start: float = 0.15,
    lr_end: float = 0.001,
) -> tuple[list[float], float]:
    """
    Adam optimisation on the simplex.
    Returns the best sequence (as list) and its exact objective value.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # initialise a on the simplex
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
    else:
        a = rng.random(n)
    a = np.clip(a, 0.0, None)
    if a.sum() == 0.0:
        a = np.ones(n) / n
    else:
        a = a / a.sum()

    x = np.log(a + 1e-16)          # logits
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    beta_start = 0.5
    best_a = a.copy()
    best_obj = float('inf')
    it = 0

    while time.time() - start < time_limit:
        it += 1
        a = _softmax(x)                     # stay on the simplex

        conv = _conv_self_fft(a)
        M = conv.max()
        obj = float(2.0 * n * M)            # sum(a)=1 ⇒ denominator = 1
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        # schedule (time‑based)
        t = min((time.time() - start) / time_limit, 1.0)
        beta = beta_start * np.exp(np.log(beta_end / beta_start) * t)
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t)

        # smooth‑max weighting
        if beta > 40000.0:              # close to true max, use hard 1‑hot
            kmax = int(np.argmax(conv))
            w = np.zeros_like(conv)
            w[kmax] = 1.0
        else:
            shifted = conv - M                # ≤ 0
            w = np.exp(beta * shifted)
            s = w.sum()
            if s == 0.0:
                w = np.ones_like(conv) / conv.size
            else:
                w /= s

        # gradient of (smooth) max w.r.t. a
        g_a = 2.0 * _correlation_fft(w, a)

        # natural gradient in log‑it space
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape plateaus
        if it % 3000 == 0:
            jitter = rng.normal(scale=0.005, size=n)
            x += jitter

        # restart from current best now and then
        if it % 60000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.003, size=n)
            m.fill(0.0)
            v.fill(0.0)

    # ------------------------------------------------------------------
    # prune negligible entries (helps the final objective)
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a /= best_a.sum()
        best_obj = float(2.0 * best_a.size * _conv_self_fft(best_a).max())
        print(f"[compress] reduced length → {best_a.size}, O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # very short deterministic refinement (mass transfer) if time permits
    remaining = time_limit - (time.time() - start)
    if remaining > 0.5:
        a_loc = best_a.copy()
        obj_loc = best_obj
        n_loc = a_loc.size
        while time.time() - start < time_limit * 0.95:
            # locate the current maximum‑convolution index
            conv_loc = _conv_self_fft(a_loc)
            kmax = int(np.argmax(conv_loc))

            # contributions to that entry
            i_idx = np.arange(n_loc)
            j_idx = kmax - i_idx
            mask = (j_idx >= 0) & (j_idx < n_loc)
            contrib = np.zeros_like(a_loc)
            contrib[mask] = a_loc[i_idx[mask]] * a_loc[j_idx[mask]]

            i = int(np.argmax(contrib))                 # heavy contributor
            # choose a light contributor (bottom 20 % quantile)
            low_thr = np.percentile(contrib, 20)
            low_candidates = np.where(contrib <= low_thr)[0]
            if low_candidates.size == 0:
                j = rng.integers(n_loc - 1)
                if j >= i:
                    j += 1
            else:
                j = int(rng.choice(low_candidates))

            # transfer a modest fraction of mass
            delta = a_loc[i] * rng.random() * 0.03   # ≤ 3 %
            if delta <= 0.0:
                continue
            a_new = a_loc.copy()
            a_new[i] -= delta
            a_new[j] += delta
            a_new = np.clip(a_new, 0.0, None)
            a_new /= a_new.sum()
            obj_new = float(2.0 * n_loc * _conv_self_fft(a_new).max())
            if obj_new < obj_loc - 1e-12:
                a_loc, obj_loc = a_new, obj_new
                if obj_new < best_obj:
                    best_a, best_obj = a_new, obj_new
                    print(f"[local] improvement O={best_obj:.6f}")
        # final compression after local steps
        mask = best_a > thresh
        if mask.sum() < best_a.size:
            best_a = best_a[mask]
            best_a /= best_a.sum()
            best_obj = float(2.0 * best_a.size * _conv_self_fft(best_a).max())
            print(f"[local compress] reduced length → {best_a.size}, O={best_obj:.6f}")

    return best_a.tolist(), best_obj

# Targeted refinement (mass‑transfer + tiny simulated annealing)
def _run_refine(
    init_seq: list[float],
    time_limit: float,
    seed: int = 0,
) -> tuple[list[float], float]:
    """
    Deterministic / annealed refinement that moves mass away from the indices
    that contribute most to the current maximal convolution entry.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    a = np.asarray(init_seq, dtype=float)
    if a.sum() < 1e-12:
        a = np.ones_like(a) / a.size
    else:
        a = a / a.sum()
    n = a.shape[0]

    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())
    best_seq = a.copy()

    # temperature schedule (simple exponential decay)
    T = best_obj * 0.1 + 1e-9
    Tmin = best_obj * 1e-6 + 1e-12
    decay = 0.996

    it = 0
    while time.time() - start < time_limit:
        it += 1
        k_max = int(np.argmax(conv))

        # contributions to the current maximal entry
        i_idx = np.arange(n)
        j_idx = k_max - i_idx
        mask = (j_idx >= 0) & (j_idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i_idx[mask]] * a[j_idx[mask]]

        # pick a heavy and a light index
        i = int(np.argmax(contrib))
        low_thr = np.percentile(contrib, 10)
        low_cand = np.where(contrib <= low_thr)[0]
        if low_cand.size == 0:
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
        else:
            j = int(rng.choice(low_cand))

        # transfer a small chunk of mass
        delta = a[i] * rng.random() * 0.04   # ≤ 4 %
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.clip(a_new, 0.0, None)
        a_new /= a_new.sum()

        conv_new = _conv_self_fft(a_new)
        obj_new = float(2.0 * n * conv_new.max())

        # deterministic improvement or Metropolis acceptance
        if obj_new < best_obj - 1e-13:
            a, conv, best_obj, best_seq = a_new, conv_new, obj_new, a_new.copy()
            print(f"[refine it={it}] improve O={best_obj:.6f}")
        else:
            prob = np.exp(-(obj_new - best_obj) / T) if T > 0 else 0.0
            if rng.random() < prob:
                a, conv = a_new, conv_new

        # temperature decay
        if it % 1500 == 0:
            T = max(T * decay, Tmin)

        # occasional global jitter (helps escape flat regions)
        if it % 8000 == 0:
            noise = rng.normal(scale=0.008, size=n)
            a = np.clip(a + noise, 0.0, None)
            a /= a.sum()
            conv = _conv_self_fft(a)

    # final pruning of tiny entries
    thresh = 1e-12
    mask = best_seq > thresh
    if mask.sum() < best_seq.size:
        best_seq = best_seq[mask]
        best_seq /= best_seq.sum()
        best_obj = float(2.0 * best_seq.size * _conv_self_fft(best_seq).max())
        print(f"[refine compress] reduced length → {best_seq.size}, O={best_obj:.6f}")

    return best_seq.tolist(), best_obj
