def propose_candidate(seed=123, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.1

    # Determine n and base a
    try:
        base = np.array(height_sequence_1, dtype=float)
        n = base.size
        a0 = base / base.sum()
        print("[init] using height_sequence_1, n=", n)
    except NameError:
        n = 4000
        a0 = np.full(n, 1.0/n)
        print("[init] no height_sequence_1, uniform, n=", n)

    L = 1 << ((3*n - 2).bit_length())

    # helpers
    def softmax(z):
        maxz = np.max(z)
        e = np.exp(z - maxz)
        return e / e.sum()

    def compute_conv(a):
        fft_a = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(fft_a * fft_a, n=L)
        return conv_full[:2*n - 1]

    def compute_objective(a):
        conv = compute_conv(a)
        maxb = np.max(conv)
        s = np.sum(a)
        if s < 0.01:
            return np.inf
        return 2.0 * n * maxb / (s*s)

    # gradient of variance objective (flattens conv)
    def grad_variance_z(z):
        a = softmax(z)
        conv = compute_conv(a)
        mu = 1.0 / (2 * n - 1)
        diff = conv - mu
        # correlation diff with a via FFT
        a_rev = a[::-1]
        fft_diff = np.fft.rfft(diff, n=L)
        fft_a_rev = np.fft.rfft(a_rev, n=L)
        corr_rev = np.fft.irfft(fft_diff * fft_a_rev, n=L)
        corr = corr_rev[n-1 : n-1 + n]
        grad_a = 4.0 * corr  # gradient ∂/∂a of variance objective
        # backprop through softmax
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)
        # objective value
        obj = np.sum(diff * diff)
        return obj, grad_z

    # gradient of smooth max surrogate
    def grad_surrogate_z(z, alpha):
        a = softmax(z)
        conv = compute_conv(a)
        # compute surrogate value (2n/alpha) * logsumexp(alpha*conv)
        scaled = alpha * conv
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        sumexp = exp_scaled.sum()
        lse = max_scaled + np.log(sumexp)
        val = (2.0 * n / alpha) * lse
        # weights w_k
        w = exp_scaled / sumexp   # shape (2n-1,)

        # correlation w with a
        a_rev = a[::-1]
        fft_w = np.fft.rfft(w, n=L)
        fft_a_rev = np.fft.rfft(a_rev, n=L)
        corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
        corr = corr_rev[n-1 : n-1 + n]
        grad_a = 4.0 * n * corr    # from derivative
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)
        return val, grad_z, a, conv

    # Keep track of best
    best_a = a0.copy()
    best_obj = compute_objective(best_a)
    print("[init] initial objective O =", best_obj)

    # --- Stage 0: variance flattening with L-BFGS ---
    # We'll do a few restarts
    restarts = 3
    max_iter_var = 100
    for r in range(restarts):
        if time.time() > deadline:
            break
        # initialize z
        if r == 0:
            # start from base a0
            eps = 1e-12
            z = np.log(best_a + eps)      # log-softmax start
        else:
            z = np.random.randn(n) * 0.5   # random start

        # use L-BFGS
        def fun(z):
            val, grad = grad_variance_z(z)
            return val, grad

        res = None
        try:
            from scipy.optimize import minimize
            res = minimize(fun, z, method='L-BFGS-B', jac=True,
                           options={'maxiter': max_iter_var, 'gtol': 1e-12})
        except Exception as e:
            print("[stage0] L-BFGS failed:", e)
            continue

        if res is not None and res.success:
            a = softmax(res.x)
            obj = compute_objective(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
                print(f"[stage0] restart {r} improved O={best_obj:.6f}")

        # time check
        if time.time() > deadline:
            break

    # --- Stage 1: smooth max surrogate annealing ---
    # schedule of alphas
    alphas = [2.0, 5.0, 15.0, 50.0, 200.0, 800.0, 3000.0]
    max_iter_sur = 80  # per alpha stage
    # start from best_a
    z = np.log(best_a + 1e-12)

    for alpha in alphas:
        if time.time() > deadline:
            break
        # define function for this alpha
        def fun_alpha(z):
            val, grad, a, conv = grad_surrogate_z(z, alpha)
            return val, grad

        try:
            res = minimize(fun_alpha, z, method='L-BFGS-B', jac=True,
                           options={'maxiter': max_iter_sur, 'gtol': 1e-12})
        except Exception as e:
            print("[stage1] L-BFGS alpha", alpha, "failed:", e)
            continue

        if res is not None and res.success:
            z = res.x   # continue with this z as init for next alpha
            a = softmax(z)
            obj = compute_objective(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
                print(f"[stage1] alpha {alpha:.1f} new best O={best_obj:.6f}")

    # --- Stage 2: stochastic local search (mass transfer) ---
    # Use best_a as starting point
    a = best_a.copy()
    cur_obj = best_obj
    # hyperparameters
    max_iters_local = 200000  # will be limited by time
    T0 = 0.05
    decay = 0.9995
    T = T0
    it = 0
    while time.time() < deadline and it < max_iters_local:
        # pick i != j
        i = np.random.randint(n)
        j = np.random.randint(n-1)
        if j >= i:
            j += 1
        if a[i] < 1e-15:
            it += 1
            continue
        # transfer up to 10% of mass from i to j
        frac = np.random.rand() * 0.1
        delta = a[i] * frac
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # compute new objective fast
        conv_new = compute_conv(a_new)
        maxb = np.max(conv_new)
        obj_new = 2.0 * n * maxb / (1.0**2)  # sum =1
        # Metropolis acceptance
        if obj_new < cur_obj or np.random.rand() < math.exp((cur_obj - obj_new) / max(T, 1e-12)):
            a = a_new
            cur_obj = obj_new
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a.copy()
                print(f"[local] iter {it:,} new best O={best_obj:.6f}")
        # occasional smoothing
        if it % 5000 == 0 and it > 0:
            kernel = np.array([0.125, 0.25, 0.25, 0.25, 0.125])
            a = np.convolve(a, kernel, mode='same')
            a = a / a.sum()  # renorm
        it += 1
        T *= decay

    # Final sanity check using the provided evaluate_sequence
    try:
        final_val = evaluate_sequence(best_a.tolist())
        print("[final] evaluate_sequence O=", final_val)
    except Exception as e:
        print("[final] evaluate_sequence failed:", e)

    return best_a.tolist()

"""
Propose a non‑negative step function that (approximately) minimises

    O(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

Algorithm Overview
------------------
1. **Re‑parameterisation** – we represent a ∈ Rⁿ as a soft‑max of an
   unconstrained vector `z`.  This guarantees a_i ≥ 0 and Σa_i = 1
   (the scale does not matter for the objective).

2. **Two‑phase continuous optimisation**
   * **Phase 0 – flattening** – minimise the variance of the convolution
     (Σ (c_k‑μ)²).  This drives the whole autocorrelation towards a uniform
     value and typically reduces the peak much faster than optimising the
     hard maximum directly.
   * **Phase 1 – smooth‑max** – minimise a smooth surrogate of the peak:
       (2·n/α)·log∑ e^{α·c_k}.
     The surrogate is gradually sharpened by increasing α (annealing).  
     Both phases use L‑BFGS‑B with an analytic gradient (derived via FFT).

3. **Stochastic local refinement**
   * Random “mass‑transfer” moves (transfer a small fraction of the mass
     from one index to another) are accepted with a Metropolis rule.
   * Every few thousand steps a sub‑gradient step on the current maximum
     entry of the convolution is performed.
   * Occasionally the current sequence is smoothed with a tiny kernel.
   * All updates keep the vector on the probability simplex.

The whole search respects the 1000 s budget; intermediate best values are
printed and the final best sequence is returned as a plain Python list.
"""

import time
import math
import numpy as np

# optional, but SciPy gives us a fast L‑BFGS implementation
try:
    from scipy.optimize import minimize
    _SCIPY_OK = True
except Exception:      # pragma: no cover
    _SCIPY_OK = False


# ----------------------------------------------------------------------
# Basic numerical helpers (all top‑level – no closures)
# ----------------------------------------------------------------------
def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(z)
    e = np.exp(z - m)
    return e / e.sum()


def _conv_fft(a: np.ndarray, n: int, L: int) -> np.ndarray:
    """Full linear convolution a★a (length 2·n‑1) via FFT of length L."""
    fa = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fa * fa, n=L)
    return conv_full[: 2 * n - 1]


def _correlation_fft(w: np.ndarray, a: np.ndarray, n: int, L: int) -> np.ndarray:
    """
    Correlation (w ⋆ a)[i] = Σ_k w_{i+k}·a_k for i=0…n‑1.
    Implemented via FFT using a reversed copy of a.
    """
    a_rev = a[::-1]
    fw = np.fft.rfft(w, n=L)
    fa_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fw * fa_rev, n=L)
    start = n - 1
    return corr_rev[start: start + n]


def _objective(a: np.ndarray, n: int, L: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)²)."""
    conv = _conv_fft(a, n, L)
    maxb = np.max(conv)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    return 2.0 * n * maxb / (s * s)


def _grad_variance(z: np.ndarray, n: int, L: int):
    """
    Gradient of the variance flattening objective
        f(a) = Σ_k (conv_k - μ)²    (μ = 1/(2n‑1))
    with respect to the soft‑max parameters z.
    Returns (objective value, gradient w.r.t. z, current a).
    """
    a = _softmax(z)
    conv = _conv_fft(a, n, L)
    mu = 1.0 / (2 * n - 1)
    diff = conv - mu
    # correlation of (conv‑μ) with a
    corr = _correlation_fft(diff, a, n, L)
    grad_a = 4.0 * corr                     # ∂f/∂a
    dot = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot)             # chain rule through soft‑max
    val = np.sum(diff * diff)
    return val, grad_z, a


def _grad_surrogate(z: np.ndarray, n: int, L: int, alpha: float):
    """
    Gradient of the smooth‑max surrogate
        S(a) = (2·n/α)·log∑_k exp(α·conv_k)
    with respect to the soft‑max parameters z.
    Returns (surrogate value, gradient w.r.t. z, current a, conv).
    """
    a = _softmax(z)
    conv = _conv_fft(a, n, L)

    # soft‑max weights over the convolution entries
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)           # stabilised
    sumexp = np.sum(exp_scaled)
    lse = max_scaled + np.log(sumexp)                  # log‑sum‑exp
    val = (2.0 * n / alpha) * lse

    w = exp_scaled / sumexp                            # probabilities, length 2n‑1
    corr = _correlation_fft(w, a, n, L)
    grad_a = 4.0 * n * corr                             # ∂S/∂a
    dot = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot)                        # chain rule
    return val, grad_z, a, conv


# Callable wrappers required by ``scipy.optimize.minimize`` ---------
class _VarianceObj:
    def __init__(self, n, L):
        self.n = n
        self.L = L

    def __call__(self, z):
        val, grad, _ = _grad_variance(z, self.n, self.L)
        return val, grad


class _SurrogateObj:
    def __init__(self, n, L, alpha):
        self.n = n
        self.L = L
        self.alpha = alpha

    def __call__(self, z):
        val, grad, _, _ = _grad_surrogate(z, self.n, self.L, self.alpha)
        return val, grad

    def _apply_move(conv_vec, a_vec, i, j, delta):
