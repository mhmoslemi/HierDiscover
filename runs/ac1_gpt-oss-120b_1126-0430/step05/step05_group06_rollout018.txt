    # loop attempts
    attempts = 0
    while time.time() < deadline:
        attempts += 1
        # random order of indices
        order = np.random.permutation(n)
        S = []
        sums = set()

        # maximum possible remaining candidates = n - current_index (but we can just continue)
        for idx in order:
            # quick prune: if len(S) + (n - idx) <= best_m: break (can't improve)
            # but idx is index in random order not sorted, ignore.
            # check conflict
            conflict = False
            # compute sums
            s2 = idx + idx
            if s2 in sums:
                conflict = True
            else:
                # loop through existing S
                for s in S:
                    if idx + s in sums:
                        conflict = True
                        break
            if not conflict:
                # add idx to S
                S.append(idx)
                # add new sums
                sums.add(s2)
                for s in S[:-1]:  # all previous
                    sums.add(idx + s)

                    best_S = S.copy()
                    # if we reach theoretical maximum, break
                    if best_m >= int(np.sqrt(2 * n)) + 5:  # near limit
                        # we can break out of outer while loops
                        # We'll break inner iter; also break outer after we check time.
                        pass
        # maybe break if time limited
        if best_m >= int(np.sqrt(2 * n)):
            # can't get larger than sqrt(2n)
            break

    # After search
    print(f"[search] best Sidon set size={best_m} (target >= {required_m}) after {attempts} attempts")

    # Construct sequence
    a = np.zeros(n, dtype=float)
    if best_m == 0:  # fallback uniform

"""
Improved search for a step‑function that minimises

    O(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm follows three complementary ideas:

1️⃣  *Many short random restarts* – draw many Dirichlet vectors with a
    small concentration (very sparse) and perform a few cheap
    “mass‑transfer” hill‑climbs.  The best seed is kept.

2️⃣  *Adam on a smooth log‑sum‑exp surrogate* – minimise a smooth
    approximation of the max‑operator while staying on the probability
    simplex.  The schedule for the smoothness parameter α is
    exponential, ending at a very large value so that the surrogate
    approaches the true max.

3️⃣  *Long simulated‑annealing refinement* – repeatedly move a small
    fraction of probability mass from an index that participates in the
    current maximal convolution entry to a random destination; moves
    are accepted by a Metropolis rule with a slowly cooling temperature.
    Every few hundred steps an exact projected sub‑gradient step is
    performed to keep the walk directed.

All phases respect a 1000 s wall‑time budget, constantly update the
best sequence seen and finally verify it with the provided
``evaluate_sequence`` function.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, ∑x=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # all non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear self‑convolution via FFT (real‑valued)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return a*a (full convolution) using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Exact objective for a simplex‑normalised vector (∑a = 1)
def _objective(a: np.ndarray, n: int, L: int) -> float:
    """2·n·max(conv(a,a))  (a is assumed to sum to 1)."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Gradient of the log‑sum‑exp (smooth max) surrogate
def _smooth_gradient(a: np.ndarray, conv: np.ndarray,
                     alpha: float, L: int) -> np.ndarray:
    """
    ∇_a ( (2·n/α)·log∑_k e^{α·conv_k} ).
    The result reduces to 2·corr(w, a) where w_k ∝ e^{α·conv_k}.
    """
    scaled = alpha * conv
    # subtract the max for numerical stability
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / exp_scaled.sum()          # soft‑max weights, ∑w=1

    # correlation w ⋆ a   (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)

    n_len = a.size
    grad = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]
    return grad

# Adam optimiser on the smooth surrogate
def _adam_phase(a: np.ndarray,
                n: int,
                L: int,
                max_iters: int,
                deadline: float,
                alpha_start: float = 5.0,
                alpha_end: float = 5e4,
                lr0: float = 0.6,
                beta1: float = 0.9,
                beta2: float = 0.999,
                eps: float = 1e-12,
                prune_every: int = 4000,
                prune_percentile: float = 0.5) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Adam on the smoothed surrogate (log‑sum‑exp of the convolution).
    Returns (final a, best objective value seen, best a seen).
    """
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_obj = _objective(a, n, L)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            if it % 2000 == 0:
                print(f"[adam] iter {it:8d}  best O={best_obj:.6f}")

        # exponential schedule for α – sharper towards the end
        frac = it / max_iters
        alpha = alpha_start * (alpha_end / alpha_start) ** frac

        grad = _smooth_gradient(a, conv, alpha, L)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        lr = lr0 / math.sqrt(it + 1)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = _project_to_simplex(a_new)

        # occasional pruning of tiny entries – helps sparsity
        if prune_every and it % prune_every == 0 and it > 0:
            thresh = np.percentile(a_new, prune_percentile)
            a_new[a_new < thresh] = 0.0
            a_new = _project_to_simplex(a_new)

        a = a_new

    return a, best_obj, best_a

# Exact projected sub‑gradient descent (true max)
def _subgradient_phase(a: np.ndarray,
                       n: int,
                       L: int,
                       max_iters: int,
                       deadline: float,
                       eta0: float = 0.25,
                       line_search: bool = True,
                       verbose: bool = False) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Projected sub‑gradient descent on the true objective.
    Returns (final a, best objective, best a).
    """
    best_a = a.copy()
    best_obj = _objective(a, n, L)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            if verbose and it % 3000 == 0:
                print(f"[sub‑grad] iter {it:8d}  best O={best_obj:.6f}")

        # sub‑gradient of the maximal entry (average over all maximisers)
        max_val = conv.max()
        ks = np.flatnonzero(conv == max_val)
        g = np.zeros_like(a)
        for k in ks:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_idx = np.arange(i_start, i_end + 1, dtype=int)
            j_idx = k - i_idx
            # ∂/∂a_i of a_i * a_j + a_j * a_i = 2 a_j
            g[i_idx] += 2.0 * a[j_idx]
        if ks.size:
            g /= ks.size

        # geometric decay of step size
        eta = eta0 * (0.9999 ** it)

        if line_search:
            step = eta
            for _ in range(5):
                cand = _project_to_simplex(a - step * g)
                cand_obj = _objective(cand, n, L)
                if cand_obj < cur_obj:
                    a = cand
                    break
                step *= 0.5
            else:
                a = _project_to_simplex(a - eta * g)
        else:
            a = _project_to_simplex(a - eta * g)

    return a, best_obj, best_a

# Simulated annealing with focussed mass transfers
def _simulated_annealing_phase(a: np.ndarray,
                               n: int,
                               L: int,
                               max_iters: int,
                               deadline: float,
                               temperature0: float = 0.07,
                               no_improve_limit: int = 6000,
                               verbose: bool = True) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Local refinement: repeatedly move a modest fraction of mass from an index
    that participates in the current maximal convolution entry to a random
    destination; accept moves with Metropolis rule.
    Every ninth iteration a true sub‑gradient step tightens the direction.
    """
    best_a = a.copy()
    best_obj = _objective(a, n, L)
    cur_obj = best_obj
    conv = _conv_fft(a, L)

    T = temperature0
    no_improve = 0

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # which convolution entry is maximal?
        kmax = int(np.argmax(conv))

        # indices i that actually contribute to that maximal entry
        i_candidates = np.arange(max(0, kmax - (n - 1)), min(n - 1, kmax) + 1, dtype=int)
        i_candidates = i_candidates[a[i_candidates] > 1e-14]
        if i_candidates.size == 0:
            i = np.random.randint(n)
        else:
            i = np.random.choice(i_candidates)

        # random destination different from i
        j = np.random.randint(n)
        while j == i:
            j = np.random.randint(n)

        # move a modest fraction (≤9 %) of the source mass
        delta = a[i] * np.random.rand() * 0.09
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = _project_to_simplex(a_new)

        conv_new = _conv_fft(a_new, L)
        new_obj = 2.0 * n * conv_new.max()

        # Metropolis acceptance
        if new_obj < cur_obj or np.random.rand() < math.exp((cur_obj - new_obj) / max(T, 1e-12)):
            a = a_new
            conv = conv_new
            cur_obj = new_obj
            no_improve = 0
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                if verbose:
                    print(f"[SA] iter {it:8d}  new global best O={best_obj:.6f}")
        else:
            no_improve += 1

        # occasional exact sub‑gradient step (keeps a guided walk)
        if it % 9 == 0:
            # recompute sub‑gradient for the current maximal entry
            conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max()
            max_val = conv.max()
            ks = np.flatnonzero(conv == max_val)
            g = np.zeros_like(a)
            for k in ks:
                i_start = max(0, k - (n - 1))
                i_end = min(n - 1, k)
                i_idx = np.arange(i_start, i_end + 1, dtype=int)
                j_idx = k - i_idx
                g[i_idx] += 2.0 * a[j_idx]
            if ks.size:
                g /= ks.size
            eta_loc = 0.07 * (1.0 - it / max_iters)
            a = _project_to_simplex(a - eta_loc * g)

            conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                if verbose:
                    print(f"[SA‑sub] iter {it:8d}  improved O={best_obj:.6f}")

        # temperature decay (slow)
        T *= 0.9997

        # restart when stagnating for a while
        if no_improve >= no_improve_limit:
            a = best_a.copy()
            a += np.random.randn(n) * 0.0012          # cheap jitter
            a = np.clip(a, 0, None)
            a = _project_to_simplex(a)
            conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max()
            no_improve = 0

    return a, best_obj, best_a

# Main entry point expected by the evaluation harness
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence that minimises the evaluation function.
    Returns the sequence as a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin – stop a little before the deadline
    deadline = start_time + budget_s - 0.2

    # ------------------------------------------------------------------
    # Problem size
    if 'height_sequence_1' in globals():
        a0 = np.array(height_sequence_1, dtype=float)
        a0 = np.maximum(a0, 0.0)
        if a0.sum() == 0.0:
            a0 = np.full_like(a0, 1.0)
        a0 = a0 / a0.sum()
        n = a0.size
        a = a0.copy()
        print(f"[init] using supplied sequence, n={n}")
    else:
        n = 4600
        a = np.full(n, 1.0 / n)           # uniform start on the simplex
        print(f"[init] using uniform start, n={n}")

    # FFT length – next power‑of‑two greater than full convolution length
    L = 1 << ((3 * n - 2).bit_length())

    # ------------------------------------------------------------------
    # Phase‑0 – cheap random restarts + short hill‑climbs
    phase0_deadline = start_time + budget_s * 0.12
    best_seq = a.copy()
    best_score = _objective(best_seq, n, L)

    restarts = 0
    while time.time() < phase0_deadline:
        restarts += 1
        conc = 0.35                     # Dirichlet concentration – sparsity
        a_rand = np.random.gamma(conc, 1.0, size=n)
        a_rand = a_rand / a_rand.sum()

        # a few fast Adam steps (500 iters)
        sub_dead = time.time() + 3.0
        a_rand, _, best_rand = _adam_phase(
            a_rand, n, L,
            max_iters=800,
            deadline=sub_dead,
            alpha_start=5.0,
            alpha_end=2000.0,
            lr0=0.8)

        cur_obj = _objective(best_rand, n, L)
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = best_rand.copy()
            print(f"[phase0] new best O={best_score:.6f} after {restarts} restarts")

    # ------------------------------------------------------------------
    # Phase‑1 – long Adam optimisation on the smooth surrogate
    phase1_deadline = start_time + budget_s * 0.42
    a, obj_adam, best_adam = _adam_phase(
        best_seq, n, L,
        max_iters=600_000,
        deadline=phase1_deadline,
        alpha_start=5.0,
        alpha_end=80_000.0,
        lr0=0.6)

    if obj_adam < best_score:
        best_score = obj_adam
        best_seq = best_adam.copy()
        print(f"[phase‑1] improved best O={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase‑2 – exact projected sub‑gradient descent (helps crispify)
    phase2_deadline = start_time + budget_s * 0.70
    a, obj_sub, best_sub = _subgradient_phase(
        best_seq, n, L,
        max_iters=350_000,
        deadline=phase2_deadline,
        eta0=0.22,
        line_search=True,
        verbose=False)

    if obj_sub < best_score:
        best_score = obj_sub
        best_seq = best_sub.copy()
        print(f"[phase‑2] improved best O={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase‑3 – intensive simulated‑annealing refinement
    phase3_deadline = deadline
    a, obj_sa, best_sa = _simulated_annealing_phase(
        best_seq, n, L,
        max_iters=500_000,
        deadline=phase3_deadline,
        temperature0=0.09,
        no_improve_limit=8000,
        verbose=True)

    if obj_sa < best_score:
        best_score = obj_sa
        best_seq = best_sa.copy()
        print(f"[phase‑3] final best O={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final verification with the provided evaluator
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence failed:", exc)

    return best_seq.tolist()
