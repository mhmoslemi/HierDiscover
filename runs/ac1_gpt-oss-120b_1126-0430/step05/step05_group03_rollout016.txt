        # For each j, f_{k-j}

"""
Robust step‑function optimiser.

The optimiser works in three stages:

1. **Adam on the simplex** – we run a soft‑max (log‑it) parametrisation of the
   sequence with a smooth approximation of the ``max`` in the objective.
   The schedule for the smoothness (β) and the learning‑rate is aggressive
   and the optimiser is restarted periodically to escape plateaus.

2. **Incremental simulated‑annealing (SA)** – starting from the best Adam
   result we perform many tiny mass‑transfer moves.
   The convolution is recomputed with a fast FFT, the temperature is
   exponentially annealed and moves that increase the objective are accepted
   with the usual Metropolis probability.  This phase is very cheap because
   each move only needs a single FFT.

3. **Final cheap local search** – a few random “mass‑transfer’’ moves similar
   to the original code are performed in the remaining time.

All sequences are kept non‑negative, normalised to sum = 1 (the evaluation
function is scale‑invariant) and finally returned as a plain ``list[float]``.
"""

import time
import math
import numpy as np

# -------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# -------------------------------------------------------------
def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv


# -------------------------------------------------------------
def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation of w (length 2·n‑1) with a (length n).
    Returns a vector of length n.
    """
    n = a.shape[0]
    m = w.shape[0]                 # = 2·n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


# -------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


# -------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))  (sum(a)=1)."""
    n = a.size
    conv = _conv_self_fft(a)
    return float(2.0 * n * conv.max())


# -------------------------------------------------------------
def _run_adam(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the simplex for a given length ``n``.
    Returns the best vector (numpy array, sum≈1) and its exact objective.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # initialise on the simplex
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
        a = np.clip(a, 0.0, None)
    else:
        a = rng.random(n)
    a = a / a.sum()

    # work in log‑it space (automatically respects the simplex)
    x = np.log(a + 1e-16)

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # schedule parameters (may be tuned)
    beta_start = 0.5
    beta_end = 5_000.0          # smoothing ends sharp enough
    lr_start = 0.30
    lr_end = 0.001
    target_iters = 900_000      # iteration budget (stopped by time)

    best_a = a.copy()
    best_obj = _objective(best_a)

    it = 0
    while time.time() - start < time_limit:
        it += 1
        # probabilities from logits
        a = _softmax(x)

        # exact objective
        conv = _conv_self_fft(a)
        M = conv.max()
        obj = 2.0 * n * M
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        # schedule
        t_frac = min(it / target_iters, 1.0)
        beta = beta_start * math.exp(math.log(beta_end / beta_start) * t_frac)
        lr = lr_start * math.exp(math.log(lr_end / lr_start) * t_frac)

        # smooth‑max weight vector
        if beta > 3_000.0:
            # use true arg‑max (sub‑gradient)
            w = np.zeros_like(conv)
            w[int(np.argmax(conv))] = 1.0
        else:
            shifted = conv - M          # ≤ 0
            w = np.exp(beta * shifted)
            w = w / w.sum()

        # gradient w.r.t. a (smooth max)
        g_a = 2.0 * _correlation_fft(w, a)   # (n,)

        # natural gradient in log‑it space (projected onto simplex)
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to leave flat plateaus
        if it % 5000 == 0:
            jitter = rng.normal(scale=0.015, size=n)
            x += jitter

        # periodic restart from the current best
        if it % 60000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.003, size=n)
            m.fill(0.0)
            v.fill(0.0)
            print(f"[n={n}] restart at iter {it}")

    # -----------------------------------------------------------------
    # compress negligible entries
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        n_new = best_a.size
        best_obj = _objective(best_a)
        print(f"[compress] length {n} → {n_new}, O={best_obj:.6f}")

    return best_a, best_obj


# -------------------------------------------------------------
def _simulated_annealing(
    a0: np.ndarray,
    time_limit: float,
    seed: int = 0,
) -> tuple[np.ndarray, float]:
    """
    Simple SA that transfers a small fraction of mass between two random
    positions.  The full convolution is recomputed with an FFT for each
    proposal (fast for the moderate lengths we use).  Returns the best vector
    found and its objective.
    """
    rng = np.random.default_rng(seed)
    n = a0.size
    a = a0.copy()
    conv = _conv_self_fft(a)
    cur_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    best_obj = cur_obj

    start = time.time()
    # temperature schedule
    T_start = max(cur_obj * 0.05, 1e-3)
    T_end = 1e-6

    # maximum fraction of donor mass that may be moved in one step
    max_frac = 0.20

    it = 0
    while time.time() - start < time_limit:
        it += 1
        # linearised temperature
        t_frac = (time.time() - start) / time_limit
        T = T_start * (T_end / T_start) ** t_frac

        # pick two distinct indices
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        if a[i] <= 0.0:
            continue
        delta = a[i] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # recompute convolution via FFT (exact)
        conv_new = _conv_self_fft(a_new)
        obj_new = 2.0 * n * conv_new.max()

        # Metropolis acceptance
        if obj_new < cur_obj or rng.random() < math.exp(-(obj_new - cur_obj) / T):
            a = a_new
            conv = conv_new
            cur_obj = obj_new
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                # occasional progress report
                if it % 5000 == 0:
                    print(f"[SA n={n} it={it}] new best O={best_obj:.6f}")

    return best_a, best_obj

def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **_ignore,
) -> list[float]:
    """
    Entry point used by the evaluation harness.
    Runs Adam optimisation for a handful of lengths, then refines the
    best result with simulated annealing and a short stochastic local search.
    Returns the final sequence (list of non‑negative floats).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # -----------------------------------------------------------------
    # Determine which lengths to try.
    # We start from the length of the previously known best (if any) and
    # then explore a geometric ladder up to a few‑thousand entries.
    lengths = [256, 512, 1024, 2048, 4096, 8192]

    if "height_sequence_1" in globals():
        seq0 = np.asarray(height_sequence_1, dtype=float).copy()
        seq0 = np.clip(seq0, 0.0, None)
        if seq0.sum() < 0.01:
            seq0 = np.ones_like(seq0)
        seq0 = seq0 / seq0.sum()
        n0 = seq0.size
        if n0 not in lengths:
            # insert the previous length near the front
            lengths = [n0] + [l for l in lengths if l != n0]
    else:
        seq0 = None

    best_seq = None
    best_obj = float("inf")

    # -----------------------------------------------------------------
    # Main Adam optimisation loop (most of the budget goes to the first,
    # largest length – that tends to give the lowest objective).
    for idx, n in enumerate(lengths):
        remaining = deadline - time.time()
        if remaining <= 2.0:
            break

        # allocate time: 70 % of the remaining for the first length,
        # the rest split among the others.
        if idx == 0:
            run_time = remaining * 0.70
        else:
            # distribute equally
            run_time = (remaining - 0.5) / (len(lengths) - idx)

        # initialise from previous global solution if possible
        init = None
        if idx == 0 and seq0 is not None:
            if seq0.size != n:
                if seq0.size > n:
                    init = seq0[:n]
                else:
                    pad = np.zeros(n - seq0.size)
                    init = np.concatenate([seq0, pad])
                init = init / init.sum()
            else:
                init = seq0

        seq, val = _run_adam(
            n=n,
            time_limit=run_time,
            init_a=init,
            seed=seed + idx * 19,
        )
        if val < best_obj:
            best_obj = val
            best_seq = seq
            print(f"[global] new overall best O={best_obj:.6f} (n={n})")

    # -----------------------------------------------------------------
    # Simulated annealing refinement on the overall best vector.
    if best_seq is not None:
        remaining = deadline - time.time()
        if remaining > 5.0:
            a = np.array(best_seq, dtype=float)
            a = np.clip(a, 0.0, None)
            a = a / a.sum()
            a, sa_obj = _simulated_annealing(
                a0=a,
                time_limit=remaining * 0.60,
                seed=seed + 999,
            )
            if sa_obj < best_obj:
                best_obj = sa_obj
                best_seq = a.tolist()
                print(f"[SA] finished with O={best_obj:.6f}")

    # -----------------------------------------------------------------
    # Final cheap stochastic “mass‑transfer’’ local search.
    if best_seq is not None:
        remaining = deadline - time.time()
        if remaining > 1.0:
            a = np.array(best_seq, dtype=float)
            n = a.size
            cur_obj = _objective(a)
            rng = np.random.default_rng(seed + 12345)
            end_local = time.time() + remaining * 0.30
            while time.time() < end_local:
                i, j = rng.choice(n, size=2, replace=False)
                if a[i] <= 0.0:
                    continue
                delta = a[i] * rng.random() * 0.02
                a[i] -= delta
                a[j] += delta
                obj = _objective(a)
                if obj < cur_obj:
                    cur_obj = obj
                    best_seq = a.tolist()
                    best_obj = obj
            print(f"[final‑local] O={best_obj:.6f}")

    # -----------------------------------------------------------------
    # Final sanity check via the provided evaluate_sequence routine.
    try:
        final_score = evaluate_sequence(best_seq)
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    # guarantee a non‑empty return
    if not best_seq:
        best_seq = [1.0]

    return best_seq
