"""
Propose a non‑negative step‑function that (approximately) minimises

    O(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

Algorithmic outline
-------------------
1. **Normalisation** – Work on the probability simplex (∑a_i = 1) so that
   the objective reduces to `2·n·max(conv)`.  A small entropy term
   keeps the iterates strictly interior.

2. **Phase 1 – variance flattening** – Gradient descent on the variance
   of the convolution vector (`Σ (c_k–µ)²`).  This drives all entries of
   the autocorrelation towards the same value and gives a good
   starting point for the next phase.

3. **Phase 2 – smooth‑max optimisation** – Replace the hard `max`
   by a log‑sum‑exp surrogate
   `smooth_max_α(c) = (1/α)·log Σ exp(α·c_k)`.  With a large `α` the
   surrogate is arbitrarily close to the true maximum.  An Adam optimiser
   on this smooth objective (augmented by a tiny entropy gradient) is
   run for a long time, with `α` annealed exponentially from 5 up to
   40000.

4. **Phase 3 – simulated‑annealing local refinement** – Repeated
   “mass‑transfer’’ Monte‑Carlo moves are attempted.  A Metropolis
   acceptance rule with a slowly cooling temperature explores the
   neighbourhood of the best solution found so far.  Occasionally a
   sub‑gradient step on the true `max(conv)` is performed to jump out of
   plateaus.  The best sequence discovered during the whole run is
   returned.

All steps respect the security checks of `evaluate_sequence`.
The routine adapts automatically to the available time budget
(≤ 1000 s) and prints intermediate objective values.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex {x≥0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        # all entries non‑positive – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution via FFT (returns length 2·n‑1)
def conv_full(a: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution a★a using FFT of length L (L ≥ 2·n‑1)."""
    fft_a = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft_a * fft_a, n=L)
    return conv[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Objective on the simplex (exact max)
def max_conv_objective(a: np.ndarray, n: int, L: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) when ∑a = 1."""
    conv = conv_full(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Gradient of the variance of the convolution (used in Phase 1)
def grad_variance(a: np.ndarray, conv: np.ndarray, n: int, L: int) -> np.ndarray:
    """
    Gradient of  Σ_k (conv_k – μ)², where μ = 1/(2·n‑1).
    Closed form: 4·corr(conv−μ , a) .
    """
    mu = 1.0 / (2 * n - 1)
    diff = conv - mu                     # length 2n‑1
    a_rev = a[::-1]

    fft_diff = np.fft.rfft(diff, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_diff * fft_a_rev, n=L)

    start = n - 1
    grad = 4.0 * corr_rev[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Smooth (log‑sum‑exp) surrogate and its gradient (Phase 2)
def smooth_max(conv: np.ndarray, alpha: float) -> float:
    """log‑sum‑exp surrogate of max(conv) with temperature 1/α."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    # subtract max for numerical stability
    exp_scaled = np.exp(scaled - max_scaled)
    return (np.log(exp_scaled.sum()) + max_scaled) / alpha


def smooth_gradient(a: np.ndarray,
                    conv: np.ndarray,
                    alpha: float,
                    L: int,
                    lam: float) -> np.ndarray:
    """
    Gradient of   2·n·smooth_max(conv) + lam·∑ a_i·log a_i .
    The first term is 2·corr(w,a) with w ∝ exp(α·conv).
    """
    # soft‑max weights for the surrogate
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / exp_scaled.sum()          # shape (2n‑1,)

    # correlation part (same trick as in grad_variance)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)

    n = a.size
    grad_corr = 2.0 * corr_rev[n - 1:n - 1 + n]   # length n

    # tiny entropy term (helps keep a strictly positive)
    eps = 1e-20
    grad_entropy = lam * (np.log(a + eps) + 1.0)

    return grad_corr + grad_entropy


# ----------------------------------------------------------------------
# Sub‑gradient of the true max (used in Phase 3)
def subgradient_max(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Returns a sub‑gradient g of  max(conv)  w.r.t. a.
    For each index k attaining the maximum,
        g_i = Σ_{j : i+j=k} 2·a_j .
    The result is averaged over all maximal indices.
    """
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        jdx = k - idx
        g[idx] += 2.0 * a[jdx]
    if ks.size > 0:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises the evaluation score.
    Returns the best sequence (as a plain Python list) found before the
    time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # tiny safety margin

    # ------------------------------------------------------------------
    # Determine problem size and starting point
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
        n = base.size
        a = base.copy()
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= a_sum
        print(f"[init] Using supplied sequence, n = {n}")
    except NameError:
        n = 4600                     # a safe default size
        a = np.full(n, 1.0 / n)
        print(f"[init] No supplied sequence – using uniform, n = {n}")

    # FFT length – a power of two ≥ 3·n‑2 (covers all convolutions we need)
    L = 1 << ((3 * n - 2).bit_length())

    # ------------------------------------------------------------------
    # Phase 1 – variance flattening (fast Adam on Σ (c‑µ)²)
    best_seq = a.copy()
    best_score = max_conv_objective(best_seq, n, L)
    print(f"[phase‑1] start O = {best_score:.6f}")

    # Adam parameters for Phase 1
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    lr0 = 0.7
    lam_entropy = 1e-4
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    phase1_deadline = start_time + budget_s * 0.20   # 20 % of total budget
    it = 0
    while time.time() < phase1_deadline:
        conv = conv_full(a, L)

        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[phase‑1] iter {it:,} new best O = {best_score:.6f}")

        # gradient of variance
        grad = grad_variance(a, conv, n, L)
        # tiny entropy term for interior feasibility
        grad += lam_entropy * (np.log(a + 1e-20) + 1.0)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        lr = lr0 / math.sqrt(it + 1)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = project_to_simplex(a_new)

        it += 1
        if it % 25000 == 0:
            print(f"[phase‑1] iter {it:,} O = {cur_obj:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – smooth‑max optimisation (log‑sum‑exp surrogate)
    print(f"[phase‑2] start, best O = {best_score:.6f}")

    # Adam parameters for Phase 2
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    lr0 = 0.5
    lam = 2e-4               # entropy weight (decays later)
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    alpha_start = 5.0
    alpha_end = 40000.0      # fairly large to approximate true max

    phase2_deadline = start_time + budget_s * 0.75   # up to 75 % of total budget
    it = 0
    max_iter_phase2 = 800_000
    while time.time() < phase2_deadline and it < max_iter_phase2:
        conv = conv_full(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[phase‑2] iter {it:,} new best O = {best_score:.6f}")

        # exponential annealing of α
        frac = it / max_iter_phase2
        alpha = alpha_start * (alpha_end / alpha_start) ** frac

        # gradient of smooth surrogate + tiny entropy
        grad = smooth_gradient(a, conv, alpha, L, lam)

        # Adam update (same as before)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        lr = lr0 / math.sqrt(it + 1)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to shake out of plateaus
        if np.random.rand() < 0.001:
            a_new += np.random.randn(n) * 1e-4

        a = project_to_simplex(a_new)
        it += 1

        if it % 50000 == 0:
            print(f"[phase‑2] iter {it:,} O = {cur_obj:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated‑annealing local refinement
    print(f"[phase‑3] start, best O = {best_score:.6f}")

    temperature0 = 0.06                # initial temperature for Metropolis
    min_temperature = 5e-5
    cooling_rate = 0.9999
    max_local_iters = 300_000
    no_improve_limit = 5000           # early‑restart trigger

    it_total = 0
    while time.time() < deadline:
        # ----- initialise a fresh candidate (jittered best or random) -----
        if np.random.rand() < 0.7:
            a = best_seq.copy()
            a += np.random.randn(n) * 0.0012
            a = np.clip(a, 0.0, None)
            a = project_to_simplex(a)
        else:
            a = np.random.rand(n)
            a = project_to_simplex(a)

        conv = conv_full(a, L)
        cur_obj = 2.0 * n * conv.max()

        it_loc = 0
        no_improve = 0
        T = temperature0

        while time.time() < deadline and it_loc < max_local_iters:
            # ---- random mass‑transfer move ---------------------------------
            i, j = np.random.randint(n, size=2)
            if i != j and a[i] > 1e-14:
                # move a random fraction (up to 15 %) of the source mass
                delta = a[i] * np.random.rand() * 0.15
                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta
                a_new = project_to_simplex(a_new)

                conv_new = conv_full(a_new, L)
                new_obj = 2.0 * n * conv_new.max()

                if new_obj < cur_obj or np.random.rand() < math.exp((cur_obj - new_obj) / max(T, 1e-12)):
                    a, conv, cur_obj = a_new, conv_new, new_obj
                    no_improve = 0
                    if cur_obj < best_score:
                        best_score = cur_obj
                        best_seq = a.copy()
                        now = time.time() - start_time
                        print(f"[{now:.1f}s] new global best O={best_score:.6f} (iter={it_loc})")
                else:
                    no_improve += 1
            else:
                no_improve += 1

            # ---- sub‑gradient step on the exact max (periodic) ------------
            if it_loc % 7 == 0:
                g = subgradient_max(a, conv)
                eta = 0.06 * (1.0 - it_loc / max_local_iters)
                a = a - eta * g
                a = project_to_simplex(a)
                conv = conv_full(a, L)
                cur_obj = 2.0 * n * conv.max()
                if cur_obj < best_score:
                    best_score = cur_obj
                    best_seq = a.copy()
                    now = time.time() - start_time
                    print(f"[{now:.1f}s] sub‑grad improved O={best_score:.6f}")

            # ---- occasional smoothing of the whole vector -----------------
            if it_loc % 4000 == 0 and it_loc > 0:
                kernel = np.array([0.125, 0.25, 0.25, 0.25, 0.125])
                a_smooth = np.convolve(a, kernel, mode="same")
                a = project_to_simplex(0.85 * a + 0.15 * a_smooth)

            # decay temperature
            T = max(min_temperature, T * cooling_rate)
            it_loc += 1
            it_total += 1

            # restart if stuck for too long
            if no_improve >= no_improve_limit:
                break

    # ------------------------------------------------------------------
    # Final sanity check using the provided evaluate_sequence()
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O = {final_score:.6f}")
    except Exception as e:
        print("[final] evaluate_sequence failed:", e)

    return best_seq.tolist()
