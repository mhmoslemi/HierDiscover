           best_obj = cur_obj
           best_a = a.copy()
           if it%1000==0: print(...)

       frac = it / max_iters
       alpha = alpha_start * ((alpha_end / alpha_start) ** frac)

       # step size schedule
       eta = eta0 / np.sqrt(it+1)

       sum_a = a.sum()
       if sum_a == 0:

           a[a < thresh] = 0.0

           else: a = a / sum_a

"""
Proposes a non‑negative step‑function sequence that minimizes

    O(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm consists of several stages:

1️⃣  **Initial seeding** – start from the supplied sequence (if any),
    from a uniform vector and from a handful of handcrafted
    parametric families (U‑shapes, triangular, sinusoidal, power‑law).
    The best among these seeds is kept.

2️⃣  **Random Dirichlet restarts** – draw many vectors from Dirichlet
    distributions with a variety of concentration parameters,
    perform a few dozen Adam steps on a smooth log‑sum‑exp surrogate,
    and keep the best result.

3️⃣  **Long Adam optimisation** – minimise the smooth surrogate with a
    schedule that slowly sharpens the approximation of the max‑operator.
    The iterates are projected onto the probability simplex.

4️⃣  **Mirror‑descent (exponentiated‑gradient) refinement** – another
    gradient‑based pass on the same surrogate, but using a KL‑mirror map
    (multiplicative updates) which often yields finer control on the
    simplex.

5️⃣  **Exact sub‑gradient descent with Polyak step‑size** – descent on the
    true non‑smooth objective, using a step size that automatically
    adapts to the current optimality gap.

6️⃣  **Targeted simulated annealing** – local stochastic moves that
    transfer a small fraction of mass from indices participating in the
    current maximal convolution entry to random destinations.
    The Metropolis rule with a slowly cooling temperature finishes the
    search.

All phases constantly update the global best sequence and stop well
before the 1000 s wall‑time limit.  The final sequence is returned as a
plain Python list; the evaluator normalises it internally, so the
sequence is kept on the simplex (sum = 1) throughout the search.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
#  Utility functions (unchanged from the reference implementation)
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # all non‑positive – return uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full self‑convolution using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


def _objective(a: np.ndarray, n: int, L: int) -> float:
    """Exact objective for a simplex‑normalised vector (∑a = 1)."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. a (ignores constant 2·n factor)."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


def _smooth_gradient(a: np.ndarray, conv: np.ndarray,
                     alpha: float, L: int) -> np.ndarray:
    """
    Gradient of the smooth (log‑sum‑exp) surrogate.
    Returns ∇_a ( (2·n/α)·log∑_k e^{α·conv_k} ).
    """
    # soft‑max over convolution entries
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)      # stabilise
    w = exp_scaled / exp_scaled.sum()

    # correlation w ⋆ a   (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n_len = a.size
    grad = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]
    return grad


# ----------------------------------------------------------------------
#  Optimisation phases
# ----------------------------------------------------------------------
def _adam_phase(a: np.ndarray, n: int, L: int,
                max_iters: int, deadline: float,
                alpha_start: float = 5.0,
                alpha_end: float = 5e4,
                lr0: float = 0.6,
                beta1: float = 0.9,
                beta2: float = 0.999,
                eps: float = 1e-12,
                prune_every: int = 5000,
                prune_percentile: float = 1.0,
                verbose: bool = False) -> tuple[np.ndarray, float, np.ndarray]:
    """Adam optimiser on the smooth log‑sum‑exp surrogate."""
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_obj = _objective(a, n, L)

    total_iters = max_iters
    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            if verbose:
                print(f"[adam] iter {it} new best O={best_obj:.6f}")

        # schedule for the smoothness parameter α
        frac = it / total_iters
        alpha = alpha_start * ((alpha_end / alpha_start) ** frac)

        grad = _smooth_gradient(a, conv, alpha, L)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        lr = lr0 / math.sqrt(it + 1)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # tiny jitter to help escape plateaus
        if np.random.rand() < 0.001:
            a_new += np.random.randn(n) * 1e-4

        a_new = _project_to_simplex(a_new)

        # occasional pruning of extremely small entries
        if prune_every and it % prune_every == 0 and it > 0:
            thresh = np.percentile(a_new, prune_percentile)
            a_new[a_new < thresh] = 0.0
            a_new = _project_to_simplex(a_new)

        a = a_new

    return a, best_obj, best_a


def _mirror_descent_phase(a: np.ndarray, n: int, L: int,
                          max_iters: int, deadline: float,
                          eta0: float = 0.02,
                          alpha_start: float = 5.0,
                          alpha_end: float = 1e6,
                          prune_every: int = 3000,
                          prune_percentile: float = 0.5,
                          verbose: bool = False) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Mirror‑descent (exponentiated‑gradient) on the smooth surrogate.
    Updates are performed multiplicatively and renormalised to the simplex.
    """
    best_a = a.copy()
    best_obj = _objective(a, n, L)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            if verbose:
                print(f"[mirror] iter {it} new best O={best_obj:.6f}")

        # smoothness schedule – start easy, end sharp
        frac = it / max_iters
        alpha = alpha_start * ((alpha_end / alpha_start) ** frac)

        grad = _smooth_gradient(a, conv, alpha, L)

        eta = eta0 / math.sqrt(it + 1)          # decreasing step size

        # multiplicative update (KL‑mirror)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        # pruning to keep sparsity manageable
        if prune_every and it % prune_every == 0 and it > 0:
            thresh = np.percentile(a, prune_percentile)
            a[a < thresh] = 0.0
            s = a.sum()
            if s == 0.0:
                a = np.full(n, 1.0 / n)
            else:
                a /= s

    return a, best_obj, best_a


def _subgradient_polyak_phase(a: np.ndarray, n: int, L: int,
                              max_iters: int, deadline: float,
                              eta_max: float = 0.4,
                              verbose: bool = False) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Projected sub‑gradient descent on the exact objective.
    Uses a Polyak step size (gap / ‖g‖²) capped by eta_max.
    """
    best_a = a.copy()
    best_obj = _objective(a, n, L)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            if verbose and it % 1000 == 0:
                print(f"[sub‑grad] iter {it} new best O={best_obj:.6f}")

        g = _subgradient_max_conv(a, conv)
        norm_sq = np.dot(g, g)
        if norm_sq == 0.0:
            break

        # Polyak step – guaranteed descent if the optimum is known.
        # Here we use the best known objective as a proxy.
        step = (cur_obj - best_obj) / (norm_sq + 1e-12)
        # step may become negative or tiny; clamp to a safe interval
        step = min(max(step, 0.0), eta_max)

        a = _project_to_simplex(a - step * g)

    return a, best_obj, best_a


def _simulated_annealing_phase(a: np.ndarray, n: int, L: int,
                               max_iters: int, deadline: float,
                               temperature0: float = 0.08,
                               no_improve_limit: int = 8000,
                               verbose: bool = True) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Local refinement by stochastic mass‑transfer moves.
    Moves are accepted if they lower the exact objective or with a
    Metropolis probability that cools with temperature T.
    """
    best_a = a.copy()
    best_obj = _objective(a, n, L)
    cur_obj = best_obj
    conv = _conv_fft(a, L)

    T = temperature0
    no_improve = 0

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # index (or indices) that achieve the current maximal convolution entry
        kmax = int(np.argmax(conv))
        i_candidates = np.arange(max(0, kmax - (n - 1)), min(n - 1, kmax) + 1)
        i_candidates = i_candidates[a[i_candidates] > 1e-14]
        if i_candidates.size == 0:
            i = np.random.randint(n)
        else:
            i = np.random.choice(i_candidates)

        # destination index (different from source)
        j = np.random.randint(n)
        while j == i:
            j = np.random.randint(n)

        # transfer a modest fraction of the source mass
        delta = a[i] * np.random.rand() * 0.12   # up to 12 % of source
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = _project_to_simplex(a_new)

        conv_new = _conv_fft(a_new, L)
        new_obj = 2.0 * n * conv_new.max()

        # Metropolis acceptance
        if new_obj < cur_obj or np.random.rand() < math.exp((cur_obj - new_obj) / max(T, 1e-12)):
            a = a_new
            conv = conv_new
            cur_obj = new_obj
            no_improve = 0
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                if verbose:
                    print(f"[SA] iter {it} new global best O={best_obj:.6f}")
        else:
            no_improve += 1

        # occasional exact sub‑gradient step to keep a directed
        if it % 7 == 0:
            g = _subgradient_max_conv(a, conv)
            eta_loc = 0.06 * (1.0 - it / max_iters)
            a = _project_to_simplex(a - eta_loc * g)
            conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                if verbose:
                    print(f"[SA‑sub] iter {it} improved O={best_obj:.6f}")

        # temperature decay (slow exponential)
        T *= 0.9997

        # restart from best if we stagnate too long
        if no_improve >= no_improve_limit:
            a = best_a.copy()
            a += np.random.randn(n) * 0.0017
            a = np.clip(a, 0.0, None)
            a = _project_to_simplex(a)
            conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max()
            no_improve = 0

    return a, best_obj, best_a

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _arcsine_like(n: int) -> np.ndarray:
    """Discrete approximation of a symmetric 1/√(x(1‑x)) shape."""
    i = np.arange(n, dtype=float)
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    a /= a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    """Symmetric triangular shape (peak in the middle)."""
    i = np.arange(n, dtype=float)
    mid = (n - 1) / 2.0
    a = (mid - np.abs(i - mid) + 1.0)   # linear increase then decrease
    a /= a.sum()
    return a


def _sinusoidal(n: int) -> np.ndarray:
    """Positive sinusoid (zero at the ends, max in centre)."""
    i = np.arange(n, dtype=float)
    a = np.sin(np.pi * (i + 0.5) / n)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _u_shape(n: int, power: float = 0.7) -> np.ndarray:
    """U‑shaped family a_i ∝ (i+1)^{-p} + (n‑i)^{-p}."""
    i = np.arange(n, dtype=float)
    a = (i + 1) ** (-power) + (n - i) ** (-power)
    a /= a.sum()
    return a


def _bell_shape(n: int, power: float = 1.2) -> np.ndarray:
    """Bell‑shaped family a_i ∝ ((i+1)*(n‑i))^{p}."""
    i = np.arange(n, dtype=float)
    a = ((i + 1) * (n - i)) ** power
    a /= a.sum()
    return a
