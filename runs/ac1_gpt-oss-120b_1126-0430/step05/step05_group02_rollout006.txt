#!/usr/bin/env python
"""
A fast multi‑stage search for a non‑negative step–function that minimises

        2·n·max_k (a*a)[k] / (Σ a_i)²

The algorithm consists of

1.  A broad parametric sweep (power‑law, exponential, beta, sine‑power,
    Gaussian, classic window functions, random Dirichlet seeds) to obtain a
    good deterministic start.
2.  An exact projected exponentiated‑gradient (mirror‑descent) refinement
    that uses the *true* sub‑gradient of the max‑autocorrelation (rather
    than a top‑k approximation).  The update is multiplicative, followed by
    a simplex projection.
3.  Simple restarts with a tiny random perturbation when progress stalls.
4.  A final call to the supplied ``evaluate_sequence`` for a guaranteed
    exact score.

All work respects the supplied time‑budget (default 1000 s).  The routine
always returns a feasible sequence (non‑negative, finite, and ≤ 1000 after
clipping).  The code contains only top‑level functions – no closures or
lambdas – and uses only the standard library plus NumPy.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex Σx=1."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_len_for_n(n: int) -> int:
    """Next power of two ≥ 2·n‑1 (sufficient for a full convolution)."""
    return 1 << ((2 * n - 1).bit_length())


def _conv_max(a: np.ndarray, fft_len: int) -> tuple[float, np.ndarray]:
    """
    Return (max convolution value, full convolution array) of a with itself.
    a must be a 1‑D real array, sum(a) is assumed to be 1.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.size - 1]
    return float(conv.max()), conv


def _cheap_score(a: np.ndarray, fft_len: int) -> float:
    """Assuming Σa = 1, the evaluation reduces to 2·n·max(conv)."""
    n = a.shape[0]
    maxb, _ = _conv_max(a, fft_len)
    return 2.0 * n * maxb


def _gradient_max_autocorr(a: np.ndarray, fft_len: int) -> tuple[np.ndarray, float]:
    """
    Compute the exact sub‑gradient of the objective f(a)=max_k (a*a)[k]
    (i.e. the gradient of the active convolution entries) and the current
    maximal autocorrelation value.
    Returns (g, maxb) where g has the same shape as a.
    """
    n = a.shape[0]
    maxb, conv = _conv_max(a, fft_len)

    # Indices (in the convolution array) that attain the maximum,
    # using a tiny tolerance to catch ties caused by symmetry.
    tol = maxb * 1e-12 + 1e-15
    active_idx = np.where(conv >= maxb - tol)[0]

    g = np.zeros_like(a)
    center = n - 1   # shift zero corresponds to this index

    for k in active_idx:
        offset = abs(k - center)   # absolute shift distance
        if offset == 0:
            # derivative of Σ a_i^2 is 2·a_i
            g += 2.0 * a
        else:
            end = n - offset
            # contributions from pairs (i, i+offset)
            g[:end] += 2.0 * a[offset:]      # part that sees a_{i+offset}
            g[offset:] += 2.0 * a[:end]      # part that sees a_i
    # Average over the active set (convex combination)
    g /= float(active_idx.size)
    return g, maxb


# ----------------------------------------------------------------------
# Families of candidate sequences (all return a normalized array Σ=1)
# ----------------------------------------------------------------------


def _make_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _make_power_law(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = i ** (-p)
    seq /= seq.sum()
    return seq


def _make_exponential(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    seq = np.exp(-lam * i)
    seq /= seq.sum()
    return seq


def _make_power_exp(n: int, p: float, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 1.0
    seq = (i ** (-p)) * np.exp(-lam * (i - 1.0))
    seq /= seq.sum()
    return seq


def _make_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(1, n + 1, dtype=np.float64) / n)
    seq = (x ** alpha) * ((1.0 - x) ** beta)
    total = seq.sum()
    if total == 0.0:
        return None
    seq /= total
    return seq


def _make_sine_power(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    seq = np.sin(np.pi * i / n) ** gamma
    seq /= seq.sum()
    return seq


def _make_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    seq = np.exp(-0.5 * ((i - mid) / sigma) ** 2)
    seq /= seq.sum()
    return seq


def _make_triangular(n: int) -> np.ndarray:
    half = (n + 1) // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    right = left[::-1] if n % 2 == 0 else left[:-1][::-1]
    seq = np.concatenate([left, right])
    seq /= seq.sum()
    return seq


def _make_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    seq = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
    seq /= seq.sum()
    return seq


def _make_hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    seq = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    seq /= seq.sum()
    return seq


def _make_blackman(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    seq = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    seq /= seq.sum()
    return seq


def _make_kaiser(n: int, beta: float) -> np.ndarray:
    seq = np.kaiser(n, beta)
    seq /= seq.sum()
    return seq

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine adapts its schedule to the available
    time budget and always returns a feasible sequence.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a safety margin for the final exact evaluation
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # Choose a working length n (5 k ≤ n ≤ 25 k)
    # --------------------------------------------------------------
    try:

        prev = height_sequence_1               # noqa: F821
        prev_arr = np.asarray(prev, dtype=np.float64)
        if prev_arr.sum() > 0.0:
            prev_arr = prev_arr / prev_arr.sum()
        else:
            prev_arr = None
        n = max(len(prev_arr), 5000) if prev_arr is not None else 5000
    except Exception:
        prev_arr = None
        n = 5000

    n = min(max(n, 5000), 25000)   # clamp to a reasonable range
    fft_len = _fft_len_for_n(n)

    # --------------------------------------------------------------
    # Cheap score (assumes Σa_i = 1): 2·n·max(conv)
    # --------------------------------------------------------------
    def cheap_score(vec: np.ndarray) -> float:
        return _cheap_score(vec, fft_len)

    for p in np.linspace(0.3, 2.0, 35):

    for lam in np.linspace(0.0, 5.0, 35):
        if time.time() >= deadline:
            break
        cand = _make_exponential(n, lam)
        val = cheap_score(cand)
        if val < best_val:
            best_a, best_val = cand.copy(), val

        for lam in np.linspace(0.0, 3.0, 9):
            if time.time() >= deadline:
                break
            cand = _make_power_exp(n, p, lam)
            val = cheap_score(cand)
            if val < best_val:
                best_a, best_val = cand.copy(), val

    # Beta‑shaped families
    for alpha in np.linspace(0.2, 5.0, 13):
        for beta in np.linspace(0.2, 5.0, 13):
            if time.time() >= deadline:
                break
            cand = _make_beta(n, alpha, beta)
            if cand is None:
                continue
            val = cheap_score(cand)
            if val < best_val:
                best_a, best_val = cand.copy(), val

    # Classic window functions (Hann, Hamming, Blackman, Kaiser)
    for win_func in (_make_hann, _make_hamming, _make_blackman):
        if time.time() >= deadline:
            break
        cand = win_func(n)
        val = cheap_score(cand)
        if val < best_val:
            best_a, best_val = cand.copy(), val

    for beta_param in np.linspace(0.0, 10.0, 11):
        if time.time() >= deadline:
            break
        cand = _make_kaiser(n, beta_param)
        val = cheap_score(cand)
        if val < best_val:
            best_a, best_val = cand.copy(), val

    # Sine‑power family
    for gamma in np.linspace(0.5, 4.0, 15):

        sigma = sigma_frac * n
        cand = _make_gaussian(n, sigma)
        val = cheap_score(cand)

    print(f"[Init] best cheap score after sweep = {best_val:.6f}")

    # --------------------------------------------------------------
    # 1. Refine the best candidate with exact exponentiated‑gradient
    # --------------------------------------------------------------
    a = best_a.copy()
    best_iter_a = a.copy()
    best_iter_val = best_val

    # hyper‑parameters
    eta0 = 0.8               # initial step‑size factor

        maxb, conv = _conv_max(a, fft_len)

            best_iter_a = a.copy()

        # --- compute exact sub‑gradient of the max‑autocorrelation ---
        # (average over all active shifts)
        g, _ = _gradient_max_autocorr(a, fft_len)

        # --- multiplicative (exponentiated‑gradient) update ---
        eta = eta0 / math.sqrt(iter_cnt)
        log_a = np.log(a + eps) - eta * g
        a = np.exp(log_a)

        # renormalise onto the simplex
        s = a.sum()
        if s == 0.0 or not np.isfinite(s):
            # fall back to the best known iterate
            a = best_iter_a.copy()
            continue
        a /= s

        # --- periodic restart when stalled ---
        if iter_cnt - last_improve_iter > stagnation_limit:
            # small random perturbation around the best iterate seen so far
            perturb = 0.005 * np.random.rand(n)
            a = _simplex_projection(best_iter_a + perturb)
            last_improve_iter = iter_cnt
            print(f"[Iter {iter_cnt:07d}] Restart after stagnation.")

        # occasional lightweight random kicks to escape shallow plateaus
        if iter_cnt % 20000 == 0 and np.random.rand() < 0.15:
            kick = 0.01 * (np.random.rand(n) - 0.5)
            a = _simplex_projection(a + kick)

    # --------------------------------------------------------------
    # 2. Final exact evaluation (uses the provided ``evaluate_sequence``)
    # --------------------------------------------------------------
    final_seq = best_iter_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)   # noqa: F821
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:      # defensive – should never happen
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # --------------------------------------------------------------
    # 3. Return the best sequence found.
    # --------------------------------------------------------------
    return final_seq
