"""
Discrete arcsine distribution search.
We generate candidate step functions by sampling the continuous arcsine
density f(x)=1/(π√(x(1−x))) on a uniform grid of size n.
The convolution f*f is constant in the continuous limit, so a finer grid
produces a self‑convolution that is almost uniform.  The score
2·n·max(conv)/sum² therefore tends to 1 as n grows.
The implementation tries a small list of grid sizes (up to a few
hundred thousand points), evaluates an inexpensive FFT‑based upper bound
and keeps the best vector found before the time budget expires.
"""
import time
import math
import numpy as np

def _conv_max_fft(a: np.ndarray) -> float:
    """Return the maximum entry of the linear self‑convolution a*a
    using an FFT of minimal power‑of‑two length."""
    n = a.shape[0]
    L = 1 << ((3 * n - 2).bit_length())
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())

    FFT‑based upper bound on the evaluation score.
    """

    start = time.time()
    deadline = start + budget_s - 0.5   # small safety margin

    #    We limit the maximum length to keep the per‑element value below
    #    the 1000‑clamp enforced by evaluate_sequence.

    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that minimises the evaluation metric.

    The function builds several discrete arcsine vectors, evaluates a fast
    FFT‑based upper bound for each, keeps the best one and finally returns it.
    """
    np.random.seed(seed)
    start = time.time()
    # keep a small safety margin for logging / final evaluation
    deadline = start + budget_s - 0.5

    # ----------------------------------------------------------------------
    # 0. Candidate grid sizes.
    #    We stay well below the value where the arcsine singularity would be
    #    clipped to 1000 (that happens only beyond ~5·10⁶ points).
    # ----------------------------------------------------------------------
    candidate_ns = [
        2_000, 5_000, 10_000, 20_000,
        50_000, 100_000, 200_000, 500_000
    ]

    best_score = float("inf")
    best_vec = None
    best_n = None

    for n in candidate_ns:
        if time.time() >= deadline:
            break

        # --------------------------------------------------------------
        # 1. Sample the arcsine density on the mid‑point rule.
        # --------------------------------------------------------------
        idx = np.arange(n, dtype=np.float64)
        x = (idx + 0.5) / n                # points in (0,1)
        raw = 1.0 / np.sqrt(x * (1.0 - x)) # 1/√(x(1‑x))

        # Normalise to a probability vector (sum = 1).  The absolute scale
        # does not influence the final score because the numerator and
        # denominator of the metric both scale with the square of the factor.
        a = raw / raw.sum()

        # --------------------------------------------------------------
        # 2. Rapid score estimate using the FFT routine.
        # --------------------------------------------------------------
        max_conv = _conv_max_fft(a)
        approx_score = 2.0 * n * max_conv
        print(f"candidate n={n:7d}, approx score={approx_score:.6f}")

        if approx_score < best_score:
            best_score = approx_score
            best_vec = a.copy()
            best_n = n

        # --------------------------------------------------------------
        # 3. (Optional) small random jitter to escape pathological ties.
        #    For the arcsine construction this is unnecessary, so we skip it.
        # --------------------------------------------------------------

    # ----------------------------------------------------------------------
    # 4. Fallback in the unlikely event that no candidate was generated.
    # ----------------------------------------------------------------------
    if best_vec is None:
        n = 20_000
        best_vec = np.ones(n, dtype=np.float64) / n
        best_n = n

    # ----------------------------------------------------------------------
    # 5. Logging of the exact score (uses the provided evaluate_sequence).
    # ----------------------------------------------------------------------
    try:
        exact = evaluate_sequence(best_vec.tolist())
        print(f"final selected n={best_n}, exact score={exact:.6f}")
    except Exception as exc:
        # Should never happen – we just log and continue.
        print(f"error during final evaluation: {exc}")

    # ----------------------------------------------------------------------
    # 6. Return the sequence as a plain Python list.
    # ----------------------------------------------------------------------
    return best_vec.tolist()
