def _run_mirror_descent(n, time_limit, init_a=None, seed=0):
    start = time.time()
    rng = np.random.default_rng(seed)
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        a = np.clip(a, 0, None)
        if np.sum(a) < 1e-12:
            a = np.ones(n) / n
        else:
            a = a / a.sum()
    else:
        a = rng.random(n)
        a = a / a.sum()
    # parameters
    eta0 = 0.5  # initial learning rate
    eta_min = 1e-4
    decay = 0.9995  # each iteration multiply by decay
    eta = eta0
    best_a = a.copy()
    conv = _conv_self_fft(a)
    max_conv = conv.max()
    best_obj = float(2 * n * max_conv)
    it = 0
    while time.time() - start < time_limit:
        it += 1
        # compute conv and max index
        conv = _conv_self_fft(a)
        max_val = conv.max()
        obj = 2.0 * n * max_val
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            # optional print
        # gradient
        k_max = int(np.argmax(conv))
        shifts = k_max - np.arange(n)
        mask = (shifts >= 0) & (shifts < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[shifts[mask]]
        # mirror descent step
        a_new = a * np.exp(-eta * g)
        # avoid underflow => add epsilon
        a_new = np.clip(a_new, 1e-20, None)
        a_new = a_new / a_new.sum()
        a = a_new
        # update eta
        eta = max(eta * decay, eta_min)
        # occasional random jitter
        if it % 5000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            a = np.clip(a + noise, 1e-20, None)
            a = a / a.sum()
            # optionally reset eta?
        # occasional restart: if no improvement for many iter, jitter bigger
    # after loop: compression of tiny entries
    thresh = 1e-12

        best_a = best_a[mask]

        # recompute obj
        conv = _conv_self_fft(best_a)

    return best_a.tolist(), best_obj

def _project_simplex(v):
    # v can be any real vector
    # returns vector on simplex (nonnegative sum=1)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Non‑negative step‑function optimiser.

The routine works in three stages while respecting the overall 1000 s budget.

1. QUICK SEARCH – projected‑gradient descent (PGD) on the probability simplex.
   The objective is  max(conv(a,a))  (the denominator of evaluate_sequence
   is constant after normalising sum(a)=1).  At every iteration we compute the
   exact sub‑gradient of the current maximal convolution entry and perform a
   Euclidean projection onto the simplex.  Learning‑rate η decays
   exponentially, occasional jitter and restarts help to escape plateaus.

2. SCALE‑UP – the best sequence from the short runs is interpolated to a
   larger support (up to ≈ 8000 entries) and refined again with PGD,
   allowing a finer balance between many constraints.

3. POLISH – a targeted simulated‑annealing (SA) step moves a tiny amount of
   mass away from the indices that contribute most to the current maximal
   convolution entry.  Non‑improving moves are accepted with a Metropolis
   probability that cools slowly.

The final sequence is returned as a plain list of floats suitable for
``evaluate_sequence``.  All helper functions are top‑level, contain no
closures and use only NumPy and the standard library.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return conv_full[: 2 * n - 1]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # Implements the algorithm of Duchi et al. (ICML 2008)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative – fall back to uniform
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _interpolate(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a (length m) to a new length new_n."""
    if a.size == new_n:
        return a.copy()
    old_n = a.size
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.clip(a_new, 0.0, None)
    s = a_new.sum()
    if s <= 0.0:
        a_new = np.ones(new_n) / new_n
    else:
        a_new /= s
    return a_new


# ----------------------------------------------------------------------
# Projected gradient descent on the simplex (exact sub‑gradient)
# ----------------------------------------------------------------------


def _run_pg(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
) -> tuple[list[float], float]:
    """
    Projected gradient descent (PGD) on the simplex with the exact
    sub‑gradient of max(conv(a,a)).
    Returns the best sequence (as list) and its objective value
    O = 2·n·max(conv).
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # initialise a on the simplex
    if init_a is not None:
        a = np.asarray(init_a, dtype=float).copy()
        a = np.clip(a, 0.0, None)
        if a.sum() < 1e-12:
            a = rng.random(n)
        a = a / a.sum()
    else:
        a = rng.random(n)
        a = a / a.sum()

    best_a = a.copy()
    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())

    # hyper‑parameters
    eta_start = 20.0          # initial step size (large to move quickly)
    eta_end = 1e-4            # final step size
    decay = 0.9995            # exponential decay factor per iteration
    eta = eta_start
    it = 0

    while time.time() - start < time_limit:
        it += 1

        conv = _conv_self_fft(a)
        max_val = conv.max()
        obj = float(2.0 * n * max_val)

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[pg n={n:5d} it={it:06d}] new best O={best_obj:.6f}")

        # sub‑gradient of the maximal convolution entry
        k_max = int(np.argmax(conv))
        shifts = k_max - np.arange(n)
        mask = (shifts >= 0) & (shifts < n)
        g = np.zeros_like(a)
        if mask.any():
            g[mask] = 2.0 * a[shifts[mask]]

        # projected gradient step
        v = a - eta * g
        a = _project_simplex(v)

        # learning‑rate schedule
        eta = max(eta * decay, eta_end)

        # occasional jitter to escape flat regions
        if it % 5000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            a = np.clip(a + noise, 0.0, None)
            a /= a.sum()

        # occasional restart from the current best (with tiny perturbation)
        if it % 20000 == 0:
            a = best_a + 0.001 * rng.random(n)
            a = _project_simplex(a)

    # compress negligible entries (helps the final objective)
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a /= best_a.sum()
        best_obj = float(2.0 * best_a.size * _conv_self_fft(best_a).max())
        print(f"[pg compress] reduced length → {best_a.size}, O={best_obj:.6f}")

    return best_a.tolist(), best_obj


# ----------------------------------------------------------------------
# Targeted simulated‑annealing refinement (same ideas as the reference)
# ----------------------------------------------------------------------


def _run_sa(
    init_seq: list[float],
    time_limit: float,
    seed: int = 0,
) -> tuple[list[float], float]:
    """
    Simple SA that moves a small fraction of mass away from the indices
    contributing most to the current maximal convolution entry.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    a = np.asarray(init_seq, dtype=float)
    if a.sum() < 1e-12:
        a = np.ones_like(a) / a.size
    else:
        a = a / a.sum()
    n = a.size

    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())
    best_seq = a.copy()

    # temperature schedule (log‑linear)
    T = best_obj * 0.2 + 1e-6
    T_min = best_obj * 1e-6 + 1e-9
    decay = 0.995

    it = 0
    while time.time() - start < time_limit:
        it += 1
        k_max = int(np.argmax(conv))

        # contribution of each index to the maximal entry
        i_vals = np.arange(n)
        j_vals = k_max - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i_vals[mask]] * a[j_vals[mask]]
        total = contrib.sum()
        if total > 0.0:
            probs = contrib / total
            i = rng.choice(n, p=probs)
        else:
            i = rng.integers(n)

        # destination index (uniformly random, different from i)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.05  # up to 5 % of a_i
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.clip(a_new, 0.0, None)
        a_new /= a_new.sum()

        conv_new = _conv_self_fft(a_new)
        obj_new = float(2.0 * n * conv_new.max())

        if obj_new < best_obj - 1e-12:
            a, conv, best_obj, best_seq = a_new, conv_new, obj_new, a_new.copy()
            print(f"[sa it={it}] new best O={best_obj:.6f}")
        else:
            prob = np.exp(-(obj_new - best_obj) / T) if T > 0 else 0.0
            if rng.random() < prob:
                a, conv = a_new, conv_new

        if it % 2000 == 0:
            T = max(T * decay, T_min)

        if it % 10000 == 0:
            noise = rng.normal(scale=0.005, size=n)
            a = np.clip(a + noise, 0.0, None)
            a /= a.sum()
            conv = _conv_self_fft(a)

    # final pruning of tiny entries
    thresh = 1e-12
    mask = best_seq > thresh
    if mask.sum() < best_seq.size:
        comp = best_seq[mask]
        comp /= comp.sum()
        obj_comp = float(2.0 * comp.size * _conv_self_fft(comp).max())
        if obj_comp < best_obj:
            best_seq, best_obj = comp, obj_comp
            print(f"[sa compress] reduced length → {comp.size}, O={best_obj:.6f}")

    return best_seq.tolist(), best_obj

def propose_candidate(

    """

    * a few quick PGD runs on modest support sizes,
    * interpolation to a larger support and a longer PGD run,
    * a final targeted SA polishing stage.

    """

    deadline = time.time() + budget_s - 0.5   # safety margin
