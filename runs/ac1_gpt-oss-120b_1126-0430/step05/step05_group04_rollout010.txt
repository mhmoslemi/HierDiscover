def run_adam(n, time_limit, seed, init_seq=None):
    start = time.time()
    rng = np.random.default_rng(seed)
    if init_seq is not None and init_seq.size == n:
        a = np.asarray(init_seq, dtype=float)
        a = np.clip(a,0,None)
        if a.sum()< 1e-12:
            a = rng.random(n)
        a = a / a.sum()
    else:
        a = rng.random(n)
        a = a / a.sum()
    x = np.log(a + 1e-16)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # schedule: lr start 0.2, lr end 0.001
    lr_start = 0.2
    lr_end = 0.001
    beta_start = 0.5
    beta_end = 1e5

    best_a = a.copy()
    best_obj = compute_obj(best_a)

    it = 0
    while time.time() - start < time_limit:
        it += 1
        a = _softmax(x)
        conv = _conv_self_fft(a)
        M = conv.max()
        obj = 2.0 * n * M   # denominator sum=1

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[Adam n={n} it={it}] new best O={best_obj:.6f}")

        # schedule
        t = min((time.time() - start) / time_limit, 1.0)
        beta = beta_start * np.exp(np.log(beta_end / beta_start) * t)
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t)

        # compute weighting w for smooth max
        if beta > 50000:
            kmax = int(np.argmax(conv))
            w = np.zeros_like(conv)
            w[kmax] = 1.0
        else:
            shifted = conv - M
            w = np.exp(beta * shifted)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.ones_like(conv) / conv.size
            else:
                w /= w_sum

        # gradient w.r.t a
        g_a = 2.0 * _correlation_fft(w, a)  # as before

        # natural gradient in logit
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * g_x
        v = beta2 * v + (1 - beta2) * (g_x ** 2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter
        if it % 2000 == 0:
            jitter = rng.normal(scale=0.005, size=n)
            x += jitter

        # restart from best every 50000 iterations
        if it % 50000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.003, size=n)
            m.fill(0.0)
            v.fill(0.0)

    # prune small entries
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = compute_obj(best_a)
        print(f"[Adam prune] reduced length→{best_a.size}, O={best_obj:.6f}")

    return best_a.tolist(), best_obj

    # Phase 1: short Adam runs to get good starting point.
    phase1_time = total_time * 0.2
    n = int(5000)
    if phase1_time > 5.0:
        # multiple restarts within phase1_time
        start_phase1 = time.time()
        best_seq = None
        best_obj = float('inf')

        while time.time() - start_phase1 < phase1_time - 0.5:
            # random init mode
            mode = rng.choice(['random', 'power', 'exp'])
            init_seq = None
            if mode == 'random':
                init_seq = rng.random(n)
            elif mode == 'power':
                p = rng.uniform(0.5, 3.0)

                init_seq = idx**(-p)

"""
Non‑negative step‑function optimiser (Adam + long Simulated Annealing).

Phase 1 – a short Adam run on a moderate support size (≈5 k) from several
randomised initialisations (uniform, power‑law or exponential).  The Adam
optimiser works on the soft‑max (log‑it) parametrisation of the simplex and
uses an exponentially increasing β to approximate the max‑convolution.

Phase 2 – a prolonged targeted simulated‑annealing refinement.  At each
iteration the index k with maximal self‑convolution is located, the
contributing pairs a[i]·a[k‑i] are weighted, and a modest amount of mass is
moved from a high‑contributing index to a random destination.  Improvements
are always accepted; worsening moves are accepted with a Metropolis
probability that slowly cools.  The temperature schedule is tuned to explore
thoroughly for the bulk of the 1000 s budget.

The routine respects the overall time budget, returns a plain list of non‑
negative floats, and prints intermediate objective values for monitoring.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (same as in the reference implementation)
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return conv_full[: 2 * n - 1]


def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation g_i = Σ_t w[i+t]·a[t]  for i=0,…,n‑1.
    Implemented via FFT.
    """
    n = a.shape[0]
    m = w.shape[0]                     # = 2n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


# ----------------------------------------------------------------------
# Objective helpers (exact objective, no clipping – identical to evaluate_sequence)
def _objective(a: np.ndarray) -> float:
    """Exact objective  O = 2·n·max(conv(a,a)) / (sum(a)^2)."""
    n = a.size
    conv = _conv_self_fft(a)
    max_conv = conv.max()
    sum_a = a.sum()
    if sum_a < 1e-12:
        return float('inf')
    return float(2.0 * n * max_conv / (sum_a ** 2))

# Adam optimiser on the simplex (log‑it parametrisation)
def _run_adam(
    n: int,
    time_limit: float,
    seed: int = 0,
    init_seq: np.ndarray | None = None,
) -> tuple[list[float], float]:
    """Adam optimisation on the log‑it simplex, returns best sequence and its O."""
    start = time.time()
    rng = np.random.default_rng(seed)

    if init_seq is not None and init_seq.size == n:
        a = np.asarray(init_seq, dtype=float)
        a = np.clip(a, 0.0, None)
        if a.sum() < 1e-12:
            a = rng.random(n)
        a = a / a.sum()
    else:
        a = rng.random(n)
        a = a / a.sum()

    # logits
    x = np.log(a + 1e-16)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # schedule (more aggressive than the reference)
    lr_start, lr_end = 0.25, 0.0005
    beta_start, beta_end = 0.5, 1e5

    best_a = a.copy()
    best_obj = _objective(best_a)

    it = 0
    while time.time() - start < time_limit:
        it += 1
        a = _softmax(x)                     # stay on the simplex

        conv = _conv_self_fft(a)
        M = conv.max()
        obj = 2.0 * n * M / (a.sum() ** 2)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[Adam n={n:5d} it={it:06d}] new best O={best_obj:.6f}")

        # time‑based schedules
        t = min((time.time() - start) / time_limit, 1.0)
        beta = beta_start * np.exp(np.log(beta_end / beta_start) * t)
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t)

        # smooth‑max weighting
        if beta > 50000.0:
            kmax = int(np.argmax(conv))
            w = np.zeros_like(conv)
            w[kmax] = 1.0
        else:
            shifted = conv - M                 # ≤ 0
            w = np.exp(beta * shifted)
            s = w.sum()
            if s == 0.0:
                w = np.ones_like(conv) / conv.size
            else:
                w /= s

        # gradient of the (smooth) max w.r.t. a
        g_a = 2.0 * _correlation_fft(w, a)

        # natural gradient in log‑it space
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape plateaus
        if it % 3000 == 0:
            jitter = rng.normal(scale=0.008, size=n)
            x += jitter

        # occasional restart from the current best
        if it % 60000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.004, size=n)
            m.fill(0.0)
            v.fill(0.0)

    # compress negligible entries (helps the final O)
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = _objective(best_a)
        print(f"[Adam prune] reduced length → {best_a.size}, O={best_obj:.6f}")

    return best_a.tolist(), best_obj

# Simulated‑annealing refinement (targeted, many iterations)
def _run_sa(
    init_seq: list[float],
    time_limit: float,
    seed: int = 0,
) -> tuple[list[float], float]:
    """
    Targeted simulated‑annealing refinement.
    Focuses mass movement on positions that contribute most to the current
    maximal convolution entry.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    a = np.asarray(init_seq, dtype=float)
    if a.sum() < 1e-12:
        a = np.ones_like(a) / a.size
    else:
        a = a / a.sum()
    n = a.shape[0]

    conv = _conv_self_fft(a)
    best_obj = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_seq = a.copy()

    # temperature schedule
    T = best_obj * 0.2 + 1e-9
    T_min = best_obj * 1e-6 + 1e-12
    decay = 0.9985

    it = 0
    max_transfer = 0.25                     # up to 25 % of the selected weight

    while time.time() - start < time_limit:
        it += 1

        # locate the current maximal convolution entry
        k_max = int(np.argmax(conv))

        # contributions a[i]·a[k_max‑i]
        i_vals = np.arange(n, dtype=int)
        j_vals = k_max - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i_vals[mask]] * a[j_vals[mask]]
        total_contrib = contrib.sum()

        # choose source index i proportionally to its contribution (if any)
        if total_contrib > 0 and rng.random() < 0.9:
            probs = contrib / total_contrib
            i = rng.choice(n, p=probs)
        else:
            i = rng.integers(n)

        # pick a destination index j ≠ i uniformly
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # amount of mass to transfer
        delta = a[i] * rng.random() * max_transfer
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # keep the vector on the simplex (sum unchanged, non‑negative)
        a_new = np.clip(a_new, 0.0, None)

        conv_new = _conv_self_fft(a_new)
        obj_new = 2.0 * n * conv_new.max() / (a_new.sum() ** 2)

        # Metropolis acceptance
        if obj_new < best_obj - 1e-14:
            best_obj, best_seq = obj_new, a_new.copy()
            a, conv = a_new, conv_new
            print(f"[SA it={it}] new best O={best_obj:.6f}")
        else:
            # accept with probability exp(-(Δ)/T)
            prob = np.exp(-(obj_new - best_obj) / T) if T > 0 else 0.0
            if rng.random() < prob:
                a, conv = a_new, conv_new

        # temperature schedule (geometric decay)
        if it % 2000 == 0:
            T = max(T * decay, T_min)

        # occasional global jitter to escape deep valleys
        if it % 15000 == 0:
            noise = rng.normal(scale=0.012, size=n)
            a = np.clip(a + noise, 0.0, None)
            a = a / a.sum()
            conv = _conv_self_fft(a)

    # final compression of tiny entries
    thresh = 1e-12
    mask = best_seq > thresh
    if mask.sum() < best_seq.size:
        comp = best_seq[mask]
        comp = comp / comp.sum()
        comp_obj = 2.0 * comp.size * _conv_self_fft(comp).max() / (comp.sum() ** 2)
        if comp_obj < best_obj:
            best_seq, best_obj = comp, comp_obj
            print(f"[SA compress] reduced length → {comp.size}, O={best_obj:.6f}")

    return best_seq.tolist(), best_obj

# Main entry point
def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **_ignore,
) -> list[float]:
    """
    Entry point for the optimisation harness.
    Performs a short Adam phase (several random restarts) to obtain a solid
    starting distribution and then spends the bulk of the time budget on a
    dedicated simulated‑annealing refinement.
    Returns a list of non‑negative floats suitable for evaluate_sequence().
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Phase 1: short Adam runs (≈20 % of total budget)
    phase1_budget = (budget_s - 1.0) * 0.2
    n = 5000                                 # moderate support size
    best_seq = None
    best_obj = float('inf')
    start_phase1 = time.time()

    # number of independent Adam restarts is limited by remaining time
    restart = 0
    while time.time() - start_phase1 < phase1_budget - 0.5:
        # pick an initialisation strategy
        mode = rng.choice(['random', 'power', 'exp'])
        if mode == 'random':
            init = rng.random(n)
        elif mode == 'power':
            p = rng.uniform(0.5, 3.0)
            init = (np.arange(1, n + 1, dtype=float) ** (-p))
        else:  # exponential
            lam = rng.uniform(0.001, 0.1)
            init = np.exp(-lam * np.arange(n, dtype=float))

        init = init / init.sum()
        # allocate a small slice of the budget to each run (≈10 s)
        run_time = min(10.0, phase1_budget - (time.time() - start_phase1) - 0.5)
        seq, obj = _run_adam(
            n=n,
            time_limit=run_time,
            seed=seed + restart * 97,
            init_seq=init,
        )
        if obj < best_obj:
            best_obj, best_seq = obj, np.asarray(seq, dtype=float)
            print(f"[Phase 1] new overall best O={best_obj:.6f} (restart {restart})")
        restart += 1
        if time.time() >= deadline - 0.5:
            break

    # Fallback if Phase 1 produced nothing
    if best_seq is None:
        best_seq = rng.random(n)
        best_seq = best_seq / best_seq.sum()
        best_obj = _objective(best_seq)

    # ------------------------------------------------------------------
    # Phase 2: exhaustive simulated‑annealing (majority of the budget)
    remaining = deadline - time.time()
    if remaining > 6.0:
        seq_refined, obj_refined = _run_sa(
            best_seq.tolist(),
            time_limit=remaining,
            seed=seed + 9999,
        )
        if obj_refined < best_obj:
            best_obj, best_seq = obj_refined, np.asarray(seq_refined, dtype=float)
            print(f"[Phase 2] after SA new best O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Final sanity checks / tiny‑entry pruning
    best_seq = np.clip(best_seq, 0.0, None)
    thresh = 1e-12
    mask = best_seq > thresh
    if mask.sum() < best_seq.size:
        best_seq = best_seq[mask]
        best_seq = best_seq / best_seq.sum()

    # one final evaluation for logging
    try:
        final_score = _objective(best_seq)
        print(f"[final] estimated O={final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] internal objective evaluation failed:", exc)

    return best_seq.tolist()
