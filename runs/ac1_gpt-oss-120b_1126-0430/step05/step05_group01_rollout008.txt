def _spectral_flatten(a: np.ndarray, L: int=None) -> np.ndarray:

   if L is None:

   # FFT

#!/usr/bin/env python
"""
Non‑negative step‑function optimiser (alternative to the reference).

The search proceeds in three phases:

1. **Initialisation** – a large suite of parametric families (power‑law,
   symmetric product‑law, exponential, logistic, sine‑window, Hann, Beta…) is
   scanned on a moderate length (≈ 25000).  The best candidate (according to
   the cheap score 2·n·max(a*a)) is kept.

2. **Spectral flattening** – the current distribution is iteratively
   transformed by fixing the DC component of its FFT and forcing all other
   magnitudes to a common value.  After each flattening step the cheap score
   is recomputed and improvements are retained.  This operation pushes the
   autocorrelation towards a flat shape.

3. **Mirror‑descent optimisation** – a mirror‑descent on the probability
   simplex (Adam on the logits) is run with the exact sub‑gradient of the
   non‑smooth objective 2·n·max(a*a).  The gradient for a given maximal
   convolution lag `k_max` is simply a shifted copy of the current vector,
   which can be computed in O(n) time.  Adaptive step‑size, back‑tracking line
   search, occasional jitter and stagnation‑restarts are employed.  The
   support size is optionally increased if time permits.

The routine respects the supplied time budget (~1000 s) and returns the best
sequence discovered (as a plain list of floats).  All numbers are non‑negative,
the sum is automatically normalised to one, and the final exact evaluation
uses the provided `evaluate_sequence` function.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (no closures, no lambdas)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max, returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    s = e.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / z.size)
    return e / s


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it that reproduces `a` after a soft‑max."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """Smallest power‑of‑two ≥ 3·n‑2 (enough for all needed convolutions)."""
    return 1 << ((3 * n - 2).bit_length())


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    s = new_a.sum()
    if s == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= s
    return new_a


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(‑p), normalised."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _sym_product_power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ ((i+1)*(n‑i))^(‑p), symmetric, normalised."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) * (n - i)) ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _exp_decay_distribution(n: int, lam: float) -> np.ndarray:
    """a_i ∝ e^(‑lam·i)+e^(‑lam·(n‑1‑i)), symmetric exponential."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _logistic_distribution(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ 1/(1+e^{α·(i‑mid)}), normalised; heavy on the left side."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = 1.0 / (1.0 + np.exp(alpha * (i - mid)))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad_exact(a: np.ndarray):
    """
    Exact max of the linear self‑convolution and its sub‑gradient
    (∇_a 2·n·max(a*a)).
    Returns (max_conv, grad_a) where
        grad_a[i] = 4·n·a[k_max‑i]   if 0 ≤ k_max‑i < n,
                     0               otherwise.
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]

    max_conv = float(conv.max())
    k_max = int(conv.argmax())

    i = np.arange(n, dtype=np.int64)
    idx = k_max - i
    mask = (idx >= 0) & (idx < n)

    grad = np.zeros_like(a)
    # Vectorised assignment
    grad[mask] = 4.0 * n * a[idx[mask]]
    return max_conv, grad


def _spectral_flatten(a: np.ndarray, L: int = None) -> np.ndarray:
    """
    One iteration of “spectral flattening”:
    keep the DC component of the FFT unchanged and force all other
    magnitudes to a common value (the RMS of the non‑DC part).  The result
    is projected back onto the probability simplex.
    """
    n = a.shape[0]
    if L is None:
        L = _fft_len_for_corr(n)

    A = np.fft.rfft(a, L)

    # Keep DC (frequency 0) untouched
    dc = A[0]

    # Target magnitude for all other frequencies
    mags = np.abs(A[1:])
    if mags.size == 0:
        # Degenerate case (n == 1)
        new_A = np.array([dc], dtype=A.dtype)
    else:
        target_mag = np.sqrt(np.mean(mags ** 2))
        phases = np.angle(A[1:])
        new_A = np.empty_like(A)
        new_A[0] = dc
        new_A[1:] = target_mag * np.exp(1j * phases)

    a_new = np.fft.irfft(new_A, L)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s == 0.0:
        a_new[:] = 1.0 / n
    else:
        a_new /= s
    return a_new

    # ------------------------------------------------------------------

    n = n_initial

    # ------------------------------------------------------------------

    # Logistic windows
    for alpha in np.linspace(0.01, 5.0, 31):
        families.append(_logistic_distribution(n, alpha))

    # Sine‑squared window
    i = np.arange(n, dtype=np.float64)
    a_sin2 = np.sin(np.pi * (i + 0.5) / n) ** 2
    a_sin2 /= a_sin2.sum()
    families.append(a_sin2)

    # Hann window
    a_hann = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
    a_hann /= a_hann.sum()
    families.append(a_hann)

    # Symmetric Beta‑like shapes (parameter p > 0, zero at the ends for p>1)
    for p in (0.5, 1.0, 2.0, 3.0):
        a_beta = ((i + 1) * (n - i)) ** (p - 1)
        a_beta /= a_beta.sum()
        families.append(a_beta)

    # If a global variable is present, try to use it as a starting point.
    try:
        global height_sequence_1  # type: ignore[name-defined]
        init_global = np.array(height_sequence_1, dtype=np.float64)
        if init_global.size >= 5000:  # avoid extremely short globals
            init_global = np.maximum(init_global, 0.0)
            s = init_global.sum()
            if s == 0.0:
                init_global[:] = 1.0 / init_global.size
            else:
                init_global /= s
            families.append(_interpolate_distribution(init_global, n))
    except Exception:
        pass  # no viable global variable

    # Evaluate all candidates (cheap bound) and keep the best.
    best_a = None
    best_score = float("inf")
    for cand in families:
        if time.time() >= deadline:
            break
        cheap = 2.0 * n * _conv_max(cand)
        if cheap < best_score:
            best_score = cheap
            best_a = cand.copy()

    if best_a is None:
        best_a = np.full(n, 1.0 / n)

    print(f"[Init] Chosen length={n}, initial cheap score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Spectral flattening phase (quick deterministic refinement).
    # ------------------------------------------------------------------
    a = best_a
    L_fft = _fft_len_for_corr(n)
    flatten_iters = 120
    for it_flat in range(flatten_iters):
        if time.time() >= deadline:
            break
        a_new = _spectral_flatten(a, L_fft)
        cheap_new = 2.0 * n * _conv_max(a_new)
        if cheap_new < best_score:
            a = a_new
            best_score = cheap_new
            best_a = a.copy()
        if it_flat % 20 == 0:
            print(f"[Flatten {it_flat:03d}] cheap={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Adam‑based mirror‑descent on the simplex using the exact sub‑gradient.
    # ------------------------------------------------------------------
    # Initialise logits with a tiny random jitter.
    z = _logits_from_distribution(a)
    z += 0.001 * np.random.randn(n)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    base_alpha = 0.45  # base learning‑rate for logits
    report_every = 10000
    last_improve_iter = 0

    best_seq = _softmax(z)
    best_score = 2.0 * n * _conv_max(best_seq)

    it = 0
    max_iters = 600_000

    while time.time() < deadline and it < max_iters:
        it += 1

        # Current point on the simplex
        a = _softmax(z)

        # Exact max‑convolution and its sub‑gradient
        max_conv, grad_a = _conv_max_and_grad_exact(a)

        cheap_score = 2.0 * n * max_conv

        # Record best seen so far
        if cheap_score < best_score - 1e-12:
            best_score = cheap_score
            best_seq = a.copy()
            last_improve_iter = it
            if it % report_every == 0:
                print(f"[Iter {it:07d}] NEW BEST cheap = {cheap_score:.6f}")

        # Mirror‑descent (Adam on logits)
        mu = np.dot(grad_a, a)                # <grad, a>
        grad_z = a * (grad_a - mu)           # ∇_z objective (KL‑mirror)

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step_dir = - m_hat / (np.sqrt(v_hat) + eps_adam)

        # Simple back‑tracking line‑search (max 4 halvings)
        accepted = False
        trial_alpha = alpha_t
        for _ in range(4):
            z_new = z + trial_alpha * step_dir
            np.clip(z_new, -50.0, 50.0, out=z_new)
            a_new = _softmax(z_new)
            new_score = 2.0 * n * _conv_max(a_new)
            if new_score <= cheap_score:
                z = z_new
                accepted = True
                break
            trial_alpha *= 0.5

        if not accepted:
            # No improvement – take a modest step anyway.
            z += alpha_t * step_dir
            np.clip(z, -50.0, 50.0, out=z)

        # Random jitter now and then (helps to escape plateaux)
        if it % 3000 == 0 and np.random.rand() < 0.14:
            z += 0.015 * np.random.randn(n)
            m *= 0.6
            v *= 0.6

        # Stagnation detection
        if it - last_improve_iter > 180_000:
            print(f"[Iter {it}] Stagnation – restarting from best.")
            z = _logits_from_distribution(best_seq) + 0.03 * np.random.randn(n)
            m.fill(0.0)
            v.fill(0.0)
            last_improve_iter = it

        # Optional up‑sampling (if enough time remains)
        if (it % 120_000 == 0) and (n < 30000) and (time.time() + 30.0 < deadline):
            new_n = min(30000, int(n * 1.20))
            print(f"[Iter {it}] Upsampling support: {n} → {new_n}")
            best_seq = _interpolate_distribution(best_seq, new_n)
            z = _logits_from_distribution(best_seq) + 0.005 * np.random.randn(new_n)
            m = np.zeros_like(z)
            v = np.zeros_like(z)
            n = new_n
            best_score = 2.0 * n * _conv_max(best_seq)
            it = 0
            last_improve_iter = 0

    # ------------------------------------------------------------------
    # 4. Final exact evaluation with the provided `evaluate_sequence`.
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # ------------------------------------------------------------------
    # 5. Return the best sequence discovered.
    # ------------------------------------------------------------------
    return final_seq
