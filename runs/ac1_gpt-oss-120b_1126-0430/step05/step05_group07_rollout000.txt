"""
Propose a step‑function that (approximately) minimises

    O(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

Algorithm (≈ 1000 s)
-------------------------------------------------
1️⃣  Initialise a (≈ 8 k points).  Use the provided ``height_sequence_1``
    if available, otherwise start from a uniform distribution.
2️⃣  **Phase 1 – variance flattening**  
    Optimise the variance of the full convolution (∑(c‑μ)²) with Adam on the
    simplex.  This pushes all convolution entries towards the same value.
3️⃣  **Phase 2 – smooth‑max refinement**  
    Replace the hard max by a log‑sum‑exp surrogate; anneal the temperature
    (α) from 10 to 4·10⁴ while continuing Adam updates (soft‑max gradient
    + tiny entropy).  The surrogate converges to the true max.
4️⃣  **Phase 3 – sub‑gradient + stochastic mass transfer**  
    Repeatedly compute the exact sub‑gradient of ``max(conv)`` (via FFT
    correlation), move a small fraction of mass from the largest‑gradient
    index to the smallest‑gradient index (Metropolis acceptance), and
    occasionally take a deterministic sub‑gradient step.  Periodic
    smoothing and restarts keep the search from stagnating.
5️⃣  Return the best sequence found (still on the simplex, i.e. sum = 1).

The code keeps a tight time budget, prints progress and finally checks the
result with the supplied ``evaluate_sequence`` function.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, sum=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:                     # all entries are zero → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """
    Full linear convolution a★a (length 2·n‑1) using an FFT of length L.
    ``L`` must be a power of two ≥ 2·len(a)‑1.
    """
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
def _gradient_variance(a: np.ndarray, conv: np.ndarray, n: int, L: int) -> np.ndarray:
    """
    Gradient of   f(a)=∑_k (conv_k – μ)²   with μ = 1/(2·n‑1)   (sum(a)=1).
    Closed form: g = 4·corr(conv‑μ , a).
    """
    mu = 1.0 / (2 * n - 1)
    diff = conv - mu                     # length 2·n‑1
    a_rev = a[::-1]

    fft_diff = np.fft.rfft(diff, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_diff * fft_a_rev, n=L)

    start = n - 1
    grad = 4.0 * corr_rev[start:start + n]
    return grad


# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, conv: np.ndarray,
                     alpha: float, L: int, lam: float) -> np.ndarray:
    """
    Gradient of the smooth surrogate
        (2·n/α)·log Σ_k e^{α·conv_k}  +  λ·∑ a_i·log a_i
    The first term equals 2·corr(w, a) where w ∝ exp(α·conv).
    """
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)          # avoid overflow
    w = exp_scaled / exp_scaled.sum()

    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)

    grad_corr = 2.0 * corr_rev[n - 1:n - 1 + n]

    eps = 1e-20
    grad_ent = lam * (np.log(a + eps) + 1.0)

    return grad_corr + grad_ent


# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, conv: np.ndarray, L: int) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a.
    If M = max(conv) and 𝕀 = {k | conv_k = M},
    then ∂max/∂a_i = 2·∑_{k∈𝕀} a_{k‑i}   (with out‑of‑range indices ignored).
    Implemented via FFT correlation for speed.
    """
    max_val = conv.max()
    mask = (conv == max_val).astype(float)          # shape 2·n‑1
    # correlation(mask, a)
    fft_mask = np.fft.rfft(mask, n=L)
    fft_a_rev = np.fft.rfft(a[::-1], n=L)
    corr = np.fft.irfft(fft_mask * fft_a_rev, n=L)
    start = a.size - 1
    g = 2.0 * corr[start:start + a.size]
    return g
