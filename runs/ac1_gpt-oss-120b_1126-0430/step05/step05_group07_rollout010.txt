"""
Propose a non‑negative step‑function that (approximately) minimises

    O(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

Algorithm
---------
1️⃣  Initialise a symmetric probability vector (∑a_i = 1, a_i ≥ 0).  If a
    reference construction ``height_sequence_1`` is available we start from
    it, otherwise we use a random Dirichlet vector on a power‑of‑two size
    (≈8192) which is convenient for FFT.

2️⃣  **Smooth max‑descent** – minimise a log‑sum‑exp surrogate of the
    convolution peak on the simplex using Adam with an exponential
    temperature schedule (α grows from ~0.1 to several thousand).  After each
    Adam step we project onto the simplex and enforce exact symmetry
    (a↔a[::-1]).  This rapidly flattens the convolution.

3️⃣  **Deterministic sub‑gradient mass transfer** – repeatedly compute the
    sub‑gradient of the true max (via FFT), pick the index that contributes
    most (largest gradient) and the index that contributes least, and move a
    small amount of mass from the former to the latter.  The move is kept
    only if it does not increase the objective (or with a tiny
    Metropolis‑type probability).  After every move we re‑symmetrise and
    re‑project.

4️⃣  **Light stochastic polishing** – occasional random jitter and a
    short Gaussian smoothing keep the search from stagnating.

The three phases share the 1000 s budget; the best sequence found at any
time is returned.  Only the final sequence is passed to ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution via FFT (length 2·n‑1)
def conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return the full convolution a★a (length 2·n‑1) using FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
def objective_simplex(a: np.ndarray, n: int, L: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a already on the simplex."""
    conv = conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def smooth_gradient(a: np.ndarray, conv: np.ndarray, alpha: float,
                    L: int, lam: float) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp surrogate + tiny entropy term.

    The surrogate is
        (2·n/α)·log Σ_k exp(α·conv_k)

    Its gradient w.r.t. a is
        2·corr(w, a) + λ·(log a + 1),
    where w ∝ exp(α·conv).
    """
    # soft‑max weights
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / exp_scaled.sum()                 # shape (2n‑1,)

    # correlation term  (same trick as in variance‑gradient)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n_len = a.size
    grad_corr = 2.0 * corr_rev[n_len - 1:n_len - 1 + n_len]

    # tiny entropy (helps avoid exact zeros)
    eps = 1e-20
    grad_ent = lam * (np.log(a + eps) + 1.0)

    return grad_corr + grad_ent


# ----------------------------------------------------------------------
def subgradient_max(a: np.ndarray, conv: np.ndarray, L: int) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a.

    For each index i,
        g_i = 2·∑_{k : conv_k = max} a_{k-i}
    (interpreted as 0 when the index is out of range).
    Computed efficiently via FFT.
    """
    # binary mask of the active (max) positions
    tol = 1e-14
    mask = (conv >= conv.max() - tol).astype(float)

    a_rev = a[::-1]
    fft_mask = np.fft.rfft(mask, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_mask * fft_a_rev, n=L)
    n_len = a.size
    g = 2.0 * corr_rev[n_len - 1:n_len - 1 + n_len]
    return g


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    the provided evaluation function.  Returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.3          # safety margin

    # --------------------------------------------------------------
    # Determine problem size and initialise a
    if "height_sequence_1" in globals():
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
        n = base.size
        a = base.copy()
        a = np.maximum(a, 0.0)
        a = a / a.sum()
        print(f"[init] start from supplied sequence, n={n}")
    else:
        # a power‑of‑two size eases FFT padding
        n = 8192
        a = np.random.rand(n)
        a = project_to_simplex(a)
        print(f"[init] start from random uniform, n={n}")

    # enforce exact symmetry (the problem is invariant under reversal)
    a = (a + a[::-1]) * 0.5

    # FFT length – next power of two ≥ 2·n‑1
    L = 1 << ((2 * n - 1).bit_length())

    # --------------------------------------------------------------
    # Track the best solution seen so far
    best_seq = a.copy()
    best_score = objective_simplex(best_seq, n, L)
    print(f"[status] initial O={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – smooth‑max (log‑sum‑exp) descent with Adam
    phase1_deadline = start_time + budget_s * 0.35   # ≈ 35 % of the budget
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    lr0 = 0.5
    lam_entropy = 1e-5
    alpha_start = 0.1
    alpha_end = 5000.0
    it = 0

    while time.time() < phase1_deadline:
        conv = conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[phase‑1] iter {it:,} new best O={best_score:.6f}")

        # exponential schedule for α (geometric increase)
        frac = (time.time() - start_time) / (phase1_deadline - start_time)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac

        grad = smooth_gradient(a, conv, alpha, L, lam_entropy)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        lr = lr0 / math.sqrt(it + 1)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # project onto simplex and re‑symmetrise
        a_new = project_to_simplex(a_new)
        a_new = (a_new + a_new[::-1]) * 0.5
        a = a_new
        it += 1

        if it % 20000 == 0:
            print(f"[phase‑1] iter {it:,} O={cur_obj:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – deterministic sub‑gradient mass‑transfer
    print(f"[phase‑2] start, best O={best_score:.6f}")
    it = 0
    while time.time() < deadline:
        conv = conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = a.copy()
            now = time.time() - start_time
            print(f"[phase‑2] time {now:.1f}s iter {it:,} new best O={best_score:.6f}")

        # compute sub‑gradient of the max using FFT (fast)
        g = subgradient_max(a, conv, L)

        # indices with largest / smallest influence on the current peak
        i_max = int(np.argmax(g))
        i_min = int(np.argmin(g))

        # transfer a modest amount of mass from i_max to i_min
        delta = min(0.12 * a[i_max], 0.01)   # cap for stability
        if delta > 1e-12:
            a_candidate = a.copy()
            a_candidate[i_max] -= delta
            a_candidate[i_min] += delta

            # keep feasibility
            a_candidate = np.clip(a_candidate, 0.0, None)
            a_candidate = project_to_simplex(a_candidate)
            a_candidate = (a_candidate + a_candidate[::-1]) * 0.5

            conv_new = conv_fft(a_candidate, L)
            new_obj = 2.0 * n * conv_new.max()

            # accept if not worse (or with a tiny probability)
            if new_obj <= cur_obj or np.random.rand() < 0.01:
                a = a_candidate
                cur_obj = new_obj
                if new_obj < best_score:
                    best_score = new_obj
                    best_seq = a.copy()
                    now = time.time() - start_time
                    print(f"[phase‑2] time {now:.1f}s iter {it:,} accepted O={best_score:.6f}")

        # occasional random jitter to escape plateaus
        if np.random.rand() < 0.008:
            jitter = np.random.randn(n) * 0.001
            a = a + jitter
            a = np.clip(a, 0.0, None)
            a = project_to_simplex(a)
            a = (a + a[::-1]) * 0.5

        # occasional mild smoothing (Gaussian‑like kernel)
        if np.random.rand() < 0.004:
            kernel = np.array([0.15, 0.7, 0.15])
            a = np.convolve(a, kernel, mode="same")
            a = project_to_simplex(a)
            a = (a + a[::-1]) * 0.5

        it += 1
        if it % 25000 == 0:
            print(f"[phase‑2] iter {it:,} O={cur_obj:.6f}")

    # --------------------------------------------------------------
    # Final sanity check and return
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:
        print("[final] evaluate_sequence failed:", e)

    return best_seq.tolist()
