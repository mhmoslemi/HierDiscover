    n = a.size
    best_a = a.copy()
    best_obj = _objective(a)
    cur_a = a.copy()
    cur_obj = best_obj
    start = time.time()
    it = 0
    # parameters

    init_temp = best_obj * 0.1

"""
Hybrid Adam‑SA optimiser for the step‑function bound.

The routine first runs a small number of Adam optimisation runs (soft‑max
parametrisation of the simplex, smooth‑max approximation of the
convolution max) on several support sizes.  The best sequence found is
then refined by a dedicated simulated‑annealing phase that performs
mass‑transfer moves directly on the simplex.  Finally the best sequence
is sanity‑checked with the provided ``evaluate_sequence`` routine.

All numbers are kept non‑negative, the sum is normalised to 1 (the
objective is homogeneous, so any scaling is equivalent) and the final
return is guaranteed to be a non‑empty list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation of w with a (g_i = Σ_t w[i+t]·a[t]), 0 ≤ i < len(a).
    Implemented via FFT.
    """
    n = a.shape[0]
    m = w.shape[0]                 # = 2·n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) (sum(a) is assumed = 1)."""
    n = a.size
    conv = _conv_self_fft(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


# ----------------------------------------------------------------------
def _adam_opt(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
) -> tuple[list[float], float]:
    """
    Adam optimisation of a length‑``n`` sequence (probability vector)
    for at most ``time_limit`` seconds.  The implementation is a slightly
    tuned version of the code shipped with the competition (larger beta
    end‑point and a slower learning‑rate decay).
    Returns the best sequence as a plain list together with its exact
    objective value.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # ----- initialisation -------------------------------------------------
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
        a = a / a.sum()
    else:
        a = rng.random(n)
        a = a / a.sum()

    # work in log‑it space (automatically respects the simplex)
    x = np.log(a + 1e-16)

    m = np.zeros_like(x)          # Adam first moment
    v = np.zeros_like(x)          # Adam second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # ----- schedules ------------------------------------------------------
    beta_start = 0.5
    beta_end   = 12000.0           # sharper soft‑max
    lr_start   = 0.20
    lr_end     = 0.0005
    target_iters = 1_200_000       # a very generous iteration budget

    best_a = a.copy()
    best_obj = _objective(best_a)

    it = 0
    while time.time() - start < time_limit:
        it += 1

        # probabilities from current logits
        a = _softmax(x)

        # exact objective
        conv = _conv_self_fft(a)
        M = conv.max()
        obj = 2.0 * n * M
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[adam n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        # schedule parameters
        t_frac = min(it / target_iters, 1.0)
        beta = beta_start * np.exp(np.log(beta_end / beta_start) * t_frac)
        lr   = lr_start   * np.exp(np.log(lr_end   / lr_start)   * t_frac)

        # smooth‑max weight vector
        if beta > 8000.0:                         # sharp enough → true arg‑max
            kmax = int(np.argmax(conv))
            w = np.zeros_like(conv)
            w[kmax] = 1.0
        else:
            shifted = conv - M                     # ≤ 0
            w = np.exp(beta * shifted)
            w = w / w.sum()

        # gradient w.r.t. a (smooth max)
        g_a = 2.0 * _correlation_fft(w, a)   # (n,)

        # natural gradient in log‑it space (projected onto simplex)
        dot = np.dot(a, g_a)                 # ⟨a, g_a⟩
        g_x = a * (g_a - dot)                # (n,)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # jitter to escape plateaus
        if it % 5000 == 0:
            jitter = rng.normal(scale=0.025, size=n)
            x += jitter

        # periodic restart from the current best
        if it % 60000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.006, size=n)
            m.fill(0.0)
            v.fill(0.0)
            print(f"[adam n={n}] restart at iter {it}")

    # ----- compress negligible entries ------------------------------------
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        n_new = best_a.size
        best_obj = _objective(best_a)
        print(f"[compress] length {n} → {n_new}, O={best_obj:.6f}")

    # ----- short stochastic local search ----------------------------------
    remaining = time_limit - (time.time() - start)
    if remaining > 0.5:
        a_loc = best_a.copy()
        obj_loc = best_obj
        n_loc = a_loc.size
        local_deadline = time.time() + remaining * 0.35
        while time.time() < local_deadline:
            i, j = rng.choice(n_loc, size=2, replace=False)
            delta = a_loc[i] * rng.random() * 0.02
            if delta <= 0.0:
                continue
            a_new = a_loc.copy()
            a_new[i] -= delta
            a_new[j] += delta
            conv_new = _conv_self_fft(a_new)
            obj_new = 2.0 * n_loc * conv_new.max()
            if obj_new < obj_loc:
                a_loc = a_new
                obj_loc = obj_new
                if obj_new < best_obj:
                    best_a = a_new
                    best_obj = obj_new
        print(f"[local] final O={best_obj:.6f}")

    return best_a.tolist(), best_obj


# ----------------------------------------------------------------------
def _simulated_annealing(
    a: np.ndarray,
    time_limit: float,
    rng: np.random.Generator,
) -> np.ndarray:
    """
    Simple simulated‑annealing refinement on the simplex.
    Mass is transferred between two coordinates; moves are accepted
    according to a temperature that linearly decays to zero.
    """
    n = a.size
    cur_a = a.copy()
    cur_obj = _objective(cur_a)

    best_a = cur_a.copy()
    best_obj = cur_obj

    # temperature schedule (initial temperature = 10 % of current obj)
    T0 = max(1e-12, cur_obj * 0.10)
    start = time.time()
    it = 0

    while time.time() - start < time_limit:
        it += 1
        # linear temperature decay
        frac = (time.time() - start) / time_limit
        T = T0 * max(0.0, 1.0 - frac) + 1e-12

        # propose a transfer of mass
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        if cur_a[i] == 0.0:
            continue
        # move up to 20 % of the donor's mass
        delta = cur_a[i] * rng.random() * 0.20
        if delta <= 0.0:
            continue

        # apply move
        cur_a[i] -= delta
        cur_a[j] += delta

        obj_new = _objective(cur_a)

        # Metropolis acceptance
        if obj_new < cur_obj or rng.random() < np.exp(-(obj_new - cur_obj) / T):
            cur_obj = obj_new
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = cur_a.copy()
        else:
            # revert
            cur_a[i] += delta
            cur_a[j] -= delta

        # occasional global jitter (helps escape deep local minima)
        if it % 20000 == 0:
            logit = np.log(cur_a + 1e-16)
            logit += rng.normal(scale=0.12, size=n)
            cur_a = _softmax(logit)
            cur_obj = _objective(cur_a)
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = cur_a.copy()

    # prune tiny entries before returning
    mask = best_a > 1e-13
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
    return best_a

def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **_ignore,
) -> list[float]:
    """
    Entry point for the harness.
    Runs several Adam optimisations on a ladder of lengths and finishes
    with a simulated‑annealing refinement.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # list of lengths to explore
    lengths = [256, 512, 1024, 2048, 4096, 6144]
    if "height_sequence_1" in globals():
        seq0 = np.asarray(height_sequence_1, dtype=float).copy()
        seq0 = np.clip(seq0, 0.0, None)
        n0 = seq0.size
        if n0 not in lengths:
            lengths.insert(0, n0)

    best_seq = None
    best_obj = float("inf")

    # ------------------------------------------------------------------
    # Adam optimisation phase (most of the time budget)
    for idx, n in enumerate(lengths):
        remaining = deadline - time.time()
        if remaining <= 2.0:
            break

        # allocate a larger share to the first (largest) length
        if idx == 0:
            run_time = remaining * 0.65
        else:
            run_time = (remaining - 0.5) / (len(lengths) - idx)

        # initialise from the global best when possible
        init = None
        if idx == 0 and best_seq is not None and len(best_seq) == n:
            init = best_seq
        elif idx == 0 and best_seq is not None:
            # adapt size by truncating / padding with zeros
            if len(best_seq) > n:
                init = best_seq[:n]
            else:
                pad = np.zeros(n - len(best_seq))
                init = np.concatenate([best_seq, pad])
            init = np.asarray(init, dtype=float)
            init = init / init.sum()

        seq, val = _adam_opt(
            n=n,
            time_limit=run_time,
            init_a=init,
            seed=seed + idx * 37,
        )
        if val < best_obj:
            best_obj = val
            best_seq = seq
            print(f"[global] new overall best O={best_obj:.6f} (n={n})")

    # ------------------------------------------------------------------
    # Simulated‑annealing refinement on the overall best
    if best_seq is not None:
        remaining = deadline - time.time()
        if remaining > 1.0:
            a = np.asarray(best_seq, dtype=float)
            a = a / a.sum()
            rng = np.random.default_rng(seed + 9999)
            a_refined = _simulated_annealing(a, remaining, rng)
            best_seq = a_refined.tolist()
            best_obj = _objective(np.asarray(best_seq))
            print(f"[SA] final refined O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check via the provided evaluate_sequence routine.
    try:
        final_score = evaluate_sequence(best_seq)
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    # guarantee a non‑empty return
    if not best_seq:
        best_seq = [1.0]

    return best_seq
