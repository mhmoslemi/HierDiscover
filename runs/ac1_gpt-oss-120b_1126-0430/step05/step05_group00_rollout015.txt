def _run_adam_opt(a_start, n, deadline, max_iters=200000):
    # a_start is probability vector sum 1
    # returns best_a, best_score
    z = _logits_from_distribution(a_start)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_alpha = 0.6
    eps_factor = 0.5
    eps_factor_min = 1e-4
    eps_factor_decay = 0.9999

    it = 0
    best_a = a_start.copy()
    best_score = 2.0 * n * _conv_max(best_a)
    while time.time() < deadline and it < max_iters:
        it += 1
        a_cur = _softmax(z)
        max_conv, grad_a = _conv_max_and_grad(a_cur, eps_factor)
        cur_score = 2.0 * n * max_conv
        # check improvement
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_cur.copy()
        # compute gradient on logits
        mu = np.dot(grad_a, a_cur)
        grad_z = a_cur * (grad_a - mu)
        # Adam
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        alpha_t = base_alpha / (np.sqrt(it) + 1.0)
        step = - alpha_t * m_hat / (np.sqrt(v_hat) + eps_adam)
        z = z + step
        # clip logits
        z = np.clip(z, -50.0, 50.0)
        # decay eps_factor
        eps_factor = max(eps_factor_min, eps_factor * eps_factor_decay)
        # optional jitter occasional
        if it % 8000 == 0 and np.random.rand() < 0.2:
            z += 0.01 * np.random.randn(n)
        # early exit if no improvement for many iters?
        # But we cannot know.
    return best_a, best_score

#!/usr/bin/env python
"""
Improved arcsine‑integrated construction + lightweight Adam refinement.

The core idea:
    * Use the exact integral of the continuous arcsine density over each
      equally spaced bin.  For x∈[0,1] the CDF of the arcsine law is
          G(x) = (2/π)·arcsin(√x).
      Setting a_i = G((i+1)/n)−G(i/n) yields a probability vector whose
      self‑convolution approximates the constant ½ much more accurately
      than the simple midpoint sampling used before.
    * Starting from that distribution we run a few thousand Adam steps on
      the simplex (logits → soft‑max) while smoothing the max‑convolution
      with a temperature that decays to near‑hard max.  This usually
      pushes the cheap score a little lower.
    * The code respects the 1000 s budget, works for any length n (taken
      from the existing global `height_sequence_1` if present, otherwise
      defaults to 20000), and returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (pure, no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max (maps logits to a probability simplex)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it representation that reproduces `a` after soft‑max."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """
    Smallest power‑of‑two ≥ 3·n‑2.
    Needed for linear convolution / correlation of length up to 3·n‑2.
    """
    return 1 << ((3 * n - 2).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a (FFT based)."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Smoothed max‑convolution and its gradient w.r.t. the probability vector.

    Parameters
    ----------
    a : np.ndarray
        Probability vector (sum = 1, non‑negative).
    eps_factor : float
        Multiplicative factor for the smoothing constant:
        eps = eps_factor * max(conv).

    Returns
    -------
    max_conv : float
        Maximum of the self‑convolution a*a.
    grad_a : np.ndarray
        Gradient of 2·n·smooth_max w.r.t. a (already multiplied by 4·n).
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    a_fft = np.fft.rfft(a, L)

    # linear self‑convolution (length 2n‑1)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # smoothed soft‑max weights
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv          # for numerical stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or np.isnan(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # gradient = 4·n * (a ⋆ w)  (linear correlation length n)
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    grad_a = 4.0 * n * corr
    return max_conv, grad_a


def _arcsine_integrated_distribution(n: int) -> np.ndarray:
    """
    Exact integrated arcsine distribution on n equal bins.
    a_i = G((i+1)/n) - G(i/n),  G(x) = (2/π)·arcsin(√x).
    Returns a probability vector (sum ≈ 1).
    """
    i = np.arange(n + 1, dtype=np.float64)  # 0 … n
    sqrt_vals = np.sqrt(i / n)
    arcsin_vals = np.arcsin(sqrt_vals)
    G = (2.0 / np.pi) * arcsin_vals
    a = G[1:] - G[:-1]                     # size n
    # Normalise to avoid round‑off drift
    total = a.sum()
    if total != 0.0:
        a /= total
    return a


def _beta_integrated_distribution(alpha: float, n: int) -> np.ndarray:
    """
    Integrated Beta(α,α) distribution using SciPy's regularised incomplete beta.
    a_i = I_{(i+1)/n}(α,α) - I_{i/n}(α,α).
    """
    try:
        from scipy.special import betainc
    except Exception as exc:
        # Fallback to midpoint if SciPy is unavailable
        print("SciPy not available, falling back to midpoint Beta:", exc)
        idx = np.arange(n, dtype=np.float64)
        x = (idx + 0.5) / n
        raw = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
        a = raw / raw.sum()
        return a

    # regularised incomplete beta at the bin edges
    edges = np.linspace(0.0, 1.0, n + 1, dtype=np.float64)
    cdf_vals = betainc(alpha, alpha, edges)   # shape (n+1,)
    a = cdf_vals[1:] - cdf_vals[:-1]
    total = a.sum()
    if total != 0.0:
        a /= total
    return a


def _run_adam_opt(a_start: np.ndarray, n: int, deadline: float,
                  max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Perform Adam optimisation (logits → soft‑max) on the smoothed max‑conv.
    Returns the best probability vector seen and its cheap score
    (2·n·max(conv)).
    """
    # Initialise Adam on logits
    z = _logits_from_distribution(a_start)
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_alpha = 0.6           # base learning‑rate for logits
    eps_factor = 0.5           # initial smoothing factor
    eps_factor_min = 5e-4
    eps_factor_decay = 0.9999

    best_a = a_start.copy()
    best_score = 2.0 * n * _conv_max(best_a)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        a_cur = _softmax(z)

        max_conv, grad_a = _conv_max_and_grad(a_cur, eps_factor)
        cur_score = 2.0 * n * max_conv

        if cur_score < best_score:
            best_score = cur_score
            best_a = a_cur.copy()

        # Mirror‑descent gradient on logits (softmax Jacobian)
        mu = np.dot(grad_a, a_cur)
        grad_z = a_cur * (grad_a - mu)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)

        # Bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # Step‑size schedule (inverse‑sqrt decay)
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step = -alpha_t * m_hat / (np.sqrt(v_hat) + eps_adam)

        z = z + step
        # Keep logits in a safe range for exp()
        z = np.clip(z, -50.0, 50.0)

        # Decay the smoothing constant
        eps_factor = max(eps_factor_min, eps_factor * eps_factor_decay)

        # Occasional tiny jitter to shake off tiny plateaus
        if it % 8000 == 0 and np.random.rand() < 0.2:
            z += 0.01 * np.random.randn(n)

        # Very early stop if we already beat an ambitious target
        if best_score < 1.2:
            break

    return best_a, best_score

    betas = [0.35, 0.45, 0.55, 0.65, 0.8]
    for alpha in betas:
        a_beta = _beta_integrated_distribution(alpha, n)
        init_vectors.append(a_beta)
        print(f"Created integrated Beta(α={alpha}) start vector.")

    # Random Dirichlet starts (to increase diversity)
    for _ in range(2):
        conc = np.random.uniform(0.3, 1.5)
        a_rand = np.random.dirichlet(np.full(n, conc))
        init_vectors.append(a_rand)
        print(f"Created random Dirichlet start (conc={conc:.3f}).")

    # --------------------------------------------------------------
    # 2. Evaluate cheap scores for the raw starts and keep the best.
    # --------------------------------------------------------------
    best_a = None
    best_score = float("inf")
    for idx, a in enumerate(init_vectors):
        cur_score = 2.0 * n * _conv_max(a)
        print(f"Start {idx+1}/{len(init_vectors)} cheap score = {cur_score:.6f}")
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
    print(f"Best initial cheap score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3. Run Adam refinements, respecting the remaining time budget.
    # --------------------------------------------------------------
    # Shuffle the start list to give each a fair chance.
    np.random.shuffle(init_vectors)

    for sv_idx, start_vec in enumerate(init_vectors):
        if time.time() >= deadline:
            print("Time budget exhausted – stopping early.")
            break
        print(f"Optimisation run {sv_idx+1}/{len(init_vectors)} (n={n})")
        # Run Adam with a per‑run time budget (approx. remaining/2)
        per_run_deadline = min(deadline, time.time() + (deadline - time.time()) * 0.5)
        a_opt, score_opt = _run_adam_opt(start_vec, n, per_run_deadline)
        print(f"  Run result cheap score = {score_opt:.6f}")

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"  New global best cheap score = {best_score:.6f}")

        # If we have already reached a very good bound we can stop.
        if best_score < 1.2:
            print("Excellent bound reached – terminating early.")
            break

    # --------------------------------------------------------------
    # 4. Return the best sequence found (as a plain Python list).
    # --------------------------------------------------------------
    final_seq = best_a.tolist()

    # Final exact evaluation (only for logging – not required for correctness)
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    return final_seq
