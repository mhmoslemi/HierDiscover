def propose_candidate(...):

    set seed, start time, deadline.

    # Determine n
    try:
        prev = height_sequence_1
        n0 = len(prev)
    except:
        prev = None
        n0 = 5000

    # Choose n: maybe min(max(n0, 8000), 25000)
    n = min(max(n0, 8000), 25000)   # aim bigger than previous.

    fft_len = 1 << ((3*n - 2).bit_length())
    # Preallocate arrays?

    best_a = None
    best_score = float('inf')

    # Helper functions

    def softmax(y):
        max_y = np.max(y)
        ex = np.exp(y - max_y)
        return ex / ex.sum()

    # Compute cheap_score function
    def cheap_score(a):
        # assume sum(a)=1
        a_fft = np.fft.fft(a, fft_len)
        conv = np.fft.ifft(a_fft * a_fft).real[:2*n-1]
        return 2.0 * n * np.max(conv)

    # initial seeds: uniform, previous (scaled), random Dirichlet, beta families maybe.

    candidates = []

    # uniform
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # previous upsample if length differs:
    if prev is not None:
        if len(prev) != n:
            # upsample via linear interpolation of cumulative distribution
            x_prev = np.arange(len(prev))
            x_new = np.linspace(0, len(prev)-1, n)
            prev_arr = np.interp(x_new, x_prev, prev)
            prev_arr = np.maximum(prev_arr, 0)
            s = prev_arr.sum()
            if s > 0:
                prev_arr /= s
            candidates.append(prev_arr)
        else:
            arr = np.array(prev, dtype=np.float64)
            arr = np.maximum(arr,0)
            s = arr.sum()
            if s>0:
                arr/=s
            candidates.append(arr)

    # Dirichlet seeds
    for conc in [0.01, 0.1, 0.5, 1.0]:
        g = np.random.gamma(conc, 1.0, size=n)
        g /= g.sum()
        candidates.append(g)

    # Beta shape seeds (coarse)
    x = np.arange(1, n+1, dtype=np.float64)/n
    for alpha in [0.5, 1.0, 2.0]:
        for beta in [0.5, 1.0, 2.0]:
            cand = (x**alpha) * ((1-x)**beta)
            s=cand.sum()
            if s>0:
                cand/=s
                candidates.append(cand)

    # Evaluate each candidate cheap_score, keep best.
    for a in candidates:
        val = cheap_score(a)
        if val < best_score:
            best_score = val
            best_a = a.copy()
    print(f"[Init] basic best cheap_score = {best_score:.6f}")

    # Optimization loop: multiple restarts?
    num_restarts = 3
    for restart in range(num_restarts):
        # Initialize y from a random Dirichlet.
        if restart == 0:
            y = np.log(best_a + 1e-30)   # start from best found
        else:
            g = np.random.gamma(1.0, 1.0, size=n)
            a0 = g / g.sum()
            y = np.log(a0 + 1e-30)

        # Adam parameters
        m = np.zeros_like(y)
        v = np.zeros_like(y)
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8
        lr0 = 0.1   # base learning rate
        iter_count = 0
        last_best_iter = 0
        best_iter_a = None
        best_iter_val = float('inf')
        # schedule for α (softmax temperature)
        alpha_start = 5.0
        alpha_end = 150.0
        total_iters_target = 200000   # approximate

        while time.time() < deadline and iter_count < 500000:
            iter_count += 1
            # compute a
            max_y = np.max(y)
            a = np.exp(y - max_y)
            sum_a = a.sum()
            a /= sum_a

            # compute fft of a
            a_fft = np.fft.fft(a, fft_len)

            # autocorrelation conv
            conv = np.fft.ifft(a_fft * a_fft).real[:2*n-1]

            # cheap score
            maxb = conv.max()
            cheap = 2.0 * n * maxb

            # update best
            if cheap < best_iter_val - 1e-12:
                best_iter_val = cheap
                best_iter_a = a.copy()
                last_best_iter = iter_count
                print(f"[Restart {restart} iter {iter_count}] New best cheap = {cheap:.6f}")

            # compute softmax weights for conv
            alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, iter_count / total_iters_target)
            # scale conv values to avoid overflow
            c_scaled = alpha * conv
            c_max = c_scaled.max()
            w = np.exp(c_scaled - c_max)
            w /= w.sum()

            # compute gradient wrt a: g_a = 2 * convolution of a with reversed w
            w_rev = w[::-1]
            # convolution using FFT
            w_rev_fft = np.fft.fft(w_rev, fft_len)
            grad_full = np.fft.ifft(a_fft * w_rev_fft).real
            # grad_full length = n + len(w_rev) -1 = n + (2n-1) -1 = 3n-2
            # extract g_i = 2 * grad_full[(len(w_rev)-1) - i] i=0..n-1
            offset = len(w_rev) - 1  # =2n-2
            g_a = 2.0 * grad_full[offset - np.arange(n)]

            # chain rule for softmax param y
            dot = np.dot(a, g_a)
            grad_y = a * (g_a - dot)

            # Adam update
            m = beta1 * m + (1 - beta1) * grad_y
            v = beta2 * v + (1 - beta2) * (grad_y ** 2)
            m_hat = m / (1 - beta1 ** iter_count)
            v_hat = v / (1 - beta2 ** iter_count)
            lr = lr0 / np.sqrt(iter_count)
            y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

            # occasional perturb if no improvement for many steps
            if iter_count - last_best_iter > 50000:
                perturb = 0.01 * np.random.randn(n)
                y += perturb
                last_best_iter = iter_count
                print(f"[Restart {restart} iter {iter_count}] Perturb due to stagnation")

        # after loop, compare best_iter_a with global best
        if best_iter_a is not None:
            val = cheap_score(best_iter_a)
            if val < best_score:
                best_score = val
                best_a = best_iter_a.copy()
                print(f"[Restart {restart}] Improvement after optimization: {val:.6f}")

        # If time is low, break
        if time.time() >= deadline:
            break

    # Final evaluation using evaluate_sequence
    final_seq = best_a.tolist()
    try:
        final_val = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_val:.6f}")
    except Exception as e:
        final_val = float('inf')
        print(f"Error final evaluation: {e}")

    return final_seq

#!/usr/bin/env python
"""
Propose a non‑negative step function (list of heights) that minimises

        2·n·max_k (a*a)[k] / (Σ a_i)²

The algorithm consists of three stages:

1. **Initialisation** – generate a few sensible seeds (uniform, the
   previously‑found sequence, Dirichlet samples and simple beta‑shaped
   families).  The best of these seeds provides a starting point.

2. **Smooth max optimisation** – we treat the objective
   ``max(conv)`` as a smoothed ``log‑sum‑exp`` of the convolution entries.
   The sequence is parametrised by unrestricted log‑weights ``y`` with
   ``a = softmax(y)`` so the sum constraint is always satisfied.
   An Adam optimiser updates ``y`` using the gradient of the
   smoothed objective, which can be written as a convolution of the
   current weights with the softmax‑distribution over the convolution
   indices.  The temperature (the ``α`` factor) is increased
   gradually so the optimisation first sees a smooth landscape and
   later focuses on the true maximum.

3. **Refinement** – a short mirror‑descent (exponentiated‑gradient)
   phase that directly uses the active (top‑k) convolution entries,
   similar to the previous implementation, to push the last few
   decimal places down.

The whole routine respects the supplied time budget (default 1000 s)
and always returns a feasible sequence.  The implementation uses only
NumPy and the provided ``evaluate_sequence`` function.  No closures or
file/network access are used.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (full length 2·n‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa).real[: 2 * a.size - 1]
    return conv


def _conv_fft(a: np.ndarray, b: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*b using FFT (full length len(a)+len(b)-1)."""
    fa = np.fft.fft(a, fft_len)
    fb = np.fft.fft(b, fft_len)
    conv = np.fft.ifft(fa * fb).real[: a.size + b.size - 1]
    return conv


def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def _cheap_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Cheap objective assuming Σa_i = 1 → 2·n·max(conv)."""
    conv = _conv_full_fft(a, fft_len)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that
    minimises ``evaluate_sequence``.  The routine adapts its schedule to
    the available time budget and always returns a feasible sequence.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a safety margin for the final exact evaluation
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # Choose a working length.
    # --------------------------------------------------------------
    try:
        prev = height_sequence_1                      # global from earlier runs
        prev_len = len(prev)
    except Exception:
        prev = None
        prev_len = 0

    # We aim for a longer vector than before (more degrees of freedom)
    n = max(prev_len, 8000)          # at least 8k entries
    n = min(n, 25000)                # never exceed the safe upper bound

    # FFT length – next power of two ≥ 3·n (covers both a*a and a*w_rev)
    fft_len = 1 << ((3 * n).bit_length())

    # --------------------------------------------------------------
    # Initialise a set of candidate sequences (all normalised to Σ=1)
    # --------------------------------------------------------------
    candidates = []

    # uniform
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # previous solution (up‑sample / tile if needed)
    if prev is not None:
        prev_arr = np.asarray(prev, dtype=np.float64)
        prev_arr = np.maximum(prev_arr, 0.0)
        if prev_arr.sum() > 0.0:
            prev_arr /= prev_arr.sum()
        else:
            prev_arr = None
        if prev_arr is not None:
            if prev_len != n:
                # simple tiling to reach length n
                repeats = n // prev_len + 1
                temp = np.tile(prev_arr, repeats)[:n]
                temp /= temp.sum()
                candidates.append(temp)
            else:
                candidates.append(prev_arr.copy())

    # a few Dirichlet seeds (wide range of concentration)
    for conc in (0.001, 0.01, 0.1, 0.5, 1.0, 2.0):
        if time.time() >= deadline:
            break
        g = np.random.gamma(conc, 1.0, size=n)
        g /= g.sum()
        candidates.append(g)

    # beta‑shaped families: a_i ∝ x_i^α (1−x_i)^β
    x = (np.arange(1, n + 1, dtype=np.float64) / n)
    for alpha in np.linspace(0.2, 5.0, 9):
        for beta in np.linspace(0.2, 5.0, 9):
            if time.time() >= deadline:
                break
            cand = (x ** alpha) * ((1.0 - x) ** beta)
            s = cand.sum()
            if s == 0.0:
                continue
            cand /= s
            candidates.append(cand)

    # keep the best of the cheap candidates
    best_a = None
    best_val = float("inf")
    for a in candidates:
        val = _cheap_score(a, fft_len, n)
        if val < best_val:
            best_val = val
            best_a = a.copy()
    print(f"[Init] best cheap score = {best_val:.6f}")

    # --------------------------------------------------------------
    # 1. Smooth‑max optimisation (Adam on log‑softmax parametrisation)
    # --------------------------------------------------------------
    # Global best (across restarts)
    global_best_a = best_a.copy()
    global_best_val = best_val

    # optimisation hyper‑parameters
    total_iters_target = 200_000          # schedule length for temperature
    alpha_start = 0.5                     # low temperature → smooth
    alpha_end = 30.0                      # higher temperature → focus on max
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.12                            # base learning rate for Adam
    max_restarts = 3
    topk_restart = 50                    # for occasional hard‑max phase

    for restart in range(max_restarts):
        if time.time() >= deadline:
            break

        # Initialise log‑weights y
        if restart == 0 and best_a is not None:
            y = np.log(best_a + 1e-30)
        else:
            # fresh random Dirichlet start
            g = np.random.gamma(1.0, 1.0, size=n)
            a0 = g / g.sum()
            y = np.log(a0 + 1e-30)

        m = np.zeros_like(y)
        v = np.zeros_like(y)

        iter_cnt = 0
        last_improve = 0
        best_iter_a = None
        best_iter_val = float("inf")

        while time.time() < deadline and iter_cnt < 800_000:
            iter_cnt += 1

            # ------- obtain probability vector a -------
            a = _softmax(y)                     # guarantees Σa = 1, a_i > 0

            # ------- convolution a*a (FFT) -------
            a_fft = np.fft.fft(a, fft_len)
            conv = np.fft.ifft(a_fft * a_fft).real[: 2 * n - 1]

            # cheap objective (max(conv) with Σ=1)
            maxb = conv.max()
            cheap = 2.0 * n * maxb

            # record best during this run
            if cheap < best_iter_val - 1e-12:
                best_iter_val = cheap
                best_iter_a = a.copy()
                last_improve = iter_cnt
                print(f"[Restart {restart} iter {iter_cnt}] NEW BEST cheap = {cheap:.6f}")

            # ------- temperature (α) schedule -------
            alpha = alpha_start + (alpha_end - alpha_start) * min(1.0,
                                                               iter_cnt / total_iters_target)

            # ------- soft‑max weights over convolution entries -------
            # scale convolution so that differences are O(1)
            conv_scaled = conv * n
            cmax = conv_scaled.max()
            # exponential weights (numerically stable)
            w = np.exp(alpha * (conv_scaled - cmax))
            w_sum = w.sum()
            if w_sum == 0.0 or not np.isfinite(w_sum):
                # fall back to uniform weighting
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # ------- gradient wrt a (2 * a * w_rev) -------
            w_rev = w[::-1]                     # length 2·n‑1
            w_rev_fft = np.fft.fft(w_rev, fft_len)
            grad_full = np.fft.ifft(a_fft * w_rev_fft).real
            grad_full = grad_full[: n + (2 * n - 1) - 1]   # linear convolution length 3·n‑2
            offset = (2 * n - 2)                # = len(w_rev) - 1
            g_a = 2.0 * grad_full[offset - np.arange(n)]

            # ------- chain rule for softmax parametrisation -------
            dot = np.dot(a, g_a)
            grad_y = a * (g_a - dot)

            # ------- Adam update -------
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
            m_hat = m / (1.0 - beta1 ** iter_cnt)
            v_hat = v / (1.0 - beta2 ** iter_cnt)
            lr = lr0 / math.sqrt(iter_cnt + 1)
            y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

            # ------- occasional hard‑max restarts (avoid stagnation) -------
            if iter_cnt - last_improve > 120_000:
                # inject a small random perturbation and project back onto simplex
                perturb = 0.02 * np.random.randn(n)
                y = np.log(_simplex_projection(np.exp(y) + perturb) + 1e-30)
                last_improve = iter_cnt
                print(f"[Restart {restart} iter {iter_cnt}] Perturb applied (stagnation).")

        # after this restart, compare its best with global best
        if best_iter_a is not None:
            val = _cheap_score(best_iter_a, fft_len, n)
            if val < global_best_val:
                global_best_val = val
                global_best_a = best_iter_a.copy()
                print(f"[Restart {restart}] global improvement: cheap = {val:.6f}")

    # --------------------------------------------------------------
    # 2. Final mirror‑descent refinement (hard max, top‑k sub‑gradient)
    # --------------------------------------------------------------
    a = global_best_a.copy()
    best_ref_a = a.copy()
    best_ref_val = global_best_val
    last_improve = 0
    iter_cnt = 0
    topk = 5
    eta0 = 0.6

    while time.time() < deadline and iter_cnt < 200_000:
        iter_cnt += 1

        conv = _conv_full_fft(a, fft_len)
        maxb = conv.max()
        cheap = 2.0 * n * maxb

        if cheap < best_ref_val - 1e-12:
            best_ref_val = cheap
            best_ref_a = a.copy()
            last_improve = iter_cnt
            print(f"[Refine iter {iter_cnt}] NEW BEST cheap = {cheap:.6f}")

        # sub‑gradient from the top‑k convolution entries
        top_idx = np.argpartition(conv, -topk)[-topk:]
        g = np.zeros_like(a)
        for idx in top_idx:
            lo = max(0, idx - (n - 1))
            hi = min(idx, n - 1)
            rng = np.arange(lo, hi + 1, dtype=np.int64)
            g[rng] += 2.0 * a[idx - rng]
        g /= float(topk)

        # exponentiated‑gradient step
        eta = eta0 / math.sqrt(iter_cnt)
        log_a = np.log(a + 1e-30)
        log_a -= eta * g
        a = np.exp(log_a)
        a /= a.sum()

        # occasional restart if no improvement for a while
        if iter_cnt - last_improve > 80_000:
            pert = 0.01 * np.random.rand(n)
            a = _simplex_projection(a + pert)
            last_improve = iter_cnt
            print(f"[Refine iter {iter_cnt}] Restart due to stagnation.")

    # --------------------------------------------------------------
    # 3. Final exact evaluation (uses the provided ``evaluate_sequence``)
    # --------------------------------------------------------------
    final_seq = best_ref_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:      # defensive – should never happen
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # --------------------------------------------------------------
    # Return the best sequence found.
    # --------------------------------------------------------------
    return final_seq
