"""
Stepped‐function optimiser (budget‑aware).

The optimiser works in three stages:

1.  A fast “Adam on the simplex” optimiser is run for a few different
    support sizes.  The objective is the exact value
    2·n·max(conv(a,a)) (a is kept normalised to sum = 1).  The max is
    approximated by a smooth soft‑max whose sharpness β is annealed
    from a moderate value to a very large one (≈10⁵).  Adam’s
    adaptive moments together with a natural‑gradient step in log‑it
    space give a stable descent on the simplex.

2.  The best sequence from stage 1 is refined by a focussed simulated
    annealing routine.  At every iteration the index `k` where the
    convolution attains its maximum is identified; the contribution
    of each coefficient to that entry is used as a proposal distribution
    for moving a small amount of mass from a “high‑contributing’’ index
    to a random other index.  A Metropolis acceptance rule with an
    exponentially decaying temperature is applied.  In addition to the
    basic move, occasional “swap’’ moves (exchanging the values of two
    indices) and global Gaussian jitter are performed to escape plateaus.

3.  Tiny entries (≤1e‑12) are pruned, the sequence is re‑normalised and
    the final objective is reported via ``evaluate_sequence``.

All work respects the overall time budget (default 1000 s).  The
implementation uses only NumPy and the standard library, runs on a
single process (max 2 CPU threads can be used implicitly by NumPy’s FFT)
and returns a plain ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper routines (FFT convolution, soft‑max, objective, etc.)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a * a using real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return conv_full[: 2 * n - 1]


def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation g_i = Σ_t w[t]·a[t‑i]  (i from 0 to len(a)-1).
    Implemented via FFT (real‑valued).
    """
    n = a.shape[0]
    m = w.shape[0]                       # = 2·n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) / (sum(a)**2).
    The sequence ``a`` is assumed non‑negative.
    """
    n = a.shape[0]
    sum_a = a.sum()
    if sum_a < 1e-12:
        return np.inf
    conv = _conv_self_fft(a)
    return float(2.0 * n * conv.max() / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑it parametrisation)
# ----------------------------------------------------------------------


def _run_adam(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
    beta_start: float = 1.0,
    beta_end: float = 1e5,
    lr_start: float = 0.25,
    lr_end: float = 0.001,
) -> tuple[list[float], float]:
    """
    Adam optimiser for a fixed support size ``n``.
    The sequence is kept on the probability simplex (sum = 1).
    Returns the best sequence (as list) and its exact objective.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # initialise a (either provided or random)
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
        a = np.clip(a, 0.0, None)
        a_sum = a.sum()
        if a_sum < 1e-12:
            a = np.ones_like(a) / n
        else:
            a = a / a_sum
    else:
        a = rng.random(n)
        a = a / a.sum()

    # logits for the soft‑max parametrisation
    x = np.log(a + 1e-16)

    # Adam momentums
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # bookkeeping
    best_a = a.copy()
    best_obj = _objective(best_a)

    it = 0
    while time.time() - start < time_limit:
        it += 1

        # ----- simplex step -------------------------------------------------
        a = _softmax(x)                     # a stays non‑negative, sum = 1

        # ----- convolution & smooth max weight -------------------------------
        conv = _conv_self_fft(a)
        M = conv.max()
        obj = float(2.0 * n * M)           # denominator is 1 (sum(a)=1)
        if obj < best_obj - 1e-14:         # record improvement
            best_obj = obj
            best_a = a.copy()
            print(f"[Adam n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        # ----- annealing schedules -----------------------------------------
        t = min((time.time() - start) / time_limit, 1.0)
        beta = beta_start * np.exp(np.log(beta_end / beta_start) * t)
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t)

        # smooth‑max weighting (log‑sum‑exp approximation)
        if beta > 5e4:                     # very sharp → hard max
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            # shifted values are ≤0, so exponentials are ≤1
            shifted = conv - M
            w = np.exp(beta * shifted)
            s = w.sum()
            if s == 0.0:
                w = np.ones_like(conv) / conv.size
            else:
                w /= s

        # ----- gradient of smooth max wrt a (2·corr) ------------------------
        g_a = 2.0 * _correlation_fft(w, a)

        # natural gradient in log‑it space (mirrored gradient)
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)

        # ----- Adam update -------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- occasional jitter / restart ---------------------------------
        if it % 3000 == 0:
            jitter = rng.normal(scale=0.008, size=n)
            x += jitter
        if it % 50000 == 0:
            # restart from the current best (adds a small random perturbation)
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.003, size=n)
            m.fill(0.0)
            v.fill(0.0)

    # ---- post‑processing: prune negligible entries -------------------------
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = _objective(best_a)
        print(f"[Adam compress] length → {best_a.size}, O={best_obj:.6f}")

    return best_a.tolist(), best_obj


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (targeted mass moves)
# ----------------------------------------------------------------------


def _run_sa(
    init_seq: list[float],
    time_limit: float,
    seed: int = 12345,
) -> tuple[list[float], float]:
    """
    Targeted simulated‑annealing starting from ``init_seq``.
    Moves are of two kinds:
        * transfer a small fraction of mass from a high‑contributing index
          (w.r.t. the current maximal convolution entry) to a random index;
        * swap the values of two randomly chosen indices.
    Returns the best sequence found and its exact objective.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    a = np.asarray(init_seq, dtype=float)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-12:
        a = np.ones_like(a) / a.size
    else:
        a = a / s
    n = a.shape[0]

    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())
    best_seq = a.copy()

    # temperature schedule (exponential decay)
    T = max(best_obj * 0.2, 1e-6)
    T_min = best_obj * 1e-6 + 1e-9
    decay = 0.997

    it = 0
    while time.time() - start < time_limit:
        it += 1

        # --- locate the current maximal convolution entry -----------------
        k_max = int(np.argmax(conv))

        # contributions a[i]·a[k_max‑i] (zero outside range)
        i_arr = np.arange(n)
        j_arr = k_max - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i_arr[mask]] * a[j_arr[mask]]
        total_contrib = contrib.sum()

        # source index: biased toward large contributions
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = rng.choice(n, p=probs)
        else:
            i = rng.integers(n)

        # destination index: uniform but ≠ i
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # magnitude of transfer (temperature‑dependent)
        # At high temperature we allow larger jumps.
        max_fraction = 0.07 * (T / (best_obj + 1e-12) + 0.1)
        delta = a[i] * rng.random() * max_fraction
        if delta <= 0.0:
            continue

        # Decide whether to perform a swap instead of a transfer
        if rng.random() < 0.02:          # 2 % chance of a pure swap
            a_new = a.copy()
            a_new[i], a_new[j] = a_new[j], a_new[i]
        else:
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta

        # re‑normalise (protect against tiny negatives)
        a_new = np.clip(a_new, 0.0, None)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-12:
            continue
        a_new /= a_new_sum

        conv_new = _conv_self_fft(a_new)
        obj_new = float(2.0 * n * conv_new.max())

        # --- Metropolis acceptance ----------------------------------------
        if obj_new < best_obj - 1e-12:
            a, conv, best_obj, best_seq = a_new, conv_new, obj_new, a_new.copy()
            print(f"[SA it={it}] new best O={best_obj:.6f}")
        else:
            prob = np.exp(-(obj_new - best_obj) / (T + 1e-16))
            if rng.random() < prob:
                a, conv = a_new, conv_new

        # --- temperature update -------------------------------------------
        if it % 2500 == 0:
            T = max(T * decay, T_min)

        # --- occasional global jitter (helps escape plateaus) -------------
        if it % 15000 == 0:
            noise = rng.normal(scale=0.008, size=n)
            a = np.clip(a + noise, 0.0, None)
            a_sum = a.sum()
            if a_sum > 0.0:
                a /= a_sum
            conv = _conv_self_fft(a)

    # ---- final compression -------------------------------------------------
    thresh = 1e-12
    mask = best_seq > thresh
    if mask.sum() < best_seq.size:
        comp = best_seq[mask]
        comp /= comp.sum()
        comp_obj = float(2.0 * comp.size * _conv_self_fft(comp).max())
        if comp_obj < best_obj:
            best_seq, best_obj = comp, comp_obj
            print(f"[SA compress] length → {comp.size}, O={best_obj:.6f}")

    return best_seq.tolist(), best_obj


# ----------------------------------------------------------------------
# Main entry point expected by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **_ignore,
) -> list[float]:
    """
    Orchestrates the multi‑stage search.
    - Several Adam runs for different support sizes (mall to large).
    - The best candidate is refined by the simulated‑annealing routine.
    - Returns the final list of non‑negative heights.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Load a previously‑found construction if available (fallback: None)
    # ------------------------------------------------------------------
    init_seq = None
    if 'height_sequence_1' in globals():
        seq0 = np.asarray(height_sequence_1, dtype=float)
        seq0 = np.clip(seq0, 0.0, None)
        if seq0.sum() < 0.01:
            seq0 = np.ones_like(seq0)
        seq0 = seq0 / seq0.sum()
        init_seq = seq0

    # ------------------------------------------------------------------
    # Define a set of support sizes.  Larger sizes give more flexibility
    # but take longer per iteration.  We allocate a modest share of the
    # budget to the small sizes (they converge quickly) and the bulk to
    # the largest size.
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 6000, 8000, 10000]      # increasing flexibility
    if init_seq is not None and init_seq.size not in lengths:
        lengths.insert(0, init_seq.size)

    best_seq = None
    best_obj = float('inf')

    # ------------------------------------------------------------------
    # Phase 1 – Adam optimiser runs (one run per length, different seeds)
    # ------------------------------------------------------------------
    for idx, n in enumerate(lengths):
        now = time.time()
        remaining = deadline - now
        if remaining <= 2.0:
            break

        # allocate time: a larger share for the longest length
        if idx == len(lengths) - 1:
            run_time = remaining * 0.55
        else:
            run_time = remaining * 0.08   # small quick runs for the short lengths

        # initialise from previous construction if length matches
        init_a = None
        if init_seq is not None and init_seq.size == n:
            init_a = init_seq

        seq, val = _run_adam(
            n=n,
            time_limit=run_time,
            init_a=init_a,
            seed=seed + idx * 271,
            beta_start=1.0,
            beta_end=1e5,
            lr_start=0.25,
            lr_end=0.001,
        )
        if val < best_obj:
            best_obj, best_seq = val, np.asarray(seq, dtype=float)
            print(f"[global] new overall best O={best_obj:.6f} (n={n})")

        if time.time() >= deadline - 0.5:
            break

    # ------------------------------------------------------------------
    # Phase 2 – targeted simulated‑annealing refinement (if time permits)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if best_seq is not None and remaining > 5.0:
        seq_refined, val_refined = _run_sa(
            best_seq.tolist(),
            time_limit=remaining,
            seed=seed + 9999,
        )
        if val_refined < best_obj:
            best_obj, best_seq = val_refined, np.asarray(seq_refined, dtype=float)
            print(f"[final] SA improved O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Fallback guard – should never happen, but ensures a non‑empty result.
    # ------------------------------------------------------------------
    if best_seq is None or best_seq.size == 0:
        best_seq = np.ones(100, dtype=float) / 100.0
        best_obj = _objective(best_seq)

    # ------------------------------------------------------------------
    # Final sanity check (calls the provided evaluate_sequence)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_seq.tolist()
