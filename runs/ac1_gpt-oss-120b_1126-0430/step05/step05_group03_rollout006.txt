def _run_opt_l2(n, time_limit, seed, init_a=None):
    rng = np.random.default_rng(seed)
    if init_a is None:
        a = rng.random(n)
    else:
        a = np.asarray(init_a)
    a = np.maximum(a, 0)
    a = a / a.sum()
    x = np.log(a + 1e-16)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1,beta2=0.9,0.999
    eps=1e-12

    start = time.time()
    best_a = a.copy()
    best_obj = _objective(a)
    it=0

    # schedule steps
    target_iters = 500000  # high
    while time.time() - start < time_limit:
        it += 1
        a = _softmax(x)      # compute probabilities
        # compute conv, eval objective
        conv = _conv_self_fft(a)
        obj = 2.0 * n * conv.max()
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            print(f"[n={n}, it={it}] new best O={best_obj}")

        # compute gradient of J
        avg = 1.0 / (2 * n - 1)
        diff = conv - avg
        # correlation gradient
        grad_J = 4.0 * _correlation_fft(diff, a)  # shape (n,)
        # natural gradient on simplex
        dot = np.dot(a, grad_J)
        g_x = a * ( -grad_J + dot )  # negative because we want descent

        # update Adam
        m = beta1 * m + (1-beta1) * g_x
        v = beta2 * v + (1-beta2) * (g_x ** 2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        # learning rate schedule decays
        lr = 0.2 * (1 - it/target_iters)**2 + 0.001 # maybe polynomial decay
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # jitter etc.
        if it % 3000 == 0:
            jitter = rng.normal(scale=0.01, size=n)
            x += jitter
        if time.time() - start > time_limit * 0.9:
            # maybe break early
            pass

    # compress negligible entries
    mask = best_a > 1e-12
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = _objective(best_a)
    # final local refinement (mass transfer) as before
    # ...
    return best_a.tolist(), best_obj

"""
Propose a step‑function sequence with a low evaluation score.

Algorithm
---------
1. Pick a support size `n` (default 8000).  Larger `n` gives a lower
   theoretical lower bound 2 n/(2 n‑1) ≈ 1, but each iteration costs
   O(n log n).

2. Stage 1 – *flatten* the convolution:
   Minimise the squared L2 distance between the full self‑convolution
   `c = a * a` and the constant value `μ = 1/(2 n‑1)` (the average of `c`
   when `∑a = 1`).  The objective

        J(a) = Σ_k (c_k – μ)²

   has gradient `∇J = 4 * corr(c‑μ, a)`.  We perform Adam on the
   log‑it parametrisation of the simplex, with occasional jitter and
   restarts, keeping the best vector found.

3. Stage 2 – *tighten* the maximum:
   Initialise the original Adam optimiser (which directly minimises the
   smoothed max of the convolution) with the result of stage 1 and run
   it for the remaining time budget.

4. Final cheap stochastic local search (random mass‑transfer moves)
   refines the best candidate.

All numbers are projected onto the simplex (non‑negative, sum = 1) after
each update, satisfying the security checks of `evaluate_sequence`.
The function returns the best sequence before the time limit expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using real FFT (length 2·n‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv


def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation w ⋆ a, 0 ≤ i < len(a).
    Implements Σ_t w[i+t]·a[t] via FFT.
    """
    n = a.shape[0]
    m = w.shape[0]                     # = 2·n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


def _objective(a: np.ndarray) -> float:
    """Exact evaluation 2·n·max(convolution) (∑a is assumed 1)."""
    n = a.size
    conv = _conv_self_fft(a)
    return 2.0 * n * conv.max()

# Stage 2 – original Adam optimiser that works on the smoothed max.
def _run_opt(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
) -> tuple[list[float], float]:
    """Adam optimisation of a length‑`n` sequence (max‑convolution objective)."""
    start = time.time()
    rng = np.random.default_rng(seed)

    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
        a = a / a.sum()
    else:
        a = rng.random(n)
        a = a / a.sum()

    x = np.log(a + 1e-16)               # log‑it parametrisation

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # schedules
    beta_start, beta_end = 0.5, 4000.0       # smoothing of the max
    lr_start, lr_end = 0.15, 0.001          # learning‑rate decay
    target_iters = 800_000

    best_a = a.copy()
    best_obj = _objective(best_a)
    it = 0

    while time.time() - start < time_limit:
        it += 1
        a = _softmax(x)

        conv = _conv_self_fft(a)
        M = conv.max()
        obj = 2.0 * n * M
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[max‑opt n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        # smoothed max weight vector
        t_frac = min(it / target_iters, 1.0)
        beta = beta_start * np.exp(np.log(beta_end / beta_start) * t_frac)
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t_frac)

        if beta > 3000.0:                     # sharp enough → true arg‑max
            kmax = int(np.argmax(conv))
            w = np.zeros_like(conv)
            w[kmax] = 1.0
        else:
            shifted = conv - M
            w = np.exp(beta * shifted)
            w = w / w.sum()

        g_a = 2.0 * _correlation_fft(w, a)   # gradient wrt a
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)                # natural gradient in log‑it space

        # Adam update (gradient descent)
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # jitter to escape plateaus
        if it % 6000 == 0:
            jitter = rng.normal(scale=0.02, size=n)
            x += jitter

        # periodic restart from the best seen so far
        if it % 70_000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.005, size=n)
            m.fill(0.0)
            v.fill(0.0)
            print(f"[max‑opt n={n}] restart at iter {it}")

    # prune negligible entries
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = _objective(best_a)
        print(f"[max‑opt compress] reduced length {n} → {best_a.size}, O={best_obj:.6f}")

    # short stochastic local search
    remaining = time_limit - (time.time() - start)
    if remaining > 0.5:
        a_loc = best_a.copy()
        obj_loc = best_obj
        n_loc = a_loc.size
        local_deadline = time.time() + remaining * 0.35
        while time.time() < local_deadline:
            i, j = rng.choice(n_loc, size=2, replace=False)
            delta = a_loc[i] * rng.random() * 0.02
            if delta <= 0.0:
                continue
            a_new = a_loc.copy()
            a_new[i] -= delta
            a_new[j] += delta
            conv_new = _conv_self_fft(a_new)
            obj_new = 2.0 * n_loc * conv_new.max()
            if obj_new < obj_loc:
                a_loc = a_new
                obj_loc = obj_new
                if obj_new < best_obj:
                    best_a = a_new
                    best_obj = obj_new
        print(f"[max‑opt local] final O={best_obj:.6f}")

    return best_a.tolist(), best_obj

def _run_opt_l2(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
) -> tuple[list[float], float]:
    """Adam optimiser that minimises Σ (conv - μ)² (μ = 1/(2n‑1))."""
    start = time.time()
    rng = np.random.default_rng(seed)

    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
        a = np.maximum(a, 0.0)
        a = a / a.sum()
    else:
        a = rng.random(n)
        a = a / a.sum()

    x = np.log(a + 1e-16)

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # schedules (more aggressive at the start)
    lr_start, lr_end = 0.25, 0.001
    target_iters = 600_000

    best_a = a.copy()
    best_obj = _objective(best_a)
    it = 0

    μ = 1.0 / (2 * n - 1)               # target constant value of the convolution

    while time.time() - start < time_limit:
        it += 1
        a = _softmax(x)

        conv = _conv_self_fft(a)
        obj = 2.0 * n * conv.max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[l2‑opt n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        # gradient of J = Σ (conv - μ)²
        diff = conv - μ                     # length 2n‑1
        grad_J = 4.0 * _correlation_fft(diff, a)   # shape (n,)

        # natural gradient on the simplex (log‑it space)
        dot = np.dot(a, grad_J)
        g_x = a * (grad_J - dot)

        # learning‑rate schedule (polynomial decay)
        t_frac = min(it / target_iters, 1.0)
        lr = lr_start * (1.0 - t_frac) ** 2 + lr_end

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter
        if it % 5000 == 0:
            jitter = rng.normal(scale=0.01, size=n)
            x += jitter

        # restart from best every 50 000 iterations
        if it % 50_000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.005, size=n)
            m.fill(0.0)
            v.fill(0.0)
            print(f"[l2‑opt n={n}] restart at iter {it}")

    # prune tiny entries
    mask = best_a > 1e-12
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = _objective(best_a)
        print(f"[l2‑opt compress] reduced length {n} → {best_a.size}, O={best_obj:.6f}")

    # final cheap random mass‑transfer refinement
    remaining = time_limit - (time.time() - start)
    if remaining > 0.5:
        a_loc = best_a.copy()
        obj_loc = best_obj
        n_loc = a_loc.size
        local_deadline = time.time() + remaining * 0.35
        while time.time() < local_deadline:
            i, j = rng.choice(n_loc, size=2, replace=False)
            delta = a_loc[i] * rng.random() * 0.02
            if delta <= 0.0:
                continue
            a_new = a_loc.copy()
            a_new[i] -= delta
            a_new[j] += delta
            conv_new = _conv_self_fft(a_new)
            obj_new = 2.0 * n_loc * conv_new.max()
            if obj_new < obj_loc:
                a_loc = a_new
                obj_loc = obj_new
                if obj_new < best_obj:
                    best_a = a_new
                    best_obj = obj_new
        print(f"[l2‑opt local] final O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – tighten the max using the original optimiser.
    remaining = deadline - time.time()
    if remaining > 4.0:
        seq, final_obj = _run_opt(
            n=n,
            time_limit=remaining,
            init_a=init_seq,
            seed=seed + 12345,
        )
    else:
        # fall back to whatever we have from stage 1
        seq = init_seq
        final_obj = _objective(np.asarray(seq, dtype=float))

    # ------------------------------------------------------------------
    # Final sanity check via the provided evaluate_sequence routine.
    try:
        score = evaluate_sequence(seq)
        print(f"[final] evaluate_sequence reports O={score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    # Guarantee a non‑empty return.
    if not seq:
        seq = [1.0]

    return seq
