def propose_candidate(seed=0, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0

    # parameter for length
    n = 25000

    # define families:
    def exp_one_side(n, lam):
        i = np.arange(n, dtype=np.float64)
        a = np.exp(-lam * i)
        s = a.sum()
        if s == 0:
            a[:] = 1.0 / n
        else:
            a /= s
        return a

    # also other families
    families = []
    # power law decreasing
    for p in [0.5, 0.7, 0.9, 1.2, 1.5]:
        families.append(('pow', p))
    # exp decays
    for loglam in [-5, -6, -7, -8, -9]:
        families.append(('exp', 10**loglam))
    # random dirichlet
    families.append(('rand', None))
    # also symmetric families from earlier code
    # but add as needed.

    best_a = None
    best_score = float('inf')
    # Scan families
    for name, param in families:
        if time.time() >= deadline:
            break
        if name == 'pow':
            a = _power_law_distribution(n, param)
        elif name == 'exp':
            a = exp_one_side(n, param)
        elif name == 'rand':
            # Dirichlet with alpha=1 (uniform) or maybe others
            alpha = np.random.uniform(0.1, 5.0)  # concentration
            a = np.random.gamma(alpha, 1.0, size=n)
            a_sum = a.sum()
            if a_sum == 0:
                a[:] = 1.0 / n
            else:
                a /= a_sum
        else:
            continue
        score = 2.0 * n * _conv_max(a)
        if score < best_score:
            best_score = score
            best_a = a

    # ensure best_a not None
    if best_a is None:
        best_a = np.full(n, 1.0 / n)

    # start gradient descent
    # convert to logits
    z = _logits_from_distribution(best_a)
    z += 0.005 * np.random.randn(n)

    # Adam parameters
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999

    base_alpha = 0.5

    eps_factor_decay = 0.9992

    best_seq = _softmax(z)
    best_score = 2.0 * n * _conv_max(best_seq)
    last_improve = 0
    it = 0

    while time.time() < deadline:
        it += 1
        a = _softmax(z)
        max_conv, grad_a = _conv_max_and_grad(a, eps_factor)
        cheap_score = 2.0 * n * max_conv

        if cheap_score < best_score:
            best_score = cheap_score
            best_seq = a.copy()
            last_improve = it

        # mirror descent gradient on logits
        mu = np.dot(grad_a, a)
        grad_z = a * (grad_a - mu)

        # Adam updates
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * (grad_z ** 2)

        v_hat = v / (1 - beta2 ** it)

        # step size
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)

        # line search

        trial_alpha = alpha_t

            v *= 0.7

        # restart if stagnation

            z = _logits_from_distribution(best_seq) + 0.02 * np.random.randn(n)
            m.fill(0.0)
            v.fill(0.0)

            last_improve = it

    while time.time() < deadline - 5:
        # random move
        i = np.random.randint(n)

        delta = best_seq[i] * frac

        # normalize not needed since sum preserved
        new_score = 2.0 * n * _conv_max(a_new)
        if new_score < best_score:
            best_score = new_score
            best_seq = a_new

    # final evaluation
    final_seq = best_seq.tolist()
    try:

        print(f"Final exact score = {final_score:.6f}")
    except Exception as e:
        final_score = float('inf')
        print(f"Error during evaluate_sequence: {e}")

    return final_seq

#!/usr/bin/env python
"""
Non‑negative step‑function optimiser (alternative to the reference code).

The optimiser works in three stages:

1. **Diversified initialisation** – a handful of parametric families
   (power‑law, asymmetric exponential, logistic) and many random
   Dirichlet draws are sampled; the one with the smallest cheap score
   `2·n·max(a*a)` is kept.

2. **Adam mirror‑descent on the probability simplex** – the selected start
   point is refined by mirror‑descent on the logits `z` (softmax(z) = a).
   A smoothed max (log‑sum‑exp) is used to obtain a gradient that drives the
   peak of the self‑convolution down.  The learning‑rate schedule,
   smoothing factor schedule and occasional restarts are tuned for fast
   convergence within the 1000 s budget.

3. **Local stochastic hill‑climbing** – after the gradient phase a few
   cheap random mass‑transfer moves are tried; any move that improves the
   cheap objective is kept.  This often yields a final tweak that the
   smooth optimiser misses.

The routine respects the supplied time budget, prints progress and finally
calls the provided `evaluate_sequence` to obtain the exact score.
"""

# ----------------------------------------------------------------------
# Imports
# ----------------------------------------------------------------------
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max, returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    s = e.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / z.size)
    return e / s


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑its that reproduce `a` after a soft‑max (adds a tiny epsilon)."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """Smallest power‑of‑two ≥ 3·n‑2 (enough for all needed convolutions)."""
    return 1 << ((3 * n - 2).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Returns (max_conv, grad_a) where

        grad_a = ∇_a [ 2·n·smooth_max(a*a) ]

    `eps_factor` multiplies the smoothing constant ε = eps_factor·max_conv.
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    # FFT of a (zero‑padded)
    a_fft = np.fft.rfft(a, L)

    # Linear self‑convolution a*a
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # Smoothed soft‑max weights
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv                # ≤ 0 for numerical stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # Correlation a ⋆ w   (gradient of the smoothed max)
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # Gradient (already multiplied by 4·n)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{‑p}, normalised."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _sym_product_power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ ((i+1)*(n‑i))^{‑p}, symmetric, normalised."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) * (n - i)) ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _exp_decay_one_side_distribution(n: int, lam: float) -> np.ndarray:
    """asymmetric exponential decay a_i ∝ exp(-lam·i), normalised."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _logistic_distribution(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ 1/(1+e^{α·(i‑mid)}), normalised; heavy on the left side."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = 1.0 / (1.0 + np.exp(alpha * (i - mid)))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _adam_optimize(a_init: np.ndarray, n: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (Adam) on the simplex starting from `a_init`.
    Returns the best distribution encountered and its cheap score.
    """
    # initialise logits with a tiny jitter
    z = _logits_from_distribution(a_init)
    z += 0.01 * np.random.randn(n)

    # Adam state
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    # schedules
    base_alpha = 0.5          # learning‑rate for logits
    eps_factor = 0.15         # start smoothing factor
    eps_factor_min = 5e-6
    eps_factor_decay = 0.9994

    best_seq = _softmax(z)
    best_score = 2.0 * n * _conv_max(best_seq)
    last_improve = 0
    it = 0

    # main optimisation loop
    while time.time() < deadline and it < 800_000:
        it += 1

        a = _softmax(z)

        max_conv, grad_a = _conv_max_and_grad(a, eps_factor)
        cheap_score = 2.0 * n * max_conv

        if cheap_score < best_score - 1e-14:
            best_score = cheap_score
            best_seq = a.copy()
            last_improve = it

        # mirror‑descent gradient on logits
        mu = np.dot(grad_a, a)
        grad_z = a * (grad_a - mu)

        # Adam moment updates (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step‑size schedule (inverse‑sqrt decay)
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step_dir = - m_hat / (np.sqrt(v_hat) + eps_adam)

        # back‑tracking line search (up to 4 halvings)
        accepted = False
        trial_alpha = alpha_t
        for _ in range(4):
            z_new = z + trial_alpha * step_dir
            z_new = np.clip(z_new, -60.0, 60.0)   # keep exp(z) stable
            a_new = _softmax(z_new)
            new_score = 2.0 * n * _conv_max(a_new)
            if new_score <= cheap_score:
                z = z_new
                accepted = True
                break
            trial_alpha *= 0.5

        if not accepted:
            # modest step anyway
            z = z + alpha_t * step_dir
            z = np.clip(z, -60.0, 60.0)

        # occasional jitter to escape plateaux
        if it % 5000 == 0 and np.random.rand() < 0.22:
            z += 0.012 * np.random.randn(n)
            m *= 0.6
            v *= 0.6

        # decay smoothing factor
        eps_factor = max(eps_factor_min, eps_factor * eps_factor_decay)

        # restart when stagnated for a long time
        if it - last_improve > 180_000:
            # re‑initialise around the current best candidate
            z = _logits_from_distribution(best_seq) + 0.02 * np.random.randn(n)
            m.fill(0.0)
            v.fill(0.0)
            eps_factor = 0.15
            last_improve = it

    return best_seq, best_score


def _local_hill_climb(a: np.ndarray, n: int, deadline: float) -> np.ndarray:
    """
    Very cheap stochastic hill‑climbing: move a tiny amount of mass from a
    random index i to a random index j and keep the move if it improves the
    cheap objective.  Stops when time runs out.
    """
    best_seq = a.copy()
    best_score = 2.0 * n * _conv_max(best_seq)

    while time.time() < deadline:
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j or best_seq[i] < 1e-15:
            continue
        # move at most a few percent of the source mass
        delta = best_seq[i] * np.random.rand() * 0.02
        cand = best_seq.copy()
        cand[i] -= delta
        cand[j] += delta
        new_score = 2.0 * n * _conv_max(cand)
        if new_score < best_score:
            best_score = new_score
            best_seq = cand
    return best_seq

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) improves the
    `evaluate_sequence` score.  The routine respects the supplied time budget
    (~1000 s) and combines random initialisation, Adam‑based mirror descent,
    and a final cheap hill‑climbing stage.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0   # small safety margin

    # ------------------------------------------------------------------
    # 0. Fixed problem size
    # ------------------------------------------------------------------
    n = 25000

    # ------------------------------------------------------------------
    # 1. Scan a few parametric families for a quick decent start
    # ------------------------------------------------------------------
    best_a = None
    best_score = float("inf")

    families = []

    # power‑law (decreasing)
    families.append(("pow", None))
    # asymmetric exponential decay
    families.append(("exp_one", None))
    # logistic‑type left‑heavy
    families.append(("logi", None))
    # symmetric product‑power‑law (just in case)
    families.append(("sym_pow", None))

    for name, _ in families:
        if time.time() >= deadline:
            break
        if name == "pow":
            for p in np.linspace(0.5, 1.3, 12):
                cand = _power_law_distribution(n, p)
                sc = 2.0 * n * _conv_max(cand)
                if sc < best_score:
                    best_score = sc
                    best_a = cand
        elif name == "exp_one":
            for lam in np.logspace(-5, -2, 12):
                cand = _exp_decay_one_side_distribution(n, lam)
                sc = 2.0 * n * _conv_max(cand)
                if sc < best_score:
                    best_score = sc
                    best_a = cand
        elif name == "logi":
            for alpha in np.linspace(0.2, 4.0, 15):
                cand = _logistic_distribution(n, alpha)
                sc = 2.0 * n * _conv_max(cand)
                if sc < best_score:
                    best_score = sc
                    best_a = cand
        elif name == "sym_pow":
            for p in np.linspace(0.5, 1.2, 12):
                cand = _sym_product_power_law_distribution(n, p)
                sc = 2.0 * n * _conv_max(cand)
                if sc < best_score:
                    best_score = sc
                    best_a = cand

    if best_a is None:
        best_a = np.full(n, 1.0 / n)

    print(f"[Init] Best parametric candidate cheap score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Random Dirichlet restarts + Adam refinement
    # ------------------------------------------------------------------
    # Keep the best sequence found so far
    overall_best_seq = best_a.copy()
    overall_best_score = best_score

    # Keep doing random restarts until the budget is almost exhausted
    while time.time() < deadline - 30.0:   # reserve ~30 s for final phases
        # Random Dirichlet – concentration parameter chosen logarithmically
        conc = 10 ** np.random.uniform(-1.0, 1.0)   # between 0.1 and 10
        draws = np.random.gamma(conc, 1.0, size=n)
        cand = draws / draws.sum()
        cheap = 2.0 * n * _conv_max(cand)

        # Keep the random candidate if it is already competitive
        if cheap < overall_best_score:
            overall_best_seq = cand.copy()
            overall_best_score = cheap

        # Refine this candidate with the Adam‑based optimizer
        remain = deadline - time.time()
        if remain < 5.0:
            break
        refined, refined_score = _adam_optimize(cand, n, deadline)

        if refined_score < overall_best_score:
            overall_best_seq = refined.copy()
            overall_best_score = refined_score
            print(f"[Restart] New best cheap = {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. A longer Adam run starting from the best found so far
    # ------------------------------------------------------------------
    print("[Phase] Long Adam refinement from best candidate.")
    overall_best_seq, overall_best_score = _adam_optimize(overall_best_seq, n, deadline)

    # ------------------------------------------------------------------
    # 4. Very cheap stochastic hill‑climbing for the last few seconds
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        overall_best_seq = _local_hill_climb(overall_best_seq, n, deadline)

    # ------------------------------------------------------------------
    # 5. Final exact evaluation with the provided evaluate_sequence()
    # ------------------------------------------------------------------
    final_seq = overall_best_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # ------------------------------------------------------------------
    # 6. Return the best sequence discovered.
    # ------------------------------------------------------------------
    return final_seq
