def _run_opt_hardmax(n, time_limit, init_a=None, seed=0):
    start = time.time()
    rng = np.random.default_rng(seed)
    if init_a is not None:
        a = np.copy(init_a)
        if a.size != n:
            a = rng.random(n)
    else:
        a = rng.random(n)
    a = np.clip(a, 0, None)
    a_sum = a.sum()
    if a_sum < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / a_sum

    x = np.log(a + 1e-16) # logits
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    best_a = a.copy()
    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())

    it = 0
    while time.time() - start < time_limit:
        it += 1
        # Softmax to get a
        a = _softmax(x)

        conv = _conv_self_fft(a)
        M = conv.max()
        obj = float(2.0 * n * M)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            print(f"[n={n} it={it}] new best O={best_obj:.6f}")

        k_max = int(np.argmax(conv))

        # compute gradient of max term w.r.t a
        # g_i = 2 * a_{k_max - i} if 0 <= k_max-i < n else 0
        idx = np.arange(n)
        j = k_max - idx
        mask = (j >= 0) & (j < n)
        g_a = np.zeros_like(a)
        if mask.any():
            g_a[mask] = 2.0 * a[j[mask]]
        # natural gradient in logit space
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)

        # schedule
        elapsed = time.time() - start
        t_frac = elapsed / time_limit
        # learning rate schedule (exponential)
        lr_start = 0.5
        lr_end = 0.001
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t_frac)

        # Adam update
        m = beta1 * m + (1 - beta1) * g_x
        v = beta2 * v + (1 - beta2) * (g_x ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # optional jitter
        if it % 5000 == 0:
            jitter = rng.normal(scale=0.01, size=n)
            x += jitter

    # compress small entries
    mask = best_a > 1e-12
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = float(2.0 * best_a.size * _conv_self_fft(best_a).max())
        print(f"[compress] reduced length → {best_a.size}, O={best_obj:.6f}")

    return best_a.tolist(), best_obj

"""
Non‑negative step‑function optimiser (hard‑max Adam).

The algorithm performs a projected Adam optimisation directly on the
hard max of the self‑convolution (no soft‑max smoothing).  The gradient
of the maximal convolution entry is computed analytically in O(n) time,
so each iteration requires only a single FFT for the convolution.
Most of the time budget is spent on a large support (≈12 k) to exploit
the extra flexibility of long sequences.  The result is then refined
by a targeted simulated‑annealing local search as in the previous
implementation.

Key ideas
---------
* Use the exact sub‑gradient of the maximal convolution entry:
      ∂(a*a)[k]/∂a_i = 2·a_{k−i}  (if the index is in range, else 0).
  This avoids the extra FFT that the smooth‑max version needed.
* Adam updates are performed in log‑it space (exponentiated gradient)
  which respects the simplex constraint.
* A simple learning‑rate schedule and occasional jitter/​re‑start
  help escape plateaus.
* After the main optimisation the negligible entries are pruned and a
  short simulated‑annealing phase moves a few percent of mass from
  high‑contributing indices to random locations.

The routine respects the overall time budget and returns a list of
non‑negative floats suitable for ``evaluate_sequence``.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Helper functions – FFT based convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return conv_full[: 2 * n - 1]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


def _grad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """
    Exact gradient of the k‑th convolution entry w.r.t. ``a``.
    Returns 2·a_{k‑i} for indices where it is defined, otherwise 0.
    """
    n = a.shape[0]
    idx = np.arange(n)
    j = k - idx
    mask = (j >= 0) & (j < n)
    g = np.zeros_like(a)
    if mask.any():
        g[mask] = 2.0 * a[j[mask]]
    return g


# ----------------------------------------------------------------------
# Hard‑max Adam optimiser
# ----------------------------------------------------------------------


def _run_opt_hardmax(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
) -> Tuple[List[float], float]:
    """
    Adam optimiser that always uses the exact max‑convolution gradient.
    Returns the best sequence (as a list) and its objective value
    2·n·max(conv(a,a)).
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # initialise a on the simplex
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
    else:
        a = rng.random(n)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n, dtype=float) / n
    else:
        a = a / s

    # logits for exponentiated (mirror) gradient updates
    x = np.log(a + 1e-16)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # learning‑rate schedule (exponential decay)
    lr_start = 0.5
    lr_end = 0.001

    best_a = a.copy()
    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())

    it = 0
    while time.time() - start < time_limit:
        it += 1

        # current point on the simplex
        a = _softmax(x)

        conv = _conv_self_fft(a)
        M = conv.max()
        obj = float(2.0 * n * M)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        # index of the maximal convolution entry
        k_max = int(np.argmax(conv))

        # exact sub‑gradient of that entry
        g_a = _grad_max_conv(a, k_max)

        # natural gradient in log‑it space (mirrored / exponentiated)
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)

        # time‑based learning‑rate (exponential)
        elapsed = time.time() - start
        t = min(elapsed / time_limit, 1.0)
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # Adam step in log‑it space
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to jump out of flat regions
        if it % 5000 == 0:
            jitter = rng.normal(scale=0.01, size=n)
            x += jitter

        # occasional full restart from the current best (helps escape)
        if it % 80000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.005, size=n)
            m.fill(0.0)
            v.fill(0.0)

    # ------------------------------------------------------------------
    # prune negligible entries (they increase n without helping)
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = float(2.0 * best_a.size * _conv_self_fft(best_a).max())
        print(f"[compress] reduced length → {best_a.size}, O={best_obj:.6f}")

    return best_a.tolist(), best_obj

def _run_sa(
    init_seq: List[float],
    time_limit: float,
    seed: int = 0,
) -> Tuple[List[float], float]:
    """Refine a sequence by small mass‑transfer moves with a Metropolis schedule."""
    start = time.time()
    rng = np.random.default_rng(seed)
    a = np.asarray(init_seq, dtype=float)
    if a.sum() < 1e-12:
        a = np.ones_like(a) / a.size
    else:
        a = a / a.sum()
    n = a.shape[0]

    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())
    best_seq = a.copy()

    # temperature schedule
    T = best_obj * 0.1 + 1e-6
    T_min = best_obj * 1e-6 + 1e-9
    decay = 0.995
    it = 0

    while time.time() - start < time_limit:
        it += 1
        k_max = int(np.argmax(conv))

        # contribution of each index to the maximal entry
        i_vals = np.arange(n)
        j_vals = k_max - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i_vals[mask]] * a[j_vals[mask]]
        total_contrib = contrib.sum()

        if total_contrib > 0:
            probs = contrib / total_contrib
            i = rng.choice(n, p=probs)
        else:
            i = rng.integers(n)

        # choose a distinct destination uniformly
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # transfer a modest fraction of mass from i to j
        delta = a[i] * rng.random() * 0.05  # up to 5 % of a[i]
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.clip(a_new, 0.0, None)
        a_new /= a_new.sum()

        conv_new = _conv_self_fft(a_new)
        obj_new = float(2.0 * n * conv_new.max())

        # Metropolis acceptance
        if obj_new < best_obj - 1e-12:
            a, conv, best_obj, best_seq = a_new, conv_new, obj_new, a_new.copy()
            print(f"[SA it={it}] new best O={best_obj:.6f}")
        else:
            prob = np.exp(-(obj_new - best_obj) / T) if T > 0 else 0.0
            if rng.random() < prob:
                a, conv = a_new, conv_new

        # temperature decay
        if it % 2000 == 0:
            T = max(T * decay, T_min)

        # occasional global jitter
        if it % 10000 == 0:
            noise = rng.normal(scale=0.01, size=n)
            a = np.clip(a + noise, 0.0, None)
            a /= a.sum()
            conv = _conv_self_fft(a)

    # final compression of tiny entries
    thresh = 1e-12
    mask = best_seq > thresh
    if mask.sum() < best_seq.size:
        comp = best_seq[mask]
        comp /= comp.sum()
        comp_obj = float(2.0 * comp.size * _conv_self_fft(comp).max())
        if comp_obj < best_obj:
            best_seq, best_obj = comp, comp_obj
            print(f"[SA compress] reduced length → {comp.size}, O={best_obj:.6f}")

    return best_seq.tolist(), best_obj

def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **_ignore,
) -> List[float]:
    """
    Find a non‑negative step‑function minimizing the evaluation metric.
    The routine spends most of the budget on a hard‑max Adam optimisation
    with a long support (≈12 k).  The resulting sequence is optionally
    refined by simulated annealing before being returned.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5  # small safety margin

    # ------------------------------------------------------------------
    # Initialise from a previously discovered construction if available.
    init_seq = None
    if 'height_sequence_1' in globals():
        seq0 = np.asarray(height_sequence_1, dtype=float)
        seq0 = np.clip(seq0, 0.0, None)
        if seq0.sum() < 0.01:
            seq0 = np.ones_like(seq0)
        seq0 = seq0 / seq0.sum()
        init_seq = seq0

    # ------------------------------------------------------------------
    # Choose a target length.
    # If enough time is left we go for a large support (≈12k),
    # otherwise we fall back to a more modest size.
    remaining = deadline - time.time()
    main_budget = remaining * 0.85
    sa_budget = remaining - main_budget

    target_n = 12000 if main_budget > 30.0 else 8000

    # Upsample any initial sequence to the target length.
    if init_seq is not None and init_seq.size != target_n:
        x_old = np.linspace(0.0, 1.0, init_seq.size, endpoint=False)
        x_new = np.linspace(0.0, 1.0, target_n, endpoint=False)
        init_a = np.interp(x_new, x_old, init_seq)
        init_a = np.clip(init_a, 0.0, None)
        init_a = init_a / init_a.sum()
    else:
        init_a = init_seq

    # ------------------------------------------------------------------
    # Run the hard‑max Adam optimisation.
    seq, obj = _run_opt_hardmax(
        n=target_n,
        time_limit=main_budget,
        init_a=init_a,
        seed=seed + 123,
    )
    best_seq = np.asarray(seq, dtype=float)
    best_obj = obj
    print(f"[global] after Adam best O={best_obj:.6f} (n={target_n})")

    # ------------------------------------------------------------------
    # If time permits, refine with simulated‑annealing.
    remaining = deadline - time.time()
    if remaining > 5.0:
        seq_ref, obj_ref = _run_sa(best_seq.tolist(), time_limit=remaining, seed=seed + 999)
        if obj_ref < best_obj:
            best_obj, best_seq = obj_ref, np.asarray(seq_ref, dtype=float)
            print(f"[final] after SA new best O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check – call the supplied evaluator.
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_seq.tolist()
