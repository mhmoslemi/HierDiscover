assistantfinal**Solution Overview**

The code below implements a *hybrid* optimiser for the problem  

\[
\min_{a_i\ge 0}\;O(a)=\frac{2\,n\;\max_k (a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\]

where “\(*\)” denotes the discrete convolution.  
All candidates are normalised to `∑a = 1`; the objective is then simply  

\[
O(a)=2\,n\;\max_k (a*a)_k .
\]

The optimiser works in four stages:

| stage | idea | why it helps |
|------|------|--------------|
| **0 – random restarts** | many Dirichlet draws, a *very short* exponentiated‑gradient run | diversity – you never get stuck in the basin of a single bad seed |
| **1 – exponentiated (mirror) gradient** on a *smooth* log‑sum‑exp surrogate of the max | mirror descent (KL‑projection) is the natural geometry of the simplex; the smooth surrogate makes the gradient well‑behaved and the schedule `α` → large yields an increasingly accurate max‑approximation |
| **2 – exact projected sub‑gradient descent** with a back‑tracking line‑search | once we are close to a (non‑smooth) optimum the true sub‑gradient gives a reliable descent direction |
| **3 – targeted simulated annealing** | small stochastic mass‑transfer moves that always try to touch the currently maximal convolution entry, plus occasional exact sub‑gradient corrective steps | explores the discrete neighbourhood of a good point and can escape shallow local minima |
| **4 – simple convex mixes** (uniform & reversed) | often the convex hull of two good points yields a still better point | cheap, sometimes yields the final push below the target bound |

All steps respect the supplied time budget (`budget_s`).  The implementation is **fully self‑contained**, uses only `numpy`, `math`, and the already‑imported `evaluate_sequence`, and follows the required API (`propose_candidate`).  Progress is printed so you can see how the bound evolves.

```python
"""
Hybrid optimiser for the step‑function bound.

The algorithm consists of:
  0 – cheap random restarts,
  1 – exponentiated (mirror) gradient descent on a log‑sum‑exp
      surrogate (α grows from a few to a very large value),
  2 – exact projected sub‑gradient descent with back‑tracking,
  3 – targeted simulated‑annealing with mass‑transfer moves,
  4 – a few convex mixes (uniform / reversed) as a final polish.

All updates keep the vector on the probability simplex (∑a = 1,
a_i ≥ 0).  The code is written for a single‑CPU environment but can
use up to the full 1000 s wall‑time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# 1.  Helper utilities --------------------------------------------------
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # all non‑positive → uniform point
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


def _objective(a: np.ndarray, n: int, L: int) -> float:
    """Exact objective for a simplex‑normalised vector (∑a = 1)."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. a (the true non‑smooth objective)."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


def _smooth_gradient(a: np.ndarray, conv: np.ndarray,
                     alpha: float, L: int) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp surrogate:
        f(a) = (2·n/α)·log Σ_k exp(α·conv_k) .
    The formula reduces to 2·corr(w, a) where w_k ∝ e^{α·conv_k}.
    """
    # soft‑max weights
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)      # stabilise
    w = exp_scaled / exp_scaled.sum()

    # correlation w ⋆ a   (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n_len = a.size
    grad = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]
    return grad


# ----------------------------------------------------------------------
# 2.  Exponentiated (mirror) gradient descent -------------------------
# ----------------------------------------------------------------------
def _expgradient_phase(a: np.ndarray,
                       n: int,
                       L: int,
                       max_iters: int,
                       deadline: float,
                       eta0: float = 0.6,
                       eta_decay: float = 0.9999,
                       alpha_start: float = 5.0,
                       alpha_end: float = 50000.0,
                       prune_every: int = 2000,
                       prune_percentile: float = 0.5) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Mirror descent on the smooth surrogate.
    Updates are multiplicative (a_i ← a_i·exp(‑η·∇_i)) and a is renormalised.
    Returns (final a, best objective seen, best a).
    """
    best_a = a.copy()
    best_obj = _objective(a, n, L)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()

        # schedule for the smoothing parameter α
        frac = it / max_iters
        alpha = alpha_start * (alpha_end / alpha_start) ** frac

        # smooth gradient
        grad = _smooth_gradient(a, conv, alpha, L)
        # clipping prevents overflow in the exponential
        grad = np.clip(grad, -30.0, 30.0)

        # step size (geometric decay)
        eta = eta0 * (eta_decay ** it)

        # multiplicative update + renormalisation (KL‑projection)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s <= 0.0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a /= s

        # occasional sparsification – speeds up later phases
        if prune_every and (it % prune_every == 0) and it > 0:
            thresh = np.percentile(a, prune_percentile)
            a[a < thresh] = 0.0
            a_sum = a.sum()
            if a_sum > 0:
                a /= a_sum

    return a, best_obj, best_a


# ----------------------------------------------------------------------
# 3.  Exact projected sub‑gradient descent -----------------------------
# ----------------------------------------------------------------------
def _subgradient_phase(a: np.ndarray,
                       n: int,
                       L: int,
                       max_iters: int,
                       deadline: float,
                       eta0: float = 0.3,
                       line_search: bool = True,
                       verbose: bool = False) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Projected sub‑gradient descent on the true (non‑smooth) objective.
    A simple back‑tracking line‑search guarantees descent.
    """
    best_a = a.copy()
    best_obj = _objective(a, n, L)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            if verbose and it % 2000 == 0:
                print(f"[sub‑grad] iter {it:,} – new best O={best_obj:.6f}")

        g = _subgradient_max_conv(a, conv)

        # geometric decay of the nominal step size
        eta = eta0 * (0.9999 ** it)

        if line_search:
            step = eta
            # up to ten back‑tracking tries (very cheap)
            for _ in range(10):
                cand = _project_to_simplex(a - step * g)
                cand_obj = _objective(cand, n, L)
                if cand_obj < cur_obj:
                    a = cand
                    break
                step *= 0.5
            else:
                a = _project_to_simplex(a - eta * g)
        else:
            a = _project_to_simplex(a - eta * g)

    return a, best_obj, best_a


# ----------------------------------------------------------------------
# 4.  Targeted simulated annealing ------------------------------------
# ----------------------------------------------------------------------
def _annealing_phase(a: np.ndarray,
                     n: int,
                     L: int,
                     max_iters: int,
                     deadline: float,
                     temperature0: float = 0.07,
                     no_improve_limit: int = 5000,
                     verbose: bool = True) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Local refinement by moving a modest fraction of mass from an index
    that participates in the current maximal convolution entry to a
    random destination.  Acceptance follows the Metropolis rule.
    Every few steps a *exact* sub‑gradient step is performed to keep
    the walk directed.
    """
    best_a = a.copy()
    best_obj = _objective(a, n, L)
    cur_obj = best_obj
    conv = _conv_fft(a, L)

    T = temperature0
    stagnation = 0

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # index (or indices) of the current convolution peak
        kmax = int(np.argmax(conv))

        # indices that actually contribute to that peak (non‑zero mass)
        i_cand = np.arange(max(0, kmax - (n - 1)), min(n - 1, kmax) + 1)
        i_cand = i_cand[a[i_cand] > 1e-14]
        if i_cand.size == 0:
            i = np.random.randint(n)
        else:
            i = np.random.choice(i_cand)

        # random destination (different from i)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # transfer up to 15 % of the source mass (larger moves than before)
        delta = a[i] * np.random.rand() * 0.15
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = _project_to_simplex(a_new)

        conv_new = _conv_fft(a_new, L)
        new_obj = 2.0 * n * conv_new.max()

        # Metropolis acceptance
        if new_obj < cur_obj or np.random.rand() < math.exp((cur_obj - new_obj) / max(T, 1e-12)):
            a = a_new
            conv = conv_new
            cur_obj = new_obj
            stagnation = 0
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                if verbose:
                    print(f"[SA]   iter {it:,} – new global best O={best_obj:.6f}")
        else:
            stagnation += 1

        # periodic exact sub‑gradient correction
        if it % 10 == 0:
            g = _subgradient_max_conv(a, conv)
            eta_loc = 0.07 * (1.0 - it / max_iters)
            a = _project_to_simplex(a - eta_loc * g)
            conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                if verbose:
                    print(f"[SA‑sg] iter {it:,} – sub‑grad improved O={best_obj:.6f}")

        # temperature decay (very slow)
        T *= 0.9999

        # restart if stuck for a while
        if stagnation >= no_improve_limit:
            a = best_a.copy()
            a += np.random.randn(n) * 0.0015
            a = np.clip(a, 0, None)
            a = _project_to_simplex(a)
            conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max()
            stagnation = 0

    return a, best_obj, best_a


# ----------------------------------------------------------------------
# 5.  Main entry point -------------------------------------------------
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function sequence that minimises
    evaluate_sequence().  The routine follows the four‑stage plan
    described in the module docstring.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a thin safety margin
    deadline = start_time + budget_s - 0.2

    # ------------------------------------------------------------------
    # initialise problem size
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        a0 = np.array(height_sequence_1, dtype=float)
        a0 = np.maximum(a0, 0.0)
        if a0.sum() == 0.0:
            a0 = np.full_like(a0, 1.0)
        a0 = a0 / a0.sum()
        n = a0.size
        a = a0.copy()
        print(f"[init] using supplied sequence, n={n}")
    else:
        n = 4600                     # default size used by the reference code
        a = np.full(n, 1.0 / n)      # uniform start on the simplex
        print(f"[init] using uniform start, n={n}")

    # FFT length – next power of two > 3·n‑2 (full convolution needs 2·n‑1)
    L = 1 << ((3 * n - 2).bit_length())

    # ------------------------------------------------------------------
    # Phase 0 – cheap random restarts
    # ------------------------------------------------------------------
    phase0_deadline = start_time + budget_s * 0.08
    best_seq = a.copy()
    best_score = _objective(best_seq, n, L)

    while time.time() < phase0_deadline:
        conc = np.random.choice([0.2, 0.4, 0.8, 2.0])
        a_rand = np.random.gamma(conc, 1.0, size=n)
        a_rand /= a_rand.sum()

        # a very short exponential‑gradient run (≈1 s)
        sub_dead = time.time() + 1.0
        a_rand, _, best_rand = _expgradient_phase(
            a_rand, n, L,
            max_iters=1500,
            deadline=sub_dead,
            eta0=1.0,
            alpha_start=5.0,
            alpha_end=2000.0,
            prune_every=0
        )
        cur_obj = _objective(best_rand, n, L)
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = best_rand.copy()
            print(f"[phase0] new best O={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – long exponentiated‑gradient optimisation (smooth surrogate)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + budget_s * 0.40
    a, obj_adam, best_adam = _expgradient_phase(
        best_seq, n, L,
        max_iters=300_000,
        deadline=phase1_deadline,
        eta0=0.6,
        eta_decay=0.9999,
        alpha_start=5.0,
        alpha_end=50000.0,
        prune_every=2000,
        prune_percentile=0.5
    )
    if obj_adam < best_score:
        best_score = obj_adam
        best_seq = best_adam.copy()
        print(f"[phase‑1] improved best O={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – exact sub‑gradient descent (non‑smooth)
    # ------------------------------------------------------------------
    phase2_deadline = start_time + budget_s * 0.65
    a, obj_sub, best_sub = _subgradient_phase(
        a, n, L,
        max_iters=200_000,
        deadline=phase2_deadline,
        eta0=0.25,
        line_search=True,
        verbose=False
    )
    if obj_sub < best_score:
        best_score = obj_sub
        best_seq = best_sub.copy()
        print(f"[phase‑2] improved best O={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – targeted simulated‑annealing refinement
    # ------------------------------------------------------------------
    phase3_deadline = deadline
    a, obj_sa, best_sa = _annealing_phase(
        best_seq, n, L,
        max_iters=500_000,
        deadline=phase3_deadline,
        temperature0=0.07,
        no_improve_limit=5000,
        verbose=True
    )
    if obj_sa < best_score:
        best_score = obj_sa
        best_seq = best_sa.copy()
        print(f"[phase‑3] final best O={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – cheap convex mixes (uniform & reversed)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    reversed_seq = best_seq[::-1]
    for t in (0.2, 0.4, 0.6, 0.8):
        # mix with uniform
        cand = t * best_seq + (1 - t) * uniform
        cand /= cand.sum()
        obj = _objective(cand, n, L)
        if obj < best_score:
            best_score = obj
            best_seq = cand.copy()
            print(f"[mix‑uni] t={t:.1f} – new best O={best_score:.6f}")

        # mix with reversed version
        cand = t * best_seq + (1 - t) * reversed_seq
        cand /= cand.sum()
        obj = _objective(cand, n, L)
        if obj < best_score:
            best_score = obj
            best_seq = cand.copy()
            print(f"[mix‑rev] t={t:.1f} – new best O={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final verification through the provided evaluator
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence failed:", exc)

    return best_seq.tolist()
```
