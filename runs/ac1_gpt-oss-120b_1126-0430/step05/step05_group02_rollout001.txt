#!/usr/bin/env python
"""
A convex‑gradient based optimiser for the non‑negative step‑function problem.

The algorithm proceeds in three stages:

1. **Broad parametric sweep** – cheap evaluation of several families
   (uniform, power‑law, exponential, power‑law × exponential, beta, Dirichlet)
   on a reasonably large length (5 k … 25 k) to obtain a good deterministic start.

2. **Smoothed max‑autocorrelation descent** – minimise
   `2·n·max_k (a*a)[k]` under the simplex (∑a=1, a≥0).
   The non‑differentiable max is replaced by a log‑sum‑exp (temperature α),
   whose gradient can be obtained by a single FFT‑based convolution.
   An exponentiated‑gradient (mirror‑descent) step with a decreasing stepsize
   and a linear increase of the temperature drives the iterates.
   Restarts with small random perturbations are performed when progress stalls.

3. **Exact evaluation** – the best iterate (according to the cheap score)
   is finally evaluated by the supplied ``evaluate_sequence`` function and
   returned.

All helper utilities are top‑level, there are no closures or lambdas,
and the routine respects the given time budget (default 1000 s).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:                     # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho_arr[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using FFT (returns length 2·n‑1 array)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa).real[: 2 * a.size - 1]
    return conv


def _logsumexp(x: np.ndarray, axis=None) -> float:
    """Stable log‑sum‑exp."""
    m = np.max(x, axis=axis, keepdims=True)
    return float(m + np.log(np.sum(np.exp(x - m), axis=axis)))

    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine adapts its schedule to the time budget
    and always returns a feasible sequence.
    """

        else:
            prev_arr = None
    except Exception:
        prev_arr = None
        prev_len = 0

    # Use a fairly large n (5 k … 25 k) – larger n gives more flexibility.
    n = min(max(prev_len, 8000), 25000)    # at least 8 k, at most 25 k

    # --------------------------------------------------------------
    # Cheap score (assumes Σa_i = 1): 2·n·max(conv)
    # --------------------------------------------------------------
    # FFT length for a*a (full convolution) – next power of two ≥ 2·n‑1
    fft_len_conv = 1 << ((2 * n - 1).bit_length())
    # FFT length for gradient cross‑correlation – next power of two ≥ 3·n‑2
    fft_len_grad = 1 << ((3 * n - 2).bit_length())

    def cheap_score(vec: np.ndarray) -> float:
        """Return 2·n·max(a*a) for a simplex‑normalised vector."""
        conv = _conv_full_fft(vec, fft_len_conv)
        return 2.0 * n * float(conv.max())

    # --------------------------------------------------------------
    # 0. Initialise with cheap parametric families
    # --------------------------------------------------------------
    best_a = None
    best_val = float("inf")

    # uniform baseline
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    val = cheap_score(uniform)
    best_a, best_val = uniform, val

    # previous construction, if any (resampled to length n)
    if prev_arr is not None:
        if prev_arr.size == n:
            cand = prev_arr
        else:

            cand = prev_arr[idx]
        cand = cand / cand.sum()
        val = cheap_score(cand)

    # power‑law sweep
    for p in np.linspace(0.4, 1.4, 61):

        cand = i ** (-p)

        val = cheap_score(cand)

        idx = np.arange(n, dtype=np.float64)
        cand = np.exp(-lam * idx)
        cand /= cand.sum()
        val = cheap_score(cand)
        if val < best_val:
            best_a, best_val = cand.copy(), val

    # power‑law × exponential sweep (coarse grid)

        cand = g / g.sum()
        val = cheap_score(cand)
        if val < best_val:
            best_a, best_val = cand.copy(), val

    print(f"[Init] best cheap score = {best_val:.6f}")

    # --------------------------------------------------------------
    # 1. Smoothed‑max gradient descent (exponentiated mirror descent)
    # --------------------------------------------------------------
    a = best_a.copy()
    best_iter_a = a.copy()
    best_iter_val = best_val
    last_improve_iter = 0
    iter_cnt = 0

    # hyper‑parameters

    alpha0 = 0.5             # initial temperature (smoothness)
    alpha_max = 25.0         # final temperature – focuses on the true max
    alpha_ramp_iters = 200000  # iterations over which temperature ramps
    max_no_improve = 150000  # restart after this many stagnant iterations

    while time.time() < deadline:
        iter_cnt += 1

        # --- compute full autocorrelation and cheap objective ---
        fa = np.fft.fft(a, fft_len_conv)
        conv_full = np.fft.ifft(fa * fa).real[: 2 * n - 1]
        maxb = float(conv_full.max())
        cheap = 2.0 * n * maxb

        # keep the best ever seen
        if cheap < best_iter_val - 1e-12:
            best_iter_val = cheap
            best_iter_a = a.copy()
            last_improve_iter = iter_cnt
            print(f"[Iter {iter_cnt:07d}] NEW BEST cheap = {cheap:.6f}")

        # --- temperature schedule (linear ramp) ---
        if iter_cnt <= alpha_ramp_iters:
            alpha = alpha0 + (alpha_max - alpha0) * (iter_cnt / alpha_ramp_iters)
        else:
            alpha = alpha_max

        # --- soft‑max weights for the smoothed max (log‑sum‑exp) ---
        shifted = conv_full - maxb                # improve numerical stability
        exp_w = np.exp(alpha * shifted)
        w = exp_w / exp_w.sum()                  # probability distribution over shifts

        # --- gradient of the smoothed objective ---
        # grad = 4·n * (cross‑correlation of a with w)
        rev_w = w[::-1]                           # reversed weights
        revF = np.fft.fft(rev_w, fft_len_grad)
        fa_full = np.fft.fft(a, fft_len_grad)    # reuse for both products
        cross_full = np.fft.ifft(fa_full * revF).real

        # cross_full indices: cross_full[L-1 - j] corresponds to Σ_i w_i a_{i-j}
        grad = 4.0 * n * cross_full[L - 1 - np.arange(n)]

        # --- mirror‑descent (exponentiated‑gradient) step ---
        eta = eta0 / math.sqrt(iter_cnt)          # decreasing stepsize

        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # 4. Return the best sequence found.
    # --------------------------------------------------------------
    return final_seq
