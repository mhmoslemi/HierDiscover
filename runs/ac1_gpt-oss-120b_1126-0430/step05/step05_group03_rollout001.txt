def _lp_optimize(f, M):
    # solve LP to maximize sum(g) s.t. conv(f,g) <= M
    n = f.shape[0]
    # Build A_ub matrix
    rows = []
    for k in range(2*n-1):
        row = np.zeros(n)
        i_min = max(0, k-(n-1))
        i_max = min(k, n-1)
        # i from i_min to i_max inclusive
        row[i_min: i_max+1] = f[i_min: i_max+1]  # but careful: indices of g correspond to (k-i)
        # Actually row for column i = f_i if (k-i) in [0,n-1] else 0. That condition yields exactly i range as above.
        rows.append(row)
    A_ub = np.vstack(rows) # shape (2n-1, n)
    b_ub = np.full(2*n-1, M)
    res = linprog(c=-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        return None
    g = res.x
    sum_g = g.sum()
    return g, sum_g
```

from scipy.optimize import linprog

def _lp_max_sum(f, M):
    n = f.shape[0]
    # build constraints
    rows = []
    cols = []
    data = []
    for k in range(2*n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # indices i_start:i_end+1 correspond to f[i]
        length = i_end - i_start + 1
        rows.extend([k] * length)
        cols.extend(range(i_start, i_end+1))
        data.extend(f[i_start:i_end+1].tolist())
    A = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    b = np.full(2*n-1, M)
    c = -np.ones(n) # maximize sum
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        return None, None
    g = res.x
    sum_g = g.sum()
    return g, sum_g
```

"""
Propose a step‑function sequence that minimises
    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm has three phases:

1. **Global search** – several Adam optimiser runs on different
   support sizes.  The optimiser works on the soft‑max (log‑it)
   parametrisation of the simplex and uses an annealed soft‑max
   approximation of the max‑convolution.  Periodic jitter and restarts
   help to escape plateaus.

2. **LP refinement** – the best candidate is pruned to its
   non‑negligible support (≤~2000 entries).  With this reduced support
   we solve a linear programme
        maximise   Σ g_i
        subject to  g_i ≥ 0  and   (f ∗ g)[k] ≤ M   (∀k)
   where `M` is the current maximal self‑convolution of `f`.
   The normalised optimiser `g` is then mixed with `f` (simple averaging)
   which provably does not increase `M`.  A few such LP steps usually
   shrink the objective a little.

3. **Stochastic‑annealing local search** – a cheap random‑mass‑transfer
   neighbourhood is explored with a simulated‑annealing acceptance rule
   for the remaining time budget.  This phase can still improve the
   objective by a few‑hundredths and is cheap because the convolution is
   computed with an FFT.

All numbers are forced to be non‑negative, normalised to sum = 1 and
finally passed to the provided ``evaluate_sequence`` routine for a sanity
check.  The function respects the 1000 s time limit.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# -------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# -------------------------------------------------------------
def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv


# -------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective (assumes sum(a)=1)."""
    n = a.size
    conv = _conv_self_fft(a)
    return float(2.0 * n * conv.max())


# -------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


# -------------------------------------------------------------
def _adam_opt(
    n: int,
    time_limit: float,
    init: np.ndarray | None = None,
    seed: int = 0,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for a single support size `n`.
    Returns the best vector (sum = 1) and its exact objective.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # initialise a on the simplex
    if init is not None:
        a = np.asarray(init, dtype=float)
        if a.size != n:
            a = rng.random(n)
        a = a / a.sum()
    else:
        a = rng.random(n)
        a = a / a.sum()

    # work in log‑it space (automatically stays on simplex)
    x = np.log(a + 1e-16)

    # Adam state
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # schedules
    beta_start, beta_end = 0.5, 4000.0   # smoothing for soft‑max
    lr_start, lr_end = 0.15, 0.001      # learning‑rate decay

    best_a = a.copy()
    best_obj = _objective(best_a)

    it = 0
    while time.time() - start < time_limit:
        it += 1

        # current vector on the simplex
        a = _softmax(x)

        # exact convolution and objective
        conv = _conv_self_fft(a)
        M = conv.max()
        obj = 2.0 * n * M
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[adam n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        # annealing of the smooth‑max temperature
        t_frac = min(it / 800_000.0, 1.0)
        beta = beta_start * np.exp(np.log(beta_end / beta_start) * t_frac)
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t_frac)

        # soft‑max weights for the (smooth) max of the convolution
        if beta > 3000.0:               # essentially exact max
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            shifted = conv - M           # ≤ 0
            w = np.exp(beta * shifted)
            w = w / w.sum()

        # gradient of 2·n·max(conv) w.r.t. a (in the original space)
        g_a = 2.0 * _correlation_fft(w, a)     # shape (n,)

        # natural gradient in log‑it space (project onto simplex)
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape flat regions
        if it % 6000 == 0:
            jitter = rng.normal(scale=0.02, size=n)
            x += jitter

        # periodic restart from current best
        if it % 70_000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.005, size=n)
            m.fill(0.0)
            v.fill(0.0)
            print(f"[adam n={n}] restart at iter {it}")

    # prune negligible entries and renormalise
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        print(f"[adam prune] length reduced to {best_a.size}")

    # final exact objective
    best_obj = _objective(best_a)
    return best_a, best_obj


# -------------------------------------------------------------
def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation of w with a (∑_t w[i+t]·a[t]), 0 ≤ i < len(a).
    Implemented via FFT.  Works for real vectors.
    """
    n = a.shape[0]
    m = w.shape[0]                 # = 2·n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


# -------------------------------------------------------------
def _lp_refine(f: np.ndarray, max_iters: int, time_limit: float) -> np.ndarray:
    """
    Perform a small number of LP‑based refinements.
    Each iteration:
        - compute current M = max(conv(f,f))
        - solve maximise Σ g   s.t. conv(f,g) ≤ M , g ≥ 0
        - if Σ g > 1, mix f ← (f + g/Σg)/2
    The loop stops when no improvement is possible or time runs out.
    """
    start = time.time()
    n = f.shape[0]

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        conv = _conv_self_fft(f)
        M = conv.max()
        # Build the constraint matrix A (size (2n‑1)×n) such that
        #   (A @ g)_k = (f * g)[k]
        rows = []
        cols = []
        data = []
        for k in range(2 * n - 1):
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            length = i_end - i_start + 1
            rows.extend([k] * length)
            cols.extend(range(i_start, i_end + 1))
            data.extend(f[i_start : i_end + 1].tolist())
        A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
        b = np.full(2 * n - 1, M, dtype=float)

        # Linear programme: maximise 1ᵀg   (i.e. minimise -1ᵀg)
        c = -np.ones(n, dtype=float)

        res = linprog(
            c,
            A_ub=A,
            b_ub=b,
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            # fallback – cannot improve further
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-9:
            # No useful direction – stop
            break

        # Normalise g to unit sum
        g_norm = g / sum_g

        # Simple averaging (t = 0.5) – guaranteed not to increase M
        f_new = 0.5 * (f + g_norm)
        obj_old = _objective(f)
        obj_new = _objective(f_new)

        if obj_new < obj_old - 1e-12:
            f = f_new
            print(f"[lp it={it}] improvement O={obj_new:.6f}")
        else:
            # No improvement – stop
            break

    return f


# -------------------------------------------------------------
def _annealing_local_search(
    a: np.ndarray,
    time_limit: float,
    seed: int = 0,
    max_frac: float = 0.05,
) -> np.ndarray:
    """
    Stochastic‑annealing neighbourhood search.
    Randomly moves a fraction (≤max_frac) of mass from one index to another.
    Accepts worsening moves with a temperature that cools linearly.
    """
    rng = np.random.default_rng(seed)
    n = a.size
    cur = a.copy()
    cur_obj = _objective(cur)
    start = time.time()
    it = 0
    while time.time() - start < time_limit:
        it += 1
        # linear cooling: temperature goes from 0.1 down to 1e-4
        elapsed = time.time() - start
        frac = elapsed / time_limit
        temp = 0.1 * (1.0 - frac) + 1e-4

        i, j = rng.choice(n, size=2, replace=False)
        if cur[i] == 0.0:
            continue
        delta = cur[i] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        # keep numerical safety
        cand = np.maximum(cand, 0.0)

        cand_obj = _objective(cand)
        if cand_obj < cur_obj:
            cur, cur_obj = cand, cand_obj
        else:
            # Metropolis acceptance
            if rng.random() < np.exp((cur_obj - cand_obj) / temp):
                cur, cur_obj = cand, cand_obj

        if it % 5000 == 0:
            print(f"[anneal] it={it} O={cur_obj:.6f}")

    return cur


# -------------------------------------------------------------
def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **_ignore,
) -> list[float]:
    """
    Entry point used by the evaluation harness.
    Runs a multi‑length Adam search, followed by an LP refinement
    and a short simulated‑annealing local search.
    Returns the best sequence found (as a plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # -----------------------------------------------------------------
    # lengths to explore – geometric ladder, plus possible reuse of a
    # previously discovered sequence (height_sequence_1) if available.
    lengths = [256, 512, 1024, 2048, 4096]
    seq0 = None
    if "height_sequence_1" in globals():
        seq0 = np.asarray(height_sequence_1, dtype=float).copy()
        seq0 = np.clip(seq0, 0.0, None)
        if seq0.sum() < 0.01:
            seq0 = np.ones_like(seq0)
        seq0 = seq0 / seq0.sum()
        n0 = seq0.size
        if n0 not in lengths:
            lengths.insert(0, n0)

    best_seq = None
    best_obj = float("inf")

    # -----------------------------------------------------------------
    # Global Adam optimisation phase
    for idx, n in enumerate(lengths):
        remaining = deadline - time.time()
        if remaining <= 1.0:
            break

        # allocate the majority of time to the first (largest) length,
        # the rest is split among the remaining candidates.
        if idx == 0:
            run_time = remaining * 0.70
        else:
            run_time = (remaining - 0.5) / (len(lengths) - idx)

        init = None
        if idx == 0 and seq0 is not None:
            if seq0.size != n:
                if seq0.size > n:
                    init = seq0[:n]
                else:
                    pad = np.zeros(n - seq0.size)
                    init = np.concatenate([seq0, pad])
                init = init / init.sum()
            else:
                init = seq0

        seq, val = _adam_opt(
            n=n,
            time_limit=run_time,
            init=init,
            seed=seed + idx * 13,
        )
        if val < best_obj:
            best_obj = val
            best_seq = seq.tolist()
            print(f"[global] new overall best O={best_obj:.6f} (n={n})")

    # -----------------------------------------------------------------
    # LP refinement (only if we have a candidate and some time left)
    if best_seq is not None:
        remaining = deadline - time.time()
        if remaining > 5.0:
            a = np.array(best_seq, dtype=float)
            # prune very small entries first – makes the LP tractable
            a = a[a > 1e-10]
            a = a / a.sum()
            # limit the support size for the LP to about 2000 entries
            if a.size > 2000:
                # keep the largest 2000 components
                idxs = np.argpartition(a, -2000)[-2000:]
                a = a[idxs]
                a = a / a.sum()
                print(f"[prune] reduced support to {a.size} for LP")

            lp_time = min(remaining * 0.40, 45.0)   # never spend more than ~45 s
            a = _lp_refine(a, max_iters=8, time_limit=lp_time)
            best_seq = a.tolist()
            best_obj = _objective(a)
            print(f"[lp] post‑refinement O={best_obj:.6f}")

    # -----------------------------------------------------------------
    # Simulated‑annealing local search for the final few seconds
    if best_seq is not None:
        remaining = deadline - time.time()
        if remaining > 2.0:
            a = np.array(best_seq, dtype=float)
            a = a / a.sum()
            a = _annealing_local_search(
                a,
                time_limit=remaining * 0.55,
                seed=seed + 999,
                max_frac=0.07,
            )
            best_seq = a.tolist()
            best_obj = _objective(a)
            print(f"[anneal] final O={best_obj:.6f}")

    # -----------------------------------------------------------------
    # Final sanity check via the provided evaluate_sequence routine.
    try:
        final_score = evaluate_sequence(best_seq)
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    # guarantee a non‑empty return
    if not best_seq:
        best_seq = [1.0]

    return best_seq
