Propose a step‑function sequence that (approximately) minimises

    O(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm consists of three stages:

1️⃣  **Parametric seeding** – generate a handful of simple analytic families
    (exponential, power‑law, sinusoidal, random) and keep the best candidate.

2️⃣  **Exponentiated‑gradient sub‑gradient descent** – iteratively
    minimise the exact max‑convolution using the sub‑gradient of the
    max‑operator.  The update is a multiplicative mirror‑descent
    (a_i ← a_i·exp(−η·g_i)) followed by a projection onto the simplex.
    Adaptive step‑size, occasional jitter, smoothing and restarts are used
    to escape plateaus.

3️⃣  **Final Adam polishing** – a few hundred ADAM steps on the smooth
    log‑sum‑exp surrogate (very large α, zero entropy) to fine‑tune the

All work respects the 1000 s budget and prints progress.  The returned
list is a plain Python list of non‑negative floats ready for `evaluate_sequence`.

# Helper: project onto probability simplex (Euclidean)
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
# Full linear convolution via FFT (length 2·n‑1)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return full convolution a⋆a (length 2·n‑1) using FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# --------------------------------------------------------------
# Sub‑gradient of max(conv) (exact, not smoothed)
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a.
    Returns vector g where g_i = 2·∑_{k∈K} a_{k-i} (bounds respected),
    K = indices where conv attains its maximum.
    """
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv >= max_val - 1e-12)   # active constraints
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size > 1:
        g /= ks.size
    return g


# --------------------------------------------------------------
# Gradient of the smooth surrogate (log‑sum‑exp) plus entropy term
def _smooth_gradient_plus_entropy(
    a: np.ndarray,
    conv: np.ndarray,
    alpha: float,
    L: int,
    lam: float,
) -> np.ndarray:
    """
    Returns ∇_a ( (2·n/α)·log∑ e^{α·conv_k} + λ·∑ a_i·log a_i ).
    The first part reduces to 2·corr(w, a) where w_k ∝ e^{α·conv_k};
    the second part contributes λ·(log a_i + 1).
    """
    # ---- soft‑max weights ----------
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)          # stability
    w = exp_scaled / exp_scaled.sum()                 # shape (2n‑1,)

    # ---- correlation w ⋆ a (g_i = 2·∑_j w_{i+j} a_j) ----
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n_len = a.size
    grad_corr = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]

    # ---- entropy part ----------
    eps = 1e-20
    grad_ent = lam * (np.log(a + eps) + 1.0)

    return grad_corr + grad_ent
