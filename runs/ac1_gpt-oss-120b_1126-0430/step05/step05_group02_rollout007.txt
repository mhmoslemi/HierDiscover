    original top‑k heuristic.
*   Dynamically increasing the soft‑max sharpness (α) and decreasing the step

def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project an arbitrary vector onto the probability simplex
    {x ≥ 0, Σ x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # all non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    Fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(Fa * Fa).real[: 2 * a.size - 1]
    return conv


def _cheap_score(a: np.ndarray, fft_len: int) -> float:
    """Cheap surrogate of the evaluation function for Σa = 1:
    2·n·max(conv)."""
    n = a.shape[0]
    conv = _conv_full_fft(a, fft_len)
    return 2.0 * n * conv.max()


def _gradient_softmax(a: np.ndarray,
                      conv: np.ndarray,
                      fft_len: int,
                      alpha: float) -> np.ndarray:
    """
    Gradient of the smooth ``soft‑max`` objective
        F(a) = ( Σ_k exp(α·(conv_k-max)) )^{-1} * Σ_k conv_k·exp(α·(conv_k-max))

    The expression reduces to a weighted sum of the individual convolution
    gradients; the weights w_k = exp(α·(conv_k-max)) are normalised to sum to 1.
    The gradient is computed efficiently via FFT.
    """
    maxb = conv.max()
    # weights concentrate on the largest convolution entries
    w = np.exp(alpha * (conv - maxb))
    w_sum = w.sum()
    if w_sum == 0 or not np.isfinite(w_sum):
        w = np.ones_like(w) / w.size
    else:
        w /= w_sum

    Fa = np.fft.fft(a, fft_len)
    Fw = np.fft.fft(w, fft_len)
    # gradient = 2·Re{ IFFT( Fa * conj(Fw) ) }[0:n]
    g = 2.0 * np.real(np.fft.ifft(Fa * np.conj(Fw)))[: a.size]
    return g

                best_up_val = cheap
                best_up_a = a.copy()
                print(f"[Upscale {up_iter:06d}] NEW BEST cheap = {cheap:.6f}")

            # use a fixed sharper α for the fine stage
            alpha = 40.0
            g = _gradient_softmax(a, conv, fft_len2, alpha)
            eta = 0.6 / math.sqrt(up_iter + 1)
            log_a = np.log(a + 1e-30) - eta * g
            a = np.exp(log_a)
            a /= a.sum()

        final_a = best_up_a

    # ------------------------------------------------------------------
    # 4. Final exact evaluation (uses the provided ``evaluate_sequence``)
    # ------------------------------------------------------------------
    final_seq = final_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:      # defensive – should never happen
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # ------------------------------------------------------------------
    # 5. Return the best sequence found.
    # ------------------------------------------------------------------
    return final_seq
