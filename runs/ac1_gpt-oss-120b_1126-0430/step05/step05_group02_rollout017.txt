def subgradient_max(a, idx):
    n = a.shape[0]
    lo = max(0, idx - (n - 1))
    hi = min(idx, n - 1)
    # g[i] = a[idx-i] * (2 - (i == idx-i))
    # Using vectorized:
    i = np.arange(lo, hi+1)
    j = idx - i
    g = np.zeros_like(a)
    # where i != j
    neq = i != j
    g[i[neq]] = 2 * a[j[neq]]
    # where i == j
    eq = i == j
    g[i[eq]] = a[i[eq]]
    return g
```

#!/usr/bin/env python
"""
Propose a non‑negative step function that minimises

        2·n·max_k (a*a)[k] / (Σ a_i)²

The algorithm works in three phases:

1. **Broad initial sampling** – many parametric families (power‑law,
   exponential, beta, Dirichlet, etc.) are sampled and the best candidate
   is kept.  This gives a good deterministic starting point.
2. **Exponentiated‑gradient (mirror‑descent) refinement** – the exact
   active convolution index (the one attaining the maximum) is identified
   and a sub‑gradient of that entry is used to perform an Adam‑style
   multiplicative update on the vector (log‑space mirror descent).  The
   iterate is re‑projected onto the probability simplex after each step.
   Adaptive step‑sizes and periodic noisy restarts help escape plateaus.
3. **Final exact evaluation** – the best sequence seen during the search
   is evaluated with the provided ``evaluate_sequence`` function and
   returned.

All helper utilities are defined at top level (no closures) and the
routine respects a 1000 s time budget.  The code works for any length
up to a few × 10⁴ (default 5 000 ≤ n ≤ 25 000).  Print statements report
progress and intermediate bounds.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full linear convolution a*a using a real FFT.
    Returns an array of length 2·n‑1 (the linear convolution, not circular).
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.size - 1]


def _conv_max_fft(a: np.ndarray, fft_len: int) -> tuple[float, int]:
    """Maximum value and its index of the autocorrelation a*a (full convolution)."""
    conv = _conv_full_fft(a, fft_len)
    idx = int(conv.argmax())
    return float(conv[idx]), idx


def _subgradient_max(a: np.ndarray, idx_max: int) -> np.ndarray:
    """
    Sub‑gradient of the active convolution entry (the one attaining the maximum).
    For a given idx k,  (a*a)[k] = Σ_i a_i a_{k-i}.
    The gradient w.r.t. a_j is:
          a_{k-j}      if j ≠ k‑j
          a_j          if j = k‑j
    The implementation returns the vector g of shape (n,).
    """
    n = a.shape[0]
    lo = max(0, idx_max - (n - 1))
    hi = min(idx_max, n - 1)
    i = np.arange(lo, hi + 1, dtype=np.int64)
    j = idx_max - i
    g = np.zeros_like(a)
    eq = i == j
    neq = ~eq
    # diagonal terms (i == j)
    if np.any(eq):
        g[i[eq]] = a[i[eq]]
    # off‑diagonal terms appear twice in the sum
    if np.any(neq):
        g[i[neq]] = 2.0 * a[j[neq]]
    return g


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that
    minimises ``evaluate_sequence``.  The routine adapts its schedule to
    the available time budget and always returns a feasible sequence.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the final exact evaluation
    deadline = start_time + budget_s - 10.0

    # ------------------------------------------------------------------
    # Choose a working length.
    # ------------------------------------------------------------------
    try:

        n = max(len(prev), 5000)
        prev_arr = np.asarray(prev, dtype=np.float64)
        if prev_arr.sum() > 0.0:
            prev_arr = prev_arr / prev_arr.sum()
        else:
            prev_arr = None
    except Exception:
        prev_arr = None
        n = 5000

    n = min(max(n, 5000), 25000)                     # 5 k ≤ n ≤ 25 k
    # FFT length – next power of two ≥ 2·n‑1
    fft_len = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # Cheap score (assumes Σa_i = 1): 2·n·max(conv)
    # ------------------------------------------------------------------
    def cheap_score(vec: np.ndarray) -> float:
        maxb, _ = _conv_max_fft(vec, fft_len)
        return 2.0 * n * maxb

    # ------------------------------------------------------------------
    # 0. Initialise with a diverse set of parametric families
    # ------------------------------------------------------------------
    best_a = None

        val = cheap_score(prev_arr)

        if val < best_val:

    # ------------------------------------------------------------------
    a = best_a.copy()
    best_iter_a = a.copy()
    best_iter_val = best_val
    last_improve = 0
    iter_cnt = 0

    # Adam‑style moments (all zero at start)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    eta0 = 1.0                     # initial learning‑rate factor

    max_iters = 2_000_000          # safety bound – will stop on time

    while time.time() < deadline and iter_cnt < max_iters:
        iter_cnt += 1

        # ---- full convolution & current cheap bound ----
        conv = _conv_full_fft(a, fft_len)
        maxb = float(conv.max())
        idx_max = int(conv.argmax())
        cheap = 2.0 * n * maxb

        # keep best seen so far
        if cheap < best_iter_val - 1e-12:
            best_iter_val = cheap
            best_iter_a = a.copy()
            last_improve = iter_cnt
            print(f"[Iter {iter_cnt:07d}] NEW BEST cheap = {cheap:.6f}")

        g = _subgradient_max(a, idx_max)

        # ---- Adam update in log‑space (multiplicative) ----
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)

        m_hat = m / (1 - beta1 ** iter_cnt)
        v_hat = v / (1 - beta2 ** iter_cnt)

        # step‑size schedule
        eta = eta0 / math.sqrt(iter_cnt)

        # multiplicative factor
        factor = np.exp(-eta * m_hat / (np.sqrt(v_hat) + eps))
        a = a * factor

        # avoid exact zeros and renormalise onto the simplex
        a = np.maximum(a, 1e-30)
        a /= a.sum()

        # ---- occasional restart when progress stalls ----
        if iter_cnt - last_improve > 250_000:
            # add modest Gaussian noise in log‑space
            log_a = np.log(a + 1e-30)
            log_a += np.random.normal(scale=0.02, size=n)
            a = np.exp(log_a)
            a = _simplex_projection(a)
            # reset Adam moments
            m.fill(0.0)
            v.fill(0.0)
            last_improve = iter_cnt
            print(f"[Iter {iter_cnt:07d}] Restart (stagnation).")

    # ------------------------------------------------------------------
    # 2. Final exact evaluation (uses the provided ``evaluate_sequence``)
    # ------------------------------------------------------------------
    final_seq = best_iter_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:      # defensive – should never happen

    # 4. Return the best sequence found.
    # ------------------------------------------------------------------
