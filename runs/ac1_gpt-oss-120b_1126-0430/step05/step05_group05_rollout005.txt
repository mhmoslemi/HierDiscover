def propose_candidate(seed=42, budget_s=1000, **kwargs): 
    import numpy as np, time, math

    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5 # safety margin

    # Determine initial n and a.
    if 'height_sequence_1' in globals():
        base = np.array(height_sequence_1, dtype=float)
        n0 = base.size
        # clip negative and handle sum
        base = np.clip(base, 0, None) # nonnegative
        if base.sum() < 1e-12:
            n = max(5000, n0) # fallback
            a = np.full(n, 1.0/n)
        else:
            # If too short (<5000), pad zeros to reach min_n
            n_target = max(6000, n0)
            if n0 < n_target:
                a = np.concatenate([base, np.zeros(n_target-n0)])
            else:
                a = base.copy()
            # normalize to sum =1
            a = a / a.sum()
            n = a.size
    else:
        n = 8000
        a = np.full(n, 1.0/n)

    # Ensure a not all zeros; possible large values >1000? not probable as sum=1.

    # FFT length
    L = 1 << (2*n-1).bit_length()

    # Helper functions (already defined above): _project_to_simplex, _conv_fft, _objective, _smooth_gradient_plus_entropy, _p_norm_gradient, _subgradient_max_conv.

    # Keep best
    best_a = a.copy()
    conv_best = _conv_fft(best_a, L)
    best_score = 2.0 * n * conv_best.max()
    print(f"[init] n={n}, best_score={best_score:.6f}")

    # Phase 1: Annealed Adam with log-sum-exp
    phase1_end = start_time + budget_s * 0.30
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    lr0 = 0.6
    alpha_start = 5.0
    alpha_end = 200_000.0
    lam_start = 0.2
    lam_end = 0.0
    it = 0
    max_iter_phase1 = 500_000
    while time.time() < phase1_end and it < max_iter_phase1:
        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_score:
            best_score = cur_obj
            best_a = a.copy()
            print(f"[phase1] it {it:,} new best O={best_score:.6f}")
        # annealing
        frac = it / max_iter_phase1
        alpha = alpha_start + (alpha_end - alpha_start) * frac
        lam = lam_start * (1 - frac)
        # gradient
        grad = _smooth_gradient_plus_entropy(a, conv, alpha, L, lam)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        lr = lr0 / math.sqrt(it + 1)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        # occasional jitter
        if np.random.rand() < 0.001:
            a_new += np.random.randn(n) * 5e-5
        # projection
        a_new = _project_to_simplex(a_new)
        a = a_new
        it += 1

    # Phase 2: p-norm gradient descent
    phase2_end = start_time + budget_s * 0.60
    # Use Adam again, with lambda=0; replace _smooth_gradient_plus_entropy with p-norm gradient.
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    lr0 = 0.5
    p_start = 4
    p_end = 80
    max_iter_phase2 = 300_000
    it = 0
    while time.time() < phase2_end and it < max_iter_phase2:
        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_score:
            best_score = cur_obj
            best_a = a.copy()
            print(f"[phase2] it {it:,} new best O={best_score:.6f}")
        # p schedule
        frac = it / max_iter_phase2
        p = p_start + (p_end - p_start) * frac
        # compute p-norm gradient
        # compute Bp = conv ** (p-1)
        # compute f = (sum conv**p) ** (1/p)
        conv_pow_p = conv ** p
        f = conv_pow_p.sum() ** (1.0/p)  # should be 1**(1/p?) but not exactly due to rounding.
        # Avoid division by zero: ensure f>0
        # Compute w = conv**(p-1) / f**(p-1)
        with np.errstate(divide='ignore', invalid='ignore'):
            w = np.where(conv > 0, conv ** (p - 1), 0.0) / (f ** (p - 1) + eps)
        # gradient via correlation: 2 * corr(w, a)
        a_rev = a[::-1]
        fft_w = np.fft.rfft(w, n=L)
        fft_a_rev = np.fft.rfft(a_rev, n=L)
        corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
        grad = 2.0 * corr_rev[n-1:n-1+n]  # shape n
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        lr = lr0 / math.sqrt(it + 1)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        # projection
        a_new = _project_to_simplex(a_new)
        a = a_new
        it += 1

    # Phase 3: subgradient descent on max
    phase3_end = start_time + budget_s * 0.85
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    lr0 = 0.4
    max_iter_phase3 = 200_000
    it = 0
    while time.time() < phase3_end and it < max_iter_phase3:
        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_score:
            best_score = cur_obj
            best_a = a.copy()
            print(f"[phase3] it {it:,} new best O={best_score:.6f}")
        # subgradient of max
        g = _subgradient_max_conv(a, conv)
        # perhaps add small random noise
        if np.random.rand() < 0.005:
            g += np.random.randn(n) * 1e-5
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        lr = lr0 / math.sqrt(it + 1)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = _project_to_simplex(a_new)
        a = a_new
        it += 1

    # Phase 4: Simulated annealing with mass transfer (optional)
    # While time remains, perform random moves.
    # We'll use restarts and move fraction adaptation.

    restarts = 0
    # maintain current best as a; also maintain global best.
    while time.time() < deadline:
        # initialise local a as best with jitter or random.
        if np.random.rand() < 0.7:
            a = best_a.copy()
            a += np.random.randn(n) * 0.001  # jitter
            a = np.clip(a, 0, None)
            a = _project_to_simplex(a)
        else:
            a = np.random.rand(n)
            a = _project_to_simplex(a)
        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        # SA parameters
        T0 = 0.08
        T = T0
        temp_decay = 0.9999
        f_no_improve = 0
        max_no_improve = 5000
        max_iter_sa = 100_000
        it_sa = 0
        while time.time() < deadline and it_sa < max_iter_sa:
            # pick random i,j
            i, j = np.random.randint(n, size=2)
            if i == j or a[i] < 1e-12:
                # skip
                pass
            else:
                # transfer a fraction
                max_frac = 0.4 * (1 - it_sa / max_iter_sa) + 0.01
                frac = np.random.rand() * max_frac
                delta = a[i] * frac
                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta
                a_new = _project_to_simplex(a_new)
                conv_new = _conv_fft(a_new, L)
                new_obj = 2.0 * n * conv_new.max()
                if new_obj < cur_obj or np.random.rand() < math.exp((cur_obj - new_obj) / max(T, 1e-12)):
                    a, conv, cur_obj = a_new, conv_new, new_obj
                    f_no_improve = 0
                    if cur_obj < best_score:
                        best_score = cur_obj
                        best_a = a.copy()
                        print(f"[SA] new global best O={best_score:.6f} restart={restarts} iter={it_sa}")
                else:
                    f_no_improve += 1
            # occasional subgradient step to keep direction
            if it_sa % 2000 == 0:
                g = _subgradient_max_conv(a, conv)
                a = a - 0.1 * g
                a = _project_to_simplex(a)
                conv = _conv_fft(a, L)
                cur_obj = 2.0 * n * conv.max()
                if cur_obj < best_score:
                    best_score = cur_obj
                    best_a = a.copy()
                    print(f"[SA] subgrad improvement O={best_score:.6f}")

            # temperature decay
            T *= temp_decay
            it_sa += 1
            if f_no_improve >= max_no_improve:
                break
        restarts += 1

    # Final polishing (if time left)
    if time.time() < deadline:

"""
Propose a non‑negative step‑function (list of heights) that (approximately)
minimises

    O(a) = 2·n·max(conv(a,a)) / (sum(a)²)

The algorithm works in several time‑budgeted phases:

1️⃣  **Annealed Adam on a smooth log‑sum‑exp surrogate** – the surrogate
    fα,λ(a) = (2·n/α)·log Σ exp(α·conv) + λ·Σ aᵢ·log aᵢ is minimised while
    projecting onto the probability simplex (∑aᵢ=1, aᵢ≥0).  α is annealed
    from a few to ~2·10⁵, λ from 0.2 to 0.

2️⃣  **p‑norm gradient descent** – for a schedule p∈[4,80] we minimise the
    Lᵖ‑norm of the convolution (which converges to the max‑norm).  The
    gradient is obtained analytically via a correlation computed with FFTs
    and an Adam‑style update.

3️⃣  **Sub‑gradient Adam on the exact max‑conv** – the true max‑conv is
    differentiable a.e.; we use its sub‑gradient together with Adam.

4️⃣  **Intensive simulated‑annealing local search** – random “mass‑transfer’’
    moves are accepted with a Metropolis rule; occasional sub‑gradient
    nudges keep the search directed.

5️⃣  **Final polishing** – a last Adam run with an extremely large smoothing
    parameter (α≈5·10⁵) to clean up any remaining slack.

All phases respect the overall 1000 s budget; intermediate best scores are
printed and the best sequence found is returned as a plain ``list`` of
floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – keeps a on the probability simplex.
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all entries are zero
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution via FFT (returns length 2·n‑1)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Gradient of the smooth log‑sum‑exp surrogate plus (optional) entropy.
def _smooth_gradient_plus_entropy(
    a: np.ndarray,
    conv: np.ndarray,
    alpha: float,
    L: int,
    lam: float,
) -> np.ndarray:
    # soft‑max weights w_k ∝ e^{α·conv_k}
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / exp_scaled.sum()

    # correlation w ⋆ a  (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n = a.size
    grad_corr = 2.0 * corr_rev[n - 1 : n - 1 + n]

    # entropy gradient λ·(log a_i + 1)
    eps = 1e-20
    grad_ent = lam * (np.log(a + eps) + 1.0)

    return grad_corr + grad_ent


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) (averaged over all maximisers)
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)          # indices achieving the max
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
# Gradient of the L^p‑norm of the convolution (p>1)
def _gradient_p_norm(a: np.ndarray, conv: np.ndarray, p: float, L: int) -> np.ndarray:
    # f = ( Σ conv^p )^{1/p}
    conv_pow_p = conv ** p
    f = conv_pow_p.sum() ** (1.0 / p) + 1e-30  # avoid zero

    # weights w_i = conv_i^{p-1} / f^{p-1}
    with np.errstate(divide="ignore", invalid="ignore"):
        w = np.where(conv > 0, conv ** (p - 1), 0.0) / (f ** (p - 1))

    # correlation w ⋆ a  (gradient = 2·corr(w,a))
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n = a.size
    grad = 2.0 * corr_rev[n - 1 : n - 1 + n]
    return grad
