"""
Propose a step‑function (non‑negative vector) that minimises

    O(a) = 2·n·max(conv(a,a)) / (sum(a)²)

The algorithm consists of the following stages (all respect a 1000 s wall‑time budget)

1️⃣  *Geometric seeds* – generate a family of exponentially decaying vectors
    a_i ∝ r^i, normalise them onto the simplex and keep the best one.

2️⃣  *Random restarts* – draw Dirichlet vectors, run a few fast Adam steps,
    keep the best candidate (the geometric seed is also used as a start).

3️⃣  *Adam on the smooth log‑sum‑exp surrogate* for a long period.

4️⃣  *Exact sub‑gradient descent* with back‑tracking line‑search.

5️⃣  *Targeted deterministic refinement* – repeatedly move a sizable fraction
    of probability mass from an index participating in the current maximal
    convolution entry to the current smallest entry; accept only if the exact
    objective improves (greedy descent).  Occasional sub‑gradient steps keep
    the walk directed.

6️⃣  Final verification via the provided ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x≥0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full convolution via FFT (fast, length L = next pow2(3*n-2))
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return a*a (full convolution) using FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Exact objective for a simplex‑normalised vector (Σa = 1)
def _objective(a: np.ndarray, n: int, L: int) -> float:
    """2·n·max(conv(a,a))   (a is assumed to sum to 1)."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Sub‑gradient of the maximal convolution entry
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. a."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
# Gradient of the log‑sum‑exp (smooth max) surrogate
def _smooth_gradient(a: np.ndarray, conv: np.ndarray,
                     alpha: float, L: int) -> np.ndarray:
    """∇a ( (2·n/α)·log Σ_k exp(α·conv_k) )."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)      # stabilise
    w = exp_scaled / exp_scaled.sum()             # soft‑max weights

    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n_len = a.size
    grad = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the smooth surrogate
def _adam_phase(a: np.ndarray, n: int, L: int,
                max_iters: int, deadline: float,
                alpha_start: float = 5.0,
                alpha_end: float = 20000.0,
                lr0: float = 0.6,
                beta1: float = 0.9,
                beta2: float = 0.999,
                eps: float = 1e-12,
                prune_every: int = 5000,
                prune_percentile: float = 1.0) -> tuple[np.ndarray, float, np.ndarray]:
    """Adam on the soft‑max surrogate."""
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_obj = _objective(a, n, L)

    total_iters = max_iters
    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[adam] iter {it} new best O={best_obj:.6f}")

        # exponential schedule for α – sharper towards the end
        frac = it / total_iters
        alpha = alpha_start * (alpha_end / alpha_start) ** frac

        grad = _smooth_gradient(a, conv, alpha, L)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        lr = lr0 / math.sqrt(it + 1)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter
        if np.random.rand() < 0.001:
            a_new += np.random.randn(n) * 1e-4

        a_new = _project_to_simplex(a_new)

        # prune tiny entries every few thousand steps
        if prune_every and it % prune_every == 0 and it > 0:
            thresh = np.percentile(a_new, prune_percentile)
            a_new[a_new < thresh] = 0.0
            a_new = _project_to_simplex(a_new)

        a = a_new

    return a, best_obj, best_a


# ----------------------------------------------------------------------
# Exact sub‑gradient descent with simple back‑tracking line search
def _subgradient_phase(a: np.ndarray, n: int, L: int,
                       max_iters: int, deadline: float,
                       eta0: float = 0.3,
                       line_search: bool = True,
                       verbose: bool = False) -> tuple[np.ndarray, float, np.ndarray]:
    """Projected sub‑gradient descent on the true objective."""
    best_a = a.copy()
    best_obj = _objective(a, n, L)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            if verbose and it % 5000 == 0:
                print(f"[sub‑grad] iter {it} new best O={best_obj:.6f}")

        g = _subgradient_max_conv(a, conv)

        # geometric decay of the step size
        eta = eta0 * (0.9999 ** it)

        if line_search:
            step = eta
            for _ in range(5):
                cand = _project_to_simplex(a - step * g)
                cand_obj = _objective(cand, n, L)
                if cand_obj < cur_obj:
                    a = cand
                    break
                step *= 0.5
            else:
                a = _project_to_simplex(a - eta * g)
        else:
            a = _project_to_simplex(a - eta * g)

    return a, best_obj, best_a


# ----------------------------------------------------------------------
# Generate a geometric (exponential) seed a_i ∝ r**i, normalised.
def _geometric_seed(n: int, r: float) -> np.ndarray:
    """Return a simplex‑normalised exponential vector with ratio r (0<r<1)."""
    idx = np.arange(n, dtype=float)
    w = np.power(r, idx)
    # avoid the all‑zero case when r is tiny
    if w.sum() == 0.0:
        w = np.ones_like(w)
    a = w / w.sum()
    return a

# Targeted deterministic refinement (greedy mass transfers)
def _deterministic_refinement(a: np.ndarray, n: int, L: int,
                             max_iters: int, deadline: float,
                             frac_transfer: float = 0.5,
                             subgrad_every: int = 20,
                             verbose: bool = True) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Greedy loop that moves a fraction of probability mass from an index
    participating in the current maximal convolution entry to the smallest
    entry.  Accepts only improvements.  Every ``subgrad_every`` iterations a
    small exact sub‑gradient step is taken to keep directionality.
    """
    best_a = a.copy()
    best_obj = _objective(a, n, L)
    cur_obj = best_obj
    conv = _conv_fft(a, L)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        kmax = int(np.argmax(conv))
        max_val = conv[kmax]

        # indices i that actually contribute (both sides > 0)
        i_low = max(0, kmax - (n - 1))
        i_high = min(n - 1, kmax)
        i_range = np.arange(i_low, i_high + 1, dtype=int)
        mask = (a[i_range] > 1e-16) & (a[kmax - i_range] > 1e-16)
        i_candidates = i_range[mask]

        if i_candidates.size == 0:
            # fallback: pick the global maximum entry
            i = np.argmax(a)
        else:
            # pick the pair that contributes the most to the peak
            prod = a[i_candidates] * a[kmax - i_candidates]
            i = i_candidates[np.argmax(prod)]

        # destination = smallest entry (different from i)
        j = np.argmin(a)
        if j == i:
            # second smallest
            sorted_idx = np.argsort(a)
            for cand in sorted_idx:
                if cand != i:
                    j = cand
                    break

        # transfer a * fraction (clamp to keep non‑negative)
        delta = a[i] * frac_transfer
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = _project_to_simplex(a_new)

        conv_new = _conv_fft(a_new, L)
        new_obj = 2.0 * n * conv_new.max()

        # accept only if improvement
        if new_obj < cur_obj:
            a = a_new
            conv = conv_new
            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_a = a.copy()
                if verbose:
                    print(f"[det] iter {it:6d} new best O={best_obj:.6f}")

        # occasional genuine sub‑gradient step to avoid stalls
        if it % subgrad_every == 0:
            g = _subgradient_max_conv(a, conv)
            eta_loc = 0.03 * (1.0 - it / max_iters)
            a = _project_to_simplex(a - eta_loc * g)
            conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                if verbose:
                    print(f"[det‑sub] iter {it:6d} sub‑grad improved O={best_obj:.6f}")

    return a, best_obj, best_a
