"""
Propose a non‑negative step function that approximately minimises

    O(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

The optimizer consists of three stages:

1️⃣  **Variance flattening** – Adam on the variance of the full convolution.
    This pushes the auto‑correlation towards a flat (almost constant) shape.

2️⃣  **Smooth‑max with spectral regularisation** – Adam on a log‑sum‑exp
    surrogate of the maximum together with a cheap penalty on high‑frequency
    Fourier components (encouraging a smooth time‑domain sequence).  The
    temperature α of the soft‑max is annealed from a small to a large value.

3️⃣  **Deterministic max‑reduction moves** – a fast hill‑climbing phase that
    repeatedly reduces the currently maximal convolution entry by moving a
    small amount of mass from a “high‑contributing’’ index to a “low‑contributing’’
    one (derived from the sub‑gradient of the max).  Small random restarts keep
    the search lively.

The whole routine respects the 1000 s budget, keeps the sequence on the
probability simplex (∑aᵢ = 1) and finally returns the best list of heights
found.  The evaluation is scale‑invariant, therefore fixing the sum to one
does not affect the objective.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex {x ≥ 0, sum(x)=1}
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        # all zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution a ★ a (length 2·n‑1) using FFT of length L
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return the full convolution a★a (length 2·n‑1) using FFT."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Gradient of the variance of the convolution vector
def _grad_variance(a: np.ndarray, conv: np.ndarray, n: int, L: int) -> np.ndarray:
    """
    Gradient of   f(a) = Σ_k (conv_k - μ)²   where μ = 1/(2n‑1) (for sum(a)=1).
    Closed form: g = 4·corr(conv‑μ , a).
    """
    mu = 1.0 / (2 * n - 1)
    diff = conv - mu                      # length 2n‑1
    a_rev = a[::-1]

    fft_diff = np.fft.rfft(diff, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_diff * fft_a_rev, n=L)

    start = n - 1
    grad = 4.0 * corr_rev[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Gradient of the log‑sum‑exp surrogate + tiny entropy term
def _smooth_gradient_plus_entropy(
    a: np.ndarray,
    conv: np.ndarray,
    alpha: float,
    L: int,
    lam: float,
) -> np.ndarray:
    """
    ∇_a ( (2·n/α)·log∑_k e^{α·conv_k} + λ·∑ a_i·log a_i ).
    The first part equals 2·corr(w,a) where w ∝ exp(α·conv).
    """
    # soft‑max weights
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / exp_scaled.sum()                 # shape (2n‑1,)

    # correlation term (same trick as in _grad_variance)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n_len = a.size
    grad_corr = 2.0 * corr_rev[n_len - 1:n_len - 1 + n_len]

    # tiny entropy (helps avoid exact zeros)
    eps = 1e-20
    grad_ent = lam * (np.log(a + eps) + 1.0)

    return grad_corr + grad_ent
