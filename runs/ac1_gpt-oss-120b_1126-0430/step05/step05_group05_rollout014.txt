"""
Propose a non‑negative step‑function sequence that (approximately) minimises

    O(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm consists of four stages:

1️⃣  **Randomised sub‑gradient descent** – simple additive sub‑gradient updates
    on the exact max‑convolution objective, with a decaying step‑size and occasional
    low‑pass smoothing.  This quickly breaks symmetry of the uniform start and
    drives the convolution peak down.

2️⃣  **Annealed Adam on a log‑sum‑exp surrogate** – optimises a smooth approximation
    of the max‑convolution (with an entropy regulariser) while projecting onto the
    probability simplex.  The temperature α is annealed from a small to a very large
    value, λ (entropy weight) is annealed to zero and the Adam step‑size decays as
    1/√t.

3️⃣  **Intensive simulated‑annealing refinement** – random “mass‑transfer’’ moves
    are accepted with a Metropolis rule; a cheap sub‑gradient step is performed
    periodically to keep pushing the peak down.

4️⃣  **Final polishing** – a short high‑α Adam run to fine‑tune the remaining slack.

All phases respect the 1000 s budget, print progress and finally return the best
sequence (as a plain Python list) found.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------
# Euclidean projection onto the probability simplex {x≥0, sum(x)=1}
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
# Full linear convolution via FFT (length 2·n‑1)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# --------------------------------------------------------------
# Exact objective (sum(a) assumed to be 1)
def _objective(a: np.ndarray, n: int, L: int) -> float:
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# --------------------------------------------------------------
# Sub‑gradient of max(conv) w.r.t. a (averaged over all maximisers)
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


# --------------------------------------------------------------
# Gradient of the smooth surrogate (log‑sum‑exp) plus entropy term
def _smooth_gradient_plus_entropy(
    a: np.ndarray,
    conv: np.ndarray,
    alpha: float,
    L: int,
    lam: float,
) -> np.ndarray:
    # ---- soft‑max weights ----------
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)          # for stability
    w = exp_scaled / exp_scaled.sum()                 # shape (2n‑1,)

    # ---- correlation w ⋆ a (g_i = 2·∑_j w_{i+j} a_j) ----
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n_len = a.size
    grad_corr = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]

    # ---- entropy part ----------
    eps = 1e-20
    grad_ent = lam * (np.log(a + eps) + 1.0)

    return grad_corr + grad_ent

            kernel = np.array([0.25, 0.5, 0.25])
            a = np.convolve(a, kernel, mode="same")
            a = a / a.sum()

        it += 1

    # ----------------------------------------------------------
    # Phase 2 – annealed Adam on smooth surrogate
    print("[phase‑2] annealed Adam start")
    a = best_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    it = 0
    max_iter_adam = 300_000
    alpha_start, alpha_end = 5.0, 200_000.0
    lam_start, lam_end = 0.2, 0.0
    while time.time() < deadline and it < max_iter_adam:
        conv = _conv_fft(a, L)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[adam] iter {it:,} new best O = {best_score:.6f}")

        frac = it / max_iter_adam
        alpha = alpha_start + (alpha_end - alpha_start) * frac
        lam = lam_start * (1.0 - frac)

        grad = _smooth_gradient_plus_entropy(a, conv, alpha, L, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        lr = 0.9 / math.sqrt(it + 1.0)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_to_simplex(a_new)

        it += 1

    # ----------------------------------------------------------
    # Phase 3 – simulated‑annealing local refinement
    print("[phase‑3] simulated annealing start")
    a = best_seq.copy()
    temperature0 = 0.05
    temp_decay = 0.99985
    max_iter_sa = 300_000
    it = 0
    T = temperature0
    while time.time() < deadline and it < max_iter_sa:
        # current convolution and score (kept up‑to‑date)
        conv_cur = _conv_fft(a, L)
        cur_score = 2.0 * n * conv_cur.max()

        # ----- random mass‑transfer move -----
        i, j = np.random.randint(n, size=2)
        if i != j and a[i] > 1e-14:
            # fraction of source mass to move (decreases over time)
            max_frac = 0.5 * (1.0 - it / max_iter_sa) + 0.02
            frac = np.random.rand() * max_frac
            delta = a[i] * frac
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            # sum stays 1, only need to keep non‑negativity
            a_new = np.clip(a_new, 0.0, None)

            conv_new = _conv_fft(a_new, L)
            new_score = 2.0 * n * conv_new.max()

            # Metropolis acceptance
            if new_score < cur_score or np.random.rand() < math.exp((cur_score - new_score) / max(T, 1e-16)):
                a = a_new
                cur_score = new_score
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(f"[SA] iter {it:,} new global best O = {best_score:.6f}")

        # ----- occasional additive sub‑gradient step -----
        if it % 100 == 0:
            conv_cur = _conv_fft(a, L)
            g = _subgradient_max_conv(a, conv_cur)
            step = 0.12 * (1.0 - it / max_iter_sa)
            a = a - step * g
            a = np.clip(a, 0.0, None)
            a = a / a.sum()
            conv_cur = _conv_fft(a, L)
            cur_score = 2.0 * n * conv_cur.max()
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[SA‑sub] iter {it:,} new best O = {best_score:.6f}")

        # ----- gentle smoothing every few thousand iterations -----
        if it % 5000 == 0 and it > 0:
            kernel = np.array([0.125, 0.25, 0.25, 0.25, 0.125])
            a = np.convolve(a, kernel, mode="same")
            a = a / a.sum()

        T *= temp_decay
        it += 1

    # ----------------------------------------------------------
    # Phase 4 – final high‑α Adam polishing
    print("[phase‑4] final polishing")
    a = best_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    it = 0
    max_iter_polish = 30_000
    alpha = 500_000.0
    lam = 0.0
    while time.time() < deadline and it < max_iter_polish:
        conv = _conv_fft(a, L)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[polish] iter {it:,} new best O = {best_score:.6f}")

        grad = _smooth_gradient_plus_entropy(a, conv, alpha, L, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        lr = 0.5 / math.sqrt(it + 1.0)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_to_simplex(a_new)

        it += 1

    # ----------------------------------------------------------
    # Final sanity check with the provided evaluator
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O = {final_score:.6f}")
    except Exception as e:
        print("[final] evaluation failed:", e)

    return best_seq.tolist()
