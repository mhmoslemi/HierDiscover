def _proj_simplex(v):
    # ensure v is np array
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1

    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _run_proj_opt(n, time_limit, init_a=None, seed=0):
    start = time.time()
    rng = np.random.default_rng(seed)
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
        a = a / np.sum(a)
    else:
        a = rng.random(n)
        a = a / np.sum(a)

    # define parameters: initial learning rate, decay schedule
    lr0 = 0.5
    lr_end = 1e-4
    # schedule based on elapsed proportion
    best_a = a.copy()
    best_obj = _objective(best_a)

    it = 0
    while time.time() - start < time_limit:
        it += 1
        # compute self-convolution
        conv = _conv_self_fft(a)
        kmax = int(np.argmax(conv))

        # compute gradient (exact subgradient)
        # gradient_i = 2 * a[kmax - i] if 0 <= kmax - i < n else 0
        grad = np.zeros_like(a)
        low = max(0, kmax - (n - 1))
        high = min(n - 1, kmax)
        # i from low..high correspond to kmax - i in 0..n-1
        i_range = np.arange(low, high + 1)
        grad[i_range] = a[kmax - i_range] * 2.0

        # learning rate decay
        t = min((time.time() - start) / time_limit, 1.0)
        lr = lr0 * (lr_end / lr0) ** t  # exponential decay

        # gradient step with projection onto simplex
        a_new = a - lr * grad
        a = _proj_simplex(a_new)

        # evaluate objective
        obj = _objective(a)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[n={n:5d} it={it:06d}] new best O={best_obj:.6f}")

        # occasional jitter to avoid stagnation
        if it % 5000 == 0:
            jitter = rng.normal(scale=0.01, size=n)
            a = _proj_simplex(a + jitter)

        # restart from best periodically
        if it % 20000 == 0:
            a = best_a.copy()
            jitter = rng.normal(scale=0.005, size=n)
            a = _proj_simplex(a + jitter)

    # compress negligible entries
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = _objective(best_a)
        print(f"[compress] reduced length → {best_a.size}, O={best_obj:.6f}")

    return best_a.tolist(), best_obj

def _proj_simplex(v):
    """Project onto the probability simplex (sum==1, v>=0)."""
    # From Duchi et al. 2008
    v = np.asarray(v, dtype=float)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        # all <=0, return uniform?
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Hybrid optimiser for the step‑function attack.

The routine consists of three stages that respect the 1000 s budget.

1. **Fast Adam optimiser** on the log‑it (soft‑max) parametrisation of the
   simplex.  A smoothly‑annealed soft‑max (log‑sum‑exp) approximation of the
   max‑convolution is used, with a much larger final β (≈5·10⁴) than the
   reference implementation.  The learning‑rate also decays exponentially.
   Several support sizes are tried; the longest run (≈60 % of the total time)
   uses a large support (up to 16 k entries) which gives the optimiser many
   degrees of freedom.

2. **Power‑law reshaping** – after the Adam run the current sequence is
   re‑weighted by several exponents p∈{0.5,0.7,0.9,1.1,1.3}.  The sequence is
   re‑normalised and the best exponent is kept.  This cheap non‑linear
   transformation often reduces the L∞‑norm of the self‑convolution.

3. **Targeted simulated‑annealing** (SA) fine‑tuning.  The SA moves a small
   amount of mass away from indices that contribute most to the current
   maximal convolution entry, accepting non‑improving moves with a
   Metropolis probability.  The SA runs only on the best candidate found
   so far and finishes with a final pruning of negligible entries.

All helper functions are top‑level, no closures or filesystem access is used.
The entry point `propose_candidate` returns a plain list of floats suitable
for `evaluate_sequence`.
"""

import time
import numpy as np
from scipy.optimize import linprog  # (optional, not used in the final version)

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using real FFT (length 2·len(a)-1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return conv_full[: 2 * n - 1]


def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation g_i = Σ_t w[i+t]·a[t] for i=0,…,n‑1.
    Implemented via FFT.
    """
    n = a.shape[0]
    m = w.shape[0]                     # = 2n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


# ----------------------------------------------------------------------
# Simple objective: 2·n·max(conv)  (sum of a is 1 → denominator = 1)
# ----------------------------------------------------------------------


def _objective(a: np.ndarray) -> float:
    """Exact objective for a simplex‑normalised vector."""
    n = a.size
    conv = _conv_self_fft(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Adam optimiser on the log‑it simplex (improved hyper‑parameters)
# ----------------------------------------------------------------------


def _run_adam_opt(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
) -> tuple[list[float], float]:
    """
    Adam optimisation on the log‑it simplex.
    Returns best sequence (list) and its exact objective.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # initialise a on the simplex
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
        a = a / a.sum()
    else:
        a = rng.random(n)
        a = a / a.sum()

    x = np.log(a + 1e-16)               # logits
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # schedule parameters – more aggressive than the reference run
    beta_start = 0.5
    beta_end = 50000.0                 # much sharper smoothing
    lr_start = 0.25
    lr_end = 0.001

    best_a = a.copy()
    best_obj = _objective(best_a)

    it = 0
    while time.time() - start < time_limit:
        it += 1
        a = _softmax(x)                     # stay on the simplex

        conv = _conv_self_fft(a)
        M = conv.max()
        obj = float(2.0 * n * M)            # denominator = 1
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[n={n:5d} it={it:06d}] new best O={best_obj:.6f}")

        # annealing / learning‑rate schedule (time‑based)
        t = min((time.time() - start) / time_limit, 1.0)
        beta = beta_start * np.exp(np.log(beta_end / beta_start) * t)
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t)

        # smooth‑max weights
        if beta > 5000.0:                # switch to exact max for the tail
            kmax = int(np.argmax(conv))
            w = np.zeros_like(conv)
            w[kmax] = 1.0
        else:
            shifted = conv - M            # ≤ 0
            w = np.exp(beta * shifted)
            s = w.sum()
            if s == 0.0:
                w = np.ones_like(conv) / conv.size
            else:
                w /= s

        # gradient of (smooth) max w.r.t. a
        g_a = 2.0 * _correlation_fft(w, a)

        # natural gradient in log‑it space (mirrored gradient)
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape plateaus
        if it % 4000 == 0:
            jitter = rng.normal(scale=0.008, size=n)
            x += jitter

        # restart from current best now and then
        if it % 60000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.004, size=n)
            m.fill(0.0)
            v.fill(0.0)

    # -----------------------------------------------------------------
    # prune negligible entries (helps later local search)
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = _objective(best_a)
        print(f"[compress] reduced length → {best_a.size}, O={best_obj:.6f}")

    # -----------------------------------------------------------------
    # optional short greedy local refinement (mass‑transfer moves)
    remaining = time_limit - (time.time() - start)
    if remaining > 1.0:
        local_start = time.time()
        a_loc = best_a.copy()
        obj_loc = best_obj
        n_loc = a_loc.size
        rng_local = np.random.default_rng(seed + 5555)
        while time.time() - local_start < remaining * 0.2:
            i, j = rng_local.choice(n_loc, size=2, replace=False)
            delta = a_loc[i] * rng_local.random() * 0.015
            if delta <= 0.0:
                continue
            a_new = a_loc.copy()
            a_new[i] -= delta
            a_new[j] += delta
            conv_new = _conv_self_fft(a_new)
            obj_new = float(2.0 * n_loc * conv_new.max())
            if obj_new < obj_loc:
                a_loc, obj_loc = a_new, obj_new
                if obj_new < best_obj:
                    best_a, best_obj = a_new, obj_new
        print(f"[local] final local O={best_obj:.6f}")

    return best_a.tolist(), best_obj


# ----------------------------------------------------------------------
# Power‑law reshaping (cheap but often beneficial)
# ----------------------------------------------------------------------


def _power_transform_refine(a: np.ndarray, current_obj: float) -> tuple[np.ndarray, float]:
    """
    Try a few exponents p and keep the best transformed sequence.
    """
    best_a = a
    best_obj = current_obj
    for p in (0.5, 0.7, 0.9, 1.1, 1.3):
        a_p = np.power(a, p)
        s = a_p.sum()
        if s <= 0.0:
            continue
        a_p /= s
        obj = _objective(a_p)
        if obj < best_obj:
            best_a, best_obj = a_p, obj
    return best_a, best_obj


# ----------------------------------------------------------------------
# Targeted simulated‑annealing refinement (unchanged)
# ----------------------------------------------------------------------


def _run_sa(
    init_seq: list[float],
    time_limit: float,
    seed: int = 0,
) -> tuple[list[float], float]:
    """
    Targeted simulated‑annealing refinement.
    Moves small amounts of mass from positions that contribute most
    to the current maximal convolution entry.
    """
    start = time.time()
    rng = np.random.default_rng(seed)
    a = np.asarray(init_seq, dtype=float)
    if a.sum() < 1e-12:
        a = np.ones_like(a) / a.size
    else:
        a = a / a.sum()
    n = a.shape[0]

    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())
    best_seq = a.copy()

    # temperature schedule
    T = best_obj * 0.12 + 1e-7
    T_min = best_obj * 1e-6 + 1e-9
    decay = 0.997
    it = 0

    while time.time() - start < time_limit:
        it += 1
        k_max = int(np.argmax(conv))

        i_vals = np.arange(n)
        j_vals = k_max - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i_vals[mask]] * a[j_vals[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            probs = contrib / total_contrib
            i = rng.choice(n, p=probs)
        else:
            i = rng.integers(n)

        # different destination index
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.07   # up to 7 % of a_i
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.clip(a_new, 0.0, None)
        a_new /= a_new.sum()

        conv_new = _conv_self_fft(a_new)
        obj_new = float(2.0 * n * conv_new.max())

        if obj_new < best_obj - 1e-13:
            a, conv, best_obj, best_seq = a_new, conv_new, obj_new, a_new.copy()
            print(f"[SA it={it}] new best O={best_obj:.6f}")
        else:
            prob = np.exp(-(obj_new - best_obj) / T) if T > 0 else 0.0
            if rng.random() < prob:
                a, conv = a_new, conv_new

        if it % 2500 == 0:
            T = max(T * decay, T_min)

        if it % 12000 == 0:
            noise = rng.normal(scale=0.008, size=n)
            a = np.clip(a + noise, 0.0, None)
            a /= a.sum()
            conv = _conv_self_fft(a)

    # final compression
    thresh = 1e-12
    mask = best_seq > thresh
    if mask.sum() < best_seq.size:
        comp = best_seq[mask]
        comp /= comp.sum()
        comp_obj = float(2.0 * comp.size * _conv_self_fft(comp).max())
        if comp_obj < best_obj:
            best_seq, best_obj = comp, comp_obj
            print(f"[SA compress] reduced length → {comp.size}, O={best_obj:.6f}")

    return best_seq.tolist(), best_obj

# Main entry point – orchestrates all phases
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **_ignore,
) -> list[float]:
    """
    Return a step‑function sequence that (hopefully) beats the previous
    best upper bound.  The routine respects the overall time budget,
    runs several Adam optimisations on increasing support sizes,
    refines the best candidate with a cheap power‑law transform and a
    focused simulated‑annealing search, and finally returns the best
    sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # -----------------------------------------------------------------
    # Load a previous construction if available.
    init_seq = None
    if 'height_sequence_1' in globals():
        seq0 = np.asarray(height_sequence_1, dtype=float)
        seq0 = np.clip(seq0, 0.0, None)
        if seq0.sum() < 0.01:
            seq0 = np.ones_like(seq0)
        seq0 = seq0 / seq0.sum()
        init_seq = seq0

    # Candidate lengths – longest first (more flexibility)
    lengths = [16000, 12000, 8000, 6000, 4000, 2000]
    if init_seq is not None and init_seq.size not in lengths:
        lengths.insert(0, init_seq.size)

    best_seq = None
    best_obj = float('inf')

    # -----------------------------------------------------------------
    # Run the Adam optimiser for each length, allocating most time to
    # the longest run.
    for idx, n in enumerate(lengths):
        now = time.time()
        remaining = deadline - now
        if remaining <= 3.0:
            break

        # time budget for this size
        if idx == 0:
            run_time = remaining * 0.60          # bulk of the time on the largest n
        else:
            # split the rest roughly evenly
            run_time = (remaining * 0.35) / (len(lengths) - idx)

        # initialise from the previous best if dimensions match
        init = None
        if best_seq is not None:
            if best_seq.size == n:
                init = best_seq
            elif best_seq.size < n:
                # pad with tiny uniform mass
                pad = np.full(n - best_seq.size, 1e-12)

        if idx == 0 and init_seq is not None and n == init_seq.size:

        # ---- Adam optimisation -------------------------------------------------
        seq, val = _run_adam_opt(

        a_arr = np.asarray(seq, dtype=float)
        a_arr, val = _power_transform_refine(a_arr, val)

        if val < best_obj:

        # stop early if we are out of time

    # -----------------------------------------------------------------
    # If time permits, perform a focussed simulated‑annealing refinement.
    remaining = deadline - time.time()

            best_seq.tolist(),
            time_limit=remaining,
            seed=seed + 7777,
        )
        if val_refined < best_obj:
            best_obj, best_seq = val_refined, np.asarray(seq_refined, dtype=float)
            print(f"[final] after SA new best O={best_obj:.6f}")

    # -----------------------------------------------------------------
    # Fallback – never expected to happen.
    if best_seq is None or best_seq.size == 0:
        best_seq = np.ones(100, dtype=float) / 100.0
        best_obj = _objective(best_seq)

    # -----------------------------------------------------------------
    # Final sanity check / reporting.
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_seq.tolist()
