def propose_candidate(seed=42, budget_s=1000, **_ignore):

    import time, numpy as np

    rng = np.random.default_rng(seed)

    deadline = time.time() + budget_s - 0.5

    # set lengths
    lengths = [256, 512, 1024, 2048, 4096]
    # add previous sequence if defined:
    try:
        height_sequence_1
    except NameError:
        height_sequence_1 = None

    if 'height_sequence_1' in globals():
        seq0 = np.asarray(globals()['height_sequence_1'], dtype=float)
        seq0 = np.clip(seq0, 0, None)
        if seq0.sum() < 0.01:
            seq0 = np.ones_like(seq0)
        seq0 = seq0 / seq0.sum()
        n_seq0 = seq0.size
        if n_seq0 not in lengths:
            lengths.insert(0, n_seq0)

    best_seq = None
    best_obj = float('inf')

    # Phase 1: quick search
    for idx, n in enumerate(lengths):
        remaining = deadline - time.time()
        if remaining < 5.0:
            break
        # allocate small time per length (maybe 5% of remaining or min)
        run_time = min(5.0, remaining * 0.07)  # ~5 sec.

        # generate initial guess
        init = None
        # try edge heavy shape
        p = -1.0  # exponent
        i = np.arange(n)
        a_edge = (i+1)**p + (n-i)**p
        a_edge = a_edge / a_edge.sum()
        # combine with uniform
        a_comb = 0.6 * a_edge + 0.4 * rng.random(n)
        a_comb = a_comb / a_comb.sum()
        init = a_comb.astype(float)

        seq, val = _run_adam_opt(n=n, time_limit=run_time, init_a=init, seed=seed + idx*13)
        if val < best_obj:
            best_obj = val
            best_seq = seq
            print(f"[quick] new best O={best_obj:.6f} at n={n}")

    # Phase 2: refined optimization on best length
    if best_seq is not None:
        n_best = len(best_seq)
        remaining = deadline - time.time()
        if remaining > 5.0:
            # allocate most of remaining to refined run
            refined_time = remaining * 0.85
            seq_ref, val_ref = _run_adam_opt(n=n_best, time_limit=refined_time, init_a=best_seq, seed=seed + 999)
            if val_ref < best_obj:
                best_obj = val_ref
                best_seq = seq_ref
                print(f"[refined] improved O={best_obj:.6f} (n={n_best})")
    else:
        # fallback: random initialization length 512
        n_best = 512
        seq_ref, val_ref = _run_adam_opt(n=n_best, time_limit=budget_s-1.0, init_a=None, seed=seed)
        best_seq = seq_ref
        best_obj = val_ref

    # Phase 3: targeted local search
    remaining = deadline - time.time()
    if remaining > 1.0:
        print(f"[local] starting targeted local search for {remaining:.1f}s")
        seq_final, obj_final = _targeted_local_search(np.array(best_seq, dtype=float), time_limit=remaining, seed=seed + 777)
        best_seq = seq_final
        best_obj = obj_final
        print(f"[final local] O={best_obj:.6f}")

    # final sanity check
    try:
        final_score = evaluate_sequence(best_seq)
        print(f"[final] evaluate_sequence O={final_score:.6f}")
    except Exception as e:
        print("[final] evaluate_sequence failed:", e)

    if not best_seq:
        best_seq = [1.0]

    return best_seq

                a = a[:n]
            else:
                pad = np.zeros(n - a.size)
                a = np.concatenate([a, pad])
        a = np.clip(a, 0, None)
        if a.sum() < 1e-6:
            a = rng.random(n)
        a = a / a.sum()
    else:
        a = rng.random(n)
        a = a / a.sum()
    x = np.log(a + 1e-16)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999

"""
Step‑function optimisation for the “max self‑convolution” bound.

The algorithm works in three stages:

1. **Exploratory Adam runs** on several support sizes using a
   log‑it (soft‑max) parametrisation of the simplex.
   Initialisations are edge‑heavy (≈ 1/(i+1)+1/(n‑i)) which
   encourages the convolution to flatten early.

2. **Refined Adam optimisation** on the best‑performing length,
   allocating the majority of the remaining budget.
   The optimiser uses a smooth‑max approximation of the
   convolution maximum that is annealed to a hard max
   (β grows from ~0.3 to 4000).  Adam’s moments are reset on
   periodic restarts and jitter is added to escape plateaus.

3. **Targeted local search** that repeatedly moves a small
   fraction of mass from the index participating most in the
   current maximal convolution entry to an opposite edge.
   The move is accepted only if the exact objective improves.

All numbers are kept non‑negative and normalised to sum = 1,
so the security checks of ``evaluate_sequence`` are satisfied.
The routine returns the best sequence found before the 1000 s
deadline.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution of a (real‑valued) using FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv


def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation of w (length 2·n‑1) with a (length n):
    (g_i = Σ_t w[i+t]·a[t]), 0 ≤ i < n.
    Implemented via FFT.
    """
    n = a.shape[0]
    m = w.shape[0]                 # = 2·n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) under the assumption Σa=1."""
    n = a.size
    conv = _conv_self_fft(a)
    return 2.0 * n * conv.max()


def _run_adam_opt(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
) -> tuple[list[float], float]:
    """
    Adam optimisation of a length‑`n` sequence for at most `time_limit` seconds.
    Returns the best sequence (plain list) and its exact objective.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # -----------------------------------------------------------------
    # initialise a on the simplex
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            if a.size > n:
                a = a[:n]
            else:
                a = np.concatenate([a, np.zeros(n - a.size)])
        a = np.clip(a, 0, None)
        if a.sum() < 1e-6:
            a = rng.random(n)
    else:
        a = rng.random(n)

    a = a / a.sum()
    # work in log‑it space (automatically respects the simplex)
    x = np.log(a + 1e-16)

    m = np.zeros_like(x)          # Adam first moment
    v = np.zeros_like(x)          # Adam second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # schedule parameters
    beta_start = 0.3               # smoothness start
    beta_end   = 4000.0            # very sharp at the end
    lr_start   = 0.12
    lr_end     = 0.001
    # a very generous iteration budget – we stop by time, not by count
    target_iters = 800_000

    best_a = a.copy()
    best_obj = _objective(best_a)

    it = 0
    while time.time() - start < time_limit:
        it += 1

        # probabilities from current logits
        a = _softmax(x)

        # exact convolution & objective
        conv = _conv_self_fft(a)
        M = conv.max()
        obj = 2.0 * n * M
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            if it % 5000 == 0 or it == 1:
                print(f"[n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        # ---- annealing schedules ------------------------------------
        t_frac = min(it / target_iters, 1.0)
        beta = beta_start * np.exp(np.log(beta_end / beta_start) * t_frac)
        lr   = lr_start   * np.exp(np.log(lr_end   / lr_start)   * t_frac)

        # ---- smooth‑max weight --------------------------------------
        if beta > 2000.0:                      # sharp enough → treat as hard max
            kmax = int(np.argmax(conv))
            # direct sub‑gradient for the hard max (no FFT needed)
            g_a = np.zeros_like(a)
            i_start = max(0, kmax - (n - 1))
            i_end   = min(n - 1, kmax)
            i_idx = np.arange(i_start, i_end + 1)
            j_idx = kmax - i_idx
            # ∂conv_kmax/∂a_i = 2 * a_j (j = kmax - i)
            g_a[i_idx] = 2.0 * a[j_idx]
        else:
            shifted = conv - M                     # ≤ 0
            w = np.exp(beta * shifted)
            w_sum = w.sum()
            if w_sum == 0:
                w = np.ones_like(conv) / conv.size
            else:
                w = w / w_sum
            g_a = 2.0 * _correlation_fft(w, a)    # (n,)

        # ---- natural gradient on the simplex (log‑it space) -----
        dot = np.dot(a, g_a)                 # ⟨a, g_a⟩
        g_x = a * (g_a - dot)                # projected onto tangent space

        # ---- Adam update -------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---- occasional jitter --------------------------------------
        if it % 4000 == 0:
            jitter = rng.normal(scale=0.02, size=n)
            x += jitter

        # ---- periodic full restart ----------------------------------
        if it % 60000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.005, size=n)
            m.fill(0.0)
            v.fill(0.0)
            print(f"[n={n}] restart at iter {it}")

    # -----------------------------------------------------------------
    # discard negligible entries (helps with very long runs)
    mask = best_a > 1e-12
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = _objective(best_a)
        print(f"[compress] reduced length {n} → {best_a.size}, O={best_obj:.6f}")

    return best_a.tolist(), best_obj


def _targeted_local_search(
    a: np.ndarray,
    time_limit: float,
    seed: int = 0,
) -> tuple[list[float], float]:
    """
    Directed hill‑climbing: repeatedly move a small fraction of mass
    from the index that contributes most to the current maximal
    convolution entry to a far‑edge position.
    Stops after `time_limit` seconds.
    """
    start = time.time()
    rng = np.random.default_rng(seed)
    n = a.size
    best_a = a.copy()
    best_obj = _objective(best_a)

    while time.time() - start < time_limit:
        conv = _conv_self_fft(a)
        max_val = conv.max()
        k = int(conv.argmax())                     # shift with maximal overlap

        # indices that actually contribute to conv[k]
        i_start = max(0, k - (n - 1))
        i_end   = min(n - 1, k)
        if i_start > i_end:
            # should not happen, but guard against it
            continue
        i_range = np.arange(i_start, i_end + 1)
        j_range = k - i_range
        contrib = a[i_range] * a[j_range]

        # pick the index with the largest product a_i * a_j
        idx = i_range[np.argmax(contrib)]
        jdx = k - idx

        # choose an edge far from idx (and also not equal to jdx)
        if idx < n // 2:
            target = n - 1
        else:
            target = 0
        if target == idx or target == jdx:
            target = 0 if target == n - 1 else n - 1

        # move a small fraction of mass from idx towards the edge
        delta = a[idx] * rng.random() * 0.03   # ≤ 3 % of its weight
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[idx] -= delta
        a_new[target] += delta
        if a_new[idx] < 0.0:
            continue  # discard illegal move

        obj_new = _objective(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            # Occasionally print progress (very cheap)
            if rng.random() < 0.005:
                print(f"[local] improvement O={best_obj:.6f}")

        # optionally also try to shift a little from jdx (second end of the pair)
        delta2 = a[jdx] * rng.random() * 0.03
        if delta2 > 0.0:
            a2 = a.copy()
            a2[jdx] -= delta2
            a2[target] += delta2
            if a2[jdx] >= 0.0:
                obj2 = _objective(a2)
                if obj2 < best_obj:
                    best_obj = obj2
                    best_a = a2.copy()
                    a = a2

    return best_a.tolist(), best_obj


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **_ignore,
) -> list[float]:
    """
    Entry point for the competition harness.

    The function explores several support sizes, runs a short Adam
    optimisation for each, keeps the best result and then performs a
    long refined Adam optimisation on that length followed by a
    targeted local search.  The final sequence is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # -------------------------------------------------------------
    # Determine lengths to try – start with a geometric ladder.
    lengths = [256, 512, 1024, 2048, 4096]

    # If a previous construction exists, try its length first.
    if "height_sequence_1" in globals():
        seq0 = np.asarray(globals()["height_sequence_1"], dtype=float)
        seq0 = np.clip(seq0, 0.0, None)
        if seq0.sum() < 0.01:
            seq0 = np.ones_like(seq0)
        seq0 = seq0 / seq0.sum()
        n0 = seq0.size
        if n0 not in lengths:
            lengths.insert(0, n0)
    else:
        seq0 = None

    best_seq = None
    best_obj = float("inf")

    # -------------------------------------------------------------
    # Phase 1 – quick exploratory runs on all chosen lengths.
    for idx, n in enumerate(lengths):
        remaining = deadline - time.time()
        if remaining <= 2.0:
            break

        # allocate a modest slice of the remaining budget (≈ 5 s or 7 % of the rest)
        run_time = min(5.0, remaining * 0.07)

        # build an edge‑heavy initial guess (p = –1)
        i = np.arange(n, dtype=float)
        p = -1.0
        a_edge = (i + 1) ** p + (n - i) ** p
        a_edge = a_edge / a_edge.sum()
        # blend with a tiny random component to avoid exact symmetry
        init_a = 0.7 * a_edge + 0.3 * np.random.default_rng(seed + idx).random(n)
        init_a = init_a / init_a.sum()

        seq, val = _run_adam_opt(
            n=n,
            time_limit=run_time,
            init_a=init_a,
            seed=seed + idx * 13,
        )
        if val < best_obj:
            best_obj = val
            best_seq = seq
            print(f"[quick] new overall best O={best_obj:.6f} (n={n})")

    # -------------------------------------------------------------
    # Phase 2 – long refined optimisation on the currently best length.
    if best_seq is not None:
        n_best = len(best_seq)
        remaining = deadline - time.time()
        if remaining > 5.0:
            refined_time = remaining * 0.85
            seq_ref, val_ref = _run_adam_opt(
                n=n_best,
                time_limit=refined_time,
                init_a=best_seq,
                seed=seed + 999,
            )
            if val_ref < best_obj:
                best_obj = val_ref
                best_seq = seq_ref
                print(f"[refined] improved O={best_obj:.6f} (n={n_best})")
    else:
        # fallback: a reasonable default length
        n_best = 1024
        best_seq, best_obj = _run_adam_opt(
            n=n_best,
            time_limit=budget_s - 1.0,
            init_a=None,
            seed=seed,
        )
        print(f"[fallback] O={best_obj:.6f} (n={n_best})")

    # -------------------------------------------------------------
    # Phase 3 – targeted local search for the final polish.
    remaining = deadline - time.time()
    if remaining > 1.0:
        print(f"[local] starting targeted search for {remaining:.1f}s")
        a_arr = np.array(best_seq, dtype=float)
        seq_final, obj_final = _targeted_local_search(
            a=a_arr,
            time_limit=remaining,
            seed=seed + 777,
        )
        best_seq = seq_final
        best_obj = obj_final
        print(f"[final local] O={best_obj:.6f}")

    # -------------------------------------------------------------
    # Final sanity check via the provided evaluation routine.
    try:
        final_score = evaluate_sequence(best_seq)
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    # guarantee a non‑empty return
    if not best_seq:
        best_seq = [1.0]

    return best_seq
