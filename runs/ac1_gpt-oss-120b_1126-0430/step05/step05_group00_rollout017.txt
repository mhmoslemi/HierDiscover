def project_onto_simplex(v, s=1):
    """Projects vector v onto the simplex {x>=0, sum x = s}."""
    # implement Duchi algorithm.
    if np.sum(v) == s and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

Will compute O(n log n) sorting each iteration.

def project_simplex(v, s=1.0):
    n = v.shape[0]
    # shift sorted desc
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

import time, math, numpy as np

def _fft_len_for_corr(n):
    return 1 << ((3 * n - 2).bit_length())

def _softmax(z):
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()

def _logits_from_distribution(a, eps=1e-16):
    return np.log(a + eps)

def _conv_max(a):
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[:2*n-1]
    return float(conv.max())

def _conv_max_and_grad(a, eps_factor):
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[:2*n-1]
    max_conv = float(conv.max())
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or np.isnan(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]
    grad_a = 4.0 * n * corr
    return max_conv, grad_a

def _max_conv_and_argmax(a):
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[:2*n-1]
    max_val = float(conv.max())
    idx = int(np.argmax(conv))
    return max_val, idx

def _project_simplex(v, s=1.0):
    # Duchi et al. projection onto simplex.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond].max()
        theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _adam_optimize(a0, n, deadline, start_time):
    # returns best a
    a = a0.copy()
    best_a = a.copy()
    L = _fft_len_for_corr(n)
    best_score = 2.0 * n * _conv_max(a)  # cheap approximate with sum=1
    # logits
    z = _logits_from_distribution(a)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_alpha = 0.8
    eps_factor = 0.08
    eps_factor_min = 5e-4
    eps_factor_decay = 0.9996
    it = 0
    max_iters = 250_000
    while time.time() < deadline and it < max_iters:
        it += 1
        a = _softmax(z)
        max_conv, grad_a = _conv_max_and_grad(a, eps_factor)
        cheap_score = 2.0 * n * max_conv
        if cheap_score < best_score:
            best_score = cheap_score
            best_a = a.copy()
        # gradient w.r.t logits
        mu = np.dot(grad_a, a)
        grad_z = a * (grad_a - mu)
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step = - alpha_t * m_hat / (np.sqrt(v_hat) + eps_adam)
        z = z + step
        z = np.clip(z, -50.0, 50.0)
        eps_factor = max(eps_factor_min, eps_factor * eps_factor_decay)
        # jitter occasionally
        if it % 4000 == 0 and np.random.rand() < 0.1:
            z += 0.003 * np.random.randn(n)
        # optional early break
        if it % 20000 == 0:
            print(f"Adam it {it}, best_score {best_score:.6f}")
        # break if time near end:
        if time.time() > deadline - 0.5:
            break
    return best_a, best_score

def _subgradient_optimize(a0, n, deadline, start_time):
    a = a0.copy()
    best_a = a.copy()
    best_score = 2.0 * n * _conv_max(a)
    it = 0
    step0 = 0.2

            # optional print
            if it % 10000 == 0:
                print(f"Subgrad it {it}, new best {best_score:.6f}")
        if time.time() > deadline - 0.2:
            break
    return best_a, best_score

#!/usr/bin/env python
"""
Optimise a non‑negative step‑function (height sequence) for the evaluation
metric

    J(a) = 2·n·max(a*a) / (sum(a))² .

The algorithm proceeds as follows:

1.  Choose a length `n`.  If the platform already provided a sequence
    (`height_sequence_1`) we reuse its length, otherwise we default to
    20 000 (the length used in the reference solution).

2.  Initialise the vector with the symmetric beta‑distribution
    a(x) ∝ x^{α‑1} (1‑x)^{α‑1}.  The classical arcsine distribution corresponds
    to α = 0.5.  A quick sweep over several α values and simple mixtures with a
    uniform vector is performed – the best candidate of this cheap search is
    taken as the starting point.

3.  Refine the candidate on the probability simplex using Adam on the logits
    (soft‑max parametrisation).  The objective used inside Adam is a smooth
    approximation of the max‑convolution; the smoothing constant is reduced
    gradually and the optimisation runs until the time budget is exhausted.

4.  Finish with a large number of cheap sub‑gradient iterations.  Each
    iteration computes the exact location of the current maximal convolution
    entry, takes a step opposite to its sub‑gradient and projects back onto the
    simplex.  This phase is inexpensive (one FFT per iteration) and therefore
    can perform many more updates than the Adam phase.

5.  The best vector found during the whole run is returned (as a plain list)
    after a final exact call to ``evaluate_sequence`` for logging.

Both the Adam and the sub‑gradient phases are time‑aware – they stop as soon
as the remaining budget is too small to guarantee another safe iteration.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------


def _fft_len_for_corr(n: int) -> int:
    """
    Smallest power‑of‑two ≥ 3·n‑2, sufficient for the linear convolution
    (length 2·n‑1) and the correlation (length 3·n‑2) used in the gradient.
    """
    return 1 << ((3 * n - 2).bit_length())


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max."""
    zmax = np.max(z)
    e = np.exp(z - zmax)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it representation that yields ``a`` after soft‑max."""
    return np.log(a + eps)


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Smoothed max‑convolution and its gradient w.r.t. ``a``.
    The smoothing constant is eps = eps_factor * max(conv).
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    a_fft = np.fft.rfft(a, L)

    # linear self‑convolution (length 2n‑1)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # smoothed soft‑max weights
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv          # for numerical stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or np.isnan(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # cross‑correlation a ⋆ w (linear, length 3n‑2)
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # gradient of the smoothed max (already multiplied by 4·n)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a


def _max_conv_and_argmax(a: np.ndarray):
    """Return (max_conv, argmax_index) for the linear convolution a*a."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_val = float(conv.max())
    arg = int(np.argmax(conv))
    return max_val, arg


def _project_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """
    Projet v onto the simplex {x ≥ 0, sum x = s}.
    Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond].max()
        theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Optimisation phases
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray, n: int, deadline: float, start: float):
    """
    Adam optimiser on the logits (soft‑max parametrisation) of the simplex.
    Returns the best vector found and its cheap score (2·n·max_conv)
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = 2.0 * n * _conv_max(a)

    # logits that generate a
    z = _logits_from_distribution(a)
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_alpha = 0.8
    eps_factor = 0.08
    eps_factor_min = 5e-4
    eps_factor_decay = 0.9996

    it = 0
    while time.time() < deadline:
        it += 1
        a = _softmax(z)

        max_conv, grad_a = _conv_max_and_grad(a, eps_factor)
        cheap_score = 2.0 * n * max_conv
        if cheap_score < best_score:
            best_score = cheap_score
            best_a = a.copy()

        # gradient on logits (mirror descent on simplex)
        mu = np.dot(grad_a, a)
        grad_z = a * (grad_a - mu)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step‑size schedule (inverse‑sqrt decay)
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step = -alpha_t * m_hat / (np.sqrt(v_hat) + eps_adam)
        z = z + step
        # keep the exponentials numerically safe
        z = np.clip(z, -50.0, 50.0)

        # gradually tighten the smoothing constant
        eps_factor = max(eps_factor_min, eps_factor * eps_factor_decay)

        # occasional tiny jitter to avoid stagnation
        if it % 4000 == 0 and np.random.rand() < 0.1:
            z += 0.003 * np.random.randn(n)

        # occasional progress report
        if it % 20000 == 0:
            print(f"[Adam] it={it}, best_score={best_score:.6f}")

        # stop when only a fraction of a second remains
        if time.time() > deadline - 0.5:
            break

    return best_a, best_score


def _subgradient_optimize(a0: np.ndarray, n: int, deadline: float, start: float):
    """
    Very cheap sub‑gradient descent on the *exact* max‑convolution.
    Each iteration needs one forward+inverse FFT only.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = 2.0 * n * _conv_max(a)

    it = 0
    step0 = 0.2
    while time.time() < deadline:
        it += 1
        max_conv, i_max = _max_conv_and_argmax(a)

        # sub‑gradient for the active index i_max
        g = np.zeros_like(a)
        if i_max < n:
            # range 0 … i_max is active
            g[: i_max + 1] = 2.0 * a[i_max :: -1]
        else:
            start_idx = i_max - n + 1
            idx = np.arange(start_idx, n)
            g[idx] = 2.0 * a[i_max - idx]

        # diminishing step size
        step = step0 / (math.sqrt(it) + 1.0)
        a_new = a - step * g
        a = _project_simplex(a_new)

        score = 2.0 * n * _conv_max(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            if it % 10000 == 0:
                print(f"[Subgrad] it={it}, new best_score={best_score:.6f}")

        # safety break
        if time.time() > deadline - 0.2:
            break

    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that minimises the evaluation metric.
    The routine combines a cheap beta‑distribution sweep, Adam optimisation on
    the simplex and a large number of cheap sub‑gradient steps.
    """
    np.random.seed(seed)
    start = time.time()
    # keep a small safety margin for the final evaluation call
    deadline = start + budget_s - 2.0

    # --------------------------------------------------------------
    # Determine the support size.
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
        print(f"Re‑using external sequence length: n = {n}")
    except Exception:
        n = 20000
        print(f"No external sequence – using default n = {n}")

    # --------------------------------------------------------------
    # 0. Base arcsine distribution (β(0.5,0.5)).
    # --------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)
    x = (idx + 0.5) / n                # mid‑point rule on (0,1)
    raw = 1.0 / np.sqrt(x * (1.0 - x))  # arcsine density (up to const.)
    a0 = raw / raw.sum()
    base_score = 2.0 * n * _conv_max(a0)
    print(f"Initial arcsine approximation: cheap J = {base_score:.6f}")

    # --------------------------------------------------------------
    # 1. Simple sweep over symmetric beta‑distributions and mixtures.
    # --------------------------------------------------------------
    best_a = a0.copy()
    best_score = base_score

    # uniform vector (purely for mixture tests)
    uniform = np.full(n, 1.0 / n)

    # mixture with uniform (λ up to 0.4)
    for lam in np.linspace(0.0, 0.4, 9):
        cand = (1.0 - lam) * a0 + lam * uniform
        cand /= cand.sum()
        score = 2.0 * n * _conv_max(cand)
        if score < best_score:
            best_score = score
            best_a = cand.copy()

    # symmetric beta(α,α) sweep (α ∈ [0.4, 1.5])
    alphas = np.linspace(0.4, 1.5, 12)
    for alpha in alphas:
        shape = (x ** (alpha - 1.0)) * ((1.0 - x) ** (alpha - 1.0))
        cand = shape / shape.sum()
        score = 2.0 * n * _conv_max(cand)
        if score < best_score:
            best_score = score
            best_a = cand.copy()

        # also try a small mixture of this shape with the arcsine base
        for lam in (0.1, 0.2):
            mix = (1.0 - lam) * a0 + lam * cand
            mix /= mix.sum()
            score = 2.0 * n * _conv_max(mix)
            if score < best_score:
                best_score = score
                best_a = mix.copy()

    print(f"After cheap parameter sweep: best cheap J = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2. Adam optimisation on the simplex (soft‑max parametrisation).
    # --------------------------------------------------------------
    if time.time() < deadline - 1.0:
        best_a, best_score = _adam_optimize(best_a, n, deadline, start)
        print(f"After Adam optimisation: best cheap J = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3. Large‑scale cheap sub‑gradient refinement.
    # --------------------------------------------------------------
    if time.time() < deadline - 1.0:
        best_a, best_score = _subgradient_optimize(best_a, n, deadline, start)
        print(f"After sub‑gradient refinement: best cheap J = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Return the best candidate (plain Python list) and log final score.
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:   # defensive – should never happen
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    return final_seq
