def propose_candidate(seed=42, budget_s=1000, **kwargs):

    np.random.seed(seed)

    start = time.time()

    deadline = start + budget_s - 0.5

    n = 4096

    # get initial candidates
    init_candidates = []

    # uniform
    init_candidates.append(np.full(n, 1.0 / n))

    # previous global if exists
    if 'height_sequence_1' in globals():
        prev = np.array(height_sequence_1, dtype=float)
        if prev.size > 0:
            # truncate or pad to length n
            if len(prev) < n:
                # pad zeros
                prev = np.concatenate([prev, np.zeros(n - len(prev))])
            elif len(prev) > n:
                prev = prev[:n]
            prev = np.clip(prev, 0, None)
            total = prev.sum()
            if total > 0:
                prev = prev / total
                init_candidates.append(prev)
    # random Dirichlet
    for _ in range(2):
        a = np.random.dirichlet(np.ones(n))
        init_candidates.append(a)

    # power law random
    for _ in range(2):
        alpha = np.random.uniform(0.3, 0.7)
        beta = np.random.uniform(0.0, 20.0)
        a = _shape_power(alpha, beta, n)
        init_candidates.append(a)

    # keep track best overall
    best_a = None
    best_score = float('inf')
    # evaluate initial candidates
    for a0 in init_candidates:
        score = _objective(a0)
        if score < best_score:
            best_score = score
            best_a = a0.copy()
    print(f"[init] best O={best_score:.6f}")

    # Phase 1: short mirror descent on each candidate (few iterations)
    short_iters = 5000
    for idx, a0 in enumerate(init_candidates):
        if time.time() >= deadline:
            break
        a, sc = _mirror_descent(a0, max_iters=short_iters, deadline=deadline)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[short] new best O={best_score:.6f} from candidate {idx}")

    # Phase 2: long polishing
    remaining = deadline - time.time()
    if remaining > 5.0:
        long_iters = int(min(300000, remaining * 200))  # adjust iterations
        print(f"[polish] starting with max_iters={long_iters}")
        a_opt, score_opt = _mirror_descent(best_a, max_iters=long_iters, deadline=deadline)
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[polish] improved O={best_score:.6f}")

    try:

def _mirror_descent(a0, max_iters, deadline):
    a = a0.copy()
    n = len(a)
    best_a = a.copy()
    best_score = _objective(a)
    # schedule parameters
    eta0 = 0.4
    gamma0 = 2.0
    gamma_end = 30.0
    stagnation = 0
    patience = max_iters // 20
    for t in range(max_iters):
        if time.time() >= deadline:
            break
        # compute schedules
        prog = t / max_iters
        eta = eta0 * (1.0 - prog)  # linear decay
        gamma = gamma0 + (gamma_end - gamma0) * prog

        # gradient
        grad = _gradient_softmax(a, gamma)
        # exponentiated update
        a = a * np.exp(-eta * grad)
        # normalize
        total = a.sum()
        if total <= 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= total

        cur_score = _objective(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            stagnation = 0
            # optionally print improvement
            if t % 1000 == 0:

"""
Propose a non‑negative step‑function (list of heights) that minimises

        O = 2·n·max(conv(a,a)) / (sum(a)²)

The optimiser works in three stages:

1. **Initialisation** – several simple candidates are built
   (uniform, previous best, Dirichlet draws, random power‑law shapes)
   and the best of them is kept.

2. **Mirror‑descent polishing** – an exponentiated‑gradient (mirror)
   descent is performed on the *soft‑max* surrogate of the max‑convolution.
   The surrogate weight γ is annealed from a small value (smooth) to a
   larger value (close to the true max).  A decreasing learning‑rate η
   together with occasional jitter keeps the iterates away from plateaus.
   The iteration stops when the time budget is exhausted.

3. **Fine‑grained random swaps** – after the continuous optimisation a
   cheap hill‑climbing loop repeatedly moves a tiny amount of mass from a
   random index to another one.  Improvements are kept, giving the algorithm
   a chance to escape the limited expressivity of the gradient step.

All steps respect the 1000 s budget, print occasional progress information
and finally return the best sequence found as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a power‑of‑two FFT (full length)."""
    n = a.size
    size = 1 << ((2 * n - 2).bit_length())        # next power of two ≥ 2n‑1
    A = np.fft.fft(a, n=size)
    conv = np.fft.ifft(A * A).real[: 2 * n - 1]
    return conv


def _conv_fft_two(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Convolution of two arbitrary 1‑D arrays using FFT (full length)."""
    n = len(x) + len(y) - 1
    size = 1 << ((n - 1).bit_length())
    X = np.fft.fft(x, n=size)
    Y = np.fft.fft(y, n=size)
    conv = np.fft.ifft(X * Y).real[:n]
    return conv


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective   O = 2·n·max(conv(a,a)) / (sum(a)²)."""
    n = a.size
    conv = _conv_fft(a)
    max_val = np.max(conv)
    s = np.sum(a)
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_val / (s * s))


# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, gamma: float) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate   fγ(a)=2·n·(1/γ)·log Σ exp(γ·conv)
    w.r.t. a.  The formula reduces to 2·(w * a)[:n] where
    w_k ∝ exp(γ·conv_k) and * denotes linear convolution.
    """
    n = a.size
    conv = _conv_fft(a)                     # length 2n‑1
    # soft‑max weights (numerically stable)
    cmax = np.max(conv)
    w = np.exp(gamma * (conv - cmax))
    w_sum = np.sum(w)
    if w_sum == 0:
        w = np.full_like(conv, 1.0 / conv.size)
    else:
        w /= w_sum

    # gradient = 2 * (w * a)[:n]
    conv_wa = _conv_fft_two(w, a)           # length 3n‑2
    grad = 2.0 * conv_wa[:n]
    return grad


# ----------------------------------------------------------------------
def _exp_update(a: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:
    """
    Exponentiated‑gradient (mirror) update on the simplex.
    Works in log‑space for numerical stability.
    """
    log_a = np.log(a + 1e-15)
    log_a -= eta * grad
    log_a -= np.max(log_a)                 # shift to keep numbers moderate
    a_new = np.exp(log_a)
    a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
def _mirror_descent(a0: np.ndarray,
                    max_iters: int,
                    deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the soft‑max surrogate.
    Returns the best vector seen and its exact objective value.
    """
    a = a0.copy()
    n = a.size
    best_a = a.copy()
    best_score = _objective(a)

    eta0 = 0.45          # initial learning rate
    gamma_start = 2.0    # start with a smooth surrogate
    gamma_end = 30.0     # finish close to the true max

    stagnation = 0
    patience = max(500, max_iters // 30)

    for t in range(max_iters):
        if time.time() >= deadline:
            break

        prog = t / max_iters
        eta = eta0 * (1.0 - prog)               # linear decay
        gamma = gamma_start + (gamma_end - gamma_start) * prog

        grad = _gradient_softmax(a, gamma)
        a = _exp_update(a, grad, eta)

        cur_score = _objective(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            stagnation = 0
            if t % 2000 == 0:
                print(f"[md] iter {t:6d}  new best O={best_score:.6f}")
        else:
            stagnation += 1
            if stagnation >= patience:
                # small jitter to leave flat regions
                jitter = np.random.randn(n) * 1e-5
                a = best_a + jitter
                a = np.clip(a, 0.0, None)
                a /= a.sum()
                stagnation = 0

    return best_a, best_score


# ----------------------------------------------------------------------
def _shape_power(alpha: float, beta: float, n: int) -> np.ndarray:
    """Monotone decreasing power‑law a_i ∝ (i+β)^(-α)  (i = 1..n)."""
    i = np.arange(1, n + 1, dtype=float)
    a = (i + beta) ** (-alpha)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
def _local_swap_search(a_start: np.ndarray,
                       cur_best: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: move a tiny fraction of mass from one index
    to another and keep the improvement.  The step size slowly shrinks.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = cur_best
    n = a.size

    # schedule for the amount that is moved (relative to the source entry)
    # start with 5 % and decay to 0.1 %
    init_frac = 0.05
    final_frac = 0.001
    total_swaps = 0

    while time.time() < deadline:
        # linearly decay the fraction depending on elapsed time
        elapsed = time.time() - (deadline - (deadline - time.time()))
        # To avoid dealing with absolute time we just decay every 1000 swaps
        frac = max(final_frac,
                   init_frac * (0.5 ** (total_swaps // 5000)))
        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        delta = a[i] * frac
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # maintain feasibility
        if a_new[i] < 0:
            continue
        score = _objective(a_new)
        total_swaps += 1
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            # occasional reporting
            if total_swaps % 5000 == 0:
                print(f"[swap] swaps {total_swaps:7d}  new best O={best_score:.6f}")

    return best_a, best_score
