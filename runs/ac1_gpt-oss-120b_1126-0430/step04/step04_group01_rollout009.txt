def propose_candidate(seed=0, budget_s=1000):
    import time, math, numpy as np
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.1

    # set n based on time
    n = 12000  # try bigger length
    a = np.full(n, 1.0/n)
    L = 1 << ((3*n - 2).bit_length())

    # Phase1: Adam on smoothed surrogate
    best_seq = a.copy()
    conv = _conv_fft(a, L)
    best_score = 2.0 * n * conv.max()
    print("[init] n=", n, "init O=", best_score)

    # hyperparameters
    beta1=0.9; beta2=0.999; eps=1e-12
    m = np.zeros_like(a); v = np.zeros_like(a)
    lr0 = 0.5

    # schedule
    phase1_deadline = start + budget_s * 0.55
    max_iter = 600_000   # high bound
    alpha_start = 5.0
    alpha_end = 20000.0
    lam_start = 0.05
    lam_end = 0.005

    it = 0
    while time.time() < phase1_deadline and it < max_iter:
        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[phase1] iter {it:,} new best O={best_score:.6f}")

        frac = it / max_iter
        alpha = alpha_start + (alpha_end - alpha_start) * frac
        lam = lam_start + (lam_end - lam_start) * (1 - frac)   # decreasing lam maybe

        grad = _smooth_gradient_plus_entropy(a, conv, alpha, L, lam)
        # Adam update
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * (grad*grad)
        m_hat = m / (1 - beta1**(it+1))
        v_hat = v / (1 - beta2**(it+1))
        lr = lr0 / math.sqrt(it + 1)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        # jitter
        if np.random.rand() < 0.001:
            a_new += np.random.randn(n) * 1e-3

        # projection onto simplex
        a_new = _project_to_simplex(a_new)

        # prune small values occasionally
        if it % 4000 == 0 and it > 0:
            cutoff = np.percentile(a_new, 0.1)   # remove bottom 0.1%
            a_new[a_new < cutoff] = 0.0
            a_new = _project_to_simplex(a_new)

        a = a_new
        it += 1

    # Phase2: Simulated annealing + subgradient
    print("[phase2] start O", best_score)
    max_restarts = 0
    subgrad_interval = 3
    temperature0 = 0.03
    max_iter_loc = 200_000
    no_improve_limit = 4000

    while time.time() < deadline:
        # restart
        if np.random.rand() < 0.7:
            a = best_seq.copy()
            a += np.random.randn(n) * 0.001
            a = np.clip(a, 0, None)
            a = _project_to_simplex(a)
        else:
            a = np.random.rand(n)
            a = _project_to_simplex(a)

        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        it_loc = 0
        no_improve = 0
        T = temperature0
        while time.time() < deadline and it_loc < max_iter_loc:
            # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j and a[i] > 1e-14:
                # move a random fraction
                delta = a[i] * np.random.rand() * 0.2
                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta
                a_new = _project_to_simplex(a_new)

                conv_new = _conv_fft(a_new, L)
                new_obj = 2.0 * n * conv_new.max()

                if new_obj < cur_obj or np.random.rand() < math.exp((cur_obj - new_obj) / max(T, 1e-12)):
                    a = a_new
                    conv = conv_new
                    cur_obj = new_obj
                    no_improve = 0
                    if cur_obj < best_score:
                        best_score = cur_obj
                        best_seq = a.copy()
                        print(f"[global] new best O={best_score:.6f} (restart {max_restarts}, iter {it_loc})")
                else:
                    no_improve += 1
            else:
                no_improve += 1

            # subgradient step
            if it_loc % subgrad_interval == 0:
                g = _subgradient_max_conv(a, conv)
                eta = 0.1 * (1 - it_loc / max_iter_loc)
                a = a - eta * g
                a = _project_to_simplex(a)
                conv = _conv_fft(a, L)
                cur_obj = 2.0 * n * conv.max()
                if cur_obj < best_score:
                    best_score = cur_obj
                    best_seq = a.copy()
                    print(f"[subgrad] improved O={best_score:.6f}")

            # occasional smoothing
            if it_loc % 5000 == 0 and it_loc > 0:
                kernel = np.array([0.0625, 0.1875, 0.5, 0.1875, 0.0625])
                a_s = np.convolve(a, kernel, mode='same')
                a = _project_to_simplex(0.85 * a + 0.15 * a_s)

            T *= 0.9999
            it_loc += 1

            if no_improve >= no_improve_limit:
                break

        max_restarts += 1
        if time.time() >= deadline:
            break

    # final evaluation
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print("[final] O=", final_score)
    except Exception as e:
        print("[final] evaluation error:", e)

    return best_seq.tolist()

"""
Propose a step‑function (non‑negative vector) that approximately minimises

    O(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm consists of three stages:

1️⃣ **Large‑scale smooth optimisation** – Adam on a
   smoothed log‑sum‑exp surrogate of the max‑convolution together
   with a tiny entropy term.  The smoothing parameter α is annealed
   from a modest start up to a very large value (≈2·10⁴) so that the
   surrogate slowly turns into the true max‑operator.

2️⃣ **Stochastic local refinement** – Simulated‑annealing style mass‑transfer
   moves combined with periodic projected sub‑gradient steps on the exact
   objective (the max‑convolution entry).  A simple geometric temperature
   schedule encourages exploration early and convergence later.

3️⃣ **Final greedy polishing & trimming** – After the time budget is almost
   exhausted we perform a short greedy hill‑climb (only accept improving
   moves) and finally strip leading/trailing zero entries.  Trimming reduces
   the length *n* and therefore the factor *2·n* in the objective without
   changing the convolution values, giving a final boost.

All operations are performed on the probability simplex (∑a_i = 1) – the
objective is invariant under scaling, so this constraint is harmless.
Fast FFT‑based convolutions (full length 2·n‑1) keep each iteration cheap
even for n≈12 000.  The routine respects the 1000 s budget and returns the
best sequence found as a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full convolution a * a (real FFT, length 2·n‑1)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return the full a⨂a convolution (length 2·n‑1) using FFT length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) w.r.t. a (averaged over all maximisers)
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
# Gradient of the smooth surrogate (log‑sum‑exp + tiny entropy)
def _smooth_gradient_plus_entropy(
    a: np.ndarray,
    conv: np.ndarray,
    alpha: float,
    L: int,
    lam: float,
) -> np.ndarray:
    """
    Gradient of  (2·n/α)·log∑_k e^{α·conv_k}  +  λ·∑ a_i·log a_i .
    First term reduces to 2·corr(w,a) with w_k ∝ e^{α·conv_k}.
    """
    # soft‑max weights over the convolution entries
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)          # stability
    w = exp_scaled / exp_scaled.sum()                 # shape (2n‑1,)

    # correlation w ⋆ a (via FFT)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n_len = a.size
    grad_corr = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]

    # entropy part (log barrier)
    eps = 1e-20
    grad_ent = lam * (np.log(a + eps) + 1.0)

    return grad_corr + grad_ent


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Search for a low‑value step‑function within the given time budget.

    Returns a plain Python list of non‑negative heights.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 0.2

    # ------------------------------------------------------------------
    # Problem size – we pick a fairly large n (≈12k) to exploit the 2·n factor.
    n = 12000
    a = np.full(n, 1.0 / n)          # start from the uniform distribution
    L = 1 << ((3 * n - 2).bit_length())   # FFT length ≥ 3n‑2 (for correlation)

    # ------------------------------------------------------------------
    # Phase‑1 : Adam on the smoothed surrogate (log‑sum‑exp + entropy)
    best_seq = a.copy()
    conv = _conv_fft(a, L)
    best_score = 2.0 * n * conv.max()
    print(f"[init] n={n}, O(start)={best_score:.6f}")

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    lr0 = 0.6                     # base learning rate
    alpha_start = 5.0
    alpha_end = 20000.0
    lam_start = 0.04
    lam_end = 0.001

    max_iter_phase1 = 500_000
    phase1_deadline = start_time + budget_s * 0.55   # ~55 % of total budget
    it = 0
    while time.time() < phase1_deadline and it < max_iter_phase1:
        # exact objective for bookkeeping
        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[phase‑1] iter {it:,} new best O={best_score:.6f}")

        # annealing schedules
        frac = it / max_iter_phase1
        alpha = alpha_start + (alpha_end - alpha_start) * frac
        lam = lam_start * (1 - frac) + lam_end * frac

        # smoothed gradient
        grad = _smooth_gradient_plus_entropy(a, conv, alpha, L, lam)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        lr = lr0 / math.sqrt(it + 1)   # simple 1/√t decay
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional tiny jitter to escape flat regions
        if np.random.rand() < 0.001:
            a_new += np.random.randn(n) * 1e-4

        # projection onto the simplex
        a_new = _project_to_simplex(a_new)

        # prune extremely tiny entries every few thousand steps
        if it % 5000 == 0 and it > 0:
            thresh = np.percentile(a_new, 0.2)   # drop the lowest 0.2 %
            a_new[a_new < thresh] = 0.0
            a_new = _project_to_simplex(a_new)

        a = a_new
        it += 1

    # ------------------------------------------------------------------
    # Phase‑2 : Stochastic local refinement (mass‑transfer + sub‑gradient)
    print(f"[phase‑2] entering local refinement, best O={best_score:.6f}")

    max_restarts = 0
    subgrad_interval = 4
    temperature0 = 0.04
    max_iter_local = 180_000
    no_improve_limit = 3500

    while time.time() < deadline:
        # initialise a restart – jittered best or fresh random start
        if np.random.rand() < 0.6:
            a = best_seq.copy()
            a += np.random.randn(n) * 0.0015
            a = np.clip(a, 0, None)
            a = _project_to_simplex(a)
        else:
            a = np.random.rand(n)
            a = _project_to_simplex(a)

        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        it_loc = 0
        no_improve = 0
        T = temperature0

        while time.time() < deadline and it_loc < max_iter_local:
            # ---------------- random mass‑transfer move -----------------
            i, j = np.random.randint(n, size=2)
            if i != j and a[i] > 1e-14:
                delta = a[i] * np.random.rand() * 0.18   # up to 18 % of source mass
                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta
                a_new = _project_to_simplex(a_new)

                conv_new = _conv_fft(a_new, L)
                new_obj = 2.0 * n * conv_new.max()

                # Metropolis acceptance
                if new_obj < cur_obj or np.random.rand() < math.exp((cur_obj - new_obj) / max(T, 1e-14)):
                    a, conv, cur_obj = a_new, conv_new, new_obj
                    no_improve = 0
                    if cur_obj < best_score:
                        best_score = cur_obj
                        best_seq = a.copy()
                        print(f"[{time.time():.1f}] new global best O={best_score:.6f} "
                              f"(restart={max_restarts}, iter={it_loc})")
                else:
                    no_improve += 1
            else:
                no_improve += 1

            # ---------------- sub‑gradient step -----------------
            if it_loc % subgrad_interval == 0:
                g = _subgradient_max_conv(a, conv)
                eta = 0.09 * (1.0 - it_loc / max_iter_local)   # linearly decaying
                a = a - eta * g
                a = _project_to_simplex(a)
                conv = _conv_fft(a, L)
                cur_obj = 2.0 * n * conv.max()
                if cur_obj < best_score:
                    best_score = cur_obj
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] sub‑grad improved O={best_score:.6f}")

            # ---------------- occasional smoothing -----------------
            if it_loc % 4000 == 0 and it_loc > 0:
                kernel = np.array([0.0625, 0.1875, 0.5, 0.1875, 0.0625])
                a_smooth = np.convolve(a, kernel, mode="same")
                a = _project_to_simplex(0.85 * a + 0.15 * a_smooth)

            # temperature decay
            T *= 0.9998
            it_loc += 1

            if no_improve >= no_improve_limit:
                break

        max_restarts += 1
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Final greedy polishing (pure improvement moves) while time permits
    if time.time() < deadline - 0.6:
        a = best_seq.copy()
        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        while time.time() < deadline - 0.5:
            i, j = np.random.randint(n, size=2)
            if i == j or a[i] < 1e-14:
                continue
            delta = a[i] * np.random.rand() * 0.10   # up to 10 %
            a_try = a.copy()
            a_try[i] -= delta
            a_try[j] += delta
            a_try = _project_to_simplex(a_try)

            conv_try = _conv_fft(a_try, L)
            obj_try = 2.0 * n * conv_try.max()
            if obj_try < cur_obj:
                a, conv, cur_obj = a_try, conv_try, obj_try
                if cur_obj < best_score:
                    best_score = cur_obj
                    best_seq = a.copy()
                    print(f"[final‑hill] improved O={best_score:.6f}")

    # ------------------------------------------------------------------
    # Trim leading / trailing near‑zero entries – this reduces n without
    # affecting the convolution values (they are shift‑invariant).
    eps_trim = 1e-12
    nz = np.where(best_seq > eps_trim)[0]
    if nz.size:
        trimmed = best_seq[nz[0] : nz[-1] + 1]
    else:
        trimmed = best_seq
    # Ensure we still satisfy the simplex (numerical drift)
    trimmed = _project_to_simplex(trimmed)

    # ------------------------------------------------------------------
    # Final sanity check via the provided evaluator
    try:
        final_score = evaluate_sequence(trimmed.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:
        print("[final] evaluate_sequence failed:", e)

    # Return the best (trimmed) sequence as a plain list
    return trimmed.tolist()
