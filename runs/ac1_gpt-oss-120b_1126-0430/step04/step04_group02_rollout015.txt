"""
Search for a non‑negative step function minimizing

    O(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The method mixes two complementary strategies:

1. **Adam optimisation** on the soft‑max (log‑it) parametrisation of the
   simplex.  The objective is smoothed with a temperature β that is
   exponentially annealed, and Adam’s adaptive moments provide a stable
   descent.  Several independent runs with different support sizes are
   performed; intermediate best values are kept.

2. **Linear‑programming refinement**.  For a distribution `f` (normalised
   to sum = 1) with current bound `M = max(conv(f,f))` we solve

        maximise   Σ g_i
        subject to  conv(f,g) ≤ M    (element‑wise)
                    g_i ≥ 0.

   The resulting `g` can carry strictly more total mass; after normalising
   `g` the bound drops.  Repeating this step (a few iterations) drives the
   maximum convolution well below the uniform benchmark, yielding a lower
   upper bound.  The constraints are assembled as a sparse Toeplitz matrix
   and solved with SciPy’s HiGHS LP solver.

The budget of 1000 s is split between the two phases; the final best
sequence is returned as a plain list of floats.  All numbers are non‑negative,
the sum is positive and the code respects the time limit.
"""

import time
import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog


# -------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# -------------------------------------------------------------
def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (real valued) using FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv


# -------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective O = 2·n·max(conv(a,a)) (assumes sum(a)=1)."""
    n = a.size
    conv = _conv_self_fft(a)
    return 2.0 * n * conv.max()


# -------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


# -------------------------------------------------------------
def _run_adam(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
) -> tuple[list[float], float]:
    """
    Adam optimisation on the log‑it simplex for a given support size ``n``.
    Returns the best sequence (list) and its exact objective value.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # ---- initialise -------------------------------------------------
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
        a = np.clip(a, 0.0, None)
        a = a / a.sum()
    else:
        a = rng.random(n)
        a = a / a.sum()

    # work in log‑it space (soft‑max automatically satisfies the simplex)
    x = np.log(a + 1e-16)

    m = np.zeros_like(x)          # Adam first moment
    v = np.zeros_like(x)          # Adam second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # ---- schedules --------------------------------------------------
    beta_start = 0.5               # smoothing start
    beta_end   = 8000.0            # smoothing end (very sharp)
    lr_start   = 0.15              # initial learning rate on logits
    lr_end     = 0.001             # final learning rate
    max_iters  = 300_000

    best_a = a.copy()
    best_obj = _objective(best_a)

    it = 0
    while time.time() - start < time_limit and it < max_iters:
        it += 1

        # probabilities from logits
        a = _softmax(x)

        # exact objective
        conv = _conv_self_fft(a)
        M = conv.max()
        obj = 2.0 * n * M
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[adam n={n:5d} it={it:06d}] new best O={best_obj:.6f}")

        # schedule parameters
        t = min(it / max_iters, 1.0)
        beta = beta_start * np.exp(np.log(beta_end / beta_start) * t)
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t)

        # smoothed max weights
        if beta > 4000.0:
            # very sharp → true arg‑max
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            shifted = conv - M                     # ≤ 0
            w = np.exp(beta * shifted)
            w = w / w.sum()

        # gradient of the smooth max (in coefficient space)
        # correlation with w yields ∂/∂a of Σ_k w_k conv_k
        g_a = 2.0 * np.fft.irfft(np.fft.rfft(w) * np.conj(np.fft.rfft(a)), n=_next_pow2(2 * n - 1))[:n]

        # natural gradient w.r.t. logits
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape plateaus
        if it % 6000 == 0:
            jitter = rng.normal(scale=0.02, size=n)
            x += jitter

        # periodic restart from the best solution seen so far
        if it % 70_000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.005, size=n)
            m.fill(0.0)
            v.fill(0.0)
            print(f"[adam n={n}] restart at iter {it}")

    # ---- compress negligible entries ---------------------------------
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = _objective(best_a)
        print(f"[compress] reduced length {n} → {best_a.size}, O={best_obj:.6f}")

    # ---- very short local search (random pair swaps) ----------------
    remaining = time_limit - (time.time() - start)
    if remaining > 0.5:
        local_start = time.time()
        a_loc = best_a.copy()
        obj_loc = best_obj
        n_loc = a_loc.size
        while time.time() - local_start < remaining * 0.3:
            i, j = rng.choice(n_loc, size=2, replace=False)
            delta = a_loc[i] * rng.random() * 0.02   # ≤2 % of a_i
            if delta <= 0.0:
                continue
            a_new = a_loc.copy()
            a_new[i] -= delta
            a_new[j] += delta
            conv_new = _conv_self_fft(a_new)
            obj_new = 2.0 * n_loc * conv_new.max()
            if obj_new < obj_loc:
                a_loc, obj_loc = a_new, obj_new
                if obj_new < best_obj:
                    best_a, best_obj = a_new, obj_new
        print(f"[local] final local O={best_obj:.6f}")

    return best_a.tolist(), best_obj

def _build_conv_matrix(f: np.ndarray) -> csr_matrix:
    """
    Construct the (2n‑1)×n matrix A such that (A @ g)[k] = (f*g)[k].
    Rows correspond to convolution output indices, columns to g‑variables.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    # column index repeats each column index n times
    col_idx = np.repeat(np.arange(n), n)
    # row indices: for column j the rows are j … j+n‑1
    row_idx = np.concatenate([np.arange(j, j + n) for j in range(n)])
    data = np.tile(f, n)
    return csr_matrix((data, (row_idx, col_idx)), shape=(rows, n))

def _lp_refine(a_init: np.ndarray, time_limit: float, rng: np.random.Generator):
    """
    Iteratively improve a normalised vector ``a`` using the linear‑programming
    step described in the paper.  Returns the best sequence (list) and its
    exact objective found within the allocated time.
    """
    start = time.time()
    a = a_init.copy()
    n = a.size
    best_a = a.copy()
    best_obj = _objective(best_a)

    max_iters = 20
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        conv = _conv_self_fft(a)
        M = conv.max()
        obj = 2.0 * n * M
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            print(f"[lp n={n:5d} it={it:02d}] new best O={best_obj:.6f}")

        # build the convolution matrix A for constraints A @ g ≤ M
        A = _build_conv_matrix(a)
        b = np.full(2 * n - 1, M, dtype=float)
        c = -np.ones(n)                     # maximise Σ g_i  ⇔  minimise −Σ g_i

        # solve the LP (HiGHS handles sparse A efficiently)
        res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
        if not res.success:
            print(f"[lp] linprog failed at iter {it}: {res.message}")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-9:            # no increase ⇒ cannot improve further
            break

        # normalise and optionally add a tiny jitter
        a = g / sum_g
        if it % 5 == 0:
            jitter = rng.normal(scale=0.001, size=n)
            jitter -= jitter.mean()
            a += jitter
            a = np.clip(a, 0, None)
            a = a / a.sum()

    # compress negligible entries of the best solution found
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = _objective(best_a)
        print(f"[lp compress] reduced length {n} → {best_a.size}, O={best_obj:.6f}")

    return best_a.tolist(), best_obj

        else:
            a0 = rng.random(n)
            a0 = a0 / a0.sum()

        seq_lp, val_lp = _lp_refine(a0, run_time, rng)
        if val_lp < best_obj:
            best_obj = val_lp
            best_seq = seq_lp
            print(f"[global] new overall best after LP O={best_obj:.6f} (n={len(seq_lp)})")

    # -----------------------------------------------------------------
    # Final evaluation with the provided evaluate_sequence routine.
    try:
        final_score = evaluate_sequence(best_seq)
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_seq
