"""
Propose a candidate step‑function that minimises

    O = 2·n·max(conv(a,a)) / (sum(a)**2)

using an Adam optimiser on the log‑probabilities (exponentiated‑gradient)
with a smoothly increasing soft‑max temperature β that eventually turns into a
hard‑max (active‑set) weight.  The algorithm runs for up to the given time
budget, restarts from the current best solution when progress stalls and
occasionally adds jitter to escape plateaus.  It works on a single
probability vector (the sum of the coefficients is irrelevant for the bound)
and returns the best sequence found as a plain ``list`` of ``float``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper FFT tools (top‑level – no closures)
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a real FFT (real input)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv


def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Linear correlation g_i = Σ_t w[i+t]·a[t] for i = 0 … n‑1.
    Implemented via FFT.  ``w`` has length 2·n‑1.
    """
    n = a.shape[0]
    m = w.shape[0]                # = 2·n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list:
    """
    Search for a non‑negative step‑function that minimises the bound
    returned by ``evaluate_sequence``.  The function respects a time budget
    (default 1000 s) and returns the best coefficient list found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1) Choose length and initialise log‑probabilities.
    # Use a moderately large length; larger n gives more freedom.
    n = 8000

    if 'height_sequence_1' in globals():
        # Use the known solution as a warm‑start if it has the same length.
        hs = np.asarray(height_sequence_1, dtype=float)
        if hs.size == n:
            a0 = hs / hs.sum()
        else:
            # Otherwise initialise from a Dirichlet‑like random vector.
            a0 = np.random.rand(n)
            a0 = a0 / a0.sum()
    else:
        a0 = np.random.rand(n)
        a0 = a0 / a0.sum()

    # Log‑variables for exponentiated‑gradient / Adam.
    u = np.log(a0 + 1e-12)

    # Adam moments.
    m = np.zeros_like(u)
    v = np.zeros_like(u)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # Schedules (tuned for the 1000 s budget).
    max_iters_est = 600_000            # generous upper bound
    beta_min = 2.0
    beta_max = 50000.0                # final soft‑max temperature
    eta_start = 0.6
    eta_end   = 0.02                  # final learning‑rate in log‑space

    # Restart / jitter parameters.
    restart_interval = 90_000          # iterations without improvement -> restart
    jitter_interval  = 5_000
    jitter_scale     = 5e-4

    best_u = u.copy()
    best_score = np.inf
    best_seq = None
    last_improve_iter = 0

    # ------------------------------------------------------------------
    # 2) Main optimisation loop.
    it = 0
    while time.time() < deadline and it < max_iters_est:
        it += 1

        # ---- (a) Normalise to a probability vector.
        a_raw = np.exp(u - u.max())          # stable exponentiation
        a = a_raw / a_raw.sum()               # Σ a = 1

        # ---- (b) Convolution and current bound.
        conv = _conv_self_fft(a)              # length 2n‑1
        M = conv.max()
        cur_score = 2.0 * n * M               # because Σ a = 1

        # ---- (c) Record best solution.
        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_u = u.copy()
            best_seq = a.copy()
            last_improve_iter = it
            print(f"[iter {it:07d}] new best O={best_score:.6f}")

        # ---- (d) Annealing schedules.
        frac = min(it / max_iters_est, 1.0)
        # soft‑max temperature β (exponential schedule)
        beta = beta_min * (beta_max / beta_min) ** frac
        # learning‑rate η (exponential decay)
        eta = eta_start * (eta_end / eta_start) ** frac

        # ---- (e) Build weight vector over convolution entries.
        # Early phase: smooth soft‑max, later phase: hard‑max (active set).
        if frac < 0.75:          # soft‑max phase
            shifted = conv - M               # ≤ 0, maximum at 0
            w = np.exp(beta * shifted)
            w_sum = w.sum()
            if w_sum == 0 or np.isnan(w_sum):
                # fallback: pure hard‑max
                w = np.zeros_like(conv)
                w[conv.argmax()] = 1.0
            else:
                w = w / w_sum
        else:                    # active‑set (hard‑max) phase
            tol = max(1e-12, M * 1e-9)
            active = np.where(conv >= M - tol)[0]
            if active.size == 0:
                w = np.zeros_like(conv)
                w[conv.argmax()] = 1.0
            else:
                w = np.zeros_like(conv)
                w[active] = 1.0 / active.size

        # ---- (f) Gradient of the (smoothed) max‑convolution.
        # g = 2 * correlation(w, a)   (shape = n)
        g = 2.0 * _correlation_fft(w, a)

        # ---- (g) Adam update on the log‑variables.
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        u = u - eta * m_hat / (np.sqrt(v_hat) + eps)

        # ---- (h) Jitter to escape flat plateaus.
        if it % jitter_interval == 0:
            u += np.random.randn(n) * jitter_scale

        # ---- (i) Restart if no improvement for a long stretch.
        if it - last_improve_iter > restart_interval:
            # Re‑initialise from the current best with a tiny perturbation.
            u = best_u.copy()
            u += np.random.randn(n) * (jitter_scale * 0.5)
            print(f"[restart] iter {it}, re‑initialised from best O={best_score:.6f}")
            last_improve_iter = it  # reset counter

        # Periodic progress report (does not affect performance much).
        if it % 100_000 == 0:
            elapsed = time.time() - (deadline - budget_s + 0.5)
            print(f"[iter {it:07d}] O={cur_score:.6f}, best={best_score:.6f}, "
                  f"time={elapsed:.1f}s")

    # ------------------------------------------------------------------
    # 3) Final evaluation with the provided evaluator.
    if best_seq is None:
        # Fallback – should never happen, but keep the API contract.
        best_seq = np.exp(best_u - best_u.max())
        best_seq = best_seq / best_seq.sum()

    final_list = best_seq.tolist()
    try:
        final_score = evaluate_sequence(final_list)
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:      # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return final_list
