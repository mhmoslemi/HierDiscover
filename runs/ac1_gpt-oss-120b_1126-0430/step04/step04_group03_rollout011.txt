def _optimize_softmax(a0, max_iters, eta0, lambda0, deadline):
    a = a0.copy()
    best = a.copy()
    best_score = _objective_norm1(a)
    for t in range(max_iters):
        if time.time()>=deadline: break
        conv = _conv_fft(a)
        max_val = conv.max()
        w = np.exp(lambda0*(conv - max_val))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(w)/w.size
        else:
            w /= w_sum
        # compute gradient via convolution
        grad_full = np.convolve(a[::-1], w, mode='full')
        grad = 2 * grad_full[len(a)-1:len(a)-1+len(a)]
        # learning rate schedule: decaying
        eta = eta0 * (1 - t/max_iters)
        # exponentiated update
        log_a = np.log(a + 1e-15) - eta * grad
        # shift to avoid overflow
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()
        cur_score = _objective_norm1(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            # optional: print progress
    return best, best_score

def _exp_update(a,g,eta):
    log_a = np.log(a + 1e-15) - eta * g
    # offset to keep numerics stable
    log_a -= np.max(log_a)
    a_new = np.exp(log_a)
    a_new = np.clip(a_new, 0, None)
    s = a_new.sum()
    if s == 0:
        a_new = np.full_like(a, 1.0 / a.shape[0])
    else:
        a_new /= s

"""
Propose a non‑negative step‑function (list of heights) that minimises

    O = 2·n·max(conv(a,a)) / (sum(a)²)

The implementation combines three ideas:

1. **Coarse parametric sweep** – power‑law shapes (a_i ∝ (i+β)^‑α)
   on a modest grid give a good starting point.

2. **Soft‑max mirror descent** – instead of the non‑smooth sub‑gradient
   of the max‑convolution we minimise a smooth surrogate
   max_λ(x)= (1/λ)·log∑exp(λ·x).  A weighted‑sum gradient is obtained
   efficiently via FFTs, and an exponentiated‑gradient (EG) update
   keeps the iterate on the simplex.

   The temperature λ is increased from a low value (≈5) to a higher one
   (≈30) so that early iterations move quickly while later iterations
   focus on the true peak.

3. **Fine polishing** – a short run of the original sub‑gradient EG
   (as in the reference implementation) removes any remaining
   artefacts.

The routine respects the 1000 s budget, prints progress, and finally
returns the best sequence it discovered as a plain ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, Σx=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a power‑of‑two FFT."""
    n = a.size
    size = 1 << ((2 * n - 2).bit_length())
    A = np.fft.fft(a, n=size)
    conv = np.fft.ifft(A * A).real[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def _conv_fft_full(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """Full linear convolution of two vectors using FFT."""
    size = 1 << ((len(u) + len(v) - 2).bit_length())
    U = np.fft.fft(u, n=size)
    V = np.fft.fft(v, n=size)
    conv = np.fft.ifft(U * V).real[: len(u) + len(v) - 1]
    return conv


# ----------------------------------------------------------------------
def _objective_norm1(a: np.ndarray) -> float:
    """Objective 2·n·max(conv) assuming Σa = 1."""
    n = a.size
    conv = _conv_fft_self(a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
def _grad_softmax(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate of max(conv(a,a)).
    Returns ∂/∂a (∑ w_k conv_k) where w_k ∝ exp(lam·(conv_k−max)).
    """
    n = a.size
    conv = _conv_fft_self(a)
    max_val = conv.max()
    # soft‑max weights (already ≤1)
    w = np.exp(lam * (conv - max_val))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    # gradient_i = 2·∑_j a_j·w_{i+j}
    # this is a cross‑correlation of w with a.
    grad_full = _conv_fft_full(a[::-1], w)          # length = 3n‑2
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]       # slice the relevant n entries
    return grad


# ----------------------------------------------------------------------
def _exp_update(a: np.ndarray, g: np.ndarray, eta: float) -> np.ndarray:
    """
    Exponentiated‑gradient (EG) update on the simplex.
    Works in log‑space for numerical stability.
    """
    log_a = np.log(a + 1e-15) - eta * g
    log_a -= np.max(log_a)               # shift to keep numbers moderate
    a_new = np.exp(log_a)
    s = a_new.sum()
    if s == 0.0:
        a_new = np.full_like(a, 1.0 / a.size)
    else:
        a_new /= s
    return a_new


# ----------------------------------------------------------------------
def _optimize_softmax(a0: np.ndarray,
                      max_iters: int,
                      eta0: float,
                      lam_start: float,
                      lam_end: float,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent optimisation using the soft‑max surrogate of the max‑conv.
    Returns the best vector seen and its objective value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective_norm1(a)

    n = a.size
    no_improve = 0
    patience = max(1, max_iters // 30)

    for t in range(max_iters):
        if time.time() >= deadline:
            break

        # linearly increase λ
        lam = lam_start + (lam_end - lam_start) * (t / max_iters)

        g = _grad_softmax(a, lam)
        eta = eta0 * (1.0 - t / max_iters)

        a = _exp_update(a, g, eta)

        cur_score = _objective_norm1(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            print(f"[soft] iter {t:6d}  new best O={best_score:.6f}")
        else:
            no_improve += 1
            if no_improve >= patience:
                # small jitter to escape plateaus
                jitter = np.random.randn(n) * 1e-4
                a = best_a + jitter
                a = np.clip(a, 1e-15, None)
                a /= a.sum()
                no_improve = 0

    return best_a, best_score


# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv(a,a)) w.r.t. a (on the simplex)."""
    n = a.size
    conv = _conv_fft_self(a)
    max_val = conv.max()
    ks = np.where(np.abs(conv - max_val) < 1e-14)[0]
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
def _exp_update_subgrad(a: np.ndarray, g: np.ndarray, eta: float) -> np.ndarray:
    """EG update using a sub‑gradient (identical to _exp_update)."""
    return _exp_update(a, g, eta)

def _optimize_subgrad(a0: np.ndarray,
                     max_iters: int,

    """

    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective_norm1(a)

# ----------------------------------------------------------------------
def _shape_power(alpha: float, beta: float, n: int) -> np.ndarray:
    """
    Monotone decreasing shape a_i ∝ (i+β)^(-α), i=1..n, normalised to sum=1.
    """
    i = np.arange(1, n + 1, dtype=float)
    a = (i + beta) ** (-alpha)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
def _shape_power_exp(alpha: float, beta: float, gamma: float, n: int) -> np.ndarray:
    """
    Power‑law with exponential tail:
        a_i ∝ (i+β)^(-α) * exp(-γ·(i‑1)/n)
    """
    i = np.arange(1, n + 1, dtype=float)
    a = (i + beta) ** (-alpha) * np.exp(-gamma * (i - 1) / n)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a

            score = _objective_norm1(seq)
            if score < local_best_score - 1e-12:
                local_best_score = score
                local_best_a = seq
                local_best_alpha, local_best_beta, local_best_gamma = a, b, g
                print(f"[rand] n={n}  O={local_best_score:.6f} (α={a:.4f}, β={b:.4f}, γ={g:.4f})")

        # ------------------- 3️⃣  include any previously known sequence ---
        if 'height_sequence_1' in globals():
            try:
                prev = np.asarray(height_sequence_1, dtype=float)
                if prev.size > 0:
                    prev = np.maximum(prev, 0.0)
                    prev = prev[:n]                     # truncate / pad if needed
                    if prev.size < n:
                        prev = np.pad(prev, (0, n - prev.size), constant_values=0.0)
                    prev /= prev.sum()
                    prev_score = _objective_norm1(prev)
                    if prev_score < local_best_score:
                        local_best_score = prev_score
                        local_best_a = prev.copy()
                        print(f"[prev] n={n}  using previous construction O={prev_score:.6f}")
            except Exception:  # pragma: no cover
                pass

        # ------------------- 4️⃣  soft‑max mirror descent ----------------
        remaining = deadline - time.time()
        if remaining > 5.0:
            # Empirical guideline: ≈1500–2000 iterations per second for n≈4096.
            it_per_sec = 1500
            max_iters = int(min(180_000, remaining * it_per_sec))
            print(f"[soft] n={n}  start soft‑max EG, max_iters={max_iters}")
            a_opt, score_opt = _optimize_softmax(
                local_best_a,
                max_iters,
                eta0=0.6,
                lam_start=5.0,
                lam_end=30.0,
                deadline=deadline,
            )
            if score_opt < local_best_score:
                local_best_score = score_opt
                local_best_a = a_opt
                print(f"[soft] n={n}  improved to O={local_best_score:.6f}")

        # ------------------- 5️⃣  optional sub‑gradient polishing ---------
        remaining = deadline - time.time()
        if remaining > 5.0:
            max_iters = int(min(150_000, remaining * 150))
            print(f"[sub ] n={n}  polishing with sub‑gradient EG, max_iters={max_iters}")
            a_opt2, score_opt2 = _optimize_subgrad(
                local_best_a,
                max_iters,
                eta0=0.5,
                deadline=deadline,
            )
            if score_opt2 < local_best_score:
                local_best_score = score_opt2
                local_best_a = a_opt2
                print(f"[sub ] n={n}  polished best O={local_best_score:.6f}")

        # ------------------- 6️⃣  update global best --------------------
        if local_best_score < best_score:
            best_score = local_best_score
            best_seq = local_best_a.copy()
            print(f"[global] new overall best O={best_score:.6f} (n={n})")

    # ------------------------------------------------------------------
    #  If, for any reason, we did not manage to produce a sequence,
    #  fall back to a simple uniform distribution (O=2).
    if best_seq is None:
        n = 1024
        best_seq = np.full(n, 1.0 / n)

    # ------------------- final exact evaluation -----------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:  # pragma: no cover
        print("[final] evaluate_sequence call failed:", e)
        final_score = None

    return best_seq.tolist()
