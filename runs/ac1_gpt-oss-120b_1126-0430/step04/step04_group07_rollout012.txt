"""
Propose a non‑negative step sequence that minimises the bound

    O = 2·n·max(conv(a,a)) / (sum(a)**2)

using a mirrored Adam optimiser on the log‑probabilities (the
exponentiated‑gradient update).  The algorithm

* works on the probability simplex via log‑variables `u = log(a)`;
* evaluates the full self‑convolution with a real FFT;
* builds a smooth weighting of the convolution entries (soft‑max early,
  hard‑max later) to obtain a sub‑gradient of the (smoothed) max‑operator;
* updates `u` with Adam (first/second moment estimates) and a decaying
  stepsize;
* injects occasional Gaussian jitter and restarts from the best‑so‑far
  solution to avoid stagnation;
* optionally up‑samples a previously found solution (`height_sequence_1`);
* respects the 1000 s budget and returns the best list of coefficients.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv


def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Linear correlation g_i = Σ_t w[i+t]·a[t]  for i = 0,…,n‑1.
    Implemented via FFT.
    """
    n = a.shape[0]
    m = w.shape[0]                 # = 2·n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` bound.  Returns the best sequence as a plain
    ``list`` of ``float``.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # 1) Length of the sequence & initial log‑probabilities
    # --------------------------------------------------------------
    target_len = 12000                   # a larger length gives more flexibility
    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=float)
        # Upsample / downsample to the target length
        if a0.size != target_len:
            x_old = np.linspace(0.0, 1.0, a0.size)
            x_new = np.linspace(0.0, 1.0, target_len)
            a0 = np.interp(x_new, x_old, a0)
        a0 = a0 / a0.sum()
        eps = 1e-12
        u = np.log(a0 + eps)               # log‑probability vector
    else:
        # Uniform start (log(1/target_len) for each entry)
        u = np.full(target_len, -np.log(target_len), dtype=float)

    n = target_len
    best_u = u.copy()
    best_score = np.inf
    best_iter = 0

    # --------------------------------------------------------------
    # 2) Adam hyper‑parameters and schedules
    # --------------------------------------------------------------
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    m = np.zeros(n, dtype=float)          # first moment
    v = np.zeros(n, dtype=float)          # second moment

    # learning‑rate schedule (exponential decay)
    eta_start = 0.6
    eta_end = 0.04

    # soft‑max / hard‑max schedule
    beta_start = 2.0
    beta_end = 300_000.0

    max_iters_est = 350_000                # used only for schedule scaling
    jitter_interval = 4000
    jitter_scale = 5e-4
    restart_interval = 25_000
    stagnation_limit = 8000                # restart if no improvement for this many iters

    it = 0
    it_since_last_best = 0

    while time.time() < deadline:
        it += 1
        it_since_last_best += 1

        # ----------------------------------------------------------
        # (a) Build probability vector a from log‑variables
        # ----------------------------------------------------------
        # log‑softmax (stable)
        u_shift = u - np.max(u)
        a = np.exp(u_shift)
        a_sum = a.sum()
        if a_sum == 0.0:                # numerical guard
            a = np.full_like(a, 1.0 / n)
            a_sum = 1.0
        a = a / a_sum                     # now Σ a_i = 1

        # ----------------------------------------------------------
        # (b) Convolution and current bound
        # ----------------------------------------------------------
        conv = _conv_self_fft(a)          # length 2·n‑1
        M = conv.max()
        cur_score = 2.0 * n * M          # because Σ a_i = 1

        # ----------------------------------------------------------
        # (c) Keep the best solution seen so far
        # ----------------------------------------------------------
        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_u = u.copy()
            best_iter = it
            it_since_last_best = 0
            print(f"[iter {it:07d}] new best O={best_score:.6f}")

        # ----------------------------------------------------------
        # (d) Schedule parameters (fraction of total planned iterations)
        # ----------------------------------------------------------
        fraction = min(it / max_iters_est, 1.0)
        beta = beta_start + (beta_end - beta_start) * fraction
        eta = eta_start * ((eta_end / eta_start) ** fraction)

        # ----------------------------------------------------------
        # (e) Build the weight vector w over convolution entries
        # ----------------------------------------------------------
        if fraction < 0.2:                         # warm‑up: smooth soft‑max
            shifted = conv - M                     # ≤ 0, max is 0
            w = np.exp(beta * shifted)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.zeros_like(conv)
                w[conv.argmax()] = 1.0
            else:
                w = w / w_sum
        else:                                      # near‑end: hard‑max active set
            tol = max(1e-12, M * 1e-8)
            active = np.where(conv >= M - tol)[0]
            if active.size == 0:
                w = np.zeros_like(conv)
                w[conv.argmax()] = 1.0
            else:
                w = np.zeros_like(conv)
                w[active] = 1.0 / active.size

        # ----------------------------------------------------------
        # (f) Gradient of the (smoothed) max‑convolution w.r.t a
        # ----------------------------------------------------------
        g = 2.0 * _correlation_fft(w, a)            # shape (n,)

        # Optional gradient clipping for numerical stability
        # (clip to a few standard deviations of the current gradient)
        g_norm = np.linalg.norm(g)
        if g_norm > 0:
            max_norm = 10.0
            if g_norm > max_norm:
                g = g * (max_norm / g_norm)

        # ----------------------------------------------------------
        # (g) Adam update on the log‑variables u
        # ----------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        bias_correction1 = 1.0 - beta1 ** it
        bias_correction2 = 1.0 - beta2 ** it
        m_hat = m / bias_correction1
        v_hat = v / bias_correction2

        step = eta * m_hat / (np.sqrt(v_hat) + eps_adam)
        u = u - step

        # ----------------------------------------------------------
        # (h) Occasional random jitter to escape flat plateaux
        # ----------------------------------------------------------
        if it % jitter_interval == 0:
            u += np.random.randn(n) * jitter_scale

        # ----------------------------------------------------------
        # (i) Cheap restart if we have been stagnant for a while
        # ----------------------------------------------------------
        if it % restart_interval == 0 or it_since_last_best >= stagnation_limit:
            # Re‑initialise from the best solution with a small perturbation.
            u = best_u.copy()
            u += np.random.randn(n) * (jitter_scale * 0.5)
            it_since_last_best = 0
            print(f"[restart] iter {it}, re‑initialised from best (O={best_score:.6f})")

    # ------------------------------------------------------------------
    # 3) Finalise the best sequence and evaluate with the provided API
    # ------------------------------------------------------------------
    best_seq = np.exp(best_u - np.max(best_u))
    best_seq = best_seq / best_seq.sum()
    best_seq_list = best_seq.tolist()

    try:
        final_score = evaluate_sequence(best_seq_list)
        print(f"[final] evaluate_sequence reports O={final_score:.6f} "
              f"(best iteration {best_iter})")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_seq_list
