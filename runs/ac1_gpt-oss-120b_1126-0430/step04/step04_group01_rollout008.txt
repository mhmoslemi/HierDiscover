"""
Propose a step‑function (non‑negative vector) that (approximately) minimises

        O(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm is a **mirror‑descent + simulated‑annealing** hybrid:

*  The vector is always kept on the probability simplex (∑a_i = 1,
   a_i ≥ 0).  With this normalisation  O(a) = 2·n·max(conv(a,a)).

*  The dominant part of the optimisation is a *mirror‑descent* on the
   L‑∞‑norm of the self‑convolution.  The sub‑gradient of
   max(conv(a,a)) is computed exactly (averaged over all maximising
   indices) and the update uses an exponentiated‑gradient step

        a_i ← a_i·exp(−η·g_i)   ,   a ← a / ∑a .

   The step size η decays as 1/√t.

*  To avoid stagnation a low‑probability **random mass‑transfer** move
   (Metropolis acceptance) is mixed in – a fraction of probability mass
   is moved from a “heavy’’ index to a random destination.

*  Every few thousand iterations a tiny **smoothing** (5‑point moving
   average) is applied; it helps to keep the shape smooth without
   destroying the progress made by the mirror step.

*  When the objective has not improved for a while the current iterate is
   *re‑jittered* around the best solution found so far – this provides a
   cheap multi‑restart scheme inside the same budget.

All heavy lifting (convolution, sub‑gradient) is performed with FFTs;
the code fits comfortably into the 1000 s budget even for the
≈ 4600‑element vectors used in the reference runs.

The function returns a plain Python list; the caller will evaluate it
with the provided ``evaluate_sequence`` function.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility: projection onto the probability simplex (Euclidean)
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex {x≥0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution via FFT (returns the *linear* convolution,
# i.e. length 2·n‑1, not circular).
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return a⨂a (full linear convolution) using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution (averaged over all maximisers).
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. a (average over all arg‑max positions)."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)          # indices attaining the max
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g
