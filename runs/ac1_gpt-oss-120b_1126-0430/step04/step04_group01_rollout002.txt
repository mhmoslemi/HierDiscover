"""
Propose a non‑negative step‑function (list of heights) that (approximately)
minimises

    O(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

The optimizer works in two phases:

* Phase‑1 – smooth surrogate optimisation.
  We minimise a log‑sum‑exp approximation of the max‑convolution plus a
  (decaying) entropy term using Adam.  The smoothness parameter α is
  annealed from a small start to a very large value so the surrogate
  converges to the true max.  After each Adam step the vector is projected
  onto the probability simplex (∑a=1, a≥0).

* Phase‑2 – targeted stochastic refinement.
  Starting from the best candidate of phase‑1 we perform many cheap
  “mass‑transfer’’ moves that are biased towards the indices that actually
  contribute to the current maximal convolution entry.  Moves are accepted
  with a Metropolis rule that slowly cools down.  Every few iterations a
  true sub‑gradient step for the exact max‑convolution is taken and a tiny
  smoothing move is occasionally applied.

The whole search respects the 1000 s wall‑time limit and returns the best
sequence (as a plain Python list) found within the budget.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i >= 0, sum_i x_i = 1}
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# --------------------------------------------------------------
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution a * a using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    # length 2*n-1, keep only the relevant prefix
    return conv_full[: 2 * a.size - 1]


# --------------------------------------------------------------
def _objective_fast(a: np.ndarray, n: int, L: int) -> float:
    """
    Fast exact objective for a vector that already satisfies sum(a)=1.
    Returns 2·n·max(conv(a,a)).
    """
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# --------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a (averaged over all maximisers).
    The gradient is
        g_i = 2 * mean_j a_j   over all pairs (i,j) with i+j = k_max .
    """
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)          # all maximising shifts
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


# --------------------------------------------------------------
def _smooth_gradient_plus_entropy(
    a: np.ndarray,
    conv: np.ndarray,
    alpha: float,
    L: int,
    lam: float,
) -> np.ndarray:
    """
    Gradient of the smooth surrogate:
        (2·n/α)·log∑_k exp(α·conv_k) + λ·∑ a_i·log a_i .
    The first part reduces to 2·corr(w,a) with w_k ∝ exp(α·conv_k);
    the second part contributes λ·(log a_i + 1).
    """
    # soft‑max weights w_k ∝ exp(α·conv_k)
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)      # improve stability
    w = exp_scaled / exp_scaled.sum()             # shape (2n‑1,)

    # correlation w ⋆ a (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n_len = a.size
    grad_corr = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]

    # entropy part λ·∑ a_i·log a_i  -> gradient λ·(log a_i + 1)
    eps = 1e-20
    grad_ent = lam * (np.log(a + eps) + 1.0)

    return grad_corr + grad_ent


# --------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that minimises the provided evaluation
    function.  Returns the best sequence found (plain list of floats).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2          # safety margin

    # ------------------------------------------------------------------
    # Choose problem size
    if "height_sequence_1" in globals():
        base = np.array(height_sequence_1, dtype=float)
        n = base.size
        a = base.copy()
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-12:
            a = np.full(n, 1.0 / n)
        else:
            a = a / a.sum()
        print(f"[init] start from supplied sequence, n={n}")
    else:
        n = 5000
        a = np.full(n, 1.0 / n)
        print(f"[init] start from uniform, n={n}")

    # FFT length – next power of two covering 2·n‑1
    L = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # Phase‑1 : Adam on smooth surrogate (log‑sum‑exp + entropy)
    best_seq = a.copy()
    best_score = _objective_fast(best_seq, n, L)
    print(f"[phase‑1] start O≈{best_score:.6f}")

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # schedule for smoothness and entropy
    alpha_start, alpha_end = 5.0, 20000.0
    lam_start, lam_end = 0.05, 1e-4

    max_iter_phase1 = 300_000
    it = 0
    phase1_deadline = start_time + budget_s * 0.48   # ~48 % of total budget

    while time.time() < phase1_deadline and it < max_iter_phase1:
        # exact objective for logging
        if it % 10000 == 0:
            cur_obj = _objective_fast(a, n, L)
            if cur_obj < best_score:
                best_score = cur_obj
                best_seq = a.copy()
                print(f"[phase‑1] iter {it:,} new best O≈{best_score:.6f}")

        # annealing
        frac = it / max_iter_phase1
        alpha = alpha_start + (alpha_end - alpha_start) * frac
        lam = lam_start * (lam_end / lam_start) ** frac

        # convolution for gradient
        conv = _conv_fft(a, L)

        # smooth gradient
        grad = _smooth_gradient_plus_entropy(a, conv, alpha, L, lam)

        # Adam update with bias correction
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        # simple 1/√t learning‑rate decay
        lr = 0.6 / math.sqrt(it + 1)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape plateaus
        if np.random.rand() < 0.001:
            a_new += np.random.randn(n) * 1e-4

        # projection onto the simplex
        a_new = _project_to_simplex(a_new)

        # optional pruning of ultra‑small entries every few thousand steps
        if it % 5000 == 0 and it > 0:
            thresh = np.percentile(a_new, 0.5)   # drop lowest half‑percentile
            a_new[a_new < thresh] = 0.0
            a_new = _project_to_simplex(a_new)

        a = a_new
        it += 1

    # ------------------------------------------------------------------
    # Phase‑2 : Stochastic, target‑aware local refinement
    print(f"[phase‑2] entering refinement, best O≈{best_score:.6f}")

    # Parameters for the stochastic search
    max_iter_local = 180_000
    subgrad_interval = 5
    temperature0 = 0.04
    no_improve_limit = 3000
    restarts = 0

    while time.time() < deadline:
        # Initialise a new restart (biased towards current best)
        if np.random.rand() < 0.6:
            a = best_seq.copy()
            a += np.random.randn(n) * 0.0015
            a = np.clip(a, 0.0, None)
            a = _project_to_simplex(a)
        else:
            a = np.random.rand(n)
            a = _project_to_simplex(a)

        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        it_loc = 0
        no_improve = 0
        T = temperature0

        while time.time() < deadline and it_loc < max_iter_local:
            # ---------- mass‑transfer move targeting the current max shift ----------
            k_max = int(np.argmax(conv))
            # compute contributions for the current max shift
            contrib = np.zeros(n)
            i_start = max(0, k_max - (n - 1))
            i_end = min(n - 1, k_max)
            i_idx = np.arange(i_start, i_end + 1, dtype=int)
            j_idx = k_max - i_idx
            contrib[i_idx] = a[i_idx] * a[j_idx]

            # pick a source index with probability proportional to its contribution
            total_contrib = contrib.sum()
            if total_contrib > 0 and np.random.rand() < 0.7:
                probs = contrib / total_contrib
                src = np.random.choice(n, p=probs)
            else:
                # fallback: choose any index with non‑negligible mass
                nz = np.flatnonzero(a > 1e-14)
                src = np.random.choice(nz) if nz.size else np.random.randint(n)

            # pick a destination where mass is relatively small (to spread it out)
            low = np.flatnonzero(a < np.median(a))
            if low.size:
                dst = np.random.choice(low)
            else:
                dst = np.random.randint(n)

            if src != dst and a[src] > 1e-14:
                delta = a[src] * np.random.rand() * 0.12   # up to 12 % of source mass
                a_new = a.copy()
                a_new[src] -= delta
                a_new[dst] += delta
                a_new = _project_to_simplex(a_new)

                conv_new = _conv_fft(a_new, L)
                new_obj = 2.0 * n * conv_new.max()

                # Metropolis acceptance
                if new_obj < cur_obj or np.random.rand() < math.exp((cur_obj - new_obj) / max(T, 1e-14)):
                    a, conv, cur_obj = a_new, conv_new, new_obj
                    no_improve = 0
                    if cur_obj < best_score:
                        best_score = cur_obj
                        best_seq = a.copy()
                        elapsed = time.time() - start_time
                        print(f"[{elapsed:.1f}s] new global best O≈{best_score:.6f} (restart={restarts}, iter={it_loc})")
                else:
                    no_improve += 1
            else:
                no_improve += 1

            # ---------- occasional sub‑gradient step on the exact max ----------
            if it_loc % subgrad_interval == 0:
                g = _subgradient_max_conv(a, conv)
                eta = 0.09 * (1.0 - it_loc / max_iter_local) + 0.001
                a = a - eta * g
                a = _project_to_simplex(a)
                conv = _conv_fft(a, L)
                cur_obj = 2.0 * n * conv.max()
                if cur_obj < best_score:
                    best_score = cur_obj
                    best_seq = a.copy()
                    elapsed = time.time() - start_time
                    print(f"[{elapsed:.1f}s] sub‑grad improved O≈{best_score:.6f}")

            # ---------- tiny smoothing move every few thousand iterations ----------
            if it_loc % 4000 == 0 and it_loc > 0:
                kernel = np.array([0.125, 0.25, 0.25, 0.25, 0.125])
                a_smooth = np.convolve(a, kernel, mode="same")
                a = _project_to_simplex(0.85 * a + 0.15 * a_smooth)

            # temperature decay
            T *= 0.99975

            it_loc += 1

            # restart if stuck
            if no_improve >= no_improve_limit:
                break

        restarts += 1
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Final sanity check with the provided evaluator (just for logging)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:  # pragma: no cover
        print("[final] evaluate_sequence failed:", e)

    return best_seq.tolist()
