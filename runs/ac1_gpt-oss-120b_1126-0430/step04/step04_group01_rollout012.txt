"""
Propose a non‑negative step‑function that (approximately) minimises

    O(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

Algorithm
---------
1️⃣  **Flatten the convolution** – we minimise the variance of the
    convolution vector `c = a★a` (i.e.  Σ (c_k‑μ)² where μ is the
    uniform mean).  This objective directly drives all entries of the
    convolution towards the same value, which is the right thing to do
    because `max(c)` is then forced to be close to the average.
    The gradient of this smooth objective is
        g = 4·corr(c‑μ , a)                (correlation via FFT)
    and we run an Adam optimiser on the simplex (∑a_i = 1, a_i ≥ 0).

2️⃣  **Refine the maximum only** – after a first flattening phase we
    switch to the soft‑max (log‑sum‑exp) surrogate used in the
    reference implementation.  This focuses the optimisation on the
    actual peak of the convolution.

3️⃣  **Stochastic local improvement** – a cheap “mass‑transfer’’ Monte‑
    Carlo loop with a Metropolis acceptance rule, plus occasional
    sub‑gradient steps on `max(conv)`.  This clean‑up phase is identical
    to the one that gave the previous best result.

The three phases share the same time budget (≈ 1000 s).  The best
sequence found in any phase is returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution via FFT (length 2·n‑1)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return the full convolution a★a (length 2·n‑1) using FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Exact objective (sum of a is already 1)
def _objective_simplex(a: np.ndarray, n: int, L: int) -> float:
    """2·n·max(conv(a,a)) for sum(a)=1."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Gradient of the variance‐of‑convolution objective
def _grad_variance(a: np.ndarray, conv: np.ndarray, n: int, L: int) -> np.ndarray:
    """
    Gradient of   f(a) = Σ_k (conv_k - μ)²   where μ = 1/(2n‑1) (for sum(a)=1).
    Closed form: g = 4·corr(conv‑μ , a) .
    """
    mu = 1.0 / (2 * n - 1)
    diff = conv - mu                     # length 2n‑1
    a_rev = a[::-1]

    fft_diff = np.fft.rfft(diff, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_diff * fft_a_rev, n=L)

    # central slice corresponds to the correlation values for each index i
    start = n - 1
    grad = 4.0 * corr_rev[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Gradient of the smooth (log‑sum‑exp) surrogate + tiny entropy term
def _smooth_gradient_plus_entropy(
    a: np.ndarray,
    conv: np.ndarray,
    alpha: float,
    L: int,
    lam: float,
) -> np.ndarray:
    """
    ∇_a ( (2·n/α)·log∑_k e^{α·conv_k} + λ·∑ a_i·log a_i ).
    The first part is 2·corr(w,a) with w ∝ exp(α·conv).
    """
    # soft‑max weights
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / exp_scaled.sum()                 # shape (2n‑1,)

    # correlation term  (same trick as in _grad_variance)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n_len = a.size
    grad_corr = 2.0 * corr_rev[n_len - 1:n_len - 1 + n_len]

    # tiny entropy (helps avoid exact zeros)
    eps = 1e-20
    grad_ent = lam * (np.log(a + eps) + 1.0)

    return grad_corr + grad_ent

            a += np.random.randn(n) * 0.0015

            a = _project_to_simplex(a)
