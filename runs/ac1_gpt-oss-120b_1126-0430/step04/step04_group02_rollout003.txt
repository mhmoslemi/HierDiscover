import cvxpy as cp

def solve_lp(f, M):
    n = len(f)
    b = cp.Variable(n, nonneg=True)
    conv = cp.conv(f, b)  # shape 2n-1
    constraints = [conv <= M]
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, verbose=False, max_iters=5000)
    # If prob.status not optimal, maybe fallback.
    b_val = b.value
    b_val = np.maximum(b_val, 0) # just to avoid negative small values
    return b_val

        # Use cvxpy

        # If b is None or solver fails, break

        sum_b = b.sum()
        if sum_b <= 0:
            break

        g = b / sum_b

        f = (1 - t) * f + t * g

        f = f / f.sum()

            jitter = np.random.normal(scale=0.005, size=n)
            f = f + jitter

        if time.time() - start > time_budget * 0.95:

"""
Multi‑length Adam optimiser for the step‑function problem.

The algorithm explores several support sizes.  For each size we run an
Adam optimiser on the soft‑max (log‑it) parametrisation of the simplex,
with an annealed smooth‑max approximation of the convolution maximum.
Every few thousand iterations a tiny Gaussian jitter is added to avoid
plateaus and every 70 000 iterations we restart from the current best.
After the optimiser finishes we prune entries that fell below a tiny
threshold, renormalise the vector and run a cheap stochastic local search
(Random mass‑transfer moves) for the remaining budget.  The best sequence
found over all lengths is finally returned.

All numbers are kept non‑negative, normalised to sum = 1 and the result
conforms to the security checks of ``evaluate_sequence``.
"""
import time
import numpy as np

# -------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# -------------------------------------------------------------
def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using FFT (real‑valued)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv


# -------------------------------------------------------------
def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation of w with a (g_i = Σ_t w[i+t]·a[t]), 0 ≤ i < len(a).
    Implemented via FFT.
    """
    n = a.shape[0]
    m = w.shape[0]                 # = 2·n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


# -------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)), assumes sum(a)=1."""
    n = a.size
    conv = _conv_self_fft(a)
    return 2.0 * n * conv.max()


# -------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


# -------------------------------------------------------------
def _run_opt(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
) -> tuple[list[float], float]:
    """
    Adam optimisation of a length‑``n`` sequence for at most ``time_limit`` seconds.
    Returns the best sequence (as a plain list) and its exact objective.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # ---- initialisation -------------------------------------------------
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
        a = a / a.sum()
    else:
        a = rng.random(n)
        a = a / a.sum()

    # work in log‑it space (automatically respects simplex)
    x = np.log(a + 1e-16)

    m = np.zeros_like(x)          # Adam first moment
    v = np.zeros_like(x)          # Adam second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # ---- schedules ------------------------------------------------------
    beta_start = 0.5               # smoothing start
    beta_end   = 4000.0            # smoothing end (sharp enough)
    lr_start   = 0.15
    lr_end     = 0.001
    # a very large iteration budget – we stop by time, not by count
    target_iters = 800_000

    best_a = a.copy()
    best_obj = _objective(best_a)

    it = 0
    while time.time() - start < time_limit:
        it += 1

        # probabilities from current logits
        a = _softmax(x)

        # convolution & exact objective
        conv = _conv_self_fft(a)
        M = conv.max()
        obj = 2.0 * n * M
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        # schedule parameters (smooth annealing & lr decay)
        t_frac = min(it / target_iters, 1.0)
        beta = beta_start * np.exp(np.log(beta_end / beta_start) * t_frac)
        lr   = lr_start   * np.exp(np.log(lr_end   / lr_start)   * t_frac)

        # weight vector for the smoothed max
        if beta > 3000.0:                       # sharp enough → true arg‑max
            kmax = int(np.argmax(conv))
            w = np.zeros_like(conv)
            w[kmax] = 1.0
        else:
            shifted = conv - M                # ≤ 0
            w = np.exp(beta * shifted)
            w = w / w.sum()                  # length 2·n‑1, sums to 1

        # gradient w.r.t. a (smooth max)
        g_a = 2.0 * _correlation_fft(w, a)   # (n,)

        # natural gradient in log‑it space (projected onto simplex)
        dot = np.dot(a, g_a)                 # scalar ⟨a, g_a⟩
        g_x = a * (g_a - dot)                # (n,)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to shake out of plateaus
        if it % 6000 == 0:
            jitter = rng.normal(scale=0.02, size=n)
            x += jitter

        # periodic restart from the current best (helps escape long stalls)
        if it % 70_000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.005, size=n)
            m.fill(0.0)
            v.fill(0.0)
            print(f"[n={n}] restart at iter {it}")

    # ---- compress negligible entries ------------------------------------
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        n_new = best_a.size
        best_obj = _objective(best_a)
        print(f"[compress] reduced length {n} → {n_new}, O={best_obj:.6f}")

    # ---- short stochastic local search ----------------------------------
    remaining = time_limit - (time.time() - start)
    if remaining > 0.5:
        a_loc = best_a.copy()
        obj_loc = best_obj
        n_loc = a_loc.size
        local_deadline = time.time() + remaining * 0.35
        while time.time() < local_deadline:
            i, j = rng.choice(n_loc, size=2, replace=False)
            # move a random 0–2 % of mass from i to j
            delta = a_loc[i] * rng.random() * 0.02
            if delta <= 0.0:
                continue
            a_new = a_loc.copy()
            a_new[i] -= delta
            a_new[j] += delta
            conv_new = _conv_self_fft(a_new)
            obj_new = 2.0 * n_loc * conv_new.max()
            if obj_new < obj_loc:
                a_loc = a_new
                obj_loc = obj_new
                if obj_new < best_obj:
                    best_a = a_new
                    best_obj = obj_new
        print(f"[local] final local O={best_obj:.6f}")

    return best_a.tolist(), best_obj


# -------------------------------------------------------------
def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **_ignore,
) -> list[float]:
    """
    Entry point for the competition harness.
    Runs several independent Adam optimisations for different support
    sizes, keeps the best sequence and finishes with a brief random
    local‐search refinement.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # -----------------------------------------------------------------
    # Decide which lengths to try.  We start with the length of the
    # previous best (if available) and then explore a geometric ladder.
    lengths = [200, 400, 800, 1600, 3200]
    if "height_sequence_1" in globals():
        seq0 = np.asarray(height_sequence_1, dtype=float).copy()
        seq0 = np.clip(seq0, 0.0, None)
        if seq0.sum() < 0.01:
            seq0 = np.ones_like(seq0)
        seq0 = seq0 / seq0.sum()
        n0 = seq0.size
        if n0 not in lengths:
            lengths.insert(0, n0)
    else:
        seq0 = None
        n0 = None

    best_seq = None
    best_obj = float("inf")

    # -----------------------------------------------------------------
    # Allocate most of the remaining budget to the first (largest) length,
    # then share the rest among the others.
    for idx, n in enumerate(lengths):
        remaining = deadline - time.time()
        if remaining <= 1.0:
            break

        if idx == 0:
            run_time = remaining * 0.70
        else:
            # share the rest equally
            run_time = (remaining - 0.5) / (len(lengths) - idx)

        # initialise from the previous global solution for the first run
        init = None
        if idx == 0 and seq0 is not None:
            if seq0.size != n:
                if seq0.size > n:
                    init = seq0[:n]
                else:
                    pad = np.zeros(n - seq0.size)
                    init = np.concatenate([seq0, pad])
                init = init / init.sum()
            else:
                init = seq0

        seq, val = _run_opt(
            n=n,
            time_limit=run_time,
            init_a=init,
            seed=seed + idx * 13,
        )
        if val < best_obj:
            best_obj = val
            best_seq = seq
            print(f"[global] new overall best O={best_obj:.6f} (n={n})")

    # -----------------------------------------------------------------
    # Final short random local search on the overall best (if time permits)
    if best_seq is not None:
        remaining = deadline - time.time()
        if remaining > 1.0:
            a = np.array(best_seq, dtype=float)
            n = a.size
            best_obj = _objective(a)
            rng = np.random.default_rng(seed + 999)
            local_end = time.time() + remaining * 0.30
            while time.time() < local_end:

                delta = a[i] * rng.random() * 0.02
                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta
                obj_new = _objective(a_new)
                if obj_new < best_obj:
                    a = a_new
                    best_obj = obj_new
                    best_seq = a.tolist()
            print(f"[final local] O={best_obj:.6f}")

    # -----------------------------------------------------------------
    # Final sanity check via the provided evaluate_sequence routine.

        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    # guarantee a non‑empty return
    if not best_seq:
        best_seq = [1.0]

    return best_seq
