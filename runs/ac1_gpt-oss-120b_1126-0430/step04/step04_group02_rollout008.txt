"""
Optimiser for the non‑negative step‑function problem.

The code runs a few Adam‑optimisation phases on the soft‑max
parameterisation of the simplex, starting from several sensible
initialisations (including an exponential shape).  After the
gradient phase a fast greedy local‑search (with occasional simulated‑
annealing acceptance) is performed.  Different support sizes are tried
within the given time budget and the best sequence found (as a plain
list of floats) is returned.

All helper functions are defined at module level – no closures or
lambdas – and the routine respects the 1000 s timeout.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a power‑of‑2 FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv


def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation of w (length 2n‑1) with a (length n):
    g_i = Σ_t w[i+t]·a[t]   for i = 0,…,n‑1
    Implemented by FFT (linear convolution).
    """
    n = a.shape[0]
    m = w.shape[0]                     # = 2n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


def _obj_exact(a: np.ndarray) -> float:
    """Exact objective  O = 2·n·max(conv(a,a))  (sum(a)=1)."""
    n = a.size
    conv = _conv_self_fft(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# initialisation helpers
# ----------------------------------------------------------------------
def _exponential_initialisation(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a few exponential‑shaped candidates and keep the best.
    The distribution is a_i ∝ r^i for a random decay factor r∈(0,1).
    """
    best_a = None
    best_val = np.inf
    # try a handful of r‑values (log‑uniform sampling)
    for _ in range(6):
        # avoid r≈0 or r≈1 which give pathological vectors
        log_r = rng.uniform(math.log(0.02), math.log(0.98))
        r = math.exp(log_r)
        a = np.power(r, np.arange(n, dtype=float))
        s = a.sum()
        if s == 0.0:
            continue
        a /= s
        val = _obj_exact(a)
        if val < best_val:
            best_val = val
            best_a = a
    # add a pure random Dirichlet start
    a_rand = rng.random(n)
    a_rand /= a_rand.sum()
    val_rand = _obj_exact(a_rand)
    if val_rand < best_val:
        best_a = a_rand
    return best_a


# ----------------------------------------------------------------------
# Adam optimiser on the log‑it (soft‑max) parametrisation
# ----------------------------------------------------------------------
def _adam_optimize(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None,
    rng: np.random.Generator,
    beta_start: float,
    beta_end: float,
    lr_start: float,
    lr_end: float,
    jitter_interval: int = 6000,
    restart_interval: int = 70000,
) -> tuple[np.ndarray, float]:
    """
    Optimise a length‑n probability vector for the given time budget.
    Returns the best vector (numpy array) and its exact objective.
    """
    start = time.time()
    # ---------- initial vector ------------------------------------------------
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = _exponential_initialisation(n, rng)
    else:
        a = _exponential_initialisation(n, rng)

    a = a / a.sum()
    # work in log‑it space (logits of a simplex)
    x = np.log(a + 1e-16)

    # Adam moments
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    best_a = a.copy()
    best_obj = _obj_exact(best_a)

    it = 0
    # we set a rough upper bound on iterations to keep schedule stable
    # (the schedule is based on relative progress, not absolute count)
    max_iters = 300_000

    while time.time() - start < time_limit and it < max_iters:
        it += 1

        # ----- probabilities from logits --------------------------------------
        a = _softmax(x)

        # ----- exact objective (for best‑track only) ----------------------------
        conv = _conv_self_fft(a)
        max_conv = conv.max()
        obj = 2.0 * n * max_conv
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[n={n:5d} it={it:06d}] new best O={best_obj:.6f}")

        # ----- schedule (fraction of allocated time) ---------------------------
        t = min(it / max_iters, 1.0)
        beta = beta_start * np.exp(np.log(beta_end / beta_start) * t)
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t)

        # ----- soft‑max weights for smoothed max --------------------------------
        if beta > 5000.0:
            kmax = int(np.argmax(conv))
            w = np.zeros_like(conv)
            w[kmax] = 1.0
        else:
            shifted = conv - max_conv               # ≤ 0
            w = np.exp(beta * shifted)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(conv, 1.0 / conv.size)
            else:
                w /= w_sum

        # ----- gradient of the smoothed max (see analysis) --------------------
        g_a = 2.0 * _correlation_fft(w, a)          # shape (n,)

        # ----- transform to log‑it space (natural gradient) --------------------
        dot = np.dot(a, g_a)                         # scalar ⟨a,g_a⟩
        g_x = a * (g_a - dot)                        # shape (n,)

        # ----- Adam update ------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- occasional jitter to kick the optimiser out of plateaus ----------
        if it % jitter_interval == 0:
            jitter = rng.normal(scale=0.02, size=n)
            x += jitter

        # ----- restart from the current best if we have been stuck ----------
        if it % restart_interval == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.005, size=n)
            m.fill(0.0)
            v.fill(0.0)
            print(f"[n={n}] restart at iteration {it}")

    # ----------------------------------------------------------------------
    # post‑processing: drop near‑zero entries, renormalise and recompute O
    # ----------------------------------------------------------------------
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a /= best_a.sum()
        best_obj = _obj_exact(best_a)
        print(f"[compress] length reduced {n}->{best_a.size}, O={best_obj:.6f}")

    return best_a, best_obj


# ----------------------------------------------------------------------
# Simple greedy / simulated‑annealing local search
# ----------------------------------------------------------------------
def _local_search(
    a_start: np.ndarray,
    time_budget: float,
    rng: np.random.Generator,
    move_frac: float = 0.02,
) -> tuple[np.ndarray, float]:
    """
    Perform many tiny mass‑transfer moves.
    Accept improvements always; worse moves are accepted with a
    temperature‑dependent probability (simulated annealing).
    """
    start = time.time()
    a = a_start.copy()
    n = a.size
    best_obj = _obj_exact(a)

    # linear cooling: start temperature roughly proportional to the excess above the theoretical lower bound (≈1)
    init_temp = max(best_obj - 1.0, 1e-3)

    while time.time() - start < time_budget:
        # pick two distinct positions
        i, j = rng.choice(n, size=2, replace=False)
        if a[i] <= 1e-18:
            continue
        # move a small fraction of the mass from i to j
        delta = a[i] * rng.random() * move_frac
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        obj_new = _obj_exact(a_new)

        # temperature decays linearly with time
        elapsed = time.time() - start
        frac = elapsed / time_budget
        temp = init_temp * (1.0 - frac) + 1e-9

        if obj_new < best_obj or rng.random() < math.exp(-(obj_new - best_obj) / temp):
            a = a_new
            if obj_new < best_obj:
                best_obj = obj_new
    return a, best_obj

        leftover = deadline - now
        if leftover > 0.6:
            # give roughly 20 % of the leftover time to a focussed local search
            ls_time = min(0.2 * leftover, 30.0)   # cap to keep overall run time reasonable
            a_arr = np.asarray(best_seq, dtype=float) if best_seq is not None else np.random.rand(n)
            a_arr = a_arr / a_arr.sum()
            seq2, val2 = _local_search(a_arr, ls_time, rng)
            if val2 < best_obj:
                best_obj = val2
                best_seq = seq2.tolist()
                print(f"[local‑search] improved O={best_obj:.6f} (n={len(seq2)})")

    # ------------------------------------------------------------------
    # Final verification with the provided evaluate_sequence routine.
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq)
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
        # the returned list must be the raw candidate (not the clipped version)
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_seq
