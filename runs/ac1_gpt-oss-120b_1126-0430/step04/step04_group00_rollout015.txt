#!/usr/bin/env python
"""
A multi‑restart, smoothed‑max Adam optimiser for the step‑function problem.

The algorithm works on the probability simplex (∑a_i = 1, a_i ≥ 0).
For a given distribution *a* the cheap objective is
    J(a) = 2 * n * max_k (a * a)[k]               (n = len(a))

We minimise J using a smooth log‑sum‑exp surrogate for the max,
compute gradients via FFT, and perform Adam updates on the log‑it
(soft‑max) parametrisation.  Several random restarts (uniform,
power‑law, Dirichlet, the previous best sequence) are tried within
the time budget and the best candidate is locally refined by tiny
random mass transfers.  The final sequence is returned and its exact
score is printed with ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Core linear‑algebra utilities (FFT based)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max, returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it (inverse soft‑max) representation of a probability vector."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """
    Smallest power of two ≥ 3·n‑2.
    This length is sufficient for a linear correlation of length n vectors.
    """
    return 1 << ((3 * n - 2).bit_length())


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    total = new_a.sum()
    if total == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= total
    return new_a


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """Normalised power‑law a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _conv_max_and_grad(a: np.ndarray, eps_factor: float) -> tuple[float, np.ndarray]:
    """
    Compute max entry of a*a and the gradient of the smoothed objective.

    Parameters
    ----------
    a : np.ndarray
        Probability vector (sum = 1, non‑negative).
    eps_factor : float
        Multiplicative factor for the smoothing constant:
        eps = eps_factor * max(conv).

    Returns
    -------
    max_conv : float
        Maximum of the self‑convolution a*a.
    grad_a : np.ndarray
        Gradient of 2·n·smooth_max w.r.t. a (already multiplied by 4·n).
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    # FFT of a (zero‑padded)
    a_fft = np.fft.rfft(a, L)

    # -------- self‑convolution a*a --------
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = conv.max()

    # -------- smooth‑max weights --------
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv        # for numerical stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or np.isnan(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # -------- cross‑correlation a ⋆ w --------
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # gradient (4·n * correlation)
    grad_a = 4.0 * n * corr
    return float(max_conv), grad_a


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a (no gradient)."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _optimize_distribution(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the logits of ``a0`` respecting the simplex.
    Runs until ``deadline`` and returns the best distribution found
    together with its cheap score (2·n·max_conv).
    """
    n = a0.shape[0]
    # initialise logits with a tiny jitter
    z = _logits_from_distribution(a0)
    z += 0.01 * np.random.randn(n)

    # Adam state
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    base_alpha = 0.20                # base learning rate for logits
    eps_factor = 0.10                # initial smoothing factor
    eps_factor_min = 5e-5
    eps_factor_decay = 0.9997

    best_a = a0.copy()
    best_score = 2.0 * n * _conv_max(best_a)
    it = 0
    last_improve = 0

    while time.time() < deadline and it < 5_000_000:
        it += 1

        # current point on simplex
        a = _softmax(z)

        # cheap evaluation + smooth gradient
        max_conv, grad_a = _conv_max_and_grad(a, eps_factor)
        cheap_score = 2.0 * n * max_conv

        # keep best
        if cheap_score < best_score:
            best_score = cheap_score
            best_a = a.copy()
            last_improve = it

        # gradient w.r.t. logits (mirror descent on the simplex)
        mu = np.dot(grad_a, a)
        grad_z = a * (grad_a - mu)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step‑size schedule (inverse‑sqrt decay)
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step_dir = - m_hat / (np.sqrt(v_hat) + eps_adam)

        # back‑tracking line search (max 4 halvings)
        trial_alpha = alpha_t
        accepted = False
        for _ in range(4):
            z_new = z + trial_alpha * step_dir
            z_new = np.clip(z_new, -50.0, 50.0)   # keep exp(z) safe
            a_new = _softmax(z_new)
            new_score = 2.0 * n * _conv_max(a_new)
            if new_score <= cheap_score:
                z = z_new
                accepted = True
                break
            trial_alpha *= 0.5

        if not accepted:
            # no improvement – take a modest step anyway
            z += alpha_t * step_dir
            z = np.clip(z, -50.0, 50.0)

        # occasional random jitter to escape plateaux
        if it % 3000 == 0 and np.random.rand() < 0.12:
            z += 0.015 * np.random.randn(n)
            m *= 0.6
            v *= 0.6

        # decay smoothing factor
        eps_factor = max(eps_factor_min, eps_factor * eps_factor_decay)

        # early stagnation break (if we made no improvement for a long time)
        if it - last_improve > 250_000:
            # No progress for a while – stop early to preserve budget
            break

    return best_a, best_score


def _local_refine_distribution(a: np.ndarray, deadline: float) -> np.ndarray:
    """
    Very cheap hill‑climbing: repeatedly transfer a tiny amount of mass
    from a randomly chosen “heavy’’ index to a randomly chosen “light’’
    index and keep the move if the cheap score improves.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = 2.0 * n * _conv_max(best_a)

    # relative step size – slowly reduced with time
    step_factor = 0.02

    while time.time() < deadline:
        i, j = np.random.choice(n, 2, replace=False)
        if best_a[i] <= 1e-12:
            continue
        delta = step_factor * best_a[i]
        a_new = best_a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # ensure non‑negativity (numerical safety)
        if a_new[i] < -1e-12:
            continue
        new_score = 2.0 * n * _conv_max(a_new)
        if new_score < best_score:
            best_a = a_new
            best_score = new_score
            # make future steps a little smaller when we improve
            step_factor = max(0.002, step_factor * 0.97)
    return best_a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) beats the target bound.
    The routine respects the given time budget (~1000 s) and uses several
    random restarts of a smoothed‑max Adam optimiser on the probability simplex.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the final evaluation
    deadline = start_time + budget_s - 3.0

    # ------------------------------------------------------------------
    # 0. Determine a base support size (use the previous best if available)
    # ------------------------------------------------------------------
    target_n = 20000
    try:
        # ``height_sequence_1`` may be a global list from the previous run
        init_global = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if init_global.shape[0] < target_n:
            # pad with zeros to reach the target length
            pad_len = target_n - init_global.shape[0]
            init_global = np.concatenate([init_global, np.zeros(pad_len)])
        a0 = np.maximum(init_global, 0.0)
        s = a0.sum()
        if s == 0.0:
            a0 = np.full_like(a0, 1.0 / a0.shape[0])
        else:
            a0 = a0 / s
        print(f"Initial sequence taken from global variable, length = {a0.shape[0]}")
    except Exception:
        # No global variable – fall back to a uniform start
        a0 = np.full(target_n, 1.0 / target_n)
        print(f"Fallback to uniform start, length = {target_n}")

    # ------------------------------------------------------------------
    # 1. Prepare a few diverse initial distributions for random restarts
    # ------------------------------------------------------------------
    initial_distributions = []

    # a) the (possibly) informed start from the global variable
    initial_distributions.append(a0)

    # b) pure uniform
    initial_distributions.append(np.full(target_n, 1.0 / target_n))

    # c) power‑law family (a few exponents)
    for p in (0.55, 0.60, 0.65):
        initial_distributions.append(_power_law_distribution(target_n, p))

    # d) a sparse Dirichlet (concentration 0.01 encourages sparsity)
    alpha = np.full(target_n, 0.01)
    initial_distributions.append(np.random.dirichlet(alpha))

    # ------------------------------------------------------------------
    # 2. Run the optimiser on each start, sharing the remaining budget
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    n_restarts = len(initial_distributions)
    time_per_restart = max(0.0, (deadline - time.time()) / max(1, n_restarts))

    for idx, a_start in enumerate(initial_distributions):
        if time.time() >= deadline:
            break
        sub_deadline = time.time() + time_per_restart
        print(f"[Restart {idx+1}/{n_restarts}] Optimising (time budget ≈ {time_per_restart:.1f}s)…")
        seq, score = _optimize_distribution(a_start, sub_deadline)
        print(f"    → best cheap score in this run = {score:.6f}")
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
        # small buffer to make sure we keep the global deadline
        if time.time() + 2.0 > deadline:
            break

    # ------------------------------------------------------------------
    # 3. Final local refinement on the overall best candidate
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("Starting final local refinement …")
        best_seq = _local_refine_distribution(best_seq, deadline)

    # ------------------------------------------------------------------
    # 4. Final exact evaluation with the provided evaluate_sequence()
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist() if best_seq is not None else [1.0]  # fallback
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:   # defensive – should never happen
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # ------------------------------------------------------------------
    # 5. Return the best sequence found.
    # ------------------------------------------------------------------
    return final_seq
