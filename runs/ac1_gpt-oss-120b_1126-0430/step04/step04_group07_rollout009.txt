"""
Propose a step‑function (non‑negative vector) that minimizes the bound

    O = 2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm is a *mirrored Adam* optimiser on the probability simplex.
For each iteration:

1. Build the probability vector `a` from log‑probabilities `u`
   (a = softmax(u);  sum(a)=1).
2. Compute the full self‑convolution `c = a * a` with a real FFT.
3. Form a weight vector `w` over the convolution entries:
   – early on a soft‑max (temperature β) concentrates the weight on
     the larger entries,
   – later a hard active set (all entries within a tiny tolerance of
     the current maximum) is used.
4. The gradient of the (smoothed) max‑convolution w.r.t. the log‑variables
   is `g = 2·correlation(w, a)`.  Correlation is also computed by FFT.
5. An Adam update (first‑ and second‑moment estimates) is applied to `u`
   with a slowly decaying learning‑rate schedule.
6. Small random jitter and cheap restarts from the best‑so‑far solution
   avoid stagnation.
7. The whole optimisation is run several times with different lengths
   `n` and random seeds; the best sequence found before the time budget
   expires is returned.

The routine respects the 1000 s budget, prints progress and finally
returns the best list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv


def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Linear correlation g_i = Σ_t w[i+t]·a[t]  (i=0,…,n‑1).
    `w` has length 2·n‑1, `a` length n.
    """
    n = a.shape[0]
    m = w.shape[0]                     # = 2·n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


# ----------------------------------------------------------------------
# Core optimisation for a fixed length
# ----------------------------------------------------------------------
def _optimise_fixed_length(
    n: int,
    seed: int,
    time_budget: float,
    init_from_known: bool = True,
) -> tuple[list, float]:
    """
    Run mirrored‑Adam on a vector of length ``n`` for at most
    ``time_budget`` seconds.  Returns the best sequence (list of floats)
    and its bound value.
    """
    np.random.seed(seed)
    deadline = time.time() + time_budget

    # ---- initial log‑probabilities ------------------------------------
    if init_from_known and "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=float)
        # truncate / pad to the desired length
        if a0.size > n:
            a0 = a0[:n]
        elif a0.size < n:
            a0 = np.concatenate([a0, np.zeros(n - a0.size)])
        a0 = np.maximum(a0, 0.0)
        if a0.sum() == 0.0:
            a0 = np.full(n, 1.0 / n)
        else:
            a0 = a0 / a0.sum()
        u = np.log(a0 + 1e-12)
    else:
        # random Dirichlet start (uniform concentration)
        a0 = np.random.dirichlet(np.ones(n))
        u = np.log(a0 + 1e-12)

    # Adam state
    m = np.zeros_like(u)
    v = np.zeros_like(u)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # scheduling parameters
    max_iters = 500_000
    eta_start, eta_end = 0.5, 0.02          # exponential decay
    beta_start, beta_end = 0.5, 15000.0    # temperature for soft‑max
    jitter_interval = 2000
    jitter_scale = 1e-3
    restart_patience = 30_000               # iterations without improvement

    best_u = u.copy()
    best_score = np.inf
    best_iter = 0
    last_improve = 0

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # ---- (a) probability vector & its convolution -----------------
        a = np.exp(u - u.max())               # stable softmax numerator
        a_sum = a.sum()
        if a_sum == 0.0:                      # safety fallback
            a = np.full_like(a, 1.0 / n)
            a_sum = 1.0
        a = a / a_sum                          # enforce Σa = 1

        conv = _conv_self_fft(a)               # length 2·n‑1
        M = conv.max()
        cur_score = 2.0 * n * M                 # because Σa = 1

        # ---- (b) keep the best solution --------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_u = u.copy()
            best_iter = it
            last_improve = it
            print(f"[n={n}] iter {it:07d}  new best O={best_score:.6f}")

        # ---- (c) schedule parameters ------------------------------------
        fraction = min(it / max_iters, 1.0)
        beta = beta_start + (beta_end - beta_start) * fraction
        eta = eta_start * (eta_end / eta_start) ** fraction   # exponential decay

        # ---- (d) weight vector over convolution entries -----------------
        if fraction < 0.4:                     # soft‑max phase
            shifted = conv - M                 # ≤ 0, zero at the maxima
            w = np.exp(beta * shifted)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.zeros_like(conv)
                w[conv.argmax()] = 1.0
            else:
                w /= w_sum
        else:                                  # active‑set (hard‑max) phase
            tol = max(1e-12, M * 1e-8)
            active = np.where(conv >= M - tol)[0]
            if active.size == 0:
                w = np.zeros_like(conv)
                w[conv.argmax()] = 1.0
            else:
                w = np.zeros_like(conv)
                w[active] = 1.0 / active.size

        # ---- (e) gradient of the (smoothed) max‑convolution -----------
        g = 2.0 * _correlation_fft(w, a)       # shape (n,)

        # ---- (f) Adam update on log‑variables ---------------------------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        u = u - eta * m_hat / (np.sqrt(v_hat) + eps)

        # ---- (g) occasional jitter --------------------------------------
        if it % jitter_interval == 0:
            u += np.random.randn(n) * jitter_scale

        # ---- (h) cheap restart if stagnant -------------------------------
        if it - last_improve > restart_patience:
            u = best_u.copy()
            u += np.random.randn(n) * (jitter_scale * 0.5)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = it
            print(f"[n={n}] restart at iter {it}, re‑initialised from best "
                  f"(O={best_score:.6f})")

    # ----- final sequence (probabilities) ---------------------------------
    best_seq = np.exp(best_u - best_u.max())
    best_seq = best_seq / best_seq.sum()
    # final bound (using the evaluator for safety)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[n={n}] final evaluation O={final_score:.6f} "
              f"(best iter {best_iter})")
    except Exception as exc:  # pragma: no cover
        print(f"[n={n}] evaluate_sequence failed: {exc}")
        final_score = best_score

    return best_seq.tolist(), float(final_score)
