"""
Optimization of non‑negative step function via Adam‑based mirror descent
with temperature annealing. We start from a Gaussian‑shaped candidate (or
an up‑sampled previous best if available), then perform exponentiated‑gradient
updates on the log‑probabilities using an Adam‑style adaptive step‑size.
A soft‑max weighting of the convolution entries is used early on,
switching to a hard‑max active‑set later. Random jitter and periodic
re‑starts from the best‑so‑far solution help avoid stagnation.
The routine respects a given time budget and returns the best sequence
found.
"""
import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()

def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv

def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute the linear correlation g_i = Σ_t w[i+t]·a[t] for i=0,…,n‑1.
    Implemented via FFT.
    """
    n = a.shape[0]
    m = w.shape[0]                # = 2·n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` bound. Returns the best sequence as a plain
    list of floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    # -----------------------------------------------------------------
    # 1) Choose length and initialise log‑probabilities.
    # Use a moderately large length; if a previous construction is
    # available we will up‑sample it and compare with a Gaussian start.
    if 'height_sequence_1' in globals():
        old = np.asarray(height_sequence_1, dtype=float)
        old_len = old.size
        # Target length: at least 8 000 and at most 15 000.
        n = max(8000, min(15000, old_len * 2))
    else:
        n = 12000

    # -----------------------------------------------------------------
    # 2) Build a set of candidate initialisations and pick the best.
    candidates = []

    # a) Gaussian‑shaped candidates (a few different widths)
    center = (n - 1) / 2.0
    sigma_vals = np.linspace(n / 8.0, n / 3.0, num=7)
    for sigma in sigma_vals:
        x = np.arange(n, dtype=float)
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
        a_sum = a.sum()
        if a_sum == 0:
            continue
        a = a / a_sum
        candidates.append(a)

    # b) Up‑sampled previous solution if present
    if 'height_sequence_1' in globals():
        old = np.asarray(height_sequence_1, dtype=float)
        old_len = old.size
        # Linear interpolation onto the new grid
        old_x = np.linspace(0, n - 1, num=old_len)
        new_x = np.arange(n, dtype=float)
        a_up = np.interp(new_x, old_x, old)
        a_up = np.maximum(a_up, 0.0)
        s = a_up.sum()
        if s > 0:
            a_up = a_up / s
            candidates.append(a_up)

    # Choose the candidate with smallest current bound (no call to evaluate_sequence)
    best_a = None
    best_score = np.inf
    for a in candidates:
        conv = _conv_self_fft(a)
        M = conv.max()
        score = 2.0 * n * M     # sum(a) == 1
        if score < best_score:

1. Choose a target length *n* (≈8 k–15 k).  Initialise a probability vector

import time

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv

    """

    Implemented via FFT.
