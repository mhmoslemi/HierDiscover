"""
Proposes a non‑negative step‑function (list of heights) that minimizes

    O(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm has two stages and runs within the given time budget.

Stage 1 – Smoothed log‑sum‑exp (Adam) on the simplex.
    •   Starts from an “arcsine‑like” distribution which already gives a
        fairly flat self‑convolution.
    •   Uses the surrogate
            fα(a) = (2·n/α)·log Σk exp(α·convk(a)) + λ·Σi ai·log ai
        whose gradient is cheap (FFT‑based) and whose smoothing parameter α
        is annealed from a small value to a large one (≈20000).  The Adam step
        is projected onto the probability simplex after each update.

Stage 2 – Fast local refinement.
    •   Keeps a full convolution array and updates it *incrementally* when a
        small amount of mass δ is moved from index i to j:
            conv ← conv + 2·δ·(shiftj(a) – shifti(a))
                    + δ²·(e2j – 2·ei+j + e2i)
        which is O(n) but vectorised in NumPy and far cheaper than a full FFT.
    •   Repeatedly performs random mass‑transfer moves, accepting improvements
        or (with a slowly cooling temperature) occasional uphill moves
        (Metropolis rule).
    •   Every few hundred iterations a sub‑gradient step for the exact
        max‑convolution entry is taken, and the convolution is occasionally
        recomputed via FFT to avoid drift.
    •   Multiple restarts (jittered copies of the current best) are used.

The best sequence found is returned as a plain Python list.
"""

def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution a * a using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


def _objective_simplex(a: np.ndarray, n: int, L: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a on the simplex."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) at a (averaged over all maximisers)."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


def _smooth_gradient_plus_entropy(
    a: np.ndarray,
    conv: np.ndarray,
    alpha: float,
    L: int,
    lam: float,
) -> np.ndarray:
    """
    Gradient of the smooth surrogate:
        (2·n/α)·log Σk exp(α·convk) + λ·Σi ai·log ai .
    """
    # soft‑max weights for the log‑sum‑exp term
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)          # for numerical stability
    w = exp_scaled / exp_scaled.sum()                 # shape (2n‑1,)

    # correlation w ⋆ a  (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n_len = a.size
    grad_corr = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]

    # entropy part λ·(log a_i + 1)
    eps = 1e-20
    grad_ent = lam * (np.log(a + eps) + 1.0)

    return grad_corr + grad_ent


def _apply_move(a: np.ndarray, conv: np.ndarray, src: int, dst: int, delta: float) -> None:
    """
    Transfer a mass `delta` from index `src` to `dst` (src != dst).
    Updates the convolution `conv` in‑place using the exact formula:
        conv ← conv + 2·δ·(shift_dst(a) – shift_src(a))
               + δ²·(e_{2·dst} – 2·e_{src+dst} + e_{2·src})
    The vector `a` is updated afterwards.
    """
    n = a.size
    # shift contributions (use a *before* the move)
    conv[dst:dst + n] += 2.0 * delta * a
    conv[src:src + n] -= 2.0 * delta * a

    # quadratic correction terms
    conv[2 * dst] += delta * delta
    conv[2 * src] += delta * delta
    conv[src + dst] -= 2.0 * delta * delta

    # finally update a
    a[src] -= delta
    a[dst] += delta

    best_seq = a.copy()
    best_score = _objective_simplex(best_seq, n, L)
    print(f"[phase‑1] start O={best_score:.6f}")

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    lr0 = 0.8                 # base learning rate
    alpha_start = 5.0
    alpha_end = 20000.0       # stronger focus on the true max
    lam = 0.005               # tiny entropy to keep the simplex interior

    max_iter_phase1 = 250_000
    it = 0
    phase1_deadline = start_time + budget_s * 0.45   # ~45 % of the budget

    while time.time() < phase1_deadline and it < max_iter_phase1:
        # exact conv for bookkeeping (used for the surrogate gradient as well)
        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[phase‑1] iter {it:,} new best O={best_score:.6f}")

        # anneal α linearly
        frac = it / max_iter_phase1
        alpha = alpha_start + (alpha_end - alpha_start) * frac

        # gradient of the smooth surrogate + entropy
        grad = _smooth_gradient_plus_entropy(a, conv, alpha, L, lam)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        lr = lr0 / math.sqrt(it + 1)          # simple 1/√t decay

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # tiny jitter to escape plateaus
        if np.random.rand() < 0.001:
            a_new += np.random.randn(n) * 1e-4

        # simplex projection
        a_new = _project_to_simplex(a_new)

        # occasional pruning of extremely tiny entries
        if it % 5000 == 0 and it > 0:
            thresh = np.percentile(a_new, 0.5)   # drop the lowest half‑percent
            a_new[a_new < thresh] = 0.0
            a_new = _project_to_simplex(a_new)

        a = a_new
        it += 1

    # --------------------------------------------------------------
    # ---------- Phase 2 : Fast stochastic local refinement ----------
    print(f"[phase‑2] entering local refinement, best O={best_score:.6f}")

    # parameters for the local search
    max_restarts = 0
    subgrad_interval = 30          # sub‑gradient step every 30 moves
    temp0 = 0.05                   # initial Metropolis temperature
    max_iter_local = 300_000       # generous upper bound per restart
    no_improve_limit = 4000        # restart if stuck

    while time.time() < deadline:
        # ----- initialise a restart (jittered best or fresh) -----
        if np.random.rand() < 0.7:
            a = best_seq.copy()
            a += np.random.randn(n) * 0.001
            a = np.clip(a, 0, None)
            a = _project_to_simplex(a)
        else:
            a = np.random.rand(n)
            a = _project_to_simplex(a)

        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        it_loc = 0
        no_improve = 0
        T = temp0

        while time.time() < deadline and it_loc < max_iter_local:
            # -------- random mass‑transfer move --------
            src = np.random.randint(n)
            if a[src] <= 1e-14:
                it_loc += 1
                continue
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1

            # move up to 12 % of the source mass
            delta = a[src] * np.random.rand() * 0.12
            _apply_move(a, conv, src, dst, delta)

            new_obj = 2.0 * n * conv.max()

            # Metropolis acceptance
            if new_obj < cur_obj or np.random.rand() < math.exp((cur_obj - new_obj) / max(T, 1e-14)):
                cur_obj = new_obj
                no_improve = 0
                if cur_obj < best_score:
                    best_score = cur_obj
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] new global best O={best_score:.6f} "
                          f"(restart={max_restarts}, iter={it_loc})")
            else:
                # revert move
                _apply_move(a, conv, dst, src, delta)   # opposite direction
                no_improve += 1

            # -------- sub‑gradient step on the current worst index --------
            if it_loc % subgrad_interval == 0:
                g = _subgradient_max_conv(a, conv)
                eta = 0.06 * (1.0 - it_loc / max_iter_local)   # linear decay
                a -= eta * g
                a = _project_to_simplex(a)
                conv = _conv_fft(a, L)          # recompute to avoid drift
                cur_obj = 2.0 * n * conv.max()
                if cur_obj < best_score:
                    best_score = cur_obj
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] sub‑grad improved O={best_score:.6f}")

            # -------- occasional smoothing (helps to escape tiny spikes) --------
            if it_loc % 5000 == 0 and it_loc > 0:
                kernel = np.array([0.125, 0.25, 0.25, 0.25, 0.125])
                a_smooth = np.convolve(a, kernel, mode="same")
                a = _project_to_simplex(0.85 * a + 0.15 * a_smooth)

            # temperature decay
            T *= 0.99985
            it_loc += 1

            # restart if no improvement for a while
            if no_improve >= no_improve_limit:
                break

        max_restarts += 1

    # --------------------------------------------------------------
    # Final sanity check with the provided evaluator
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:
        print("[final] evaluate_sequence failed:", e)

    return best_seq.tolist()
