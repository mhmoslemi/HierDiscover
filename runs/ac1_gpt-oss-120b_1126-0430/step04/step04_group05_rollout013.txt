"""
Propose a step‑function that minimises

    O(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm works on probability vectors (sum = 1) so that
O(a) = 2·n·max(conv(a,a)).  It consists of three phases:

1️⃣  Initialise a with a “flatten‑convolution’’ shape (discrete arcsine).
2️⃣  Adam optimisation on a smooth log‑sum‑exp surrogate of the max
    (the surrogate becomes the true max gradually).
3️⃣  Exact sub‑gradient descent for a short finishing pass.

All iterates are projected onto the probability simplex, the best
vector seen so far is kept and finally returned.  The whole run
automatically respects the 1000 s wall‑time limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# 1.  Simplex projection (Euclidean)
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# 2.  Full self‑convolution via FFT (length 2·n‑1)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return a*a (full convolution) using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    # keep only the first (2·n‑1) entries
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# 3.  Smooth (log‑sum‑exp) surrogate gradient
def _smooth_gradient(a: np.ndarray, conv: np.ndarray,
                     alpha: float, L: int) -> np.ndarray:
    """
    Gradient of   (2·n/α)·log Σ_k exp(α·conv_k)   w.r.t. a.
    The result is   2·corr(w,a)   with   w_k ∝ exp(α·conv_k).
    """
    # soft‑max weights over the convolution entries
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)        # stabilise
    w = exp_scaled / exp_scaled.sum()

    # correlation w ⋆ a  (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)   # length ≥ 3·n‑2
    n_len = a.size
    grad = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]
    return grad


# ----------------------------------------------------------------------
# 4.  Sub‑gradient of the true max‑convolution objective
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a
    (averaged if several diagonals attain the maximum).
    """
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
# 5.  A handy arcsine‑like initialisation (flattens the convolution)
def _init_arcsine(n: int) -> np.ndarray:
    """Discrete analogue of the arcsine density on {0,…,n‑1}."""
    i = np.arange(n, dtype=float)
    # shift by 0.5 to avoid division by zero at the ends
    denom = np.sqrt((i + 0.5) * (n - 0.5 - i))
    a = 1.0 / denom
    a = np.nan_to_num(a, nan=0.0, posinf=0.0)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **_):
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  Returns the sequence as a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin before the wall‑time limit
    deadline = start_time + budget_s - 0.2

    # ------------------------------------------------------------------
    # Problem size & initial vector
    if 'height_sequence_1' in globals():
        a = np.array(height_sequence_1, dtype=float)
        a = np.maximum(a, 0.0)
        if a.sum() == 0:
            a = np.full_like(a, 1.0)
        a = a / a.sum()
        n = a.size
        print(f"[init] Using supplied sequence, n={n}")
    else:
        # Choose a power‑of‑two length – cheap FFT and enough resolution
        n = 4096
        a = _init_arcsine(n)
        print(f"[init] Using arcsine start, n={n}")

    # FFT length – next power of two ≥ 3·n (covers conv & correlation)
    L = 1 << ((3 * n - 2).bit_length())

    # ------------------------------------------------------------------
    # Helper to evaluate the exact objective (sum = 1 ⇒ O = 2·n·max(conv))
    def _objective(a_vec: np.ndarray) -> float:
        conv = _conv_fft(a_vec, L)
        return 2.0 * n * conv.max()

    best_seq = a.copy()
    best_score = _objective(best_seq)
    print(f"[start] O={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam on the smoothed surrogate
    # --------------------------------------------------------------
    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    lr0 = 0.7               # base learning‑rate
    alpha_start = 5.0
    alpha_end = 200_000.0   # very sharp surrogate at the end

    max_iter_phase1 = 800_000          # generous upper bound; time will stop it earlier
    it = 0
    # allocate ~80 % of the budget to phase 1
    phase1_deadline = start_time + budget_s * 0.80

    while it < max_iter_phase1 and time.time() < phase1_deadline:
        # full self‑convolution and exact objective (only for bookkeeping)
        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[phase‑1] iter {it:06d}  new best O={best_score:.6f}")

        # geometric annealing of the smoothing parameter α
        frac = it / max_iter_phase1
        alpha = alpha_start * ((alpha_end / alpha_start) ** frac)

        # smooth gradient
        grad = _smooth_gradient(a, conv, alpha, L)

        # Adam bias‑corrected updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        # learning‑rate schedule: 1/√t decay
        lr = lr0 / math.sqrt(it + 1)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional tiny jitter to escape flat plateaus
        if np.random.rand() < 0.001:
            a_new += np.random.randn(n) * 1e-5

        # project back onto the simplex
        a_new = _project_to_simplex(a_new)

        # prune a negligible tail every few thousand steps – helps sparsity
        if it % 5000 == 0 and it > 0:
            thresh = np.percentile(a_new, 0.2)   # drop the lowest 0.2 %
            a_new[a_new < thresh] = 0.0
            a_new = _project_to_simplex(a_new)

        a = a_new
        it += 1

    # ------------------------------------------------------------------
    # Phase 2 – Exact sub‑gradient descent (pure max‑convolution)
    # --------------------------------------------------------------
    subgrad_iters = 150_000
    eta0 = 0.25           # initial step‑size
    it2 = 0
    phase2_deadline = start_time + budget_s * 0.95
    while it2 < subgrad_iters and time.time() < phase2_deadline:
        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[sub‑grad] iter {it2:06d}  new best O={best_score:.6f}")

        # sub‑gradient of the exact max‑convolution
        g = _subgradient_max_conv(a, conv)

        # geometric decay of the step‑size
        eta = eta0 * (0.9997 ** it2)
        a = a - eta * g
        a = _project_to_simplex(a)

        it2 += 1

    # ------------------------------------------------------------------
    # Phase 3 – final sanity check (optional local jitter)
    # --------------------------------------------------------------
    # tiny random perturbation around the best point – sometimes uncovers
    # a marginally better neighbour.
    a = best_seq.copy()
    jitter = np.random.randn(n) * 5e-5
    a = np.clip(a + jitter, 0, None)
    a = _project_to_simplex(a)
    conv = _conv_fft(a, L)
    cur_obj = 2.0 * n * conv.max()
    if cur_obj < best_score:
        best_score = cur_obj
        best_seq = a.copy()
        print(f"[final jitter] improved O={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final verification through the provided evaluator
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence failed:", exc)

    return best_seq.tolist()
