def _adam_optimize(init_a, time_limit, seed):
    start = time.time()
    rng = np.random.default_rng(seed)
    n = init_a.size
    # init logits
    epsilon = 1e-16
    x = np.log(np.maximum(init_a, epsilon))
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1=0.9; beta2=0.999; eps=1e-12

    best_a = init_a.copy()
    best_obj = _objective(best_a)

    it = 0
    max_iters = 300_000

    # schedule parameters
    lr_start = 0.2
    lr_end = 0.001
    # no smoothing; use exact max
    while time.time() - start < time_limit and it < max_iters:
        it += 1
        # compute a from softmax logits
        a = np.exp(x)
        a = a / a.sum()
        conv = _conv_self_fft(a)
        i_max = int(np.argmax(conv))
        # compute objective
        obj = 2.0 * n * conv[i_max] / (a.sum()**2)  # a.sum ~=1
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            # print progress every maybe 5000 it
            if it % 5000 == 0:
                print(f"[n={n} it={it}] new best O={best_obj:.6f}")

        # compute gradient
        grad = _grad_max(a, i_max)  # shape n
        # natural gradient for simplex:
        dot = np.dot(a, grad)
        g_x = a * (grad - dot)

        # learning rate schedule: exponential decay
        t = min(it / max_iters, 1.0)
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # jitter occasional
        if it % 5000 == 0:
            jitter = rng.normal(scale=0.01, size=n)
            x += jitter

        # reinitialize occasional (like restart)
        if it % 100_000 == 0:
            x = np.log(np.maximum(best_a, epsilon))
            jitter = rng.normal(scale=0.005, size=n)
            x += jitter
            m.fill(0.0)
            v.fill(0.0)
            print(f"[n={n}] restart at iter {it}")

    # compress small entries
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = _objective(best_a)
        print(f"[compress] reduced n {n} -> {best_a.size}, O={best_obj:.6f}")

    return best_a.tolist(), best_obj

"""
Propose a non‑negative step‑function (a list of heights) that minimises

    O(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm combines several ideas:

* A variety of deterministic initialisations (random Dirichlet, uniform,
  Hann/Blackman windows, DPSS windows and the previous best sequence if
  available) are evaluated cheaply and the most promising one is selected.
* An Adam optimiser works directly on the log‑it (soft‑max) parametrisation
  of the simplex, using the *exact* sub‑gradient of the max‑convolution
  (no smooth approximation).  The natural‑gradient step ensures the simplex
  constraints are respected.
* Occasional jitter, restarts from the best‑so‑far solution and a simple
  pruning of negligible entries keep the search diverse and fast.
* After the main optimisation stage a lightweight random local search
  (mass transfers between coordinates) refines the solution further.
* The whole routine is time‑aware: a 1000 s budget is split among a few
  support sizes (500 – 8000) and the remaining time is used for final
  polishing.  Intermediate progress and the current best objective value are
  printed.

The returned object is a plain ``list[float]`` that can be fed into
``evaluate_sequence``.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# -------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# -------------------------------------------------------------
def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (fast for large n)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = rfft(a, n=L)
    conv = irfft(af * af, n=L)[: 2 * n - 1]
    return conv


# -------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)**2)."""
    n = a.size
    conv = _conv_self_fft(a)
    return 2.0 * n * conv.max() / (a.sum() ** 2)


# -------------------------------------------------------------
def _grad_max(a: np.ndarray, i_max: int) -> np.ndarray:
    """
    Sub‑gradient of the maximal self‑convolution term.
    For i_max = argmax(conv), the gradient w.r.t. a is
        g[k] = 2·a[i_max‑k]   if 0 ≤ i_max‑k < n, else 0.
    """
    n = a.shape[0]
    idx = i_max - np.arange(n)
    mask = (idx >= 0) & (idx < n)
    g = np.zeros_like(a)
    g[mask] = 2.0 * a[idx[mask]]
    return g


# -------------------------------------------------------------
def _adam_optimize(init_a: np.ndarray, time_limit: float, seed: int):
    """
    Adam optimisation on log‑it space using the exact max‑convolution gradient.
    Returns the best sequence (as a list) and its exact objective value.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    n = init_a.size
    eps = 1e-16
    x = np.log(np.maximum(init_a, eps))            # logits
    m = np.zeros_like(x)                           # first moment
    v = np.zeros_like(x)                           # second moment
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-12

    # learning‑rate schedule (exponential decay)
    lr_start = 0.2
    lr_end = 0.001
    max_iters = 300_000

    best_a = np.exp(x)
    best_a /= best_a.sum()
    best_obj = _objective(best_a)

    it = 0
    while time.time() - start < time_limit and it < max_iters:
        it += 1

        # current simplex point
        a = np.exp(x)
        a /= a.sum()

        # convolution and max index
        conv = _conv_self_fft(a)
        i_max = int(np.argmax(conv))

        # exact objective
        obj = 2.0 * n * conv[i_max] / (a.sum() ** 2)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            if it % 2000 == 0:
                print(f"[n={n:5d} it={it:06d}] new best O={best_obj:.6f}")

        # sub‑gradient of the max term
        grad = _grad_max(a, i_max)

        # natural‑gradient on the simplex (mirror descent in log‑it space)
        dot = np.dot(a, grad)
        g_x = a * (grad - dot)

        # learning‑rate schedule (exponential decay)
        t = min(it / max_iters, 1.0)
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t)

        # Adam update in log‑it space
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional jitter to escape plateaus
        if it % 5000 == 0:
            jitter = rng.normal(scale=0.01, size=n)
            x += jitter

        # restart from the best solution every so often
        if it % 100_000 == 0:
            x = np.log(np.maximum(best_a, eps))
            x += rng.normal(scale=0.005, size=n)
            m.fill(0.0)
            v.fill(0.0)
            print(f"[n={n}] restart at iter {it}")

    # prune negligible entries
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a /= best_a.sum()
        best_obj = _objective(best_a)
        print(f"[compress] reduced length {n} → {best_a.size}, O={best_obj:.6f}")

    return best_a.tolist(), best_obj

def _random_local_search(init_seq: list, time_limit: float, seed: int):
    """
    Very light‑weight hill‑climbing: random mass transfers between coordinates.
    Returns a (possibly) improved sequence and its objective value.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    a = np.asarray(init_seq, dtype=float)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-12:
        a = np.ones_like(a)
    a /= a.sum()

    n = a.size
    best_a = a.copy()
    best_obj = _objective(best_a)

    max_frac = 0.2          # up‑to‑20 % of source mass may be moved
    it = 0
    while time.time() - start < time_limit:
        it += 1
        # choose a source with non‑negligible mass
        src_candidates = np.where(best_a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)
        # choose an arbitrary target (may be the same index – we skip that)
        j = rng.integers(0, n)
        if i == j:
            continue
        # transfer a random fraction of the source mass
        frac = rng.random() * max_frac
        delta = best_a[i] * frac
        if delta <= 0.0:
            continue
        a_new = best_a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        obj = _objective(a_new)
        if obj < best_obj:
            best_a = a_new
            best_obj = obj
            if it % 2000 == 0:
                print(f"[local] iter {it} improved O={best_obj:.6f}")

    # final pruning of tiny entries
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a /= best_a.sum()
        best_obj = _objective(best_a)
        print(f"[compress‑local] reduced length {n} → {best_a.size}, O={best_obj:.6f}")

    return best_a.tolist(), best_obj

def _make_initial_candidates(n: int, rng: np.random.Generator):
    """
    Produce a small diverse set of deterministic & random initial vectors.
    All candidates are returned as NumPy arrays (not yet normalised).
    """
    init = []

    # random Dirichlet‑like point
    init.append(rng.random(n))

    # uniform distribution
    init.append(np.ones(n))

    # Hann window
    if n > 1:
        i = np.arange(n)
        hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        init.append(hann)

        # Blackman window (non‑negative)
        black = 0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) \
                + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
        init.append(black)

    # exponential decay (roughly triangular)
    exp_decay = np.exp(-np.arange(n) / (n / 10.0))
    init.append(exp_decay)

    # DPSS window (if SciPy is available)
    try:
        from scipy.signal.windows import dpss
        dps = dpss(n, NW=2.5, sym=True)[0]          # first DPSS eigenvector
        init.append(np.abs(dps))                    # enforce non‑negativity
    except Exception:
        pass

    return init
