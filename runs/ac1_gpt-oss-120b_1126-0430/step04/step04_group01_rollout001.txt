"""
Propose a non‑negative step‑function sequence that (approximately) minimises

    O(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

Algorithm
---------

* **Initialisation** – start from the supplied `height_sequence_1` (if any) or a
  uniform simplex vector of length ≈5000.  All vectors are projected onto the
  probability simplex ∑a_i = 1, so the objective reduces to   O = 2·n·max(conv).

* **Phase 1 – Adam on a smooth log‑sum‑exp surrogate**  
  minimise  (2·n/α)·log∑_k exp(α·conv_k)  + λ·∑ a_i·log a_i.  
  The surrogate concentrates on the true maximum when α grows, while the
  tiny entropy term (λ) keeps the search away from degenerate spikes.
  Adam with bias‐correction and a slowly decaying learning‑rate is used.
  After each step the vector is projected onto the simplex and a tiny
  pruning of negligible entries is applied.

* **Phase 2 – Targeted stochastic refinement**  
  Repeatedly move a small amount of probability mass **from** an index that
  heavily contributes to the current maximal convolution entry **to** an index
  that contributes little.  The move is accepted if it improves the exact
  objective, otherwise with a Metropolis probability governed by a slowly
  decreasing temperature.  Every few iterations a sub‑gradient step on the
  exact max‑convolution is performed, and occasional smoothing is applied.

* **Final check** – the best sequence found is evaluated with the provided
  `evaluate_sequence` routine and returned as a plain Python list.

The whole routine respects a 1000 s wall‑clock budget (a small safety margin
is kept).  All helper functions are defined at module level, use only NumPy,
and avoid any filesystem or network access.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex {x ≥ 0, sum(x)=1}
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = cssv[-1] / n
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution via FFT (full length 2·n‑1)
def conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution a⨂a using FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Exact objective for a vector already on the simplex (∑a=1)
def objective_simplex(a: np.ndarray, n: int, L: int) -> float:
    conv = conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) w.r.t. a (averaged over all maximisers)
def subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
# Gradient of the smooth surrogate (log‑sum‑exp) plus entropy term
def smooth_gradient_plus_entropy(a: np.ndarray,
                                 conv: np.ndarray,
                                 alpha: float,
                                 L: int,
                                 lam: float) -> np.ndarray:
    # soft‑max weights
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)      # for numerical stability
    w = exp_scaled / exp_scaled.sum()            # shape (2·n‑1,)

    # correlation w ⋆ a   (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n_len = a.size
    grad_corr = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]

    # entropy part
    eps = 1e-20
    grad_ent = lam * (np.log(a + eps) + 1.0)

    return grad_corr + grad_ent


# ----------------------------------------------------------------------
# Targeted mass‑transfer move: move mass from a high‑contributor to a low‑contributor
def targeted_move(a: np.ndarray,
                  conv: np.ndarray,
                  k_max: int,
                  step_factor: float = 0.1) -> np.ndarray:
    """Return a new vector after moving a small amount of mass."""
    n = a.size
    # compute contributions to the maximal entry
    contrib = np.zeros_like(a)
    i_start = max(0, k_max - (n - 1))
    i_end = min(n - 1, k_max)
    idx = np.arange(i_start, i_end + 1, dtype=int)
    partner = k_max - idx
    contrib[idx] = a[idx] * a[partner]

    # source: index with largest contribution (if tie pick randomly)
    src_candidates = np.flatnonzero(contrib == contrib.max())
    if src_candidates.size == 0:
        src = np.random.randint(n)
    else:
        src = np.random.choice(src_candidates)

    # target: pick among indices with low contribution (below 25‑th percentile)
    thresh = np.percentile(contrib, 25)
    low_idx = np.flatnonzero(contrib <= thresh)
    # ensure we have at least one candidate different from src
    if low_idx.size == 0 or (low_idx.size == 1 and low_idx[0] == src):
        tgt = (src + np.random.randint(1, n)) % n
    else:
        # choose a target distinct from src
        possible = low_idx[low_idx != src]
        if possible.size == 0:
            tgt = (src + np.random.randint(1, n)) % n
        else:
            tgt = np.random.choice(possible)

    # amount to move (fraction of source mass, capped)
    max_delta = a[src]
    delta = step_factor * max_delta * np.random.rand()
    if delta <= 0.0:
        return a  # nothing to do

    a_new = a.copy()
    a_new[src] -= delta
    a_new[tgt] += delta

    # projection is cheap because sum is unchanged, but we keep it for stability
    a_new = project_to_simplex(a_new)
    return a_new


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    evaluate_sequence(...).  Returns the best sequence found as a plain list.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    # keep a small safety cushion (the caller penalises >budget_s)
    deadline = start_time + budget_s - 0.3

    # --------------------------------------------------------------
    # Initialise a – use supplied sequence if present, otherwise uniform
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=float)
        if base.size == 0:
            n = 5000
            a = np.full(n, 1.0 / n)
        else:
            a = base.copy()
            a = np.maximum(a, 0.0)
            a = a / a.sum()                     # simplex
            n = a.size
            print(f"[init] start from provided sequence, n={n}")
    else:
        n = 5000
        a = np.full(n, 1.0 / n)
        print(f"[init] start from uniform distribution, n={n}")

    # FFT length – next power of two >= 2·n
    L = 1 << ((2 * n - 1).bit_length())

    # --------------------------------------------------------------
    # Phase‑1 : Adam on the smooth log‑sum‑exp surrogate
    best_seq = a.copy()
    best_score = objective_simplex(best_seq, n, L)
    print(f"[phase‑1] start O={best_score:.6f}")

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    lr0 = 0.8                     # slightly larger base LR
    alpha_start = 5.0
    alpha_end = 20000.0           # push the surrogate closer to true max
    lam = 0.005                    # smaller entropy -> crisper shapes
    max_iter_phase1 = 350_000     # generous, exit on time
    it = 0
    phase1_deadline = start_time + budget_s * 0.50  # half the budget

    while time.time() < phase1_deadline and it < max_iter_phase1:
        # exact objective (for bookkeeping)
        conv = conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[phase‑1] iter {it:,} new best O={best_score:.6f}")

        # anneal smoothing parameter α
        frac = it / max_iter_phase1
        alpha = alpha_start + (alpha_end - alpha_start) * frac

        # compute smooth gradient
        grad = smooth_gradient_plus_entropy(a, conv, alpha, L, lam)

        # Adam update with bias correction
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        # learning‑rate schedule: 1/√t decay
        lr = lr0 / math.sqrt(it + 1)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape plateaus
        if np.random.rand() < 0.001:
            a_new += np.random.randn(n) * 1e-4

        # simplex projection
        a_new = project_to_simplex(a_new)

        # prune extremely tiny entries every few thousand iterations
        if it % 5000 == 0 and it > 0:
            thresh = np.percentile(a_new, 0.5)   # drop lowest half‑percent
            a_new[a_new < thresh] = 0.0
            a_new = project_to_simplex(a_new)

        a = a_new
        it += 1

    # --------------------------------------------------------------
    # Phase‑2 : Targeted stochastic refinement + sub‑gradient steps
    print(f"[phase‑2] entering refinement, best O={best_score:.6f}")

    # Parameters for refinement
    max_iter_local = 150_000
    subgrad_interval = 6
    temperature0 = 0.02
    no_improve_limit = 3000
    step_factor = 0.12           # magnitude of a targeted mass move
    restart_counter = 0

    while time.time() < deadline:
        # Create a restart – either jittered best or fresh random
        if np.random.rand() < 0.6:
            a = best_seq.copy()
            a += np.random.randn(n) * 0.0013
            a = np.clip(a, 0, None)
            a = project_to_simplex(a)
        else:
            a = np.random.rand(n)
            a = project_to_simplex(a)

        conv = conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        it_loc = 0
        no_improve = 0
        T = temperature0

        while time.time() < deadline and it_loc < max_iter_local:
            # ---- targetted mass‑transfer move ----
            k_max = int(np.argmax(conv))
            a_new = targeted_move(a, conv, k_max, step_factor=step_factor)

            conv_new = conv_fft(a_new, L)
            new_obj = 2.0 * n * conv_new.max()

            # Metropolis acceptance
            if new_obj < cur_obj or np.random.rand() < math.exp((cur_obj - new_obj) / max(T, 1e-14)):
                a, conv, cur_obj = a_new, conv_new, new_obj
                no_improve = 0
                if cur_obj < best_score:
                    best_score = cur_obj
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] new global best O={best_score:.6f} (restart={restart_counter}, it={it_loc})")
            else:
                no_improve += 1

            # ---- occasional sub‑gradient step (exact max) ----
            if it_loc % subgrad_interval == 0:
                g = subgradient_max_conv(a, conv)          # shape (n,)
                eta = 0.05 * (1.0 - it_loc / max_iter_local)   # decaying step size
                a = a - eta * g
                a = project_to_simplex(a)
                conv = conv_fft(a, L)
                cur_obj = 2.0 * n * conv.max()
                if cur_obj < best_score:
                    best_score = cur_obj
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] sub‑grad improved O={best_score:.6f}")

            # ---- occasional smoothing (low‑cost) ----
            if it_loc % 4000 == 0 and it_loc > 0:
                kernel = np.array([0.125, 0.25, 0.25, 0.25, 0.125])
                a_smooth = np.convolve(a, kernel, mode="same")
                a = project_to_simplex(0.85 * a + 0.15 * a_smooth)
                conv = conv_fft(a, L)
                cur_obj = 2.0 * n * conv.max()
                if cur_obj < best_score:
                    best_score = cur_obj
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] smoothing improved O={best_score:.6f}")

            # temperature decay
            T *= 0.9998
            it_loc += 1

            # restart if stagnation detected
            if no_improve >= no_improve_limit:
                break

        restart_counter += 1
        if time.time() >= deadline:
            break

    # --------------------------------------------------------------
    # Final sanity check using the provided evaluator (if available)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:
        print("[final] evaluate_sequence failed:", e)

    # Return the best sequence as a plain list of Python floats
    return best_seq.tolist()
