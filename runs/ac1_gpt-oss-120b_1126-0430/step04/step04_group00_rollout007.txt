Now we need to implement projection onto simplex. Implementation algorithm:

def _project_simplex(v):
    """Project v onto simplex {x|x>=0, sum x = 1}."""
    # using algorithm from Duchi et al.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w

#!/usr/bin/env python
"""
Hybrid hard‑max Adam optimiser for the convolution‑max objective.

The algorithm
    * builds a few sensible initial probability vectors (global solution,
      power‑law, edge‑heavy, uniform);
    * picks the best among them (cheap score = 2·n·max(a*a));
    * runs an Adam optimiser on the soft‑max logits of the probabilities.
      The gradient is the exact *hard* gradient of the max‑convolution
      (∂/∂a_i J = 4 n a_{k_max‑i}) which drives mass away from the
      index k_max where the convolution attains its maximum.
    * a back‑tracking line‑search guarantees monotone decrease of the cheap
      score; occasional random jitter helps escape plateaux.
    * the support is enlarged progressively (interpolation) and the whole
      routine respects the 1000 s budget.

The routine returns the best sequence found (as a plain list of floats).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable softmax; output sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it representation that reproduces `a` after softmax."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """Smallest power‑of‑two ≥ 3·n‑2 (sufficient for linear correlation)."""
    return 1 << ((3 * n - 2).bit_length())


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    total = new_a.sum()
    if total == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= total
    return new_a


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """Normalized a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _edge_heavy_distribution(n: int, p: float) -> np.ndarray:
    """
    Concentrates mass towards the two ends.
    a_i ∝ (|i−(n−1)/2| + 1)^{p}.
    """
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    dist = np.abs(idx - centre)
    a = (dist + 1.0) ** p
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _conv_max(a: np.ndarray) -> float:
    """Return max entry of the linear self‑convolution a*a."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _hard_grad_max_conv(a: np.ndarray) -> tuple[float, int, np.ndarray]:
    """
    Compute max convolution value, the index where it occurs,
    and the exact gradient ∂J/∂a (J = 2·n·max_conv).
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]

    max_val = float(conv.max())
    k_max = int(conv.argmax())

    # Gradient of J = 2·n·max_conv  →  ∂J/∂a_i = 4·n·a_{k_max‑i} (if defined)
    grad = np.zeros_like(a)
    i_start = max(0, k_max - (n - 1))
    i_end = min(n - 1, k_max)
    idxs = np.arange(i_start, i_end + 1, dtype=np.int64)
    js = k_max - idxs
    grad[idxs] = 4.0 * n * a[js]
    return max_val, k_max, grad

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights (a step function) that minimises
    the provided evaluation function.  The routine respects a ~1000 s
    time budget and uses a hard‑max Adam optimiser with several
    initialisations and progressive support enlargement.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0            # safety margin

    # ------------------------------------------------------------------
    # 0. Determine problem size and an initial distribution.
    # ------------------------------------------------------------------
    try:
        # Global variable from the previous run (if present)
        init_global = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        n = init_global.shape[0]
        print(f"Using global sequence, length = {n}")
        candidate0 = np.maximum(init_global, 0.0)
        if candidate0.sum() == 0.0:
            candidate0[:] = 1.0 / n
        else:
            candidate0 /= candidate0.sum()
        init_candidates = [("global", candidate0)]
    except Exception:
        n = 20000                                 # default support size
        init_candidates = []
        print(f"No global sequence found, default length = {n}")

    # Add a few hand‑crafted starts
    for p in (0.58, 0.60, 0.62, 0.65):
        init_candidates.append((f"pow_{p:.2f}", _power_law_distribution(n, p)))
    for p in (0.5, 1.0, 2.0):
        init_candidates.append((f"edge_{p:.1f}", _edge_heavy_distribution(n, p)))
    init_candidates.append(("uniform", np.full(n, 1.0 / n)))

    # ------------------------------------------------------------------
    # 1. Pick the best amongst the cheap initial candidates.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    for name, a in init_candidates:
        score = 2.0 * n * _conv_max(a)
        print(f"Init {name:10s} → cheap score {score:.6f}")
        if score < best_score:
            best_score = score
            best_seq = a.copy()
    if best_seq is None:
        # fallback – uniform
        best_seq = np.full(n, 1.0 / n)
        best_score = 2.0 * n * _conv_max(best_seq)

    print(f"Chosen initial cheap score = {best_score:.6f}")

    # Slight jitter to avoid exact zero gradients
    z = _logits_from_distribution(best_seq)
    z += 0.01 * np.random.randn(n)

    # ------------------------------------------------------------------
    # 2. Adam optimiser on the logits (softmax parametrisation).
    # ------------------------------------------------------------------
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_alpha = 0.20                     # base learning‑rate for logits

    it = 0
    last_improve_iter = 0
    best_seq_record = best_seq.copy()
    best_score_record = best_score

    # Pre‑compute FFT length for the current support size
    L = _fft_len_for_corr(n)

    # ------------------------------------------------------------------
    # 3. Main optimisation loop.
    # ------------------------------------------------------------------
    while time.time() < deadline and it < 2_000_000:
        it += 1

        # --- current point on the simplex ---
        a = _softmax(z)

        # --- hard max‑convolution and its gradient ---
        max_val, k_max, grad_a = _hard_grad_max_conv(a)
        cheap_score = 2.0 * n * max_val

        # --- keep the best ever seen ---
        if cheap_score < best_score_record:
            best_score_record = cheap_score
            best_seq_record = a.copy()
            last_improve_iter = it
            if it % 2000 == 0:
                print(f"[Iter {it:07d}] NEW BEST cheap = {cheap_score:.6f}")

        # --- gradient w.r.t. logits (mirror descent on simplex) ---
        mu = np.dot(grad_a, a)
        grad_z = a * (grad_a - mu)

        # --- Adam moment updates (bias‑corrected) ---
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # --- step‑size schedule (inverse‑sqrt decay) ---
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step_dir = - m_hat / (np.sqrt(v_hat) + eps_adam)
        trial_alpha = alpha_t

        # --- back‑tracking line search (max 4 halvings) ---
        accepted = False
        for _ in range(4):
            z_new = z + trial_alpha * step_dir
            z_new = np.clip(z_new, -50.0, 50.0)           # keep exp(z) numerically safe
            a_new = _softmax(z_new)

            # cheap evaluation of the trial point
            a_fft = np.fft.rfft(a_new, L)
            conv_new = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
            new_score = 2.0 * n * float(conv_new.max())

            if new_score <= cheap_score:
                z = z_new
                accepted = True
                break
            trial_alpha *= 0.5

        if not accepted:
            # No improvement – take a modest step anyway
            z += alpha_t * step_dir
            z = np.clip(z, -50.0, 50.0)

        # --- occasional random perturbations to escape plateaux ---
        if it % 3000 == 0 and np.random.rand() < 0.15:
            z += 0.02 * np.random.randn(n)
            m *= 0.6
            v *= 0.6

        # ------------------------------------------------------------------
        # Optional support enlargement (if time permits).
        # ------------------------------------------------------------------
        if (it % 15000 == 0) and (n < 30000) and (time.time() + 30 < deadline):
            new_n = min(30000, int(n * 1.4) + 1)
            print(f"[Iter {it:07d}] Expanding support: {n} → {new_n}")
            a_interp = _interpolate_distribution(best_seq_record, new_n)

            # Reset optimiser for the new problem size
            z = _logits_from_distribution(a_interp)
            m = np.zeros_like(z)
            v = np.zeros_like(z)
            n = new_n
            L = _fft_len_for_corr(n)
            best_seq_record = a_interp.copy()
            best_score_record = 2.0 * n * _conv_max(a_interp)
            it = 0
            last_improve_iter = 0
            continue

        # ------------------------------------------------------------------
        # Early termination if stagnation persists at maximum size.
        # ------------------------------------------------------------------
        if n >= 30000 and (it - last_improve_iter) > 500_000:
            print(f"[Iter {it:07d}] Stagnation detected – terminating early.")
            break

    # ----------------------------------------------------------------------
    # 4. Final exact evaluation (once, at the end).
    # ----------------------------------------------------------------------
    final_seq = best_seq_record.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:   # defensive – should never happen
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # ----------------------------------------------------------------------
    # 5. Return the best sequence found.
    # ----------------------------------------------------------------------
    return final_seq
