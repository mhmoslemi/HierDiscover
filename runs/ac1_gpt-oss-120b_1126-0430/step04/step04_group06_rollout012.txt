#!/usr/bin/env python
"""
A mirror‑descent optimiser for the step‑function bound.
The algorithm works on the probability simplex via logits.
At each iteration it:
  * computes the full self‑convolution `c = a*a` (FFT, O(n log n));
  * selects the `K` largest entries of `c`;
  * builds a weight vector `w` from a soft‑max on those entries;
  * obtains the (weighted) sub‑gradient `∇a  Σ w_k (a*a)_k`
    efficiently with a second FFT;
  * maps the gradient to log‑it space and applies Adam.
The temperature `θ` is annealed from a small value to a large one,
so the optimisation first follows a smooth landscape and later
focuses on the true maximum.  The routine runs for up to the
provided time budget (≈1000 s) and returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable softmax; returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Logits that produce `a` after softmax."""
    return np.log(a + eps)


def _next_pow_two(x: int) -> int:
    """Smallest power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution using a pre‑chosen FFT length."""
    fa = np.fft.rfft(a, fft_len)
    fb = np.fft.rfft(b, fft_len)
    conv = np.fft.irfft(fa * fb, fft_len)
    # slice to exact length
    return conv[: a.shape[0] + b.shape[0] - 1]


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full self‑convolution a*a."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _weighted_subgradient_fft(a: np.ndarray,
                              conv: np.ndarray,
                              K: int,
                              theta: float,
                              fft_len: int) -> np.ndarray:
    """
    Compute a weighted sub‑gradient of `max(conv)` using the top K entries.
    The gradient is obtained with a second FFT for efficiency.

    Parameters
    ----------
    a : np.ndarray
        Current probability vector (sum = 1).
    conv : np.ndarray
        Full self‑convolution a*a (length 2·n‑1).
    K : int
        Number of largest convolution entries to use.
    theta : float
        Temperature for the soft‑max weights (larger ≈ sharper).
    fft_len : int
        FFT length that accommodates a (length n) and w (length 2·n‑1).

    Returns
    -------
    g : np.ndarray
        Sub‑gradient w.r.t. `a` (same shape as `a`).
    """
    n = a.shape[0]
    # ensure the absolute maximum participates
    max_idx = int(np.argmax(conv))
    # pick K largest (unsorted) indices
    idx = np.argpartition(conv, -K)[-K:]
    if max_idx not in idx:
        idx = np.append(idx, max_idx)

    # values corresponding to the selected indices
    vals = conv[idx]
    # soft‑max weighting (shifted by the global maximum for stability)
    max_val = conv[max_idx]
    exps = np.exp(theta * (vals - max_val))
    weights = exps / exps.sum()

    # build weight vector w (same length as conv)
    w = np.zeros_like(conv)
    w[idx] = weights

    # reverse w to obtain the kernel required for the gradient
    rev_w = w[::-1]

    # convolution a * rev_w gives the weighted gradient (up to the factor 2)
    fa = np.fft.rfft(a, fft_len)
    fw = np.fft.rfft(rev_w, fft_len)
    conv2 = np.fft.irfft(fa * fw, fft_len)
    # conv2 length = n + (2n‑1) – 1 = 3n‑2, we need entries (2n‑2‑p)
    slice_start = (2 * n - 2)
    g = 2.0 * conv2[slice_start - np.arange(n)]
    return g


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    s = new_a.sum()
    if s == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= s
    return new_a


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """Normalised power‑law a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a

# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Optimise a non‑negative step‑function under the bound described in
    ``evaluate_sequence``.  The routine respects the given time budget
    (≈1000 s) and returns the best sequence discovered.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0  # safety margin for the final call

    # ------------------------------
    # 0. Choose a working support size.
    # ------------------------------
    init_global = globals().get("height_sequence_1", None)
    if init_global is not None:
        init_global = np.array(init_global, dtype=np.float64)
        n0 = max(8000, init_global.shape[0])
        print(f"[Init] Using supplied global sequence, length = {n0}")
        a0 = np.maximum(init_global, 0.0)
        s = a0.sum()
        if s == 0.0:
            a0 = np.full_like(a0, 1.0 / a0.shape[0])
        else:
            a0 /= s
    else:
        n0 = 15000
        print(f"[Init] No global sequence – starting with length = {n0}")
        a0 = _power_law_distribution(n0, p=0.6)

    n_max = 30000
    n = n0

    # ---------------------------------------------------------------
    # 1. Initialise logits and Adam state.
    # ---------------------------------------------------------------
    eps = 1e-16
    z = _logits_from_distribution(a0, eps)

    m = np.zeros_like(z)  # 1st moment
    v = np.zeros_like(z)  # 2nd moment
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    base_alpha = 0.5  # base learning‑rate

    # ---------------------------------------------------------------
    # 2. Optimisation loop.
    # ---------------------------------------------------------------
    best_seq = a0.copy()
    best_score = float("inf")
    it = 0
    last_improve = 0
    report_every = 2000

    # hyper‑parameters for the smoothed sub‑gradient
    K_start = 30
    K_end = 300
    theta_start = 0.3
    theta_end = 15.0
    # linear schedules (per iteration)
    max_iters = 1_200_000  # just a safe upper bound
    K_slope = (K_end - K_start) / max_iters
    theta_slope = (theta_end - theta_start) / max_iters

    while time.time() < deadline and it < max_iters:
        it += 1

        # probability vector from logits
        a = _softmax(z)

        # FFT length that fits both a*a (2n‑1) and a*rev_w (3n‑2)
        fft_len = _next_pow_two(3 * n - 2)

        # self‑convolution (full)
        conv = _conv_self_fft(a, fft_len)            # length 2n‑1

        maxb = float(conv.max())
        cheap_score = 2.0 * n * maxb

        # record the best (exact) score is cheap_score because Σa = 1
        if cheap_score < best_score:
            best_score = cheap_score
            best_seq = a.copy()
            last_improve = it
            if it % report_every == 0:
                print(f"[Iter {it:07d}] NEW BEST cheap = {cheap_score:.6f}")

        # -----------------------------------------------------------
        # Smoothed sub‑gradient (full FFT version)
        # -----------------------------------------------------------
        K = int(K_start + K_slope * it)
        K = min(max(K, K_start), K_end)

        theta = theta_start + theta_slope * it
        # gradient w.r.t a
        g_a = _weighted_subgradient_fft(a, conv, K, theta, fft_len)

        # Convert to gradient w.r.t logits (mirror descent on simplex)
        mu = float(np.dot(g_a, a))
        grad_z = a * (g_a - mu)

        # Adam moment updates (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # direction (negative Adam step)
        step_dir = - m_hat / (np.sqrt(v_hat) + eps_adam)

        # learning‑rate schedule (inverse‑sqrt decay)
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)

        # update logits
        z = z + alpha_t * step_dir
        np.clip(z, -80.0, 80.0, out=z)

        # -----------------------------------------------------------
        # Occasional random perturbations
        # -----------------------------------------------------------
        if it % 5000 == 0 and np.random.rand() < 0.25:
            perturb = 0.02 * np.random.randn(n)
            z += perturb
            m *= 0.5
            v *= 0.5

        # -----------------------------------------------------------
        # Progressive enlargement of the support (if time permits)
        # -----------------------------------------------------------
        if (it % 25000 == 0) and (n < n_max) and (time.time() + 30.0 < deadline):
            new_n = min(n_max, n + 5000)
            print(f"[Iter {it:07d}] Expanding support: {n} → {new_n}")

            # Interpolate the best distribution obtained so far
            a = _interpolate_distribution(best_seq, new_n)

            # Re‑initialise logits and Adam state for the new size
            z = _logits_from_distribution(a, eps)
            m = np.zeros_like(z)
            v = np.zeros_like(z)

            n = new_n
            it = 0
            last_improve = 0
            # recompute fft_len with new n and continue
            continue

        # -----------------------------------------------------------
        # Early stop if stagnation after reaching max size
        # -----------------------------------------------------------
        if n >= n_max and it - last_improve > 300_000:
            print(f"[Iter {it:07d}] No improvement for {it-last_improve} iters – early stop.")
            break

    # ---------------------------------------------------------------
    # 3. Final evaluation (platform will also evaluate)
    # ---------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[Result] Final exact score = {final_score:.6f}")
    except Exception as exc:
        final_score = float("inf")
        print(f"[Result] Evaluation error: {exc}")

    # ---------------------------------------------------------------
    # 4. Return the best sequence found.
    # ---------------------------------------------------------------
    return final_seq

        mu = float(np.dot(g_a, a))
        grad_z = a * (g_a - mu)

        # Adam moment updates (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # direction (negative Adam step)
        step_dir = - m_hat / (np.sqrt(v_hat) + eps_adam)

        # simple inverse‑sqrt schedule
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)

        # perform the update in log‑it space
        z = z + alpha_t * step_dir
        np.clip(z, -80.0, 80.0, out=z)

        # -----------------------------------------------------------
        # Occasional random perturbations (helps escape plateaus)
        # -----------------------------------------------------------
        if it % 5000 == 0 and np.random.rand() < 0.25:
            perturb = 0.02 * np.random.randn(n)
            z += perturb
            m *= 0.5
            v *= 0.5

        # -----------------------------------------------------------
        # Progressive enlargement of the support (if time permits)
        # -----------------------------------------------------------
        if (it % 25000 == 0) and (n < n_max) and (time.time() + 30.0 < deadline):
            new_n = min(n_max, n + 5000)
            print(f"[Iter {it:07d}] Expanding support: {n} → {new_n}")

            a = _interpolate_distribution(best_seq, new_n)

            z = _logits_from_distribution(a, eps)
            m = np.zeros_like(z)
            v = np.zeros_like(z)

            n = new_n
            it = 0
            last_improve = 0
            continue

        # -----------------------------------------------------------
        # Early stop if stagnation after reaching max size
        # -----------------------------------------------------------
        if n >= n_max and it - last_improve > 300_000:
            print(f"[Iter {it:07d}] No improvement for {it-last_improve} iters – early stop.")
            break

    # ---------------------------------------------------------------
    # 3. Final evaluation (the platform will also evaluate)
    # ---------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[Result] Final exact score = {final_score:.6f}")
    except Exception as exc:
        final_score = float("inf")
        print(f"[Result] Evaluation error: {exc}")

    # ---------------------------------------------------------------
    # 4. Return the best sequence found.
    # ---------------------------------------------------------------
    return final_seq
