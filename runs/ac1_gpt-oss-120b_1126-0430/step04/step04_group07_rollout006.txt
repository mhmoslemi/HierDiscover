"""
Propose a candidate step‑function that minimizes
    O = 2·n·max(conv(a,a))  (with sum(a)=1)

The algorithm is a **Adam‑accelerated exponentiated‑gradient** optimiser
on the probability simplex:

* a = exp(u) / Σexp(u)  (u – log‑probabilities)
* a smooth (log‑sum‑exp) approximation of the max‑convolution is used
  early on, switching to an active‑set (hard‑max) later.
* Adam’s first‑ and second‑moment estimates are kept for the raw gradient
  ∂M/∂a (M ≈ max(conv)).
* Learning‑rate and soft‑max temperature (β) are annealed exponentially.
* Periodic jitter and cheap restarts from the best‑so‑far solution help
  escaping plateaus.
* The convolution and the correlation needed for the gradient share a
  single forward FFT per iteration (size ≈ 3·n) to reduce overhead.

The routine respects a 1000 s budget, prints progress and returns the
best sequence found (as a plain list of floats).  If a previous solution
is available via the global variable ``height_sequence_1`` it is used as
the starting point; otherwise a uniform distribution of length 8000 is
used.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper FFT utilities – share one forward FFT of 'a' per iteration.
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_and_corr(a: np.ndarray, w: np.ndarray | None = None):
    """
    Compute full linear convolution a*a and, optionally, the linear
    correlation of weight vector w with a.

    Parameters
    ----------
    a : np.ndarray, shape (n,)
        Non‑negative probability vector (sum = 1).
    w : np.ndarray or None, shape (2·n‑1,)
        Weight over convolution entries; if None only the convolution is
        returned.

    Returns
    -------
    conv : np.ndarray, shape (2·n‑1,)
        Full linear convolution a*a.
    corr : np.ndarray or None, shape (n,)
        Linear correlation (w * a) if w is provided, else None.
    """
    n = a.shape[0]
    # Length sufficient for both convolution (2n‑1) and correlation (3n‑2)
    L = _next_pow2(3 * n - 2)
    af = np.fft.rfft(a, n=L)                     # forward FFT of a
    conv_full = np.fft.irfft(af * af, n=L)       # a*a
    conv = conv_full[: 2 * n - 1]                # truncate to linear conv

    if w is None:
        return conv, None

    wf = np.fft.rfft(w, n=L)                     # forward FFT of w
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    corr = corr_full[:n]                         # correlation (size n)
    return conv, corr

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list:
    """
    Search for a non‑negative step‑function minimising the bound ``evaluate_sequence``.
    Returns the best found sequence (list of floats).

    Parameters
    ----------
    seed : int (default = 42)
        Random seed for reproducibility.
    budget_s : int (default = 1000)
        Time budget in seconds; the function stops before this deadline.

    Returns
    -------
    list[float]
        The best probability vector (sums to 1) discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # 1) Initialise the log‑probabilities u.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=float)
        a0 = a0 / a0.sum() if a0.sum() > 0 else np.full_like(a0, 1.0 / a0.size)
        n = a0.size
        eps = 1e-12
        u = np.log(a0 + eps)
    else:
        n = 8000
        u = np.full(n, -np.log(n))                 # uniform log‑probabilities

    best_u = u.copy()
    best_score = np.inf
    best_iter = 0

    # ------------------------------------------------------------------
    # 2) Adam state.
    # ------------------------------------------------------------------
    m = np.zeros_like(u)          # first moment
    v = np.zeros_like(u)          # second moment
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    # ------------------------------------------------------------------
    # 3) Schedules (exponential annealing).
    # ------------------------------------------------------------------
    # learning‑rate (log‑domain) → from 0.5 to 0.02
    lr_start, lr_end = 0.5, 0.02
    # soft‑max temperature β → from 2.0 to 5·10⁴
    beta_start, beta_end = 2.0, 5e4

    # switch from soft‑max to hard‑max at this fraction of the run
    soft_fraction = 0.3

    max_iters_est = 400_000      # rough upper bound (used for scheduling)
    jitter_interval = 3000
    jitter_scale = 5e-4
    restart_interval = 25000

    it = 0
    while time.time() < deadline and it < 800_000:
        it += 1

        # --------------------------------------------------------------
        # (a) Build probability vector a from the log‑weights.
        # --------------------------------------------------------------
        a = np.exp(u - u.max())           # stable exponentiation
        a_sum = a.sum()
        if a_sum == 0.0:                  # fallback in case of under‑flow
            a = np.full_like(a, 1.0 / n)
            a_sum = 1.0
        a = a / a_sum

        # --------------------------------------------------------------
        # (b) Convolution and max value.
        # --------------------------------------------------------------
        conv, _ = _conv_and_corr(a)        # mark w=None for now
        M = conv.max()
        cur_score = 2.0 * n * M            # O = 2·n·M  (sum(a)=1)

        # --------------------------------------------------------------
        # (c) Keep track of best solution.
        # --------------------------------------------------------------
        if cur_score < best_score - 1e-14:   # strict improvement
            best_score = cur_score
            best_u = u.copy()
            best_iter = it
            print(f"[iter {it:07d}] new best O={best_score:.6f}")

        # --------------------------------------------------------------
        # (d) Scheduling of β (soft‑max temperature) and η (learning rate).
        # --------------------------------------------------------------
        fraction = min(it / max_iters_est, 1.0)
        beta = beta_start * (beta_end / beta_start) ** fraction
        lr = lr_start * (lr_end / lr_start) ** fraction

        # --------------------------------------------------------------
        # (e) Build weight vector w over convolution entries.
        # --------------------------------------------------------------
        if fraction < soft_fraction:                     # smooth phase
            # use log‑sum‑exp trick: exp(β*(conv-M)) ≤ 1
            shifted = conv - M
            w = np.exp(beta * shifted)
            w_sum = w.sum()
            if w_sum == 0.0:         # numerical safety for huge β
                w = np.zeros_like(conv)
                w[conv.argmax()] = 1.0
            else:
                w /= w_sum
        else:                                            # active set phase
            tol = max(1e-12, M * 1e-8)                    # tight tolerance
            active = np.where(conv >= M - tol)[0]
            if active.size == 0:
                w = np.zeros_like(conv)
                w[conv.argmax()] = 1.0
            else:
                w = np.zeros_like(conv)
                w[active] = 1.0 / active.size

        # --------------------------------------------------------------
        # (f) Gradient of (smoothed) max‑convolution w.r.t a.
        # --------------------------------------------------------------
        _, corr = _conv_and_corr(a, w)   # correlation = ∂M/∂a
        g = 2.0 * corr                   # factor 2 from derivative of a*a

        # --------------------------------------------------------------
        # (g) Adam update on the log‑variables u.
        # --------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)

        # bias‑corrected moments
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # exponentiated‑gradient step (u ← u – η·preconditioned g)
        u = u - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # --------------------------------------------------------------
        # (h) Occasional jitter to escape flat plateaus.
        # --------------------------------------------------------------
        if it % jitter_interval == 0:
            u += np.random.randn(n) * jitter_scale

        # --------------------------------------------------------------
        # (i) Cheap restart if progress stalls.
        # --------------------------------------------------------------
        if it % restart_interval == 0:
            # Re‑initialise from the best point with a tiny perturbation.
            u = best_u.copy()
            u += np.random.randn(n) * (jitter_scale * 0.5)
            print(f"[restart] iter {it:,} – re‑initialised from best (O={best_score:.6f})")

    # ------------------------------------------------------------------
    # 4) Final evaluation with the provided ``evaluate_sequence``.
    # ------------------------------------------------------------------
    best_seq = np.exp(best_u - best_u.max())
    best_seq = best_seq / best_seq.sum()
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f} "
              f"(best iteration {best_iter})")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_seq.tolist()
