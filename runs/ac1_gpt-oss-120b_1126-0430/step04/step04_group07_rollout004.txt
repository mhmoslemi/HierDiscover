"""
Propose a high‑quality step‑function using an Adam‑accelerated
exponentiated‑gradient (mirror‑descent) on the log‑probabilities.
The objective  O = 2·n·max(conv(a,a))  (with Σa=1) is smoothed by a
soft‑max over the convolution entries; the smoothing temperature (β)
is increased slowly, ending with an active‑set (hard‑max) phase.
A per‑coordinate Adam optimiser, occasional random jitter and a
restart‑from‑best strategy keep the search from stagnating.
The routine respects a 1000 s budget, prints progress and returns the
best coefficient list found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper FFT utilities ----------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two greater than or equal to x."""
    return 1 << (x - 1).bit_length()


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * a using a real FFT.
    Returns an array of length 2·len(a)-1.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv


def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Linear correlation g_i = Σ_t w[i+t]·a[t]  for i=0,…,n‑1,
    where w has length 2·n‑1.
    Implemented via FFT (single forward transform of a reused).
    """
    n = a.shape[0]
    m = w.shape[0]                     # = 2·n‑1
    L = _next_pow2(m + n - 1)
    # forward transforms
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    # correlation via product with conj(af)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` bound.  Returns the best coefficient list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # 1) Choose sequence length and initialise log‑probabilities.
    # If a previous solution exists we keep its length, otherwise
    # we use a convenient power‑of‑two size.
    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=float)
        a0 = np.maximum(a0, 0.0)           # guard against negatives
        if a0.sum() == 0.0:
            a0 = np.full_like(a0, 1.0 / a0.size)
        a0 = a0 / a0.sum()
        n = a0.size
        eps = 1e-12
        u = np.log(a0 + eps)               # log‑variables
    else:
        n = 16384                           # power of two for efficient FFT
        u = np.full(n, -np.log(n))          # uniform start (log(1/n))

    # Adam state
    m = np.zeros_like(u)   # first moment
    v = np.zeros_like(u)   # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0

    # bookkeeping
    best_u = u.copy()
    best_score = np.inf
    best_iter = 0
    last_improve = 0

    # --------------------------------------------------------------
    # 2) Hyper‑parameter schedules
    max_iters_est = 300_000          # used for schedule normalisation
    beta_start = 0.5
    beta_end = 8e4
    lr_start = 1.0
    lr_end = 0.02
    jitter_interval = 4000
    jitter_scale = 5e-3
    restart_interval = 30_000

    it = 0
    while time.time() < deadline and it < 800_000:
        it += 1

        # -----------------------------------------------------------
        # (a) Build probability vector a (always normalised to sum 1)
        a = np.exp(u - u.max())               # shift for stability
        a_sum = a.sum()
        if a_sum == 0.0:                      # numerical safety net
            a = np.full_like(a, 1.0 / n)
            a_sum = 1.0
        a = a / a_sum

        # -----------------------------------------------------------
        # (b) Convolution and current objective
        conv = _conv_self_fft(a)
        M = conv.max()
        cur_score = 2.0 * n * M               # because Σa = 1

        # -----------------------------------------------------------
        # (c) Record best
        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_u = u.copy()
            best_iter = it
            last_improve = it
            print(f"[iter {it:07d}] new best O={best_score:.6f}")

        # -----------------------------------------------------------
        # (d) Schedule fraction
        fraction = it / max_iters_est
        if fraction > 1.0:
            fraction = 1.0

        # temperature for the soft‑max over conv entries
        beta = beta_start * (beta_end / beta_start) ** fraction

        # learning‑rate for the Adam update (applied to log‑variables)
        lr = lr_start * (lr_end / lr_start) ** fraction

        # -----------------------------------------------------------
        # (e) Build weight vector w over conv entries
        if fraction < 0.6:                     # soft‑max phase
            shifted = conv - M                 # ≤0, max becomes 0
            # using exp(β·shifted) is safe – exponent ≤ 0
            w = np.exp(beta * shifted)
            w_sum = w.sum()
            if w_sum == 0.0:                   # defensive fallback
                w = np.zeros_like(conv)
                w[conv.argmax()] = 1.0
            else:
                w /= w_sum
        else:                                 # active‑set (hard‑max) phase
            tol = max(1e-14, M * 1e-9)
            active = np.where(conv >= M - tol)[0]
            if active.size == 0:               # should not happen
                w = np.zeros_like(conv)
                w[conv.argmax()] = 1.0
            else:
                w = np.zeros_like(conv)
                w[active] = 1.0 / active.size

        # -----------------------------------------------------------
        # (f) Gradient of the smoothed max‑convolution
        g = 2.0 * _correlation_fft(w, a)      # same shape as u

        # -----------------------------------------------------------
        # (g) Adam update on log‑variables u
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        u = u - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # optional clipping to keep numbers in a safe range
        u = np.clip(u, -50.0, 50.0)

        # -----------------------------------------------------------
        # (h) Occasional random jitter to escape plateaus
        if it % jitter_interval == 0:
            u += rng.normal(scale=jitter_scale, size=n)

        # -----------------------------------------------------------
        # (i) Restart from the best‑so‑far if stagnation detected
        if it % restart_interval == 0:
            if it - last_improve > restart_interval:
                u = best_u.copy()
                # small perturbation to explore a neighbourhood
                u += rng.normal(scale=jitter_scale * 0.5, size=n)
                # reset Adam moments to avoid stale momentum
                m.fill(0.0)
                v.fill(0.0)
                beta1_pow = 1.0
                beta2_pow = 1.0
                print(f"[restart] iter {it}, re‑initialised from best O={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Produce final sequence from the best log‑vector
    best_a = np.exp(best_u - best_u.max())
    best_a = best_a / best_a.sum()

    # 4) Final evaluation (the official scorer)
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f} "
              f"(best iteration {best_iter})")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_a.tolist()
