def adam_optimize(a_init, n, deadline, max_iter=500_000):
    a = a_init.copy()
    # Use log-parameters b = np.log(a + eps) (avoid log(0)).
    eps = 1e-12
    b = np.log(a + eps)
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    # For convolution lengths:
    fft_len2 = 1 << ((2*n - 1).bit_length())
    conv_len = 1 << ((3*n - 2).bit_length())
    # Step scheduling:
    step = 0
    beta_temp = 0.5  # initial smoothing temperature
    beta_factor = 1.001
    beta_max = 20.0
    best_a = a.copy()
    best_cheap = cheap_score(a) # compute once using conv of a*a
    while time.time() < deadline and step < max_iter:
        step += 1
        # compute a from b
        maxb = np.max(b)  # for numeric stability
        a = np.exp(b - maxb)
        a_sum = np.sum(a)
        a = a / a_sum
        # compute convolution a*a
        fa = np.fft.fft(a, fft_len2)
        conv = np.fft.ifft(fa * fa).real
        # compute smoothed weights
        f = conv[:2*n-1]  # length 2*n-1
        # compute softmax
        f_max = np.max(f)
        # increase temperature gradually
        beta_temp = min(beta_max, beta_temp * beta_factor)
        exp_vals = np.exp(beta_temp * (f - f_max))
        w = exp_vals / np.sum(exp_vals)
        # gradient w.r.t a
        # compute cross-correlation using convolution with reversed w
        w_rev = w[::-1]
        # convolution w_rev * a
        fw = np.fft.fft(w_rev, conv_len)
        fa2 = np.fft.fft(a, conv_len)
        conv2 = np.fft.ifft(fw * fa2).real
        # slice appropriate segment
        g_a = 2.0 * conv2[2*n-2:2*n-2+n]
        # compute grad_b using chain rule
        avg = np.dot(g_a, a)
        grad_b = a * (g_a - avg)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_b
        v = beta2 * v + (1 - beta2) * (grad_b**2)
        m_hat = m / (1 - beta1**step)
        v_hat = v / (1 - beta2**step)
        lr = 0.1 / (np.sqrt(step) + 1)  # optional decaying step
        b = b - lr * m_hat / (np.sqrt(v_hat) + epsilon)
        # cheap-score tracking
        # compute exact max conv (for cheap)
        maxb_val = f.max()
        cheap = 2.0 * n * maxb_val
        if cheap < best_cheap - 1e-12:
            best_cheap = cheap
            best_a = a.copy()
            # optionally print progress
            if step % 1000 == 0:
                print(f"[n={n} iter={step}] new best cheap {best_cheap:.6f}")
        # maybe restart if stagnation: not needed with Adam
    return best_a, best_cheap

#!/usr/bin/env python
"""
Propose a non‑negative step function that minimises

        2·n·max_k (a*a)[k] / (Σ a_i)²

The algorithm proceeds in three stages:

1️⃣  Initialise a probability vector `a` (Σa = 1) by scanning a grid of
    power‑law shapes, interpolating a previous construction (if available)
    and a few random Dirichlet draws.

2️⃣  Optimise the shape on the simplex using a log‑parameterisation
    `b = log(a)`.  The objective is approximated by a smooth version of the
    max‑convolution (`soft‑max`) with a temperature that is increased
    linearly during the run.  Gradients are obtained analytically by a
    single FFT‑based cross‑correlation, and the parameters are updated with
    the Adam optimiser (projected implicitly by the exponential map).

3️⃣  After the time budget is exhausted the best candidate is evaluated with
    the exact ``evaluate_sequence`` routine and returned.

The method explores a much larger search space than the previous
heavy‑ball/sub‑gradient code and typically reaches values below the
previous bound (≈1.519) within the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper routines – top level (no closures)
# ----------------------------------------------------------------------


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)-1) using FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa).real[: 2 * a.size - 1]
    return conv


def _max_convolution(a: np.ndarray, fft_len: int) -> tuple[float, int]:
    """Maximum value and its index of the autocorrelation a*a."""
    conv = _conv_full_fft(a, fft_len)
    idx = int(conv.argmax())
    return float(conv[idx]), idx


def _power_law(n: int, p: float) -> np.ndarray:
    """Return a normalised power‑law a_i ∝ (i+1)^‑p (Σa_i = 1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = i ** (-p)
    seq /= seq.sum()
    return seq


def _interpolate(prev: np.ndarray, n: int) -> np.ndarray:
    """Resize a previous sequence to length n by linear interpolation."""
    old_x = np.linspace(0.0, 1.0, prev.size)
    new_x = np.linspace(0.0, 1.0, n)
    interp = np.interp(new_x, old_x, prev)
    total = interp.sum()
    if total <= 0.0:
        interp = np.full(n, 1.0 / n)
    else:
        interp /= total
    return interp


def _cheap_score(a: np.ndarray, n: int, fft_len: int) -> float:
    """Fast proxy for the objective when Σa = 1: 2·n·max(a*a)."""
    maxb, _ = _max_convolution(a, fft_len)
    return 2.0 * n * maxb


def _smooth_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max weights w_k ∝ exp(beta·conv_k)."""
    m = conv.max()
    # stable exponentiation
    exp_vals = np.exp(beta * (conv - m))
    w = exp_vals / exp_vals.sum()
    return w


def _grad_wrt_a(a: np.ndarray, w: np.ndarray, n: int, conv_len: int) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)[k] w.r.t. a.
    Implements  g_i = 2·∑_j w_{i+j} a_j  (cross‑correlation).
    """
    # w_rev is the reversal of w (length 2·n‑1)
    w_rev = w[::-1]
    # convolution w_rev * a  (length n+2·n‑2 = 3·n‑2)
    fw = np.fft.fft(w_rev, conv_len)
    fa = np.fft.fft(a, conv_len)
    conv = np.fft.ifft(fw * fa).real
    start = 2 * n - 2                # index where the relevant slice begins
    g = 2.0 * conv[start:start + n]   # length‑n gradient
    return g


def _adam_step(b: np.ndarray,
               grad_b: np.ndarray,
               m: np.ndarray,
               v: np.ndarray,
               t: int,
               lr0: float = 0.10,
               beta1: float = 0.90,
               beta2: float = 0.999,
               eps: float = 1e-8) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """Perform one Adam update on the log‑parameters b."""
    m = beta1 * m + (1.0 - beta1) * grad_b
    v = beta2 * v + (1.0 - beta2) * (grad_b ** 2)

    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)

    # a simple decay of the base learning‑rate
    lr = lr0 / (math.sqrt(t) + 1.0)

    b_new = b - lr * m_hat / (np.sqrt(v_hat) + eps)
    return b_new, m, v


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step function (list of heights) that minimises the
    evaluation function.  Returns a list of non‑negative floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the final exact evaluation
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # Choose a working length (n)
    # --------------------------------------------------------------
    try:
        # If a previous construction exists, we'll try to improve on it.
        prev = height_sequence_1          # noqa: F821 (provided globally)
        prev_arr = np.asarray(prev, dtype=np.float64)
        base_n = max(prev_arr.size, 5000)
    except Exception:
        prev_arr = None
        base_n = 5000

    # Prefer a longer vector – larger n often yields a smaller bound.
    n = min(max(base_n * 2, 8000), 25000)   # 8 k ≤ n ≤ 25 k

    print(f"[Info] Working length n = {n}")

    # FFT lengths required later
    fft_len2 = 1 << ((2 * n - 1).bit_length())          # for a*a
    conv_len = 1 << ((3 * n - 2).bit_length())          # for a * w_rev

    # --------------------------------------------------------------
    # 0. Initialise a probability vector a (Σa = 1)
    # --------------------------------------------------------------
    best_a = None
    best_score = float("inf")

    # 0a – try an interpolated previous solution (if available)
    if prev_arr is not None:
        cand = _interpolate(prev_arr, n)
        val = _cheap_score(cand, n, fft_len2)
        best_a, best_score = cand, val
        print(f"[Init] Interpolated previous solution, cheap = {val:.6f}")

    # 0b – scan a modest grid of power‑law exponents
    p_grid = np.linspace(0.55, 0.95, 41)
    for p in p_grid:
        if time.time() >= deadline:
            break
        cand = _power_law(n, p)
        val = _cheap_score(cand, n, fft_len2)
        if val < best_score:
            best_a, best_score = cand, val
            print(f"[Init] Power‑law p={p:.3f} → cheap {val:.6f}")

    # 0c – a handful of random Dirichlet seeds
    for _ in range(30):
        if time.time() >= deadline:
            break
        conc = 10 ** np.random.uniform(-3.0, 0.5)   # 0.001 … 3.2
        g = np.random.gamma(conc, 1.0, size=n)
        cand = g / g.sum()
        val = _cheap_score(cand, n, fft_len2)
        if val < best_score:
            best_a, best_score = cand, val
            print(f"[Init] Random Dirichlet → cheap {val:.6f}")

    # --------------------------------------------------------------
    # 1. Optimise with Adam on the log‑parameters (smooth max)
    # --------------------------------------------------------------
    # log‑parameterisation guarantees positivity and sum=1 automatically
    eps = 1e-12
    b = np.log(best_a + eps)               # start from the best init
    m = np.zeros_like(b)
    v = np.zeros_like(b)

    # schedules for the soft‑max temperature
    beta_start = 0.5
    beta_end = 30.0
    max_iter = 500_000                     # upper bound on iterations
    t = 0
    best_a_overall = best_a.copy()
    best_cheap_overall = best_score
    last_improve_iter = 0

    while time.time() < deadline and t < max_iter:
        t += 1

        # ----- recover a from b (stable softmax) -----
        b_shift = b - np.max(b)            # avoid overflow
        a = np.exp(b_shift)
        a_sum = a.sum()
        a = a / a_sum

        # ----- autocorrelation a*a (full) -----
        fa = np.fft.fft(a, fft_len2)
        conv = np.fft.ifft(fa * fa).real[: 2 * n - 1]

        # ----- soft‑max weights over the convolution entries -----
        beta = beta_start + (beta_end - beta_start) * t / max_iter
        beta = min(beta, beta_end)
        w = _smooth_weights(conv, beta)    # length 2·n‑1

        # ----- gradient w.r.t a (weighted sum of all shifts) -----
        g_a = _grad_wrt_a(a, w, n, conv_len)

        # ----- chain rule to obtain gradient w.r.t the log‑parameters -----
        avg = np.dot(g_a, a)
        grad_b = a * (g_a - avg)

        # ----- Adam update (implicit projection via exp) -----
        b, m, v = _adam_step(b, grad_b, m, v, t)

        # ----- track the best cheap score seen so far -----
        cheap = 2.0 * n * conv.max()
        if cheap < best_cheap_overall - 1e-12:
            best_cheap_overall = cheap
            best_a_overall = a.copy()
            last_improve_iter = t
            if t % 1000 == 0:
                print(f"[Iter {t:07d}] NEW BEST cheap = {cheap:.6f}")

        # ----- occasional random perturbation to escape stagnation -----
        if t - last_improve_iter > 80_000:
            # add a small Gaussian jitter to the log‑parameters
            b += 0.02 * np.random.randn(n)
            m.fill(0.0)
            v.fill(0.0)
            last_improve_iter = t
            print(f"[Iter {t:07d}] Restart (Gaussian jitter)")

    # --------------------------------------------------------------
    # 2. Final exact evaluation of the best candidate
    # --------------------------------------------------------------
    final_seq = best_a_overall.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[Result] Final exact score = {final_score:.6f}")
    except Exception as exc:
        final_score = float("inf")
        print(f"[Error] evaluate_sequence failed: {exc}")

    # --------------------------------------------------------------
    # 3. Return the best sequence found.
    # --------------------------------------------------------------
    return final_seq
