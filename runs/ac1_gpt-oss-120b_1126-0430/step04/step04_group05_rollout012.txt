    hundred mirror‑descent steps (multiplicative update a←a·exp(−η·g))
    where g is the sub‑gradient of the current maximal convolution
    entry, and keeps the best vector found.

2️⃣  Refinement with Adam on a smooth log‑sum‑exp surrogate of the
    max‑operator.  The smoothing parameter α is annealed from a modest
    value to a very large one, turning the surrogate into the exact
    max.

3️⃣  Targeted stochastic refinement.  The shift k* attaining the
    maximal convolution is identified; the corresponding sub‑gradient
    points to indices that contribute most to that peak.  A small
    fraction of probability mass is moved from the “worst’’ index to
    the “best’’ one (mass‑transfer move) and accepted with a
    Metropolis rule.  Occasionally a random jitter is added and a
    simple restart is performed if progress stalls.

All steps respect the 1000 s wall‑time budget and maintain the
vector on the probability simplex.  The best sequence encountered
is finally checked with the provided evaluate_sequence() and returned
as a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Projection onto the probability simplex {x ≥ 0, sum(x)=1}
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:                     # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full self‑convolution of a (length 2·n‑1) via FFT.
def _conv_full(a: np.ndarray, L: int) -> np.ndarray:
    """Return a*a (full convolution) using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Sub‑gradient of the maximal entry of the convolution.
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a.
    For every maximal shift k we add the contribution
        g_i += 2·a_{k-i}
    (averaged over all maximal shifts).
    """
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
# Gradient of the soft‑max (log‑sum‑exp) surrogate of max(conv).
def _smooth_gradient(a: np.ndarray, conv: np.ndarray,
                     alpha: float, L: int) -> np.ndarray:
    """
    ∇_a   ( (2·n/α)·log∑_k e^{α·conv_k} )
    The formula reduces to 2·corr(w,a) where
        w_k ∝ e^{α·conv_k} .
    """
    # soft‑max weights (stabilised)
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / exp_scaled.sum()

    # correlation w ⋆ a  (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)   # length ≥ 3·n‑2
    n_len = a.size
    grad = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]
    return grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence that minimises the evaluation
    function.  Returns the sequence as a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2          # safety margin

    # --------------------------------------------------------------
    # Initialise a (probability vector on the simplex)
    if 'height_sequence_1' in globals():
        a = np.array(height_sequence_1, dtype=float)
        a = np.maximum(a, 0.0)
        if a.sum() == 0.0:
            a = np.full_like(a, 1.0)
        a = a / a.sum()
        n = a.size
        print(f"[init] supplied sequence, n={n}")
    else:
        n = 5000                                    # a size giving a good trade‑off
        a = np.full(n, 1.0 / n)
        print(f"[init] uniform start, n={n}")

    L = 1 << ((3 * n - 2).bit_length())          # length for all FFTs (≥3·n‑2)

    # initial objective
    conv = _conv_full(a, L)
    best_obj = 2.0 * n * conv.max()
    best_seq = a.copy()
    print(f"[init] O={best_obj:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – many random restarts with mirror‑descent on the exact objective
    phase1_deadline = start_time + budget_s * 0.30
    restart_id = 0
    while time.time() < phase1_deadline:
        # ---- random Dirichlet start (conc = 0.5 gives a diverse set) ----
        a = np.random.gamma(0.5, 1.0, size=n)
        a /= a.sum()

        # occasional jitter around the current best – helps to explore a neighbourhood
        if np.random.rand() < 0.12 and restart_id % 5 == 0:
            a = best_seq + 0.015 * np.random.randn(n)
            a = np.clip(a, 0, None)
            a /= a.sum()

        # ---- mirror‑descent (multiplicative update) ----
        max_iter = 2000
        eta0 = 0.65
        for it in range(max_iter):
            conv = _conv_full(a, L)
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_seq = a.copy()
                print(f"[phase‑1] restart{restart_id} iter{it} new best O={best_obj:.6f}")

            # sub‑gradient of the current maximal convolution entry
            g = _subgradient_max_conv(a, conv)

            # step‑size (classic 1/√t schedule)
            eta = eta0 / math.sqrt(it + 1)

            # mirror‑descent update (multiplicative)
            a = a * np.exp(-eta * g)

            # project back onto the simplex
            s = a.sum()
            if s == 0.0:
                a = np.full(n, 1.0 / n)
            else:
                a = a / s

            # tiny random jitter to avoid dead‑locks
            if np.random.rand() < 0.001:
                a += 0.001 * np.random.randn(n)
                a = np.clip(a, 0, None)
                a /= a.sum()
        restart_id += 1

    # --------------------------------------------------------------
    # Phase 2 – Adam on the smoothed log‑sum‑exp surrogate (sharpening α)
    a = best_seq.copy()
    beta1, beta2, eps = 0.9, 0.999, 1e-12
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    lr0 = 0.6
    alpha_start, alpha_end = 5.0, 5e4
    max_iter_phase2 = 500_000
    phase2_deadline = start_time + budget_s * 0.70
    it = 0
    while it < max_iter_phase2 and time.time() < phase2_deadline:
        conv = _conv_full(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_seq = a.copy()
            print(f"[phase‑2] iter{it} new best O={best_obj:.6f}")

        # anneal the smoothing parameter α (linear schedule)
        frac = it / max_iter_phase2
        alpha = alpha_start + (alpha_end - alpha_start) * frac

        # smooth gradient of the surrogate
        grad = _smooth_gradient(a, conv, alpha, L)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        lr = lr0 / math.sqrt(it + 1)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the simplex
        a = _project_to_simplex(a)

        # occasional pruning of tiny entries (helps sparsity)
        if it % 4000 == 0 and it > 0:
            thresh = np.percentile(a, 0.5)
            a[a < thresh] = 0.0
            a = _project_to_simplex(a)

        it += 1

    # --------------------------------------------------------------
    # Phase 3 – targeted stochastic refinement
    a = best_seq.copy()
    conv = _conv_full(a, L)
    cur_obj = 2.0 * n * conv.max()
    print(f"[phase‑3] start stochastic refinement, O={cur_obj:.6f}")

    max_iter_local = 400_000
    no_improve = 0
    T0 = 0.08                              # initial temperature for Metropolis
    T = T0

    it_loc = 0
    while time.time() < deadline and it_loc < max_iter_local:
        # recompute convolution only when needed
        conv = _conv_full(a, L)
        cur_obj = 2.0 * n * conv.max()

        # --- mass‑transfer move guided by the sub‑gradient ---
        g = _subgradient_max_conv(a, conv)
        i_high = int(np.argmax(g))               # index that hurts the current peak the most
        i_low = int(np.argmin(g))                # index that hurts it the least
        if i_high != i_low and a[i_high] > 1e-14:
            # move a modest fraction (≤12 %) of the source mass
            delta = a[i_high] * (np.random.rand() * 0.12)
            a[i_high] -= delta
            a[i_low] += delta
            a = _project_to_simplex(a)

            conv_new = _conv_full(a, L)
            new_obj = 2.0 * n * conv_new.max()

            # Metropolis acceptance
            if new_obj < cur_obj or np.random.rand() < math.exp((cur_obj - new_obj) / max(T, 1e-12)):
                cur_obj = new_obj
                conv = conv_new
                no_improve = 0
                if cur_obj < best_obj:
                    best_obj = cur_obj
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] New global best O={best_obj:.6f} (iter={it_loc})")
            else:
                no_improve += 1
        else:
            no_improve += 1

        # occasional random jitter (helps escape shallow plateaus)
        if it_loc % 500 == 0:
            idx = np.random.randint(n)
            a[idx] = max(0.0, a[idx] + (np.random.rand() - 0.5) * 0.02)
            a = _project_to_simplex(a)

        # temperature decay
        T *= 0.9996

        # restart if we have been stuck for a while
        if no_improve >= 8_000:
            a = best_seq.copy()
            a += 0.002 * np.random.randn(n)
            a = np.clip(a, 0, None)
            a = _project_to_simplex(a)
            no_improve = 0

        it_loc += 1

    # ------------------------------------------------------------------
    # Final verification with the provided evaluator
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence failed:", exc)

    return best_seq.tolist()
