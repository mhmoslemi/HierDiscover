def _power_law_distribution(n, p):
    idx = np.arange(1, n+1, dtype=np.float64)
    a = idx**(-p)
    a_sum = a.sum()
    if a_sum == 0:
        a[:] = 1.0 / n
    else:

    return a

def _interpolate_distribution(a, new_n):
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0)
    s = new_a.sum()
    if s == 0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= s
    return new_a

#!/usr/bin/env python
"""
Optimisation of a non‑negative step‑function via a full‑gradient
log‑sum‑exp smoothing of the self‑convolution.

* The vector `a` lives on the probability simplex (Σa = 1, a_i ≥ 0).
* The objective is 2·n·max(a*a) (the denominator Σa² = 1 after normalising).
* We minimise the smooth approximation
        J(a) = (1/θ)·log Σ_k exp(θ·(a*a)_k)
  whose gradient w.r.t. a is
        ∇J = 2·corr(a, w)   with   w_k ∝ exp(θ·(a*a)_k).
  Correlation is computed with a single pair of FFTs (O(n log n)).
* Adam on the log‑it representation implements mirror descent on the simplex.
* The temperature θ is increased linearly from 0.5 to 30.
* The support size is gradually enlarged (15000 → 20000 → 25000 → 30000)
  by linear interpolation of the best distribution found so far.
* Random jitter and Adam‑state damping help escape plateaus.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max; returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it representation that yields `a` after soft‑max."""
    return np.log(a + eps)


def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    s = new_a.sum()
    if s == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= s
    return new_a


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """Normalised power‑law a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _self_convolution_fft(a: np.ndarray, M: int) -> tuple[np.ndarray, np.ndarray]:
    """
    Full linear self‑convolution a*a via FFT of length M.
    Returns the convolution (length 2·len(a)‑1) and the FFT of a.
    """
    a_fft = np.fft.rfft(a, M)
    conv = np.fft.irfft(a_fft * a_fft, M)[: 2 * a.size - 1]
    return conv, a_fft


def _weighted_gradient(a: np.ndarray,
                       conv: np.ndarray,
                       a_fft: np.ndarray,
                       theta: float,
                       M: int) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp smoothed max‑convolution:
        ∇J = 2·corr(a, w)   with   w_k ∝ exp(θ·conv_k)

    The correlation is evaluated with FFT in O(n log n).
    """
    # soft‑max weights over the full convolution
    max_c = conv.max()
    exps = np.exp(theta * (conv - max_c))
    w = exps / exps.sum()

    # correlation a ⋆ w  (linear, not circular) via FFT
    w_fft = np.fft.rfft(w, M)
    corr_full = np.fft.irfft(a_fft * np.conj(w_fft), M)
    # first n entries correspond to lags 0 … n‑1
    g = 2.0 * corr_full[: a.shape[0]]
    return g.real  # discard possible tiny imaginary part

# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) beats the
    target bound.  The routine respects the given time budget (~1000 s)
    and uses a smoothed‑gradient Adam optimiser on the simplex.
    """
    np.random.seed(seed)
    start = time.time()
    # a small safety margin – the platform will also evaluate the result
    deadline = start + budget_s - 2.0

    # ------------------------------------------------------------------
    # 0. Working support size
    # ------------------------------------------------------------------
    n_max = 30000
    n_start = 15000          # start small for cheap iterations

    # ------------------------------------------------------------------
    # 1. Initialise a probability vector (ℓ₁‑normalised)
    # ------------------------------------------------------------------
    global_seq = globals().get("height_sequence_1")
    if global_seq is not None:
        a0 = np.array(global_seq, dtype=np.float64)
        a0 = np.maximum(a0, 0.0)
        s0 = a0.sum()
        if s0 == 0.0:
            a0 = np.full_like(a0, 1.0 / a0.shape[0])
        else:
            a0 /= s0
        if a0.shape[0] > n_start:
            a = _interpolate_distribution(a0, n_start)
            print(f"[Init] Global sequence down‑sampled to {n_start} entries")
        else:
            a = a0.copy()

    else:

        print(f"[Init] No global – power‑law start with n={n_start}")

    n = a.shape[0]

    # ------------------------------------------------------------------
    eps = 1e-16

    m = np.zeros_like(z)                # first‑moment estimate

    beta1 = 0.9
    beta2 = 0.999

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    theta_start = 0.5
    theta_end = 30.0

    # ------------------------------------------------------------------
    # 4. Tracking best solution
    # ------------------------------------------------------------------
    M = _next_pow2(3 * n)
    conv, a_fft = _self_convolution_fft(a, M)
    best_score = 2.0 * n * conv.max()
    best_seq = a.copy()
    last_improve = 0

    # ------------------------------------------------------------------

    while time.time() < deadline and it < 2_000_000:
        it += 1

        # probability vector from logits

        # ------------------------------------------------------------------
        g = _weighted_gradient(a, conv, a_fft, theta, M)

        # ------------------------------------------------------------------

        z = z + alpha_t * step_dir
        np.clip(z, -80.0, 80.0, out=z)          # keep logits in a safe numeric range

        # ------------------------------------------------------------------
        # Occasional random jitter (helps escape plateaus)
        # ------------------------------------------------------------------
        if it % 5000 == 0 and np.random.rand() < 0.30:

        # ------------------------------------------------------------------

            new_n = min(n_max, n + 5000)
            print(f"[Iter {it:07d}] Expanding support {n} → {new_n}")
            # interpolate the best distribution obtained so far
            a = _interpolate_distribution(best_seq, new_n)
            z = _logits_from_distribution(a, eps)
            m = np.zeros_like(z)
            v = np.zeros_like(z)
            n = new_n
            # reset counters for the new problem size
            it = 0
            last_improve = 0
            # recompute best_score for the enlarged support

            conv, a_fft = _self_convolution_fft(a, M)
            best_score = 2.0 * n * conv.max()
            best_seq = a.copy()

        # ------------------------------------------------------------------
        if n >= n_max and it - last_improve > 300_000:

    # 6. Final exact evaluation (the platform will also evaluate)
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:

        print(f"[Result] Final exact score = {final_score:.6f}")
