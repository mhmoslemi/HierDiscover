    n0 = 8000

    # maybe use global init if exists else use random start.

    # logit representation
    eps = 1e-16

    # Adam
    m = np.zeros_like(z); v = np.zeros_like(z)
    beta1=0.9; beta2=0.999; eps_adam=1e-8
    base_alpha = 0.2

    best_seq = a.copy()
    best_score = float('inf')
    iteration = 0
    last_improve_iter = 0
    # schedule parameters
    alpha_start = 0.5
    alpha_end = 20.0
    # compute slope per second
    total_time = budget_s - 10
    alpha_slope = (alpha_end - alpha_start) / max(1.0, total_time)

    # loop
    while time.time() < deadline and iteration < 2_000_000:
        iteration += 1

        a = _softmax(z)
        # optional symmetrize
        a = (a + a[::-1]) * 0.5
        a /= a.sum()  # renorm after sym

        # compute conv
        fft_len = _fft_len_for_n(len(a))
        conv = _conv_full_fft(a, fft_len)  # length 2n-1

        # cheap score = 2*n*max(conv)
        n = len(a)
        cheap_score = 2.0 * n * conv.max()
        if cheap_score < best_score:
            best_score = cheap_score
            best_seq = a.copy()
            last_improve_iter = iteration
            print(f"[Iter {iteration}] New best cheap={cheap_score:.6f}")

        # compute gradient
        # temperature / alpha schedule
        elapsed = time.time() - start
        alpha = min(alpha_end, alpha_start + alpha_slope * elapsed) # used for smoothing parameter

        # compute softmax weights
        max_conv = conv.max()
        exps = np.exp(alpha * (conv - max_conv))
        w = exps / exps.sum()
        # gradient in a-space
        # compute gradient = alpha * 2 * conv_a_w using FFT
        # w_rev
        w_rev = w[::-1]
        # convolution length needed: conv of a (n) with w_rev (2n-1) -> length 3n-2
        conv_len = 3 * n - 2
        L = 1 << ((conv_len - 1).bit_length())
        Fa = np.fft.rfft(a, L)
        Fw = np.fft.rfft(w_rev, L)
        d = np.fft.irfft(Fa * Fw, L)  # length L
        # take slice n-1 to 2n-2 inclusive (size n). then reverse.
        grad_a = alpha * 2.0 * d[n-1:2*n-1][::-1]

        # Mirror descent conversion to logits
        mu = np.dot(grad_a, a)
        grad_z = a * (grad_a - mu)

        # Update Adam
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * (grad_z ** 2)
        m_hat = m / (1 - beta1 ** iteration)
        v_hat = v / (1 - beta2 ** iteration)

        # step direction
        step = - m_hat / (np.sqrt(v_hat) + eps_adam)

        # step-size schedule
        alpha_t = base_alpha / (math.sqrt(iteration) + 1)
        z = z + alpha_t * step

        # clip logits
        np.clip(z, -80, 80, out=z)

        # random perturb occasionally
        if iteration % 5000 == 0 and np.random.rand() < 0.2:
            z += 0.02 * np.random.randn(n)
            m *= 0.5; v *= 0.5

        # enlarge support condition
        if (iteration % 25000 == 0) and (n < n_max) and (time.time() + 30 < deadline):
            new_n = min(n_max, n + 5000)
            print(f"[Iter {iteration}] Expanding support: {n} -> {new_n}")
            # interpolate best_seq to new_n
            a = _interpolate_distribution(best_seq, new_n)
            # reinitialize z, m, v
            z = _logits_from_distribution(a, eps)
            m = np.zeros_like(z)
            v = np.zeros_like(z)
            # reset iteration counters for new phase
            iteration = 0
            last_improve_iter = 0
            best_seq = a.copy()
            best_score = 2.0 * new_n * (_conv_full_fft(a, _fft_len_for_n(new_n)).max())
            continue

        # early stopping
        if n >= n_max and iteration - last_improve_iter > 200_000:
            print(f"[Iter {iteration}] No improvement for long time, stop.")
            break

    # return best seq
    final_seq = best_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[Result] Final exact score = {final_score:.6f}")
    except Exception as exc:
        print(f"[Result] Evaluation error: {exc}")

    return final_seq

#!/usr/bin/env python
"""
Propose a non‑negative step‑function (list of heights) that minimises

    2·n·max(convolve(a, a)) / (sum(a)^2)

where n = len(a).  The expression is scale‑invariant, so we restrict
∑a_i = 1 and search over probability vectors.

Algorithm
---------

* Initialise a probability vector of moderate size (≈ 8 000) by mixing a
  power‑law and a Hann window – both give a smooth, non‑negative shape.
* Represent the vector in the log‑it (softmax) parametrisation
  `a = softmax(z)`; this guarantees the simplex constraints.
* At each iteration compute the full self‑convolution `c = a*a` (FFT‑based
  O(n log n)).  A smoothed approximation of the maximum is the
  log‑sum‑exp:
        f(a) = (1/α)·log Σ_i exp(α·c_i).
  Its gradient w.r.t. a is a simple weighted convolution:
        ∇_a f = 2·a * w_rev   (reversed weights)
  where `w_i = softmax(α·c_i)`.
* Perform Adam updates on the logits `z` using the mirror‑descent
  conversion `∇_z f = a·(∇_a f – ⟨∇_a f, a⟩)`.
* The smoothing temperature `α` is annealed from a small value (≈ 5)
  up to a large value (≈ 5 000) so that the optimisation first sees a
  smooth landscape and finally focuses on the true maximum.
* Every few thousand steps a tiny random perturbation is added to
  escape plateaus.  When progress stalls we either re‑initialise from a
  fresh power‑law or enlarge the support (interpolating the current
  best distribution) up to a hard cap of 30 000 entries.
* The best sequence (lowest cheap bound `2·n·max(conv)`) is kept and
  finally returned.  The platform will evaluate it with `evaluate_sequence`.

The whole routine respects a time budget of ≈ 1000 s and never uses
closures or file I/O.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable softmax returning a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Logit vector that produces `a` after a softmax."""
    return np.log(a + eps)


def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two ≥ 2·n‑1 (efficient size for real FFT)."""
    return 1 << ((2 * n - 1).bit_length())


def _fft_len_for_conv_len(m: int) -> int:
    """Smallest power‑of‑two ≥ m (used for arbitrary convolution lengths)."""
    return 1 << ((m - 1).bit_length())


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full linear self‑convolution a*a using an FFT of length `fft_len`.
    Returns an array of length 2·len(a)‑1.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.size - 1]
    return conv


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    total = new_a.sum()
    if total == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= total
    return new_a


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """Normalised power‑law a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _hann_distribution(n: int) -> np.ndarray:
    """Normalised Hann window (raised cosine) on [0,n‑1]."""
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _gradient_logsumexp_conv(a: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the smoothed objective
        f(a) = (1/α) * log Σ_i exp(α * (a*a)_i)
    with respect to the probability vector `a`.
    Returned gradient has the same shape as `a`.
    """
    n = a.shape[0]

    # 1. self‑convolution
    fft_len = _fft_len_for_n(n)
    conv = _conv_full_fft(a, fft_len)                     # length 2n‑1

    # 2. soft‑max weights w_i ∝ exp(α·conv_i)
    max_c = conv.max()
    # shift by max for numerical stability
    exps = np.exp(alpha * (conv - max_c))
    w = exps / exps.sum()                                 # length 2n‑1

    # 3. weighted gradient  g_a = 2 * (a * w_rev)  (reversed w)
    w_rev = w[::-1]                                       # length 2n‑1
    conv_len = 3 * n - 2
    L = _fft_len_for_conv_len(conv_len)                   # power of two ≥ 3n‑2
    Fa = np.fft.rfft(a, L)
    Fw = np.fft.rfft(w_rev, L)
    d = np.fft.irfft(Fa * Fw, L)                         # length L
    # slice the relevant n entries (indices n‑1 … 2n‑2) and reverse
    grad_a = 2.0 * d[n - 1 : 2 * n - 1][::-1]
    return grad_a


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights (a step‑function) that aims
    to minimise the evaluation function.  The routine respects a ~1000 s
    time budget and uses a smoothed max‑convolution gradient (Adam on
    the probability simplex) with progressive support enlargement.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # safety margin for final eval

    # ------------------------------------------------------------------
    # 0. Choose initial support size and initialise a probability vector.
    # ------------------------------------------------------------------
    init_global = globals().get("height_sequence_1", None)
    if init_global is not None:
        init_global = np.array(init_global, dtype=np.float64)
        n = max(2000, init_global.shape[0])
        a = np.maximum(init_global, 0.0)
        s = a.sum()
        a = a / s if s > 0.0 else np.full_like(a, 1.0 / a.shape[0])
        print(f"[Init] Using supplied global sequence, length = {n}")
    else:
        n = 8000                              # a comfortable default
        # mixture of Hann window and power‑law for a good generic start
        a = 0.5 * _hann_distribution(n) + 0.5 * _power_law_distribution(n, 0.60)
        a /= a.sum()
        print(f"[Init] Started from mixed Hann + power‑law, length = {n}")

    # log‑it parametrisation
    eps = 1e-16
    z = _logits_from_distribution(a, eps)

    # Adam state
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    base_alpha = 0.25                      # base learning‑rate

    # ------------------------------------------------------------------
    # 1. Optimisation parameters
    # ------------------------------------------------------------------
    n_max = 30000
    best_seq = a.copy()
    best_score = float("inf")
    it = 0
    last_improve = 0

    # smoothing (log‑sum‑exp) temperature schedule
    alpha_start = 5.0
    alpha_end = 5000.0
    alpha_slope = (alpha_end - alpha_start) / max(1.0, budget_s - 10.0)

    # ------------------------------------------------------------------
    # 2. Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline and it < 2_000_000:
        it += 1

        # probability vector from logits
        a = _softmax(z)

        # enforce symmetry (the problem is invariant under reversal)
        a = (a + a[::-1]) * 0.5
        a /= a.sum()

        # self‑convolution (cheap to compute, needed for the bound)
        fft_len = _fft_len_for_n(len(a))
        conv = _conv_full_fft(a, fft_len)            # length 2n‑1

        # cheap bound = 2·n·max(conv) (since sum(a)=1)
        cheap_score = 2.0 * len(a) * conv.max()
        if cheap_score < best_score:
            best_score = cheap_score
            best_seq = a.copy()
            last_improve = it
            print(f"[Iter {it:07d}] NEW BEST cheap = {cheap_score:.6f}")

        # --------------------------------------------------------------
        # Smoothed gradient of max‑convolution (log‑sum‑exp surrogate)
        # --------------------------------------------------------------
        elapsed = time.time() - start
        alpha = min(alpha_end, alpha_start + alpha_slope * elapsed)

        grad_a = _gradient_logsumexp_conv(a, alpha)   # shape (n,)

        # mirror‑descent conversion to log‑it space
        mu = float(np.dot(grad_a, a))
        grad_z = a * (grad_a - mu)

        # Adam moment updates (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step direction (negative Adam step)
        step_dir = - m_hat / (np.sqrt(v_hat) + eps_adam)

        # basic step‑size schedule: inverse‑sqrt decay
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)

        # perform the update in log‑it space
        z = z + alpha_t * step_dir
        # keep logits bounded for numerical stability
        np.clip(z, -80.0, 80.0, out=z)

        # --------------------------------------------------------------
        # Random perturbations (helps leave plateaus)
        # --------------------------------------------------------------
        if it % 5000 == 0 and np.random.rand() < 0.2:
            z += 0.02 * np.random.randn(len(a))
            m *= 0.5
            v *= 0.5

        # --------------------------------------------------------------
        # Progressive enlargement of the support (if time permits)
        # --------------------------------------------------------------
        if (it % 25000 == 0) and (len(a) < n_max) and (time.time() + 30.0 < deadline):
            new_n = min(n_max, len(a) + 5000)
            print(f"[Iter {it:07d}] Expanding support: {len(a)} → {new_n}")
            a = _interpolate_distribution(best_seq, new_n)
            a = (a + a[::-1]) * 0.5
            a /= a.sum()
            z = _logits_from_distribution(a, eps)
            m = np.zeros_like(z)
            v = np.zeros_like(z)
            it = 0
            last_improve = 0
            best_seq = a.copy()
            best_score = 2.0 * new_n * _conv_full_fft(a, _fft_len_for_n(new_n)).max()
            continue

        # --------------------------------------------------------------
        # Detect stagnation – occasional full restart from a fresh power‑law
        # --------------------------------------------------------------
        if it - last_improve > 200_000:
            print(f"[Iter {it:07d}] Stagnation detected – re‑initialising.")
            a = _power_law_distribution(len(a), np.random.uniform(0.3, 0.9))
            a = (a + a[::-1]) * 0.5
            a /= a.sum()
            z = _logits_from_distribution(a, eps)
            m = np.zeros_like(z)
            v = np.zeros_like(z)
            it = 0
            last_improve = 0
            continue

    # ------------------------------------------------------------------
    # 3. Final evaluation (the platform will also evaluate)
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[Result] Final exact score = {final_score:.6f}")
    except Exception as exc:
        print(f"[Result] Evaluation error: {exc}")

    # ------------------------------------------------------------------
    # 4. Return the best sequence found.
    # ------------------------------------------------------------------
    return final_seq
