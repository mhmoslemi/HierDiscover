def evaluate_sequence(sequence: list[float]) -> float:
    # security: convert to floats, enforce nonnegative, clip to [0,1000]

#!/usr/bin/env python
"""
Propose a non‑negative height sequence that minimises

    2 * n * max(convolve(a, a)) / (sum(a)**2)

using a full‑gradient Adam optimiser on the probability simplex.
The gradient of the smooth approximation of the max‑convolution is
computed analytically via FFT‑based correlation, i.e. we use the exact
sub‑gradient of  log‑sum‑exp(θ·conv)  (θ is increased linearly from a
small to a large value).  The optimiser works on logits `z` so that
`a = softmax(z)` automatically enforces non‑negativity and unit sum.
A short random‑perturbation scheme and optional support enlargement are
used to escape plateaus.  The best sequence found before the time‑budget
expires is returned (the external ``evaluate_sequence`` will finally
evaluate it).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it representation that yields `a` after soft‑max."""
    return np.log(a + eps)


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (useful for FFT lengths)."""
    if x <= 1:
        return 1
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, pad_len: int) -> np.ndarray:
    """
    Full linear self‑convolution a*a using a real FFT of length ``pad_len``.
    Returns an array of length 2·len(a)‑1.
    """
    fa = np.fft.rfft(a, pad_len)
    conv = np.fft.irfft(fa * fa, pad_len)[: 2 * a.size - 1]
    return conv


def _weighted_full_subgradient(a: np.ndarray,
                               conv: np.ndarray,
                               theta: float,
                               pad_len: int) -> np.ndarray:
    """
    Exact sub‑gradient of the smoothed max‑convolution.

    Parameters
    ----------
    a : np.ndarray
        Current probability vector (∑a = 1).
    conv : np.ndarray
        Full linear self‑convolution a*a (length 2·n‑1).
    theta : float
        Smoothing temperature – larger values concentrate the weight on the
        absolute maximum of the convolution.
    pad_len : int
        FFT length used for the correlation (≥ 3·n‑2).

    Returns
    -------
    g : np.ndarray
        Weighted sub‑gradient w.r.t. `a` (same shape as ``a``).
    """
    n = a.shape[0]
    L = conv.shape[0]               # L = 2·n‑1
    maxc = conv.max()
    if maxc <= 0.0:
        return np.zeros_like(a)

    # Normalise the convolution so that its maximum equals 1.
    conv_norm = conv / maxc
    # Soft‑max weights on the normalised values.
    # conv_norm ≤ 1, the maximum has value 1 → exponent = 0.
    vals = conv_norm - 1.0            # ≤ 0, with a single zero at the peak
    exps = np.exp(theta * vals)      # safe, no overflow
    w = exps / exps.sum()            # shape (L,)

    # ---- correlation a with w  (g_raw_j = Σ_i w_i a_{i-j}) ----
    # This is a linear correlation which can be obtained by a convolution
    # with the reversed weight vector.
    rev_w = w[::-1]                  # length L
    fa = np.fft.rfft(a, pad_len)
    fw = np.fft.rfft(rev_w, pad_len)
    corr_full = np.ipt(np.fft.irfft(fa * fw, pad_len))[: n + L - 1]

    # Extract the needed slice: g_raw[j] = corr_full[L‑1‑j]  (j = 0…n‑1)
    idx = np.arange(n, dtype=np.int64)
    g_raw = corr_full[(L - 1) - idx]

    # Diagonal correction: each offset i=2·j contributes only once
    # to the derivative of the self‑convolution.
    diag = w[2 * idx] * a

    # Full weighted sub‑gradient
    g = 2.0 * g_raw - diag
    return g


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    tot = new_a.sum()
    if tot == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= tot
    return new_a


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """Normalised power‑law a_i ∝ (i+1)^{‑p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _run_optimizer(init_a: np.ndarray,
                   deadline: float,
                   max_n: int,
                   max_iters: int = 1_500_000,
                   allow_expand: bool = False) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex with full weighted sub‑gradient.
    Returns the best probability vector found and its cheap score
    (2·n·max(conv)).
    """
    np.random.seed()                # keep stochasticity independent between phases
    a = np.maximum(init_a, 0.0)
    a_sum = a.sum()
    if a_sum == 0.0:
        a[:] = 1.0 / a.shape[0]
    else:
        a /= a_sum

    n = a.shape[0]
    # Adam state
    eps = 1e-16
    z = _logits_from_distribution(a, eps)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.90
    beta2 = 0.999
    eps_adam = 1e-8
    base_alpha = 0.25

    # temperature schedule for the smoothed max
    theta_start = 0.5
    theta_end = 50.0   # fairly high – drives the weight onto the true maximum
    total_budget = deadline - time.time()
    theta_slope = (theta_end - theta_start) / max(1.0, total_budget - 5.0)

    # FFT length for correlation (≥ 3·n‑2)
    pad_len_corr = _next_pow2(3 * n - 2)

    best_seq = a.copy()
    # cheap proxy = exact objective because ∑a = 1
    conv = _conv_self_fft(a, pad_len_corr)
    best_score = 2.0 * n * conv.max()
    it = 0
    last_improve = 0
    report_every = 2000

    while time.time() < deadline and it < max_iters:
        it += 1

        # probability vector from logits
        a = _softmax(z)

        # full self‑convolution (FFT)
        conv = _conv_self_fft(a, pad_len_corr)
        maxb = float(conv.max())
        cheap_score = 2.0 * n * maxb

        # record improvement
        if cheap_score < best_score:
            best_score = cheap_score
            best_seq = a.copy()
            last_improve = it
            if it % report_every == 0:
                print(f"[Iter {it:07d}] NEW BEST cheap = {cheap_score:.6f}")

        # -----------------------------------------------------------
        # Weighted full sub‑gradient (smooth approximation of max)
        # -----------------------------------------------------------
        elapsed = time.time() - (deadline - total_budget)
        theta = min(theta_end, theta_start + theta_slope * elapsed)

        g_a = _weighted_full_subgradient(a, conv, theta, pad_len_corr)

        # Mirror‑descent (natural gradient) conversion to log‑it space
        mu = float(np.dot(g_a, a))
        grad_z = a * (g_a - mu)

        # Adam moment updates (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # direction (negative Adam step)
        step_dir = - m_hat / (np.sqrt(v_hat) + eps_adam)

        # inverse‑sqrt decay of the step size
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)

        # update logits
        z = z + alpha_t * step_dir
        np.clip(z, -80.0, 80.0, out=z)   # keep numbers numerically safe

        # -----------------------------------------------------------
        # Random perturbations (helps escape plateaux)
        # -----------------------------------------------------------
        if it % 5000 == 0 and np.random.rand() < 0.25:
            perturb = 0.02 * np.random.randn(n)
            z += perturb
            m *= 0.5
            v *= 0.5

        # -----------------------------------------------------------
        # Early stop on stagnation (when max size already reached)
        # -----------------------------------------------------------
        if not allow_expand and it - last_improve > 300_000:
            print(f"[Iter {it:07d}] Stagnation – early exit (no improvement for {it-last_improve} iters).")
            break

    return best_seq, best_score


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function height vector that minimises the provided
    evaluation function.  The routine respects the time budget (≈1000 s)
    and returns the best sequence discovered.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0   # small safety margin

    # ---------------------------------------------------------------
    # 0. Determine a starting distribution.
    # ---------------------------------------------------------------
    init_global = globals().get("height_sequence_1", None)
    if isinstance(init_global, (list, tuple, np.ndarray)):
        init_global = np.asarray(init_global, dtype=np.float64)
    if init_global is not None and init_global.size > 0:
        a0 = np.maximum(init_global, 0.0)
        a0_sum = a0.sum()
        if a0_sum == 0.0:
            a0 = np.full_like(a0, 1.0 / a0.shape[0])
        else:
            a0 /= a0_sum
        n0 = max(2000, a0.shape[0])
        print(f"[Init] Using supplied global sequence, length = {n0}")
    else:
        n0 = 20000
        a0 = _power_law_distribution(n0, 0.60)
        print("[Init] No global sequence – using power‑law start")

    # ---------------------------------------------------------------
    # 1. Run optimisation on the initial support size.
    # ---------------------------------------------------------------
    best_seq, best_score = _run_optimizer(a0, deadline, max_n=n0)

    # ---------------------------------------------------------------
    # 2. If time permits, enlarge the support (up to 30 000) and re‑optimise.
    # ---------------------------------------------------------------
    now = time.time()
    if now < deadline - 5.0:
        n_max = 30000
        if best_seq.shape[0] < n_max:
            a_big = _interpolate_distribution(best_seq, n_max)
            print(f"[Phase] Expanding support to {n_max}")
            seq_big, score_big = _run_optimizer(a_big, deadline, max_n=n_max)
            if score_big < best_score:
                best_seq, best_score = seq_big, score_big

    # ---------------------------------------------------------------
    # 3. Optional short random restart (helps avoid unlucky local minima).
    # ---------------------------------------------------------------
    now = time.time()

        # Allocate a modest number of iterations for the restart.

        if score_rand < best_score:

    # ---------------------------------------------------------------
    # 4. Final output (the external platform will evaluate it).
    # ---------------------------------------------------------------
    final_seq = best_seq.tolist()
    print(f"[Result] Best cheap score = {best_score:.6f}")
    try:

        print(f"[Result] Exact evaluation = {exact:.6f}")
    except Exception as exc:                     # pragma: no cover
        print(f"[Result] Evaluation error: {exc}")

    return final_seq
