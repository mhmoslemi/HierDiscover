"""
Propose a non‑negative step function (list of heights) that minimises

    O = 2·n·max(conv(a,a)) / (sum(a)²)

The algorithm works in three phases (under a 1000 s budget)

1. **Broad parametric sweep** – a few analytically‑motivated families
   (power‑law, Beta‑shaped, half‑sine) are sampled on a coarse grid.
   The best‑scoring vector is kept.

2. **Stochastic local refinement** – the current best vector is jittered
   with Gaussian noise, projected back onto the probability simplex and
   kept if it improves the true objective.  A few thousand cheap trials
   are performed.

3. **Smooth‑objective mirror descent** – we minimise a smoothed version
   of the max‑convolution using an Adam‑style optimiser.
   The smoothing is a soft‑max (log‑sum‑exp) with a temperature that
   grows from β≈0.5 to β≈200, so the optimisation gradually focuses on
   the true supremum.  Gradients are obtained analytically via a
   cross‑correlation computed with FFTs; after each step the iterate is
   projected onto the simplex (ensuring non‑negativity and sum=1).

The routine respects the time limit, prints occasional progress messages
and finally returns the best sequence found (as a plain ``list[float]``).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # fallback to uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a power‑of‑two FFT."""
    n = a.size
    size = 1 << ((2 * n - 2).bit_length())      # next power of two ≥ 2n‑1
    A = np.fft.fft(a, n=size)
    conv = np.fft.ifft(A * A).real[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def _objective_norm1(a: np.ndarray) -> float:
    """Objective 2·n·max(conv) assuming Σa = 1."""
    n = a.size
    conv = _conv_fft(a)
    return float(2.0 * n * np.max(conv))

# ---------- families of parametric shapes --------------------------------
def _shape_power(alpha: float, beta: float, n: int) -> np.ndarray:
    """Power‑law a_i ∝ (i+β)^(-α), i = 1..n, normalised to sum‑1."""
    i = np.arange(1, n + 1, dtype=float)
    a = (i + beta) ** (-alpha)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _shape_beta(alpha: float, beta: float, n: int) -> np.ndarray:
    """Beta‑shaped a_i ∝ i^{α‑1}·(n‑i+1)^{β‑1}, i = 1..n, normalised."""
    i = np.arange(1, n + 1, dtype=float)
    a = (i ** (alpha - 1.0)) * ((n + 1 - i) ** (beta - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _shape_sine(p: float, n: int) -> np.ndarray:
    """Half‑sine a_i ∝ sin(π·(i‑½)/n)^p, i = 1..n, normalised."""
    i = np.arange(0, n, dtype=float) + 0.5               # i = 0.5 … n‑0.5
    a = np.sin(np.pi * i / n) ** p
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _shape_power_exp(alpha: float, beta: float, gamma: float, n: int) -> np.ndarray:
    """Power‑law with exponential tail a_i ∝ (i+β)^(-α)·exp(-γ·(i‑1)/n)."""
    i = np.arange(1, n + 1, dtype=float)

    s = a.sum()

    else:

def _shape_beta_exp(alpha: float, beta: float, gamma: float, n: int) -> np.ndarray:
    """Beta‑shape with exponential tail."""
    i = np.arange(1, n + 1, dtype=float)
    a = (i ** (alpha - 1.0)) * ((n + 1 - i) ** (beta - 1.0)) * np.exp(-gamma * (i - 1) / n)
    s = a.sum()

# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max of β·conv (stable version)."""
    z = beta * conv
    m = np.max(z)
    exp_z = np.exp(z - m)
    return exp_z / exp_z.sum()

def _grad_softmax(a: np.ndarray, w: np.ndarray, L: int) -> np.ndarray:
    """
    Gradient of the smoothed objective F(a)= (1/β)·logΣexp(β·conv).
    It equals 2·(a ⋆ w) where ⋆ denotes linear cross‑correlation.
    Computed via FFT of length L ≥ 3·n‑2.
    """
    n = a.shape[0]
    A = np.fft.fft(a, n=L)
    W = np.fft.fft(w, n=L)
    # cross‑correlation: conj(A) * W
    G = 2.0 * np.conj(A) * W
    g_full = np.fft.ifft(G).real
    # first n entries correspond to lags 0..n‑1
    return g_full[:n]


# ----------------------------------------------------------------------
def _optimize_smooth(a0: np.ndarray,
                     max_iters: int,
                     deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam‑style optimisation of the smoothed max‑convolution.
    Returns the best vector seen and its true objective value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective_norm1(a)

    n = a.size
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # schedules
    beta_start, beta_end = 0.5, 200.0   # smoothing temperature
    lr_start, lr_end = 0.5, 0.01       # learning‑rate decay

    # FFT length for the cross‑correlation (linear, no wrap‑around)
    L = 1 << ((3 * n - 2).bit_length())

    no_improve = 0
    patience = max(500, max_iters // 30)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- smoothing parameter (soft‑max temperature) -----
        beta = beta_start + (beta_end - beta_start) * (t / max_iters)

        # ----- current convolution and soft‑max weights -----
        conv = _conv_fft(a)                     # length 2n‑1
        w = _softmax_weights(conv, beta)        # same length

        # ----- gradient of the smoothed objective -----
        g = _grad_softmax(a, w, L)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr_start + (lr_end - lr_start) * (t / max_iters)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- maintain feasibility (non‑negative, sum = 1) -----
        a = _project_to_simplex(a)

        # ----- true objective evaluation -----
        cur_score = _objective_norm1(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            print(f"[opt] iter {t:6d}  new best O={best_score:.6f}")
        else:
            no_improve += 1
            if no_improve >= patience:
                # small random kick to escape plateaus
                jitter = np.random.randn(n) * 1e-4
                a = best_a + jitter
                a = np.clip(a, 0.0, None)
                a = _project_to_simplex(a)
                no_improve = 0

    return best_a, best_score

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the supplied
    ``evaluate_sequence`` bound.  Returns the sequence as a plain list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Length of the candidate (power‑of‑two works well with FFT)
    n = 4096

    best_seq = None
    best_score = float("inf")
    # ------------------------------------------------------------------
    # 1️⃣  Coarse grid search over several families
    print("[grid] starting parametric sweep")

    # ---- power‑law family ----
    alphas = np.linspace(0.35, 0.65, 31)   # 0.35 … 0.65
    betas = np.linspace(0.0, 20.0, 31)    # 0 … 20
    for a in alphas:
        for b in betas:
            seq = _shape_power(a, b, n)
            score = _objective_norm1(seq)
            if score < best_score:
                best_score, best_seq = score, seq
                best_alpha, best_beta = a, b
    print(f"[grid] power‑law best O={best_score:.6f} (α={best_alpha:.4f}, β={best_beta:.4f})")

    # ---- Beta‑shape family (symmetric & asymmetric) ----
    alphas_beta = np.linspace(0.2, 3.0, 21)
    betas_beta = np.linspace(0.2, 3.0, 21)
    for a in alphas_beta:
        for b in betas_beta:
            seq = _shape_beta(a, b, n)
            score = _objective_norm1(seq)
            if score < best_score:
                best_score, best_seq = score, seq
                best_alpha, best_beta = a, b
    print(f"[grid] beta‑shape best O={best_score:.6f} (α={best_alpha:.4f}, β={best_beta:.4f})")

    # ---- half‑sine family ----
    p_vals = np.linspace(0.2, 5.0, 21)
    for p in p_vals:
        seq = _shape_sine(p, n)
        score = _objective_norm1(seq)
        if score < best_score:
            best_score, best_seq = score, seq
            best_p = p
    print(f"[grid] sine‑power best O={best_score:.6f} (p={best_p:.4f})")

    # ------------------------------------------------------------------
    # 2️⃣  Local random refinement (perturb the current best vector)
    print("[rand] starting local random refinement")
    for _ in range(4000):
        if time.time() >= deadline:
            break
        # Gaussian jitter, then re‑project onto the simplex
        cand = best_seq + np.random.randn(n) * 0.02
        cand = np.clip(cand, 0.0, None)
        cand = _project_to_simplex(cand)
        score = _objective_norm1(cand)
        if score < best_score:
            best_score, best_seq = score, cand
            print(f"[rand] improvement O={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Incorporate a previously obtained construction if present
    if 'height_sequence_1' in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=float)
            if prev.size > 0:
                prev = np.maximum(prev, 0.0)
                prev = _project_to_simplex(prev)
                prev_score = _objective_norm1(prev)
                if prev_score < best_score:
                    best_seq, best_score = prev, prev_score
                    print(f"[prev] using previous construction O={best_score:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 4️⃣  Smoothing‑objective Adam optimisation (the heavy work)
    remaining = deadline - time.time()
    if remaining > 5.0:
        # estimate possible iterations (≈150‑200 iters per second on a modest VM)
        max_iters = int(min(200_000, remaining * 180))
        print(f"[opt] polishing – max_iters={max_iters}")
        a_opt, score_opt = _optimize_smooth(best_seq, max_iters, deadline)
        if score_opt < best_score:
            best_seq, best_score = a_opt, score_opt
            print(f"[opt] polished best O={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Final evaluation with the provided ``evaluate_sequence``.
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:  # pragma: no cover
        print("[final] evaluate_sequence call failed:", e)
        final_score = None

    return best_seq.tolist()
