        w = np.exp(shifted)

        dot = np.dot(a, g_a)

        m = beta1 * m + (1 - beta1) * g_z
        v = beta2 * v + (1 - beta2) * (g_z * g_z)
        m_hat = m / (1 - beta1 ** iteration)
        v_hat = v / (1 - beta2 ** iteration)
        lr = 0.05 / (1 + 0.0001 * iteration)  # decaying learning rate
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

    final_seq = best_a.tolist()
    try:

        print(f"Final exact score = {final_score:.6f}")
    except Exception as e:

#!/usr/bin/env python
"""
Propose a non‑negative step function (list of heights) that minimises

        2·n·max_k (a*a)[k] / (Σ a_i)²

The algorithm works entirely in the probability simplex (Σ a_i = 1) and
optimises a smooth approximation of the objective

    F(a) = log Σ_k exp(α·(a*a)[k])          (α → ∞ recovers the max)

using Adam on the log‑softmax parametrisation
    a_i = exp(z_i) / Σ_j exp(z_j).

A slowly increasing smoothing parameter α focuses the optimisation on the
largest convolution entries.  Convolutions and correlations are computed
with a real FFT (length a power of two ≥ 3·n) which makes each iteration
≈ O(n log n).  The best sequence found during the time budget is returned
and its exact score is printed via ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Top‑level helpers (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable softmax: returns a probability vector that sums to 1."""
    m = np.max(z)
    e = np.exp(z - m)
    return e / e.sum()


def _interpolate_sequence(old_seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linearly up‑sample a sequence to a new length and renormalise."""
    old_len = old_seq.shape[0]
    if old_len == new_len:
        return old_seq / old_seq.sum()
    old_x = np.linspace(0.0, 1.0, old_len, dtype=np.float64)
    new_x = np.linspace(0.0, 1.0, new_len, dtype=np.float64)
    up = np.interp(new_x, old_x, old_seq)
    s = up.sum()
    return up / s if s > 0 else up


def _initial_sequence(seed: int, n: int) -> np.ndarray:
    """
    Produce an initial feasible sequence of length n.
    Prefer the globally available ``height_sequence_1`` if possible,
    otherwise fall back to a power‑law shape.
    """
    np.random.seed(seed)
    # Try to reuse a previous construction (if the variable exists)
    try:
        prev = height_sequence_1               # type: ignore[name-defined]
        prev_arr = np.asarray(prev, dtype=np.float64)
        if prev_arr.size > 0 and prev_arr.sum() > 0:
            # up‑sample / down‑sample to the target length
            a0 = _interpolate_sequence(prev_arr, n)
            return a0
    except Exception:
        pass

    # No good previous sequence – use a power‑law seed
    p = 0.65
    i = np.arange(1, n + 1, dtype=np.float64)
    a0 = i ** (-p)
    a0 /= a0.sum()
    return a0


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Optimise a step‑function using smoothed‑max gradient descent on the
    simplex (Adam on the log‑softmax parametrisation).  The routine respects
    the given time budget and always returns a feasible (non‑negative,
    non‑zero) sequence.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a few seconds for the final exact evaluation
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # Choose a working length n (5k ≤ n ≤ 25k) and initialise a.
    # --------------------------------------------------------------
    # Prefer a relatively large n – more degrees of freedom usually help.
    n = 20000
    a = _initial_sequence(seed, n)               # a already normalised (sum = 1)

    # --------------------------------------------------------------
    # Pre‑compute constant sizes for FFTs.
    # --------------------------------------------------------------
    # Length needed for linear convolution of a_rev (size n) and w (size 2n‑1)
    L = _next_pow2(3 * n - 2)

    # Pre‑allocate temporary buffers to avoid repeated allocation.
    pad_a = np.empty(L, dtype=np.float64)
    pad_arev = np.empty(L, dtype=np.float64)
    pad_w = np.empty(L, dtype=np.float64)

    # Adam state
    z = np.log(a + 1e-12)                         # log‑softmax parameters
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.90
    beta2 = 0.999
    eps_adam = 1e-8

    # Tracking the best solution
    best_a = a.copy()
    # cheap version of the objective (exact when sum=1)
    #   cheap = 2·n·max_k (a*a)[k]
    # compute once for the initial a
    pad_a.fill(0)
    pad_a[:n] = a
    A = np.fft.rfft(pad_a)                       # real FFT of a
    conv_full = np.fft.irfft(A * A, n=L)
    conv = conv_full[: 2 * n - 1]
    best_val = 2.0 * n * conv.max()
    print(f"[Init] n={n}, initial cheap score = {best_val:.6f}")

    # --------------------------------------------------------------
    # Optimisation loop – smoothed max via Adam on the simplex.
    # --------------------------------------------------------------
    iteration = 0
    last_report = start_time
    no_improve = 0
    RESTART_AFTER = 200_000                      # iterations without progress

    while True:
        now = time.time()
        if now >= deadline:
            break
        iteration += 1

        # ----- 1) softmax to obtain current a (sum = 1, a_i ≥ 0) -----
        a = _softmax(z)

        # ----- 2) full autocorrelation a*a (linear convolution) -----
        pad_a.fill(0)
        pad_a[:n] = a
        A = np.fft.rfft(pad_a)
        conv_full = np.fft.irfft(A * A, n=L)
        conv = conv_full[: 2 * n - 1]

        conv_max = conv.max()
        cheap = 2.0 * n * conv_max

        # ----- 3) keep the best sequence seen so far -----
        if cheap + 1e-12 < best_val:
            best_val = cheap
            best_a = a.copy()
            no_improve = 0
            print(f"[Iter {iteration:07d}] NEW BEST cheap = {best_val:.6f}")
        else:
            no_improve += 1

        # ----- 4) smoothing parameter α (linear schedule) -----
        elapsed_frac = (now - start_time) / (deadline - start_time)
        alpha = 2.0 + 28.0 * elapsed_frac          # from 2 → 30

        # ----- 5) build the weight vector w (soft‑max over conv) -----
        # shift by conv_max to avoid overflow
        shifted = alpha * (conv - conv_max)
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            wnorm = np.full_like(w, 1.0 / w.size)
        else:
            wnorm = w / w_sum

        # ----- 6) gradient of the smoothed objective w.r.t. a -----
        #   g_a = 2 * (wnorm ⋆ a)   (⋆ = correlation)
        pad_w.fill(0)
        pad_w[: 2 * n - 1] = wnorm
        pad_arev.fill(0)
        pad_arev[:n] = a[::-1]

        W = np.fft.rfft(pad_w)
        Arev = np.fft.rfft(pad_arev)
        corr_full = np.fft.irfft(Arev * W, n=L)    # length L, real
        g_a = 2.0 * corr_full[n - 1 : n - 1 + n]   # slice the central n entries

        # ----- 7) gradient w.r.t. the softmax parameters z -----
        dot = np.dot(a, g_a)                      # scalar = a·g_a
        g_z = a * (g_a - dot)                     # ∂F/∂z_i = a_i (g_a_i - a·g_a)

        # ----- 8) Adam update on z -----
        m = beta1 * m + (1.0 - beta1) * g_z
        v = beta2 * v + (1.0 - beta2) * (g_z * g_z)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        # simple decaying learning‑rate schedule
        lr = 0.30 / (1.0 + 0.0001 * iteration)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- 9) occasional progress report -----
        if iteration % 5000 == 0:
            elapsed = now - start_time
            print(f"[Iter {iteration:07d}] time {elapsed:.1f}s, best cheap {best_val:.6f}")

        # ----- 10) restart heuristic to escape stagnation -----
        if no_improve >= RESTART_AFTER:
            # perturb the best solution slightly and re‑initialise Adam
            perturb = best_a * (1.0 + 0.02 * np.random.randn(n))
            perturb = np.clip(perturb, 0.0, None)
            if perturb.sum() == 0.0:
                perturb = best_a.copy()
            z = np.log(perturb + 1e-12)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[Iter {iteration:07d}] Restart – small random perturbation applied.")

    # --------------------------------------------------------------
    # Final exact evaluation (uses the provided ``evaluate_sequence``)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:                     # defensive – should never happen
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # --------------------------------------------------------------
    # Return the best sequence found.
    # --------------------------------------------------------------
    return final_seq
