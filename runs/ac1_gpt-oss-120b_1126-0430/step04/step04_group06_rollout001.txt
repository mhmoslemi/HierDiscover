#!/usr/bin/env python
"""
Smooth‑max convolution optimizer.

Implements a mirror‑descent (Adam) optimizer on the simplex
parameterised by logits `z` (softmax → a).  The objective is the
smoothed max of the self‑convolution `c = a*a`:

    L(a) = (1/θ)·log Σ_i exp(θ·c_i) ,

which approaches the true worst‑case `max(c)` as θ grows.
The gradient of L w.r.t. a can be written in closed form:

    ∂L/∂a = 2·(w * a_rev)[n‑1 : n‑1+n] ,

where `w` are the softmax weights of `θ·c` and `a_rev` is a
reversed copy of `a`.  This gradient is computed with two real FFTs
per iteration (O(n log n)).  A linear schedule raises θ from a
small smoothing constant (≈0.5) to a large value (≈2·10⁵) so the
optimizer ultimately focuses on the true maximum.  The support size
is gradually increased (up to 30 000) by linear interpolation of the
best distribution found so far.

The routine respects a time budget (~1000 s) and returns the best
sequence (list of non‑negative heights) discovered.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable softmax, returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it representation that yields `a` after softmax."""
    return np.log(a + eps)


def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two ≥ 2·n‑1 (efficient size for real FFT)."""
    return 1 << ((2 * n - 1).bit_length())


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full linear self‑convolution a*a using an FFT of length `fft_len`.
    Returns an array of length 2·len(a)‑1.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.size - 1]
    return conv


def _smooth_max_gradient(a: np.ndarray, theta: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Compute the smoothed max‑convolution value and its gradient w.r.t. `a`
    using the closed‑form expression.

    Returns
    -------
    conv : np.ndarray
        Full self‑convolution a*a (length 2·n‑1).
    grad_a : np.ndarray
        Gradient of (1/θ)·log‑sum‑exp(θ·conv) w.r.t. a (length n).
    """
    n = a.shape[0]

    # 1) self‑convolution a*a
    fft_len_c = _fft_len_for_n(n)
    conv = _conv_full_fft(a, fft_len_c)

    # 2) softmax weights w_i ∝ exp(θ·conv_i)
    scaled = theta * conv
    max_scaled = scaled.max()
    exps = np.exp(scaled - max_scaled)          # stability
    w = exps / exps.sum()

    # 3) gradient ∂L/∂a = 2·(w * a_rev)[n‑1 : n‑1+n]
    a_rev = a[::-1]

    # length needed for linear convolution of a_rev (len=n) and w (len=2n‑1)
    L_needed = len(w) + n - 1                     # = 3·n‑2
    fft_len_g = 1 << (L_needed.bit_length())      # next power of two
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    start = n - 1
    grad_a = 2.0 * conv_rev[start:start + n]      # length n
    return conv, grad_a


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    total = new_a.sum()
    if total == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= total
    return new_a


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """Normalised power‑law a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) beats the
    current best bound.  The routine respects the given time budget
    (≈1000 s) and uses a smoothed‑max convolution Adam optimiser on the
    probability simplex.
    """
    np.random.seed(seed)
    start = time.time()
    # keep a small safety margin for the final evaluate()
    deadline = start + budget_s - 2.0

    # ---------------------------------------------------------------
    # 0. Choose a working support size.
    # ---------------------------------------------------------------
    global_seq = globals().get("height_sequence_1", None)
    if global_seq is not None:
        init_a = np.array(global_seq, dtype=np.float64)
        n = max(2000, init_a.shape[0])
        use_global = True
        print(f"[Init] Using supplied global sequence, length = {n}")
    else:
        init_a = None
        n = 20000                     # default initial size
        use_global = False
        print(f"[Init] No global sequence – starting with length = {n}")

    n_max = 30000                     # hard cap on the support size

    # ---------------------------------------------------------------
    # 1. Initialise a probability vector a (sum = 1).
    # ---------------------------------------------------------------
    if use_global and init_a is not None:
        a = np.maximum(init_a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / a.shape[0])
        else:
            a /= s
    else:
        # a modest power‑law is a good generic start.
        a = _power_law_distribution(n, 0.60)
        print("[Init] Power‑law start with p = 0.60")

    # ---------------------------------------------------------------
    # 2. Log‑it parametrisation + Adam state.
    # ---------------------------------------------------------------
    eps = 1e-16
    z = _logits_from_distribution(a, eps)

    m = np.zeros_like(z)                # 1st moment
    v = np.zeros_like(z)                # 2nd moment
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    base_alpha = 0.25                   # base learning‑rate

    # ---------------------------------------------------------------
    # 3. Optimisation loop.
    # ---------------------------------------------------------------
    best_seq = a.copy()
    best_score = float("inf")           # cheap_score = 2·n·max(conv)
    it = 0
    last_improve = 0
    report_every = 2500

    # schedule for the smoothing temperature θ
    theta_start = 0.5
    theta_end = 2.0e5                    # large enough to approximate the hard max
    budget = budget_s

    while time.time() < deadline and it < 2_000_000:
        it += 1

        # probability vector from the current logits
        a = _softmax(z)
        n = a.shape[0]

        # elapsed time & temperature
        elapsed = time.time() - start
        theta = theta_start + (theta_end - theta_start) * min(1.0, elapsed / budget)

        # full convolution and gradient of the smoothed max
        conv, grad_a = _smooth_max_gradient(a, theta)

        max_conv = float(conv.max())
        cheap_score = 2.0 * n * max_conv

        if cheap_score < best_score:
            best_score = cheap_score
            best_seq = a.copy()
            last_improve = it
            if it % report_every == 0:
                print(f"[Iter {it:07d}] NEW BEST cheap = {cheap_score:.6f}")

        # -----------------------------------------------------------
        # Mirror‑descent gradient on logits (Adam)
        # -----------------------------------------------------------
        mu = float(np.dot(grad_a, a))
        grad_z = a * (grad_a - mu)

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step_dir = - m_hat / (np.sqrt(v_hat) + eps_adam)

        # inverse‑sqrt decay of the step‑size
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)

        # update logits and keep them numerically stable
        z = z + alpha_t * step_dir
        np.clip(z, -80.0, 80.0, out=z)

        # -----------------------------------------------------------
        # Occasional random perturbations (helps escape plateaus)
        # -----------------------------------------------------------
        if it % 5000 == 0 and np.random.rand() < 0.25:
            perturb = 0.02 * np.random.randn(n)
            z += perturb
            m *= 0.5
            v *= 0.5

        # -----------------------------------------------------------
        # Progressive enlargement of the support (if we have time)
        # -----------------------------------------------------------
        if (it % 25000 == 0) and (n < n_max) and (time.time() + 20.0 < deadline):
            new_n = min(n_max, n + 5000)
            print(f"[Iter {it:07d}] Expanding support: {n} → {new_n}")
            a_interp = _interpolate_distribution(best_seq, new_n)
            z = _logits_from_distribution(a_interp, eps)
            m = np.zeros_like(z)
            v = np.zeros_like(z)
            n = new_n
            # reset counters for the new problem size
            it = 0
            last_improve = 0
            # recompute a cheap_score for the enlarged support
            conv, _ = _smooth_max_gradient(a_interp, theta_start)  # cheap preview
            best_score = 2.0 * n * conv.max()
            best_seq = a_interp.copy()
            continue

        # -----------------------------------------------------------
        # Early stop if stagnation after reaching maximal size
        # -----------------------------------------------------------
        if n >= n_max and it - last_improve > 300_000:
            print(f"[Iter {it:07d}] Stagnation – early exit.")
            break

    # ---------------------------------------------------------------
    # 4. Final exact evaluation (the platform will also evaluate)
    # ---------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[Result] Final exact score = {final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        final_score = float("inf")
        print(f"[Result] Evaluation error: {exc}")

    # make the best sequence available for possible subsequent runs
    globals()["height_sequence_1"] = final_seq

    # ---------------------------------------------------------------
    # 5. Return the best sequence found.
    # ---------------------------------------------------------------
    return final_seq
