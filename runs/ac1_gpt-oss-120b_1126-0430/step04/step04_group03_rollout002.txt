def shape_product_power(alpha: float, beta: float, n: int) -> np.ndarray:
    i = np.arange(1, n+1, dtype=float)
    a = ( (i + beta) * (n - i + 1 + beta) ) ** (-alpha)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a

"""
propose_candidate
-----------------

Search for a non‑negative step function (list of heights) that minimises

    O = 2·n·max(conv(a,a)) / (sum(a)²)

The algorithm proceeds in four stages:

1. **Coarse grid search** over three families of analytic shapes  
   – simple power law, – edge‑heavy product power law, – product
   power law with an exponential tail.  
   Each candidate is normalised to Σa = 1 and its objective is
   evaluated by a fast FFT based convolution.

2. **Random local perturbations** of the current best vector.  
   Small Gaussian noise is added, the vector is projected back onto the
   probability simplex and kept if it improves the bound.

3. **Mirror‑descent polishing** using exponentiated‑gradient updates
   on the simplex.  The true sub‑gradient of `max(conv(a,a))` is used,
   a linearly decaying learning rate is applied and occasional jitter
   avoids flat plateaus.  The polishing runs for the remaining time
   budget.

4. **Exact evaluation** with the supplied ``evaluate_sequence`` function.

All steps respect the overall 1000 s deadline.  The returned list
contains non‑negative floats and can be fed directly to
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a power‑of‑two FFT."""
    n = a.size
    size = 1 << ((2 * n - 2).bit_length())  # next power of two ≥ 2n‑1
    A = np.fft.fft(a, n=size)
    conv = np.fft.ifft(A * A).real[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def _objective_norm1(a: np.ndarray) -> float:
    """
    Objective 2·n·max(conv) assuming Σa = 1.
    The function works directly on the (already normalised) vector.
    """
    n = a.size
    conv = _conv_fft(a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) w.r.t. a on the simplex.
    Returns a vector of the same shape as ``a``.
    """
    n = a.size
    conv = _conv_fft(a)
    max_val = conv.max()
    # tolerance for “active” indices (relative to max)
    tol = max(1e-12, max_val * 1e-12)
    ks = np.where(conv >= max_val - tol)[0]

    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        # each pair (i,j) appears twice in the sum, hence the factor 2
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
def _exp_update(a: np.ndarray, g: np.ndarray, eta: float) -> np.ndarray:
    """
    Exponentiated‑gradient update on the simplex.
    Operates in log‑space for numerical stability.
    """
    log_a = np.log(a + 1e-15)          # avoid log(0)
    log_a -= eta * g
    log_a -= np.max(log_a)            # shift to keep numbers moderate
    a_new = np.exp(log_a)
    a_new /= a_new.sum()
    return a_new

def _optimize(a0: np.ndarray,
              eta0: float,
              deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) optimisation.
    Runs until ``deadline`` is reached.
    Returns the best vector seen and its objective value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective_norm1(a)

    n = a.size
    no_improve = 0
    patience = 2000                     # number of iterations without progress
    iteration = 0

    while time.time() < deadline:
        g = _subgradient_max_conv(a)

        # linearly decaying learning rate
        eta = eta0 * max(0.0, 1.0 - iteration / 300_000)

        a = _exp_update(a, g, eta)

        cur_score = _objective_norm1(a)
        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            print(f"[opt] iter {iteration:6d}  new best O={best_score:.6f}")
        else:

                a = np.clip(a, 0.0, None)
                a = _project_to_simplex(a)
                no_improve = 0
        iteration += 1

    return best_a, best_score


# ----------------------------------------------------------------------
def _shape_power(alpha: float, beta: float, n: int) -> np.ndarray:
    """Monotone decreasing power‑law shape: a_i ∝ (i+β)^(-α) (i=1..n)."""
    i = np.arange(1, n + 1, dtype=float)
    a = (i + beta) ** (-alpha)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
def _shape_product(alpha: float, beta: float, n: int) -> np.ndarray:
    """
    Edge‑heavy product power‑law:
        a_i ∝ ((i+β)*(n+1-i+β))^(-α) ,  i = 1..n
    Symmetric and decreasing towards the centre.
    """
    i = np.arange(1, n + 1, dtype=float)
    a = ((i + beta) * (n + 1 - i + beta)) ** (-alpha)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
def _shape_product_exp(alpha: float, beta: float, gamma: float, n: int) -> np.ndarray:
    """
    Product power‑law with an exponential tail:
        a_i ∝ ((i+β)*(n+1-i+β))^(-α) * exp(-γ·(i‑1)/n)
    """
    i = np.arange(1, n + 1, dtype=float)
    a = ((i + beta) * (n + 1 - i + beta)) ** (-alpha)
    a *= np.exp(-gamma * (i - 1) / n)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a

    if remaining > 5.0:
        print(f"[opt] polishing – eta0=0.6, remaining≈{remaining:.1f}s")
        a_opt, score_opt = _optimize(best_seq, eta0=0.6, deadline=deadline)
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt
            print(f"[opt] polished best O={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Final exact evaluation with the provided checker.
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)
        final_score = None

    # Return a plain Python list as required.
    return best_seq.tolist()
