"""
Optimisation of a non‑negative step function to minimise

    O(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm works in three phases on a single support size (default 6000):
  1. Adam optimisation on the log‑it (soft‑max) parametrisation with an
     exponentially annealed soft‑max approximation of the max‑convolution.
  2. A short exact‑gradient phase that descends directly on the true max‑shift.
  3. Simulated‑annealing style local search (random mass transfers) for fine‑tuning.

The sequence is kept normalised to ``sqrt(2·n)`` – for this normalisation the
objective equals the maximum entry of the self‑convolution, which the optimiser
minimises directly.  The final list is returned; ``evaluate_sequence`` will
re‑scale it automatically (its result equals the objective we have minimised).

The implementation respects the 1000 s budget, prints progress and
automatically falls back to a simple uniform distribution if the budget is
exhausted too early.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (real, FFT‑based)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """Correlation  g_i = Σ_t w[i+t]·a[t]  (FFT implementation)."""
    n = a.shape[0]
    m = w.shape[0]                     # = 2n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


# ----------------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


# ----------------------------------------------------------------------
def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return _conv_self_fft(a).max()


# ----------------------------------------------------------------------
def _adam_phase(a: np.ndarray,
                target_sum: float,
                time_limit: float,
                rng: np.random.Generator,
                seed: int = 0) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Adam optimisation on the log‑it representation.
    Returns (best_a, final_a, best_obj).
    """
    n = a.size
    # initialise logits
    x = np.log(a + 1e-16)

    m = np.zeros_like(x)          # Adam first moment
    v = np.zeros_like(x)          # Adam second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # schedule parameters
    beta_start, beta_end = 0.5, 8000.0
    lr_start, lr_end = 0.20, 0.001
    max_iters = 500_000

    best_a = a.copy()
    best_obj = _max_conv(best_a)

    start = time.time()
    it = 0
    while time.time() - start < time_limit and it < max_iters:
        it += 1

        # probabilities from logits and renormalisation to target_sum
        a = _softmax(x) * target_sum

        # exact convolution and objective
        conv = _conv_self_fft(a)
        M = conv.max()
        if M < best_obj:
            best_obj = M
            best_a = a.copy()
            print(f"[Adam n={n:5d} it={it:06d}] new best max‑conv={best_obj:.6e}")

        # schedule
        t = min(it / max_iters, 1.0)
        beta = beta_start * np.exp(np.log(beta_end / beta_start) * t)
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t)

        # smoothed max – soft‑max weights over convolution entries
        if beta > 4000.0:                     # almost a hard max
            kmax = int(np.argmax(conv))
            w = np.zeros_like(conv)
            w[kmax] = 1.0
        else:
            shifted = conv - M                 # ≤0
            w = np.exp(beta * shifted)
            w /= w.sum()

        # gradient of the smooth max (in the a‑space)
        g_a = 2.0 * _correlation_fft(w, a)    # shape (n,)

        # natural‑gradient transform to log‑it space
        dot = np.dot(a, g_a)                  # scalar ⟨a,g_a⟩
        g_x = a * (g_a - dot)                 # shape (n,)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to avoid plateaus
        if it % 5000 == 0:
            jitter = rng.normal(scale=0.02, size=n)
            x += jitter

        # hard restart from the current best every now and then
        if it % 70_000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.005, size=n)
            m.fill(0.0)
            v.fill(0.0)
            print(f"[Adam n={n:5d}] restart at iter {it}")

    # final a after Adam (may be different from best_a)
    final_a = _softmax(x) * target_sum
    return best_a, final_a, best_obj


# ----------------------------------------------------------------------
def _exact_gradient_phase(a: np.ndarray,
                          target_sum: float,
                          time_limit: float,
                          rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Gradient descent using the *true* max‑shift (no smoothing).
    Returns (best_a, best_obj).
    """
    n = a.size
    best_a = a.copy()
    best_obj = _max_conv(best_a)

    start = time.time()
    it = 0
    while time.time() - start < time_limit:
        it += 1
        conv = _conv_self_fft(a)
        M = conv.max()
        k_max = int(np.argmax(conv))

        # gradient of the max‑convolution (exact)
        g = np.zeros_like(a)
        # for each i we need a_{k_max-i} if that index exists
        idx = np.arange(n)
        j = k_max - idx
        mask = (j >= 0) & (j < n)
        g[mask] = 2.0 * a[j[mask]]

        # simple decreasing step size
        lr = 0.02 / (1.0 + it * 1e-4)   # starts at 0.02, slowly decays
        a_new = a - lr * g
        a_new = np.maximum(a_new, 0.0)

        # re‑normalise to the fixed sum
        s = a_new.sum()
        if s == 0.0:                     # can happen only if everything vanished
            a_new = np.full_like(a, target_sum / n)
        else:
            a_new *= target_sum / s

        a = a_new
        M_new = _max_conv(a)

        if M_new < best_obj:
            best_obj = M_new
            best_a = a.copy()
            print(f"[Exact n={n:5d} it={it:06d}] new best max‑conv={best_obj:.6e}")

    return best_a, best_obj


# ----------------------------------------------------------------------
def _simulated_annealing_phase(a: np.ndarray,
                               target_sum: float,
                               time_limit: float,
                               rng: np.random.Generator,
                               current_best_obj: float) -> tuple[np.ndarray, float]:
    """
    Very simple simulated‑annealing hill‑climb using random mass transfers.
    Returns (best_a, best_obj).
    """
    n = a.size
    best_a = a.copy()
    best_obj = current_best_obj

    start = time.time()
    # Annealing schedule: linear cooling
    while time.time() - start < time_limit:
        # temperature proportional to remaining time
        elapsed = time.time() - start
        frac = max(0.0, 1.0 - elapsed / time_limit)
        T = best_obj * 0.02 * max(frac, 0.01)   # a small absolute temperature

        # propose a random transfer
        i, j = rng.choice(n, size=2, replace=False)
        max_transfer = 0.25 * a[i]               # move up to 25 % of the source mass
        if max_transfer <= 0.0:
            continue
        delta = rng.random() * max_transfer

        a_prop = a.copy()
        a_prop[i] -= delta
        a_prop[j] += delta

        # sum stays constant, only need the max‑convolution
        M_prop = _max_conv(a_prop)

        if M_prop < best_obj:
            # accept improvement
            a = a_prop
            best_obj = M_prop
            best_a = a_prop.copy()
            print(f"[SA] improved to {best_obj:.6e}")
        else:
            # possibly accept a worsening move
            if rng.random() < np.exp(-(M_prop - best_obj) / (T + 1e-15)):
                a = a_prop

    return best_a, best_obj


# ----------------------------------------------------------------------
def _run_optimisation(n: int,
                      time_budget: float,
                      init_a: np.ndarray | None = None,
                      seed: int = 0) -> tuple[list[float], float]:
    """
    Full optimisation pipeline for a given support size ``n``.
    Returns (best_sequence, exact_objective).
    """
    rng = np.random.default_rng(seed)
    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # initialise a (non‑negative, correct sum)
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            # pad / truncate to the required length
            if a.size < n:
                pad = rng.random(n - a.size)
                a = np.concatenate([a, pad])
            else:
                a = a[:n]
    else:
        a = rng.random(n)

    a = np.maximum(a, 0.0)
    a = a * target_sum / a.sum()

    # ------------------------------------------------------------------
    # split the allocated budget
    t_start = time.time()
    t_adam = time_budget * 0.55
    t_exact = time_budget * 0.30
    t_sa = max(0.0, time_budget - t_adam - t_exact)

    # ---------- Adam phase ----------
    best_a_adam, a_adam, best_obj = _adam_phase(a, target_sum, t_adam, rng, seed)

    # optional compression – drop negligible entries to speed later phases
    thresh = 1e-12
    mask = best_a_adam > thresh
    if mask.sum() < best_a_adam.size:
        best_a_adam = best_a_adam[mask]
        a_adam = a_adam[mask]
        # re‑normalise (the target sum stays the same)
        best_a_adam *= target_sum / best_a_adam.sum()
        a_adam *= target_sum / a_adam.sum()
        print(f"[compress] reduced support from {n} to {best_a_adam.size}")

    # ---------- Exact‑gradient phase ----------
    best_a_exact, best_obj = _exact_gradient_phase(a_adam, target_sum, t_exact, rng)

    # ---------- Simulated‑annealing phase ----------
    best_a_final, best_obj = _simulated_annealing_phase(best_a_exact,
                                                        target_sum,
                                                        t_sa,
                                                        rng,
                                                        best_obj)

    # final normalisation (just in case)
    best_a_final = np.maximum(best_a_final, 0.0)
    best_a_final *= target_sum / best_a_final.sum()
    return best_a_final.tolist(), best_obj


# ----------------------------------------------------------------------
def propose_candidate(
        seed: int = 42,
        budget_s: int = 1000,
        **_ignore,
) -> list[float]:
    """
    Entry point used by the evaluation harness.
    One long optimisation run (default length 6000) is performed;
    the best sequence found within the time budget is returned.
    """
    np.random.seed(seed)
    end_time = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # support size – 6000 gives a good trade‑off between quality and speed
    n = 6000

    # starting point: use the previous construction if available
    init = None
    if 'height_sequence_1' in globals():
        seq0 = np.asarray(height_sequence_1, dtype=float).copy()
        seq0 = np.clip(seq0, 0.0, None)
        # keep the same total mass as required by the optimiser
        if seq0.sum() < 0.01:
            seq0 = np.ones_like(seq0)
        init = seq0

    remaining = max(0.0, end_time - time.time())
    if remaining < 5.0:          # not enough time – fall back to a uniform sequence
        uniform = np.full(n, np.sqrt(2.0 * n) / n)
        return uniform.tolist()

    seq, val = _run_optimisation(
        n=n,
        time_budget=remaining,
        init_a=init,
        seed=seed,
    )

    # ------------------------------------------------------------------
    # final evaluation (mainly for diagnostics)
    try:
        final_score = evaluate_sequence(seq)
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return seq
