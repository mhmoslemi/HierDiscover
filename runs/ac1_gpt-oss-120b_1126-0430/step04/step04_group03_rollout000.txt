"""
Propose a non‑negative step‑function (list of heights) that minimises

    O = 2·n·max(conv(a,a)) / (sum(a)²)

The search is a multi‑phase hybrid:

1. **Coarse parametric sweep** – power‑law shapes a_i ∝ (i+β)^(-α)
   on a modest grid.  The best shape is kept as a warm start.

2. **Random local refinement** – tiny random perturbations around the best
   (α,β) (and an optional exponential tail) to escape the grid.

3. **Multi‑restart mirror‑descent** – many restarts from random
   symmetric initial vectors (Dirichlet draws and Gaussian absolute
   draws).  Each restart runs exponentiated‑gradient (mirror) descent on
   the simplex, with a linear learning‑rate schedule and occasional jitter.
   The iterate is symmetrised after every update to stay in the
   (a_i = a_{n-1-i}) sub‑space, which empirically yields lower autocorrelation
   peaks.

4. **Final polishing** – a longer mirror‑descent run from the current best.

All heavy work (convolution, sub‑gradient) uses an FFT of size
next‑power‑of‑two ≥ 2·n‑1.  The routine respects the 1000 s budget,
printing progress and finally returning the best sequence as a plain
list of floats.  The returned vector is normalised to sum = 1, so the
objective computed internally coincides with ``evaluate_sequence`` (which
clips to [0,1000] and discards negatives).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _symmetrize(a: np.ndarray) -> np.ndarray:
    """Force a to be symmetric about the centre (a_i = a_{n-1-i})."""
    return (a + a[::-1]) / 2.0


# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using an FFT of power‑of‑two length."""
    n = a.size
    size = 1 << ((2 * n - 2).bit_length())   # next power of two ≥ 2n‑1
    A = np.fft.fft(a, n=size)
    conv = np.fft.ifft(A * A).real[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def _objective_norm1(a: np.ndarray) -> float:
    """Objective 2·n·max(conv) assuming Σa = 1."""
    n = a.size
    conv = _conv_fft(a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) w.r.t. a (on the simplex).
    Returns a vector of size n.
    """
    n = a.size
    conv = _conv_fft(a)
    max_val = conv.max()
    # tolerance to capture all (near‑)maximal indices
    ks = np.where(np.abs(conv - max_val) < 1e-14)[0]
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        # derivative contributes 2*a[j] for each i
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
def _exp_update(a: np.ndarray, g: np.ndarray, eta: float) -> np.ndarray:
    """
    Exponentiated‑gradient (mirror‑descent) update on the simplex.
    Operates in log‑space for stability.
    """
    log_a = np.log(a + 1e-15)          # avoid log(0)
    log_a -= eta * g
    log_a -= np.max(log_a)            # shift to keep numbers moderate
    a_new = np.exp(log_a)
    a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
def _optimize(a0: np.ndarray,
              max_iters: int,
              eta0: float,
              deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) optimisation on the simplex.
    The iterate is symmetrised after each step.  Returns the best vector
    encountered and its objective value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective_norm1(a)
    n = a.size

    # early‑stop patience
    no_improve = 0
    patience = max(1, max_iters // 30)

    for t in range(max_iters):
        if time.time() >= deadline:
            break

        # sub‑gradient of the max‑convolution
        g = _subgradient_max_conv(a)

        # linear learning‑rate decay
        eta = eta0 * (1.0 - t / max_iters)

        a = _exp_update(a, g, eta)
        a = _symmetrize(a)                      # stay in symmetric sub‑space

        cur_score = _objective_norm1(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            print(f"[opt] iter {t:6d}  new best O={best_score:.6f}")
        else:
            no_improve += 1
            if no_improve >= patience:
                # small random jitter to escape flat plateaus
                jitter = np.random.randn(n) * 1e-4
                a = best_a + jitter
                a = np.clip(a, 0.0, None)
                a = _project_to_simplex(a)
                a = _symmetrize(a)
                no_improve = 0

    return best_a, best_score


# ----------------------------------------------------------------------
def _shape_power(alpha: float, beta: float, n: int) -> np.ndarray:
    """Monotone decreasing power‑law shape a_i ∝ (i+β)^(-α), normalised."""
    i = np.arange(1, n + 1, dtype=float)
    a = (i + beta) ** (-alpha)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
def _shape_power_exp(alpha: float, beta: float, gamma: float, n: int) -> np.ndarray:
    """Power‑law with exponential tail a_i ∝ (i+β)^(-α)·exp(-γ·(i‑1)/n)."""
    i = np.arange(1, n + 1, dtype=float)
    a = (i + beta) ** (-alpha) * np.exp(-gamma * (i - 1) / n)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


# ----------------------------------------------------------------------

    """

    # ------------------------------------------------------------------

    n = 8192

    alphas = np.linspace(0.30, 0.80, 26)   # 26 values

            if score < best_score:

                best_alpha, best_beta = a, b
    print(f"[grid] best O={best_score:.6f} (α={best_alpha:.4f}, β={best_beta:.4f})")

    # ------------------------------------------------------------------

    for _ in range(2000):
