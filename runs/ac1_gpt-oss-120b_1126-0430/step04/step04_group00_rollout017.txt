#!/usr/bin/env python
"""
Robust step‑function search via multi‑stage smoothed‑max descent.

The routine
* tries several simple initial distributions (power‑law, inverse‑sqrt, exponential);
* runs a smoothed‑max Adam optimiser on the simplex;
* enlarges the support up to 50 000 entries;
* performs cheap local restarts when progress stalls;
* finishes by returning the best sequence found (as a plain Python list).

The implementation is fully self‑contained, uses only NumPy and the supplied
`evaluate_sequence` for the final score, and respects the ~1000 s time budget.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# 1. Utility functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max; returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it representation that yields `a` after soft‑max."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """
    Smallest power‑of‑two ≥ 3·n‑2,
    which is enough for a linear convolution of length (2·n‑1) with a
    vector of length n (i.e. correlation a⋆w of size 3·n‑2).
    """
    return 1 << ((3 * n - 2).bit_length())


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    total = new_a.sum()
    if total == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= total
    return new_a


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """Normalised power‑law a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _inverse_sqrt_distribution(n: int) -> np.ndarray:
    """Normalised a_i ∝ 1 / sqrt(i+1)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(idx)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _exponential_distribution(n: int, lam: float) -> np.ndarray:
    """Normalised a_i ∝ exp(-lam * i)."""
    idx = np.arange(0, n, dtype=np.float64)
    a = np.exp(-lam * idx)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _conv_max_and_grad(a: np.ndarray, eps_factor: float) -> tuple[float, np.ndarray]:
    """
    Compute max entry of a*a and the gradient of the smoothed objective.

    Parameters
    ----------
    a : np.ndarray
        Probability vector (sum = 1, non‑negative).
    eps_factor : float
        Multiplicative factor for the smoothing constant:
        eps = eps_factor * max(conv).

    Returns
    -------
    max_conv : float
        Maximum of the self‑convolution a*a.
    grad_a : np.ndarray
        Gradient of 2·n·smooth_max w.r.t. a (already multiplied by 4·n).
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    # FFT of a (zero‑padded)
    a_fft = np.fft.rfft(a, L)

    # --- self‑convolution a*a (linear, length 2n‑1) ---
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # --- smoothing weights (soft‑max over conv) ---
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or np.isnan(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # --- cross‑correlation of a and w (linear, length 3n‑2) ---
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    grad_a = 4.0 * n * corr
    return max_conv, grad_a


def _conv_max(a: np.ndarray) -> float:
    """Return max entry of a*a (no gradient)."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# 2. Core optimisation routine
# ----------------------------------------------------------------------


def _run_adam(
    a0: np.ndarray,
    budget_s: float,
    start_time: float,
    max_n: int = 50000,
    n_growth_factor: float = 1.5,
    seed: int = 0,
) -> np.ndarray:
    """
    Adam optimiser on the simplex that respects a total time budget.
    Returns the best distribution found (as a NumPy array that sums to 1).
    """
    np.random.seed(seed)
    deadline = start_time + budget_s - 2.0  # small safety margin

    # initialise
    n = a0.shape[0]
    a = a0.copy()
    z = _logits_from_distribution(a) + 0.01 * np.random.randn(n)

    # Adam state
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_alpha = 0.25  # a slightly larger base learning rate

    best_seq = _softmax(z)
    best_score = 2.0 * n * _conv_max(best_seq)

    it = 0
    last_improve_iter = 0
    eps_factor = 0.10          # start with a relatively large smoothing factor
    eps_factor_min = 4e-4
    eps_factor_decay = 0.999    # decay a bit slower than before
    report_every = 5000

    while time.time() < deadline and it < 2_000_000:
        it += 1

        # current point on simplex
        a = _softmax(z)

        # cheap evaluation + gradient (smooth max)
        max_conv, grad_a = _conv_max_and_grad(a, eps_factor)
        cheap_score = 2.0 * n * max_conv

        # keep best
        if cheap_score < best_score:
            best_score = cheap_score
            best_seq = a.copy()
            last_improve_iter = it
            if it % report_every == 0:
                print(f"[Iter {it:07d}] new best cheap = {cheap_score:.6f}")

        # gradient w.r.t logits (mirror descent on simplex)
        mu = np.dot(grad_a, a)
        grad_z = a * (grad_a - mu)

        # Adam moment updates (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step size schedule (inverse‑sqrt decay)
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step_dir = - m_hat / (np.sqrt(v_hat) + eps_adam)

        # back‑tracking line search (up to 5 halvings)
        accepted = False
        trial_alpha = alpha_t
        for _ in range(5):
            z_new = z + trial_alpha * step_dir
            z_new = np.clip(z_new, -60.0, 60.0)
            a_new = _softmax(z_new)
            new_max = _conv_max(a_new)
            new_score = 2.0 * n * new_max
            if new_score <= cheap_score:
                z = z_new
                accepted = True
                break
            trial_alpha *= 0.5

        if not accepted:
            # take a modest step anyway (helps escape flat regions)
            z += alpha_t * step_dir
            z = np.clip(z, -60.0, 60.0)

        # occasional random perturbation
        if it % 4000 == 0 and np.random.rand() < 0.12:
            z += 0.02 * np.random.randn(n)
            m *= 0.6
            v *= 0.6

        # decay smoothing factor
        eps_factor = max(eps_factor_min, eps_factor * eps_factor_decay)

        # support enlargement if we have time

#!/usr/bin/env python
"""
Robust step‑function search via multi‑stage smoothed‑max descent.

The routine
* tries several simple initial distributions (power‑law, inverse‑sqrt,
  exponential);
* runs a smoothed‑max Adam optimiser on the simplex;
* enlarges the support up to 50 000 entries;
* performs cheap local restarts when progress stalls;
* finishes by returning the best sequence found (as a plain Python list).

Only NumPy and the supplied ``evaluate_sequence`` are used.  The algorithm
honours the ~1000 s time budget.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level – no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max; returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it representation that yields ``a`` after ``softmax``."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """
    Smallest power‑of‑two ≥ 3·n‑2,
    enough for a linear convolution of length (2·n‑1).
    """
    return 1 << ((3 * n - 2).bit_length())


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    tot = new_a.sum()
    if tot == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= tot
    return new_a


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """Normalised a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _inverse_sqrt_distribution(n: int) -> np.ndarray:
    """Normalised a_i ∝ 1 / √(i+1)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(idx)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _exponential_distribution(n: int, lam: float) -> np.ndarray:
    """Normalised a_i ∝ exp(-lam·i)."""
    idx = np.arange(0, n, dtype=np.float64)
    a = np.exp(-lam * idx)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _conv_max_and_grad(a: np.ndarray, eps_factor: float):
    """
    Compute max entry of a*a and the gradient of the smoothed objective.

    Returns
    -------
    max_conv : float
        Maximum entry of the self‑convolution a*a.
    grad_a : np.ndarray
        Gradient of 2·n·smooth_max w.r.t. a (already multiplied by 4·n).
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    a_fft = np.fft.rfft(a, L)

    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or np.isnan(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    grad_a = 4.0 * n * corr
    return max_conv, grad_a


def _conv_max(a: np.ndarray) -> float:
    """Return max entry of a*a (no gradient)."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (supports gradual enlargement)
# ----------------------------------------------------------------------


def _run_adam(
    a0: np.ndarray,
    budget_s: float,
    start_time: float,
    max_n: int = 50000,
    n_growth_factor: float = 1.5,
    seed: int = 0,
) -> np.ndarray:
    """
    Adam optimiser on the probability simplex with a smoothed‑max objective.
    Returns the best distribution found (sums to 1).
    """
    np.random.seed(seed)
    deadline = start_time + budget_s - 2.0  # safety margin

    n = a0.shape[0]
    z = _logits_from_distribution(a0) + 0.01 * np.random.randn(n)

    # Adam state
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_alpha = 0.25

    best_seq = _softmax(z)
    best_score = 2.0 * n * _conv_max(best_seq)

    it = 0
    last_improve_iter = 0
    eps_factor = 0.10
    eps_factor_min = 4e-4
    eps_factor_decay = 0.999
    report_every = 5000

    while time.time() < deadline and it < 2_000_000:
        it += 1

        a = _softmax(z)
        max_conv, grad_a = _conv_max_and_grad(a, eps_factor)
        cheap_score = 2.0 * n * max_conv

        if cheap_score < best_score:
            best_score = cheap_score
            best_seq = a.copy()
            last_improve_iter = it
            if it % report_every == 0:
                print(f"[Iter {it:07d}] NEW BEST cheap = {cheap_score:.6f}")

        # gradient w.r.t. logits (mirror descent on simplex)
        mu = np.dot(grad_a, a)
        grad_z = a * (grad_a - mu)

        # Adam updates (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step_dir = -m_hat / (np.sqrt(v_hat) + eps_adam)

        # simple back‑tracking line‑search
        accepted = False
        trial_alpha = alpha_t
        for _ in range(5):
            z_new = z + trial_alpha * step_dir
            z_new = np.clip(z_new, -60.0, 60.0)
            a_new = _softmax(z_new)
            new_max = _conv_max(a_new)
            new_score = 2.0 * n * new_max
            if new_score <= cheap_score:
                z = z_new
                accepted = True
                break
            trial_alpha *= 0.5

        if not accepted:
            z += alpha_t * step_dir
            z = np.clip(z, -60.0, 60.0)

        # occasional random jitter
        if it % 4000 == 0 and np.random.rand() < 0.12:
            z += 0.02 * np.random.randn(n)
            m *= 0.6
            v *= 0.6

        # decay smoothing constant
        eps_factor = max(eps_factor_min, eps_factor * eps_factor_decay)

        # support enlargement
        if (it % 25000 == 0) and (n < max_n) and (time.time() + 30 < deadline):
            new_n = min(max_n, int(n * n_growth_factor) + 1)
            print(f"[Iter {it:07d}] Expanding support: {n} → {new_n}")
            a_interp = _interpolate_distribution(best_seq, new_n)
            z = _logits_from_distribution(a_interp)
            m = np.zeros_like(z)
            v = np.zeros_like(z)
            n = new_n
            best_seq = a_interp.copy()
            best_score = 2.0 * n * _conv_max(best_seq)
            it = 0
            last_improve_iter = 0
            eps_factor = 0.10
            continue

        # restart on long stagnation (helps escape plateaux)
        if n >= max_n and it - last_improve_iter > 250_000:
            print(f"[Iter {it:07d}] Stagnation – applying fresh jitter.")
            z = _logits_from_distribution(best_seq) + 0.03 * np.random.randn(n)
            m = np.zeros_like(z)
            v = np.zeros_like(z)
            last_improve_iter = it

    return best_seq
