def grad_smoothed_max(a, tau):
    A = np.fft.rfft(a, n=m)  # shape (m//2+1,)
    conv = np.fft.irfft(A * A, n=m)[:2*n-1]
    max_conv = conv.max()

    # subtract max to avoid large exponents
    exps = np.exp(tau * (conv - max_conv))

"""
Propose a step‑function (non‑negative list) that minimises

    O = 2·n·max(conv(a,a)) / (sum(a)²)

Algorithm
---------
1. **Parameter sweep** – generate simple monotone shapes
   a_i ∝ (i+β)^(‑α)   and a_i ∝ exp(‑λ·i)   on a grid.
   Keep the best candidate (normalised to sum = 1).

2. **Local random refinement** – perturb the best (α,β,λ) a few thousand
   times and keep any improvement.

3. **Smoothed‑max Adam optimisation** – minimise the smooth surrogate
   Sτ(a)= (1/τ)·log Σ_k exp(τ·conv_k) with an Adam‑style update.
   The gradient of Sτ is the (τ‑softened) sub‑gradient of max(conv).
   τ is increased slowly so the optimisation starts smooth and ends
   with the true max‑objective.  After each step the vector is projected
   onto the probability simplex (Σa=1, a≥0).  The whole loop runs until
   the time budget expires, keeping the best sequence seen.

4. **Optional random restarts** – if time permits, jitter the best
   sequence and run a short Adam optimisation again.

5. **Final evaluation** – report the value returned by the supplied
   ``evaluate_sequence`` function.

The implementation uses real‑FFT (rfft/irfft) for fast O(n log n)
convolutions, a simple simplex projection, and respects the 1000 s
budget.  All numbers are kept non‑negative; the scale is irrelevant
because the objective is scale‑invariant.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the probability simplex {x ≥ 0, Σx = 1}.
    Implements the O(n) algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_full_fft(a: np.ndarray, m: int) -> np.ndarray:
    """
    Linear convolution a*a (full length 2·len(a)-1) using a power‑of‑two FFT.
    Returns the first 2·n-1 entries of the convolution.
    """
    A = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(A * A, n=m)
    return conv[: 2 * a.size - 1]


# ----------------------------------------------------------------------
def _objective_norm1(a: np.ndarray, m: int) -> float:
    """Objective 2·n·max(conv) assuming Σa = 1."""
    n = a.size
    conv = _conv_full_fft(a, m)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
def _shape_power(alpha: float, beta: float, n: int) -> np.ndarray:
    """Monotone decreasing power‑law: a_i ∝ (i+β)^(-α)."""
    i = np.arange(1, n + 1, dtype=float)
    a = (i + beta) ** (-alpha)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
def _shape_power_exp(alpha: float, beta: float, gamma: float, n: int) -> np.ndarray:
    """Power‑law with an exponential tail."""
    i = np.arange(1, n + 1, dtype=float)
    a = (i + beta) ** (-alpha) * np.exp(-gamma * (i - 1) / n)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
def _shape_exp(lmbda: float, n: int) -> np.ndarray:
    """Pure exponential decay: a_i ∝ exp(-λ·i)."""
    i = np.arange(n, dtype=float)
    a = np.exp(-lmbda * i)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
def _grad_smoothed_max(a: np.ndarray, tau: float, m: int):
    """
    Gradient of the smoothed max (log‑sum‑exp) of the convolution.
    Returns (gradient, max_conv) where gradient has the same shape as a.
    """
    n = a.size
    # FFT of a (real)
    A = np.fft.rfft(a, n=m)

    # Full convolution a*a
    conv = np.fft.irfft(A * A, n=m)[: 2 * n - 1]

    max_conv = conv.max()

    # τ‑softmax weights on the convolution entries
    # shift by max_conv for numerical stability
    exp_terms = np.exp(tau * (conv - max_conv))
    w = exp_terms / exp_terms.sum()                     # length 2n‑1

    # Gradient = 2 * correlation(a, w)
    W = np.fft.rfft(w, n=m)                            # FFT of weights
    corr = np.fft.ifft(A * np.conj(W)).real[:n]        # first n entries
    grad = 2.0 * corr
    return grad, max_conv


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   max_iters: int,
                   deadline: float,
                   m: int,
                   lr: float = 0.05,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8,
                   tau0: float = 0.5,
                   tau_inc: float = 0.001,
                   tau_max: float = 30.0,
                   use_monotone: bool = False) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smoothed‑max objective.
    Returns the best vector encountered and its exact O‑value.
    """
    a = a0.copy()
    n = a.size

    # Adam moment estimates
    m_vec = np.zeros_like(a)
    v_vec = np.zeros_like(a)

    # best tracking
    best_a = a.copy()
    # initial exact objective
    best_O = _objective_norm1(a, m)

    tau = tau0

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        grad, max_conv = _grad_smoothed_max(a, tau, m)

        # Adam update (bias‑corrected)
        m_vec = beta1 * m_vec + (1.0 - beta1) * grad
        v_vec = beta2 * v_vec + (1.0 - beta2) * (grad * grad)
        m_hat = m_vec / (1.0 - beta1 ** t)
        v_hat = v_vec / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step

        # Project onto the simplex
        a = _project_to_simplex(a)

        if use_monotone:
            # Optional monotone (non‑increasing) projection via PAVA.
            a = _project_monotone_decreasing(a)

        # Exact objective for the current iterate
        cur_O = 2.0 * n * max_conv

        if cur_O < best_O - 1e-12:
            best_O = cur_O
            best_a = a.copy()
            print(f"[opt] iter {t:6d}  new best O={best_O:.6f}")

        # schedule for τ (smoothness) – slowly increase
        tau = min(tau_max, tau + tau_inc)

    return best_a, best_O


# ----------------------------------------------------------------------
def _project_monotone_decreasing(v: np.ndarray) -> np.ndarray:
    """
    Isotonic regression onto the cone {x₀ ≥ x₁ ≥ … ≥ xₙ₋₁ ≥ 0}.
    Uses the Pool Adjacent Violators Algorithm (PAVA) on -v.
    The result is re‑normalised to sum = 1 (preserving monotonicity).
    """
    n = v.shape[0]
    x = -v.copy()                       # we want non‑decreasing on -x
    # PAVA initial blocks
    sums = list(x)
    weights = [1] * n
    i = 0
    while i < len(sums) - 1:
        if sums[i] / weights[i] > sums[i + 1] / weights[i + 1]:
            # merge blocks i and i+1
            sums[i] += sums[i + 1]
            weights[i] += weights[i + 1]
            del sums[i + 1]
            del weights[i + 1]
            i = max(i - 1, 0)
        else:
            i += 1
    # expand back to full vector
    out = np.empty(n, dtype=float)
    idx = 0
    for s, w in zip(sums, weights):
        avg = s / w
        out[idx:idx + w] = -avg
        idx += w
    # enforce non‑negativity and re‑normalise
    out = np.maximum(out, 0.0)
    total = out.sum()
    if total > 0.0:
        out /= total
    else:
        out = np.full_like(v, 1.0 / n)
    return out


# ----------------------------------------------------------------------
