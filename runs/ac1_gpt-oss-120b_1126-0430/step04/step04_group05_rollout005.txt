"""
Propose a non‑negative step‑function that minimizes

    O(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

Algorithm
---------

1️⃣  Normalise the search vector to the simplex (sum = 1).  
    If a previous construction ``height_sequence_1`` is present we
    down‑sample it to a manageable size (≈2000 entries).

2️⃣  **Iterated linearisation (convex–concave procedure).**  
    For the current vector `a` we compute the full self‑convolution
    `c = conv(a,a)`.  The true constraints are

        ∀k :   aᵀ B_k a ≤ t ,   where  B_k[i,j] = 1 if i+j=k else 0

    (t is the value of `max(conv)` we would like to minimise).  
    Each quadratic constraint is linearised around the current point:

        2·(B_k a)ᵀ a – aᵀ B_k a ≤ t     →   2·c_k·a – t ≤ c[k]

    where `c_k = B_k a` is just a shifted copy of `a`.  
    The resulting linear programme

        minimise   t
        s.t.       2·c_k·a – t ≤ c[k]   (for every shift k)
                   a_i ≥ 0
                   sum a = 1
                   t   ≥ 0

    is solved with `scipy.optimize.linprog` (HiGHS).  The solution
    replaces `a` (projected back onto the simplex).  Repeating the
    linearisation 10‑30 times drives the maximal convolution down
    steadily.

3️⃣  **Stochastic local refinement.**  
    After the LP phase we perform a cheap simulated‑annealing style
    mass‑transfer hill‑climber (move a random fraction of probability
    mass from one index to another).  Improved candidates are kept
    and the temperature slowly decays.

4️⃣  The best sequence found is finally checked with the provided
    ``evaluate_sequence`` function and returned as a plain list of
    floats.

The whole search respects the 1000 s wall‑time limit and prints
intermediate objective values so the progress can be inspected.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        # all entries are zero – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full self‑convolution of a using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


def _downsample_sequence(seq: np.ndarray, target_n: int) -> np.ndarray:
    """Aggregate `seq` into `target_n` blocks, preserving total mass."""
    N = len(seq)
    if target_n >= N:
        a = np.array(seq, dtype=float)
    else:
        # compute block sizes (as equal as possible)
        base = N // target_n
        remainder = N % target_n
        block_sizes = np.full(target_n, base, dtype=int)
        block_sizes[:remainder] += 1
        a = []
        start = 0
        for sz in block_sizes:
            a.append(np.sum(seq[start:start + sz]))
            start += sz
        a = np.array(a, dtype=float)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / a.sum()
    return a


def _build_lp_matrices(a: np.ndarray, conv: np.ndarray) -> tuple:
    """
    Build the matrices for the linear programme that linearises the
    quadratic constraints  aᵀ B_k a ≤ t .
    Returns (A_ub, b_ub, A_eq, b_eq, bounds).
    """
    n = a.size
    m = 2 * n - 1                     # number of convolution entries
    # coefficient for the variable t (last column)
    minus_one_col = -np.ones((m, 1), dtype=float)

    # reversed a – used to create the shifted rows efficiently
    a_rev = a[::-1]

    # allocate space for the a‑coefficients (all rows share the same shape)
    A_a = np.zeros((m, n), dtype=float)

    # Fill each row k with a shifted copy of a_rev
    # The shift amount s = k - (n-1)  (negative ⇒ left shift, positive ⇒ right shift)
    for k in range(m):
        s = k - (n - 1)
        if s >= 0:
            # shift right by s → zeros on the left
            A_a[k, s:] = 2.0 * a_rev[: n - s]
        else:
            # shift left by -s → zeros on the right
            shift = -s
            A_a[k, : n - shift] = 2.0 * a_rev[shift:]
    # Assemble the full inequality matrix
    A_ub = np.hstack([A_a, minus_one_col])
    b_ub = conv.copy()

    # Equality: sum_i a_i = 1  (t has no role in the equality)
    A_eq = np.zeros((1, n + 1), dtype=float)
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0], dtype=float)

    # Variable bounds: a_i ≥ 0,  t ≥ 0
    bounds = [(0.0, None)] * n + [(0.0, None)]

    return A_ub, b_ub, A_eq, b_eq, bounds


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence that minimises the objective
    used by ``evaluate_sequence``.
    Returns the sequence as a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2          # small safety margin

    # ------------------------------------------------------------------
    # 0️⃣  Choose problem size and initialise a
    target_n = 2000                                 # fast enough for LPs
    if 'height_sequence_1' in globals():
        old = np.array(height_sequence_1, dtype=float)
        a = _downsample_sequence(old, target_n)
        print(f"[init] down‑sampled supplied sequence to n={a.size}")
    else:
        a = np.full(target_n, 1.0 / target_n, dtype=float)
        print(f"[init] uniform start, n={a.size}")

    n = a.size
    L = 1 << ((3 * n - 2).bit_length())            # FFT length (next power of two)

    # ------------------------------------------------------------------
    # Helper to evaluate the exact objective (sum is 1 ⇒ O = 2·n·maxconv)
    def _exact_obj(a_vec: np.ndarray) -> float:
        conv = _conv_fft(a_vec, L)
        return 2.0 * n * conv.max()

    best_seq = a.copy()
    best_score = _exact_obj(best_seq)
    print(f"[start] O = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Iterated linear‑concave (LP) refinement
    # Allocate roughly the first 55 % of the wall‑time to this phase.
    phase1_deadline = start_time + budget_s * 0.55
    lp_iter = 0
    max_lp_iter = 30

    while lp_iter < max_lp_iter and time.time() < phase1_deadline:
        # current convolution and exact objective
        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_score - 1e-9:
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[LP] iter {lp_iter:02d} new best O={best_score:.6f}")

        # Build and solve the linear programme
        A_ub, b_ub, A_eq, b_eq, bounds = _build_lp_matrices(a, conv)

        # minimise t (last variable)
        c = np.zeros(n + 1, dtype=float)
        c[-1] = 1.0

        # Solve – HiGHS is fast for this size
        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': max(0, phase1_deadline - time.time())})

        if not res.success:
            print(f"[LP] iteration {lp_iter} failed (status {res.status}), aborting LP phase.")
            break

        a_new = res.x[:n]
        # tiny numerical errors may violate the simplex condition – project back
        a_new = _project_to_simplex(a_new)

        # Optionally mix a little with the previous iterate to stabilise
        blend = 0.9
        a = blend * a + (1.0 - blend) * a_new

        # Normalise explicitly (project_to_simplex already does)
        a = _project_to_simplex(a)

        lp_iter += 1

    # ------------------------------------------------------------------
    # 2️⃣  Stochastic local refinement (simulated‑annealing style)
    # Allocate the remaining time (~45 %) to this phase.
    print(f"[refine] start stochastic refinement, best O={best_score:.6f}")
    T0 = 0.08
    T = T0
    max_refine_iters = 300_000
    it_refine = 0
    no_improve = 0
    no_improve_limit = 10_000
    a = best_seq.copy()                     # start from the best LP solution

    # pre‑compute convolution once – it will be updated lazily
    conv = _conv_fft(a, L)
    cur_obj = 2.0 * n * conv.max()

    while time.time() < deadline and it_refine < max_refine_iters:
        # ---- mass‑transfer move ----
        i, j = np.random.randint(n, size=2)
        if i != j and a[i] > 1e-14:
            # move a modest fraction (≤10 %) of the source mass
            delta = a[i] * np.random.rand() * 0.10
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            a_new = _project_to_simplex(a_new)

            conv_new = _conv_fft(a_new, L)
            new_obj = 2.0 * n * conv_new.max()

            # Metropolis acceptance
            improve = new_obj < cur_obj
            if improve or np.random.rand() < math.exp((cur_obj - new_obj) / max(T, 1e-12)):
                a = a_new
                conv = conv_new
                cur_obj = new_obj
                no_improve = 0
                if cur_obj < best_score - 1e-9:
                    best_score = cur_obj
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] new global best O={best_score:.6f} (iter={it_refine})")
            else:
                no_improve += 1
        else:
            no_improve += 1

        # ---- occasional exact sub‑gradient step (helps escape plateaus) ----
        if it_refine % 11 == 0:
            # max‑convolution index
            k_max = np.argmax(conv)
            # sub‑gradient of max‑conv (same as in the reference code)
            i_start = max(0, k_max - (n - 1))
            i_end = min(n - 1, k_max)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            idx_j = k_max - idx_i
            g = np.zeros_like(a)
            g[idx_i] += 2.0 * a[idx_j]
            eta = 0.06 * (1.0 - it_refine / max_refine_iters)
            a = a - eta * g
            a = _project_to_simplex(a)
            conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_score - 1e-9:
                best_score = cur_obj
                best_seq = a.copy()
                print(f"[sub‑grad] iter {it_refine} improved O={best_score:.6f}")

        # temperature decay
        T *= 0.9996
        it_refine += 1

        # restart if we have been stuck too long
        if no_improve >= no_improve_limit:
            a = best_seq.copy()
            a += np.random.randn(n) * 0.0012
            a = np.clip(a, 0, None)
            a = _project_to_simplex(a)
            conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max()
            no_improve = 0

    # ------------------------------------------------------------------
    # Final sanity check with the provided evaluator
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence failed:", exc)

    return best_seq.tolist()
