"""
Propose a non‑negative step function (list of heights) that minimises

        2·n·max_k (a*a)[k] / (Σ a_i)²

The algorithm builds on the previous heuristic but replaces the
sub‑gradient with a smooth “soft‑max” surrogate of the max‑convolution
and employs the Adam optimiser on the simplex (projection after each
update).  A schedule for the soft‑max temperature β gradually focuses
the optimisation on the true maximum.  Multiple diverse initialisations
(power‑law, exponential, uniform and Dirichlet draws) are screened,
the best seed is then refined.  Stagnation triggers a small random
perturbation and a moment reset.  All operations use FFTs (O(n log n))
so thousands of iterations fit comfortably into the 1000 s budget.

The routine always returns a feasible sequence (non‑negative,
Σa_i>0) and prints progress information.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        # all entries are non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa).real[: 2 * a.size - 1]
    return conv


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Compute numerically stable soft‑max weights w_i ∝ exp(β·conv_i)."""
    # subtract the maximum to avoid overflow
    maxc = conv.max()
    exp_term = np.exp(beta * (conv - maxc))
    w = exp_term / exp_term.sum()
    return w


def _softmax_grad(a: np.ndarray, conv: np.ndarray, beta: float,
                  fft_len_grad: int) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate of max(conv) w.r.t. a.
    Returns a vector of length len(a).
    """
    n = a.shape[0]
    # soft‑max weights over the full convolution (length 2·n‑1)
    w = _softmax_weights(conv, beta)

    # convolution w * a_rev (a reversed) gives the required gradient
    a_rev = a[::-1]
    # FFT‑based convolution (length 3·n‑2)
    fw = np.fft.fft(w, fft_len_grad)
    fa_rev = np.fft.fft(a_rev, fft_len_grad)
    full = np.fft.ifft(fw * fa_rev).real

    # grad_k = 2 * full[n‑1 + k] for k = 0 … n‑1
    start = n - 1
    grad = 2.0 * full[start:start + n]
    return grad


def _power_law_seq(n: int, p: float, reverse: bool = False) -> np.ndarray:
    """Return a normalised power‑law a_i ∝ (i+1)^{-p} (or increasing if reverse)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    if reverse:
        seq = i ** p            # increasing power law
    else:
        seq = i ** (-p)         # decreasing power law
    seq /= seq.sum()
    return seq


def _exp_decay_seq(n: int, lam: float) -> np.ndarray:
    """Return a normalised exponential decay a_i ∝ exp(-lam·i)."""
    i = np.arange(n, dtype=np.float64)
    seq = np.exp(-lam * i)
    seq /= seq.sum()
    return seq


def _dirichlet_seq(n: int, conc: float) -> np.ndarray:
    """Draw a Dirichlet sample with given concentration (scaled to sum‑1)."""
    g = np.random.gamma(conc, 1.0, size=n)
    seq = g / g.sum()
    return seq
