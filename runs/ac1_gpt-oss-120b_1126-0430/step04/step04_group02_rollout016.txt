def _conv_self(a):
    return np.convolve(a, a, mode='full')

def _objective(a):

    conv = _conv_self(a)
    return 2.0 * n * conv.max()

def solve_lp(f, M):
    # Given f and M, solve LP described.
    n = len(f)
    g = cp.Variable(n)
    f_const = cp.Parameter(n, nonneg=True)
    M_param = cp.Parameter(nonneg=True)
    f_const.value = f
    M_param.value = M
    constraints = [g >= 0, cp.conv(f_const, g) <= M_param]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    prob.solve(solver='ECOS', warm_start=True, verbose=False, max_iters=2000)
    if g.value is None:

    # ensure nonnegative
    g_val = np.maximum(np.array(g.value, dtype=float), 0.0)

"""
Propose a step‑function (non‑negative sequence) that minimises

    O(a) = 2·n·max(conv(a,a)) ,   sum(a)=1 , a_i≥0

The implementation mixes three complementary optimisation ideas
within the 1000 s budget:

1. **Adam on the log‑it (softmax) parametrisation** – a fast
   gradient descent with an annealed smooth max.  The schedule is
   tuned for a very long run (≈0.85 s of the total budget) on a
   fairly large support (≈5000 entries).  The optimiser keeps the
   best‑ever vector and occasionally restarts from it.

2. **Optional linear‑programming refinement** – when `cvxpy` is
   available we solve a small LP that, for the current vector `f`,
   finds a vector `g` with maximal total mass while keeping the
   convolution `f*g` below the current worst convolution value.
   A tiny convex combination `f←(1‑t)f+t g` (t≈0.05) is then applied.
   A few LP steps often lower the maximum convolution a little
   further.

3. **Final local hill‑climbing** – with the remaining time we perform
   random mass‑transfer moves (up to 5 % of the source entry) and
   accept only improvements.  The convolution is recomputed by FFT
   on every trial, which is still cheap for a few‑ten‑thousand
   moves.  The best vector from this phase replaces the global best
   if it improves the objective.

All steps respect the simplex constraint by normalising after each
change.  The routine finally returns the best list of floats it has
found.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full convolution of a with itself using real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv


def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """Correlation (w ⋆ a) for 1‑D real vectors using FFT."""
    n = a.shape[0]
    m = w.shape[0]                     # = 2n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) (assumes sum(a)=1)."""
    n = a.size
    conv = _conv_self_fft(a)
    return 2.0 * n * conv.max()


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


# ----------------------------------------------------------------------
def _run_opt(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
) -> tuple[list[float], float]:
    """
    Adam optimisation on the simplex (log‑it parameterisation).

    Returns the best vector found (as a plain list) and its exact
    objective value.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # initial vector
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
        a = a / a.sum()
    else:
        a = rng.random(n)
        a = a / a.sum()

    # logits (log‑it) – keep positivity automatically
    x = np.log(a + 1e-16)

    m = np.zeros_like(x)          # Adam 1st moment
    v = np.zeros_like(x)          # Adam 2nd moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # schedules (more aggressive than the reference)
    beta_start = 0.1
    beta_end = 15000.0
    lr_start = 0.20
    lr_end = 0.001
    max_iters = 800_000

    best_a = a.copy()
    best_obj = _objective(best_a)

    it = 0
    while time.time() - start < time_limit and it < max_iters:
        it += 1

        # probabilities from logits
        a = _softmax(x)

        # exact objective
        conv = _conv_self_fft(a)
        M = conv.max()
        obj = 2.0 * n * M
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[n={n:5d} it={it:06d}] new best O={best_obj:.6f}")

        # schedule parameters
        t = min(it / max_iters, 1.0)
        beta = beta_start * np.exp(np.log(beta_end / beta_start) * t)
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t)

        # smooth‑max weights
        if beta > 13000.0:                     # effectively hard max
            kmax = int(np.argmax(conv))
            w = np.zeros_like(conv)
            w[kmax] = 1.0
        else:
            shifted = conv - M
            w = np.exp(beta * shifted)
            w = w / w.sum()

        # gradient of the smooth objective (w.r.t. a)
        g_a = 2.0 * _correlation_fft(w, a)

        # natural‑gradient wrt logits
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape plateaus
        if it % 7000 == 0:
            jitter = rng.normal(scale=0.025, size=n)
            x += jitter

        # periodic restart from the best solution found so far
        if it % 120_000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.006, size=n)
            m.fill(0.0)
            v.fill(0.0)
            print(f"[n={n:5d}] restart at iter {it}")

    # ------------------------------------------------------------------
    # compress negligible entries
    thr = 1e-12
    mask = best_a > thr
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = _objective(best_a)
        print(f"[compress] reduced length {n} → {best_a.size}, O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # final local hill‑climbing (random mass transfers)
    rem = time_limit - (time.time() - start)
    if rem > 0.8:
        a_loc = best_a.copy()
        obj_loc = best_obj
        n_loc = a_loc.size
        rng_local = np.random.default_rng(seed + 9999)
        end_local = time.time() + rem * 0.6
        while time.time() < end_local:
            i, j = rng_local.choice(n_loc, size=2, replace=False)
            if a_loc[i] <= 0.0:
                continue
            # move a small random fraction (up to 4 %)
            delta = a_loc[i] * rng_local.random() * 0.04
            a_new = a_loc.copy()
            a_new[i] -= delta
            a_new[j] += delta
            conv_new = _conv_self_fft(a_new)
            obj_new = 2.0 * n_loc * conv_new.max()
            if obj_new < obj_loc:
                a_loc, obj_loc = a_new, obj_new
                if obj_new < best_obj:
                    best_a, best_obj = a_new, obj_new
        print(f"[local] final local O={best_obj:.6f}")

    return best_a.tolist(), best_obj


# ----------------------------------------------------------------------
def _lp_refine(a: np.ndarray, time_budget: float, seed: int = 0) -> np.ndarray:
    """
    Optional refinement using a tiny linear programme.

    For the current vector `a` computes a vector `g` with maximal
    ℓ₁‑mass under the constraint `conv(a,g) ≤ max(conv(a,a))` and mixes
    a small step `t` of `g` into `a`.  Returns the possibly‑improved
    vector (still normalised to sum‑one).  If `cvxpy` is unavailable the
    input is returned unchanged.
    """
    try:
        import cvxpy as cp
    except Exception:                      # pragma: no cover
        return a

    n = a.size
    # build a single LP problem (warm‑started) -------------------------------------------------
    g = cp.Variable(n)
    f_par = cp.Parameter(n, nonneg=True)
    M_par = cp.Parameter(nonneg=True)
    constraints = [
        g >= 0,
        cp.conv(f_par, g) <= M_par
    ]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    rng = np.random.default_rng(seed)
    t = 0.05                               # mixing weight

    end = time.time() + time_budget
    while time.time() < end:
        conv_a = _conv_self_fft(a)
        M = conv_a.max()
        f_par.value = a
        M_par.value = M
        prob.solve(solver='ECOS', warm_start=True, max_iters=2000, verbose=False)
        if g.value is None:
            break
        g_val = np.maximum(np.array(g.value, dtype=float), 0.0)
        if g_val.sum() < 1e-15:
            break
        g_norm = g_val / g_val.sum()
        a_new = (1.0 - t) * a + t * g_norm
        a_new = np.clip(a_new, 0.0, None)
        a_new /= a_new.sum()
        # accept if objective improves
        if _objective(a_new) < _objective(a):
            a = a_new
        else:
            # reduce step size a bit and try again
            t = max(t * 0.7, 0.005)

    return a

                    best_obj, best_seq = new_obj, a_new.tolist()
        print(f"[final local] best O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # final evaluation (for logging only)
    try:
        final_score = evaluate_sequence(best_seq)
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence failed:", exc)

    return best_seq
