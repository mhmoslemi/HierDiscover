"""
propose_candidate
-----------------
Search for a non‑negative step‑function (a list of floats) that minimises

    O(a) = 2·n·max(conv(a,a)) / (sum(a)²)

The algorithm consists of several independent restarts, each containing
three phases:

1️⃣  *Adam on a smooth log‑sum‑exp surrogate* – the true max is replaced
    by a soft‑max with a temperature α that is slowly increased.  After
    each Adam step the vector is projected onto the probability simplex
    (non‑negative and sum = 1).

2️⃣  *Exact sub‑gradient descent* – once α is large enough we switch to
    the genuine non‑smooth objective and perform a few hundred‑thousand
    projected sub‑gradient steps.

3️⃣  *Stochastic local refinement* – a simulated‑annealing style “mass‑transfer”
    move (shift a random fraction of probability mass from one index to
    another) is repeatedly tried; moves are accepted if they improve the
    exact objective or with a Metropolis probability that cools.
    Occasional exact sub‑gradient steps keep the search directed.

All three phases respect the 1000 s wall‑time budget, keep track of the
best sequence encountered and finally verify it with the provided
``evaluate_sequence`` function.

The search uses a larger problem size (≈ 8000 coefficients) than the
previous best run; increasing the length gives a finer “frequency grid”
and allows the optimizer to approach the theoretical lower bound
O≈1.0.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex {x ≥ 0, sum(x)=1}
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        # all entries are non‑positive → return uniform point
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear (full‑length) self‑convolution using FFT
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return a*a (full convolution) using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    # keep only the first 2·len(a)‑1 entries (the linear convolution)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Exact objective when the vector is already normalised (sum == 1)
def _objective_simplex(a: np.ndarray, n: int, L: int) -> float:
    """Compute 2·n·max(conv(a,a))  (a is assumed to sum to 1)."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Sub‑gradient of the maximal convolution entry (used in the exact phase)
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Return a sub‑gradient of max(conv) w.r.t. a.
    Works for the convex‑concave max of the quadratic form.
    """
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)          # all maximisers
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        # each pair (i,j) contributes 2·a_j to the derivative w.r.t. a_i
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
# Gradient of the smooth log‑sum‑exp surrogate of the max
def _smooth_gradient(a: np.ndarray, conv: np.ndarray, alpha: float, L: int) -> np.ndarray:
    """
    Compute ∇_a ( (2·n/α)·log Σ_k exp(α·conv_k) ).
    The formula reduces to 2·corr(w, a) where w_k ∝ exp(α·conv_k).
    """
    # soft‑max (numerically stable)
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / exp_scaled.sum()

    # correlation w ⋆ a   (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)   # length ≥ 3·n‑2
    n_len = a.size
    grad = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]
    return grad


# ----------------------------------------------------------------------
# Initialisation helpers -------------------------------------------------
def _init_uniform(n: int) -> np.ndarray:
    """All entries equal (probability simplex)."""
    return np.full(n, 1.0 / n)


def _init_dirichlet(n: int, alpha: float = 0.3) -> np.ndarray:
    """Random Dirichlet (Gamma) vector, then normalised."""
    x = np.random.gamma(alpha, size=n)
    x /= x.sum()
    return x


def _init_u_shape(n: int, power: float = 2.0) -> np.ndarray:
    """
    U‑shaped distribution: mass concentrates at the two ends.
    a_i ∝ (|i - centre| + ε)^power, normalised to sum = 1.
    """
    centre = (n - 1) / 2.0
    dist = np.abs(np.arange(n) - centre) + 0.5          # avoid zero
    a = np.power(dist, power)
    a /= a.sum()
    return a


def _init_sparse_random(n: int, density: float = 0.07) -> np.ndarray:
    """
    Random sparse support: choose a fraction 'density' of the indices,
    assign equal mass to those entries.
    """
    s = max(2, int(density * n))
    idx = np.random.choice(n, size=s, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / s
    return a


# ----------------------------------------------------------------------
def _run_one_restart(a0: np.ndarray, time_limit: float, n: int, L: int):
    """
    Perform the three‑phase optimisation starting from a0.
    Returns the best sequence (numpy array) and its exact objective.
    """
    start = time.time()
    deadline = start + time_limit

    # --------------------------------------------------------------
    # Phase 1 – Adam on the smoothed surrogate (log‑sum‑exp)
    # --------------------------------------------------------------
    a = a0.copy()
    best_seq = a.copy()
    best_obj = _objective_simplex(a, n, L)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    lr0 = 0.8                # a bit larger than the reference
    alpha_start, alpha_end = 5.0, 20000.0
    max_iter_phase1 = 250_000

    for it in range(max_iter_phase1):
        if time.time() >= deadline:
            break

        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_seq = a.copy()
        # anneal α linearly
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iter_phase1)

        # gradient of smooth surrogate
        grad = _smooth_gradient(a, conv, alpha, L)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        # step‑size decays like 1/√t
        lr = lr0 / math.sqrt(it + 1)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional tiny jitter to escape plateaus
        if np.random.rand() < 0.001:
            a_new += np.random.randn(n) * 1e-4

        # project onto the simplex
        a_new = _project_to_simplex(a_new)

        # pruning of tiny entries – keep the support manageable
        if it % 4000 == 0 and it > 0:
            thresh = np.percentile(a_new, 0.5)          # drop bottom 0.5 %
            a_new[a_new < thresh] = 0.0
            a_new = _project_to_simplex(a_new)

        a = a_new

    # --------------------------------------------------------------
    # Phase 2 – exact sub‑gradient descent (non‑smooth objective)
    # --------------------------------------------------------------
    eta0 = 0.30
    subgrad_iters = 120_000
    for it in range(subgrad_iters):
        if time.time() >= deadline:
            break
        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_seq = a.copy()
        g = _subgradient_max_conv(a, conv)

        # geometrically decaying step size
        eta = eta0 * (0.9995 ** it)
        a = a - eta * g
        a = _project_to_simplex(a)

    # --------------------------------------------------------------
    # Phase 3 – stochastic local refinement (simulated annealing)
    # --------------------------------------------------------------
    cur_conv = _conv_fft(a, L)
    cur_obj = 2.0 * n * cur_conv.max()
    if cur_obj < best_obj:
        best_obj = cur_obj
        best_seq = a.copy()

    T0 = 0.07
    T = T0
    it_loc = 0
    no_improve = 0
    max_no_improve = 7000

    while time.time() < deadline:
        # mass‑transfer move
        i, j = np.random.randint(n, size=2)
        if i != j and a[i] > 1e-14:
            delta = a[i] * np.random.rand() * 0.09   # ≤9 % of source mass
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            a_new = _project_to_simplex(a_new)

            conv_new = _conv_fft(a_new, L)
            new_obj = 2.0 * n * conv_new.max()

            # Metropolis acceptance
            if new_obj < cur_obj or np.random.rand() < math.exp((cur_obj - new_obj) / max(T, 1e-12)):
                a = a_new
                cur_conv = conv_new
                cur_obj = new_obj
                no_improve = 0
                if cur_obj < best_obj:
                    best_obj = cur_obj
                    best_seq = a.copy()
            else:
                no_improve += 1
        else:
            no_improve += 1

        # occasional exact sub‑gradient step (keeps directionality)
        if it_loc % 11 == 0:
            g = _subgradient_max_conv(a, cur_conv)
            eta_loc = 0.07 * (1.0 - it_loc / 200_000)
            a = a - eta_loc * g
            a = _project_to_simplex(a)
            cur_conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * cur_conv.max()
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_seq = a.copy()

        # decay temperature
        T *= 0.9997
        it_loc += 1

        # restart if stuck for a while
        if no_improve >= max_no_improve:
            a = best_seq.copy()
            a += np.random.randn(n) * 0.0012
            a = np.clip(a, 0, None)
            a = _project_to_simplex(a)
            cur_conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * cur_conv.max()
            no_improve = 0

    return best_seq, best_obj

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Entry point required by the evaluation harness.
    Returns a list of non‑negative floats (the step‑function heights)
    that (hopefully) yields a low value of ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2                # safety margin

    # ------------------------------------------------------------------
    # Problem size -------------------------------------------------------
    # Use a larger sequence than the previous best (≈ 4600).  8192 is a
    # power‑of‑two‑friendly size that still fits comfortably into memory.
    n = 8192
    if 'height_sequence_1' in globals():
        # If a previous construction is supplied, we keep its length as a
        # fallback (it might be useful for a quick warm‑start).
        n = max(n, len(height_sequence_1))

    # FFT length – next power of two larger than the longest convolution/correlation
    L = 1 << ((3 * n - 2).bit_length())

    init_functions = [
        lambda: _init_uniform(n),

        lambda: _init_dirichlet(n, alpha=0.8),
        lambda: _init_u_shape(n, power=1.8),
        lambda: _init_u_shape(n, power=2.5),
        lambda: _init_sparse_random(n, density=0.06),
        lambda: _init_sparse_random(n, density=0.10),
    ]

    # Store the overall best result
    best_overall_seq = None
    best_overall_score = float('inf')

    # Allocate roughly equal time slices to each restart (but stop early
    # if the global deadline is approaching).
    n_restarts = len(init_functions)
    for idx, init_fn in enumerate(init_functions):
        now = time.time()

        if time_left < 5.0:                     # not enough time for another run

        # Spend at most a quarter of the remaining wall‑time on a single restart
        max_time_this_restart = min(time_left * 0.33, 210.0)   # also cap at ~3 min

        if score < best_overall_score:

            best_overall_seq = seq.copy()

        # early exit if we have already reached the target bound
        if best_overall_score <= 1.503:
            print("[early stop] target bound reached")
            break

    # ------------------------------------------------------------------
    # Final sanity check with the provided evaluator
    final_seq = best_overall_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence failed:", exc)

    return final_seq
