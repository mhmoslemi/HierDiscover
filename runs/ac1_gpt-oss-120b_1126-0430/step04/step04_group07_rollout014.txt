"""
Propose a candidate step function that (approximately) flattens its self‑convolution.
The algorithm consists of two stages:

1. **Random shape sampling** – many families of analytic windows (Kaiser,
   Gaussian, raised‑cosine, asymmetric Beta, exponential, sparse spikes …)
   are sampled with random parameters and a range of lengths.  Each candidate
   is normalised to a probability vector (sum = 1) and its bound
   ``O = 2·n·max(conv(a,a))`` is evaluated with a fast FFT‑based convolution.
   The best sample is kept.

2. **Local refinement** – starting from the best sampled vector a,
   a mirror‑descent / exponentiated‑gradient (Adam‑style) optimiser
   minimises the same objective.  The gradient of the (smoothed) max‑convolution
   is obtained by correlating the current distribution with a weight vector
   that concentrates on the active (maximal) convolution entries.  The
   optimisation runs until the time budget is exhausted (≈ 1000 s).

The combination of diverse initialisations and a solid local optimiser
typically beats the previous mirror‑descent baseline and pushes the bound
below the target 1.503.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers (identical to the ones used by the baseline)

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv


def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Linear correlation g_i = Σ_t w[i+t]·a[t]   for i=0,…,n‑1.
    Implemented via FFT.  ``w`` has length 2·n‑1.
    """
    n = a.shape[0]
    m = w.shape[0]                # = 2·n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


# ----------------------------------------------------------------------
# Shape generators -----------------------------------------------------

def _gen_kaiser(n: int, beta: float) -> np.ndarray:
    """Kaiser window (beta ≥ 0)."""
    return np.kaiser(n, float(beta))


def _gen_gaussian(n: int, sigma: float) -> np.ndarray:
    """Centered Gaussian, sigma > 0."""
    x = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
    return a


def _gen_cosine_pow(n: int, p: float) -> np.ndarray:
    """Raised cosine: (0.5 - 0.5*cos(2πi/(n-1)))**p."""
    i = np.arange(n, dtype=float)
    base = 0.5 - 0.5 * np.cos(2.0 * math.pi * i / (n - 1))
    a = np.power(base, float(p))
    return a


def _gen_exponential(n: int, decay: float, right: bool = False) -> np.ndarray:
    """Exponential decay from the left (or right if ``right``)."""
    x = np.arange(n, dtype=float)
    if right:
        x = n - 1 - x
    a = np.exp(-float(decay) * x)
    return a


def _gen_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric Beta distribution on a discrete grid."""
    i = np.arange(n, dtype=float) + 0.5
    x = i / n
    a = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    return a


def _gen_uniform_interval(n: int, frac: float) -> np.ndarray:
    """Uniform mass on a central interval of relative length ``frac``."""
    a = np.zeros(n, dtype=float)
    interior = max(1, int(round(frac * n)))
    start = (n - interior) // 2
    a[start:start + interior] = 1.0
    return a


def _gen_sparse(n: int, k: int) -> np.ndarray:
    """k random spikes with random positive weights."""
    a = np.zeros(n, dtype=float)
    idx = np.random.choice(n, size=min(k, n), replace=False)
    vals = np.random.rand(len(idx)) + 1e-6  # avoid exact zeros
    a[idx] = vals
    return a


# ----------------------------------------------------------------------
# Utility: compute the bound O = 2·n·max(conv(a,a)) (a is assumed normalised)

def _score_of(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a normalised probability vector."""
    conv = _conv_self_fft(a)
    return 2.0 * a.size * float(conv.max())


# ----------------------------------------------------------------------
# Main entry point -----------------------------------------------------

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list:
    """
    Search for a non‑negative step‑function that minimises the bound
    ``evaluate_sequence``.  The routine mixes random analytic windows with a
    mirror‑descent (Adam) refinement and respects the 1000 s budget.
    Returns the best sequence as a plain list of floats.
    """
    # ------------------------------------------------------------------
    # 0. General set‑up
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    best_seq = None
    best_score = math.inf

    # If a previous construction is available, use it as an initial guess
    if 'height_sequence_1' in globals():
        tmp = np.asarray(height_sequence_1, dtype=float)
        if tmp.size > 0:
            tmp = np.maximum(0.0, tmp)
            if tmp.sum() > 0:
                best_seq = tmp / tmp.sum()
                best_score = _score_of(best_seq)
                print(f"[init] using provided height_sequence_1, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1. Random shape sampling
    # ------------------------------------------------------------------
    # lengths to try – larger lengths give a tighter theoretical lower bound
    lengths = [2000, 4000, 8000, 16000, 24000]

    # mapping of shape name → (generator, parameter sampler)
    shape_families = [
        ("kaiser",
         _gen_kaiser,
         lambda n: {"beta": np.random.uniform(0.0, 25.0)}),

        ("gaussian",
         _gen_gaussian,
         lambda n: {"sigma": np.random.uniform(0.05 * n, 0.5 * n)}),

        ("cosine_pow",
         _gen_cosine_pow,
         lambda n: {"p": np.random.uniform(0.3, 5.0)}),

        ("exponential_left",
         lambda n, decay: _gen_exponential(n, decay, right=False),
         lambda n: {"decay": np.random.uniform(0.0, 30.0 / max(1, n))}),

        ("exponential_right",
         lambda n, decay: _gen_exponential(n, decay, right=True),
         lambda n: {"decay": np.random.uniform(0.0, 30.0 / max(1, n))}),

        ("beta",
         _gen_beta,
         lambda n: {
             "alpha": np.random.exponential(1.0) + 0.2,
             "beta":  np.random.exponential(1.0) + 0.2
         }),

        ("uniform_interval",
         _gen_uniform_interval,
         lambda n: {"frac": np.random.uniform(0.2, 0.9)}),

        ("sparse",
         _gen_sparse,
         lambda n: {"k": np.random.randint(5, min(200, max(5, n // 5)))})
    ]

    # sampling loop – leave a few seconds for the local refinement
    while time.time() < deadline - 8.0:
        n = int(random.choice(lengths))
        name, gen, sampler = random.choice(shape_families)
        try:
            params = sampler(n)
            a = gen(n, **params)
        except Exception as exc:  # pragma: no cover
            print(f"[sample] generation error for {name}: {exc}")
            continue

        # make sure we have a non‑negative vector, normalise to sum‑1
        a = np.maximum(0.0, a.astype(float))
        s = a.sum()
        if s <= 0.0 or not np.isfinite(s):
            continue
        a /= s

        # evaluate the bound
        try:
            score = _score_of(a)
        except Exception as exc:  # pragma: no cover
            print(f"[sample] evaluation error: {exc}")
            continue

        if score < best_score - 1e-12:
            best_score = score
            best_seq = a.copy()
            print(f"[sample] n={n:5d} shape={name:<18} score={score:.6f}")

    # ------------------------------------------------------------------
    # 2. Local refinement (Adam‑style mirror descent)
    # ------------------------------------------------------------------
    if best_seq is None:
        # should never happen, but fall back to uniform if it does
        n = 8000
        best_seq = np.full(n, 1.0 / n, dtype=float)
        best_score = _score_of(best_seq)
        print("[fallback] using uniform distribution as a start")

    # work in log‑probability space to keep positivity automatically
    eps = 1e-14
    u = np.log(best_seq + eps)      # log‑variables
    n = best_seq.size

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    m = np.zeros_like(u)
    v = np.zeros_like(u)
    eta = 0.05                      # base learning rate
    it = 0
    jitter_interval = 3000
    jitter_scale = 1e-4
    restart_interval = 20000

    # keep a snapshot of the best solution found during refinement
    best_u = u.copy()
    best_ref_score = best_score
    best_iter = 0

    while time.time() < deadline - 0.3:
        it += 1

        # ---------- construct probability vector ----------
        a = np.exp(u - u.max())      # subtract max for numerical stability
        a_sum = a.sum()
        if a_sum <= 0.0:
            a = np.full_like(a, 1.0 / n)
            a_sum = 1.0
        a /= a_sum

        # ---------- evaluate ----------
        conv = _conv_self_fft(a)
        M = conv.max()
        cur_score = 2.0 * n * float(M)

        # ---------- keep the best ----------
        if cur_score < best_ref_score - 1e-12:
            best_ref_score = cur_score
            best_u = u.copy()
            best_iter = it
            print(f"[refine] iter {it:07d} new best O={cur_score:.6f}")

        # ---------- gradient of smoothed max ----------
        # identify (near) active entries of the convolution
        tol = max(1e-14, M * 1e-8)
        active = np.where(conv >= M - tol)[0]
        if active.size == 0:
            # fallback – should not happen
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
        else:
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.size

        # correlation gives sub‑gradient of the (soft) max
        g = 2.0 * _correlation_fft(w, a)

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        # bias‑corrected moments
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = eta * m_hat / (np.sqrt(v_hat) + 1e-8)
        u = u - step

        # ---------- occasional jitter ----------
        if it % jitter_interval == 0:
            u += np.random.randn(n) * jitter_scale

        # ---------- cheap restart ----------
        if it % restart_interval == 0:
            u = best_u.copy()
            u += np.random.randn(n) * (jitter_scale * 0.5)
            print(f"[restart] iter {it}, re‑initialised from current best (O={best_ref_score:.6f})")

    # ------------------------------------------------------------------
    # 3. Return the best sequence found (converted back from log‑space)
    # ------------------------------------------------------------------
    best_a = np.exp(best_u - best_u.max())
    best_a = best_a / best_a.sum()

    # Final sanity check with the provided evaluator (optional)
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f} "
              f"(best during refinement iter {best_iter})")
        # keep the evaluator's score as an extra indicator
        best_score = min(best_score, final_score)
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_a.tolist()
