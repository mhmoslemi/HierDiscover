"""
Optimise a non‑negative step‑function to minimise

    O = 2·n·max(conv(a,a)) / (sum(a)²)

The algorithm proceeds in several stages within the 1000 s budget:

1. **Multi‑scale random shape search** – try four parametric families
   (power law, power + exponential tail, Beta‑density, power × polynomial)
   on several lengths n ∈ {1024,2048,4096,8192,16384}.  Thousands of
   random parameterisations are evaluated using an FFT‑based convolution.
   The best candidate (lowest O) is kept.

2. **Mirror‑descent polishing** – projected exponentiated‑gradient descent
   on the true objective (sub‑gradient of the max‑convolution) with a
   linearly decaying learning rate and occasional jitter.

3. **Hill‑climbing refinement** – small mass‑transfer moves between random
   indices are accepted only if they improve the objective.

4. **Final verification** – the best sequence is passed to the supplied
   `evaluate_sequence` function.

All helper routines are top‑level, the code respects the time‑budget and
returns a plain ``list[float]``.
"""

import time
import numpy as np
from scipy.optimize import linprog  # optional, not used in the main loop


# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a power‑of‑two FFT, returning length 2·n‑1."""
    n = a.size
    size = 1 << ((2 * n - 2).bit_length())   # next power of two ≥ 2n‑1
    A = np.fft.fft(a, n=size)
    conv = np.fft.ifft(A * A).real[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def _objective_norm1(a: np.ndarray) -> float:
    """Objective 2·n·max(conv) under the constraint Σa = 1."""
    n = a.size
    conv = _conv_fft(a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) w.r.t. a (on the simplex).
    For every index k attaining the maximum we add the contribution
    `2 a_j` to each a_i that participates in the term a_i a_j.
    The result is averaged over all maximal k.
    """
    n = a.size
    conv = _conv_fft(a)
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    ks = np.where(conv >= max_val - eps)[0]

    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]

    if ks.size:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
def _exp_update(a: np.ndarray, g: np.ndarray, eta: float) -> np.ndarray:
    """
    Exponentiated‑gradient (mirror) update on the simplex.
    Operates in log‑space for numerical stability.
    """
    log_a = np.log(a + 1e-15)
    log_a -= eta * g
    log_a -= np.max(log_a)          # shift to avoid overflow
    a_new = np.exp(log_a)
    a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
def _optimize(a0: np.ndarray,
              max_iters: int,
              eta0: float,
              deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) optimisation.
    Returns the best vector seen and its objective value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective_norm1(a)

    n = a.size
    no_improve = 0
    patience = max_iters // 20 if max_iters >= 20 else 1

    for t in range(max_iters):
        if time.time() >= deadline:
            break

        g = _subgradient_max_conv(a)

        # linearly decaying learning rate
        eta = eta0 * (1.0 - t / max_iters)

        a = _exp_update(a, g, eta)

        cur_score = _objective_norm1(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            print(f"[opt] iter {t:6d}  new best O={best_score:.6f}")
        else:
            no_improve += 1
            if no_improve >= patience:
                # small jitter to escape plateaus
                jitter = np.random.randn(n) * 1e-4
                a = best_a + jitter
                a = np.clip(a, 0.0, None)
                a = _project_to_simplex(a)
                no_improve = 0

    return best_a, best_score

# Parametric families ----------------------------------------------------
def _shape_power(alpha: float, beta: float, n: int) -> np.ndarray:
    """a_i ∝ (i+β)^(-α) , i=1..n, then normalised to sum 1."""
    i = np.arange(1, n + 1, dtype=float)
    a = (i + beta) ** (-alpha)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _shape_power_exp(alpha: float, beta: float, gamma: float, n: int) -> np.ndarray:
    """Power‑law with exponential tail:
       a_i ∝ (i+β)^(-α)·exp(-γ·(i‑1)/n)
    """
    i = np.arange(1, n + 1, dtype=float)
    a = (i + beta) ** (-alpha) * np.exp(-gamma * (i - 1) / n)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _shape_beta(p: float, q: float, n: int) -> np.ndarray:
    """Beta‑density shape on the discrete grid."""
    x = np.arange(1, n + 1, dtype=float) / (n + 1)   # lies in (0,1)
    a = (x ** (p - 1.0)) * ((1.0 - x) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _shape_power_poly(alpha: float, beta: float, delta: float, n: int) -> np.ndarray:
    """Power‑law multiplied by a polynomial taper:
       a_i ∝ (i+β)^(-α)·(1‑(i‑1)/(n‑1))^δ
    """
    i = np.arange(1, n + 1, dtype=float)
    taper = 1.0 - (i - 1.0) / (n - 1.0)
    a = (i + beta) ** (-alpha) * (taper ** delta)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the supplied
    ``evaluate_sequence`` bound.  Returns the sequence as a plain ``list[float]``.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # a tiny safety margin

    # ------------------------------------------------------------------
    # Candidate lengths – powers of two work nicely with the FFT.
    lengths = [1024, 2048, 4096, 8192, 16384]

    best_seq = None
    best_score = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # 1️⃣  Multi‑scale random parametric sweep
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[search] exploring n = {n}")

        # ---- power law -------------------------------------------------
        for _ in range(1200):
            if time.time() >= deadline:
                break
            alpha = np.random.uniform(0.2, 0.9)
            beta = np.random.uniform(0.0, 30.0)
            seq = _shape_power(alpha, beta, n)
            score = _objective_norm1(seq)
            if score < best_score:
                best_score = score
                best_seq = seq
                best_n = n
                print(f"[update] n={n} O={score:.6f} (power α={alpha:.4f} β={beta:.4f})")
                if best_score <= 1.503:
                    break

        # ---- power law + exponential tail -------------------------------
        for _ in range(1200):
            if time.time() >= deadline:
                break
            alpha = np.random.uniform(0.2, 0.9)
            beta = np.random.uniform(0.0, 30.0)
            gamma = np.random.uniform(0.0, 2.0)
            seq = _shape_power_exp(alpha, beta, gamma, n)
            score = _objective_norm1(seq)
            if score < best_score:
                best_score = score
                best_seq = seq
                best_n = n
                print(f"[update] n={n} O={score:.6f} (powexp α={alpha:.4f} β={beta:.4f} γ={gamma:.4f})")
                if best_score <= 1.503:
                    break

        # ---- Beta‑density -----------------------------------------------
        for _ in range(800):
            if time.time() >= deadline:
                break
            p = np.random.uniform(0.1, 2.0)
            q = np.random.uniform(0.1, 5.0)
            seq = _shape_beta(p, q, n)
            score = _objective_norm1(seq)
            if score < best_score:
                best_score = score
                best_seq = seq
                best_n = n
                print(f"[update] n={n} O={score:.6f} (beta p={p:.4f} q={q:.4f})")
                if best_score <= 1.503:
                    break

        # ---- power law × polynomial taper -------------------------------
        for _ in range(800):
            if time.time() >= deadline:
                break
            alpha = np.random.uniform(0.2, 0.9)
            beta = np.random.uniform(0.0, 30.0)
            delta = np.random.uniform(0.0, 4.0)
            seq = _shape_power_poly(alpha, beta, delta, n)
            score = _objective_norm1(seq)
            if score < best_score:
                best_score = score
                best_seq = seq
                best_n = n
                print(f"[update] n={n} O={score:.6f} (powpoly α={alpha:.4f} β={beta:.4f} δ={delta:.4f})")
                if best_score <= 1.503:
                    break

        if best_score <= 1.503:
            break   # early exit – we already hit the target

    # ------------------------------------------------------------------
    # 2️⃣  Incorporate a previously discovered construction (if any)
    if 'height_sequence_1' in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=float)
            if prev.size > 0:
                prev = np.maximum(prev, 0.0)
                prev = prev / prev.sum()
                # Resample to the current best length if needed
                if best_n is not None and prev.size != best_n:
                    xs = np.linspace(0.0, 1.0, prev.size)
                    xt = np.linspace(0.0, 1.0, best_n)
                    prev = np.interp(xt, xs, prev)
                    prev = prev / prev.sum()
                prev_score = _objective_norm1(prev)
                if prev_score < best_score:
                    best_seq = prev
                    best_score = prev_score
                    best_n = best_n or prev.size
                    print(f"[prev] using previous construction O={prev_score:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 3️⃣  Mirror‑descent polishing (exponentiated‑gradient)
    remaining = deadline - time.time()
    if best_seq is not None and remaining > 5.0:
        max_iters = int(min(200_000, remaining * 200))   # up to ~200 iters per sec
        print(f"[opt] polishing – n={best_n} start O={best_score:.6f} max_iters={max_iters}")
        a_opt, score_opt = _optimize(best_seq, max_iters, eta0=0.6, deadline=deadline)
        if score_opt < best_score:
            best_seq = a_opt
            best_score = score_opt
            print(f"[opt] after polishing best O={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Local stochastic refinement (small mass swaps)
    remaining = deadline - time.time()
    if best_seq is not None and remaining > 5.0:
        n = best_seq.size
        max_steps = int(min(150_000, remaining * 400))   # cheap O(n) updates
        print(f"[local] hill‑climbing up to {max_steps} steps (n={n})")
        a = best_seq.copy()
        cur_score = best_score
        for step in range(max_steps):
            if time.time() >= deadline:
                break
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            # move a small fraction of mass from i to j
            delta = a[i] * np.random.uniform(0.0, 0.05)
            if delta <= 0.0:
                continue
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            # enforce feasibility (non‑negativity is guaranteed by construction)
            score = _objective_norm1(a_new)
            if score < cur_score - 1e-12:
                a = a_new
                cur_score = score
                best_seq = a_new
                best_score = score
                print(f"[local] step {step:6d}  new best O={score:.6f}")
                if best_score <= 1.503:
                    break

        # One final quick mirror‑descent pass after the swaps
        remaining = deadline - time.time()
        if remaining > 2.0:
            max_iters = int(min(30_000, remaining * 100))
            a_opt, score_opt = _optimize(best_seq, max_iters, eta0=0.3, deadline=deadline)
            if score_opt < best_score:
                best_seq = a_opt
                best_score = score_opt
                print(f"[opt2] final polishing best O={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Exact evaluation via the provided security‑checked function.
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)
        final_score = None

    # ------------------------------------------------------------------
    return best_seq.tolist()
