"""
Propose a candidate step‑function that minimises the bound
    O = 2·n·max(conv(a,a))   (with sum(a)=1)

Algorithm
---------
* Uses exponentiated‑gradient (mirror descent) on the log‑weights u_i = log a_i.
* The gradient of the (smoothed) max‑convolution is obtained via a
  correlation computed with a single FFT pair.
* Adam‑style adaptive learning rates are applied to u for faster convergence.
* A soft‑max weighting of the convolution entries is used early on;
  later the algorithm switches to an active‑set (hard‑max) weighting.
* Periodic jitter and occasional restarts from the best‑so‑far solution
  avoid stagnation.
* The length is fixed to 16000 (the largest that still fits a power‑of‑2 FFT
  of size 32768).  This gives the optimizer many degrees of freedom while
  keeping each iteration cheap (<~1 ms).

The function respects the 1000 s time budget, prints progress and finally
returns the best sequence (as a plain ``list[float]``) found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution a * a using a real FFT of length L."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return conv


def _correlation_fft(w: np.ndarray, a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear correlation g_i = Σ_t w[i+t]·a[t] ,  i = 0,…,n‑1.
    Implemented via FFT of length L (≥ len(w)+len(a)‑1).
    """
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[: a.size]

# Main search routine
# ----------------------------------------------------------------------

                     budget_s: int = 1000,
                     **kwargs) -> list:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence found as a plain list of floats.
    """

    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # 1) Length and initialisation
    # --------------------------------------------------------------
    # The FFT length for n=16000 (2n‑1=31999) is 32768 = 2**15.
    n = 16000
    L = _next_pow2(2 * n - 1)               # = 32768 for n <= 16384

    # Initialise from a known solution if it exists, otherwise uniform.
    try:
        a0 = np.asarray(height_sequence_1, dtype=float)

        a0 = np.maximum(a0, 0.0)
