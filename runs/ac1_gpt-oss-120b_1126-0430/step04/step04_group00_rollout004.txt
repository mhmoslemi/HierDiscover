def propose_candidate(seed=0, budget_s=1000):
    import time, numpy as np, math
    # configuration
    n_initial = 12000  # start size
    max_n = 25000
    # store global best across runs
    best_score = float('inf')
    best_seq = None
    start = time.time()
    dead = start + budget_s - 5.0
    run_id=0
    while time.time()+0.5 < dead:
        # randomly pick exponent p
        p = np.random.uniform(0.45, 0.75)
        n = n_initial
        a = _power_law_distribution(n, p)
        # random jitter with softmax transform: we work in logits
        z = _logits_from_distribution(a) + 0.01 * np.random.randn(n)
        # Adam state
        m = np.zeros_like(z)
        v = np.zeros_like(z)
        beta1, beta2 = 0.9, 0.999
        eps_adam = 1e-8
        base_alpha = 0.2
        eps_factor = 0.08
        eps_factor_decay = 0.9998
        # loop for this run
        it=0
        last_improve_iter=0
        while time.time() < dead and it < 200000:
            it+=1
            a = _softmax(z)
            max_conv, grad_a = _conv_max_and_grad(a, eps_factor)
            cheap_score = 2.0 * n * max_conv
            if cheap_score < best_score:
                best_score = cheap_score
                best_seq = a.copy()
                last_improve_iter = it
                print(f"[Run {run_id} iter {it}] NEW GLOBAL BEST cheap={cheap_score:.6f}")
            mu = np.dot(grad_a, a)
            grad_z = a * (grad_a - mu)
            m = beta1*m + (1-beta1)*grad_z
            v = beta2*v + (1-beta2)*(grad_z**2)
            m_hat = m/(1-beta1**it)
            v_hat = v/(1-beta2**it)
            alpha_t = base_alpha/(math.sqrt(it)+1.0)
            step_dir = - m_hat/(np.sqrt(v_hat)+eps_adam)
            # simple line search
            trial_alpha = alpha_t
            for _ in range(5):
                z_new = z + trial_alpha*step_dir
                z_new = np.clip(z_new, -50, 50)
                a_new = _softmax(z_new)
                new_score = 2.0 * n * _conv_max(a_new)
                if new_score <= cheap_score:
                    z = z_new
                    break
                trial_alpha *= 0.5
            else:
                z += alpha_t * step_dir
                z = np.clip(z, -50, 50)
            # random perturbation

                z += 0.02*np.random.randn(n)
                m *= 0.5
                v *= 0.5
            # smoothing decay

            # maybe enlarge support

#!/usr/bin/env python
"""
Multi‑restart smoothed‑max Adam optimiser on the probability simplex.

The objective (after normalising Σa_i = 1) is
    J(a) = 2·n·max_k (a*a)[k] .
The nonsmooth max is approximated by a log‑sum‑exp (soft‑max) with a
temperature ε = ε_factor·max(b).  The gradient of the smooth objective
is 4·n·corr(a,w) where w_k ∝ exp(b_k/ε).  Adam updates are applied to the
log‑it (soft‑max) parametrisation a = softmax(z).  Several independent
restarts (different random power‑law exponents, optional use of a previously
found construction) are performed within the 1000 s budget.  The best
candidate is periodically enlarged (linear interpolation) up to a support
size of at most 25000 and fine‑tuned again.  Finally the exact
``evaluate_sequence`` is called and the best sequence is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – must be top‑level (no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable softmax; returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it representation that yields `a` after softmax."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """
    Smallest power‑of‑two ≥ 3·n‑2,
    enough for a linear convolution of length (2·n‑1) with a vector of length n.
    """
    return 1 << ((3 * n - 2).bit_length())


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    total = new_a.sum()
    if total == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= total
    return new_a


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """Normalised power‑law a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _conv_max_and_grad(a: np.ndarray, eps_factor: float) -> tuple[float, np.ndarray]:
    """
    Compute max entry of a*a and the gradient of the smoothed objective.

    Parameters
    ----------
    a : np.ndarray
        Probability vector (sum = 1, non‑negative).
    eps_factor : float
        Multiplicative factor for the smoothing constant:
        eps = eps_factor * max(conv).

    Returns
    -------
    max_conv : float
        Maximum of the self‑convolution a*a.
    grad_a : np.ndarray
        Gradient of 2·n·smooth_max w.r.t. a (already multiplied by 4·n).
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    # FFT of a (zero‑padded)
    a_fft = np.fft.rfft(a, L)

    # ----- self‑convolution a*a (linear, length 2n‑1) -----
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # ----- smoothing weights w (soft‑max over conv) -----
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv           # for numerical stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or np.isnan(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # ----- cross‑correlation of a and w (linear, length 3n‑2) -----
    w_fft = np.fft.rfft(w, L)
    # correlation = ifft( conj(a_fft) * w_fft )
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # gradient of the smoothed objective (already includes factor 4·n)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a


def _conv_max(a: np.ndarray) -> float:
    """Return max entry of a*a (no gradient)."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())

# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) beats the target bound.
    The routine respects the given time budget (~1000 s) and uses several
    independent Adam optimisations on the probability simplex, occasionally
    enlarging the support up to 25000 entries.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # small safety margin

    # ------------------------------------------------------------------
    # 0. Global configuration
    # ------------------------------------------------------------------
    n_start = 8000               # start size for the cheap restarts
    n_max = 25000                # absolute maximum length we will allow
    eps_factor_start = 0.09      # initial smoothing factor (ε = eps·max)
    eps_factor_min = 5e-4
    eps_factor_decay = 0.9996
    base_alpha_start = 0.22      # initial learning‑rate for logits
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    best_score = float("inf")
    best_seq = None
    run_id = 0

    # ------------------------------------------------------------------
    # Helper: run a single Adam optimisation instance
    # ------------------------------------------------------------------
    def _run_adam(a0: np.ndarray, n: int, time_allow: float):
        """Perform Adam optimisation on a given start distribution."""
        nonlocal best_score, best_seq, run_id

        # Initialise logits with a small jitter
        z = _logits_from_distribution(a0) + 0.01 * np.random.randn(n)

        m = np.zeros_like(z)
        v = np.zeros_like(z)

        eps_factor = eps_factor_start
        it = 0
        last_improve_it = 0
        report_interval = 4000

        # optimisation loop
        while time.time() < deadline and (time.time() - start_time) < time_allow:
            it += 1

            a = _softmax(z)                     # always normalised to sum = 1
            max_conv, grad_a = _conv_max_and_grad(a, eps_factor)
            cheap_score = 2.0 * n * max_conv

            if cheap_score < best_score:
                best_score = cheap_score
                best_seq = a.copy()
                last_improve_it = it
                print(f"[Run {run_id} it {it:07d}] NEW GLOBAL BEST = {cheap_score:.6f}")

            # ---- gradient w.r.t. logits (mirror descent on the simplex) ----
            mu = np.dot(grad_a, a)
            grad_z = a * (grad_a - mu)

            # Adam moment updates (bias‑corrected)
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            # step‑size schedule (inverse‑sqrt decay)
            alpha_t = base_alpha_start / (math.sqrt(it) + 1.0)
            step_dir = - m_hat / (np.sqrt(v_hat) + eps_adam)

            # back‑tracking line search (up to 5 halvings)
            trial_alpha = alpha_t
            accepted = False
            for _ in range(5):
                z_new = z + trial_alpha * step_dir
                z_new = np.clip(z_new, -50.0, 50.0)
                a_new = _softmax(z_new)
                new_score = 2.0 * n * _conv_max(a_new)
                if new_score <= cheap_score:
                    z = z_new
                    accepted = True
                    break
                trial_alpha *= 0.5

            if not accepted:
                # fall back to a modest step
                z += alpha_t * step_dir
                z = np.clip(z, -50.0, 50.0)

            # occasional random perturbation to escape plateaus
            if it % 5000 == 0 and np.random.rand() < 0.12:
                z += 0.02 * np.random.randn(n)
                m *= 0.6
                v *= 0.6

            # smooth‑max temperature decay
            eps_factor = max(eps_factor_min, eps_factor * eps_factor_decay)

            # periodic progress report (cheap)
            if it % report_interval == 0:
                print(f"[Run {run_id} it {it:07d}] score={cheap_score:.6f}")

            # ------------------------------------------------------------------
            # Optional support enlargement (once per run, when time permits)
            # ------------------------------------------------------------------
            if it == int(1.2e5) and n < n_max and (time.time() + 30) < deadline:
                new_n = min(n_max, int(n * 1.5) + 1)
                print(f"[Run {run_id}] enlarging support {n} → {new_n}")
                a_interp = _interpolate_distribution(best_seq, new_n)
                # Reset optimiser for the larger problem
                z = _logits_from_distribution(a_interp)
                m = np.zeros_like(z)
                v = np.zeros_like(z)
                n = new_n
                # reset schedule variables for the enlarged problem
                eps_factor = eps_factor_start
                it = 0
                last_improve_it = 0

    # ------------------------------------------------------------------
    # 1. Quick scan of a few power‑law exponents (very cheap) – provides a
    #    decent starting point if no global variable is available.
    # ------------------------------------------------------------------
    try:
        # If the platform already supplied a previous construction, use it.
        init_global = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        n0 = max(n_start, init_global.shape[0])
        a0 = np.maximum(init_global, 0.0)
        s = a0.sum()
        if s == 0.0:
            a0 = np.full_like(a0, 1.0 / a0.shape[0])
        else:
            a0 /= s
        print(f"Initial sequence taken from global variable, length = {n0}")
    except Exception:
        # No predefined sequence – create a quick power‑law candidate.
        n0 = n_start
        p_grid = np.linspace(0.45, 0.75, 15)
        a0 = None
        best_tmp = float("inf")
        for p in p_grid:
            if time.time() >= deadline:
                break
            cand = _power_law_distribution(n0, p)
            sc = 2.0 * n0 * _conv_max(cand)
            if sc < best_tmp:
                best_tmp = sc
                a0 = cand
        if a0 is None:
            a0 = _power_law_distribution(n0, 0.60)
        print(f"Power‑law start (p≈{p:.3f}), length = {n0}")

    # ------------------------------------------------------------------
    # 2. Run several independent restarts with random power‑law exponents.
    #    Each restart gets a modest time budget (≈ 80 s).  The best
    #    candidate across all runs is kept globally.
    # ------------------------------------------------------------------
    restart_time = (budget_s - 200) / max(1, 6)   # split remaining time
    for run_id in range(6):
        if time.time() >= deadline:

        a_start = _power_law_distribution(n_start, p)

        # Occasionally initialise from the best‑so‑far (to refine it)
        if np.random.rand() < 0.3 and best_seq is not None:
            a_start = _interpolate_distribution(best_seq, n_start)

        # Occasionally try a Dirichlet random start (more exploratory)
        if np.random.rand() < 0.15:
            a_start = np.random.dirichlet(np.ones(n_start) * 0.8)

        # Run Adam optimisation on this start
        _run_adam(a_start, n_start, time_allow=restart_time)

    # ------------------------------------------------------------------
    # 3. If the best sequence found so far is still far from the maximal
    #    support, enlarge it (once) and fine‑tune again until time runs out.
    # ------------------------------------------------------------------
    if best_seq is not None and best_seq.shape[0] < n_max and time.time() < deadline - 30:
        print(f"Enlarging best sequence {best_seq.shape[0]} → {n_max}")
        a_big = _interpolate_distribution(best_seq, n_max)
        _run_adam(a_big, n_max, time_allow=budget_s * 0.2)

    # 4. Final exact evaluation with the provided evaluate_sequence().
    # ------------------------------------------------------------------
    if best_seq is None:

        best_seq = _power_law_distribution(5000, 0.60)

        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    return final_seq
