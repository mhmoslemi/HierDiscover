* A polishing phase that runs exponentiated‑gradient (mirror descent)

  larger one (n≈8000).  Random restarts, jitter and simple time‑capped
  loops keep the whole search inside the 1000 s budget.

All helpers are defined at top level, no closures are used.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a power‑of‑two FFT."""
    n = a.size
    size = 1 << ((2 * n - 2).bit_length())     # next power of two ≥ 2n‑1
    A = np.fft.fft(a, n=size)
    conv = np.fft.ifft(A * A).real[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def _objective_norm(a: np.ndarray) -> float:
    """
    Objective 2·n·max(conv) assuming Σa = 1.
    a is expected to be a non‑negative 1‑D array.
    """
    n = a.size
    conv = _conv_fft(a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) w.r.t. a (a lies on the simplex).
    The sub‑gradient is the average of the gradients belonging to all
    (almost) maximising shifts.
    """
    n = a.size
    conv = _conv_fft(a)
    max_val = conv.max()
    # treat values within a tiny tolerance as maximal
    ks = np.where(np.abs(conv - max_val) < 1e-14)[0]
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        # each product a_i*a_j appears twice in the derivative
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
def _exp_update(a: np.ndarray, g: np.ndarray, eta: float) -> np.ndarray:
    """
    Exponentiated‑gradient update on the simplex.
    Operates in log‑space for numerical stability.
    """
    eps = 1e-15
    log_a = np.log(a + eps)
    log_a -= eta * g
    log_a -= np.max(log_a)          # shift to keep numbers moderate
    a_new = np.exp(log_a)
    a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
def _optimize(a0: np.ndarray, max_iters: int, eta0: float,
              deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) optimisation.
    Returns the best vector seen and its objective value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective_norm(a)

    n = a.size
    no_improve = 0
    patience = max_iters // 20 if max_iters >= 20 else 5

    for t in range(max_iters):
        if time.time() >= deadline:
            break

        g = _subgradient_max_conv(a)
        eta = eta0 * (1.0 - t / max_iters)   # linear decay

        a = _exp_update(a, g, eta)

        cur_score = _objective_norm(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            print(f"[opt] iter {t:6d}  new best O={best_score:.6f}")
        else:
            no_improve += 1
            if no_improve >= patience:
                # jitter to escape flat regions
                jitter = np.random.randn(n) * 1e-4
                a = best_a + jitter
                a = np.clip(a, 0.0, None)
                a = _project_to_simplex(a)
                no_improve = 0

    return best_a, best_score


# ----------------------------------------------------------------------
def _shape(alpha: float, beta: float, gamma: float, w: float,
           n: int) -> np.ndarray:
    """
    Generate a step‑function of length n:
        a_i ∝ (i+β)^(-α)·exp(-γ·(i‑1)/n)  (power–law + exponential tail)
        mixed with a uniform component of weight w.
    The result is normalised to sum‑1.
    """
    i = np.arange(1, n + 1, dtype=float)        # i = 1 … n
    base = (i + beta) ** (-alpha) * np.exp(-gamma * (i - 1) / n)

    uniform = np.full(n, 1.0 / n)
    a = (1.0 - w) * base + w * uniform

    s = a.sum()
    if s == 0.0:
        a = uniform
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the bound
    used by ``evaluate_sequence``.  The routine respects a 1000 s budget,
    prints intermediate progress and finally returns the best sequence.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Small size for fast exploration, large size for final polishing
    n_small = 2048
    n_large = 8192

    # Parameter bounds
    min_alpha, max_alpha = 0.2, 0.9
    min_beta, max_beta   = 0.0, 50.0
    min_gamma, max_gamma = 0.0, 1.0
    min_w, max_w         = 0.0, 0.3

    # ------------------------------------------------------------------
    # 1️⃣  Coarse grid search on the small size
    alphas = np.linspace(0.30, 0.70, 9)          # 9 values
    betas  = np.linspace(0.0, 30.0, 16)         # 16 values
    gammas = np.linspace(0.0, 0.60, 7)          # 7 values
    ws     = [0.0, 0.05, 0.10, 0.15]            # 4 values

    best_score = float("inf")
    best_params = None
    best_seq_small = None

    print("[grid] starting coarse search (small size)")
    for alpha in alphas:
        for beta in betas:
            for gamma in gammas:
                for w in ws:
                    if time.time() >= deadline:
                        break
                    seq = _shape(alpha, beta, gamma, w, n_small)
                    score = _objective_norm(seq)
                    if score < best_score:
                        best_score = score
                        best_params = (alpha, beta, gamma, w)
                        best_seq_small = seq
                        print(f"[grid] new best O={best_score:.6f} "
                              f"(α={alpha:.3f}, β={beta:.1f}, γ={gamma:.3f}, w={w:.2f})")
                if time.time() >= deadline:
                    break
            if time.time() >= deadline:
                break
        if time.time() >= deadline:
            break

    # If no candidate survived (unlikely), fall back to uniform
    if best_seq_small is None:
        best_seq_small = np.full(n_small, 1.0 / n_small)
        best_score = _objective_norm(best_seq_small)
        best_params = (0.5, 0.0, 0.0, 0.0)

    # unpack for convenience
    best_alpha, best_beta, best_gamma, best_w = best_params

    # ------------------------------------------------------------------
    # 2️⃣  Random local refinement (still on small size)
    print("[rand] starting random local refinement")
    while time.time() < deadline:
        # Gaussian perturbation of the current best parameters
        alpha = best_alpha + np.random.randn() * 0.02
        beta  = best_beta  + np.random.randn() * 0.5
        gamma = best_gamma + np.random.randn() * 0.04
        w     = best_w     + np.random.randn() * 0.02

        # Clip to the allowed ranges
        alpha = np.clip(alpha, min_alpha, max_alpha)
        beta  = np.clip(beta,  min_beta,  max_beta)
        gamma = np.clip(gamma, min_gamma, max_gamma)
        w     = np.clip(w,     min_w,     max_w)

        seq = _shape(alpha, beta, gamma, w, n_small)
        score = _objective_norm(seq)

        if score < best_score:
            best_score = score
            best_params = (alpha, beta, gamma, w)
            best_seq_small = seq
            best_alpha, best_beta, best_gamma, best_w = best_params
            print(f"[rand] improved O={best_score:.6f} "
                  f"(α={alpha:.3f}, β={beta:.2f}, γ={gamma:.3f}, w={w:.3f})")

        # occasional pure random restart
        if np.random.rand() < 0.001:
            alpha = np.random.uniform(min_alpha, max_alpha)
            beta  = np.random.uniform(min_beta, max_beta)
            gamma = np.random.uniform(min_gamma, max_gamma)
            w     = np.random.uniform(min_w, max_w)

    # ------------------------------------------------------------------
    # 3️⃣  Incorporate a previously discovered construction, if present
    if 'height_sequence_1' in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=float)
            prev = np.maximum(prev, 0.0)
            # resize to the small dimension if necessary
            if prev.size != n_small:
                prev = np.interp(np.linspace(0, prev.size - 1, n_small),
                                 np.arange(prev.size), prev)
            prev /= prev.sum()
            prev_score = _objective_norm(prev)
            if prev_score < best_score:
                best_seq_small = prev
                best_score = prev_score
                print(f"[prev] adopted previous sequence O={best_score:.6f}")
        except Exception:          # pragma: no cover
            pass

    # ------------------------------------------------------------------
    # 4️⃣  Polishing on the small size with mirror‑descent
    remaining = deadline - time.time()
    if remaining > 5.0:
        max_iters = int(min(100_000, remaining * 150))
        print(f"[opt_small] polishing start – max_iters={max_iters}")
        a_opt_small, score_opt_small = _optimize(best_seq_small,
                                                max_iters,
                                                eta0=0.7,
                                                deadline=deadline)
        if score_opt_small < best_score:
            best_seq_small = a_opt_small
            best_score = score_opt_small
            print(f"[opt_small] improved O={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Upscale to a larger dimension for the final phase
    # generate a fresh shape with the best parameters at the large size
    seq_large = _shape(best_alpha, best_beta, best_gamma, best_w, n_large)

    # also interpolate the best small‑size vector (it may be slightly better)
    seq_interp = np.interp(np.linspace(0, n_small - 1, n_large),
                           np.arange(n_small), best_seq_small)
    seq_interp /= seq_interp.sum()

    # blend both ideas and renormalise
    a0 = 0.5 * seq_large + 0.5 * seq_interp
    a0 = a0 / a0.sum()

    # ------------------------------------------------------------------
    # 6️⃣  Polishing on the large size
    remaining = deadline - time.time()
    if remaining > 5.0:
        # large‑size iterations are slower; use a conservative budget
        max_iters_large = int(min(50_000, remaining * 80))
        print(f"[opt_large] polishing start – max_iters={max_iters_large}")
        a_opt_large, score_opt_large = _optimize(a0,
                                                max_iters_large,
                                                eta0=0.5,
                                                deadline=deadline)
        # keep whichever is best
        if score_opt_large < best_score:
            best_seq = a_opt_large
            best_score = score_opt_large
            print(f"[opt_large] improved O={best_score:.6f}")
        else:
            best_seq = a0
    else:
        best_seq = a0

    # ------------------------------------------------------------------
    # 7️⃣  Final exact evaluation with the provided oracle
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:            # pragma: no cover
        print("[final] evaluate_sequence call failed:", e)
        final_score = None

    return best_seq.tolist()
