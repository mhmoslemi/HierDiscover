Improved three‑phase optimizer for the step‑function objective

    O(a) = 2·n·max(conv(a,a)) / (sum(a)^2) .

The optimizer is based on the previous Adam + sub‑gradient + annealing
pipeline but with several practical upgrades:

* longer and less aggressive annealing (α grows to 5·10⁵) – the surrogate
  becomes almost the true max‑operator.
* a slightly larger base learning‑rate and no hard pruning of tiny
  components – this keeps useful degrees of freedom.
* the stochastic refinement (phase 3) is run for many more iterations,
  uses two kinds of moves (mass‑transfer and pair‑swap), and employs a
  slower temperature decay, giving the search more chance to escape
  plateaus.
* multiple independent restarts are performed until the wall‑time is
  exhausted; the globally best sequence is returned.

All helper functions are defined at module level (no closures) and the
code respects the 1000 s wall‑time limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:                     # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full self‑convolution via FFT (length L is a power of two)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return a * a (full convolution) using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Exact objective for a simplex‑normalised vector (sum == 1)
def _objective_simplex(a: np.ndarray, n: int, L: int) -> float:
    """2·n·max(conv(a,a))  (a is assumed to sum to 1)."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv)  (used in the non‑smooth phases)
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of the maximal convolution entry w.r.t. a."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)          # all maximising indices
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
# Gradient of the log‑sum‑exp (smooth‑max) surrogate
def _smooth_gradient(a: np.ndarray, conv: np.ndarray, alpha: float, L: int) -> np.ndarray:
    """
    Gradient of  (2·n/α)·log∑_k exp(α·conv_k) .
    The result reduces to 2·corr(w,a) where w_k ∝ exp(α·conv_k).
    """
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)          # stabilise
    w = exp_scaled / exp_scaled.sum()

    # correlation w ⋆ a   (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n_len = a.size
    grad = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]
    return grad

    Returns (best_sequence, best_score) found before time expires.

    a = a0.copy()
    best_seq = a.copy()
    best_score = _objective_simplex(best_seq, n, L)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    m = np.zeros_like(a)
    v = np.zeros_like(a)

                best_seq = a.copy()
                print(f"[{time.time():.1f}] Sub‑grad improved O={best_score:.6f}")

        # temperature decay (slow)
        T *= 0.99985
        it_loc += 1

        # restart if stagnation lasts too long
        if no_improve >= no_improve_limit:
            a = best_seq.copy()
            a += np.random.randn(n) * 0.0012
            a = np.clip(a, 0, None)
            a = _project_to_simplex(a)
            conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max()
            no_improve = 0

    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimizes the evaluation
    function.  The routine runs several independent optimisation runs,
    each consisting of:

        1️⃣  Adam on a log‑sum‑exp surrogate (smooth max)
        2️⃣  exact sub‑gradient descent
        3️⃣  long stochastic refinement with annealing

    The best sequence seen during the whole wall‑time is returned as a
    plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.15          # tiny safety margin

    # ------------------------------------------------------------------
    # Determine a reasonable default length.
    # If a previous construction is supplied we keep its length,
    # otherwise we use 4600 (a good trade‑off between quality and speed).
    if 'height_sequence_1' in globals():
        a0 = np.array(height_sequence_1, dtype=float)
        a0 = np.maximum(a0, 0.0)
        if a0.sum() == 0.0:
            a0 = np.full_like(a0, 1.0)
        a0 = a0 / a0.sum()
        n = a0.size
        print(f"[init] using supplied sequence, n={n}")
    else:
        n = 4600
        a0 = np.full(n, 1.0 / n)
        print(f"[init] using uniform start, n={n}")

    # FFT length – next power‑of‑two that comfortably covers the correlation length
    L = 1 << ((3 * n - 2).bit_length())

    # Global best across all restarts
    global_best_seq = a0.copy()
    global_best_score = _objective_simplex(global_best_seq, n, L)

    # Allocate roughly 90 % of the wall‑time to optimisation runs,
    # the remaining 10 % to the final sanity check.
    optimisation_budget = (deadline - start_time) * 0.9

    while time.time() - start_time < optimisation_budget:
        # each restart gets about 8 % of the overall budget
        remaining = deadline - time.time()
        run_budget = min(optimisation_budget * 0.08, remaining)

        # initialise either from the global best (with jitter) or from a fresh uniform vector
        if np.random.rand() < 0.6:
            a_start = global_best_seq.copy()
            a_start += np.random.randn(n) * 0.001
            a_start = np.clip(a_start, 0, None)
            a_start = _project_to_simplex(a_start)
        else:
            a_start = np.random.exponential(scale=1.0, size=n)
            a_start /= a_start.sum()

        seq, score = _run_optimisation(a_start, n, L, run_budget, time.time())
        if score < global_best_score:
            global_best_score = score
            global_best_seq = seq.copy()
            print(f"[restart] new global best O={global_best_score:.6f}")

    # ------------------------------------------------------------------
    # final verification with the provided evaluator
    try:
        final_score = evaluate_sequence(global_best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence failed:", exc)

    return global_best_seq.tolist()
