def _project_simplex(v):
    # ensure v is numpy array
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zeros?
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1  # index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

#!/usr/bin/env python
"""
A two‑phase optimizer for the step‑function bound.

Phase 1 – Adam on the log‑it (soft‑max) parametrisation.
  * Uses the smoothed log‑sum‑exp approximation to the max‑convolution.
  * Larger initial learning‑rate (0.3) and a slower decay.
  * Frequent back‑tracking line‑search guarantees a non‑increasing cheap
    score (2·n·max(conv)).
  * Random jitter and momentum restarts help escape plateaus.

Phase 2 – lightweight simulated‑annealing refine on the simplex.
  * Small mass transfers between two random indices are proposed.
  * Accept improvements always; otherwise accept with Metropolis probability.
  * Temperature decays geometrically, allowing occasional uphill steps.

Multiple independent restarts (up to the time budget) are performed;
the best sequence seen overall is finally evaluated with ``evaluate_sequence``.
All operations respect the 1000 s budget.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level, no closures
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max; returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Convert a probability vector to log‑it space (inverse softmax)."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """
    Smallest power‑of‑two ≥ 3·n‑2, sufficient for a linear convolution of
    length (2·n‑1) and the subsequent correlation of length (3·n‑2).
    """
    return 1 << ((3 * n - 2).bit_length())


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """Normalised power‑law a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    total = new_a.sum()
    if total == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= total
    return new_a


def _conv_max(a: np.ndarray) -> float:
    """Return the maximum entry of the (linear) self‑convolution a*a."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float) -> tuple[float, np.ndarray]:
    """
    Compute max entry of a*a and the gradient of the smoothed objective.

    Parameters
    ----------
    a : np.ndarray
        Probability vector (sum = 1, non‑negative).
    eps_factor : float
        Multiplicative factor for the smoothing constant:
        eps = eps_factor * max(conv).

    Returns
    -------
    max_conv : float
        Maximum of the self‑convolution a*a.
    grad_a : np.ndarray
        Gradient of 2·n·smooth_max w.r.t. a (already multiplied by 4·n).
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    # FFT of a (zero‑padded)
    a_fft = np.fft.rfft(a, L)

    # self‑convolution (linear, length 2n‑1)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # smoothed soft‑max weights
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv                       # for numerical stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or np.isnan(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # cross‑correlation a⋆w (linear, length 3n‑2)
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # gradient of the smoothed objective (4·n * correlation)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights (the step‑function) that minimises
    ``evaluate_sequence``.  The routine respects a ~1000 s time budget.
    """
    np.random.seed(seed)
    start = time.time()
    # keep a small safety margin for the final evaluation
    deadline = start + budget_s - 5.0

    # ------------------------------------------------------------------
    # 0. Problem size and initialisation
    # ------------------------------------------------------------------
    n = 20000                     # fixed support length (large enough)
    try:
        # Prefer the previously discovered sequence if available
        init_seq = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if init_seq.size != n:
            # Interpolate / trim to the desired length
            init_seq = _interpolate_distribution(init_seq, n)
        a0 = np.maximum(init_seq, 0.0)
        s0 = a0.sum()
        if s0 == 0.0:
            a0 = np.full(n, 1.0 / n)
        else:
            a0 /= s0
        print(f"Initial sequence taken from global variable, length={n}")
    except Exception:
        # Fall back to a random power‑law start
        p0 = np.random.uniform(0.4, 0.8)
        a0 = _power_law_distribution(n, p0)
        print(f"Power‑law start p={p0:.3f}, length={n}")

    # Record the best sequence seen so far (global)
    best_seq = a0.copy()
    best_score = 2.0 * n * _conv_max(best_seq)
    print(f"Initial cheap score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1. Optimisation parameters
    # ------------------------------------------------------------------
    # Adam on logits
    base_alpha = 0.30          # larger base learning‑rate
    eps_factor = 0.15          # initial smoothing factor (ε = eps_factor·max)
    eps_factor_min = 5e-5
    eps_factor_decay = 0.9997
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    # Loop control
    max_iters_per_restart = 400_000
    restart_budget = 2          # at most two restarts (adjustable)
    restarts_done = 0

    # ------------------------------------------------------------------
    # 2. Main optimisation loop (multiple restarts)
    # ------------------------------------------------------------------
    while time.time() < deadline and restarts_done < restart_budget:
        restarts_done += 1
        print(f"\n=== Restart {restarts_done}/{restart_budget} ===")

        # ----- pick a starting point -------------------------------------------------
        if restarts_done == 1:
            a = best_seq.copy()                     # continue from the global best
        else:
            p = np.random.uniform(0.3, 0.9)
            a = _power_law_distribution(n, p)       # fresh random start
            print(f"  New random start p={p:.3f}")

        # small jitter to avoid exact replicas of a stationary point
        a += 0.01 * np.random.randn(n)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        # ----- convert to logits ------------------------------------------------------
        z = _logits_from_distribution(a)
        m = np.zeros_like(z)
        v = np.zeros_like(z)

        # local bookkeeping
        it = 0
        local_best_seq = a.copy()
        local_best_score = 2.0 * n * _conv_max(a)
        last_improve = 0

        # ----- Adam optimisation -------------------------------------------------------
        while time.time() < deadline and it < max_iters_per_restart:
            it += 1

            # current point on simplex
            a = _softmax(z)

            # cheap evaluation + smoothed gradient
            max_conv, grad_a = _conv_max_and_grad(a, eps_factor)
            cheap_score = 2.0 * n * max_conv

            # keep global / local bests
            if cheap_score < local_best_score:
                local_best_score = cheap_score
                local_best_seq = a.copy()
                last_improve = it
                if cheap_score < best_score:
                    best_score = cheap_score
                    best_seq = a.copy()
                    print(f"[Restart {restarts_done} it {it}] NEW GLOBAL BEST cheap = {cheap_score:.6f}")

            # gradient w.r.t. logits (mirrored descent on the simplex)
            mu = np.dot(grad_a, a)
            grad_z = a * (grad_a - mu)

            # Adam moment updates (bias‑corrected)
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            # step size schedule (inverse‑sqrt decay)
            alpha_t = base_alpha / (math.sqrt(it) + 1.0)
            step_dir = - m_hat / (np.sqrt(v_hat) + eps_adam)
            trial_alpha = alpha_t

            # back‑tracking line search (up to 5 halvings)
            accepted = False
            for _ in range(5):
                z_new = z + trial_alpha * step_dir
                z_new = np.clip(z_new, -50.0, 50.0)
                a_new = _softmax(z_new)
                new_score = 2.0 * n * _conv_max(a_new)
                if new_score <= cheap_score:
                    z = z_new
                    accepted = True
                    break
                trial_alpha *= 0.5

            if not accepted:
                # fallback: a modest Adam step without line‑search guarantee
                z = z + alpha_t * step_dir
                z = np.clip(z, -50.0, 50.0)

            # occasional random perturbation to shake out of plateaus
            if it % 8000 == 0 and np.random.rand() < 0.2:
                z += 0.02 * np.random.randn(n)
                m *= 0.7
                v *= 0.7

            # lower the smoothing constant gradually
            eps_factor = max(eps_factor_min, eps_factor * eps_factor_decay)

            # early‑stop condition for this restart (no improvement for long)
            if it - last_improve > 200_000:
                print(f"[Restart {restarts_done}] Stagnation after {it} iterations.")
                break

        # ----- Phase 2: simulated‑annealing refinement on the simplex -----------------
        # Use the best point found in this restart as the annealing start.
        a_refine = local_best_seq.copy()
        score_refine = local_best_score

        temp = 0.01                 # initial temperature
        temp_decay = 0.99995
        anneal_iters = 80_000

        for r_it in range(anneal_iters):
            if time.time() >= deadline:
                break

            # propose moving a tiny amount of mass from i -> j
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            # delta is a small fraction of a[i]; ensure non‑negativity
            delta = 0.001 * a_refine[i] * np.random.rand()
            if delta <= 0.0:
                continue
            a_new = a_refine.copy()
            a_new[i] -= delta
            a_new[j] += delta
            if a_new[i] < 0.0:
                continue      # keep feasibility

            new_score = 2.0 * n * _conv_max(a_new)

            # Metropolis acceptance
            if new_score < score_refine or np.random.rand() < math.exp((score_refine - new_score) / max(temp, 1e-12)):
                a_refine = a_new
                score_refine = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_seq = a_refine.copy()
                    print(f"[Anneal] New GLOBAL BEST cheap = {new_score:.6f}")

            # temperature decay
            temp *= temp_decay

        # record the outcome of this restart
        if score_refine < best_score:
            best_score = score_refine
            best_seq = a_refine.copy()
            print(f"[Restart {restarts_done}] End‑restart best cheap = {score_refine:.6f}")

    # ----------------------------------------------------------------------
    # 3. Final evaluation with the provided ``evaluate_sequence``
    # ----------------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)  # assumed to be in the environment
        print(f"\nFinal exact score = {final_score:.6f}")
    except Exception as exc:
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # ----------------------------------------------------------------------
    # 4. Return the best sequence found
    # ----------------------------------------------------------------------
    return final_seq
