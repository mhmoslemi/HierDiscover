#!/usr/bin/env python
"""
propose_candidate
-----------------

A two‑stage optimizer for the step‑function problem.

Stage 1 – a linear‑programming “direction step”.  For the current
probability vector `a` we compute `M = max(conv(a,a))`.  Keeping `M`
fixed we solve the LP

    maximize   sum(b)
    subject to b ≥ 0,
               conv(a,b) ≤ M                 (elementwise)

The solution `b` tries to put as much probability mass as possible
while respecting the current worst‑case convolution value `M`.
After normalising `b` to the simplex we move a fraction `t` towards it:
    a ← (1−t)·a + t·b .
Repeated application drives `M` down.

Stage 2 – cheap stochastic hill‑climbing.  From the best vector found
by the LP stage we perform many tiny “mass‑transfer’’ moves
(i←j) accepted by a Metropolis rule, together with occasional exact
sub‑gradient steps for the true nonsmooth objective.
The whole search stops after the 1000 s budget and returns the best
sequence seen.

The algorithm respects all constraints of ``evaluate_sequence`` and
does not use any file I/O.  It works on a single process but may be
run with any number of CPU cores (the heavy LP solves are done by
the native solver).
"""

import time
import math
import numpy as np
import cvxpy as cp

# ------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x >= 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        # all non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# FFT‑based full self‑convolution (length 2·n‑1)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return full convolution a*a using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ------------------------------------------------------------
# Sub‑gradient of max(conv) w.r.t. a (used in the local stage)
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) (the infinity‑norm of the autocorrelation)."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)          # indices where the max is attained
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]                # derivative of a_i a_j + a_j a_i
    if ks.size:
        g /= ks.size
    return g


# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that
    minimises the evaluation function.  Returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a tiny safety margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 0.15

    # ------------------------------------------------------------
    # Initialise the probability vector `a`
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=float)
        a = np.clip(a, 0.0, None)
        if a.sum() == 0.0:
            a = np.full_like(a, 1.0)
        a = a / a.sum()
        n = a.size
        print(f"[init] using supplied sequence, n={n}")
    else:
        # moderate size – enough degrees of freedom, small enough for the LP
        n = 2500
        a = np.full(n, 1.0 / n)
        print(f"[init] uniform start, n={n}")

    # FFT length – next power of two larger than 3·n‑2 (convolution + correlation)
    L = 1 << ((3 * n - 2).bit_length())

    # ------------------------------------------------------------
    # Simple objective (sum = 1) :  O = 2·n·max(conv)
    def _objective(a_vec: np.ndarray) -> float:
        conv = _conv_fft(a_vec, L)
        return 2.0 * n * conv.max()

    best_a = a.copy()
    best_score = _objective(best_a)
    print(f"[init] O={best_score:.6f}")

    # ------------------------------------------------------------
    # Stage 1 – LP direction steps
    # Build a reusable cvxpy problem: maximize sum(b) subject to conv(a,b) ≤ M
    a_param = cp.Parameter(n, value=a)
    M_param = cp.Parameter(nonneg=True)
    b_var = cp.Variable(n, nonneg=True)
    # cvxpy's conv returns a vector of length 2·n‑1
    conv_expr = cp.conv(a_param, b_var)
    constraints = [conv_expr <= M_param]
    lp_problem = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

    max_lp_iters = 12                     # a small fixed budget for the LP stage
    lp_iter = 0
    while time.time() < deadline * 0.55 and lp_iter < max_lp_iters:
        conv_vals = _conv_fft(a, L)
        M = conv_vals.max()
        # Update parameters
        a_param.value = a
        M_param.value = M

        # Solve the LP (ECOS is the default interior‑point solver for LPs)
        try:
            lp_problem.solve(solver='ECOS', warm_start=True, max_iters=5000, verbose=False)
        except Exception as exc:
            print(f"[LP] solver exception at iter {lp_iter}: {exc}")
            break

        b_val = b_var.value
        if b_val is None:
            print(f"[LP] no solution returned at iter {lp_iter}")
            break

        b_val = np.maximum(b_val, 0.0)

        # Normalise b onto the simplex (preserves feasibility of the next step)

        # Mixing coefficient – start aggressive, then become cautious
        t = 0.7 if lp_iter < 4 else 0.3

        a = _project_to_simplex(a)   # numerical safety

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[LP] iter {lp_iter} new best O={best_score:.6f}")

        lp_iter += 1

    # ------------------------------------------------------------
    # Stage 2 – cheap stochastic refinement (mass‑transfer + sub‑gradient)
    print(f"[refine] start local search, best O={best_score:.6f}")
    cur_a = best_a.copy()
    cur_conv = _conv_fft(cur_a, L)
    cur_obj = _objective(cur_a)

    # Simulated‑annealing style parameters
    temperature = 0.12
    it_local = 0
    max_local_iters = int(2e5)          # will be limited by the remaining time

    while time.time() < deadline and it_local < max_local_iters:
        # ---- random mass transfer move ----
        i, j = np.random.randint(n, size=2)
        if i != j and cur_a[i] > 1e-14:
            # move at most 9 % of the source mass
            delta = cur_a[i] * np.random.rand() * 0.09
            candidate = cur_a.copy()
            candidate[i] -= delta
            candidate[j] += delta
            candidate = _project_to_simplex(candidate)

            conv_cand = _conv_fft(candidate, L)
            obj_cand = _objective(candidate)

            # Metropolis acceptance
            if obj_cand < cur_obj or np.random.rand() < math.exp((cur_obj - obj_cand) / max(temperature, 1e-14)):
                cur_a = candidate
                cur_conv = conv_cand
                cur_obj = obj_cand
                if cur_obj < best_score:
                    best_score = cur_obj
                    best_a = cur_a.copy()
                    print(f"[refine] iter {it_local} new best O={best_score:.6f}")

                # slowly cool
                temperature *= 0.9997

        # ---- occasional exact sub‑gradient step ----
        if it_local % 15 == 0:
            subgrad = _subgradient_max_conv(cur_a, cur_conv)
            # step size decays linearly with the remaining budget
            eta = 0.03 * (1.0 - it_local / max_local_iters)
            cand = cur_a - eta * subgrad
            cand = _project_to_simplex(cand)

            conv_cand = _conv_fft(cand, L)
            obj_cand = _objective(cand)

            if obj_cand < cur_obj:
                cur_a, cur_conv, cur_obj = cand, conv_cand, obj_cand
                if cur_obj < best_score:
                    best_score = cur_obj
                    best_a = cur_a.copy()
                    print(f"[sgd] iter {it_local} new best O={best_score:.6f}")

        it_local += 1

    # ------------------------------------------------------------
    # Final verification with the provided evaluator
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence failed:", exc)

    # Return a plain Python list (the evaluator expects a list of floats)
    return best_a.tolist()
