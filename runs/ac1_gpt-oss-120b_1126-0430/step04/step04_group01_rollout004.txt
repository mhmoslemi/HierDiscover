We minimise the squared L₂‑error of the full auto‑convolution to the
theoretically optimal constant value 1/(2n‑1).  The objective

        J(a) = ‖a∗a – 1/(2n‑1)·𝟙‖₂²

has a simple gradient  ∇J = 4·( (a∗a – avg)  ⋆  a ),
where “⋆” denotes cross‑correlation.  Both the convolution and the
cross‑correlation are evaluated by a single FFT (size ≈ 3n) per
iteration.  An Adam optimiser with a decaying entropy term (λ·∑a·log a)
keeps the iterates on the probability simplex (∑a=1, a≥0).

Stage B – *local refinement*  
Starting from the best vector found in Stage A we run a cheap stochastic
search that repeatedly transfers a random fraction of mass from one index
to another.  Moves are accepted if they improve the exact objective or,
with a Metropolis probability, if they are slightly worse (simulated‑
annealing).  Every few iterations we also take a sub‑gradient step on the
exact max‑convolution entry (the gradient of max(conv) is the average of
the usual auto‑correlation vectors of the active indices).  The current
best solution is kept and returned.

Both stages respect the 1000 s wall‑time limit.  All operations are
vectorised and use only NumPy/FFT, so the routine runs comfortably within
the budget on a single CPU.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:               # numerical safety
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution (zero‑padded) via FFT
def _conv_fft(a: np.ndarray, n: int, L: int) -> np.ndarray:
    """
    Return the full linear convolution a∗a (length 2·n‑1) using a real FFT
    of length L (L ≥ 2·n‑1).  The caller must supply n = len(a).
    """
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * n - 1]


# ----------------------------------------------------------------------
# Exact objective for a vector that is already normalised (Σa = 1)
def _objective_simplex(a: np.ndarray, n: int, L: int) -> float:
    conv = _conv_fft(a, n, L)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) (averaged over all maximising indices)
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Gradient of max(conv) w.r.t. a (averaged if several indices tie)."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)      # all ties
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
# Gradient of the L2‑flattening objective J(a) = ||a∗a - avg||²
def _grad_flattening(a: np.ndarray, conv: np.ndarray, avg: float,
                     n: int, L: int) -> np.ndarray:
    """
    Compute ∇J = 4·( (conv - avg) ⋆ a ), where “⋆” is cross‑correlation.
    Both convolutions are performed with the same FFT length L.
    """
    e = conv - avg                               # length 2n‑1
    a_rev = a[::-1]

    # FFTs of e and a_rev (zero‑padded to the same length L)
    fft_e = np.fft.rfft(e, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)

    # cross‑correlation = irfft(fft(e) * fft(a_rev))
    corr_full = np.fft.irfft(fft_e * fft_a_rev, n=L)

    # extract the n entries that correspond to the gradient
    grad = 4.0 * corr_full[n - 1: n - 1 + n]
    return grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that minimises the provided evaluation
    function.  Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.3   # safety margin

    # --------------------------------------------------------------
    # Problem size ---------------------------------------------------
    # If a previous construction is available we start from it,
    # otherwise we create a fresh uniform vector.
    if "height_sequence_1" in globals():
        base = np.array(height_sequence_1, dtype=float)
        n0 = base.size
        # we use a slightly larger n to give the optimiser more freedom
        n = max(8000, n0)                     # 8 k is still cheap for an FFT
        a = np.zeros(n, dtype=float)
        a[:n0] = base
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-12:                    # fallback if the base is all zeros
            a = np.full(n, 1.0 / n)
        else:
            a = a / a.sum()
        print(f"[init] start from supplied sequence (n0={n0}), padded to n={n}")
    else:
        n = 8000
        a = np.full(n, 1.0 / n)
        print(f"[init] start from uniform, n={n}")

    # FFT length – must accommodate the longest convolution we need
    L = 1 << ((3 * n - 2).bit_length())      # next power of two ≥ 3n‑2

    # --------------------------------------------------------------
    # Stage A – flattening the auto‑convolution --------------------
    best_seq = a.copy()
    best_score = _objective_simplex(best_seq, n, L)
    print(f"[stage‑A] initial O={best_score:.6f}")

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    max_iter_A = 250_000                     # generous upper bound
    it_A = 0
    avg_const = 1.0 / (2 * n - 1)            # target constant value for a∗a

    # entropy weight – large at the start to keep the distribution spread,
    # linearly annealed to zero
    lam_start = 0.12
    lam_end = 0.0

    # we allocate roughly 35 % of the time budget to this stage
    stageA_deadline = start_time + 0.35 * budget_s

    while time.time() < stageA_deadline and it_A < max_iter_A:
        # ----- full convolution -----
        conv = _conv_fft(a, n, L)                     # length 2n‑1
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[stage‑A] iter {it_A:,} new best O={best_score:.6f}")

        # ----- gradient of the L2‑flattening objective -----
        grad_flat = _grad_flattening(a, conv, avg_const, n, L)

        # ----- annealed entropy term -----
        lam = lam_start * max(0.0, 1.0 - it_A / max_iter_A)
        grad_ent = lam * (np.log(a + 1e-20) + 1.0)

        grad = grad_flat + grad_ent

        # ----- Adam update (bias‑corrected) -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it_A + 1))
        v_hat = v / (1.0 - beta2 ** (it_A + 1))

        # simple 1/√t learning‑rate decay
        lr = 0.7 / math.sqrt(it_A + 1)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape tiny plateaus
        if np.random.rand() < 0.001:
            a_new += np.random.randn(n) * 1e-4

        # projection onto the probability simplex
        a_new = _project_to_simplex(a_new)

        # prune extremely tiny entries every few thousand iters
        if it_A % 5000 == 0 and it_A > 0:
            thresh = np.percentile(a_new, 0.5)   # drop the lowest half‑percent
            a_new[a_new < thresh] = 0.0
            a_new = _project_to_simplex(a_new)

        a = a_new
        it_A += 1

    # --------------------------------------------------------------
    # Stage B – stochastic local refinement -------------------------
    print(f"[stage‑B] entering local refinement, best O so far = {best_score:.6f}")

    max_restarts = 0
    subgrad_interval = 5          # sub‑gradient step every 5 iterations
    temperature0 = 0.05
    max_iter_local = 200_000
    no_improve_limit = 4000

    while time.time() < deadline:
        # ---- initialise a restart ------------------------------------------------
        if np.random.rand() < 0.65:
            a = best_seq.copy()
            a += np.random.randn(n) * 0.0010
            a = np.clip(a, 0.0, None)
            a = _project_to_simplex(a)
        else:
            a = np.random.rand(n)
            a = _project_to_simplex(a)

        conv = _conv_fft(a, n, L)
        cur_obj = 2.0 * n * conv.max()
        it_loc = 0
        no_improve = 0
        T = temperature0

        while time.time() < deadline and it_loc < max_iter_local:
            # ------------------------------------------------------------------
            # 1️⃣ random mass‑transfer move
            i, j = np.random.randint(n, size=2)
            if i != j and a[i] > 1e-14:
                delta = a[i] * np.random.rand() * 0.18          # up to 18 % of source
                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta
                a_new = _project_to_simplex(a_new)

                conv_new = _conv_fft(a_new, n, L)
                new_obj = 2.0 * n * conv_new.max()

                # Metropolis acceptance
                if new_obj < cur_obj or np.random.rand() < math.exp(
                        (cur_obj - new_obj) / max(T, 1e-14)):
                    a, conv, cur_obj = a_new, conv_new, new_obj
                    no_improve = 0
                    if cur_obj < best_score:
                        best_score = cur_obj
                        best_seq = a.copy()
                        now = time.time() - start_time
                        print(f"[{now:.1f}s] new global best O={best_score:.6f}"
                              f" (restart={max_restarts}, iter={it_loc})")
                else:
                    no_improve += 1
            else:
                no_improve += 1

            # ------------------------------------------------------------------
            # 2️⃣ sub‑gradient step on the current max‑convolution entry
            if it_loc % subgrad_interval == 0:
                g = _subgradient_max_conv(a, conv)
                eta = 0.07 * (1.0 - it_loc / max_iter_local)   # linearly decaying
                a = a - eta * g
                a = _project_to_simplex(a)
                conv = _conv_fft(a, n, L)
                cur_obj = 2.0 * n * conv.max()
                if cur_obj < best_score:
                    best_score = cur_obj
                    best_seq = a.copy()
                    now = time.time() - start_time
                    print(f"[{now:.1f}s] sub‑grad improved O={best_score:.6f}")

            # ------------------------------------------------------------------
            # 3️⃣ occasional smoothing (tiny moving‑average blend)
            if it_loc % 4000 == 0 and it_loc > 0:
                kernel = np.array([0.125, 0.25, 0.25, 0.25, 0.125])
                a_smooth = np.convolve(a, kernel, mode="same")
                a = _project_to_simplex(0.85 * a + 0.15 * a_smooth)

            # ------------------------------------------------------------------
            # temperature decay
            T *= 0.9998
            it_loc += 1

            # restart if we have been stuck for a while
            if no_improve >= no_improve_limit:
                break

        max_restarts += 1
        if time.time() >= deadline:
            break

    # --------------------------------------------------------------
    # Final sanity check with the provided evaluator

        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:
        print("[final] evaluate_sequence raised an exception:", e)

    # Return a plain Python list
    return best_seq.tolist()
