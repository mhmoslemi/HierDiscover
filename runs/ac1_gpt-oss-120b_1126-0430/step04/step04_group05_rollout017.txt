    # a * (grad_a - np.dot(a, grad_a))
    dot = np.dot(a, grad_a)
    grad_z = a * (grad_a - dot)
    return obj, conv, a, grad_z

"""
Propose a non‑negative step‑function (probability vector) that minimizes

    O(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm works in three time‑controlled phases:

1️⃣  *Smooth Adam* – optimise a differentiable soft‑max surrogate of the
    max‑operator.  The variables are the logits ``z`` of a soft‑max
    parametrisation ``a = softmax(z)``; this automatically satisfies the
    simplex constraints.  The smoothing parameter α is increased
    exponentially from 5 to 2·10⁴ while Adam adapts the step‑size.

2️⃣  *Exact Adam* – continue the optimisation on the true objective.
    The sub‑gradient of ``max(conv)`` is obtained analytically and the
    same Adam machinery (now on the logits) is used with a smaller
    learning‑rate.

3️⃣  *Guided local refinement* – switch back to the probability vector
    itself.  At each iteration the sub‑gradient tells which indices
    contribute most to the current maximal convolution entry.  Mass is
    transferred from the “worst’’ index to the “best’’ one; moves are
    accepted if they improve the exact objective (or very rarely by a
    tiny Metropolis probability).  Stagnation triggers a small jittered
    restart around the current best.

All phases stop automatically after the allotted wall‑time (≤1000 s)
and the best sequence found is returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution a*a using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(z)
    e = np.exp(z - m)
    return e / e.sum()


# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the maximal convolution entry.
    For each index i,  g_i = 2·∑_{k∈A} a_{k-i}   (A = set of maximisers).
    """
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
def _smooth_obj_and_grad(z: np.ndarray, alpha: float, n: int, L: int):
    """
    Compute the smooth surrogate
        fα(a) = (2·n/α)·log∑_k exp(α·conv_k)
    together with its gradient w.r.t. the logits ``z``.
    Returns (obj_soft, conv, a, grad_z).
    """
    a = _softmax(z)

    # full convolution a*a
    conv = _conv_fft(a, L)                     # length 2*n-1

    # ----- smooth objective -----
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)   # stabilise
    sum_exp = exp_scaled.sum()
    log_sum = max_scaled + np.log(sum_exp)
    obj_soft = (2.0 * n / alpha) * log_sum

    # ----- gradient w.r.t. a (soft‑max surrogate) -----
    w = exp_scaled / sum_exp                    # soft‑max weights over conv‑entries

    # correlation w ⋆ a  (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    grad_a = 2.0 * corr_rev[n - 1:n - 1 + n]    # slice to length n

    # ----- chain rule to logits -----
    dot = np.dot(a, grad_a)
    grad_z = a * (grad_a - dot)

    return obj_soft, conv, a, grad_z

    # --------------------------------------------------------------
    # ------------------------------------------------------------------
    # Initialise a probability vector (sum = 1)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.array(height_sequence_1, dtype=float)
        a0 = np.maximum(a0, 0.0)
        if a0.sum() == 0.0:
            a0 = np.full_like(a0, 1.0)
        a0 = a0 / a0.sum()
        n = a0.size
        print(f"[init] using supplied sequence, n={n}")
    else:
        n = 4600                     # a reasonable default size
        a0 = np.full(n, 1.0 / n)     # uniform start
        print(f"[init] using uniform start, n={n}")

    # FFT length – next power of two ≥ 3·n
    L = 1 << ((3 * n - 2).bit_length())

    # ------------------------------------------------------------------
    # Helper: exact objective (no smoothing)
    def _exact_obj(conv):
        return 2.0 * n * conv.max()

    # ------------------------------------------------------------------
    # Phase 1 – Adam on the smooth surrogate
    # ------------------------------------------------------------------
    # logits for the soft‑max parametrisation
    eps_log = 1e-16
    z = np.log(a0 + eps_log)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    # smoothing schedule
    alpha_start, alpha_end = 5.0, 20000.0
    max_iter_phase1 = 300_000                # generous upper bound
    phase1_deadline = start_time + budget_s * 0.45

    # tracking best found sequence
    best_seq = a0.copy()
    best_score = _exact_obj(_conv_fft(a0, L))
    print(f"[phase‑1] start O={best_score:.6f}")

    # Adam bias‑correction factors (updated multiplicatively)
    bc1, bc2 = 1.0, 1.0
    it = 0
    while it < max_iter_phase1 and time.time() < phase1_deadline:
        # exponential increase of α, guaranteeing a very sharp surrogate at the end
        frac = it / max_iter_phase1
        alpha = alpha_start * (alpha_end / alpha_start) ** frac

        obj_soft, conv, a, grad_z = _smooth_obj_and_grad(z, alpha, n, L)

        # exact objective for bookkeeping
        cur_obj = _exact_obj(conv)
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[phase‑1] iter {it} new best O={best_score:.6f}")

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)

        bc1 *= beta1
        bc2 *= beta2
        m_hat = m / (1.0 - bc1)
        v_hat = v / (1.0 - bc2)

        # learning‑rate decays like 1/√t
        lr = 0.12 / math.sqrt(it + 1)

        z -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional tiny jitter to escape flat plateaus
        if np.random.rand() < 0.001:
            z += np.random.randn(n) * 1e-4

        # keep logits in a numeric range to avoid overflow/underflow
        if it % 5000 == 0 and it > 0:
            z = np.clip(z, -50.0, 50.0)

        it += 1

    # ------------------------------------------------------------------
    # Phase 2 – Adam on the exact (non‑smooth) objective
    # ------------------------------------------------------------------
    print(f"[phase‑2] start exact Adam, best O={best_score:.6f}")
    # re‑initialise Adam moments for the new objective
    m.fill(0.0)
    v.fill(0.0)
    bc1, bc2 = 1.0, 1.0

    phase2_deadline = start_time + budget_s * 0.75
    it2 = 0
    max_iter_phase2 = 250_000
    while it2 < max_iter_phase2 and time.time() < phase2_deadline:
        a = _softmax(z)
        conv = _conv_fft(a, L)
        cur_obj = _exact_obj(conv)
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[phase‑2] iter {it2} new best O={best_score:.6f}")

        # sub‑gradient of max(conv)
        g = _subgradient_max_conv(a, conv)               # shape (n,)

        # chain‑rule to logits
        dot = np.dot(a, g)
        grad_z = a * (g - dot)

        # Adam step
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)

        bc1 *= beta1
        bc2 *= beta2
        m_hat = m / (1.0 - bc1)
        v_hat = v / (1.0 - bc2)

        # smaller learning‑rate for the non‑smooth phase
        lr = 0.04 / math.sqrt(it2 + 1)

        z -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # keep logits bounded
        if it2 % 4000 == 0 and it2 > 0:
            z = np.clip(z, -50.0, 50.0)

        it2 += 1

    # ------------------------------------------------------------------
    # Phase 3 – deterministic, gradient‑guided local refinement
    # ------------------------------------------------------------------
    print(f"[phase‑3] stochastic local refinement, best O={best_score:.6f}")
    a = best_seq.copy()                               # work directly with probabilities
    conv = _conv_fft(a, L)
    cur_obj = _exact_obj(conv)

    best_seq_local = a.copy()
    best_score_local = cur_obj

    it_loc = 0
    no_improve = 0
    while time.time() < deadline and it_loc < 400_000:
        # --- mass‑transfer move guided by the sub‑gradient ---
        g = _subgradient_max_conv(a, conv)
        i_src = int(np.argmax(g))                     # index that most contributes to the peak
        i_tgt = int(np.argmin(g))                     # index that contributes least

        if i_src != i_tgt and a[i_src] > 1e-14:
            # move a modest fraction of the source mass
            delta = a[i_src] * np.random.rand() * 0.09
            a_new = a.copy()
            a_new[i_src] -= delta
            a_new[i_tgt] += delta

            # maintain exact simplex (numerical drift may appear)
            a_new = np.clip(a_new, 0.0, None)
            a_new /= a_new.sum()

            conv_new = _conv_fft(a_new, L)
            new_obj = _exact_obj(conv_new)

            # Metropolis‑style acceptance
            accept = False
            if new_obj < cur_obj:
                accept = True
            else:
                # occasional uphill move (tiny temperature)
                T = max(0.01, 0.01 * (1.0 - it_loc / 400_000))
                if np.random.rand() < math.exp((cur_obj - new_obj) / T):
                    accept = True

            if accept:
                a = a_new
                conv = conv_new
                cur_obj = new_obj
                no_improve = 0
                if cur_obj < best_score_local:
                    best_score_local = cur_obj
                    best_seq_local = a.copy()
                    print(f"[local] iter {it_loc} new best O={best_score_local:.6f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        # --- occasional jittered restart if stuck for a while ---
        if no_improve >= 8_000:
            a = best_seq_local.copy()
            a += np.random.randn(n) * 0.0015
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            conv = _conv_fft(a, L)
            cur_obj = _exact_obj(conv)
            no_improve = 0
            print(f"[restart] re‑jittered after {it_loc} iterations, O={cur_obj:.6f}")

        # tiny regular jitter to keep the dynamics alive
        if it_loc % 7_000 == 0 and it_loc > 0:
            a += np.random.randn(n) * 5e-4
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            conv = _conv_fft(a, L)
            cur_obj = _exact_obj(conv)

        it_loc += 1

    # ------------------------------------------------------------------
    # Final sanity check with the provided evaluator
    try:
        final_score = evaluate_sequence(best_seq_local.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence failed:", exc)

    return best_seq_local.tolist()
