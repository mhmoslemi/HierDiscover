Simplex projection algorithm:

def simplex_projection(v):
    # Project v onto probability simplex (sum=1, v>=0)
    # from https://github.com/inducer/pyproj or others
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def project_simplex(v):
    # Algorithm from Duchi et al. 2008.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

#!/usr/bin/env python
"""
Projected‐gradient / Adam optimiser for the self‑convolution max problem.

The objective after normalising ∑a_i = 1 is
    J(a) = 2·n·max_k (a*a)[k] .
We minimise J by working in the soft‑max (log‑it) parametrisation
    a = softmax(z)
which automatically enforces non‑negativity and the sum‑to‑one
constraint.  The optimisation proceeds in two phases:

* Phase 1 (≈ 60 % of the time budget) – minimise a least‑squares
  distance to the perfectly flat convolution (the “uniform” target).
  This quickly drives the distribution towards one whose self‑convolution
  is nearly constant.

* Phase 2 – directly reduce the true maximum using the exact sub‑gradient
  of the max‑operator (computed via a cheap FFT‑correlation).

Both phases use Adam with a decaying step‑size and a simple back‑track
line‑search that guarantees a non‑increasing cheap score
    cheap = 2·n·max(conv).

The best sequence seen so far is finally evaluated with the provided
``evaluate_sequence`` function and returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level helpers (no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector (sum=1)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it vector that yields ``a`` after ``softmax``."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """
    Smallest power‑of‑two ≥ 3·n‑2.
    This length is sufficient for the linear convolution (size 2·n‑1)
    and the subsequent correlation (size up to 3·n‑2).
    """
    return 1 << ((3 * n - 2).bit_length())


def _conv_full(a: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution a*a (full, length 2·n‑1) using FFT of length L."""
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * a.shape[0] - 1]
    return conv


def _corr_with_signal(a: np.ndarray, s: np.ndarray, L: int) -> np.ndarray:
    """
    Correlation of a (length n) with a signal s (length 2·n‑1).

    Returns the first n entries of
        corr[i] = Σ_j a[j]·s[i+j] .
    """
    a_fft = np.fft.rfft(a, L)
    s_fft = np.fft.rfft(s, L)
    corr = np.fft.irfft(np.conj(a_fft) * s_fft, L)[: a.shape[0]]
    return corr


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Gradient utilities
# ----------------------------------------------------------------------


def _grad_l2(a: np.ndarray, conv: np.ndarray, target: np.ndarray, L: int):
    """
    Gradient of ½‖conv−target‖² w.r.t. a.
    Returns (grad, conv) where conv is the already‑computed convolution.
    """
    diff = conv - target
    grad = 4.0 * _corr_with_signal(a, diff, L)
    return grad, conv


def _grad_max(a: np.ndarray, conv: np.ndarray, L: int):
    """
    Sub‑gradient of max_k conv_k.
    Returns (grad, max_val) where grad_i = 2·Σ_{k∈M} a_{k−i}
    and M is the set of indices where conv attains its maximum.
    """
    max_val = conv.max()
    # indicator of maximal entries (exact max, ties are kept)
    w = np.zeros_like(conv)
    w[conv >= max_val - 1e-14] = 1.0
    grad = 2.0 * _corr_with_signal(a, w, L)
    return grad, max_val

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) beats the
    target bound.  The routine respects the given time budget (~1000 s) and
    uses a two‑phase Adam optimiser on the probability simplex.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # small safety margin

    # --------------------------------------------------------------
    # 0. Determine problem size and an initial distribution.
    # --------------------------------------------------------------
    try:
        # If a previous construction is available, start from it.
        init = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        n = init.shape[0]
        a = np.maximum(init, 0.0)
    except Exception:
        # Fallback: a random Dirichlet vector (α=0.8 gives a mildly
        # sparse start) with the desired length.
        n = 20000
        a = np.random.gamma(0.8, 1.0, size=n)
    # Normalise to a probability vector.
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a_sum

    # Initialise log‑it parameters.
    z = _logits_from_distribution(a)
    # Small jitter to avoid exact replicas of previous optima.
    z += 0.01 * np.random.randn(n)

    # Adam state.
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_alpha = 0.25          # base learning‑rate for logits

    # FFT length (constant throughout the run).
    L = _fft_len_for_corr(n)

    # Target distribution for the L2 phase: flat convolution.
    target_conv = np.full(2 * n - 1, 1.0 / (2 * n - 1), dtype=np.float64)

    # --------------------------------------------------------------
    # 1. Book‑keeping.
    # --------------------------------------------------------------
    a = _softmax(z)                    # current probability vector
    conv = _conv_full(a, L)
    cheap_score = 2.0 * n * conv.max()
    best_a = a.copy()
    best_score = cheap_score

    it = 0
    last_improve_it = 0
    phase = 0                          # 0 = L2‑flattening, 1 = max‑reduction
    phase_switch_time = start + 0.6 * (deadline - start)

    # --------------------------------------------------------------
    # 2. Optimisation loop.
    # --------------------------------------------------------------
    while time.time() < deadline and it < 1_500_000:
        it += 1

        # Switch phase once the allocated portion of the budget is used.
        if phase == 0 and time.time() >= phase_switch_time:
            phase = 1
            # Reset Adam moments for the new phase (helps stability).
            m.fill(0.0)
            v.fill(0.0)
            it = 0
            last_improve_it = 0
            print(f"[{time.time() - start:6.1f}s] Switching to max‑phase (iteration {it})")

        # ----------------------------------------------------------
        # 2.1  Compute convolution and the appropriate gradient.
        # ----------------------------------------------------------
        conv = _conv_full(a, L)

        if phase == 0:
            # L2‑flattening phase.
            grad_a, _ = _grad_l2(a, conv, target_conv, L)
        else:
            # Max‑reduction phase.
            grad_a, max_val = _grad_max(a, conv, L)

        # Transform gradient to the log‑it (softmax) space.
        mu = np.dot(grad_a, a)
        grad_z = a * (grad_a - mu)

        # Adam moment updates (bias‑corrected).
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # Step‑size schedule (inverse‑sqrt decay).
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        direction = -m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----------------------------------------------------------
        # 2.2  Back‑tracking line search (up to 5 halvings).
        # ----------------------------------------------------------
        accepted = False
        trial_alpha = alpha_t
        for _ in range(5):
            z_new = z + trial_alpha * direction
            # Clip logits to avoid overflow in the exponential.
            z_new = np.clip(z_new, -50.0, 50.0)
            a_new = _softmax(z_new)
            conv_new = _conv_full(a_new, L)
            new_score = 2.0 * n * conv_new.max()
            if new_score <= cheap_score:
                # Accept the step.
                z = z_new
                a = a_new
                cheap_score = new_score
                accepted = True
                break
            trial_alpha *= 0.5  # halve and try again

        if not accepted:
            # No improvement – still take a modest step to keep moving.
            z = z + alpha_t * direction
            z = np.clip(z, -50.0, 50.0)
            a = _softmax(z)
            conv = _conv_full(a, L)
            cheap_score = 2.0 * n * conv.max()

        # ----------------------------------------------------------
        # 2.3  Record the best candidate seen so far.
        # ----------------------------------------------------------
        if cheap_score < best_score:
            best_score = cheap_score
            best_a = a.copy()
            last_improve_it = it
            if it % 2000 == 0:
                print(f"[Iter {it:07d}] NEW BEST cheap = {best_score:.6f}")

        # ----------------------------------------------------------
        # 2.4  Occasional random perturbations to escape plateaux.
        # ----------------------------------------------------------
        if it % 6000 == 0 and np.random.rand() < 0.12:
            z += 0.02 * np.random.randn(n)          # jitter in log‑it space
            m *= 0.7
            v *= 0.7

        # ----------------------------------------------------------
        # 2.5  Early termination if stagnation is observed (after
        #       the max‑phase is active).
        # ----------------------------------------------------------
        if phase == 1 and it - last_improve_it > 250_000:
            print(f"[Iter {it:07d}] Stagnation detected – terminating early.")
            break

    # --------------------------------------------------------------
    # 3. Final exact evaluation with the provided evaluate_sequence().
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:   # defensive – should never happen
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # --------------------------------------------------------------
    # 4. Return the best sequence.
    # --------------------------------------------------------------
    return final_seq
