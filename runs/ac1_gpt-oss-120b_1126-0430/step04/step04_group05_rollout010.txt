propose_candidate
-----------------

A three‑stage optimizer for the step‑function bound

    O(a) = 2·n·max( a * a ) / (sum(a)²)

The algorithm works with a probability simplex representation
(sum(a)=1, a_i ≥ 0) because the objective is scale‑invariant.

Stage 0 – cheap random restarts:
    Sample a few Dirichlet‑type vectors (including sparse,
    medium‑dense and uniform candidates) and keep the best one.

Stage 1 – “flattening” gradient descent:
    Minimise the L2‑error of the self‑convolution to a flat target.
    The target is the constant 1/(2·n−1), which is the average value
    of the convolution (since ∑_k (a*a)_k = 1).  The gradient of
    ‖conv−avg‖₂² can be written as a convolution and is obtained
    efficiently by FFT.  After each step we project back onto the
    simplex.

Stage 2 – stochastic refinement:
    Simulated‑annealing style mass‑transfer moves (move a random
    fraction of the probability mass from one index to another)
    accept improvements or Metropolis steps, together with occasional
    exact sub‑gradient steps for the true max‑operator.  Restarts are
    performed when progress stalls.

All phases respect a total wall‑time of 1000 s and keep the best
candidate found.  At the end the candidate is checked with the
provided ``evaluate_sequence`` function.

The code is written for clarity, uses only NumPy (no external
optimisers) and runs on a single CPU core.

import math

# ----------------------------------------------------------------------
#  Helper: Euclidean projection onto the probability simplex.
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex {x≥0, sum(x)=1}."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        # all non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
#  Full self‑convolution via FFT (length 2·n‑1)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return the full convolution a*a using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
#  Objective on the simplex (sum(a)=1):  O = 2·n·max(conv)
def _objective_simplex(a: np.ndarray, n: int, L: int) -> float:
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()

        # occasionally try the pure uniform vector (may be useful after a restart)
        if np.random.rand() < 0.03:
            a_unif = np.full(n, 1.0 / n)
            cur = _objective_simplex(a_unif, n, L)
            if cur < best_score:
                best_score = cur
                best_seq = a_unif.copy()
                print(f"[rnd] uniform improved O={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – flattening gradient descent (≈45 % of total budget)
    a = best_seq.copy()
    phase1_deadline = start_time + budget_s * 0.6
    it = 0
    eta0 = 0.6   # initial learning rate
    max_iter_phase1 = 500_000

    while it < max_iter_phase1 and time.time() < phase1_deadline:
        # current convolution and objective
        conv = _conv_fft(a, L)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[flat] iter {it} new best O={best_score:.6f}")

        # gradient of ‖conv−avg‖₂²  (see discussion in the module docstring)
        err = conv - avg_target                     # length 2·n‑1
        a_rev = a[::-1]
        fft_err = np.fft.rfft(err, n=L)
        fft_a_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(fft_err * fft_a_rev, n=L)
        grad = 4.0 * corr[n - 1 : n - 1 + n]        # same length as a

        # step‑size schedule (1/√t)
        eta = eta0 / math.sqrt(it + 1.0)
        a_new = a - eta * grad
        a_new = _project_to_simplex(a_new)

        # occasional pruning of tiny entries (helps numerical stability)
        if it % 5000 == 0 and it > 0:
            thresh = np.percentile(a_new, 0.5)      # keep top‑99.5 %
            a_new[a_new < thresh] = 0.0
            a_new = _project_to_simplex(a_new)

        a = a_new
        it += 1

    # ------------------------------------------------------------------
    # Stage 2 – stochastic local refinement (remaining budget)
    a = best_seq.copy()
    conv = _conv_fft(a, L)
    cur_score = 2.0 * n * conv.max()
    best_score = cur_score
    best_seq = a.copy()

    T0 = 0.08                # initial temperature for Metropolis moves
    T = T0
    it_loc = 0
    max_iter_loc = 300_000
    no_improve = 0
    no_improve_limit = 8000

    while time.time() < deadline and it_loc < max_iter_loc:
        # ----- mass‑transfer move -----
        i, j = np.random.randint(n, size=2)
        if i != j and a[i] > 1e-15:
            # transfer up to 12 % of the source mass
            delta = a[i] * np.random.rand() * 0.12
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            a_new = _project_to_simplex(a_new)

            conv_new = _conv_fft(a_new, L)
            new_score = 2.0 * n * conv_new.max()

            # Metropolis acceptance
            if new_score < cur_score or np.random.rand() < math.exp((cur_score - new_score) / max(T, 1e-12)):
                a = a_new
                conv = conv_new
                cur_score = new_score
                no_improve = 0
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(f"[SA] iter {it_loc} new best O={best_score:.6f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        # ----- occasional exact sub‑gradient step (max‑operator) -----
        if it_loc % 9 == 0:
            # sub‑gradient of the maximal convolution entry
            max_val = conv.max()
            ks = np.flatnonzero(conv == max_val)
            g = np.zeros_like(a)
            for k in ks:
                i_start = max(0, k - (n - 1))
                i_end = min(n - 1, k)
                idx = np.arange(i_start, i_end + 1, dtype=int)
                jdx = k - idx
                g[idx] += 2.0 * a[jdx]
            if ks.size:
                g /= ks.size
            # decreasing step‑size
            eta_loc = 0.07 * (1.0 - it_loc / max_iter_loc)
            a = a - eta_loc * g
            a = _project_to_simplex(a)
            conv = _conv_fft(a, L)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[sub‑grad] iter {it_loc} improved O={best_score:.6f}")

        # temperature decay
        T *= 0.9996

        # restart if stuck
        if no_improve >= no_improve_limit:
            a = best_seq.copy()
            a += np.random.randn(n) * 0.0012
            a = np.clip(a, 0.0, None)
            a = _project_to_simplex(a)
            conv = _conv_fft(a, L)
            cur_score = 2.0 * n * conv.max()
            no_improve = 0

        it_loc += 1

    # ------------------------------------------------------------------
    # Final sanity check with the provided evaluator
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence failed:", exc)

    return best_seq.tolist()
