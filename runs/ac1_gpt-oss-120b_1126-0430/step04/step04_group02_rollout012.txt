"""
Improved search for a non‑negative step‑function that minimises

    O(a) = 2·n·max(conv(a,a))    (the sum of a is forced to 1 via soft‑max)

The method consists of several independent Adam optimisations on the
log‑it (soft‑max) parametrisation of the simplex.  For each support size we
try a number of diverse initialisations (uniform, edge‑heavy,
alternating, sparse, random).  The Adam optimiser is equipped with an
annealing “soft‑max” that smoothly approximates the true max; the annealing
parameter β grows exponentially while the learning rate decays exponentially.
During optimisation we keep the best vector seen so far, occasionally jitter
the logits to escape plateaus and occasionally restart from the current best.

After the gradient phase we compress entries that are numerically zero, then
run a short simulated‑annealing style local search that repeatedly moves a
small random fraction of mass between two positions and accepts the move if
the objective improves (or with a tiny temperature‑dependent chance).

All runs share the global 1000 s wall‑clock budget; the longest run (largest
n) receives the biggest share of the time.  The final best sequence is
checked with the provided ``evaluate_sequence`` routine before being returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a power‑of‑2 FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation of w (length 2n‑1) with a (length n):
    (w ⋆ a)[i] = Σ_t w[i+t]·a[t]   for i=0,…,n‑1.
    Implemented via FFT.
    """
    n = a.shape[0]
    m = w.shape[0]               # = 2n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


# ----------------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces a probability vector)."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) (expects Σa = 1)."""
    n = a.size
    conv = _conv_self_fft(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _make_initial_vector(n: int, strategy: str, rng: np.random.Generator) -> np.ndarray:
    """
    Produce an initial probability vector (sum = 1) of length n
    according to the requested strategy.
    """
    if strategy == "uniform":
        a = np.ones(n, dtype=float)
    elif strategy == "edge":
        idx = np.arange(n, dtype=float)
        # large at both ends, small in the middle: 1/( (i+1)*(n-i) )
        a = 1.0 / ((idx + 1) * (n - idx) + 1e-6)
    elif strategy == "alternating":
        a = ((np.arange(n) % 2) + 1).astype(float)   # pattern 2,1,2,1,...
    elif strategy == "sparse":
        # put equal mass on a small random subset (≈ √n elements)
        m = max(2, int(np.sqrt(n) * 0.8))
        a = np.zeros(n, dtype=float)
        chosen = rng.choice(n, size=m, replace=False)
        a[chosen] = 1.0
    else:   # "random"
        a = rng.random(n)
    a = np.clip(a, 0.0, None)
    if a.sum() == 0.0:
        a = np.ones_like(a)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
def _run_opt(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None,
    seed: int,
    strategy: str | None = None,
) -> tuple[list[float], float]:
    """
    Optimise a length‑n probability vector for at most ``time_limit`` seconds.
    Returns the best sequence (as a plain list) and its exact objective.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # ---- initialise ----------------------------------------------------
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = _make_initial_vector(n, "random", rng)
    else:
        a = _make_initial_vector(n, strategy or "random", rng)

    a = np.clip(a, 0.0, None)
    a = a / a.sum()                     # ensure sum = 1

    # work in log‑it space (soft‑max guarantees feasibility)
    x = np.log(a + 1e-16)

    # ---- Adam state ----------------------------------------------------
    m = np.zeros_like(x)                # first moment
    v = np.zeros_like(x)                # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # ---- schedules -----------------------------------------------------
    beta_start = 0.5
    beta_end   = 30000.0
    lr_start   = 0.25
    lr_end     = 0.001

    best_a = a.copy()
    best_obj = _objective(best_a)

    it = 0
    max_iters = 2_000_000   # just a safety cap

    while time.time() - start < time_limit and it < max_iters:
        it += 1

        # probability vector from logits
        a = _softmax(x)

        # exact objective (sum(a)=1 already)
        conv = _conv_self_fft(a)
        M = conv.max()
        obj = 2.0 * n * M

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        # ---- schedule parameters (based on iteration fraction) ----------
        t_frac = it / max_iters
        beta = beta_start * np.exp(np.log(beta_end / beta_start) * t_frac)
        lr   = lr_start   * np.exp(np.log(lr_end   / lr_start)   * t_frac)

        # ---- soft‑max weights for the (smoothed) max --------------------
        if beta > 25000.0:
            # very sharp → true arg‑max
            kmax = int(np.argmax(conv))
            w = np.zeros_like(conv)
            w[kmax] = 1.0
        else:
            shifted = conv - M                # ≤ 0
            w = np.exp(beta * shifted)
            w = w / w.sum()                  # length 2n‑1, sums to 1

        # ---- gradient of the smoothed objective w.r.t. a ----------------
        g_a = 2.0 * _correlation_fft(w, a)   # shape (n,)

        # natural gradient on the simplex (log‑it space)
        dot = np.dot(a, g_a)                 # scalar ⟨a, g_a⟩
        g_x = a * (g_a - dot)                # shape (n,)

        # ---- Adam update ------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---- occasional jitter to break stagnation --------------------
        if it % 4000 == 0:
            jitter = rng.normal(scale=0.015, size=n)
            x += jitter

        # ---- restart from current best every now and then ---------------
        if it % 60_000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.005, size=n)
            m.fill(0.0)
            v.fill(0.0)
            print(f"[n={n:5d}] restart at iter {it}")

    # ----- compress negligible entries -----------------------------------
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        n_new = best_a.size
        best_obj = _objective(best_a)
        print(f"[compress] length {n} → {n_new}, O={best_obj:.6f}")

    # ----- brief simulated‑annealing local search ------------------------
    remaining = time_limit - (time.time() - start)
    if remaining > 0.6:
        local_start = time.time()
        a_loc = best_a.copy()
        obj_loc = best_obj
        n_loc = a_loc.size
        temperature = 0.001   # tiny temperature, acceptance only for tiny ups

        while time.time() - local_start < remaining * 0.35:
            i, j = rng.choice(n_loc, size=2, replace=False)
            # transfer a random small fraction from i to j
            delta = a_loc[i] * rng.random() * 0.03   # up to 3 % of a_i
            if delta <= 0.0:
                continue
            a_new = a_loc.copy()
            a_new[i] -= delta
            a_new[j] += delta
            conv_new = _conv_self_fft(a_new)
            obj_new = 2.0 * n_loc * conv_new.max()
            if obj_new < obj_loc or rng.random() < np.exp(-(obj_new - obj_loc) / temperature):
                a_loc, obj_loc = a_new, obj_new
                if obj_new < best_obj:
                    best_a, best_obj = a_new, obj_new
        print(f"[local] final O={best_obj:.6f}")

    return best_a.tolist(), best_obj


# ----------------------------------------------------------------------
def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **_ignore,
) -> list[float]:
    """
    Main entry point.  Runs several Adam‑optimisation attempts with a
    portfolio of support sizes and initialisations, applies a short
    local search and returns the best step‑function found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Candidate lengths – we explore a small geometric progression.
    lengths = [500, 1000, 2000, 4000, 8000]

    # If a previous construction exists, try its length first.
    if 'height_sequence_1' in globals():
        seq0 = np.asarray(height_sequence_1, dtype=float).copy()
        seq0 = np.clip(seq0, 0.0, None)
        if seq0.sum() < 0.01:
            seq0 = np.ones_like(seq0)
        seq0 = seq0 / seq0.sum()
        n0 = seq0.size
        if n0 not in lengths:
            lengths.insert(0, n0)

    best_seq = None
    best_obj = float('inf')

    # --------------------------------------------------------------
    # Give the longest run (≈60 % of remaining time) to the largest n,
    # the rest is split among the other candidates.
    for idx, n in enumerate(lengths):
        remaining = deadline - time.time()
        if remaining <= 1.0:
            break

        # allocate time budget for this length
        if idx == 0:          # longest length gets the biggest share
            run_time = remaining * 0.60
        else:
            # share the rest equally (minus a tiny safety margin)
            run_time = (remaining - 0.5) / (len(lengths) - idx)

        # -----------------------------------------------------------------
        # Try several initialisation strategies for robustness.
        strategies = ["uniform", "edge", "alternating", "sparse", None]
        for strat_idx, strat in enumerate(strategies):
            # If we run out of time, bail out early.
            if time.time() >= deadline - 0.3:
                break

            # Give each strategy a fraction of the allocated time.
            sub_time = run_time / len(strategies)

            init_a = None
            if idx == 0 and strat is None and 'height_sequence_1' in globals():
                init_a = seq0

            seq, val = _run_opt(
                n=n,
                time_limit=sub_time,
                init_a=init_a,
                seed=seed + idx * 31 + (strat_idx or 0) * 7,
                strategy=strat,
            )
            if val < best_obj:
                best_obj = val
                best_seq = seq
                print(f"[global] new overall best O={best_obj:.6f} (n={n}, strat={strat})")

    # ------------------------------------------------------------------
    # Final safety check with the official evaluator.
    try:
        final_score = evaluate_sequence(best_seq)
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_seq
