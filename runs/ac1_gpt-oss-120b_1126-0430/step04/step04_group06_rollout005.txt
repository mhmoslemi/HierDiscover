#!/usr/bin/env python
"""
Propose a non‑negative step‑function (probability vector) that minimises
    f(a) = 2·n·max(a∗a)   (with Σa = 1)

The algorithm is a mirror‑descent optimiser on the probability simplex.
It uses a smooth approximation of the max (log‑sum‑exp) and the full
gradient w.r.t. the logits (`z` with a = softmax(z)).  Adam with an
inverse‑sqrt schedule updates the logits.  A temperature parameter
θ gradually increases, focusing the gradient onto the true maximum.
The optimiser runs for the whole time budget (≈1000 s) and keeps the
best sequence seen.  If a global variable `height_sequence_1` is
available it is used as a warm start; otherwise a power‑law start is
employed.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max; returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it representation that yields `a` after soft‑max."""
    return np.log(a + eps)


def _fft_len_for_n(L: int) -> int:
    """Smallest power‑of‑two ≥ L (efficient size for real FFT)."""
    return 1 << ((L - 1).bit_length())


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full linear self‑convolution a*a using a real FFT of length `fft_len`.
    Returns an array of length 2·len(a)‑1.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.size - 1]
    return conv


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    total = new_a.sum()
    if total == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= total
    return new_a


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """Normalised power‑law a_i ∝ (i+1)^{‑p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Core optimisation routine
# ----------------------------------------------------------------------


def _run_opt_on_fixed_n(
    n: int,
    init_a: np.ndarray,
    time_deadline: float,
    *,
    theta_start: float = 0.5,
    theta_end: float = 15.0,
    base_alpha: float = 0.22,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps_adam: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Optimises a probability vector of length `n` using Adam on the logits.
    Returns the best vector found (as np.ndarray) and the best cheap score
    (2·n·max(conv)) observed during the run.
    """
    # Prepare FFT lengths (once)
    fft_len_conv = _fft_len_for_n(2 * n - 1)
    fft_len_grad = _fft_len_for_n(3 * n - 2)

    # Initialise logits and Adam moments
    a = np.maximum(init_a, 0.0)
    if a.sum() == 0.0:
        a = np.full_like(a, 1.0 / n)
    else:
        a = a / a.sum()
    z = _logits_from_distribution(a)

    m = np.zeros_like(z)
    v = np.zeros_like(z)

    best_a = a.copy()
    best_score = float("inf")
    it = 0
    last_improve_it = 0
    report_every = 3000

    # Temperature schedule (linear in real time)
    start_time = time.time()
    total_budget = max(1.0, time_deadline - start_time)
    theta_slope = (theta_end - theta_start) / total_budget

    while time.time() < time_deadline:
        it += 1

        # ---- probability vector -------------------------------------------------
        a = _softmax(z)

        # ---- self‑convolution ----------------------------------------------------
        conv = _conv_full_fft(a, fft_len_conv)          # length 2n‑1
        max_conv = float(conv.max())
        cheap_score = 2.0 * n * max_conv

        # ---- keep track of the best ------------------------------------------------
        if cheap_score < best_score:
            best_score = cheap_score
            best_a = a.copy()
            last_improve_it = it
            if it % report_every == 0:
                elapsed = time.time() - start_time
                print(
                    f"[Iter {it:07d}] new best cheap={cheap_score:.6f} "
                    f"(θ≈{theta:.2f}) elapsed={elapsed:.1f}s"
                )

        # ---- smooth max (log‑sum‑exp) ---------------------------------------------
        elapsed = time.time() - start_time
        theta = min(theta_end, theta_start + theta_slope * elapsed)

        # Soft‑max weights over all convolution entries
        # (subtract max to avoid overflow)
        exps = np.exp(theta * (conv - max_conv))
        w = exps / exps.sum()                     # shape (2n‑1,)

        # ---- gradient of smooth max w.r.t. a --------------------------------------
        # g_i = 2 * Σ_k w_k * a_{k‑i}
        a_rev = a[::-1]                           # reversed a
        Fa_rev = np.fft.rfft(a_rev, fft_len_grad)
        Fw = np.fft.rfft(w, fft_len_grad)
        conv_aw = np.fft.irfft(Fa_rev * Fw, fft_len_grad)
        # slice the central n entries: indices n‑1 … n‑1+n‑1
        start = n - 1
        g = 2.0 * conv_aw[start : start + n]      # shape (n,)

        # Gradient of the cheap objective (2·n·max) ≈ 2·n·g
        grad_a = (2.0 * n) * g

        # ---- mirror‑descent onto simplex (logits gradient) ------------------------
        mu = np.dot(grad_a, a)
        grad_z = a * (grad_a - mu)

        # ---- Adam update ---------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step_dir = - m_hat / (np.sqrt(v_hat) + eps_adam)

        # Inverse‑sqrt step‑size schedule (slightly decaying)
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)

        z = z + alpha_t * step_dir
        # keep logits numerically safe
        np.clip(z, -80.0, 80.0, out=z)

        # ---- occasional random perturbation (helps escape plateaus) -------------
        if it % 4000 == 0 and np.random.rand() < 0.2:
            z += 0.015 * np.random.randn(n)
            m *= 0.5
            v *= 0.5

        # ---- early stop after long stagnation ------------------------------------
        if it - last_improve_it > 250_000 and elapsed > total_budget * 0.4:
            # after a while without improvement we stop this run
            break

    return best_a, best_score


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Entry point required by the platform.
    Returns a list of non‑negative heights (float) that (hopefully) beats the
    target bound.  The routine respects the given time budget (~1000 s) and
    uses a gradient‑descent optimiser on the simplex with a smooth max
    approximation.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.5  # safety margin

    # ------------------------------------------------------------------
    # Decide on the working support size.
    # ------------------------------------------------------------------
    global_seq = globals().get("height_sequence_1", None)
    if isinstance(global_seq, (list, tuple, np.ndarray)):
        # Use the supplied sequence as a warm‑start (may be long)
        init = np.array(global_seq, dtype=np.float64)
        n0 = max(10000, init.shape[0])          # at least 10 k entries
        # If it's short we will interpolate later
        print(f"[Init] Using global sequence as start, length={init.shape[0]}")
    else:
        init = None
        n0 = 25000
        print("[Init] No global sequence – will start from power‑law")

    # Cap the length – very large vectors are too costly
    n_max = 30000
    n = min(n_max, n0)

    # ------------------------------------------------------------------
    # Prepare a few different initialisations (global, power‑law, random)
    # ------------------------------------------------------------------
    init_candidates = []

    if init is not None:
        # Interpolate or truncate the global one to the target length
        if init.shape[0] != n:
            a0 = _interpolate_distribution(init, n)
        else:
            a0 = init.copy()
        init_candidates.append(a0)

    # a power‑law start (p≈0.6 works well)
    init_candidates.append(_power_law_distribution(n, 0.60))

    # a random Dirichlet start
    rand_vec = np.random.rand(n)
    init_candidates.append(rand_vec / rand_vec.sum())

    # ------------------------------------------------------------------
    # Run optimisation on each start, respecting the global deadline
    # ------------------------------------------------------------------
    best_overall_a = None
    best_overall_score = float("inf")

    for idx, a_start in enumerate(init_candidates):
        if time.time() >= deadline:
            break

        time_left = deadline - time.time()
        print(
            f"[Run {idx + 1}/{len(init_candidates)}] "
            f"starting optimisation with n={n} (time left {time_left:.1f}s)"
        )

        a_opt, score_opt = _run_opt_on_fixed_n(
            n=n,
            init_a=a_start,
            time_deadline=deadline,
            theta_start=0.5,
            theta_end=15.0,
            base_alpha=0.22,
        )

        print(
            f"[Run {idx + 1}] finished with best cheap score = {score_opt:.6f}"
        )
        if score_opt < best_overall_score:
            best_overall_score = score_opt
            best_overall_a = a_opt.copy()

    # ------------------------------------------------------------------
    # Optional fine‑tuning on the best candidate (higher temperature)
    # ------------------------------------------------------------------
    if best_overall_a is not None and time.time() < deadline - 5.0:
        # Brief high‑precision run: higher θ, smaller step‑size
        print("[Fine‑tune] Running a short high‑θ refinement")
        a_fine, score_fine = _run_opt_on_fixed_n(
            n=n,
            init_a=best_overall_a,
            time_deadline=deadline,
            theta_start=8.0,
            theta_end=30.0,
            base_alpha=0.12,
        )
        if score_fine < best_overall_score:
            best_overall_score = score_fine
            best_overall_a = a_fine.copy()
        print(f"[Fine‑tune] refined cheap score = {score_fine:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation (the platform will also evaluate)
    # ------------------------------------------------------------------
    final_seq = best_overall_a.tolist() if best_overall_a is not None else [1.0]
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[Result] Final exact score = {final_score:.6f}")
    except Exception as exc:
        final_score = float("inf")
        print(f"[Result] Evaluation error: {exc}")

    # ------------------------------------------------------------------
    # Return the best sequence found.
    # ------------------------------------------------------------------
    return final_seq
