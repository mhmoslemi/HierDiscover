def _smooth_gradient(a, conv, alpha, L):
    # conv is full convolution length 2*n-1
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / np.sum(exp_scaled)
    # correlation w * a (like cross-correlation)
    # Let's compute using FFT: corr = w ⋆ a
    # cross-correlation (w * a)[i] = sum_j w_{i+j} a_j
    # The code from earlier uses reversed a and FFT multiplication.
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n = a.size
    grad = 2.0 * corr_rev[n-1:n-1+n]   # slice length n
    return grad
```

Propose a non‑negative step‑function that (empirically) beats the previous
best O≈1.5189.  The algorithm consists of four stages:

1️⃣  Parameter sweep over a few simple families (power‑law, exponential,
    truncated linear) to obtain a good initial shape.
2️⃣  Exponentiated‑gradient optimisation of a smooth log‑sum‑exp (soft‑max)
    surrogate of the max‑convolution.  Adam is replaced by a multiplicative
    update that stays on the probability simplex.
3️⃣  Multiplicative sub‑gradient descent on the exact objective (max

4️⃣  Stochastic mass‑transfer hill‑climbing (simulated‑annealing style) that
    refines the best solution found so far.

All phases respect the overall 1000 s wall‑time budget and keep track of the
best sequence seen.  The final candidate is verified with the provided

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full self‑convolution a∗a using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
def _objective_simplex(a: np.ndarray, n: int, L: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a already simplex‑normalised."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. a (valid for a on the simplex)."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, conv: np.ndarray, alpha: float, L: int) -> np.ndarray:
    """
    Gradient of the smoothed surrogate
        (2·n/α)·log Σ_k exp(α·conv_k)
    with respect to a.
    """
    # soft‑max weights on the convolution entries
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / exp_scaled.sum()

    # correlation w ⋆ a   (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n_len = a.size
    grad = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]
    return grad

def _generate_powerlaw(p: float, n: int) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)  (p≥0)."""
    idx = np.arange(1, n + 1, dtype=float)
    a = idx ** (-p)
    return a / a.sum()


def _generate_exponential(r: float, n: int) -> np.ndarray:
    """a_i ∝ r**i  (0<r<1)."""
    i = np.arange(n, dtype=float)
    a = r ** i
    return a / a.sum()


def _generate_linear(support: int, n: int) -> np.ndarray:
    """
    Linear decreasing shape that is positive on the first `support` entries:
        a_i = max(0, support - i)
    """
    i = np.arange(n, dtype=float)
    a = np.maximum(0.0, support - i)
    if a.sum() == 0:
        return np.full_like(a, 1.0 / n)
    return a / a.sum()
