"""
propose_candidate
-----------------

Search for a non‑negative step‑function (list of heights) that minimises

        O(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm works in three stages:

1️⃣  **Fast parametric sweep** – a few analytic families (inverse‑power,
    exponential, piecewise‑uniform) are sampled on the given length *n*.
    The best vector from this cheap sweep seeds the optimiser.

2️⃣  **Adam on a smooth surrogate** – we optimise the log‑sum‑exp
    approximation of the max together with a tiny entropy term.
    The variables are the unrestricted logits *b* such that
    a = softmax(b) (automatically satisfies a_i≥0 and Σa_i=1).
    A linear schedule slowly raises the smoothing parameter α.

3️⃣  **Exact‑objective refinement** – the Adam machinery is kept but the
    gradient is now the true sub‑gradient of the max‑convolution entry.
    This drives the solution to a (local) minimiser of the real objective.

All phases respect the global time budget (default 1000 s).  The function
returns the best sequence discovered as a plain Python list.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    y = np.exp(x - np.max(x))
    return y / y.sum()


def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution a⨂a (length 2·n‑1) via FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


def _smooth_gradient_plus_entropy(
    a: np.ndarray,
    conv: np.ndarray,
    alpha: float,
    L: int,
    lam: float,
) -> np.ndarray:
    """
    Gradient of   (2·n/α)·log∑_k e^{α·conv_k}   +   λ·∑ a_i·log a_i
    with respect to a (the probabilities).  The first term yields
    2·corr(w,a) where w_k ∝ e^{α·conv_k}.
    """
    # soft‑max weights from the log‑sum‑exp term
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)   # stability
    w = exp_scaled / exp_scaled.sum()          # shape (2n‑1,)

    # correlation w ⋆ a (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n_len = a.size
    grad_corr = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]

    # entropy contribution λ·(log a + 1)
    eps = 1e-20
    grad_ent = lam * (np.log(a + eps) + 1.0)

    return grad_corr + grad_ent


def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. a (averaged over all maximisers)."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)            # indices attaining the max
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


def _objective_from_a(a: np.ndarray, n: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a already on the simplex."""
    conv = _conv_fft(a, L=_next_fft_len(2 * n - 1))
    return 2.0 * n * conv.max()


def _next_fft_len(min_len: int) -> int:
    """Smallest power‑of‑two ≥ min_len."""
    return 1 << (min_len - 1).bit_length()

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    O(a) = 2·n·max(conv(a,a)) / (sum(a)**2).  Returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin
    deadline = start_time + budget_s - 0.3

    # ----------------------------------------------------------
    # Problem size
    if "height_sequence_1" in globals():
        base = np.array(height_sequence_1, dtype=float)
        n = base.size
        init_a = base / base.sum()
        print(f"[init] Using supplied sequence, n={n}")
    else:
        n = 5000                     # default length (a power‑of‑two neighbour is fine)
        init_a = np.full(n, 1.0 / n)  # uniform start
        print(f"[init] No supplied sequence, using uniform, n={n}")

    L = _next_fft_len(3 * n - 2)

    best_a = init_a.copy()
    best_score = 2.0 * n * _conv_fft(best_a, L).max()
    print(f"[sweep] start O={best_score:.6f}")

    # ----------- helper to evaluate a candidate --------------------
    def _evaluate_candidate(a_vec: np.ndarray):
        nonlocal best_a, best_score
        a_norm = a_vec
        if a_norm.sum() == 0:
            return
        a_norm = a_norm / a_norm.sum()
        conv = _conv_fft(a_norm, L)
        cur = 2.0 * n * conv.max()
        if cur < best_score:
            best_score = cur
            best_a = a_norm.copy()
            print(f"[sweep] new best O={best_score:.6f}")

    # uniform (already evaluated)
    # ------ inverse‑power family a_i ∝ (i+1)^{-p} + (n-i)^{-p} ----------
    p_vals = np.linspace(0.0, 2.5, 13)   # 0 → uniform, 2.5 → strong U‑shape
    idx = np.arange(n, dtype=float)
    for p in p_vals:
        if time.time() >= deadline:
            break
        tmp = (idx + 1) ** (-p) + (n - idx) ** (-p)
        _evaluate_candidate(tmp)

    # ------ exponential decay a_i ∝ exp(-p·dist_to_edge) ----------
    p_vals_exp = np.linspace(0.0, 5.0, 13)
    dist = np.minimum(idx, n - 1 - idx).astype(float)
    for p in p_vals_exp:
        if time.time() >= deadline:
            break
        tmp = np.exp(-p * dist)
        _evaluate_candidate(tmp)

    # ------ piecewise‑uniform (first/last m entries) ---------------
    for frac in [0.01, 0.02, 0.04, 0.06, 0.08, 0.10, 0.15, 0.20]:
        if time.time() >= deadline:

        m = max(1, int(frac * n))
        if 2 * m > n:

        tmp = np.zeros(n, dtype=float)
        tmp[:m] = 1.0 / (2 * m)
        tmp[-m:] = 1.0 / (2 * m)
        _evaluate_candidate(tmp)

    for _ in range(6):

            break
        tmp = np.random.rand(n)
        _evaluate_candidate(tmp)

    print(f"[phase‑2] starting smooth Adam, best O={best_score:.6f}")

    # start from the best vector we have
    eps = 1e-12
    b = np.log(best_a + eps)                # logits (unconstrained)
    m = np.zeros_like(b)
    v = np.zeros_like(b)
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.6
    lam = 0.02                               # tiny entropy regulariser

    # smoothing schedule
    alpha_start, alpha_end = 5.0, 8000.0

    phase2_deadline = start_time + budget_s * 0.45   # ~45 % of total time
    it = 0
    max_iter_phase2 = 200_000                       # generous upper bound

    while time.time() < phase2_deadline and it < max_iter_phase2:
        # current probabilities
        a = _softmax(b)

        # exact convolution (needed for evaluating true objective)
        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_score:
            best_score = cur_obj
            best_a = a.copy()
            print(f"[phase‑2] iter {it:,} new best O={best_score:.6f}")

        # anneal smoothing parameter
        frac = it / max_iter_phase2
        alpha = alpha_start + (alpha_end - alpha_start) * frac

        # gradient of the smooth surrogate w.r.t. a
        grad_a = _smooth_gradient_plus_entropy(a, conv, alpha, L, lam)

        # chain rule: ∂/∂b = a·(∂/∂a – (a·∂/∂a))
        dot = float(np.dot(a, grad_a))
        grad_b = a * (grad_a - dot)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_b
        v = beta2 * v + (1.0 - beta2) * (grad_b * grad_b)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        lr = lr0 / math.sqrt(it + 1.0)

        b = b - lr * m_hat / (np.sqrt(v_hat) + 1e-12)

        # tiny random jitter to help escape flat plateaus
        if np.random.rand() < 0.001:
            b += np.random.randn(n) * 1e-4

        it += 1

    # ----------------------------------------------------------
    # 3️⃣  Adam on the exact max‑convolution (sub‑gradient) ---------------
    print(f"[phase‑3] exact‑objective refinement, best O={best_score:.6f}")

    # keep the same Adam state (but re‑initialise moments for safety)
    m[:] = 0.0
    v[:] = 0.0
    beta1, beta2 = 0.9, 0.999
    lr0_3 = 0.5
    it3 = 0
    max_iter_phase3 = 250_000

    phase3_deadline = deadline
    while time.time() < phase3_deadline and it3 < max_iter_phase3:
        a = _softmax(b)

        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_score:
            best_score = cur_obj
            best_a = a.copy()
            print(f"[phase‑3] iter {it3:,} new best O={best_score:.6f}")

        # sub‑gradient of the true max‑convolution entry
        subgrad_a = _subgradient_max_conv(a, conv)

        # map to logits space
        dot = float(np.dot(a, subgrad_a))
        grad_b = a * (subgrad_a - dot)

        # Adam step on logits
        m = beta1 * m + (1.0 - beta1) * grad_b
        v = beta2 * v + (1.0 - beta2) * (grad_b * grad_b)
        m_hat = m / (1.0 - beta1 ** (it3 + 1))
        v_hat = v / (1.0 - beta2 ** (it3 + 1))
        lr = lr0_3 / math.sqrt(it3 + 1.0)

        b = b - lr * m_hat / (np.sqrt(v_hat) + 1e-12)

        # occasional tiny perturbation
        if np.random.rand() < 0.001:
            b += np.random.randn(n) * 5e-5

        it3 += 1

    # ----------------------------------------------------------
    # Final sanity check with the provided evaluator
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:
        print("[final] evaluate_sequence failed:", e)

    return best_a.tolist()
