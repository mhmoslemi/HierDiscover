"""
Optimization of a non‑negative step‑function sequence a∈ℝⁿ  (sum(a)=√(2n))
to minimise          O(a)=2·n·max(conv(a,a)) / (sum(a)²)
which, after normalising the L¹‑norm, reduces to minimising
M(a)=max(conv(a,a)).

The algorithm consists of three phases and several safety
features:

* **Phase 1 – Adam on a smooth soft‑max surrogate**.
  We minimise  (2·n/α)·log∑ₖexp(α·convₖ)  using Adam with a
  linearly‑annealed smoothing parameter α (5 → 2·10⁵).  After each
  update the vector is projected onto the probability simplex
  (∑=1) and re‑scaled to the required L¹‑norm.  The learning
  rate decays as 1/√t, a small jitter is added regularly and a
  very mild pruning of the smallest 0.1 % of the mass is performed
  every 10 000 steps.

* **Phase 2 – exact sub‑gradient descent** on the true objective
  M(a).  A geometrically decaying step‑size is used and the best
  iterate is kept.

* **Phase 3 – stochastic refinement**.
  Two kinds of moves are tried:
      – random mass‑transfer (as before);
      – a *guided* transfer that moves a fraction of the mass from
        the index with the largest sub‑gradient component to the
        index with the smallest one (these are the entries that
        contribute most/least to the current maximal convolution).
  Moves are accepted if they improve the exact objective or with a
  Metropolis probability that slowly cools.  Periodically a
  sub‑gradient step is performed to keep the search directed.
  If the search stalls for many iterations a jittered restart
  around the current best point is triggered.

All three phases respect the global time‑budget (default 1000 s)
and the best sequence found so far is finally checked with the
provided ``evaluate_sequence`` function before being returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:                     # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full self‑convolution (length 2·n‑1)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return a*a (full convolution) using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Exact objective for a simplex‑normalised vector (Σ=1)
def _objective_simplex(a: np.ndarray, n: int, L: int) -> float:
    """2·n·max(conv(a,a))  (a is assumed to sum to 1)."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Sub‑gradient of the maximal convolution entry (as used in the reference code)
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Return the sub‑gradient of max(conv) w.r.t. a."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)          # possibly several maximisers
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
# Gradient of the log‑sum‑exp (soft‑max) surrogate of the max
def _smooth_gradient(a: np.ndarray, conv: np.ndarray, alpha: float, L: int) -> np.ndarray:
    """
    Compute ∇_a ( (2·n/α)·log∑_k e^{α·conv_k} ).
    The formula reduces to 2·corr(w, a) where w_k ∝ e^{α·conv_k}.
    """
    # soft‑max weights
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)      # stabilise
    w = exp_scaled / exp_scaled.sum()

    # correlation w ⋆ a   (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)   # length ≥ 3·n‑2
    n_len = a.size
    grad = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]
    return grad

    # Allocate roughly 25 % of the total budget to this phase
    print(f"[phase‑2] start, best O={best_score:.6f}")
    phase2_deadline = start_time + budget_s * 0.8
    eta0 = 0.30                     # initial step‑size for sub‑gradient
    it2 = 0
    while time.time() < phase2_deadline:
        conv = _conv_fft(a / sum_target, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[sub‑grad] iter {it2:06d} new best O={best_score:.6f}")

        # sub‑gradient (on simplex‑scaled vector)
        g = _subgradient_max_conv(a / sum_target, conv) * sum_target  # chain rule
        # geometric decay of step‑size
        eta = eta0 * (0.9995 ** it2)
        a = a - eta * g
        a = _project_to_simplex(a) * sum_target
        it2 += 1

    # ------------------------------------------------------------------
    # Phase 3 – stochastic refinement (simulated annealing + guided moves)
    print(f"[phase‑3] stochastic refinement, best O={best_score:.6f}")
    phase3_deadline = deadline
    max_iter_local = 300_000
    no_improve_limit = 7_000
    temperature0 = 0.07
    T = temperature0
    it_loc = 0
    no_improve = 0
    cur_conv = _conv_fft(a / sum_target, L)
    cur_obj = 2.0 * n * cur_conv.max()

    # keep a copy of the very best sequence found so far
    a_best = best_seq.copy()

    while time.time() < phase3_deadline and it_loc < max_iter_local:
        # ----- decide which move to attempt -----
        guided = np.random.rand() < 0.6   # majority of moves are guided

        if guided:
            # compute sub‑gradient once (used for guiding)
            g = _subgradient_max_conv(a / sum_target, cur_conv) * sum_target

            # indices with biggest / smallest contribution
            i = int(np.argmax(g))
            j = int(np.argmin(g))
            if i == j or a[i] < 1e-14:
                # fall back to a random move if guiding fails
                i, j = np.random.randint(n, size=2)
        else:
            i, j = np.random.randint(n, size=2)

        if i != j and a[i] > 1e-14:
            # move a modest fraction (up to 15 %) of the source mass
            delta = a[i] * np.random.rand() * 0.15
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            a_new = _project_to_simplex(a_new) * sum_target

            conv_new = _conv_fft(a_new / sum_target, L)
            new_obj = 2.0 * n * conv_new.max()

            # Metropolis acceptance
            if new_obj < cur_obj or np.random.rand() < math.exp((cur_obj - new_obj) / max(T, 1e-12)):
                a = a_new
                cur_conv = conv_new
                cur_obj = new_obj
                no_improve = 0
                if cur_obj < best_score:
                    best_score = cur_obj
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] new global best O={best_score:.6f} (iter={it_loc})")
            else:
                no_improve += 1
        else:
            no_improve += 1

        # ----- occasional exact sub‑gradient step (keeps direction) -----
        if it_loc % 7 == 0:
            g = _subgradient_max_conv(a / sum_target, cur_conv) * sum_target
            eta_local = 0.06 * (1.0 - it_loc / max_iter_local)
            a = a - eta_local * g
            a = _project_to_simplex(a) * sum_target
            cur_conv = _conv_fft(a / sum_target, L)
            cur_obj = 2.0 * n * cur_conv.max()
            if cur_obj < best_score:
                best_score = cur_obj
                best_seq = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved O={best_score:.6f}")

        # ----- temperature decay -----
        T *= 0.9993

        # ----- restart if stagnation persists -----
        if no_improve >= no_improve_limit:
            # jittered restart around the best point seen so far
            a = a_best.copy()
            a += np.random.randn(n) * 0.0012
            a = np.clip(a, 0.0, None)
            a = _project_to_simplex(a) * sum_target
            cur_conv = _conv_fft(a / sum_target, L)
            cur_obj = 2.0 * n * cur_conv.max()
            no_improve = 0

        it_loc += 1

    # ------------------------------------------------------------------
    # Final sanity check with the provided evaluator
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence failed:", exc)

    # Return a plain Python list (values are already non‑negative and respect the caps)
    return best_seq.tolist()
