#!/usr/bin/env python
"""
Optimising the step‑function bound J(a)=2·n·max(a*a)/(∑a)².

The optimiser works in three phases:
  1)  A cheap scan over a few parametric families (power‑law,
      geometric) to obtain a good initial distribution.
  2)  An Adam optimiser on the soft‑max (log‑it) parametrisation
      with a smoothly‑approximated max (log‑sum‑exp).  The smoothing
      factor ε is driven to 0 so that the algorithm finally follows
      the true sub‑gradient of max(a*a).  The optimiser enlarges the
      support up to a configurable maximum (≈ 25 000) and performs
      occasional random perturbations to escape plateaux.
  3)  A short L‑BFGS polishing step (SciPy) that works directly
      on the logits with a very small ε, followed by a stochastic
      local‑search that moves tiny amounts of mass between two
      coordinates.

The routine respects the supplied time budget (≈ 1000 s).  All
intermediate scores are printed; the final exact score is obtained
by calling ``evaluate_sequence`` on the returned list.
"""

# ----------------------------------------------------------------------
# Imports
# ----------------------------------------------------------------------
import time
import math
import numpy as np

# Scipy is optional – it speeds up the final polishing step
try:
    from scipy.optimize import minimize  # type: ignore
    _has_scipy = True
except Exception:  # pragma: no cover
    _has_scipy = False

# Helper utilities – top‑level and without closures
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Returns logits that produce ``a`` after soft‑max."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """
    Smallest power‑of‑two ≥ 3·n‑2.
    Guarantees a length that can hold the linear convolution of size 2·n‑1
    and the subsequent correlation of size 3·n‑2.
    """
    return 1 << ((3 * n - 2).bit_length())


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    total = new_a.sum()
    if total == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= total
    return new_a


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """Normalised power‑law a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _geom_distribution(n: int, r: float) -> np.ndarray:
    """Normalised geometric a_i ∝ r**i (0<r<1)."""
    if r <= 0.0:
        return np.full(n, 1.0 / n)
    idx = np.arange(n, dtype=np.float64)
    a = r ** idx
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _conv_max(a: np.ndarray) -> float:
    """Return the maximal entry of the full linear convolution a*a."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float) -> tuple[float, np.ndarray]:
    """
    Compute max entry of a*a and the gradient of the smoothed objective
    2·n·smooth_max.  ``eps_factor`` controls the smoothing constant:
    ε = eps_factor·max_conv.
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    # FFT of a (zero‑padded)
    a_fft = np.fft.rfft(a, L)

    # --- self‑convolution a*a (linear) ---
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # --- smoothing weights w (soft‑max over conv) ---
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv           # for numerical stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or np.isnan(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # --- cross‑correlation of a and w (linear) ---
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # gradient of the smoothed objective (4·n * correlation)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a


def _cheap_score(a: np.ndarray) -> float:
    """2·n·max_conv for a normalised to sum=1 (the value returned by evaluate_sequence)."""
    n = a.shape[0]
    return 2.0 * n * _conv_max(a)
