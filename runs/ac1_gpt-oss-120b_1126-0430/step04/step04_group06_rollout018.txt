#!/usr/bin/env python
"""
Improved attack on the step‑function bound.

The optimisation problem is:
    minimise  2·n·max_k (a*a)[k]   subject to   a_i ≥ 0 , Σ a_i = 1.
(The external ``evaluate_sequence`` rescales the vector, so we may treat
 ``a`` as a probability distribution.)

The code below runs a mirror‑descent / Adam optimiser on the log‑it
parameterisation of a probability vector.  The gradient is obtained from a
*soft‑max* approximation of the max‑convolution:
    w_k ∝ exp(θ·(conv_k−max(conv)))   (θ is a temperature that is
    increased during the run).  The exact gradient of the smoothed
    max is
        g_i = 2·∑_k w_k·a_{k−i},
    which can be computed by a second FFT (convolution of the weight
    vector with the current distribution).

The optimiser works on a fixed support size (≈30 000 – the largest size
allowed without becoming prohibitively slow).  The algorithm keeps the
best sequence seen (according to the cheap proxy 2·n·max(conv)) and
returns it at the end of the time budget.

The method is deterministic apart from a single random seed that is used
to initialise the logits; a few random perturbations are injected during
the run to help escape plateaux.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top‑level, no closures
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max; returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it representation that yields ``a`` after soft‑max."""
    return np.log(a + eps)


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full linear self‑convolution a*a using an FFT of length ``fft_len``.
    Result length = 2·len(a)‑1.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.size - 1]
    return conv


def _smooth_max_gradient(a: np.ndarray, conv: np.ndarray, theta: float) -> np.ndarray:
    """
    Gradient of the soft‑max approximation to ``max(conv)``.
    Returns ∂/∂a_i  (log‑sum‑expₜ conv)  where the temperature is ``theta``.
    """
    n = a.shape[0]
    # ----- soft‑max weights over all convolution entries -----------------
    max_val = conv.max()
    # subtract max for numerical stability
    w_raw = np.exp(theta * (conv - max_val))
    w = w_raw / w_raw.sum()                # shape (2·n‑1,)

    # ----- compute g_i = 2·∑_k w_k·a_{k‑i} using a second FFT ------------
    # Reverse the weight vector because of the index shift in the formula.
    w_rev = w[::-1]

    # Length needed for the convolution w_rev * a
    conv_len_needed = n + w_rev.size - 1          # = 3·n‑2
    fft_len = _next_pow2(conv_len_needed)

    fa = np.fft.rfft(a, fft_len)
    fw = np.fft.rfft(w_rev, fft_len)
    conv_rev = np.fft.irfft(fa * fw, fft_len)[:conv_len_needed]

    # slice that yields g_i (see derivation in the docstring)
    start = w_rev.size - 1            # 2·n‑2
    g = 2.0 * conv_rev[start: start + n]
    return g


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """Normalised power‑law a_i ∝ (i+1)^{‑p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) beats the current
    best bound.  The routine respects the given time budget (≈1000 s) and
    uses a smoothed‑max gradient Adam optimiser on the probability simplex.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # safety margin for final eval

    # ---------------------------------------------------------------
    # 0. Choose a working support size (the larger the better, up to limit)
    # ---------------------------------------------------------------
    n = 30000                     # hard‑coded maximum that still fits in memory
    print(f"[Init] Optimising with support size n = {n}")

    # ---------------------------------------------------------------
    # 1. Initialise a probability vector.
    # ---------------------------------------------------------------
    # Start from a mild power‑law – this is a generic, non‑trivial start.
    a = _power_law_distribution(n, 0.5)
    print("[Init] Power‑law start with exponent = 0.5")

    # ---------------------------------------------------------------
    # 2. Log‑it parametrisation + Adam state.
    # ---------------------------------------------------------------
    eps = 1e-16
    z = _logits_from_distribution(a, eps)

    m = np.zeros_like(z)        # first‑moment estimate
    v = np.zeros_like(z)        # second‑moment estimate
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    base_alpha = 0.30           # base learning‑rate (soft‑max gradient is smoother)

    # ---------------------------------------------------------------
    # 3. Pre‑compute FFT length for the self‑convolution (used every step).
    # ---------------------------------------------------------------
    fft_len_a = _next_pow2(2 * n - 1)

    # ---------------------------------------------------------------
    # 4. Main optimisation loop.
    # ---------------------------------------------------------------
    best_seq = a.copy()
    best_score = float("inf")
    it = 0
    last_improve_it = 0
    report_every = 5000

    # temperature schedule for the soft‑max (from gentle smoothing to sharp)
    theta_start = 0.6
    theta_end = 20.0
    theta_slope = (theta_end - theta_start) / max(1.0, budget_s - 10.0)

    while time.time() < deadline and it < 2_000_000:
        it += 1

        # ----- probability vector from the logits --------------------
        a = _softmax(z)

        # ----- self‑convolution (full) --------------------------------
        conv = _conv_full_fft(a, fft_len_a)

        max_conv = float(conv.max())
        cheap_score = 2.0 * n * max_conv

        # ----- record best ever seen ---------------------------------
        if cheap_score < best_score:
            best_score = cheap_score
            best_seq = a.copy()
            last_improve_it = it
            if it % report_every == 0:
                print(f"[Iter {it:07d}] NEW BEST cheap = {cheap_score:.6f}")

        # ----- smoothed max gradient ----------------------------------
        elapsed = time.time() - start
        theta = min(theta_end, theta_start + theta_slope * elapsed)

        g_a = _smooth_max_gradient(a, conv, theta)

        # ----- mirror‑descent conversion to log‑it space (KL geometry)
        mu = float(np.dot(g_a, a))
        grad_z = a * (g_a - mu)

        # ----- Adam moment updates (bias‑corrected) -------------------
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # ----- adaptive step size (inverse‑sqrt decay) ---------------
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)

        # ----- perform the update in log‑it space --------------------
        z = z + alpha_t * (- m_hat / (np.sqrt(v_hat) + eps_adam))
        np.clip(z, -80.0, 80.0, out=z)       # keep logits in a safe range

        # ----- occasional random perturbations (helps escape plateaux)
        if it % 3000 == 0 and np.random.rand() < 0.20:
            perturb = 0.015 * np.random.randn(n)
            z += perturb
            m *= 0.5
            v *= 0.5

        # ----- early stopping if no progress for a long while --------
        if it - last_improve_it > 400_000:
            print(f"[Iter {it:07d}] No improvement for {it-last_improve_it} iters – terminating.")
            break

    # ---------------------------------------------------------------
    # 5. Final exact evaluation (the platform will also evaluate)
    # ---------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[Result] Final exact score = {final_score:.6f}")
    except Exception as exc:
        final_score = float("inf")
        print(f"[Result] Evaluation error: {exc}")

    # ---------------------------------------------------------------
    # 6. Return the best sequence found.
    # ---------------------------------------------------------------
    return final_seq
