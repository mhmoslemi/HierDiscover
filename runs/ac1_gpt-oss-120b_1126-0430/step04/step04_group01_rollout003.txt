def _corr_w_a(w: np.ndarray, a: np.ndarray, L: int) -> np.ndarray:
    n = a.size
    # correlation of w with a reversed
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    # pick appropriate segment
    return 2.0 * corr_rev[n-1:n-1+n]
```

Propose a step‑function sequence that (approximately) minimises

    O(a) = 2·n·max(conv(a,a)) / (sum a)^2

The routine runs inside the given time budget (default 1000 s) and
returns the best sequence found as a plain list of floats.

Algorithm (three phases)

1️⃣ Random sampling (Dirichlet draws) – cheap way to obtain a decent
   starting point on the probability simplex.

2️⃣ Mirror‑descent (exponentiated‑gradient) on a smooth surrogate of the
   max‑convolution:  fα(a)= (2·n/α)·log∑k exp(α·convk(a)).
   The gradient of the soft‑max part is 2·corr(w,a) where
   w_k ∝ exp(α·convk).  A KL‑mirror step (log‑space update + renormalisation)
   guarantees non‑negativity and the simplex constraint.

   α (the temperature) is annealed from a small value (≈5) up to a large
   one (≈15000).  The step‑size η is decreased linearly.

3️⃣ Stochastic local refinement – cheap “mass‑transfer’’ moves accepted
   with a Metropolis rule, plus occasional sub‑gradient steps on the
   exact max‑convolution entry.  This phase tightens the bound further.

All heavy operations use FFT‑based convolution (O(n log n)).  The code
automatically respects the global time limit; if the budget is exhausted
the current best sequence is returned.

# Helper utilities

def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # fallback – uniform vector
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution a⨂a (length 2·n‑1) using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


def _objective_simplex(a: np.ndarray, n: int, L: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a already normalised (∑a=1)."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. a (averaged over all maximisers)."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)  # indices where max is attained
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


def _correlation_w_a(w: np.ndarray, a: np.ndarray, L: int) -> np.ndarray:
    """
    Compute g_i = 2·∑_j w_{i+j}·a_j   (i=0..n‑1).
    This is the gradient of the smooth surrogate w.r.t. a.
    """
    n = a.size
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    # extract the n relevant entries (centered part of the correlation)
    g = 2.0 * corr_rev[n - 1 : n - 1 + n]
    return g

        # Dirichlet with small concentration → sparse vectors
        x = np.random.gamma(0.5, 1.0, size=n)
        a = x / x.sum()
        conv = _conv_fft(a, L)
        cur = 2.0 * n * conv.max()
        if cur < best_score:
            best_score = cur
            best_seq = a.copy()
            print(f"[sample] #{s:03d} new best O={best_score:.6f}")

    # If the random stage did not produce anything (unlikely), initialise uniformly.
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = _objective_simplex(best_seq, n, L)
        print("[init] fallback to uniform start")

    # ------------------------------------------------------------------
    # Phase 1 – mirror‑descent on the smooth log‑sum‑exp surrogate
    # ------------------------------------------------------------------
    a = best_seq.copy()
    max_iter_phase1 = 300_000
    for it in range(max_iter_phase1):
        if time.time() >= deadline:
            break

        # --- compute current convolution ---
        conv = _conv_fft(a, L)

        # --- anneal the temperature α ---
        alpha_start, alpha_end = 5.0, 15000.0
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iter_phase1)

        # --- soft‑max weights w_k ∝ exp(α·conv_k)  (stable) ---
        scaled = alpha * conv
        max_scaled = scaled.max()
        exp_scaled = np.exp(scaled - max_scaled)
        w = exp_scaled / exp_scaled.sum()

        # --- gradient of the smooth surrogate (2·corr) ---
        grad = _correlation_w_a(w, a, L)

        # --- KL‑mirror (exponentiated‑gradient) update ---
        #   log‑space update: log a ← log a – η·grad, then renormalise.
        eta = 0.15 * (1.0 - it / max_iter_phase1) + 0.001
        log_a = np.log(a + 1e-30) - eta * grad
        a = np.exp(log_a)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / a_sum

        # --- occasional bookkeeping ---
        if it % 5000 == 0:
            conv_check = _conv_fft(a, L)
            cur_obj = 2.0 * n * conv_check.max()
            if cur_obj < best_score:
                best_score = cur_obj
                best_seq = a.copy()
                print(f"[phase‑1] iter {it:,} new best O={best_score:.6f}")

    print(f"[phase‑1] finished, best O={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – stochastic local refinement
    # ------------------------------------------------------------------
    a = best_seq.copy()
    conv = _conv_fft(a, L)
    cur_obj = 2.0 * n * conv.max()
    best_seq = a.copy()
    best_score = cur_obj

    subgrad_interval = 5
    temperature = 0.03
    max_iter_local = 200_000
    no_improve_limit = 2000
    it_loc = 0
    no_improve = 0

    while time.time() < deadline and it_loc < max_iter_local:
        # ---- cheap mass‑transfer move ----
        i, j = np.random.randint(n, size=2)
        if i != j and a[i] > 1e-15:
            delta = a[i] * np.random.rand() * 0.20  # up to 20 % of source mass
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            a_new /= a_new.sum()
            conv_new = _conv_fft(a_new, L)
            new_obj = 2.0 * n * conv_new.max()
            # Metropolis acceptance
            if new_obj < cur_obj or np.random.rand() < math.exp(
                (cur_obj - new_obj) / max(temperature, 1e-15)
            ):
                a, conv, cur_obj = a_new, conv_new, new_obj
                no_improve = 0
                if cur_obj < best_score:
                    best_score = cur_obj
                    best_seq = a.copy()
                    print(
                        f"[local] new global best O={best_score:.6f} iter={it_loc}"
                    )
            else:
                no_improve += 1
        else:
            no_improve += 1

        # ---- sub‑gradient step on the exact max‑convolution entry ----
        if it_loc % subgrad_interval == 0:
            g = _subgradient_max_conv(a, conv)
            eta = 0.05 * (1.0 - it_loc / max_iter_local) + 1e-4
            a = a - eta * g
            a = np.clip(a, 0.0, None)
            s = a.sum()
            if s == 0.0:
                a = np.full(n, 1.0 / n)
            else:
                a = a / s
            conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_score:
                best_score = cur_obj
                best_seq = a.copy()
                print(f"[local] sub‑grad improved O={best_score:.6f}")

        # ---- temperature decay ----
        temperature *= 0.9999
        it_loc += 1

        # ---- restart if stuck for a while ----
        if no_improve >= no_improve_limit:
            a = best_seq.copy()
            a += np.random.randn(n) * 0.001
            a = np.clip(a, 0.0, None)
            a = a / a.sum()
            conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max()
            no_improve = 0

    print(f"[phase‑2] finished, best O={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check with the provided evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:
        print("[final] evaluation failed:", exc)

    return final_seq
