"""
Propose a low‑value step function for the security‑evaluation task.

The algorithm works in three stages on a simplex (non‑negative entries that sum to 1).

1. **Smooth Adam descent** – we optimise a soft‑max approximation of the
   max‑convolution using Adam in the log‑it (log‑softmax) space.
   The smoothing parameter β is increased exponentially (annealing) so the
   optimisation first sees a smooth landscape and later concentrates on the
   true maximum.

2. **Exact max‑subgradient mirror descent** – after the annealing phase we
   switch to the true max (binary weight vector) and perform a few hundred
   iterations of mirror‑descent on the simplex:
        a ← a·exp(−η·∇)  (· denotes element‑wise product)
   followed by renormalisation.  This aggressively reduces the exact objective.

3. **Light‑weight greedy local search** – random mass‑transfer moves are
   tried while time permits; any improvement is kept.

Multiple support sizes are tried (powers of two are preferred for fast FFTs);
the best sequence found within the overall 1000 s budget is returned.
All numbers are non‑negative; the routine never returns an empty list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full a*a convolution (length 2·n−1) using a real FFT.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv


def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation g_i = Σ_t w[i+t]·a[t] for i = 0…n−1.
    Implemented via FFT (linear convolution, then truncation).
    """
    n = a.shape[0]
    # length needed for linear convolution of w (len 2n-1) with a (len n)
    L = _next_pow2(w.shape[0] + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


def _objective(a: np.ndarray) -> float:
    """
    Exact evaluation ``2·n·max(conv(a,a))``; assumes ``sum(a) == 1``.
    """
    n = a.size
    conv = _conv_self_fft(a)
    return 2.0 * n * conv.max()


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


# ----------------------------------------------------------------------
def _run_opt(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
) -> tuple[list[float], float]:
    """
    Optimise a length‑``n`` step function within ``time_limit`` seconds.

    Returns a plain list of floats (sum = 1) and the exact objective value.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # ---- initialise ----------------------------------------------------
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
        a = np.clip(a, 0.0, None)
    else:
        a = rng.random(n)

    a = a / a.sum()                     # simplex constraint

    # We work in the log‑it space (log‑softmax) because the simplex
    # constraint is automatically satisfied after the soft‑max.
    x = np.log(a + 1e-16)               # avoid log(0)

    # Adam moments
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # ---- annealing schedule for the smooth max ------------------------
    beta_start, beta_end = 0.1, 25000.0          # smoothing parameter
    lr_start, lr_end = 0.20, 0.001               # learning rate on logits
    max_iters = 600_000

    best_a = a.copy()
    best_obj = _objective(best_a)

    it = 0
    while time.time() - start < time_limit * 0.70 and it < max_iters:
        it += 1

        # current probabilities
        a = _softmax(x)

        # exact convolution & objective (for bookkeeping)
        conv = _conv_self_fft(a)
        M = conv.max()
        obj = 2.0 * n * M
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a.copy()
            print(f"[{n:5d} it={it:06d}] new best O={best_obj:.6f}")

        # schedule parameters (fraction of total iterations)
        t = min(it / max_iters, 1.0)
        beta = beta_start * np.exp(np.log(beta_end / beta_start) * t)
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t)

        # ----- smooth‑max weights ---------------------------------------
        if beta > 20000.0:                     # treat as hard max
            kmax = int(np.argmax(conv))
            w = np.zeros_like(conv)
            w[kmax] = 1.0
        else:
            shifted = conv - M                 # ≤ 0
            w = np.exp(beta * shifted)
            w_sum = w.sum()
            if w_sum == 0.0:                    # numerical safety
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
                w_sum = 1.0
            w = w / w_sum

        # ----- gradient of the smoothed max (w.r.t. a) ------------------
        g_a = 2.0 * _correlation_fft(w, a)    # shape (n,)

        # ----- natural gradient in log‑it space (mirror) ---------------
        dot = np.dot(a, g_a)                  # ⟨a, g⟩
        g_x = a * (g_a - dot)                 # ∇ in log‑it coordinates

        # ----- Adam update ---------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape flat regions
        if it % 8000 == 0:
            x += rng.normal(scale=0.015, size=n)

        # restart from best every so often (helps escape plateaus)
        if it % 100_000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.004, size=n)
            m.fill(0.0)
            v.fill(0.0)
            print(f"[{n}] restart at iter {it}")

    # ------------------------------------------------------------------
    # Phase 2 – exact max subgradient (mirror descent)
    remaining = time_limit - (time.time() - start)
    if remaining > 1.5:
        # use the current best as starting point
        a = best_a.copy()
        eta_start, eta_end = 0.5, 0.001
        max_iters2 = 4000
        it2 = 0
        while time.time() - start < time_limit * 0.95 and it2 < max_iters2:
            it2 += 1
            conv = _conv_self_fft(a)
            M = conv.max()
            obj = 2.0 * n * M
            if obj < best_obj - 1e-15:
                best_obj = obj
                best_a = a.copy()
                print(f"[{n:5d} sub it={it2:04d}] new best O={best_obj:.6f}")

            # indicator of the (possibly multiple) max positions
            eps_max = 1e-12
            max_inds = np.where(conv >= M - eps_max)[0]
            w = np.zeros_like(conv)
            w[max_inds] = 1.0 / max_inds.size

            # exact subgradient
            g_a = 2.0 * _correlation_fft(w, a)

            # mirror‑descent (exponentiated gradient) step
            frac = it2 / max_iters2
            eta = eta_start * np.exp(np.log(eta_end / eta_start) * frac)
            a = a * np.exp(-eta * g_a)
            a_sum = a.sum()
            if a_sum == 0.0:                     # safety – re‑initialise
                a = rng.random(n)
                a /= a.sum()
            else:
                a /= a_sum

    # ------------------------------------------------------------------
    # Phase 3 – greedy random mass‑transfer local search
    remaining = time_limit - (time.time() - start)
    if remaining > 0.5:
        a_loc = best_a.copy()
        obj_loc = best_obj
        n_loc = a_loc.size
        local_start = time.time()
        while time.time() - local_start < remaining * 0.4:
            i, j = rng.choice(n_loc, size=2, replace=False)
            # move a small fraction from i to j
            delta = a_loc[i] * rng.random() * 0.02
            if delta <= 0.0:
                continue
            a_new = a_loc.copy()
            a_new[i] -= delta
            a_new[j] += delta
            conv_new = _conv_self_fft(a_new)
            obj_new = 2.0 * n_loc * conv_new.max()
            if obj_new < obj_loc:
                a_loc, obj_loc = a_new, obj_new
                if obj_new < best_obj:
                    best_a, best_obj = a_new, obj_new
        print(f"[local] final O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Remove entries that are numerically zero to keep the sequence short.
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = _objective(best_a)
        print(f"[compress] length reduced to {best_a.size}, O={best_obj:.6f}")

    return best_a.tolist(), best_obj


# ----------------------------------------------------------------------
def _resize_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Resize a sequence to a new length using linear interpolation
    on the cumulative distribution (preserves total mass = 1).
    """
    old_n = seq.size
    cdf = np.cumsum(seq)
    cdf = np.insert(cdf, 0, 0.0)            # length old_n+1, last entry =1
    xs = np.linspace(0.0, 1.0, new_n + 1)
    new_cdf = np.interp(xs, np.linspace(0.0, 1.0, old_n + 1), cdf)
    new_seq = np.diff(new_cdf)
    # numerical cleanup
    new_seq[new_seq < 0] = 0.0
    new_seq /= new_seq.sum()
    return new_seq


# ----------------------------------------------------------------------
def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **_ignore,
) -> list[float]:
    """
    Main entry point for the scoring harness.

    Runs several optimisation attempts for different support sizes
    (powers of two are preferred) within the allotted time budget and
    returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.4   # safety margin

    # --------------------------------------------------------------
    # Determine candidate lengths (powers of two, plus any historic length)
    lengths = [1024, 2048, 4096, 8192]         # 8192 is still fast with FFTs
    if 'height_sequence_1' in globals():
        prev = np.asarray(height_sequence_1, dtype=float)
        prev = np.clip(prev, 0.0, None)
        if prev.sum() > 0.01:
            n_prev = prev.size
            if n_prev not in lengths:
                lengths.insert(0, n_prev)

    best_seq = None
    best_obj = float('inf')

    # --------------------------------------------------------------
    for idx, n in enumerate(lengths):
        now = time.time()
        if deadline - now < 2.0:
            break

        # allocate time: first (largest) run gets the lion's share
        if idx == 0:
            run_time = (deadline - now) * 0.75
        else:
            remaining_runs = len(lengths) - idx
            run_time = (deadline - now - 0.5) / remaining_runs

        init = None
        if idx == 0 and 'height_sequence_1' in globals():
            # reuse the previous construction, possibly resized
            init = _resize_sequence(prev, n)

        seq, obj = _run_opt(
            n=n,
            time_limit=run_time,
            init_a=init,
            seed=seed + idx * 31,
        )
        if obj < best_obj:
            best_obj, best_seq = obj, seq
            print(f"[global] new overall best O={best_obj:.6f} (n={n})")

    # --------------------------------------------------------------
    # Final sanity check with the official evaluation routine.
    try:
        final_score = evaluate_sequence(best_seq)   # type: ignore
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:                         # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_seq
