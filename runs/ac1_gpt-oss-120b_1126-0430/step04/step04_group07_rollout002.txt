"""
Propose a non‑negative step function (list of floats) that minimizes the
bound  O = 2·n·max(conv(a,a)) / (sum(a)^2)  (the provided evaluate_sequence).

Algorithm
---------

* **Initialization**  
  • If a previous construction `height_sequence_1` is present, start from it
    (with a tiny random jitter).  
  • Otherwise use a smooth sinusoidal shape – a good generic starter for the
    convolution‑flattening problem.

* **Exponentiated‑gradient optimisation on the simplex**  
  The probabilities `a_i` are represented by log‑variables `u_i` (`a_i ∝ e^{u_i}`).
  At each iteration:
    1. Build the probability vector `a` (stable normalisation).
    2. Compute the full self‑convolution `c = a*a` via a real FFT.
    3. Determine a weight vector `w` over the convolution entries:
        – early iterations use a soft‑max (`β` small) to give a smooth gradient,
        – later iterations switch to a hard‑max (uniform weight on the active
          maximal entries) for a true sub‑gradient of the max‑norm.
    4. Obtain the gradient `g = 2·corr(w, a)` (the correlation implements the
       derivative of the weighted convolution).
    5. Apply **momentum** to the gradient (`g_mom = γ·g_mom + (1‑γ)·g`).
    6. Mirror‑descent step on the log‑variables
          `u ← u – η·g_mom`
       with a learning‑rate `η` that decays exponentially.
    7. Every few thousand iterations add a small random jitter to `u` and,
       periodically, restart from the current best solution with a modest
       perturbation (helps escaping plateaus).

* **Book‑keeping**  
  The best vector seen so far (in terms of the raw bound `2·n·max(conv)`) is
  stored and printed.  After the time budget expires the best vector is
  normalised to sum‑to‑one and returned as a plain Python list.

* **Parameters** (chosen empirically for the 1000 s budget)  
  - Length `n = 8192` (a power of two → fast FFT).  
  - Soft‑max temperature `β` grows from 4 to 30000.  
  - Learning rate `η` decays from 1.0 to 0.01.  
  - Momentum coefficient `γ = 0.9`.  
  - Jitter every 3000 iters (scale 1e‑3).  
  - Full restart every 30 000 iters.

The routine respects all constraints of `evaluate_sequence` (non‑negative,
finite values, sum‑to‑one invariance) and prints progress for debugging.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv


def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Linear correlation g_i = Σ_t w[i+t]·a[t]   for i = 0,…,n‑1.
    w has length 2·n‑1, a has length n.
    Implemented via FFT.
    """
    n = a.shape[0]
    m = w.shape[0]                # = 2·n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Returns the best sequence (list of floats, sum = 1).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # 1) Choose length and initialise log‑probabilities (u)
    # --------------------------------------------------------------
    n = 8192                     # power‑of‑two length → FFT is fast
    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=float)
        a0 = a0 / a0.sum()
        # pad or truncate to length n
        if a0.size < n:
            a0 = np.pad(a0, (0, n - a0.size), mode='constant')
        elif a0.size > n:
            a0 = a0[:n]
        eps = 1e-12
        u = np.log(a0 + eps)
    else:
        # sinusoidal start: a_i ∝ sin(π·(i+0.5)/n)
        idx = np.arange(n, dtype=float)
        a0 = np.sin(math.pi * (idx + 0.5) / n)
        a0 = np.maximum(a0, 0.0)
        a0 = a0 / a0.sum()
        eps = 1e-12
        u = np.log(a0 + eps)

    # auxiliary data
    best_u = u.copy()
    best_score = np.inf
    best_iter = 0
    g_mom = np.zeros_like(u)          # momentum term

    # --------------------------------------------------------------
    # 2) Hyper‑parameter schedules
    # --------------------------------------------------------------
    # number of iterations we roughly aim for; the loop will stop earlier
    max_iters_est = 300_000

    # learning‑rate schedule (exponential decay)
    eta_start = 1.0
    eta_end = 0.01

    # soft‑max temperature schedule (increase → sharper)
    beta_start = 4.0
    beta_end = 30000.0

    # momentum coefficient
    gamma = 0.9

    # jitter / restart settings
    jitter_interval = 3000
    jitter_scale = 1e-3
    restart_interval = 30_000

    it = 0
    while time.time() < deadline and it < 700_000:
        it += 1

        # ----------------------------------------------------------
        # (a) Build probability vector a from log‑variables u
        # ----------------------------------------------------------
        a = np.exp(u - u.max())          # stable exponentiation
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full_like(a, 1.0 / n)
            a_sum = 1.0
        a = a / a_sum                    # enforce Σ a_i = 1

        # ----------------------------------------------------------
        # (b) Self‑convolution and current bound
        # ----------------------------------------------------------
        conv = _conv_self_fft(a)         # length = 2·n‑1
        M = conv.max()
        cur_score = 2.0 * n * M           # because Σa = 1

        # ----------------------------------------------------------
        # (c) Record best solution
        # ----------------------------------------------------------
        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_u = u.copy()
            best_iter = it
            print(f"[iter {it:07d}] new best O={best_score:.6f}")

        # ----------------------------------------------------------
        # (d) Schedule the parameters for this iteration
        # ----------------------------------------------------------
        fraction = min(it / max_iters_est, 1.0)
        eta = eta_start * (eta_end / eta_start) ** fraction
        beta = beta_start + (beta_end - beta_start) * fraction

        # ----------------------------------------------------------
        # (e) Construct weight vector w over convolution entries
        # ----------------------------------------------------------
        if fraction < 0.4:           # early stage: smooth soft‑max
            shifted = conv - M       # ≤ 0, max = 0
            # exponentials stay ≤ 1, safe from overflow
            w = np.exp(beta * shifted)
            w_sum = w.sum()
            if w_sum == 0.0:        # numerical guard
                w = np.zeros_like(conv)
                w[conv.argmax()] = 1.0
            else:
                w /= w_sum
        else:                         # later stage: hard‑max active set
            tol = max(1e-12, M * 1e-8)           # tiny tolerance
            active = np.where(conv >= M - tol)[0]
            if active.size == 0:
                w = np.zeros_like(conv)
                w[conv.argmax()] = 1.0
            else:
                w = np.zeros_like(conv)
                w[active] = 1.0 / active.size

        # ----------------------------------------------------------
        # (f) Gradient of the (smoothed) max‑convolution
        # ----------------------------------------------------------
        g = 2.0 * _correlation_fft(w, a)   # shape (n,)

        # ----------------------------------------------------------
        # (g) Momentum update
        # ----------------------------------------------------------
        g_mom = gamma * g_mom + (1.0 - gamma) * g

        # ----------------------------------------------------------
        # (h) Mirror‑descent / exponentiated‑gradient step on u
        # ----------------------------------------------------------
        u = u - eta * g_mom

        # ----------------------------------------------------------
        # (i) Random jitter to avoid stagnation
        # ----------------------------------------------------------
        if it % jitter_interval == 0:
            u += np.random.randn(n) * jitter_scale

        # ----------------------------------------------------------
        # (j) Periodic cheap restart from the best‑so‑far solution
        # ----------------------------------------------------------
        if it % restart_interval == 0:
            u = best_u.copy()
            u += np.random.randn(n) * (jitter_scale * 0.5)
            print(f"[restart] iter {it}, re‑initialised from best (O={best_score:.6f})")

    # ------------------------------------------------------------------
    # 3) Return the best vector found (normalised to sum = 1)
    # ------------------------------------------------------------------
    best_seq = np.exp(best_u - best_u.max())
    best_seq = best_seq / best_seq.sum()
    # Convert to plain Python list as required by the evaluator
    return best_seq.tolist()
