class SurrogateObjective:
    def __init__(self, alpha, n, L):

        self.n = n
        self.L = L
    def __call__(self, u):

        conv = conv_fft(a, a, self.L)  # length 2n-1
        # compute log-sum-exp with scaling
        scaled = self.alpha * conv
        max_scaled = scaled.max()
        sum_exp = np.exp(scaled - max_scaled).sum()
        f = (2.0 * self.n / self.alpha) * (max_scaled + np.log(sum_exp))
        # gradient
        w = np.exp(scaled - max_scaled) / sum_exp  # softmax weight
        # compute grad_a

        # compute grad_u
        dot = np.dot(a, grad_a)
        grad_u = a * (grad_a - dot)
        return f, grad_u

"""
Propose a non‑negative step‑function that (approximately) minimizes

    O(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm has three parts:

1️⃣  *Smooth surrogate minimisation* – we minimise the log‑sum‑exp
    surrogate of the max‑operator using L‑BFGS‑B on the
    soft‑max (probability‑simplex) parametrisation a_i = exp(u_i)/∑exp(u).
    The smoothing parameter α is increased geometrically from 5 up to 2·10⁴.

2️⃣  *Exact sub‑gradient steps* – occasional projected sub‑gradient moves
    on the true objective keep the iterates focused on the nonsmooth
    landscape.

3️⃣  *Stochastic annealing* – a simulated‑annealing–style mass‑transfer
    move (move a small fraction of probability mass from one index to
    another) is performed until the time budget runs out.  All moves are
    accepted if they improve the exact objective or with a Metropolis
    probability that cools over time.

The best sequence encountered is returned (as a plain list of floats).
"""

import time
import math
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Basic utilities -------------------------------------------------------
def softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max that returns a probability vector."""
    vmax = np.max(v)
    e = np.exp(v - vmax)
    return e / e.sum()


def conv_fft(x: np.ndarray, y: np.ndarray, L: int) -> np.ndarray:
    """
    Full linear convolution of real vectors x and y using an FFT of length L.
    The result has length len(x)+len(y)-1.
    """
    fx = np.fft.rfft(x, n=L)
    fy = np.fft.rfft(y, n=L)
    conv_full = np.fft.irfft(fx * fy, n=L)
    return conv_full[: len(x) + len(y) - 1]


def exact_objective(a: np.ndarray, n: int, L: int) -> float:
    """Return O(a) = 2·n·max(conv(a,a)) (sum(a) is 1 because of the soft‑max)."""
    conv = conv_fft(a, a, L)                # length 2·n‑1
    return 2.0 * n * conv.max()


def _subgradient_max_conv(a: np.ndarray, conv_full: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a (used for exact sub‑gradient steps).
    conv_full must be the *full* convolution (length 2·n‑1).
    """
    n = a.size
    max_val = conv_full.max()
    ks = np.flatnonzero(conv_full == max_val)          # all maximisers
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
# Surrogate objective (smooth approximation of the max‑operator) -----
class SurrogateObjective:
    """
    Callable used by scipy.optimize.minimize.
    Given a smoothing parameter α returns the value and gradient of

        Fα(u) = (2·n/α)·log Σ_k exp(α·conv_k(u))

    where a = softmax(u) and conv = a*a.
    The gradient is returned w.r.t. the free variables u.
    """

    def __init__(self, alpha: float, n: int, L: int):
        self.alpha = alpha
        self.n = n
        self.L = L

    def __call__(self, u: np.ndarray):
        # a lives on the simplex automatically
        a = softmax(u)

        # full self‑convolution a*a
        conv = conv_fft(a, a, self.L)                # length 2·n‑1

        # log‑sum‑exp of α·conv (stable)
        scaled = self.alpha * conv
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        sum_exp = exp_scaled.sum()
        f = (2.0 * self.n / self.alpha) * (max_scaled + np.log(sum_exp))

        # soft‑max weights w_k = exp(α·conv_k) / Σ exp(α·conv)
        w = exp_scaled / sum_exp

        # gradient w.r.t. a : grad_a = 2·(w ⋆ a)
        a_rev = a[::-1]
        corr = conv_fft(w, a_rev, self.L)          # length 3·n‑2
        grad_a = 2.0 * corr[self.n - 1: self.n - 1 + self.n]

        # chain rule to obtain gradient w.r.t. u (soft‑max Jacobian)
        dot = np.dot(a, grad_a)
        grad_u = a * (grad_a - dot)

        return f, grad_u
