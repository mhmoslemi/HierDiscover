def _next_pow2(x):
    return 1 << ((x - 1).bit_length())

def _conv_self_fft(a):
    n = a.shape[0]
    L = _next_pow2(2*n - 1)
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return conv_full[: 2*n - 1]

def _softmax(x):
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()

def _run_sa(initial_seq, time_limit, seed):
    start = time.time()
    rng = np.random.default_rng(seed)
    a = np.asarray(initial_seq, dtype=float)
    # ensure normalized sum = 1
    sum_a = a.sum()
    if sum_a < 1e-12:
        a = np.ones_like(a) / a.size
    else:
        a = a / sum_a
    n = a.shape[0]
    conv = _conv_self_fft(a)
    max_conv = conv.max()
    best_obj = float(2.0 * n * max_conv)
    best_seq = a.copy()
    # temperature schedule
    T = best_obj * 0.1 + 1e-6
    T_min = best_obj * 1e-6 + 1e-9
    decay = 0.995
    iteration = 0
    while time.time() - start < time_limit:
        iteration += 1
        # find index of max conv
        k_max = int(np.argmax(conv))
        # compute contributions
        contrib = np.zeros_like(a)
        # vectorized compute for i where j in range
        # j = k_max - i, condition 0 <= j < n
        i_vals = np.arange(n)
        j_vals = k_max - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        contrib[mask] = a[i_vals[mask]] * a[j_vals[mask]]
        contrib_sum = contrib.sum()
        if contrib_sum > 0:
            probs = contrib / contrib_sum
            i = rng.choice(n, p=probs)
        else:
            i = rng.integers(n)
        # choose j uniformly (different)
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        # mass to move
        delta = a[i] * rng.random() * 0.05  # up to 5% of a[i]
        if delta <= 0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # small numerical fix
        if a_new[i] < 0:
            a_new[i] = 0.0
        # compute new objective via FFT
        conv_new = _conv_self_fft(a_new)
        obj_new = float(2.0 * n * conv_new.max())
        # acceptance
        if obj_new < best_obj - 1e-12:
            # improvement
            a = a_new
            conv = conv_new
            best_obj = obj_new
            best_seq = a_new.copy()
            print(f"[SA it={iteration}] new best O={best_obj:.6f}")
        else:
            # Metropolis
            if T > 0:
                prob = np.exp(-(obj_new - best_obj) / T)
                if rng.random() < prob:
                    a = a_new
                    conv = conv_new
        # anneal temperature
        if iteration % 2000 == 0:
            T = max(T * decay, T_min)
        # occasional large jitter
        if iteration % 10000 == 0:
            # add Gaussian noise to all entries, keep normalized
            noise = rng.normal(scale=0.01, size=n)
            a = a + noise
            a = np.clip(a, 0, None)
            sum_a = a.sum()
            if sum_a < 1e-12:
                a = np.ones_like(a) / n
            else:
                a = a / sum_a

"""
Non‑negative step‑function optimiser.

The algorithm proceeds in two phases:

1.  A fast Adam optimiser works on the soft‑max (log‑it) parametrisation of
    the simplex.  An annealed soft‑max approximation of the max‑convolution
    is used (β grows exponentially) and Adam’s adaptive moments give a
    stable descent.  Several support sizes are tried; the longest run gets
    most of the time budget.

2.  The best sequence is refined by a targeted simulated‑annealing
    local‑search.  At each step the index k where the convolution attains
    its maximum is identified, contributions a[i]·a[k‑i] are weighted,
    and a small amount of mass is moved from a high‑contributing index
    to a random other index.  Non‑improving moves are accepted with a
    Metropolis probability that slowly cools.  Finally negligible entries
    are pruned.

The whole routine respects the 1000 s budget and returns a plain list of
floats suitable for ``evaluate_sequence``.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return conv_full[: 2 * n - 1]


def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation g_i = Σ_t w[i+t]·a[t] for i=0,…,n‑1.
    Implemented via FFT.
    """
    n = a.shape[0]
    m = w.shape[0]                     # = 2n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.size
    conv = _conv_self_fft(a)
    return float(2.0 * n * conv.max())


def _run_opt(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
) -> tuple[list[float], float]:
    """
    Adam optimisation on the log‑it simplex.
    Returns the best sequence (list) and its exact objective.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # initialise
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
        a = a / a.sum()
    else:
        a = rng.random(n)
        a = a / a.sum()

    x = np.log(a + 1e-16)       # logits
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # schedule parameters
    beta_start = 0.5
    beta_end = 20000.0          # sharper smoothing than the previous run
    lr_start = 0.15
    lr_end = 0.0005

    best_a = a.copy()
    best_obj = _objective(best_a)

    it = 0
    while time.time() - start < time_limit:
        it += 1
        a = _softmax(x)                     # stay on the simplex

        conv = _conv_self_fft(a)
        M = conv.max()
        obj = float(2.0 * n * M)            # sum(a) = 1 ⇒ denominator = 1
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[n={n:5d} it={it:06d}] new best O={best_obj:.6f}")

        # annealing / learning‑rate schedule (time‑based)
        t = min((time.time() - start) / time_limit, 1.0)
        beta = beta_start * np.exp(np.log(beta_end / beta_start) * t)
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t)

        # soft‑max weighting for the smooth max
        if beta > 5000.0:
            kmax = int(np.argmax(conv))
            w = np.zeros_like(conv)
            w[kmax] = 1.0
        else:
            shifted = conv - M                # ≤ 0
            w = np.exp(beta * shifted)
            s = w.sum()
            if s == 0.0:
                w = np.ones_like(conv) / conv.size
            else:
                w /= s

        # gradient of (smooth) max w.r.t. a
        g_a = 2.0 * _correlation_fft(w, a)

        # natural gradient in log‑it space (mirrored gradient)
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape plateaus
        if it % 5000 == 0:
            jitter = rng.normal(scale=0.01, size=n)
            x += jitter

        # restart from current best every so often
        if it % 80000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.005, size=n)
            m.fill(0.0)
            v.fill(0.0)

    # -----------------------------------------------------------------
    # compress negligible entries (helps the final O)
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = _objective(best_a)
        print(f"[compress] reduced length → {best_a.size}, O={best_obj:.6f}")

    # -----------------------------------------------------------------
    # very short greedy local search (mass‑transfer moves) if time permits
    remaining = time_limit - (time.time() - start)
    if remaining > 0.5:
        local_start = time.time()
        a_loc = best_a.copy()
        obj_loc = best_obj
        n_loc = a_loc.size
        while time.time() - local_start < remaining * 0.3:
            i, j = rng.choice(n_loc, size=2, replace=False)
            delta = a_loc[i] * rng.random() * 0.02   # ≤2 % of a_i
            if delta <= 0.0:
                continue
            a_new = a_loc.copy()
            a_new[i] -= delta
            a_new[j] += delta
            conv_new = _conv_self_fft(a_new)
            obj_new = float(2.0 * n_loc * conv_new.max())
            if obj_new < obj_loc:
                a_loc, obj_loc = a_new, obj_new
                if obj_new < best_obj:
                    best_a, best_obj = a_new, obj_new
        print(f"[local] final local O={best_obj:.6f}")

    return best_a.tolist(), best_obj


def _run_sa(
    init_seq: list[float],
    time_limit: float,
    seed: int = 0,
) -> tuple[list[float], float]:
    """
    Targeted simulated‑annealing refinement.
    Moves small amounts of mass from positions that contribute most
    to the current maximal convolution entry.
    """
    start = time.time()
    rng = np.random.default_rng(seed)
    a = np.asarray(init_seq, dtype=float)
    if a.sum() < 1e-12:
        a = np.ones_like(a) / a.size
    else:
        a = a / a.sum()
    n = a.shape[0]

    conv = _conv_self_fft(a)
    best_obj = float(2.0 * n * conv.max())
    best_seq = a.copy()

    # temperature schedule
    T = best_obj * 0.1 + 1e-6
    T_min = best_obj * 1e-6 + 1e-9
    decay = 0.995
    it = 0

    while time.time() - start < time_limit:
        it += 1
        # index where convolution is maximal
        k_max = int(np.argmax(conv))

        # contributions a[i]·a[k_max‑i] (zero outside range)
        i_vals = np.arange(n)
        j_vals = k_max - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i_vals[mask]] * a[j_vals[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            probs = contrib / total_contrib
            i = rng.choice(n, p=probs)
        else:
            i = rng.integers(n)

        # pick a different destination index uniformly
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # transfer a modest fraction of mass from i to j
        delta = a[i] * rng.random() * 0.05   # up to 5 % of a[i]
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.clip(a_new, 0.0, None)    # safeguard against tiny negatives
        a_new /= a_new.sum()

        conv_new = _conv_self_fft(a_new)
        obj_new = float(2.0 * n * conv_new.max())

        # Metropolis acceptance
        if obj_new < best_obj - 1e-12:
            a, conv, best_obj, best_seq = a_new, conv_new, obj_new, a_new.copy()
            print(f"[SA it={it}] new best O={best_obj:.6f}")
        else:
            prob = np.exp(-(obj_new - best_obj) / T) if T > 0 else 0.0
            if rng.random() < prob:
                a, conv = a_new, conv_new

        # temperature decay
        if it % 2000 == 0:
            T = max(T * decay, T_min)

        # occasional global jitter
        if it % 10000 == 0:
            noise = rng.normal(scale=0.01, size=n)
            a = np.clip(a + noise, 0.0, None)
            a /= a.sum()
            conv = _conv_self_fft(a)

    # final compression of tiny entries
    thresh = 1e-12
    mask = best_seq > thresh
    if mask.sum() < best_seq.size:
        comp = best_seq[mask]
        comp /= comp.sum()
        comp_obj = float(2.0 * comp.size * _conv_self_fft(comp).max())
        if comp_obj < best_obj:
            best_seq, best_obj = comp, comp_obj
            print(f"[SA compress] reduced length → {comp.size}, O={best_obj:.6f}")

    return best_seq.tolist(), best_obj

    seed: int = 42,
    budget_s: int = 1000,
    **_ignore,
) -> list[float]:
    """
    Main entry point used by the evaluation harness.

    keeps the best sequence, and finally refines it with a targeted

    """

    deadline = time.time() + budget_s - 0.5   # safety margin
