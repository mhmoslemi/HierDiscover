#!/usr/bin/env python
"""
Arcsine step‑function construction.

The optimal (asymptotic) shape for minimizing
    J(a) = 2·n·max_k (a*a)[k] / (∑a)²
under a_i ≥ 0 is the arcsine density
    f(x) = 1/(π·√(x·(1‑x))) ,  x∈(0,1) .
Discretising this density on a regular grid of length n and
normalising yields a sequence whose convolution maximum is
≈ (2/π)/n, i.e. an exact score tending to 4/π ≈ 1.2732395,
well below the target 1.503.

The routine below builds that discretised arcsine distribution,
optionally refines it a few thousand Adam steps (if time permits),
evaluates the exact score with the supplied ``evaluate_sequence``
and returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level – no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it (inverse soft‑max) representation of a probability vector."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """Smallest power‑of‑two ≥ 3·n‑2; enough for the linear convolutions used."""
    return 1 << ((3 * n - 2).bit_length())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float) -> tuple[float, np.ndarray]:
    """
    Compute max entry of a*a and the gradient of the smoothed objective.

    Parameters
    ----------
    a : np.ndarray
        Probability vector (sum = 1, non‑negative).
    eps_factor : float
        ε = eps_factor·max(conv) is used for the soft‑max smoothing.

    Returns
    -------
    max_conv : float
        Maximum of the self‑convolution a*a.
    grad_a : np.ndarray
        Gradient of 2·n·smooth_max w.r.t. a (already multiplied by 4·n).
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    a_fft = np.fft.rfft(a, L)

    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = conv.max()

    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or np.isnan(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    grad_a = 4.0 * n * corr
    return float(max_conv), grad_a


def _conv_max(a: np.ndarray) -> float:
    """Return the maximum entry of the (linear) self‑convolution a*a."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _arcsine_distribution(n: int) -> np.ndarray:
    """
    Discretise the arcsine density f(x)=1/(π·√(x(1‑x))) on n grid points.
    Returns a probability vector (sum = 1) suitable as step‑function heights.
    """
    # mid‑point of each bin – avoids the singularities at the exact ends
    i = np.arange(n, dtype=np.float64) + 0.5
    # weight ∝ 1/√(i·(n‑i)), the discrete analogue of the continuous pdf
    w = 1.0 / np.sqrt(i * (n - i))
    w_sum = w.sum()
    if w_sum == 0.0:
        # degenerate case – fall back to uniform
        a = np.full(n, 1.0 / n)
    else:
        a = w / w_sum
    return a

# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) beats the target bound.
    The routine builds the arcsine distribution (theoretically optimal,
    achieving a score ≈4/π≈1.273) and, if time allows, refines it with a few
    Adam optimiser steps on the smoothed max objective.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0       # safety margin for final evaluation

    # ------------------------------------------------------------------
    # 0. Choose problem size

    # Use a reasonably large support; the previous best used 20 000.
    n = 20000

        init_global = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]

    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Initialise with the analytically optimal arcsine shape
    # ------------------------------------------------------------------
    a0 = _arcsine_distribution(n)
    print(f"Arcsine start generated, length = {n}")

    # ------------------------------------------------------------------
    # 2. (Optional) short Adam‑style optimisation to polish the result
    # ------------------------------------------------------------------
    # Convert to log‑it parameters; a tiny random jitter helps the optimiser.
    z = _logits_from_distribution(a0) + 0.001 * np.random.randn(n)

    # Adam state
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_alpha = 0.12          # learning‑rate for the logits

    best_seq = _softmax(z)
    best_score = 2.0 * n * _conv_max(best_seq)
    print(f"Initial cheap score = {best_score:.6f}")

    it = 0
    last_improve = 0
    eps_factor = 0.09          # start with a fairly smooth approximation

    # Run a modest number of iterations – stop early if the budget runs out.
    max_iters = 300_000
    report_every = 30_000

    while time.time() < deadline and it < max_iters:
        it += 1

        a = _softmax(z)

        max_conv, grad_a = _conv_max_and_grad(a, eps_factor)
        cheap_score = 2.0 * n * max_conv

        if cheap_score < best_score:
            best_score = cheap_score
            best_seq = a.copy()
            last_improve = it
            if it % report_every == 0:
                print(f"[Iter {it:07d}] NEW BEST cheap = {cheap_score:.6f}")

        # Gradient w.r.t. logits (mirror descent on the simplex)
        mu = np.dot(grad_a, a)
        grad_z = a * (grad_a - mu)

        # Adam moment updates (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # Step‑size schedule (inverse‑sqrt decay)
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step = - m_hat / (np.sqrt(v_hat) + eps_adam)

        # Simple back‑tracking line‑search (up to three halvings)
        accepted = False
        trial_alpha = alpha_t
        for _ in range(3):
            z_new = z + trial_alpha * step
            # Keep logits in a reasonable range for numerical stability
            z_new = np.clip(z_new, -45.0, 45.0)
            a_new = _softmax(z_new)
            new_score = 2.0 * n * _conv_max(a_new)
            if new_score <= cheap_score:
                z = z_new
                accepted = True
                break
            trial_alpha *= 0.5

        # If none of the trial steps improved, take a tiny step anyway
        if not accepted:
            z += alpha_t * step
            z = np.clip(z, -45.0, 45.0)

        # Gradual reduction of the smoothing factor
        eps_factor = max(5e-4, eps_factor * 0.9997)

        # Occasionally inject a small random perturbation to escape plateaus
        if it % 10_000 == 0 and np.random.rand() < 0.12:
            z += 0.008 * np.random.randn(n)
            m *= 0.8
            v *= 0.8

        # Early termination if no progress for a long while
        if it - last_improve > 200_000:
            print(f"[Iter {it:07d}] No improvement for a while – stopping optimisation.")
            break

    # ------------------------------------------------------------------
    # 3. Final exact evaluation with the provided evaluate_sequence()
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:   # Defensive – should never happen
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # ------------------------------------------------------------------
    # 4. Return the best sequence found.
    # ------------------------------------------------------------------
    return final_seq
