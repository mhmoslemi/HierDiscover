"""

# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max; result sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Return log‑it vector that yields `a` after soft‑max."""
    return np.log(a + eps)


def _fft_len_for_len(L: int) -> int:
    """Smallest power‑of‑two ≥ L (ideal size for real FFT)."""
    return 1 << ((L - 1).bit_length())


def _linconv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear (full) convolution of two real vectors using a real FFT.
    Returns an array of length len(x)+len(y)-1.
    """
    n = len(x) + len(y) - 1
    f_len = _fft_len_for_len(n)
    fx = np.fft.rfft(x, f_len)
    fy = np.fft.rfft(y, f_len)
    conv = np.fft.irfft(fx * fy, f_len)[:n]
    return conv


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    s = new_a.sum()
    if s == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= s
    return new_a


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """Normalised power‑law a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights (probability vector, sum≈1)
    that minimises the evaluate_sequence() score.  The routine runs for
    up to ~budget_s seconds and uses a soft‑max Adam optimiser with a
    smooth max‑convolution surrogate.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # safety margin

    # ---------------------------------------------------------------
    # 0. Choose the initial support size.
    # ---------------------------------------------------------------
    # If a previous construction is available, start from it,
    # otherwise fall back to a power‑law.
    init_global = globals().get("height_sequence_1", None)
    if init_global is not None and isinstance(init_global, list):
        init_arr = np.array(init_global, dtype=np.float64)
        n = max(2000, init_arr.shape[0])      # guarantee enough slots
        use_global = True
        print(f"[Init] Using supplied global sequence, length = {n}")
    else:
        init_arr = None
        n = 20000
        use_global = False
        print(f"[Init] No global sequence – starting with length = {n}")

    n_max = 30000                               # hard cap

    # ---------------------------------------------------------------
    # 1. Initialise a probability vector.
    # ---------------------------------------------------------------
    if use_global and init_arr is not None:
        a = np.maximum(init_arr, 0.0)
        s = a.sum()
        a = a / s if s > 0 else np.full_like(a, 1.0 / a.shape[0])
    else:
        # a modest power‑law is a good generic starter.
        a = _power_law_distribution(n, 0.60)
        print("[Init] Power‑law start (p=0.60)")

    # ---------------------------------------------------------------
    # 2. Log‑it parametrisation + Adam state.
    # ---------------------------------------------------------------
    eps = 1e-16
    z = _logits_from_distribution(a, eps)

    m = np.zeros_like(z)          # first moment
    v = np.zeros_like(z)          # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    base_alpha = 0.5              # a slightly larger base learning rate

    # ---------------------------------------------------------------
    # 3. Hyper‑parameters for the smoothed max‑convolution.
    # ---------------------------------------------------------------
    theta_start = 0.5
    theta_end = 20.0
    theta_slope = (theta_end - theta_start) / max(1.0, budget_s - 10.0)

    # Small timers for reporting
    report_every = 5000
    best_seq = a.copy()
    best_score = float("inf")
    it = 0
    last_improve = 0

    # ---------------------------------------------------------------
    # 4. Main optimisation loop.
    # ---------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # ----- probability vector from logits ---------------------------------
        a = _softmax(z)

        # ----- full self‑convolution -----------------------------------------
        conv = _linconv_fft(a, a)               # length = 2*n-1
        max_conv = float(conv.max())

        # cheap proxy for the true objective (since sum(a)=1)
        cheap_score = 2.0 * n * max_conv
        if cheap_score < best_score:
            best_score = cheap_score
            best_seq = a.copy()
            last_improve = it
            if it % report_every == 0:
                print(f"[Iter {it:07d}] NEW BEST cheap = {cheap_score:.6f}")

        # ----- temperature for the soft‑max (smooth max) ----------------------
        elapsed = time.time() - start
        theta = min(theta_end, theta_start + theta_slope * elapsed)

        # ----- smooth weighting of convolution entries -------------------------
        # w_k ∝ exp(theta * (conv_k - max_conv))  (max subtracted for stability)
        w = np.exp(theta * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0.0:           # pathological – fall back to uniform weights
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum               # now ∑ w_k = 1

        # ----- gradient of the smoothed objective -----------------------------
        # g_i = 2 * Σ_j a_j * w_{i+j}  (linear convolution of rev(a) with w)
        rev_a = a[::-1]
        conv_rev = _linconv_fft(rev_a, w)          # length = 3*n-2
        start_idx = n - 1
        g = 2.0 * conv_rev[start_idx:start_idx + n]   # shape (n,)

        # ----- mirror‑descent conversion to logits (KL geometry) -------------
        mu = float(np.dot(g, a))
        grad_z = a * (g - mu)                     # gradient w.r.t. logits

        # ----- Adam moment updates (bias‑corrected) ---------------------------
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # ----- learning‑rate schedule (inverse‑sqrt decay) -------------------
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)

        # ----- update logits (negative Adam step) ----------------------------
        step = - m_hat / (np.sqrt(v_hat) + eps_adam)
        z = z + alpha_t * step
        np.clip(z, -80.0, 80.0, out=z)            # keep numbers in a safe range

        # ----- occasional random perturbations --------------------------------
        if it % 4000 == 0 and np.random.rand() < 0.20:
            perturb = 0.02 * np.random.randn(n)
            z += perturb
            m *= 0.5
            v *= 0.5

        # ----- progressive enlargement of the support -------------------------
        if (it % 20000 == 0) and (n < n_max) and (time.time() + 30.0 < deadline):
            new_n = min(n_max, n + 5000)
            print(f"[Iter {it:07d}] Expanding support: {n} → {new_n}")
            a = _interpolate_distribution(best_seq, new_n)
            z = _logits_from_distribution(a, eps)
            m = np.zeros_like(z)
            v = np.zeros_like(z)
            n = new_n
            # reset counters for the new problem size
            it = 0
            last_improve = 0
            # recompute best_score for the new size (using the interpolated seq)
            conv = _linconv_fft(a, a)
            best_score = 2.0 * n * float(conv.max())
            best_seq = a.copy()
            continue

        # ----- early termination if stagnation after reaching max size -----
        if n >= n_max and it - last_improve > 300_000:
            print(f"[Iter {it:07d}] No improvement for {it-last_improve} iters – stopping.")
            break

    # -----------------------------------------------------------------------
    # 5. Final exact evaluation (the platform will also evaluate)
    # -----------------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[Result] Final exact score = {final_score:.6f}")
    except Exception as exc:                     # pragma: no‑cover
        final_score = float("inf")
        print(f"[Result] Evaluation error: {exc}")

    # -----------------------------------------------------------------------
    # 6. Return the best sequence found.
    # -----------------------------------------------------------------------
    return final_seq
