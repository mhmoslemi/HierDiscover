def simplex_proj(v):
    """Project vector v onto probability simplex."""
    # Ensure v is a numpy array.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function (non‑negative sequence) that minimises
    2·n·max(conv(sequence,sequence)) / (sum(sequence)**2)

The optimiser works directly on the probability simplex:
* the vector a (∑a_i = 1, a_i ≥ 0) is represented by logits z
  (soft‑max guarantees the constraints);
* at each iteration we evaluate the full self‑convolution a∗a
  (FFT, O(n log n));
* a smooth approximation of the max‑operator – the
  log‑sum‑exp of the convolution – is used as a surrogate loss.
  Its gradient w.r.t. a is a weighted sum of the sub‑gradients of the
  individual convolution entries; the weights are a soft‑max over the
  convolution entries (temperature θ is increased linearly).
* Adam updates are performed on the logit space (mirror descent on the
  simplex);
* occasional random perturbations and a progressive increase of the
  support size (up to 30 000) help to escape plateaus.
The best probability vector found during the run is returned; the
external ``evaluate_sequence`` routine rescales it automatically.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max → probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it representation that yields `a` after soft‑max."""
    return np.log(a + eps)


def _next_pow_two(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full linear convolution of two real vectors via FFT.
    Returns an array of length len(x)+len(y)-1.
    """
    fx = np.fft.rfft(x, fft_len)
    fy = np.fft.rfft(y, fft_len)
    conv = np.fft.irfft(fx * fy, fft_len)
    return conv[: len(x) + len(y) - 1]


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Full self‑convolution a*a (length 2·len(a)‑1) using a minimal FFT length.
    """
    n = a.shape[0]
    fft_len = _next_pow_two(2 * n - 1)
    return _full_convolution(a, a, fft_len)


def _gradient_via_logsumexp(a: np.ndarray, conv: np.ndarray, theta: float) -> np.ndarray:
    """
    Gradient of the smooth max (log‑sum‑exp) of `conv` w.r.t. the probability vector `a`.

    Parameters
    ----------
    a : np.ndarray
        Current probability vector (∑a = 1, a_i ≥ 0).
    conv : np.ndarray
        Full self‑convolution a*a (length 2·n‑1).
    theta : float
        Temperature – larger values concentrate the weight on the absolute maximum.

    Returns
    -------
    g : np.ndarray
        Weighted sub‑gradient of size n.
    """
    # soft‑max over the convolution entries (numerically stable)
    max_c = conv.max()
    exps = np.exp(theta * (conv - max_c))
    weights = exps / exps.sum()                     # shape (2·n‑1,)

    # weighted sum of the basic sub‑gradients:
    #   ∂/∂a_i (a*a)[k] = 2·a_{k-i}   (zero when index out of range)
    # The weighted sum can be written as a convolution of `weights` with the
    # reversed probability vector.
    a_rev = a[::-1]
    L = _next_pow_two(len(weights) + len(a_rev) - 1)
    grad_full = _full_convolution(weights, a_rev, L)       # length = 3·n‑2
    start = len(weights) - 1                               # index of the first relevant element
    g = 2.0 * grad_full[start:start + a.shape[0]]
    return g


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """Normalised power‑law a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    a_sum = a.sum()
    if a_sum == 0.0:
        a[:] = 1.0 / n
    else:
        a /= a_sum
    return a


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    s = new_a.sum()
    if s == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= s
    return new_a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) beats the target bound.
    The routine respects the given time budget (≈1000 s) and uses an Adam optimiser
    on the simplex with a full‑soft‑max smoothing of the max‑convolution.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0           # safety margin for final evaluation

    # ---------------------------------------------------------------
    # 0. Initialise support size (use supplied global if present)
    # ---------------------------------------------------------------
    glob = globals().get("height_sequence_1", None)
    if glob is not None:
        init_seq = np.array(glob, dtype=np.float64)
        n = max(2000, init_seq.shape[0])
        use_global = True
        print(f"[Init] Using supplied global sequence, length = {n}")
    else:
        init_seq = None
        n = 20000                                    # default starting length
        use_global = False
        print(f"[Init] No global sequence – starting with length = {n}")

    n_max = 30000                                  # hard limit on support size

    # ---------------------------------------------------------------
    # 1. Initialise a probability vector
    # ---------------------------------------------------------------
    if use_global and init_seq is not None:
        a = np.maximum(init_seq, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / a.shape[0])
        else:
            a /= s
        print("[Init] Re‑using supplied distribution (renormalised).")
    else:
        # power‑law start – works well as a generic prior
        a = _power_law_distribution(n, p=0.60)
        print("[Init] Power‑law start with exponent p=0.60")

    # ---------------------------------------------------------------
    # 2. Log‑it parametrisation + Adam state
    # ---------------------------------------------------------------
    eps = 1e-16
    z = _logits_from_distribution(a, eps)

    m = np.zeros_like(z)                # first moment
    v = np.zeros_like(z)                # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    base_alpha = 0.30                  # base learning‑rate

    # ---------------------------------------------------------------
    # 3. Optimisation loop
    # ---------------------------------------------------------------
    best_seq = a.copy()
    best_score = float("inf")
    it = 0
    last_improve = 0
    report_interval = 2000

    # temperature schedule for the smoothed max
    theta_start = 0.2
    theta_end = 15.0
    theta_slope = (theta_end - theta_start) / max(1.0, budget_s - 10.0)

    while time.time() < deadline and it < 2_000_000:
        it += 1

        # probability vector from logits
        a = _softmax(z)

        # full self‑convolution (FFT)
        conv = _conv_self(a)

        maxb = float(conv.max())
        cheap_score = 2.0 * n * maxb

        # record best ever seen
        if cheap_score < best_score:
            best_score = cheap_score
            best_seq = a.copy()
            last_improve = it
            if it % report_interval == 0:
                print(f"[Iter {it:07d}] NEW BEST  score = {cheap_score:.6f}")

        # -----------------------------------------------------------
        # Smoothed sub‑gradient of the max‑convolution via log‑sum‑exp
        # -----------------------------------------------------------
        elapsed = time.time() - start_time
        theta = min(theta_end, theta_start + theta_slope * elapsed)

        g_a = _gradient_via_logsumexp(a, conv, theta)

        # Mirror‑descent conversion to log‑it space (KL geometry)
        mu = float(np.dot(g_a, a))
        grad_z = a * (g_a - mu)

        # Adam moments (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # direction (negative Adam step)
        step_dir = - m_hat / (np.sqrt(v_hat) + eps_adam)

        # simple inverse‑sqrt learning‑rate decay
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)

        # update logits
        z = z + alpha_t * step_dir
        np.clip(z, -80.0, 80.0, out=z)      # keep numbers in a safe range

        # -----------------------------------------------------------
        # Random perturbations (help escape plateaus)
        # -----------------------------------------------------------
        if it % 4000 == 0 and np.random.rand() < 0.30:
            perturb = 0.015 * np.random.randn(n)
            z += perturb
            m *= 0.5
            v *= 0.5

        # -----------------------------------------------------------
        # Progressive support enlargement (if time permits)
        # -----------------------------------------------------------
        if (it % 30000 == 0) and (n < n_max) and (time.time() + 30.0 < deadline):
            new_n = min(n_max, n + 5000)
            print(f"[Iter {it:07d}] Expanding support: {n} → {new_n}")
            a = _interpolate_distribution(best_seq, new_n)
            z = _logits_from_distribution(a, eps)
            m = np.zeros_like(z)
            v = np.zeros_like(z)
            n = new_n
            it = 0
            last_improve = 0
            # recompute best score after enlargement
            conv = _conv_self(a)
            best_score = 2.0 * n * conv.max()
            best_seq = a.copy()
            continue

        # -----------------------------------------------------------
        # Early stop if stagnation after reaching maximum size
        # -----------------------------------------------------------
        if n >= n_max and it - last_improve > 400_000:
            print(f"[Iter {it:07d}] No improvement for {it-last_improve} iters – terminating.")
            break

    # ---------------------------------------------------------------
    # 4. Final exact evaluation (the platform will also evaluate)
    # ---------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[Result] Final exact score = {final_score:.6f}")
    except Exception as exc:
        final_score = float("inf")
        print(f"[Result] Evaluation error: {exc}")

    # ---------------------------------------------------------------
    # 5. Return the best sequence found.
    # ---------------------------------------------------------------
    return final_seq
