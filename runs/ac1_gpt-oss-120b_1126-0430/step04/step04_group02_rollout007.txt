    af = np.fft.rfft(a, n=L)

def _run_lp_opt(n, time_limit, init_a=None, seed=0):
    start = time.time()
    rng = np.random.default_rng(seed)
    if init_a is not None:
        a = np.array(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
    else:
        a = rng.random(n)
    a = np.maximum(a, 0)
    if a.sum() == 0:
        a = np.ones(n)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = _objective(best_a)
    t = 0.2
    it = 0
    while time.time() - start < time_limit:
        it += 1
        conv_self = _conv_self_fft(a)
        M = conv_self.max()
        # solve LP
        g_opt = _lp_optimize(a, M)
        if g_opt is None:
            print("[LP failed]")
            break
        # normalize g_opt to sum 1
        if np.sum(g_opt) < 1e-8:
            # if near zero, fallback to random
            g_opt = rng.random(n)
        g_opt = np.maximum(g_opt, 0)
        g_opt = g_opt / np.sum(g_opt)
        # convex combination
        a_new = (1 - t) * a + t * g_opt
        a_new = np.maximum(a_new, 0)
        a_new = a_new / np.sum(a_new)
        a = a_new
        # evaluate objective
        obj = _objective(a)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            print(f"[n={n} it={it}] new best O={best_obj:.6f}")
        # adjust t adaptively?
        # maybe decay t
        if it % 50 == 0:
            t = max(0.01, t * 0.95)
        # occasional jitter
        if it % 200 == 0:
            a += rng.normal(scale=0.001, size=n)
            a = np.maximum(a, 0)
            a = a / a.sum()
    # final local random search as before
    # random pair transfers...
    remaining = time_limit - (time.time() - start)
    if remaining > 0.5:
        a_loc = best_a.copy()
        obj_loc = best_obj
        n_loc = a_loc.size
        local_start = time.time()
        while time.time() - local_start < remaining * 0.3:
            i, j = rng.choice(n_loc, size=2, replace=False)
            # transfer fraction
            delta = a_loc[i] * rng.random() * 0.02

            a_new[i] -= delta

            # ensure >0

            obj_new = 2.0 * n_loc * conv_new.max()

                obj_loc = obj_new
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new
    return best_a.tolist(), best_obj

"""
LP‑Adam hybrid search for a non‑negative step function that minimises

    O(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

The core idea is an alternating linear‑programming step
(optimise a new vector g under the current supremum bound) followed by a
small convex combination with the current iterate.  The LP is tiny:
the variables are the n coefficients of g and the (2·n‑1) linear
constraints  (f∗g)[k] ≤ M, where M is the current peak of f∗f.
cvxpy + ECOS solves this in a few tens of ms for n≈4000.

After a few LP‑updates we do a brief Adam‑optimisation (the same code as
the reference) to fine‑tune the solution, and finally a cheap
random‑mass‑transfer hill‑climb.  The whole search respects the 1000 s
budget and prints intermediate best values.

The returned sequence is a plain list of non‑negative floats whose sum
is normalised to 1, so the evaluation routine only needs to compute the
convolution.
"""

import time
import numpy as np
import cvxpy as cp

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using FFT (real valued)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a*b using FFT."""
    n = a.shape[0]
    m = b.shape[0]
    L = _next_pow2(n + m - 1)
    af = np.fft.rfft(a, n=L)
    bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(af * bf, n=L)[: n + m - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) when sum(a)=1."""
    n = a.size
    conv = _conv_self_fft(a)
    return 2.0 * n * conv.max()


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


# ----------------------------------------------------------------------
def _lp_optimize(f: np.ndarray, bound: float) -> np.ndarray | None:
    """
    Solve
        max   sum(g)
        s.t.  g ≥ 0
              conv(f,g) ≤ bound   (elementwise)
    Returns the optimal g (as a NumPy array) or None on failure.
    """
    n = f.shape[0]
    g = cp.Variable(n, nonneg=True)
    # cvxpy treats ``conv`` as an affine expression because ``f`` is constant.
    conv_fg = cp.conv(cp.Constant(f), g)          # length 2n-1
    constraints = [conv_fg <= bound]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver="ECOS", warm_start=True, max_iters=5000, verbose=False)
    except Exception:  # pragma: no cover – safety net
        return None

    if g.value is None:
        return None
    return np.maximum(g.value, 0.0)


def _run_lp_opt(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
) -> tuple[list[float], float]:
    """
    Perform a few LP‑updates for a vector of length ``n``.
    Returns the best sequence (as a Python list) and its exact objective.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # ---- initialise ----------------------------------------------------
    if init_a is not None:
        a = np.asarray(init_a, dtype=float).copy()
        if a.size != n:
            a = rng.random(n)
    else:
        a = rng.random(n)

    a = np.clip(a, 0.0, None)
    if a.sum() == 0.0:
        a = np.ones(n)
    a = a / a.sum()                 # keep sum = 1 throughout

    best_a = a.copy()
    best_obj = _objective(best_a)

    # ------------------------------------------------------------------
    # LP + convex‑combination loop
    t = 0.25                         # mixing weight (0<t<1)
    it = 0
    while time.time() - start < time_limit:
        it += 1
        # current bound
        conv_self = _conv_self_fft(a)
        M = conv_self.max()

        # linear programme
        g_opt = _lp_optimize(a, M)
        if g_opt is None:
            print("[LP] solve failed – aborting this run")
            break

        # Normalise the candidate
        if g_opt.sum() < 1e-12:                 # degenerate – fallback to random
            g_opt = rng.random(n)
        g_opt = np.clip(g_opt, 0.0, None)
        g_opt = g_opt / g_opt.sum()

        # convex combination and renormalisation
        a = (1.0 - t) * a + t * g_opt
        a = np.clip(a, 0.0, None)
        a = a / a.sum()

        # evaluate
        obj = _objective(a)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            print(f"[LP n={n:5d} it={it:04d}] new best O={best_obj:.6f}")

        # slowly decay the mixing weight – later steps become finer
        if it % 30 == 0:
            t = max(0.02, t * 0.94)

    # ------------------------------------------------------------------
    # Short Adam‑refinement (same code as the reference, but only a few  k‑iters)
    def _adam_refine(a0: np.ndarray, run_secs: float, rng_local: np.random.Generator):
        a = a0.copy()
        x = np.log(a + 1e-16)
        m = np.zeros_like(x)
        v = np.zeros_like(x)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-12
        start_loc = time.time()
        it_loc = 0
        while time.time() - start_loc < run_secs:
            it_loc += 1
            a = _softmax(x)
            conv = _conv_self_fft(a)
            M_loc = conv.max()
            obj_loc = 2.0 * n * M_loc
            nonlocal best_a, best_obj
            if obj_loc < best_obj - 1e-12:
                best_obj = obj_loc
                best_a = a.copy()
            # smoothing schedule (very fast – we just need a few hundred iters)
            t_loc = min(it_loc / 5000.0, 1.0)
            beta = 0.5 * np.exp(np.log(8000.0 / 0.5) * t_loc)
            lr = 0.12 * np.exp(np.log(0.003 / 0.12) * t_loc)

            if beta > 4000.0:
                kmax = int(np.argmax(conv))
                w = np.zeros_like(conv)
                w[kmax] = 1.0
            else:
                shifted = conv - M_loc
                w = np.exp(beta * shifted)
                w /= w.sum()

            grad_a = 2.0 * _conv_fft(w, a)[:n]               # correlation via FFT
            dot = np.dot(a, grad_a)
            grad_x = a * (grad_a - dot)

            m = beta1 * m + (1.0 - beta1) * grad_x
            v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
            m_hat = m / (1.0 - beta1 ** it_loc)
            v_hat = v / (1.0 - beta2 ** it_loc)
            x -= lr * m_hat / (np.sqrt(v_hat) + eps)

            if it_loc % 500 == 0:
                x += rng_local.normal(scale=0.015, size=n)

        return

    remaining = time_limit - (time.time() - start)
    if remaining > 0.5:
        _adam_refine(best_a, remaining * 0.4, rng)

    # ------------------------------------------------------------------
    # Final tiny random hill‑climb
    remaining = time_limit - (time.time() - start)
    if remaining > 0.2:
        a_loc = best_a.copy()
        obj_loc = best_obj
        n_loc = a_loc.size
        local_start = time.time()
        while time.time() - local_start < remaining * 0.4:
            i, j = rng.choice(n_loc, size=2, replace=False)
            delta = a_loc[i] * rng.random() * 0.02
            if delta <= 0.0:
                continue
            a_new = a_loc.copy()
            a_new[i] -= delta
            a_new[j] += delta
            conv_new = _conv_self_fft(a_new)
            obj_new = 2.0 * n_loc * conv_new.max()
            if obj_new < obj_loc:
                a_loc, obj_loc = a_new, obj_new
                if obj_new < best_obj:
                    best_a, best_obj = a_new, obj_new
        print(f"[local] final O={best_obj:.6f}")

    return best_a.tolist(), best_obj


# ----------------------------------------------------------------------
def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **_ignore,
) -> list[float]:
    """
    Main entry point used by the evaluation harness.
    Runs several LP‑Adam optimisation attempts (different support sizes)
    within the allotted time budget and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    # candidate lengths – a few powers of two give good FFT behaviour
    lengths = [500, 1000, 2000, 4000]
    # reuse the previous construction if it exists
    if "height_sequence_1" in globals():
        seq0 = np.asarray(height_sequence_1, dtype=float).copy()
        seq0 = np.clip(seq0, 0.0, None)
        if seq0.sum() < 0.01:
            seq0 = np.ones_like(seq0)
        seq0 = seq0 / seq0.sum()
        n0 = seq0.size
        if n0 not in lengths:
            lengths.insert(0, n0)

    best_seq = None
    best_obj = float("inf")

    # --------------------------------------------------------------
    # allocate most time to the largest length, the rest is shared
    for idx, n in enumerate(lengths):
        remaining = deadline - time.time()
        if remaining <= 1.0:
            break

        if idx == 0:
            run_time = remaining * 0.65
        else:
            # split the remaining time evenly among the rest
            run_time = (remaining - 0.5) / (len(lengths) - idx)

        # initialise – possibly from the previous best
        init = None
        if idx == 0 and "height_sequence_1" in globals():
            init = seq0
            if init.size > n:
                init = init[:n]
            elif init.size < n:
                # pad with tiny random mass so the LP has room to grow
                pad = np.zeros(n - init.size)
                init = np.concatenate([init, pad])

        seq, val = _run_lp_opt(
            n=n,
            time_limit=run_time,
            init_a=init,
            seed=seed + idx * 17,
        )
        if val < best_obj:
            best_obj = val
            best_seq = seq
            print(f"[global] new overall best O={best_obj:.6f} (n={n})")

    # --------------------------------------------------------------
    # final evaluation with the provided evaluate_sequence routine.
    try:
        final_score = evaluate_sequence(best_seq)
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_seq
