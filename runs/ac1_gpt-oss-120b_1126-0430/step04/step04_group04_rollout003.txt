#!/usr/bin/env python
"""
Projected Adam optimisation of the maximum autocorrelation.

The algorithm works as follows:

1. Choose a length `n` (use a previous construction if present, otherwise a
   moderate size ≈ 10 000).  Initialise the vector with a few cheap families
   (uniform, power‑law, exponential) and keep the best one w.r.t.
   `2·n·max(a*a)` (the objective for a normalised sequence).

2. Repeatedly:
   * compute the full linear convolution `c = a*a` via a real FFT;
   * locate all shifts where `c` attains the current maximum (active set);
   * build the sub‑gradient of the max‑convolution (for a shift `k` the
     gradient is `2·a_{k‑i}` on the overlapping indices);
   * perform a projected Adam step on the simplex (the simplex projection
     preserves non‑negativity and the sum‑to‑one constraint);
   * keep the best vector seen so far; if no improvement occurs for a long
     while, inject a small random perturbation and restart the Adam moments.

3. When the time budget (minus a safety margin) is exhausted evaluate the
   best candidate with the provided `evaluate_sequence` and return it.

The code respects the 1000 s budget, works on a single thread (or up to two
if NumPy uses BLAS) and never accesses the filesystem or network.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        # all non‑positive – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using real FFT (returns length 2·n‑1)."""
    # real FFT is faster for real data
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def _cheap_score(a: np.ndarray, n: int, fft_len: int) -> float:
    """Fast evaluation of 2·n·max(a*a) for a vector already normalised (sum=1)."""
    conv = _conv_full_fft(a, fft_len)
    return 2.0 * n * conv.max()


def _power_law_sequence(n: int, p: float) -> np.ndarray:
    """Create a normalised decreasing power‑law a_i ∝ (i+1)⁻ᵖ (Σa=1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = i ** (-p)
    seq /= seq.sum()
    return seq


def _exponential_sequence(n: int, r: float) -> np.ndarray:
    """Create a normalised geometric sequence a_i ∝ rⁱ (0<r<1, Σa=1)."""
    i = np.arange(n, dtype=np.float64)
    seq = r ** i
    seq /= seq.sum()
    return seq


def _subgradient_max_conv(a: np.ndarray, active_shifts: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)[k] for the active shift set.
    For each active shift k we add 2·a_{k‑i} on the overlapping indices.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in active_shifts:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g[idx] += 2.0 * a[k - idx]
    return g


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function minimising ``evaluate_sequence``.
    The routine adapts to the supplied time budget; it always returns a
    feasible sequence (non‑negative, sum > 0).
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a few seconds for the final exact evaluation
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # 0. Choose a working length `n`
    # ------------------------------------------------------------------
    try:
        prev = height_sequence_1          # noqa: F821 (provided globally if any)
        prev_arr = np.asarray(prev, dtype=np.float64)
        prev_len = prev_arr.size
    except Exception:
        prev_arr = None
        prev_len = 0

    # target range 8k … 20k – large enough for refinement but still cheap
    n = max(prev_len, 8000)
    n = min(n, 20000)

    # FFT length – next power of two ≥ 2·n‑1
    fft_len = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # 1. Build a few cheap candidate families and pick the best
    # ------------------------------------------------------------------
    best_a = None
    best_val = float("inf")

    # uniform
    cand = np.full(n, 1.0 / n, dtype=np.float64)
    val = _cheap_score(cand, n, fft_len)
    best_a, best_val = cand.copy(), val
    print(f"[Init] uniform cheap score = {val:.6f}")

    # power‑law sweep
    for p in np.linspace(0.3, 2.0, 36):  # 0.3 … 2.0 in steps of 0.05
        if time.time() >= deadline:
            break
        cand = _power_law_sequence(n, p)
        val = _cheap_score(cand, n, fft_len)
        if val < best_val:
            best_a, best_val = cand.copy(), val
            print(f"[Init] power‑law p={p:.3f} cheap = {val:.6f}")

    # exponential sweep (geometric series)
    for r in np.linspace(0.90, 0.9999, 30):
        if time.time() >= deadline:
            break
        cand = _exponential_sequence(n, r)
        val = _cheap_score(cand, n, fft_len)
        if val < best_val:
            best_a, best_val = cand.copy(), val
            print(f"[Init] exponential r={r:.5f} cheap = {val:.6f}")

    # take previous construction if present (interpolate to current length)
    if prev_arr is not None and prev_len > 0:
        # linear interpolation onto the new grid
        x_old = np.linspace(0.0, 1.0, prev_len, endpoint=False)
        x_new = np.linspace(0.0, 1.0, n, endpoint=False)
        cand = np.interp(x_new, x_old, prev_arr)
        cand = np.maximum(cand, 0.0)
        if cand.sum() > 0.0:
            cand /= cand.sum()
            val = _cheap_score(cand, n, fft_len)
            if val < best_val:
                best_a, best_val = cand.copy(), val
                print(f"[Init] up‑scaled previous cheap = {val:.6f}")

    print(f"[Init] best cheap score after seeding = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. Projected Adam optimisation of the max‑convolution
    # ------------------------------------------------------------------
    a = best_a.copy()
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.90
    beta2 = 0.999
    eps = 1e-12
    base_lr = 0.8     # initial learning‑rate factor
    iter_cnt = 0
    last_improve = 0
    best_iter_a = a.copy()
    best_iter_val = best_val

    # stagnation parameters
    stagnation_iters = 120_000

    while time.time() < deadline:
        iter_cnt += 1

        # ---- convolution and cheap score ----
        conv = _conv_full_fft(a, fft_len)
        maxb = conv.max()
        cheap = 2.0 * n * maxb

        # Keep the best seen so far
        if cheap < best_iter_val - 1e-12:
            best_iter_val = cheap
            best_iter_a = a.copy()
            last_improve = iter_cnt
            print(f"[Iter {iter_cnt:07d}] NEW BEST cheap = {cheap:.6f}")

        # ---- sub‑gradient of the active max shift(s) ----
        # tolerance to collect all shifts that are (numerically) maximal
        tol = 1e-12
        active_shifts = np.where(conv >= maxb - tol)[0]
        g = _subgradient_max_conv(a, active_shifts)

        # ---- Adam update (projected) ----
        lr = base_lr / math.sqrt(iter_cnt)          # diminishing step size
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        # bias‑correction
        m_hat = m / (1.0 - beta1 ** iter_cnt)
        v_hat = v / (1.0 - beta2 ** iter_cnt)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_candidate = a - step
        a_candidate = _simplex_projection(a_candidate)
        a = a_candidate

        # ---- occasional multiplicative (exponentiated‑gradient) move ----
        if np.random.rand() < 0.07:       # ~7 % chance
            mult = np.exp(-lr * g)
            a = a * mult
            a = a / a.sum()

        # ---- restart if stuck ----
        if iter_cnt - last_improve > stagnation_iters:
            # small random perturbation of the best vector
            perturb = 0.015 * np.random.rand(n)
            a = _simplex_projection(best_iter_a + perturb)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = iter_cnt
            print(f"[Iter {iter_cnt:07d}] Restart after stagnation (perturb).")

    # ------------------------------------------------------------------
    # 3. Final exact evaluation (uses the provided ``evaluate_sequence``)
    # ------------------------------------------------------------------
    final_seq = best_iter_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:  # defensive – should never happen
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # ------------------------------------------------------------------
    # 4. Return the best sequence found.
    # ------------------------------------------------------------------
    return final_seq
