def propose_candidate(seed=..., budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5

    # Determine base length
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)
        n0 = len(base_seq)
    except Exception:
        base_seq = None
        n0 = None

    # Choose n
    n = max(5000, n0 if n0 is not None else 8000)
    n = min(n, 25000)

    # Prepare base values
    if base_seq is not None and base_seq.size >= n:
        values = base_seq[:n].copy()
        values = np.maximum(values, 0)
    else:
        # draw random Dirichlet
        conc = 0.1
        g = np.random.gamma(conc, 1.0, size=n)
        values = g / g.sum()

    # Normalize
    values = normalize(values)
    # Clip to [0,1000] not needed.

    # Setup FFT length
    fft_len = 1 << ((2 * n - 1).bit_length())

    # Cheap score function
    def cheap_score(arr):
        maxb = conv_max(arr, fft_len)
        return 2.0 * n * maxb

    # Evaluate base
    best_seq = values.copy()
    best_score = cheap_score(best_seq)
    print(f"[Init] n={n}, base cheap score={best_score:.6f}")

    # Perform random permutation batches
    def random_permutations(arr, trials):
        nonlocal best_seq, best_score
        for _ in range(trials):
            perm = np.random.permutation(n)
            cand = arr[perm]
            score = cheap_score(cand)
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                print(f"[Perm] New best cheap={best_score:.6f}")

    # Hill climbing swaps
    def hill_climb(start_arr, steps, temperature0):
        nonlocal best_seq, best_score
        cur = start_arr.copy()
        cur_score = cheap_score(cur)
        t0 = temperature0
        for step in range(steps):
            # time check
            if time.time() >= deadline:
                break
            # propose swap
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            cand = cur.copy()
            cand[i], cand[j] = cand[j], cand[i]
            cand_score = cheap_score(cand)
            if cand_score < cur_score:
                cur, cur_score = cand, cand_score
                if cand_score < best_score:
                    best_score = cand_score
                    best_seq = cand.copy()
                    print(f"[Hill] New global best cheap={best_score:.6f} (step {step})")
            else:
                # accept with probability
                T = t0 * (1 - step/steps)
                if T < 1e-12:
                    T = 1e-12
                prob = math.exp(-(cand_score - cur_score) / T)
                if np.random.rand() < prob:
                    cur, cur_score = cand, cand_score

    # Main loop
    # We'll allocate budget: initial random permutations, then hill climbs interleaved.
    # e.g., while time remains > 20 sec: 
    #   perform random_permutations on best_seq (or original values)
    #   hill_climb starting from best_seq for some steps (like 20000)
    #   maybe reinitialize with new random permutation of base values

    while time.time() < deadline:
        remaining = deadline - time.time()
        # If enough time for a batch of permutations
        if remaining > 5:
            # Estimate about 0.001 sec per conv, so 2000 trials ~ 2 sec
            trials = int(min(5000, max(100, remaining * 1000))) # adjust
            random_permutations(best_seq, trials)
        if time.time() >= deadline:
            break
        # Hill climb
        steps = int(min(50000, max(1000, remaining * 200))) # steps per second estimate = 200
        hill_climb(best_seq, steps, temperature0=1e-3)  # small temperature for fine search
        # maybe restart with a fresh random permutation after some iterations
        # check time
        if time.time() >= deadline:
            break

    # final evaluation
    final_seq = best_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as e:
        final_score = float('inf')
        print(f"Evaluation error: {e}")

    return final_seq

def _simplex_projection(v):
    """Project onto probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]  # descending
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # Find rho = max i such that u_i - (cssv_i)/i >0.
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non-positive
        w = np.full_like(v, 1.0/n)
        return w
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

Understanding of gradient: The gradient of conv[k] w.r.t a_i = a_{k-i} + a_{k-i}? Actually (a*a)[k] = Σ_{i} a_i a_{k-i}. Partial derivative w.r.t a_i appears twice: from term a_i a_{k-i} when i appears as first factor, plus term a_{k-i} a_i when i appears as second factor. But these are the same term, so gradient is: ∂/∂ a_i (Σ_j a_j a_{k-j}) = a_{k-i} + a_{i - k}? Not exactly. In discrete convolution, for given shift k, the term a_i a_{k-i} exists only if 0 ≤ i < n and 0 ≤ k-i < n. So derivative w.r.t a_i picks contributions where a_i appears in either factor: appears in term a_i * a_{k-i} (if k-i valid) as first factor; also appears in term a_{k-i} * a_i (when i = k - (some index)). But noting that sum includes both i and j symmetrical i↔j, but there is double counting if we treat both i and j contributions. def _topk_subgradient(a, conv, k=5):
    n = a.shape[0]
    # indices of top k conv values
    if k >= conv.size:
        top_indices = np.arange(conv.size)
    else:
        top_indices = np.argpartition(conv, -k)[-k:]
    g = np.zeros_like(a)
    for idx in top_indices:
        lo = max(0, idx - (n - 1))
        hi = min(idx, n - 1)
        rng = np.arange(lo, hi + 1, dtype=np.int64)
        # a[idx - rng] corresponds to a_{k-i} where i=rng
        g[rng] += 2.0 * a[idx - rng]
    g /= float(k)
    return g

#!/usr/bin/env python
"""
Propose a non‑negative step function that minimises

        2·n·max_k (a*a)[k] / (Σ a_i)²

The algorithm combines three ideas:

1. **Starting pool** – a few Dirichlet‑drawn vectors and, if
   available, the previously found ``height_sequence_1``.  All vectors are
   normalised to sum‑one (the objective is scale‑invariant).

2. **Permutation search** – for each base vector we repeatedly randomise
   its ordering and improve it with a simulated‑annealing hill‑climber
   that swaps two positions or applies a small multiplicative perturbation.
   The cheap surrogate score is ``2·n·max(conv)`` where the convolution
   is obtained by an FFT (≈ O(n log n) per evaluation).

3. **Fine‑grained refinement** – after the permutation phase the current
   best candidate is polished by a few projected sub‑gradient steps that
   use the gradients of the *k* largest convolution entries (the same
   “top‑k’’ idea used in the reference code).

The whole search respects a user supplied time‑budget (default 1000 s)
and always returns a feasible sequence (non‑negative, sum>0).  Progress
and intermediate cheap scores are printed for diagnostics.

The code is completely self‑contained – only ``numpy`` and the already
imported ``evaluate_sequence`` are required.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – **must be top level** (no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the probability simplex
    {x ≥ 0, Σ x = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending sort
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)                # 1‑based index
    cond = u - cssv / ind > 0
    if not np.any(cond):                     # all non‑positive => uniform
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full linear convolution a*a using a real FFT.
    Returns an array of length 2·len(a)‑1.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _conv_max_fft(a: np.ndarray, fft_len: int) -> float:
    """
    Return max_k (a*a)[k] using an FFT.
    """
    conv = _conv_full_fft(a, fft_len)
    return float(conv.max())


def _cheap_score(a: np.ndarray, n: int, fft_len: int) -> float:
    """
    Cheap surrogate of the true objective: 2·n·max(conv)  (assumes Σa_i=1).
    """
    return 2.0 * n * _conv_max_fft(a, fft_len)


def _topk_subgradient(a: np.ndarray, conv: np.ndarray, k: int = 5) -> np.ndarray:
    """
    Build a sub‑gradient from the k largest convolution entries.
    For each selected shift s we add the gradient 2·a_{s‑i}.
    The result is the average over the k contributions.
    """
    n = a.shape[0]
    if k >= conv.size:
        top_idx = np.arange(conv.size)
    else:
        top_idx = np.argpartition(conv, -k)[-k:]
    g = np.zeros_like(a)
    for idx in top_idx:
        lo = max(0, idx - (n - 1))
        hi = min(idx, n - 1)
        rng = np.arange(lo, hi + 1, dtype=np.int64)
        g[rng] += 2.0 * a[idx - rng]
    g /= float(k)
    return g


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The routine adapts its schedule to the available time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a safety margin for the final exact evaluation
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # 1. Choose a working length and a pool of base vectors.
    # --------------------------------------------------------------
    try:
        # Use the previously discovered construction if it exists.
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore
        prev_len = prev.size
    except Exception:
        prev = None
        prev_len = None

    # guaranteed range 5 k … 25 k
    n = max(prev_len if prev_len is not None else 5000, 5000)
    n = min(max(n, 5000), 25000)

    # create a pool of starting distributions (Dirichlet draws + possible prev)
    base_pool = []

    if prev is not None and prev_len >= n:
        base = prev[:n].copy()
        base = np.maximum(base, 0.0)
        base /= max(base.sum(), 1e-30)
        base_pool.append(base)

    # add a few random Dirichlet vectors with a wide range of concentrations
    for _ in range(4):
        conc = 10 ** np.random.uniform(-3.0, 0.5)   # ~0.001 … 3.2
        g = np.random.gamma(conc, 1.0, size=n)
        v = g / max(g.sum(), 1e-30)
        base_pool.append(v)

    # --------------------------------------------------------------
    # 2. Pre‑compute FFT length and helper closures.
    # --------------------------------------------------------------
    fft_len = 1 << ((2 * n - 1).bit_length())    # next power of two ≥ 2·n‑1

    # --------------------------------------------------------------
    # 3. Evaluate the very first candidate (the first element of the pool).
    # --------------------------------------------------------------
    best_seq = base_pool[0].copy()
    best_score = _cheap_score(best_seq, n, fft_len)
    print(f"[Init] n={n}, initial cheap score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Core search: repeated random permutations + simulated‑annealing
    #    hill‑climbing, followed by a short sub‑gradient polishing.
    # --------------------------------------------------------------
    iteration = 0
    while time.time() < deadline:
        # ---- pick a base vector and randomise its ordering -----------------
        base_vec = base_pool[np.random.randint(len(base_pool))]
        cur_seq = base_vec[np.random.permutation(n)].copy()
        cur_score = _cheap_score(cur_seq, n, fft_len)

        # ---- determine how many inner steps we can afford ------------------
        time_left = deadline - time.time()
        # Rough budget: ≈200 cheap evaluations per second
        steps = int(min(20000, max(500, time_left * 200)))
        temperature0 = 5e-3            # initial temperature for SA
        p_swap = 0.70                  # probability of a pure swap move
        topk_grad = 5                  # used in the optional gradient polish

        # ---- simulated‑annealing hill‑climber -------------------------------
        for step in range(steps):
            if time.time() >= deadline:
                break

            # propose a neighbour
            if np.random.rand() < p_swap:
                # swap two positions
                i, j = np.random.randint(0, n, size=2)
                cand_seq = cur_seq.copy()
                cand_seq[i], cand_seq[j] = cand_seq[j], cand_seq[i]
            else:
                # multiplicative perturbation (keeps the simplex automatically)
                eps = 0.02
                mult = np.exp(eps * np.random.randn(n))
                cand_seq = cur_seq * mult
                cand_seq /= max(cand_seq.sum(), 1e-30)

            cand_score = _cheap_score(cand_seq, n, fft_len)

            # acceptance rule
            if cand_score < cur_score:
                cur_seq, cur_score = cand_seq, cand_score
                # global best?
                if cand_score < best_score:
                    best_score = cand_score
                    best_seq = cand_seq.copy()
                    elapsed = time.time() - start_time
                    print(f"[Global] New best cheap = {best_score:.6f} (t={elapsed:.1f}s)")
            else:
                # simulated‑annealing acceptance of worse moves
                T = temperature0 * (1.0 - step / steps)
                T = max(T, 1e-12)
                prob = math.exp(-(cand_score - cur_score) / T)
                if np.random.rand() < prob:
                    cur_seq, cur_score = cand_seq, cand_score

        # ------------------------------------------------------------------
        # 5. Optional fine‑grained sub‑gradient polishing of the best local
        #    candidate obtained in the previous hill‑climb.
        # ------------------------------------------------------------------
        if time.time() < deadline:
            a = cur_seq.copy()
            momentum = np.zeros_like(a)
            eta0 = 0.4
            mu = 0.85
            # Allocate a few thousand gradient steps (≈200 per second)
            grad_steps = int(min(3000, max(500, (deadline - time.time()) * 200)))
            for it in range(grad_steps):
                if time.time() >= deadline:
                    break
                conv_full = _conv_full_fft(a, fft_len)
                g = _topk_subgradient(a, conv_full, k=topk_grad)
                eta = eta0 / math.sqrt(it + 1.0)
                momentum = mu * momentum - eta * g
                cand = a + momentum
                cand = _simplex_projection(cand)

                # occasional exponentiated‑gradient step (10 % chance)
                if np.random.rand() < 0.10:
                    mult = np.exp(-eta * g)
                    cand = a * mult
                    cand /= max(cand.sum(), 1e-30)

                a = cand
                cheap = _cheap_score(a, n, fft_len)
                if cheap < best_score:
                    best_score = cheap
                    best_seq = a.copy()
                    elapsed = time.time() - start_time
                    print(f"[Polish] New best cheap = {best_score:.6f} (t={elapsed:.1f}s)")

        iteration += 1

    # --------------------------------------------------------------
    # 6. Final exact evaluation (uses the provided evaluate_sequence)
    # --------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:  # defensive – should never happen
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # --------------------------------------------------------------
    # 7. Return the best sequence found.
    # --------------------------------------------------------------
    return final_seq
