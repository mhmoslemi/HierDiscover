We'll need generic convolution function to compute convolution of w and reversed a. We'll create `_conv_full_fft_general(x, y, fft_len)` that returns convolution via FFT (zero padded). Use real FFT.

Implementation:

```
def _conv_full_fft_general(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    fx = np.fft.rfft(x, fft_len)
    fy = np.fft.rfft(y, fft_len)
    conv = np.fft.irfft(fx * fy, fft_len)
    # Trim to length len(x) + len(y) - 1
    return conv[:len(x) + len(y) - 1]
```

Compute conv_ws = _conv_full_fft_general(w, a_rev, fft_len2). Then g = 2 * conv_ws[(n-1):(n-1+n)].

Then compute gradient in logit space: mu = np.dot(g, a); grad_z = a * (g - mu).

#!/usr/bin/env python
"""
Propose a non‑negative step function (a list of heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

where conv is the full linear convolution.  The optimizer works on the
probability simplex (sum(a)=1) and uses an Adam‑style mirror descent with a
smooth log‑sum‑exp approximation of the max‑convolution.  Several random
initialisations (power‑law, exponential, or an interpolated previous best)
are tried within the allotted time budget and the best sequence found
is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level helpers (no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max; output sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑its that produce `a` after softmax."""
    return np.log(a + eps)


def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two ≥ 2·n‑1 (good for real FFT)."""
    return 1 << ((2 * n - 1).bit_length())


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full self‑convolution a*a using an FFT of length `fft_len`."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.size - 1]
    return conv


def _conv_full_fft_general(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of two real vectors using FFT.
    Returns length len(x)+len(y)-1.
    """
    n = len(x) + len(y) - 1
    fft_len = 1 << ((n - 1).bit_length())
    fx = np.fft.rfft(x, fft_len)
    fy = np.fft.rfft(y, fft_len)
    conv = np.fft.irfft(fx * fy, fft_len)[:n]
    return conv


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to length `new_n`."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    s = new_a.sum()
    if s == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= s
    return new_a


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) , then normalised."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _exp_distribution(n: int, c: float) -> np.ndarray:
    """a_i ∝ exp(-c * i / n) , then normalised."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-c * i / n)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a

# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) gives a low
    evaluation score.  The routine respects the given time budget (≈1000 s)
    and repeatedly runs a smoothed‑max Adam optimiser from diverse
    starting points, keeping the best sequence found.
    """

    start_time = time.time()

        conv_ = _conv_full_fft(a_vec, fft_len_)

                # also update global best
                nonlocal best_seq, best_score
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(f"[{time.time() - start_time:6.1f}s] NEW GLOBAL BEST = {best_score:.6f}")

            # ---- temperature (smooth max) ---------------------------------------
            theta = theta_start + (theta_end - theta_start) * min(1.0, it / 10000.0)

            # ---- smooth weights (log‑sum‑exp) ------------------------------------
            # subtract the max for numerical stability
            max_conv = cur_max
            exp_terms = np.exp(theta * (conv - max_conv))
            w = exp_terms / exp_terms.sum()  # shape = 2n-1, sum = 1

            # ---- gradient of smooth max -----------------------------------------
            # g = 2 * (w convolve reversed a)[n-1 : n-1+n]
            a_rev = a[::-1]
            # convolution length = (2n-1) + n - 1 = 3n-2
            conv_full = _conv_full_fft_general(w, a_rev)
            g = 2.0 * conv_full[(n - 1) : (n - 1 + n)]

            # ---- mirror‑descent conversion to logits (KL geometry) -------------
            mu = float(np.dot(g, a))
            grad_z = a * (g - mu)

            # ---- Adam moment updates (bias‑corrected) ---------------------------
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            # ---- step‑size schedule (inverse‑sqrt decay) -----------------------
            alpha_t = base_alpha / (math.sqrt(it) + 1.0)

            # ---- update logits --------------------------------------------------
            step = - alpha_t * m_hat / (np.sqrt(v_hat) + eps_adam)
            z = z + step
            np.clip(z, -80.0, 80.0, out=z)

            # ---- occasional random perturbation --------------------------------
            if it % 4000 == 0 and np.random.rand() < 0.2:
                perturb = 0.01 * np.random.randn(n)
                z += perturb
                m *= 0.5
                v *= 0.5

        return local_best_a, local_best_score

    # ------------------------------------------------------------------
    # Main search loop (multiple restarts, dynamic lengths)
    # ------------------------------------------------------------------
    support_options = [8000, 12000, 20000, 30000]

    while time.time() < deadline:
        # decide length for this run (prefer larger when time permits)
        if deadline - time.time() > 30.0:
            n = np.random.choice(support_options)
        else:
            # near the end just try a quick refinement on the current best size
            n = best_seq.shape[0] if best_seq is not None else 12000

        # build an initial distribution for this length
        init_type = np.random.choice(["power", "exp", "mix", "global"])
        if init_type == "global" and init_from_global is not None:
            a0 = _interpolate_distribution(init_from_global, n)
        elif init_type == "mix":
            # mix two exponentials with random weights
            c1 = np.random.uniform(0.5, 3.0)
            c2 = np.random.uniform(0.5, 3.0)
            w = np.random.rand()
            a_tmp = w * _exp_distribution(n, c1) + (1.0 - w) * _exp_distribution(n, c2)[::-1]
            a0 = a_tmp / a_tmp.sum()
        elif init_type == "exp":
            c = np.random.uniform(0.5, 3.0)
            a0 = _exp_distribution(n, c)
        else:   # power‑law
            p = np.random.uniform(0.3, 1.5)
            a0 = _power_law_distribution(n, p)

        # allocate a slice of the remaining time for this run
        time_for_run = min(deadline - time.time(), 12.0)  # at most ~12 s per restart
        if time_for_run < 2.0:
            break

        # run the optimiser
        _ = run_optimisation(a0, n, time.time() + time_for_run)

        # attempt a cheap up‑sampling of the current global best if we have time
        if best_seq is not None and best_seq.shape[0] < N_MAX:
            # upscale to the maximum allowed length
            new_len = N_MAX
            if time.time() + 3.0 < deadline:
                up_seq = _interpolate_distribution(best_seq, new_len)
                up_score = cheap_score_of(up_seq)
                if up_score < best_score:
                    best_score = up_score
                    best_seq = up_seq
                    print(f"[{time.time() - start_time:6.1f}s] UPSCALED BEST = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final exact evaluation (the platform will also evaluate)
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist() if best_seq is not None else [1.0]
    try:
        final_val = evaluate_sequence(final_seq)
        print(f"[Result] Final exact score = {final_val:.6f}")
    except Exception as e:
        print(f"[Result] Evaluation failed: {e}")

    return final_seq
