    n = a.shape[0]
    size = 1 << ((2*n - 2).bit_length())  # for conv a*a
    A = np.fft.fft(a, n=size)

"""
Propose a non‑negative step‑function (list of heights) that minimises

        O = 2·n·max(conv(a,a)) / (sum(a)²)

The algorithm consists of three phases:

1. **Coarse parametric sweep** – generate power‑law shapes
   a_i ∝ (i+β)⁻ᵅ (optionally multiplied by an exponential tail)
   on a fine grid and keep the best candidate.

2. **Smooth‑max mirror descent** – minimise a smooth approximation of
   the max‐convolution using exponentiated‑gradient updates.
   The smoothness parameter λ is increased geometrically (from ≈0.5
   to ≈50) so that early iterations move quickly on a smooth landscape
   while later iterations focus on the true max.

3. **Annealed random perturbations** – wander around the best solution
   with a decreasing temperature to escape shallow local minima.

All updates stay on the probability simplex (Σa = 1) by working in log‑space.
The routine respects a 1000 s budget, prints progress, and finally
returns the best sequence found as a plain ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x≥0, Σx=1} (J. Duchi et al.)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray, size: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of given size."""
    A = np.fft.fft(a, n=size)
    conv = np.fft.ifft(A * A).real[: 2 * a.size - 1]
    return conv


# ----------------------------------------------------------------------
def _objective_norm1(a: np.ndarray) -> float:
    """Exact objective for a normalised to sum = 1:
       O = 2·n·max(conv(a,a))."""
    n = a.size
    size = 1 << ((2 * n - 2).bit_length())
    conv = _conv_fft(a, size)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
def _shape_power(alpha: float, beta: float, n: int) -> np.ndarray:
    """Monotone decreasing power‑law a_i ∝ (i+β)⁻ᵅ, normalised to sum = 1."""
    i = np.arange(1, n + 1, dtype=float)
    a = (i + beta) ** (-alpha)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
def _shape_power_exp(alpha: float, beta: float, gamma: float, n: int) -> np.ndarray:
    """Power‑law with exponential tail:
       a_i ∝ (i+β)⁻ᵅ · exp(‑γ·(i‑1)/n) , normalised to sum = 1."""
    i = np.arange(1, n + 1, dtype=float)
    a = (i + beta) ** (-alpha) * np.exp(-gamma * (i - 1) / n)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, lam: float) -> tuple[float, np.ndarray]:
    """
    Returns a smooth approximation of max(conv(a,a)) and its gradient
    with respect to a (scaled for the full objective O = 2·n·max).
    The smooth max is   (1/lam)·log Σ exp(lam·conv_k).
    """
    n = a.shape[0]

    # ----- 1) compute the (linear) self‑convolution a*a -----
    size1 = 1 << ((2 * n - 2).bit_length())
    A = np.fft.fft(a, n=size1)
    conv = np.fft.ifft(A * A).real[: 2 * n - 1]

    # ----- 2) smooth max (log‑sum‑exp) -----
    scaled = lam * conv
    max_scaled = np.max(scaled)
    exps = np.exp(scaled - max_scaled)          # stability trick
    sum_exps = np.sum(exps)
    smooth_max = (np.log(sum_exps) + max_scaled) / lam

    # soft‑max weights w_k = exp(lam·conv_k) / Σ exp(lam·conv_j)
    w = exps / sum_exps          # length 2n‑1, non‑negative, Σ w = 1

    # ----- 3) gradient of smooth max:
    #   ∂smooth_max/∂a_i = 2 * Σ_j a_j w_{i+j}
    # we compute this cross‑correlation via an FFT.
    w_rev = w[::-1]                       # reversed version of w
    size2 = 1 << ((3 * n - 2).bit_length())
    A2 = np.fft.fft(a, n=size2)
    W_rev = np.fft.fft(w_rev, n=size2)
    conv_lr = np.fft.ifft(A2 * W_rev).real[: 3 * n - 2]   # linear conv a * w_rev

    # extract Σ_j a_j w_{i+j} for i = 0..n‑1
    idx = 2 * n - 2 - np.arange(n)        # mapping explained in the docstring
    cross = conv_lr[idx]                 # shape (n,)

    # gradient of the *exact* objective O = 2·n·smooth_max
    grad_O = 4.0 * n * cross              # 2·n * (2·cross) = 4·n·cross
    return smooth_max, grad_O

    else:

    best_a = a0.copy()

                    if obj < best_obj - 1e-12:
