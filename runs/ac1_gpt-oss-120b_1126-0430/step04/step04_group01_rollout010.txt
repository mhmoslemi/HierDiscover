"""
Propose a non‑negative step‑function (list of heights) that (approximately)
minimises

        O(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm works in two stages inside the available 1000 s budget.

Stage 1 – “flattening”  
We fix Σaᵢ = 1 (simplex projection) so that O = 2·n·max(conv).  
Let b = a * a (full convolution, length 2·n‑1) and
    μ = 1/(2·n‑1)  (the average value of b when Σa = 1).

The ideal situation is b ≡ μ.  We therefore minimise the quadratic penalty

        P(a) = Σₖ (bₖ – μ)²

whose gradient is

        ∇P = 4·corr(a , b‑μ)

where corr denotes the (non‑reversed) cross‑correlation.
The gradient is obtained efficiently with three FFTs per iteration.
We run an Adam optimiser on  (P + λ·∑aᵢ·log aᵢ)  (the tiny entropy term
prevents premature zero‑clipping) and project onto the probability simplex
after each step.  This drives the convolution towards a flat profile,
drastically reducing the maximum entry.

Stage 2 – stochastic local refinement  
From the best point of stage 1 we perform many cheap “mass‑transfer’’ moves:
a random fraction of the mass at position i is moved to position j.
Moves are accepted if they lower the exact objective O, otherwise with a
Metropolis probability that slowly cools down.  Every few iterations we
take a sub‑gradient step for the true max‑convolution term
(∇max conv = average of the sub‑gradients coming from all maximising
indices).  This fine‑tunes the solution and helps escape tiny plateaux.

All arithmetic respects the non‑negativity and Σa = 1 constraints, and the
final sequence is returned as a plain Python list.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – projects onto {x ≥ 0, Σx = 1}
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:                     # all entries are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution via FFT (returns the central 2·n‑1 values)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return a * a (full convolution) using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    # keep only the first 2·n‑1 entries (the genuine convolution)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Exact objective for a vector already normalised (Σa = 1)
def _objective_O(a: np.ndarray, n: int, L: int) -> float:
    """O(a) = 2·n·max(conv(a,a))  (since Σa = 1)."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) – averaged over all maximising indices
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
# Gradient of the flattening penalty  P(a) = Σ(b‑μ)²
def _penalty_gradient(a: np.ndarray, n: int, L: int, mu: float, lam: float) -> np.ndarray:
    """
    Returns ∇ ( Σ(bₖ‑μ)² + λ·∑aᵢ·log aᵢ )
    where b = a⨂a, μ = 1/(2·n‑1).
    The quadratic part yields 4·corr(a , b‑μ).
    """
    # convolution a*a
    conv = _conv_fft(a, L)                     # length 2·n‑1
    diff = conv - mu                           # b‑μ

    # cross‑correlation a ⋆ (b‑μ)  =  conv_rev[n‑1 : n‑1+n]  where conv_rev = a_rev * diff
    a_rev = a[::-1]
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    fft_diff = np.fft.rfft(diff, n=L)
    corr_rev = np.fft.irfft(fft_a_rev * fft_diff, n=L)
    grad_quad = 4.0 * corr_rev[n - 1 : n - 1 + n]   # shape (n,)

    # entropy regulariser (tiny)
    eps = 1e-20
    grad_ent = lam * (np.log(a + eps) + 1.0)

    return grad_quad + grad_ent
