"""
Propose a non‑negative step‑function (list of heights) that minimises

        2·n·max_k (a*a)[k] / (Σ a_i)²

The algorithm starts from the best known construction (or a random
Dirichlet seed), then performs a large‑scale projected Adam optimisation
on a smooth approximation of the ℓ∞‑norm of the autocorrelation.
The smooth approximation is the log‑sum‑exp of the convolution entries,
controlled by a temperature parameter α that is slowly increased so that
the optimisation first smooths the whole convolution and later focuses on
the true maximum.  Every update is projected onto the probability simplex
to keep the sequence non‑negative and normalised.  Simple restarts with a
small random perturbation are performed when the score stalls.

The whole search respects a user supplied time budget (default 1000 s)
and always returns a feasible sequence.  The final exact score is obtained
by calling the already imported ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # all non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float) -> np.ndarray:
    """Return a normalised power‑law a_i ∝ (i+1)^‑p (Σa_i = 1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = i ** (-p)
    seq /= seq.sum()
    return seq


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·n‑1) using FFT."""
    A = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(A * A))[: 2 * a.size - 1]
    return conv


def _conv_full_and_fft(a: np.ndarray, fft_len: int):
    """Return (full convolution a*a, FFT of a) for a given length."""
    A = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(A * A))[: 2 * a.size - 1]
    return conv, A


def _correlation_weighted(a: np.ndarray, w: np.ndarray, A_fft: np.ndarray,
                         fft_len: int) -> np.ndarray:
    """
    Compute g = 2 * sum_i a[i] * w[i + j]   (j = 0..n‑1)
    using FFT correlation.  Returns a vector of length n.
    """
    W = np.fft.fft(w, fft_len)
    corr = np.real(np.fft.ifft(np.conj(A_fft) * W))[: a.shape[0]]
    return 2.0 * corr
