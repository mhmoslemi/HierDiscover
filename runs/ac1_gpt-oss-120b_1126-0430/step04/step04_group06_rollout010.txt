def _full_weighted_gradient(a, conv, theta):
    # a: np.ndarray length n
    # conv: np.ndarray length 2n-1
    # compute weight vector w (length 2n-1) via softmax
    max_val = conv.max()
    exps = np.exp(theta * (conv - max_val))
    w = exps / exps.sum()
    # compute gradient: g = 2 * conv_full_fft(w, a_rev) with appropriate slice
    a_rev = a[::-1]
    # compute convolution via FFT
    # Determine fft length for conv of lengths len(w) and len(a_rev)
    L = 1 << ((len(w) + len(a_rev) - 1).bit_length())
    # compute FFT
    fw = np.fft.rfft(w, L)
    fa_rev = np.fft.rfft(a_rev, L)
    conv_w_arev = np.fft.irfft(fw * fa_rev, L)
    # slice
    start = len(w) // 2   # since len(w)=2n-1, but index n-1 is the center
    # Actually we derived start = n-1
    n = a.shape[0]
    start = n - 1  # because len(w) = 2n-1, and n-1 is the centre index
    g = 2.0 * conv_w_arev[start:start + n]
    return g

        conv_wr = np.fft.irfft(fw * fa_rev, Lw)
        start_idx = n - 1
        g_a = 2.0 * conv_wr[start_idx:start_idx + n]

        # mirror descent to logits
        mu = np.dot(g_a, a)
        grad_z = a * (g_a - mu)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * (grad_z ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step_dir = - m_hat / (np.sqrt(v_hat) + eps_adam)

        # step size
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)

        z = z + alpha_t * step_dir
        np.clip(z, -80.0, 80.0, out=z)

        # occasional random perturb
        if it % 5000 == 0 and np.random.rand() < 0.25:
            perturb = 0.02 * np.random.randn(n)
            z += perturb
            m *= 0.5
            v *= 0.5

        # early stop if no improvement for many iters
        if it - last_improve > 200000:
            break

    return best_seq, best_maxb

"""
Optimiser for the step‑function L∞‑convolution problem.

Strategy
--------
1. **Hierarchical search** – first optimise a modest‑size (≈8 k) vector for several
   power‑law exponents, then interpolate the best candidate to the full size
   (30 k).  This cheap phase explores the shape space quickly.
2. **Smooth max‑approximation** – the objective `max(conv)` is approximated by the
   log‑sum‑exp (soft‑max) of the self‑convolution.  The temperature is increased
   linearly during the run so the optimisation starts smooth and ends with an
   almost exact sub‑gradient of the maximum.
3. **Full‑gradient via FFT** – for a weight vector *w* (the soft‑max of the
   convolution) the gradient w.r.t. the probability vector `a` is
   `g = 2·(w * a_rev)` where `a_rev` is `a` reversed.  Both the self‑convolution
   and this weighted convolution are computed with real FFTs (O(n log n)).
4. **Mirror descent on the simplex** – `a` is parametrised by logits `z`
   (`a = softmax(z)`).  Adam updates are applied to `z` with an inverse‑sqrt
   learning‑rate schedule.
5. **Perturbations & early stopping** – occasional random noise helps escape
   plateaus; the loop stops if no improvement is seen for many iterations or
   the time budget expires.

The returned list is a non‑negative vector (sum≈1); `evaluate_sequence`
is scale‑invariant, so this normalisation does not affect the score.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max; returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it representation that yields `a` after soft‑max."""
    return np.log(a + eps)


def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x (x≥1)."""
    return 1 << ((x - 1).bit_length())


def _fft_len_for_n(n: int) -> int:
    """FFT length needed for a full self‑convolution of length `n`."""
    return _next_pow2(2 * n - 1)


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full linear self‑convolution a*a using an FFT of length `fft_len`.
    Returns an array of length 2·len(a)‑1.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.size - 1]
    return conv


def _full_weighted_gradient(a: np.ndarray,
                            conv: np.ndarray,
                            theta: float,
                            fft_len_w: int) -> np.ndarray:
    """
    Weighted gradient of the smoothed max (log‑sum‑exp) of `conv`.

    Parameters
    ----------
    a       : probability vector (sum = 1)
    conv    : full self‑convolution a*a (length 2·n‑1)
    theta   : smoothing parameter (large → sharper)
    fft_len_w : FFT length for the convolution w * a_rev  (≥ 3·n‑2)

    Returns
    -------
    g_a : gradient w.r.t. `a` (same shape as `a`)
    """
    n = a.shape[0]

    # soft‑max weights w_i ∝ exp(theta·(conv_i - max(conv)))
    max_val = conv.max()
    exps = np.exp(theta * (conv - max_val))
    w = exps / exps.sum()

    # convolution w * a_rev (full length)
    a_rev = a[::-1]
    fw = np.fft.rfft(w, fft_len_w)
    fa_rev = np.fft.rfft(a_rev, fft_len_w)
    conv_wr = np.fft.irfft(fw * fa_rev, fft_len_w)

    # slice the part that corresponds to each index of `a`
    start = n - 1
    g = 2.0 * conv_wr[start:start + n]
    return g


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    tot = new_a.sum()
    if tot == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= tot
    return new_a


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """Normalised power‑law a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _run_optimizer(initial_a: np.ndarray,
                   n: int,
                   max_iters: int,
                   start_time: float,
                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Optimise a probability vector of length `n` using the full weighted
    gradient described above.  Returns the best vector seen and its
    corresponding max‑convolution value.
    """
    eps = 1e-16
    a = initial_a.copy()
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    z = _logits_from_distribution(a, eps)

    # Adam state
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_alpha = 0.5

    # FFT lengths (fixed for the whole run)
    fft_len_self = _fft_len_for_n(n)                 # for a*a
    fft_len_w = _next_pow2(3 * n - 2)                # for w * a_rev

    best_seq = a.copy()
    best_maxb = float("inf")
    it = 0
    last_improve = 0

    # temperature schedule (linear in elapsed wall‑time)
    theta_start = 1.0
    theta_end = 5e6

    total_allowed = max(1e-12, deadline - start_time)

    while time.time() < deadline and it < max_iters:
        it += 1

        # current probability vector
        a = _softmax(z)

        # self‑convolution a*a
        fa = np.fft.rfft(a, fft_len_self)
        conv = np.fft.irfft(fa * fa, fft_len_self)[: 2 * n - 1]

        maxb = conv.max()
        if maxb < best_maxb:
            best_maxb = maxb
            best_seq = a.copy()
            last_improve = it

        # temperature based on elapsed wall‑time
        elapsed = time.time() - start_time
        theta = theta_start + (theta_end - theta_start) * min(1.0, elapsed / total_allowed)

        # full weighted gradient
        g_a = _full_weighted_gradient(a, conv, theta, fft_len_w)

        # mirror‑descent conversion to logits (KL geometry)
        mu = float(np.dot(g_a, a))
        grad_z = a * (g_a - mu)

        # Adam updates (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step_dir = - m_hat / (np.sqrt(v_hat) + eps_adam)

        # inverse‑sqrt step‑size schedule
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        z = z + alpha_t * step_dir
        np.clip(z, -80.0, 80.0, out=z)

        # occasional random perturbation
        if it % 5000 == 0 and np.random.rand() < 0.25:
            perturb = 0.02 * np.random.randn(n)
            z += perturb
            m *= 0.5
            v *= 0.5

        # early stop if stagnation
        if it - last_improve > 200_000:
            break

    return best_seq, best_maxb


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) beats the target bound.
    The routine respects the given time budget (≈1000 s) and uses a two‑stage
    optimisation: a cheap exploratory phase on a small support followed by a
    high‑precision phase on the full 30 000‑element support.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Stage A – cheap exploration on a smaller support
    # ------------------------------------------------------------------
    n_small = 8000
    p_candidates = [0.4, 0.55, 0.7, 0.9]
    best_small_seq = None
    best_small_maxb = float("inf")

    # If a global sequence is available, prefer it for the small phase
    if "height_sequence_1" in globals():
        seq = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if seq.size > 0:
            seq = np.maximum(seq, 0.0)
            s = seq.sum()
            if s > 0.0:
                seq /= s
            best_small_seq = seq
            best_small_maxb = None
            print("[Stage A] Using supplied global sequence as start.")
    else:
        for p in p_candidates:
            if time.time() >= deadline:
                break
            a0 = _power_law_distribution(n_small, p)
            seq, maxb = _run_optimizer(a0, n_small,
                                       max_iters=12_000,
                                       start_time=start,
                                       deadline=deadline)
            print(f"[Stage A] p={p:.2f} → max‑conv {maxb:.8e}")
            if maxb < best_small_maxb:
                best_small_maxb = maxb
                best_small_seq = seq.copy()

    # Fallback if nothing was produced (should not happen)
    if best_small_seq is None:
        best_small_seq = _power_law_distribution(n_small, 0.6)

    # ------------------------------------------------------------------
    # Stage B – full‑scale optimisation (n = 30000)
    # ------------------------------------------------------------------
    n_big = 30000
    a_big = _interpolate_distribution(best_small_seq, n_big)

    final_seq, final_maxb = _run_optimizer(
        a_big,
        n_big,
        max_iters=10 ** 9,               # effectively “run until time is up”
        start_time=start,
        deadline=deadline
    )

    final_list = final_seq.tolist()

    # ---------------------------------------------------------------
    # 4. Final exact evaluation (the platform will also evaluate)
    # ---------------------------------------------------------------
    try:
        final_score = evaluate_sequence(final_list)
        print(f"[Result] Final exact score = {final_score:.6f}")
    except Exception as exc:               # safety – platform will re‑evaluate anyway
        print(f"[Result] Evaluation error: {exc}")

    return final_list
