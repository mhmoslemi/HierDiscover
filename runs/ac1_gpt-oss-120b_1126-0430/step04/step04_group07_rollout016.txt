import time, numpy as np, math
from scipy.optimize import linprog

def _next_pow2(x):
    return 1 << (x-1).bit_length()

def _conv_self_fft(a):
    n = a.shape[0]
    L = _next_pow2(2*n-1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af*af, n=L)[:2*n-1]
    return conv

def _correlation_fft(w, a):
    n = a.shape[0]
    m = w.shape[0]
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]

def propose_candidate(seed=42, budget_s=1000):
    """Hybrid exponentiated‑gradient + active‑set linear‑programming optimizer
    for the max‑convolution bound.  Starts from the best known sequence (if
    ``height_sequence_1`` is defined), runs a mirror‑descent phase using an
    Adam‑style step‑size schedule, then repeatedly solves a tiny LP that
    maximises the total mass of a vector ``g`` under the current
    convolution bound.  The new vector is mixed with the current one; the
    process monotonically reduces the bound.  The routine respects the
    time budget and prints progress.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # -----------------------------------------------------------------
    # 1) Choose length and initialise a
    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=float)
        if a0.ndim != 1:
            a0 = a0.ravel()
        # protect against non‑positive sum after security checks
        total = a0.sum()
        if total <= 0:
            a0 = np.full_like(a0, 1.0)
            total = a0.sum()
        a0 = a0 / total   # normalise to sum 1
        n = a0.size
        u = np.log(a0 + 1e-12)   # log‑probabilities
        print(f"[init] start from height_sequence_1, n={n}")
    else:
        n = 5000
        u = np.full(n, -np.log(n))   # uniform log‑probability
        print(f"[init] start from uniform, n={n}")

    # -----------------------------------------------------------------
    # 2) Optimisation state
    best_u = u.copy()
    best_score = np.inf
    best_iter = 0

    # Adam moments
    m = np.zeros_like(u)
    v = np.zeros_like(u)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # schedules
    max_iters = 500_000
    beta_start = 2.0
    beta_end   = 30000.0
    eta_start  = 0.7
    eta_end    = 0.02
    jitter_interval = 4000
    jitter_scale = 5e-4
    restart_interval = 25000

    it = 0
    # -------------------------------------------------------------
    # Phase 1 – mirror descent with Adam
    while time.time() < deadline and it < max_iters:
        it += 1

        # (a) probability vector
        # stable soft‑max
        max_u = u.max()
        a = np.exp(u - max_u)
        a_sum = a.sum()
        a = a / a_sum

        conv = _conv_self_fft(a)
        M = conv.max()
        cur_score = 2.0 * n * M   # O = 2 n M  (since sum = 1)

            best_score = cur_score
            best_u = u.copy()
            best_iter = it
            print(f"[iter {it:07d}] new best O={best_score:.6f}")

        # (d) schedule
        frac = min(it / max_iters, 1.0)
        beta = beta_start + (beta_end - beta_start) * frac
        eta = eta_start * (eta_end / eta_start) ** frac

        if frac < 0.4:   # soft‑max mode

            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.zeros_like(conv)
                w[conv.argmax()] = 1.0
            else:
                w = w / w_sum

                w = np.zeros_like(conv)
                w[conv.argmax()] = 1.0

"""
Hybrid mirror‑descent / Adam optimiser with an active‑set linear‑program
refinement for the max‑convolution bound.

The algorithm works in two phases:

1.  An exponentiated‑gradient (mirror‑descent) optimiser on the probability
    simplex reduces the bound quickly.  We use an Adam‑style adaptive step‑size,
    a soft‑max weighting of the convolution early on and a hard‑max (active set)
    weighting later.  Periodic jitter and cheap restarts from the best‑so‑far
    solution keep the search from stagnating.

2.  When enough progress has been made we switch to a small linear programme.
    For the current vector `f` we maximise `∑ g_i` subject to the linear
    constraints `f * g ≤ M_f` only at the (few) *active* lags where the
    convolution attains its maximum `M_f`.  The resulting vector `g` is scaled
    to the same ℓ₁‑norm as `f` and mixed with `f`.  This monotone mixing
    provably does not increase the bound and often yields a noticeable drop.

The routine respects the 1000 s time budget, prints progress, evaluates the
final sequence with the provided ``evaluate_sequence`` function and returns the
best list of non‑negative coefficients found.
"""

import time
import numpy as np
from scipy.optimize import linprog


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a via a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv


def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Linear correlation  g_i = Σ_t w[i+t]·a[t]  for i = 0,…,n‑1,
    implemented via FFT.
    """
    n = a.shape[0]
    m = w.shape[0]                # = 2·n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]

        else:
            t_mix *= 0.5
            if t_mix < 1e-3:
                print("[LP] mixing weight too small – exiting LP phase")
                break

        if (deadline - time.time()) < 5:
            break

    # -----------------------------------------------------------------
    # 3) Final evaluation and return
    best_seq = np.exp(best_u - best_u.max())
    best_seq = best_seq / best_seq.sum()

    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f} "
              f"(best iteration {best_iter})")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_seq.tolist()
```
