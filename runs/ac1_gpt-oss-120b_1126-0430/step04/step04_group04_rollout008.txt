#!/usr/bin/env python
"""
Propose a non‑negative step function (list of heights) that minimises
``evaluate_sequence``.

Algorithm
---------
1. Choose a working length *n* (reuse the previous construction if
   available, otherwise 8 k–20 k elements).
2. Initialise a candidate by a cheap scan of power‑law shapes and a
   collection of random Dirichlet draws (wide range of concentration
   parameters).  Keep the best candidate according to the cheap surrogate
   ``2·n·max(conv)`` (the denominator is 1 because we keep the vector on the
   simplex).
3. Refine the best candidate with a *soft‑max* approximation of the
   ``max`` in the objective:

      • For the current vector a (∑a = 1) compute the full convolution
        c = a∗a (FFT, linear convolution).

      • Form a probability distribution over the convolution indices
        w_i ∝ exp(τ·(c_i−max(c))) with a temperature τ that is slowly
        increased from τ≈0.2 to τ≈4.0.

      • The gradient of  Σ_i w_i·c_i  w.r.t. a is
        g = 2·(a ∗ w_rev) where w_rev is w reversed.  The convolution
        is evaluated with a second FFT (length ≥3n) for O(n log n) work.

      • Update a by an exponentiated‑gradient step
        a ← a·exp(−η·g)  (η decays as 1/√t) and renormalise to the simplex.
        Very occasional additive “momentum” perturbations are added to
        avoid stagnation.

   The loop runs until the time budget is exhausted (≈1000 s).  The best
   vector seen (lowest cheap score) is remembered.

4. Finally evaluate the best vector with the provided ``evaluate_sequence``
   and return it (as a plain Python list).

The method combines a smooth global approximation of the max‑operator,
efficient FFT‑based gradients and a multiplicative update that respects
non‑negativity without an explicit projection, which improves over the
previous top‑K heavy‑ball approach.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the probability simplex {x ≥ 0, Σx = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # all non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float) -> np.ndarray:
    """
    Return a normalised power‑law a_i ∝ (i+1)^‑p  (∑a_i = 1).
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = i ** (-p)
    seq /= seq.sum()
    return seq


def _cheap_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Compute the cheap surrogate 2·n·max(conv)  for a vector a on the simplex.
    """
    A = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(A * A))
    conv = conv_full[: 2 * n - 1]
    return 2.0 * n * conv.max()

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The routine adapts its schedule to the available time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0  # safety margin for final eval

    # --------------------------------------------------------------
    # Choose a working length.
    # --------------------------------------------------------------
    try:
        prev = height_sequence_1          # noqa: F821 – may be supplied globally
        n0 = len(prev)
        prev_arr = np.asarray(prev, dtype=np.float64)
        if prev_arr.sum() > 0.0:
            prev_arr = prev_arr / prev_arr.sum()
        else:
            prev_arr = None
    except Exception:
        prev = None
        prev_arr = None
        n0 = 8000

    # Keep n modest enough for fast FFTs but large enough to give a good bound.
    n = max(min(max(n0, 6000), 20000), 6000)

    # FFT lengths (next power of two)
    fft_len = 1 << ((2 * n - 1).bit_length())          # for a*a
    grad_fft_len = 1 << ((3 * n - 2).bit_length())    # for a * w_rev

    # --------------------------------------------------------------
    # 0. Initialise candidates (power‑law scan + Dirichlet draws)
    # --------------------------------------------------------------
    best_a = None
    best_score = float("inf")

    # power‑law sweep
    for p in np.linspace(0.55, 0.80, 81):
        if time.time() >= deadline:
            break
        cand = _power_law_sequence(n, p)
        val = _cheap_score(cand, fft_len, n)
        if val < best_score:
            best_score, best_a = val, cand.copy()
    # include previous construction if length matches
    if prev_arr is not None and prev_arr.size == n:
        val = _cheap_score(prev_arr, fft_len, n)
        if val < best_score:
            best_score, best_a = val, prev_arr.copy()

    # random Dirichlet seeds with varying concentration
    for it in range(400):
        if time.time() >= deadline:
            break
        conc = 10 ** np.random.uniform(-4.0, 0.6)   # from ~1e‑4 to ~4
        g = np.random.gamma(conc, 1.0, size=n)
        cand = g / g.sum()
        val = _cheap_score(cand, fft_len, n)
        if val < best_score:
            best_score, best_a = val, cand.copy()
            print(f"[Rand {it:03d}] new best cheap = {val:.6f}")

    # --------------------------------------------------------------
    # 1. Refine using a soft‑max approximation of the max‑operator.
    # --------------------------------------------------------------
    a = best_a.copy()
    a = a / a.sum()                     # ensure we are on the simplex

    # Hyper‑parameters
    eta0 = 0.8                          # initial step‑size for EG
    beta_momentum = 0.88                # momentum coefficient for occasional additive step
    tau0 = 0.2                          # initial temperature for soft‑max
    tau_max = 4.0
    max_iters = 500_000
    momentum = np.zeros_like(a)

    # linear schedules
    tau_inc = (tau_max - tau0) / max_iters
    best_iter_a = a.copy()
    best_iter_val = best_score
    last_improve = 0

    iter_cnt = 0
    while time.time() < deadline and iter_cnt < max_iters:
        iter_cnt += 1

        # ---- convolution a*a (full linear convolution) ----
        A_fft = np.fft.fft(a, grad_fft_len)
        conv_full = np.real(np.fft.ifft(A_fft * A_fft))
        conv = conv_full[: 2 * n - 1]
        max_conv = conv.max()
        cheap = 2.0 * n * max_conv

        # ---- keep best cheap score observed ----
        if cheap < best_iter_val - 1e-12:
            best_iter_val = cheap
            best_iter_a = a.copy()
            last_improve = iter_cnt
            print(f"[Iter {iter_cnt:07d}] NEW BEST cheap = {cheap:.6f}")

        # ---- soft‑max weights over convolution entries ----
        tau = min(tau_max, tau0 + tau_inc * iter_cnt)
        # stability: subtract max before exponentiating
        exp_arg = tau * (conv - max_conv)
        # avoid overflow/underflow
        exp_arg = np.clip(exp_arg, -80.0, 80.0)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of Σ w_i·conv_i w.r.t. a:  g = 2·(a ∗ w_rev) ----
        w_rev = w[::-1]                       # reverse
        W_fft = np.fft.fft(w_rev, grad_fft_len)
        grad_full = np.real(np.fft.ifft(A_fft * W_fft))
        # linear part of length 3n‑2 is at the start of the circular result
        linear_grad = grad_full[: 3 * n - 2]
        start_idx = 2 * n - 2
        g = 2.0 * linear_grad[start_idx - np.arange(n)]

        # ---- exponentiated‑gradient update (multiplicative) ----
        eta = eta0 / math.sqrt(iter_cnt)
        # clip gradient to avoid overflow in exp
        g_clip = np.clip(g, -100.0, 100.0)
        a_mul = a * np.exp(-eta * g_clip)
        s = a_mul.sum()
        if s == 0.0 or not np.isfinite(s):
            a_mul = np.full_like(a, 1.0 / n)
            s = 1.0
        a_mul /= s

        # ---- occasional additive (heavy‑ball) step to escape plateaus ----
        if np.random.rand() < 0.07:
            momentum = beta_momentum * momentum - eta * g
            a_add = a + momentum
            a_add = _simplex_projection(a_add)
            # decide which candidate is better (cheap evaluation only)
            cheap_add = _cheap_score(a_add, fft_len, n)
            if cheap_add < cheap:
                a = a_add
            else:
                a = a_mul
        else:
            a = a_mul

        # ---- occasional random perturbation after long stagnation ----
        if iter_cnt - last_improve > 120_000:
            pert = 0.01 * np.random.rand(n)
            a = _simplex_projection(best_iter_a + pert)
            momentum = np.zeros_like(a)
            last_improve = iter_cnt
            print(f"[Iter {iter_cnt:07d}] Restart after stagnation (perturb).")

    # --------------------------------------------------------------
    # 2. Final exact evaluation with ``evaluate_sequence``
    # --------------------------------------------------------------
    final_seq = best_iter_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # --------------------------------------------------------------
    # 3. Return the best sequence found.
    # --------------------------------------------------------------
    return final_seq
