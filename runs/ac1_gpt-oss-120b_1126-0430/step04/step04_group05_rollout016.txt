"""
Propose a non‑negative step‑function sequence that (approximately) minimises

    O(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The code uses three complementary phases inside many short restarts:

* Phase 1 – Adam on a smooth log‑sum‑exp surrogate of the max‑operator.
  The smoothing parameter α grows exponentially, allowing a gradual
  sharpening of the objective.  After each Adam step the vector is
  projected onto the probability simplex (sum = 1, a_i ≥ 0).

* Phase 2 – Exact projected sub‑gradient descent on the true non‑smooth
  objective.  The step‑size decays geometrically.

* Phase 3 – Simulated‑annealing style mass‑transfer moves combined with
  occasional exact sub‑gradient steps (the “local refinement”).  A
  Metropolis acceptance criterion with a slowly cooling temperature
  helps to escape plateaus.

The whole procedure is repeated with fresh random starts until the
overall time budget (≈ 1000 s) is exhausted.  The best sequence seen
so far is returned as a plain list of floats.  All helper functions are
top‑level, there are no closures, and the code respects the required
interface.
"""
import math
import time
import numpy as np

# ----------------------------------------------------------------------
# Projection onto the probability simplex {x ≥ 0, sum(x)=1}
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending sort
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:                        # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full self‑convolution using FFT (length 2·n‑1)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return a*a (full convolution) using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Exact objective (no scaling – works for any a, not only simplex)
def _objective(a: np.ndarray, n: int, L: int) -> float:
    conv = _conv_fft(a, L)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a <= 0.0:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a ** 2))


# ----------------------------------------------------------------------
# Gradient of the log‑sum‑exp (soft‑max) surrogate of max(conv)
def _smooth_gradient(a: np.ndarray, conv: np.ndarray, alpha: float, L: int) -> np.ndarray:
    """
    ∇_a [ (2·n/α)·log Σ_k exp(α·conv_k) ].
    The gradient reduces to 2·corr(w,a) where w_k ∝ exp(α·conv_k).
    """
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)          # stabilise
    w = exp_scaled / exp_scaled.sum()                 # soft‑max weights

    # correlation w ⋆ a (ordered pairs)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)   # length ≥ 3·n‑2
    n_len = a.size
    grad = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]
    return grad


# ----------------------------------------------------------------------
# Sub‑gradient of the maximal convolution entry (exact, non‑smooth)
def _subgrad_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. a."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)   # all maximising indices
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        # contribution 2·a_j for each i (ordered pair (i,j) and (j,i))
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g

                        print(f"[restart {restart} SA‑sub] iter {it_loc} improved O={best_score:.6f}")

                # temperature decay
                T *= 0.9998
                it_loc += 1
            else:
                it_loc += 1

        # --------------------------------------------------------------
        # End of this restart – report status and possibly exit early
        remaining_total = deadline - time.time()
        print(f"[restart {restart}] finished – best O={best_score:.6f}, time left≈{remaining_total:.1f}s")
        if best_score <= 1.48:                     # a comfortable safety margin
            break

    # ------------------------------------------------------------------
    # Final verification with the provided evaluator
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence failed:", exc)

    return best_seq.tolist()
