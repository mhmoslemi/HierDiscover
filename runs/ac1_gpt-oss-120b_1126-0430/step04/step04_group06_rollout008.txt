#!/usr/bin/env python
"""
Propose a step‑function that minimizes the evaluation metric

    2·n·max(a*a) / (sum a)^2

using a mirror‑descent (Adam) optimiser on the probability simplex.
The gradient is the exact gradient of a *smoothed* max‑convolution
obtained from the log‑sum‑exp of the full self‑convolution.  This avoids
the “top‑K” approximation of the previous runner and gives a more
accurate descent direction, which typically yields a lower
∞‑norm of the convolution (hence a smaller score).

The optimisation works entirely in the log‑it space:
    a = softmax(z)  guarantees a_i ≥ 0 and Σ a_i = 1.
A schedule raises the temperature θ from a small value (smooth)
to a larger one (sharp) so that the algorithm first follows a gentle
landscape and finally concentrates on the true maximum.
Random perturbations and Adam’s adaptive steps help escape plateaus.
The code respects a 1000 s time budget and returns the best sequence
found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max (returns a probability vector)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it representation that yields `a` after soft‑max."""
    return np.log(a + eps)


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """Normalised power‑law a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
# Gradient of the smooth max‑convolution (log‑sum‑exp)
# ----------------------------------------------------------------------


def _smooth_max_gradient(a: np.ndarray, conv: np.ndarray,
                         A_fft: np.ndarray, theta: float,
                         L: int) -> np.ndarray:
    """
    Compute ∇_a  (  log‑sum‑exp_{k}(θ·conv_k)  )
    where conv = a*a (full linear convolution).

    Returns a gradient vector of length n = len(a).
    """
    n = a.shape[0]
    # soft‑max weights over the convolution entries
    max_val = conv.max()
    exp_vals = np.exp(theta * (conv - max_val))
    w = exp_vals / exp_vals.sum()                     # length = 2·n‑1

    # FFT of the weight vector (same length L as the a‑FFT)
    W_fft = np.fft.rfft(w, L)

    # Correlation w ⋆ a (full, linear).  Using the identity
    #   corr = ifft( FFT(w) * conj(FFT(a)) )
    corr_full = np.fft.irfft(W_fft * np.conj(A_fft), L)

    # Keep only the “valid” part where w fully overlaps a:
    #   start = n‑1  →  end = start + n
    full_len = len(w) + n - 1            # = 3·n‑2
    corr_full = corr_full[:full_len]     # discard trailing padding
    start = n - 1
    end = start + n
    corr_valid = corr_full[start:end]    # shape (n,)

    # ∂/∂a_i  (θ·log‑sum‑exp) = 2·∑_k w_k a_{k‑i}
    grad = 2.0 * corr_valid
    return grad


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Returns a list of non‑negative heights that (hopefully) beat the
    current best bound.  The routine respects the given time budget
    (≈1000 s) and uses a smoothed‑max convolution gradient with Adam
    on the simplex.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin for final eval

    # ---------------------------------------------------------------
    # 0. Initialise a probability vector (sum = 1)
    # ---------------------------------------------------------------
    global_seq = globals().get("height_sequence_1", None)
    if global_seq is not None:
        a = np.array(global_seq, dtype=np.float64)
        a = np.maximum(a, 0.0)                # guard against stray negatives
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / a.shape[0])
        else:
            a = a / s
        n = a.shape[0]
        print(f"[Init] Using supplied global sequence, length = {n}")
    else:
        n = 30000                               # default maximal support
        a = _power_law_distribution(n, 0.60)   # a gentle start
        print(f"[Init] No global sequence – starting with length = {n}")

    # ---------------------------------------------------------------
    # 1. Log‑it parametrisation + Adam state
    # ---------------------------------------------------------------
    eps = 1e-16
    z = _logits_from_distribution(a, eps)

    m = np.zeros_like(z)               # first‑moment estimate
    v = np.zeros_like(z)               # second‑moment estimate
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    base_alpha = 0.5                    # base learning‑rate

    # ---------------------------------------------------------------
    # 2. Pre‑compute FFT length (big enough for both self‑convolution
    #    and the correlation needed for the gradient)
    # ---------------------------------------------------------------
    L = _next_pow2(3 * n - 2)           # ≥ len(conv) + len(a) – 1

    # ---------------------------------------------------------------
    # 3. Main optimisation loop
    # ---------------------------------------------------------------
    best_seq = a.copy()
    best_score = float("inf")
    it = 0
    last_improve = 0
    report_every = 5000

    # temperature (smoothing) schedule
    theta_start = 0.2
    theta_end = 12.0
    theta_slope = (theta_end - theta_start) / max(1.0, budget_s - 20.0)

    while time.time() < deadline and it < 800_000:
        it += 1

        # probability vector from logits
        a = _softmax(z)

        # FFT of a (padded to length L)
        A_fft = np.fft.rfft(a, L)

        # self‑convolution a*a (full, linear)
        conv = np.fft.irfft(A_fft * A_fft, L)[: 2 * n - 1]

        # cheap (exact for Σa = 1) evaluation of the objective
        maxb = float(conv.max())
        cheap_score = 2.0 * n * maxb
        if cheap_score < best_score:
            best_score = cheap_score
            best_seq = a.copy()
            last_improve = it
            if it % report_every == 0:
                print(f"[Iter {it:07d}] NEW BEST cheap = {cheap_score:.9f}")

        # -----------------------------------------------------------
        # Smoothed gradient of the max‑convolution
        # -----------------------------------------------------------
        elapsed = time.time() - start_time
        theta = theta_start + theta_slope * elapsed
        if theta > theta_end:
            theta = theta_end

        grad_a = _smooth_max_gradient(a, conv, A_fft, theta, L)

        # -----------------------------------------------------------
        # Mirror‑descent step (KL geometry) on the simplex
        # -----------------------------------------------------------
        mu = float(np.dot(grad_a, a))
        grad_z = a * (grad_a - mu)

        # Adam moment updates (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # direction (negative Adam step)
        step_dir = - m_hat / (np.sqrt(v_hat) + eps_adam)

        # inverse‑sqrt learning‑rate decay
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)

        # update logits and keep them in a safe numeric range
        z = z + alpha_t * step_dir
        np.clip(z, -80.0, 80.0, out=z)

        # -----------------------------------------------------------
        # Occasional random perturbations (helps escape plateaus)
        # -----------------------------------------------------------
        if it % 4000 == 0 and np.random.rand() < 0.30:
            z += 0.015 * np.random.randn(n)
            m *= 0.5
            v *= 0.5

        # -----------------------------------------------------------
        # Early‑stop on stagnation
        # -----------------------------------------------------------
        if it - last_improve > 300_000:
            print(f"[Iter {it:07d}] No improvement for {it-last_improve} iters – stopping.")
            break

    # ---------------------------------------------------------------
    # 4. Final exact evaluation (the platform will also evaluate)
    # ---------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[Result] Final exact score = {final_score:.9f}")
    except Exception as exc:
        final_score = float("inf")
        print(f"[Result] Evaluation error: {exc}")

    # Store the sequence globally for possible reuse in later runs
    globals()["height_sequence_1"] = final_seq

    # ---------------------------------------------------------------
    # 5. Return the best sequence found.
    # ---------------------------------------------------------------
    return final_seq
