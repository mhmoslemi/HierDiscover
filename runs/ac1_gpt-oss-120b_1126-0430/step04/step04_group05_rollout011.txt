"""
propose_candidate
=================

A multi‑restart, simplex‑projected optimiser for the step‑function objective

    O(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm works on the probability simplex (∑a = 1) – the objective is
scale‑invariant – and combines three ideas:

1️⃣  Adam on a log‑sum‑exp surrogate of the max‑operator with an annealed
    smoothing parameter α.  The surrogate converges to the true max as
    α→∞, providing a smooth landscape for fast optimisation.

2️⃣  Exact projected sub‑gradient descent on the genuine (non‑smooth) objective.

3️⃣  A cheap linear‑programming refinement (when SciPy is available) that
    solves
        maximise   ∑ b_j
        subject to  conv(a,b) ≤ M ,   b ≥ 0
    where M = max(conv(a,a)).  The optimal b is normalised back to the simplex
    and mixed with the current iterate; this is the “alternating LP’’ method
    described in the reference material and often yields a noticeable drop in
    the bound.

The whole search runs for at most ``budget_s`` seconds (≈1000 s).  Several
different initialisations (uniform, cosine‑taper, sine‑square, random Dirichlet)
are tried; the best sequence found overall is returned as a plain list of
floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# 1.  Simplex utilities
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0,∑x=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    # Find the largest rho satisfying u[rho] > cssv[rho] / (rho+1)
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:                     # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# 2.  FFT‑based full convolution (length 2·n‑1)
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Return the full convolution a*a using an FFT of length L."""
    n = a.size
    if L is None:
        L = 1 << ((2 * n - 2).bit_length())      # next power of two ≥ 2·n‑1
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * n - 1]


# ----------------------------------------------------------------------
# 3.  Objective (scale‑invariant form)
# ----------------------------------------------------------------------
def _objective_O(a: np.ndarray, L: int) -> float:
    """2·n·max(conv(a,a))  (a is assumed to sum to 1)."""
    n = a.size
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# 4.  Sub‑gradient of the maximal convolution entry
# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. a (used for exact sub‑gradient steps)."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)          # indices attaining the max
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
# 5.  Gradient of the log‑sum‑exp surrogate
# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, conv: np.ndarray, alpha: float, L_corr: int) -> np.ndarray:
    """
    ∇_a ( (2·n/α)·log∑_k e^{α·conv_k} ).
    Implements the closed‑form 2·corr(w,a) where w_k ∝ e^{α·conv_k}.
    """
    # soft‑max weights
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)      # stabilise
    w = exp_scaled / exp_scaled.sum()

    # correlation w ⋆ a   (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L_corr)
    fft_a_rev = np.fft.rfft(a_rev, n=L_corr)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L_corr)
    n = a.size
    grad = 2.0 * corr_rev[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# 6.  Simple initialisers (all return a probability vector of length n)
# ----------------------------------------------------------------------
def _init_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _init_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    # raised‑cosine taper, values are non‑negative
    a = np.cos(np.pi * (i - (n - 1) / 2) / (n - 1))
    a = a ** 2                     # make everything non‑negative
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _init_sine_square(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.sin(np.pi * (i + 0.5) / n) ** 2
    a /= a.sum()
    return a


def _init_random_dirichlet(n: int, concentration: float = 0.5) -> np.ndarray:
    # Gamma draws give a Dirichlet(α) distribution
    a = np.random.gamma(concentration, 1.0, size=n)
    a /= a.sum()
    return a

    Returns the best sequence (as a numpy array) and its objective value.
    """

    L_corr = 1 << ((3 * n - 2).bit_length())          # for correlation in surrogate

    best_seq = a.copy()
    best_obj = _objective_O(best_seq, L_conv)

    # ------------------------------------------------------------------
    # Phase‑1: Adam on the smooth surrogate
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    lr0 = 0.6
    alpha_start = 5.0
    alpha_end = 200_000.0                     # a very sharp surrogate

    max_iter_phase1 = 300_000                 # generous ceiling – real limit is time
    it = 0
    phase1_deadline = deadline * 0.45 + time.time()  # allocate ≈45 % of remaining time

    while it < max_iter_phase1 and time.time() < phase1_deadline:
        conv = _conv_fft(a, L_conv)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_seq = a.copy()
            # occasional progress report
            if it % 10000 == 0:
                print(f"[adam] iter {it:6d} O={best_obj:.6f}")

        # linear annealing of α
        frac = it / max_iter_phase1

        grad = _smooth_gradient(a, conv, alpha, L_corr)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        lr = lr0 / math.sqrt(it + 1)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        it += 1

        conv = _conv_fft(a, L_conv)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_seq = a.copy()
            if it2 % 5000 == 0:

        # geometric decay of step‑size

        a = a - eta * g
        a = _project_to_simplex(a)

        it2 += 1

    cur_obj = 2.0 * n * _conv_fft(a, L_conv).max()
    it_loc = 0
    max_iter_local = 120_000
    no_improve = 0
    no_improve_limit = 6000

    while time.time() < deadline and it_loc < max_iter_local:
        # pick a source and a destination
        i, j = np.random.randint(n, size=2)
        if i != j and a[i] > 1e-15:
            delta = a[i] * np.random.rand() * 0.09   # ≤9 % of source mass

            a_new[j] += delta
            a_new = _project_to_simplex(a_new)

        else:
            no_improve += 1

        # occasional exact sub‑gradient step (keeps a directed)
        if it_loc % 13 == 0:
            conv = _conv_fft(a, L_conv)
            g = _subgradient_max_conv(a, conv)
            a = a - 0.06 * g
            a = _project_to_simplex(a)

        temperature *= 0.9997
        it_loc += 1
        if no_improve >= no_improve_limit:
            # gentle restart around the incumbent best
            a = best_seq.copy()
            a += np.random.randn(n) * 0.0012
            a = np.clip(a, 0, None)
            a = _project_to_simplex(a)
            no_improve = 0

    return best_seq, best_obj


# ----------------------------------------------------------------------
# 8.  Optional LP refinement (requires SciPy)
# ----------------------------------------------------------------------
def _lp_refinement(a: np.ndarray, max_seconds: float) -> tuple[np.ndarray, float]:
    """
    Solve  max ∑b   s.t.  conv(a,b) ≤ M (M = max(conv(a,a))) , b≥0,
    then mix the optimal b with a.  Returns an improved (a, O) pair
    or the original pair if the LP fails or exceeds the time budget.
    """
    try:
        from scipy.optimize import linprog
    except Exception:
        # SciPy unavailable – skip refinement
        return a, _objective_O(a, 1 << ((2 * a.size - 2).bit_length()))

    start = time.time()
    n = a.size
    conv_a = _conv_fft(a)               # full convolution, length 2·n‑1
    M = conv_a.max()

    # Build dense constraint matrix C where C[k,i] = a[i] if k-i ∈ [0,n‑1] else 0
    rows = 2 * n - 1
    C = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        C[i:i + n, i] = a
    b_ub = np.full(rows, M, dtype=np.float64)

    # Linear programme: minimise -1·b   (i.e. maximise ∑b)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None) for _ in range(n)]

    # Solve – give the LP a modest time budget
    lp_time_limit = max_seconds - (time.time() - start) - 0.1
    if lp_time_limit < 0.5:
        return a, _objective_O(a, 1 << ((2 * n - 2).bit_length()))
    res = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': lp_time_limit})

    if not res.success:
        return a, _objective_O(a, 1 << ((2 * n - 2).bit_length()))

    b_opt = res.x
    if b_opt.sum() == 0.0:
        return a, _objective_O(a, 1 << ((2 * n - 2).bit_length()))

    g = b_opt / b_opt.sum()                     # bring back to the simplex
    # try a few convex mixes of a and g, keep the best
    L_conv = 1 << ((2 * n - 2).bit_length())
    best_local = a
    best_val = _objective_O(a, L_conv)
    for t in (1.0, 0.75, 0.5, 0.33, 0.2, 0.1):
        cand = (1 - t) * a + t * g
        cand = _project_to_simplex(cand)        # safety
        val = _objective_O(cand, L_conv)
        if val < best_val:
            best_val = val
            best_local = cand
    return best_local, best_val

# 9.  Top‑level entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function sequence that minimises
    the evaluation function.  Returns the sequence as a plain ``list``.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.05          # tiny safety margin

    # --------------------------------------------------------------
    # Choose problem size and an initial vector
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=float)
        a0 = np.maximum(a0, 0.0)
        if a0.sum() == 0.0:
            a0 = np.full_like(a0, 1.0)
        a0 = a0 / a0.sum()
        n = a0.size
        print(f"[init] using supplied sequence, n={n}")
    else:
        # default length – a compromise between speed and quality
        n = 3000
        a0 = _init_uniform(n)
        print(f"[init] using uniform start, n={n}")

    # ------------------------------------------------------------------
    # Prepare a pool of diverse initialisers
    # ------------------------------------------------------------------
    init_fns = [
        lambda: a0.copy(),
        lambda: _init_uniform(n),
        lambda: _init_cosine(n),
        lambda: _init_sine_square(n),
        lambda: _init_random_dirichlet(n, concentration=0.5),
        lambda: _init_random_dirichlet(n, concentration=1.0),
        lambda: _init_random_dirichlet(n, concentration=2.0),
    ]

    # ------------------------------------------------------------------
    # Run a few independent restarts (time‑controlled)
    # ------------------------------------------------------------------
    best_seq = a0.copy()
    best_obj = _objective_O(best_seq, 1 << ((2 * n - 2).bit_length()))
    restart = 0
    while time.time() < deadline - 5.0:      # keep ~5 s for final polishing
        # pick the next initialiser cyclically
        a_start = init_fns[restart % len(init_fns)]()
        remaining = deadline - time.time()
        # allocate a slice of the remaining budget to this run
        run_deadline = time.time() + min(120.0, remaining * 0.5)

        seq, val = _optimise_one_start(a_start, run_deadline)

        if val < best_obj:
            best_obj = val
            best_seq = seq.copy()
            print(f"[restart {restart}] new global best O={best_obj:.6f}")

        restart += 1

    # ------------------------------------------------------------------
    # Final polish: an (optional) LP refinement if we have a little time left
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 4.0:
        print("[final] LP refinement phase")
        best_seq, best_obj = _lp_refinement(best_seq, max_seconds=time_left)

    # ------------------------------------------------------------------
    # Verify via the provided evaluator and return
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence failed:", exc)

    return best_seq.tolist()
