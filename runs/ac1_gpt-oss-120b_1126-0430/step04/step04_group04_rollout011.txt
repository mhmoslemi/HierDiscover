#!/usr/bin/env python
"""
Propose a non‑negative step function (list of heights) that minimises

        2·n·max_k (a*a)[k] / (Σ a_i)²

The algorithm is deliberately different from the previous heavy‑ball
implementation:

*  a quick exhaustive scan of several parametric families
   (power‑law, exponential, power‑law × exponential and beta‑shaped
   distributions) to obtain a good deterministic starting point;
*  a stochastic mirror‑descent (exponentiated‑gradient) refinement that
   directly follows the sub‑gradient of the active convolution entries;
*  simple restarts with small perturbations when progress stalls;
*  a final exact call to ``evaluate_sequence``.

All helper utilities are defined at top‑level, no closures are used,
and the routine respects the supplied time budget (default 1000 s).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float) -> np.ndarray:
    """Return a normalised power‑law a_i ∝ (i+1)^‑p (Σa_i = 1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = i ** (-p)
    seq /= seq.sum()
    return seq


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using FFT (returns length 2·n‑1 array)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa).real[: 2 * a.size - 1]
    return conv


def _conv_max_fft(a: np.ndarray, fft_len: int) -> tuple[float, int]:
    """Maximum value and its index of the autocorrelation a*a."""
    conv = _conv_full_fft(a, fft_len)
    idx = int(conv.argmax())
    return float(conv[idx]), idx

# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that
    minimises ``evaluate_sequence``.  The routine adapts its schedule to
    the available time budget and always returns a feasible sequence.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a safety margin for the final exact evaluation
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # Choose a working length.
    # --------------------------------------------------------------
    try:
        prev = height_sequence_1                      # provided globally
        n = max(len(prev), 5000)
        prev_arr = np.asarray(prev, dtype=np.float64)
        if prev_arr.sum() > 0.0:
            prev_arr = prev_arr / prev_arr.sum()
        else:
            prev_arr = None
    except Exception:
        prev_arr = None
        n = 5000

    # keep n within a reasonable range for the time budget
    n = min(max(n, 5000), 25000)                     # 5 k ≤ n ≤ 25 k

    # FFT length – next power of two ≥ 2·n‑1
    fft_len = 1 << ((2 * n - 1).bit_length())

    # --------------------------------------------------------------
    # Cheap score (assumes Σa_i = 1): 2·n·max(conv)
    # --------------------------------------------------------------
    def cheap_score(vec: np.ndarray) -> float:
        maxb, _ = _conv_max_fft(vec, fft_len)

    best_a = None

    # previous construction, if any
    if prev_arr is not None and prev_arr.size == n:

        if val < best_val:

    # power‑law sweep
    for p in np.linspace(0.4, 1.2, 81):
        if time.time() >= deadline:

        cand = _power_law_sequence(n, p)
        val = cheap_score(cand)

    # pure exponential sweep
    for lam in np.linspace(0.0, 3.0, 61):
        if time.time() >= deadline:

        idx = np.arange(n, dtype=np.float64)
        cand = np.exp(-lam * idx)
        cand /= cand.sum()
        val = cheap_score(cand)

    # power‑law × exponential sweep (coarse grid)

            cand /= cand.sum()

    # beta‑shaped families: a_i ∝ x_i^α (1-x_i)^β
    x = (np.arange(1, n + 1, dtype=np.float64) / n)

        for beta in np.linspace(0.2, 5.0, 21):

            cand = (x ** alpha) * ((1.0 - x) ** beta)
            cand_sum = cand.sum()
            if cand_sum == 0.0:

            cand /= cand_sum

    for conc in [0.001, 0.01, 0.1, 0.5, 1.0, 3.0]:
        if time.time() >= deadline:

        cand = g / g.sum()

    print(f"[Init] best cheap score = {best_val:.6f}")

    # --------------------------------------------------------------
    # 1. Refine the best candidate with mirror‑descent (exponentiated‑gradient)
    # --------------------------------------------------------------
    a = best_a.copy()
    # keep the best ever seen during refinement
    best_iter_a = a.copy()
    best_iter_val = best_val
    last_improve = 0
    iter_cnt = 0
    # hyper‑parameters
    eta0 = 0.6               # initial step size factor
    topk = 5                 # number of largest convolution entries used for g
    max_iters = 1_200_000    # safety upper bound

    while time.time() < deadline and iter_cnt < max_iters:
        iter_cnt += 1

        # ----- compute full convolution and cheap score -----
        conv = _conv_full_fft(a, fft_len)
        maxb = conv.max()
        cheap = 2.0 * n * maxb

        # ----- keep the best cheap score -----
        if cheap < best_iter_val - 1e-12:
            best_iter_val = cheap
            best_iter_a = a.copy()
            last_improve = iter_cnt
            print(f"[Iter {iter_cnt:07d}] NEW BEST cheap = {cheap:.6f}")

        # ----- sub‑gradient from the top‑k shifts -----
        # (average of the gradients of the k largest convolution entries)
        top_idx = np.argpartition(conv, -topk)[-topk:]
        g = np.zeros_like(a)
        for idx in top_idx:
            lo = max(0, idx - (n - 1))
            hi = min(idx, n - 1)
            rng = np.arange(lo, hi + 1, dtype=np.int64)
            g[rng] += 2.0 * a[idx - rng]
        g /= float(topk)

        eta = eta0 / math.sqrt(iter_cnt)

        a = np.exp(log_a)
        sum_a = a.sum()
        if sum_a == 0.0 or not np.isfinite(sum_a):
            # fall back to the last good iterate
            a = best_iter_a.copy()
            continue
        a /= sum_a

        if iter_cnt - last_improve > 200_000:

            a = _simplex_projection(best_iter_a + perturb)
            last_improve = iter_cnt
            print(f"[Iter {iter_cnt:07d}] Restart after stagnation.")

    # 2. Final exact evaluation (uses the provided ``evaluate_sequence``)
    # --------------------------------------------------------------
    final_seq = best_iter_a.tolist()
    try:

        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:      # defensive – should never happen
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # --------------------------------------------------------------
    # 4. Return the best sequence found.
    # --------------------------------------------------------------
    return final_seq
