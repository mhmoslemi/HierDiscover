```
def compute_objective(a):
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    conv = np.fft.irfft(np.fft.rfft(a, n=L) * np.fft.rfft(a, n=L), n=L)[:2*n-1]

    sum_a = a.sum()
    if sum_a < 1e-12:

    return 2.0 * n * max_b / (sum_a * sum_a)
```

The objective  

\[
O(a)=\frac{2\,n\;\max (a*a)}{(\sum a)^2}
\]

is *scale invariant* – we may enforce \(\sum a_i = 1\) and simply minimise  
\(\max (a*a)\).  The convolution of a non‑negative vector with itself is a
discrete analogue of the autocorrelation of a probability density.
If a density behaves like \(x^{-1/2}\) near the interval ends, its
self‑convolution is almost constant (the continuous identity  

\[
\int_0^t \frac{dx}{\sqrt{x(t-x)}} = \pi .
\]

Therefore a **U‑shaped** discrete sequence  

\[
a_i\;\propto\; \bigl(\min\{i+1,n-i\}\bigr)^{-p},
\qquad p\approx\frac12,
\]

produces a very flat convolution and a much smaller maximum than the
uniform start used by the reference code.  The algorithm below

1. builds such power‑law shapes for several exponents `p`,
2. optionally mixes them with a uniform vector,
3. keeps the best candidate,
4. performs a short projected sub‑gradient polish,
5. finally returns the best sequence it has seen.

All steps are performed with FFT‑based convolutions, respect the simplex
constraint, and stay well within the 1000 s wall‑time.

```python
"""
Propose a non‑negative step‑function sequence that minimises

    O(a) = 2·n·max(conv(a,a)) / (sum(a)²)

The algorithm builds a symmetric power‑law (“U‑shaped”) distribution
a_i ∝ min(i+1, n‑i)^{-p} (p≈0.5), optionally mixes it with a uniform
vector, and finishes with a short projected sub‑gradient refinement.
All operations use FFT‑based convolutions and simplex projection.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x >= 0, sum(x)=1}.  Returns a new array.
    """
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho_candidates = np.where(rho_arr > 0)[0]
    if rho_candidates.size == 0:
        # all non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho_candidates[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """
    Full convolution a*a using an FFT of length L (L is a power of two).
    Returns a vector of length 2*len(a)-1.
    """
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
def _objective_simplex(a: np.ndarray, n: int, L: int) -> float:
    """
    Compute O(a) assuming sum(a)=1.
    """
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a (valid for non‑negative a).
    """
    n = a.size
    max_val = conv.max()
    # all indices where the maximum is attained
    ks = np.flatnonzero(conv == max_val)
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence that minimises O(a).
    Returns the sequence as a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # stop a little before the deadline so we can return a result
    deadline = start_time + budget_s - 0.2

    # --------------------------------------------------------------
    # Determine problem size
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
        print(f"[init] Using supplied length n={n}")
    else:
        n = 5000                     # a comfortable size for the time budget
        print(f"[init] No supplied sequence – using n={n}")

    # FFT length – next power of two >= 2·n‑1
    L = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # Baseline uniform vector (just for reference)
    uniform = np.full(n, 1.0 / n)
    best_a = uniform.copy()
    best_val = _objective_simplex(best_a, n, L)
    print(f"[baseline] uniform O={best_val:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Build power‑law “U‑shaped” candidates
    idx = np.arange(n)
    d = np.minimum(idx + 1, n - idx)              # distance to nearest end (1‑based)
    p_candidates = np.linspace(0.10, 1.20, 23)     # try a fine grid of exponents

    for p in p_candidates:
        a = d.astype(float) ** (-p)
        a_sum = a.sum()
        if a_sum == 0.0:
            continue
        a /= a_sum                                 # simplex normalisation
        cur = _objective_simplex(a, n, L)
        if cur < best_val:
            best_val = cur
            best_a = a.copy()
            print(f"[search-p] p={p:.3f} → O={cur:.6f}")
        if time.time() > deadline:
            break

    # ------------------------------------------------------------------
    # 2️⃣  Mix the best power‑law shape with a uniform component
    if time.time() < deadline:
        for alpha in np.linspace(0.0, 0.5, 11):     # weight of the uniform part
            a_mix = alpha * uniform + (1.0 - alpha) * best_a
            # still on the simplex because both parts sum to 1
            cur = _objective_simplex(a_mix, n, L)
            if cur < best_val:
                best_val = cur
                best_a = a_mix.copy()
                print(f"[mix] α={alpha:.2f} → O={cur:.6f}")
            if time.time() > deadline:
                break

    # ------------------------------------------------------------------
    # 3️⃣  Very short projected sub‑gradient polish (few thousand steps)
    if time.time() < deadline:
        a = best_a.copy()
        subgrad_steps = 2000
        eta0 = 0.15                                 # initial step size
        for it in range(subgrad_steps):
            conv = _conv_fft(a, L)
            cur = 2.0 * n * conv.max()
            if cur < best_val:
                best_val = cur
                best_a = a.copy()
                print(f"[local] iter {it:04d} O={cur:.6f}")
            # sub‑gradient of the max‑entry of the convolution
            g = _subgradient_max_conv(a, conv)
            # geometrically decaying step size
            eta = eta0 * (0.99 ** it)
            a = a - eta * g
            # keep feasibility
            a = _project_to_simplex(a)

            if time.time() > deadline:
                break

    # ------------------------------------------------------------------
    # Final sanity check with the provided evaluator
    best_seq = best_a.tolist()
    try:
        final_score = evaluate_sequence(best_seq)
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence failed:", exc)

    return best_seq
```
