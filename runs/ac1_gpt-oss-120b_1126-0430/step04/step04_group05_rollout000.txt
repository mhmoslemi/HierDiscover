"""
Propose a step‑function (non‑negative list) that minimises

    O(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm consists of two main phases:

1️⃣ **Polyak sub‑gradient descent** – the exact objective is
   O = 2·n·max(conv) (the sum is normalised to 1).  The sub‑gradient of
   `max(conv)` is inexpensive to compute.  Using a Polyak step‑size
   η = (max‑avg) / ((2·n)·‖g‖²) (where *avg* = 1/(2n‑1) is the
   trivial lower bound) yields a provably decreasing sequence
   without any hand‑tuned decay schedule.

2️⃣ **Stochastic annealing fine‑tuning** – after the sub‑gradient phase
   we perform many cheap mass‑transfer moves (move a fraction of the
   probability mass from one index to another).  Moves are accepted
   if they improve the exact objective or with a Metropolis probability
   that slowly cools.  Occasionally a small exact sub‑gradient step is
   taken to keep the search directed.

Both phases respect a 1000 s wall‑time budget, keep the best sequence
encountered, and finally verify it with the provided ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x≥0, ∑x=1}
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:                     # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full self‑convolution using FFT (returns length 2·n‑1)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return a*a (full convolution) using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) w.r.t. a (as used in the reference code)
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) at a."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)          # indices attaining the max
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        # derivative of a_i*a_j w.r.t a_i is a_j (ordered pair) → multiplied by 2
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size                # average over all active constraints
    return g

    # a small safety margin before the external timeout
