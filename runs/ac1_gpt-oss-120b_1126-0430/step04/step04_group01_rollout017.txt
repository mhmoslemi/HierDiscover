    O(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm consists of three stages:

1️⃣  **Annealed Adam on a smooth log‑sum‑exp surrogate**  
    We minimise (2·n/α)·log Σ exp(α·conv) + λ·Σ a_i·log a_i  
    while projecting onto the probability simplex (∑a_i=1, a_i≥0).  
    α is annealed from a small to a very large value, λ is annealed from
    a modest entropy weight down to zero.  The Adam step‑size decays as 1/√t.

2️⃣  **Intensive simulated‑annealing local search**  
    Random “mass‑transfer’’ moves (move a random fraction of probability mass
    from one index to another) are accepted if they improve the exact objective,
    otherwise with a Metropolis probability that cools down.  A sub‑gradient step
    on the exact max‑convolution is performed every iteration.  Restarts are
    triggered after a stretch without improvement.

3️⃣  **Final polishing** – a few extra Adam steps at the end of the budget
    with an extremely large smoothing parameter to fine‑tune the remaining
    slack.

The routine respects the 1000 s budget, prints progress, and finally returns
the best sequence as a plain ``list`` of ``float``.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------
# Simplex projection (Euclidean)
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
# Full linear convolution via FFT (length 2·n‑1)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return the full a⨂a convolution (length 2·n‑1) using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# --------------------------------------------------------------
# Exact objective for a vector that is already normalised (∑a=1)
def _objective_simplex(a: np.ndarray, n: int, L: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for sum(a)=1."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# --------------------------------------------------------------
# Sub‑gradient of max(conv) (used in phase 2)
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. a (averaged over all maximisers)."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)          # indices achieving the max
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


# --------------------------------------------------------------
# Gradient of the smooth surrogate (log‑sum‑exp) plus entropy term
def _smooth_gradient_plus_entropy(
    a: np.ndarray,
    conv: np.ndarray,
    alpha: float,
    L: int,
    lam: float,
) -> np.ndarray:
    """
    Returns ∇_a ( (2·n/α)·log∑_k e^{α·conv_k} + λ·∑ a_i·log a_i ).
    The first part reduces to 2·corr(w,a) with w_k ∝ e^{α·conv_k};
    the second part contributes λ·(log a_i + 1).
    """
    # ---- soft‑max weights ----------
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)          # numerical stability
    w = exp_scaled / exp_scaled.sum()                 # shape (2n‑1,)

    # ---- correlation w ⋆ a (g_i = 2·∑_j w_{i+j} a_j) ----
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n_len = a.size
    grad_corr = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]

    # ---- entropy part ----------
    eps = 1e-20
    grad_ent = lam * (np.log(a + eps) + 1.0)

    return grad_corr + grad_ent


# --------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    the provided evaluation function.  Returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # reserve a small safety margin
    deadline = start_time + budget_s - 0.2

    # --------------------------------------------------------------
    # Determine problem size -------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.array(height_sequence_1, dtype=float)
        n = base.size
        a = np.maximum(base, 0.0)
        tot = a.sum()
        if tot < 1e-12:
            a = np.full(n, 1.0 / n)
        else:
            a /= tot
        print(f"[init] start from supplied sequence, n={n}")
    else:
        # a slightly larger n than used before gives more freedom
        n = 6000
        a = np.full(n, 1.0 / n)
        print(f"[init] start from uniform, n={n}")

    # FFT length – next power of two ≥ 2·n‑1
    L = 1
    while L < 2 * n - 1:
        L <<= 1

    # --------------------------------------------------------------
    # Phase‑1 : Annealed Adam on the smooth surrogate -----------------------
    best_seq = a.copy()
    best_score = _objective_simplex(best_seq, n, L)
    print(f"[phase‑1] start O={best_score:.6f}")

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    lr0 = 0.9                      # base learning rate
    alpha_start = 2.0
    alpha_end = 20000.0
    lambda_start = 0.2
    lambda_end = 0.0

    max_iter_phase1 = 600_000
    it = 0
    # allocate roughly 60 % of the time budget to phase‑1
    phase1_deadline = start_time + budget_s * 0.60

    while time.time() < phase1_deadline and it < max_iter_phase1:
        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[phase‑1] iter {it:,} new best O={best_score:.6f}")

        # ----- annealing schedule -----
        frac = it / max_iter_phase1
        alpha = alpha_start + (alpha_end - alpha_start) * frac
        lam = max(0.0, lambda_start * (1.0 - frac))

        # ----- smooth gradient -----
        grad = _smooth_gradient_plus_entropy(a, conv, alpha, L, lam)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        lr = lr0 / math.sqrt(it + 1)          # 1/√t decay

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional tiny jitter to escape plateaus
        if np.random.rand() < 0.001:
            a_new += np.random.randn(n) * 5e-5

        # project back onto the simplex
        a_new = _project_to_simplex(a_new)

        a = a_new
        it += 1

    # --------------------------------------------------------------
    # Phase‑2 : Simulated‑annealing intensive local refinement ---------------
    print(f"[phase‑2] entering local refinement, best O={best_score:.6f}")

    max_restarts = 0
    temperature0 = 0.10
    temp_decay = 0.99985
    max_iter_local = 300_000
    no_improve_limit = 2000
    subgrad_interval = 1       # sub‑gradient every iteration for aggressive descent

    while time.time() < deadline:
        # ----- initialise a restart -----
        if np.random.rand() < 0.6:
            a = best_seq.copy()
            a += np.random.randn(n) * 0.0010   # small Gaussian jitter
            a = np.clip(a, 0, None)
            a = _project_to_simplex(a)
        else:
            a = np.random.rand(n)
            a = _project_to_simplex(a)

        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        it_loc = 0
        no_improve = 0
        T = temperature0

        while time.time() < deadline and it_loc < max_iter_local:
            # ---------- random mass‑transfer move ----------
            i, j = np.random.randint(n, size=2)
            if i != j and a[i] > 1e-14:
                # fraction of source mass to move – annealed max fraction
                max_frac = 0.5 * (1.0 - it_loc / max_iter_local) + 0.02
                frac = np.random.rand() * max_frac
                delta = a[i] * frac
                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta
                a_new = _project_to_simplex(a_new)

                conv_new = _conv_fft(a_new, L)
                new_obj = 2.0 * n * conv_new.max()

                # Metropolis acceptance
                if new_obj < cur_obj or np.random.rand() < math.exp((cur_obj - new_obj) / max(T, 1e-16)):
                    a, conv, cur_obj = a_new, conv_new, new_obj
                    no_improve = 0
                    if cur_obj < best_score:
                        best_score = cur_obj
                        best_seq = a.copy()
                        print(f"[{time.time():.1f}] new global best O={best_score:.6f} (restart={max_restarts}, iter={it_loc})")
                else:
                    no_improve += 1
            else:
                no_improve += 1

            # ---------- sub‑gradient step ----------
            if it_loc % subgrad_interval == 0:
                g = _subgradient_max_conv(a, conv)
                # step size decays linearly
                eta = 0.15 * (1.0 - it_loc / max_iter_local)
                a = a - eta * g
                a = _project_to_simplex(a)
                conv = _conv_fft(a, L)
                cur_obj = 2.0 * n * conv.max()
                if cur_obj < best_score:
                    best_score = cur_obj
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] sub‑grad improved O={best_score:.6f}")

            # ---------- occasional smoothing ----------
            if it_loc % 5000 == 0 and it_loc > 0:
                kernel = np.array([0.125, 0.25, 0.25, 0.25, 0.125])
                a_smooth = np.convolve(a, kernel, mode="same")
                a = _project_to_simplex(0.85 * a + 0.15 * a_smooth)

            # temperature decay
            T *= temp_decay
            it_loc += 1

            # restart if stagnating
            if no_improve >= no_improve_limit:
                break

        max_restarts += 1
        if time.time() >= deadline:
            break

    # --------------------------------------------------------------
    # Phase‑3 : Final Adam polishing (very high α) -------------------------
    print(f"[phase‑3] final Adam polishing, best O={best_score:.6f}")
    # Re‑use the Adam machinery with a huge α and zero entropy
    a = best_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    lr0 = 0.6
    alpha = 50000.0
    lam = 0.0
    polish_iters = 30_000
    it = 0
    while time.time() < deadline and it < polish_iters:
        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[phase‑3] iter {it:,} new best O={best_score:.6f}")

        grad = _smooth_gradient_plus_entropy(a, conv, alpha, L, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        lr = lr0 / math.sqrt(it + 1)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = _project_to_simplex(a_new)
        a = a_new
        it += 1

    except Exception as e:
        print("[final] evaluate_sequence failed:", e)

    return best_seq.tolist()
