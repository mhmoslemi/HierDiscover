"""
Propose candidate sequence optimizing the max‑self‑convolution bound.

The algorithm is a continuation‑gradient method on the probability simplex
using an Adam‑style exponentiated‑gradient update.  A temperature β
smoothes the max‑operator:  w_i ∝ exp(β·(conv_i−max(conv))).  β is
gradually increased (soft‑max → hard‑max) while the learning‑rate η is
exponentially decayed.  Random jitter and periodic restarts from the
best‑so‑far solution help to escape plateaus.  The procedure respects a
1000 s time budget and returns the best sequence found.
"""
import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv

def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Linear correlation g_i = Σ_t w[i+t]·a[t]   for i = 0,…,n‑1.
    w must have length 2·n‑1.
    """
    n = a.shape[0]
    m = w.shape[0]        # = 2·n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]

Propose a step‑function that minimizes the bound

        O = 2·n·max(conv(a,a))   with   sum(a)=1 , a_i ≥ 0 .

The algorithm is a continuation‑gradient (soft‑max → hard‑max) optimiser on
the probability simplex.  It works in the log‑domain (exponentiated‑gradient)
and employs Adam‑style moments for an adaptive step‑size.  A temperature
parameter β smooths the max‑operator:
        w_i ∝ exp(β·(conv_i−max(conv))) .
β is increased during the run while the base learning‑rate η is decay‑ed.

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv

    Linear correlation g_i = Σ_t w[i+t]·a[t] for i = 0,…,n‑1.
    ``w`` must have length 2·n‑1 (full self‑convolution length).
