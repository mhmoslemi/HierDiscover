"""
Mirror‑descent / Adam optimizer with a final stochastic refinement.

The routine tries a few different lengths (the known best length and a
larger one) and runs an Adam‑style mirror‑descent on the log‑probabilities
of the step function.  The gradient of the smooth approximation of
maxₖ (a*a)ₖ is obtained by a soft‑max over the convolution entries and a
correlation computed with a real FFT (O(n log n) per iteration).

A simple stochastic “pair‑wise mass transfer” local search is applied after
the gradient phase to squeeze a little more out of the solution.

The whole optimisation respects the 1000 s budget; the best sequence found
is returned as a plain list of floats.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv


def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Linear correlation g_i = Σ_t w[i+t]·a[t] for i = 0 … n‑1.
    Both arguments are real; the result is real.
    """
    n = a.shape[0]
    m = w.shape[0]                # = 2·n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Soft‑max over the convolution entries.
    beta controls the sharpness; larger beta → harder max.
    """
    M = conv.max()
    shifted = conv - M          # ≤ 0, max = 0
    # exponentials are never > 1, no overflow risk
    exp_vals = np.exp(beta * shifted)
    s = exp_vals.sum()
    if s == 0.0:                # extreme case – fall back to hard max
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w
    return exp_vals / s


def _score_from_a(a: np.ndarray) -> float:
    """
    Compute the objective O = 2·n·max(conv(a,a))    (sum(a) is assumed 1)
    """
    conv = _conv_self_fft(a)
    return 2.0 * a.size * conv.max()


def _adam_optimize(
    n: int,
    deadline: float,
    init_u: np.ndarray = None,
) -> tuple[np.ndarray, float]:
    """
    Adam‑style mirror descent on the log‑probabilities u_i = log a_i.
    Returns the logits of the best solution (still in log‑domain) and its
    objective value.
    """
    # initialise logits (log‑probabilities)
    if init_u is not None:
        u = init_u.copy()
    else:
        u = np.full(n, -np.log(n))   # uniform distribution → log(1/n)

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0

    # best solution tracking
    best_u = u.copy()
    best_score = np.inf
    best_iter = 0
    it = 0
    iters_since_best = 0

    # hyper‑parameters (empirically tuned)
    eta_start = 0.7            # initial step‑size in log‑domain
    eta_end = 0.02             # final step‑size
    beta_start = 1.0           # smoothing start
    beta_end = 100000.0        # very sharp end
    max_iters_est = 500_000    # used to compute schedules
    jitter_interval = 3000
    jitter_scale = 5e-4
    restart_interval = 20000   # restart if no improvement for this many iters

    while time.time() < deadline:
        it += 1

        # -----------------------------------------------------------------
        # (a) probability vector a (simplex) and its self‑convolution
        a = np.exp(u - u.max())
        a_sum = a.sum()
        if a_sum == 0.0:                # numerical safety net
            a = np.full_like(a, 1.0 / n)
            a_sum = 1.0
        a = a / a_sum                     # now ∑a = 1

        conv = _conv_self_fft(a)
        M = conv.max()
        cur_score = 2.0 * n * M

        # -----------------------------------------------------------------
        # (b) best‑solution bookkeeping
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_u = u.copy()
            best_iter = it
            iters_since_best = 0
            print(f"[iter {it:07d} n={n}] new best O={best_score:.6f}")
        else:
            iters_since_best += 1

        # -----------------------------------------------------------------
        # (c) schedule for learning‑rate and beta
        frac = min(it / max_iters_est, 1.0)
        eta = eta_start * ((eta_end / eta_start) ** frac)   # exponential decay
        beta = beta_start + (beta_end - beta_start) * frac

        # -----------------------------------------------------------------
        # (d) soft‑max weight over convolution entries
        w = _softmax_weights(conv, beta)

        # (e) gradient of the smoothed max‑convolution
        g = 2.0 * _correlation_fft(w, a)   # shape (n,)

        # -----------------------------------------------------------------
        # (f) Adam update on the log‑variables
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        u = u - eta * m_hat / (np.sqrt(v_hat) + eps)

        # -----------------------------------------------------------------
        # (g) occasional jitter to keep the dynamics alive
        if it % jitter_interval == 0:
            u += np.random.randn(n) * jitter_scale

        # -----------------------------------------------------------------
        # (h) restart if we have been stuck for a while
        if iters_since_best >= restart_interval:
            u = best_u + np.random.randn(n) * (jitter_scale * 0.5)
            m.fill(0.0)
            v.fill(0.0)
            iters_since_best = 0
            print(f"[restart] iter {it}, re‑init from best (O={best_score:.6f})")

        # (optional) break early if we have exhausted almost all time
        if time.time() >= deadline:
            break

    return best_u, best_score


def _local_refine(
    a: np.ndarray,
    deadline: float,
    step_frac: float = 0.01,
    max_trials: int = 1_000_000,
) -> tuple[np.ndarray, float]:
    """
    Simple stochastic refinement: repeatedly move a small fraction of mass
    from a random source index to a random destination index and keep the
    change if it lowers the objective.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score_from_a(best_a)
    trials = 0

    while time.time() < deadline and trials < max_trials:
        trials += 1
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        if best_a[i] < 1e-18:
            continue
        delta = best_a[i] * step_frac
        a_new = best_a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # numerical safety
        if a_new[i] < 0.0 or a_new[j] < 0.0:
            continue
        score = _score_from_a(a_new)
        if score < best_score:
            best_a = a_new
            best_score = score
            # small progress report (optional)
            print(f"[refine] trial {trials:07d} O={best_score:.6f}")

    return best_a, best_score


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list:
    """
    Main entry point required by the evaluation framework.

    The function explores a couple of candidate lengths, runs an Adam‑based
    mirror‑descent optimiser on the log‑probabilities of the step function,
    and finally performs a lightweight stochastic post‑processing.  The best
    sequence (as a plain Python list) is returned.
    """
    np.random.seed(seed)
    # leave a small safety margin for the post‑processing code
    global_deadline = time.time() + budget_s - 0.5

    # -----------------------------------------------------------------
    # Lengths to try.  The known construction (4587) is tried first;
    # a larger length is attempted if time permits.
    lengths = [4587, 8000]

    best_seq = None
    best_overall_score = np.inf

    for n in lengths:
        if time.time() >= global_deadline:
            break

        # -----------------------------------------------------------------
        # Initialise logits (log‑probabilities)
        if 'height_sequence_1' in globals():
            hs = np.asarray(height_sequence_1, dtype=float)
            if hs.size == n:
                a0 = hs / hs.sum()
                u0 = np.log(a0 + 1e-12)
            elif hs.size < n:
                # pad the known vector with tiny values and renormalise
                pad = np.full(n - hs.size, 1e-12)
                a0 = np.concatenate([hs, pad])
                a0 = a0 / a0.sum()
                u0 = np.log(a0 + 1e-12)
            else:   # truncate
                a0 = hs[:n]
                a0 = a0 / a0.sum()
                u0 = np.log(a0 + 1e-12)
        else:
            u0 = None

        # -----------------------------------------------------------------
        # Adam optimisation on the log‑domain
        best_u, _ = _adam_optimize(n, global_deadline, init_u=u0)

        # Convert logits back to a probability vector
        a_curr = np.exp(best_u - best_u.max())
        a_curr = a_curr / a_curr.sum()
        score_curr = _score_from_a(a_curr)
        print(f"[length {n}] after Adam O={score_curr:.6f}")

        # -----------------------------------------------------------------
        # Stochastic local refinement (only if we still have enough time)
        if time.time() < global_deadline:
            # give the refinement at most ~5 % of the remaining budget
            refine_deadline = time.time() + 0.05 * (global_deadline - time.time())
            a_refined, score_refined = _local_refine(a_curr, refine_deadline)
            print(f"[length {n}] after local refinement O={score_refined:.6f}")
            a_curr, score_curr = a_refined, score_refined

        # -----------------------------------------------------------------
        # Keep the best across all tried lengths
        if score_curr < best_overall_score:
            best_overall_score = score_curr
            best_seq = a_curr.tolist()
            print(f"[global] new best overall O={best_overall_score:.6f}")

    # -----------------------------------------------------------------
    # Final safety: if nothing was produced (should never happen) fall back
    if best_seq is None:
        best_seq = [1.0] * 10

    # -----------------------------------------------------------------
    # Final call to the provided evaluator (the result is only printed,
    # the function must return the sequence)
    try:
        final_val = evaluate_sequence(best_seq)
        print(f"[final] evaluate_sequence reports O={final_val:.6f}")
    except Exception as exc:          # pragma: no cover
        print("[final] evaluate_sequence failed:", exc)

    return best_seq
