"""
Optimisation of a non‑negative step function.

The routine runs an Adam‑style mirror‑descent on the log‑variables
(u_i = log a_i) on the probability simplex.  The objective is the
max‑entry of the self‑convolution (the bound 2·n·max(conv)/sum²).
Early iterations use a soft‑max weighting of the convolution entries,
later iterations focus on the single worst entry (hard‑max).  Small
random jitter and periodic restarts from the best‑so‑far solution
help escape plateaus.  The algorithm works fully in NumPy, uses FFTs
for O(n log n) convolutions and respects a 1000 s time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv


def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Linear correlation g_i = Σ_t w[i+t]·a[t]  for i=0,…,n‑1.
    Implemented via FFT (real‑valued).
    """
    n = a.shape[0]
    m = w.shape[0]                     # = 2·n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]

# ----------------------------------------------------------------------
# Main optimisation entry point

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list:
    """
    Search for a non‑negative step function minimising ``evaluate_sequence``.
    Returns the best sequence as a plain list of floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # 1) Initialise the log‑probabilities (u = log a)
    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=float)
        a0 = np.maximum(a0, 0.0)                # enforce non‑negative
        if a0.sum() == 0.0:
            a0 = np.full_like(a0, 1.0)
        a0 = a0 / a0.sum()                      # normalise to sum=1
        n = a0.shape[0]
        eps = 1e-12
        u = np.log(a0 + eps)
    else:
        n = 8192                                 # a moderately large length
        u = np.full(n, -np.log(n))               # uniform start (log(1/n))

    best_u = u.copy()
    best_score = np.inf
    best_iter = 0

    # --------------------------------------------------------------
    # Adam accumulator initialisation
    m = np.zeros_like(u)        # first moment
    v = np.zeros_like(u)        # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    # Schedules (tuned to favour fast convergence then fine‑tuning)
    max_iters_est = 400_000
    beta_start = 2.0
    beta_end   = 100_000.0
    eta_start  = 0.6
    eta_end    = 0.02
    jitter_interval = 4000
    jitter_scale = 5e-4
    restart_interval = 20000

    it = 0
    while time.time() < deadline and it < 800_000:
        it += 1

        # ----------------------------------------------------------
        # (a) build the probability vector a from the log‑variables
        a = np.exp(u - u.max())
        a_sum = a.sum()

            a = np.full_like(a, 1.0 / n)
            a_sum = 1.0
        a = a / a_sum                         # normalise → sum(a)=1

        # ----------------------------------------------------------

        conv = _conv_self_fft(a)
        M = conv.max()
        cur_score = 2.0 * n * M               # because sum(a)=1

        # (c) keep the best solution seen so far
        if cur_score < best_score - 1e-13:

        fraction = min(it / max_iters_est, 1.0)

            u += np.random.randn(n) * (jitter_scale * 0.5)
            print(f"[restart] iter {it}, re‑initialised from best (O={best_score:.6f})")

    # ------------------------------------------------------------------
    # 3) Final evaluation with the provided evaluator
    best_seq = np.exp(best_u - best_u.max())

    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f} "
              f"(best iteration {best_iter})")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_seq.tolist()
