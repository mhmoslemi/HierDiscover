"""
Propose a non‑negative step‑function (list of heights) that approximately
minimises

    O(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm consists of two phases.

Phase 1 – Exponentiated‑gradient optimisation of a smooth surrogate.
We minimise a smoothed version of the max‑convolution together with an
ℓ₂‑penalty on the full convolution:

   F(a) = (2·n/α)·log∑_k exp(α·conv_k(a))  +  λ₂·‖conv(a)‖₂² .

Both terms are differentiable; the gradient is obtained via FFT‑based
correlations.  An exponentiated‑gradient (multiplicative) update keeps the
variables strictly positive and automatically satisfies the simplex
constraint (∑a_i = 1).  The smoothing parameter α is annealed from a small
value to a large one, while the ℓ₂‑weight λ₂ is annealed down to zero,
so that the optimisation first flattens the whole convolution and then
focuses on the true maximum.

Phase 2 – Stochastic local refinement.
Starting from the best candidate of phase 1 we perform many cheap
mass‑transfer moves (move a random fraction of probability mass from one
index to another).  Moves are accepted if they improve the exact
objective, otherwise with a Metropolis probability that slowly cools
down.  Every few iterations a projected sub‑gradient step on the exact
max‑convolution is taken.  A few restarts (jittered copies of the current
best or fresh random points) improve exploration.

Both phases respect a total wall‑clock budget of 1000 s.  The final
best sequence is returned as a plain Python list.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------
# Full linear convolution via FFT (length 2·n‑1)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return the full a⨂a convolution (length 2·n‑1) using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# --------------------------------------------------------------
# Correlation (x ⋆ a)[i] = ∑_j x[i + j]·a_j  ,  i = 0 … n‑1
def _correlation(x: np.ndarray, a: np.ndarray, L: int) -> np.ndarray:
    """Correlation of x with a (definition above) using FFT of length L."""
    a_rev = a[::-1]
    fft_x = np.fft.rfft(x, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_x * fft_a_rev, n=L)
    n = a.size
    return corr_rev[n - 1 : n - 1 + n]


# --------------------------------------------------------------
# Exact objective for a probability vector (∑a = 1)
def _objective_exact(a: np.ndarray, n: int, L: int) -> float:
    """Exact O = 2·n·max(conv) for a normalised (∑a = 1) vector."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# --------------------------------------------------------------
# Phase‑1: exponentiated‑gradient on the smooth surrogate.
def _phase1_expgd(
    a: np.ndarray,
    n: int,
    L: int,
    budget_end: float,
    *,
    alpha_start: float = 5.0,
    alpha_end: float = 8000.0,
    eta0: float = 0.5,
    lam2_start: float = 0.5,
    lam2_end: float = 0.0,
) -> tuple[np.ndarray, float]:
    """
    Run EG optimisation until time reaches ``budget_end``.
    Returns (best_a, best_score).
    """
    best_a = a.copy()
    best_score = _objective_exact(a, n, L)

    it = 0
    # time budget for phase‑1 ≈ 55 % of total
    phase_deadline = budget_end - 0.3  # small safety margin

    while time.time() < phase_deadline:
        # --- current convolution and exact objective (for bookkeeping) ---
        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_score:
            best_score = cur_obj
            best_a = a.copy()
            print(f"[phase‑1] iter {it:,} new best O={best_score:.6f}")

        # --- annealing schedules ---
        frac = it / max(1, (phase_deadline - (budget_end - 1000.0)))  # normalised 0‑1
        alpha = alpha_start + (alpha_end - alpha_start) * frac
        lam2 = lam2_start + (lam2_end - lam2_start) * frac
        eta = eta0 / math.sqrt(it + 1.0)

        # --- smooth max (log‑sum‑exp) weights ---
        scaled = alpha * conv
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)          # numeric stability
        w = exp_scaled / exp_scaled.sum()                 # shape (2n‑1,)

        # --- gradient of smooth max term (2·corr(w,a)) ---
        grad_max = 2.0 * _correlation(w, a, L)

        # --- gradient of ℓ₂‑penalty term  λ₂·‖conv‖₂²  ( = λ₂·4·corr(conv,a) ) ---
        if lam2 != 0.0:
            grad_l2 = 4.0 * lam2 * _correlation(conv, a, L)
        else:
            grad_l2 = np.zeros_like(a)

        # --- total gradient ---
        grad = grad_max + grad_l2

        # --- exponentiated‑gradient update (multiplicative) ---
        # Clip gradient to avoid overflow in the exponent
        grad_clip = np.clip(grad, -50.0, 50.0)
        a = a * np.exp(-eta * grad_clip)

        # --- renormalise onto the probability simplex (guarantees ∑a = 1) ---
        sum_a = a.sum()
        if sum_a == 0.0:            # extremely unlikely, but guard against division by zero
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / sum_a

        # --- occasional jitter to escape flat plateaus ---
        if np.random.rand() < 0.001:
            a += np.random.randn(n) * 1e-4
            a = np.maximum(a, 0.0)
            a = a / a.sum()

        it += 1

    return best_a, best_score


# --------------------------------------------------------------
# Phase‑2: stochastic local refinement (mass‑transfer + sub‑gradient)
def _phase2_local_refine(
    start_seq: np.ndarray,
    n: int,
    L: int,
    deadline: float,
    *,
    max_iter_local: int = 150_000,
    temperature0: float = 0.04,
    subgrad_interval: int = 5,
    no_improve_limit: int = 3_000,
) -> tuple[np.ndarray, float]:
    """
    Perform a stochastic local search starting from ``start_seq``.
    Returns (best_seq, best_score) discovered before ``deadline``.
    """
    best_seq = start_seq.copy()
    best_score = _objective_exact(best_seq, n, L)

    # helper to compute exact objective fast
    def _obj(a):
        conv = _conv_fft(a, L)
        return 2.0 * n * conv.max(), conv

    # ----------------------------------------------------------------------
    # Main restart loop
    while time.time() < deadline:
        # ----- initialise a restart ----------
        if np.random.rand() < 0.6:
            a = best_seq.copy()
            a += np.random.randn(n) * 0.0015
            a = np.clip(a, 0.0, None)
            a = a / a.sum()
        else:
            a = np.random.rand(n)
            a = a / a.sum()

        cur_score, conv = _obj(a)
        it_loc = 0
        no_improve = 0
        T = temperature0

        # ----- inner optimisation loop ----------
        while time.time() < deadline and it_loc < max_iter_local:
            # ---- random mass‑transfer move ----
            i, j = np.random.randint(n, size=2)
            if i != j and a[i] > 1e-14:
                delta = a[i] * np.random.rand() * 0.15  # up to 15 % of source mass
                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta
                a_new = a_new / a_new.sum()  # stays on the simplex

                new_score, _ = _obj(a_new)

                # Metropolis acceptance
                if new_score < cur_score or np.random.rand() < math.exp((cur_score - new_score) / max(T, 1e-14)):
                    a, cur_score = a_new, new_score
                    conv = _conv_fft(a, L)
                    no_improve = 0
                    if cur_score < best_score:
                        best_score = cur_score
                        best_seq = a.copy()
                        print(f"[{time.time():.1f}] new global best O={best_score:.6f}")
                else:
                    no_improve += 1
            else:
                no_improve += 1

            # ---- sub‑gradient step on the current max entry ----
            if it_loc % subgrad_interval == 0:
                # compute sub‑gradient of max(conv)
                max_val = conv.max()
                ks = np.flatnonzero(conv == max_val)
                g = np.zeros_like(a)
                for k in ks:
                    i_start = max(0, k - (n - 1))
                    i_end = min(n - 1, k)
                    idx = np.arange(i_start, i_end + 1, dtype=int)
                    jdx = k - idx
                    g[idx] += 2.0 * a[jdx]
                if ks.size:
                    g /= ks.size
                # simple diminishing step size
                eta = 0.07 * (1.0 - it_loc / max_iter_local)
                a = a - eta * g
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                conv = _conv_fft(a, L)
                cur_score = 2.0 * n * conv.max()
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] sub‑grad improved O={best_score:.6f}")

            # ---- occasional smoothing move (tiny moving average) ----
            if it_loc % 4000 == 0 and it_loc > 0:
                kernel = np.array([0.125, 0.25, 0.25, 0.25, 0.125])
                a_smooth = np.convolve(a, kernel, mode="same")
                a = 0.85 * a + 0.15 * a_smooth
                a = a / a.sum()
                conv = _conv_fft(a, L)
                cur_score = 2.0 * n * conv.max()

            # ---- temperature decay ----
            T *= 0.9998
            it_loc += 1

            # ---- restart if stuck for a while ----
            if no_improve >= no_improve_limit:
                break

    return best_seq, best_score
