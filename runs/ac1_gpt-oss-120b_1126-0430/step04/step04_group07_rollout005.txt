"""
Propose a low‑value step‑function for the bound
    O = 2·n·max(conv(a,a)) / (sum(a)²)

The algorithm runs several short Adam‑based mirror‑descent searches on the
log‑probabilities of the coefficients.  For each candidate length we try a few
initialisations (random, U‑shaped, optionally the previous best solution) and
keep the best sequence found.  The objective is approximated by a
soft‑max/active‑set weighting of the convolution entries, exactly as in the
reference implementation, but with adaptive Adam steps, occasional jitter,
symmetry enforcement and cheap restarts.  The total runtime respects the
1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv


def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute the linear correlation  g_i = Σ_t w[i+t]·a[t]   for i=0,…,n‑1.
    Implemented via FFT.  ``w`` has length 2·n‑1.
    """
    n = a.shape[0]
    m = w.shape[0]  # = 2·n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


# ----------------------------------------------------------------------
# Core optimisation routine for a fixed length
# ----------------------------------------------------------------------
def _optimize_sequence(
    n: int,
    init_a: np.ndarray,
    budget_seconds: float,
    verbose_prefix: str = "",
) -> tuple[np.ndarray, float]:
    """
    Perform a limited‑time Adam optimisation on a probability vector
    of length `n`.  Returns the best probability vector discovered and its
    exact objective value as reported by ``evaluate_sequence``.
    """
    # Normalise the initial vector and move to log‑space
    a0 = np.maximum(0.0, init_a)
    s = a0.sum()
    if s == 0.0:
        a0 = np.full(n, 1.0 / n)
    else:
        a0 = a0 / s
    eps = 1e-12
    u = np.log(a0 + eps)   # log‑probabilities (unnormalised)

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    # Book‑keeping
    best_u = u.copy()
    best_score = np.inf
    best_iter = 0

    # Schedules (tuned for this problem)
    max_iters_est = 300_000            # a generous upper bound
    jitter_interval = 4000
    jitter_scale = 6e-4
    restart_interval = 50_000

    deadline = time.time() + budget_seconds

    for it in range(1, max_iters_est + 1):
        if time.time() >= deadline:
            break

        # --------------------------------------------------------------
        # (a) Build the probability vector from log‑variables
        u_max = u.max()
        a = np.exp(u - u_max)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, 1.0 / n)
            a_sum = 1.0
        a = a / a_sum

        # Optional symmetry enforcement (helps to keep the shape balanced)
        if it % 3 == 0:
            a = (a + a[::-1]) * 0.5
            # Refresh log‑variables to stay consistent
            u = np.log(a + eps)

        # --------------------------------------------------------------
        # (b) Convolution and objective value
        conv = _conv_self_fft(a)
        M = conv.max()
        cur_score = 2.0 * n * M      # since sum(a)=1

        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_u = u.copy()
            best_iter = it
            print(f"{verbose_prefix}[iter {it:07d}] new best O={best_score:.6f}")

        # --------------------------------------------------------------
        # (c) Scheduling parameters
        fraction = min(it / max_iters_est, 1.0)

        # temperature for the soft‑max (β)
        beta_start = 0.8
        beta_end = 20000.0
        beta = beta_start + (beta_end - beta_start) * fraction

        # base learning‑rate (η)
        eta_start = 0.6
        eta_end = 0.02
        eta = eta_start * (eta_end / eta_start) ** fraction

        # --------------------------------------------------------------
        # (d) Build the weight vector over convolution entries
        if fraction < 0.35:                     # smooth phase
            shifted = conv - M                  # ≤ 0, max is 0
            w = np.exp(beta * shifted)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.zeros_like(conv)
                w[conv.argmax()] = 1.0
            else:
                w /= w_sum
        else:                                   # hard‑max / active‑set phase
            tol = max(1e-12, M * 1e-8)
            active = np.where(conv >= M - tol)[0]
            if active.size == 0:
                w = np.zeros_like(conv)
                w[conv.argmax()] = 1.0
            else:
                w = np.zeros_like(conv)
                w[active] = 1.0 / active.size

        # --------------------------------------------------------------
        # (e) Gradient of the (smoothed) max‑convolution
        g = 2.0 * _correlation_fft(w, a)   # shape (n,)

        # --------------------------------------------------------------
        # (f) Adam update on log‑variables
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        bias_correction1 = 1.0 - beta1 ** it
        bias_correction2 = 1.0 - beta2 ** it
        m_hat = m / bias_correction1
        v_hat = v / bias_correction2
        step = eta / (np.sqrt(v_hat) + eps_adam)
        u = u - step * m_hat

        # --------------------------------------------------------------
        # (g) Random jitter to escape plateaus
        if it % jitter_interval == 0:
            u += np.random.randn(n) * jitter_scale

        # --------------------------------------------------------------
        # (h) Cheap restart from the current best solution
        if it % restart_interval == 0:
            u = best_u.copy()
            u += np.random.randn(n) * (jitter_scale * 0.5)
            print(f"{verbose_prefix}[restart] iter {it}, O={best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best probability vector found (normalised to sum=1)
    a_best = np.exp(best_u - best_u.max())
    a_best = a_best / a_best.sum()

    # Exact evaluation with the provided function (for reporting only)
    try:
        exact = evaluate_sequence(a_best.tolist())
        print(f"{verbose_prefix}[final] exact O={exact:.6f} (best iter {best_iter})")
    except Exception as exc:  # pragma: no cover
        exact = best_score
        print(f"{verbose_prefix}[final] evaluate_sequence failed:", exc)

    return a_best, float(exact)

            break

        # ---- build a few plausible initial probability vectors -----------------
        init_variants = []

        # a) random Dirichlet‑like start
        a_rand = np.random.rand(n)
        a_rand = a_rand / a_rand.sum()
        init_variants.append(a_rand)

        # b) U‑shaped (heavier on the ends)
        p = 0.6
        idx = np.arange(n, dtype=float)
        a_u = ( (idx + 1) ** (-p) + (n - idx) ** (-p) )
        a_u = a_u / a_u.sum()
        init_variants.append(a_u)

        # c) reuse the previous construction if available
        if "height_sequence_1" in globals():
            prev = np.asarray(height_sequence_1, dtype=float)
            if prev.size > n:
                prev = prev[:n]
            elif prev.size < n:
                # pad with tiny values to keep the total sum close to 1
                pad = np.full(n - prev.size, 0.0)
                prev = np.concatenate([prev, pad])
            prev = prev / max(prev.sum(), 1e-12)
            init_variants.append(prev)

        # ---- allocate a slice of the remaining budget for this length -------
        remaining_total = deadline - time.time()
        # each length gets at most ~20 % of the remaining time, capped at 90 s
        length_budget = min(remaining_total * 0.20, 90.0)

        # run each initial variant for an equal share of the length budget
        per_variant = length_budget / len(init_variants)

        for init_a in init_variants:
            if time.time() >= deadline:
                break
            seq, score = _optimize_sequence(
                n,
                init_a,
                per_variant,
                verbose_prefix=f"[len {n}] ",
            )
            if score < best_score:
                best_score = score
                best_seq = seq
                print(f"[global] new global best O={best_score:.6f} (len={n})")

    # ------------------------------------------------------------------
    # 3) Fallback – if nothing was found (should never happen) use a uniform vector
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 5000
        best_seq = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # 4) Final evaluation (the scoreboard will call evaluate_sequence again,
    #    but we print it here for diagnostics)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_seq.tolist()
