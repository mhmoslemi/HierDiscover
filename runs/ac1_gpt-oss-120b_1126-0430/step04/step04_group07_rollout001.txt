"""
Propose a candidate step‐function sequence that (approximately) minimizes

    O = 2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm is an Adam‑accelerated mirror‑descent on the log‑variables
u_i = log(a_i).  At each iteration we

1. normalize exp(u) to obtain a probability vector a,
2. compute the self‑convolution a*a via an FFT,
3. build a weight vector w that concentrates on the (soft) maximal
   convolution entries (early iterations use a smooth soft‑max, later
   iterations switch to a hard active‑set),
4. obtain the gradient ∂max/∂a by correlating w with a,
5. perform an Adam update on the log‑variables u,
6. periodically add jitter and restart from the best solution seen so far.

The schedule for the soft‑max temperature (beta) and the learning rate (eta)
is exponential, driving the optimisation from a smooth to a hard
maximisation while reducing the step size.  The best sequence found before
the time budget expires is returned as a plain ``list[float]``.  Progress is
printed to stdout.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * a using a real FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv


def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Linear correlation g_i = Σ_t w[i+t] * a[t]   for i = 0,…,n‑1.
    Implemented via FFT. ``w`` must have length 2*n‑1.
    """
    n = a.shape[0]
    m = w.shape[0]                # = 2·n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list:
    """
    Search for a non‑negative step‑function sequence that minimises the
    provided ``evaluate_sequence`` bound.  Returns the best sequence found
    as a plain ``list`` of ``float``.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    # -----------------------------------------------------------------
    # 1) Choose length and initialise log‑probabilities (u = log a)
    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=float)
        if a0.size == 0:
            raise ValueError("height_sequence_1 is empty")
        a0 = a0 / a0.sum()                     # normalise to sum 1
        n = a0.size
        # add a tiny jitter to avoid exact zeroes
        eps = 1e-12
        u = np.log(a0 + eps)
    else:
        # a moderately large length (larger than the previous 4600)
        n = 6000
        u = np.full(n, -np.log(n))            # uniform log‑probabilities

    # Adam state
    m = np.zeros_like(u)
    v = np.zeros_like(u)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0

    # Best solution tracking
    best_u = u.copy()
    best_score = np.inf
    best_iter = 0

    # -----------------------------------------------------------------
    # 2) Hyper‑parameter schedules
    max_iters_est = 300_000          # rough schedule horizon
    eta_start = 0.6
    eta_end = 0.02
    beta_start = 0.5
    beta_end = 30_000.0
    jitter_interval = 4000
    jitter_scale = 1e-3
    restart_interval = 80_000

    it = 0
    while time.time() < deadline and it < 800_000:
        it += 1

        # -------------------------------------------------------------
        # (a) Normalise to probability vector a
        u_shift = u - np.max(u)               # shift for numerical stability
        a = np.exp(u_shift)
        a_sum = a.sum()
        if a_sum == 0.0:                      # safeguard
            a = np.full_like(a, 1.0 / n)
            a_sum = 1.0
        a = a / a_sum

        # -------------------------------------------------------------
        # (b) Self‑convolution and current objective
        conv = _conv_self_fft(a)              # length 2n‑1
        M = conv.max()
        cur_score = 2.0 * n * M                # sum(a)=1 → O = 2 n M
        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_u = u.copy()
            best_iter = it
            print(f"[iter {it:07d}] new best O={best_score:.6f}")

        # -------------------------------------------------------------
        # (c) Schedules for soft‑max temperature (beta) and learning rate
        fraction = min(it / max_iters_est, 1.0)
        beta = beta_start * (beta_end / beta_start) ** fraction
        eta = eta_start * (eta_end / eta_start) ** fraction

        # -------------------------------------------------------------
        # (d) Build weight vector w over convolution entries
        #     early in the run use a smooth soft‑max, later a hard active set
        if fraction < 0.55:                               # soft‑max phase
            shifted = conv - M                            # ≤ 0, max at 0
            w = np.exp(beta * shifted)                    # safe: exponent ≤ 0
            w_sum = w.sum()
            if w_sum > 0.0:
                w /= w_sum
            else:
                # fallback – should be extremely rare
                w = np.zeros_like(conv)
                w[conv.argmax()] = 1.0
        else:                                            # active‑set phase
            tol = max(1e-12, M * 1e-8)                     # very tight tolerance
            active = np.where(conv >= M - tol)[0]
            if active.size == 0:
                w = np.zeros_like(conv)
                w[conv.argmax()] = 1.0
            else:
                w = np.zeros_like(conv)
                inv = 1.0 / active.size
                w[active] = inv

        # -------------------------------------------------------------
        # (e) Gradient of (smoothed) max‑convolution w.r.t. a
        #     dM/da ≈ 2 * correlation(w, a)
        g = 2.0 * _correlation_fft(w, a)          # shape (n,)

        # -------------------------------------------------------------
        # (f) Adam update on log‑variables u  (we minimise → step opposite gradient)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        u = u - eta * m_hat / (np.sqrt(v_hat) + eps_adam)

        # -------------------------------------------------------------
        # (g) Random jitter to avoid plateaus
        if it % jitter_interval == 0:
            u += np.random.randn(n) * jitter_scale

        # -------------------------------------------------------------
        # (h) Cheap restart from best solution if progress stalls
        if it % restart_interval == 0:
            u = best_u.copy()
            u += np.random.randn(n) * (jitter_scale * 0.5)
            print(f"[restart] iter {it}, re‑initialised from best (O={best_score:.6f})")

    # -----------------------------------------------------------------
    # 3) Return the best sequence found (normalised)
    best_a = np.exp(best_u - np.max(best_u))
    best_a = best_a / best_a.sum()
    # final evaluation (just for information, not required for the answer)
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f} "
              f"(best iteration {best_iter})")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_a.tolist()
