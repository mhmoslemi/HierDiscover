"""
propose_candidate
-----------------

A fast, robust optimiser that searches for a non‑negative step function (a
probability vector) minimising

    O = 2·n·max(conv(a, a))          (a is normalised: sum(a)=1)

The optimisation is performed on the log‑probabilities `u = log a` by an
Adam‑style exponentiated‑gradient update.  A temperature parameter `β` is used
to turn the hard max‑operator over the convolution into a smooth soft‑max
early in the run, then into an active‑set (hard‑max) later on.  Occasional
random jitter and cheap restarts from the best‑so‑far solution keep the
search from stagnating.

The algorithm works with a moderate sequence length (default 12 000).  If a
previously discovered construction is available in the global variable
`height_sequence_1` it is interpolated to the target length and used as a
warm start; otherwise the optimiser starts from a uniform distribution
with a tiny random perturbation.

All heavy lifting (full self‑convolution and correlation) is performed with
real FFTs (`np.fft.rfft`), giving an O(n log n) cost per iteration and allowing
hundreds of thousands of iterations within the 1000 s budget.

The function returns the best sequence (as a plain list of floats) that was
found before the time budget expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper FFT routines ----------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two greater than or equal to *x*."""
    return 1 << (x - 1).bit_length()


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv


def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Linear correlation g_i = Σ_t w[i+t]·a[t]  for i=0,…,n‑1.
    Implemented via FFT. ``w`` has length 2·n‑1.
    """
    n = a.shape[0]
    m = w.shape[0]                # = 2·n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


# ----------------------------------------------------------------------
# Main search routine ----------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the bound
    evaluated by ``evaluate_sequence``.  Returns the best sequence as a list
    of floats.
    """
    np.random.seed(seed)
    # safety margin to avoid overrunning the time limit
    deadline = time.time() + budget_s - 0.5

    # ------------------------------------------------------------------
    # 1) Choose a working length and initialise log‑probabilities.
    #    If a previous construction exists, interpolate it to the target
    #    length and use it as a warm start.
    TARGET_N = 12000                     # a size that still fits comfortably
    eps = 1e-12

    if 'height_sequence_1' in globals():
        old = np.asarray(height_sequence_1, dtype=float)
        if old.size == 0:
            # fallback to uniform
            n = TARGET_N
            u = np.full(n, -np.log(n))
        else:
            # interpolate the old sequence to TARGET_N points
            old = old / old.sum()
            x_old = np.linspace(0.0, 1.0, old.size)
            x_new = np.linspace(0.0, 1.0, TARGET_N)
            a0 = np.interp(x_new, x_old, old)
            a0 = a0 / a0.sum()
            n = a0.size
            u = np.log(a0 + eps)
    else:
        n = TARGET_N
        u = np.full(n, -np.log(n))

    # ------------------------------------------------------------------
    # 2) Adam state
    m = np.zeros_like(u)          # first moment
    v = np.zeros_like(u)          # second moment
    beta1 = 0.9
    beta2 = 0.999
    adam_eps = 1e-8

    # ------------------------------------------------------------------
    # 3) Global best bookkeeping
    best_u = u.copy()
    best_score = np.inf
    best_iter = 0
    total_iters = 0

    # ------------------------------------------------------------------
    # 4) Hyper‑parameter schedules
    max_iters_est = 5_00_000          # used only for the β / η schedule
    beta_start = 4.0
    beta_end   = 20000.0
    eta_start  = 0.5                # initial step‑size (log‑domain)
    eta_end    = 0.03               # final step‑size
    jitter_interval = 5000
    jitter_scale = 5e-4
    restart_interval = 30_000

    # ------------------------------------------------------------------
    # 5) Main optimisation loop
    while time.time() < deadline:
        total_iters += 1

        # ----- (a) build the probability vector -----------------------
        a = np.exp(u - np.max(u))        # stable softmax numerator
        a_sum = a.sum()
        if a_sum == 0.0:                 # numerical safety
            a = np.full_like(a, 1.0 / n)
            a_sum = 1.0
        a = a / a_sum

        # ----- (b) self‑convolution and current score ----------------
        conv = _conv_self_fft(a)
        M = conv.max()
        cur_score = 2.0 * n * M           # because sum(a)=1

        # ----- (c) update the global best ----------------------------
        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_u = u.copy()
            best_iter = total_iters
            print(f"[iter {total_iters:07d}] new best O={best_score:.6f}")

        # ----- (d) schedule parameters --------------------------------
        fraction = min(total_iters / max_iters_est, 1.0)
        beta = beta_start + (beta_end - beta_start) * fraction
        eta = eta_start * (eta_end / eta_start) ** fraction

        # ----- (e) build the weight vector over convolution entries ----
        if fraction < 0.4:                           # soft‑max phase
            shifted = conv - M                       # ≤ 0, max is 0
            w = np.exp(beta * shifted)
            w_sum = w.sum()
            if w_sum == 0.0:
                # fallback – should not happen
                w = np.zeros_like(conv)
                w[conv.argmax()] = 1.0
            else:
                w = w / w_sum
        else:                                        # hard‑max / active set
            tol = max(1e-12, M * 1e-8)
            active = np.where(conv >= M - tol)[0]
            if active.size == 0:
                w = np.zeros_like(conv)
                w[conv.argmax()] = 1.0
            else:
                w = np.zeros_like(conv)
                w[active] = 1.0 / active.size

        # ----- (f) gradient of the (smoothed) max‑convolution ---------
        g = 2.0 * _correlation_fft(w, a)            # shape (n,)

        # ----- (g) Adam update on log‑variables -----------------------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        # bias‑corrected moments
        bc1 = 1.0 - beta1 ** total_iters
        bc2 = 1.0 - beta2 ** total_iters
        m_hat = m / bc1
        v_hat = v / bc2

        u = u - eta * m_hat / (np.sqrt(v_hat) + adam_eps)

        # ----- (h) occasional jitter (helps escape plateaus) ---------
        if total_iters % jitter_interval == 0:
            u += np.random.randn(n) * jitter_scale

        # ----- (i) cheap restart from the best‑so‑far solution ---------
        if total_iters % restart_interval == 0:
            u = best_u.copy()
            u += np.random.randn(n) * (jitter_scale * 0.5)
            m.fill(0.0)
            v.fill(0.0)
            print(f"[restart] iter {total_iters}, re‑initialised from best "
                  f"(O={best_score:.6f})")

        # ----- (j) break if time is about to run out -----------------
        if time.time() >= deadline - 0.2:
            break

    # ------------------------------------------------------------------
    # 6) Final normalisation and evaluation with the provided function
    best_seq = np.exp(best_u - np.max(best_u))
    best_seq = best_seq / best_seq.sum()
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f} "
              f"(best iteration {best_iter})")
    except Exception as exc:                               # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_seq.tolist()
