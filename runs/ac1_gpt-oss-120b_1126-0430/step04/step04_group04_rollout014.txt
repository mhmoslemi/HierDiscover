#!/usr/bin/env python
"""
Gradient–based optimisation of a step‑function.

We minimise
        2·n·max_k (a*a)[k] / (∑ a_i)²        (a_i ≥ 0, ∑ a_i = 1)

Instead of the top‑K sub‑gradient used previously we employ a
*soft–max* approximation of the max‑operator.  For a temperature τ
the smoothed objective is

        φ(a) = τ·log ∑_k exp( (a*a)[k] / τ ),

whose gradient can be written in closed form and evaluated by two
FFT based convolutions only.  The iteration is a mirror‑descent /
exponentiated‑gradient step on the probability simplex, which avoids
the O(n log n) simplex‑projection in every iteration.  Several
initialisations (uniform, power‑law, Dirichlet draws) are tried and
the best candidate is kept.  A simple restart mechanism perturbs the
current best when progress stalls.

The routine respects the given time budget (default 1000 s) and always
returns a feasible non‑negative sequence.  The final exact value is
obtained by calling the provided ``evaluate_sequence`` function.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, Σx = 1}."""
    # Implements the algorithm of Wang & Carreira‑Perpinán (2013)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float) -> np.ndarray:
    """Return a normalised power‑law a_i ∝ (i+1)^‑p (∑a_i = 1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = i ** (-p)
    seq /= seq.sum()
    return seq


def _dirichlet_sequence(n: int, conc: float) -> np.ndarray:
    """Draw a Dirichlet vector with concentration ``conc`` and normalise."""
    g = np.random.gamma(conc, 1.0, size=n)
    seq = g / g.sum()
    return seq


def _cheap_score_and_conv(a: np.ndarray, fft_len: int) -> tuple[float, np.ndarray]:
    """Return cheap objective 2·n·max(conv) and the full convolution a*a."""
    n = a.shape[0]
    Fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(Fa * Fa).real[: 2 * n - 1]   # linear convolution
    maxb = conv.max()
    return 2.0 * n * maxb, conv


def _smooth_gradient(a: np.ndarray,
                     conv: np.ndarray,
                     maxb: float,
                     tau: float,
                     fft_len: int) -> np.ndarray:
    """
    Gradient of τ·log∑exp((a*a)[k]/τ) w.r.t. a.
    Implements the closed form 2·(a * w_rev) where w are soft‑max weights.
    """
    n = a.shape[0]
    M = 2 * n - 2                               # last index of conv
    # soft‑max weights (numerically stable)
    shifted = conv - maxb                       # ≤ 0
    w = np.exp(shifted / tau)
    w /= w.sum()
    # reversed weight vector
    w_rev = w[::-1]

    Fa = np.fft.fft(a, fft_len)
    Fwr = np.fft.fft(w_rev, fft_len)
    conv_rev = np.fft.ifft(Fa * Fwr).real       # length = fft_len

    # gradient_i = 2 * conv_rev[M - i]
    idx = M - np.arange(n)
    grad = 2.0 * conv_rev[idx]
    return grad


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The algorithm:

    1. Choose a working length ``n`` (≈10 k, bounded by 5‑k … 25‑k).
    2. Initialise several candidates (uniform, power‑law, Dirichlet draws);
       keep the best according to the cheap objective.
    3. Perform a long mirror‑descent run using a soft‑max smoothed max.
       The update is an exponentiated‑gradient step on the simplex.
    4. Restart with a small random perturbation when progress stalls.
    5. Return the best sequence and report its exact score.

    The routine always respects the time budget and returns a feasible
    sequence (non‑negative, sum > 0).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0  # a safety margin for the final eval

    # --------------------------------------------------------------
    # Choose a working length.
    # --------------------------------------------------------------
    try:
        prev = height_sequence_1          # noqa: F821  (global if present)
        prev_arr = np.asarray(prev, dtype=np.float64)
        if prev_arr.sum() > 0.0:
            prev_arr = prev_arr / prev_arr.sum()
        else:
            prev_arr = None
    except Exception:
        prev_arr = None

    # Desired length – a compromise between quality and speed
    n = 10000
    # If a previous sequence exists and is not too short/long, reuse its size
    if prev_arr is not None and 5000 <= len(prev_arr) <= 25000:
        n = len(prev_arr)

    # --------------------------------------------------------------
    # FFT length – next power of two ≥ 3·n‑2 (needed for the gradient step)
    # --------------------------------------------------------------
    fft_len = 1 << ((3 * n - 2).bit_length())

    # --------------------------------------------------------------
    # 0. Initialise with a small scan of power‑law shapes and a few
    #    Dirichlet draws – keep the best cheap candidate.
    # --------------------------------------------------------------
    best_a = None
    best_val = float("inf")
    best_conv = None

    # uniform (p = 0) is included in the grid for completeness
    p_grid = np.linspace(0.3, 1.2, 30)
    for p in p_grid:
        if time.time() >= deadline:
            break
        cand = _power_law_sequence(n, p)
        val, conv = _cheap_score_and_conv(cand, fft_len)
        if val < best_val:
            best_val, best_a, best_conv = val, cand.copy(), conv.copy()

    # a few random Dirichlet samples with a wide range of concentrations
    for _ in range(30):
        if time.time() >= deadline:
            break
        conc = 10 ** np.random.uniform(-3.0, 0.5)      # ~0.001 … 3.2
        cand = _dirichlet_sequence(n, conc)
        val, conv = _cheap_score_and_conv(cand, fft_len)
        if val < best_val:
            best_val, best_a, best_conv = val, cand.copy(), conv.copy()

    # previous construction, if compatible
    if prev_arr is not None and prev_arr.size == n:
        val, conv = _cheap_score_and_conv(prev_arr, fft_len)
        if val < best_val:
            best_val, best_a, best_conv = val, prev_arr.copy(), conv.copy()

    print(f"[Init] best cheap score = {best_val:.6f} (n = {n})")

    # --------------------------------------------------------------
    # 1. Mirror‑descent with soft‑max smoothing.
    # --------------------------------------------------------------
    a = best_a.copy()
    momentum = np.zeros_like(a)          # not used for EG, kept for possible restarts
    best_iter_a = a.copy()
    best_iter_val = best_val
    last_improve_iter = 0

    iter_cnt = 0
    max_iter = 2_000_000                # hard cap – the loop also stops at deadline

    # hyper‑parameters (tuned empirically for the 1000 s budget)
    eta0 = 1.0                           # base step‑size for exponentiated‑gradient
    temp_factor = 0.5                    # controls initial temperature (relative to max)
    stagnation_limit = 150_000           # restart after this many iterations without improvement

    while time.time() < deadline and iter_cnt < max_iter:
        iter_cnt += 1

        # ----- compute full convolution and cheap score -----
        Fa = np.fft.fft(a, fft_len)
        conv_full = np.fft.ifft(Fa * Fa).real[: 2 * n - 1]
        maxb = conv_full.max()
        cheap = 2.0 * n * maxb

        # ----- keep best cheap score -----
        if cheap < best_iter_val - 1e-12:
            best_iter_val = cheap
            best_iter_a = a.copy()
            last_improve_iter = iter_cnt
            print(f"[Iter {iter_cnt:07d}] NEW BEST cheap = {cheap:.6f}")

        # ----- temperature schedule (decreases slowly) -----
        # start with a fraction of the current maximum, then decay with sqrt(iter)
        tau = maxb * max(temp_factor / math.sqrt(iter_cnt + 1), 1e-8)

        # ----- gradient of the smoothed max -----
        grad = _smooth_gradient(a, conv_full, maxb, tau, fft_len)

        # ----- exponentiated‑gradient step (mirror descent on simplex) -----
        # Clip the gradient to avoid overflow in the exponentials
        grad_clip = np.clip(grad, -10.0, 10.0)
        eta = eta0 / math.sqrt(iter_cnt + 1)
        mult = np.exp(-eta * grad_clip)
        a = a * mult
        s = a.sum()
        if s == 0.0 or np.isnan(s):
            # rare numerical disaster – fall back to uniform
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        # ----- simple stagnation detection and restart -----
        if iter_cnt - last_improve_iter > stagnation_limit:
            # perturb the best known vector slightly and continue
            perturb = 0.02 * np.random.rand(n)
            a = best_iter_a + perturb
            a = _simplex_projection(a)   # cheap projection for the restart
            last_improve_iter = iter_cnt
            print(f"[Iter {iter_cnt:07d}] Restart (perturbation) due to stagnation.")

    # --------------------------------------------------------------
    # 2. Final exact evaluation.
    # --------------------------------------------------------------
    final_seq = best_iter_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:      # defensive – should never happen
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # --------------------------------------------------------------
    # 3. Return the best sequence found.
    # --------------------------------------------------------------
    return final_seq
