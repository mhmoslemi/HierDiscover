    (`sum = 1`) and evaluated with a fast FFT‑based convolution.
    The best candidate (lowest exact objective) is kept.

2.  **Adam optimisation on the simplex** – starting from the best
    candidate we run an Adam optimiser in log‑it space (softmax
    parameterisation).  A smooth “soft‑max’’ approximation of the
    max‑operator is used (controlled by an annealing parameter `β`
    that grows exponentially).  The gradient is obtained via FFT‑based
    correlation, the learning‐rate decays exponentially and we occasionally
    jitter the logits to escape plateaus.  The optimiser runs for the
    bulk of the remaining time budget.

3.  **Tiny local search** – a few random mass‑transfer moves are tried
    (move a small random fraction of the mass from one index to another)
    to squeeze out any remaining improvement.

All helper functions are defined at the top level; the entry point
`propose_candidate` is the only function the evaluation harness calls.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (real‑valued) using an FFT of length power‑of‑2."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv


def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation of w (length 2n‑1) with a (length n) → result length n.
    Implements   g_i = Σ_t w[i+t]·a[t] .
    """
    n = a.shape[0]
    m = w.shape[0]                     # = 2n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))  (assumes sum(a)=1)."""
    n = a.size
    conv = _conv_self_fft(a)
    return 2.0 * n * conv.max()


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


def _run_opt(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
) -> tuple[list[float], float]:
    """
    Adam optimisation of a length‑`n` vector for at most ``time_limit`` seconds.
    The vector lives on the probability simplex (soft‑max of logits).
    Returns the best sequence found (as a plain list) and its exact objective.
    """

    start = time.time()
    rng = np.random.default_rng(seed)

    # -------- initialise ----------
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
        a = a / a.sum()
    else:
        a = rng.random(n)
        a = a / a.sum()

    # work in log‑it space (unconstrained)
    x = np.log(a + 1e-16)

    m = np.zeros_like(x)          # Adam first moment
    v = np.zeros_like(x)          # Adam second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # -------- schedules ----------
    beta_start = 0.5
    beta_end = 8000.0
    lr_start = 0.15
    lr_end = 0.001
    max_iters = 400_000

    best_a = a.copy()
    best_obj = _objective(best_a)

    it = 0
    while time.time() - start < time_limit and it < max_iters:
        it += 1

        # current probabilities
        a = _softmax(x)

        # exact objective (fast via FFT)
        conv = _conv_self_fft(a)
        M = conv.max()
        obj = 2.0 * n * M
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[n={n:5d} it={it:06d}] new best O={best_obj:.6f}")

        # ----- schedule parameters -----
        t = min(it / max_iters, 1.0)
        beta = beta_start * np.exp(np.log(beta_end / beta_start) * t)
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t)

        # ----- smooth max weights -----
        if beta > 4000.0:                       # very sharp → true arg‑max
            kmax = int(np.argmax(conv))
            w = np.zeros_like(conv)
            w[kmax] = 1.0
        else:
            shifted = conv - M                  # ≤ 0
            w = np.exp(beta * shifted)
            w /= w.sum()

        # ----- gradient w.r.t a (smooth max) -----
        g_a = 2.0 * _correlation_fft(w, a)       # shape (n,)

        # ----- transform gradient to log‑it space (natural gradient) -----
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to break plateaus
        if it % 6000 == 0:
            x += rng.normal(scale=0.02, size=n)

        # restart from the current best every so often
        if it % 70_000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.005, size=n)
            m.fill(0.0)
            v.fill(0.0)
            print(f"[n={n}] restart at iter {it}")

    # -------- compress negligible entries ----------
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = _objective(best_a)
        print(f"[compress] reduced length → {best_a.size}, O={best_obj:.6f}")

    # -------- tiny local random search ----------
    remaining = time_limit - (time.time() - start)
    if remaining > 0.5:
        a_loc = best_a.copy()
        obj_loc = best_obj
        n_loc = a_loc.size
        rng_local = np.random.default_rng(seed + 12345)
        local_end = time.time() + remaining * 0.3
        while time.time() < local_end:
            i, j = rng_local.choice(n_loc, size=2, replace=False)
            delta = a_loc[i] * rng_local.random() * 0.02
            a_new = a_loc.copy()
            a_new[i] -= delta
            a_new[j] += delta
            conv_new = _conv_self_fft(a_new)
            obj_new = 2.0 * n_loc * conv_new.max()
            if obj_new < obj_loc:
                a_loc, obj_loc = a_new, obj_new
                if obj_new < best_obj:
                    best_a, best_obj = a_new, obj_new
        print(f"[local] final local O={best_obj:.6f}")

    return best_a.tolist(), best_obj


# ----------------------------------------------------------------------
def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **_ignore,
) -> list[float]:
    """
    Main entry point.
    1) Scan a few symmetric power‑law shapes for several lengths.
    2) Optimise the best shape with Adam (log‑it softmax).
    3) A short random‑mass‑transfer local search.
    The returned list satisfies the constraints of ``evaluate_sequence``.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------- stage 1 : parametric sweep --------------------
    lengths = [2000, 4000, 8000, 12000, 16000, 20000]

    # include the length of a previously found solution if available
    if 'height_sequence_1' in globals():
        prev = np.asarray(height_sequence_1, dtype=float)
        prev = np.clip(prev, 0.0, None)
        if prev.sum() >= 0.01:
            n_prev = prev.size
            if n_prev not in lengths:
                lengths.insert(0, n_prev)

    exponents = [0.25, 0.35, 0.45, 0.55, 0.70, 0.90, 1.10]

    best_seq = None
    best_obj = float('inf')
    best_n = None

    for n in lengths:
        # stop early if we run out of time
        if time.time() > deadline - 5.0:
            break
        idx = np.arange(1, n + 1, dtype=float)
        prod = idx * (n + 1 - idx)          # i·(n+1-i)
        for p in exponents:
            a = prod ** (-p)
            a_sum = a.sum()
            if a_sum == 0.0:
                continue
            a /= a_sum                         # normalize to sum = 1
            # fast exact objective (no clipping needed)
            obj = _objective(a)
            if obj < best_obj:
                best_obj = obj
                best_seq = a.tolist()
                best_n = n
                print(f"[init] n={n:5d} p={p:4.2f} → O={obj:.6f}")

    # --------------------- stage 2 : Adam optimisation --------------------
    if best_seq is None:
        # as a fallback start from a random uniform vector
        n = lengths[0]
        best_seq = np.random.rand(n).tolist()
        best_n = n
        best_obj = _objective(np.asarray(best_seq))

    # remaining time for the main optimisation
    remaining = deadline - time.time()
    if remaining > 5.0:
        # give the optimiser about 80 % of the remaining budget
        opt_time = remaining * 0.80
        seq_opt, obj_opt = _run_opt(
            n=best_n,
            time_limit=opt_time,
            init_a=best_seq,
            seed=seed + 777,
        )
        if obj_opt < best_obj:
            best_obj = obj_opt
            best_seq = seq_opt
            print(f"[opt] improved overall O={best_obj:.6f} (n={best_n})")

    # --------------------- stage 3 : tiny random local search ------------
    remaining = deadline - time.time()
    if remaining > 1.0:
        a = np.array(best_seq, dtype=float)
        obj = best_obj
        n = a.size
        rng = np.random.default_rng(seed + 9999)
        while time.time() < deadline - 0.1:
            i, j = rng.choice(n, size=2, replace=False)
            delta = a[i] * rng.random() * 0.01   # move ≤1 % of mass
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            conv_new = _conv_self_fft(a_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < obj:
                a, obj = a_new, obj_new
                best_seq = a.tolist()
                best_obj = obj
                print(f"[local] new O={obj:.6f}")
        print(f"[final‑local] O={best_obj:.6f}")

    # ----------------------------------------------------------------------
    # Final verification (optional – any exception is ignored)
    try:
        final_score = evaluate_sequence(best_seq)
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:                      # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_seq
