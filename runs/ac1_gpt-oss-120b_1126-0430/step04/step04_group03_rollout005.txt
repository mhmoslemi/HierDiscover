def _exp_update_momentum(a, g, m, eta, mu):
    m = mu*m + (1-mu)*g
    log_a = np.log(a + 1e-15) - eta * m
    log_a -= np.max(log_a)
    a_new = np.exp(log_a)
    a_new /= a_new.sum()
    return a_new, m
```

def _optimize_expg(a0, max_iters, eta0, mu, deadline):
    a = a0.copy()

    β∈[0,30] for a fixed length n (chosen as a power‑of‑two to keep the
    FFT fast).  Keep the best shape.

2️⃣  **Fine random refinement** – perturb the best (α,β) together with a
    tiny exponential tail γ, evaluate many candidates until time runs
    out.

3️⃣  **Multi‑start mirror‑descent** – initialise a set of random
    Dirichlet vectors with different concentration parameters, then run
    an exponentiated‑gradient (mirror) descent with momentum on the true
    objective.  Each run is limited by the remaining time; the best
    result among all starts is retained.

4️⃣  **Simulated‑annealing polishing** – starting from the current best,
    perform a temperature‑controlled random walk on the simplex,
    accepting uphill moves with a probability that decays with the
    temperature.

5️⃣  **Final exact check** – call the provided ``evaluate_sequence``
    routine on the best candidate and return it as a plain ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x≥0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a power‑of‑two FFT (real output)."""
    n = a.size
    size = 1 << ((2 * n - 2).bit_length())   # next power of two ≥ 2n‑1
    A = np.fft.fft(a, n=size)
    conv = np.fft.ifft(A * A).real[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def _objective_norm1(a: np.ndarray) -> float:
    """Objective 2·n·max(conv) assuming Σa = 1."""
    n = a.size
    conv = _conv_fft(a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) w.r.t. a (on the simplex).
    For every index k attaining the maximal convolution value the
    gradient contribution equals 2·a_{k‑i} for each feasible i.
    """
    n = a.size
    conv = _conv_fft(a)
    max_val = conv.max()
    ks = np.where(np.abs(conv - max_val) < 1e-14)[0]
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
def _exp_update_momentum(a: np.ndarray,
                         g: np.ndarray,
                         m: np.ndarray,
                         eta: float,
                         mu: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Exponentiated‑gradient update with exponential‑moving‑average momentum.
    Returns the updated vector and the new momentum term.
    """
    # momentum update
    m = mu * m + (1.0 - mu) * g
    # log‑space mirror step
    log_a = np.log(a + 1e-15) - eta * m
    log_a -= np.max(log_a)                # numerical stabilisation
    a_new = np.exp(log_a)
    a_new /= a_new.sum()
    return a_new, m


# ----------------------------------------------------------------------
def _optimize_exponential(a0: np.ndarray,
                          max_iters: int,
                          eta0: float,
                          mu: float,
                          deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent optimisation with momentum.
    Returns the best vector seen and its objective value.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    best_a = a.copy()
    best_score = _objective_norm1(a)

    n = a.size
    patience = max_iters // 20
    no_improve = 0

    for t in range(max_iters):
        if time.time() >= deadline:
            break

        g = _subgradient_max_conv(a)
        eta = eta0 * (1.0 - t / max_iters)          # linear decay
        a, m = _exp_update_momentum(a, g, m, eta, mu)

        cur = _objective_norm1(a)
        if cur < best_score - 1e-12:
            best_score = cur
            best_a = a.copy()
            no_improve = 0
            print(f"[opt] iter {t:6d}  new best O={best_score:.6f}")
        else:
            no_improve += 1
            if no_improve >= patience:
                # small jitter to escape flat plateaus
                jitter = np.random.randn(n) * 1e-4
                a = best_a + jitter
                a = np.clip(a, 0.0, None)
                a /= a.sum()
                m = np.zeros_like(a)
                no_improve = 0

    return best_a, best_score


# ----------------------------------------------------------------------
def _shape_power(alpha: float, beta: float, n: int) -> np.ndarray:
    """Monotone decreasing shape a_i ∝ (i+β)^(-α), normalised to sum = 1."""
    i = np.arange(1, n + 1, dtype=float)
    a = (i + beta) ** (-alpha)
    s = a.sum()
    if s == 0.0:
        a.fill(1.0 / n)
    else:
        a /= s
    return a


def _shape_power_exp(alpha: float, beta: float, gamma: float, n: int) -> np.ndarray:
    """
    Power‑law with exponential tail:
        a_i ∝ (i+β)^(-α)·exp(-γ·(i‑1)/n)
    """
    i = np.arange(1, n + 1, dtype=float)
    a = (i + beta) ** (-alpha) * np.exp(-gamma * (i - 1) / n)
    s = a.sum()
    if s == 0.0:
        a.fill(1.0 / n)
    else:
        a /= s
    return a


def _shape_gaussian(sigma: float, n: int) -> np.ndarray:
    """Centered Gaussian (truncated) normalised to unit sum."""
    x = np.arange(n, dtype=float) - (n - 1) / 2.0
    a = np.exp(-0.5 * (x / sigma) ** 2)
    s = a.sum()
    a /= s
    return a


def _shape_triangle(n: int) -> np.ndarray:
    """Symmetric triangular shape (peak in the centre) normalised."""
    half = (n + 1) // 2
    left = np.linspace(1, half, half, endpoint=False)
    right = left[::-1] if n % 2 == 0 else np.append(left[:-1][::-1], 0)
    a = np.concatenate([left, right])[:n]
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising the bound.
    Returns the best sequence as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # a small safety margin

    # ------------------------------------------------------------------
    # Problem size – a power of two works nicely with the FFT.
    n = 4096

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1️⃣  Coarse power‑law grid search
    alphas = np.linspace(0.30, 0.80, 51)      # 51 × 51 ≈ 2600 evaluations
    betas  = np.linspace(0.0, 30.0, 51)

    print("[grid] start coarse sweep")
    for a in alphas:
        for b in betas:
            seq = _shape_power(a, b, n)
            score = _objective_norm1(seq)
            if score < best_score:
                best_score = score
                best_seq = seq
                best_alpha, best_beta = a, b
    print(f"[grid] best O={best_score:.6f} (α={best_alpha:.4f}, β={best_beta:.4f})")

    # ------------------------------------------------------------------
    # 2️⃣  Fine random refinement of (α,β,γ)
    print("[refine] random local search around best power‑law")
    while time.time() < deadline - 2.0:
        a = best_alpha + np.random.randn() * 0.018
        b = best_beta  + np.random.randn() * 0.4
        if a <= 0 or b < 0:
            continue
        g = np.abs(np.random.randn()) * 0.25   # γ ∈ [0,~0.5]
        seq = _shape_power_exp(a, b, g, n)
        score = _objective_norm1(seq)
        if score < best_score:
            best_score = score
            best_seq = seq
            best_alpha, best_beta, best_gamma = a, b, g
            print(f"[refine] new best O={best_score:.6f} (α={a:.4f}, β={b:.4f}, γ={g:.4f})")

    # ------------------------------------------------------------------
    # 3️⃣  Incorporate any previously discovered construction
    if 'height_sequence_1' in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=float)
            if prev.size > 0:
                prev = np.maximum(prev, 0.0)
                prev /= prev.sum()
                prev_score = _objective_norm1(prev)
                if prev_score < best_score:
                    best_seq = prev
                    best_score = prev_score
                    print(f"[prev] using previous construction O={best_score:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 4️⃣  Multi‑start mirror‑descent (exponentiated‑gradient with momentum)
    print("[opt] multi‑start mirror descent")
    # allocate the remaining budget proportionally to the number of starts
    while time.time() < deadline - 5.0:
        # pick a random Dirichlet concentration; small values give sparse starts,
        # large values give near‑uniform starts.
        conc = 10 ** np.random.uniform(-1.5, 1.5)   # roughly 0.03 … 30
        a0 = np.random.dirichlet(np.full(n, conc))
        # limit the number of iterations to what fits the remaining time
        secs_left = deadline - time.time()
        max_iters = max(200, int(min(5000, secs_left * 150)))  # ~150 iters per second
        a_opt, score_opt = _optimize_exponential(a0,
                                                 max_iters=max_iters,
                                                 eta0=0.7,
                                                 mu=0.85,
                                                 deadline=deadline)
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt
            print(f"[opt] new global best O={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Simulated‑annealing polishing
    if time.time() < deadline - 2.0:
        print("[anneal] start simulated‑annealing polishing")
        a = best_seq.copy()
        cur_score = best_score
        # temperature schedule
        T0 = 2e-2
        T_end = 5e-6
        total_steps = 30000
        for step in range(total_steps):
            if time.time() >= deadline:
                break
            # geometric cooling
            T = T0 * ((T_end / T0) ** (step / total_steps))
            # propose a neighbour by adding zero‑mean noise
            delta = np.random.randn(n) * (0.02 * (1.0 - step / total_steps))
            a_new = a + delta
            a_new = np.clip(a_new, 0.0, None)
            if a_new.sum() == 0.0:
                continue
            a_new /= a_new.sum()
            new_score = _objective_norm1(a_new)
            # Metropolis acceptance
            if new_score < cur_score or np.random.rand() < np.exp(-(new_score - cur_score) / max(T, 1e-12)):
                a = a_new
                cur_score = new_score
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(f"[anneal] new best O={best_score:.6f}")

        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:  # pragma: no cover
        print("[final] evaluate_sequence failed:", e)
        final_score = None

    return best_seq.tolist()
