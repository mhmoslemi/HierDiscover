def _subgradient_max_conv(a, conv):
    # conv shape 2n-1
    k = int(np.argmax(conv))
    n = a.shape[0]
    g = np.zeros_like(a)
    # part1: a_{k-i}
    if k < n:
        # i from 0..k
        # g_i += a_{k-i}
        # This is reversed slice a[0:k+1] reversed.
        g[:k+1] = a[k::-1]
    else:
        # k >= n
        offset = k - n + 1
        # i from offset..n-1
        # g_i += a_{k - i}
        # For i = offset, k-i = n-1; for i = n-1, k-i = k-(n-1)=offset
        # So reversed slice a[n-1:offset-1:-1]
        g[offset:] = a[n-1:offset-1:-1]   # Works if offset>0
    # part2: a_{i - k}
    if k < n:
        # i from k..n-1
        # g_i += a[i - k]
        g[k:] += a[:n - k]
    # else part2 no contribution
    return g, k
```

"""
Exponential‑gradient minimiser for 2·n·max(conv(a)).

The objective for a non‑negative step function a of length n (with sum a>0) is

    J(a) = 2·n·max_k (a*a)[k]   (evaluated by `evaluate_sequence`).

Because scaling cancels, we restrict a to the probability simplex (∑a=1).
At each iteration we
  1. compute the full self‑convolution by FFT,
  2. locate the index k* attaining the maximum,
  3. obtain the sub‑gradient of (a*a)[k*] w.r.t. a
       g_i = a_{k*‑i} + a_{i‑k*}   (zero when an index is out of range),
  4. perform an exponentiated‑gradient update
       a ← a·exp(−η·g)  and renormalise,
  5. keep the best sequence seen so far.

The step size η decays as η₀/√t.  Random jitter is injected periodically
to escape plateaux and a small amount of additive noise is applied after
long stagnation periods.  The optimiser is run many times from
different random initialisations (including the provided global
`height_sequence_1` when available) and the best result is finally
checked with `evaluate_sequence`.

The whole routine respects the supplied time budget (≈ 1000 s) and
returns a list of non‑negative heights.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers (top level – no closures)
# ----------------------------------------------------------------------


def _fft_len_for_conv(n: int) -> int:
    """
    Smallest power‑of‑two ≥ 2·n‑1, sufficient for a linear convolution
    of two length‑n vectors.
    """
    return 1 << ((2 * n - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using a real FFT of length L.
    Returns the convolution of length 2·n‑1 (truncated).
    """
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * a.shape[0] - 1]
    return conv


def _max_conv_and_subgrad(a: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Compute the maximum entry of the self‑convolution of a and the
    sub‑gradient of that entry (with respect to a).

    Parameters
    ----------
    a : np.ndarray
        Probability vector (sum = 1, non‑negative).

    Returns
    -------
    max_conv : float
        Maximum value of (a*a)[k] over all k.
    g : np.ndarray
        Sub‑gradient of the active convolution entry.
    """
    n = a.shape[0]
    L = _fft_len_for_conv(n)

    # convolution and the active index
    conv = _conv_self_fft(a, L)
    k = int(np.argmax(conv))
    max_conv = float(conv[k])

    # sub‑gradient of c_k = Σ_i a_i a_{k-i}
    g = np.zeros_like(a)

    if k < n:
        # term a_{k-i} for i=0..k  (reversed prefix)
        g[: k + 1] = a[k :: -1]          # a[k], a[k‑1], …, a[0]
        # term a_{i-k} for i=k..n‑1  (suffix)
        g[k:] += a[: n - k]
    else:
        # k ≥ n  → only the first term contributes
        offset = k - n + 1                # first i where k‑i becomes valid
        # i runs from offset … n‑1  → a_{k-i} = a[n‑1 … offset] (reversed)
        g[offset:] = a[n - 1 : offset - 1 : -1]

    return max_conv, g


# ----------------------------------------------------------------------
# Simple exponentiated‑gradient optimiser on the simplex
# ----------------------------------------------------------------------


def _eg_optimize(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Run exponentiated‑gradient descent on the simplex starting from `a0`.

    The routine stops either when the time limit is reached or when a
    long period without improvement elapses.  It returns the best
    sequence found (still a probability vector) together with its cheap
    score 2·n·max(conv).
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = float("inf")
    last_improve = 0
    eta0 = 0.5                     # initial step size
    it = 0

    while time.time() < deadline:
        it += 1

        # --- convolution, max entry and sub‑gradient ---
        max_conv, g = _max_conv_and_subgrad(a)
        cheap_score = 2.0 * n * max_conv

        # --- keep the best ever seen ---
        if cheap_score < best_score:
            best_score = cheap_score
            best_a = a.copy()
            last_improve = it
            if it % 5000 == 0:
                print(f"[Iter {it:07d}] NEW BEST cheap = {cheap_score:.6f}")

        # --- step size (inverse‑sqrt decay) ---
        eta = eta0 / (math.sqrt(it) + 1.0)

        # --- exponentiated‑gradient update (mirror descent w.r.t. KL) ---
        a = a * np.exp(-eta * g)

        # numerical safety – avoid exact zeros before renormalisation
        a = np.maximum(a, 1e-15)

        # --- project back onto the simplex ---
        s = a.sum()
        if s == 0.0 or np.isnan(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # --- occasional random jitter to escape plateaux ---
        if it % 20000 == 0:
            a += 1e-5 * np.random.randn(n)
            a = np.clip(a, 0.0, None)
            a /= a.sum()

        # --- early termination on long stagnation ---
        if it - last_improve > 400_000:
            # give the outer controller a chance to restart
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple linear interpolation of a probability vector (used for
# enlarging/re‑shaping the support when a global seed of different
# length is available).
# ----------------------------------------------------------------------


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    total = new_a.sum()
    if total == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= total
    return new_a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights (a step function) that aims to
    minimise `evaluate_sequence`.  The search runs for at most
    ``budget_s`` seconds, respects the time budget, and uses a
    lightweight exponentiated‑gradient optimiser with several random
    restarts.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0   # safety margin for final evaluation

    # ------------------------------------------------------------------
    # 0. Choose a working support size (the length of the step function)
    # ------------------------------------------------------------------
    TARGET_N = 20000                     # a reasonable size for the budget
    try:
        # Prefer the previously discovered construction if it exists
        init_seq = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if init_seq.shape[0] != TARGET_N:
            init_seq = _interpolate_distribution(init_seq, TARGET_N)
        a0 = np.maximum(init_seq, 0.0)
        s0 = a0.sum()
        if s0 == 0.0:
            a0 = np.full(TARGET_N, 1.0 / TARGET_N)
        else:
            a0 = a0 / s0
        print(f"Initial sequence taken from global variable, length = {TARGET_N}")
    except Exception:
        # Fall back to the uniform distribution
        a0 = np.full(TARGET_N, 1.0 / TARGET_N)
        print(f"Uniform start, length = {TARGET_N}")

    # ------------------------------------------------------------------
    # 1. Multi‑restart optimisation loop
    # ------------------------------------------------------------------
    best_global_a = a0.copy()
    best_global_score = float("inf")
    restart_counter = 0

    # Allocate a small buffer for the final evaluation
    while time.time() < deadline - 30.0:
        restart_counter += 1

        # Random perturbation of the current best to initialise a new run
        a_start = best_global_a + 0.01 * np.random.rand(TARGET_N)
        a_start = np.clip(a_start, 0.0, None)
        a_start /= a_start.sum()

        # Give each inner run a modest time slice (up to 4 min)
        inner_deadline = min(deadline, time.time() + 240.0)

        a_opt, score_opt = _eg_optimize(a_start, inner_deadline)

        if score_opt < best_global_score:
            best_global_score = score_opt
            best_global_a = a_opt.copy()
            print(f"[Restart {restart_counter}] NEW OVERALL cheap = {score_opt:.6f}")

        # If only a few seconds remain, break early
        if time.time() + 5.0 >= deadline:
            break

    # ------------------------------------------------------------------
    # 2. Final exact evaluation with the provided function
    # ------------------------------------------------------------------
    final_seq = best_global_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:   # defensive
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    return final_seq
