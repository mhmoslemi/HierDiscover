propose_candidate – convex‐subgradient / Adam optimiser for the step‑function
objective

    O(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm consists of three stages:

1️⃣  **Adam on a smooth log‑sum‑exp surrogate**  
    ⇢  fα(a) = (2·n/α)·log Σk exp(α·convk(a)) + λ·Σi a_i·log a_i  
    This yields a differentiable objective whose gradient is a
    correlation with a soft‑max weight vector.  We run Adam with a
    decreasing learning‑rate, anneal the smoothing parameter α from a
    small value up to a large one (≈2·10⁴) and keep the best exact
    objective seen.

2️⃣  **Deterministic sub‑gradient descent on the true max‑convolution**  
    The exact objective 2·n·max(conv) is convex; a sub‑gradient is
    obtained from the active convolution indices.  We perform a projected
    sub‑gradient step with a back‑tracking line‑search (step size
    decays ∝1/√t) and keep the current best sequence.

3️⃣  **Stochastic mass‑transfer refinement**  
    Randomly move a fraction of mass between two bins (Metropolis
    acceptance with a slowly cooling temperature).  This helps to
    escape plateaus that the deterministic phase may get stuck in.

The routine adapts to the available time (default budget 1000 s) and
automatically up‑samples a supplied start‑up sequence (``height_sequence_1``)
to a power‑of‑two length (default 8192).  All updates are projected onto
the probability simplex to guarantee feasibility for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution via FFT (returns length 2·n‑1)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Exact objective for a vector that is already normalised (∑a = 1)
def _objective(a: np.ndarray, n: int, L: int) -> float:
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) (averaged over all maximisers)
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)          # indices attaining the max
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        mask = i_idx != j_idx
        # off‑diagonal terms contribute 2·a_j
        g[i_idx[mask]] += 2.0 * a[j_idx[mask]]
        # diagonal terms contribute a_j
        g[i_idx[~mask]] += a[j_idx[~mask]]
    if ks.size:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
# Gradient of the smooth surrogate (log‑sum‑exp) plus entropy term
def _smooth_gradient_plus_entropy(
    a: np.ndarray,
    conv: np.ndarray,
    alpha: float,
    L: int,
    lam: float,
) -> np.ndarray:
    """
    Returns ∇_a ( (2·n/α)·log∑_k e^{α·conv_k} + λ·∑_i a_i·log a_i ).
    The first part reduces to 2·corr(w,a) with w_k ∝ e^{α·conv_k};
    the second part contributes λ·(log a_i + 1).
    """
    # ---- soft‑max weights ----------
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)          # for numerical stability
    w = exp_scaled / exp_scaled.sum()                 # shape (2n‑1,)

    # ---- correlation w ⋆ a ----------
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n_len = a.size
    grad_corr = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]

    # ---- entropy part ----------
    eps = 1e-20
    grad_ent = lam * (np.log(a + eps) + 1.0)

    return grad_corr + grad_ent

        while time.time() < deadline and it_loc < max_iter_local:
            # -------- random mass‑transfer move ------------------------------------
            i, j = np.random.randint(n, size=2)
            if i != j and a[i] > 1e-15:
                delta = a[i] * np.random.rand() * max_move_frac
                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta
                a_new = _project_to_simplex(a_new)

                conv_new = _conv_fft(a_new, L)
                new_obj = 2.0 * n * conv_new.max()

                # Metropolis acceptance
                if new_obj < cur_obj or np.random.rand() < math.exp((cur_obj - new_obj) / max(T, 1e-14)):
                    a, conv, cur_obj = a_new, conv_new, new_obj
                    no_improve = 0
                    if cur_obj < best_obj:
                        best_obj = cur_obj
                        best_seq = a.copy()
                        t_now = time.time() - start_time
                        print(f"[{t_now:.1f}s] new global best O={best_obj:.6f}")
                else:
                    no_improve += 1
            else:
                no_improve += 1

            # -------- deterministic sub‑gradient step ------------------------------
            if it_loc % subgrad_interval == 0:
                # compute sub‑gradient of the exact max‑convolution
                g = _subgradient_max_conv(a, conv)

                # back‑tracking line search (simple geometric decay)
                eta = 0.12 * (1.0 - it_loc / max_iter_local) + 1e-4
                for _ in range(5):
                    a_trial = _project_to_simplex(a - eta * g)
                    conv_trial = _conv_fft(a_trial, L)
                    obj_trial = 2.0 * n * conv_trial.max()
                    if obj_trial <= cur_obj:
                        a, conv, cur_obj = a_trial, conv_trial, obj_trial
                        if cur_obj < best_obj:
                            best_obj = cur_obj
                            best_seq = a.copy()
                            t_now = time.time() - start_time
                            print(f"[{t_now:.1f}s] sub‑grad improved O={best_obj:.6f}")
                        break
                    eta *= 0.5   # shrink step if not improving

            # -------- occasional smoothing of the profile -------------------------
            if it_loc % 4000 == 0 and it_loc > 0:
                kernel = np.array([0.125, 0.25, 0.25, 0.25, 0.125])
                a_smooth = np.convolve(a, kernel, mode="same")
                a = _project_to_simplex(0.86 * a + 0.14 * a_smooth)

            # temperature decay
            T *= 0.99985
            it_loc += 1

            # restart if we have been stuck for a while
            if no_improve >= no_improve_limit:
                break

        # finish this restart – go to the next one if time permits
        if time.time() >= deadline:
            break

    # --------------------------------------------------------------
    # Final sanity check using the provided evaluator
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:
        print("[final] evaluate_sequence failed:", e)

    # Return the best sequence discovered (as a plain Python list)
    return best_seq.tolist()
