def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        w = np.full_like(v, 1.0 / n)
        return w
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using a power‑of‑two FFT (real‑FFT version)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 2).bit_length())   # next power of two ≥ 2n‑1
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return conv


def _conv_linear_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT (output size = len(x)+len(y)-1)."""
    m = len(x) + len(y) - 1
    size = 1 << ((m - 1).bit_length())
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)[:m]
    return conv

    eta0 = 0.7

    lambda_start = 1.0e3
    lambda_end   = 2.0e5

    # rough iteration budget (≈150 iters per second on typical hardware)
    max_iters = int(min(200_000,
                        max(0.0, (deadline - time.time())) * 150))
    print(f"[opt] will run up to {max_iters} iterations "
          f"(≈{max_iters/150:.1f}s)")

    stagnation_patience = 3000
    stagnation_counter = 0

    # ---------- main optimisation loop ----------
    for t in range(max_iters):
        if time.time() >= deadline:
            print("[opt] time budget exhausted")
            break

        # evaluate current objective
        conv = _conv_fft(a0)
        max_conv = conv.max()
        O = 2.0 * n * max_conv

        if O < best_O - 1e-12:
            best_O = O
            best_a = a0.copy()
            stagnation_counter = 0
            print(f"[iter {t:6d}] new best O={best_O:.6f}")
        else:
            stagnation_counter += 1

        # annealed soft‑max weights
        frac = t / max_iters
        lam = lambda_start * ((lambda_end / lambda_start) ** frac)

        scaled = lam * conv
        max_scaled = np.max(scaled)
        w = np.exp(scaled - max_scaled)
        w_sum = w.sum()
        w = w / w_sum if w_sum > 0 else np.full_like(w, 1.0 / w.size)

        # gradient of the smoothed max (cross‑correlation)
        corr = _cross_corr_fft(w, a0)          # Σ_k w_k a_{k-i}
        grad = 2.0 * corr                       # factor 2 = d(conv)/da

        # normalise direction to keep the log‑update stable
        grad_norm = np.linalg.norm(grad) + eps
        direction = grad / grad_norm

        # Adam update on log‑space (exponentiated‑gradient style)
        m = beta1 * m + (1.0 - beta1) * direction
        v = beta2 * v + (1.0 - beta2) * (direction ** 2)

        m_hat = m / (1.0 - beta1 ** (t + 1))
        v_hat = v / (1.0 - beta2 ** (t + 1))

        eta = eta0 * (1.0 - frac)               # linear decay

        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        log_a = np.log(a0 + eps) - step
        a0 = np.exp(log_a)
        s = a0.sum()
        a0 = a0 / s if s > 0 else np.full_like(a0, 1.0 / n)

        # jitter to escape plateaux
        if stagnation_counter >= stagnation_patience:
            jitter = np.random.randn(n) * 1e-4
            a0 = np.clip(a0 + jitter, 0.0, None)
            a0 = a0 / a0.sum()
            stagnation_counter = 0
            print(f"[jitter] injected noise at iter {t}")

    # ---------- final exact evaluation ----------
    final_seq = best_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:  # pragma: no cover
        print("[final] evaluate_sequence call failed:", e)
        final_score = None

    return final_seq
