#!/usr/bin/env python
"""
Propose a non‑negative step function (list of heights) that minimises

    2·n·max_k (a*a)[k] / (Σ a_i)²

The routine expands the previous approach with a richer parametric
initialisation (Beta‑distribution shaped sequences) and a smooth
approximation of the ∞‑norm using a soft‑max.  The gradient of this
surrogate is obtained efficiently by FFT‑based cross‑correlation.
An Adam‑style (exponentially‑averaged) update on the probability simplex
is performed, with occasional multiplicative (exponentiated‑gradient)
steps.  The optimiser runs until the time budget is exhausted (or a
target value of 1.503 is reached) and returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using FFT (returns length 2·n‑1 array)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa).real[: 2 * a.size - 1]
    return conv


def _cross_corr_fft(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Cross‑correlation g[k] = Σ_i w_i a_{i‑k} for k = 0 … n‑1.
    Computed via FFT by convolving w with the reversed a.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    full = np.fft.ifft(fw * fa_rev).real
    start = n - 1                      # slice that yields the required n values
    return full[start:start + n]


def _beta_sequence(n: int, alpha: float, beta: float) -> np.ndarray:
    """Return a normalised Beta‑shaped sequence on {0,…,n‑1}."""
    # centre the points to avoid zero at the boundaries
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    seq = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    if seq.sum() <= 0.0 or np.any(np.isnan(seq)):
        # fallback to uniform if something goes wrong
        seq = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        seq /= seq.sum()
    return seq


def _power_law_sequence(n: int, p: float, offset: int = 0) -> np.ndarray:
    """Return a normalised power‑law a_i ∝ (i‑offset+1)^‑p (Σa_i = 1)."""
    seq = np.zeros(n, dtype=np.float64)
    i = np.arange(offset, n, dtype=np.float64)
    if i.size == 0:
        return seq
    seq[offset:] = (i + 1.0) ** (-p)
    if seq.sum() == 0.0:
        seq = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        seq /= seq.sum()
    return seq


# ----------------------------------------------------------------------
# Main entry point required by the evaluation platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The routine builds a diverse set of parametric initial candidates,
    keeps the best, and refines it with a smooth‑max gradient descent
    (Adam‑style) on the simplex.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin for final eval

    # --------------------------------------------------------------
    # Choose a working length (larger n usually yields a smaller constant).
    # --------------------------------------------------------------
    # Try to reuse a previous construction if it exists.
    try:
        prev = list(height_sequence_1)            # noqa: F821 (provided globally)
        n = max(len(prev), 15000)
    except Exception:
        prev = None
        n = 15000

    n = min(max(n, 8000), 25000)                 # keep n in a comfortable range
    print(f"[Info] Using n = {n}")

    # FFT lengths – we need one for the convolution a*a (2·n‑1) and one for
    # the cross‑correlation (2·n‑1 + n‑1 = 3·n‑2).  Use the larger one for both.
    fft_len = 1 << ((3 * n - 2).bit_length())

    # --------------------------------------------------------------
    # Cheap evaluation (sum = 1 → score = 2·n·max(conv))
    # --------------------------------------------------------------
    def cheap_score(vec: np.ndarray) -> float:
        conv = _conv_full_fft(vec, fft_len)
        return 2.0 * n * conv.max()

    # --------------------------------------------------------------
    # 0. Initialise with a diverse set of parametric candidates.
    # --------------------------------------------------------------
    best_a = None
    best_val = float("inf")

    # 0a) previous construction (if length matches)
    if prev is not None and len(prev) == n:
        cand = np.asarray(prev, dtype=np.float64)
        if cand.sum() > 0.0:
            cand = cand / cand.sum()
            val = cheap_score(cand)
            best_a, best_val = cand, val
            print(f"[Init] previous sequence cheap = {val:.6f}")

    # 0b) Beta‑distribution shapes (α,β) – covers monotone & symmetric shapes
    alphas = np.linspace(0.2, 2.0, 13)   # 13 values
    betas = np.linspace(0.2, 2.5, 13)
    for alpha in alphas:
        for beta in betas:
            if time.time() >= deadline:
                break
            cand = _beta_sequence(n, alpha, beta)
            val = cheap_score(cand)
            if val < best_val:
                best_val, best_a = val, cand.copy()
                print(f"[Beta] α={alpha:.2f} β={beta:.2f} cheap={val:.6f}")

    # 0c) Power‑law shapes with a small left offset (to make a0 small)
    p_grid = np.linspace(0.5, 1.2, 25)
    offset_grid = np.arange(0, int(0.20 * n) + 1, max(1, int(0.05 * n)))
    for p in p_grid:
        for off in offset_grid:
            if time.time() >= deadline:
                break
            cand = _power_law_sequence(n, p, offset=off)
            val = cheap_score(cand)
            if val < best_val:
                best_val, best_a = val, cand.copy()
                print(f"[Power] p={p:.3f} off={off} cheap={val:.6f}")

    # 0d) Random Dirichlet draws (wide concentration range)
    for conc in np.logspace(-3.0, 0.5, 15):          # 15 concentration levels
        for _ in range(5):
            if time.time() >= deadline:
                break
            g = np.random.gamma(conc, 1.0, size=n)
            cand = g / g.sum()
            val = cheap_score(cand)
            if val < best_val:
                best_val, best_a = val, cand.copy()
                print(f"[Dir] conc={conc:.5f} cheap={val:.6f}")

    # Fallback to uniform if nothing was found (should never happen)
    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = cheap_score(best_a)
        print("[Init] fallback uniform")

    print(f"[Init] best cheap = {best_val:.6f}")

    # --------------------------------------------------------------
    # 1. Refine the best candidate with smooth‑max Adam optimisation.
    # --------------------------------------------------------------
    a = best_a.copy()
    m = np.zeros_like(a)            # first moment
    v = np.zeros_like(a)            # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.6                       # initial learning‑rate factor
    gamma0 = 12.0                   # initial soft‑max temperature
    max_iters = 200_000
    best_iter_a = a.copy()
    best_iter_val = best_val
    last_improve = 0

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ---- convolution a*a (full) ----
        conv_full = _conv_full_fft(a, fft_len)        # length 2·n‑1
        max_conv = conv_full.max()
        cheap = 2.0 * n * max_conv

        # keep best cheap seen so far
        if cheap < best_iter_val - 1e-12:
            best_iter_val = cheap
            best_iter_a = a.copy()
            last_improve = it
            print(f"[Iter {it:07d}] NEW BEST cheap = {cheap:.6f}")

        # ---- soft‑max weights over the convolution entries ----
        gamma = min(200.0, gamma0 + 0.001 * it)          # slowly increase temperature
        # stability: subtract max before exponentiation
        w = np.exp(gamma * (conv_full - max_conv))
        w /= w.sum()

        # ---- cross‑correlation term (gradient of smooth‑max) ----
        cross = _cross_corr_fft(w, a, fft_len)         # length n
        grad = 2.0 * cross                              # gradient of (1/γ)log‑sum‑exp

        # ---- Adam averages ----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # ---- step size schedule ----
        lr = lr0 / math.sqrt(it)

        # ---- occasionally use multiplicative (exponentiated‑gradient) update ----
        if np.random.rand() < 0.10:
            a = a * np.exp(-lr * grad)
        else:
            a = a - lr * (m_hat / (np.sqrt(v_hat) + eps))

        # ---- projection back onto the simplex ----
        a = _simplex_projection(a)

        # ---- restart if stuck for a long time ----
        if it - last_improve > 60_000:
            # small random perturbation of the current best vector
            noise = 0.01 * np.random.rand(n)
            a = _simplex_projection(best_iter_a + noise)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = it
            print(f"[Iter {it:07d}] Restart after stagnation (perturb).")

        # stop early if we have already beaten the target
        if best_iter_val < 1.503:
            print(f"[Info] Target reached (cheap={best_iter_val:.6f}) – stopping early.")
            break

    # --------------------------------------------------------------
    # 2. Final exact evaluation (the cheap score already equals the exact score
    #    because the sequence is already normalised to sum = 1).
    # --------------------------------------------------------------
    final_seq = best_iter_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    return final_seq
