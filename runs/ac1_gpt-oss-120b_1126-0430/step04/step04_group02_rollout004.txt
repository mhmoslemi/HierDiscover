def upscale(seq, new_n):
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    new_seq = np.interp(new_x, old_x, seq)
    # ensure nonneg and sum to 1
    new_seq = np.clip(new_seq, 0, None)
    if new_seq.sum() == 0:
        new_seq = np.ones_like(new_seq) / len(new_seq)
    else:
        new_seq = new_seq / new_seq.sum()
    return new_seq
```

"""
Propose a non‑negative step‑function minimising

    O(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

using a multi‑scale Adam optimiser on the simplex (log‑it/soft‑max
parameterisation) with an exponentially increasing smoothing of the
max‑operator, aggressive restarts, jitter and a final greedy local
search.  The algorithm runs several support sizes (including the length
of a previously found construction if present) and always respects the
global time budget (default 1000 s).  The best sequence is returned as a
plain list of floats.
"""

import time
import numpy as np

# -------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two greater than or equal to x."""
    return 1 << (x - 1).bit_length()


# -------------------------------------------------------------
def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (real‑valued) using FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv


# -------------------------------------------------------------
def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation needed for the gradient:
        g_i = Σ_t w[i+t]·a[t]      for i = 0,…,n‑1
    Implemented via FFT (linear convolution).
    """
    n = a.shape[0]
    m = w.shape[0]                     # = 2n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


# -------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)**2)."""
    sum_a = a.sum()
    if sum_a < 1e-12:
        return np.inf
    n = a.size
    conv = _conv_self_fft(a)
    return 2.0 * n * conv.max() / (sum_a ** 2)


# -------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


# -------------------------------------------------------------
def _upscale(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Interpolate a probability vector to a new length.
    The result is re‑normalised to sum 1 and clipped to be non‑negative.
    """
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(new_x, old_x, seq)
    new_seq = np.clip(new_seq, 0.0, None)
    s = new_seq.sum()
    if s == 0.0:
        new_seq = np.ones(new_n, dtype=float) / new_n
    else:
        new_seq = new_seq / s
    return new_seq


# -------------------------------------------------------------
def _run_opt(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
    beta_start: float = 0.5,
    beta_end: float = 30000.0,
    lr_start: float = 0.15,
    lr_end: float = 0.001,
) -> tuple[list[float], float]:
    """
    Adam optimisation on the log‑it simplex for a fixed support size ``n``.
    Returns the best sequence (as a plain list) and its exact objective.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # ----- initialise -------------------------------------------------
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.shape != (n,):
            a = rng.random(n)
    else:
        a = rng.random(n)

    a = a / a.sum()
    # work in log‑it space (log‑softmax)
    x = np.log(a + 1e-16)

    m = np.zeros_like(x)          # first moment
    v = np.zeros_like(x)          # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    best_a = a.copy()
    best_obj = _objective(best_a)
    last_improve_it = 0
    it = 0
    restart_gap = 25000           # iterations without improvement before restart
    jitter_interval = 5000

    while time.time() - start < time_limit:
        it += 1

        # ----- probabilities from logits --------------------------------
        a = _softmax(x)

        # ----- exact objective -----------------------------------------
        conv = _conv_self_fft(a)
        M = conv.max()
        obj = 2.0 * n * M / (a.sum() ** 2)   # a.sum() == 1 here, but keep general

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            last_improve_it = it
            # occasional progress report
            print(f"[n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        # ----- schedule parameters --------------------------------------
        elapsed = time.time() - start
        frac = min(elapsed / time_limit, 1.0)  # [0,1]

        # exponential schedule for smoothing parameter β
        beta = beta_start * np.exp(np.log(beta_end / beta_start) * frac)

        # exponential decay for learning rate
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * frac)

        # ----- smooth (or exact) max weights ----------------------------
        if beta > 20000.0:            # treat as exact arg‑max
            kmax = int(np.argmax(conv))
            w = np.zeros_like(conv)
            w[kmax] = 1.0
        else:
            shifted = conv - M               # ≤ 0
            # prevent overflow for very negative values (they become zero)
            w = np.exp(beta * shifted)
            w_sum = w.sum()
            if w_sum == 0.0:                # numerical underflow – fall back to arg‑max
                kmax = int(np.argmax(conv))
                w = np.zeros_like(conv)
                w[kmax] = 1.0
            else:
                w = w / w_sum

        # ----- gradient w.r.t. a (smooth max) ---------------------------
        g_a = 2.0 * _correlation_fft(w, a)   # shape (n,)

        # ----- natural gradient in log‑it space -------------------------
        dot = np.dot(a, g_a)                # scalar = ⟨a, g_a⟩
        g_x = a * (g_a - dot)               # shape (n,)

        # ----- Adam update ---------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- jitter to avoid plateaus --------------------------------
        if it % jitter_interval == 0:
            jitter = rng.normal(scale=0.02, size=n)
            x += jitter

        # ----- restart if stuck -----------------------------------------
        if it - last_improve_it > restart_gap:
            # re‑initialise logits from the current best plus tiny noise
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.005, size=n)
            m.fill(0.0)
            v.fill(0.0)
            last_improve_it = it
            print(f"[n={n:5d}] restart at iter {it}")

    # ----- compress negligible entries ---------------------------------
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        n_new = best_a.size
        best_obj = _objective(best_a)
        print(f"[compress] reduced length {n} → {n_new}, O={best_obj:.6f}")

    # ----- short greedy local search (random mass transfers) ---------
    remaining = time_limit - (time.time() - start)
    if remaining > 0.5:
        # use a small fraction of the remaining budget for a cheap hill‑climb
        local_time = remaining * 0.3
        local_start = time.time()
        a_loc = best_a.copy()
        obj_loc = best_obj
        n_loc = a_loc.size
        # pre‑compute current convolution once
        conv_loc = _conv_self_fft(a_loc)

        while time.time() - local_start < local_time:
            i, j = rng.choice(n_loc, size=2, replace=False)
            # move a small random fraction of mass from i to j
            delta = a_loc[i] * rng.random() * 0.02   # ≤ 2 % of a_i
            if delta <= 0.0:
                continue
            a_new = a_loc.copy()
            a_new[i] -= delta
            a_new[j] += delta
            conv_new = _conv_self_fft(a_new)
            obj_new = 2.0 * n_loc * conv_new.max() / (a_new.sum() ** 2)
            if obj_new < obj_loc:
                a_loc, obj_loc = a_new, obj_new
                if obj_new < best_obj:
                    best_a, best_obj = a_new, obj_new
        print(f"[local] final local O={best_obj:.6f}")

    return best_a.tolist(), best_obj
