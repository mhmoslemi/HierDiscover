def _project_onto_simplex(v):
    """Project v onto the probability simplex (sum=1, v>=0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]  # last True
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _subgrad_max_and_conv(a):
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2*n-1]
    max_conv = conv.max()
    max_idx = np.flatnonzero(conv == max_conv)  # could be multiple
    grad = np.zeros_like(a)
    for k in max_idx:
        # left
        left = k - np.arange(n)
        mask = (left >= 0) & (left < n)
        grad[mask] += a[left[mask]]
        # right
        right = np.arange(n) - k
        mask2 = (right >= 0) & (right < n)
        grad[mask2] += a[right[mask2]]
    return max_conv, grad

#!/usr/bin/env python
"""
Hybrid optimisation for the step‑function bound.

* Phase 1 – Smoothed‑max Adam optimiser (as in the reference
  implementation) but with an adaptive time budget.  It quickly finds a
  good region of the simplex.
* Phase 2 – Sub‑gradient descent on the *exact* max‑convolution.
  The true objective 2·n·max(a*a) is used, together with a projection
  onto the probability simplex after each step.
* Random perturbations and a simple “no‑improvement” restart keep the
  search from getting stuck.
* The whole routine respects the provided time budget (≈ 1000 s) and
  finally returns the best sequence seen; its exact value is obtained
  with ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level – no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max (returns a probability vector that sums to 1)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it representation that yields `a` after soft‑max."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """
    Smallest power‑of‑two ≥ 3·n‑2, enough for a linear correlation of length
    3·n‑2 (used for a⋆w and a*a).
    """
    return 1 << ((3 * n - 2).bit_length())


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """Normalised power‑law a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a*a (no gradient)."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float) -> tuple[float, np.ndarray]:
    """
    Smoothed version of max(a*a) and its gradient.
    The smoothing constant is ε = eps_factor·max(conv).
    Returns (max_conv, grad) where grad = 4·n·corr(a,w)
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    a_fft = np.fft.rfft(a, L)

    # linear self‑convolution
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = conv.max()

    # soft‑max weights
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or np.isnan(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # correlation a⋆w  (only first n entries needed)
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    grad = 4.0 * n * corr
    return float(max_conv), grad


def _subgrad_max_and_conv(a: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Exact max of a*a and a sub‑gradient of that max.
    If several lags attain the maximum the gradients are summed.
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = conv.max()
    max_idx = np.flatnonzero(conv == max_conv)

    grad = np.zeros_like(a)

    # vectorised handling of each maximal lag
    for k in max_idx:
        # left contribution: i where 0 ≤ k‑i < n  →  i ∈ [max(0,k‑n+1), min(k,n‑1)]
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_range = k - i_range
        grad[i_range] += a[j_range]

        # right contribution: i where 0 ≤ i‑k < n  →  i ∈ [k, min(k+n‑1,n‑1)]
        i_start2 = max(k, 0)
        i_end2 = min(k + n - 1, n - 1)
        i_range2 = np.arange(i_start2, i_end2 + 1, dtype=np.int64)
        j_range2 = i_range2 - k
        grad[i_range2] += a[j_range2]

    return float(max_conv), grad


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    Implementation follows the O(n) algorithm of
    Duchi et al. (2008).
    """
    if v.ndim != 1:
        raise ValueError("v must be a one‑dimensional array")
    n = v.shape[0]

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
