def _project_simplex(v):
    """Projects vector v onto the probability simplex (sum=1, v>=0)."""
    # From Duchi et al. 2008
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a step‑function (non‑negative heights) that minimizes the
evaluation function 2·n·max(conv)/(sum(a)**2).  

The algorithm builds a probability vector (sum = 1) and optimises it on the
simplex with an Adam optimiser on the soft‑max (log‑it) parameters.
Key ideas:

* start from a cheap power‑law distribution (or the provided
  ``height_sequence_1``) and normalise to a probability vector;
* use a smooth approximation of the max (soft‑max over the self‑convolution)
  to obtain a usable gradient;
* Adam updates on the logits together with a projection (clipping) keep the
  parameters well‑behaved;
* occasional Gaussian jitter and a modest line‑search‑like back‑off avoid
  plateaux;
* the support size is enlarged progressively (up to 40 000 entries) by linear
  interpolation of the current best sequence;
* a final short fine‑tuning phase with a very small smoothing factor refines the
  solution.
The best sequence found within the time budget is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max; returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it representation that yields `a` after soft‑max."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """
    Smallest power‑of‑two ≥ 3·n‑2,
    sufficient for a linear convolution of length (2·n‑1) followed by a
    correlation of length (3·n‑2).
    """
    return 1 << ((3 * n - 2).bit_length())


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    total = new_a.sum()
    if total == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= total
    return new_a


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """Normalised power‑law a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _conv_max_and_grad(a: np.ndarray, eps_factor: float) -> tuple[float, np.ndarray]:
    """
    Compute the maximal entry of the self‑convolution a*a and the gradient of the
    smoothed objective (2·n·smooth_max).

    Parameters
    ----------
    a : np.ndarray
        Probability vector (sum = 1, non‑negative).
    eps_factor : float
        Multiplicative factor for the smoothing constant:
        eps = eps_factor·max(conv).

    Returns
    -------
    max_conv : float
        Maximum of the self‑convolution a*a.
    grad_a : np.ndarray
        Gradient of 2·n·smooth_max w.r.t. a (already multiplied by 4·n).
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    # FFT of a (zero‑padded)
    a_fft = np.fft.rfft(a, L)

    # self‑convolution a*a (linear, length 2n‑1)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # soft‑max weights over the convolution entries
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv               # for numeric stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or np.isnan(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # cross‑correlation of a with the weight vector w
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # gradient of the smoothed objective (4·n·corr)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a


def _conv_max(a: np.ndarray) -> float:
    """Return the maximal entry of a*a (no gradient)."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())

# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) beats the current
    bound.  The routine respects the given time budget (~1000 s) and uses a
    smoothed‑max Adam optimiser on the probability simplex, with progressive
    support enlargement (up to 40 000 entries) and occasional random perturbations
    to escape plateaux.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # 0. Choose an initial support size and an initial distribution.
    # ------------------------------------------------------------------
    try:
        # Global variable may already contain a good candidate.
        init_global = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        n = max(5000, init_global.shape[0])
        a0 = np.maximum(init_global, 0.0)
        s = a0.sum()
        if s == 0.0:
            a0 = np.full_like(a0, 1.0 / a0.shape[0])
        else:
            a0 /= s
        print(f"Initial sequence taken from global variable, length = {n}")
    except Exception:
        # No global variable – start from a power‑law distribution.
        n = 8000
        # Quick scan over a few exponents to obtain a decent starter.
        p_grid = np.linspace(0.55, 0.75, 21)
        best_a = None
        best_score = float("inf")
        for p in p_grid:
            if time.time() >= deadline:
                break
            cand = _power_law_distribution(n, p)
            max_c = _conv_max(cand)
            score = 2.0 * n * max_c
            if score < best_score:
                best_score = score
                best_a = cand
        if best_a is None:
            best_a = np.full(n, 1.0 / n)
        a0 = best_a
        print(f"Power‑law start, length = {n}")

    # Small random jitter on the log‑it parameters.
    z = _logits_from_distribution(a0)
    z += 0.01 * np.random.randn(n)

    # ------------------------------------------------------------------
    # 1. Adam state on the logits.
    # ------------------------------------------------------------------
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_alpha = 0.25                     # base step‑size for logits

    # ------------------------------------------------------------------
    # 2. Optimisation loop (main phase).
    # ------------------------------------------------------------------
    best_seq = _softmax(z)                # best probability vector seen so far
    best_score = 2.0 * n * _conv_max(best_seq)

    it = 0
    last_improve_it = 0
    eps_factor = 0.20                     # initial smoothing factor (ε = eps_factor·max)
    eps_factor_min = 0.005
    eps_decay = 0.9998
    report_every = 2000
    max_n = 40000
    growth_factor = 1.5
    expand_check = 12000                  # try to grow every this many iterations

    while time.time() < deadline and it < 2_000_000:
        it += 1

        # ----- current point on the simplex -----
        a = _softmax(z)

        # ----- cheap objective + smoothed gradient -----
        max_conv, grad_a = _conv_max_and_grad(a, eps_factor)
        cheap_score = 2.0 * n * max_conv

        # ----- record the best ever seen -----
        if cheap_score < best_score:
            best_score = cheap_score
            best_seq = a.copy()
            last_improve_it = it
            if it % report_every == 0:
                print(f"[Iter {it:07d}] NEW BEST cheap = {cheap_score:.6f}")

        # ----- gradient w.r.t. logits (mirror‑descent on the simplex) -----
        mu = np.dot(grad_a, a)
        grad_z = a * (grad_a - mu)

        # ----- Adam moment updates (bias‑corrected) -----
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # ----- step‑size schedule (inverse‑sqrt decay) -----
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step = alpha_t * m_hat / (np.sqrt(v_hat) + eps_adam)
        z_new = z - step
        z_new = np.clip(z_new, -50.0, 50.0)   # keep exp(z) safe

        # ----- accept the step (unconditionally) -----
        z = z_new

        # ----- occasional random jitter to escape plateaus -----
        if it % 2500 == 0 and np.random.rand() < 0.2:
            z += 0.02 * np.random.randn(n)
            m *= 0.6
            v *= 0.6

        # ----- decay smoothing factor -----
        eps_factor = max(eps_factor_min, eps_factor * eps_decay)

        # ----- progressive support enlargement (if time permits) -----
        if (it % expand_check == 0) and (n < max_n) and (time.time() + 20 < deadline):
            new_n = min(max_n, int(n * growth_factor) + 1)
            print(f"[Iter {it:07d}] Expanding support: {n} → {new_n}")
            a_interp = _interpolate_distribution(best_seq, new_n)
            z = _logits_from_distribution(a_interp)
            m = np.zeros_like(z)
            v = np.zeros_like(z)
            n = new_n
            best_seq = a_interp.copy()
            best_score = 2.0 * n * _conv_max(best_seq)
            it = 0
            last_improve_it = 0
            eps_factor = 0.20
            continue

        # ----- early stop if stagnation after maximal size -----
        if n >= max_n and it - last_improve_it > 400_000:
            print(f"[Iter {it:07d}] Stagnation detected – terminating early.")
            break

    # ------------------------------------------------------------------
    # 3. Short fine‑tuning phase with very small smoothing (high precision).
    # ------------------------------------------------------------------
    fine_iters = 2000
    for i in range(fine_iters):
        if time.time() >= deadline:
            break
        a = _softmax(z)
        max_conv, grad_a = _conv_max_and_grad(a, eps_factor_min)
        cheap_score = 2.0 * n * max_conv
        if cheap_score < best_score:
            best_score = cheap_score
            best_seq = a.copy()

        mu = np.dot(grad_a, a)
        grad_z = a * (grad_a - mu)

        # Adam moments (reuse same buffers)
        tot_it = it + i + 1
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** tot_it)
        v_hat = v / (1.0 - beta2 ** tot_it)

        alpha_t = base_alpha / (math.sqrt(tot_it) + 1.0)
        step = alpha_t * m_hat / (np.sqrt(v_hat) + eps_adam)
        z = z - step
        z = np.clip(z, -50.0, 50.0)

    # ------------------------------------------------------------------
    # 4. Final exact evaluation with the provided evaluate_sequence().
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:   # defensive – should never happen
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # ------------------------------------------------------------------
    # 5. Return the best sequence found.
    # ------------------------------------------------------------------
    return final_seq
