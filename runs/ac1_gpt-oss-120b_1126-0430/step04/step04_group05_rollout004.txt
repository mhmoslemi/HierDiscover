"""
Propose a non‑negative step‑function (a list of heights) that minimizes

    O(a) = 2·n·max(conv(a,a)) / (sum(a)²)

The optimiser consists of four stages

1️⃣  Adam on a smooth log‑sum‑exp surrogate of the max‑operator, with an
    exponential annealing of the smoothing parameter α.  After each step the
    vector is projected onto the probability simplex.

2️⃣  Projected sub‑gradient descent on the exact (non‑smooth) objective.

3️⃣  Simulated‑annealing style mass‑transfer moves together with occasional
    exact sub‑gradient steps.

4️⃣  A single linear‑programming refinement: maximise the ℓ₁‑norm of a new
    vector g under the linearised convolution constraints conv(a,g)≤M.
    Mixing the current a with the normalised optimiser g typically reduces M
    further.

All phases respect a total wall‑time of 1000 s.  The best sequence found is
returned as a plain ``list[float]`` and is finally checked with the provided
``evaluate_sequence`` function.
"""

import time
import math
import numpy as np
from typing import List

# optional LP support (SciPy is available in the execution environment)
try:
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix
except Exception:      # pragma: no cover
    linprog = None
    csr_matrix = None


# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:                 # all non‑positive → uniform simplex point
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full (linear) convolution a*a using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
def _objective_simplex(a: np.ndarray, n: int, L: int) -> float:
    """Exact objective O = 2·n·max(conv(a,a)) under the assumption Σa=1."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. a (used for the exact objective)."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)          # possibly several maximisers
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, conv: np.ndarray,
                    alpha: float, L: int) -> np.ndarray:
    """
    Gradient of the smoothed surrogate
        fα(a) = (2·n/α)·log Σ_k exp(α·conv_k)
    using the identity ∇fα = 2·corr(w,a) with soft‑max weights w.
    """
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)      # stable exp
    w = exp_scaled / exp_scaled.sum()             # soft‑max over k

    # correlation w ⋆ a (length at least 3·n‑2)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)

    n_len = a.size
    grad = 2.0 * corr_rev[n_len - 1: n_len - 1 + n_len]
    return grad


# ----------------------------------------------------------------------
def _lp_refine(a: np.ndarray, M: float, n: int, max_iter: int = 1) -> np.ndarray:
    """
    One (or a few) LP refinements.
    Maximise Σ g_j  s.t.  (a * g)_k ≤ M  for all k,  g ≥ 0.
    Returns a new candidate obtained by mixing a with the normalised LP solution.
    """
    if linprog is None:
        return a                     # fallback – LP not available

    # Build a * g ≤ M  (Toeplitz) only from entries of a that are > eps.
    eps = 1e-12
    idx = np.where(a > eps)[0]
    if idx.size == 0:                # degenerate – fall back to uniform
        idx = np.arange(n)

    # Sparse matrix construction
    rows = []
    cols = []
    data = []
    for i in idx:
        a_i = a[i]
        # rows = i + j   (j = 0 … n‑1)
        row_inds = i + np.arange(n, dtype=np.int64)
        rows.extend(row_inds.tolist())
        cols.extend(np.arange(n, dtype=np.int64).tolist())
        data.extend([a_i] * n)

    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    b = np.full(2 * n - 1, M)

    # maximise Σ g_j  → minimise  −Σ g_j
    c = -np.ones(n)

    # Use HiGHS (fast interior‑point/simplex hybrid)
    res = linprog(c, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n,
                  method='highs',
                  options={'presolve': True, 'time_limit': 30})

    if not res.success:
        return a                         # LP failed – keep current a

    g = res.x
    # normalise g to the simplex (same total mass as a)
    g = g / g.sum()
    tau = 0.15                          # modest mixing
    a_new = (1 - tau) * a + tau * g
    a_new = _project_to_simplex(a_new)
    return a_new


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a step‑function that minimises the evaluation function.
    Returns the sequence as a list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2          # tiny safety margin

    # --------------------------------------------------------------
    # Problem size & initial point
    if 'height_sequence_1' in globals():
        a = np.array(height_sequence_1, dtype=float)
        a = np.maximum(a, 0.0)
        if a.sum() == 0.0:
            a = np.full_like(a, 1.0)
        a = a / a.sum()                     # simplex normalisation
        n = a.size
        print(f"[init] start from supplied sequence, n={n}")
    else:
        n = 5000                               # a good trade‑off
        a = np.full(n, 1.0 / n)                # uniform start on the simplex
        print(f"[init] uniform start, n={n}")

    # FFT length – next power of two larger than the longest convolution/correlation
    L = 1 << ((3 * n - 2).bit_length())

    # --------------------------------------------------------------
    # Helper to keep the best solution found so far
    best_seq = a.copy()
    best_score = _objective_simplex(best_seq, n, L)
    print(f"[init] O={best_score:.6f}")

    # ==============================================================
    # Phase 1 – Adam on a smoothed surrogate
    # ==============================================================

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    lr0 = 0.7                     # a slightly larger start‑learning‑rate
    alpha_start, alpha_end = 5.0, 5e4   # exponential anneal up to 5·10⁴

    max_iter_phase1 = 500_000
    phase1_deadline = start_time + budget_s * 0.40   # 40 % of the budget

    prune_every = 8000          # prune periodically
    prune_percent = 0.5         # drop the smallest 0.5 % of the mass

    it = 0
    while it < max_iter_phase1 and time.time() < phase1_deadline:
        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[phase‑1] iter {it:,} new best O={best_score:.6f}")

        # exponential schedule for α
        frac = it / max_iter_phase1
        alpha = alpha_start * ((alpha_end / alpha_start) ** frac)

        grad = _smooth_gradient(a, conv, alpha, L)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        lr = lr0 / math.sqrt(it + 1.0)       # 1/√t decay

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to help escape flat regions
        if np.random.rand() < 0.001:
            a_new += np.random.randn(n) * 1e-4

        # project back to the simplex
        a_new = _project_to_simplex(a_new)

        # modest pruning – remove the tiniest entries every few thousand steps
        if it % prune_every == 0 and it > 0:
            thresh = np.percentile(a_new, prune_percent)
            a_new[a_new < thresh] = 0.0
            a_new = _project_to_simplex(a_new)

        a = a_new
        it += 1

        # early‑stop if we are already past the global deadline
        if time.time() >= deadline:
            break

    # ==============================================================
    # Phase 2 – exact sub‑gradient descent
    # ==============================================================

    subgrad_iters = 300_000
    eta0 = 0.35
    phase2_deadline = start_time + budget_s * 0.65   # up to 65 % of total

    it2 = 0
    while it2 < subgrad_iters and time.time() < phase2_deadline:
        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[sub‑grad] iter {it2:,} new best O={best_score:.6f}")

        g = _subgradient_max_conv(a, conv)

        # step‑size decays like 1/√t (more aggressive than pure geometric)
        eta = eta0 / math.sqrt(it2 + 1.0)
        a = a - eta * g
        a = _project_to_simplex(a)

        it2 += 1
        if time.time() >= deadline:
            break

    # ==============================================================
    # Phase 3 – stochastic local refinement
    # ==============================================================

    max_iter_local = 200_000
    no_improve_limit = 6_000
    temperature0 = 0.07
    T = temperature0
    it_loc = 0
    no_improve = 0

    # start local search from the best sequence found so far
    a = best_seq.copy()
    conv = _conv_fft(a, L)
    cur_obj = 2.0 * n * conv.max()

    print(f"[phase‑3] start stochastic refinement, best O={best_score:.6f}")

    while time.time() < deadline and it_loc < max_iter_local:
        # ---- mass‑transfer move (random i → j) ----
        i, j = np.random.randint(n, size=2)
        if i != j and a[i] > 1e-14:
            # move up to 9 % of the source mass
            delta = a[i] * np.random.rand() * 0.09
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            a_new = _project_to_simplex(a_new)

            conv_new = _conv_fft(a_new, L)
            new_obj = 2.0 * n * conv_new.max()

            # Metropolis acceptance
            if new_obj < cur_obj or np.random.rand() < math.exp((cur_obj - new_obj) / max(T, 1e-12)):
                a = a_new
                conv = conv_new
                cur_obj = new_obj
                no_improve = 0
                if cur_obj < best_score:
                    best_score = cur_obj
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] NEW GLOBAL BEST O={best_score:.6f} (iter={it_loc})")
            else:
                no_improve += 1
        else:
            no_improve += 1

        # occasional exact sub‑gradient nudges (keeps directionality)
        if it_loc % 13 == 0:
            g = _subgradient_max_conv(a, conv)
            eta_loc = 0.07 * (1.0 - it_loc / max_iter_local)   # linear decay
            a = a - eta_loc * g
            a = _project_to_simplex(a)
            conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_score:
                best_score = cur_obj
                best_seq = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved O={best_score:.6f}")

        # temperature annealing
        T *= 0.9997

        # restart if stalled for a while
        if no_improve >= no_improve_limit:
            a = best_seq.copy()
            a += np.random.randn(n) * 0.0012
            a = np.clip(a, 0, None)
            a = _project_to_simplex(a)
            conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max()
            no_improve = 0

        it_loc += 1
        if time.time() >= deadline:
            break

    # ==============================================================
    # Phase 4 – linear‑programming refinement (optional, cheap)
    # ==============================================================

    if linprog is not None:
        # perform a few LP‑based improvement steps (very cheap compared to FFT loops)
        for lp_step in range(3):
            if time.time() >= deadline:
                break
            M = conv.max()
            a = _lp_refine(a, M, n)
            conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_score:
                best_score = cur_obj
                best_seq = a.copy()
                print(f"[LP‑refine {lp_step}] new best O={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final check with the provided evaluator
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:          # pragma: no cover
        print("[final] evaluate_sequence failed:", exc)

    return best_seq.tolist()
