def propose_candidate(seed=42, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0
    
    # Determine length n
    try:
        prev = height_sequence_1
        n = max(len(prev), 5000)
        prev_arr = np.asarray(prev, dtype=np.float64)
    except Exception:
        prev_arr = None
        n = 15000
    n = min(max(n, 5000), 25000)
    fft_len = 1 << ((2*n - 2).bit_length())
    
    def cheap_score(a):
        maxb = np.fft.ifft(np.fft.fft(a, fft_len)**2).real[:2*n-1].max()
        return 2.0 * n * maxb
    
    # initial candidates
    best_a = None
    best_val = np.inf
    
    # power law scan
    for p in np.linspace(0.5, 1.2, 141):
        if time.time() >= deadline: break
        a = np.arange(1, n+1, dtype=np.float64) ** (-p)
        a /= a.sum()
        val = cheap_score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
    if prev_arr is not None and prev_arr.size == n:
        a = prev_arr / prev_arr.sum()  # ensure normalized
        val = cheap_score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
    
    print("[Init] best cheap", best_val)
    
    # Random Dirichlet seeds
    for conc in np.logspace(-3, 0, 200):
        if time.time() >= deadline: break
        g = np.random.gamma(conc, 1.0, size=n)
        a = g / g.sum()
        val = cheap_score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
    
    # Projected Adam
    a = best_a.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 1.0   # base learning rate for simplex - will be scaled
    best_iter_a = a.copy()
    best_iter_val = best_val
    iter_cnt = 0
    while time.time() < deadline:
        iter_cnt += 1
        # compute conv via FFT
        conv = np.fft.ifft(np.fft.fft(a, fft_len)**2).real[:2*n-1]
        maxb = conv.max()
        val = 2.0 * n * maxb
        # update best
        if val < best_iter_val:
            best_iter_val = val
            best_iter_a = a.copy()
        # compute gradient for top K shifted terms
        k = 10
        # indices of top k conv entries
        if k >= conv.size:
            top_idx = np.arange(conv.size)
        else:
            top_idx = np.argpartition(conv, -k)[-k:]
        grad = np.zeros_like(a)
        for idx in top_idx:
            lo = max(0, idx - (n-1))
            hi = min(idx, n-1)
            rng = np.arange(lo, hi+1, dtype=np.int64)
            grad[rng] += 2.0 * a[idx - rng]
        grad /= float(k)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad**2)
        m_hat = m / (1 - beta1**iter_cnt)
        v_hat = v / (1 - beta2**iter_cnt)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        # step size scaling with iteration for stability
        step = step / (np.sqrt(iter_cnt) * 5.0)  # shrink over time
        a_new = a - step
        a_new = _simplex_projection(a_new)
        a = a_new
        # occasional random perturbation if stuck
        if iter_cnt % 200000 == 0:
            pert = 0.01 * np.random.rand(n)
            a = _simplex_projection(a + pert)
    
    # Local refinement: mixing with uniform or reversed
    uniform = np.full(n, 1.0/n)
    rev = best_iter_a[::-1]
    for mix in np.linspace(0.0, 0.4, 41):
        if time.time() >= deadline: break
        # blend best with uniform
        a2 = (1-mix) * best_iter_a + mix * uniform
        a2 = a2 / a2.sum()
        val = cheap_score(a2)
        if val < best_iter_val:
            best_iter_val = val
            best_iter_a = a2.copy()
        # blend with reversed
        a3 = (1-mix) * best_iter_a + mix * rev
        a3 = a3 / a3.sum()
        val = cheap_score(a3)
        if val < best_iter_val:
            best_iter_val = val
            best_iter_a = a3.copy()
    
    # final evaluation
    final_seq = best_iter_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print("Final score", final_score)
    except Exception as e:
        final_score = float('inf')
        print("Error", e)
    return final_seq

#!/usr/bin/env python
"""
Propose a non‑negative step function (list of heights) that minimises

        2·n·max_k (a*a)[k] / (Σ a_i)²

The algorithm is a lightweight “projected Adam” optimiser on the simplex,
augmented with:
  * a quick scan of power‑law families,
  * many Dirichlet seeds,
  * an adaptive top‑K sub‑gradient (the gradient of the k‑largest
    convolution entries),
  * occasional exponential‑gradient steps and small random perturbations
    to escape plateaus,
  * a final local mixing with the uniform and the reversed sequence.

All heavy work is done with FFT‑based convolutions, which makes
≈10 000 iterations on a 5 000‑element vector inexpensive – well within the
1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top level, no closures)
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the probability simplex
    { x ≥ 0 , Σ x = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:          # all entries are non‑positive → uniform point
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def power_law(n: int, p: float) -> np.ndarray:
    """Return a normalised power‑law a_i ∝ (i+1)^‑p (Σ a_i = 1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p)
    a /= a.sum()
    return a


def conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    fa = np.fft.fft(a, fft_len)
    c = np.fft.ifft(fa * fa).real
    return c[: 2 * a.size - 1]


def conv_max(a: np.ndarray, fft_len: int) -> float:
    """Maximum value of the convolution a*a."""
    return float(conv_fft(a, fft_len).max())


def cheap_score(a: np.ndarray, n: int, fft_len: int) -> float:
    """2·n·max(a*a) – the quantity that evaluate_sequence returns when Σa=1."""
    return 2.0 * n * conv_max(a, fft_len)


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The routine adapts its schedule to the available time budget and
    always returns a feasible (non‑negative) list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # --------------------------------------------------------------
    # Determine a working length.
    # --------------------------------------------------------------
    try:
        prev = height_sequence_1                     # provided globally
        n = max(len(prev), 5000)
        prev_arr = np.asarray(prev, dtype=np.float64)
    except Exception:
        prev_arr = None
        n = 15000

    n = min(max(n, 5000), 25000)                     # 5 k ≤ n ≤ 25 k
    fft_len = 1 << ((2 * n - 2).bit_length())       # next power of two ≥ 2·n‑1

    # --------------------------------------------------------------
    # 0 – initial scan (power laws + Dirichlet seeds)
    # --------------------------------------------------------------
    best_a = None
    best_val = np.inf

    # power‑law sweep
    for p in np.linspace(0.55, 0.80, 81):
        if time.time() >= deadline:
            break
        a = power_law(n, p)
        val = cheap_score(a, n, fft_len)
        if val < best_val:
            best_val = val
            best_a = a.copy()

    # previous construction, if compatible
    if prev_arr is not None and prev_arr.size == n:
        a = np.maximum(prev_arr, 0.0)                # safety
        if a.sum() == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / a.sum()
        val = cheap_score(a, n, fft_len)
        if val < best_val:
            best_val = val
            best_a = a.copy()

    # random Dirichlet seeds – wide range of concentrations
    for conc in np.logspace(-3.0, 0.5, 200):
        if time.time() >= deadline:
            break
        g = np.random.gamma(conc, 1.0, size=n)
        a = g / g.sum()
        val = cheap_score(a, n, fft_len)
        if val < best_val:
            best_val = val
            best_a = a.copy()

    print(f"[Init] best cheap = {best_val:.6f}")

    # --------------------------------------------------------------
    # 1 – Projected Adam on the simplex (sub‑gradient based)
    # --------------------------------------------------------------
    a = best_a.copy()
    m = np.zeros_like(a)            # first moment
    v = np.zeros_like(a)            # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 1.0                   # will be scaled down each iteration
    iter_cnt = 0
    best_iter_a = a.copy()
    best_iter_val = best_val
    last_improve = 0

    # hyper‑parameters
    topk_start = 15                 # be a little more aggressive than before
    topk_end = 5
    switch_iter = 200_000           # after this we tighten the gradient

    while time.time() < deadline:

        # ----- convolution & cheap score -----
        conv = conv_fft(a, fft_len)
        maxb = conv.max()
        cheap = 2.0 * n * maxb

        # ----- keep the best -----
        if cheap < best_iter_val - 1e-12:

            best_iter_a = a.copy()

        # ----- adaptive top‑k -----
        k_now = topk_end if iter_cnt > switch_iter else topk_start
        if k_now >= conv.size:

        else:

        grad = np.zeros_like(a)
        for idx in top_idx:

            hi = min(idx, n - 1)

            grad[rng] += 2.0 * a[idx - rng]
        grad /= float(k_now)

        # ----- Adam update (projected) -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iter_cnt)
        v_hat = v / (1.0 - beta2 ** iter_cnt)

        # learning‑rate schedule (decreases with √t)
        lr = base_lr / (np.sqrt(iter_cnt) * 5.0)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasionally use an exponentiated‑gradient step (10 % chance)
        if np.random.rand() < 0.10:
            mult = np.exp(-step)          # multiplicative update stays on simplex
            a = a * mult
            a = a / a.sum()
        else:
            a = a - step
            a = simplex_projection(a)

        # ----- restart if we are stuck for a while -----
        if iter_cnt - last_improve > 150_000:
            perturb = 0.02 * np.random.rand(n)
            a = simplex_projection(best_iter_a + perturb)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = iter_cnt
            print(f"[Iter {iter_cnt:07d}] Restart after stagnation.")

    # --------------------------------------------------------------
    # 2 – Small local mixing with uniform / reversed sequences
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    rev = best_iter_a[::-1]
    mixes = np.linspace(0.0, 0.35, 36)  # 0 % … 35 % in steps of 1 %

    for alpha in mixes:
        if time.time() >= deadline:
            break
        # blend with uniform
        cand = (1.0 - alpha) * best_iter_a + alpha * uniform
        cand = cand / cand.sum()
        val = cheap_score(cand, n, fft_len)
        if val < best_iter_val:
            best_iter_val = val
            best_iter_a = cand.copy()
            print(f"[Mix U {alpha:.3f}] improved cheap = {val:.6f}")

        # blend with reversed version
        cand = (1.0 - alpha) * best_iter_a + alpha * rev
        cand = cand / cand.sum()
        val = cheap_score(cand, n, fft_len)
        if val < best_iter_val:
            best_iter_val = val
            best_iter_a = cand.copy()
            print(f"[Mix R {alpha:.3f}] improved cheap = {val:.6f}")

    # --------------------------------------------------------------
    # 3 – Final exact evaluation (uses the provided evaluate_sequence)
    # --------------------------------------------------------------
    final_seq = best_iter_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:      # defensive – should never happen
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # --------------------------------------------------------------
    # 4 – Return the best sequence found.
    # --------------------------------------------------------------
    return final_seq
