"""
Optimiser for the step‑function bound  O = 2·n·max(conv(a,a)) / (sum a)^2.

The core of the method is a stochastic gradient descent on a soft‑max
parameterisation of the simplex (log‑its).  The loss uses a smooth
approximation of the max (soft‑max with an annealing temperature β that
grows exponentially).  Adam’s adaptive moments give a stable descent.
A set of diverse initialisations (uniform, exponential, double‑exponential
and the previous best solution) is tried for several support sizes.
When the budget is almost exhausted a fairly aggressive random‑move
local‑search (mass transfers, swaps and occasional uphill moves via a
simple annealing schedule) is performed.  The routine respects the 1000 s
budget, prints intermediate best values and finally returns the best
sequence found.

All numbers are non‑negative; the routine works directly with the
evaluate_sequence function supplied by the harness.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a power‑of‑two FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation of w (length 2n‑1) with a (length n).
    Returns vector of length n:  (w ⋆ a)[i] = Σ_t w[i+t]·a[t].
    """
    n = a.shape[0]
    m = w.shape[0]               # = 2n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


# ----------------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


# ----------------------------------------------------------------------
def _smooth_max_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Compute the soft‑max weights for a given β.
    When β is very large we simply return a one‑hot vector at the arg‑max.
    """
    if beta > 5000.0:
        w = np.zeros_like(conv)
        w[int(np.argmax(conv))] = 1.0
        return w
    M = conv.max()
    shifted = conv - M            # ≤ 0
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum == 0.0:             # numeric safety
        w = np.zeros_like(conv)
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= w_sum
    return w


# ----------------------------------------------------------------------
def _exact_objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))  (assumes sum(a)>0)."""
    n = a.size
    conv = _conv_self_fft(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _adam_step(x, g, m, v, it, lr, beta1=0.9, beta2=0.999, eps=1e-12):
    """Perform a single Adam update on the log‑it vector x."""
    m = beta1 * m + (1.0 - beta1) * g
    v = beta2 * v + (1.0 - beta2) * (g * g)

    m_hat = m / (1.0 - beta1 ** it)
    v_hat = v / (1.0 - beta2 ** it)

    x_new = x - lr * m_hat / (np.sqrt(v_hat) + eps)
    return x_new, m, v


# ----------------------------------------------------------------------
def _run_opt(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    rng: np.random.Generator | None = None,
    seed: int = 0,
) -> tuple[list[float], float]:
    """
    Optimise a length‑n non‑negative vector under the simplex constraint.
    Returns (best_sequence, best_exact_objective).
    """

    start = time.time()
    if rng is None:
        rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # initial log‑it vector ------------------------------------------------
    if init_a is not None:
        a0 = np.asarray(init_a, dtype=float)
        if a0.ndim != 1 or a0.size != n:
            a0 = rng.random(n)
        a0 = np.clip(a0, 0.0, None)
        if a0.sum() < 1e-12:
            a0 = rng.random(n)
        a0 = a0 / a0.sum()
    else:
        # try a few handcrafted seeds if no explicit init is given
        choice = rng.integers(4)
        if choice == 0:
            a0 = np.full(n, 1.0 / n)
        elif choice == 1:
            # exponential decay
            gamma = rng.uniform(1.0 / n, 5.0 / n)
            a0 = np.exp(-gamma * np.arange(n))
            a0 = a0 / a0.sum()
        elif choice == 2:
            # double‑exponential (sharper front, long tail)
            gamma = rng.uniform(1.0 / n, 3.0 / n)
            a0 = np.exp(-gamma * np.arange(n) ** 1.2)
            a0 = a0 / a0.sum()
        else:
            a0 = rng.random(n)
            a0 = a0 / a0.sum()

    # work in log‑it space (always keep probabilities positive)
    x = np.log(a0 + 1e-16)

    # Adam moments
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    # schedules
    beta_start = 0.5          # start temperature for soft‑max
    beta_end   = 20000.0      # very sharp approximation of max
    lr_start   = 0.18
    lr_end     = 0.0008
    max_iters  = 400_000

    best_a = None
    best_obj = np.inf

    it = 0
    while time.time() - start < time_limit and it < max_iters:
        it += 1

        # --------------------------------------------------------------
        # current probability vector (simplex)
        a = _softmax(x)

        # --------------------------------------------------------------
        # exact objective (for bookkeeping)
        conv = _conv_self_fft(a)
        M = conv.max()
        obj = 2.0 * n * M
        if obj < best_obj - 1e-16:
            best_obj = float(obj)
            best_a = a.copy()
            print(f"[n={n:5d} it={it:06d}] new best O={best_obj:.6f}")

        # --------------------------------------------------------------
        # schedule values (exponential)
        t = min(it / max_iters, 1.0)
        beta = beta_start * np.exp(np.log(beta_end / beta_start) * t)
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t)

        # --------------------------------------------------------------
        # smooth max weights
        w = _smooth_max_weights(conv, beta)

        # --------------------------------------------------------------
        # gradient w.r.t. a of the smooth objective (2×corr)
        g_a = 2.0 * _correlation_fft(w, a)

        # natural gradient for the soft‑max mapping
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)

        # --------------------------------------------------------------
        # Adam update on logits
        x, m, v = _adam_step(x, g_x, m, v, it, lr)

        # --------------------------------------------------------------
        # occasional random jitter to escape plateaus
        if it % 5000 == 0:
            x += rng.normal(scale=0.015, size=n)

        # --------------------------------------------------------------
        # restart from the current best every 60 000 iterations
        if it % 60_000 == 0 and best_a is not None:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.003, size=n)
            m.fill(0.0)
            v.fill(0.0)
            print(f"[n={n}] restart at iter {it}")

    # ------------------------------------------------------------------
    # sanity check – if optimiser never produced a point, fall back to init
    if best_a is None:
        best_a = _softmax(x)
        best_obj = _exact_objective(best_a)

    # ------------------------------------------------------------------
    # prune negligible entries (helps the final local search)
    thresh = 1e-13
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        best_obj = _exact_objective(best_a)
        print(f"[compress] length → {best_a.size}, O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # ==== final random local search (mass transfers & swaps) ===========
    remaining = time_limit - (time.time() - start)
    if remaining > 1.0:
        # simple simulated‑annealing schedule
        T0 = 0.02
        T_end = 0.0005
        local_start = time.time()
        a_loc = best_a.copy()
        obj_loc = best_obj
        n_loc = a_loc.size

        while time.time() - local_start < remaining * 0.7:
            # pick move type
            if rng.random() < 0.7:
                # mass transfer
                i, j = rng.choice(n_loc, size=2, replace=False)
                # move a random fraction of a_i (up to 8 %)
                delta = a_loc[i] * rng.random() * 0.08
                if delta <= 0:
                    continue
                a_new = a_loc.copy()
                a_new[i] -= delta
                a_new[j] += delta
            else:
                # swap two entries
                i, j = rng.choice(n_loc, size=2, replace=False)
                a_new = a_loc.copy()
                a_new[i], a_new[j] = a_new[j], a_new[i]

            # evaluate exact objective
            conv_new = _conv_self_fft(a_new)
            obj_new = 2.0 * n_loc * conv_new.max()

            # acceptance rule (Metropolis)
            if obj_new < obj_loc:
                a_loc, obj_loc = a_new, obj_new
                if obj_new < best_obj:
                    best_a, best_obj = a_new, obj_new
            else:
                # probabilistic uphill move
                T = T0 * (1 - (time.time() - local_start) / (remaining * 0.7))
                T = max(T, T_end)
                prob = np.exp(-(obj_new - obj_loc) / T)
                if rng.random() < prob:
                    a_loc, obj_loc = a_new, obj_new

        print(f"[local] final O={best_obj:.6f}")

    # final normalisation safeguard (evaluate_sequence expects any scale)
    best_seq = best_a.tolist()
    return best_seq, float(best_obj)


# ----------------------------------------------------------------------
def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **_ignore,
) -> list[float]:
    """
    Entry point used by the evaluation harness.
    Runs a suite of short Adam‑optimisations for several support sizes,
    followed by an aggressive random local search.  The best sequence
    encountered (as a plain list) is returned.
    """

    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # Possible lengths – we focus on the range where the previous run
    # performed well but also try a smaller and a larger size.
    lengths = [1500, 3000, 4000, 6000, 8000]

    # reuse previously discovered sequence if present
    prev_seq = None
    if 'height_sequence_1' in globals():
        prev_arr = np.asarray(height_sequence_1, dtype=float).copy()
        prev_arr = np.clip(prev_arr, 0.0, None)
        if prev_arr.sum() >= 0.01:
            prev_seq = prev_arr / prev_arr.sum()
            n0 = prev_seq.size
            if n0 not in lengths:
                lengths.insert(0, n0)

    best_seq = None
    best_val = float('inf')
    rng = np.random.default_rng(seed + 1234)

    for idx, n in enumerate(lengths):
        remaining = deadline - time.time()
        if remaining <= 1.0:
            break

        # time split: first (largest) run gets most budget
        if idx == 0:
            run_time = remaining * 0.70
        else:
            # distribute the rest equally among the remaining runs
            run_time = (remaining - 0.5) / (len(lengths) - idx)

        # ---- choose initialisation ------------------------------------
        init = None
        if idx == 0 and prev_seq is not None and prev_seq.size == n:
            init = prev_seq
        elif idx == 0 and prev_seq is not None and prev_seq.size > n:
            # down‑sample previous best (simple take first n entries)
            init = prev_seq[:n]
            init = init / init.sum()
        else:
            # no explicit init – let optimiser pick its own random start
            init = None

        seq, val = _run_opt(
            n=n,
            time_limit=run_time,
            init_a=init,
            rng=rng,
            seed=seed + idx * 17,
        )

        if val < best_val:
            best_val = val
            best_seq = seq
            print(f"[global] new overall best O={best_val:.6f} (n={n})")

    # ------------------------------------------------------------------
    # Final evaluation via the provided hook (mostly for debugging)
    try:
        final_score = evaluate_sequence(best_seq)
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_seq
