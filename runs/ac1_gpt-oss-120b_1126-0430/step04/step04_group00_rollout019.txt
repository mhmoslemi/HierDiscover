#!/usr/bin/env python
"""
Optimising the step‑function bound.

The objective (for a non‑negative vector a with sum = 1) is
    J(a) = 2·n·max_k (a*a)[k] .
Because the convolution (a*a) is non‑negative it can be written as a
linear combination of the entries of a, therefore we can optimise J by
projected gradient descent on the probability simplex.
The gradient of the *exact* max is easy to compute:
    grad_a[i] = 4·n·a[k*‑i]   (k* = argmax conv)
and the gradient w.r.t. the logits `z` used for a soft‑max
parametrisation is  grad_z = a·(grad_a‑μ)  with μ = ⟨grad_a,a⟩.

To avoid the nondifferentiability of the max we start with a smooth
approximation (log‑sum‑exp) and decay the smoothing factor to zero.
Multiple families of initial points (uniform, power‑law, geometric,
U‑shaped) are surveyed, the best is then refined with Adam on the
logits.  The routine respects the 1000 s budget, uses only NumPy and
no external files.

The final sequence (list of non‑negative floats) is returned.  It will
be evaluated with the provided ``evaluate_sequence`` function.
"""
import time
import math
import numpy as np

# -------------------------------------------------------------
# Helper utilities – must be top level (no closures)
# -------------------------------------------------------------

def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it representation that yields ``a`` after soft‑max."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """
    Smallest power‑of‑two ≥ 3·n‑2,
    sufficient for a linear convolution of length (2·n‑1).
    """
    return 1 << ((3 * n - 2).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """Return the maximum entry of the linear self‑convolution a*a."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    # Numerical noise may create tiny negatives – clip.
    conv = np.maximum(conv, 0.0)
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float) -> tuple[float, np.ndarray]:
    """
    Smoothed max (log‑sum‑exp) together with its gradient w.r.t. ``a``.
    The smoothing constant is eps = eps_factor·max(conv).
    Returns (max_conv, grad_a) where grad_a = ∂(2·n·smooth_max)/∂a .
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]

    max_conv = conv.max()
    eps = max(max_conv * eps_factor, 1e-12)

    # soft‑max weights over the convolution entries
    shifted = conv - max_conv          # for numerical stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or np.isnan(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # correlation a ⋆ w  (linear, length n)
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # gradient of 2·n·smooth_max w.r.t. a (already multiplied by 4·n)
    grad_a = 4.0 * n * corr
    return float(max_conv), grad_a


def _conv_max_and_subgrad(a: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Exact max of the convolution together with a sub‑gradient.
    For the arg‑max index k* the gradient is
        grad_a[i] = 4·n·a[k*‑i]   if 0≤k*‑i<n, else 0.
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    conv = np.maximum(conv, 0.0)
    k_star = int(np.argmax(conv))
    max_conv = float(conv[k_star])

    # build sub‑gradient
    idx = np.arange(n)
    j = k_star - idx
    mask = (j >= 0) & (j < n)
    grad_a = np.zeros_like(a)
    grad_a[mask] = 4.0 * n * a[j[mask]]
    return max_conv, grad_a


def _initial_candidates(n: int) -> list[np.ndarray]:
    """Generate a collection of plausible starting distributions."""
    cand = []

    # 1. uniform
    cand.append(np.full(n, 1.0 / n))

    # 2. power‑law a_i ∝ (i+1)^{-p}
    p_grid = np.linspace(0.3, 1.2, 19)
    idx = np.arange(1, n + 1, dtype=np.float64)
    for p in p_grid:
        a = idx ** (-p)
        a /= a.sum()
        cand.append(a)

    # 3. geometric a_i ∝ r^i  (r close to 1)
    r_grid = np.linspace(0.999, 0.99999, 15)
    for r in r_grid:
        a = np.power(r, np.arange(n, dtype=np.float64))
        a /= a.sum()
        cand.append(a)

    # 4. symmetric geometric (distance from nearest end)
    for r in r_grid:
        d = np.minimum(np.arange(n), np.arange(n)[::-1])
        a = np.power(r, d.astype(np.float64))
        a /= a.sum()
        cand.append(a)

    # 5. U‑shaped: a_i ∝ (i+1)^p + (n-i)^p   (p>0)
    p_grid_u = np.linspace(0.5, 2.0, 7)
    i = np.arange(n, dtype=np.float64)
    for p in p_grid_u:
        a = (i + 1) ** p + (n - i) ** p
        a /= a.sum()
        cand.append(a)

    # 6. double‑power‑law (ends heavier, centre lighter)
    p_grid_dp = np.linspace(0.5, 2.0, 7)
    for p in p_grid_dp:
        a = (i + 1) ** (-p) + (n - i) ** (-p)
        a /= a.sum()
        cand.append(a)

    return cand


def _best_candidate_from_pool(candidates: list[np.ndarray], n: int) -> np.ndarray:
    """Select the candidate with the smallest cheap score."""
    best = None
    best_score = float("inf")
    for a in candidates:
        max_c = _conv_max(a)
        cheap = 2.0 * n * max_c
        if cheap < best_score:
            best_score = cheap
            best = a
    return best

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) beats the current bound.
    The routine respects a ~1000 s time budget and performs a
    multistart smoothed‑max Adam optimisation on the probability simplex.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0   # small safety margin

    # ------------------------------------------------------------------
    # 0. Problem size – a few ten‑thousands of entries give a good trade‑off.
    # ------------------------------------------------------------------
    n = 20000
    max_n = 20000       # we keep the support size fixed; enlarging later gave no benefit

    # ------------------------------------------------------------------
    # 1. Generate a pool of promising initial points and pick the best.
    # ------------------------------------------------------------------
    print("[Setup] Generating initial candidates (n=%d) …" % n)
    candidates = _initial_candidates(n)
    a = _best_candidate_from_pool(candidates, n)
    print("[Setup] Best initial cheap score = %.6f" % (2.0 * n * _conv_max(a)))

    # Small random jitter to break symmetry
    z = _logits_from_distribution(a)
    z += 0.01 * np.random.randn(n)

    # ------------------------------------------------------------------
    # 2. Optimisation loop – Adam on logits with a decreasing smoothing factor.
    # ------------------------------------------------------------------
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_alpha = 0.20          # a slightly larger base learning‑rate
    eps_factor = 0.12          # initial smoothing factor (ε = eps_factor·max)
    eps_factor_min = 5e-4
    eps_factor_decay = 0.9985  # decay a bit faster than the original code

    best_seq = _softmax(z)                 # current best (probability vector)
    best_score = 2.0 * n * _conv_max(best_seq)
    print("[Optim] Starting cheap score = %.6f" % best_score)

    it = 0
    last_improve_iter = 0
    report_every = 3000
    max_iters = 1_200_000

    while time.time() < deadline and it < max_iters:
        it += 1

        # current point on the simplex
        a = _softmax(z)

        # cheap evaluation + smooth gradient
        max_conv, grad_a = _conv_max_and_grad(a, eps_factor)

        cheap_score = 2.0 * n * max_conv

        # keep the best ever seen
        if cheap_score < best_score:
            best_score = cheap_score
            best_seq = a.copy()
            last_improve_iter = it
            if it % report_every == 0:
                print(f"[Iter {it:07d}] NEW BEST cheap = {cheap_score:.6f}")

        # gradient w.r.t. logits (mirror‑descent on the simplex)
        mu = np.dot(grad_a, a)
        grad_z = a * (grad_a - mu)

        # Adam moment updates (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step‑size schedule (inverse‑sqrt decay)
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step_dir = - m_hat / (np.sqrt(v_hat) + eps_adam)
        trial_alpha = alpha_t

        # back‑tracking line‑search (up to 4 halvings)
        accepted = False
        for _ in range(4):
            z_new = z + trial_alpha * step_dir
            # keep exp(z) safe
            z_new = np.clip(z_new, -50.0, 50.0)
            a_new = _softmax(z_new)
            new_max = _conv_max(a_new)
            new_score = 2.0 * n * new_max
            if new_score <= cheap_score:
                z = z_new
                accepted = True
                break
            trial_alpha *= 0.5

        if not accepted:
            # no improvement – still take a modest step
            z += alpha_t * step_dir
            z = np.clip(z, -50.0, 50.0)

        # occasional random perturbations to escape plateaux
        if it % 5000 == 0 and np.random.rand() < 0.20:
            z += 0.018 * np.random.randn(n)
            m *= 0.6
            v *= 0.6

        # gradual reduction of the smoothing factor
        eps_factor = max(eps_factor_min, eps_factor * eps_factor_decay)

        # early termination if stagnation after many iterations
        if it - last_improve_iter > 300_000:
            print(f"[Iter {it:07d}] Stagnation – terminating early.")
            break

    # ------------------------------------------------------------------
    # 3. Final exact evaluation with the provided evaluate_sequence()
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:

        print(f"[Result] Final exact score = {final_score:.6f}")
    except Exception as exc:
        final_score = float("inf")
        print(f"[Error] evaluate_sequence raised an exception: {exc}")

    # ------------------------------------------------------------------
    # 4. Return best sequence.
    # ------------------------------------------------------------------
    return final_seq
