"""
search for a non‑negative step function minimizing

    O(a) = 2·n·max(conv(a,a)) / (sum(a)²)

The optimizer works in two stages.

Stage 1 – multiple short Adam runs.
    * Different initialisations are tried:
        – random dense vectors (soft‑max of Gaussian logits)
        – random sparse vectors (few non‑zero entries, Dirichlet‑distributed)
    * The Adam schedule uses an exponentially increasing “soft‑max” temperature
      (β) to approximate the ℓ∞‑norm of the convolution and a decreasing learning rate.
    * The best found vector for the current length is kept.

Stage 2 – a fast greedy hill‑climber.
    * It repeatedly swaps a non‑zero entry with a zero entry (or moves a tiny
      fraction of mass) and accepts the move only if the exact objective improves.
    * The remaining time budget is spent here, after which the final sequence is
      returned.

Both stages respect the 1000 s budget.  The code prints intermediate best
objective values and finally the value reported by ``evaluate_sequence``.
"""

import time
import numpy as np

# -------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT lengths)."""
    return 1 << (x - 1).bit_length()


# -------------------------------------------------------------
def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full (non‑circular) self‑convolution using real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv


# -------------------------------------------------------------
def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation of the (full) convolution result `w` with the vector `a`.
    Implements g_i = Σ_t w[i+t]·a[t] for i = 0…n‑1 using FFT.
    """
    n = a.shape[0]
    m = w.shape[0]                     # = 2n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


# -------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) assuming sum(a)=1.
    The function is scale‑invariant (the denominator cancels for sum≠1).
    """
    n = a.size
    conv = _conv_self_fft(a)
    return 2.0 * n * conv.max()


# -------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


# -------------------------------------------------------------
def _run_opt(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
    beta_end: float = 8000.0,
    lr_end: float = 0.001,
    max_iters: int = 300_000,
) -> tuple[list[float], float]:
    """
    Adam optimiser on the log‑it (soft‑max) parameters.
    Returns the best sequence (as a Python list) and its exact objective.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # ---- initialise -------------------------------------------------
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
        a = a / a.sum()
    else:
        a = rng.random(n)
        a = a / a.sum()

    # work in log‑it space (simplex is enforced by soft‑max)
    x = np.log(a + 1e-16)          # avoid log(0)

    m = np.zeros_like(x)          # Adam first moment
    v = np.zeros_like(x)          # Adam second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # ---- schedules --------------------------------------------------
    beta_start = 0.5               # start of the smooth‑max temperature
    lr_start = 0.15                # initial learning‑rate (logits)

    best_a = a.copy()
    best_obj = _objective(best_a)

    it = 0
    while time.time() - start < time_limit and it < max_iters:
        it += 1

        # ---- probabilities (simplex) ---------------------------------
        a = _softmax(x)

        # ---- exact objective -----------------------------------------
        conv = _conv_self_fft(a)
        M = conv.max()
        obj = 2.0 * n * M
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[n={n:5d} it={it:06d}] new best O={best_obj:.6f}")

        # ---- schedule parameters --------------------------------------
        t = min(it / max_iters, 1.0)
        beta = beta_start * np.exp(np.log(beta_end / beta_start) * t)
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t)

        # ---- smooth‑max weights ---------------------------------------
        if beta > 4000.0:
            # very sharp → use true arg‑max
            kmax = int(np.argmax(conv))
            w = np.zeros_like(conv)
            w[kmax] = 1.0
        else:
            shifted = conv - M                # ≤0
            w = np.exp(beta * shifted)
            w = w / w.sum()                  # length 2n‑1, sums to 1

        # ---- gradient w.r.t. a (smooth max) -------------------------
        g_a = 2.0 * _correlation_fft(w, a)   # shape (n,)

        # ---- natural gradient (log‑it space) ------------------------
        dot = np.dot(a, g_a)                # scalar ⟨a,g_a⟩
        g_x = a * (g_a - dot)               # shape (n,)

        # ---- Adam update ---------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---- occasional jitter (helps escape plateaus) -------------
        if it % 6000 == 0:
            jitter = rng.normal(scale=0.02, size=n)
            x += jitter

        # ---- restart from current best every so often ---------------
        if it % 70_000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.005, size=n)
            m.fill(0.0)
            v.fill(0.0)
            print(f"[n={n:5d}] restart at iter {it}")

    # ---- compress negligible entries ---------------------------------
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        n_new = best_a.size
        best_obj = _objective(best_a)
        print(f"[compress] reduced length {n} → {n_new}, O={best_obj:.6f}")

    # ---- light‑weight local search -----------------------------------
    remaining = time_limit - (time.time() - start)
    if remaining > 0.5:
        a_loc = best_a.copy()
        obj_loc = best_obj
        n_loc = a_loc.size
        while time.time() - start < time_limit * 0.95:
            # pick a random pair (i in support, j outside) and swap their masses
            supp = np.where(a_loc > 0)[0]
            if supp.size == 0:
                break
            i = rng.choice(supp)
            j = rng.choice(np.setdiff1d(np.arange(n_loc), supp), replace=False)
            a_new = a_loc.copy()
            a_new[j] = a_new[i]
            a_new[i] = 0.0
            # renormalise (the total mass stays 1)
            conv_new = _conv_self_fft(a_new)
            obj_new = 2.0 * n_loc * conv_new.max()
            if obj_new < obj_loc:
                a_loc, obj_loc = a_new, obj_new
                if obj_new < best_obj:
                    best_a, best_obj = a_new, obj_new
        print(f"[local] final local O={best_obj:.6f}")

    return best_a.tolist(), best_obj

def _random_sparse_init(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """
    Return a vector of length `n` with exactly `k` non‑zero entries.
    The non‑zero entries are placed at distinct random positions and
    Dirichlet‑distributed (exponential draws) so that they sum to 1.
    """
    if k > n:
        k = n
    idx = rng.choice(n, size=k, replace=False)
    weights = rng.exponential(scale=1.0, size=k)
    weights /= weights.sum()
    a = np.zeros(n, dtype=float)
    a[idx] = weights
    return a


# -------------------------------------------------------------
def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **_ignore,
) -> list[float]:
    """
    Main entry point.
    Runs several optimisation attempts (different lengths and
    initialisations) inside the allotted time budget and returns the best
    sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ---- decide which lengths to try ---------------------------------
    lengths = [500, 1000, 2000, 4000]          # baseline set
    if 'height_sequence_1' in globals():
        seq0 = np.asarray(height_sequence_1, dtype=float).copy()
        seq0 = np.clip(seq0, 0.0, None)
        if seq0.sum() < 0.01:
            seq0 = np.ones_like(seq0)
        seq0 = seq0 / seq0.sum()
        n0 = seq0.size
        if n0 not in lengths:
            lengths.insert(0, n0)            # try the previous length first

    # ensure monotonic decreasing (largest first)
    lengths = sorted(set(lengths), reverse=True)

    best_seq = None
    best_obj = float('inf')

    # ---- iterate over the chosen lengths -------------------------------
    for idx, n in enumerate(lengths):
        remaining = deadline - time.time()
        if remaining <= 1.0:
            break

        # allocate time: the first (largest) length gets the bulk,
        # the rest share the rest.
        if idx == 0:
            run_time = remaining * 0.70
        else:
            # split the leftover equally among remaining lengths
            run_time = (remaining - 0.3) / (len(lengths) - idx)

        print(f"\n[global] start length n={n}, allocated time {run_time:.1f}s")

        # ---- number of independent Adam runs for this length -----------
        # keep it small so that we also have time for hill‑climbing later
        n_runs = max(1, int(run_time // 30))   # roughly 30 s per run
        run_time_per = run_time / n_runs

        # ---- try a few different initialisations -----------------------
        for run_id in range(n_runs):
            rng = np.random.default_rng(seed + idx * 100 + run_id)

            # choose initialisation
            if run_id % 2 == 0:
                # dense random init
                init = None
            else:
                # sparse init: size roughly 0.6·√n (heuristic)
                k = max(2, int(0.6 * np.sqrt(n)))
                init = _random_sparse_init(n, k, rng)

            # slightly different hyper‑parameters for the last run
            beta_end = 12000.0 if run_id == n_runs - 1 else 8000.0
            lr_end = 0.0008 if run_id == n_runs - 1 else 0.001
            max_iters = 400_000

            seq, val = _run_opt(
                n=n,
                time_limit=run_time_per,
                init_a=init,
                seed=seed + idx * 1000 + run_id,
                beta_end=beta_end,
                lr_end=lr_end,
                max_iters=max_iters,
            )
            if val < best_obj:
                best_obj = val
                best_seq = seq
                print(f"[global] new overall best O={best_obj:.6f} (n={n})")

        # ---- after the Adam runs, perform a focused hill‑climb ----------
        # give the remaining slice (≈20 % of the original allocation) to
        # a cheap swap‑based optimiser.
        extra_time = deadline - time.time()
        if extra_time > 1.0:
            print(f"[global] hill‑climb on best seq (n={n}) for {extra_time:.1f}s")
            a = np.asarray(best_seq, dtype=float)
            # ensure sum=1 (tiny numerical drift may appear)
            a /= a.sum()
            # simple swap‑based local search
            n_loc = a.size
            rng = np.random.default_rng(seed + 9999 + idx)
            start_hc = time.time()
            while time.time() - start_hc < extra_time * 0.80:
                supp = np.where(a > 1e-12)[0]
                if supp.size == 0:
                    break
                i = rng.choice(supp)
                # pick a zero entry
                zeros = np.where(a <= 1e-12)[0]
                if zeros.size == 0:
                    break
                j = rng.choice(zeros)

                # propose swapping the whole mass at i with a zero at j
                a_new = a.copy()
                a_new[j] = a_new[i]
                a_new[i] = 0.0

                # evaluate objective (exact)
                conv_new = _conv_self_fft(a_new)
                obj_new = 2.0 * n_loc * conv_new.max()
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_seq = a_new.tolist()
                    a = a_new
                    # optional early stop if we beat the target
                    if best_obj < 1.503:
                        break
            print(f"[global] after hill‑climb O={best_obj:.6f}")

    # ---- final verification with the provided evaluate_sequence ----------
    try:
        final_score = evaluate_sequence(best_seq)
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_seq
