"""
Propose a step‑function (non‑negative list) that minimises

        O(a) = 2·n·max(conv(a,a)) / (sum(a)²)

The optimizer works inside the 1000 s wall‑time budget and consists of
four phases:

1️⃣  *Smooth surrogate (log‑sum‑exp) minimisation* – an Adam optimiser
    on the smoothed objective  (2·n/α)·log∑ₖexp(α·convₖ).  The smoothing
    parameter α is increased exponentially from 5 to 1e5, turning the
    surrogate into the true max‑operator.

2️⃣  *Projected sub‑gradient descent* – after the surrogate is sharp we
    switch to the genuine non‑smooth objective and perform classic
    sub‑gradient steps with a geometrically decaying step‑size, always
    projecting back onto the probability simplex.

3️⃣  *Guided stochastic refinement* – a simulated‑annealing loop where
    a mass‑transfer move is performed from the index that contributes the
    most to the current maximal convolution entry (obtained from the
    sub‑gradient) to an index that contributes the least.  The move is
    accepted if it improves the exact objective or with a Metropolis
    probability that slowly cools.  Occasionally a pure sub‑gradient step
    is inserted to keep the search directed.

4️⃣  *Final polishing* – a few extra Adam steps with a tiny learning‑rate
    to clean up tiny numerical artefacts.

All intermediate solutions are kept in `best_seq`; the best one found
before the deadline is returned as a plain list of floats.
"""

import time
import math
import numpy as np
from scipy import sparse
# scipy.optimize.linprog is imported only if the optional LP polishing
# is activated (it is cheap for the problem sizes we use).
from scipy.optimize import linprog


# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:                     # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """
    Full (linear) convolution a*a using an FFT of length L.
    Returns the first 2·len(a)-1 entries.
    """
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray, L: int) -> float:
    """Exact objective 2·n·max(conv) (the denominator = 1 because we keep sum=1)."""
    n = a.size
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a.
    Returns a vector of the same size as a.
    """
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)          # indices where the max is attained
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, conv: np.ndarray,
                     alpha: float, L: int) -> np.ndarray:
    """
    Gradient of (2·n/α)·log Σₖ exp(α·convₖ) w.r.t. a.
    The formula reduces to 2·corr(w, a) where wₖ ∝ exp(α·convₖ).
    """
    # soft‑max weights over the convolution entries
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)          # stabilise
    w = exp_scaled / exp_scaled.sum()

    # correlation w ⋆ a (g_i = 2·Σⱼ w_{i+j} aⱼ)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)   # length ≥ 3·n‑2
    n_len = a.size
    grad = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]
    return grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function sequence that minimises the
    evaluation function.  Returns the sequence as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.1          # tiny safety margin

    # --------------------------------------------------------------
    # 1. problem size & initial point
    if 'height_sequence_1' in globals():
        a = np.array(height_sequence_1, dtype=float)
        a = np.maximum(a, 0.0)
        if a.sum() == 0:
            a = np.full_like(a, 1.0)
        a = a / a.sum()
        n = a.size
        print(f"[init] Using supplied sequence, n={n}")
    else:
        # default size – 4600 gave a good trade‑off in earlier runs
        n = 4600
        # start from an arcsine‑like shape (known to flatten the sum
        # distribution) plus a tiny random jitter
        i = np.arange(n, dtype=float)
        a = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
        a += 1e-4 * np.random.rand(n)               # jitter
        a = np.maximum(a, 0.0)
        a = a / a.sum()
        print(f"[init] Uniform‑arcsine start, n={n}")

    # FFT length – next power of two large enough for all convolutions/correlations
    L = 1 << ((3 * n - 2).bit_length())

    # --------------------------------------------------------------
    # bookkeeping
    best_seq = a.copy()
    best_score = _objective_exact(best_seq, L)
    print(f"[init] O={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam on the smoothed log‑sum‑exp surrogate
    phase1_end = start_time + budget_s * 0.30
    beta1, beta2, eps = 0.9, 0.999, 1e-12
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    lr0 = 0.6
    alpha_start, alpha_end = 5.0, 1e5

    it1 = 0
    while time.time() < phase1_end:
        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[phase‑1] iter {it1} new best O={best_score:.6f}")

        # exponential schedule for α
        frac = (time.time() - start_time) / (phase1_end - start_time)
        alpha = alpha_start * ((alpha_end / alpha_start) ** frac)

        # gradient of smooth surrogate
        grad = _smooth_gradient(a, conv, alpha, L)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it1 + 1))
        v_hat = v / (1.0 - beta2 ** (it1 + 1))

        lr = lr0 / math.sqrt(it1 + 1)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape flat plateaus
        if np.random.rand() < 0.001:
            a_new += np.random.randn(n) * 1e-5

        a_new = _project_to_simplex(a_new)

        # prune tiny entries every few thousand steps – helps speed‑up later phases
        if it1 % 5000 == 0 and it1 > 0:
            thresh = np.percentile(a_new, 0.5)   # drop ~0.5 % smallest mass
            a_new[a_new < thresh] = 0.0
            a_new = _project_to_simplex(a_new)

        a = a_new
        it1 += 1

    # ------------------------------------------------------------------
    # Phase 2 – exact sub‑gradient descent
    phase2_end = start_time + budget_s * 0.55
    eta0 = 0.30
    it2 = 0
    while time.time() < phase2_end:
        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[phase‑2] iter {it2} new best O={best_score:.6f}")

        g = _subgradient_max_conv(a, conv)

        # geometrically decaying step size
        eta = eta0 * (0.9995 ** it2)
        a = a - eta * g
        a = _project_to_simplex(a)

        # occasional pruning (keep the vector reasonably sparse)
        if it2 % 3000 == 0 and it2 > 0:
            thresh = np.percentile(a, 0.5)
            a[a < thresh] = 0.0
            a = _project_to_simplex(a)

        it2 += 1

    # ------------------------------------------------------------------
    # Phase 3 – guided stochastic refinement (simulated annealing)
    phase3_end = start_time + budget_s * 0.95
    T = 0.15                     # initial temperature
    it3 = 0
    no_improve = 0
    max_no_improve = 10000
    a = best_seq.copy()         # start from the best found so far
    conv = _conv_fft(a, L)
    cur_obj = 2.0 * n * conv.max()

    while time.time() < phase3_end and it3 < 500_000:
        # --------------------------------------------------------------
        # Determine the most/least harmful indices via the sub‑gradient
        subg = _subgradient_max_conv(a, conv)
        src = np.argmax(subg)                     # contributes most to the peak
        # choose a destination with very small (or zero) contribution
        low_candidates = np.where(subg <= np.percentile(subg, 10))[0]
        if low_candidates.size == 0:
            dst = (src + 1) % n
        else:
            dst = np.random.choice(low_candidates)

        # propose a mass‑transfer move
        if a[src] > 1e-12:
            delta = a[src] * np.random.rand() * 0.07   # up to 7 % of source mass
        else:
            delta = 0.0

        # fallback to a completely random move if the guided move is impossible
        if delta <= 0.0 or src == dst:
            i, j = np.random.randint(n, size=2)
            if i != j and a[i] > 1e-14:
                delta = a[i] * np.random.rand() * 0.09
                src, dst = i, j
            else:
                it3 += 1
                continue

        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta
        # projection is not needed – the operation preserves sum and non‑negativity

        conv_new = _conv_fft(a_new, L)
        new_obj = 2.0 * n * conv_new.max()

        # Metropolis acceptance
        accept = False
        if new_obj < cur_obj:
            accept = True
        else:
            prob = math.exp((cur_obj - new_obj) / max(T, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a = a_new
            conv = conv_new
            cur_obj = new_obj
            no_improve = 0
            if cur_obj < best_score:
                best_score = cur_obj
                best_seq = a.copy()
                print(f"[phase‑3] iter {it3} new global best O={best_score:.6f}")
        else:
            no_improve += 1

        # occasional exact sub‑gradient step (keeps the search directed)
        if it3 % 11 == 0:
            g = _subgradient_max_conv(a, conv)
            eta_loc = 0.07 * (1.0 - it3 / 500_000)
            a = a - eta_loc * g
            a = _project_to_simplex(a)
            conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_score:
                best_score = cur_obj
                best_seq = a.copy()
                print(f"[phase‑3‑sub] iter {it3} improved O={best_score:.6f}")

        # temperature decay
        T *= 0.9997

        # restart if we have been stuck for a while
        if no_improve >= max_no_improve:
            a = best_seq.copy()
            a += np.random.randn(n) * 1e-4
            a = np.clip(a, 0, None)
            a = _project_to_simplex(a)
            conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max()
            no_improve = 0
            print(f"[restart] after {no_improve} non‑improving steps")

        # prune very tiny entries every few thousand iterations
        if it3 % 5000 == 0 and it3 > 0:
            thresh = np.percentile(a, 0.3)
            a[a < thresh] = 0.0
            a = _project_to_simplex(a)
            conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max()

        it3 += 1

    # ------------------------------------------------------------------
    # Phase 4 – final light Adam polishing (optional, cheap)
    # This cleans up any remaining high‑frequency noise.
    phase4_end = deadline
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    it4 = 0
    while time.time() < phase4_end and it4 < 80_000:
        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[phase‑4] iter {it4} new best O={best_score:.6f}")

        # use a *very* small learning‑rate, α kept moderate (≈1e4)
        alpha = 2e4
        grad = _smooth_gradient(a, conv, alpha, L)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it4 + 1))
        v_hat = v / (1.0 - beta2 ** (it4 + 1))
        lr = 0.12 / math.sqrt(it4 + 1)      # tiny step
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_to_simplex(a)

        it4 += 1

    # ------------------------------------------------------------------
    # Return the best candidate found
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence failed:", exc)

    return best_seq.tolist()
