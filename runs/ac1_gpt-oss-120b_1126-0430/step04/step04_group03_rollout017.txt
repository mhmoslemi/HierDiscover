def _weighted_conv_gradient(a, w):
    """Given a vector a (len n) and weight w (len 2n-1)
    returns gradient of sum_i conj_i w_i wrt a, which equals
    2 * sum_j a_j w[i+j] for i=0..n-1.
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv = _conv_linear(a, w_rev)  # length 3n-2
    offset = 2 * n - 2
    grad = 2.0 * conv[offset - np.arange(n)]
    return grad

    best_a = a.copy()

    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    beta_soft = 1.0
    beta_soft_max = 200.0
    beta_inc = (beta_soft_max - beta_soft) / max_iters

    for t in range(1, max_iters+1):
        if time.time() >= deadline: break

        conv = _conv_fft(a)
        # update softmax temperature
        beta_soft = min(beta_soft_max, beta_soft + beta_inc)

        # compute softmax weights
        maxc = conv.max()
        exp_term = np.exp(beta_soft * (conv - maxc))
        w = exp_term / exp_term.sum()

        # compute gradient
        g = _weighted_conv_gradient(a, w)

        # Adam update on a (not log)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)

"""
Propose a step‑function (non‑negative list of heights) that minimises

    O = 2·n·max(conv(a,a)) / (sum(a)²)

The search consists of three stages:

1. **Parametric sweep** – test two simple families of shapes
       *power law*      a_i ∝ (i+β)^(-α)
       *U‑shape*        a_i ∝ ((i+β+1)*(n−i+β))^(-α)
   on a modest grid and keep the best candidate.

2. **Local random refinement** – perturb the best parameters and
   keep any improvement.

3. **Smooth‑max mirror descent** – minimise a smooth approximation of
   the max‑convolution using Adam on the simplex.  The smoothness
   parameter β grows from 1 to ≈200 so that the algorithm first sees
   a mellow landscape and later focuses on the true max.  Jitter is
   added when progress stalls.  The whole routine respects the 1000‑s
   budget, prints progress and finally returns the best vector as a
   plain ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_linear(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using a power‑of‑two FFT."""
    n = x.shape[0]
    m = y.shape[0]
    size = 1 << ((n + m - 1).bit_length())
    X = np.fft.fft(x, n=size)
    Y = np.fft.fft(y, n=size)
    conv = np.fft.ifft(X * Y).real[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (size 2·n‑1) via FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 2).bit_length())
    A = np.fft.fft(a, n=size)
    conv = np.fft.ifft(A * A).real[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def _objective_norm1(a: np.ndarray) -> float:
    """Objective 2·n·max(conv) assuming Σa = 1."""
    n = a.shape[0]
    conv = _conv_fft(a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
def _weighted_conv_gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·conv_k(a) w.r.t. a (a ≥ 0, Σa = 1).
    Result: g_i = 2·∑_j a_j·w_{i+j},  i = 0..n‑1.
    """
    n = a.shape[0]
    w_rev = w[::-1]                     # reverse to turn the sum into a convolution
    conv = _conv_linear(a, w_rev)       # length = n + (2n‑1) – 1 = 3n‑2
    offset = 2 * n - 2
    g = 2.0 * conv[offset - np.arange(n)]
    return g


# ----------------------------------------------------------------------
def _shape_power(alpha: float, beta: float, n: int) -> np.ndarray:
    """Monotone decreasing power‑law shape."""
    i = np.arange(1, n + 1, dtype=float)
    a = (i + beta) ** (-alpha)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


# ----------------------------------------------------------------------
def _shape_u(alpha: float, beta: float, n: int) -> np.ndarray:
    """U‑shaped (large at the ends, small in the centre) shape."""
    i = np.arange(n, dtype=float)
    # product (i+1+β)*(n-i+β) is smallest at the ends → larger values there
    a = ((i + 1.0 + beta) * (n - i + beta)) ** (-alpha)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


# ----------------------------------------------------------------------
def _smooth_optimize(a0: np.ndarray,
                     max_iters: int,
                     deadline: float,
                     lr: float = 0.1,
                     beta_start: float = 1.0,
                     beta_end: float = 200.0) -> tuple[np.ndarray, float]:
    """
    Adam‑style optimisation of the smooth‑max objective.
    Returns the best vector seen and its objective value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective_norm1(a)

    n = a.size
    m = np.zeros_like(a)            # first moment
    v = np.zeros_like(a)            # second moment
    b1, b2 = 0.9, 0.999
    eps = 1e-8

    beta = beta_start
    beta_inc = (beta_end - beta_start) / max(1, max_iters)

    # patience for jitter
    patience = 0
    patience_limit = max(500, max_iters // 30)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- compute smooth max weights -----
        conv = _conv_fft(a)               # length 2n‑1
        maxc = conv.max()
        # stable soft‑max
        exp_term = np.exp(beta * (conv - maxc))
        w = exp_term / exp_term.sum()     # w_k ≥ 0, Σ w = 1

        # ----- gradient of the smooth objective -----
        grad = _weighted_conv_gradient(a, w)   # shape (n,)

        # ----- Adam update (Euclidean) -----
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)

        # bias‑corrected
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step                      # move opposite to gradient (minimise)

        # ----- projection onto the simplex -----
        a = np.clip(a, 0.0, None)
        a = _project_to_simplex(a)

        # ----- bookkeeping -----
        cur_score = _objective_norm1(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            patience = 0
            print(f"[smooth_opt] iter {t:6d}  new best O={best_score:.6f}")
        else:
            patience += 1
            if patience >= patience_limit:
                # small random kick to escape plateaus
                jitter = np.random.randn(n) * 1e-4
                a = best_a + jitter
                a = np.clip(a, 0.0, None)
                a = _project_to_simplex(a)
                patience = 0

        # increase smoothness
        beta = min(beta_end, beta + beta_inc)

    return best_a, best_score
