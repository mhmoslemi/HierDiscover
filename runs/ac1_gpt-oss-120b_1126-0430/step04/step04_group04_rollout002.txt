def _gradient_from_weights(w, a_rev, fft_len2):
    wf = np.fft.fft(w, fft_len2)
    ar_f = np.fft.fft(a_rev, fft_len2)
    conv = np.fft.ifft(wf * ar_f).real
    n = len(a_rev)
    # gradient entries are at indices n-1 .. n-1+n-1
    grad = 2.0 * conv[n-1:n-1+n]
    return grad

#!/usr/bin/env python
"""
Propose a non‑negative step function (list of heights) that minimises

        2·n·max_k (a*a)[k] / (Σ a_i)²

The algorithm works in three phases:

1. **Broad initialisation** – scan a few simple families (power‑law,
   exponential) and many random Dirichlet vectors, keeping the best
   cheap score `2·n·max(conv)`.  The best candidate is stored as `best_a`.

2. **Smooth max optimisation** – minimise a smooth approximation of the
   max‑convolution using a temperature‐controlled soft‑max:
        f_β(a) = (1/β)·log Σ_k exp(β·conv_k(a)) .
   The gradient of `f_β` can be written as a convolution, so each
   iteration only needs a few FFTs.  An exponentiated‑gradient update
   (which automatically respects non‑negativity and the simplex) is
   applied, the temperature β is slowly increased, and a learning‐rate
   η decays as 1/√t.  Stagnation triggers a small random perturbation.

3. **Final evaluation** – the best vector found during the search is
   evaluated with the provided `evaluate_sequence` and returned.

The routine respects the supplied time budget (default 1000 s) and
always returns a feasible sequence (non‑negative, sum > 0).  No file
or network access is performed.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa).real[: 2 * a.size - 1]
    return conv


def _power_law_seq(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^‑p, normalised so Σa_i = 1."""
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = i ** (-p)
    seq /= seq.sum()
    return seq


def _exponential_seq(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i), normalised."""
    i = np.arange(n, dtype=np.float64)
    seq = np.exp(-lam * i)
    seq /= seq.sum()
    return seq


def _gradient_from_weights(
    w: np.ndarray, a_rev: np.ndarray, fft_len2: int
) -> np.ndarray:
    """
    Gradient of the smooth max w.r.t. a.
    w – soft‑max weights (length 2·n‑1)
    a_rev – reversed a (length n)
    Returns vector of length n.
    """
    wf = np.fft.fft(w, fft_len2)
    arf = np.fft.fft(a_rev, fft_len2)
    full = np.fft.ifft(wf * arf).real
    n = a_rev.size
    # valid linear convolution occupies indices n‑1 … n‑1+n‑1
    grad = 2.0 * full[n - 1 : n - 1 + n]
    return grad


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
