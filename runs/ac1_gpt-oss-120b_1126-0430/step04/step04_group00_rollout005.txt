#!/usr/bin/env python
"""
Simple multi‑stage optimiser for the max‑convolution bound.

The objective (after fixing Σaᵢ = 1) is
    J(a) = 2·n·max_k (a*a)[k] .
We minimise J by

1.  Choosing a good deterministic start (power‑law, exponential or a
    previously found construction if the global variable ``height_sequence_1``
    is present).

2.  Optimising the smooth approximation of the max using an Adam optimiser
    on the log‑it (soft‑max) parametrisation of a probability vector.
    The smoothing constant is ε = eps_factor·max_conv and is driven towards
    zero (≈10⁻⁶) so that the optimiser eventually targets the true max.

3.  Random “jiggle” moves and occasional restarts to escape plateaus.

4.  A short post‑processing step that locally smooths the best vector
    (moving a little mass to neighbour entries) – a cheap way of reducing
    the peak of the convolution.

The routine respects a ~1000 s time budget, prints progress messages and
finally returns the best non‑negative sequence (as a plain Python list) it
found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level (no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max, returns a probability vector."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Inverse of softmax – returns logits that produce ``a`` after softmax."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """
    Return the smallest power‑of‑two ≥ 3·n‑2.
    This length is sufficient for a linear convolution of length 2·n‑1
    and for a correlation of length 3·n‑2 (used in the gradient).
    """
    return 1 << ((3 * n - 2).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_and_grad(a: np.ndarray, eps_factor: float) -> tuple[float, np.ndarray]:
    """
    Compute the (unsmoothed) max of a*a and the gradient of the
    smoothed objective 2·n·softmax_ε(conv)  w.r.t. the probability vector a.

    Returns
    -------
    max_conv : float
        The exact max of the convolution (used for bookkeeping).
    grad_a : np.ndarray
        Gradient of the smoothed objective (already multiplied by 4·n).
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    # FFT of a (zero‑padded)
    a_fft = np.fft.rfft(a, L)

    # ----- linear convolution a*a (length 2n‑1) -----
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # ----- smoothing weights (soft‑max over conv) -----
    eps = max(eps_factor * max_conv, 1e-12)
    shifted = conv - max_conv          # numerical shift for stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or np.isnan(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # ----- cross‑correlation a ⋆ w  (length n) -----
    # correlation = ifft( conj(a_fft) * w_fft )
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # Gradient of the smoothed objective (4·n·correlation)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """Normalised a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    a_sum = a.sum()
    if a_sum == 0.0:
        a[:] = 1.0 / n
    else:
        a /= a_sum
    return a


def _exponential_distribution(n: int, r: float) -> np.ndarray:
    """Normalised a_i ∝ r**i  (0<r<1)."""
    i = np.arange(n, dtype=np.float64)
    a = r ** i
    a_sum = a.sum()
    if a_sum == 0.0:
        a[:] = 1.0 / n
    else:
        a /= a_sum
    return a


def _random_dirichlet(n: int, alpha: float = 0.5) -> np.ndarray:
    """Draw a random point from the Dirichlet simplex with concentration α."""
    x = np.random.gamma(alpha, 1.0, size=n)
    s = x.sum()
    if s == 0.0:
        x[:] = 1.0 / n
    else:
        x /= s
    return x


def _smooth_local_step(a: np.ndarray, radius: int = 3, strength: float = 0.02) -> np.ndarray:
    """
    Perform a tiny local smoothing of `a`:
        a'[i] = (1‑strength)·a[i] + strength·average of neighbours within radius.
    The operation keeps the sum unchanged (by renormalising afterwards).
    """
    if radius <= 0 or strength <= 0.0:
        return a
    kernel = np.ones(2 * radius + 1, dtype=np.float64)
    kernel /= kernel.sum()
    smooth = np.convolve(a, kernel, mode="same")
    a_new = (1.0 - strength) * a + strength * smooth
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s == 0.0:
        a_new[:] = 1.0 / a.shape[0]
    else:
        a_new /= s
    return a_new
