"""
Propose a step‑function (non‑negative list of heights) that minimises

        O = 2·n·max(conv(a,a)) / (sum(a)²)

The algorithm has three phases:

1. **Broad parametric sweep** – search a family
        a_i ∝ (i+β)^(-α)·exp(-γ·i/n)
   on a coarse grid (α up to 2, β up to 20, γ up to 1) and keep the best.
   This quickly finds a shape with O≈1.55.

2. **Smooth max mirror‑descent** – treat the objective
   `O = 2·n·max(conv)` as the limit of a smooth soft‑max
   `M_λ = (1/λ)·log Σ_k exp(λ·conv_k)`.
   Using the soft‑max we obtain a dense gradient:
        ∂M_λ/∂a = 2·(a ⋆ w)   with   w_k = exp(λ·conv_k)/Σ exp
   where “⋆” denotes discrete correlation.
   The variables are the logits of a softmax (i.e. a = softmax(logits));
   an Adam update on the logits yields an exponentiated‑gradient
   step that respects the simplex constraint.
   λ is increased linearly during the run so that the optimisation
   gradually focuses on the true max.

3. **Tiny hill‑climbing jitter** – if the optimizer stalls we add a
   small random perturbation and continue.

All work is time‑capped (budget 1000 s).  The best sequence seen is
returned as a plain ``list[float]``.  The routine prints progress
including intermediate objective values.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector onto the probability simplex (non‑negative, sum=1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow_two(x: int) -> int:
    """Small helper – next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray, size: int = None) -> np.ndarray:
    """Linear convolution a*a using an FFT of appropriate length."""
    n = a.size
    if size is None:
        size = _next_pow_two(2 * n - 1)
    A = np.fft.fft(a, n=size)
    conv = np.fft.ifft(A * A).real[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def _correlation_fft(a: np.ndarray, w: np.ndarray, size: int = None) -> np.ndarray:
    """
    Discrete correlation (a ⋆ w)[i] = Σ_j a_j·w_{i+j}
    for i = 0 … len(a)-1.  Implemented via FFT.
    """
    n = a.size
    m = w.size
    if size is None:
        size = _next_pow_two(n + m - 1)
    A = np.fft.fft(a, n=size)
    W = np.fft.fft(w, n=size)
    corr = np.fft.ifft(A * np.conj(W)).real[:n]   # first n entries = required correlation
    return corr


# ----------------------------------------------------------------------
def _objective_norm1(a: np.ndarray) -> float:
    """Objective 2·n·max(conv) under the assumption Σ a = 1."""
    n = a.size
    conv = _conv_fft(a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
def _shape_power_exp(alpha: float, beta: float, gamma: float, n: int) -> np.ndarray:
    """Monotone decreasing shape: (i+β)^(-α)·exp(-γ·i/n), i=0…n‑1."""
    i = np.arange(n, dtype=float)
    a = (i + beta) ** (-alpha) * np.exp(-gamma * i / n)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
def _smooth_optimize(a0: np.ndarray,
                     max_iters: int,
                     deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (Adam on logits) using a smooth soft‑max approximation
    of the max‑convolution.  Returns the best vector seen and its exact
    (unsmoothed) objective.
    """
    n = a0.size
    # FFT size for correlation (needs to accommodate len(a)+len(conv)-1)
    fft_size = _next_pow_two(3 * n - 2)

    # initialise logits (log‑softmax parameters)
    logits = np.log(a0 + 1e-15)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    best_a = a0.copy()
    best_score = _objective_norm1(best_a)

    # linear schedule for λ (smoothness)
    lam_start, lam_end = 5.0, 80.0

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ---------- forward pass ----------
        a = np.exp(logits - np.max(logits))
        a /= a.sum()

        # convolution a*a
        conv = _conv_fft(a, size=fft_size)                        # length 2n‑1
        # smooth max via soft‑max
        lam = lam_start + (lam_end - lam_start) * (t / max_iters)
        scaled = lam * conv
        max_scaled = np.max(scaled)
        w = np.exp(scaled - max_scaled)                           # stable soft‑max numerator
        w /= w.sum()                                              # w_k are probabilities

        # ---------- gradient ----------
        # ∂M/∂a = 2·(a ⋆ w)
        grad_a = 2.0 * _correlation_fft(a, w, size=fft_size)

        # gradient w.r.t. logits (softmax chain rule)
        avg = np.dot(a, grad_a)
        grad_logits = a * (grad_a - avg)

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = 0.12 / (np.sqrt(v_hat) + eps)        # per‑coordinate step size
        logits -= lr * m_hat

        # Occasionally enforce numeric stability
        if t % 5000 == 0:
            logits = np.clip(logits, -30.0, 30.0)

        # ---------- bookkeeping ----------
        if t % 2000 == 0:
            cur_score = _objective_norm1(np.exp(logits - np.max(logits)))
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = np.exp(logits - np.max(logits))
                best_a /= best_a.sum()
                print(f"[smooth_opt] iter {t:6d} new best O={best_score:.6f}")

    # final sanity check: one more evaluation of the best found point
    final_score = _objective_norm1(best_a)
    return best_a, final_score

    print(f"[grid] best O={best_score:.6f} (α={best_alpha:.4f}, β={best_beta:.4f}, γ={best_gamma:.4f})")

    # ------------------------------------------------------------------

                prev_score = _objective_norm1(prev)

    except Exception as e:  # pragma: no cover
        print("[final] evaluate_sequence call failed:", e)
        final_score = None

    return best_seq.tolist()
