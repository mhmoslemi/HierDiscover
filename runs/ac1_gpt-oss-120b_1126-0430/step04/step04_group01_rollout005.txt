"""
Propose a non‑negative step‑function that (approximately) minimises

    O(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

Algorithm
---------

* **Phase 1 – Mirror‑descent on a smoothed max‑convolution.**
  We minimise the surrogate
        f_α(a) = (2·n/α)·log∑_k exp(α·conv_k(a)) + λ·∑ a_i·log a_i
  using an exponentiated‑gradient (KL‑mirror) update.
  The smoothing parameter α is increased linearly from a small value
  to a large one, so the optimisation gradually focuses on the true
  infinity‑norm.  After each step the vector is projected back onto the
  probability simplex (∑a_i=1, a_i≥0).  Gradient clipping and a
  decreasing stepsize η give stability.

* **Phase 2 – Stochastic local refinement.**
  Starting from the best point of phase 1 we perform cheap Monte‑Carlo moves:
  – mass‑transfer moves (biased by the sub‑gradient of the max‑convolution);
  – swaps of two entries;
  – shuffles of a short contiguous block.
  Moves are accepted if they improve the exact objective, otherwise
  with a Metropolis probability that cools down.  Every few thousand
  iterations a deterministic sub‑gradient step is taken.

* **Phase 3 – Final polish.**
  When the time budget is almost exhausted a few extra mirror‑descent
  steps with a very large α are performed.

The routine respects the 1000 s wall‑clock budget, prints progress, and
returns the best sequence it found as a plain list of floats.

# Helper: Euclidean projection onto the probability simplex
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full convolution a * a using FFT (length 2·n‑1)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return the full self‑convolution of a (length 2·n‑1)."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Exact objective (handles arbitrary scaling)
def _objective(a: np.ndarray) -> float:
    """Compute O(a) = 2·n·max(conv) / (sum(a)^2)."""
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(a, L=_fft_len_for_n(n))
    return 2.0 * n * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Choose a safe FFT length for a given n (needs ≥ 3n‑2 for correlation)
def _fft_len_for_n(n: int) -> int:
    """Next power‑of‑two ≥ 3·n‑2 (sufficient for both convolution and correlation)."""
    return 1 << ((3 * n - 2).bit_length())


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) (used in Phase 2)
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. a (averaged over all maximisers)."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)          # indices where the max is attained
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
# Gradient of the smooth surrogate (log‑sum‑exp) plus entropy term
def _smooth_gradient_plus_entropy(
    a: np.ndarray,
    conv: np.ndarray,
    alpha: float,
    L: int,
    lam: float,
) -> np.ndarray:
    """
    ∇_a ( (2·n/α)·log∑_k e^{α·conv_k} + λ·∑ a_i·log a_i ).
    First term = 2·corr(w, a) where w_k ∝ e^{α·conv_k}.
    """
    # soft‑max weights over the convolution entries
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)          # for numerical stability
    w = exp_scaled / exp_scaled.sum()                 # shape (2n‑1,)

    # correlation w ⋆ a (using FFT, see notes above)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n_len = a.size
    grad_corr = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]

    # entropy regulariser
    eps = 1e-20
    grad_ent = lam * (np.log(a + eps) + 1.0)

    return grad_corr + grad_ent
