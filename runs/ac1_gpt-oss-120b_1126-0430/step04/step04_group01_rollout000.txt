"""
Propose a non‑negative step‑function that (approximately) minimises

    O(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm consists of two stages:

1️⃣  **Smooth surrogate optimisation (Adam)**  
    We minimise a smooth approximation of the max‑convolution:
        f_α(a) = (2·n/α)·log Σ_k exp(α·conv_k(a)) + λ·Σ_i a_i·log a_i .
    The first term is a soft‑max (log‑sum‑exp) that converges to the true max
    as α→∞, while the entropy term (λ>0) keeps the distribution spread out.
    Adam with bias‑correction updates the parameters, and after each step we
    project onto the probability simplex (∑a_i=1, a_i≥0).  The smoothing
    parameter α is increased exponentially from a small start to a large
    final value, and λ is slowly annealed to zero.

2️⃣  **Exact sub‑gradient descent with exponentiated updates**  
    Starting from the best candidate of stage 1 we run many iterations of
    exponentiated gradient descent on the exact objective
        g(a) = max_k conv_k(a) .
    The sub‑gradient of the max is computed analytically; the EG update
        a_i ← a_i·exp(−η·g_i)
    automatically respects non‑negativity, and we renormalise to keep the
    sum equal to one.  The step‑size η decays as 1/√t.  Periodically we perform
    cheap random “mass‑transfer” moves (Metropolis acceptance) and a small
    moving‑average smoothing to escape plateaus.

The whole search respects a 1000 s wall‑clock budget.  The best sequence
found (as a plain Python list) is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full convolution a*a using FFT of length L (returns length 2·n‑1)."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


def _objective_simplex(a: np.ndarray, n: int, L: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a already on the simplex."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a (averaged over all maximisers).
    """
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


def _smooth_gradient_plus_entropy(
    a: np.ndarray,
    conv: np.ndarray,
    alpha: float,
    L: int,
    lam: float,
) -> np.ndarray:
    """
    Gradient of the smooth surrogate:
        (2·n/α)·log Σ exp(α·conv)  +  λ·Σ a_i·log a_i
    """
    # soft‑max weights w_k ∝ exp(α·conv_k)
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)  # numeric stability
    w = exp_scaled / exp_scaled.sum()  # shape (2n‑1,)

    # correlation w ⋆ a (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n_len = a.size
    grad_corr = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]

    # entropy gradient λ·(log a_i + 1)
    eps = 1e-30
    grad_ent = lam * (np.log(a + eps) + 1.0)

    return grad_corr + grad_ent

# Main search routine
# ----------------------------------------------------------------------

    Search for a low‑O step‑function.  Returns the best sequence found
    as a plain Python list of floats.

    start_time = time.time()
    deadline = start_time + budget_s - 0.2  # safety margin

    # --------------------------------------------------------------

    else:
