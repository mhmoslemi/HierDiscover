    deadline = start + budget_s - 5.0

    n = 8000  # moderate

    best_score = float('inf')
    best_iter = 0
    last_improve = 0

    iter_cnt = 0

    eps_decay = 0.9995  # per iter multiplicative decay
    eps = eps0

            best_a = a.copy()
            last_improve = iter_cnt
            print(f"[Iter {iter_cnt:07d}] new best cheap={cheap:.6f}")

        thresh = maxval * (1.0 - eps)
        mask = conv >= thresh
        k = mask.sum()

#!/usr/bin/env python
"""
Optimise a non‑negative step function (probability vector) for

        2·n·max_k (a*a)[k] / (∑ a_i)²

The programme uses a *smoothed* sub‑gradient of the max‑convolution:
all convolution entries that are close to the current maximum are
averaged (weights from a soft‑max).  The gradient is obtained by a second
FFT‑convolution, then the coefficients are updated with a mixture of an
additive projected gradient step and an exponentiated‑gradient step.
A simple annealing schedule shrinks the tolerance that defines “close
to the maximum” and occasional random restarts help escape local minima.
The routine respects the time budget (default 1000 s) and always returns
a feasible sequence (non‑negative, sum > 0.01).  The final exact score is
printed before the sequence is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Small helper – smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution of a and b using FFT (length = len(a)+len(b)-1)."""
    fa = np.fft.fft(a, fft_len)
    fb = np.fft.fft(b, fft_len)
    conv = np.fft.ifft(fa * fb).real[: len(a) + len(b) - 1]
    return conv


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full convolution a*a (length = 2·len(a)-1) using FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa).real[: 2 * a.size - 1]
    return conv


def _gradient_from_mask(a: np.ndarray,
                        mask: np.ndarray,
                        fft_len_full: int) -> np.ndarray:
    """
    Gradient of the (soft‑max) max–convolution w.r.t. a.
    ``mask`` is a binary (or weighted) mask of length 2·n‑1.
    The result has shape (n,).
    """
    w = mask.astype(np.float64)
    w_sum = w.sum()
    if w_sum == 0.0:               # should never happen, but be safe
        return np.zeros_like(a)
    w /= w_sum                     # normalise to a probability distribution
    w_rev = w[::-1]
    # convolution a * w_rev  (length = n + (2n‑1) – 1 = 3n‑2)
    conv2 = _conv_fft(a, w_rev, fft_len_full)
    n = a.shape[0]
    # slice n values centred around index 2n‑2, then reverse order
    g = 2.0 * conv2[n - 1 : 2 * n - 1][::-1]
    return g


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The routine adapts its schedule to the available time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin for final eval

    # --------------------------------------------------------------
    # Choose a working length.
    # --------------------------------------------------------------
    try:
        prev = height_sequence_1                      # noqa: F821 (global, optional)
        prev_arr = np.asarray(prev, dtype=np.float64)
    except Exception:
        prev_arr = None

    # Target length – a compromise between expressiveness and speed.
    n = 8000
    if prev_arr is not None and prev_arr.size > 0:
        # If a previous construction has a similar size, reuse it.
        if prev_arr.size == n:
            a = prev_arr / prev_arr.sum()
        else:
            # Otherwise initialise from a Dirichlet – the previous record is kept for later use.
            a = None
    else:
        a = None

    if a is None:
        # Random Dirichlet start (concentration ≈ 0.5 gives a modestly sparse start)
        conc = 0.5
        g = np.random.gamma(conc, 1.0, size=n)
        a = g / g.sum()

    # --------------------------------------------------------------
    # Pre‑compute FFT lengths
    # --------------------------------------------------------------
    fft_len_self = _next_pow2(2 * n - 1)          # for a*a
    fft_len_full = _next_pow2(3 * n - 2)          # for a * mask_rev

    # --------------------------------------------------------------
    # Helper for cheap upper‑bound (sum = 1 ⇒ score = 2·n·max(conv))
    # --------------------------------------------------------------
    def cheap_score_from_max(max_val: float) -> float:
        return 2.0 * n * max_val

    # --------------------------------------------------------------
    # Initial evaluation
    # --------------------------------------------------------------
    conv_self = _conv_self_fft(a, fft_len_self)
    max_self = conv_self.max()
    best_score = cheap_score_from_max(max_self)
    best_a = a.copy()
    best_iter = 0
    print(f"[Init] cheap score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Optimisation loop
    # --------------------------------------------------------------
    iter_cnt = 0
    last_improve = 0

    # hyper‑parameters (chosen experimentally)
    eta0 = 0.80                # initial step‑size factor
    eps0 = 0.20                # initial tolerance for “near‑max’’ mask
    eps_decay = 0.9995         # multiplicative decay of eps each iteration
    eps = eps0

    while time.time() < deadline and iter_cnt < 500_000:
        iter_cnt += 1

        # --- compute full self‑convolution and current cheap score ---
        conv_self = _conv_self_fft(a, fft_len_self)
        max_self = conv_self.max()
        cheap = cheap_score_from_max(max_self)

        # --- keep the best vector seen so far ---
        if cheap < best_score - 1e-12:
            best_score = cheap
            best_a = a.copy()
            last_improve = iter_cnt
            best_iter = iter_cnt
            print(f"[Iter {iter_cnt:07d}] NEW BEST cheap = {cheap:.6f}")

        # --- build mask of entries close to the maximum -----------------
        # tolerance shrinks over time, allowing a finer focus later.
        thresh = max_self * (1.0 - eps)
        mask = conv_self >= thresh
        # Avoid pathological situation where mask is tiny (e.g. only one entry)
        if mask.sum() < 5:
            # fall back to the 5 absolute largest entries
            top_idx = np.argpartition(conv_self, -5)[-5:]
            mask = np.zeros_like(conv_self, dtype=bool)
            mask[top_idx] = True

        # --- gradient w.r.t. the smoothed max ---
        g = _gradient_from_mask(a, mask, fft_len_full)

        # --- step size ---
        eta = eta0 / math.sqrt(iter_cnt)

        # --- stochastic mix of additive and multiplicative updates ----------
        if np.random.rand() < 0.15:
            # exponentiated‑gradient (always stays on the simplex)
            mult = np.exp(-eta * g)
            a = a * mult
            a_sum = a.sum()
            if a_sum == 0.0:                     # numeric safety
                a = np.random.rand(n)
                a_sum = a.sum()
            a = a / a_sum
        else:
            # additive projected gradient descent
            a_candidate = a - eta * g
            a = _simplex_projection(a_candidate)

        # --- anneal the mask tolerance ------------------------------------
        eps = max(1e-6, eps * eps_decay)

        # --- restart if we have been stagnant for a while -----------------
        if iter_cnt - last_improve > 120_000:
            # small random perturbation of the current best vector
            perturb = 0.02 * np.random.rand(n)
            a = _simplex_projection(best_a + perturb)
            last_improve = iter_cnt
            eps = eps0
            print(f"[Iter {iter_cnt:07d}] Restart after stagnation (perturb).")

    # --------------------------------------------------------------
    #  Final exact evaluation (uses the provided ``evaluate_sequence``)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:      # defensive – should never happen
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # --------------------------------------------------------------
    #  Return the best sequence found.
    # --------------------------------------------------------------
    return final_seq
