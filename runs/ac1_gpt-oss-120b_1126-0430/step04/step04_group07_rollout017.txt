"""
Propose a candidate step‑function that minimizes the bound

    O = 2·n·max(conv(a,a)) / (sum(a)**2)

The optimiser works on the log‑probabilities `u = log(a)` and uses an
Adam‑style mirror‑descent on the probability simplex.  The weight vector
over the convolution entries starts as a smooth soft‑max (large temperature)
and gradually concentrates on the few active constraints (hard‑max).  A
learning‑rate schedule, periodic jitter and cheap restarts from the best‑so‑far
solution keep the search from stagnating.  After the main run a short
local swap‑based hill‑climbing phase tries to improve the ordering of the
coefficients (the values are fixed, only their positions are permuted).

The routine respects a 1000 s time budget, prints progress, evaluates the
final solution with the provided ``evaluate_sequence`` and returns the best
list of coefficients found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – pure top‑level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv


def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Linear correlation g_i = Σ_t w[i+t]·a[t] for i = 0,…,n‑1.
    Implemented via FFT (zero‑padding to power‑of‑two).
    """
    n = a.shape[0]
    m = w.shape[0]                # = 2·n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    bound.  Returns the best sequence as a list of floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5      # safety margin

    # ------------------------------------------------------------------
    # 1) initialise length and log‑probabilities
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=float)
        a0 = a0 / a0.sum()
        n = a0.size
        u = np.log(a0 + 1e-12)                    # tiny eps to avoid log(0)
    else:
        # a moderately large length; empirical work shows n≈6000‑8000 works well
        n = 6000
        u = np.full(n, -np.log(n), dtype=float)   # uniform log‑probability

    best_u = u.copy()
    best_score = np.inf
    best_iter = 0

    # ------------------------------------------------------------------
    # 2) Adam state
    # ------------------------------------------------------------------
    m = np.zeros_like(u)           # first moment
    v = np.zeros_like(u)           # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    # ------------------------------------------------------------------
    # 3) hyper‑parameter schedules
    # ------------------------------------------------------------------
    max_iters_est = 1_200_000          # generous upper bound; loop stops earlier
    beta_start = 2.0
    beta_end = 50_000.0
    eta_start = 0.5
    eta_end = 0.005
    softmax_phase = 0.45               # fraction of run where we use a soft‑max weight
    jitter_interval = 4000
    jitter_scale = 2e-3
    restart_interval = 100_000

    it = 0
    while time.time() < deadline and it < max_iters_est:
        it += 1

        # --------------------------------------------------------------
        # (a) build probability vector a and its self‑convolution
        # --------------------------------------------------------------
        u_shift = u - u.max()                      # avoid overflow
        a = np.exp(u_shift)
        a = a / a.sum()                            # Σa = 1  (scale‑invariant)

        conv = _conv_self_fft(a)                   # length 2·n‑1
        M = conv.max()
        cur_score = 2.0 * n * M                     # O = 2·n·max(conv) (Σa=1)

        # --------------------------------------------------------------
        # (b) record best solution
        # --------------------------------------------------------------
        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_u = u.copy()
            best_iter = it
            print(f"[iter {it:07d}] new best O={best_score:.9f}")

        # --------------------------------------------------------------
        # (c) schedule parameters
        # --------------------------------------------------------------
        fraction = min(it / max_iters_est, 1.0)
        beta = beta_start + (beta_end - beta_start) * fraction
        eta = eta_start * (eta_end / eta_start) ** fraction   # exponential decay

        if it % restart_interval == 0:

            m.fill(0.0)
            v.fill(0.0)
            print(f"[restart] iter {it}, re‑initialised from best O={best_score:.9f}")

    # ------------------------------------------------------------------
    # 4) post‑process the best log‑vector
    # ------------------------------------------------------------------
    best_seq = np.exp(best_u - best_u.max())
    best_seq = best_seq / best_seq.sum()   # ensure exact Σ=1

    # ------------------------------------------------------------------
    # 5) short local permutation hill‑climb (uses remaining budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 12.0:
        a = best_seq.copy()
        cur_val = best_score
        # try a few hundred random swaps
        while time.time() < deadline - 5.0:
            i, j = np.random.randint(0, n, size=2)
            if i == j:

            a_new[i], a_new[j] = a_new[j], a_new[i]           # swap positions

            M_new = conv_new.max()
            val_new = 2.0 * n * M_new

                a = a_new
                cur_val = val_new
                best_seq = a_new
                best_score = val_new
                print(f"[swap] improvement to O={best_score:.9f}")

    # 6) final evaluation with the provided evaluator
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.9f} "
              f"(best iteration {best_iter})")
    except Exception as exc:      # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_seq.tolist()
