def max_convolution_subgradient(a):

    conv = np.convolve(a, a)  # full

    maxb = conv.max()

    # epsilon relative tolerance

    tolerance = maxb * 1e-12 + 1e-15

    active = np.where(conv >= maxb - tolerance)[0]

    # weights uniform

    w = 1.0 / len(active)

    # compute gradient

    n = a.shape[0]

    g = np.zeros_like(a)

    for off in active:

        lo = max(0, off - (n - 1))

        hi = min(off, n - 1)

        i_range = np.arange(lo, hi + 1)

        g[i_range] += w * 2.0 * a[off - i_range]

    return g, maxb

#!/usr/bin/env python
"""
Improved search for a low‑value step‑function sequence.

The algorithm combines three ideas that were not all used together in the
reference implementation:

1. **Exact sub‑gradient of the max‑convolution** – instead of a smoothed
   approximation we identify the active offset(s) where the convolution attains
   its maximum and build the true sub‑gradient (averaged when several offsets
   are active).

2. **Projected (mirror) Adam on the simplex** – the probability vector is
   represented by logits `z`.  Adam updates are performed in log‑it space,
   which corresponds to mirror descent w.r.t. the KL‑divergence.

3. **Multi‑phase schedule with restarts and support expansion** – we start
   with a modest length, optimise, then interpolate to a larger support.
   If progress stalls we randomise the logits (a cheap restart) and continue.
   All steps respect the supplied time budget (≈1000 s).

The routine returns a plain list of non‑negative floats that (hopefully) beats
the previously reported bound of 1.519133 when evaluated with the provided
`evaluate_sequence` function.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level – no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max; returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it representation that yields `a` after soft‑max."""
    return np.log(a + eps)


def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two ≥ 2·n‑1 (efficient size for real FFT)."""
    return 1 << ((2 * n - 1).bit_length())


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full linear self‑convolution a*a using a real FFT of length `fft_len`.
    Returns an array of length 2·len(a)‑1.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.size - 1]
    return conv


def _max_convolution_subgradient(a: np.ndarray,
                                 conv: np.ndarray,
                                 tol: float = 1e-14) -> (np.ndarray, float):
    """
    Exact sub‑gradient of `max(conv)` at the current probability vector `a`.

    Parameters
    ----------
    a : np.ndarray
        Current probability vector (sum = 1).
    conv : np.ndarray
        Full self‑convolution a*a (length 2·n‑1).
    tol : float
        Tolerance for determining the active set (relative).

    Returns
    -------
    g : np.ndarray
        Sub‑gradient w.r.t. `a` (same shape as `a`).
    maxb : float
        The maximal convolution value.
    """
    n = a.shape[0]
    maxb = float(conv.max())

    # active set – offsets where conv is within tolerance of the maximum
    # use an absolute tolerance scaled to the magnitude of maxb
    eps = max(tol, tol * maxb)
    active = np.where(conv >= maxb - eps)[0]

    # average the gradients of all active offsets (any convex combination is valid)
    weight = 1.0 / len(active)
    g = np.zeros_like(a)

    for off in active:
        lo = max(0, off - (n - 1))
        hi = min(off, n - 1)
        i = np.arange(lo, hi + 1, dtype=np.int64)
        # contribution of offset `off`: ∂/∂a_i (a*a)[off] = 2·a_{off‑i}
        g[i] += weight * 2.0 * a[off - i]
    return g, maxb


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)        # guard against tiny negatives
    total = new_a.sum()
    if total == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= total
    return new_a


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """Normalised power‑law a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) beats the target
    bound.  The routine respects the allocated time budget (≈1000 s) and
    uses a projected‑Adam optimiser on the probability simplex combined with
    occasional restarts and a progressive increase of the support size.
    """
    np.random.seed(seed)
    start = time.time()
    # keep a small safety margin for the final evaluation
    deadline = start + budget_s - 2.0

    # ---------------------------------------------------------------
    # 0. Working support size
    # ---------------------------------------------------------------
    # If a previous construction is supplied as a global variable we reuse it.
    init_global = globals().get("height_sequence_1", None)
    if init_global is not None:
        init_global = np.array(init_global, dtype=np.float64)
        n = max(2000, init_global.shape[0])
        a = np.maximum(init_global, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / a.shape[0])
        else:
            a /= s
        print(f"[Init] Using supplied global sequence, length = {n}")
    else:
        # start from a modest power‑law – works well as a generic seed.
        n = 12000                     # initial size, will be enlarged later
        a = _power_law_distribution(n, 0.60)
        print(f"[Init] Power‑law start, length = {n}")

    # ---------------------------------------------------------------
    # 1. Log‑it parametrisation + Adam state
    # ---------------------------------------------------------------
    z = _logits_from_distribution(a)
    m = np.zeros_like(z)                # first moment
    v = np.zeros_like(z)                # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    base_alpha = 0.25

    # ---------------------------------------------------------------
    # 2. Book‑keeping
    # ---------------------------------------------------------------
    best_seq = a.copy()
    best_score = float("inf")
    it = 0
    last_improve = 0
    RESTART_WAIT = 150_000           # iterations without improvement → restart
    REPORT_EVERY = 5_000
    MAX_SUPPORT = 30000

    # ---------------------------------------------------------------
    # 3. Main optimisation loop
    # ---------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # probability vector from logits (guarantees non‑negativity & sum=1)
        a = _softmax(z)

        # full convolution using FFT (fast for large n)
        fft_len = _fft_len_for_n(n)
        conv = _conv_full_fft(a, fft_len)

        # cheap proxy (the exact objective when sum(a)=1)
        maxb = float(conv.max())
        cheap_score = 2.0 * n * maxb

        # record best (using cheap proxy – equals the true objective here)
        if cheap_score < best_score:
            best_score = cheap_score
            best_seq = a.copy()
            last_improve = it
            print(f"[Iter {it:07d}] NEW BEST cheap = {cheap_score:.6f}")

        # -----------------------------------------------------------
        # Exact sub‑gradient of the max‑convolution
        # -----------------------------------------------------------
        g_a, _ = _max_convolution_subgradient(a, conv, tol=1e-14)

        # Mirror‑descent conversion to log‑it space (KL geometry)
        mu = float(np.dot(g_a, a))
        grad_z = a * (g_a - mu)

        # Adam moments (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # direction (negative Adam step)
        step_dir = - m_hat / (np.sqrt(v_hat) + eps_adam)

        # simple inverse‑sqrt decay for the learning‑rate
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)

        # update logits
        z = z + alpha_t * step_dir
        np.clip(z, -80.0, 80.0, out=z)   # keep logits numerically safe

        # -----------------------------------------------------------
        # Occasional random perturbations – helps escape plateaus
        # -----------------------------------------------------------
        if it % 7_000 == 0 and np.random.rand() < 0.30:
            perturb = 0.02 * np.random.randn(n)
            z += perturb
            m *= 0.5
            v *= 0.5

        # -----------------------------------------------------------
        # Early‑stop / restart logic
        # -----------------------------------------------------------
        if it - last_improve > RESTART_WAIT:
            # restart with a fresh random Dirichlet‑like start
            print(f"[Iter {it:07d}] Stagnation – restarting.")
            a = np.random.dirichlet(alpha=np.full(n, 0.8))
            z = _logits_from_distribution(a)
            m.fill(0.0)
            v.fill(0.0)
            it = 0
            last_improve = 0
            continue

        # -----------------------------------------------------------
        # Progressive enlargement of the support (if time permits)
        # -----------------------------------------------------------
        if it % 30_000 == 0 and n < MAX_SUPPORT and (time.time() + 25.0 < deadline):
            new_n = min(MAX_SUPPORT, n + 6000)
            print(f"[Iter {it:07d}] Expanding support: {n} → {new_n}")

            # interpolate the best distribution found so far
            a = _interpolate_distribution(best_seq, new_n)

            # re‑initialise Adam state for the larger problem
            z = _logits_from_distribution(a)
            m = np.zeros_like(z)
            v = np.zeros_like(z)
            n = new_n
            it = 0
            last_improve = 0

            # compute a fresh cheap score for the new size
            fft_len = _fft_len_for_n(n)
            conv = _conv_full_fft(a, fft_len)
            best_score = 2.0 * n * conv.max()
            best_seq = a.copy()
            continue

        # -----------------------------------------------------------
        # Reporting progress
        # -----------------------------------------------------------
        if it % REPORT_EVERY == 0:
            elapsed = time.time() - start
            print(f"[Iter {it:07d}] elapsed={elapsed:.1f}s "
                  f"cheap={cheap_score:.6f} best={best_score:.6f}")

    # ---------------------------------------------------------------
    # 4. Final exact evaluation (the platform will also evaluate)
    # ---------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[Result] Final exact score = {final_score:.6f}")
    except Exception as exc:
        final_score = float("inf")
        print(f"[Result] Evaluation error: {exc}")

    # ---------------------------------------------------------------
    # 5. Return the best sequence found.
    # ---------------------------------------------------------------
    return final_seq
