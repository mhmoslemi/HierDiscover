Optimising non‑negative step‑function for minimal O = 2·n·max(conv(a,a)).

Strategy:
1. Multi‑scale optimisation.  Start with a modest support size,
   optimise it by Adam in the log‑it (softmax) parametrisation
   using a smooth soft‑max approximation of the max (exponential
   temperature β that is annealed up to 1e5).  After convergence the
   solution is up‑sampled to a larger size and refined again.
2. A final simulated‑annealing style local search moves a small
   fraction of mass between random indices and accepts improvements
   (or worsening moves with a decreasing temperature) to escape
   plateaus.
3. The best sequence seen during the whole run is returned.
"""
import time
import numpy as np

# ----- FFT utilities ----------------------------------------------------
def _next_pow2(x: int) -> int:
    return 1 << (x - 1).bit_length()

def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv

def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    m = w.shape[0]                 # = 2n-1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]

def _softmax(x: np.ndarray) -> np.ndarray:
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()

# ----- objective ---------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    n = a.size
    conv = _conv_self_fft(a)
    return float(2.0 * n * conv.max())

# ----- optimisation core -------------------------------------------------
def _adam_optimize(
        n: int,
        time_limit: float,
        init_a: np.ndarray | None = None,
        rng: np.random.Generator | None = None,
        seed: int = 0,
) -> tuple[np.ndarray, float]:
    """Adam optimisation on the simplex for a given support size.

    Returns the best vector (sum to 1) and its exact objective value.
    """
    start = time.time()
    if rng is None:
        rng = np.random.default_rng(seed)

    # initialise a on the simplex
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
        a = a / a.sum()
    else:
        a = rng.random(n)
        a = a / a.sum()

    # work in log‑it space
    eps = 1e-16
    x = np.log(a + eps)

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    adam_eps = 1e-12

    # schedules
    beta_start = 0.5
    beta_end = 1e5          # sharper soft‑max at the end
    lr_start = 0.2
    lr_end = 0.001
    max_iters = 300_000

    best_a = a.copy()
    best_obj = _objective_exact(best_a)

    it = 0
    while time.time() - start < time_limit and it < max_iters:
        it += 1

        # probabilities from logits
        a = _softmax(x)                     # sum to 1

        # exact objective
        conv = _conv_self_fft(a)
        M = conv.max()
        obj = 2.0 * n * M
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[n={n:5d} it={it:06d}] new best O={best_obj:.6f}")

        # schedule parameters
        t = min(it / max_iters, 1.0)
        beta = beta_start * np.exp(np.log(beta_end / beta_start) * t)
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t)

        # soft‑max weights for smoothed max
        if beta > 2e4:          # use exact arg‑max when temperature is huge
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            # shift for numerical stability (max already subtracted)
            shifted = conv - M
            w = np.exp(beta * shifted)
            w_sum = w.sum()
            # avoid division by zero (should not happen)
            if w_sum == 0.0:
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w /= w_sum

        # gradient w.r.t. a (gradient of weighted sum of conv entries)
        g_a = 2.0 * _correlation_fft(w, a)    # shape (n,)

        # natural gradient in log‑it space (mirrored gradient for softmax)
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + adam_eps)

        # jitter every few thousand steps to escape flat regions
        if it % 5000 == 0:
            jitter = rng.normal(scale=0.02, size=n)
            x += jitter

        # restart from best solution if no improvement for a while
        if it % 60000 == 0:
            x = np.log(best_a + eps)
            x += rng.normal(scale=0.005, size=n)
            m.fill(0.0)
            v.fill(0.0)
            print(f"[n={n}] restart at iteration {it}")

    thr = 1e-12
    mask = best_a > thr
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        n_new = best_a.size
        best_obj = _objective_exact(best_a)
        print(f"[compress] length {n} -> {n_new}, O={best_obj:.6f}")

    if remaining > 0.6:
        a_loc = best_a.copy()
        obj_loc = best_obj

        T_end = 1e-6

        while time.time() - local_start < remaining * 0.4:
            # linear temperature decay based on elapsed fraction
            frac = (time.time() - local_start) / (remaining * 0.4)
            T = T_start * (1 - frac) + T_end * frac

            i, j = rng.choice(n_loc, size=2, replace=False)
            # propose transferring a small random fraction of mass
            delta = a_loc[i] * rng.random() * 0.05   # up to 5 % of source
            if delta <= 0.0:
                continue
            a_new = a_loc.copy()
            a_new[i] -= delta
            a_new[j] += delta
            # re‑normalise (total mass unchanged)
            conv_new = _conv_self_fft(a_new)
            obj_new = 2.0 * n_loc * conv_new.max()
            if obj_new < obj_loc or rng.random() < np.exp(-(obj_new - obj_loc) / max(T, 1e-12)):
                a_loc, obj_loc = a_new, obj_new
                if obj_new < best_obj:
                    best_a, best_obj = a_new, obj_new
        print(f"[local] final O={best_obj:.6f}")

    return best_a, best_obj

# ----- up‑sampling helper ------------------------------------------------
def _upsample_to(a_small: np.ndarray, n_target: int) -> np.ndarray:
    """Linearly interpolate a_small (sum=1) to length n_target and renormalise."""
    x_old = np.linspace(0.0, 1.0, num=a_small.size, endpoint=False)
    x_new = np.linspace(0.0, 1.0, num=n_target, endpoint=False)
    a_new = np.interp(x_new, x_old, a_small, left=0.0, right=0.0)
    s = a_new.sum()
    if s <= 0.0:
        a_new = np.full(n_target, 1.0 / n_target)
    else:
        a_new /= s
    return a_new

        obj = 2.0 * n * M
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[n={n:5d} it={it:06d}] new best O={best_obj:.6f}")

        # schedule parameters (exponential annealing)
        t = min(it / max_iters, 1.0)
        beta = beta_start * np.exp(np.log(beta_end / beta_start) * t)
        lr = lr_start * np.exp(np.log(lr_end / lr_start) * t)

        # smooth‑max weights
        if beta > 2e4:                     # hard arg‑max when very sharp
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            shifted = conv - M                # ≤ 0
            w = np.exp(beta * shifted)
            w_sum = w.sum()
            if w_sum == 0.0:                  # numerical safety
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w /= w_sum

        # gradient w.r.t. a (gradient of Σ_k w_k conv_k)
        g_a = 2.0 * _correlation_fft(w, a)

        # natural gradient in log‑it space (mirror descent)
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + adam_eps)

        # occasional jitter to escape flat regions
        if it % 5000 == 0:
            x += rng.normal(scale=0.02, size=n)

        # restart from current best every so often
        if it % 60000 == 0:
            x = np.log(best_a + eps)
            x += rng.normal(scale=0.005, size=n)
            m.fill(0.0)
            v.fill(0.0)
            print(f"[n={n}] restart at iteration {it}")

    # ---- compress negligible entries ------------------------------------
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        n_new = best_a.size
        best_obj = _objective_exact(best_a)
        print(f"[compress] length {n} → {n_new}, O={best_obj:.6f}")

    # ---- final local search (simulated annealing) --------------------
    remaining = time_limit - (time.time() - start)
    if remaining > 0.6:
        a_loc = best_a.copy()
        obj_loc = best_obj
        n_loc = a_loc.size
        T_start, T_end = 1e-3, 1e-6
        ls_start = time.time()
        while time.time() - ls_start < remaining * 0.4:
            frac = (time.time() - ls_start) / (remaining * 0.4)
            T = T_start * (1 - frac) + T_end * frac

            i, j = rng.choice(n_loc, size=2, replace=False)
            delta = a_loc[i] * rng.random() * 0.05   # up to 5 % of a_i
            if delta <= 0.0:
                continue
            a_new = a_loc.copy()
            a_new[i] -= delta
            a_new[j] += delta

            conv_new = _conv_self_fft(a_new)
            obj_new = 2.0 * n_loc * conv_new.max()
            if obj_new < obj_loc or rng.random() < np.exp(-(obj_new - obj_loc) / max(T, 1e-12)):
                a_loc, obj_loc = a_new, obj_new
                if obj_new < best_obj:
                    best_a, best_obj = a_new, obj_new
        print(f"[local] final O={best_obj:.6f}")

    return best_a, best_obj


# ----------------------------------------------------------------------
def _upsample_to(a_small: np.ndarray, n_target: int) -> np.ndarray:
    """Linearly interpolate a_small (sum = 1) to length n_target and renormalise."""
    x_old = np.linspace(0.0, 1.0, num=a_small.size, endpoint=False)
    x_new = np.linspace(0.0, 1.0, num=n_target, endpoint=False)
    a_new = np.interp(x_new, x_old, a_small, left=0.0, right=0.0)
    s = a_new.sum()
    if s <= 0.0:
        a_new = np.full(n_target, 1.0 / n_target)
    else:
        a_new /= s
    return a_new

def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **_ignore,
) -> list[float]:
    """
    Main entry point used by the evaluation harness.

    Runs a multi‑scale optimisation, finishes with a short simulated‑annealing
    local search and returns the best sequence found (as a plain Python list).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # support sizes – grow progressively; the last one gets the bulk of the budget
    lengths = [256, 512, 1024, 2048, 4096, 8192]

    # warm‑start from a previously supplied global variable if present
    init_seq = None
    if 'height_sequence_1' in globals():
        seq0 = np.asarray(height_sequence_1, dtype=float)
        seq0 = np.clip(seq0, 0.0, None)
        if seq0.sum() < 0.01:
            seq0 = np.ones_like(seq0)
        seq0 = seq0 / seq0.sum()
        init_seq = seq0
        if seq0.size not in lengths:
            lengths = [seq0.size] + lengths

    best_seq = None
    best_obj = float('inf')

    for idx, n in enumerate(lengths):
        now = time.time()
        remaining = deadline - now
        if remaining < 2.0:
            break

        # allocate time: early stages get a small fixed budget, final stage gets the rest
        if idx < len(lengths) - 1:
            run_time = min(30.0, remaining * 0.08)
        else:
            run_time = remaining - 0.5   # keep a tiny safety margin

        # initialise: up‑sample the best solution obtained so far (if any)
        init_a = None
        if best_seq is not None:
            init_a = _upsample_to(np.asarray(best_seq, dtype=float), n)
        if idx == 0 and init_seq is not None:
            init_a = _upsample_to(init_seq, n)

        seq, val = _adam_optimize(
            n=n,
            time_limit=run_time,
            init_a=init_a,
            rng=rng,
            seed=seed + idx * 17,
        )
        if val < best_obj:
            best_obj = val
            best_seq = seq
            print(f"[global] new overall best O={best_obj:.6f} (n={n})")

    # final sanity check via the provided evaluate_sequence (mainly for debugging)
    try:
        final_score = evaluate_sequence(best_seq)
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_seq.tolist()
