"""
Propose a non‑negative step‑function (list of heights) that minimises

    O = 2·n·max(conv(a,a)) / (sum(a)²)

The method consists of four stages:

1. **Rich parametric sweep** – generate candidates from three families
   (pure power‑law, power‑law with exponential tail, and discretised
   Beta‑distributions).  The best candidate (lowest O) is kept.

2. **Mirror‑descent polishing** – starting from that candidate we run a
   projected exponentiated‑gradient descent (KL‑mirror descent) with a
   decaying learning rate and occasional jitter.  The loop is time‑capped
   and the best point seen is retained.

3. **Permutation hill‑climbing** – the values of the best vector are
   frozen but their order is allowed to change.  Swapping two entries
   and re‑evaluating O is cheap via an FFT‑based convolution; we accept
   only improving swaps (simulated‑annealing style acceptance is added).
   This often flattens the self‑convolution further.

4. **Final exact evaluation** – the best vector is handed to the
   provided ``evaluate_sequence`` function and returned as a plain list.

All helper functions are top‑level, no closures or lambdas are used.
The routine respects the 1000 s budget and prints progress
information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of a vector onto the probability simplex. """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # numerical safety
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_full_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """ Linear convolution of two real vectors using a power‑of‑two FFT. """
    size = 1 << ((x.size + y.size - 2).bit_length())
    X = np.fft.fft(x, n=size)
    Y = np.fft.fft(y, n=size)
    conv = np.fft.ifft(X * Y).real[: x.size + y.size - 1]
    return conv


# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """ Shortcut for a*a (full linear convolution). """
    return _conv_full_fft(a, a)


# ----------------------------------------------------------------------
def _objective_norm1(a: np.ndarray) -> float:
    """Objective 2·n·max(conv) under the convention Σa = 1."""
    n = a.size
    conv = _conv_fft(a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv(a,a)) w.r.t. a (valid on the simplex)."""
    n = a.size
    conv = _conv_fft(a)
    max_val = conv.max()
    ks = np.where(np.abs(conv - max_val) < 1e-14)[0]
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
def _exp_update(a: np.ndarray, g: np.ndarray, eta: float) -> np.ndarray:
    """Exponentiated‑gradient (mirror‑descent) update on the simplex."""
    log_a = np.log(a + 1e-15)
    log_a -= eta * g
    log_a -= np.max(log_a)               # prevent overflow
    a_new = np.exp(log_a)
    a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
def _optimize(a0: np.ndarray,
              max_iters: int,
              eta0: float,
              deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected exponentiated‑gradient descent with jitter.
    Returns the best vector seen and its objective value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective_norm1(a)

    n = a.size
    no_improve = 0
    patience = max(10, max_iters // 30)   # early‑stop patience

    for t in range(max_iters):
        if time.time() >= deadline:
            break

        g = _subgradient_max_conv(a)
        eta = eta0 * (1.0 - t / max_iters)   # linear decay

        a = _exp_update(a, g, eta)

        cur_score = _objective_norm1(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            print(f"[opt] iter {t:6d}  new best O={best_score:.6f}")
        else:
            no_improve += 1
            if no_improve >= patience:
                # small random jitter to escape flat regions
                jitter = np.random.randn(n) * 1e-4
                a = best_a + jitter
                a = np.clip(a, 0.0, None)
                a = _project_to_simplex(a)
                no_improve = 0

    return best_a, best_score


# ----------------------------------------------------------------------
def _shape_power(alpha: float, beta: float, n: int) -> np.ndarray:
    """Monotone decreasing a_i ∝ (i+β)^(-α), i = 1..n, normalised to sum‑1."""
    i = np.arange(1, n + 1, dtype=float)
    a = (i + beta) ** (-alpha)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


# ----------------------------------------------------------------------
def _shape_power_exp(alpha: float, beta: float, gamma: float, n: int) -> np.ndarray:
    """Power‑law with an exponential tail."""
    i = np.arange(1, n + 1, dtype=float)
    a = (i + beta) ** (-alpha) * np.exp(-gamma * (i - 1) / n)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


# ----------------------------------------------------------------------
def _shape_beta(p: float, q: float, n: int) -> np.ndarray:
    """Beta‑distribution shape on {0,…,n‑1}: a_i ∝ (i+1)^{p‑1}·(n‑i)^{q‑1}."""
    i = np.arange(1, n + 1, dtype=float)          # i = 1..n
    a = (i) ** (p - 1) * (n - i + 1) ** (q - 1)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


# ----------------------------------------------------------------------
def _random_dirichlet(n: int, conc: float = 0.5) -> np.ndarray:
    """Draw a random point on the simplex using a symmetric Dirichlet."""
    a = np.random.gamma(conc, 1.0, size=n)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


# ----------------------------------------------------------------------
def _permutation_hill_climb(a_start: np.ndarray,
                            max_swaps: int,
                            deadline: float) -> tuple[np.ndarray, float]:
    """
    Try to lower the objective by swapping positions of the entries.
    Simple simulated‑annealing style acceptance is used.
    """
    a = a_start.copy()
    n = a.size
    best_a = a.copy()
    best_score = _objective_norm1(a)

    # annealing schedule (temperature decays to ~0)
    T0 = best_score * 0.02
    for t in range(max_swaps):
        if time.time() >= deadline:
            break

        # choose a pair (prefer swapping a large with a small entry)
        i = np.random.choice(n, p=a)                # heavy index
        j = np.random.randint(0, n)                # uniform index
        if i == j:
            continue

        a_candidate = a.copy()
        a_candidate[i], a_candidate[j] = a_candidate[j], a_candidate[i]

        cur_score = _objective_norm1(a_candidate)
        delta = cur_score - best_score
        T = T0 * (0.9995 ** t)                     # exponential cooling

        if delta < 0 or np.random.rand() < np.exp(-delta / max(T, 1e-12)):
            # accept the swap
            a = a_candidate
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                print(f"[perm] swap {t:6d}  new best O={best_score:.6f}")

    return best_a, best_score

    # --------------------------------------------------------------

    best_seq = None
