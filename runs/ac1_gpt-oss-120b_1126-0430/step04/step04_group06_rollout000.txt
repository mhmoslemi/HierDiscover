#!/usr/bin/env python
"""
Improved optimisation of a non‑negative step function.

* The vector is represented by logits `z`; `a = softmax(z)` guarantees
  a_i ≥ 0 and Σ a_i = 1 (the scale is irrelevant for the evaluation).
* At each iteration we compute the full self‑convolution `c = a*a` with a
  real FFT (O(n log n)).
* Instead of using a limited set of top offsets we form the *exact*
  smoothed max of the convolution via a full soft‑max:
        w_k = exp(θ·c_k) / Σ_j exp(θ·c_j) .
  The gradient of  Σ_k w_k c_k with respect to `a` can be written as

        ∇_a Σ w_k c_k = (w * a)[:n] + a ,

  i.e. a single convolution of the weight vector `w` with `a`.  This
  gives an unbiased, smooth sub‑gradient of the max‑convolution and
  avoids the heuristic “K‑largest‑offset” approach.
* Adam (with a √t decay) performs mirror‑descent on the logits.
* The optimiser works on a sequence of increasing support sizes:
  5 000 → 10 000 → 20 000 → 30 000 (or the maximum possible given the
  remaining time).  At each enlargement the current best distribution
  is interpolated to the larger grid, logits are rebuilt and optimisation
  continues.
* Occasionally a small random perturbation is added to the logits (helps
  escape plateaus) and if no improvement is seen for a long period the
  Adam state is reset around the current best.
* The best probability vector found is returned as a plain Python list.
  `evaluate_sequence` rescales it automatically, so we keep the sum = 1.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max; returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    s = e.sum()
    if s == 0.0:
        # fallback – uniform distribution
        return np.full_like(z, 1.0 / z.size)
    return e / s


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it representation that yields `a` after soft‑max."""
    return np.log(a + eps)


def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two ≥ n (efficient size for real FFT)."""
    return 1 << ((n - 1).bit_length())


def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution `x*y` using a real FFT.
    Returns an array of length len(x)+len(y)-1.
    """
    n = x.shape[0] + y.shape[0] - 1
    fft_len = _fft_len_for_n(n)
    fx = np.fft.rfft(x, fft_len)
    fy = np.fft.rfft(y, fft_len)
    conv = np.fft.irfft(fx * fy, fft_len)[:n]
    return conv


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)        # guard against tiny negatives
    total = new_a.sum()
    if total == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= total
    return new_a


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """Normalised power‑law a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Optimise a step‑function by minimising the `evaluate_sequence`
    bound.  The routine respects the given time budget (~1000 s) and
    returns a list of non‑negative heights (sum ≈ 1).  The optimiser
    works on a sequence of growing support sizes and uses a fully
    smoothed max‑convolution gradient together with Adam on the
    soft‑max logits.
    """
    np.random.seed(seed)
    start_time = time.time()
    # Keep a small safety margin for the final evaluation
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # 0. Determine the sequence of support sizes (progressive enlargement)
    # ------------------------------------------------------------------
    max_n = 30000                     # hard cap
    # Choose a geometric schedule that respects the time budget
    stage_sizes = [5000, 10000, 20000, 30000]
    stage_sizes = [s for s in stage_sizes if s <= max_n]

    # ------------------------------------------------------------------
    # 1. Global best (across all stages)
    # ------------------------------------------------------------------
    best_overall_seq = None
    best_overall_score = float("inf")

    # ------------------------------------------------------------------
    # 2. Optimisation loop over stages
    # ------------------------------------------------------------------
    for stage_idx, n in enumerate(stage_sizes, 1):
        if time.time() >= deadline:
            break

        print(f"[Stage {stage_idx}/{len(stage_sizes)}] n = {n}")

        # --------------------------------------------------------------
        # 2.1 initialise a probability vector of length n
        # --------------------------------------------------------------
        if best_overall_seq is not None:
            # Interpolate the current best distribution to the larger grid
            a = _interpolate_distribution(best_overall_seq, n)
        else:
            # Try to reuse a previously supplied global sequence, otherwise power‑law
            init_global = globals().get("height_sequence_1", None)
            if init_global is not None:
                a0 = np.array(init_global, dtype=np.float64)
                if a0.size != n:
                    a = _interpolate_distribution(a0, n)
                else:
                    a = np.maximum(a0, 0.0)
            else:
                a = _power_law_distribution(n, 0.60)

        # Normalise to a probability vector (sum = 1)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / a_sum

        # --------------------------------------------------------------
        # 2.2 Log‑it parametrisation + Adam state
        # --------------------------------------------------------------
        z = _logits_from_distribution(a)
        m = np.zeros_like(z)                # first moment
        v = np.zeros_like(z)                # second moment
        beta1, beta2 = 0.9, 0.999
        eps_adam = 1e-8
        base_alpha = 0.30                    # base step size

        # --------------------------------------------------------------
        # 2.3 Stage‑specific optimisation parameters
        # --------------------------------------------------------------
        # Temperature schedule for the soft‑max over the convolution
        theta_start = 0.05
        theta_end = 15.0
        theta_slope = (theta_end - theta_start) / max(1.0, budget_s - 5.0)

        # Stopping criteria inside a stage
        best_seq_stage = a.copy()
        best_score_stage = float("inf")
        it = 0
        last_improve = 0
        report_every = 2000

        while time.time() < deadline and it < 1_200_000:
            it += 1

            # ----------------------------------------------------------
            # 2.3.1 Current probability vector
            # ----------------------------------------------------------
            a = _softmax(z)

            # ----------------------------------------------------------
            # 2.3.2 Self‑convolution a*a (full length = 2n‑1)
            # ----------------------------------------------------------
            conv_self = _full_convolution_fft(a, a)
            max_conv = float(conv_self.max())
            cheap_score = 2.0 * n * max_conv  # == exact score because sum(a)=1

            # ----------------------------------------------------------
            # 2.3.3 Record best candidate in this stage
            # ----------------------------------------------------------
            if cheap_score < best_score_stage:
                best_score_stage = cheap_score
                best_seq_stage = a.copy()
                last_improve = it
                if it % report_every == 0:
                    print(f"[Stage {n} Iter {it:07d}] NEW BEST cheap = {cheap_score:.6f}")

            # ----------------------------------------------------------
            # 2.3.4 Smoothed max‑convolution gradient (full soft‑max)
            # ----------------------------------------------------------
            elapsed = time.time() - start_time
            theta = min(theta_end, theta_start + theta_slope * elapsed)

            # stable soft‑max over the convolution entries
            vals = theta * conv_self
            max_val = vals.max()
            exps = np.exp(vals - max_val)
            w = exps / exps.sum()                     # length = 2n‑1, ∑w = 1

            # gradient of Σ_k w_k (a*a)[k] w.r.t. a is (w * a)[:n] + a
            conv_w_a = _full_convolution_fft(w, a)    # length = 3n‑2
            grad_a = conv_w_a[:n] + a

            # ----------------------------------------------------------
            # 2.3.5 Mirror‑descent conversion to log‑it space
            # ----------------------------------------------------------
            mu = float(np.dot(grad_a, a))
            grad_z = a * (grad_a - mu)

            # ----------------------------------------------------------
            # 2.3.6 Adam moment updates (bias‑corrected)
            # ----------------------------------------------------------
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            # step size with inverse‑sqrt decay
            alpha_t = base_alpha / (math.sqrt(it) + 1.0)
            step = - alpha_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            # ----------------------------------------------------------
            # 2.3.7 Logit update
            # ----------------------------------------------------------
            z = z + step
            # keep logits within a safe numeric window
            np.clip(z, -80.0, 80.0, out=z)

            # ----------------------------------------------------------
            # 2.3.8 Occasional random perturbation (helps escape plateaus)
            # ----------------------------------------------------------
            if it % 5000 == 0 and np.random.rand() < 0.25:
                z += 0.02 * np.random.randn(n)
                m *= 0.5
                v *= 0.5

            # ----------------------------------------------------------
            # 2.3.9 Stagnation handling – reset Adam around current best
            # ----------------------------------------------------------
            if it - last_improve > 250_000:
                print(f"[Stage {n} Iter {it}] Stagnation detected – resetting Adam.")
                # re‑initialise logits from the best seen so far and add tiny noise
                z = _logits_from_distribution(best_seq_stage) + 0.01 * np.random.randn(n)
                m = np.zeros_like(z)
                v = np.zeros_like(z)
                last_improve = it  # give it a fresh chance

        # End of while loop for the current stage
        print(f"[Stage {n}] Best cheap score = {best_score_stage:.6f}")

        # Update the global best (across stages)
        if best_score_stage < best_overall_score:
            best_overall_score = best_score_stage
            best_overall_seq = best_seq_stage.copy()

        # If we have barely any time left, stop expanding further
        if time.time() + 5.0 >= deadline:
            break

    # ------------------------------------------------------------------
    # 3. Final exact evaluation (the platform will also evaluate)
    # ------------------------------------------------------------------
    final_seq = best_overall_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[Result] Final exact score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        final_score = float("inf")
        print(f"[Result] Evaluation error: {exc}")

    # ------------------------------------------------------------------
    # 4. Return the best sequence found.
    # ------------------------------------------------------------------
    return final_seq
```
