"""
Propose a step‑function (non‑negative vector) that gives a low value of

    O(a) = 2·n·max(conv(a,a)) / (sum(a)²)

where conv is the linear (full) convolution.  The algorithm combines three
ingredients:

1.  A soft‑max (log‑it) parametrisation of the simplex and Adam optimisation
    on a smoothed approximation of the max‑operator (controlled by an
    annealing β that grows exponentially).

2.  Periodic restarts from the current best solution with a small random
    jitter – this helps escape plateaus.

3.  A cheap simulated‑annealing style “mass‑transfer” local search that
    runs continuously alongside Adam.  The move transfers a tiny random
    fraction of mass from one index to another and is accepted either if
    it improves the exact objective or with a probability exp(Δ/T) where the
    temperature T decays over time.

The routine runs a handful of independent optimisation runs for several
candidate lengths (the longest length receives the bulk of the time
budget).  After each run the vector is compressed (tiny entries are dropped)
and a short pure local‑search phase is performed.  The best sequence found
over all runs is returned.

All helper functions are top‑level; there are no closures or lambdas.
"""

import time
import numpy as np

# -------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# -------------------------------------------------------------
def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear self‑convolution a*a using a real FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv


# -------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) (a is assumed to sum to 1)."""
    n = a.size
    conv = _conv_self_fft(a)
    return 2.0 * n * conv.max()


# -------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


# -------------------------------------------------------------
def _run_opt(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
) -> tuple[list[float], float]:
    """
    Optimise a vector of length ``n`` for at most ``time_limit`` seconds.
    Returns the best sequence (as a plain list) and its exact objective.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # ---- initialise -------------------------------------------------
    if init_a is not None:
        a0 = np.asarray(init_a, dtype=float)
        if a0.size != n:
            a0 = rng.random(n)
    else:
        a0 = rng.random(n)

    a0 = np.clip(a0, 0.0, None)
    if a0.sum() < 0.01:
        a0 = np.ones_like(a0) * (1.0 / n)
    a0 = a0 / a0.sum()                     # ensure a valid simplex point

    # work in log‑it space (log‑softmax) – this automatically enforces
    # a_i ≥ 0 and Σ a_i = 1 after the soft‑max step.
    x = np.log(a0 + 1e-16)

    # Adam moments
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # ---- annealing / learning‑rate schedules -----------------------
    max_iters = 400_000
    beta_start = 2.0          # start with a modest smoothing
    beta_end   = 1e4          # end with a very sharp max‑approximation
    lr_start   = 0.12
    lr_end     = 0.0008

    best_a = None
    best_obj = np.inf

    # temperature for the auxiliary simulated‑annealing moves
    # (decays from 1e‑2 to 1e‑5)
    temp_start = 1e-2
    temp_end   = 1e-5

    it = 0
    while time.time() - start < time_limit and it < max_iters:
        it += 1

        # ---- obtain probabilities from logits -----------------------
        a = _softmax(x)                     # a sums to 1, a_i ≥ 0

        # ---- exact objective ---------------------------------------
        conv = _conv_self_fft(a)
        M = conv.max()
        obj = 2.0 * n * M
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            print(f"[n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        # ---- schedule values ----------------------------------------
        t = min(it / max_iters, 1.0)
        beta = beta_start * np.exp(np.log(beta_end / beta_start) * t)
        lr   = lr_start   * np.exp(np.log(lr_end   / lr_start)   * t)
        temp = temp_start * np.exp(np.log(temp_end / temp_start) * t)

        # ---- smooth approximation of the max -----------------------
        if beta > 5000.0:                     # very sharp → use true arg‑max
            kmax = int(np.argmax(conv))
            w = np.zeros_like(conv)
            w[kmax] = 1.0
        else:
            shifted = conv - M               # ≤0
            w = np.exp(beta * shifted)
            w /= w.sum()                     # sums to 1

        # ---- gradient of the smoothed max (w.r.t. a) ---------------
        g_a = 2.0 * _conv_self_fft(w)   # correlation of w with a (same as correlation)
        # natural‑gradient wrt logits (soft‑max manifold)
        dot = np.dot(a, g_a)            # ⟨a,g_a⟩
        g_x = a * (g_a - dot)

        # ---- Adam update --------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---- occasional jitter (to shake out of tiny plateaus) ------
        if it % 5000 == 0:
            jitter = rng.normal(scale=0.015, size=n)
            x += jitter
            m.fill(0.0)
            v.fill(0.0)

        # ---- mass‑transfer simulated‑annealing move -----------------
        # perform it with probability ~0.1 per iteration
        if rng.random() < 0.1:
            i, j = rng.choice(n, size=2, replace=False)
            # propose moving a small random fraction of mass from i to j
            delta = a[i] * rng.random() * 0.025
            if delta > 0.0:
                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta
                # evaluate exact objective for the proposed move
                obj_new = _objective(a_new)
                accept = False
                if obj_new < obj:
                    accept = True
                else:
                    # Metropolis acceptance
                    if np.log(rng.random()) < (obj - obj_new) / temp:
                        accept = True
                if accept:
                    # re‑encode the accepted a_new into logits for continuity
                    x = np.log(a_new + 1e-16)
                    a = a_new
                    obj = obj_new
                    if obj_new < best_obj:
                        best_obj = obj_new
                        best_a = a_new.copy()
        # ---- restart from the best solution every 80k iters ----------
        if it % 80_000 == 0:
            if best_a is not None:
                x = np.log(best_a + 1e-16)
                x += rng.normal(scale=0.006, size=n)
                m.fill(0.0)
                v.fill(0.0)
                print(f"[n={n}] restart at it {it}")

    # ---- compress negligible entries --------------------------------
    thresh = 1e-13
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        n_new = best_a.size
        best_obj = _objective(best_a)
        print(f"[compress] length {n} → {n_new}, O={best_obj:.6f}")

    # ---- final pure local‑search (random pair swaps) ---------------
    remaining = time_limit - (time.time() - start)
    if remaining > 0.5:
        loc_start = time.time()
        a_loc = best_a.copy()
        obj_loc = best_obj
        n_loc = a_loc.size
        # simple greedy hill‑climb without temperature (purely improving)
        while time.time() - loc_start < remaining * 0.4:
            i, j = rng.choice(n_loc, size=2, replace=False)
            delta = a_loc[i] * rng.random() * 0.01
            if delta <= 0.0:
                continue
            a_new = a_loc.copy()
            a_new[i] -= delta
            a_new[j] += delta
            obj_new = _objective(a_new)
            if obj_new < obj_loc:
                a_loc, obj_loc = a_new, obj_new
                if obj_new < best_obj:
                    best_a, best_obj = a_new, obj_new
        print(f"[local] final O={best_obj:.6f}")

    return best_a.tolist(), best_obj
