import numpy as np

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     length: int | None = None,
                     **kwargs) -> list[float]:
    """
    Return a non‑negative step‑function that minimises the provided
    ``evaluate_sequence`` bound.

    Parameters
    ----------
    seed : int, optional
        Random seed (kept for API compatibility – not used here).
    budget_s : int, optional
        Time budget in seconds – not needed for the analytic construction.
    length : int | None, optional
        Desired length of the sequence.  If omitted a default of 8000 is
        used (any length ≥2 works, larger n gives a bound closer to 4/3).

    Returns
    -------
    list[float]
        The height sequence.
    """
    # ------------------------------------------------------------------
    # 1) Choose a length.
    #    The analytic bound approaches 4/3 from above as n → ∞,
    #    so a few thousand entries already give a very good score.
    if length is None:
        n = 8000
    else:
        n = int(length)
    if n < 2:
        n = 2

    # ------------------------------------------------------------------
    # 2) Build the decreasing linear sequence.
    #    a_i = 2·(n‑i) / (n·(n+1))   →  Σ a_i = 1 exactly.
    denom = n * (n + 1)
    a = [(2.0 * (n - i)) / denom for i in range(n)]

    # Numerical round‑off can make the sum differ from 1 by ~1e‑15.
    # Renormalise to guarantee the invariant expected by the evaluator.
    s = float(np.sum(a))
    if s == 0.0:                     # safety – should never happen
        a = [1.0 / n] * n
    else:
        a = [x / s for x in a]

    # ------------------------------------------------------------------
    # 3) Evaluate the bound (optional, but helpful for logging).
    try:
        bound = evaluate_sequence(a)
        print(f"[linear‑decreasing] n = {n}, bound O = {bound:.6f}")
    except Exception as exc:  # pragma: no cover – evaluator should be present
        print("[linear‑decreasing] evaluate_sequence failed:", exc)

    # ------------------------------------------------------------------
    # 4) Return the raw Python list as required by the framework.
    return a
```
