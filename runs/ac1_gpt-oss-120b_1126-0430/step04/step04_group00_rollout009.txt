"""
Propose a step‑function sequence with a low evaluation score.

Algorithm
---------
1. Initialise a probability distribution `a` on the simplex.
   * If the global variable ``height_sequence_1`` exists, use it (after
     normalising and clipping); otherwise start from a power‑law.
2. Optimise the distribution using mirror‑descent on the *soft‑max* of the
   self‑convolution (the usual “smoothed‑max” trick) with Adam on the logits.
   * The smoothing constant ε = eps_factor·max(conv) is slowly reduced.
   * When ε becomes very small we switch to the exact sub‑gradient of the
     max‑convolution (indicator over the active indices).
3. Periodically enlarge the support (up to ``max_n``) by linear interpolation
   of the best distribution seen so far; the optimiser is re‑initialised.
4. Random perturbations and occasional restarts prevent stagnation.
5. Throughout the run we keep the best sequence (according to the cheap
   objective 2·n·max(conv)).  At the end we evaluate it with the provided
   ``evaluate_sequence`` and return it.

The implementation respects the ~1000 s budget, uses only NumPy, and keeps
all helper functions at the top level (no closures).  The code is a
self‑contained improvement over the previous baseline and is expected to
beat the target bound 1.503.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level, no closures
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable softmax; returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it representation that yields `a` after softmax."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """
    Smallest power‑of‑two ≥ 3·n‑2,
    which is enough for the linear correlation length (3·n‑2).
    """
    return 1 << ((3 * n - 2).bit_length())


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    total = new_a.sum()
    if total == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= total
    return new_a


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """Normalised power‑law a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _conv_max_and_grad(a: np.ndarray, eps_factor: float) -> tuple[float, np.ndarray]:
    """
    Compute max entry of a*a and the gradient of the *smoothed* objective.

    Parameters
    ----------
    a : np.ndarray
        Probability vector (sum = 1, non‑negative).
    eps_factor : float
        Multiplicative factor for the smoothing constant:
        ε = eps_factor * max(conv).

    Returns
    -------
    max_conv : float
        Maximum of the self‑convolution a*a.
    grad_a : np.ndarray
        Gradient of 2·n·smooth_max w.r.t. a (already multiplied by 4·n).
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    # FFT of a (zero‑padded)
    a_fft = np.fft.rfft(a, L)

    # self‑convolution (linear) – keep the first 2·n‑1 entries
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # – smoothing weights (soft‑max over conv) –
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv                     # for numeric stability
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or np.isnan(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # – correlation a ⋆ w (linear, length 3·n‑2) –
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    grad_a = 4.0 * n * corr
    return max_conv, grad_a


def _max_subgrad(a: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Compute max(conv) and a sub‑gradient of the *exact* max‑convolution.

    This is the limit of the smoothed version when ε→0: the weight vector
    w becomes the normalised indicator of the active (maximal) indices.
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # active set – indices whose value is (numerically) maximal
    tol = 1e-12
    active = np.where(conv >= max_conv - tol)[0]
    w = np.zeros_like(conv)
    if active.size == 0:                     # should not happen
        w[:] = 1.0 / w.size
    else:
        w[active] = 1.0 / active.size

    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    grad_a = 4.0 * n * corr
    return max_conv, grad_a


def _conv_max(a: np.ndarray) -> float:
    """Return max entry of the self‑convolution a*a (no gradient)."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) beats the
    target bound.  The routine respects the given time budget (~1000 s) and
    uses a smoothed‑max Adam optimiser on the probability simplex, with
    progressive support enlargement, occasional random perturbations and a
    final exact evaluation.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0  # safety margin for the final eval

    # ------------------------------------------------------------------
    # 0. Initialise a probability vector `a0`
    # ------------------------------------------------------------------
    try:
        # Global variable from a previous run (list of floats)
        init = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        init = np.clip(init, 0.0, 1000.0)
        s = init.sum()
        if s == 0.0:
            n0 = max(8000, init.shape[0])
            a0 = np.full(n0, 1.0 / n0)
        else:
            a0 = init / s
        n = max(8000, a0.shape[0])
        print(f"Initial sequence taken from global variable, length = {n}")
    except Exception:
        # No global – fall back to a power‑law start.
        n = 8000
        a0 = _power_law_distribution(n, 0.60)
        print(f"Power‑law start, length = {n}")

    # Ensure we have exactly `n` entries
    if a0.shape[0] != n:
        a0 = _interpolate_distribution(a0, n)

    # Log‑it representation (soft‑max parametrisation)
    z = _logits_from_distribution(a0) + 0.01 * np.random.randn(n)

    # ------------------------------------------------------------------
    # 1. Adam state on logits
    # ------------------------------------------------------------------
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    # ------------------------------------------------------------------
    # 2. Optimisation loop
    # ------------------------------------------------------------------
    # hyper‑parameters
    base_alpha = 0.22
    eps_factor = 1.0          # start with strong smoothing
    eps_factor_min = 5e-5
    eps_decay = 0.9998
    max_n = 40000             # allow a larger support than the previous run
    n_growth_factor = 1.3
    expand_interval = 13000   # iterations between support enlargements
    report_every = 5000
    random_perturb_prob = 0.18
    perturb_scale = 0.022

    # best record (cheap score = 2·n·max_conv)
    a = _softmax(z)
    max_c, _ = _conv_max_and_grad(a, eps_factor)
    best_seq = a.copy()
    best_score = 2.0 * n * max_c
    last_improve_iter = 0

    it = 0
    while time.time() < deadline and it < 2_500_000:
        it += 1

        # ----- current point on the simplex -----
        a = _softmax(z)

        # ----- gradient (smooth or exact) -----
        if eps_factor > 0.01:
            max_c, grad_a = _conv_max_and_grad(a, eps_factor)
        else:
            max_c, grad_a = _max_subgrad(a)

        cheap_score = 2.0 * n * max_c

        # ----- keep best ever seen -----
        if cheap_score < best_score - 1e-12:
            best_score = cheap_score
            best_seq = a.copy()
            last_improve_iter = it
            if it % report_every == 0:
                print(f"[Iter {it:07d}] NEW BEST cheap = {cheap_score:.6f}")

        # ----- gradient w.r.t. logits (mirror descent on simplex) -----
        mu = np.dot(grad_a, a)
        grad_z = a * (grad_a - mu)

        # ----- Adam moment updates (bias‑corrected) -----
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # ----- step size schedule (inverse‑sqrt decay) -----
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)
        step_dir = - m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- back‑tracking line search (up to 5 halvings) -----
        trial_alpha = alpha_t
        accepted = False
        for _ in range(5):
            z_new = z + trial_alpha * step_dir
            z_new = np.clip(z_new, -50.0, 50.0)
            a_new = _softmax(z_new)
            new_max = _conv_max(a_new)
            new_score = 2.0 * n * new_max
            if new_score <= cheap_score:
                z = z_new
                accepted = True
                break
            trial_alpha *= 0.5

        if not accepted:
            # take a modest step even if we could not improve
            z = z + alpha_t * step_dir
            z = np.clip(z, -50.0, 50.0)

        # ----- occasional random perturbations (to escape plateaux) -----
        if (it % 4000 == 0) and (np.random.rand() < random_perturb_prob):
            z += perturb_scale * np.random.randn(n)
            m *= 0.6
            v *= 0.6

        # ----- reduce smoothing factor -----
        eps_factor = max(eps_factor_min, eps_factor * eps_decay)

        # ----- progressive support enlargement (if time permits) -----
        if (it % expand_interval == 0) and (n < max_n) and (time.time() + 30 < deadline):
            new_n = min(max_n, int(n * n_growth_factor) + 1)
            print(f"[Iter {it:07d}] Expanding support: {n} → {new_n}")
            a_interp = _interpolate_distribution(best_seq, new_n)
            # reset optimiser for the larger problem
            z = _logits_from_distribution(a_interp) + 0.01 * np.random.randn(new_n)
            m = np.zeros_like(z)
            v = np.zeros_like(z)
            n = new_n
            best_seq = a_interp.copy()
            best_score = 2.0 * n * _conv_max(best_seq)
            last_improve_iter = it
            # keep the smoothing schedule fresh
            eps_factor = 1.0
            continue

        # ----- early stopping if stagnation after reaching max size -----
        if n >= max_n and it - last_improve_iter > 400_000:
            print(f"[Iter {it:07d}] Stagnation detected – terminating early.")
            break

    # ------------------------------------------------------------------
    # 3. Final exact evaluation with the provided evaluate_sequence()
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    # safeguard against the “sum < 0.01” guard
    if sum(final_seq) < 0.01:
        final_seq = [1.0 / len(final_seq)] * len(final_seq)

    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:   # defensive – should never happen
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # ------------------------------------------------------------------
    # 4. Return the best sequence found.
    # ------------------------------------------------------------------
    return final_seq
