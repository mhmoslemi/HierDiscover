"""
Improved step‑function search

The core idea is a projected‑gradient descent on the simplex.
The exact sub‑gradient of the max‑convolution is

    ∂ max(conv(a,a)) / ∂ a = a + shift_k(a)      (k = an arg‑max index)

The previous implementation used only the shift term, which is
sub‑optimal.  Here we add the missing “a’’ term, keep the Adam
optimizer on the log‑it (soft‑max) parametrisation, and use a more
aggressive annealing schedule for the smooth‑max (β).  After the main
optimisation we compress negligible entries, run a short, aggressive
random‑mass‑transfer local search and finally return the best sequence
found within the 1000 s budget.

The code respects the required API: a top‑level ``propose_candidate``
function that returns a plain list of non‑negative floats.
"""

import time
import numpy as np

# -------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# -------------------------------------------------------------
def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (real‑valued) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv


# -------------------------------------------------------------
def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation of w (length 2n‑1) with a (length n):
        g_i = Σ_t w[i+t]·a[t]   for i = 0,…,n‑1
    Implemented by FFT.
    """
    n = a.shape[0]
    m = w.shape[0]                     # = 2n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


# -------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


# -------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))  (assumes sum(a)=1)."""
    n = a.size
    conv = _conv_self_fft(a)
    return 2.0 * n * conv.max()


# -------------------------------------------------------------
def _run_opt(
    n: int,
    time_limit: float,
    init_a: np.ndarray | None = None,
    seed: int = 0,
) -> tuple[list[float], float]:
    """
    Optimise a non‑negative vector of length ``n`` for ``time_limit`` seconds.
    Returns the best sequence (plain list) and its exact objective.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # ----- initialise -------------------------------------------------
    if init_a is not None:
        a = np.asarray(init_a, dtype=float)
        if a.size != n:
            a = rng.random(n)
    else:
        a = rng.random(n)

    # protect against pathological sums
    if a.sum() < 1e-12:
        a = np.ones(n, dtype=float)

    a = a / a.sum()                     # work on the simplex
    x = np.log(a + 1e-16)               # logits (log‑its) – avoids log(0)

    # Adam state
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # ----- schedules --------------------------------------------------
    beta_start = 0.1                    # smoothing start (soft‑max temperature)
    beta_end   = 12000.0                # final temperature, very sharp
    lr_start   = 0.25                   # initial learning rate on logits
    lr_end     = 0.001                  # final learning rate
    max_iters  = 600_000                # safety stop‑gap

    best_a = a.copy()
    best_obj = _objective(best_a)

    it = 0
    while time.time() - start < time_limit and it < max_iters:
        it += 1

        # ----- probabilities from logits --------------------------------
        a = _softmax(x)                 # always sum‑to‑1, non‑negative

        # ----- exact objective -----------------------------------------
        conv = _conv_self_fft(a)
        M = conv.max()
        obj = 2.0 * n * M
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            print(f"[n={n:5d} it={it:06d}] new best O={best_obj:.8f}")

        # ----- schedule parameters --------------------------------------
        t = min(it / max_iters, 1.0)
        beta = beta_start * np.exp(np.log(beta_end / beta_start) * t)
        lr   = lr_start   * np.exp(np.log(lr_end   / lr_start)   * t)

        # ----- soft‑max weights for the (smoothed) max -----------------
        if beta > 5000.0:                # essentially true arg‑max
            kmax = int(np.argmax(conv))
            w = np.zeros_like(conv)
            w[kmax] = 1.0
        else:
            shifted = conv - M           # ≤0
            w = np.exp(beta * shifted)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.zeros_like(conv)
                w[0] = 1.0               # fallback
            else:
                w = w / w_sum

        # ----- gradient of the (smooth) max (now with the extra “a’’ term)
        # g_a = 2 * ( a + correlation(w, a) )
        g_a = 2.0 * (a + _correlation_fft(w, a))

        # ----- natural‑gradient transform to log‑it space ---------------
        dot = np.dot(a, g_a)            # scalar ⟨a,g_a⟩
        g_x = a * (g_a - dot)           # shape (n,)

        # ----- Adam update ---------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- occasional jitter to escape plateaus --------------------
        if it % 5000 == 0:
            jitter = rng.normal(scale=0.03, size=n)
            x += jitter

        # ----- occasional hard restart from the current best ------------
        if it % 80_000 == 0:
            x = np.log(best_a + 1e-16)
            x += rng.normal(scale=0.007, size=n)
            m.fill(0.0)
            v.fill(0.0)
            print(f"[n={n:5d}] restart at iter {it}")

    # ----- compress negligible entries ---------------------------------
    thresh = 1e-12
    mask = best_a > thresh
    if mask.sum() < best_a.size:
        best_a = best_a[mask]
        best_a = best_a / best_a.sum()
        n_new = best_a.size
        best_obj = _objective(best_a)
        print(f"[compress] length {n} → {n_new}, O={best_obj:.8f}")

    # ----- short random local search (mass transfers) ------------------
    remaining = time_limit - (time.time() - start)
    if remaining > 0.8:
        local_start = time.time()
        a_loc = best_a.copy()
        obj_loc = best_obj
        n_loc = a_loc.size
        # Accept‑only hill‑climb; no simulated annealing to keep it fast
        while time.time() - local_start < remaining * 0.35:
            i, j = rng.choice(n_loc, size=2, replace=False)
            # transfer a small random fraction of mass from i to j
            max_delta = a_loc[i] * 0.05            # ≤5 % of a_i
            if max_delta <= 0.0:
                continue
            delta = rng.random() * max_delta
            a_new = a_loc.copy()
            a_new[i] -= delta
            a_new[j] += delta
            conv_new = _conv_self_fft(a_new)
            obj_new = 2.0 * n_loc * conv_new.max()
            if obj_new < obj_loc:
                a_loc, obj_loc = a_new, obj_new
                if obj_new < best_obj:
                    best_a, best_obj = a_new, obj_new
        print(f"[local] final O={best_obj:.8f}")

    return best_a.tolist(), best_obj


# -------------------------------------------------------------
def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **_ignore,
) -> list[float]:
    """
    Main entry point used by the evaluation harness.
    Several optimisation runs (different support sizes) are performed
    within the allotted budget and the best sequence is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # ---------------------------------------------------------
    # candidate lengths – larger supports give more flexibility
    lengths = [8000, 4000, 2000, 1000, 500]

    # If a previous solution exists, make sure we also try its length
    if 'height_sequence_1' in globals():
        seq0 = np.asarray(height_sequence_1, dtype=float).copy()
        seq0 = np.clip(seq0, 0.0, None)
        if seq0.sum() < 0.01:
            seq0 = np.ones_like(seq0)
        seq0 = seq0 / seq0.sum()
        n0 = seq0.size
        if n0 not in lengths:
            lengths.insert(0, n0)

    best_seq = None
    best_obj = float('inf')

    # ---------------------------------------------------------
    # Allocate most time to the first (largest) run, later runs share the rest
    for idx, n in enumerate(lengths):
        remaining = deadline - time.time()
        if remaining <= 1.0:
            break

        if idx == 0:
            run_time = remaining * 0.70
        else:
            run_time = (remaining - 0.5) / (len(lengths) - idx)

        init = None
        if idx == 0 and 'height_sequence_1' in globals():
            init = seq0

        seq, val = _run_opt(
            n=n,
            time_limit=run_time,
            init_a=init,
            seed=seed + idx * 31,
        )
        if val < best_obj:
            best_obj = val
            best_seq = seq
            print(f"[global] new overall best O={best_obj:.8f} (n={n})")

    # ---------------------------------------------------------
    # Final sanity check with the provided evaluate_sequence routine.
    try:
        final_score = evaluate_sequence(best_seq)
        print(f"[final] evaluate_sequence reports O={final_score:.8f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_seq
