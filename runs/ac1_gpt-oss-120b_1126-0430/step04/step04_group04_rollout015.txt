def _simplex_projection(v):
    """Project v onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
```

"""
Propose a non‑negative step function that minimises

        2·n·max_k (a*a)[k] / (Σ a_i)²

The implementation follows a two‑phase scheme:

1. **Warm‑start** – a short scan of power‑law shapes and a batch of
   Dirichlet draws (different concentration parameters) to get a decent
   initial vector.

2. **Smooth‑max Adam optimisation** – we minimise a smooth approximation
   of the max‑convolution using the differentiable loss

        L(a) = log‑sum‑exp( α·(a*a) ) / α

   (α controls the sharpness).  The gradient of L can be written as a
   weighted cross‑correlation, which we evaluate with FFTs in
   O(n log n) time.  An Adam optimiser (with projection onto the
   probability simplex) drives the sequence to lower values of the true
   objective.  The loop restarts and anneals parameters when progress
   stalls.

The routine respects the given time budget (default 1000 s) and always
returns a feasible sequence (non‑negative, sum > 0).  The final exact
score is printed before returning the best vector found.
"""

import time

def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float) -> np.ndarray:
    """Return a normalised power‑law a_i ∝ (i+1)^‑p (Σa_i = 1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = i ** (-p)
    seq /= seq.sum()
    return seq


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using FFT (returns length 2·n‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa).real[: 2 * a.size - 1]
    return conv


def _cross_corr_fft(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ⋆ a)[j] = Σ_i w_i a_{i+j}
    for j = 0…len(a)-1 .  The result length is len(a).
    """
    fw = np.fft.fft(w, fft_len)
    fa = np.fft.fft(a, fft_len)
    # cross‑correlation = ifft( fw * conj(fa) )
    corr_full = np.fft.ifft(fw * np.conj(fa)).real
    # linear part lives from offset = len(a)-1
    offset = a.size - 1
    return corr_full[offset: offset + a.size]


def _cheap_score(vec: np.ndarray, n: int, conv_len: int) -> float:
    """2·n·max(conv) for a vector already normalised to Σ=1."""
    conv = _conv_full_fft(vec, conv_len)
    return 2.0 * n * conv.max()

            print(f"[Init‑Dir {it:03d}] cheap = {val:.6f}")

    # previous construction if compatible
    if prev_arr is not None and prev_arr.size == n:
        val = _cheap_score(prev_arr, n, conv_len)
        if val < best_val:
            best_val = val
            best_a = prev_arr.copy()
            print(f"[Init‑Prev] cheap = {val:.6f}")

    # --------------------------------------------------------------
    # 1. Smooth‑max optimisation with Adam + simplex projection.
    # --------------------------------------------------------------
    a = best_a.copy()
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    lr0 = 0.7                     # initial learning rate
    alpha = 12.0                  # smooth‑max temperature (larger → sharper)
    iteration = 0
    last_improve = 0
    best_iter_a = a.copy()
    best_iter_val = best_val

    while time.time() < deadline and iteration < 600_000:
        iteration += 1

        # ---- full convolution and cheap score ----
        conv = _conv_full_fft(a, conv_len)
        maxb = conv.max()
        cheap = 2.0 * n * maxb

        # keep best seen
        if cheap < best_iter_val - 1e-12:
            best_iter_val = cheap
            best_iter_a = a.copy()
            last_improve = iteration
            print(f"[Iter {iteration:07d}] NEW BEST cheap = {cheap:.6f}")

        # ---- smooth‑max weighting ----
        # avoid overflow by subtracting max before exponentiation
        maxc = conv.max()
        w = np.exp(alpha * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # ---- gradient = 2 * (w ⋆ a)  (cross‑correlation) ----
        grads = 2.0 * _cross_corr_fft(w, a, corr_len)

        # ---- Adam update (minimisation) ----
        m = beta1 * m + (1.0 - beta1) * grads
        v = beta2 * v + (1.0 - beta2) * (grads * grads)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a - step                     # descent step
        a_candidate = _simplex_projection(a_candidate)

        a = a_candidate

        # ---- parameter schedule (annealing) ----
        if iteration % 5000 == 0:
            # slowly increase the sharpness of the smooth‑max
            alpha = min(alpha * 1.07, 80.0)
            # decay learning rate a little
            lr0 *= 0.97

        # ---- occasional random perturbation when stuck ----
        if iteration - last_improve > 80_000:
            # small random jitter around the best iterate so far
            jitter = 0.03 * np.random.rand(n)
            a = _simplex_projection(best_iter_a + jitter)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = iteration
            print(f"[Iter {iteration:07d}] Restart (stagnation)")

    # --------------------------------------------------------------
    # 2. Final exact evaluation.
    # --------------------------------------------------------------
    final_seq = best_iter_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:      # defensive – should never happen
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # --------------------------------------------------------------
    # 3. Return the best sequence.
    # --------------------------------------------------------------
    return final_seq
