#!/usr/bin/env python
"""
Propose a candidate step‑function (non‑negative sequence) that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

using a multi‑restart Adam optimiser on the probability simplex.
The algorithm explores several initial shapes (power‑law, random Dirichlet)
and runs a two‑phase optimisation:

* Phase 1 – smoothed sub‑gradient obtained from the K largest convolution
  entries.  A temperature θ slowly increases, concentrating the weights
  onto the true maximum.
* Phase 2 – exact sub‑gradient of the max‑convolution (average over all
  active offsets).

The optimiser works with logits (softmax → probability vector) so the
constraints a_i ≥ 0 and Σ a_i = 1 are enforced automatically.
Occasional random perturbations help escape plateaus, and the support size
is increased progressively (5 k → 15 k → 30 k) when time permits.

The best sequence found within the time budget is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities (top level – no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max, returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it representation giving `a` after soft‑max."""
    return np.log(a + eps)


def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two ≥ 2·n‑1 (optimal length for rFFT)."""
    return 1 << ((2 * n - 1).bit_length())


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of length `fft_len`."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.size - 1]
    return conv


def _weighted_subgradient(
    a: np.ndarray, conv: np.ndarray, k: int, theta: float
) -> np.ndarray:
    """
    Smoothed sub‑gradient of `max(conv)`.  The K largest convolution entries
    are selected, weighted by a soft‑max with temperature `theta`, and the
    corresponding sub‑gradients are summed.
    """
    n = a.shape[0]
    # indices of the K largest entries (unordered)
    idx = np.argpartition(conv, -k)[-k:]
    # sort them so that the largest gets the biggest weight (more stable)
    order = np.argsort(-conv[idx])
    idx = idx[order]

    # soft‑max weights
    vals = conv[idx]
    max_val = vals[0]
    exps = np.exp(theta * (vals - max_val))
    weights = exps / exps.sum()

    g = np.zeros_like(a)
    for off, w in zip(idx, weights):
        lo = max(0, off - (n - 1))
        hi = min(off, n - 1)
        i = np.arange(lo, hi + 1, dtype=np.int64)
        g[i] += w * 2.0 * a[off - i]
    return g


def _exact_subgradient(a: np.ndarray, conv: np.ndarray, max_val: float) -> np.ndarray:
    """
    Exact sub‑gradient of `max(conv)`.  All offsets attaining the maximum are
    used and their contributions are averaged.
    """
    n = a.shape[0]
    idx = np.where(conv == max_val)[0]
    g = np.zeros_like(a)
    for off in idx:
        lo = max(0, off - (n - 1))
        hi = min(off, n - 1)
        i = np.arange(lo, hi + 1, dtype=np.int64)
        g[i] += 2.0 * a[off - i]
    # average over the active set to stay inside the convex hull of sub‑gradients
    if len(idx) > 0:
        g /= len(idx)
    return g


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """Normalised power‑law a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _random_dirichlet(n: int, alpha: float = 0.3) -> np.ndarray:
    """Generate a random probability vector from a Dirichlet(alpha)."""
    x = np.random.gamma(alpha, 1.0, size=n)
    s = x.sum()
    return x / s if s > 0 else np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Core optimiser for a single support size
# ----------------------------------------------------------------------


def _optimize_distribution(
    a0: np.ndarray,
    n: int,
    start_global: float,
    time_deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex.
    Returns the best probability vector found and its cheap objective
    (2·n·max(conv)).
    """
    # initialise logits
    z = _logits_from_distribution(a0, eps=1e-16)

    # Adam state
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    base_alpha = 0.30  # a slightly larger base learning rate

    # scheduling parameters
    fft_len = _fft_len_for_n(n)

    # sub‑gradient settings
    K_max = min(200, max(10, n // 20))  # start with a fairly large K
    K_min = 1
    phase_change_iter = 150_000  # after this we use the exact sub‑gradient
    theta_start = 0.15
    theta_end = 12.0
    total_time = time_deadline - start_global
    theta_slope = (theta_end - theta_start) / max(1.0, total_time)

    best_seq = a0.copy()
    best_score = float("inf")
    it = 0
    last_improve = 0
    # early‑stop parameters
    stagnation_limit = 250_000

    while time.time() < time_deadline and it < 800_000:
        it += 1

        # current probability vector
        a = _softmax(z)

        # full convolution via FFT
        conv = _conv_full_fft(a, fft_len)
        maxb = float(conv.max())
        cheap_score = 2.0 * n * maxb

        # keep the best solution seen so far
        if cheap_score < best_score:
            best_score = cheap_score
            best_seq = a.copy()
            last_improve = it

        # ----------- sub‑gradient computation -----------------
        # linearly decay K from K_max to K_min over the first phase
        K = max(K_min, int(K_max - (K_max - K_min) * it / phase_change_iter))

        # temperature for the smoothed max (grows with elapsed time)
        elapsed = time.time() - start_global
        theta = min(theta_end, theta_start + theta_slope * elapsed)

        if it < phase_change_iter and K > 1:
            # smoothed sub‑gradient using the K largest entries
            g_a = _weighted_subgradient(a, conv, K, theta)
        else:
            # exact sub‑gradient (average over all active offsets)
            g_a = _exact_subgradient(a, conv, maxb)

        # ---------- mirror‑descent on the simplex -------------
        mu = float(np.dot(g_a, a))
        grad_z = a * (g_a - mu)  # natural gradient for KL geometry

        # Adam moment updates (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # inverse‑sqrt learning‑rate schedule
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)

        # parameter update in log‑it space
        step = -m_hat / (np.sqrt(v_hat) + eps_adam)
        z = z + alpha_t * step
        np.clip(z, -80.0, 80.0, out=z)  # keep logits numerically safe

        # ------------------- occasional noise ------------------
        if it % 5000 == 0 and np.random.rand() < 0.22:
            z += 0.015 * np.random.randn(n)
            m *= 0.5
            v *= 0.5

        # ------------------- early exit ------------------------
        if it - last_improve > stagnation_limit:
            # no improvement for a long time → stop this run
            break

    return best_seq, best_score
