"""
Generate a non‑negative step function that minimises the evaluation metric

    O(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm consists of two phases:

1. **Frank‑Wolfe refinement (LP‑based).**  
   For a fixed length `n` we start from a uniform distribution.
   With the current function `f` we compute `M = max(conv(f,f))` and solve the
   linear programme

        maximise    Σ b_j
        subject to  (f * b)_i ≤ M   for all i
                    b_j ≥ 0

   The LP can be written as `A @ b ≤ M` where `A` is the convolution matrix of
   `f`.  After solving we obtain `g = b / Σb` (renormalised to a simplex) and
   replace `f ← (1‑t)·f + t·g` with a diminishing step size `t`.  Repeating a
   few times quickly drives the maximum self‑convolution down.

2. **Projected Adam fine‑tuning.**  
   The result of phase 1 is used as a warm start for a standard Adam optimiser
   on the soft‑max (log‑it) parametrisation of the simplex.  A smooth approximation
   of the `max` is employed (annealing parameter `β`).  This phase removes the
   remaining small plateau effects.

Both phases are run for several candidate lengths (e.g. 500, 1000, 2000) within
the total time budget (≈ 1000 s).  The best sequence found is returned.
"""

# Basic convolution utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (real output)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) (assumes sum(a)=1)."""
    n = a.size
    conv = _conv_self_fft(a)
    return 2.0 * n * conv.max()

# Adam optimiser (the one from the previous state‑of‑the‑art)
def _softmax(x: np.ndarray) -> np.ndarray:
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """Correlation of w (length 2n‑1) with a (length n)."""
    n = a.shape[0]
    m = w.shape[0]
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]

    start = time.time()

        if obj < best_obj - 1e-14:

            best_a = a.copy()
            print(f"[Adam n={n:5d} it={it:07d}] new best O={best_obj:.6f}")

        # schedule
        t = min(it / max_iters, 1.0)

            print(f"[Adam n={n}] restart at iter {it}")

        print(f"[Compress] length reduced → {best_a.size}, O={best_obj:.6f}")

            delta = a_loc[i] * rng.random() * 0.02

            a_new[i] -= delta

            if obj_new < obj_loc:

                if obj_new < best_obj:

    return best_a.tolist(), best_obj

# Frank‑Wolfe (LP) refinement
def _make_conv_matrix(f: np.ndarray) -> csr_matrix:
    """Return the (2n‑1)×n sparse matrix representing convolution by `f`."""
    n = f.shape[0]
    i = np.arange(n)[:, None]
    k = np.arange(n)[None, :]
    rows = (i + k).ravel()
    cols = np.tile(np.arange(n), n)
    data = np.repeat(f, n)
    A = coo_matrix((data, (rows, cols)), shape=(2 * n - 1, n), dtype=float).tocsr()
    return A


def _solve_lp(f: np.ndarray, M: float, time_limit: float = 5.0) -> np.ndarray | None:
    """
    Solve  maximise Σ b_j   subject to  (f*b)_i ≤ M,  b≥0.
    Returns the optimal b (or None on failure).
    """
    A = _make_conv_matrix(f)
    n = f.size
    c = -np.ones(n)                     # minimise -Σb  → maximise Σb
    b_ub = np.full(A.shape[0], M)
    bounds = [(0, None)] * n
    try:
        res = linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"time_limit": time_limit, "presolve": True},
        )
    except Exception:                   # pragma: no cover
        return None
    if not res.success:
        return None
    b = np.maximum(res.x, 0.0)
    return b


def _frank_wolfe_opt(n: int, time_limit: float, seed: int = 0) -> tuple[list[float], float]:
    """
    Frank‑Wolfe style optimisation for a fixed length `n`.
    Returns the best sequence (as list) and its exact objective.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # start from the uniform distribution on the simplex
    f = np.full(n, 1.0 / n)

    best_f = f.copy()
    best_obj = _objective(best_f)

    t = 0.3                     # initial convex‑combination weight
    itr = 0
    while time.time() - start < time_limit and itr < 12:
        itr += 1
        conv_f = _conv_self_fft(f)
        M = conv_f.max()

        b = _solve_lp(f, M, time_limit=5.0)
        if b is None:
            break

        g = b / b.sum()                 # re‑normalise to the simplex
        f = (1.0 - t) * f + t * g       # convex update

        # keep the best solution seen so far
        obj = _objective(f)
        if obj < best_obj:
            best_obj = obj
            best_f = f.copy()
            print(f"[FW n={n:5d} itr={itr:02d}] new best O={best_obj:.6f}")

        t *= 0.85                       # gradually shrink step size

    return best_f.tolist(), best_obj

def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **_ignore,
) -> list[float]:
    """
    Main entry point for the evaluator.
    Tries several support sizes, performs a fast LP based Frank‑Wolfe pass
    followed by a short Adam fine‑tuning, and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # candidate lengths (keep them modest for the LP phase)
    lengths = [500, 1000, 2000, 4000]

    # If a previous construction exists, try its length first
    if 'height_sequence_1' in globals():
        seq0 = np.asarray(height_sequence_1, dtype=float).copy()
        seq0 = np.clip(seq0, 0.0, None)
        if seq0.sum() < 0.01:
            seq0 = np.ones_like(seq0)
        seq0 = seq0 / seq0.sum()
        n0 = seq0.size
        if n0 not in lengths:
            lengths.insert(0, n0)

    best_seq = None
    best_obj = float("inf")

    for idx, n in enumerate(lengths):
        remaining = deadline - time.time()
        if remaining <= 1.0:
            break

        # allocate time: the first (largest) run gets the bulk
        if idx == 0:
            fw_time = remaining * 0.5
            adam_time = remaining * 0.45
        else:
            # split the remaining budget roughly equally among the rest
            per = (remaining - 0.5) / (len(lengths) - idx)
            fw_time = per * 0.6
            adam_time = per * 0.35

        # ----- Frank‑Wolfe phase -------------------------------------------------
        init_seq = None
        if idx == 0 and 'height_sequence_1' in globals():
            init_seq = seq0
        seq_fw, val_fw = _frank_wolfe_opt(
            n=n,
            time_limit=fw_time,
            seed=seed + idx * 17,
        )
        # use the FW result as initialisation for Adam
        # (if a previous construction is better, keep it as fallback)
        if init_seq is not None and len(init_seq) == n:
            seq_fw = init_seq

        # ----- Adam refinement ----------------------------------------------------
        seq_adam, val_adam = _run_opt(
            n=n,
            time_limit=adam_time,
            init_a=seq_fw,
            seed=seed + idx * 31,
        )

        # pick the better of the two for this length
        if val_adam < val_fw:
            seq_best_len = seq_adam
            obj_best_len = val_adam
        else:
            seq_best_len = seq_fw
            obj_best_len = val_fw

        if obj_best_len < best_obj:
            best_obj = obj_best_len
            best_seq = seq_best_len
            print(f"[Global] new overall best O={best_obj:.6f} (n={n})")

    # final sanity check with the provided evaluator
    try:
        final_score = evaluate_sequence(best_seq)
        print(f"[Final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[Final] evaluate_sequence call failed:", exc)

    return best_seq
