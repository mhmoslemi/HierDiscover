"""
Propose a non‑negative step‑function that (approximately) minimises

    O(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

Algorithm
---------

1️⃣  **Initialisation** – start from the previously found sequence
    (``height_sequence_1``) if it exists, otherwise from a uniform
    distribution of a reasonable size (≈ 8000).  The vector is
    normalised to have unit L¹‑norm (the objective is scale invariant).

2️⃣  **Smooth Adam phase** – a short Adam optimisation on a
    log‑sum‑exp surrogate (same idea as the reference code).  This
    quickly produces a decent shape.

3️⃣  **Successive linear‑programming refinement** – at each iteration

        • compute the exact convolution,
        • select the *M* largest convolution entries (the most binding
          constraints),
        • linearise each selected quadratic constraint
          ``S_k(a)=∑ a_i a_{k‑i}`` around the current point,
        • solve the LP   minimise  t
          subject to  S_k(a0)+∇S_k(a0)·(a−a0) ≤ t   for the chosen indices,
          a_i ≥ 0,   ∑ a_i = 1,   t ≥ 0.

       The LP is solved with ``scipy.optimize.linprog`` (highs).  The
       solution is accepted if the true objective improves; otherwise
       the active‑set size is increased.  The process stops when the
       time budget is almost exhausted or when no improvement is seen.

4️⃣  **Deterministic sub‑gradient polishing** – a long run of projected
    sub‑gradient steps on the exact max‑convolution term, with a
    slowly decaying step‑size.

5️⃣  **Stochastic local search** – cheap “mass‑transfer’’ moves
    (move a random fraction of probability mass from one index to
    another) are performed and accepted either if they improve the
    exact objective or with a Metropolis probability that cools down.

The whole routine respects a 1000 s wall‑clock budget and returns the
best sequence found (as a plain Python list).  All numbers are forced
to be non‑negative by the final projection onto the probability simplex.
"""

import time
import math
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Helper: project onto the probability simplex (Euclidean projection)
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto the simplex {x≥0, sum(x)=1}."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: full convolution using FFT (length 2·n‑1)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return the full a⨂a convolution (length 2·n‑1) using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) (averaged over all maximisers)
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. a (averaged over all maximisers)."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)          # indices achieving the max
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        # derivative is 2 * a_{j_idx}
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g
