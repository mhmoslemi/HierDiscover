"""
Propose a step‑function (non‑negative list of heights) that minimises

    O = 2·n·max(conv(a,a)) / (sum(a)²)

The search uses three complementary ideas:

1️⃣  A coarse grid over power‑law shapes (a_i ∝ (i+β)^‑α) and a
    follow‑up random refinement (including an exponential tail).

2️⃣  A small linear‑programming (LP) “Frank‑Wolfe‑like” step:
    given the current vector f (∑f=1) we maximise ∑g subject to
    (f∗g)_k ≤ max(f∗f) for a selected set of the most active
    convolution indices.  The returned g is normalised to ∑g=1 and
    mixed with f (convex combination) – this often reduces the peak
    of the self‑convolution faster than pure gradient descent.

3️⃣  A final polishing phase using exponentiated‑gradient mirror descent
    on the true objective (sub‑gradient of the max‑convolution).

All phases are time‑capped (total budget 1000 s) and abort gracefully
when the deadline approaches.  The best sequence found is returned
as a plain ``list[float]``.
"""

import time
import math
import numpy as np

# -------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # fallback – uniform simplex
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# -------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a power‑of‑two FFT."""
    n = a.size
    size = 1 << ((2 * n - 2).bit_length())  # next power of two ≥ 2n‑1
    A = np.fft.fft(a, n=size)
    conv = np.fft.ifft(A * A).real[: 2 * n - 1]
    return conv


# -------------------------------------------------------------
def _objective_norm1(a: np.ndarray) -> float:
    """Objective 2·n·max(conv) assuming Σa = 1."""
    n = a.size
    conv = _conv_fft(a)
    return float(2.0 * n * np.max(conv))


# -------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv(a,a)) w.r.t. a (on the simplex)."""
    n = a.size
    conv = _conv_fft(a)
    max_val = conv.max()
    ks = np.where(np.abs(conv - max_val) < 1e-14)[0]
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


# -------------------------------------------------------------
def _exp_update(a: np.ndarray, g: np.ndarray, eta: float) -> np.ndarray:
    """Exponentiated‑gradient update on the simplex (log‑space)."""
    log_a = np.log(a + 1e-15)          # avoid log(0)
    log_a -= eta * g
    log_a -= np.max(log_a)            # numerical stability
    a_new = np.exp(log_a)
    a_new /= a_new.sum()
    return a_new


# -------------------------------------------------------------
def _optimize(a0: np.ndarray,
              max_iters: int,
              eta0: float,
              deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) optimisation.
    Returns the best vector seen and its objective value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective_norm1(a)

    n = a.size
    no_improve = 0
    patience = max_iters // 30   # early‑stop patience

    for t in range(max_iters):
        if time.time() >= deadline:
            break

        g = _subgradient_max_conv(a)
        eta = eta0 * (1.0 - t / max_iters)   # linear decay

        a = _exp_update(a, g, eta)

        cur_score = _objective_norm1(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            print(f"[opt] iter {t:6d}  new best O={best_score:.6f}")
        else:
            no_improve += 1
            if no_improve >= patience:
                # jitter to escape flat regions
                jitter = np.random.randn(n) * 1e-4
                a = best_a + jitter
                a = np.clip(a, 0.0, None)
                a = _project_to_simplex(a)
                no_improve = 0

    return best_a, best_score

def _shape_power(alpha: float, beta: float, n: int) -> np.ndarray:
    """Monotone decreasing a_i ∝ (i+β)^(-α), normalised to sum‑1."""
    i = np.arange(1, n + 1, dtype=float)
    a = (i + beta) ** (-alpha)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


# -------------------------------------------------------------
def _shape_power_exp(alpha: float, beta: float, gamma: float, n: int) -> np.ndarray:
    """Power‑law with exponential tail:
       a_i ∝ (i+β)^(-α)·exp(-γ·(i‑1)/n)   (normalised to sum‑1)."""
    i = np.arange(1, n + 1, dtype=float)
    a = (i + beta) ** (-alpha) * np.exp(-gamma * (i - 1) / n)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


# -------------------------------------------------------------
def _select_constraint_indices(conv: np.ndarray,
                               max_constraints: int = 80) -> np.ndarray:
    """Return up to ``max_constraints`` indices where ``conv`` is largest."""
    if max_constraints >= conv.size:
        return np.arange(conv.size)
    # use argpartition for speed
    idx = np.argpartition(-conv, max_constraints - 1)[:max_constraints]
    # sort them descending for nicer behaviour
    idx = idx[np.argsort(-conv[idx])]
    return idx


# -------------------------------------------------------------
def _lp_max_sum(f: np.ndarray,
                M: float,
                ks: np.ndarray) -> np.ndarray | None:
    """
    Solve   max Σ g_j   s.t.   (f∗g)_k ≤ M  (for all k∈ks),  g≥0.
    Returns the optimal g (may be None on failure).
    """
    n = f.shape[0]
    K = ks.shape[0]

    # Build the (K × n) constraint matrix.
    A = np.zeros((K, n), dtype=float)
    for row, k in enumerate(ks):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        j_range = np.arange(i_start, i_end + 1, dtype=int)
        # coefficient for g_j is f_{k‑j}
        A[row, j_range] = f[k - j_range]

    b = np.full(K, M, dtype=float)
    c = -np.ones(n, dtype=float)               # minimise -Σg ⇒ maximise Σg
    bounds = [(0.0, None)] * n

    try:
        from scipy.optimize import linprog
        # ``highs`` is fast and reliable; suppress output.
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs',
                      options={'presolve': True, 'disp': False})
        if res.success:
            return res.x
        else:
            return None
    except Exception:      # pragma: no cover
        # If SciPy is unavailable we cannot run the LP.
        return None

    """
    Perform a few LP‑based improvement steps.
    Returns the best vector seen and its objective.
    """
    best_f = f.copy()
    best_score = _objective_norm1(best_f)

    for it in range(max_lp_iters):
        if time.time() >= deadline:
            break

        conv_ff = _conv_fft(best_f)
        M = conv_ff.max()
        ks = _select_constraint_indices(conv_ff, max_constraints=80)

        g0 = _lp_max_sum(best_f, M, ks)
        if g0 is None:
            break

        sum_g0 = g0.sum()
        if sum_g0 <= 1e-12:
            break

        g_norm = g0 / sum_g0           # now Σg_norm = 1 and (f∗g_norm) ≤ M
        improved = False

        for t in t_vals:
            cand = (1.0 - t) * best_f + t * g_norm
            cand = np.clip(cand, 0.0, None)
            cand_sum = cand.sum()
            if cand_sum == 0.0:
                continue
            cand /= cand_sum
            score = _objective_norm1(cand)
            if score < best_score - 1e-12:
                best_score = score
                best_f = cand
                improved = True
                print(f"[lp] iter {it:3d} t={t:.2f}  O={best_score:.6f}")
                break

        if not improved:
            # No progress – stop early.
            break

    return best_f, best_score

            break
        a = best_alpha + np.random.randn() * 0.015
        b = best_beta  + np.random.randn() * 0.30
        if a <= 0 or b < 0:
            continue
        g = np.abs(np.random.randn()) * 0.25        # γ in [0,~0.5]
        seq = _shape_power_exp(a, b, g, n)
        score = _objective_norm1(seq)
        if score < best_score:
            best_score = score
            best_seq = seq
            best_alpha, best_beta, best_gamma = a, b, g
            print(f"[rand] new best O={best_score:.6f} (α={a:.4f}, β={b:.4f}, γ={g:.4f})")

    # -----------------------------------------------------------------
    # 3️⃣  LP‑based improvement (few Frank‑Wolfe style steps).
    if time.time() < deadline - 10.0:          # keep some time for polishing
        t_values = [0.1, 0.2, 0.3, 0.5, 0.7]
        best_seq, best_score = _lp_improve(best_seq,
                                           t_vals=t_values,
                                           max_lp_iters=6,
                                           deadline=deadline)
        print(f"[lp] after LP steps best O={best_score:.6f}")

    # -----------------------------------------------------------------
    # 4️⃣  Mirror‑descent polishing (exponentiated‑gradient).
    remaining = deadline - time.time()
    if remaining > 5.0:
        # Rough estimate: ~120 iterations per second on typical hardware.
        max_iters = int(min(180_000, remaining * 120))
        print(f"[opt] polishing – max_iters={max_iters}")
        best_seq, best_score = _optimize(best_seq,
                                         max_iters=max_iters,
                                         eta0=0.6,
                                         deadline=deadline)
        print(f"[opt] polished best O={best_score:.6f}")

    # -----------------------------------------------------------------
    # 5️⃣  Final exact evaluation using the provided evaluator.
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:    # pragma: no cover
        print("[final] evaluate_sequence call failed:", e)
        final_score = None

    return best_seq.tolist()
