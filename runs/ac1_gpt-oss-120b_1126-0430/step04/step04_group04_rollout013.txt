"""
Propose a non‑negative step function (list of heights) that minimises

        2·n·max_k (a*a)[k] / (Σ a_i)² .

The algorithm combines several ideas that were not used together before:

* A richer family of deterministic initialisations (power‑law, shifted
  power‑law and exponential decays) scanned on a longer sequence (up to
  25 000 points).  The best among several thousand candidates is kept.
* A large set of random Dirichlet draws with concentration parameters
  drawn from a log‑uniform distribution (covers both very sparse and
  almost uniform vectors).
* Two optimisation phases:
    – **Phase 1** – a heavy‑ball sub‑gradient method that uses the
      gradients of the *K* largest convolution entries (K=5).  Momentum
      (μ≈0.88) and a decreasing step‑size (η≈0.55/√t) are employed.
    – **Phase 2** – a fine‑grained Adam‑style mirror‑descent whose
      update uses the exact gradient of the (single) maximal convolution
      entry.  The Adam step‑size starts small (≈0.15) and decays.
* Periodic stochastic perturbations and occasional exponentiated‑gradient
  “multiplicative” steps help escape shallow plateaus.
* The whole search respects the time budget (default 1000 s) and always
  returns a feasible normalised sequence (non‑negative, sum > 0).

The final candidate is evaluated with the provided ``evaluate_sequence``
function before being returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top level – no closures)
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def power_law_sequence(n: int, p: float, shift: float = 0.0) -> np.ndarray:
    """Return a normalised power‑law a_i ∝ (i+shift+1)^‑p (Σa_i = 1)."""
    i = np.arange(n, dtype=np.float64) + shift + 1.0
    seq = i ** (-p)
    seq /= seq.sum()
    return seq


def exponential_sequence(n: int, r: float) -> np.ndarray:
    """Return a normalised exponential decay a_i ∝ r^i (Σa_i = 1)."""
    i = np.arange(n, dtype=np.float64)
    seq = r ** i
    seq /= seq.sum()
    return seq


def random_dirichlet(n: int, conc: float) -> np.ndarray:
    """Draw a random point from Dirichlet(conc,…,conc).  conc > 0."""
    g = np.random.gamma(conc, 1.0, size=n)
    seq = g / g.sum()
    return seq


def conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa).real[: 2 * a.size - 1]
    return conv


def conv_max_fft(a: np.ndarray, fft_len: int) -> tuple[float, int]:
    """Maximum value and its index of the autocorrelation a*a."""
    conv = conv_full_fft(a, fft_len)
    idx = int(conv.argmax())
    return float(conv[idx]), idx


def topk_subgradient(a: np.ndarray, conv: np.ndarray, k: int = 5) -> np.ndarray:
    """
    Sub‑gradient built from the k largest convolution entries.
    For each selected shift s we add the gradient 2·a_{s‑i}.
    The result is the average of the k contributions.
    """
    n = a.shape[0]
    # indices of the k largest values (ties are broken arbitrarily)
    top_idx = np.argpartition(conv, -k)[-k:]
    g = np.zeros_like(a)
    for idx in top_idx:
        lo = max(0, idx - (n - 1))
        hi = min(idx, n - 1)
        rng = np.arange(lo, hi + 1, dtype=np.int64)
        g[rng] += 2.0 * a[idx - rng]
    g /= float(k)
    return g


def max_conv_gradient(a: np.ndarray, conv_idx: int) -> np.ndarray:
    """
    Exact gradient of the single maximal convolution entry.
    g_i = 2 * a_{conv_idx-i} if the index is inside the array, else 0.
    """
    n = a.shape[0]
    lo = max(0, conv_idx - (n - 1))
    hi = min(conv_idx, n - 1)
    g = np.zeros_like(a)
    rng = np.arange(lo, hi + 1, dtype=np.int64)
    g[rng] = 2.0 * a[conv_idx - rng]
    return g

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):

    # Target length – a few‑ten‑thousand works well within the time budget.
    n = max(prev_len, 15000)                    # start with at least 15 k

    print(f"[Init] uniform cheap = {val:.6f}")

    # power‑law scan (including a shift parameter)
    p_grid = np.linspace(0.30, 1.30, 61)           # more aggressive range

    for shift in shift_grid:

            cand = power_law_sequence(n, p, shift)
            val = cheap_score(cand)
            if val < best_val:

                print(f"[Init] power‑law p={p:.3f} shift={shift:.1f} cheap={val:.6f}")

    # exponential decay family

    for r in r_grid:
        if time.time() >= deadline:
            break
        cand = exponential_sequence(n, r)
        val = cheap_score(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[Init] exponential r={r:.5f} cheap={val:.6f}")

    # consider the previous construction (if any) – padded / normalised
    if prev_arr is not None:
        if prev_len < n:

        else:

        padded = padded / padded.sum() if padded.sum() > 0 else padded
        val = cheap_score(padded)
        if val < best_val:
            best_val, best_a = val, padded.copy()
            print(f"[Init] previous construction cheap={val:.6f}")

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    rand_iters = 300
    for it in range(rand_iters):

        # draw concentration from a log‑uniform distribution
        conc = 10 ** np.random.uniform(-5.0, 1.0)   # 1e‑5 … 10
        cand = random_dirichlet(n, conc)
        val = cheap_score(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[Rand {it:03d}] new best cheap = {val:.6f}")

    # --------------------------------------------------------------
    # 2. Phase 1 – heavy‑ball sub‑gradient descent (top‑K gradients)
    # --------------------------------------------------------------
    a = best_a.copy()
    momentum = np.zeros_like(a)

    # hyper‑parameters (empirically tuned)
    eta0 = 0.55               # initial step‑size factor
    mu = 0.88                 # momentum coefficient
    topk = 5                  # number of active shifts used for the gradient
    iter_limit = 2_000_000    # generous upper bound – we stop on time
    iter_cnt = 0
    last_improve = 0
    best_iter_a = a.copy()
    best_iter_val = best_val

    while time.time() < deadline and iter_cnt < iter_limit:
        iter_cnt += 1

        # ---- full convolution & cheap objective ----
        conv = conv_full_fft(a, fft_len)
        maxb = conv.max()
        cheap = 2.0 * n * maxb

        # keep the best seen so far
        if cheap < best_iter_val - 1e-12:
            best_iter_val = cheap
            best_iter_a = a.copy()
            last_improve = iter_cnt
            print(f"[Iter {iter_cnt:08d}] NEW BEST cheap = {cheap:.6f}")

        eta = eta0 / math.sqrt(iter_cnt)

        a_candidate = a + momentum
        a_candidate = simplex_projection(a_candidate)

        if np.random.rand() < 0.10:
            mult = np.exp(-eta * g)
            a_candidate = a * mult
            a_candidate = a_candidate / a_candidate.sum()

        a = a_candidate

        # ---- restart if stuck for a long time ----
        if iter_cnt - last_improve > 150_000:
            # small random perturbation of the current best vector
            perturb = 0.02 * np.random.rand(n)
            a = simplex_projection(best_iter_a + perturb)
            momentum = np.zeros_like(a)
            last_improve = iter_cnt
            print(f"[Iter {iter_cnt:08d}] Restart (perturb) after stagnation.")

    # 3. Phase 2 – fine Adam‑style mirror descent (single max gradient)
    # --------------------------------------------------------------
    a = best_iter_a.copy()

    while time.time() < deadline:

        if cheap < best_iter_val - 1e-12:

            best_iter_a = a.copy()

        m = beta1 * m + (1.0 - beta1) * g

            v = np.zeros_like(a)

        print(f"[Error] final evaluation failed: {exc}")

    # --------------------------------------------------------------
    # 5. Return the best sequence found.
    # --------------------------------------------------------------
    return final_seq
