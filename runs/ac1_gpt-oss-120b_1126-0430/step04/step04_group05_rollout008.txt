"""
propose_candidate
-----------------

A three‑phase optimisation that (i) minimises a smooth log‑sum‑exp surrogate
    of the maximal self‑convolution, (ii) refines the solution with an
    Adam‑style optimiser on the exact (non‑smooth) max‑operator and (iii)
    performs a long simulated‑annealing hill‑climb to polish the best
    sequence found.

Key ideas that improve over the previous implementation
-------------------------------------------------------
* A *larger* problem size (n≈10 000).  More degrees of freedom allow the
  convolution to be flattened more effectively, moving the objective
  `2·n·max(conv)` closer to its theoretical lower bound `≈1`.
* An **exponential** schedule for the smoothing parameter α (instead of a
  linear one) – the surrogate becomes sharp much later, giving the optimiser
  more time to explore a smooth landscape.
* Adam on the smoothed surrogate (instead of plain SGD) for faster
  convergence and adaptive step‑sizes.
* A second‑phase Adam that uses the *exact* sub‑gradient of the max‑operator.
  This keeps the heavy‑ball momentum while still moving downhill.
* A final stochastic refinement that (a) moves a random fraction of mass
  between two indices, (b) accepts only improving moves or Metropolis
  moves with a slowly decreasing temperature, and (c) periodically injects
  a tiny amount of Gaussian jitter to escape plateaus.
* All phases respect the global time‑budget (default 1000 s) and keep the
  best sequence encountered.

The function returns a plain ``list[float]`` that can be fed directly to
``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex."""
    n = v.shape[0]
    # Sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:                     # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full self‑convolution via FFT (returns length 2·n‑1)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return a*a (full convolution) using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Sub‑gradient of the maximal convolution entry (as used in the reference code)
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Return the sub‑gradient of max(conv) w.r.t. a."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
# Gradient of the log‑sum‑exp (soft‑max) surrogate of the max
def _smooth_gradient(a: np.ndarray, conv: np.ndarray, alpha: float, L: int) -> np.ndarray:
    """
    Compute ∇_a ( (2·n/α)·log∑_k e^{α·conv_k} ).
    Formula reduces to 2·corr(w, a) where w_k ∝ e^{α·conv_k}.
    """
    # soft‑max weights over convolution entries
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)      # stabilise
    w = exp_scaled / exp_scaled.sum()

    # correlation w ⋆ a   (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)   # length ≥ 3·n‑2
    n_len = a.size
    grad = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]
    return grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises

        O(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

    The algorithm consists of three phases (smooth Adam, exact Adam,
    simulated‑annealing refinement).  The best sequence found before the
    wall‑time expires is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2          # safety margin

    # --------------------------------------------------------------
    # Problem size & initial point
    if 'height_sequence_1' in globals():
        a = np.array(height_sequence_1, dtype=float)
        a = np.maximum(a, 0.0)
        if a.sum() == 0.0:
            a = np.full_like(a, 1.0)
        a = a / a.sum()                         # simplex normalisation
        n = a.size
        print(f"[init] Using supplied sequence, n = {n}")
    else:
        # Larger size than the previous best – still modest enough for the LP‑free approach
        n = 10000
        a = np.full(n, 1.0 / n)                 # uniform start on the simplex
        print(f"[init] Uniform start, n = {n}")

    # FFT length – next power of two > 3·n‑2 (sufficient for all convolutions/correlations)
    L = 1 << ((3 * n - 2).bit_length())

    # --------------------------------------------------------------
    # Phase 1 – Adam on a log‑sum‑exp surrogate (smooth approximation of max)
    best_seq = a.copy()
    conv = _conv_fft(a, L)
    best_obj = 2.0 * n * conv.max()               # exact objective (sum = 1)
    print(f"[phase‑1] start O = {best_obj:.6f}")

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    lr0 = 0.8                                     # a little larger than before
    alpha_start = 5.0
    alpha_end = 200_000.0                         # more aggressive final smoothing

    # allocate roughly 50 % of total budget to this phase
    phase1_deadline = start_time + budget_s * 0.5
    it = 0
    while time.time() < phase1_deadline:
        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_seq = a.copy()
            print(f"[phase‑1] iter {it:,} new best O = {best_obj:.6f}")

        # exponential schedule for α (more gradual sharpening)
        frac = (it + 1) / (phase1_deadline - start_time)   # normalised by elapsed fraction
        alpha = alpha_start * (alpha_end / alpha_start) ** frac

        # gradient of smoothed surrogate
        grad = _smooth_gradient(a, conv, alpha, L)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        lr = lr0 / math.sqrt(it + 1)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional tiny jitter to avoid exact stagnation
        if np.random.rand() < 0.001:
            a_new += np.random.randn(n) * 1e-5

        # project back onto simplex
        a_new = _project_to_simplex(a_new)

        # prune the very smallest 0.5 % of entries every few thousand steps
        if it % 4000 == 0 and it > 0:
            thresh = np.percentile(a_new, 0.5)
            a_new[a_new < thresh] = 0.0
            a_new = _project_to_simplex(a_new)

        a = a_new
        it += 1

    # --------------------------------------------------------------
    # Phase 2 – Adam on the *exact* max‑operator (sub‑gradient version)
    print(f"[phase‑2] start exact Adam, best O = {best_obj:.6f}")

    # Adam state for sub‑gradient (reuse m, v)
    it2 = 0
    phase2_deadline = start_time + budget_s * 0.7   # another ~20 % of the budget
    while time.time() < phase2_deadline:
        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_seq = a.copy()
            print(f"[sub‑grad‑adam] iter {it2:,} new best O = {best_obj:.6f}")

        # true sub‑gradient of the max (non‑smooth)
        g = _subgradient_max_conv(a, conv)

        # Adam‑style step with decaying learning rate
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** (it2 + 1))
        v_hat = v / (1.0 - beta2 ** (it2 + 1))
        lr = lr0 / math.sqrt(it2 + 1) * 0.3          # smaller than phase‑1 learning rate

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = _project_to_simplex(a_new)

        a = a_new
        it2 += 1

    # --------------------------------------------------------------
    # Phase 3 – Stochastic hill‑climbing (simulated annealing)
    print(f"[phase‑3] stochastic refinement, best O = {best_obj:.6f}")

    a = best_seq.copy()          # start refinement from the current best
    conv = _conv_fft(a, L)
    cur_obj = 2.0 * n * conv.max()

    it_loc = 0
    T0 = 0.08
    T = T0
    no_improve = 0
    max_iter_local = 300_000
    no_improve_limit = 12_000

    while time.time() < deadline and it_loc < max_iter_local:
        # ---- mass‑transfer move ----
        i, j = np.random.randint(n, size=2)
        if i != j and a[i] > 1e-15:
            # move up to 8 % of the source mass
            delta = a[i] * np.random.rand() * 0.08
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            a_new = _project_to_simplex(a_new)

            conv_new = _conv_fft(a_new, L)
            new_obj = 2.0 * n * conv_new.max()

            # Metropolis acceptance
            if new_obj < cur_obj or np.random.rand() < math.exp((cur_obj - new_obj) / max(T, 1e-12)):
                a = a_new
                conv = conv_new
                cur_obj = new_obj
                no_improve = 0
                if cur_obj < best_obj:
                    best_obj = cur_obj
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] New global best O = {best_obj:.6f} (iter {it_loc})")
            else:
                no_improve += 1
        else:
            no_improve += 1

        # ---- occasional exact sub‑gradient Adam step (keeps direction) ----
        if it_loc % 11 == 0:
            g = _subgradient_max_conv(a, conv)
            # a tiny Adam‑style step (no momentum to keep it cheap)
            step = 0.04 / (1.0 + it_loc * 1e-5)
            a = _project_to_simplex(a - step * g)
            conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Sub‑grad Adam improved O = {best_obj:.6f}")

        # temperature decay
        T *= 0.9995
        it_loc += 1

        # restart when stuck for a while
        if no_improve >= no_improve_limit:
            a = best_seq.copy()
            a += np.random.randn(n) * 0.0012
            a = np.clip(a, 0, None)
            a = _project_to_simplex(a)
            conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max()
            no_improve = 0

    # ------------------------------------------------------------------
    # Final verification with the provided evaluator
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O = {final_score:.6f}")
    except Exception as exc:                      # pragma: no cover
        print("[final] evaluate_sequence failed:", exc)

    return best_seq.tolist()
