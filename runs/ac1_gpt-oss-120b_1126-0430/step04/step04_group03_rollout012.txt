"""
Improved search for a non‑negative step‑function that minimises

    O = 2·n·max(conv(a,a)) / (sum(a)²)

The routine consists of four phases:

1. **Coarse parametric sweep** – generate candidates of the form
       a_i ∝ (i+β)^(-α)·exp(-γ·i/n)
   on a modest grid for (α,β,γ) and keep the best one.

2. **Focused random refinement** – perturb (α,β,γ) around the current best
   and accept any improvement; this step discovers better tail‑shapes.

3. **Adam‑styled mirror‑descent on the true objective** – work directly
   with the log‑variables (log‑a) on the simplex.  The gradient is the
   sub‑gradient of the currently maximal convolution entry.  An Adam
   update (first‑ and second‑moment estimates) together with a decaying
   learning‑rate yields fast, stable convergence.  When the objective
   plateaus we add a small jitter to escape flat regions.

4. **Final exact evaluation** – the best candidate is checked with the
   provided ``evaluate_sequence`` function.

The whole search respects the supplied time budget (default 1000 s) and
prints progress.  The returned list contains non‑negative heights that
should beat the previous bound (~1.5189) and aim for the target ≤ 1.503.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using an FFT of length power‑of‑two."""
    n = a.size
    # length needed for linear convolution = 2n‑1
    size = 1 << ((2 * n - 2).bit_length())  # next power of two ≥ 2n‑1
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Return the objective O = 2·n·max(conv) assuming Σa = 1.
    (If the sum deviates from 1 we still compute the true bound.)
    """
    n = a.size
    s = a.sum()
    if s == 0.0:
        return float("inf")
    conv = _conv_fft(a / s)          # normalise internally to avoid division later
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
def _subgrad_max_conv(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) w.r.t. a.
    Only the (single) index attaining the maximum is used.
    """
    n = a.size
    conv = _conv_fft(a)
    k = int(np.argmax(conv))               # a single maximising index
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    idx = np.arange(i_start, i_end + 1, dtype=int)
    g = np.zeros_like(a)
    g[idx] = a[k - idx]                    # derivative is a_{k-i}
    return g


# ----------------------------------------------------------------------
def _adam_log_update(log_a: np.ndarray,
                    g: np.ndarray,
                    m: np.ndarray,
                    v: np.ndarray,
                    t: int,
                    lr: float,
                    beta1: float = 0.9,
                    beta2: float = 0.999,
                    eps: float = 1e-8) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """
    Adam update in the log‑domain (mirrored‑descent on the simplex).
    """
    m = beta1 * m + (1.0 - beta1) * g
    v = beta2 * v + (1.0 - beta2) * (g ** 2)

    # bias‑correction
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)

    # step on log‑variables
    log_a = log_a - lr * m_hat / (np.sqrt(v_hat) + eps)

    # keep numbers numerically stable
    log_a -= np.max(log_a)          # shift without changing the simplex point
    a = np.exp(log_a)
    a /= a.sum()
    log_a = np.log(a + 1e-15)       # re‑store safe log
    return log_a, m, v


# ----------------------------------------------------------------------
def _shape_power(alpha: float, beta: float, n: int) -> np.ndarray:
    """a_i ∝ (i+β)^(-α)   (i = 0…n‑1) and normalised to sum = 1."""
    i = np.arange(n, dtype=float) + 1.0          # i starts at 1
    a = (i + beta) ** (-alpha)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _shape_power_exp(alpha: float, beta: float, gamma: float, n: int) -> np.ndarray:
    """a_i ∝ (i+β)^(-α)·exp(-γ·i/n), normalised to sum = 1."""
    i = np.arange(n, dtype=float) + 1.0
    a = (i + beta) ** (-alpha) * np.exp(-gamma * (i - 1.0) / n)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a

            # sub‑gradient at the currently maximal convolution entry
            g = _subgrad_max_conv(a)

            # decaying learning‑rate (initial lr ≈ 0.7)
            lr = 0.7 * (1.0 - t / max_iters) + 0.02

            # Adam update in log‑space
            log_a, m, v = _adam_log_update(log_a, g, m, v, t, lr)

            # recover a from the updated log‑variables
            a = np.exp(log_a)
            a /= a.sum()          # explicit projection (numerical safety)

        # keep the best found during polishing
        if best_opt < best_score:
            best_score = best_opt
            best_seq = best_a
            print(f"[opt] polished best O={best_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣ Final exact evaluation with the provided routine.
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:  # pragma: no cover
        print("[final] evaluate_sequence call failed:", e)
        final_score = None

    return best_seq.tolist()
