def propose_move(a, k_max):
    i_start = max(0, k_max - (n-1))
    i_end = min(n-1, k_max)
    # range of i such that j = k_max - i is valid
    idxs = np.arange(i_start, i_end+1)
    # compute contributions
    contrib = a[idxs] * a[k_max - idxs]
    # find i_max where contribution is highest
    idx = idxs[np.argmax(contrib)]
    j = k_max - idx
    # choose source = idx if a[idx] > a[j] else j
    src = idx if a[idx] >= a[j] else j
    dst = np.random.randint(n)
    while dst == src:
        dst = np.random.randint(n)
    # transfer delta
    delta = a[src] * np.random.rand() * 0.3  # up to 30%
    a_new = a.copy()
    a_new[src] -= delta
    a_new[dst] += delta
    # project onto simplex to compensate rounding
    a_new = _project_to_simplex(a_new)
    return a_new
```

"""
Propose a non‑negative step‑function sequence that (approximately) minimises

    O(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm consists of three stages:

1️⃣  **Parametric warm‑start** – we generate many simple families
    (exponential, truncated exponential, linear, power‑law, logistic)
    and keep the one with the lowest exact objective.  This gives a very
    competitive starting point (often already O < 1.6).

2️⃣  **Greedy max‑convolution reduction** – the current maximal convolution
    entry `k* = argmax conv` is identified.  A mass–transfer move removes a
    fraction of the largest contributing coefficient (or its mirror partner)
    and places it at a random far‑away position.  The move is accepted with a
    Metropolis probability that slowly cools down.

3️⃣  **Sub‑gradient polishing** – every few iterations we take a step in the
    direction of the (averaged) sub‑gradient of the max‑convolution term and
    re‑project onto the probability simplex.  This refines plateaus that the
    greedy moves cannot escape.

All steps respect the global time budget (default 1000 s) and keep the
sequence normalised (`∑a_i = 1`).  The final list is returned as plain
Python floats; `evaluate_sequence` will clip to the required range automatically.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection – Euclidean (used after every modification)
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:                     # all entries are zero → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full convolution via FFT (length 2·n‑1)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return the full a⨂a convolution (length 2·n‑1) using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Exact objective for a vector that is already normalised ∑a = 1
def _objective_norm1(a: np.ndarray, n: int, L: int) -> float:
    """Exact O = 2·n·max(conv) for a probability vector (∑a = 1)."""
    conv = _conv_fft(a, L)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) (used for polishing)
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. a (average over all maximisers)."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)          # indices attaining the max
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
# Candidate families (parameterised non‑negative sequences, normalised)
def _make_exponential(n: int, r: float) -> np.ndarray:
    """a_i ∝ r**i  (full support)"""
    a = np.power(r, np.arange(n, dtype=float))
    a /= a.sum()
    return a


def _make_truncated_exponential(n: int, r: float, m_frac: float) -> np.ndarray:
    """a_i ∝ r**i for i < m; zero otherwise.  m = floor(m_frac·n)."""
    m = max(1, int(m_frac * n))
    a = np.power(r, np.arange(m, dtype=float))
    a /= a.sum()
    full = np.zeros(n, dtype=float)
    full[:m] = a
    return full


def _make_linear(n: int) -> np.ndarray:
    """a_i ∝ (n - i)  (decreasing linear)"""
    a = np.arange(n, 0, -1, dtype=float)
    a /= a.sum()
    return a


def _make_power_decay(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}  (heavy‑tail)"""
    a = 1.0 / np.arange(1, n + 1, dtype=float) ** p
    a /= a.sum()
    return a


def _make_logistic(n: int, beta: float, mu_frac: float) -> np.ndarray:
    """a_i ∝ 1/(1+exp(beta·(i‑mu)))  (sigmoidal decay)"""
    mu = mu_frac * n
    x = np.arange(n, dtype=float)
    a = 1.0 / (1.0 + np.exp(beta * (x - mu)))
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000):
    """
    Search for a step‑function (list of non‑negative heights) that minimises the
    provided evaluation function.  Returns the best sequence found (plain list).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.15               # safety margin

    # --------------------------------------------------------------
    # Problem size – keep the length the same as the previous best if
    # a sequence is supplied, otherwise use a convenient power‑of‑two size.
    if "height_sequence_1" in globals():
        base = np.array(height_sequence_1, dtype=float)
        n = base.size
        a = base / base.sum()
        print(f"[init] start from supplied sequence, n={n}")
    else:
        n = 5000
        a = np.full(n, 1.0 / n)          # uniform start (sum = 1)
        print(f"[init] start from uniform, n={n}")

    # FFT length – smallest power of two ≥ 2·n‑1
    L = 1 << ((2 * n - 2).bit_length())

    # --------------------------------------------------------------
    # 1️⃣  Parametric warm‑start: scan simple families
    best_seq = a.copy()
    best_obj = _objective_norm1(best_seq, n, L)
    print(f"[warm‑start] initial O={best_obj:.6f}")

    # ----- exponential (full support) ---------------------------------
    for r in np.linspace(0.995, 0.9999, 12):
        cand = _make_exponential(n, r)
        obj = _objective_norm1(cand, n, L)
        if obj < best_obj:
            best_obj, best_seq = obj, cand.copy()
            print(f"[warm‑start] exp r={r:.5f} → O={obj:.6f}")

    # ----- truncated exponential ---------------------------------------
    for m_frac in (0.4, 0.5, 0.6, 0.7, 0.8, 0.9, 1.0):
        for r in np.linspace(0.985, 0.9998, 10):
            cand = _make_truncated_exponential(n, r, m_frac)
            obj = _objective_norm1(cand, n, L)
            if obj < best_obj:
                best_obj, best_seq = obj, cand.copy()
                print(f"[warm‑start] trunc exp m={m_frac:.2f} r={r:.5f} → O={obj:.6f}")

    # ----- linear ------------------------------------------------------
    cand = _make_linear(n)
    obj = _objective_norm1(cand, n, L)
    if obj < best_obj:
        best_obj, best_seq = obj, cand.copy()
        print(f"[warm‑start] linear → O={obj:.6f}")

    # ----- power‑law ---------------------------------------------------
    for p in np.linspace(0.4, 2.0, 9):
        cand = _make_power_decay(n, p)
        obj = _objective_norm1(cand, n, L)
        if obj < best_obj:
            best_obj, best_seq = obj, cand.copy()
            print(f"[warm‑start] power p={p:.2f} → O={obj:.6f}")

    # ----- logistic ----------------------------------------------------
    for beta in np.linspace(0.005, 0.05, 7):
        for mu_frac in (0.2, 0.3, 0.4, 0.5):
            cand = _make_logistic(n, beta, mu_frac)
            obj = _objective_norm1(cand, n, L)
            if obj < best_obj:
                best_obj, best_seq = obj, cand.copy()
                print(f"[warm‑start] logistic β={beta:.4f} μ={mu_frac:.2f} → O={obj:.6f}")

    print(f"[warm‑start] best after scanning: O={best_obj:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Greedy max‑convolution reduction + simulated annealing
    print("[local‑refine] entering stochastic local optimisation")
    max_restarts = 0
    iteration = 0
    temperature = 0.02                # initial Metropolis temperature
    subgrad_every = 10                # how often to apply a sub‑gradient step

    while time.time() < deadline:
        # ----- occasional restart (jitter around the best) ---------------
        if np.random.rand() < 0.6:
            a = best_seq.copy()
            a += np.random.randn(n) * 1e-3          # Gaussian jitter
            a = np.clip(a, 0, None)
            a = _project_to_simplex(a)
        else:
            a = np.random.rand(n)
            a = _project_to_simplex(a)

        # pre‑compute current state
        conv = _conv_fft(a, L)
        cur_obj = float(2.0 * n * conv.max())
        k_max = int(conv.argmax())
        it_loc = 0
        no_improve = 0
        T = temperature

        # ----- inner loop ------------------------------------------------
        while time.time() < deadline:
            # 1) focused mass‑transfer move aimed at the current worst k
            i_start = max(0, k_max - (n - 1))
            i_end = min(n - 1, k_max)
            if i_start > i_end:                     # should not happen
                break
            idxs = np.arange(i_start, i_end + 1, dtype=int)
            # contributions a_i * a_{k-i}
            contrib = a[idxs] * a[k_max - idxs]
            # pick the index that participates in the largest product
            ii = idxs[np.argmax(contrib)]
            jj = k_max - ii
            # choose the larger of the two as source
            src = ii if a[ii] >= a[jj] else jj

            # destination far away (prefer opposite side of the array)
            dst = np.random.randint(n)
            while dst == src:
                dst = np.random.randint(n)

            # move up to 30 % of the source mass
            delta = a[src] * np.random.rand() * 0.30
            a_new = a.copy()
            a_new[src] -= delta
            a_new[dst] += delta
            a_new = _project_to_simplex(a_new)

            conv_new = _conv_fft(a_new, L)
            new_obj = float(2.0 * n * conv_new.max())
            k_new = int(conv_new.argmax())

            # Metropolis acceptance
            if new_obj < cur_obj or np.random.rand() < math.exp((cur_obj - new_obj) / max(T, 1e-12)):
                a, conv, cur_obj, k_max = a_new, conv_new, new_obj, k_new
                no_improve = 0
                if cur_obj < best_obj:
                    best_obj, best_seq = cur_obj, a.copy()
                    print(f"[{time.time():.1f}] NEW GLOBAL BEST O={best_obj:.6f} (restart={max_restarts}, iter={it_loc})")
            else:
                no_improve += 1

            # 2) occasional sub‑gradient polishing step
            if it_loc % subgrad_every == 0:

                eta = 0.07 * (1.0 - it_loc / 200_000)   # slowly decaying step
                a = a - eta * g
                a = _project_to_simplex(a)
                conv = _conv_fft(a, L)
                cur_obj = float(2.0 * n * conv.max())
                k_max = int(conv.argmax())
                if cur_obj < best_obj:
                    best_obj, best_seq = cur_obj, a.copy()
                    print(f"[{time.time():.1f}] sub‑grad improved O={best_obj:.6f}")

            # 3) temperature decay
            T *= 0.9998
            iteration += 1
            it_loc += 1

            if no_improve >= 4000:
                break

        max_restarts += 1
        # keep a small reserve of time for the final evaluation
        if time.time() >= deadline:
            break

        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence failed:", exc)

    # Return the best sequence discovered (as a plain Python list)
    return best_seq.tolist()
