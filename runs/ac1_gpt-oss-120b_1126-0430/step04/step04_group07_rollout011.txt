def solve_lp_refine(f, M):
    n = len(f)
    # Build sparse A_ub for constraints A @ g <= M
    # Create arrays of data, rows, cols
    row = []
    col = []
    data = []
    for i in range(2*n - 1):
        j_start = max(0, i - (n-1))
        j_end = min(n-1, i)
        # indices of f slice:
        # For each j in [j_start, j_end]:
        # column = i - j
        # value = f[j]
        # Add into arrays
        # length = j_end - j_start + 1
        # Use vectorized:
        js = np.arange(j_start, j_end+1)
        cols = i - js
        rows = np.full_like(cols, i)
        row.extend(rows.tolist())
        col.extend(cols.tolist())
        data.extend(f[js].tolist())
    A = scipy.sparse.csr_matrix((data, (row, col)), shape=(2*n-1, n))
    # Upper bound vector b = M * ones
    b = np.full(2*n-1, M)
    # Bounds for variables: (0, None)
    bounds = [(0, None) for _ in range(n)]
    # Objective c: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    # Solve
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        raise RuntimeError("LP failed")
    return res.x

"""
Hybrid mirror‑descent + active‑set linear‑program refinement for the
step‑function bound.

The core optimiser is an exponentiated‑gradient (mirror‑descent) on the
log‑probabilities u_i = log a_i, which minimises the quantity

    O(a) = 2·n·max(conv(a,a))          (sum(a) is normalised to 1)

Early iterations use a smooth soft‑max weight over the convolution entries;
later iterations concentrate the weight on the (few) maximal entries,
giving the exact sub‑gradient of the max‑operator.

Every ``lp_interval`` iterations a tiny linear‑program is solved:
given the current f, we maximise Σ g_j subject to the linear constraints

    (f * g)[i] ≤ M   for all i

where M = max(conv(f,f)).  Only the (near‑maximal) constraints are kept in an
active‑set; violated constraints are added iteratively (cut‑plane method).
The resulting g (scaled to a probability vector) is mixed with f using a
small step size ``lp_mix``.  This step is guaranteed never to increase the
max‑convolution and often yields a noticeable drop in the bound.

The routine respects a 1000 s budget, prints progress, evaluates the final
solution with the provided ``evaluate_sequence`` and returns the best list
found.
"""

import time
import numpy as np
import cvxpy as cp

# -------------------------------------------------------------
#  FFT based convolution / correlation helpers
# -------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv


def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Linear correlation g_i = Σ_t w[i+t]·a[t]  for i = 0,…,n‑1.
    Implemented via FFT.  ``w`` must have length 2·n‑1.
    """
    n = a.shape[0]
    m = w.shape[0]                # = 2·n‑1
    L = _next_pow2(m + n - 1)
    wf = np.fft.rfft(w, n=L)
    af = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(wf * np.conj(af), n=L)
    return corr_full[:n]


# -------------------------------------------------------------
#  Active‑set linear programme (maximise Σ g_j  subject to  f*g ≤ M)
# -------------------------------------------------------------
def _lp_refine_active(f: np.ndarray,
                      M: float,
                      tol: float = 1e-9,
                      max_cuts: int = 30) -> np.ndarray:
    """
    Solve the LP that maximises the total mass of ``g`` while keeping
    ``conv(f, g) ≤ M``.  Only constraints that are (nearly) active are
    kept in an active set, which is expanded with a cutting‑plane loop.
    Returns the optimal ``g`` (non‑negative, not normalised).

    Parameters
    ----------
    f : np.ndarray
        Current probability vector (sum = 1, length n).
    M : float
        Current maximal self‑convolution value of ``f``.
    tol : float
        Feasibility tolerance for the omitted constraints.
    max_cuts : int
        Maximum number of cutting‑plane iterations.

    Returns
    -------
    g : np.ndarray
        The optimal vector (may contain very small values).
    """
    n = f.size
    # start with the constraints that are already tight (or almost)
    conv_ff = _conv_self_fft(f)
    active = list(np.where(conv_ff >= M - max(tol, M * 1e-8))[0])
    if not active:                     # safety fallback
        active = [int(conv_ff.argmax())]

    for _ in range(max_cuts):
        # ----- build the (small) LP -----
        g = cp.Variable(n, nonneg=True)
        constraints = []
        for i in active:
            j_start = max(0, i - (n - 1))
            j_end = min(n - 1, i)
            f_slice = f[j_start:j_end + 1]
            idx = i - np.arange(j_start, j_end + 1)   # columns of g
            constraints.append(cp.sum(cp.multiply(f_slice, g[idx])) <= M)

        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)

        if prob.status not in ("optimal", "optimal_inaccurate"):
            # LP failed – give up and return the current feasible point
            break

        g_val = np.asarray(g.value).ravel()
        g_val = np.maximum(g_val, 0.0)          # numerical safety

        # ----- check for violated constraints -----
        conv_fg = np.convolve(f, g_val)
        viol = conv_fg - M
        max_viol = viol.max()
        if max_viol <= tol:
            return g_val

        # add the most violated index to the active set and repeat
        worst_i = int(np.argmax(viol))
        if worst_i not in active:
            active.append(worst_i)

    # Return the last feasible solution (even if a few constraints are slightly violated)
    return g_val


# -------------------------------------------------------------
#  Main search routine
# -------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list:
    """
    Search for a non‑negative step‑function that minimises the bound
    evaluated by ``evaluate_sequence``.  Returns the best sequence found
    as a plain list of floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin before termination

    # ---------------------------------------------------------
    # 1. Initialise the log‑probabilities
    # ---------------------------------------------------------
    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=float)
        a0 = a0 / a0.sum()
        n = a0.size
        # tiny jitter to avoid exact zeros in the log‑domain
        eps = 1e-12
        u = np.log(a0 + eps)
    else:
        # a moderately large length – a good trade‑off between expressivity
        # and per‑iteration cost
        n = 8000
        u = np.full(n, -np.log(n))               # corresponds to uniform a

    best_u = u.copy()
    best_score = np.inf
    best_iter = 0

    # ---------------------------------------------------------
    # 2. Hyper‑parameters & schedules
    # ---------------------------------------------------------
    # learning‑rate schedule (exponential decay)
    eta_start = 0.8
    eta_end   = 0.02

    # soft‑max temperature (beta) – starts low, ends very high
    beta_start = 2.0
    beta_end   = 25000.0

    max_iters_est = 1_200_000   # used only for schedule interpolation

    jitter_interval = 3000
    jitter_scale = 5e-4

    restart_interval = 80_000

    # LP‑refinement schedule
    lp_interval = 150_000       # run LP every this many iterations
    lp_mix = 0.08               # convex‑combination weight for the LP solution

    it = 0
    while time.time() < deadline:
        it += 1

        # -------------------------------------------------
        # (a)   build probability vector a and its self‑conv
        # -------------------------------------------------
        a = np.exp(u - u.max())               # log‑normalisation for stability
        a_sum = a.sum()
        if a_sum == 0.0:                       # safety fallback
            a = np.full_like(a, 1.0 / n)
            a_sum = 1.0
        a = a / a_sum

        conv = _conv_self_fft(a)
        M = conv.max()
        cur_score = 2.0 * n * M                 # because sum(a)=1

        # -------------------------------------------------
        # (b)   keep the best solution seen so far
        # -------------------------------------------------
        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_u = u.copy()
            best_iter = it
            print(f"[iter {it:07d}] new best O={best_score:.6f}")

        # -------------------------------------------------
        # (c)   schedule parameters
        # -------------------------------------------------
        fraction = min(it / max_iters_est, 1.0)
        beta = beta_start + (beta_end - beta_start) * fraction
        eta = eta_start * (eta_end / eta_start) ** fraction

        # -------------------------------------------------
        # (d)   build the weight vector over convolution entries
        # -------------------------------------------------
        if fraction < 0.35:                     # early soft‑max phase
            shifted = conv - M                  # ≤ 0
            w = np.exp(beta * shifted)
            w_sum = w.sum()
            if w_sum == 0.0:                    # numerical safeguard
                w = np.zeros_like(conv)
                w[conv.argmax()] = 1.0
            else:
                w /= w_sum
        else:                                   # hard‑max / active‑set phase
            # tolerance relative to M – extremely tight
            tol = max(1e-12, M * 1e-9)
            active = np.where(conv >= M - tol)[0]
            if active.size == 0:
                w = np.zeros_like(conv)
                w[conv.argmax()] = 1.0
            else:
                w = np.zeros_like(conv)
                w[active] = 1.0 / active.size

        # -------------------------------------------------
        # (e)   gradient of the (smoothed) max‑convolution
        # -------------------------------------------------
        g_grad = 2.0 * _correlation_fft(w, a)   # shape (n,)

        # -------------------------------------------------
        # (f)   mirror‑descent / exponentiated‑gradient step
        # -------------------------------------------------
        u = u - eta * g_grad

        # -------------------------------------------------
        # (g)   occasional jitter to escape plateaus
        # -------------------------------------------------
        if it % jitter_interval == 0:
            u += np.random.randn(n) * jitter_scale

        # -------------------------------------------------
        # (h)   cheap restart from the best‑so‑far solution
        # -------------------------------------------------
        if it % restart_interval == 0:
            u = best_u.copy()
            u += np.random.randn(n) * (jitter_scale * 0.5)
            print(f"[restart] iter {it}, re‑initialised from best (O={best_score:.6f})")

        # -------------------------------------------------
        # (i)   periodic LP refinement (active‑set)
        # -------------------------------------------------
        if it % lp_interval == 0:
            try:
                g_raw = _lp_refine_active(a, M, tol=1e-9, max_cuts=30)
                sum_g = g_raw.sum()
                if sum_g > 1e-12:
                    g_norm = g_raw / sum_g
                    # convex combination: a ← (1‑t)·a + t·g_norm
                    a = (1.0 - lp_mix) * a + lp_mix * g_norm
                    a = np.maximum(a, 0.0)
                    a /= a.sum()
                    # re‑initialise the log‑variables from the mixed vector
                    u = np.log(a + 1e-12)

                    # re‑evaluate after LP step (might be an immediate improvement)
                    conv = _conv_self_fft(a)
                    M = conv.max()
                    cur_score = 2.0 * n * M
                    if cur_score < best_score - 1e-13:
                        best_score = cur_score
                        best_u = np.log(a + 1e-12)
                        best_iter = it
                        print(f"[LP step] iter {it} new best O={best_score:.6f}")
                else:
                    # degenerate LP – skip
                    pass
            except Exception as exc:   # pragma: no cover
                print(f"[LP] failed at iter {it}: {exc}")

    # -----------------------------------------------------------------
    # 3. Final evaluation and return
    # -----------------------------------------------------------------
    best_seq = np.exp(best_u - best_u.max())
    best_seq = best_seq / best_seq.sum()
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f} "
              f"(best iteration {best_iter})")
    except Exception as exc:                                   # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)

    return best_seq.tolist()
