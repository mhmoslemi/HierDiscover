def _adam_optimize(a0, max_iters, lr, T_init, T_final, deadline):
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_a = a.copy()
    best_obj = _objective_norm1(a)
    for t in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        # temperature schedule
        T = T_init * ((T_final / T_init) ** (t / max_iters))
        obj, w = _objective_smooth(a, T)
        # compute gradient
        conv_slice = _conv_full_fft(w[::-1], a)[n-1:2*n-1]
        grad = 4 * n * conv_slice  # shape n
        # Adam update
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        # project to simplex
        a = _project_to_simplex(a)
        # evaluate exact objective periodically
        if t % 1000 == 0 or t == max_iters:
            cur_obj = _objective_norm1(a)
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                print(f"[adam] iter {t:6d} new best O={best_obj:.6f}")
    return best_a, best_obj

"""
Improved optimizer for the step‑function bound O = 2·n·max(conv(a,a)) / (sum(a)²).

Main ideas
---------

* Work on the *simplex* (a ≥ 0, Σa = 1).  The scaling of a does not affect the
  objective, therefore we can keep the sum fixed and only optimise the shape.
* Replace the hard “max” by a smooth log‑sum‑exp (temperature T).  This gives a
  differentiable surrogate; its gradient can be written in closed form and
  evaluated with two fast Fourier transforms.
* Optimise the smoothed objective with Adam (projected onto the simplex after
  each update).  The temperature is annealed from a relatively high value
  (strong smoothing, large steps) down to a tiny value (close to the true max).
* Use the previously found construction (if present) as a warm start and also
  try a few simple parametric families (power‑law + exponential tail) as
  baselines.
* The whole routine is time‑capped; it prints progress and finally returns the
  best sequence found (as a plain ``list[float]``).

The code follows the required API – the platform will call ``propose_candidate``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the probability simplex {x≥0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:           # all entries are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a FFT of suitable length."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    A = np.fft.fft(a, n=size)
    conv = np.fft.ifft(A * A).real[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def _conv_full_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors via FFT."""
    m = x.size + y.size - 1
    size = _next_pow2(m)
    X = np.fft.fft(x, n=size)
    Y = np.fft.fft(y, n=size)
    conv = np.fft.ifft(X * Y).real[:m]
    return conv


# ----------------------------------------------------------------------
def _objective_norm1(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a already normalised (Σa=1)."""
    n = a.size
    conv = _conv_fft(a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
def _objective_smooth(a: np.ndarray, T: float):
    """
    Smoothed objective = 2·n·( T·log Σ exp(conv/T) )
    Returns (value, softmax_weights) where the weights sum to 1.
    """
    n = a.size
    conv = _conv_fft(a)                     # length 2n-1
    m = conv.max()
    # stable softmax
    exp_shift = np.exp((conv - m) / T)      # all ≤ 1
    Z = exp_shift.sum()
    w = exp_shift / Z                       # softmax weights
    smooth_max = m + T * np.log(Z)          # = T·log Σ exp(conv/T)
    value = 2.0 * n * smooth_max
    return float(value), w


# ----------------------------------------------------------------------
def _gradient_smooth(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smoothed objective (including the outer factor 2·n).
    Implements   ∇ O = 4·n·reverse_slice( conv(w_rev , a) )
    where w_rev = w[::-1] and the slice is indices [n‑1 … 2n‑2].
    """
    n = a.size
    w_rev = w[::-1]
    conv_full = _conv_full_fft(w_rev, a)    # length 3n‑2
    # slice [n-1, 2n-1) (size n) and reverse it
    slice_part = conv_full[n - 1: 2 * n - 1][::-1]
    grad = 4.0 * n * slice_part
    return grad


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   max_iters: int,
                   lr: float,
                   T_start: float,
                   T_end: float,
                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smoothed objective, projected onto the simplex.
    Returns the best vector seen and its *exact* objective value.
    """
    a = a0.copy()
    n = a.size
    m = np.zeros_like(a)          # 1st moment
    v = np.zeros_like(a)          # 2nd moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    best_a = a.copy()
    best_score = _objective_norm1(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # temperature annealing (geometric)
        T = T_start * ((T_end / T_start) ** (t / max_iters))

        # smooth value and soft‑max weights
        _, w = _objective_smooth(a, T)

        # gradient of the *exact* smoothed objective
        grad = _gradient_smooth(a, w)

        # Adam update (in the ambient space)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # enforce simplex constraints
        a = _project_to_simplex(a)

        # evaluate exact objective every few thousand steps
        if t % 2000 == 0 or t == max_iters:
            cur = _objective_norm1(a)
            if cur < best_score - 1e-12:
                best_score = cur
                best_a = a.copy()
                print(f"[adam] iter {t:6d}  new best O={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
def _shape_power(alpha: float, beta: float, n: int) -> np.ndarray:
    """a_i ∝ (i+β)^(-α), i=1..n, normalised to sum‑1."""
    i = np.arange(1, n + 1, dtype=float)
    a = (i + beta) ** (-alpha)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
def _shape_power_exp(alpha: float, beta: float, gamma: float, n: int) -> np.ndarray:
    """Power‑law with an exponential tail."""
    i = np.arange(1, n + 1, dtype=float)
    a = (i + beta) ** (-alpha) * np.exp(-gamma * (i - 1) / n)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,

    """
    Search for a non‑negative step function that minimises the supplied
    ``evaluate_sequence`` bound.  Returns the sequence as a plain list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size – a power of two works nicely with the FFT.
    # 8192 gives a finer discretisation than the earlier 4096 and still
    # stays comfortably within the time budget.
    n = 8192

    # ------------------------------------------------------------------

    #     tail).  The grid is deliberately a bit finer than before.
    alphas = np.linspace(0.30, 0.70, 41)       # 41 values
    betas  = np.linspace(0.0, 25.0, 41)        # 41 values

    best_seq = None
    best_score = float("inf")
    best_alpha = best_beta = best_gamma = None

    print("[grid] start coarse search (power‑law only)")
    for a in alphas:
        for b in betas:
            seq = _shape_power(a, b, n)
            sc = _objective_norm1(seq)
            if sc < best_score:
                best_score = sc
                best_seq = seq
                best_alpha, best_beta = a, b
    print(f"[grid] best O={best_score:.6f}  (α={best_alpha:.4f}, β={best_beta:.4f})")

    # ------------------------------------------------------------------
    # 2️⃣  Expand the search with an exponential tail factor γ.
    print("[grid] adding exponential tail")
    for _ in range(2000):
        if time.time() >= deadline:
            break
        a = best_alpha + np.random.randn() * 0.02
        b = best_beta  + np.random.randn() * 0.4
        if a <= 0 or b < 0:

        seq = _shape_power_exp(a, b, g, n)
        sc = _objective_norm1(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq
            best_alpha, best_beta, best_gamma = a, b, g
            print(f"[rand] new best O={best_score:.6f} (α={a:.4f}, β={b:.4f}, γ={g:.4f})")

    # ------------------------------------------------------------------
    # 3️⃣  Warm‑start from any previously found construction.
    if 'height_sequence_1' in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=float)
            if prev.size > 0:
                # Rescale / interpolate to current length n
                if prev.size != n:
                    # simple linear interpolation onto the new grid
                    x_old = np.linspace(0.0, 1.0, prev.size)
                    x_new = np.linspace(0.0, 1.0, n)
                    prev = np.interp(x_new, x_old, prev)

                if prev.sum() < 0.01:      # avoid degenerate case

                prev /= prev.sum()
                prev_score = _objective_norm1(prev)
                if prev_score < best_score:
                    best_seq = prev
                    best_score = prev_score
                    print(f"[prev] using previous construction O={best_score:.6f}")
        except Exception:  # pragma: no cover
            pass

    # ------------------------------------------------------------------
    # 4️⃣  Adam optimisation on the smooth surrogate objective.
    #     Allocate the bulk of the remaining budget.
    remaining = deadline - time.time()
    if remaining > 8.0:
        # Rough speed estimate: ~120 Adam‑steps per second on typical hardware.
        max_iters = int(min(200_000, remaining * 120))
        print(f"[adam] start optimisation – max_iters={max_iters}")

        # Adam parameters (tuned empirically)
        lr = 0.08               # base learning rate
        T_start = 5e-2          # fairly smooth at the beginning
        T_end = 5e-5            # close to the true max at the end

        a_opt, score_opt = _adam_optimize(
            best_seq,
            max_iters=max_iters,
            lr=lr,
            T_start=T_start,
            T_end=T_end,
            deadline=deadline,
        )
        if score_opt < best_score:
            best_seq = a_opt
            best_score = score_opt
            print(f"[adam] polished best O={best_score:.6f}")

    # 5️⃣  Tiny random jitter after polishing – sometimes helps escape a flat
    #     plateau that the gradient descent missed.

        if time.time() >= deadline:
            break
        cand = best_seq + np.random.randn(n) * 5e-5
        cand = np.clip(cand, 0.0, None)
        cand = _project_to_simplex(cand)
        sc = _objective_norm1(cand)
        if sc < best_score:
            best_seq = cand
            best_score = sc
            print(f"[jitter] improved O={best_score:.6f}")

        final_score = None
