"""
propose_candidate
-----------------

Search for a non‑negative step‑function (a list of heights) that minimises

    O = 2·n·max(conv(a,a)) / (sum(a)²)

The method uses three stages:

1. **Coarse parametric sweep** – power‑law shapes
       a_i ∝ (i+β)^(‑α)·exp(‑γ·(i‑1)/n)
   on a modest grid (α,β,γ) for a moderately large length *n*.
2. **Local random refinement** – Gaussian perturbations of the best
   parameters from step 1.
3. **Smooth‑max mirror descent** – minimise a smoothed version of the
   max‑convolution using exponentiated‑gradient updates.
   The smoothing parameter λ is annealed (λ→large) so that the
   optimisation first sees a gentle landscape and finally focuses on the
   true maximum.  All updates keep the vector on the probability simplex
   (non‑negative, sum = 1).

The routine adapts the problem size (n = 4096 or 8192) to the available
time, respects the 1000 s budget, prints progress and finally returns the
best sequence as a plain ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x≥0, Σx=1} (used only for jitter)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a power‑of‑two FFT."""
    n = a.size
    size = 1 << ((2 * n - 2).bit_length())        # next power of two ≥ 2n‑1
    A = np.fft.fft(a, n=size)
    conv = np.fft.ifft(A * A).real[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def _objective_norm1(a: np.ndarray) -> float:
    """Objective 2·n·max(conv) assuming Σa = 1."""
    n = a.size
    conv = _conv_fft(a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
def _smooth_max_and_weights(conv: np.ndarray, lam: float):
    """
    Smoothed max:   (1/lam)·log Σ exp(lam·conv_k).
    Returns (smooth_max, weight vector w_k = exp(lam·conv_k)/Σexp).
    """
    scaled = lam * conv
    shift = np.max(scaled)                     # for numerical stability
    exp_vals = np.exp(scaled - shift)
    Z = exp_vals.sum()
    smooth = (np.log(Z) + shift) / lam
    w = exp_vals / Z
    return smooth, w


# ----------------------------------------------------------------------
def _shape_power_exp(alpha: float, beta: float, gamma: float, n: int) -> np.ndarray:
    """
    Produce a monotone decreasing sequence
        a_i ∝ (i+β)^(-α)·exp(-γ·(i‑1)/n),      i = 1..n,
    then normalise to sum‑1.
    """
    i = np.arange(1, n + 1, dtype=float)
    a = (i + beta) ** (-alpha) * np.exp(-gamma * (i - 1) / n)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
def _smooth_optimize(a0: np.ndarray,
                     deadline: float,
                     max_iters: int) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent optimisation on the smoothed objective.
    Returns the best vector seen and its (true) objective value.
    """
    a = a0.copy()
    n = a.size
    best_a = a.copy()
    best_score = _objective_norm1(a)

    # annealing parameters
    lam = 5.0               # start smooth
    lam_max = 120.0
    eta = 0.5               # step size for exponentiated‑gradient
    eta_min = 1e-4

    # early‑stop / jitter handling
    no_improve = 0
    patience = max_iters // 30 if max_iters >= 30 else 0

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # -------- gradient of smoothed max ----------
        conv = _conv_fft(a)                     # length 2n‑1
        _, w = _smooth_max_and_weights(conv, lam)

        # convolution w * a_rev gives ∂smooth_max/∂a (up to factor 2)
        a_rev = a[::-1]
        size = 1 << ((len(w) + n - 1).bit_length())
        W_fft = np.fft.fft(w, size)
        A_rev_fft = np.fft.fft(a_rev, size)
        g_conv = np.fft.ifft(W_fft * A_rev_fft).real
        grad = 2.0 * g_conv[n - 1: n - 1 + n]   # gradient of smooth max

        # -------- exponentiated‑gradient update ----------
        # a_i ← a_i·exp(-η·gradient_i)  then renormalise
        a = a * np.exp(-eta * grad)
        a = np.maximum(a, 1e-30)
        a_sum = a.sum()
        a /= a_sum

        # -------- keep track of true objective ----------
        if (it & 63) == 0:            # evaluate every 64‑th iteration
            cur_score = _objective_norm1(a)
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                no_improve = 0
                print(f"[smooth] iter {it:6d}  new best O={best_score:.6f}")
            else:
                no_improve += 1
                if patience and no_improve >= patience:
                    # small random jitter to escape flat plateaus
                    jitter = np.random.randn(n) * 1e-4
                    a = best_a + jitter
                    a = np.clip(a, 0.0, None)
                    a = _project_to_simplex(a)
                    no_improve = 0

        # -------- schedule updates ----------
        lam = min(lam * 1.025, lam_max)            # slowly tighten the max
        eta = max(eta * 0.995, eta_min)           # slowly decrease step size

    return best_a, best_score
