"""
Proposes a non‑negative step‑function sequence that minimizes 
   O(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

Algorithm:
1️⃣ Multiple short random restarts with a few Adam steps to find a promising
   region of the simplex.
2️⃣ Long Adam optimisation on a smooth log‑sum‑exp surrogate of the max,
   with simplex projection.
3️⃣ Projected sub‑gradient descent with a back‑tracking line search.
4️⃣ Simulated‑annealing style mass‑transfer moves that target the shift
   currently attaining the maximal convolution value.

All phases respect a 1000 s wall‑time budget and keep the best
candidate seen.
"""

import time, math, numpy as np

# ------------------- Helper functions ---------------------------------

def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # all non‑positive -> uniform simplex
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution a*a using FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    # first 2*len(a)-1 entries contain the linear convolution
    return conv_full[: 2 * a.size - 1]


def _objective(a: np.ndarray, n: int, L: int) -> float:
    """Exact objective 2·n·max(conv) for a that already sums to 1."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. a (see reference implementation)."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


def _smooth_gradient(a: np.ndarray, conv: np.ndarray, alpha: float, L: int) -> np.ndarray:
    """
    Gradient of (2·n/α)·log Σ_k exp(α·conv_k) w.r.t. a.
    The result is 2·corr(w, a) where w_k ∝ exp(α·conv_k).
    """
    # soft‑max weights
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)      # stabilise
    w = exp_scaled / exp_scaled.sum()

    # correlation w ⋆ a  (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n_len = a.size
    grad = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]
    return grad


# ------------------- Optimisation phases ------------------------------

def _adam_phase(a: np.ndarray, n: int, L: int,
                max_iters: int, deadline: float,
                alpha_start: float = 5.0,
                alpha_end: float = 20000.0,
                lr0: float = 0.6,
                beta1: float = 0.9,
                beta2: float = 0.999,
                eps: float = 1e-12,
                prune_every: int = 5000,
                prune_percentile: float = 1.0) -> tuple[np.ndarray, float]:
    """Adam on the smoothed surrogate; returns (a, best_obj)."""
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_obj = _objective(a, n, L)

    # pre‑compute schedule length
    total_iters = max_iters

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            # print occasional improvement
            if it % 5000 == 0:
                print(f"[adam] iter {it} new best O={best_obj:.6f}")

        # α schedule – exponential to concentrate early
        frac = it / total_iters
        alpha = alpha_start * (alpha_end / alpha_start) ** frac

        grad = _smooth_gradient(a, conv, alpha, L)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        lr = lr0 / math.sqrt(it + 1)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # small random jitter
        if np.random.rand() < 0.001:
            a_new += np.random.randn(n) * 1e-4

        # simplex projection
        a_new = _project_to_simplex(a_new)

        # occasional pruning of tiny entries
        if prune_every and it % prune_every == 0 and it > 0:
            thresh = np.percentile(a_new, prune_percentile)
            a_new[a_new < thresh] = 0.0
            a_new = _project_to_simplex(a_new)

        a = a_new

    return a, best_obj, best_a


def _subgradient_phase(a: np.ndarray, n: int, L: int,
                       max_iters: int, deadline: float,
                       eta0: float = 0.3,
                       line_search: bool = True,
                       verbose: bool = False) -> tuple[np.ndarray, float, np.ndarray]:
    """Projected sub‑gradient descent with optional back‑tracking line search."""
    best_a = a.copy()
    best_obj = _objective(a, n, L)
    cur_obj = best_obj
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            if verbose and it % 5000 == 0:
                print(f"[sub] iter {it} new best O={best_obj:.6f}")

        g = _subgradient_max_conv(a, conv)
        grad_norm_sq = np.dot(g, g) + 1e-24

        # Simple geometric decay step size
        eta = eta0 * (0.9999 ** it)

        if line_search:
            # try to find a step that reduces the objective
            step = eta
            for _ in range(5):
                a_candidate = _project_to_simplex(a - step * g)
                cand_obj = _objective(a_candidate, n, L)
                if cand_obj < cur_obj:
                    a = a_candidate
                    cur_obj = cand_obj
                    break
                step *= 0.5
            else:
                # no improvement – take the geometric step anyway
                a = _project_to_simplex(a - eta * g)
        else:
            a = _project_to_simplex(a - eta * g)

    return a, best_obj, best_a


def _simulated_annealing_phase(a: np.ndarray, n: int, L: int,
                               max_iters: int, deadline: float,
                               temperature0: float = 0.05,
                               no_improve_limit: int = 8000,
                               verbose: bool = True) -> tuple[np.ndarray, float, np.ndarray]:
    """Local refinement using random mass‑transfer moves and occasional sub‑grad steps."""
    best_a = a.copy()
    best_obj = _objective(a, n, L)
    cur_obj = best_obj
    conv = _conv_fft(a, L)

    T = temperature0
    no_improve = 0

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # ----- pick a shift with maximal convolution -----
        kmax = int(np.argmax(conv))

        # Choose a source index i that participates in this shift
        i_candidates = np.arange(max(0, kmax - (n - 1)), min(n - 1, kmax) + 1)
        # keep only indices with non‑negligible mass
        i_candidates = i_candidates[a[i_candidates] > 1e-14]
        if i_candidates.size == 0:
            # fallback to a completely random move
            i = np.random.randint(n)
        else:
            i = np.random.choice(i_candidates)

        # destination index j (different from i)
        j = np.random.randint(n)
        while j == i:
            j = np.random.randint(n)

        # move a fraction from i to j
        delta = a[i] * np.random.rand() * 0.09  # up to 9 %
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = _project_to_simplex(a_new)

        conv_new = _conv_fft(a_new, L)
        new_obj = 2.0 * n * conv_new.max()

        # Metropolis acceptance
        if new_obj < cur_obj or np.random.rand() < math.exp((cur_obj - new_obj) / max(T, 1e-12)):
            a = a_new
            conv = conv_new
            cur_obj = new_obj
            no_improve = 0
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                if verbose:
                    print(f"[{time.time():.1f}] new global best O={best_obj:.6f} (iter={it})")
        else:
            no_improve += 1

        # occasional exact sub‑gradient step to keep direction
        if it % 9 == 0:
            g = _subgradient_max_conv(a, conv)
            # linearly decay the step size during SA
            eta_loc = 0.07 * (1.0 - it / max_iters)
            a = _project_to_simplex(a - eta_loc * g)
            conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                if verbose:
                    print(f"[{time.time():.1f}] sub‑grad improved O={best_obj:.6f}")

        # temperature decay
        T *= 0.9997

        # restart if stuck
        if no_improve >= no_improve_limit:
            a = best_a.copy()
            a += np.random.randn(n) * 0.0015
            a = np.clip(a, 0, None)
            a = _project_to_simplex(a)
            conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max()
            no_improve = 0

    return a, best_obj, best_a

    if rho.size == 0:                      # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear self‑convolution via FFT
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return a*a (full convolution) using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Exact objective for a simplex‑normalised vector (∑a = 1)
def _objective(a: np.ndarray, n: int, L: int) -> float:
    """2·n·max(conv(a,a))  (a is assumed to sum to 1)."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Sub‑gradient of the maximal convolution entry
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. a."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
# Gradient of the log‑sum‑exp (smooth max) surrogate
def _smooth_gradient(a: np.ndarray, conv: np.ndarray,
                     alpha: float, L: int) -> np.ndarray:
    """
    Compute ∇_a ( (2·n/α)·log∑_k e^{α·conv_k} ).
    The formula reduces to 2·corr(w, a) where w_k ∝ e^{α·conv_k}.
    """
    # soft‑max weights
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)      # stabilise
    w = exp_scaled / exp_scaled.sum()

    # correlation w ⋆ a   (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)   # length ≥ 3·n‑2
    n_len = a.size
    grad = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the smooth surrogate
def _adam_phase(a: np.ndarray, n: int, L: int,
                max_iters: int, deadline: float,
                alpha_start: float = 5.0,
                alpha_end: float = 20000.0,
                lr0: float = 0.6,
                beta1: float = 0.9,
                beta2: float = 0.999,
                eps: float = 1e-12,
                prune_every: int = 5000,
                prune_percentile: float = 1.0) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Adam on the soft‑max surrogate.
    Returns (final a, best objective value seen, best a).
    """
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_obj = _objective(a, n, L)

    total_iters = max_iters
    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[adam] iter {it} new best O={best_obj:.6f}")

        # exponential schedule for α – sharper towards the end
        frac = it / total_iters
        alpha = alpha_start * (alpha_end / alpha_start) ** frac

        grad = _smooth_gradient(a, conv, alpha, L)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        lr = lr0 / math.sqrt(it + 1)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape flat plateaus
        if np.random.rand() < 0.001:
            a_new += np.random.randn(n) * 1e-4

        a_new = _project_to_simplex(a_new)

        # prune tiny entries every few thousand steps
        if prune_every and it % prune_every == 0 and it > 0:
            thresh = np.percentile(a_new, prune_percentile)
            a_new[a_new < thresh] = 0.0
            a_new = _project_to_simplex(a_new)

        a = a_new

    return a, best_obj, best_a


# ----------------------------------------------------------------------
# Exact sub‑gradient descent with simple back‑tracking line search
def _subgradient_phase(a: np.ndarray, n: int, L: int,
                       max_iters: int, deadline: float,
                       eta0: float = 0.3,
                       line_search: bool = True,
                       verbose: bool = False) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Projected sub‑gradient descent on the true objective.
    Returns (final a, best objective, best a).
    """
    best_a = a.copy()
    best_obj = _objective(a, n, L)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _conv_fft(a, L)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            if verbose and it % 5000 == 0:
                print(f"[sub‑grad] iter {it} new best O={best_obj:.6f}")

        g = _subgradient_max_conv(a, conv)

        # geometric decay of the step size
        eta = eta0 * (0.9999 ** it)

        if line_search:
            step = eta
            # at most five back‑tracking trials
            for _ in range(5):
                cand = _project_to_simplex(a - step * g)
                cand_obj = _objective(cand, n, L)
                if cand_obj < cur_obj:
                    a = cand
                    break
                step *= 0.5
            else:
                a = _project_to_simplex(a - eta * g)
        else:
            a = _project_to_simplex(a - eta * g)

    return a, best_obj, best_a


# ----------------------------------------------------------------------
# Simulated annealing with mass‑transfer moves that target the current
# maximal convolution entry.
def _simulated_annealing_phase(a: np.ndarray, n: int, L: int,
                               max_iters: int, deadline: float,
                               temperature0: float = 0.05,
                               no_improve_limit: int = 8000,
                               verbose: bool = True) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Local refinement: move a fraction of mass from an index that
    participates in the current maximal convolution entry to a random
    destination; accept with Metropolis rule.  Occasional exact
    sub‑gradient steps keep the walk directed.
    """
    best_a = a.copy()
    best_obj = _objective(a, n, L)
    cur_obj = best_obj
    conv = _conv_fft(a, L)

    T = temperature0
    no_improve = 0

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # index of the current peak of the convolution
        kmax = int(np.argmax(conv))

        # indices that actually contribute to that peak
        i_candidates = np.arange(max(0, kmax - (n - 1)), min(n - 1, kmax) + 1)
        i_candidates = i_candidates[a[i_candidates] > 1e-14]
        if i_candidates.size == 0:
            i = np.random.randint(n)
        else:
            i = np.random.choice(i_candidates)

        # random destination
        j = np.random.randint(n)
        while j == i:
            j = np.random.randint(n)

        # transfer a modest fraction of the source mass
        delta = a[i] * np.random.rand() * 0.09
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = _project_to_simplex(a_new)

        conv_new = _conv_fft(a_new, L)
        new_obj = 2.0 * n * conv_new.max()

        # Metropolis acceptance
        if new_obj < cur_obj or np.random.rand() < math.exp((cur_obj - new_obj) / max(T, 1e-12)):
            a = a_new
            conv = conv_new
            cur_obj = new_obj
            no_improve = 0
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                if verbose:
                    print(f"[{time.time():.1f}] new global best O={best_obj:.6f} (iter={it})")
        else:
            no_improve += 1

        # occasional exact sub‑gradient step (keeps direction)
        if it % 9 == 0:
            g = _subgradient_max_conv(a, conv)
            eta_loc = 0.07 * (1.0 - it / max_iters)
            a = _project_to_simplex(a - eta_loc * g)
            conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                if verbose:
                    print(f"[{time.time():.1f}] sub‑grad improved O={best_obj:.6f}")

        # temperature decay
        T *= 0.9997

        # restart if stagnating for a while
        if no_improve >= no_improve_limit:
            a = best_a.copy()
            a += np.random.randn(n) * 0.0015
            a = np.clip(a, 0, None)
            a = _project_to_simplex(a)
            conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max()
            no_improve = 0

    return a, best_obj, best_a

# Main entry point expected by the evaluation harness
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence that minimises the evaluation function.
    Returns the sequence as a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin – we stop a little before the deadline
    deadline = start_time + budget_s - 0.2

    # -------------------- initialise problem size --------------------
    if 'height_sequence_1' in globals():
        a0 = np.array(height_sequence_1, dtype=float)
        a0 = np.maximum(a0, 0.0)
        if a0.sum() == 0.0:
            a0 = np.full_like(a0, 1.0)
        a0 = a0 / a0.sum()
        n = a0.size
        a = a0.copy()
        print(f"[init] using supplied sequence, n={n}")
    else:
        n = 4600
        a = np.full(n, 1.0 / n)   # uniform start on the simplex
        print(f"[init] using uniform start, n={n}")

    # FFT length – next power of two greater than 3·n‑2
    L = 1 << ((3 * n - 2).bit_length())

    # -----------------------------------------------------------------
    # Phase 0 – cheap random restarts to obtain a good seed
    phase0_deadline = start_time + budget_s * 0.15
    best_seq = a.copy()
    best_score = _objective(best_seq, n, L)

    while time.time() < phase0_deadline:
        conc = 0.4                     # Dirichlet concentration (<1 ⇒ sparse)
        a_rand = np.random.gamma(conc, 1.0, size=n)
        a_rand = a_rand / a_rand.sum()
        # a few Adam steps (fast)
        sub_dead = time.time() + 4.0   # limit each trial to a few seconds
        a_rand, _, best_rand = _adam_phase(a_rand, n, L,
                                           max_iters=1500,
                                           deadline=sub_dead,
                                           alpha_start=5.0,
                                           alpha_end=2000.0,
                                           lr0=0.8)
        cur_obj = _objective(best_rand, n, L)
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = best_rand.copy()
            print(f"[phase0] new best O={best_score:.6f}")

    # -----------------------------------------------------------------
    # Phase 1 – long Adam optimisation on the smooth surrogate
    phase1_deadline = start_time + budget_s * 0.45
    a, obj_after_adam, best_adam = _adam_phase(best_seq, n, L,
                                               max_iters=500_000,
                                               deadline=phase1_deadline,
                                               alpha_start=5.0,
                                               alpha_end=50_000.0,
                                               lr0=0.6)
    # start from the best point seen in phase 1
    a = best_adam
    if obj_after_adam < best_score:
        best_score = obj_after_adam
        best_seq = best_adam.copy()
        print(f"[phase‑1] improved best O={best_score:.6f}")

    # -----------------------------------------------------------------
    # Phase 2 – exact sub‑gradient descent with line search
    phase2_deadline = start_time + budget_s * 0.75
    a, obj_sub, best_sub = _subgradient_phase(a, n, L,
                                              max_iters=300_000,
                                              deadline=phase2_deadline,
                                              eta0=0.25,
                                              line_search=True,
                                              verbose=False)
    if obj_sub < best_score:
        best_score = obj_sub
        best_seq = best_sub.copy()
        print(f"[phase‑2] improved best O={best_score:.6f}")

    # -----------------------------------------------------------------
    # Phase 3 – stochastic annealing refinement
    phase3_deadline = deadline
    a, obj_sa, best_sa = _simulated_annealing_phase(best_seq, n, L,
                                                    max_iters=300_000,
                                                    deadline=phase3_deadline,
                                                    temperature0=0.07,
                                                    no_improve_limit=7000,
                                                    verbose=True)
    if obj_sa < best_score:
        best_score = obj_sa
        best_seq = best_sa.copy()
        print(f"[phase‑3] final best O={best_score:.6f}")

    # -----------------------------------------------------------------
    # Final verification through the provided evaluator
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence failed:", exc)

    return best_seq.tolist()
