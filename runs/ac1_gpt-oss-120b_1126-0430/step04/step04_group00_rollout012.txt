def propose_candidate(seed=42, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0

    # initial sizes
    init_n = 8000   # start moderate
    max_n = 60000   # final support (choose within time)
    # Starting distributions
    candidates = []
    # Uniform
    candidates.append(_uniform_distribution(init_n))
    # Arcsine
    candidates.append(_arcsine_distribution(init_n))
    # If height_sequence_1 exists, use its normalized version truncated/padded to init_n
    try:
        global height_sequence_1
        hs = np.array(height_sequence_1, dtype=np.float64)
        if hs.size < init_n:
            hs = _interpolate_distribution(hs, init_n)
        else:
            hs = hs[:init_n]
        hs = np.maximum(hs,0.0)
        if hs.sum() == 0.0:
            hs = _uniform_distribution(init_n)
        else:
            hs /= hs.sum()
        candidates.append(hs)
    except Exception:
        pass

    best_seq = None
    best_score = float('inf')
    best_z = None
    best_n = init_n

    # Evaluate cheap scores for each candidate and use best to start
    for a0 in candidates:
        max_conv = _conv_max(a0)
        cheap = 2.0*init_n*max_conv
        print(f"Initial candidate cheap score {cheap:.6f}")
        if cheap < best_score:
            best_score = cheap
            best_seq = a0.copy()
            best_z = _logits_from_distribution(a0) + 0.01*np.random.randn(init_n)  # small jitter
            best_n = init_n

    n = best_n
    a = best_seq
    z = best_z

    # Adam state
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_alpha = 0.4 # adapt later

    # smoothing factor schedule
    eps_factor = 0.15
    eps_factor_min = 5e-4
    eps_factor_decay = 0.9997

    iteration = 0
    last_improve_iter = 0
    report_interval = 5000
    expand_interval = 20000
    # largest allowed support
    max_support = max_n

    while time.time() < deadline:
        iteration += 1

        # softmax compute a
        a = _softmax(z)

        # cheap conv and gradient
        max_conv, grad_a = _conv_max_and_grad(a, eps_factor)
        cheap = 2.0*n*max_conv

        # update best
        if cheap < best_score:
            best_score = cheap
            best_seq = a.copy()
            last_improve_iter = iteration
            if iteration % report_interval == 0:
                print(f"[Iter {iteration:07d}] New best cheap={cheap:.6f}")

        # gradient vs logits
        mu = np.dot(grad_a, a)
        grad_z = a * (grad_a - mu)

        # Adam moments
        m = beta1*m + (1-beta1)*grad_z
        v = beta2*v + (1-beta2)*(grad_z**2)
        m_hat = m/(1-beta1**iteration)
        v_hat = v/(1-beta2**iteration)

        # step size
        alpha_t = base_alpha/(math.sqrt(iteration) + 1.0)
        step_dir = - m_hat/(np.sqrt(v_hat) + eps_adam)

        # backtracking line search
        accepted = False
        trial_alpha = alpha_t
        for _ in range(8):
            z_new = z + trial_alpha*step_dir
            z_new = np.clip(z_new, -80.0, 80.0)   # safe exp
            a_new = _softmax(z_new)
            new_max = _conv_max(a_new)
            new_score = 2.0*n*new_max
            if new_score <= cheap:
                z = z_new
                accepted = True
                break
            trial_alpha *= 0.5
        if not accepted:
            # Take step anyway (but with limited size)
            z = z + alpha_t*step_dir
            z = np.clip(z, -80.0, 80.0)

        # occasional random jump
        if iteration % 4000 == 0 and np.random.rand() < 0.2:
            z += 0.01*np.random.randn(n)
            m *= 0.5
            v *= 0.5

        # decay epsilon factor
        eps_factor = max(eps_factor_min, eps_factor*eps_factor_decay)

        # support expansion
        if iteration % expand_interval == 0 and n < max_support and (time.time() + 30) < deadline:
            new_n = min(max_support, int(n*1.5)+1)
            print(f"[Iter {iteration:07d}] Expanding support: {n} -> {new_n}")
            # interpolate best_seq
            a_interp = _interpolate_distribution(best_seq, new_n)
            # reset logits and Adam
            z = _logits_from_distribution(a_interp) + 0.01*np.random.randn(new_n)
            m = np.zeros_like(z)
            v = np.zeros_like(z)
            n = new_n
            # recompute best score for new size
            best_seq = a_interp.copy()
            best_score = 2.0*n*_conv_max(best_seq)
            iteration = 0
            last_improve_iter = 0
            eps_factor = 0.15
            continue

        # early exit if no improvement for many iterations and at max support
        if n >= max_support and iteration - last_improve_iter > 400_000:
            print(f"[Iter {iteration:07d}] No improvement for long; terminating")
            break

    # final evaluation
    final_seq = best_seq.tolist()
    try:

        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:
        final_score = float('inf')
        print(f"Error in final evaluation: {exc}")

def _conv_max_and_grad(a, eps_factor):
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[:2*n-1]
    max_conv = conv.max()
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0 or np.isnan(w_sum):
        w = np.full_like(w, 1.0/w.size)
    else:
        w /= w_sum
    # correlation of a and w (length 3n-2)
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]
    grad_a = 4.0 * n * corr
    return float(max_conv), grad_a

def _arcsine_distribution(n):
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    a_sum = a.sum()
    if a_sum == 0.0:
        a[:] = 1.0/n
    else:
        a /= a_sum
    return a

"""
Gradient‑descent optimisation of a non‑negative step‑function.

The objective (after normalising the sum to 1) is
    J(a) = 2·n·max_k (a*a)[k]               (a*a = linear convolution)

We minimise J by optimisation on the probability simplex using
an entropic parametrisation a = softmax(z).  The gradient of a
smooth approximation of the max‑operator is obtained via the
soft‑max weights of the self‑convolution.  Adam is used as optimiser,
together with a simple back‑tracking line‑search that guarantees a
non‑increasing cheap score 2·n·max(conv).  The support size is
progressively enlarged (up to max_n) by linear interpolation of the
current best distribution.  Several initialisations (uniform,
arcsine‑like, and a possible historic solution) are tried and the best
is continued.

The routine respects a ~1000 s budget and finally evaluates the best
candidate with the provided ``evaluate_sequence`` function.
"""

import time
import math
import numpy as np

# Helper utilities (must be top‑level – no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max; returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it representation that yields `a` after soft‑max."""
    return np.log(a + eps)


def _fft_len_for_corr(n: int) -> int:
    """
    Smallest power‑of‑two ≥ 3·n‑2,
    enough for a linear convolution of length (2·n‑1) and a correlation of
    length (3·n‑2).
    """
    return 1 << ((3 * n - 2).bit_length())


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    total = new_a.sum()
    if total == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= total
    return new_a


def _uniform_distribution(n: int) -> np.ndarray:
    """Flat distribution (sum = 1)."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine_distribution(n: int) -> np.ndarray:
    """
    Discrete arcsine‑like law:
        a_i ∝ 1 / sqrt((i+0.5)·(n‑0.5‑i)),
    normalised to sum = 1.
    """
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _conv_max_and_grad(a: np.ndarray, eps_factor: float) -> tuple[float, np.ndarray]:
    """
    Compute the max entry of a*a and the gradient of the smoothed
    objective w.r.t. a (already multiplied by 4·n).

    Parameters
    ----------
    a : np.ndarray
        Probability vector (sum = 1, non‑negative).
    eps_factor : float
        Multiplicative factor for the smoothing constant:
        eps = eps_factor·max(conv).

    Returns
    -------
    max_conv : float
        Maximum of the self‑convolution a*a.
    grad_a : np.ndarray
        Gradient of `2·n·smooth_max` w.r.t. a.
    """
    n = a.shape[0]
    L = _fft_len_for_corr(n)

    # FFT of a (zero‑padded)
    a_fft = np.fft.rfft(a, L)

    # ---- self‑convolution a*a (linear, length 2n‑1) ----
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    max_conv = float(conv.max())

    # ---- smoothing weights (soft‑max over conv) ----
    eps = max(max_conv * eps_factor, 1e-12)
    shifted = conv - max_conv                    # stabilise exp
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0 or np.isnan(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # ---- cross‑correlation of a and w (length 3n‑2) ----
    w_fft = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(a_fft) * w_fft, L)[:n]

    # gradient of the smoothed objective (4·n·corr)
    grad_a = 4.0 * n * corr
    return max_conv, grad_a


def _conv_max(a: np.ndarray) -> float:
    """Return max entry of a*a (no gradient)."""
    n = a.shape[0]
    L = _fft_len_for_corr(n)
    a_fft = np.fft.rfft(a, L)
    conv = np.fft.irfft(a_fft * a_fft, L)[: 2 * n - 1]
    return float(conv.max())
