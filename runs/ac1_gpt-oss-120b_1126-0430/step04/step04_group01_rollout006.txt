"""
Propose a step‑function (non‑negative list) that (approximately) minimises

    O(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The optimiser works in three stages (all respect the 1000 s budget).

* Stage 0 – initialise a probability vector (∑a=1).
* Stage 1 – minimise an L2‑distance to a *flat* convolution
            (target = constant 1/(2 n‑1)).  
  This pushes the autocorrelation close to the theoretical lower bound.
* Stage 2 – switch to the smooth log‑sum‑exp surrogate used previously,
            gradually tightening the approximation of the true max.
* Stage 3 – stochastic local refinement (mass‑transfer moves + sub‑gradient
            steps) to polish the final solution.

All updates keep the vector on the probability simplex via Euclidean
projection.  The routine returns the best sequence found (as a plain list).
"""

import time
import math
import numpy as np

# --------------------------------------------------------------
# Euclidean projection onto the probability simplex {x ≥ 0, sum(x)=1}.
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
# Full linear convolution a*a (length 2·n‑1) via FFT.
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return the full self‑convolution of a (length 2·n‑1)."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# --------------------------------------------------------------
# Sub‑gradient of max(conv) (averaged over all maximisers).
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. a (averaged over all maximisers)."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


# --------------------------------------------------------------
# Gradient of the smooth surrogate (log‑sum‑exp) plus an entropy term.
def _smooth_gradient_plus_entropy(
    a: np.ndarray,
    conv: np.ndarray,
    alpha: float,
    L: int,
    lam: float,
) -> np.ndarray:
    """
    ∇_a ( (2·n/α)·log∑_k e^{α·conv_k} + λ·∑ a_i·log a_i ).
    The first part reduces to 2·corr(w,a) with w_k ∝ e^{α·conv_k}.
    """
    # soft‑max weights
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)          # numerical stability
    w = exp_scaled / exp_scaled.sum()                 # shape (2n‑1,)

    # correlation w ⋆ a (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n_len = a.size
    grad_corr = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]

    # entropy part
    eps = 1e-20
    grad_ent = lam * (np.log(a + eps) + 1.0)

    return grad_corr + grad_ent


# --------------------------------------------------------------
# L2 loss (‖conv(a) – const‖²) and its gradient.
def _l2_loss_and_grad(
    a: np.ndarray,
    n: int,
    L: int,
    target: np.ndarray,
) -> tuple[float, np.ndarray]:
    """
    Returns loss = Σ (conv_k – target_k)²
    and gradient = 4·( (conv‑target) ⋆ a )  (⋆ = correlation).
    """
    conv = _conv_fft(a, L)                 # length 2n‑1
    diff = conv - target
    loss = np.sum(diff * diff)

    # correlation diff ⋆ a  (valid length n)
    a_rev = a[::-1]
    fft_diff = np.fft.rfft(diff, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_full = np.fft.irfft(fft_diff * fft_a_rev, n=L)
    # slice the “valid” part: indices n‑1 … n‑1+n‑1
    grad = 4.0 * corr_full[n - 1 : n - 1 + n]
    return loss, grad
