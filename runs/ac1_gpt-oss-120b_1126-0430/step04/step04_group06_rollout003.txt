def propose_candidate(seed=42, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.

    n = 30000
    # Initialize a: power law p=0.6 or maybe start with smooth shape
    a = _power_law_distribution(n, 0.5) # but maybe use other initial shape

    # optionally random perturbation.
    # Use logit representation
    eps = 1e-16
    z = _logits_from_distribution(a, eps)

    # Adam state
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    # schedule for theta
    theta_start = 0.5
    theta_end = 30.0

    # hyperparameters
    base_alpha = 0.3
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    best_seq = a.copy()
    best_score = float('inf')
    it = 0

    while time.time() < deadline:
        it += 1
        # Softmax to get a
        a = _softmax(z)

        # compute conv a*a
        fft_len = _fft_len_for_n(n)
        conv = _conv_full_fft(a, fft_len)  # length 2n-1

        # Compute cheap bound
        cheap_score = 2.0 * n * float(conv.max())
        if cheap_score < best_score:
            best_score = cheap_score
            best_seq = a.copy()
            print(f"[Iter {it}] new best cheap={cheap_score:.6f}")

        # Compute smooth max loss
        maxc = conv.max()
        # compute exponent weights
        # avoid overflow: subtract maxc
        exp_term = np.exp(theta * (conv - maxc))  # note: may overflow with large theta; but conv values ~ O(??). maxc ~ something ~maybe ~1/n? For n large ~0.07? Not huge; but theta up to 30, exp(30*diff). diff can be negative upto maybe -1? So exp -30. okay.
        Z = exp_term.sum()
        # gradient w.r.t. conv: p = exp_term / Z
        p = exp_term / Z

        # compute gradient wrt a: g_a = 2 * (p * a_rev) convolution
        a_rev = a[::-1]  # reversed array

        L = 1 << ((len(p) + len(a_rev) - 1).bit_length())

        frev = np.fft.rfft(a_rev, L)

        mu = float(np.dot(g_a, a))

def _smooth_max_and_gradient(a, theta):
    n = a.shape[0]
    fft_len = _fft_len_for_n(n)   # for convolution a*a
    conv = _conv_full_fft(a, fft_len)   # result len 2n-1

"""
Propose a high‑quality non‑negative step‑function sequence.

The method is a mirror‑descent (Adam) optimiser on the probability simplex.
We work directly with the smooth approximation of the max‑convolution
(`log‑sum‑exp`), whose gradient is obtained via an FFT‑based correlation.
The logits are kept symmetric (z[i] = z[n-1-i]), which reduces the search
space and empirically yields lower convolution peaks.  A simple schedule
increases the smoothing temperature and we inject occasional random
perturbations to escape plateaus.  The best candidate seen during the
budget is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top‑level, no closures
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max; returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it representation that yields `a` after soft‑max."""
    return np.log(a + eps)


def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two ≥ 2·n‑1 (efficient size for real FFT)."""
    return 1 << ((2 * n - 1).bit_length())


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full linear self‑convolution a*a using an FFT of length `fft_len`.
    Returns an array of length 2·len(a)‑1.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.size - 1]
    return conv


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """Normalised power‑law a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _smooth_max_gradient(a: np.ndarray, theta: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Compute the full self‑convolution ``c = a*a``, the smooth max weight
    vector ``p`` (softmax with temperature theta) and the gradient
    d/d a of the smoothed max.

    Returns
    -------
    conv : np.ndarray
        Full convolution (length 2·n‑1).
    grad : np.ndarray
        Gradient vector of shape (n,).
    """
    n = a.shape[0]

    # ---------- convolution a*a ----------
    fft_len = _fft_len_for_n(n)
    conv = _conv_full_fft(a, fft_len)          # length 2n-1

    # ---------- smooth max (log‑sum‑exp) ----------
    max_c = conv.max()
    # shift by max_c to keep exponentials in [0,1]
    exp_term = np.exp(theta * (conv - max_c))
    Z = exp_term.sum()
    p = exp_term / Z                           # weights, sum(p)=1

    # ---------- gradient via correlation ----------
    # correlation(p, a) = ifft( conj(fft(p)) * fft(a) )
    L = 1 << ((len(p) + n - 1).bit_length())   # power‑of‑two >= 3n‑2
    P = np.fft.rfft(p, L)
    A = np.fft.rfft(a, L)
    corr = np.fft.irfft(np.conj(P) * A, L)      # length L, real

    center = len(p) - 1                         # index of lag 0
    idx = center - np.arange(n)                 # indices for lags 0,‑1,…,‑(n‑1)
    grad = 2.0 * corr[idx]                      # shape (n,)

    return conv, grad
