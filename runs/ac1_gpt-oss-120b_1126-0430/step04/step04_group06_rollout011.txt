"""
Propose a step‑function (non‑negative heights) that minimizes the evaluation
function.  The algorithm is a mirror‑descent (Adam) optimiser on the
probability simplex using an exact smoothed maximum of the self‑convolution.
The smoothed max is obtained with a softmax over *all* convolution entries,
which yields a more accurate gradient than the previous “top‑K” heuristic.
A linear temperature schedule slowly sharpens the softmax, allowing the
optimizer to escape plateaus before focusing on the true maximum.
The implementation respects a 1000 s time budget and returns the best
sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures, top‑level only)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max; returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it representation that yields `a` after soft‑max (inverse of softmax)."""
    return np.log(a + eps)


def _fft_len_for_n(n: int) -> int:
    """Return the smallest power‑of‑two ≥ n (efficient length for real FFT)."""
    return 1 << ((n - 1).bit_length())


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full linear self‑convolution a*a using an FFT of length `fft_len`.
    Returns an array of length 2·len(a)‑1.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.size - 1]
    return conv


def _corr_full_fft(a: np.ndarray, b: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full linear cross‑correlation (a ⋆ b) using an FFT of length `fft_len`.
    The result length is len(a)+len(b)-1.  For lag `ℓ` (possibly negative)
    we have (a ⋆ b)[ℓ] = Σ_i a_i·b_{i+ℓ}.
    """
    Fa = np.fft.rfft(a, fft_len)
    Fb = np.fft.rfft(b, fft_len)
    corr = np.fft.irfft(Fa * np.conj(Fb), fft_len)
    return corr


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """Return a normalised power‑law a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Optimise the step‑function using Adam on the simplex with an exact
    smoothed‑max gradient.  The routine respects the given time budget
    (≈1000 s) and returns the best sequence it discovered.
    """
    np.random.seed(seed)
    start = time.time()
    # leave a small safety margin for the final evaluation call
    deadline = start + budget_s - 2.0

    # ---------------------------------------------------------------
    # 0. Choose a working support size.
    # ---------------------------------------------------------------
    global_seq = globals().get("height_sequence_1", None)
    if global_seq is not None:
        init = np.array(global_seq, dtype=np.float64)
        n = max(20000, init.shape[0])
        print(f"[Init] Using supplied global sequence, length = {n}")
        use_global = True
    else:
        init = None
        n = 30000                     # default large support (good trade‑off)
        use_global = False
        print(f"[Init] No global sequence – starting with length = {n}")

    # ---------------------------------------------------------------
    # 1. Initialise a probability vector (sum = 1).
    # ---------------------------------------------------------------
    if use_global and init is not None:
        a = np.maximum(init, 0.0)
        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / a.shape[0])
        else:
            a /= total
    else:

        a = _power_law_distribution(n, 0.60)
        print("[Init] Power‑law start with exponent p=0.60")

    # ---------------------------------------------------------------
    # 2. Log‑it parametrisation + Adam state.
    # ---------------------------------------------------------------
    eps = 1e-16
    z = _logits_from_distribution(a, eps)

    v = np.zeros_like(z)          # second‑moment
    beta1 = 0.9
    beta2 = 0.999

    base_alpha = 0.5               # base learning‑rate (larger than before)

    # ---------------------------------------------------------------
    best_seq = a.copy()
    best_score = float("inf")
    it = 0
    report_every = 2000

    # temperature schedule for the softmax (smooth max)
    theta_start = 0.1
    theta_end = 20.0
    theta_slope = (theta_end - theta_start) / max(1.0, budget_s - 10.0)

    # Pre‑compute FFT lengths that will be reused many times.
    # Convolution length: 2*n-1
    conv_fft_len = _fft_len_for_n(2 * n - 1)
    # Correlation length: n + (2*n-1) - 1 = 3*n - 2
    corr_fft_len = _fft_len_for_n(3 * n - 2)

    while time.time() < deadline and it < 2_000_000:

        mu = float(np.dot(g, a))

                break

    a = best_seq.copy()
    z = _logits_from_distribution(a, eps)

        w /= w.sum()

        grad_z = a * (g - mu)
