Propose a non‑negative step function that minimises

    2·n·max_k (a*a)[k] / (Σ a_i)²

The algorithm combines three ideas:

1.  Initialise from several cheap constructions (previous best,
    uniform, power‑law, a few Dirichlet draws) and keep the best
    candidate under the cheap score 2·n·max(conv).

2.  Iteratively improve the candidate with a *soft‑max* surrogate
    for the max‑operator:
        f_τ(a) = τ·log(∑_k exp(conv_k/τ))

    The gradient of f_τ can be written in closed form and is
    evaluated via two FFT‑based convolutions per iteration.
    An exponentiated‑gradient (mirror‑descent) step
    a ← a·exp(−η·∇f_τ) followed by renormalisation keeps the
    iterate on the probability simplex (Σ a_i = 1, a_i ≥ 0).

3.  Simple exploration: every few‑tens of thousands of iterations a
    small random multiplicative perturbation of the current best is
    tried; if it improves the cheap score it is accepted.

The temperature τ is chosen as a small fraction of the current
maximum convolution value (τ = τ_factor·max_b) and the step size η
decreases like 1/√t.  The whole loop respects the supplied time
budget (default 1000 s) and finally calls the provided
``evaluate_sequence`` for the exact score.  The returned list is
guaranteed to be feasible (non‑negative, sum > 0).

The method is fast (≈ 2 FFT per iteration) and typically drives the

def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float) -> np.ndarray:
    """Return a normalised power‑law a_i ∝ (i+1)^‑p (Σa_i = 1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = i ** (-p)
    seq /= seq.sum()
    return seq


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full, length 2·n‑1) using FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa).real[: 2 * a.size - 1]
    return conv


def _conv_rev_grad(a: np.ndarray, w_rev: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute the gradient of the soft‑max surrogate.
    Returns g_i = 2 * Σ_k w_k * a_{k-i} for i=0..n‑1.
    Implements g = 2 * (a * w_rev) evaluated at positions (2n‑2‑i).
    """
    fa = np.fft.fft(a, fft_len)
    fw = np.fft.fft(w_rev, fft_len)
    conv = np.fft.ifft(fa * fw).real
    n = a.shape[0]
    # conv length = n + (2n-1) - 1 = 3n-2
    # slice indices (2n-2-i) give the needed values
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * conv[idx]
    return grad

        exp_arg = (conv - maxb) / tau
        # clip extreme negatives to keep np.exp stable
        exp_arg = np.clip(exp_arg, -700, 0)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smooth surrogate -----
        w_rev = w[::-1]
        grad = _conv_rev_grad(a, w_rev, grad_fft_len)

        # ----- exponentiated‑gradient update (mirror descent) -----
        eta = eta0 / math.sqrt(iter_cnt)
        a = a * np.exp(-eta * grad)
        # re‑normalise to the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # ----- occasional random perturbation if stuck -----
        if iter_cnt - last_improve > restart_gap:
            # small multiplicative noise centred at 1
            noise = 0.02 * np.random.randn(n)
            a = a * np.exp(noise)
            a = _simplex_projection(a)
            last_improve = iter_cnt
            print(f"[Iter {iter_cnt:07d}] Restart (perturb) after stagnation.")

        # ----- periodic pure‑random neighbour trials for exploration -----
        if iter_cnt % 20000 == 0:
            # generate a neighbour by a tiny random walk
            cand = best_iter_a * np.exp(0.01 * np.random.randn(n))
            cand /= cand.sum()
            cand_val = cheap_score(cand)
            if cand_val < best_iter_val:
                best_iter_val = cand_val
                best_iter_a = cand.copy()
                a = cand.copy()            # jump to the better neighbour
                last_improve = iter_cnt
                print(f"[Iter {iter_cnt:07d}] Random neighbour improved cheap = {cand_val:.6f}")

    # --------------------------------------------------------------
    # 2. Final exact evaluation (uses the provided ``evaluate_sequence``)
    # --------------------------------------------------------------
    final_seq = best_iter_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:      # defensive – should never happen
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    # --------------------------------------------------------------
    # 3. Return the best sequence found.
    # --------------------------------------------------------------
    return final_seq
