"""
Algorithm
---------

Searches for a non‑negative step function (a list of heights) that minimizes

        O = 2·n·max(conv(a,a)) / (sum(a)²)

The routine works under a fixed time budget (default 1000 s) and proceeds in
three phases for each chosen length *n* (powers of two up to 16384).

1. **Parametric sweep**  
   • monotone power‑law:      a_i ∝ (i+β)^(‑α)  
   • monotone power‑law with exponential tail: a_i ∝ (i+β)^(‑α)·exp(‑γ·i/n)  
   • symmetric product‑law:   a_i ∝ ((i+β)*(n+1‑i+β))^(‑α)  
   • symmetric product‑law with exponential tail.  

   A coarse grid over (α,β,γ) is evaluated with a fast FFT based objective.
   The best vector from the sweep becomes the seed for the next stage.

2. **Random local refinement**  
   Random perturbations of the best parameters (or of the vector itself) are
   generated and accepted if they improve the objective.  A few thousand
   trials are performed while respecting the remaining budget.

3. **Mirror‑descent polishing**  
   Starting from the best candidate, an exponentiated‑gradient (EG) descent
   minimises the true objective.  A sub‑gradient of the maximal convolution
   entry is used; all entries whose convolution value lies within a tiny
   tolerance of the maximum also contribute, which yields a smoother descent.
   The learning rate decays linearly, and occasional jitter prevents stagnation.

The algorithm is run for several lengths (4096, 8192, 16384) as long as time
allows; the globally best sequence is finally evaluated with the supplied
``evaluate_sequence`` and returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a power‑of‑two FFT (real output)."""
    n = a.size
    size = 1 << ((2 * n - 2).bit_length())   # next power of two ≥ 2n‑1
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def _objective_norm1(a: np.ndarray) -> float:
    """
    Objective 2·n·max(conv) under the convention Σa = 1.
    Returns a plain float.
    """
    n = a.size
    conv = _conv_fft(a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray,
                          rel_tol: float = 1e-9,
                          abs_tol: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) w.r.t. a on the simplex.
    All indices whose convolution value is within tolerance of the maximum
    are treated as active.
    """
    n = a.size
    conv = _conv_fft(a)
    max_val = conv.max()
    tol = max(abs_tol, rel_tol * max_val)
    ks = np.where(conv >= max_val - tol)[0]

    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
def _exp_update(a: np.ndarray, g: np.ndarray, eta: float) -> np.ndarray:
    """
    Standard exponentiated‑gradient (EG) update:
        a_i ← a_i·exp(‑η·g_i)   and then re‑normalise to sum‑1.
    """
    # use log‑space for extra stability when η is large
    log_a = np.log(a + 1e-15) - eta * g
    # shift to avoid overflow/underflow
    log_a -= np.max(log_a)
    a_new = np.exp(log_a)
    s = a_new.sum()
    if s == 0.0:
        a_new = np.full_like(a, 1.0 / a.size)
    else:
        a_new /= s
    return a_new


# ----------------------------------------------------------------------
def _optimize(a0: np.ndarray,
              max_iters: int,
              eta0: float,
              deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent polishing using exponentiated‑gradient.
    Returns the best vector seen and its objective value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective_norm1(a)

    n = a.size
    no_improve = 0
    patience = max_iters // 20 if max_iters >= 20 else 5

    for t in range(max_iters):
        if time.time() >= deadline:
            break

        g = _subgradient_max_conv(a)

        # linear decay of the learning rate
        eta = eta0 * (1.0 - t / max_iters)

        a = _exp_update(a, g, eta)

        cur_score = _objective_norm1(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            print(f"[opt] iter {t:6d}  new best O={best_score:.6f}")
        else:
            no_improve += 1
            if no_improve >= patience:
                # small jitter to hop out of a flat region
                jitter = np.random.randn(n) * 1e-4
                a = best_a + jitter
                a = np.clip(a, 0.0, None)
                a = _project_to_simplex(a)
                no_improve = 0

    return best_a, best_score

def _shape_power(alpha: float, beta: float, n: int) -> np.ndarray:
    """Monotone decreasing shape a_i ∝ (i+β)^(‑α) , i = 1..n."""
    i = np.arange(1, n + 1, dtype=float)
    a = (i + beta) ** (-alpha)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _shape_power_exp(alpha: float, beta: float, gamma: float, n: int) -> np.ndarray:
    """Power‑law with exponential tail."""
    i = np.arange(1, n + 1, dtype=float)
    a = (i + beta) ** (-alpha) * np.exp(-gamma * (i - 1) / n)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _shape_sym(alpha: float, beta: float, n: int) -> np.ndarray:
    """Symmetric shape a_i ∝ ((i+β)*(n+1‑i+β))^(‑α)."""
    i = np.arange(1, n + 1, dtype=float)                # 1‑based index
    left = i + beta
    right = (n + 1 - i) + beta
    a = (left * right) ** (-alpha)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _shape_sym_exp(alpha: float, beta: float, gamma: float, n: int) -> np.ndarray:
    """Symmetric product‑law with an exponential factor on the distance from the edge."""
    i = np.arange(1, n + 1, dtype=float)
    left = i + beta
    right = (n + 1 - i) + beta
    base = (left * right) ** (-alpha)
    # distance from the nearest edge, normalised by n
    dist = np.minimum(i - 1, n - i) / n
    a = base * np.exp(-gamma * dist)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:

    Entry‑point required by the judge.
    Returns a list of non‑negative step heights that (empirically) minimise
    the supplied ``evaluate_sequence`` bound.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # keep a safety margin

    # Length choices – powers of two for FFT efficiency.
    # We start with a moderate size and move to larger ones if time permits.
    length_candidates = [4096, 8192, 16384]

    global_best_seq = None
    global_best_score = float("inf")
    best_n = None

    for n in length_candidates:
        if time.time() >= deadline - 10.0:
            break
        seq, score = _search_n(n, deadline)
        if score < global_best_score:
            global_best_score = score
            global_best_seq = seq
            best_n = n
            print(f"[overall] new global best O={global_best_score:.6f} (n={best_n})")

    # Final exact evaluation using the provided security‑checked function.
    try:
        final_score = evaluate_sequence(global_best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:  # pragma: no cover
        print("[final] evaluate_sequence call failed:", e)
        final_score = None

    return global_best_seq.tolist()
