"""
Propose a non‑negative step‑function that (approximately) minimises

    O(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm works in three stages:

1️⃣  **Parametric warm‑start** – try a few simple families (exponential,
    power‑law, linear) on a fixed length `n`.  Each candidate is projected
    onto the probability simplex (∑aᵢ=1) and its exact objective
    `2·n·max(conv)` is evaluated via an FFT.  The best candidate is kept.

2️⃣  **Sub‑gradient descent** – starting from the best warm‑start,
    repeatedly move the current point in the direction of a sub‑gradient
    of `max(conv)`.  The gradient of the active convolution entry is
    computed analytically (`_subgradient_max_conv`).  After each step the
    vector is projected back onto the simplex.

3️⃣  **Stochastic refinement** – a simple “mass‑transfer’’ Metropolis
    walk (randomly move a fraction of the probability mass from one
    index to another) with a slowly cooling temperature.  Occasionally
    the vector is smoothed by a small moving‑average blend.  Any improvement
    updates the global best.

All steps respect the 1000 s time budget and use only NumPy/standard‑library
operations.  The final sequence is returned as a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex {x ≥ 0, sum(x)=1}
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # all entries are zero – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution via FFT (returns length 2·n‑1 array)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution a⨂a using an FFT of length L (L ≥ 2·len(a)‑1)."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution entry (averaged over all maximisers)
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. a (averaged if several maximisers)."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)          # indices where the max is attained
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        # derivative of sum_i a[i] * a[k-i] w.r.t. a[i] is a[j] + a[j] (i≠j) or a[i] (i=j)
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
# Exact objective for a vector already on the simplex (∑a = 1)
def _objective_simplex(a: np.ndarray, n: int, L: int) -> float:
    """Return the exact O‑value 2·n·max(conv) for sum(a)=1."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()
