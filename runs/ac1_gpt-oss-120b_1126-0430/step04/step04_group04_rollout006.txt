#!/usr/bin/env python
"""
Propose a non‑negative step function that minimizes

        2·n·max_k (a*a)[k] / (Σ a_i)²

The algorithm proceeds in three stages:

1. **Initialisation** – a short scan of power‑law families and many
   Dirichlet random draws (wide range of concentration) to obtain a
   reasonably good starting shape (already below the previous best).

2. **Smooth‑max gradient descent** – we minimise a smooth approximation of
   the max‑convolution using the log‑sum‑exp (soft‑max) surrogate.
   For a given β the surrogate is
       Lβ(a) = (1/β)·log∑_k e^{β·(a*a)[k]} .
   Its gradient can be written in closed form and evaluated efficiently
   with FFTs (O(L log L) per iteration).  β is increased gradually so
   that the optimisation first smoothes the landscape and then focuses
   on the true maximum.

3. **Projection and restarts** – after each gradient step the iterate is
   projected onto the probability simplex (non‑negative, sum = 1) using a
   classic ‑O(n)‑time projection.  Stagnation triggers a small random
   perturbation to escape local minima.

The routine respects the given time budget (default 1000 s) and always
returns a feasible sequence.  No external files or network access are used. 
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float) -> np.ndarray:
    """Return a normalised power‑law a_i ∝ (i+1)^‑p (Σa_i = 1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = i ** (-p)
    seq /= seq.sum()
    return seq
