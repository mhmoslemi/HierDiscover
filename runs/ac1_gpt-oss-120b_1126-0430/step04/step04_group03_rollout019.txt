    # conv shape (2n-1)
    # compute w = exp(conv/eps)

    max_c = conv.max()
    w = np.exp((conv - max_c) / eps)

    if w_sum == 0:
        w = np.ones_like(conv) / conv.size

        w /= w_sum

    conv, A = conv_full(a)  # compute initial conv and A (store)
    best_o = _objective_norm1(a)  # O = 2 n * maxconv
    best_seq = a.copy()

            break
        # compute current conv, A already

        eps = max(eps0 * (eps_decay ** t), 1e-8)

            # optionally print progress
            print(f"[opt] iter {t:6d} O={best_o:.6f}")

    return best_seq, best_o

"""
Propose a step‑function (non‑negative list of heights) that minimises

    O = 2·n·max(conv(a,a)) / (sum(a)²)

The algorithm works in three phases:

1️⃣  **Broad random sampling of a few‑parameter families**  
    – Power‑law shapes a_i ∝ (i+β)^(-α)  
    – Power‑law with an exponential tail a_i ∝ (i+β)^(-α)·exp(-γ·i/n)  

    Thousands of random (α,β,γ) triples are generated, the corresponding
    sequences are normalised (∑a_i=1) and the objective 2·n·max(conv) is
    evaluated via an FFT‑based convolution.  The best sequence found so far
    is kept.

2️⃣  **Local perturbations** – small mass transfers between two random
    indices are tried to improve the incumbent.  This explores the
    neighbourhood of the best candidate without any gradient information.

3️⃣  **Mirror‑descent (exponentiated‑gradient) polishing** – starting from the
    best sequence, a smoothed version of the “max‑convolution’’ objective is
    minimised:

        f_ε(a) = 2·n·ε·log Σ_k exp(conv_k(a)/ε)

    Its gradient is obtained by a correlation of the current sequence with
    the soft‑max weights w_k ∝ exp(conv_k/ε).  An exponentiated‑gradient
    update keeps the vector on the probability simplex.  The smoothing
    parameter ε is gradually reduced; a small amount of jitter is added when
    progress stalls.

The whole routine respects the 1000 s wall‑time budget, prints progress,
and finally returns the best sequence (as a plain ``list[float]``) found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small utility – next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a power‑of‑two FFT (full length)."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def _objective_norm1(a: np.ndarray) -> float:
    """Objective 2·n·max(conv) for a vector already normalised to sum‑1."""
    n = a.shape[0]
    conv = _conv_fft(a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
def _shape_power(alpha: float, beta: float, n: int) -> np.ndarray:
    """Monotone decreasing (i+β)^(-α) shape, normalised to sum‑1."""
    i = np.arange(1, n + 1, dtype=float)
    a = (i + beta) ** (-alpha)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
def _shape_power_exp(alpha: float, beta: float, gamma: float, n: int) -> np.ndarray:
    """Power‑law with exponential tail:
       a_i ∝ (i+β)^(-α)·exp(-γ·(i‑1)/n)  , normalised to sum‑1."""
    i = np.arange(1, n + 1, dtype=float)
    a = (i + beta) ** (-alpha) * np.exp(-gamma * (i - 1) / n)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, eps: float) -> np.ndarray:
    """Soft‑max over the convolution values (stabilised)."""
    max_c = conv.max()
    w = np.exp((conv - max_c) / eps)
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full_like(w, 1.0 / w.size)
    return w / w_sum


# ----------------------------------------------------------------------
def _optimize_softmax(a0: np.ndarray,
                      max_iters: int,
                      eta0: float,
                      eps0: float,
                      eps_decay: float,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the smoothed objective.
    Returns the best sequence seen and its (true) objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    size_fft = _next_pow2(3 * n - 2)          # for correlation (a * w)

    best_a = a.copy()
    best_score = _objective_norm1(a)
    no_improve = 0
    patience = max_iters // 20 if max_iters > 0 else 1000

    for t in range(max_iters):
        if time.time() >= deadline:
            break

        # ----- FFT of current a -------------------------------------------------
        A = np.fft.rfft(a, n=size_fft)

        # ----- convolution a*a --------------------------------------------------
        conv = np.fft.irfft(A * A, n=size_fft)[: 2 * n - 1]

        # ----- soft‑max weights --------------------------------------------------
        eps = max(eps0 * (eps_decay ** t), 1e-12)
        w = _softmax_weights(conv, eps)

        # ----- gradient via correlation (a ⋆ w) -------------------------------
        W = np.fft.rfft(w, n=size_fft)
        corr = np.fft.irfft(np.conj(A) * W, n=size_fft)
        start = len(w) - 1                     # = 2n‑2
        g = 2.0 * corr[start:start + n]        # ∂ max‑conv ≈ 2·(a⋆w)

        # ----- exponentiated‑gradient update ------------------------------------
        eta = eta0 * (1.0 - t / max_iters)     # linear decay
        log_a = np.log(a + 1e-15) - eta * g
        log_a -= log_a.max()                  # keep numbers moderate
        a = np.exp(log_a)
        a /= a.sum()

        # ----- evaluate true objective (without smoothing) ----------------------
        cur_score = _objective_norm1(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            print(f"[opt] iter {t:6d}  new best O={best_score:.6f}")
        else:
            no_improve += 1
            if no_improve >= patience:
                # small jitter to escape plateaus
                jitter = np.random.randn(n) * 1e-5
                a = best_a + jitter
                a = np.clip(a, 0.0, None)
                a_sum = a.sum()
                a = np.full_like(a, 1.0 / n) if a_sum == 0.0 else a / a_sum
                no_improve = 0

    return best_a, best_score

                    prev = _normalise(prev)
                candidates.append(prev)
        except Exception:       # pragma: no cover
            pass

    # a small deterministic grid (α,β)
    alphas_grid = np.linspace(0.25, 0.65, 9)          # 9 values
    betas_grid  = np.linspace(0.0, 30.0, 9)          # 9 values
    for a in alphas_grid:
        for b in betas_grid:
            candidates.append(_shape_power(a, b, n))

    # ------------------------------------------------------------------
    # 2️⃣  Large random sweep over the three‑parameter family.
    #     Guard with the deadline – stop early if we run out of time.
    while time.time() < deadline - 5.0:
        a = np.random.uniform(0.10, 0.80)            # α
        b = np.random.uniform(0.0, 80.0)            # β
        g = np.random.uniform(0.0, 0.80)            # γ
        seq = _shape_power_exp(a, b, g, n)
        candidates.append(seq)
        # limit the number of stored candidates to keep memory modest
        if len(candidates) > 25000:
            break

    # ------------------------------------------------------------------
    # 3️⃣  Evaluate all candidates on the cheap “norm‑1’’ objective.
    best_seq = None
    best_score = float("inf")
    print("[search] evaluating candidates")
    for seq in candidates:
        if time.time() >= deadline:
            break
        seq = _normalise(seq)
        score = _objective_norm1(seq)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[search] new best O={best_score:.6f}")

    # If for some reason we have no candidate, fall back to uniform.
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = _objective_norm1(best_seq)

    # ------------------------------------------------------------------
    # 4️⃣  Local random refinement – move a tiny amount of mass between two
    #     random positions.  Accept only improvements.
    print("[local] start local perturbation")
    for _ in range(3000):
        if time.time() >= deadline:
            break
        a = best_seq.copy()
        i, j = np.random.randint(0, n, size=2)
        delta = (np.random.rand() - 0.5) * 0.02    # up to ±1 % of total mass
        if a[i] + delta < 0.0 or a[j] - delta < 0.0:
            continue
        a[i] += delta
        a[j] -= delta
        a = _normalise(a)
        score = _objective_norm1(a)
        if score < best_score:
            best_score = score
            best_seq = a
            print(f"[local] improved O={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Polishing with exponentiated‑gradient (soft‑max) mirror descent.
    remaining = deadline - time.time()
    if remaining > 10.0:
        # Rough estimate: ~200 iterations per second on typical hardware.
        max_iters = int(min(200_000, remaining * 200))
        print(f"[opt] polishing start – max_iters={max_iters}")
        a_opt, score_opt = _optimize_softmax(best_seq,
                                             max_iters=max_iters,
                                             eta0=0.5,
                                             eps0=0.01,
                                             eps_decay=0.9995,
                                             deadline=deadline)
        if score_opt < best_score:
            best_seq = a_opt
            best_score = score_opt
            print(f"[opt] polished best O={best_score:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣  Final exact evaluation via the provided evaluate_sequence().
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:   # pragma: no cover
        print("[final] evaluate_sequence call failed:", e)
        final_score = None

    return best_seq.tolist()
