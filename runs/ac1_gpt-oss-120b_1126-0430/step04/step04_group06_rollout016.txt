    a /= a.sum()

    best_a = a.copy()

    iter = 0
    # parameters

    decay = 0.001

    while time.time() < deadline and iter < 500000:
        iter += 1
        # compute convolution via FFT
        fft_len = 1 << ((2*n - 1).bit_length())
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[:2*n-1]
        max_idx = int(conv.argmax())
        max_val = conv[max_idx]
        # compute gradient of max conv
        # gradient g_i = 2 * a_{max_idx - i} if 0 <= max_idx - i < n else 0
        g = np.zeros_like(a)

        hi = min(max_idx, n - 1)
        i = np.arange(lo, hi+1)
        g[i] = 2.0 * a[max_idx - i]

        # multiplicative update
        eta = eta0 / np.sqrt(iter)  # decreasing
        a = a * np.exp(-eta * g)  # elementwise
        # renormalize to sum 1 (but also to avoid underflow)
        sum_a = a.sum()
        if sum_a == 0:

        else:
            a /= sum_a

            pert = np.random.randn(n) * 0.01
            a = np.maximum(a + pert, 0)
            a /= a.sum()

        # evaluate cheap score
        cheap_score = 2.0 * n * max_val
        if cheap_score < best_score:
            best_score = cheap_score
            best_a = a.copy()

        # maybe enlarge support if time remains (but we keep same n)

#!/usr/bin/env python
"""
Simple Adam‑based optimizer for the step‑function bound.

The algorithm works directly on the probability simplex (softmax logits)
and tries to minimise the quantity

    score = 2·n·max(conv(a,a))

which, after the external ``evaluate_sequence`` routine rescales,
is exactly the value returned by the evaluator (the sum of the
coefficients is normalised to 1, therefore the division by
sum(a)**2 has no effect).

Key ideas
---------

*   Start from a reasonably good seed – the global variable
    ``height_sequence_1`` (if present) or a power‑law distribution.
*   Keep the support size fixed at the maximal allowed length
    (30 000).  The original reference code repeatedly *reset* the
    optimisation when the support was enlarged; we simply work with
    the full size from the beginning, which gives many more
    optimisation steps for the same time budget.
*   Use a smoothed sub‑gradient of ``max(conv)``.  The top‑K
    convolution entries are selected, weighted by a temperature‑controlled
    soft‑max (the temperature is increased linearly so that the
    sub‑gradient focuses on the true maximum later in the run).
*   Adam on the log‑it (mirrored) space performs fast, numerically
    stable updates on the simplex.
*   Small random perturbations are injected occasionally to help the
    optimiser escape plateaus.
*   The best sequence seen so far (according to the cheap proxy
    ``2·n·max(conv)``) is stored and finally returned.  The external
    ``evaluate_sequence`` routine will compute the exact score.

Running time
------------
The routine respects the supplied ``budget_s`` (default 1000 s).  It
stops a couple of seconds before the deadline, evaluates the best
candidate with ``evaluate_sequence`` and returns it.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (must be top‑level – no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max that returns a probability vector (sum = 1)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Logit representation that yields ``a`` after the soft‑max."""
    return np.log(a + eps)


def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two ≥ 2·n‑1 (efficient size for real FFT)."""
    return 1 << ((2 * n - 1).bit_length())


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full linear self‑convolution a*a using an FFT of length ``fft_len``.
    Returns an array of length 2·len(a)‑1.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.size - 1]
    return conv


def _weighted_subgradient(
    a: np.ndarray, conv: np.ndarray, k: int, theta: float
) -> np.ndarray:
    """
    Smoothed sub‑gradient of ``max(conv)`` for a probability vector ``a``.

    Parameters
    ----------
    a      : np.ndarray, shape (n,)
        Current probability vector (sum == 1, a >= 0).
    conv   : np.ndarray, shape (2*n-1,)
        Full linear self‑convolution a*a.
    k      : int
        Number of top convolution entries taken into account.
    theta  : float
        Temperature for the soft‑max weighting – larger values give
        a sharper focus on the true maximum.

    Returns
    -------
    g : np.ndarray, shape (n,)
        Weighted sub‑gradient w.r.t. ``a``.
    """
    n = a.shape[0]

    # indices of the k largest entries (unsorted)
    idx = np.argpartition(conv, -k)[-k:]
    # order them so the largest entry gets the largest weight (more stable)
    order = np.argsort(-conv[idx])
    idx = idx[order]

    # soft‑max style weights
    vals = conv[idx]
    max_val = vals[0]                     # the largest entry
    exps = np.exp(theta * (vals - max_val))
    weights = exps / exps.sum()

    # accumulate the weighted sub‑gradients
    g = np.zeros_like(a)
    for off, w in zip(idx, weights):
        # contribution of offset ``off``:
        #   ∂/∂a_i (a*a)[off] = 2·a_{off‑i}   (if the index is valid)
        lo = max(0, off - (n - 1))
        hi = min(off, n - 1)
        i = np.arange(lo, hi + 1, dtype=np.int64)
        g[i] += w * 2.0 * a[off - i]
    return g


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """Normalised power‑law a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Optimise a non‑negative step‑function and return the best sequence.
    The function respects the given time budget and uses an Adam optimiser
    on the probability simplex with a smoothed max‑convolution sub‑gradient.
    """
    np.random.seed(seed)
    start = time.time()
    # leave a small safety margin for the final evaluation
    deadline = start + budget_s - 2.0

    # ---------------------------------------------------------------
    # 0. Determine the working support size (fixed, no later enlargement).
    # ---------------------------------------------------------------
    n_max = 30000                # hard cap imposed by the challenge
    # Use the supplied global sequence if it exists, otherwise start fresh.
    init_global = globals().get("height_sequence_1", None)
    if init_global is not None:
        init_global = np.array(init_global, dtype=np.float64)
        # Pad / truncate to the chosen size
        if init_global.shape[0] < n_max:
            pad = np.zeros(n_max - init_global.shape[0], dtype=np.float64)
            a0 = np.concatenate([init_global, pad])
        else:
            a0 = init_global[:n_max].copy()
        print(f"[Init] Using supplied global sequence, padded to {n_max} entries")
    else:
        a0 = _power_law_distribution(n_max, 0.60)
        print(f"[Init] No global sequence – started from power‑law (p=0.60)")

    # normalise to a probability vector (sum = 1)
    a0 = np.maximum(a0, 0.0)
    s0 = a0.sum()
    if s0 == 0.0:
        a0[:] = 1.0 / n_max
    else:
        a0 /= s0

    # ---------------------------------------------------------------
    # 1. Log‑it parametrisation + Adam state.
    # ---------------------------------------------------------------
    eps = 1e-16
    z = _logits_from_distribution(a0, eps)

    m = np.zeros_like(z)          # first‑moment estimate
    v = np.zeros_like(z)          # second‑moment estimate
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    base_alpha = 0.45             # more aggressive than the reference code

    # ---------------------------------------------------------------
    # 2. Hyper‑parameters for the smoothed sub‑gradient.
    # ---------------------------------------------------------------
    K = 30                        # number of top convolution entries used
    theta_start = 0.1
    theta_end = 35.0
    theta_slope = (theta_end - theta_start) / max(1.0, budget_s - 10.0)

    # ---------------------------------------------------------------
    # 3. Main optimisation loop.
    # ---------------------------------------------------------------
    best_seq = a0.copy()
    # cheap proxy – with sum == 1 the exact score is 2·n·max(conv)
    best_score = float("inf")
    it = 0
    last_improve = 0
    report_every = 2500

    n = n_max
    fft_len = _fft_len_for_n(n)

    while time.time() < deadline and it < 2_000_000:
        it += 1

        # probability vector from the current logits
        a = _softmax(z)

        # full self‑convolution (FFT based)
        conv = _conv_full_fft(a, fft_len)

        maxb = float(conv.max())
        cheap_score = 2.0 * n * maxb

        # record best ever seen (cheap proxy)
        if cheap_score < best_score:
            best_score = cheap_score
            best_seq = a.copy()
            last_improve = it
            if it % report_every == 0:
                print(f"[Iter {it:07d}] NEW BEST cheap = {cheap_score:.6f}")

        # -----------------------------------------------------------
        # Smoothed sub‑gradient of the max‑convolution
        # -----------------------------------------------------------
        elapsed = time.time() - start
        theta = min(theta_end, theta_start + theta_slope * elapsed)

        g_a = _weighted_subgradient(a, conv, K, theta)

        # mirror‑descent conversion to log‑it space (KL geometry)
        mu = float(np.dot(g_a, a))
        grad_z = a * (g_a - mu)

        # Adam moment updates (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # direction (negative Adam step)
        step_dir = - m_hat / (np.sqrt(v_hat) + eps_adam)

        # basic step‑size schedule: inverse‑sqrt decay
        alpha_t = base_alpha / (math.sqrt(it) + 1.0)

        # perform the update in log‑it space
        z = z + alpha_t * step_dir
        # keep logits in a safe numeric range
        np.clip(z, -80.0, 80.0, out=z)

        # -----------------------------------------------------------
        # Occasional random perturbations (helps escape plateaus)
        # -----------------------------------------------------------
        if it % 5000 == 0 and np.random.rand() < 0.30:
            perturb = 0.015 * np.random.randn(n)
            z += perturb
            # damp the Adam moments to avoid huge steps after a jump
            m *= 0.6
            v *= 0.6

        # -----------------------------------------------------------
        # Early stop if stagnation for a long time
        # -----------------------------------------------------------
        if it - last_improve > 300_000:
            print(f"[Iter {it:07d}] No improvement for {it-last_improve} iters – stopping.")
            break

    # ---------------------------------------------------------------
    # 4. Final evaluation with the external routine
    # ---------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[Result] Final exact score = {final_score:.6f}")
    except Exception as exc:        # protect against unexpected errors
        final_score = float("inf")
        print(f"[Result] Evaluation error: {exc}")

    # ---------------------------------------------------------------
    # 5. Return the best sequence found.
    # ---------------------------------------------------------------
    return final_seq
