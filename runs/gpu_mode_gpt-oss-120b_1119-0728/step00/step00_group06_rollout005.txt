def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # Input shape: [B,N,N,dim]
    B,N,N2,dim = input_tensor.shape
    assert N==N2

    dim_c = config["dim"] # same as dim

    hidden_dim = config["hidden_dim"]

    eps = 1e-5

    # (1) layernorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim_c,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)

    # (2) Linear projections (no bias)
    # Using torch.nn.functional.linear
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])  # shape B,N,N,hidden
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # mask: shape [B,N,N] -> unsqueeze(-1)
    if mask is not None:
        mask_f = mask.unsqueeze(-1)  # shape B,N,N,1
        left = left * mask_f
        right = right * mask_f

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # (3) core multiplication: out = left @ right.transpose(-2,-1) across seq_len dimension
    # reshape to (B*hidden, N, N)
    left_reshape = left.view(B*hidden_dim, N, N).contiguous()
    right_reshape = right.view(B*hidden_dim, N, N).contiguous()

    # Output allocate
    out_reshape = torch.empty_like(left_reshape)

    # call Triton kernel
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32  # can tune

    grid = (tl.cdiv(N, BLOCK_M), tl.cdiv(N, BLOCK_N), B*hidden_dim)

    tri_mul_kernel[grid](
        left_reshape,
        right_reshape,
        out_reshape,
        N,
        left_reshape.stride(0), left_reshape.stride(1), left_reshape.stride(2),
        right_reshape.stride(0), right_reshape.stride(1), right_reshape.stride(2),
        out_reshape.stride(0), out_reshape.stride(1), out_reshape.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B,N,N,hidden]
    out = out_reshape.view(B, hidden_dim, N, N).permute(0,2,3,1).contiguous() # shape B,N,N,hidden

    # (4) to_out_norm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])

    # (5) gating
    out = out * out_gate

    # (6) final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    N,
    stride_l_bh, stride_l_i, stride_l_j,
    stride_r_bh, stride_r_i, stride_r_j,
    stride_o_bh, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_bh = tl.program_id(2)  # batch*hidden
    pid_m = tl.program_id(0)   # row block
    pid_n = tl.program_id(1)   # col block

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for kp in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kp * BLOCK_K + tl.arange(0, BLOCK_K)
        # Load A, shape (BLOCK_M, BLOCK_K)
        a_ptr = left_ptr + pid_bh * stride_l_bh + offs_m[:, None] * stride_l_i + offs_k[None, :] * stride_l_j
        a = tl.load(a_ptr, mask=(offs_m[:, None] < N) & (offs_k[None, :] < N), other=0.0).to(tl.float32)
        # Load B, shape (BLOCK_K, BLOCK_N) (original right as B)
        b_ptr = right_ptr + pid_bh * stride_r_bh + offs_k[:, None] * stride_r_i + offs_n[None, :] * stride_r_j
        b = tl.load(b_ptr, mask=(offs_k[:, None] < N) & (offs_n[None, :] < N), other=0.0).to(tl.float32)
        # Compute dot product
        acc += tl.dot(a, b)  # returns float32

    # Write back
    o_ptr = out_ptr + pid_bh * stride_o_bh + offs_m[:, None] * stride_o_i + offs_n[None, :] * stride_o_j
    # Store only valid positions
    mask_o = (offs_m[:, None] < N) & (offs_n[None, :] < N)
    tl.store(o_ptr, acc, mask=mask_o)

"""
TriMul (outgoing) – fast Triton implementation.

The TriMul forward consists of:
1. Layer‑norm over the channel dimension.
2. Two linear projections (left / right), masked, gated and fused with sigmoid gates.
3. A costly Einstein summation:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is exactly a batch of H×B matrix multiplications (N×N) per hidden channel.
   This N³ operation is performed in a custom Triton kernel that works on
   half‑precision data (lower memory bandwidth) while accumulating in FP32.
4. A second layer‑norm, an output gate and a final linear projection.

Only the matmul (step 3) is off‑loaded to Triton; the surrounding linear
operations and normalisations are executed with PyTorch for simplicity.
The kernel is written for the H100 (CUDA‑capable) and targets a runtime
≈ 1 ms on the provided test configurations.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: outgoing TriMul (left @ rightᵀ) for each batch & hidden dim
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr,               # *float16   [B, N, N, H]
    right_ptr,              # *float16   [B, N, N, H]
    out_ptr,                # *float16   [B, N, N, H]
    N,                      # int32      sequence length
    stride_left_batch,      # int64
    stride_left_i,          # int64
    stride_left_k,          # int64
    stride_left_h,          # int64
    stride_right_batch,     # int64
    stride_right_i,         # int64
    stride_right_k,         # int64
    stride_right_h,         # int64
    stride_out_batch,       # int64
    stride_out_i,           # int64
    stride_out_k,           # int64
    stride_out_h,           # int64
    hidden_dim,             # int32  (runtime constant)
    BLOCK_M: tl.constexpr,  # tile size in M dimension
    BLOCK_N: tl.constexpr,  # tile size in N dimension
    BLOCK_K: tl.constexpr,  # tile size in K dimension
):
    pid_m = tl.program_id(0)                # block row
    pid_n = tl.program_id(1)                # block col
    pid_z = tl.program_id(2)                # combined batch*hidden

    # ------------------------------------------------------------------
    # Resolve batch index and hidden‑channel index from pid_z
    # ------------------------------------------------------------------
    batch = pid_z // hidden_dim
    h     = pid_z %  hidden_dim

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Base pointers for the current (batch, hidden) slice
    base_left  = left_ptr  + batch * stride_left_batch  + h * stride_left_h
    base_right = right_ptr + batch * stride_right_batch + h * stride_right_h
    base_out   = out_ptr   + batch * stride_out_batch   + h * stride_out_h

    # Accumulator in FP32 (higher numeric stability)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the K dimension (the summed axis)
    # ------------------------------------------------------------------
    for kp in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kp * BLOCK_K + tl.arange(0, BLOCK_K)

        # ---------- Load left block (M x K) ----------
        a_ptrs = base_left + (offs_m[:, None] * stride_left_i) + (offs_k[None, :] * stride_left_k)
        a = tl.load(a_ptrs,
                    mask=(offs_m[:, None] < N) & (offs_k[None, :] < N),
                    other=0.0)               # dtype follows left_ptr (fp16)

        # ---------- Load right block (K x N) ----------
        # Note the transpose: we need right[j,k] → right[k,j] for the dot.
        b_ptrs = base_right + (offs_k[:, None] * stride_right_k) + (offs_n[None, :] * stride_right_i)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < N) & (offs_n[None, :] < N),
                    other=0.0)               # dtype follows right_ptr (fp16)

        # ---------- Accumulate ----------
        acc += tl.dot(a, b)                  # fp16·fp16 → fp32 accumulation

    # ------------------------------------------------------------------
    # Write the result tile back to global memory
    # ------------------------------------------------------------------
    c_ptrs = base_out + (offs_m[:, None] * stride_out_i) + (offs_n[None, :] * stride_out_k)
    mask_c = (offs_m[:, None] < N) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_c)

# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor, shape [B, N, N, dim]
            - mask:        torch.Tensor, shape [B, N, N] (bool or float)
            - weights:     dict of model weights (see code)
            - config:      dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------
    # 1. LayerNorm on the input (channel axis = dim)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Optional mask (mask is broadcast over the channel dim)
    # --------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)      # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------
    # 4. Gating (sigmoid of linear projections)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core N³ operation – off‑load to Triton
    # --------------------------------------------------------------
    B, N, _, _ = left.shape

    # Cast to half‑precision to reduce memory traffic
    left_h = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()

    # Output buffer (half‑precision)
    out_h = torch.empty_like(left_h)

    # Tile sizes – tuned for H100
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),          # rows
        triton.cdiv(N, BLOCK_N),          # cols
        B * hidden_dim,                    # batch × hidden
    )

    # Launch the kernel
    tri_mul_outgoing_kernel[grid](
        left_h,
        right_h,
        out_h,
        N,
        left_h.stride(0), left_h.stride(1), left_h.stride(2), left_h.stride(3),
        right_h.stride(0), right_h.stride(1), right_h.stride(2), right_h.stride(3),
        out_h.stride(0), out_h.stride(1), out_h.stride(2), out_h.stride(3),
        hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Cast back to FP32 for the remaining layers
    out = out_h.to(torch.float32)

    # --------------------------------------------------------------
    # 6. Post‑matmul layer‑norm, gating and final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
```
