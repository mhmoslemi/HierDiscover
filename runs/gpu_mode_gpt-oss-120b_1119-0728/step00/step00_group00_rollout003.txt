import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_Abatch, stride_Am, stride_Ak,
    stride_Bbatch, stride_Bk, stride_Bn,
    stride_Cbatch, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    pid_n = tl.program_id(1)

    # Compute batch id and block row offset
    total_blocks_m = tl.cdiv(M, BLOCK_M)
    batch_id = pid // total_blocks_m
    block_m_id = pid % total_blocks_m

    # Compute indices
    i = block_m_id * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i < M
    mask_j = j < N

    # offsets for A and B matrices for the current batch
    a_batch_offset = batch_id * stride_Abatch
    b_batch_offset = batch_id * stride_Bbatch
    c_batch_offset = batch_id * stride_Cbatch

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        k_range = k_offset + tl.arange(0, BLOCK_K)
        mask_k = k_range < K

        # Load A tile [M, K]
        a_ptrs = A_ptr + a_batch_offset + i[:, None] * stride_Am + k_range[None, :] * stride_Ak
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load B tile [K, N]
        b_ptrs = B_ptr + b_batch_offset + k_range[:, None] * stride_Bk + j[None, :] * stride_Bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(a, b)

    # Store output
    c_ptrs = C_ptr + c_batch_offset + i[:, None] * stride_Cm + j[None, :] * stride_Cn
    tl.store(c_ptrs, acc.to(tl.float32), mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """
    Triton implementation of the outgoing Triangle Multiplicative Update (TriMul).

    The kernel performs:
    1) LayerNorm on the input.
    2) Linear projections + gating (all in PyTorch).
    3) Core O(N³·H) multiplication
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
       implemented as a batched matrix multiplication using a Triton kernel.
    4) Post‑norm, gating and final linear projection.
    """
    import torch
    import torch.nn.functional as F
    import triton
    import triton.language as tl

    # Unpack arguments
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # Load weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ---------- 1. LayerNorm ----------
    # LayerNorm over last dimension (dim)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # ---------- 2. Linear proj + gating ----------
    # Projections
    left = F.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    # Apply mask before gating if present
    if not nomask:
        # mask shape [B,N,N] -> [B,N,N,1]
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)
        left = left * mask
        right = right * mask

    # Gates
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ---------- 3. Core TriMul (batched GEMM) ----------
    # Rearrange tensors to treat hidden_dim as a batch dimension:
    # shape -> [B, hidden_dim, N, N]
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    B, H, N, _ = left.shape

    left_flat = left.view(B * H, N, N)   # (BH, N, N)
    right_flat = right.view(B * H, N, N)

    # Transpose the second operand to turn the contraction into a matmul
    right_T = right_flat.transpose(1, 2).contiguous()   # (BH, N, N)

    # Allocate output
    out_flat = torch.empty_like(left_flat)

    # Triton kernel launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    M = N
    K = N

    # Strides for each tensor
    stride_a_batch, stride_a_m, stride_a_k = left_flat.stride()
    stride_b_batch, stride_b_k, stride_b_n = right_T.stride()   # after transpose the stride order is (batch, k, n)
    stride_c_batch, stride_c_m, stride_c_n = out_flat.stride()

    # Compute grid size
    total_blocks_m = (M + BLOCK_M - 1) // BLOCK_M
    total_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (B * H * total_blocks_m, total_blocks_n)

    # Launch kernel
    batched_matmul_kernel[grid](
        left_flat,
        right_T,
        out_flat,
        M,
        N,
        K,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to original layout [B,N,N,hidden_dim]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ---------- 4. Post‑norm, gating and final projection ----------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=1e-5)
    out = out * out_gate
    out = F.linear(out, to_out_weight)

    return out

"""
TriMul (outgoing) – Triton‑accelerated implementation

Algorithm
---------
1. Layer‑norm the 4‑D input tensor over its channel dimension `dim`.
2. Linear projections (dim → hidden_dim) for the left and right tensors.
   A mask (if provided) is applied **before** gating.
3. Compute three sigmoid gates (left, right, out) from the normalized input.
   Apply left/right gates to the projected tensors.
4. Core multiplicative update:
      out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   The contraction is a batched matrix multiplication with the hidden
   dimension folded into the batch axis. This step is performed by a
   custom Triton kernel (`batched_matmul_kernel`).
5. Layer‑norm on the hidden dimension, gate with `out_gate`,
   and a final linear projection back to `dim`.

The heavy O(N³·hidden_dim) work is done in Triton, while the cheap
linear / activation / normalisation steps stay in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_Abatch, stride_Am, stride_Ak,
    stride_Bbatch, stride_Bk, stride_Bn,
    stride_Cbatch, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Batched GEMM: C = A @ B   where
    A : (batch, M, K)
    B : (batch, K, N)
    C : (batch, M, N)
    """
    pid = tl.program_id(0)          # encodes (batch, block_m)
    pid_n = tl.program_id(1)        # block_n

    # Number of M‑blocks per matrix
    total_blocks_m = tl.cdiv(M, BLOCK_M)

    # Decode batch id and block‑row id
    batch_id = pid // total_blocks_m
    block_m_id = pid % total_blocks_m

    # Compute absolute row / column indices
    i = block_m_id * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i < M
    mask_j = j < N

    # Base pointers for this batch
    a_batch_off = batch_id * stride_Abatch
    b_batch_off = batch_id * stride_Bbatch
    c_batch_off = batch_id * stride_Cbatch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        k_range = k_offset + tl.arange(0, BLOCK_K)
        mask_k = k_range < K

        # Load A tile [M, K]
        a_ptrs = A_ptr + a_batch_off + i[:, None] * stride_Am + k_range[None, :] * stride_Ak
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # Load B tile [K, N]
        b_ptrs = B_ptr + b_batch_off + k_range[:, None] * stride_Bk + j[None, :] * stride_Bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        acc += tl.dot(a, b)          # (BLOCK_M, BLOCK_N)

    # Write back
    c_ptrs = C_ptr + c_batch_off + i[:, None] * stride_Cm + j[None, :] * stride_Cn
    tl.store(c_ptrs, acc.to(tl.float32), mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Triton‑backed forward pass of the outgoing Triangle Multiplicative Update.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (optional, bool/float)
        - weights_dict : mapping layer names → torch.Tensor (all on the same device)
        - config_dict  : must contain "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 1️⃣ unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 2️⃣ load weight tensors (already on the correct device)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 3️⃣ Layer‑norm over the channel dimension `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)

    # ------------------------------------------------------------------
    # 4️⃣ Linear projections (dim → hidden_dim)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)     # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 5️⃣ Optional mask (applied before gating)
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 6️⃣ Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 7️⃣ Core TriMul contraction – batched GEMM via Triton
    # ------------------------------------------------------------------
    # Fold hidden dimension into batch dimension.
    # Shape after permute: [B, hidden_dim, N, N]
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    B, H, N, _ = left.shape                     # N == N

    # Flatten batch & hidden dim -> (BH, N, N)
    left_flat = left.view(B * H, N, N)          # A
    right_flat = right.view(B * H, N, N)        # B (un‑transposed)

    # Transpose the second operand to turn the contraction into A @ Bᵀ
    right_T = right_flat.transpose(1, 2).contiguous()   # shape (BH, N, N)

    # Output buffer
    out_flat = torch.empty_like(left_flat)

    # ------------------------------------------------------------------
    # Triton launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    M = N
    K = N

    # Strides (batch, row, col) for each tensor
    stride_a_batch, stride_a_m, stride_a_k = left_flat.stride()
    stride_b_batch, stride_b_k, stride_b_n = right_T.stride()
    stride_c_batch, stride_c_m, stride_c_n = out_flat.stride()

    # Grid: (batch*row_blocks, col_blocks)
    total_blocks_m = (M + BLOCK_M - 1) // BLOCK_M
    total_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (B * H * total_blocks_m, total_blocks_n)

    # Launch the kernel
    batched_matmul_kernel[grid](
        left_flat,
        right_T,
        out_flat,
        M,
        N,
        K,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8️⃣ Post‑norm, out‑gate, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=1e-5)
    out = out * out_gate
    out = F.linear(out, to_out_weight)          # → [B, N, N, dim]

    return out
