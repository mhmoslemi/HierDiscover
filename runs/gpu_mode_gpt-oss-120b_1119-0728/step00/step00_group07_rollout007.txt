                     weight_ptr, bias_ptr,
                     eps, D,
                     stride_x_b, stride_x_i, stride_x_j, stride_x_d,
                     stride_y_b, stride_y_i, stride_y_j, stride_y_d,
                     stride_w, stride_b,
                     BLOCK_D: tl.constexpr):

   pid = tl.program_id(0)
   total_positions = B * N * N
   b = pid // (N*N)
   rem = pid % (N*N)
   i = rem // N
   j = rem % N

   # compute base ptrs
   x_base = x_ptr + b*stride_x_b + i*stride_x_i + j*stride_x_j
   y_base = y_ptr + b*stride_y_b + i*stride_y_i + j*stride_y_j

   # first pass: compute sum and sumsq
   sum_val = tl.float32(0.0)
   sum_sq_val = tl.float32(0.0)

import torch
import triton
import triton.language as tl

@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr,
    weight_ptr, bias_ptr,
    eps, DIM,
    stride_xb, stride_xi, stride_xj, stride_xd,
    stride_yb, stride_yi, stride_yj, stride_yd,
    stride_w, stride_b,
    batch, seq_len,
    BLOCK_D: tl.constexpr):
    pid = tl.program_id(0)
    total = batch * seq_len * seq_len
    # decode batch, i, j
    b = pid // (seq_len * seq_len)
    rem = pid % (seq_len * seq_len)
    i = rem // seq_len
    j = rem % seq_len

    # base pointers for this (b,i,j)
    x_base = x_ptr + b * stride_xb + i * stride_xi + j * stride_xj
    y_base = y_ptr + b * stride_yb + i * stride_yi + j * stride_yj

    # ---- first pass: compute mean and variance ----
    sum_val = tl.float32(0.0)
    sum_sq_val = tl.float32(0.0)
    ONE = tl.full((BLOCK_D,), 1.0, dtype=tl.float32)

    for off in range(0, DIM, BLOCK_D):
        offs = off + tl.arange(0, BLOCK_D)
        mask = offs < DIM
        x = tl.load(x_base + offs * stride_xd, mask=mask, other=0.0)
        sum_val += tl.dot(x, ONE)
        sum_sq_val += tl.dot(x * x, ONE)

    mu = sum_val / DIM
    var = sum_sq_val / DIM - mu * mu
    inv_std = 1.0 / tl.sqrt(var + eps)

    # ---- second pass: write normalized output ----
    for off in range(0, DIM, BLOCK_D):
        offs = off + tl.arange(0, BLOCK_D)
        mask = offs < DIM
        x = tl.load(x_base + offs * stride_xd, mask=mask, other=0.0)
        w = tl.load(weight_ptr + offs * stride_w, mask=mask, other=0.0)
        b_val = tl.load(bias_ptr + offs * stride_b, mask=mask, other=0.0)
        y = (x - mu) * inv_std
        y = y * w + b_val
        tl.store(y_base + offs * stride_yd, y, mask=mask)

"""
TriMul (outgoing) forward implementation.

Algorithm
---------
1. **LayerNorm (Triton)**  
   - For each (b, i, j) element of the 4‑D tensor `x` compute the mean and variance
     over the last dimension (`dim`).  
   - Apply the learned scale (`norm.weight`) and bias (`norm.bias`).  
   - The kernel is written in Triton and operates on one (b,i,j) position per program
     (grid size = B·N·N).  The reduction over `dim` is performed in two passes
     (mean/var → normalize → scale+bias).

2. **Projection & Gates (PyTorch)**  
   - Cast the normalized tensor to FP16 for speed (the GPU H100 can use Tensor
     Cores).  
   - Perform five linear maps on the normalized data:
        * left_proj, right_proj
        * left_gate, right_gate, out_gate
   - Apply the sigmoid non‑linearity to the three gate outputs.

3. **Mask & Gating**  
   - If a mask is present, multiply `left` and `right` by the mask (broadcasted).
   - Multiply the gated tensors: `left = left * left_gate`, `right = right * right_gate`.

4. **Tri‑multiplicative update**  
   - Reshape `left` and `right` to `(B*H, N, N)` where `H = hidden_dim`.
   - Compute `out = left @ right.transpose(-2, -1)` via a batched GEMM
     (`torch.bmm`).  This is the `einsum('... i k d, ... j k d -> ... i j d')` step.

5. **Output processing**  
   - Apply a second LayerNorm over the hidden dimension (`to_out_norm`).
   - Multiply by `out_gate`.
   - Project back to the original dimension with `to_out` linear layer.

All operations are done in FP16 except the final cast back to FP32,
ensuring the returned tensor has dtype `torch.float32`.

The Triton kernel only covers the LayerNorm; the heavy matrix multiplications
are performed by highly‑optimized cuBLAS kernels via PyTorch.
"""

    j = rem % seq_len

    # Base pointers for this (b,i,j)
    x_base = x_ptr + b * stride_xb + i * stride_xi + j * stride_xj
    y_base = y_ptr + b * stride_yb + i * stride_yi + j * stride_yj

    # ------------------------------------------------------------------
    # 1st pass: compute mean & variance over dim
    # ------------------------------------------------------------------
    sum_val = tl.float32(0.0)
    sum_sq_val = tl.float32(0.0)
    ONE = tl.full((BLOCK_D,), 1.0, dtype=tl.float32)

    for off in range(0, DIM, BLOCK_D):
        offs = off + tl.arange(0, BLOCK_D)
        mask = offs < DIM
        x = tl.load(x_base + offs * stride_xd, mask=mask, other=0.0)   # (BLOCK_D,)
        sum_val += tl.dot(x, ONE)                # Σ x
        sum_sq_val += tl.dot(x * x, ONE)         # Σ x²

    mu = sum_val / DIM
    var = sum_sq_val / DIM - mu * mu
    inv_std = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # 2nd pass: write normalized result (scale & bias)
    # ------------------------------------------------------------------
    for off in range(0, DIM, BLOCK_D):
        offs = off + tl.arange(0, BLOCK_D)
        mask = offs < DIM
        x = tl.load(x_base + offs * stride_xd, mask=mask, other=0.0)
        w = tl.load(weight_ptr + offs * stride_w, mask=mask, other=0.0)
        b_val = tl.load(bias_ptr + offs * stride_b, mask=mask, other=0.0)

        y = (x - mu) * inv_std            # normalize
        y = y * w + b_val                 # apply LayerNorm weight & bias
        tl.store(y_base + offs * stride_yd, y, mask=mask)


# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: Tuple containing
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict of model parameters (all float32)
            - config: dict with keys
                * "dim" (int)
                * "hidden_dim" (int)
                * "nomask" (bool, optional)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # Extract configuration
    # ------------------------------------------------------------------
    dim = config["dim"]               # input channel dimension
    hidden_dim = config["hidden_dim"] # hidden (output) dimension
    eps = 1e-5                         # LayerNorm epsilon (fixed)
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # Prepare mask (float32 for broadcasting)
    # ------------------------------------------------------------------
    if nomask or mask_tensor is None:
        mask = torch.ones(B, N, N, dtype=dtype, device=device)
    else:
        mask = mask_tensor.to(dtype=dtype)

    # ------------------------------------------------------------------
    # Triton LayerNorm on the input tensor
    # ------------------------------------------------------------------
    norm_out = torch.empty_like(input_tensor, dtype=torch.float32, device=device)

    # Strides (in number of elements, not bytes)
    stride_xb, stride_xi, stride_xj, stride_xd = input_tensor.stride()
    stride_yb, stride_yi, stride_yj, stride_yd = norm_out.stride()

    # LayerNorm parameters
    norm_weight = weights["norm.weight"].contiguous()
    norm_bias   = weights["norm.bias"].contiguous()

    # Launch Triton kernel
    grid = (B * N * N,)
    layernorm_kernel[grid](
        input_tensor, norm_out,
        norm_weight, norm_bias,
        eps, dim,
        stride_xb, stride_xi, stride_xj, stride_xd,
        stride_yb, stride_yi, stride_yj, stride_yd,
        norm_weight.stride(0), norm_bias.stride(0),
        B, N,
        BLOCK_D=64,                     # tile size (fits dim up to 384)
    )
    # ------------------------------------------------------------------
    # Cast to FP16 for the compute‑heavy parts (TensorCores)
    # ------------------------------------------------------------------
    norm_fp16 = norm_out.to(torch.float16)

    # Linear weights (cast to FP16)
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # Linear projections & gates (batched GEMMs)
    # ------------------------------------------------------------------
    flat_norm = norm_fp16.view(-1, dim)                # (B*N*N, dim)

    left      = torch.nn.functional.linear(flat_norm, left_proj_w)      # (B*N*N, H)
    right     = torch.nn.functional.linear(flat_norm, right_proj_w)
    left_gate = torch.sigmoid(torch.nn.functional.linear(flat_norm, left_gate_w))
    right_gate= torch.sigmoid(torch.nn.functional.linear(flat_norm, right_gate_w))
    out_gate  = torch.sigmoid(torch.nn.functional.linear(flat_norm, out_gate_w))

    # Reshape back to 4‑D tensors
    left      = left.view(B, N, N, hidden_dim)
    right     = right.view(B, N, N, hidden_dim)
    left_gate = left_gate.view(B, N, N, hidden_dim)
    right_gate= right_gate.view(B, N, N, hidden_dim)
    out_gate  = out_gate.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # Apply optional mask (broadcast on the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left  = left  * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # Gating
    # ------------------------------------------------------------------
    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Tri‑multiplicative update (batched matrix multiplication)
    # ------------------------------------------------------------------
    # (B, N, N, H) -> (B*H, N, N)
    left_2d  = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)          # L
    right_2d = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N).transpose(2, 1)  # Rᵀ

    out_2d = torch.bmm(left_2d, right_2d)                                        # (B*H, N, N)

    # Back to 4‑D
    out = out_2d.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)                 # (B,N,N,H)

    # ------------------------------------------------------------------
    # Output LayerNorm + gate
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out, (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # Final linear projection back to original dimension
    # ------------------------------------------------------------------
    out_flat = out.view(-1, hidden_dim)                # (B*N*N, H)
    result_fp16 = torch.nn.functional.linear(out_flat, to_out_w)   # (B*N*N, dim)

    result = result_fp16.view(B, N, N, dim).to(torch.float32)

    return result
