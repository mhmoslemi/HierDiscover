"""
TriMul (outgoing) forward implementation.

The heavy part of the TriMul forward pass is the
einsum
    out[i, j, d] = Σ_k left[i, k, d] * right[j, k, d]

which is a batched matrix multiplication over the sequence
dimension (N) for every hidden channel d.  The kernel below
flattens the batch‑and‑hidden dimension (B * hidden_dim) and
computes a standard GEMM (M×K)·(K×N) for each slice using
Triton.  Half‑precision (fp16) is used for the input matrices
to reduce bandwidth, while accumulation is performed in
float32 for numerical stability.  The rest of the TriMul
pipeline (LayerNorms, linear projections, gating) is handled
with PyTorch ops.

The kernel processes blocks of size (BLOCK_M, BLOCK_N) with a
reduction over K in tiles of BLOCK_K.  A third program‑id
dimension enumerates the flattened batch/hidden dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul einsum
# ----------------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel(
    a_ptr,                     # left matrix  (M, N, N)  – fp16
    b_ptr,                     # right matrix (M, N, N)  – fp16 (already transposed)
    c_ptr,                     # output matrix (M, N, N) – fp32
    N,                         # sequence length (same for rows/cols/reduction)
    stride_am, stride_ai, stride_ak,   # strides for a_ptr (batch, row, col)
    stride_bm, stride_bk, stride_bn,   # strides for b_ptr (batch, row, col)
    stride_cm, stride_ci, stride_cj,   # strides for c_ptr (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # block row index
    pid_n = tl.program_id(1)   # block col index
    pid_b = tl.program_id(2)   # which flattened (batch, hidden) matrix

    # ------------------------------------------------------------------
    # Offsets for the output tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Reduce over the K dimension (the sequence dimension)
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # pointers for A (left)   shape: (BLOCK_M, BLOCK_K)
        a_ptrs = (
            a_ptr
            + pid_b * stride_am
            + offs_m[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        # pointers for B (right)  shape: (BLOCK_K, BLOCK_N)
        b_ptrs = (
            b_ptr
            + pid_b * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0,
                    dtype=tl.float16)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # tl.dot returns fp16, cast to fp32 before accumulation
        acc += tl.dot(a, b).to(tl.float32)

    # ------------------------------------------------------------------
    # Write back the result
    # ------------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_b * stride_cm
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input, mask, weights, config)
            - input : torch.Tensor [B, N, N, dim]
            - mask  : torch.Tensor [B, N, N] (or None)
            - weights: dict with all layer parameters
            - config : dict with "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Helper to get weight tensors
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3) Optional mask
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(left.dtype)  # broadcast over hidden dim
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4) Gating values (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))   # [B,N,N,hidden_dim]

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Cast to fp16 for the heavy multiplication
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # ------------------------------------------------------------------
    # 6) Rearrange for GEMM: (B, hidden, N, N) where left rows=i, cols=k
    #    and right rows=k, cols=j (the required transpose)
    # ------------------------------------------------------------------
    left_perm = left_fp16.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]  (i, k)
    right_perm = right_fp16.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N]  (k, j)

    # Flatten batch * hidden into a single GEMM batch dimension
    left_flat = left_perm.view(-1, N, N)   # shape (M, N, N)   M = B * hidden_dim
    right_flat = right_perm.view(-1, N, N)  # same shape
    M = left_flat.shape[0]  # = bs * hidden_dim

    # Allocate output buffer (float32)
    out_flat = torch.empty((M, N, N), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # 7) Launch Triton kernel
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        M,
    )

    trimul_gemm_kernel[
        grid
    ](
        a_ptr=left_flat,
        b_ptr=right_flat,
        c_ptr=out_flat,
        N=N,
        stride_am=left_flat.stride(0),
        stride_ai=left_flat.stride(1),
        stride_ak=left_flat.stride(2),
        stride_bm=right_flat.stride(0),
        stride_bk=right_flat.stride(1),
        stride_bn=right_flat.stride(2),
        stride_cm=out_flat.stride(0),
        stride_ci=out_flat.stride(1),
        stride_cj=out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 8) Reshape back to [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    out = out_flat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 9) Output LayerNorm + gating
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate  # element‑wise gate

    # ------------------------------------------------------------------
    # 10) Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_weight)  # bias=False in the original module

    return out
```
