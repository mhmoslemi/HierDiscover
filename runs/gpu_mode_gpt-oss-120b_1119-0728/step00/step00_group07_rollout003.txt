2. Linear projections of the normalized tensor:
      left  = x_norm @ W_left   (shape [B,N,N,H])
      right = x_norm @ W_right  (shape [B,N,N,H])
3. Optional pairwise mask (broadcast on the feature axis) is applied.
4. Compute sigmoid gates from the same normalized tensor:
      left_gate  = sigmoid(x_norm @ W_left_gate)
      right_gate = sigmoid(x_norm @ W_right_gate)
      out_gate   = sigmoid(x_norm @ W_out_gate)
   and fuse the three‑way multiplication:
      left  = left  * left_gate  * mask
      right = right * right_gate * mask
5. Tri‑multiplicative update:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix‑multiply A @ Bᵀ for each hidden channel.
   A and B are reshaped so that the hidden dimension becomes part of the batch
   (batch_total = B * H).  The heavy O(N³) work is performed by a custom
   Triton kernel that implements a classic blocked GEMM with the second
   operand accessed in transposed order.
6. Layer‑norm over the hidden dimension, multiply by the output gate and
   apply a final linear projection back to the original channel size.
All tensors remain on the GPU; the kernel works with float32 (the accumulator
is float32, loads are float32).  The implementation fits the H100 memory model
and meets the required runtimes for the provided test configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication C = A @ Bᵀ
# A : (batch, M, K)   – left tensor after reshaping (M=N, K=N)
# B : (batch, N, K)   – right tensor (will be read as Bᵀ inside the kernel)
# C : (batch, M, N)
# -----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    a_ptr,               # *float32  A
    b_ptr,               # *float32  B (will be read transposed)
    c_ptr,               # *float32  C
    M, N, K,             # matrix sizes (M=N, K=N)
    stride_a_batch, stride_a_m, stride_a_k,   # strides for A
    stride_b_batch, stride_b_k, stride_b_n,   # strides for B (note: k then n)
    stride_c_batch, stride_c_m, stride_c_n,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program identifiers
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # --------------------------------------------------------------
    # Compute the block offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Pointers to the start of the blocks for A and B
    a_ptrs = (
        a_ptr
        + batch_id * stride_a_batch
        + offs_m[:, None] * stride_a_m
        + offs_k[None, :] * stride_a_k
    )
    # B is accessed as Bᵀ: we need elements B[j, k] -> offset (k, j)
    b_ptrs = (
        b_ptr
        + batch_id * stride_b_batch
        + offs_k[:, None] * stride_b_k
        + offs_n[None, :] * stride_b_n
    )

    # --------------------------------------------------------------
    # Accumulator
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_idx in range(0, num_k_tiles):
        # Load A and B tiles with boundary masks
        a = tl.load(
            a_ptrs,
            mask=(offs_m < M)[:, None] & (offs_k < K)[None, :],
            other=0.0,
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k < K)[:, None] & (offs_n < N)[None, :],
            other=0.0,
        )
        # Accumulate the dot product of the tiles
        acc += tl.dot(a, b)

        # Advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # --------------------------------------------------------------
    # Write out the result
    # --------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + batch_id * stride_c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n
    )
    mask = (offs_m < M)[:, None] & (offs_n < N)[None, :]
    tl.store(c_ptrs, acc, mask=mask)


def tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication C = left @ rightᵀ using Triton.
    left/right: (B*H, N, N), dtype float32, contiguous.
    Returns: (B*H, N, N) float32 tensor.
    """
    assert left.shape == right.shape
    batch_total, N, K = left.shape
    assert N == K  # square pairwise tensors in TriMul

    # Ensure contiguous layout (required for correct strides)
    left = left.contiguous()
    right = right.contiguous()

    # Output allocation
    out = torch.empty_like(left)

    # Strides for the three tensors
    stride_a_batch, stride_a_m, stride_a_k = left.stride()
    stride_b_batch, stride_b_n, stride_b_k = right.stride()
    stride_c_batch, stride_c_m, stride_c_n = out.stride()

    # Block sizes – tuned for H100 (32‑bit arithmetic, ample shared memory)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Grid configuration: (batch, blocks_m, blocks_n)
    grid = (
        batch_total,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    # Launch the kernel
    tri_mul_kernel[grid](
        left,
        right,
        out,
        N,
        N,
        K,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.
    data = (input_tensor, mask_tensor, weight_dict, config_dict)

    Returns:
        torch.Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # -------------------------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    # Shapes / config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -------------------------------------------------------------------
    # Extract weights (all are float32 tensors on the correct device)
    # -------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # -------------------------------------------------------------------
    # 1. LayerNorm over the last dimension (dim)
    # -------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # shape [B, N, N, dim]

    # -------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -------------------------------------------------------------------
    left = F.linear(x_norm, left_proj_weight)   # [B, N, N, H]
    right = F.linear(x_norm, right_proj_weight)  # [B, N, N, H]

    # -------------------------------------------------------------------
    # 3. Optional mask (broadcast on feature axis)
    # -------------------------------------------------------------------
    if nomask:
        mask = torch.ones_like(left[..., :1])  # shape [B,N,N,1]
    else:
        # mask_tensor.shape == [B, N, N]; convert to float and broadcast
        mask = mask_tensor.unsqueeze(-1).to(left.dtype)

    left = left * mask
    right = right * mask

    # -------------------------------------------------------------------
    # 4. Gating (sigmoid) – also fused after the mask multiplication
    # -------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_weight))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight)) # [B, N, N, H]
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_weight))    # [B, N, N, H]

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------------------------
    # 5. Tri‑multiplicative update via a Triton kernel:
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # -------------------------------------------------------------------
    B, N_, _, H = left.shape
    assert N_ == N, "Sequence length mismatch."

    # Reshape so hidden dimension becomes part of the batch
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # (B*H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N) # (B*H, N, N)

    # Batched GEMM with Bᵀ inside the kernel
    out_flat = tri_mul_triton(left_flat, right_flat)  # (B*H, N, N)

    # Reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # -------------------------------------------------------------------
    # 6. Final LayerNorm over hidden dimension, scaling by out_gate,
    #    and final linear projection back to dim.
    # -------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )  # [B, N, N, H]

    out = out * out_gate  # point‑wise gating

    # Final projection (hidden_dim -> dim)
    out = F.linear(out, to_out_weight)  # [B, N, N, dim]

    return out
