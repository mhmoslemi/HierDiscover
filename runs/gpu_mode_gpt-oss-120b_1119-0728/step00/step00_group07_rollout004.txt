@triton.jit
def bmm_tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_i, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program IDs
    pid_batch = tl.program_id(2)
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    # map pid_batch to b and d
    b = pid_batch // D
    d = pid_batch % D

    # Compute base offsets for this batch and hidden dim
    off_batch = b * stride_left_b
    off_d = d * stride_left_d

    # Compute starting indices
    i0 = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j0 = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Compute mask for i and j out-of-bound
    mask_i = i0 < N
    mask_j = j0 < N
    mask_ij = mask_i[:, None] & mask_j[None, :]

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Iterate over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k0 < N

        # Pointers for A tile
        a_ptrs = left_ptr + off_batch + off_d + i0[:, None] * stride_left_i + k0[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float16)

        # Pointers for B tile (transposed)
        b_ptrs = right_ptr + off_batch + off_d + j0[None, :] * stride_right_i + k0[:, None] * stride_right_k
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float16)

        # Compute dot product and accumulate
        acc += tl.dot(a, b)  # shape (BLOCK_M, BLOCK_N)

    # Write back result
    c_ptrs = out_ptr + off_batch + off_d + i0[:, None] * stride_out_i + j0[None, :] * stride_out_j
    tl.store(c_ptrs, acc, mask=mask_ij)

    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_i, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,

    pid_j = tl.program_id(1)

    d = pid_batch % D

    base_left = b * stride_left_b + d * stride_left_d
    base_right = b * stride_right_b + d * stride_right_d

    mask_j = j < N

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # assume dtype float16 or float32
    device = input_tensor.device
    # LayerNorm
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # Projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"]) # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # Mask
    if mask is None or config.get("nomask", False):
        mask_use = torch.ones_like(left[..., 0])
    else:
        mask_use = mask
    mask_use = mask_use.to(left.dtype).unsqueeze(-1)
    left = left * mask_use
    right = right * mask_use

    # Gates
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Convert to half for kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, _, _ = left_fp16.shape

    out_fp = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Compute strides
    stride_left = left_fp16.stride()
    stride_right = right_fp16.stride()
    stride_out = out_fp.stride()

    # Define block sizes
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = ( (N + BLOCK_M - 1) // BLOCK_M,
             (N + BLOCK_N - 1) // BLOCK_N,
             B * hidden_dim )
    batched_matmul_tri_mul_kernel[grid](
        left_fp16, right_fp16, out_fp,
        B, N, hidden_dim,
        stride_left[0], stride_left[1], stride_left[2], stride_left[3],
        stride_right[0], stride_right[1], stride_right[2], stride_right[3],
        stride_out[0], stride_out[1], stride_out[2], stride_out[3],
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8
    )

    # Apply to_out_norm
    out_norm = torch.nn.functional.layer_norm(out_fp, (hidden_dim,),
                                               weight=weights["to_out_norm.weight"],
                                               bias=weights["to_out_norm.bias"])

    out_norm = out_norm * out_gate

    output = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

    return output

"""
TriMul (outgoing) forward pass implemented with a custom Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor X ∈ (B, N, N, C).
2. Linear projections to hidden dimension H:
        L = X·W_left   (B, N, N, H)
        R = X·W_right  (B, N, N, H)
3. Optional pairwise mask (broadcast on H) is applied to L and R.
4. Gated scalings:
        L ← L ⊙ σ(X·W_left_gate)
        R ← R ⊙ σ(X·W_right_gate)
   (σ = sigmoid)
5. Core “triangle multiplication”:
        Out[b,i,j,h] = Σ_k L[b,i,k,h] * R[b,j,k,h]       (∑ over k)
   This is a batched matrix multiplication where each (b,h) slice
   computes L_slice @ R_sliceᵀ.  The heavy work is performed by the
   Triton kernel `tri_mul_bmm_kernel`.
6. Layer‑norm on the hidden dimension, gated by out_gate = σ(X·W_out_gate).
7. Final linear projection back to C dimensions.

The Triton kernel computes many independent GEMMs (B·H of size N×N)
in a single launch, handling the transpose of the right operand
by adjusting the pointer arithmetic.  Computations are carried out
in fp16 for bandwidth, accumulation in fp32 for numerical stability,
and the result is written as fp32.

All tensors are assumed to reside on the CUDA device.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the triangle‑multiplicative update
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_bmm_kernel(
    left_ptr, right_ptr, out_ptr,          # pointers
    B, N, H,                               # batch, seq_len, hidden_dim
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """ Compute (B*H) matrix products A @ Bᵀ where A and B are N×N.
        A = left[b, :, :, h]      shape (N, N)
        B = right[b, :, :, h]ᵀ   shape (N, N)
        out = result[b, :, :, h] shape (N, N)
    """
    # ------------------------------------------------------------------
    # Program IDs
    # ------------------------------------------------------------------
    pid_batch = tl.program_id(2)           # runs over B*H fused batches
    pid_i = tl.program_id(0)               # row block of the output
    pid_j = tl.program_id(1)               # column block of the output

    # ------------------------------------------------------------------
    # Decode fused batch index into (b, h)
    # ------------------------------------------------------------------
    b = pid_batch // H
    h = pid_batch % H

    # Base offsets for the current (b, h) slice
    base_left  = b * stride_left_b  + h * stride_left_h
    base_right = b * stride_right_b + h * stride_right_h
    base_out   = b * stride_out_b   + h * stride_out_h

    # ------------------------------------------------------------------
    # Row / column coordinates for this tile
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds rows / columns
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_ij = mask_i[:, None] & mask_j[None, :]

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension k
    # ------------------------------------------------------------------
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k_idx in range(0, num_k_blocks):
        offs_k = k_idx * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ------------------------------------------------------------------
        # Load A tile (left) : shape (BLOCK_M, BLOCK_K)
        # ------------------------------------------------------------------
        a_ptrs = left_ptr + base_left \
                 + offs_i[:, None] * stride_left_i \
                 + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0, dtype=tl.float16)

        # ------------------------------------------------------------------
        # Load B tile (right, transposed) : shape (BLOCK_K, BLOCK_N)
        # We need right[b, j, k, h] which corresponds to the transpose.
        # ------------------------------------------------------------------
        b_ptrs = right_ptr + base_right \
                 + offs_j[None, :] * stride_right_i \
                 + offs_k[:, None] * stride_right_k
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0, dtype=tl.float16)

        # ------------------------------------------------------------------
        # Multiply‑accumulate
        # ------------------------------------------------------------------
        acc += tl.dot(a, b)   # fp16·fp16 -> fp32 accumulation

    # ------------------------------------------------------------------
    # Store the result tile
    # ------------------------------------------------------------------
    c_ptrs = out_ptr + base_out \
             + offs_i[:, None] * stride_out_i \
             + offs_j[None, :] * stride_out_j
    tl.store(c_ptrs, acc, mask=mask_ij)


# ----------------------------------------------------------------------
# Python entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output tensor of shape (B, N, N, dim)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # LayerNorm on the input (dim is the last dimension)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # Linear projections (dim -> hidden_dim)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Apply mask if present (mask shape (B, N, N))
    # ------------------------------------------------------------------
    if mask_tensor is None or config.get("nomask", False):
        mask = torch.ones((B, N, N), dtype=left.dtype, device=device)
    else:
        mask = mask_tensor.to(left.dtype)
    mask = mask.unsqueeze(-1)                     # (B, N, N, 1)
    left = left * mask
    right = right * mask

    # ------------------------------------------------------------------
    # Gating scalars (sigmoid of linear(x))
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Convert to fp16 for the GEMM kernel (memory‑bandwidth friendly)
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # ------------------------------------------------------------------
    # Allocate output buffer (B, N, N, hidden_dim) in fp32
    # ------------------------------------------------------------------
    out_hidden = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # Strides (in elements, not bytes)
    # ------------------------------------------------------------------
    s_left = left_fp16.stride()
    s_right = right_fp16.stride()
    s_out = out_hidden.stride()
    # (B, N, N, H) layout -> strides: (N*N*H, N*H, H, 1)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = s_left
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = s_right
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = s_out

    # ------------------------------------------------------------------
    # Triton launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,                     # blocks over rows i
        (N + BLOCK_N - 1) // BLOCK_N,                     # blocks over cols j
        B * hidden_dim,                                   # fused batch / hidden dimension
    )

    # ------------------------------------------------------------------
    # Run the kernel
    # ------------------------------------------------------------------
    tri_mul_bmm_kernel[grid](
        left_fp16,
        right_fp16,
        out_hidden,
        B,
        N,
        hidden_dim,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_h,
        stride_right_b,
        stride_right_i,
        stride_right_k,
        stride_right_h,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,      # good default for fp16 matmul on H100
    )

    # ------------------------------------------------------------------
    # Post‑processing: hidden‑dim layer norm, output gate, final projection
    # ------------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out_norm = out_norm * out_gate
    output = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

    return output
