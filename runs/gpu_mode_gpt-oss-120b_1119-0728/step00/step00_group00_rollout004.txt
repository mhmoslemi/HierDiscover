def trimul_kernel(left_ptr, right_ptr, out_ptr,
                  B, N, H,
                  stride_left_b, stride_left_i, stride_left_k, stride_left_h,
                  stride_right_b, stride_right_j, stride_right_k, stride_right_h,
                  stride_out_b, stride_out_i, stride_out_j, stride_out_h,
                  BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid = tl.program_id(0)
    total_batches = B * H
    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)
    # compute block indices
    batch_id = pid // (num_i * num_j)
    pid_rem = pid % (num_i * num_j)
    i_block = pid_rem // num_j
    j_block = pid_rem % num_j
    # compute batch and hidden index
    b = batch_id // H
    h = batch_id % H
    # compute start offsets
    i0 = i_block * BLOCK_M
    j0 = j_block * BLOCK_N

    # compute offsets for left and right base pointers
    # base offset for given b and h:
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # to accumulate in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k_offset in range(0, N, BLOCK_K):
        # compute actual block size for tails
        cur_k = tl.where(k_offset + BLOCK_K <= N, BLOCK_K, N - k_offset)
        # but we can just set loads masked with k < N
        # offsets for left: i, k
        i_index = i0 + tl.arange(0, BLOCK_M)[:, None]
        k_index = k_offset + tl.arange(0, BLOCK_K)[None, :]
        left_ptrs = left_base + i_index * stride_left_i + k_index * stride_left_k
        # offsets for right: j, k (need transposed to (k,j))
        j_index = j0 + tl.arange(0, BLOCK_N)[None, :]
        # right tile as (k, j)
        right_ptrs = right_base + (k_offset + tl.arange(0, BLOCK_K)[:, None]) * stride_right_k + (j0 + tl.arange(0, BLOCK_N)[None, :]) * stride_right_j

        # use mask for out-of-bound
        mask_i = i0 + tl.arange(0, BLOCK_M) < N
        mask_j = j0 + tl.arange(0, BLOCK_N) < N

import torch
import triton
import triton.language as tl

@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    """Compute out[b,i,j,h] = sum_{k} left[b,i,k,h] * right[b,j,k,h]"""
    pid = tl.program_id(0)
    total_bh = B * H
    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)

    bh = pid // (num_i * num_j)          # combined batch*hidden id
    pid_rem = pid % (num_i * num_j)
    i_block = pid_rem // num_j
    j_block = pid_rem % num_j

    # decode b and h from bh
    b = bh // H
    h = bh % H

    # compute base pointers for this batch & hidden
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # compute the start coordinates for the tile
    i0 = i_block * BLOCK_M
    j0 = j_block * BLOCK_N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k axis
    for k_offset in range(0, N, BLOCK_K):
        # compute valid masks for load
        cur_k = k_offset + tl.arange(0, BLOCK_K)
        cur_i = i0 + tl.arange(0, BLOCK_M)[:, None]
        cur_j = j0 + tl.arange(0, BLOCK_N)[None, :]

        # mask for left tile (i,k)
        mask_i = cur_i < N
        mask_k = cur_k < N
        mask_left = mask_i & mask_k

        # mask for right tile (k,j)
        mask_j = cur_j < N
        mask_right = mask_k[:, None] & mask_j[None, :]

        # compute pointer offsets
        left_ptrs = left_base + cur_i * stride_left_i + cur_k * stride_left_k
        right_ptrs = right_base + cur_k[:, None] * stride_right_k + cur_j * stride_right_i

        # load as FP16 and promote to FP32 on the fly
        left_tile = tl.load(left_ptrs, mask=mask_left, other=0.0).to(tl.float32)
        right_tile = tl.load(right_ptrs, mask=mask_right, other=0.0).to(tl.float32)

        acc += tl.dot(left_tile, right_tile)

    # store result
    out_i = i0 + tl.arange(0, BLOCK_M)[:, None]
    out_j = j0 + tl.arange(0, BLOCK_N)[None, :]
    mask_store = (out_i < N) & (out_j < N)
    out_ptrs = out_base + out_i * stride_out_i + out_j * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_store)

    Perform the outgoing TriMul operation using a fused Triton kernel for the
    quadratic multiplication over the sequence dimension.
    """
    import torch

"""
TriMul (outgoing) forward pass.

The heavy part of the TriMul module is the quadratic ``einsum``:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

We treat every (batch, hidden) pair as an independent matrix‑multiply:
    left[b,:, :,d]  shape (N, N)
    right[b,:, :,d] shape (N, N) → we need its transpose on the
    second axis, i.e. right_T[k, j] = right[j, k].

A custom Triton kernel computes the batched GEMM
    out = left @ right_T
for all hidden dimensions in parallel.  The kernel works on
tiles (BLOCK_M × BLOCK_N) and iterates over the K‑dimension in
chunks of BLOCK_K, accumulating in FP32 while loading in FP16.
All remaining operations (layer‑norms, linear projections,
gating) are performed in PyTorch.
"""

import torch
import triton
import triton.language as tl

# --------------------------------------------------------------------------
# Triton kernel for the quadratic multiplication (batch * hidden) x N x N
# --------------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr,          # *fp16  [B, N, N, H]
    right_ptr,         # *fp16  [B, N, N, H]
    out_ptr,           # *fp32  [B, N, N, H]
    B, N, H,           # runtime scalars
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].

    The kernel tiles the i‑j plane with (BLOCK_M, BLOCK_N) blocks and
    iterates over k in steps of BLOCK_K.  All batch × hidden slices are
    flattened into a single dimension for launching.
    """
    pid = tl.program_id(0)

    total_bh = B * H                     # combined batch‑hidden dimension
    num_i = tl.cdiv(N, BLOCK_M)          # number of i‑tiles
    num_j = tl.cdiv(N, BLOCK_N)          # number of j‑tiles

    # Decode (batch, hidden) from the program id
    bh_id   = pid // (num_i * num_j)               # 0 … total_bh‑1
    rem     = pid % (num_i * num_j)
    i_block = rem // num_j
    j_block = rem % num_j

    b = bh_id // H                                 # batch index
    h = bh_id % H                                  # hidden index

    # Base pointers for this (b, h) slice
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    i0 = i_block * BLOCK_M
    j0 = j_block * BLOCK_N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the K dimension
    # ------------------------------------------------------------------
    for k_offset in range(0, N, BLOCK_K):
        # Offsets within the current tile
        i_idx = i0 + tl.arange(0, BLOCK_M)[:, None]          # (M, 1)
        j_idx = j0 + tl.arange(0, BLOCK_N)[None, :]          # (1, N)
        k_idx = k_offset + tl.arange(0, BLOCK_K)             # (K,)

        # Masks for out‑of‑bounds loads
        mask_i = i_idx < N
        mask_j = j_idx < N
        mask_k = k_idx < N

        # ---- left tile: shape (M, K) ----
        left_ptrs = left_base + i_idx * stride_left_i + k_idx[None, :] * stride_left_k
        mask_left = mask_i & mask_k[None, :]
        left_tile = tl.load(left_ptrs, mask=mask_left, other=0.0).to(tl.float32)

        # ---- right tile (transposed): shape (K, N) ----
        # right[j, k] is stored with stride_right_i for j and stride_right_k for k.
        right_ptrs = right_base + k_idx[:, None] * stride_right_k + j_idx * stride_right_i
        mask_right = mask_k[:, None] & mask_j[None, :]
        right_tile = tl.load(right_ptrs, mask=mask_right, other=0.0).to(tl.float32)

        # Accumulate the block product
        acc += tl.dot(left_tile, right_tile)   # (M, N)

    # ------------------------------------------------------------------
    # Store the result
    # ------------------------------------------------------------------
    out_i = i0 + tl.arange(0, BLOCK_M)[:, None]
    out_j = j0 + tl.arange(0, BLOCK_N)[None, :]
    mask_store = (out_i < N) & (out_j < N)

    out_ptrs = out_base + out_i * stride_out_i + out_j * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_store)


# --------------------------------------------------------------------------
# Python entry point ----------------------------------------------------------------
def custom_kernel(data):
    """
    Perform the outgoing TriMul operation.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
              - input_tensor: Tensor[B, N, N, dim] (float32)
              - mask_tensor : Tensor[B, N, N]    (float32 or bool)
              - weights_dict: dictionary with all model weights
              - config_dict : contains 'dim', 'hidden_dim' and 'nomask'

    Returns:
        Tensor[B, N, N, dim] (float32) – the forward result of the TriMul module.
    """
    import torch

    input_tensor, mask_tensor, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=dim,
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (skip if config["nomask"] is True)
    # ------------------------------------------------------------------
    if not config.get("nomask", False):
        mask_exp = mask_tensor.unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gating
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Heavy quadratic multiplication via Triton
    # ------------------------------------------------------------------
    # Cast to fp16 for bandwidth‑saving (kernel accumulates in fp32)
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Output buffer (FP32 as required by the rest of the pipeline)
    out = torch.empty((B, N, N, hidden), dtype=torch.float32, device=input_tensor.device)

    # Strides in element units (not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left_fp16.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right_fp16.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # Tile sizes – chosen to fit H100 registers and achieve good occupancy
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Grid configuration
    num_i = (N + BLOCK_M - 1) // BLOCK_M
    num_j = (N + BLOCK_N - 1) // BLOCK_N
    total_batches = B * hidden
    total_blocks = total_batches * num_i * num_j

    trimul_kernel[total_blocks](
        left_fp16,
        right_fp16,
        out,
        B,
        N,
        hidden,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_h,
        stride_right_b,
        stride_right_i,
        stride_right_k,
        stride_right_h,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. Post‑processing: LayerNorm, gating, final linear
    # ------------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out,
        normalized_shape=hidden,
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out_norm * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
