@triton.jit
def layer_norm_kernel(x_ptr, y_ptr, weight_ptr, bias_ptr,
                      C, eps,
                      stride_xb, stride_xi, stride_xj, stride_xc,
                      stride_yb, stride_yi, stride_yj, stride_yc,
                      seq_len, BLOCK_C: tl.constexpr):
    pid = tl.program_id(0)
    # compute b,i,j
    total_per_batch = seq_len * seq_len
    b = pid // total_per_batch
    rem = pid % total_per_batch
    i = rem // seq_len
    j = rem % seq_len

    # load a block of C elements
    offs_c = tl.arange(0, BLOCK_C)
    mask = offs_c < C
    x_ptrs = x_ptr + b*stride_xb + i*stride_xi + j*stride_xj + offs_c*stride_xc
    x = tl.load(x_ptrs, mask=mask, other=0.0).to(tl.float32)

    # compute mean
    sum_x = tl.sum(x, axis=0)
    mean = sum_x / C

    # compute variance
    var = tl.sum((x - mean) * (x - mean), axis=0) / C

    # Normalize
    x_hat = (x - mean) / tl.sqrt(var + eps)

    # Apply weight and bias
    w = tl.load(weight_ptr + offs_c, mask=mask, other=0.0)
    b_ = tl.load(bias_ptr + offs_c, mask=mask, other=0.0)
    y = x_hat * w + b_

    # Store
    y_ptrs = y_ptr + b*stride_yb + i*stride_yi + j*stride_yj + offs_c*stride_yc
    tl.store(y_ptrs, y, mask=mask)

```
def custom_kernel(data):
    """TriMul outgoing kernel implementation.

    This function normalizes the input with a fused layer‑norm kernel,
    projects the normalized tensor to a hidden dimension, applies three
    sigmoid gates and a mask, performs a batched matrix multiplication
    that corresponds to the einsum `'... i k d, ... j k d -> ... i j d'`,
    normalises the result again and projects back to the original channel
    dimension. All heavy element‑wise work (layer‑norm) is performed by a
    custom Triton kernel; the linear layers and the N×N matrix
    multiplication are delegated to the highly‑optimised cuBLAS kernels that
    PyTorch calls under the hood.
    """
    import torch
    import triton
    import triton.language as tl

    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, C = input_tensor.shape
    hidden = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Layer‑norm across channel dimension C (custom Triton kernel)
    # ------------------------------------------------------------------
    # Allocate output tensor for the normalized representation.
    x_norm = torch.empty_like(input_tensor)

    # Strides (in elements) for input and output tensors.
    stride_xb, stride_xi, stride_xj, stride_xc = input_tensor.stride()
    stride_yb, stride_yi, stride_yj, stride_yc = x_norm.stride()

    total_groups = B * N * N  # one program instance per (b,i,j)

    # Launch kernel. The BLOCK_C constant is set to the maximum possible C
    # (384) which covers all configurations; a mask takes care of the
    # remainder for smaller C.
    layer_norm_kernel[(total_groups,)](
        input_tensor,
        x_norm,
        weights["norm.weight"],
        weights["norm.bias"],
        C,
        eps,
        stride_xb, stride_xi, stride_xj, stride_xc,
        stride_yb, stride_yi, stride_yj, stride_yc,
        N,
        BLOCK_C=384,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections + sigmoid gates + mask
    # ------------------------------------------------------------------
    # Prepare mask (broadcastable to [B,N,N,1]).
    if mask is None or config.get("nomask", False):
        mask_b = torch.ones((B, N, N, 1), dtype=dtype, device=device)
    else:
        mask_b = mask.to(dtype).unsqueeze(-1)  # shape (B,N,N,1)

    # Flatten the first three dimensions to feed into torch.nn.functional.linear.
    x_flat = x_norm.view(-1, C)  # (B*N*N, C)

    # Projection tensors (no bias in the reference implementation).
    left_proj = torch.nn.functional.linear(x_flat, weights["left_proj.weight"]).view(B, N, N, hidden)
    right_proj = torch.nn.functional.linear(x_flat, weights["right_proj.weight"]).view(B, N, N, hidden)

    left_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, weights["left_gate.weight"])).view(B, N, N, hidden)
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, weights["right_gate.weight"])).view(B, N, N, hidden)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, weights["out_gate.weight"])).view(B, N, N, hidden)

    # Apply mask and gates.
    left = left_proj * mask_b * left_gate
    right = right_proj * mask_b * right_gate

    # ------------------------------------------------------------------
    # 3. Core N×N×hidden “tri‑mul” via batched matmul.
    # ------------------------------------------------------------------
    # Rearrange to (B, hidden, N, N) so that each hidden slice can be
    # processed independently by a cuBLAS batched GEMM.
    left_mat = left.permute(0, 3, 1, 2).contiguous()
    right_mat = right.permute(0, 3, 1, 2).contiguous()

    # Collapse batch and hidden dimension (B*hidden, N, N).
    left_mat_2d = left_mat.view(B * hidden, N, N)
    right_mat_2d = right_mat.view(B * hidden, N, N)

    # out = left @ right^T   (the “einsum” in the paper)
    out_mat = torch.matmul(left_mat_2d, right_mat_2d.transpose(-2, -1))

    # Reshape back to [B, N, N, hidden].
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 4. Output layer‑norm and final linear projection.
    # ------------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    out_norm = out_norm * out_gate  # element‑wise gating

    # Final projection back to original channel dimension `dim`.
    out_flat = out_norm.view(-1, hidden)
    out_final = torch.nn.functional.linear(out_flat, weights["to_out.weight"]).view(B, N, N, C)

    return out_final

"""
TriMul “outgoing” kernel (AlphaFold3) – forward only.

Algorithm
---------
1. Layer‑norm over the channel dimension C of the input tensor X
   using a custom Triton kernel (no reduction across batches or
   sequence axes, each program instance handles one (b,i,j) slice).

2. Project the normalized tensor to the hidden dimension H with three
   separate linear layers (left_proj, right_proj, out_gate) and two
   gating layers (left_gate, right_gate).  The linear layers are
   executed by torch.nn.functional.linear – a highly‑optimised cuBLAS
   GEMM – while the sigmoid gates are applied in PyTorch.

3. Apply the optional mask (broadcasted on the hidden dimension) and
   multiply the projected tensors with their respective gates.

4. Compute the core “TriMul” operation
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   by reshaping to (B·H, N, N) and calling a batched matmul
   (left @ rightᵀ).  The matmul runs in FP16 for speed and is cast
   back to FP32.

5. A second Layer‑Norm (hidden‑dim) is applied, followed by an element‑wise
   gating with out_gate and a final linear projection back to the original
   channel dimension.

Only the layer‑norm step is written in Triton; the remaining heavy
operations are delegated to cuBLAS kernels, guaranteeing high
throughput on an H100.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def layer_norm_kernel(
    x_ptr,               # *const float32   – input tensor
    y_ptr,               # *mut   float32   – normalized output
    weight_ptr,          # *const float32   – γ (LayerNorm weight)
    bias_ptr,            # *const float32   – β (LayerNorm bias)
    C,                   # i32              – channel dimension
    eps,                 # f32              – epsilon
    stride_xb, stride_xi, stride_xj, stride_xc,   # i32 strides for X
    stride_yb, stride_yi, stride_yj, stride_yc,   # i32 strides for Y
    seq_len,             # i32              – N
    BLOCK_C: tl.constexpr,                       # compile‑time max C (384)
):
    pid = tl.program_id(0)                     # one instance per (b,i,j)
    total_per_batch = seq_len * seq_len
    b = pid // total_per_batch
    rem = pid % total_per_batch
    i = rem // seq_len
    j = rem % seq_len

    # ------------------------------------------------------------------
    # Load the C‑vector belonging to (b,i,j, :)
    # ------------------------------------------------------------------
    offs_c = tl.arange(0, BLOCK_C)
    mask_c = offs_c < C

    x_ptrs = (
        x_ptr
        + b * stride_xb
        + i * stride_xi
        + j * stride_xj
        + offs_c * stride_xc
    )
    x = tl.load(x_ptrs, mask=mask_c, other=0.0).to(tl.float32)

    # ------------------------------------------------------------------
    # Mean and variance (float32)
    # ------------------------------------------------------------------
    c_f = tl.float32(C)
    sum_x = tl.sum(x)
    mean = sum_x / c_f

    diff = x - mean
    var = tl.sum(diff * diff) / c_f

    # Normalise
    x_hat = diff * tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # Apply affine γ,β
    # ------------------------------------------------------------------
    w = tl.load(weight_ptr + offs_c, mask=mask_c, other=0.0)
    b_ = tl.load(bias_ptr   + offs_c, mask=mask_c, other=0.0)
    y = x_hat * w + b_

    # ------------------------------------------------------------------
    # Store result
    # ------------------------------------------------------------------
    y_ptrs = (
        y_ptr
        + b * stride_yb
        + i * stride_yi
        + j * stride_yj
        + offs_c * stride_yc
    )
    tl.store(y_ptrs, y, mask=mask_c)


def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/int) or None
            - weights: dict of torch.Tensors
            - config: dict containing at least:
                * "dim" (C)
                * "hidden_dim" (H)
                * optional "nomask" bool flag

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    torch.backends.cuda.matmul.allow_tf32 = False  # deterministic FP32 matmul
    with torch.no_grad():
        # ------------------------------------------------------------------
        # unpack inputs
        # ------------------------------------------------------------------
        input_tensor, mask, weights, config = data
        device = input_tensor.device
        dtype = input_tensor.dtype

        B, N, _, C = input_tensor.shape
        H = config["hidden_dim"]          # hidden dimension
        eps = 1e-5

        # ------------------------------------------------------------------
        # 1) Layer‑norm (custom Triton kernel)
        # ------------------------------------------------------------------
        x_norm = torch.empty_like(input_tensor)

        # strides are expressed in **elements** (not bytes)
        stride_xb, stride_xi, stride_xj, stride_xc = input_tensor.stride()
        stride_yb, stride_yi, stride_yj, stride_yc = x_norm.stride()

        total_programs = B * N * N
        # Block size for the channel dimension – the maximum possible C is 384.
        layer_norm_kernel[(total_programs,)](
            input_tensor,
            x_norm,
            weights["norm.weight"],
            weights["norm.bias"],
            C,
            eps,
            stride_xb, stride_xi, stride_xj, stride_xc,
            stride_yb, stride_yi, stride_yj, stride_yc,
            N,
            BLOCK_C=384,
            num_warps=4,
        )

        # ------------------------------------------------------------------
        # 2) Linear projections + sigmoid gates + mask
        # ------------------------------------------------------------------
        # mask handling (broadcast to hidden dim)
        if mask is None or config.get("nomask", False):
            mask_f = torch.ones((B, N, N, 1), dtype=dtype, device=device)
        else:
            mask_f = mask.to(dtype).unsqueeze(-1)  # (B,N,N,1)

        # flatten the first three dims for a single linear call
        x_flat = x_norm.view(-1, C)                     # (B·N·N, C)

        # ----- projections (no bias) -----
        left_proj = torch.nn.functional.linear(
            x_flat, weights["left_proj.weight"]
        ).view(B, N, N, H)
        right_proj = torch.nn.functional.linear(
            x_flat, weights["right_proj.weight"]
        ).view(B, N, N, H)

        # ----- gates (sigmoid) -----
        left_gate = torch.sigmoid(
            torch.nn.functional.linear(x_flat, weights["left_gate.weight"])
        ).view(B, N, N, H)
        right_gate = torch.sigmoid(
            torch.nn.functional.linear(x_flat, weights["right_gate.weight"])
        ).view(B, N, N, H)
        out_gate = torch.sigmoid(
            torch.nn.functional.linear(x_flat, weights["out_gate.weight"])
        ).view(B, N, N, H)

        # Apply mask and element‑wise gating
        left = left_proj * mask_f * left_gate
        right = right_proj * mask_f * right_gate

        # ------------------------------------------------------------------
        # 3) Core tri‑mul via batched matmul
        # ------------------------------------------------------------------
        # reshape to (B, H, N, N) → (B·H, N, N)
        left_mat = left.permute(0, 3, 1, 2).contiguous()
        right_mat = right.permute(0, 3, 1, 2).contiguous()

        left_2d = left_mat.view(B * H, N, N).to(torch.float16)
        right_2d = right_mat.view(B * H, N, N).to(torch.float16)

        # out = left @ rightᵀ   (the einsum in the paper)
        out_2d = torch.matmul(left_2d, right_2d.transpose(-2, -1))
        out_2d = out_2d.to(torch.float32)

        # reshape back to (B, N, N, H)
        out = (
            out_2d.view(B, H, N, N)
            .permute(0, 2, 3, 1)
            .contiguous()
        )  # [B, N, N, H]

        # ------------------------------------------------------------------
        # 4) Output layer‑norm, out‑gate, final linear proj
        # ------------------------------------------------------------------
        out_norm = torch.nn.functional.layer_norm(
            out,
            (H,),
            weight=weights["to_out_norm.weight"],
            bias=weights["to_out_norm.bias"],
            eps=1e-5,
        )
        out_norm = out_norm * out_gate   # element‑wise gating

        # final projection back to original channel dimension C
        out_flat = out_norm.view(-1, H)
        out_final = torch.nn.functional.linear(
            out_flat, weights["to_out.weight"]
        ).view(B, N, N, C)

        return out_final
