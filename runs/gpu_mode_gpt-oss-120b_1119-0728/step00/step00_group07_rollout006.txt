@triton.jit
def batched_gemm_kernel(a_ptr, b_ptr, c_ptr,
                        M, N, K,
                        stride_a_batch, stride_a_m, stride_a_k,
                        stride_b_batch, stride_b_k, stride_b_n,
                        stride_c_batch, stride_c_m, stride_c_n,
                        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    batch_id = tl.program_id(0)
    block_i = tl.program_id(1)
    block_j = tl.program_id(2)

    # compute row and col offsets within block
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for valid positions
    mask_i = offs_i < M
    mask_j = offs_j < N

    # pointer offsets for each batch
    a_ptr_base = a_ptr + batch_id * stride_a_batch
    b_ptr_base = b_ptr + batch_id * stride_b_batch
    c_ptr_base = c_ptr + batch_id * stride_c_batch

    # accumulate in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # k range offset
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < K

        # Load A: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_ptr_base + (offs_i[:, None] * stride_a_m) + (k_offset[None, :] * stride_a_k)
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load B (transposed): B dimensions K x N, we need element B[k,j]

"""
TriMul (outgoing) forward pass with a custom Triton kernel.
---------------------------------------------------------

The module follows the reference PyTorch implementation:

1. Layer‑norm over the last channel dimension.
2. Two linear projections (left/right) and three gating linear layers.
3. Optional mask applied to the projected tensors.
4. Gated tensors are multiplied element‑wise.
5. The cubic “einsum” (`... i k d, ... j k d -> ... i j d`) is implemented as a
   batched matrix multiplication `C = A @ Bᵀ` where for each hidden channel
   d we compute `C[i,j] = Σₖ A[i,k,d] * B[j,k,d]`.  This is the expensive O(N³)
   part, and we accelerate it with a Triton GEMM kernel.
6. A second layer‑norm, an output gate, and a final linear projection back to
   the original channel dimension.

Only the heavy matrix multiplication is off‑loaded to Triton; the surrounding
operations stay in PyTorch for simplicity and correctness.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ) where A and B have shape
#   A: (batch, M, K)   -- here M = N, K = N
#   B: (batch, M, K)   -- we interpret Bᵀ on‑the‑fly
#   C: (batch, M, N)   -- N = M in our case (square output)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,              # pointers
    M, N, K,                          # matrix sizes (all == seq_len)
    stride_a_batch, stride_a_m, stride_a_k,   # A strides
    stride_b_batch, stride_b_k, stride_b_n,   # B strides (note: B is accessed as Bᵀ)
    stride_c_batch, stride_c_m, stride_c_n,   # C strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ Bᵀ  (per hidden channel)."""
    batch_id = tl.program_id(0)               # one program per (batch * hidden)
    pid_m = tl.program_id(1)                  # block-row index
    pid_n = tl.program_id(2)                  # block-col index

    # Offsets within the current block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_batch_ptr = a_ptr + batch_id * stride_a_batch
    b_batch_ptr = b_ptr + batch_id * stride_b_batch
    c_batch_ptr = c_ptr + batch_id * stride_c_batch

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_off < K

        # Load A tile [BLOCK_M, BLOCK_K]
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_a_m) + (k_off[None, :] * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Load Bᵀ tile [BLOCK_K, BLOCK_N]
        # B is stored as (batch, M, K); we need Bᵀ => element (k, j) = original (j, k)
        b_ptrs = b_batch_ptr + (k_off[:, None] * stride_b_k) + (offs_n[None, :] * stride_b_n)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Accumulate the block‑product
        acc += tl.dot(a, b)

    # Write result back to C
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_c_m) + (offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point expected by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operation.
    Arguments:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Load model parameters
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. Layer‑norm over channel dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [bs, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3. Optional mask (mask shape: [bs, N, N])
    # ------------------------------------------------------------------
    if not config.get("nomask", False):
        # mask broadcasted over channel dimension
        mask_exp = mask.unsqueeze(-1)          # [bs, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gating tensors (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Batched GEMM for the cubic term (out = Σₖ left[...,i,k,:] * right[...,j,k,:])
    # ------------------------------------------------------------------
    # Rearrange to (batch, hidden, N, N) and flatten the batch*hidden dimension.
    left_perm = left.permute(0, 3, 1, 2).contiguous()    # [bs, hidden, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [bs, hidden, N, N]

    # Merge batch and channel dimensions → (B*H, N, N)
    left_flat = left_perm.view(bs * hidden_dim, seq_len, seq_len)   # A
    right_flat = right_perm.view(bs * hidden_dim, seq_len, seq_len) # B (will be read as Bᵀ)

    # Allocate output tensor
    out_flat = torch.empty_like(left_flat)  # fp32 output

    # Compute strides (they are the same for A, B and C because tensors are contiguous)
    stride_a_batch = left_flat.stride(0)   # N * N
    stride_a_m = left_flat.stride(1)       # N
    stride_a_k = left_flat.stride(2)       # 1

    # For B we treat the inner dimension (k) as the last axis (stride = 1)
    # and the outer dimension (j) as the column axis (stride = N).
    stride_b_batch = right_flat.stride(0)  # N * N
    stride_b_k = right_flat.stride(2)      # 1   (original k axis)
    stride_b_n = right_flat.stride(1)      # N   (original j axis)

    stride_c_batch = out_flat.stride(0)    # N * N
    stride_c_m = out_flat.stride(1)        # N
    stride_c_n = out_flat.stride(2)        # 1

    # Launch Triton kernel
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    grid = (
        bs * hidden_dim,                       # batch dimension (batch * hidden)
        (seq_len + BLOCK_M - 1) // BLOCK_M,    # rows
        (seq_len + BLOCK_N - 1) // BLOCK_N,    # cols
    )
    batched_gemm_kernel[grid](
        a_ptr=left_flat.data_ptr(),
        b_ptr=right_flat.data_ptr(),
        c_ptr=out_flat.data_ptr(),
        M=seq_len,
        N=seq_len,
        K=seq_len,
        stride_a_batch=stride_a_batch,
        stride_a_m=stride_a_m,
        stride_a_k=stride_a_k,
        stride_b_batch=stride_b_batch,
        stride_b_k=stride_b_k,
        stride_b_n=stride_b_n,
        stride_c_batch=stride_c_batch,
        stride_c_m=stride_c_m,
        stride_c_n=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [bs, N, N, hidden_dim]
    out = out_flat.view(bs, hidden_dim, seq_len, seq_len)      # [bs, hidden, N, N]
    out = out.permute(0, 2, 3, 1).contiguous()                # [bs, N, N, hidden]

    # ------------------------------------------------------------------
    # 6. Post‑processing: Layer‑norm, output gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=1e-5)

    out = out * out_gate                                   # gated output

    # Final linear back to original channel dimension (dim)
    out = F.linear(out, to_out_weight)                     # [bs, N, N, dim]

    return out
```
