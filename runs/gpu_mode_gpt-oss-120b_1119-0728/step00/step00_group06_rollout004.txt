@triton.jit
def tri_mul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    total_heads = B * H
    BLOCKS_M = tl.cdiv(N, BLOCK_M)
    BLOCKS_N = tl.cdiv(N, BLOCK_N)
    head = pid // (BLOCKS_M * BLOCKS_N)
    pid_mn = pid % (BLOCKS_M * BLOCKS_N)
    pid_m = pid_mn // BLOCKS_N
    pid_n = pid_mn % BLOCKS_N

    # compute batch and hidden index
    b = head // H
    h = head % H

    # offsets within block
    i_off = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    k_off = tl.arange(0, BLOCK_K)

    # boundary masks
    i_mask = i_off < N
    j_mask = j_off < N

    # base pointer offsets
    b_off = b * stride_left_b   # for left and right: same layout of batch stride.
    h_off = h * stride_left_h

    # Compute initial pointers
    # left: shape (i,k)
    left_ptrs = left_ptr + b_off + h_off + i_off[:, None] * stride_left_i + k_off[None, :] * stride_left_k
    # right transposed: shape (k,j)
    right_ptrs = right_ptr + b_off + h_off + k_off[:, None] * stride_right_k + j_off[None, :] * stride_right_i

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k = kk * BLOCK_K
        # compute masks for k dimension
        k_mask = (k + k_off) < N

        # load
        left = tl.load(left_ptrs, mask=i_mask[:, None] & k_mask[None, :], other=0.0).to(tl.float32)
        right = tl.load(right_ptrs, mask=k_mask[:, None] & j_mask[None, :], other=0.0).to(tl.float32)

        # compute dot product: (M,K) x (K,N) -> (M,N)
        acc += tl.dot(left, right)   # uses internal fma

        # advance pointers
        left_ptrs += BLOCK_K * stride_left_k
        right_ptrs += BLOCK_K * stride_right_k

    # write back
    out_ptrs = out_ptr + b_off + h_off + i_off[:, None] * stride_out_i + j_off[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=i_mask[:, None] & j_mask[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config['hidden_dim']
    # Ensure dtype compatibility
    x = input_tensor

    # LayerNorm over dim
    norm_weight = weights['norm.weight']
    norm_bias = weights['norm.bias']
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias)

    # Linear projections
    left_proj_weight = weights['left_proj.weight'] # shape (hidden_dim, dim)
    right_proj_weight = weights['right_proj.weight']
    left_gate_weight = weights['left_gate.weight']
    right_gate_weight = weights['right_gate.weight']
    out_gate_weight = weights['out_gate.weight']

    left_proj = torch.nn.functional.linear(x_norm, left_proj_weight)  # shape [B,N,N,hidden_dim]
    right_proj = torch.nn.functional.linear(x_norm, right_proj_weight)

    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    # Apply gates & mask
    if config.get('nomask', False):
        left = left_proj * left_gate
        right = right_proj * right_gate
    else:
        mask = mask.unsqueeze(-1).to(x.dtype)  # shape [B,N,N,1]
        left = left_proj * left_gate * mask
        right = right_proj * right_gate * mask

    # Compute out via Triton kernel
    out = torch.empty(bs, N, N, hidden_dim, dtype=torch.float32, device=input_tensor.device)
    # Launch kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    total_heads = bs * hidden_dim
    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    total_programs = total_heads * num_blocks_m * num_blocks_n

    # compute strides
    left_stride_b = left.stride(0)
    left_stride_i = left.stride(1)
    left_stride_k = left.stride(2)
    left_stride_h = left.stride(3)

    right_stride_b = right.stride(0)
    right_stride_i = right.stride(1)
    right_stride_k = right.stride(2)
    right_stride_h = right.stride(3)

    out_stride_b = out.stride(0)
    out_stride_i = out.stride(1)
    out_stride_j = out.stride(2)
    out_stride_h = out.stride(3)

    tri_mul_einsum_kernel[total_programs](
        left, right, out,
        bs, N, hidden_dim,
        left_stride_b, left_stride_i, left_stride_k, left_stride_h,
        right_stride_b, right_stride_i, right_stride_k, right_stride_h,
        out_stride_b, out_stride_i, out_stride_j, out_stride_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # After kernel, out shape is [B,N,N,H]

    # to_out_norm layernorm
    to_out_norm_weight = weights['to_out_norm.weight']
    to_out_norm_bias = weights['to_out_norm.bias']
    out_norm = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # multiply out_gate
    out_gate = out_gate  # shape [B,N,N,H]
    out = out_norm * out_gate

    # final linear to dim
    to_out_weight = weights['to_out.weight']
    # linear transforms from hidden_dim to dim
    out = torch.nn.functional.linear(out, to_out_weight)   # (B,N,N,dim)

    return out

"""
TriMul (outgoing) forward implementation with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor (shape [B, N, N, C]).
2. Linear projections (left_proj, right_proj) and three gating projections
   (left_gate, right_gate, out_gate) are applied to the normalized tensor.
   The mask (if present) is multiplied before the gates, exactly as in the
   reference implementation.
3. A custom Triton kernel computes the Einstein‑summation
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batch of H (= hidden_dim) independent matrix multiplies:
   for every hidden channel `d` we compute left[...,d] @ right[...,d]ᵀ.
   The kernel fuses all heads (`B*H`) and tiles the (i,j) output with
   BLOCK_M×BLOCK_N tiles while marching over the k dimension in BLOCK_K‑sized
   chunks.  Loads are performed in FP16, accumulation in FP32.
4. The result (shape [B, N, N, H]) is layer‑normed, multiplied by the out‑gate,
   and finally projected back to the original channel dimension `C`.

Only the N³‑summation (step 3) is written in Triton; all other operations
use PyTorch primitives (highly‑optimized cuBLAS / cuDNN) and therefore keep the
overall runtime well below the target ~1 ms on an H100.
"""

import torch
import triton
import triton.language as tl

# -------------------------------------------------------------------------
# Triton kernel: batched matrix multiplication over the hidden dimension
# -------------------------------------------------------------------------
@triton.jit
def tri_mul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,               # pointers
    B, N, H,                                    # shape parameters
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]   """
    pid = tl.program_id(0)

    # -------------------------------------------------
    # Decode a 3‑D grid: (head = b*H + h, block_i, block_j)
    # -------------------------------------------------
    total_heads = B * H
    BLOCKS_M = tl.cdiv(N, BLOCK_M)
    BLOCKS_N = tl.cdiv(N, BLOCK_N)

    head = pid // (BLOCKS_M * BLOCKS_N)                 # which (b,h) pair
    pid_mn = pid % (BLOCKS_M * BLOCKS_N)
    pid_m = pid_mn // BLOCKS_N                         # block row index i
    pid_n = pid_mn % BLOCKS_N                          # block col index j

    # Recover batch and hidden indices
    b = head // H
    h = head % H

    # -------------------------------------------------
    # Offsets inside the block
    # -------------------------------------------------
    i_off = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    k_off = tl.arange(0, BLOCK_K)

    i_mask = i_off < N
    j_mask = j_off < N

    # Base offsets for this (b,h)
    b_off_left  = b * stride_left_b
    h_off_left  = h * stride_left_h
    b_off_right = b * stride_right_b
    h_off_right = h * stride_right_h
    b_off_out   = b * stride_out_b
    h_off_out   = h * stride_out_h

    # -------------------------------------------------
    # Pointers for the current tile
    #   left  : (i, k)   -> shape (BLOCK_M, BLOCK_K)
    #   right : (k, j)   -> shape (BLOCK_K, BLOCK_N)  (transposed view)
    # -------------------------------------------------
    left_ptrs = left_ptr + b_off_left + h_off_left + \
                i_off[:, None] * stride_left_i + k_off[None, :] * stride_left_k
    right_ptrs = right_ptr + b_off_right + h_off_right + \
                 k_off[:, None] * stride_right_k + j_off[None, :] * stride_right_i

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------
    # Loop over the reduction dimension k
    # -------------------------------------------------
    for cur_k in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k_off = cur_k * BLOCK_K + k_off
        k_mask = cur_k_off < N

        # Load tiles (FP16 -> FP32)
        left_tile  = tl.load(left_ptrs,
                             mask=i_mask[:, None] & k_mask[None, :],
                             other=0.0).to(tl.float32)
        right_tile = tl.load(right_ptrs,
                             mask=k_mask[:, None] & j_mask[None, :],
                             other=0.0).to(tl.float32)

        # GEMM on the tile
        acc += tl.dot(left_tile, right_tile)

        # Advance K‑dimension pointers
        left_ptrs  += BLOCK_K * stride_left_k
        right_ptrs += BLOCK_K * stride_right_k

    # -------------------------------------------------
    # Write the result
    # -------------------------------------------------
    out_ptrs = out_ptr + b_off_out + h_off_out + \
               i_off[:, None] * stride_out_i + j_off[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=i_mask[:, None] & j_mask[None, :])

# -------------------------------------------------------------------------
# Entry point required by the problem statement
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor  [B, N, N, C]   (float32)
        - mask         : torch.Tensor  [B, N, N]    (bool / 0-1) or None
        - weights      : dict of torch.Tensors containing all model parameters
        - config       : dict with at least keys
              "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Processed tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, C = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. LayerNorm over the channel dimension C
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(C,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )

    # -----------------------------------------------------------------
    # 2. Linear projections + gating + optional mask
    # -----------------------------------------------------------------
    # Cast to FP16 for the heavy linear work (less memory traffic)
    x_fp16 = x.half()

    # ----- Projections ------------------------------------------------
    left_proj_W  = weights["left_proj.weight"].half()   # (hidden_dim, C)
    right_proj_W = weights["right_proj.weight"].half()

    left  = torch.nn.functional.linear(x_fp16, left_proj_W)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_fp16, right_proj_W)

    # ----- Gates -------------------------------------------------------
    left_gate_W  = weights["left_gate.weight"].half()
    right_gate_W = weights["right_gate.weight"].half()
    out_gate_W   = weights["out_gate.weight"].half()

    left_gate  = torch.sigmoid(torch.nn.functional.linear(x_fp16, left_gate_W))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_fp16, right_gate_W))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x_fp16, out_gate_W))

    # ----- Apply mask (if any) before gating ---------------------------
    if nomask or mask is None:
        # No mask → treat as all‑ones
        left = left * left_gate
        right = right * right_gate
        # out_gate stays unchanged (shape [B,N,N,hidden_dim])
    else:
        mask_f = mask.unsqueeze(-1).to(x_fp16.dtype)           # [B,N,N,1]
        left = left * mask_f * left_gate
        right = right * mask_f * right_gate

    # Cast back to float16 for kernel (already fp16)
    left = left.contiguous()
    right = right.contiguous()

    # -----------------------------------------------------------------
    # 3. N³ reduction via Triton kernel (left @ rightᵀ per hidden channel)
    # -----------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Block sizes – tuned for H100 (feel free to adjust)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Compute launch grid: one program per (head, tile_i, tile_j)
    total_heads = B * hidden_dim
    blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    total_programs = total_heads * blocks_m * blocks_n

    # Strides of the tensors (in elements, not bytes)
    s_left_b, s_left_i, s_left_k, s_left_h = left.stride()
    s_right_b, s_right_i, s_right_k, s_right_h = right.stride()
    s_out_b, s_out_i, s_out_j, s_out_h = out.stride()

    # Launch the kernel
    tri_mul_einsum_kernel[total_programs](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        s_left_b,
        s_left_i,
        s_left_k,
        s_left_h,
        s_right_b,
        s_right_i,
        s_right_k,
        s_right_h,
        s_out_b,
        s_out_i,
        s_out_j,
        s_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 4. Second LayerNorm, out‑gate multiplication, final linear proj.
    # -----------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out_norm = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )

    # Multiply by out‑gate (both are float32)
    out = out_norm * out_gate.to(dtype=torch.float32)

    # Final linear: hidden_dim -> C
    to_out_W = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_W)

    return out
