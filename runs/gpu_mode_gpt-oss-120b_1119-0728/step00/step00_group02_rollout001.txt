@triton.jit
def triangle_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_rj, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)  # combined batch * hidden
    block_i = tl.program_id(1)
    block_j = tl.program_id(2)

    # compute b and h indices
    b = pid // H
    h = pid % H

    # compute start offsets for i and j
    i_start = block_i * BLOCK_M
    j_start = block_j * BLOCK_N

    # range of i/j indices
    i = i_start + tl.arange(0, BLOCK_M)
    j = j_start + tl.arange(0, BLOCK_N)
    i_mask = i < N
    j_mask = j < N

    # initialize accumulator 
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k axis in blocks
    for k_start in range(0, N, BLOCK_K):
        k = k_start + tl.arange(0, BLOCK_K)
        k_mask = k < N

        # load left block [M, K]
        left_ptrs = left_ptr + b * stride_lb + h * stride_lh + i[:, None] * stride_li + k[None, :] * stride_lk
        left = tl.load(left_ptrs, mask=i_mask[:, None] & k_mask[None, :], other=0.0)
        left = left.to(tl.float32)

        # load right block [N, K] (actually j x k)
        right_ptrs = right_ptr + b * stride_rb + h * stride_rh + j[:, None] * stride_rj + k[None, :] * stride_rk
        right = tl.load(right_ptrs, mask=j_mask[:, None] & k_mask[None, :], other=0.0)
        right = right.to(tl.float32)

        # accumulate dot product over K (left @ right.T)
        # left shape (M,K), right shape (N,K)
        acc += tl.dot(left, right, trans_b=True)  # M,N 

    # store accumulate to output
    out_ptrs = out_ptr + b * stride_ob + h * stride_oh + i[:, None] * stride_oi + j[None, :] * stride_oj
    tl.store(out_ptrs, acc, mask=i_mask[:, None] & j_mask[None, :])

"""
TriMul (outgoing) forward pass with a fused Triton kernel.

The kernel computes the core O(N³) operation
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
for every batch `b` and hidden channel `h`.  A 3‑D grid is used:
* program_id(0)  – combined batch × hidden dimension
* program_id(1)  – tile along the first sequence axis (i)
* program_id(2)  – tile along the second sequence axis (j)

Each tile (BLOCK_M × BLOCK_N) accumulates the inner product over K in
blocks of size BLOCK_K using `tl.dot` (which maps to a fast MMA on H100).
All other operations (LayerNorm, Linear projections, gating, final
normalisation/linear) are carried out with native PyTorch for simplicity.

The function `custom_kernel` receives a tuple
    (input_tensor, mask_tensor, weights_dict, config_dict)
and returns the processed tensor with shape [B, N, N, dim].
"""
import torch
import triton
import triton.language as tl
import torch.nn.functional as F


@triton.jit
def triangle_mul_kernel(
    left_ptr,               # const float*  left  [B,N,N,H]
    right_ptr,              # const float*  right [B,N,N,H]
    out_ptr,                # float*        out   [B,N,N,H]
    B, N, H,                # runtime sizes
    stride_lb, stride_li, stride_lk, stride_lh,   # left  strides
    stride_rb, stride_rj, stride_rk, stride_rh,   # right strides
    stride_ob, stride_oi, stride_oj, stride_oh,   # out   strides
    BLOCK_M: tl.constexpr,  # tile size along i
    BLOCK_N: tl.constexpr,  # tile size along j
    BLOCK_K: tl.constexpr,  # tile size along k
):
    # ------------------------------------------------------------------
    # Program IDs
    # ------------------------------------------------------------------
    pid = tl.program_id(0)          # combined batch * hidden
    pid_i = tl.program_id(1)        # block i  (first sequence axis)
    pid_j = tl.program_id(2)        # block j  (second sequence axis)

    # decode batch and hidden indices
    b = pid // H
    h = pid % H

    # ------------------------------------------------------------------
    # Tile coordinates
    # ------------------------------------------------------------------
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    i = i_start + tl.arange(0, BLOCK_M)      # [BLOCK_M]
    j = j_start + tl.arange(0, BLOCK_N)      # [BLOCK_N]

    i_mask = i < N
    j_mask = j < N

    # ------------------------------------------------------------------
    # Accumulator (fp32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension k
    # ------------------------------------------------------------------
    num_k = tl.cdiv(N, BLOCK_K)
    for k_iter in range(num_k):
        k_offset = k_iter * BLOCK_K
        k = k_offset + tl.arange(0, BLOCK_K)   # [BLOCK_K]
        k_mask = k < N

        # left tile   shape (BLOCK_M, BLOCK_K)
        left_ptrs = (
            left_ptr
            + b * stride_lb
            + h * stride_lh
            + i[:, None] * stride_li
            + k[None, :] * stride_lk
        )
        left = tl.load(
            left_ptrs,
            mask=i_mask[:, None] & k_mask[None, :],
            other=0.0,
        ).to(tl.float32)

        # right tile  shape (BLOCK_N, BLOCK_K)  (indexed by j, k)
        right_ptrs = (
            right_ptr
            + b * stride_rb
            + h * stride_rh
            + j[:, None] * stride_rj
            + k[None, :] * stride_rk
        )
        right = tl.load(
            right_ptrs,
            mask=j_mask[:, None] & k_mask[None, :],
            other=0.0,
        ).to(tl.float32)

        # accumulate  Σ_k left[i,k] * right[j,k]   (dot over K)
        acc += tl.dot(left, right, trans_b=True)   # (M,N)

    # ------------------------------------------------------------------
    # Write out the tile
    # ------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_ob
        + h * stride_oh
        + i[:, None] * stride_oi
        + j[None, :] * stride_oj
    )
    tl.store(out_ptrs, acc, mask=i_mask[:, None] & j_mask[None, :])

    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, on the same device).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]                # input/output channel dimension (C)
    hidden_dim = config["hidden_dim"]  # hidden dimension used inside TriMul
    eps = 1e-5

    # ------------------------------------------------------------------
    # Helper: LayerNorm using the provided weight/bias
    # ------------------------------------------------------------------
    def layer_norm(x, weight, bias):
        return F.layer_norm(x, (x.shape[-1],), weight=weight, bias=bias, eps=eps)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = layer_norm(
        input_tensor,
        weights["norm.weight"],
        weights["norm.bias"],
    )   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    if not config.get("nomask", False):

        right = right * m

    # 4) Gating (sigmoid) and element‑wise gating
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------

    left = left.contiguous()
    right = right.contiguous()

    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

        B * H,                                   # batch * hidden dimension
        (N + BLOCK_M - 1) // BLOCK_M,            # tiles along i
        (N + BLOCK_N - 1) // BLOCK_N,            # tiles along j
    )
    triangle_mul_kernel[
        grid
    ](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_lb,
        stride_li,
        stride_lk,
        stride_lh,
        stride_rb,
        stride_rj,
        stride_rk,
        stride_rh,
        stride_ob,
        stride_oi,
        stride_oj,
        stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,   # H100 enjoys high occupancy
    )

    # ------------------------------------------------------------------
    # 6) Post‑processing: LayerNorm, out‑gate, final linear
    # ------------------------------------------------------------------
    out = layer_norm(
        out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
    )
    out = out * out_gate                     # element‑wise mask
    out = F.linear(out, weights["to_out.weight"])

    return out
```

This implementation follows the reference PyTorch logic, using a custom Triton kernel for the cubic‑complexity `einsum` step while keeping all other operations in native PyTorch for simplicity and correctness. The kernel is tiled (64 × 64 output tiles) and accumulates the reduction over K in 32‑element blocks using `tl.dot`, which maps efficiently to the H100 tensor cores. The overall API matches the required `custom_kernel(data)` signature.
