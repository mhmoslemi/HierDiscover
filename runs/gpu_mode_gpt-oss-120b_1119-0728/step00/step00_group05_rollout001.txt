import triton
import triton.language as tl

@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_abatch, stride_arow, stride_acol,
    stride_bbatch, stride_brow, stride_bcol,
    stride_cbatch, stride_crow, stride_ccol,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # batch id = program id 0
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute block offsets.
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # Mask for out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator
    accumulator = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Compute block of K indices
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A block (M x K)
        a_ptrs = a_ptr + batch * stride_abatch + offs_m[:, None] * stride_arow + offs_k[None, :] * stride_acol
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load B block (K x N)
        b_ptrs = b_ptr + batch * stride_bbatch + offs_k[:, None] * stride_brow + offs_n[None, :] * stride_bcol
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # Multiply-accumulate
        accumulator += tl.dot(a, b)

    # Write back
    c_ptrs = c_ptr + batch * stride_cbatch + offs_m[:, None] * stride_crow + offs_n[None, :] * stride_ccol
    # Cast back to float16 (or keep float32 if output tensor dtype is fp16)
    # This kernel expects c dtype same as a,b; we will assume fp16.
    c = accumulator.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # Get weight tensors
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # LayerNorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )
    # Projections
    left = torch.nn.functional.linear(x, left_proj_weight)  # [B,N,N,H]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Mask handling
    if not config.get("nomask", False) and mask is not None:
        mask = mask.unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    right = right * right_gate

    left = left.to(torch.float16)
    right = right.to(torch.float16)

    right = right.permute(0, 3, 1, 2).contiguous() # shape [B, H, N, N]

    right = right.view(-1, N, N) # [B*H, N, N]

    right_t = right.transpose(1, 2).contiguous()

    stride_a_batch = left.stride(0)

        right_t,
        out,
        N, N, N,
        stride_a_batch, stride_a_row, stride_a_col,
        stride_b_batch, stride_b_row, stride_b_col,
        stride_c_batch, stride_c_row, stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,

    out = out.view(B, H, N, N).permute(0, 2, 3, 1).to(torch.float32)  # cast to float32

    out = out * out_gate

    # final linear
    out = torch.nn.functional.linear(out, to_out_weight)

"""
custom_kernel: Triton‑accelerated outgoing TriMul (AlphaFold3)

The implementation follows the reference PyTorch TriMul forward:

1. Layer‑norm over the last channel (dim).
2. Two linear projections (left/right) and three gating linear layers.
3. Optional pairwise mask applied to the projected tensors.
4. Gate‑modulated left/right tensors are multiplied by each other
   summed over the “k” sequence dimension:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication of shape
       (B·hidden, N, N) ← (B·hidden, N, N) @ (B·hidden, N, N)ᵀ .
   The heavy O(N³·hidden) work is performed by a custom Triton kernel
   that uses fp16 tensor‑cores with float32 accumulation.
5. A second Layer‑norm, an output‑gate, and a final linear projection
   back to the original channel dimension.

Only the forward pass is required; gradients are not computed.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ) for many independent matrices.
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    M, N, K,                      # matrix sizes (all == seq_len)
    stride_abatch, stride_arow, stride_acol,
    stride_bbatch, stride_brow, stride_bcol,
    stride_cbatch, stride_crow, stride_ccol,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute C = A @ Bᵀ for a batch of matrices.

    Arguments:
    * a_ptr, b_ptr, c_ptr – raw pointers to the tensors (fp16)
    * M, N, K – dimensions (all equal to seq_len)
    * stride_* – strides in element units for the three tensors
    * BLOCK_* – compile‑time tile sizes
    """
    batch_id = tl.program_id(0)            # which matrix in the batch
    pid_m    = tl.program_id(1)            # block row index
    pid_n    = tl.program_id(2)            # block col index

    # ----- block offsets -----
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- K‑loop (inner dimension) -----
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_blocks):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A block: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_ptr + batch_id * stride_abatch \
                 + offs_m[:, None] * stride_arow \
                 + offs_k[None, :] * stride_acol
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # Load B block: shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_ptr + batch_id * stride_bbatch \
                 + offs_k[:, None] * stride_brow \
                 + offs_n[None, :] * stride_bcol
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        # Matrix multiply‑accumulate
        acc += tl.dot(a, b)

    # ----- write‑back -----
    c_ptrs = c_ptr + batch_id * stride_cbatch \
             + offs_m[:, None] * stride_crow \
             + offs_n[None, :] * stride_ccol
    # Store as fp16 (the output tensor is allocated in fp16)
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (float32) or None
            - weights: dict of model parameters (all tensors on CUDA)
            - config: dict containing at least "dim", "hidden_dim" and "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # convenience aliases for weight tensors
    # ------------------------------------------------------------------
    norm_weight        = weights["norm.weight"]
    norm_bias          = weights["norm.bias"]
    left_proj_weight   = weights["left_proj.weight"]    # (hidden_dim, dim)
    right_proj_weight  = weights["right_proj.weight"]   # (hidden_dim, dim)
    left_gate_weight   = weights["left_gate.weight"]    # (hidden_dim, dim)
    right_gate_weight  = weights["right_gate.weight"]   # (hidden_dim, dim)
    out_gate_weight    = weights["out_gate.weight"]     # (hidden_dim, dim)
    to_out_norm_weight = weights["to_out_norm.weight"]  # (hidden_dim,)
    to_out_norm_bias   = weights["to_out_norm.bias"]    # (hidden_dim,)
    to_out_weight      = weights["to_out.weight"]       # (dim, hidden_dim)

    # ------------------------------------------------------------------
    # 1. LayerNorm over the channel dim
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) and optional mask
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)  # [B, N, N, hidden_dim]

    # Apply pairwise mask if it exists and nomask flag is False
    if not config.get("nomask", False) and mask is not None:
        mask_exp = mask.unsqueeze(-1)                     # [B, N, N, 1]
        left  = left  * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 3. Gates (sigmoid of another linear projection)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # Apply gates
    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Batched GEMM (the O(N³) core) using Triton
    # ------------------------------------------------------------------
    # Cast to fp16 for bandwidth / tensor‑core efficiency
    left  = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, H = left.shape   # B = batch, H = hidden_dim

    # Rearrange to (batch, hidden, N, N) → flatten batch & hidden as one axis
    left  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left  = left.view(-1, N, N)    # [B*H, N, N]
    right = right.view(-1, N, N)   # [B*H, N, N]

    # Transpose the second operand (we need rightᵀ)
    right_t = right.transpose(1, 2).contiguous()   # [B*H, N, N]

    # Allocate output buffer (fp16)
    out_fp16 = torch.empty_like(left)

    # Extract strides (in element units, not bytes)
    stride_a_batch = left.stride(0)
    stride_a_row   = left.stride(1)   # strides for the M dimension
    stride_a_col   = left.stride(2)   # strides for the K dimension

    stride_b_batch = right_t.stride(0)
    stride_b_row   = right_t.stride(1)   # K dimension after transpose
    stride_b_col   = right_t.stride(2)   # N dimension

    stride_c_batch = out_fp16.stride(0)
    stride_c_row   = out_fp16.stride(1)
    stride_c_col   = out_fp16.stride(2)

    # Tile sizes – empirically good for H100 fp16 TensorCores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        left.shape[0],                                   # batch dimension = B*H
        (N + BLOCK_M - 1) // BLOCK_M,                    # rows
        (N + BLOCK_N - 1) // BLOCK_N,                    # cols
    )

    batched_gemm_kernel[grid](
        left,
        right_t,
        out_fp16,
        N, N, N,                                         # M, N, K (all = seq_len)
        stride_a_batch, stride_a_row, stride_a_col,
        stride_b_batch, stride_b_row, stride_b_col,
        stride_c_batch, stride_c_row, stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Convert back to float32 for the remaining ops
    out = out_fp16.to(torch.float32)

    # Restore original shape: [B, N, N, hidden_dim]
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5. Second LayerNorm, output gate, final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate                      # apply output gate (still float32)

    # Final projection back to original channel dimension
    out = torch.nn.functional.linear(out, to_out_weight)   # [B, N, N, dim]

    return out
