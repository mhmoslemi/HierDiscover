def custom_kernel(data):
    import torch
    import triton
    import triton.language as tl

    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    
    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)
    
    # Linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])
    
    # mask
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_
        right = right * mask_
    
    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))
    
    left = left * left_gate
    right = right * right_gate

    # Convert to float16 for kernel
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # prepare output
    out = torch.empty_like(left, dtype=torch.float16)

    # stride calculations
    stride_left_b, stride_left_i, stride_left_k, stride_left_d = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_d = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_d = out.stride()
    
    # define block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # compute grid
    total_channels = bs * hidden_dim
    grid = (total_channels, (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)

    @triton.jit
    def trimul_kernel(
        left_ptr, right_ptr, out_ptr,
        B, N, D,
        stride_left_b, stride_left_i, stride_left_k, stride_left_d,
        stride_right_b, stride_right_j, stride_right_k, stride_right_d,
        stride_out_b, stride_out_i, stride_out_j, stride_out_d,
        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
        ):
        pid_c = tl.program_id(0)  # combined batch and hidden dim
        pid_m = tl.program_id(1)  # row block
        pid_n = tl.program_id(2)  # col block

        # decode batch and hidden dim
        b = pid_c // D
        d = pid_c % D

        # compute offsets
        offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
        offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
        # bounds check
        mask_m = offs_m < N
        mask_n = offs_n < N

        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        # loop over k dimension
        for k in range(0, tl.cdiv(N, BLOCK_K)):
            offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
            mask_k = offs_k < N

            # compute pointers
            left_ptrs = left_ptr + b * stride_left_b + d * stride_left_d + offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
            right_ptrs = right_ptr + b * stride_right_b + d * stride_right_d + offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_j

            # load tiles, with masking
            left_tile = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
            right_tile = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

            # dot product accumulate
            acc += tl.dot(left_tile, right_tile)

        # store result
        out_ptrs = out_ptr + b * stride_out_b + d * stride_out_d + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
        tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

    # launch kernel
    trimul_kernel[grid](
        left, right, out,
        bs, N, hidden_dim,
        stride_left_b, stride_left_i, stride_left_k, stride_left_d,
        stride_right_b, stride_right_j, stride_right_k, stride_right_d,
        stride_out_b, stride_out_i, stride_out_j, stride_out_d,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    
    # Continue with rest of pipeline
    out = out.to(torch.float32)
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=1e-5)
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out

import torch
import triton
import triton.language as tl

def custom_kernel(data):
    """...
    """
    input_tensor, mask, weights, config = data
    # parse dims
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # layernorm over dim
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)

    # linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # apply mask if present
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_
        right = right * mask_

    # gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # cast to half for MatMul kernel
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # allocate output
    out = torch.empty_like(left, dtype=torch.float16)

    # get strides
    stride_left_b, stride_left_i, stride_left_k, stride_left_d = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_d = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_d = out.stride()

    # block sizes (tuned for H100)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # total number of (batch, hidden) channels
    total_c = bs * hidden_dim

    grid = (total_c, (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)

    @triton.jit
    def trimul_kernel(
        left_ptr, right_ptr, out_ptr,
        B, N,
        stride_left_b, stride_left_i, stride_left_k, stride_left_d,
        stride_right_b, stride_right_j, stride_right_k, stride_right_d,
        stride_out_b, stride_out_i, stride_out_j, stride_out_d,
        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
        D: tl.constexpr,  # hidden dimension
    ):
        pid_c = tl.program_id(0)  # combined batch+channel
        pid_m = tl.program_id(1)
        pid_n = tl.program_id(2)

        # decode batch index and hidden channel
        b = pid_c // D
        d = pid_c % D

        # row/col offsets
        offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
        offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_m = offs_m < N
        mask_n = offs_n < N

        # accumulator in fp32
        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        # loop over reduction K dimension
        for kk in range(0, tl.cdiv(N, BLOCK_K)):
            offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
            mask_k = offs_k < N

            # pointer arithmetics
            left_ptrs = left_ptr + b * stride_left_b + d * stride_left_d + offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
            right_ptrs = right_ptr + b * stride_right_b + d * stride_right_d + offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_j

            # load with masking
            left_tile = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
            right_tile = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

            # dot product (fp16 * fp16 -> fp32)
            acc += tl.dot(left_tile, right_tile)

        # store result
        out_ptrs = out_ptr + b * stride_out_b + d * stride_out_d + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
        tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

    # launch
    trimul_kernel[grid](
        left, right, out,
        bs, N,
        stride_left_b, stride_left_i, stride_left_k, stride_left_d,
        stride_right_b, stride_right_j, stride_right_k, stride_right_d,
        stride_out_b, stride_out_i, stride_out_j, stride_out_d,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        D=hidden_dim,
    )
    # cast back to fp32 for following layers
    out = out.to(torch.float32)

    # LayerNorm over hidden dimension
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=eps)

    # apply out_gate (still fp32)
    out = out * out_gate

    # final linear projection back to original dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
custom_kernel: fast Triton implementation of the “outgoing” TriMul operation.

Algorithm
---------
1. Layer‑norm the input tensor (dim‑wise).
2. Project the normalized tensor to a hidden dimension with three linear layers:
   * left_proj  → left  (shape [B, N, N, H])
   * right_proj → right (shape [B, N, N, H])
3. (Optional) Apply a binary mask to left/right.
4. Compute three sigmoid‑gates (left_gate, right_gate, out_gate) from the
   *same* normalized tensor and multiply left/right by their corresponding gates.
5. Fuse the heavy O(N³·H) computation in a custom Triton kernel:
   For each batch‑hidden channel (b, h) compute
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   which is exactly a batched matrix multiplication left @ rightᵀ for every hidden
   channel.  The kernel works on FP16 tiles, accumulates in FP32 and writes the
   result back as FP16.
6. Convert the result back to FP32, apply a second layer‑norm (to_out_norm),
   multiply by the out‑gate and finally project back to the original dim
   with a linear layer (to_out).

The heavy reduction over the sequence dimension is therefore executed on‑GPU
with a highly‑tuned Triton matmul, while the surrounding linear/gate operations
remain in PyTorch for simplicity.  The kernel is compiled with block sizes
(64,64,64) which are well‑suited for the H100 tensor‑core units.
"""

import torch
import triton
import triton.language as tl

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # -----------------------------------------------------------------------
    # 1. Unpack dimensions and hyper‑parameters
    # -----------------------------------------------------------------------
    bs, N, _, dim = input_tensor.shape               # input shape [B,N,N,dim]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------------
    # 2. Layer‑norm over the last dimension (dim)
    # -----------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # -----------------------------------------------------------------------
    # 3. Linear projections (no bias)
    # -----------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------------
    # 4. Optional mask (broadcasted on the hidden dimension)
    # -----------------------------------------------------------------------
    if mask is not None:
        # mask: [B, N, N] → [B, N, N, 1] for broadcasting
        mask_ = mask.unsqueeze(-1).to(left.dtype)
        left  = left  * mask_
        right = right * mask_

    # -----------------------------------------------------------------------
    # 5. Gates (sigmoid of a linear transform of the normalized input)
    # -----------------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # Cast to FP16 for the heavy matmul (half the memory bandwidth)
    left  = left.contiguous().to(torch.float16)
    right = right.contiguous().to(torch.float16)

    # -----------------------------------------------------------------------
    # 6. Triton kernel: batched matmul of left @ rightᵀ for every hidden channel
    # -----------------------------------------------------------------------
    out = torch.empty_like(left, dtype=torch.float16)   # shape [B,N,N,H]

    # Strides in element units (not bytes)
    s_lb, s_li, s_lk, s_ld = left.stride()   # B, i, k, d
    s_rb, s_rj, s_rk, s_rd = right.stride()
    s_ob, s_oi, s_oj, s_od = out.stride()

    # Tuned block sizes – values work well on H100 tensor cores
    BLOCK_M = 64   # rows of left (i dimension)
    BLOCK_N = 64   # cols of rightᵀ (j dimension)
    BLOCK_K = 64   # reduction dimension (k)

    total_channels = bs * hidden_dim          # each (b, h) gets its own program instance
    grid = (total_channels,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    @triton.jit
    def trimul_kernel(
        left_ptr, right_ptr, out_ptr,
        B, N,
        s_lb, s_li, s_lk, s_ld,
        s_rb, s_rj, s_rk, s_rd,
        s_ob, s_oi, s_oj, s_od,
        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
        D: tl.constexpr,                     # hidden dimension (compile‑time constant)
    ):
        # -------------------------------------------------------------------
        #   Program IDs: 0 → (batch, hidden) channel, 1 → row block, 2 → col block
        # -------------------------------------------------------------------
        pid_c = tl.program_id(0)   # combined batch * hidden index
        pid_m = tl.program_id(1)   # block‑row (i)
        pid_n = tl.program_id(2)   # block‑col (j)

        # Decode batch and hidden‑channel from the combined index
        b = pid_c // D
        d = pid_c % D

        # Offsets for the current tile
        offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
        offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_i = offs_i < N
        mask_j = offs_j < N

        # Accumulator in FP32 (higher precision for the reduction)
        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        # -------------------------------------------------------------------
        #   Loop over the K dimension (the shared sequence axis)
        # -------------------------------------------------------------------
        for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
            offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
            mask_k = offs_k < N

            # Pointers to left[i,k] and rightᵀ[k,j] (note the transpose)
            left_ptrs = (left_ptr
                         + b * s_lb + d * s_ld
                         + offs_i[:, None] * s_li
                         + offs_k[None, :] * s_lk)

            right_ptrs = (right_ptr
                          + b * s_rb + d * s_rd
                          + offs_k[:, None] * s_rk
                          + offs_j[None, :] * s_rj)

            # Load tiles with out‑of‑bounds masking (return 0.0 for illegal lanes)
            left_tile = tl.load(left_ptrs,
                                mask=mask_i[:, None] & mask_k[None, :],
                                other=0.0)
            right_tile = tl.load(right_ptrs,
                                 mask=mask_k[:, None] & mask_j[None, :],
                                 other=0.0)

            # FP16 × FP16 → FP32 accumulation
            acc += tl.dot(left_tile, right_tile)

        # Store the result back as FP16
        out_ptrs = (out_ptr
                    + b * s_ob + d * s_od
                    + offs_i[:, None] * s_oi
                    + offs_j[None, :] * s_oj)

        tl.store(out_ptrs,
                 acc.to(tl.float16),
                 mask=mask_i[:, None] & mask_j[None, :])

    # Launch the kernel
    trimul_kernel[grid](
        left, right, out,
        bs, N,
        s_lb, s_li, s_lk, s_ld,
        s_rb, s_rj, s_rk, s_rd,
        s_ob, s_oi, s_oj, s_od,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        D=hidden_dim,
    )

    # -----------------------------------------------------------------------
    # 7. Post‑processing in FP32 (layer‑norm, out‑gate, final linear)
    # -----------------------------------------------------------------------
    out = out.to(torch.float32)

    # to_out_norm over the hidden dimension H
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # Apply the out‑gate (still FP32)
    out = out * out_gate

    # Final projection back to the original “dim”
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
