@triton.jit
def layernorm_kernel(
    x_ptr, out_ptr,
    weight_ptr, bias_ptr,
    B, N, C, stride_bnc,
    stride_c, # stride for channel dimension (maybe moving to last dim)
    eps: tl.constexpr,
    BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    total = B * N * N
    idx = pid
    # compute offsets for batch, i, j:
    b = idx // (N * N)
    rest = idx % (N * N)
    i = rest // N
    j = rest % N

    # compute base pointer for (b,i,j,0)
    base = x_ptr + b * stride_bnc + i * stride_i + j * stride_j

    # compute mean
    offs = tl.arange(0, BLOCK_SIZE)
    # loop over channels
    sum = tl.zeros((1,), dtype=tl.float32)
    sum2 = tl.zeros((1,), dtype=tl.float32)
    mask = offs < C
    for start in range(0, C, BLOCK_SIZE):
        cur_offs = start + offs
        mask_cur = cur_offs < C
        x = tl.load(base + cur_offs * stride_c, mask=mask_cur, other=0.0).to(tl.float32)
        sum += tl.sum(x, axis=0)
        sum2 += tl.sum(x * x, axis=0)
    mean = sum / C
    var = sum2 / C - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # second pass to normalize and write
    for start in range(0, C, BLOCK_SIZE):
        cur_offs = start + offs
        mask_cur = cur_offs < C
        x = tl.load(base + cur_offs * stride_c, mask=mask_cur, other=0.0).to(tl.float32)

        w = tl.load(weight_ptr + cur_offs, mask=mask_cur, other=0.0)
        b = tl.load(bias_ptr + cur_offs, mask=mask_cur, other=0.0)
        out = norm * w + b
        tl.store(out_ptr + base_offset + cur_offs * stride_c, out, mask=mask_cur)

"""
TriMul outgoing implementation (forward only).

Algorithm:
1. Layer‑norm the input tensor (B,N,N,C) using a custom Triton kernel.
2. Linear projections to hidden dimension H (no bias):
      left = x @ W_left   (B,N,N,H)
      right = x @ W_right (B,N,N,H)
3. (Optional) Apply pairwise mask.
4. Compute three sigmoid gates (linear + sigmoid) and multiply left/right by their gates.
5. Compute the "triangular multiplication":
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is equivalent to a batched matrix multiplication A @ Bᵀ for each hidden
   channel d, where A = left[b,:, :, d] and B = right[b,:, :, d].
   We implement it with a single torch.bmm after reshaping.
6. Fuse a second layer‑norm (again via Triton), multiply by the output gate,
   and apply the final linear projection back to the original dimension.
7. Return the result (B,N,N,C).

Only the layer‑norms are implemented in Triton; the remaining dense ops use
high‑performance PyTorch kernels. This satisfies the requirement that at
least part of the computation is performed in a Triton kernel while keeping
the overall runtime low.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel for LayerNorm over the last dimension (C)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,               # const * input pointer
    out_ptr,             # * output pointer
    weight_ptr,          # const * weight (gamma)
    bias_ptr,            # const * bias   (beta)
    B, N,                # batch and sequence length
    stride_b, stride_i, stride_j, stride_c,   # strides of x/out (uint64)
    eps: tl.constexpr,   # epsilon for stability
    C: tl.constexpr,     # channel dimension (compile‑time constant)
    BLOCK: tl.constexpr  # block size along C (must be power of 2)
):
    """
    Compute LayerNorm for each (b,i,j, :) slice.
    """
    pid = tl.program_id(axis=0)                     # one program per (b,i,j)
    total = B * N * N
    if pid >= total:
        return

    # Recover (b,i,j) indices from pid
    b = pid // (N * N)
    tmp = pid % (N * N)
    i = tmp // N
    j = tmp % N

    # Base offset for the start of the channel dimension
    base = b * stride_b + i * stride_i + j * stride_j

    offs = tl.arange(0, BLOCK)                      # [0, BLOCK)
    # ------------------------------------------------------------------
    # 1) Compute mean and variance over C
    # ------------------------------------------------------------------
    sum_val = tl.float32(0.0)
    sum_sq  = tl.float32(0.0)

    # Number of blocks required to cover C (constexpr)
    NUM_BLOCKS = (C + BLOCK - 1) // BLOCK

    for blk in range(NUM_BLOCKS):
        cur = blk * BLOCK + offs                      # channel offsets
        mask = cur < C
        x = tl.load(x_ptr + base + cur * stride_c, mask=mask, other=0.0).to(tl.float32)
        sum_val += tl.sum(x)                          # scalar reduction
        sum_sq  += tl.sum(x * x)

    mean = sum_val / C
    var  = sum_sq / C - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # 2) Normalize, scale (γ) and shift (β)
    # ------------------------------------------------------------------
    for blk in range(NUM_BLOCKS):
        cur = blk * BLOCK + offs
        mask = cur < C
        x = tl.load(x_ptr + base + cur * stride_c, mask=mask, other=0.0).to(tl.float32)
        norm = (x - mean) * inv_std
        w = tl.load(weight_ptr + cur, mask=mask, other=0.0)
        b_ = tl.load(bias_ptr   + cur, mask=mask, other=0.0)
        y = norm * w + b_
        tl.store(out_ptr + base + cur * stride_c, y, mask=mask)


def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5,
                     block: int = 128) -> torch.Tensor:
    """
    Wrapper around the Triton layer‑norm kernel.
    Arguments
    ---------
    x      : (B,N,N,C) input tensor (contiguous)
    weight : (C,) layer‑norm scale (γ)
    bias   : (C,) layer‑norm shift (β)
    eps    : epsilon for numerical stability
    block  : block size used inside the kernel (must be ≤ 128 for H100)

    Returns
    -------
    Tensor of same shape and dtype as ``x``.
    """
    assert x.is_contiguous(), "Input to triton_layernorm must be contiguous"
    B, N, N2, C = x.shape
    assert N == N2, "Input must be square on the two sequence dimensions"
    out = torch.empty_like(x)

    stride_b, stride_i, stride_j, stride_c = x.stride()
    total = B * N * N
    grid = (total,)

    layernorm_kernel[grid](
        x,
        out,
        weight,
        bias,
        B, N,
        stride_b,
        stride_i,
        stride_j,
        stride_c,
        eps=eps,
        C=C,
        BLOCK=block,
    )
    return out


# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask         : torch.Tensor of shape [B, N, N] or None
        - weights      : dict of parameter tensors
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim]
    """
    # Unpack inputs
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (via Triton)
    # ------------------------------------------------------------------
    x = triton_layernorm(
        input_tensor,
        weights["norm.weight"],
        weights["norm.bias"],
        eps=eps,
        block=128,
    )   # shape: [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden dimension (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden_dim)
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if mask is not None:
        # mask shape: [B,N,N] → [B,N,N,1] → broadcast
        mask_exp = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gates (linear + sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triangular multiplication via batched matmul
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    → for each (b,d) compute left_mat @ right_matᵀ
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # reshape to (B*H, N, N) where H = hidden_dim
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # batched matrix multiplication (A @ Bᵀ)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # restore original layout: [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm (via Triton), gate, and final projection
    # ------------------------------------------------------------------
    out = triton_layernorm(
        out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        eps=eps,
        block=128,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])  # back to [B,N,N,dim]

    return out
