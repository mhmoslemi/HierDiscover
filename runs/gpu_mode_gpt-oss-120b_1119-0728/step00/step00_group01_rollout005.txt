def layernorm_kernel(
    x_ptr, y_ptr, weight_ptr, bias_ptr,
    B, N, C,
    stride_bx, stride_ix, stride_jx, stride_cx,
    stride_by, stride_iy, stride_jy, stride_cy,
    BLOCK_C: tl.constexpr):
    pid = tl.program_id(axis=0)
    # compute overall index: each program handles one (b, i, j) triple.
    total = B * N * N
    # we have one program per row (b,i,j)
    # compute b = pid // (N*N)
    # compute i = (pid % (N*N)) // N
    # compute j = pid % N

    # compute base offset
    offset = b*stride_bx + i*stride_ix + j*stride_jx

    sum = 0.0
    sum2 = 0.0
    for c in range(0, tl.cdiv(C, BLOCK_C)):
        c_offset = offset + (c * BLOCK_C) * stride_cx
        x = tl.load(x_ptr + c_offset, mask=c*BLOCK_C + tl.arange(0, BLOCK_C) < C, other=0.0)
        sum += tl.sum(x, axis=0)
        sum2 += tl.sum(x * x, axis=0)
    # compute mean and var

def layernorm_kernel(
    in_ptr, out_ptr,
    weight_ptr, bias_ptr,
    stride_bb, stride_i, stride_j, stride_c,
    out_stride_bb, out_stride_i, out_stride_j, out_stride_c,
    N, C,
    eps,
    BLOCK_C: tl.constexpr):
    pid = tl.program_id(0)

    # compute position
    n = N
    b = pid // (n * n)
    rem = pid - b * (n * n)
    i = rem // n
    j = rem - i * n

    # base offsets
    base_off = b * stride_bb + i * stride_i + j * stride_j

    sum = 0.0
    sum_sq = 0.0

    for block in range(0, tl.cdiv(C, BLOCK_C)):
        c_off = block * BLOCK_C
        # offsets for load
        idx = c_off + tl.arange(0, BLOCK_C)
        mask = idx < C
        ptr = in_ptr + base_off + idx * stride_c
        x = tl.load(ptr, mask=mask, other=0.0)
        sum += tl.sum(x, axis=0)
        sum_sq += tl.sum(x * x, axis=0)

    inv_std = 1.0 / tl.sqrt(var + eps)

        idx = c_off + tl.arange(0, BLOCK_C)

def custom_kernel(data):
    """
    Triton-accelerated implementation of the outgoing Triangle Multiplicative Update.
    It performs:
    1. LayerNorm on input.
    2. Linear projections and gating (via torch ops).
    3. Elementwise mask.
    4. Batched matrix multiplication representing the `einsum` operation.
    5. Second LayerNorm, out gating, and final linear.
    """
    (x, mask, weights, config) = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Ensure contiguous weights
    for key in weights:
        weights[key] = weights[key].contiguous()

    # 1. First LayerNorm using Triton
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = _layernorm_triton(x, norm_weight, norm_bias)   # shape [B,N,N,dim]

    # 2. Linear projections
    left_proj_weight = weights["left_proj.weight"]   # shape [hidden_dim, dim]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x_norm, left_proj_weight)   # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # 3. Apply mask if needed
    if not nomask:
        # mask may be bool or int, convert to float
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # 4. Compute gate values
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # 5. TriMul einsum via batched matmul
    # reshape for batched GEMM: (B, hidden, N, N) x (B, hidden, N, N).transpose(-2,-1)
    left_ = left.permute(0, 3, 1, 2)   # (B, hidden, N, N)
    right_ = right.permute(0, 3, 2, 1) # (B, hidden, N, N)
    out_ = torch.matmul(left_, right_)  # (B, hidden, N, N)
    out = out_.permute(0, 2, 3, 1)   # (B, N, N, hidden)

    # 6. Second LayerNorm (hidden_dim) via Triton
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_norm = _layernorm_triton(out, to_out_norm_weight, to_out_norm_bias)

    # 7. Apply out gating
    out_norm = out_norm * out_gate

    # 8. Final linear to dim
    final = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

    return final

"""
Triton‑accelerated forward pass for the *outgoing* Triangle Multiplicative Update
(TriMul) used in AlphaFold‑3 style pairwise updates.

The implementation follows the reference PyTorch code:

    1. LayerNorm over the last dimension (dim) – done with a custom Triton kernel.
    2. Linear projections (left/right) and linear gates (left_gate/right_gate/out_gate)
       – performed with torch.nn.functional.linear + sigmoid.
    3. Optional pairwise mask applied to the projected tensors.
    4. Core TriMul operation:
           out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
       which is mathematically a batched matrix multiplication:
           out = (left.permute(0,3,1,2) @ right.permute(0,3,2,1))
           → shape (B, N, N, hidden_dim).
       The batched GEMM is executed with ``torch.matmul`` (cuBLAS/Tensor‑cores).
    5. LayerNorm over the hidden dimension (again via Triton).
    6. Out‑gate multiplication and final linear projection back to ``dim``.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: LayerNorm over the last dimension.
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    in_ptr, out_ptr,               # pointers to input / output tensors
    weight_ptr, bias_ptr,          # LayerNorm parameters (1‑D tensors)
    stride_b_in, stride_i_in, stride_j_in, stride_c_in,  # input strides
    stride_b_out, stride_i_out, stride_j_out, stride_c_out,  # output strides
    N, C,                          # sequence length and channel size
    eps,                           # epsilon for numerical stability
    BLOCK_C: tl.constexpr,         # compile‑time block size for the C dimension
):
    """Row‑wise LayerNorm.  One program instance handles one (b,i,j) slice."""
    pid = tl.program_id(0)                       # pid ∈ [0, B·N·N)

    # ------------------------------------------------------------------
    # Decode (batch, i, j) from the linear program id.
    # ------------------------------------------------------------------
    n = N
    b = pid // (n * n)
    rem = pid - b * (n * n)
    i = rem // n
    j = rem - i * n

    # Base offset for this row (the C‑dimension is handled inside the loops)
    row_offset_in = b * stride_b_in + i * stride_i_in + j * stride_j_in
    row_offset_out = b * stride_b_out + i * stride_i_out + j * stride_j_out

    # ------------------------------------------------------------------
    # First pass – compute mean and variance.
    # ------------------------------------------------------------------
    sum_val = tl.float32(0.0)
    sum_sq = tl.float32(0.0)

    num_blocks = tl.cdiv(C, BLOCK_C)
    for blk in range(0, num_blocks):
        c_idx = blk * BLOCK_C + tl.arange(0, BLOCK_C)          # [BLOCK_C]
        mask = c_idx < C
        ptr = in_ptr + row_offset_in + c_idx * stride_c_in
        x = tl.load(ptr, mask=mask, other=0.0)                # (BLOCK_C,)
        sum_val += tl.sum(x, axis=0)
        sum_sq += tl.sum(x * x, axis=0)

    mean = sum_val / C
    var = sum_sq / C - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass – normalize, apply weight & bias and store.
    # ------------------------------------------------------------------
    for blk in range(0, num_blocks):
        c_idx = blk * BLOCK_C + tl.arange(0, BLOCK_C)
        mask = c_idx < C
        # load raw input
        x_ptr = in_ptr + row_offset_in + c_idx * stride_c_in
        x = tl.load(x_ptr, mask=mask, other=0.0)

        # load weight & bias (both 1‑D)
        w = tl.load(weight_ptr + c_idx, mask=mask, other=0.0)
        b_ = tl.load(bias_ptr + c_idx, mask=mask, other=0.0)

        # LayerNorm formula
        y = ((x - mean) * inv_std) * w + b_

        # store result
        y_ptr = out_ptr + row_offset_out + c_idx * stride_c_out
        tl.store(y_ptr, y, mask=mask)


def _layernorm_triton(x: torch.Tensor,
                      weight: torch.Tensor,
                      bias: torch.Tensor,
                      eps: float = 1e-5) -> torch.Tensor:
    """
    Wrapper that runs ``layernorm_kernel`` over a tensor of shape
    [B, N, N, C] (or any tensor where the last dimension is to be normalized).
    Returns a new tensor with the same shape and dtype as ``x``.
    """
    B, N, N2, C = x.shape
    assert N == N2, "TriMul expects a square pair tensor"

    # Prepare output
    out = torch.empty_like(x)

    # Strides for input and output (in number of elements, not bytes)
    s_b_in, s_i_in, s_j_in, s_c_in = x.stride()
    s_b_out, s_i_out, s_j_out, s_c_out = out.stride()

    # Choose a block size that covers the largest possible C (384)
    BLOCK_C = 128

    total_rows = B * N * N
    grid = (total_rows,)

    layernorm_kernel[grid](
        x, out,
        weight, bias,
        s_b_in, s_i_in, s_j_in, s_c_in,
        s_b_out, s_i_out, s_j_out, s_c_out,
        N, C,
        eps,
        BLOCK_C=BLOCK_C,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config) where
            input_tensor : torch.Tensor  [B, N, N, dim]
            mask         : torch.Tensor  [B, N, N] (may be ignored)
            weights      : dict of torch.Tensor containing all linear/ln parameters
            config       : dict with keys ``dim`` and ``hidden_dim`` and optional ``nomask``

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Ensure all weight tensors are contiguous and on the same device
    # ------------------------------------------------------------------
    for k, v in weights.items():
        weights[k] = v.contiguous()

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (over ``dim``) – Triton implementation
    # ------------------------------------------------------------------
    x_norm = _layernorm_triton(
        input_tensor,
        weights["norm.weight"],
        weights["norm.bias"],
    )   # shape [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(
        x_norm, weights["left_proj.weight"]
    )                     # [B, N, N, hidden_dim]
    right = torch.nn.functional.linear(
        x_norm, weights["right_proj.weight"]
    )                     # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3️⃣ Optional pairwise mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        # ``mask`` is [B, N, N]; expand to [B, N, N, 1] for broadcasting
        mask_f = mask.to(input_tensor.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4️⃣ Gates (linear + sigmoid) – pure PyTorch
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(
        x_norm, weights["left_gate.weight"]
    ).sigmoid()          # [B, N, N, hidden_dim]
    right_gate = torch.nn.functional.linear(
        x_norm, weights["right_gate.weight"]
    ).sigmoid()
    out_gate = torch.nn.functional.linear(
        x_norm, weights["out_gate.weight"]
    ).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core TriMul: batched matmul implements the einsum
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    # Bring the hidden dimension forward so that we can use torch.matmul.
    # left_ : (B, hidden, N, N)   = left.permute(0,3,1,2)
    # right_: (B, hidden, N, N)   = right.permute(0,3,2,1)   (k ↔ j)
    left_ = left.permute(0, 3, 1, 2)          # (B, hidden, N, N)
    right_ = right.permute(0, 3, 2, 1)        # (B, hidden, N, N)
    # Batched matmul over the last two axes.
    out_ = torch.matmul(left_, right_)        # (B, hidden, N, N)
    out = out_.permute(0, 2, 3, 1)            # (B, N, N, hidden)

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension – Triton again
    # ------------------------------------------------------------------
    out_norm = _layernorm_triton(
        out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
    )   # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 7️⃣ Out‑gate multiplication
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate

    # ------------------------------------------------------------------
    # 8️⃣ Final linear projection back to ``dim`` (no bias)
    # ------------------------------------------------------------------
    final = torch.nn.functional.linear(
        out_norm, weights["to_out.weight"]
    )   # shape [B, N, N, dim]

    return final
