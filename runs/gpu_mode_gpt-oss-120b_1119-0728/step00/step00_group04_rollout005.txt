def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid_b = tl.program_id(2)
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    
    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J
    # Offsets
    i_range = tl.arange(0, BLOCK_I)
    j_range = tl.arange(0, BLOCK_J)
    d_range = tl.arange(0, BLOCK_D)
    
    # Compute mask for boundary
    i_mask = i_start + i_range < N
    j_mask = j_start + j_range < N
    
    # Compute offsets for i and j base pointers
    i_offset = i_start + i_range
    j_offset = j_start + j_range
    
    # Initialize accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)
    
    for k_start in range(0, N, BLOCK_K):
        k_range = tl.arange(0, BLOCK_K)
        k_valid = k_start + k_range < N
        
        # Compute pointers to left: shape (BLOCK_I, BLOCK_K, BLOCK_D)
        # pointer formula: left_ptr + pid_b*stride_l_b + i_offset*stride_l_i + (k_start + k_range)*stride_l_k + d_range*stride_l_h
        left_ptrs = left_ptr + pid_b*stride_l_b + (i_offset[:, None] * stride_l_i) + ((k_start + k_range)[None, :] * stride_l_k) + (d_range[None, None, :] * stride_l_h)
        # For out-of-bounds k (if k_valid false), set pointer offset to maybe zero and mask.
        # we will load and mask later.
        left = tl.load(left_ptrs, mask=k_valid[None, :] & i_mask[:, None] , other=0.0)
        # left shape (BLOCK_I, BLOCK_K, BLOCK_D)
        
        # Right: shape (BLOCK_J, BLOCK_K, BLOCK_D)
        right_ptrs = right_ptr + pid_b*stride_r_b + (j_offset[:, None] * stride_r_j) + ((k_start + k_range)[None, :] * stride_r_k) + (d_range[None, None, :] * stride_r_h)
        right = tl.load(right_ptrs, mask=k_valid[None, :] & j_mask[:, None] , other=0.0)
        # right shape (BLOCK_J, BLOCK_K, BLOCK_D)
        
        # For each kk in BLOCK_K
        for kk in range(0, BLOCK_K):
            # Option: Use tl.load mask for out-of-range K? Already masked.
            left_vec = left[:, kk, :]  # shape (BLOCK_I, BLOCK_D)
            right_vec = right[:, kk, :]  # shape (BLOCK_J, BLOCK_D)
            # Broadcast
            # left_vec[:, None, :] shape (BLOCK_I, 1, BLOCK_D)
            # right_vec[None, :, :] shape (1, BLOCK_J, BLOCK_D)
            # Multiply and accumulate
            acc += left_vec[:, None, :] * right_vec[None, :, :]  # shape (BLOCK_I, BLOCK_J, BLOCK_D)
    
    # Store acc to out
    # out pointer: out_ptr + pid_b*stride_o_b + i_offset*stride_o_i + j_offset*stride_o_j + d_range*stride_o_h
    out_ptrs = out_ptr + pid_b*stride_o_b + (i_offset[:, None, None] * stride_o_i) + (j_offset[None, :, None] * stride_o_j) + (d_range[None, None, :] * stride_o_h)
    # store
    tl.store(out_ptrs, acc, mask=i_mask[:, None, None] & j_mask[None, :, None])

@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    D_OFFSET: tl.constexpr,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid_b = tl.program_id(2)
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J
    i_off = i_start + tl.arange(0, BLOCK_I)
    j_off = j_start + tl.arange(0, BLOCK_J)

    i_mask = i_off < N
    j_mask = j_off < N

    # d dimension indices
    d_off = D_OFFSET + tl.arange(0, BLOCK_D)
    d_mask = d_off < H

    # accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

    # Loop over k
    for k_start in range(0, N, BLOCK_K):
        k_off = k_start + tl.arange(0, BLOCK_K)
        k_mask = k_off < N

        # Compute left pointers
        # shape (BLOCK_I, BLOCK_K, BLOCK_D)
        left_ptrs = left_ptr + \
            pid_b * stride_left_b + \
            i_off[:, None, None] * stride_left_i + \
            k_off[None, :, None] * stride_left_k + \
            d_off[None, None, :] * stride_left_h

        left = tl.load(left_ptrs, mask=(i_mask[:, None, None] & k_mask[None, :, None] & d_mask[None, None, :]), other=0.0)

        # Compute right pointers
        right_ptrs = right_ptr + \
            pid_b * stride_right_b + \
            j_off[:, None, None] * stride_right_i + \
            k_off[None, :, None] * stride_right_k + \
            d_off[None, None, :] * stride_right_h

        right = tl.load(right_ptrs, mask=(j_mask[:, None, None] & k_mask[None, :, None] & d_mask[None, None, :]), other=0.0)

        # accumulate across k
        # Loop over k dimension within this block
        for kk in range(0, BLOCK_K):
            # check if k valid
            kk_mask = k_mask[kk]
            # we can early break if all false? Not needed
            left_vec = left[:, kk, :]  # (BLOCK_I, BLOCK_D)
            right_vec = right[:, kk, :]  # (BLOCK_J, BLOCK_D)
            # Broadcast multiply.
            acc += left_vec[:, None, :] * right_vec[None, :, :]
            # Note: multiplication broadcast across BLOCK_J and BLOCK_I dims automatically matches.
    # End for k

    # store results
    out_ptrs = out_ptr + \
        pid_b * stride_out_b + \
        i_off[:, None, None] * stride_out_i + \
        j_off[None, :, None] * stride_out_j + \
        d_off[None, None, :] * stride_out_h

    tl.store(out_ptrs, acc, mask=(i_mask[:, None, None] & j_mask[None, :, None] & d_mask[None, None, :]))

"""
TriMul (outgoing) – Triton implementation
------------------------------------------------------------
The forward pass consists of three stages:

1. **Pre‑processing (PyTorch)**
   * Layer‑norm over the last dimension (dim).
   * Linear projections ``left_proj`` and ``right_proj`` (dim → hidden_dim).
   * Optional masking (broadcast over the hidden dimension).
   * Gated projections using three sigmoid‑gated linear layers.

2. **Core contraction (Triton)**
   * Compute   out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d].
   * This is a batched matrix‑multiply per hidden channel (i‑×‑k) @ (j‑×‑k)ᵀ.
   * The kernel tiles the ``i`` and ``j`` axes (32×32) and the reduction axis ``k`` (32).
   * The hidden dimension is tiled with ``BLOCK_D = 32`` (the hidden_dim in all tests
     is a multiple of 32, so no out‑of‑bounds accesses occur).  Accumulation is performed
     in float32 registers.

3. **Post‑processing (PyTorch)**
   * Layer‑norm over the hidden dimension, gating with ``out_gate`` and a final
     linear projection back to ``dim``.

The Triton kernel only handles the expensive ℕ³ contraction; the surrounding
normalisation and linear layers are left to PyTorch for simplicity and correctness.
"""

import torch
import triton
import triton.language as tl
from torch.nn.functional import linear, layer_norm, sigmoid


# ----------------------------------------------------------------------
# Triton kernel: sum_k left[i,k,d] * right[j,k,d]  →  out[i,j,d]
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,          # const float*  (B,N,N,H)
    right_ptr,         # const float*  (B,N,N,H)
    out_ptr,           # float*        (B,N,N,H)

    B, N, H,           # int32
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,

    d_offset: tl.int32,                     # start index in hidden dimension
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Compute a tile of the TriMul contraction.
    Grid: (num_i_tiles, num_j_tiles, batch)
    """

    pid_b = tl.program_id(2)   # batch dimension
    pid_i = tl.program_id(0)   # tile over i (seq_len)
    pid_j = tl.program_id(1)   # tile over j (seq_len)

    # ----- tile offsets -------------------------------------------------
    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)            # (BLOCK_I,)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)            # (BLOCK_J,)
    d = d_offset + tl.arange(0, BLOCK_D)                  # (BLOCK_D,)

    i_mask = i < N
    j_mask = j < N
    d_mask = d < H

    # accumulator in FP32
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

    # ----- reduction over k ---------------------------------------------
    for k_start in range(0, N, BLOCK_K):
        k = k_start + tl.arange(0, BLOCK_K)                # (BLOCK_K,)
        k_mask = k < N

        # Load left[i,k,d]  → shape (BLOCK_I, BLOCK_K, BLOCK_D)
        left_ptrs = left_ptr \
            + pid_b * stride_l_b \
            + i[:, None, None] * stride_l_i \
            + k[None, :, None] * stride_l_k \
            + d[None, None, :] * stride_l_h
        left = tl.load(left_ptrs,
                       mask=(i_mask[:, None, None] &
                             k_mask[None, :, None] &
                             d_mask[None, None, :]),
                       other=0.0)

        # Load right[j,k,d] → shape (BLOCK_J, BLOCK_K, BLOCK_D)
        right_ptrs = right_ptr \
            + pid_b * stride_r_b \
            + j[:, None, None] * stride_r_i \
            + k[None, :, None] * stride_r_k \
            + d[None, None, :] * stride_r_h
        right = tl.load(right_ptrs,
                        mask=(j_mask[:, None, None] &
                              k_mask[None, :, None] &
                              d_mask[None, None, :]),
                        other=0.0)

        # Accumulate across the k‑dimension.
        #   left_vec : (BLOCK_I, BLOCK_D)
        #   right_vec: (BLOCK_J, BLOCK_D)
        #   broadcast → (BLOCK_I, BLOCK_J, BLOCK_D)
        for kk in range(0, BLOCK_K):
            left_vec  = left[:, kk, :]          # (BLOCK_I, BLOCK_D)
            right_vec = right[:, kk, :]         # (BLOCK_J, BLOCK_D)
            acc += left_vec[:, None, :] * right_vec[None, :, :]

    # ----- store ---------------------------------------------------------
    out_ptrs = out_ptr \
        + pid_b * stride_o_b \
        + i[:, None, None] * stride_o_i \
        + j[None, :, None] * stride_o_j \
        + d[None, None, :] * stride_o_h

    tl.store(out_ptrs,
             acc,
             mask=(i_mask[:, None, None] &
                   j_mask[None, :, None] &
                   d_mask[None, None, :]))

def custom_kernel(data):
    """
    Forward pass for the "outgoing" TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, L, L, C] (float32)
        - mask         : torch.Tensor [B, L, L] (bool/int/float) or None
        - weights      : dict of torch.Tensors (see key list below)
        - config       : dict containing ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, L, L, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]               # input last dimension
    hidden_dim = config["hidden_dim"] # projection dimension
    eps = 1e-5

    # ensure contiguous layout
    x = input_tensor.contiguous()

    # --------------------------------------------------------------
    # 1. PyTorch pre‑processing
    # --------------------------------------------------------------
    # LayerNorm over the last dimension
    x = layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # Linear projections (no bias)
    left  = linear(x, weights["left_proj.weight"])
    right = linear(x, weights["right_proj.weight"])

    # Optional mask (broadcast over hidden dimension)
    if mask is not None:
        mask_f = mask.unsqueeze(-1).type_as(left)   # shape [B, L, L, 1]
        left  = left * mask_f
        right = right * mask_f

    # Gating networks (sigmoid(linear(...)))
    left_gate  = sigmoid(linear(x, weights["left_gate.weight"]))
    right_gate = sigmoid(linear(x, weights["right_gate.weight"]))
    out_gate   = sigmoid(linear(x, weights["out_gate.weight"]))

    # Apply gates
    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 2. Triton contraction
    # --------------------------------------------------------------
    B, N, _, _ = left.shape                      # B, seq_len, seq_len, hidden_dim
    out = torch.empty_like(left)                 # placeholder for the contracted result

    # tiling parameters – chosen to fit comfortably in registers on H100
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32
    BLOCK_D = 32                                   # hidden‑dim tile (multiple of 32 in all tests)

    grid = (
        (N + BLOCK_I - 1) // BLOCK_I,   # i‑tiles
        (N + BLOCK_J - 1) // BLOCK_J,   # j‑tiles
        B                               # batch
    )

    # strides of the three tensors (they are contiguous on GPU)
    s_l_b, s_l_i, s_l_k, s_l_h = left.stride()
    s_r_b, s_r_i, s_r_k, s_r_h = right.stride()
    s_o_b, s_o_i, s_o_j, s_o_h = out.stride()

    # loop over the hidden dimension in BLOCK_D‑sized chunks
    for d_off in range(0, hidden_dim, BLOCK_D):
        trimul_outgoing_kernel[grid](
            left,
            right,
            out,

            B, N, hidden_dim,
            s_l_b, s_l_i, s_l_k, s_l_h,
            s_r_b, s_r_i, s_r_k, s_r_h,
            s_o_b, s_o_i, s_o_j, s_o_h,

            d_off,                               # runtime offset inside hidden dim
            BLOCK_I=BLOCK_I,
            BLOCK_J=BLOCK_J,
            BLOCK_K=BLOCK_K,
            BLOCK_D=BLOCK_D,
        )

    # --------------------------------------------------------------
    # 3. Post‑processing (norm, gate, final linear)
    # --------------------------------------------------------------
    out = layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    output = linear(out, weights["to_out.weight"])   # hidden_dim → dim, no bias

    return output
```
