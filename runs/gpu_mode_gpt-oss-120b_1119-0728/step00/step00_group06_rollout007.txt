def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_batch_a, stride_batch_b, stride_batch_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets for this block for a, b, c
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Pointers base for this batch
    a_ptrs = a_ptr + batch_id * stride_batch_a + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_batch_a, stride_batch_b, stride_batch_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # compute masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # compute base address offsets for this batch
    a_batch_ptr = a_ptr + batch_id * stride_batch_a
    b_batch_ptr = b_ptr + batch_id * stride_batch_b
    c_batch_ptr = c_ptr + batch_id * stride_batch_c

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # config
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    eps = 1e-5

    # get weights
    norm_weight = weights['norm.weight']
    norm_bias = weights['norm.bias']
    left_proj_weight = weights['left_proj.weight']
    right_proj_weight = weights['right_proj.weight']
    left_gate_weight = weights['left_gate.weight']
    right_gate_weight = weights['right_gate.weight']
    out_gate_weight = weights['out_gate.weight']
    to_out_norm_weight = weights['to_out_norm.weight']
    to_out_norm_bias = weights['to_out_norm.bias']
    to_out_weight = weights['to_out.weight']

    # Norm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Projections
    left = torch.nn.functional.linear(x, left_proj_weight)   # (B,N,N,hidden_dim)
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Mask
    if not config.get('nomask', False):
        # ensure mask is present
        if mask is None:
            raise ValueError('Mask required but not provided')
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # Gating
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Prepare for batched gemm
    B, N, _, _ = left.shape

    # Permute tensors to (B, hidden, M, K) and (B, hidden, K, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()

    out = out * out_gate

    # Final linear projection to dim
    out = torch.nn.functional.linear(out, to_out_weight)  # shape (B,N,N,dim)

    return out

```python
"""
TriMul outgoing kernel implementation.

The forward pass performs:
1. LayerNorm over the channel dimension.
2. Linear projections to a hidden dimension and optional masked scaling.
3. Gated element-wise multiplication.
4. An “outer‑product” reduction over the second sequence dimension:
   out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d],
   which is equivalently a batched matrix multiplication
   left @ rightᵀ per hidden channel.
   This reduction is executed with a custom Triton kernel that
   fuses the K‑dimension loop and exploits Tensor‑Core arithmetic.
5. LayerNorm + output gate and a final linear projection back to the
   original channel dimension.
All heavy compute (step 4) is performed on the GPU via Triton.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM A (M×K) @ Bᵀ (K×N) → C (M×N)
#   - A: shape (batch, M, K)
#   - B: shape (batch, K, N)   (i.e. the transpose of the original right tensor)
#   - C: shape (batch, M, N)
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_batch_a, stride_batch_b, stride_batch_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)   # batch dimension
    pid_m    = tl.program_id(1)   # block row
    pid_n    = tl.program_id(2)   # block col

    # ------------------------------------------------------------------
    # Offsets for this block
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Pointers to the beginning of the batch
    # ------------------------------------------------------------------
    a_batch_ptr = a_ptr + batch_id * stride_batch_a
    b_batch_ptr = b_ptr + batch_id * stride_batch_b
    c_batch_ptr = c_ptr + batch_id * stride_batch_c

    # ------------------------------------------------------------------
    # Pointers that walk a 2‑D tile of A and B
    # ------------------------------------------------------------------
    a_ptrs = a_batch_ptr + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_batch_ptr + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # ------------------------------------------------------------------
    # Accumulator – we use float32 regardless of the input type to keep
    # numeric stability.
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K in blocks of size BLOCK_K
    # ------------------------------------------------------------------
    for _k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k[None, :] < K),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0)
        # tl.dot computes the inner product over the K axis
        acc += tl.dot(a, b)

        a_ptrs += BLOCK_K * stride_ak   # advance K dimension in A
        b_ptrs += BLOCK_K * stride_bk   # advance K dimension in B

    # ------------------------------------------------------------------
    # Store the result
    # ------------------------------------------------------------------
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc, mask=mask)


def _batched_matmul_triton(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication using the Triton kernel defined above.
    a: (batch, M, K)
    b: (batch, K, N)
    Returns c: (batch, M, N) = a @ b
    """
    assert a.is_contiguous() and b.is_contiguous(), "inputs must be contiguous"
    batch, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2, f"Incompatible inner dimensions {K} vs {K2}"

    c = torch.empty((batch, M, N), dtype=a.dtype, device=a.device)

    # Strides (in elements, not bytes)
    stride_am = a.stride(1)
    stride_ak = a.stride(2)
    stride_batch_a = a.stride(0)

    stride_bk = b.stride(1)
    stride_bn = b.stride(2)
    stride_batch_b = b.stride(0)

    stride_cm = c.stride(1)
    stride_cn = c.stride(2)
    stride_batch_c = c.stride(0)

    # Tunable block sizes – these work well on an H100 for the ranges in the test.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (batch,
            (M + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    _batched_gemm_kernel[grid](
        a, b, c,
        M, N, K,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        stride_batch_a, stride_batch_b, stride_batch_c,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return c


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Arguments:
        data : (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor[bs, seq_len, seq_len, dim] (float32)
            - mask        : torch.Tensor[bs, seq_len, seq_len] (bool/int/float) or None
            - weights     : dict of torch tensors required by the module
            - config      : dict with keys "dim", "hidden_dim", "nomask" (bool)
    Returns:
        torch.Tensor[bs, seq_len, seq_len, dim] – the transformed tensor.
    """
    # Unpack arguments
    input_tensor, mask, weights, config = data

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # LayerNorm over the channel dimension (dim)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # Linear projections to the hidden dimension (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden_dim)
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Optional mask – the reference multiplies left/right * mask before gating
    # ------------------------------------------------------------------
    if not config.get("nomask", False):
        # mask may be bool or int; convert to same dtype and broadcast
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # Gating (sigmoid of a linear projection of the normalized input)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Core TriMul reduction: out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]
    # This is implemented as a batched GEMM left @ rightᵀ per hidden channel.
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape

    # Rearrange tensors so that the hidden dimension becomes part of the batch.
    #   left_perm : (B, hidden_dim, i, k)
    #   right_perm: (B, hidden_dim, k, j)  – note the swap of the two
    #   sequence dimensions for the right tensor.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden_dim, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, hidden_dim, N, N) where dim2 is the K‑axis

    # Collapse (B, hidden_dim) into a single batch dimension accepted by the Triton kernel
    batch = B * hidden_dim
    left_flat = left_perm.view(batch, N, N)   # (batch, M=N, K=N)
    right_flat = right_perm.view(batch, N, N) # (batch, K=N, N)

    # Compute the batched matrix multiplication using Triton.
    out_flat = _batched_matmul_triton(left_flat, right_flat)  # (batch, N, N)

    # Restore the original 4‑D layout.
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # Output LayerNorm, gating and final linear projection back to dim
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate

    out = torch.nn.functional.linear(out, weights["to_out.weight"])  # (B, N, N, dim)

    return out
```

"""
TriMul outgoing kernel (forward only).

The forward pass follows the PyTorch reference:

1. Layer‑norm over the channel dimension `dim`.
2. Linear projections `left_proj`, `right_proj` (no bias) to a hidden dimension `hidden_dim`.
3. Optional mask scaling of `left`/`right`.
4. Gated element‑wise multiplication:
       left  = left  * sigmoid(linear(x, left_gate_weight))
       right = right * sigmoid(linear(x, right_gate_weight))
5. Core “tri‑multiplicative” reduction
       out[i, j, d] = Σ_k left[i, k, d] * right[j, k, d]
   This is equivalent to a batched matrix multiplication
       out_d = left_d @ right_dᵀ
   for each hidden channel `d`.  The reduction is implemented with a custom
   Triton GEMM kernel that fuses the K‑dimension loop and exploits the
   Tensor‑core friendly layout.
6. Layer‑norm + output gate on the hidden dimension, then a final linear
   projection back to the original `dim`.

All heavy computation (step 5) runs on the GPU via Triton.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
#   A : (batch, M, K)   – left tensor of shape (B, hidden, N, N) → (batch, N, N)
#   B : (batch, K, N)   – rightᵀ tensor of shape (B, hidden, N, N) → (batch, N, N)
#   C : (batch, M, N)   – result tensor (batch, N, N)
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_batch_a, stride_batch_b, stride_batch_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)   # batch dimension
    pid_m    = tl.program_id(1)   # row block
    pid_n    = tl.program_id(2)   # col block

    # ------------------------------------------------------------------
    # Offsets of the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Base pointers for the current batch
    # ------------------------------------------------------------------
    a_batch_ptr = a_ptr + batch_id * stride_batch_a
    b_batch_ptr = b_ptr + batch_id * stride_batch_b
    c_batch_ptr = c_ptr + batch_id * stride_batch_c

    # ------------------------------------------------------------------
    # Pointers to a (M×K) and b (K×N) tiles
    # ------------------------------------------------------------------
    a_ptrs = a_batch_ptr + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_batch_ptr + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # ------------------------------------------------------------------
    # Accumulator – use float32 for numerical stability
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the K‑dimension in BLOCK_K sized steps
    # ------------------------------------------------------------------
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k[None, :] < K),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0)
        acc += tl.dot(a, b)               # (BLOCK_M, BLOCK_N)

        a_ptrs += BLOCK_K * stride_ak      # advance K in A
        b_ptrs += BLOCK_K * stride_bk      # advance K in B

    # ------------------------------------------------------------------
    # Store the result tile
    # ------------------------------------------------------------------
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc, mask=mask)


def _batched_matmul_triton(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication using the Triton kernel above.
    a : (batch, M, K)
    b : (batch, K, N)
    Returns c : (batch, M, N) = a @ b
    """
    assert a.is_contiguous() and b.is_contiguous(), "inputs must be contiguous"
    batch, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2, f"inner dimensions mismatch {K} vs {K2}"

    c = torch.empty((batch, M, N), dtype=a.dtype, device=a.device)

    # Strides in element units (not bytes)
    stride_am = a.stride(1)
    stride_ak = a.stride(2)
    stride_batch_a = a.stride(0)

    stride_bk = b.stride(1)
    stride_bn = b.stride(2)
    stride_batch_b = b.stride(0)

    stride_cm = c.stride(1)
    stride_cn = c.stride(2)
    stride_batch_c = c.stride(0)

    # Tunable tile sizes – work well on H100 for the problem sizes in the tests.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch,
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _batched_gemm_kernel[grid](
        a,
        b,
        c,
        M,
        N,
        K,
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        stride_batch_a,
        stride_batch_b,
        stride_batch_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return c


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            * input_tensor : torch.Tensor[bs, seq_len, seq_len, dim] (float32)
            * mask         : torch.Tensor[bs, seq_len, seq_len] (bool/int/float) or None
            * weights      : dict of model parameters
            * config       : dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor[bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. LayerNorm over the last channel dimension (`dim`)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # shape (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2. Linear projections to `hidden_dim` (bias=False)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden)
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask – multiplied *before* gating (as in the reference)
    # ------------------------------------------------------------------
    if not config.get("nomask", False):
        # mask may be bool/int – cast to the same dtype as `left`
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of a linear projection of the normalized input)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core TriMul reduction: out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]
    #    Implemented as a batched GEMM left @ rightᵀ per hidden channel.
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape

    # Rearrange so that the hidden dimension becomes part of the batch.
    # left_perm : (B, hidden, i, k)
    # right_perm: (B, hidden, k, i)  – swap the two sequence axes.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, hidden, N, N)

    # Collapse (B, hidden) into a single batch dimension for the Triton kernel.
    batch = B * hidden_dim
    left_flat = left_perm.view(batch, N, N)   # (batch, M=N, K=N)
    right_flat = right_perm.view(batch, N, N) # (batch, K=N, N)

    # Batched matrix multiplication using Triton.
    out_flat = _batched_matmul_triton(left_flat, right_flat)  # (batch, N, N)

    # Restore the original 4‑D layout: (B, N, N, hidden)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm + output gate, then final linear back to `dim`
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate

    out = torch.nn.functional.linear(out, weights["to_out.weight"])  # (B, N, N, dim)

    return out
