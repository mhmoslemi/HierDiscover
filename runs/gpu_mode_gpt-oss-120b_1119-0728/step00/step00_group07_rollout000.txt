@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_bl, stride_il, stride_kl, stride_hl,
    stride_br, stride_jr, stride_kr, stride_hr,
    stride_bo, stride_io, stride_jo, stride_ho,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_hb = tl.program_id(2)  # batch*head index

    # compute b,h
    b = pid_hb // H
    h = pid_hb % H

    # offsets
    off_left = b * stride_bl + h * stride_hl
    off_right = b * stride_br + h * stride_hr
    off_out = b * stride_bo + h * stride_ho

    # compute block coordinates
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_idx = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_idx < N

        # compute pointers
        left_ptrs = left_ptr + off_left + i[:, None] * stride_il + k_idx[None, :] * stride_kl
        right_ptrs = right_ptr + off_right + j[:, None] * stride_jr + k_idx[None, :] * stride_kr

        left_vals = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        right_vals = tl.load(right_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

"""
TriMul kernel (outgoing version)

The operation computes, for each batch `b` and hidden channel `h`,
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

where `left` and `right` are linear projections of the input tensor.
The heavy "tri‑multiplicative" contraction (a batch of H independent
matrix‑multiplications of size N×N) is performed with a custom Triton
kernel.  All other steps – LayerNorm, linear projections, gating,
masking, final LayerNorm, and output linear projection – are carried out
with PyTorch (float‑32), while the kernel works in FP16 for speed and
accumulates in FP32 for numerical stability.

The kernel is launched on a 3‑D grid:
    (block_i, block_j, batch_head)  where batch_head = b * hidden_dim + h.
Each block computes a tile of size BLOCK_M×BLOCK_N of the output
matrix for a given `(b,h)`.  The inner‑most loop tiles over the reduction
dimension `k` with size BLOCK_K, using `tl.dot` (which performs the
dot‑product across the K‑axis).  The result is stored as FP16.
"""

import torch
import triton
import triton.language as tl
import math
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: batched matrix‑multiplication for the TriMul contraction
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,               # const * left  [B,N,N,H]  (FP16)
    right_ptr,              # const * right [B,N,N,H]  (FP16)
    out_ptr,                # * out         [B,N,N,H]  (FP16)
    B, N, H,                # int32 dimensions
    # strides (in elements) for the three tensors
    stride_bl, stride_il, stride_kl, stride_hl,
    stride_br, stride_jr, stride_kr, stride_hr,
    stride_bo, stride_io, stride_jo, stride_ho,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # -------------------------------------------------------------
    # 1) Identify the current tile
    # -------------------------------------------------------------
    pid_i = tl.program_id(0)      # block index over the output row dimension i
    pid_j = tl.program_id(1)      # block index over the output column dimension j
    pid_hb = tl.program_id(2)     # combined batch & hidden‑channel index

    # Recover original batch `b` and hidden channel `h`
    b = pid_hb // H
    h = pid_hb % H

    # Compute the starting coordinates of the tile
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds rows / columns
    mask_i = i < N
    mask_j = j < N

    # Base offsets for the three tensors (same for all rows/cols in the tile)
    off_left  = b * stride_bl + h * stride_hl          # [b,*,*,h]
    off_right = b * stride_br + h * stride_hr
    off_out   = b * stride_bo + h * stride_ho

    # -------------------------------------------------------------
    # 2) Accumulate over the reduction axis `k`
    # -------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K in tiles of size BLOCK_K
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # Pointers to the current K‑tile of left and right
        # left:   [i, k]  (shape M×K)
        # right:  [j, k]  (shape N×K) – note the second argument uses `j`
        left_ptrs  = left_ptr  + off_left  + i[:, None] * stride_il + k[None, :] * stride_kl
        right_ptrs = right_ptr + off_right + j[:, None] * stride_jr + k[None, :] * stride_kr

        # Load with masking; convert to FP32 for accumulation
        left_tile  = tl.load(left_ptrs,
                             mask=mask_i[:, None] & mask_k[None, :],
                             other=0.0).to(tl.float32)
        right_tile = tl.load(right_ptrs,
                             mask=mask_j[:, None] & mask_k[None, :],
                             other=0.0).to(tl.float32)

        # `tl.dot` sums over the K dimension (inner product of rows)
        acc += tl.dot(left_tile, right_tile)

    # -------------------------------------------------------------
    # 3) Write the tile back to global memory (store as FP16)
    # -------------------------------------------------------------
    out_ptrs = out_ptr + off_out + i[:, None] * stride_io + j[None, :] * stride_jo
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

# ----------------------------------------------------------------------
# Entry point – PyTorch wrapper that builds the full TriMul forward pass
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict]) -> torch.Tensor:
    """
    Forward pass of the "outgoing" TriMul module.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool or float) or None
        - weights_dict : mapping from weight name to torch.Tensor
        - config_dict  : contains "dim", "hidden_dim" and optional "nomask"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) – the TriMul output.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5  # LayerNorm epsilon

    # ------------------------------------------------------------------
    # Convenience: fetch all weight tensors
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    left_proj_weight  = weights["left_proj.weight"]    # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]   # (hidden_dim, dim)

    left_gate_weight  = weights["left_gate.weight"]    # (hidden_dim, dim)
    right_gate_weight = weights["right_gate.weight"]   # (hidden_dim, dim)
    out_gate_weight   = weights["out_gate.weight"]     # (hidden_dim, dim)

    to_out_norm_weight = weights["to_out_norm.weight"]  # (hidden_dim,)
    to_out_norm_bias   = weights["to_out_norm.bias"]    # (hidden_dim,)

    to_out_weight = weights["to_out.weight"]            # (dim, hidden_dim)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (over the last dim)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast on the hidden dimension)
    # ------------------------------------------------------------------
    if not config.get("nomask", False) and mask_tensor is not None:
        # Cast mask to the same dtype and add singleton dim
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid after a linear projection)
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Tri‑multiplicative contraction via Triton
    # ------------------------------------------------------------------
    # Cast to FP16 for the kernel (weights are FP32 → FP16 conversion)
    left_fp16  = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    out_fp16   = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)

    # Strides in *elements* (not bytes) – required by Triton
    stride_bl, stride_il, stride_kl, stride_hl = left_fp16.stride()
    stride_br, stride_jr, stride_kr, stride_hr = right_fp16.stride()
    stride_bo, stride_io, stride_jo, stride_ho = out_fp16.stride()

    # Tile sizes – chosen to give good occupancy on an H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # blocks over i
        (N + BLOCK_N - 1) // BLOCK_N,   # blocks over j
        B * hidden_dim,                 # combined batch & channel dimension
    )

    # Launch the kernel
    tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        N,
        hidden_dim,
        stride_bl, stride_il, stride_kl, stride_hl,
        stride_br, stride_jr, stride_kr, stride_hr,
        stride_bo, stride_io, stride_jo, stride_ho,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Back to FP32 for the remaining post‑processing
    out = out_fp16.to(torch.float32)

    # ------------------------------------------------------------------
    # 6) Output LayerNorm (over hidden_dim) + out‑gate
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 7) Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_weight)   # bias is None

    return out
