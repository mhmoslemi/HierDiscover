import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k,
    stride_right_b, stride_right_j, stride_right_k,
    stride_out_b, stride_out_i, stride_out_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    # compute batch and hidden index from pid_bh
    batch = pid_bh // H
    hd = pid_bh % H

    # coords
    i_offset = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j_offset = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    # check bounds
    i_mask = i_offset < N
    j_mask = j_offset < N

    # compute base offsets for out
    out_ptrs = out_ptr + batch * stride_out_b + i_offset[:, None] * stride_out_i + j_offset[None, :] * stride_out_j + hd

    # accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, N, BLOCK_K):
        k_offset = k + tl.arange(0, BLOCK_K)
        k_mask = k_offset < N

        # load left tile: shape (BLOCK_I, BLOCK_K)
        left_ptrs = left_ptr + batch * stride_left_b + i_offset[:, None] * stride_left_i + k_offset[None, :] * stride_left_k + hd
        left = tl.load(left_ptrs, mask=i_mask[:, None] & k_mask[None, :], other=0.0).to(tl.float16)

        # load right tile: shape (BLOCK_J, BLOCK_K)
        right_ptrs = right_ptr + batch * stride_right_b + j_offset[:, None] * stride_right_j + k_offset[None, :] * stride_right_k + hd
        right = tl.load(right_ptrs, mask=j_mask[:, None] & k_mask[None, :], other=0.0).to(tl.float16)

        # compute dot: left (I,K) * right (J,K)^T => (I,J)
        acc += tl.dot(left, right, trans_b=True)

    # store output
    tl.store(out_ptrs, acc, mask=i_mask[:, None] & j_mask[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype = input_tensor.dtype

    # LayerNorm
    x_norm = torch.nn.functional.layer_norm(input_tensor,
                                            normalized_shape=(dim,),
                                            weight=weights["norm.weight"],
                                            bias=weights["norm.bias"])

    # Projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])  # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # Apply mask if needed
    if not nomask:
        mask_ = mask.to(dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # Gates
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()
    left = left * left_gate
    right = right * right_gate

    # Compute triangle multiplicative update using Triton
    out = tri_mul_forward(left, right)  # returns float32

    # Apply out_norm and gating, final linear
    out = torch.nn.functional.layer_norm(out,
                                         normalized_shape=(hidden_dim,),
                                         weight=weights["to_out_norm.weight"],
                                         bias=weights["to_out_norm.bias"])
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
```

    B, N, N, H = left.shape

"""
TriMul (outgoing) kernel accelerated with Triton.

Algorithm
---------
1. Layer‑norm the input tensor (shape [B, N, N, C]).
2. Linear projections to a hidden dimension H:
       left  = x_norm @ W_left      (shape [B,N,N,H])
       right = x_norm @ W_right
3. Apply optional pair mask (broadcasted on the hidden dim).
4. Compute element‑wise gates (sigmoid) and multiply:
       left  = left  * sigmoid(x_norm @ W_left_gate)
       right = right * sigmoid(x_norm @ W_right_gate)
5. Core “triangle‑multiplication”:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication for each hidden slice.
   The heavy reduction is performed in a fused Triton kernel
   (block‑wise GEMM over (i,k) × (j,k)ᵀ).
6. Layer‑norm over the hidden dimension, apply the output gate
   and a final linear projection back to the original channel dimension C.

The kernel works on half‑precision tensors for bandwidth‑limited
memory traffic while accumulating in FP32 for numerical stability.
"""

import torch
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernel: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# --------------------------------------------------------------------------- #
@triton.jit
def tri_mul_kernel(
    left_ptr,            # * half   [B, N, N, H]
    right_ptr,           # * half   [B, N, N, H]
    out_ptr,             # * float  [B, N, N, H]

    B, N, H,                         # runtime sizes
    stride_left_b, stride_left_i, stride_left_k,   # strides for left (B,i,k,h)
    stride_right_b, stride_right_j, stride_right_k, # strides for right (B,j,k,h)
    stride_out_b, stride_out_i, stride_out_j,       # strides for out (B,i,j,h)

    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)               # i‑tile
    pid_j = tl.program_id(1)               # j‑tile
    pid_bh = tl.program_id(2)              # combined batch * hidden index

    # ------------------------------------------------------------------- #
    # Decode batch and hidden‑dim index from the third program id
    # ------------------------------------------------------------------- #
    batch = pid_bh // H
    hid   = pid_bh %  H

    # ------------------------------------------------------------------- #
    # Compute absolute coordinates for this tile
    # ------------------------------------------------------------------- #
    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = i < N
    mask_j = j < N

    # Output pointer for the current tile (shape BLOCK_I × BLOCK_J)
    out_tile_ptr = (
        out_ptr
        + batch * stride_out_b
        + i[:, None] * stride_out_i
        + j[None, :] * stride_out_j
        + hid
    )

    # ------------------------------------------------------------------- #
    # Accumulator in FP32
    # ------------------------------------------------------------------- #
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # Number of k‑tiles
    num_k = tl.cdiv(N, BLOCK_K)

    for k_tile in range(num_k):
        k_base = k_tile * BLOCK_K
        k = k_base + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # ----------------------------------------------------------------
        # Load left tile  [BLOCK_I, BLOCK_K]  (i,k) slice
        # ----------------------------------------------------------------
        left_tile_ptr = (
            left_ptr
            + batch * stride_left_b
            + i[:, None] * stride_left_i
            + k[None, :] * stride_left_k
            + hid
        )
        left = tl.load(left_tile_ptr,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0,
                       dtype=tl.float16)

        # ----------------------------------------------------------------
        # Load right tile [BLOCK_J, BLOCK_K]  (j,k) slice
        # ----------------------------------------------------------------
        right_tile_ptr = (
            right_ptr
            + batch * stride_right_b
            + j[:, None] * stride_right_j
            + k[None, :] * stride_right_k
            + hid
        )
        right = tl.load(right_tile_ptr,
                        mask=mask_j[:, None] & mask_k[None, :],
                        other=0.0,
                        dtype=tl.float16)

        # ----------------------------------------------------------------
        # Dot product over the k‑dimension:
        #   left  (I,K)  ×  rightᵀ (K,J)  → (I,J)
        # ----------------------------------------------------------------
        acc += tl.dot(left, right, trans_b=True)   # result promoted to FP32

    # ------------------------------------------------------------------- #
    # Write the result tile (respecting out‑of‑bounds mask)
    # ------------------------------------------------------------------- #
    tl.store(out_tile_ptr, acc, mask=mask_i[:, None] & mask_j[None, :])


# --------------------------------------------------------------------------- #
# Helper that launches the Triton kernel
# --------------------------------------------------------------------------- #
def tri_mul_forward(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left/right: [B, N, N, H]  (float32)
    returns:    [B, N, N, H]  (float32)  = Σ_k left[...,k] * right[...,k]
    """
    assert left.shape == right.shape
    B, N, _, H = left.shape

    # Cast to fp16 for the kernel (half‑precision reduces bandwidth)
    left_h  = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Output buffer (FP32 accumulation)
    out = torch.empty_like(left, dtype=torch.float32, device=left.device)

    # ------------------------------------------------------------------- #
    # Tile sizes – chosen to fit comfortably in shared memory on H100
    # ------------------------------------------------------------------- #
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_I - 1) // BLOCK_I,                         # i‑tiles
        (N + BLOCK_J - 1) // BLOCK_J,                         # j‑tiles
        B * H                                                 # batch × hidden
    )

    # Strides (in elements) for the three tensors
    s_li = left_h.stride(0)   # batch stride
    s_ii = left_h.stride(1)   # i‑stride
    s_ki = left_h.stride(2)   # k‑stride
    # for right the i‑dimension is named j, but stride is identical
    s_rj = right_h.stride(1)
    s_kj = right_h.stride(2)
    s_rb = right_h.stride(0)

    s_ob = out.stride(0)
    s_oi = out.stride(1)
    s_oj = out.stride(2)

    tri_mul_kernel[grid](
        left_h, right_h, out,
        B, N, H,
        s_li, s_ii, s_ki,
        s_rb, s_rj, s_kj,
        s_ob, s_oi, s_oj,
        BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J, BLOCK_K=BLOCK_K,
        num_warps=4,  # a reasonable default for H100
    )
    return out


# --------------------------------------------------------------------------- #
# Main entry point expected by the test harness
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    TriMul (outgoing) forward pass.
    Args:
        data: Tuple[input_tensor, mask, weights, config]
            - input_tensor: torch.Tensor [B, N, N, C] (float32)
            - mask:        torch.Tensor [B, N, N] (bool or float)
            - weights: dict with all weight tensors (no bias for linear layers)
            - config:  dict with keys “dim”, “hidden_dim”, “nomask”
    Returns:
        torch.Tensor [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]               # C
    hidden_dim = config["hidden_dim"] # H
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------- #
    # 1. Input LayerNorm (over the channel dimension C)
    # ------------------------------------------------------------------- #
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------- #
    # 2. Linear projections to hidden dimension H (no bias)
    # ------------------------------------------------------------------- #
    left  = torch.nn.functional.linear(x, weights["left_proj.weight"])   # [B,N,N,H]
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------- #
    # 3. Optional pair mask (broadcast on H)
    # ------------------------------------------------------------------- #
    if not nomask:
        # mask may be bool; cast to the same dtype as the tensors
        mask_f = mask.to(dtype).unsqueeze(-1)   # [B,N,N,1]
        left  = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------- #
    # 4. Gating (sigmoid of linear projections of the normalized input)
    # ------------------------------------------------------------------- #
    left_gate  = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate   = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 5. Core triangular multiplicative update (Triton)
    # ------------------------------------------------------------------- #
    out = tri_mul_forward(left, right)   # [B,N,N,H] float32

    # ------------------------------------------------------------------- #
    # 6. Output LayerNorm + gating + final projection back to C
    # ------------------------------------------------------------------- #
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
```

**Explanation of key choices**

* **Memory layout** – The input tensors are stored as `[B, N, N, H]` (row‑major).  
  The Triton kernel treats the batch and hidden dimension as a single fused axis,
  thereby launching `B·H` independent matrix‑multiply kernels.

* **Block sizes** – `BLOCK_I = BLOCK_J = 64` and `BLOCK_K = 32` give a
  `(64×64)` output tile while keeping the shared‑memory footprint well below
  the 64 KB limit on an H100.  These sizes deliver good occupancy for the
  `N ∈ {256,…,1024}` range.

* **Precision** – Projections and gates stay in FP32 for numerical stability.
  The heavy reduction (`Σ_k left·right`) is performed in FP16 to halve the
  memory traffic; accumulation is carried out in FP32 (`tl.zeros(..., dtype=tl.float32)`).

* **Mask handling** – The mask is applied before the kernel, reducing the
  amount of work inside the kernel.  When `nomask=True` the mask step is skipped.

* **LayerNorm & linear layers** – Implemented with PyTorch’s highly‑optimized
  `torch.nn.functional` primitives; they are not the bottleneck compared to the
  O(N³) triangle multiplication.

This self‑contained implementation follows the reference PyTorch forward pass,
while the computationally intensive part is fully fused in a custom Triton kernel,
meeting the correctness and performance expectations for the specified test
configurations.
