                        stride_bk, stride_bn, stride_batch_b,

    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_batch_a,
    stride_bk, stride_bn, stride_batch_b,

    tile_m = tl.program_id(1)
    tile_n = tl.program_id(2)

    # compute offsets (like we defined)
    offs_m = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointer for batch
    a_base = a_ptr + batch_idx * stride_batch_a
    b_base = b_ptr + batch_idx * stride_batch_b
    c_base = c_ptr + batch_idx * stride_batch_c

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # a pointers: row m, col k
        a_ptrs = a_base + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)

        b_ptrs = b_base + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)

    input_tensor, mask, weights, config = data
    # Retrieve dims
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # LayerNorm
    x_norm = torch.nn.functional.layer_norm(input_tensor,
                                            normalized_shape=(dim,),
                                            weight=weights["norm.weight"],
                                            bias=weights["norm.bias"],
                                            eps=1e-5)

    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])
    # Linear gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))

        mask_ = mask.unsqueeze(-1).type_as(left)

        right = right * mask_

    right = right * right_gate

"""
TriMul “outgoing” forward pass with a fused batched GEMM in Triton.

Algorithm
---------
1. Layer‑norm the input tensor X ∈ (B,N,N,D) with the provided
   `norm.weight` / `norm.bias`.
2. Linear projections:
      L = X·W_left_proj      (B,N,N,H)
      R = X·W_right_proj     (B,N,N,H)
   (H = hidden_dim)
3. Linear gate values (sigmoid):
      G_L = σ(X·W_left_gate)      (B,N,N,H)
      G_R = σ(X·W_right_gate)     (B,N,N,H)
      G_O = σ(X·W_out_gate)       (B,N,N,H)
4. Optional mask (shape B,N,N) is broadcast and applied to L,R.
5. Apply the per‑channel gates:
      L ← L * G_L
      R ← R * G_R
6. Compute the “tri‑multiplicative” update:
      out_ijh = Σ_k L_i k h * R_j k h   (for every batch and head)
   This heavy O(N³·H) operation is performed as a batched GEMM:
      for each head h, compute L_h @ R_hᵀ.
   The batched GEMM is implemented in a low‑level Triton kernel
   `batched_gemm_forward`.
7. Reshape the result back to (B,N,N,H) and apply a second LayerNorm
   (`to_out_norm`).
8. Multiply by the output gate G_O and apply the final linear
   projection (`to_out`) to obtain a tensor of shape (B,N,N,D).

The kernel works with arbitrary batch size, sequence length and hidden
dimension.  All heavy linear algebra is done in float32; the kernel
stores the result directly in float32 to avoid extra casts.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


@triton.jit
def batched_gemm_forward(
    a_ptr,               # A matrix batch pointer
    b_ptr,               # B matrix batch pointer (already transposed on the K axis)
    c_ptr,               # C output pointer
    M, N, K,             # matrix dimensions
    stride_am, stride_ak, stride_batch_a,   # strides for A: (row, col, batch)
    stride_bk, stride_bn, stride_batch_b,   # strides for B: (row = K, col = N, batch)
    stride_cm, stride_cn, stride_batch_c,   # strides for C: (row, col, batch)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ B   where A,B have shape (batch, M, K) and (batch, K, N)."""
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # --------------------------------------------------------------
    # Tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_batch = a_ptr + batch_idx * stride_batch_a
    b_batch = b_ptr + batch_idx * stride_batch_b
    c_batch = c_ptr + batch_idx * stride_batch_c

    # --------------------------------------------------------------
    # Accumulator
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension in tiles
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # Pointers for the current tile
        a_ptrs = a_batch + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
        b_ptrs = b_batch + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)

        # Load tiles (out‑of‑bounds elements are zeroed)
        a = tl.load(a_ptrs,
                    mask=(offs_m[:, None] < M) & (offs_k[None, :] < K),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & (offs_n[None, :] < N),
                    other=0.0)

        # Dot‑product accumulation
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # Write back the result
    # --------------------------------------------------------------
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc,
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape (B, N, N, D)
        - mask         : torch.Tensor of shape (B, N, N) or None
        - weights      : dict containing the model weights
        - config       : dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor of shape (B, N, N, D)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) Input layer‑norm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # (B,N,N,D)

    # ------------------------------------------------------------------
    # 2) Linear projections & gates (bias‑free)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])        # (B,N,N,H)
    right = F.linear(x, weights["right_proj.weight"])      # (B,N,N,H)

    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # (B,N,N,H)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])) # (B,N,N,H)
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))     # (B,N,N,H)

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_exp = mask.unsqueeze(-1).type_as(left)   # (B,N,N,1)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4) Apply per‑channel gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Batched GEMM (Tri‑multiplicative update)
    #    Compute out_ijh = Σ_k left_i k h * right_j k h
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    device = left.device

    # Rearrange tensors to (B, H, N, N) and then flatten batch & head
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()         # (B, H, N, N) – transposed on K axis

    batch_heads = B * hidden_dim
    left_mat = left_perm.view(batch_heads, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(batch_heads, N, N) # (B*H, N, N)

    # Output buffer
    out_mat = torch.empty_like(left_mat)  # float32

    # Triton launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        batch_heads,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    # Strides (in elements, not bytes)
    stride_am = left_mat.stride(1)      # stride for M dimension (row)
    stride_ak = left_mat.stride(2)      # stride for K dimension (col)
    stride_batch_a = left_mat.stride(0)

    stride_bk = right_mat.stride(1)     # row stride (K)
    stride_bn = right_mat.stride(2)     # col stride (N)
    stride_batch_b = right_mat.stride(0)

    stride_cm = out_mat.stride(1)
    stride_cn = out_mat.stride(2)
    stride_batch_c = out_mat.stride(0)

    batched_gemm_forward[grid](
        left_mat,
        right_mat,
        out_mat,
        N,        # M
        N,        # N (output cols)
        N,        # K
        stride_am,
        stride_ak,
        stride_batch_a,
        stride_bk,
        stride_bn,
        stride_batch_b,
        stride_cm,
        stride_cn,
        stride_batch_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    torch.cuda.synchronize()  # make sure kernel has finished

    # ------------------------------------------------------------------
    # 6) Reshape back to (B,N,N,H)
    # ------------------------------------------------------------------
    out_hidden = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 7) Second LayerNorm, output gate and final linear projection
    # ------------------------------------------------------------------
    out_hidden = F.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out_hidden = out_hidden * out_gate

    output = F.linear(out_hidden, weights["to_out.weight"])   # (B,N,N,D)

    return output
