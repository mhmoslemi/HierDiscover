    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    
    # Extract weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]
    
    # Compose forward
    with torch.no_grad():
        # LayerNorm
        x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)
        # Linear projections (float32)
        left_proj = torch.nn.functional.linear(x, left_proj_weight)
        right_proj = torch.nn.functional.linear(x, right_proj_weight)
        left_gate_pre = torch.nn.functional.linear(x, left_gate_weight)
        right_gate_pre = torch.nn.functional.linear(x, right_gate_weight)
        out_gate_pre = torch.nn.functional.linear(x, out_gate_weight)
        # Sigmoid gates
        left_gate = torch.sigmoid(left_gate_pre)
        right_gate = torch.sigmoid(right_gate_pre)
        out_gate = torch.sigmoid(out_gate_pre)
        # Apply mask if needed
        if not nomask and mask is not None:
            mask_unsqueeze = mask[..., None].to(x.dtype)
            left_proj = left_proj * mask_unsqueeze
            right_proj = right_proj * mask_unsqueeze
        # Apply gating
        left = left_proj * left_gate
        right = right_proj * right_gate
        # Cast to half for kernel
        left = left.to(torch.float16)
        right = right.to(torch.float16)
        # Permute to (B, hidden, N, N)
        left_perm = left.permute(0, 3, 1, 2).contiguous()
        right_perm = right.permute(0, 3, 1, 2).contiguous()
        # Output buffer
        out_perm = torch.empty_like(left_perm, dtype=torch.float16)
        # Launch kernel
        B, N, _, _ = input_tensor.shape
        # hidden = hidden_dim
        # Retrieve strides
        stride_lb, stride_lh, stride_li, stride_lk = left_perm.stride()
        stride_rb, stride_rh, stride_rj, stride_rk = right_perm.stride()
        stride_ob, stride_oh, stride_oi, stride_oj = out_perm.stride()
        # Determine block sizes
        BLOCK_M = 64
        BLOCK_N = 64
        BLOCK_K = 64

            hidden_dim,
            N,

            stride_lh,
            stride_li,
            stride_lk,
            stride_rb,
            stride_rh,
            stride_rj,
            stride_rk,
            stride_ob,
            stride_oh,
            stride_oi,
            stride_oj,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
        )
        # Permute back to (B,N,N,hidden)
        out = out_perm.permute(0, 2, 3, 1).contiguous()

        out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

"""
TriMul (outgoing) implementation with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor over the last dimension.
2. Linear projections (no bias) -> left_proj, right_proj.
3. Compute three sigmoidal gates from the same normalized input.
4. Apply optional mask, then multiply each projection by its gate.
5. The core “tri‑multiplicative” update:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is a batch of matrix multiplications left_d @ right_dᵀ
   for each hidden dimension d.  This step is performed by a custom
   Triton kernel that processes a combined (batch, head) dimension.
6. Layer‑norm over the hidden dimension, multiply by the output gate,
   and a final linear projection back to the original channel size.
All heavy arithmetic (step 5) runs in fp16 with fp32 accumulation and
produces fp16 output that is cast back to fp32 for the remaining
PyTorch operations.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr,    # ptr to left tensor (B, H, N, N)  fp16
    right_ptr,   # ptr to right tensor (B, H, N, N) fp16
    out_ptr,     # ptr to output tensor (B, H, N, N) fp16
    B: tl.constexpr,
    H: tl.constexpr,
    N: tl.constexpr,
    K: tl.constexpr,          # = N (shared reduction axis)
    stride_left_batch,  # stride for batch dimension in left
    stride_left_head,   # stride for head (hidden) dimension in left
    stride_left_i,      # stride for i (first N) dimension in left
    stride_left_k,      # stride for k (second N) dimension in left
    stride_right_batch, # stride for batch dimension in right
    stride_right_head,  # stride for head (hidden) dimension in right
    stride_right_j,     # stride for j (first N) dimension in right
    stride_right_k,     # stride for k (second N) dimension in right
    stride_out_batch,   # stride for batch dimension in out
    stride_out_head,    # stride for head dimension in out
    stride_out_i,       # stride for i dimension in out
    stride_out_j,       # stride for j dimension in out
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ Bᵀ  where
        A  = left[b, h, i, k]   (M=N, K=N)
        B  = right[b, h, j, k]  (N=N, K=N)
    Output:
        out[b, h, i, j] = Σ_k A[i,k] * B[j,k]
    The kernel processes a fused batch*head dimension (pid0) and tiles
    the M×N output plane.
    """
    pid = tl.program_id(0)                # combined batch*head
    pid_m = tl.program_id(1)              # i‑tile
    pid_n = tl.program_id(2)              # j‑tile

    # Decompose pid into batch index b and head index h
    b = pid // H
    h = pid % H

    # Starting offsets for this tile
    i_start = pid_m * BLOCK_M
    j_start = pid_n * BLOCK_N

    # Offsets within the tile (constexpr)
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_i = offs_i < N
    mask_j = offs_j < N
    num_k_blocks = tl.cdiv(K, BLOCK_K)

    # Base pointers for this (b, h) slice
    base_left = left_ptr + b * stride_left_batch + h * stride_left_head
    base_right = right_ptr + b * stride_right_batch + h * stride_right_head
    base_out = out_ptr + b * stride_out_batch + h * stride_out_head

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks
    for kk in range(0, num_k_blocks):
        cur_k = kk * BLOCK_K + offs_k
        mask_k = cur_k < K

        # A[i, k] tile   (M×K)
        a_ptrs = base_left + offs_i[:, None] * stride_left_i + cur_k[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=(mask_i[:, None] & mask_k[None, :]),
                    other=0.0).to(tl.float32)

        # B[j, k] tile   (N×K)  -> need (K×N) for dot
        b_ptrs = base_right + cur_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_j
        b = tl.load(b_ptrs,
                    mask=(mask_k[:, None] & mask_j[None, :]),
                    other=0.0).to(tl.float32)

        # Accumulate dot product over K
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N)

    # Write result back in fp16
    c_ptrs = base_out + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.
    Input:
        data = (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Load model parameters (all are assumed to be float32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. Layer‑norm over the channel dimension
    # ------------------------------------------------------------------
    with torch.no_grad():
        x = torch.nn.functional.layer_norm(
            input_tensor,
            (dim,),
            weight=norm_weight,
            bias=norm_bias,
        )  # [B, N, N, dim]  float32

        # ------------------------------------------------------------------
        # 2. Linear projections + gates (still in float32)
        # ------------------------------------------------------------------
        left_proj = torch.nn.functional.linear(x, left_proj_weight)          # [B,N,N,hidden]
        right_proj = torch.nn.functional.linear(x, right_proj_weight)
        left_gate_pre = torch.nn.functional.linear(x, left_gate_weight)
        right_gate_pre = torch.nn.functional.linear(x, right_gate_weight)
        out_gate_pre = torch.nn.functional.linear(x, out_gate_weight)

        left_gate = torch.sigmoid(left_gate_pre)
        right_gate = torch.sigmoid(right_gate_pre)
        out_gate = torch.sigmoid(out_gate_pre)

        # ------------------------------------------------------------------
        # 3. Optional mask (broadcasted on the hidden dimension)
        # ------------------------------------------------------------------
        if (not nomask) and (mask is not None):
            mask_ = mask.unsqueeze(-1).to(x.dtype)          # [B,N,N,1]
            left_proj = left_proj * mask_
            right_proj = right_proj * mask_

        # ------------------------------------------------------------------
        # 4. Apply gates
        # ------------------------------------------------------------------
        left = left_proj * left_gate          # [B,N,N,hidden]
        right = right_proj * right_gate

        # ------------------------------------------------------------------
        # 5. Cast to fp16 and permute to (B, head, N, N) for the kernel
        # ------------------------------------------------------------------
        left = left.to(torch.float16)
        right = right.to(torch.float16)
        left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
        right_perm = right.permute(0, 3, 1, 2).contiguous()

        B, N, _, _ = input_tensor.shape
        H = hidden_dim

        # Output buffer for the Triton kernel
        out_perm = torch.empty_like(left_perm, dtype=torch.float16)

        # ------------------------------------------------------------------
        # 6. Launch the fused Triton kernel (C = left @ rightᵀ per head)
        # ------------------------------------------------------------------
        stride_lb, stride_lh, stride_li, stride_lk = left_perm.stride()
        stride_rb, stride_rh, stride_rj, stride_rk = right_perm.stride()
        stride_ob, stride_oh, stride_oi, stride_oj = out_perm.stride()

        # Tunable tile sizes (empirically good for H100 FP16)
        BLOCK_M = 64
        BLOCK_N = 64
        BLOCK_K = 64

        grid = (
            B * H,                               # combined batch*head
            (N + BLOCK_M - 1) // BLOCK_M,        # i‑tiles
            (N + BLOCK_N - 1) // BLOCK_N,        # j‑tiles
        )

        tri_mul_kernel[grid](
            left_perm,
            right_perm,
            out_perm,
            B,
            H,
            N,
            N,                     # K == N
            stride_lb,
            stride_lh,
            stride_li,
            stride_lk,
            stride_rb,
            stride_rh,
            stride_rj,
            stride_rk,
            stride_ob,
            stride_oh,
            stride_oi,
            stride_oj,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
        )

        # ------------------------------------------------------------------
        # 7. Reshape back to (B, N, N, hidden) and cast to float32
        # ------------------------------------------------------------------
        out = out_perm.permute(0, 2, 3, 1).contiguous().to(torch.float32)  # [B,N,N,hidden]

        # ------------------------------------------------------------------
        # 8. Final hidden‑dim layer‑norm, gate, and projection
        # ------------------------------------------------------------------
        out = torch.nn.functional.layer_norm(
            out,
            (hidden_dim,),
            weight=to_out_norm_weight,
            bias=to_out_norm_bias,
        )
        out = out * out_gate.to(torch.float32)          # broadcast gate
        out = torch.nn.functional.linear(out, to_out_weight)  # [B,N,N,dim]

    return out
