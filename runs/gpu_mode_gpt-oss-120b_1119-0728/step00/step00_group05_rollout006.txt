    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    batch_size, seq_len, _, _ = input_tensor.shape

    # Load weights as dict
    # bias may be optional for LayerNorm but given dictionary includes bias.

    # 1. LayerNorm
    # compute norm: use torch.nn.functional.layer_norm
    # norm_weight, norm_bias
    # Since we need to apply per last dimension.
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # 2. Linear projections: left_proj, right_proj
    left = torch.nn.functional.linear(x, left_proj_weight)  # shape B,N,N,hidden_dim
    right = torch.nn.functional.linear(x, right_proj_weight)

    # 3. Apply mask if exists (mask may be None)
    if mask is not None:

        left = left * mask
        right = right * mask

    # 4. Compute gates:
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # left,right now shape B,N,N,hidden_dim

    # 5. Compute core tri-mul using Triton
    out = trimul_kernel(left, right)  # expects to return same shape

    # 6. to_out_norm + gating
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    out = out * out_gate  # \in shape B,N,N,hidden_dim

    # 7. final linear to dim
    out = torch.nn.functional.linear(out, to_out_weight)  # shape B,N,N,dim

    import torch
    import triton
    import triton.language as tl

    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # Extract weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # 1. LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # 2. Projections
    # shape: (B, N, N, hidden_dim)
    # flatten for linear
    B, N, _, _ = x.shape
    x_flat = x.reshape(-1, dim)  # (B*N*N, dim)
    left = torch.nn.functional.linear(x_flat, left_proj_weight)  # returns (B*N*N, hidden_dim)
    left = left.view(B, N, N, hidden_dim)
    right = torch.nn.functional.linear(x_flat, right_proj_weight).view(B, N, N, hidden_dim)

    # 3. mask
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)  # B,N,N,1
        left = left * mask_exp
        right = right * mask_exp

    # 4. Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, left_gate_weight)).view(B, N, N, hidden_dim)
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, right_gate_weight)).view(B, N, N, hidden_dim)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, out_gate_weight)).view(B, N, N, hidden_dim)

    left = left * left_gate
    right = right * right_gate

    # 5. Core tri-mul
    # Use Triton kernel:
    out = trimul_forward(left, right)   # returns B,N,N,hidden_dim

    # 6. Output norm and gating
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    out = out * out_gate
    # 7. Final linear
    out_flat = out.reshape(-1, hidden_dim)
    out = torch.nn.functional.linear(out_flat, to_out_weight).view(B, N, N, dim)

"""
TriMul (outgoing) forward pass – Triton‑accelerated core.

Algorithm
---------
1. Layer‑norm the input tensor over the last dimension (dim).
2. Linear projections (left_proj / right_proj) → (B,N,N,hidden_dim).
3. Optional pairwise mask multiplication.
4. Compute per‑position gates (left_gate, right_gate, out_gate) via a sigmoid‑linear,
   then apply left/right gates to the projected tensors.
5. Core “tri‑multiplicative” update:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d].
   This is a batched matrix multiply where the reduction is performed per hidden
   channel.  The reduction is realised in a custom Triton kernel that processes
   tiles of (i,j) and a block of the hidden dimension d.
6. Layer‑norm + out‑gate on the hidden dimension.
7. Final linear projection back to dim.

Only the O(N³·hidden_dim) reduction is done in Triton; all other
operations stay in PyTorch (high‑throughput linear layers, LayerNorm,
etc.).  The kernel works for any batch size B≤2, any sequence length N,
and hidden dimensions that are a multiple of the chosen BLOCK_D (128)
or will be masked for the remainder.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _trimul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,  # ptrs
    B, N, D,                       # runtime sizes
    # strides (in elements)
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_i, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    # compile‑time block sizes
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """Batched (i,k,d) × (j,k,d) → (i,j,d) reduction.
       One program processes a tile (BLOCK_I × BLOCK_J) of the
       (i,j) plane and a contiguous BLOCK_D slice of the hidden dimension.
    """
    pid_i = tl.program_id(0)          # tile in i‑direction
    pid_j = tl.program_id(1)          # tile in j‑direction
    pid_z = tl.program_id(2)          # batch * d‑block

    # -------------------------------------------------------------
    # Resolve batch index and hidden‑dim block from the third axis
    # -------------------------------------------------------------
    num_d_blocks = tl.cdiv(D, BLOCK_D)
    batch_idx = pid_z // num_d_blocks
    d_block   = pid_z % num_d_blocks

    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J
    d_start = d_block * BLOCK_D

    # -------------------------------------------------------------
    # Offsets for the current tile
    # -------------------------------------------------------------
    offs_i = i_start + tl.arange(0, BLOCK_I)
    offs_j = j_start + tl.arange(0, BLOCK_J)
    offs_d = d_start + tl.arange(0, BLOCK_D)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_d = offs_d < D

    # Accumulator for the output tile (float32 for numerical safety)
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

    # -------------------------------------------------------------
    # Loop over the reduction dimension k in blocks
    # -------------------------------------------------------------
    for start_k in range(0, N, BLOCK_K):
        cur_k = start_k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # ------------------- left tile (i,k,d) -------------------
        left_ptrs = left_ptr + \
            batch_idx * stride_left_b + \
            offs_i[:, None, None] * stride_left_i + \
            cur_k[None, :, None] * stride_left_k + \
            offs_d[None, None, :] * stride_left_d

        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
            other=0.0,
        )  # (BLOCK_I, BLOCK_K, BLOCK_D)

        # ------------------- right tile (k,j,d) ------------------
        # NOTE: we deliberately place k as the leading dimension so that
        # we can index right[kk, :, :] inside the inner loop.
        right_ptrs = right_ptr + \
            batch_idx * stride_right_b + \
            cur_k[:, None, None] * stride_right_k + \
            offs_j[None, :, None] * stride_right_i + \
            offs_d[None, None, :] * stride_right_d

        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :],
            other=0.0,
        )  # (BLOCK_K, BLOCK_J, BLOCK_D)

        # ------------------- reduction over k --------------------
        # Loop over k inside the block; each iteration does a
        # broadcasted multiply‑add across the hidden dimension.
        for kk in tl.static_range(0, BLOCK_K):
            # left_i_d : (BLOCK_I, BLOCK_D)
            # right_j_d: (BLOCK_J, BLOCK_D)
            l = left[:, kk, :]          # (BLOCK_I, BLOCK_D)
            r = right[kk, :, :]         # (BLOCK_J, BLOCK_D)
            # broadcast to (BLOCK_I, BLOCK_J, BLOCK_D) and accumulate
            acc += l[:, None, :] * r[None, :, :]

    # -------------------------------------------------------------
    # Write the tile back to global memory (float16 output)
    # -------------------------------------------------------------
    out_ptrs = out_ptr + \
        batch_idx * stride_out_b + \
        offs_i[:, None, None] * stride_out_i + \
        offs_j[None, :, None] * stride_out_j + \
        offs_d[None, None, :] * stride_out_d

    tl.store(
        out_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :],
    )


def _trimul_forward(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Triton‑accelerated implementation of
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

    Parameters
    ----------
    left, right : torch.Tensor
        Shape (B, N, N, D), dtype float32 or float16.
        D is the hidden dimension.

    Returns
    -------
    out : torch.Tensor
        Same shape & dtype as inputs (float32 final).
    """
    # Cast to float16 for the kernel (less bandwidth)
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, D = left_h.shape
    out_h = torch.empty_like(left_h, dtype=torch.float16)

    # Gather strides (in elements, not bytes)
    stride_l_b, stride_l_i, stride_l_k, stride_l_d = left_h.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_d = right_h.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_d = out_h.stride()

    # Tuning parameters – chosen to fit H100 SMs and the problem sizes.
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 8
    BLOCK_D = 128  # >= hidden_dim; excess lanes are masked out

    grid = (
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
        B * ((D + BLOCK_D - 1) // BLOCK_D),
    )

    _trimul_matmul_kernel[grid](
        left_h,
        right_h,
        out_h,
        B,
        N,
        D,
        stride_l_b,
        stride_l_i,
        stride_l_k,
        stride_l_d,
        stride_r_b,
        stride_r_i,  # this is the original 'i' stride, used as the 'j' stride
        stride_r_k,
        stride_r_d,
        stride_o_b,
        stride_o_i,
        stride_o_j,
        stride_o_d,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    # Convert back to float32 for the remaining PyTorch pipeline
    return out_h.to(torch.float32)


def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator used in AlphaFold‑3.
    The heavy O(N³·c_hidden) reduction is performed with a fused Triton
    kernel; all other linear / LayerNorm / sigmoid operations use native
    PyTorch kernels for maximum throughput.
    """
    # -----------------------------------------------------------------
    # unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # weight tensors (all are 2‑D except LayerNorm which also has bias)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # -----------------------------------------------------------------
    # 1. Input LayerNorm over the last dimension (dim)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # (B, N, N, dim)

    B, N, _, _ = x.shape

    # -----------------------------------------------------------------
    # 2. Linear projections to hidden_dim (no bias)
    #    We flatten the first three dimensions for an efficient GEMM.
    # -----------------------------------------------------------------
    x_flat = x.reshape(-1, dim)                         # (B*N*N, dim)
    left = F.linear(x_flat, left_proj_weight)           # (B*N*N, hidden_dim)
    right = F.linear(x_flat, right_proj_weight)         # (B*N*N, hidden_dim)

    left = left.view(B, N, N, hidden_dim)
    right = right.view(B, N, N, hidden_dim)

    # -----------------------------------------------------------------
    # 3. Apply optional pairwise mask (shape B,N,N)
    # -----------------------------------------------------------------
    if not nomask:
        # mask could be bool or float – convert to float32 for multiplication
        mask_f = mask.to(dtype=left.dtype).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 4. Compute per‑position gates (sigmoid(linear(x))) and apply
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_flat, left_gate_weight)).view(B, N, N, hidden_dim)
    right_gate = torch.sigmoid(F.linear(x_flat, right_gate_weight)).view(B, N, N, hidden_dim)
    out_gate = torch.sigmoid(F.linear(x_flat, out_gate_weight)).view(B, N, N, hidden_dim)

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core TriMul – custom Triton kernel
    # -----------------------------------------------------------------
    out = _trimul_forward(left, right)   # shape (B,N,N,hidden_dim), dtype float32

    # -----------------------------------------------------------------
    # 6. Output LayerNorm + out‑gate
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate

    # -----------------------------------------------------------------
    # 7. Final linear projection back to original dimension (dim)
    # -----------------------------------------------------------------
    out_flat = out.reshape(-1, hidden_dim)
    out = F.linear(out_flat, to_out_weight).view(B, N, N, dim)

    return out
