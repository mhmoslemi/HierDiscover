def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    epsilon = 1e-5

    # Access weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]
    
    # LayerNorm (norm)
    # compute mean and var over last dim dim
    x = input_tensor
    # x shape [B,N,N,dim]
    # compute mean, var
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + epsilon)
    x_norm = x_norm * norm_weight + norm_bias

    # Compute linear projections
    # shape: [B,N,N,hidden]
    left = torch.nn.functional.linear(x_norm, left_proj_weight)  # out_features=hidden_dim
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # mask
    if mask_tensor is not None:
        mask = mask_tensor.to(x.dtype)  # assume 0/1
        mask = mask.unsqueeze(-1)  # (B,N,N,1)
        left = left * mask
        right = right * mask
    # Gating
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Cast to half for multiplication
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Prepare for batched GEMM
    B, N, _, H = left.shape  # H == hidden_dim

    # Permute to shape (B*H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # allocate output half
    out_mat = torch.empty_like(left_mat, dtype=torch.float16)

    # launch Triton kernel
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32  # can tune
    grid = ( (N + BLOCK_M - 1) // BLOCK_M,
             (N + BLOCK_N - 1) // BLOCK_N,
             B * H )
    triton_matmul_kernel[
        grid,
        # set number of warps maybe
        # optionally pass num_warps=4 etc
    ](
        left_mat,
        right_mat,
        out_mat,
        left_mat.stride(0), left_mat.stride(1), left_mat.stride(2),
        right_mat.stride(0), right_mat.stride(1), right_mat.stride(2),
        out_mat.stride(0), out_mat.stride(1), out_mat.stride(2),
        N, N, N,  # M, N, K (all N)
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    # Reshape back
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # shape (B,N,N,H)

    # to_out_norm (layernorm over hidden dimension)
    out_mean = out.mean(dim=-1, keepdim=True)
    out_var = out.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out - out_mean) * torch.rsqrt(out_var + epsilon)
    out_norm = out_norm * to_out_norm_weight + to_out_norm_bias

    # multiply by out_gate
    out_norm = out_norm * out_gate

    # final linear projection to dim
    out_final = torch.nn.functional.linear(out_norm, to_out_weight)

    return out_final
```

"""
TriMul (outgoing) forward pass implemented with a fused batched GEMM in Triton.

Algorithm
---------
1. Layer‐norm the 4‑D input tensor over the last dimension (dim).
2. Linear projections → left, right  (shape [B,N,N,H]) where H = hidden_dim.
3. Apply optional pairwise mask.
4. Compute per‑position gates (sigmoid of linear projections) and multiply.
5. Cast left/right to fp16 and reshape to (B*H, N, N) so that each
   (batch, hidden) pair becomes an independent GEMM:
        out_{b,h,i,j} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
   which is a mat‑mul left @ rightᵀ.
6. Launch a custom Triton kernel that performs the batched GEMM
   (BLOCK_M=128, BLOCK_N=128, BLOCK_K=32).  The kernel loads tiles of
   A (left) and Bᵀ (right) in fp16, accumulates in fp32 and stores the
   result in fp16.
7. Reshape the result back to [B,N,N,H] and perform a second layer‑norm
   over the hidden dimension.
8. Apply the output gate, then a final linear projection back to `dim`.
9. Return the result in float32.

The heavy O(N³) work is carried out in the Triton kernel, yielding a
large speed‑up over a pure‑Python implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM computing C = A @ Bᵀ
#   A : [B_total, M, K]   (fp16)
#   B : [B_total, N, K]   (fp16)  (we read it transposed)
#   C : [B_total, M, N]   (fp16)
# ----------------------------------------------------------------------
@triton.jit
def triton_matmul_kernel(
    A, B, C,                     # pointers
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    M, N, K,                     # matrix sizes (int)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program IDs
    pid_batch = tl.program_id(2)          # batch * hidden_idx
    pid_m = tl.program_id(0)              # tile row
    pid_n = tl.program_id(1)              # tile col

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # A ptr: [batch, m, k]
        a_ptrs = (
            A
            + pid_batch * stride_a_batch
            + offs_m[:, None] * stride_a_m
            + offs_k[None, :] * stride_a_k
        )
        # Bᵀ ptr: we read B as transposed, i.e. [batch, k, n]
        b_ptrs = (
            B
            + pid_batch * stride_b_batch
            + offs_k[:, None] * stride_b_k
            + offs_n[None, :] * stride_b_n
        )

        # masks for out‑of‑bounds loads
        mask_a = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        mask_b = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)   # fp16
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)   # fp16

        # dot: fp16×fp16 → fp32 accumulation
        acc += tl.dot(a, b)

    # write back to C (fp16)
    c_ptrs = (
        C
        + pid_batch * stride_c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n
    )
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


# ----------------------------------------------------------------------
# Main entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul (outgoing) forward pass.

    Arguments
    ---------
    data : Tuple[torch.Tensor, torch.Tensor, dict, dict]
        - input_tensor : shape [B, N, N, dim] (float32)
        - mask_tensor  : shape [B, N, N] (float32 or bool) ; may be None
        - weights      : dict of model weight tensors
        - config       : dict with keys 'dim', 'hidden_dim', optional 'nomask'

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs & weights
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm parameters (dim)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # Linear weights (no bias)
    left_proj_weight = weights["left_proj.weight"]   # [hidden_dim, dim]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    # Output norm (hidden dim)
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    # Final projection (dim x hidden_dim)
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    # x: [B, N, N, dim]
    x = input_tensor
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    x_norm = x_norm * norm_weight + norm_bias  # broadcast over B,N,N

    # ------------------------------------------------------------------
    # 2) Linear projections
    # ------------------------------------------------------------------
    # [B,N,N,hidden_dim]  (float32)
    left = F.linear(x_norm, left_proj_weight)   # bias=False
    right = F.linear(x_norm, right_proj_weight)

    # ------------------------------------------------------------------
    # 3) Optional mask
    # ------------------------------------------------------------------
    if not config.get("nomask", False):
        if mask_tensor is None:
            raise ValueError("Mask expected but not provided.")
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Cast to fp16 for the massive GEMM
    # ------------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, H = left.shape   # H == hidden_dim
    # Reshape to (B*H, N, N) so each (batch, hidden) pair is a separate GEMM
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Output buffer
    out_mat = torch.empty_like(left_mat, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 6) Launch Triton batched GEMM kernel: out = left @ rightᵀ
    # ------------------------------------------------------------------
    BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 32
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )
    triton_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        # strides for A
        left_mat.stride(0),
        left_mat.stride(1),
        left_mat.stride(2),
        # strides for B
        right_mat.stride(0),
        right_mat.stride(1),
        right_mat.stride(2),
        # strides for C
        out_mat.stride(0),
        out_mat.stride(1),
        out_mat.stride(2),
        N,            # M
        N,            # N
        N,            # K
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,   # H100 supports many warps
    )

    # ------------------------------------------------------------------
    # 7) Reshape back to [B,N,N,hidden_dim]
    # ------------------------------------------------------------------
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # fp16

    # ------------------------------------------------------------------
    # 8) Second LayerNorm over hidden dimension (float32 for stability)
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)
    mean_h = out_fp32.mean(dim=-1, keepdim=True)
    var_h = out_fp32.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out_fp32 - mean_h) * torch.rsqrt(var_h + eps)
    out_norm = out_norm * to_out_norm_weight + to_out_norm_bias    # broadcast

    # ------------------------------------------------------------------
    # 9) Apply output gate and final linear projection (float32)
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate          # out_gate is float32 already
    output = F.linear(out_norm, to_out_weight)   # [B,N,N,dim]

    return output
