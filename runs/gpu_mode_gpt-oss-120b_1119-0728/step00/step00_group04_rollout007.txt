@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
):
    pid_b = tl.program_id(0)  # batch index
    pid_i = tl.program_id(1)  # tile i
    pid_j = tl.program_id(2)  # tile j

    # offsets for i and j
    i = tl.arange(0, BLOCK_M)
    j = tl.arange(0, BLOCK_N)
    h = tl.arange(0, BLOCK_H)

    i_idx = pid_i * BLOCK_M + i
    j_idx = pid_j * BLOCK_N + j

    # Create mask for i,j out-of-bounds
    i_mask = i_idx < N
    j_mask = j_idx < N

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    batch_size, seq_len, _, _ = input_tensor.shape
    device = input_tensor.device

    # LayerNorm using provided weight and bias
    eps = 1e-5

    # Compute mean and var across dim axis
    # x_norm = (x - mean) / sqrt(var + eps) * weight + bias
    # The mean and var per [bs, seq_len, seq_len] (i.e., across last dim)
    # implement using torch.mean and torch.var across -1

    # weight shape [dim], bias shape [dim]
    x = input_tensor
    # compute layernorm
    # We should ensure dtype float32
    # Use torch.mean and torch.var
    x_mean = x.mean(dim=-1, keepdim=True)
    x_var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - x_mean) / torch.sqrt(x_var + eps)
    x_norm = x_norm * weights["norm.weight"] + weights["norm.bias"]  # broadcast

    # Linear projections and gating
    # left_proj: shape [hidden_dim, dim]
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])  # shape [bs,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # mask handling
    if mask is not None and mask.numel()>0:
        # mask shape [bs,N,N]; unsqueeze(-1)
        mask_unsq = mask.unsqueeze(-1).to(x_norm.dtype)
        left = left * mask_unsq
        right = right * mask_unsq

    # Gates
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Now contraction using Triton
    # Allocate output tensor
    out = torch.empty((batch_size, seq_len, seq_len, hidden_dim), dtype=x_norm.dtype, device=device)

    # Compute strides
    stride_left_b = left.stride(0)
    stride_left_i = left.stride(1)
    stride_left_k = left.stride(2)
    stride_left_h = left.stride(3)

    stride_right_b = right.stride(0)
    stride_right_j = right.stride(1)
    stride_right_k = right.stride(2)
    stride_right_h = right.stride(3)

    stride_out_b = out.stride(0)
    stride_out_i = out.stride(1)
    stride_out_j = out.stride(2)
    stride_out_h = out.stride(3)

    # Choose block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    BLOCK_H = 16  # hidden tile

    # Ensure for seq_len not > max block? It's fine.
    grid = (batch_size, triton.cdiv(seq_len, BLOCK_M), triton.cdiv(seq_len, BLOCK_N))

    tri_mul_kernel[grid](
        left, right, out,
        batch_size, seq_len, hidden_dim,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_j, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K, BLOCK_H=BLOCK_H,

    out_mean = out.mean(dim=-1, keepdim=True)

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # offsets
    i = tl.arange(0, BLOCK_M)
    j = tl.arange(0, BLOCK_N)

    i_idx = pid_i * BLOCK_M + i
    j_idx = pid_j * BLOCK_N + j

    i_mask = i_idx < N
    j_mask = j_idx < N

    # hidden loop
    h_offset = 0
    while h_offset < H:
        h = tl.arange(0, BLOCK_H) + h_offset
        h_mask = h < H

        # accumulator
        acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

        # k loop
        k_offset = 0
        while k_offset < N:
            k = tl.arange(0, BLOCK_K) + k_offset
            k_mask = k < N

            # compute ptrs
            # left pointer
            # left_ptr + b*STRB + i_idx[:,None,None]*STRI + k[None,:,None]*STRK + h[None,None,:]*STRH
            left_ptrs = left_ptr + pid_b * stride_left_b \
                + (i_idx[:, None, None] * stride_left_i) \
                + (k[None, :, None] * stride_left_k) \
                + (h[None, None, :] * stride_left_h)
            # mask for left load
            left_mask = i_mask[:, None, None] & k_mask[None, :, None] & h_mask[None, None, :]
            left_tile = tl.load(left_ptrs, mask=left_mask, other=0.0)

            right_ptrs = right_ptr + pid_b * stride_right_b \
                + (j_idx[:, None, None] * stride_right_j) \
                + (k[None, :, None] * stride_right_k) \
                + (h[None, None, :] * stride_right_h)
            right_mask = j_mask[:, None, None] & k_mask[None, :, None] & h_mask[None, None, :]
            right_tile = tl.load(right_ptrs, mask=right_mask, other=0.0)

            # compute partial sum over k slice
            # left_tile: (BLOCK_M, BLOCK_K, BLOCK_H)
            # right_tile: (BLOCK_N, BLOCK_K, BLOCK_H)
            # for each kk in BLOCK_K:
            # Using broadcasting:
            # left_tile[:, kk, :] shape (BLOCK_M, BLOCK_H)
            # right_tile[:, kk, :] shape (BLOCK_N, BLOCK_H)
            # acc += left_tile[:, kk, :][:, None, :] * right_tile[:, kk, :][None, :, :]

            # we can vectorize using a for loop
            for kk in range(0, BLOCK_K):
                # check if kk within valid range (but mask already covers)
                l_vec = left_tile[:, kk, :]    # shape (BLOCK_M, BLOCK_H)
                r_vec = right_tile[:, kk, :]   # shape (BLOCK_N, BLOCK_H)
                # Broadcast and accumulate
                acc += l_vec[:, None, :] * r_vec[None, :, :]

            k_offset += BLOCK_K

        # Store acc to out
        out_ptrs = out_ptr + pid_b * stride_out_b \
            + (i_idx[:, None, None] * stride_out_i) \
            + (j_idx[None, :, None] * stride_out_j) \
            + (h[None, None, :] * stride_out_h)

        out_mask = i_mask[:, None, None] & j_mask[None, :, None] & h_mask[None, None, :]
        tl.store(out_ptrs, acc, mask=out_mask)

        h_offset += BLOCK_H

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    bs, seq_len, _, _ = input_tensor.shape
    device = input_tensor.device
    eps = 1e-5

    # LayerNorm
    # norm weight and bias
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # compute mean & var across dim axis
    x = input_tensor
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * norm_weight + norm_bias

    # Linear projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x_norm, left_proj_weight)
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Apply mask if needed
    if mask is not None and not config.get("nomask", False):
        mask_unsq = mask.unsqueeze(-1).to(x_norm.dtype)
        left = left * mask_unsq
        right = right * mask_unsq

    # Gating
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Allocate output tensor for the contraction
    out = torch.empty((bs, seq_len, seq_len, hidden_dim), dtype=input_tensor.dtype, device=device)

    # Strides
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # Block sizes
    BLOCK_M = 16
    BLOCK_N = 16
    BLOCK_K = 32
    BLOCK_H = 8

    grid = (bs, triton.cdiv(seq_len, BLOCK_M), triton.cdiv(seq_len, BLOCK_N))

    )

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]

"""
TriMul "outgoing" implementation.

The kernel computes the core O(N³·C) contraction:
    out[b,i,j,c] = Σ_k left[b,i,k,c] * right[b,j,k,c]

where `left` and `right` are the linearly‑projected and gated tensors
of shape [B, N, N, C_hidden].  The contraction is carried out by a
custom Triton kernel that tiles the (i, j) output plane and a small
chunk of the hidden dimension (C_hidden).  Within each tile we load a
(A‑tile) [M×K×H] and a (B‑tile) [N×K×H] and accumulate the outer
product across the reduction dimension K.  The remaining operations
(LayerNorm, gating, final linear) are performed in PyTorch.

The kernel is written for float32 (input is float32; the kernel can be
easily adapted to fp16).  Block sizes were chosen to keep register
pressure reasonable (M=N=16, K=32, H=8) while still exposing enough
parallelism to achieve sub‑millisecond runtimes on H100 for the
tested configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr,         # const float*   [B, N, N, C]
    right_ptr,        # const float*   [B, N, N, C]
    out_ptr,          # float*         [B, N, N, C]
    B, N, C,          # tensor dimensions (runtime)
    stride_left_b, stride_left_i, stride_left_k, stride_left_c,
    stride_right_b, stride_right_j, stride_right_k, stride_right_c,
    stride_out_b, stride_out_i, stride_out_j, stride_out_c,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_C: tl.constexpr,
):
    pid_b = tl.program_id(0)           # batch dimension
    pid_i = tl.program_id(1)           # tile over output i (row)
    pid_j = tl.program_id(2)           # tile over output j (col)

    # --- index vectors for the tile -----------------------------------------
    i = tl.arange(0, BLOCK_M)
    j = tl.arange(0, BLOCK_N)

    i_idx = pid_i * BLOCK_M + i        # absolute i indices
    j_idx = pid_j * BLOCK_N + j        # absolute j indices

    i_mask = i_idx < N
    j_mask = j_idx < N

    # Loop over hidden‑dimension chunks
    for c0 in range(0, C, BLOCK_C):
        c = tl.arange(0, BLOCK_C) + c0
        c_mask = c < C

        # accumulator for the current (i,j,c) sub‑tile
        acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_C), dtype=tl.float32)

        # Loop over the reduction dimension k
        for k0 in range(0, N, BLOCK_K):
            k = tl.arange(0, BLOCK_K) + k0
            k_mask = k < N

            # -------------------------------------------------------------
            # Load left tile: shape (M, K, C)
            left_ptrs = left_ptr \
                + pid_b * stride_left_b \
                + (i_idx[:, None, None] * stride_left_i) \
                + (k[None, :, None] * stride_left_k) \
                + (c[None, None, :] * stride_left_c)
            left_mask = i_mask[:, None, None] & k_mask[None, :, None] & c_mask[None, None, :]
            left_tile = tl.load(left_ptrs, mask=left_mask, other=0.0)

            # Load right tile: shape (N, K, C)
            right_ptrs = right_ptr \
                + pid_b * stride_right_b \
                + (j_idx[:, None, None] * stride_right_j) \
                + (k[None, :, None] * stride_right_k) \
                + (c[None, None, :] * stride_right_c)
            right_mask = j_mask[:, None, None] & k_mask[None, :, None] & c_mask[None, None, :]
            right_tile = tl.load(right_ptrs, mask=right_mask, other=0.0)

            # -------------------------------------------------------------
            # Compute contribution of this k‑slice:
            #   acc[i,j,c] += left[i,k,c] * right[j,k,c]   (broadcast over i and j)
            for kk in range(0, BLOCK_K):
                l_vec = left_tile[:, kk, :]      # (M, C)
                r_vec = right_tile[:, kk, :]     # (N, C)
                # Broadcast to (M, N, C) and accumulate
                acc += l_vec[:, None, :] * r_vec[None, :, :]

        # Store the computed sub‑tile to global memory
        out_ptrs = out_ptr \
            + pid_b * stride_out_b \
            + (i_idx[:, None, None] * stride_out_i) \
            + (j_idx[None, :, None] * stride_out_j) \
            + (c[None, None, :] * stride_out_c)
        out_mask = i_mask[:, None, None] & j_mask[None, :, None] & c_mask[None, None, :]
        tl.store(out_ptrs, acc, mask=out_mask)


def custom_kernel(data):
    """
    Entry point expected by the test harness.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (may be None)
            - weights: dict of model parameters (see function body)
            - config: dict with keys "dim", "hidden_dim", "nomask" ...

    Returns:
        torch.Tensor of shape [B, N, N, dim]   (the TriMul forward output)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    bs, seq_len, _, _ = input_tensor.shape
    eps = 1e-5

    # --------------------------------------------------------------------- #
    # 1. LayerNorm over the feature dimension (dim)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_weight + norm_bias                     # [B, N, N, dim]

    # --------------------------------------------------------------------- #
    # 2. Linear projections (no bias)
    left = F.linear(x, weights["left_proj.weight"])    # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------------- #
    # 3. Optional mask (applied before gating)
    if mask is not None and not config.get("nomask", False):
        mask_exp = mask.unsqueeze(-1).to(dtype)        # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # --------------------------------------------------------------------- #
    # 4. Gating (sigmoid) and apply to projections
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------- #
    # 5. Core contraction: out[b, i, j, c] = Σ_k left[b, i, k, c] * right[b, j, k, c]
    out = torch.empty((bs, seq_len, seq_len, hidden_dim), dtype=dtype, device=device)

    # strides for the Triton kernel (must be integer python scalars)
    sL_b, sL_i, sL_k, sL_c = left.stride()
    sR_b, sR_j, sR_k, sR_c = right.stride()
    sO_b, sO_i, sO_j, sO_c = out.stride()

    # tile sizes (tuned for H100, keep register pressure low)
    BLOCK_M = 16          # rows of output (i)
    BLOCK_N = 16          # cols of output (j)
    BLOCK_K = 32          # reduction tile
    BLOCK_C = 8           # hidden‑dim chunk

    grid = (
        bs,
        triton.cdiv(seq_len, BLOCK_M),
        triton.cdiv(seq_len, BLOCK_N),
    )

    tri_mul_kernel[grid](
        left, right, out,
        bs, seq_len, hidden_dim,
        sL_b, sL_i, sL_k, sL_c,
        sR_b, sR_j, sR_k, sR_c,
        sO_b, sO_i, sO_j, sO_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_C=BLOCK_C,
    )

    # --------------------------------------------------------------------- #
    # 6. LayerNorm over hidden dimension
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out_mean = out.mean(dim=-1, keepdim=True)
    out_var = out.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out - out_mean) / torch.sqrt(out_var + eps)
    out_norm = out_norm * to_out_norm_w + to_out_norm_b

    # --------------------------------------------------------------------- #
    # 7. Apply output gating
    out_norm = out_norm * out_gate

    # --------------------------------------------------------------------- #
    # 8. Final linear projection back to original dimension
    output = F.linear(out_norm, weights["to_out.weight"])

    return output
