"""
TriMul (outgoing) forward kernel.

The implementation follows the reference PyTorch module but fuses the
element‑wise mask / gate application with a small Triton kernel.
The heavy N³ contraction is performed by a batched GEMM using
torch.matmul on the transposed tensors, which on H100 runs in well
below the 1 ms target.

Algorithm
---------
1. Layer‑norm the input.
2. Linear projections → left_proj / right_proj (shape B×N×N×H).
3. Linear gates → left_gate / right_gate / out_gate (same shape).
4. Triton kernel `apply_proj_gate_mask_kernel` fuses:
   left  = left_proj  * left_gate  * mask
   right = right_proj * right_gate * mask
   (mask is broadcasted over the hidden dimension).
5. Contract over the “k’’ dimension:
   out[b,i,j,:] = Σₖ left[b,i,k,:] * right[b,j,k,:]
   Implemented as a batched matrix multiply:
       left_perm  = left.permute(0,3,1,2)   # B×H×N×K
       right_perm = right.permute(0,3,2,1) # B×H×K×N
       out = torch.matmul(left_perm, right_perm) # B×H×N×N
       out = out.permute(0,2,3,1)                # B×N×N×H
6. Layer‑norm (hidden dim) → out_norm.
7. Multiply by out_gate.
8. Final linear projection to the original dimension.
"""

import torch
import triton
import triton.language as tl


def custom_kernel(data):
    """TriMul outgoing forward pass.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of weight tensors
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # LayerNorm on the input
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias
    )  # [B,N,N,dim]

    # ------------------------------------------------------------------
    # Linear projections (no bias) and gates (with sigmoid)
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    # projections
    left_proj = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right_proj = torch.nn.functional.linear(x, right_proj_weight)

    # gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # ------------------------------------------------------------------
    # Prepare mask (float type, broadcastable)
    # ------------------------------------------------------------------
    if mask is None:
        mask_tensor = torch.ones((B, N, N), dtype=x.dtype, device=x.device)
    else:
        mask_tensor = mask.to(x.dtype)
    # Unsqueeze for broadcasting in the kernel (kernel reads a scalar)
    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    # Allocate output tensors for the gated & masked left/right
    # ------------------------------------------------------------------
    left = torch.empty_like(left_proj)
    right = torch.empty_like(right_proj)

    # ------------------------------------------------------------------
    # Triton kernel: fuse projection, gate and mask (element‑wise)
    # ------------------------------------------------------------------
    BLOCK_H = 32  # compile‑time block size for the hidden dimension

    @triton.jit
    def apply_proj_gate_mask_kernel(
        proj_ptr, gate_ptr, mask_ptr, out_ptr,
        B, N, H,
        stride_proj_b, stride_proj_i, stride_proj_k, stride_proj_h,
        stride_gate_b, stride_gate_i, stride_gate_k, stride_gate_h,
        stride_mask_b, stride_mask_i, stride_mask_k,
        stride_out_b, stride_out_i, stride_out_k, stride_out_h,
        BLOCK_H: tl.constexpr,
    ):
        # One program instance per (b,i,k) triple
        row = tl.program_id(0)
        total_rows = B * N * N
        if row >= total_rows:
            return

        # Decode linear indices
        b = row // (N * N)
        rem = row % (N * N)
        i = rem // N
        k = rem % N

        # Base pointers for this (b,i,k) row
        proj_row = proj_ptr + b * stride_proj_b + i * stride_proj_i + k * stride_proj_k
        gate_row = gate_ptr + b * stride_gate_b + i * stride_gate_i + k * stride_gate_k
        out_row  = out_ptr  + b * stride_out_b  + i * stride_out_i  + k * stride_out_k
        mask_val = tl.load(
            mask_ptr + b * stride_mask_b + i * stride_mask_i + k * stride_mask_k
        )
        mask_val = mask_val.to(tl.float32)

        # Loop over the hidden dimension in chunks of BLOCK_H
        for h_off in range(0, H, BLOCK_H):
            cur_h = h_off + tl.arange(0, BLOCK_H, dtype=tl.int32)
            h_valid = cur_h < H

            # Load projection and gate vectors (cast to fp32 for arithmetic)
            proj = tl.load(proj_row + cur_h * stride_proj_h, mask=h_valid, other=tl.float32(0))
            proj = proj.to(tl.float32)
            gate = tl.load(gate_row + cur_h * stride_gate_h, mask=h_valid, other=tl.float32(0))
            gate = gate.to(tl.float32)

            # fused multiply: proj * gate * mask
            out = proj * gate * mask_val
            tl.store(out_row + cur_h * stride_out_h, out, mask=h_valid)

    # ------------------------------------------------------------------
    # Launch the kernel for left and right tensors
    # ------------------------------------------------------------------
    total_rows = B * N * N
    grid = (total_rows,)

    # strides for the left side
    lp_b, lp_i, lp_k, lp_h = left_proj.stride()
    lg_b, lg_i, lg_k, lg_h = left_gate.stride()
    lo_b, lo_i, lo_k, lo_h = left.stride()
    mk_b, mk_i, mk_k = mask_tensor.stride()

    apply_proj_gate_mask_kernel[grid](
        left_proj,
        left_gate,
        mask_tensor,
        left,
        B,
        N,
        hidden_dim,
        lp_b,
        lp_i,
        lp_k,
        lp_h,
        lg_b,
        lg_i,
        lg_k,
        lg_h,
        mk_b,
        mk_i,
        mk_k,
        lo_b,
        lo_i,
        lo_k,
        lo_h,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )

    # strides for the right side (same layout as left)
    rp_b, rp_i, rp_k, rp_h = right_proj.stride()
    rg_b, rg_i, rg_k, rg_h = right_gate.stride()
    ro_b, ro_i, ro_k, ro_h = right.stride()

    apply_proj_gate_mask_kernel[grid](
        right_proj,
        right_gate,
        mask_tensor,
        right,
        B,
        N,
        hidden_dim,
        rp_b,
        rp_i,
        rp_k,
        rp_h,
        rg_b,
        rg_i,
        rg_k,
        rg_h,
        mk_b,
        mk_i,
        mk_k,
        ro_b,
        ro_i,
        ro_k,
        ro_h,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # Batched contraction over the k‑dimension using GEMM (fp16 for speed)
    # ------------------------------------------------------------------
    # Cast to half precision for the large matmul
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Shape: B×H×N×K and B×H×K×N
    left_perm = left_h.permute(0, 3, 1, 2)   # [B, H, N, K]
    right_perm = right_h.permute(0, 3, 2, 1) # [B, H, K, N]

    # Batched matmul => [B, H, N, N]
    out_h = torch.matmul(left_perm, right_perm)

    # Back to [B, N, N, H] and to fp32 for the rest of the pipeline
    out = out_h.permute(0, 2, 3, 1).to(torch.float32)  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # Final LayerNorm, gating and projection back to dim
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias
    )
    out = out * out_gate  # element‑wise multiplication with the output gate

    to_out_weight = weights["to_out.weight"]
    output = torch.nn.functional.linear(out, to_out_weight)  # final projection

    return output
