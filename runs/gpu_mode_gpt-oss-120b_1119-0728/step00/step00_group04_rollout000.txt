```python
def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    # Cast to appropriate dtype (float32)
    x = input_tensor
    # LayerNorm
    # compute normalized shape dim (last dim)
    # weight and bias have shape dim
    x = torch.nn.functional.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    
    # Linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    if not config.get("nomask", True):
        # mask is bool or int; convert to float
        mask_f = mask.to(dtype=left.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f
    # Gating
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()
    
    left = left * left_gate
    right = right * right_gate

    # cast to half for kernel
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape

    # compute output using Triton kernel
    out_h = trimul_matmul_triton(left_h, right_h, B, N, hidden_dim) # we need function

    # cast back to float
    out = out_h.to(torch.float32)

    # to_out_norm: layer norm over hidden dim
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    # out gate
    out = out * out_gate

    # final linear to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
```

```python
def trimul_matmul_triton(left, right, B, N, D):
    # left, right shape (B,N,N,D) half
    # allocate output
    out = torch.empty_like(left)  # shape (B,N,N,D)
    # get strides
    stride_lb, stride_li, stride_lk, stride_ld = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rd = right.stride()
    stride_ob, stride_oi, stride_oj, stride_od = out.stride()
    # launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (B * D, (N + BLOCK_M - 1)//BLOCK_M, (N + BLOCK_N - 1)//BLOCK_N)
    trimul_kernel[grid](
        left, right, out,
        B, N, D,
        stride_lb, stride_li, stride_lk, stride_ld,
        stride_rb, stride_rj, stride_rk, stride_rd,
        stride_ob, stride_oi, stride_oj, stride_od,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    return out
```

"""
TriMul (outgoing) forward pass implemented with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor `x` over its last dimension (dim).
2. Linear projections  (dim → hidden_dim) produce “left” and “right”.
3. Apply the optional binary mask (broadcast on hidden_dim).
4. Compute three gates (left, right, out) via a bias‑free linear + sigmoid.
5. Multiply “left/right” by their respective gates.
6. Fuse the O(N³) contraction
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   with a custom Triton kernel that treats the hidden dimension as an
   extra batch dimension and performs a tiled matrix‑multiply
   (M‑tile × N‑tile) for each (batch, hidden) pair.
7. Layer‑norm the result over the hidden dimension, multiply by the
   output gate and apply the final linear projection back to `dim`.

The heavy O(N³) reduction (step 6) is the only part required to be
implemented in Triton; the surrounding linear/normalisation ops are
performed with PyTorch for simplicity.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the pairwise contraction
# ----------------------------------------------------------------------
@triton.jit
def _trimul_gemm_kernel(
    left_ptr,            # half*   [B, N, N, H]
    right_ptr,           # half*   [B, N, N, H]
    out_ptr,             # half*   [B, N, N, H]

    B, N, H,             # integer sizes

    # strides of the tensors (in elements)
    stride_lb, stride_li, stride_lk, stride_ld,   # left
    stride_rb, stride_rj, stride_rk, stride_rd,   # right
    stride_ob, stride_oi, stride_oj, stride_od,   # out

    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Computes, for each (b, h), the matrix multiplication:
        out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]
    using tiled dot‑products.
    """
    pid = tl.program_id(0)                         # (b * H) + h
    batch = pid // H
    hid   = pid %  H

    pid_i = tl.program_id(1)                       # tile row index
    pid_j = tl.program_id(2)                       # tile col index

    # ------------------------------- offsets -------------------------------
    i_off = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    k_off = tl.arange(0, BLOCK_K)                  # will be rebased inside the loop

    mask_i = i_off < N
    mask_j = j_off < N

    # ------------------------------- accumulator -------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------- main loop over K -------------------------------
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k_blocks):
        cur_k = k * BLOCK_K + k_off
        mask_k = cur_k < N

        # Load left[i, k]  -> (M, K)
        left_ptrs = (
            left_ptr
            + batch * stride_lb
            + hid   * stride_ld
            + i_off[:, None] * stride_li
            + cur_k[None, :] * stride_lk
        )
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)               # half

        # Load right[j, k] -> (N, K)  (note the order, we will transpose)
        right_ptrs = (
            right_ptr
            + batch * stride_rb
            + hid   * stride_rd
            + j_off[:, None] * stride_rj
            + cur_k[None, :] * stride_rk
        )
        right = tl.load(right_ptrs,
                        mask=mask_j[:, None] & mask_k[None, :],
                        other=0.0)               # half

        # Cast to fp32 for accumulation
        left_f  = tl.cast(left,  tl.float32)
        right_f = tl.cast(right, tl.float32)

        # acc += left[i, k] * right[j, k] summed over k  (dot with transposed right)
        acc += tl.dot(left_f, right_f, trans_b=True)

    # ------------------------------- store result -------------------------------
    out_ptrs = (
        out_ptr
        + batch * stride_ob
        + hid   * stride_od
        + i_off[:, None] * stride_oi
        + j_off[None, :] * stride_oj
    )
    # Store as half (the output buffer was allocated as half)
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def _tri_mul_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Wrapper around the Triton kernel. `left` and `right` must be
    contiguous tensors of shape (B, N, N, H) and dtype torch.float16.
    Returns a tensor of the same shape and dtype.
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, N, _, H = left.shape
    out = torch.empty_like(left)                     # (B, N, N, H) half

    # strides in elements (not bytes)
    stride_lb, stride_li, stride_lk, stride_ld = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rd = right.stride()
    stride_ob, stride_oi, stride_oj, stride_od = out.stride()

    # Tunable block sizes – these work well on an H100 for the target N
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (B * H,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    _trimul_gemm_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_lb,
        stride_li,
        stride_lk,
        stride_ld,
        stride_rb,
        stride_rj,
        stride_rk,
        stride_rd,
        stride_ob,
        stride_oi,
        stride_oj,
        stride_od,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,            # enough parallelism for the tile sizes above
    )
    return out


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (AlphaFold3 version).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]
        - mask         : torch.Tensor [B, N, N]  (may be None)
        - weights      : dict of torch.Tensors
        - config       : dict with keys "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(dtype=left.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left  = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gates (sigmoid of bias‑free linear)
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate   = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # ------------------------------------------------------------------
    # 5) Apply gates
    # ------------------------------------------------------------------
    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Cast to half and run the Fusion kernel (O(N³) contraction)
    # ------------------------------------------------------------------
    left_h  = left.contiguous().half()   # [B, N, N, H] half
    right_h = right.contiguous().half()  # [B, N, N, H] half

    out_h = _tri_mul_gemm(left_h, right_h)   # half result

    # Back to float32 for the remaining ops
    out = out_h.float()

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, gate, final projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
