def custom_kernel(data):
    # unpack
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm on input
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)

    # Linear projections (float32)
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # Apply mask if present
    if mask is not None and not config.get("nomask", False):
        # mask shape [B,N,N]; ensure broadcast
        mask_f = mask.to(left.dtype).unsqueeze(-1)   # shape [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # Gating
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Convert to half for matmul
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape
    batch_head = B * hidden_dim

    # reshape for batched matmul
    left_mat = left_h.permute(0, 3, 1, 2).contiguous().view(batch_head, N, N)
    right_mat = right_h.permute(0, 3, 1, 2).contiguous().view(batch_head, N, N)
    # transpose on last two dims (K,N) = (N,N) -> we need right_T where inner dim is K and outer dim N
    right_T = right_mat.transpose(1, 2).contiguous()  # shape (batch_head, N, N)

    # allocate output
    out_mat = torch.empty_like(left_mat, dtype=torch.float32)  # float32

    # compute strides
    stride_ah = left_mat.stride(0)
    stride_am = left_mat.stride(1)
    stride_ak = left_mat.stride(2)

    stride_bh = right_T.stride(0)
    stride_bk = right_T.stride(1)
    stride_bj = right_T.stride(2)

    stride_ch = out_mat.stride(0)
    stride_ci = out_mat.stride(1)
    stride_cj = out_mat.stride(2)

    # launch Triton kernel
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    grid = (batch_head,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    batched_matmul_kernel[grid](
        left_mat.data_ptr(),
        right_T.data_ptr(),
        out_mat.data_ptr(),
        batch_head,
        N, N, N,
        stride_ah, stride_am, stride_ak,
        stride_bh, stride_bk, stride_bj,
        stride_ch, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # reshape back
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # output layernorm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=eps)

    out = out * out_gate

    # final linear projection
    output = torch.nn.functional.linear(out, weights["to_out.weight"])

    return output

"""
TriMul outgoing kernel implemented in Triton.

The implementation follows the PyTorch reference:
1. Layer‑norm over the channel dimension of the 4‑D input.
2. Linear projections for left/right tensors and three gating tensors.
3. Optional masking.
4. Element‑wise gating (sigmoid) and multiplication.
5. The core O(N³) reduction is expressed as a batched matrix‑multiply:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]  ->   out = left @ rightᵀ
   A separate matmul is performed for every (batch, head) pair.
   This matmul is executed by a custom Triton kernel that processes
   a batch of matrices in half‑precision while accumulating in FP32.
6. Layer‑norm on the hidden dimension, gating, and final linear projection.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F

@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    batch_head, M, N, K,
    stride_ah, stride_am, stride_ak,
    stride_bh, stride_bk, stride_bj,
    stride_ch, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM: C[b] = A[b] @ B[b]   where A: (M,K), B: (K,N)."""
    # 3‑D launch: (batch_head, block_i, block_j)
    bh = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # Pointers for the current tile
        a_ptrs = A_ptr + bh * stride_ah + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = B_ptr + bh * stride_bh + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bj

        # Load tiles (half precision)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k[None, :] < K), other=0.0, dtype=tl.float16)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & mask_n[None, :], other=0.0, dtype=tl.float16)

        # Low‑precision dot, accumulate in FP32
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # Write back the result
    c_ptrs = C_ptr + bh * stride_ch + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask_tensor  : torch.Tensor of shape [B, N, N] or None
        - weights_dict : mapping name -> torch.Tensor containing model weights
        - config_dict  : contains at least "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, on the same device)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (float32)
    # ------------------------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x_norm, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over the hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        mask_f = mask.to(left.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = F.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate   = F.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core O(N³) reduction – batched matmul in half‑precision
    # ------------------------------------------------------------------
    # Cast to FP16 for the heavy matmul
    left_h  = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape
    batch_head = B * hidden_dim

    # Reshape to (batch_head, M, K) where M = K = N
    left_mat  = left_h.permute(0, 3, 1, 2).contiguous().view(batch_head, N, N)
    right_mat = right_h.permute(0, 3, 1, 2).contiguous().view(batch_head, N, N)

    # Transpose the RHS so we can compute A @ Bᵀ = Σ_k A[i,k] * B[j,k]
    right_T = right_mat.transpose(1, 2).contiguous()

    # Allocate output buffer (FP32)
    out_mat = torch.empty_like(left_mat, dtype=torch.float32)

    # Extract strides (in elements)
    stride_ah = left_mat.stride(0)
    stride_am = left_mat.stride(1)
    stride_ak = left_mat.stride(2)

    stride_bh = right_T.stride(0)
    stride_bk = right_T.stride(1)
    stride_bj = right_T.stride(2)

    stride_ch = out_mat.stride(0)
    stride_ci = out_mat.stride(1)
    stride_cj = out_mat.stride(2)

    # Kernel launch configuration
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    grid = (
        batch_head,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    batched_matmul_kernel[grid](
        left_mat.data_ptr(),
        right_T.data_ptr(),
        out_mat.data_ptr(),
        batch_head,
        N,
        N,
        N,
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bk,
        stride_bj,
        stride_ch,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    # ------------------------------------------------------------------
    # 6. Reshape back to [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7. Output LayerNorm, gating and final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    output = F.linear(out, weights["to_out.weight"])

    return output

"""
TriMul (outgoing) forward pass with a custom Triton kernel.

Algorithm
---------
1. Layer‑norm over the channel dimension of the 4‑D input tensor.
2. Linear projections for “left” and “right” tensors (no bias) and three
   gating projections followed by a sigmoid.
3. Optional binary mask (broadcast over the hidden dimension) applied to
   the projected tensors.
4. Element‑wise gating:
       left  = left  * sigmoid(left_gate)
       right = right * sigmoid(right_gate)
5. The heavy O(N³) reduction is the contraction
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is equivalent to a batched matrix multiplication
       out[b,d] = left[b,d] @ right[b,d]ᵀ
   for every (batch, head) pair.  This batched GEMM is implemented in
   half‑precision with accumulation in FP32 inside a Triton kernel.
6. Layer‑norm over the hidden dimension, multiplication with the output gate,
   and a final linear projection back to the original channel size.

The kernel processes a tile of size BLOCK_M×BLOCK_N and loops over K
in chunks of BLOCK_K.  Half‑precision loads reduce memory bandwidth, while
the accumulation in FP32 preserves numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    batch_head, M, N, K,
    stride_ah, stride_am, stride_ak,
    stride_bh, stride_bk, stride_bj,
    stride_ch, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM: C[b] = A[b] @ B[b]   where A:(M,K), B:(K,N)."""
    # 3‑D grid: (batch_head, block_i, block_j)
    bh = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in BLOCK_K chunks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # Pointer arithmetic for the current tile
        a_ptrs = A_ptr + bh * stride_ah + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = B_ptr + bh * stride_bh + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bj

        # Load tiles (FP16) with bounds checking
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k[None, :] < K),
                    other=0.0,
                    dtype=tl.float16)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # Half‑precision dot, accumulate in FP32
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # Write the result (FP32)
    c_ptrs = C_ptr + bh * stride_ch + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj
