"""
custom_kernel:  Triton‑accelerated forward pass of the “outgoing” TriMul module
(AlphaFold3 style).

Algorithm
---------
1. Layer‑normalize the input tensor `x` over the last dimension (dim).
2. Linear projections → `left` and `right` (shape [B, N, N, H]) with
   weights `left_proj.weight` / `right_proj.weight`.
3. Optional pairwise mask (broadcast on the last dim) applied to the
   projected tensors.
4. Compute three gating tensors (sigmoid of a linear projection of `x`):
   `left_gate`, `right_gate`, `out_gate` (all shape [B, N, N, H]).
5. Apply the gates to the projected tensors.
6. Core operation (the expensive O(N³·H) part):
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication for each hidden channel `h`.
   Implemented with a tiled Triton kernel that:
   * processes one hidden channel (`h`) per program (grid dim‑2);
   * loops over the sum‑dimension `k` in blocks;
   * uses `tl.dot` on (M×K)·(K×N) tiles and accumulates the result.
7. Layer‑normalize the result over the hidden dimension (H) and apply the
   output gate.
8. Final linear projection back to the original feature dimension (`dim`).

The kernel operates on float32 data (you may cast inputs to float16 for
extra speed; the accumulator is kept in float32 to preserve precision).
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


@triton.jit
def trimul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,          # pointers
    N, H, B,                               # N: seq_len, H: hidden_dim, B: batch
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    One hidden channel `h` per program (program_id(2)). The sum over `k`
    is tiled with BLOCK_K.
    """
    pid_i = tl.program_id(0)      # block index along seq dimension i
    pid_j = tl.program_id(1)      # block index along seq dimension j
    pid_h = tl.program_id(2)      # hidden channel index

    # offsets for i, j and k within the block
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Loop over the batch dimension (B ≤ 2, small – cheap to loop)
    # ------------------------------------------------------------------
    for b in range(0, B):
        # base pointers for this batch and hidden channel
        left_base  = left_ptr  + b * stride_left_b  + pid_h * stride_left_h
        right_base = right_ptr + b * stride_right_b + pid_h * stride_right_h
        out_base   = out_ptr   + b * stride_out_b   + pid_h * stride_out_h

        # accumulator for the (BLOCK_M, BLOCK_N) tile
        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        # number of k‑blocks
        num_k_blocks = tl.cdiv(N, BLOCK_K)

        # ------------------------------------------------------------------
        # Sum over k
        # ------------------------------------------------------------------
        for kb in range(0, num_k_blocks):
            k = kb * BLOCK_K + offs_k               # global k index
            mask_k = k < N

            # left[i,k]  -> shape (BLOCK_M, BLOCK_K)
            left_ptrs = (
                left_base
                + offs_i[:, None] * stride_left_i
                + k[None, :]     * stride_left_k
            )
            left = tl.load(
                left_ptrs,
                mask = mask_i[:, None] & mask_k[None, :],
                other = 0.0,
            ).to(tl.float32)

            # right[j,k] but we need right transposed: (k, j)
            # → shape (BLOCK_K, BLOCK_N)
            right_ptrs = (
                right_base
                + k[:, None]     * stride_right_i   # k acts as the first dim
                + offs_j[None, :] * stride_right_k   # j as the second dim
            )
            right = tl.load(
                right_ptrs,
                mask = mask_k[:, None] & mask_j[None, :],
                other = 0.0,
            ).to(tl.float32)

            # matrix multiplication for this k‑block
            acc += tl.dot(left, right, trans_b=False)

        # ------------------------------------------------------------------
        # Store the result tile
        # ------------------------------------------------------------------
        out_ptrs = (
            out_base
            + offs_i[:, None] * stride_out_i
            + offs_j[None, :] * stride_out_j
        )
        tl.store(
            out_ptrs,
            acc,
            mask = mask_i[:, None] & mask_j[None, :]
        )


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator using a Triton kernel for the
    heavy pairwise multiplication.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim]
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]        # H in the kernel

    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )   # shape [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask – mask is [B, N, N]; broadcast on hidden dim
    # ------------------------------------------------------------------
    if mask is not None:
        # ensure mask dtype matches input and add the hidden dim
        mask_ = mask.to(dtype).unsqueeze(-1)      # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4) Gating tensors (sigmoid of linear proj of the *original* x)
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # Apply the respective gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core TriMul multiplication using Triton
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape            # N = sequence length
    H = hidden_dim

    # Ensure contiguous layout for the kernel
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left, dtype=torch.float32, device=device)   # [B, N, N, H]

    # Strides (in elements, not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # Kernel block sizes – these were chosen to give good occupancy on H100.
    BLOCK_M = 64    # i dimension
    BLOCK_N = 64    # j dimension
    BLOCK_K = 32    # reduction (k) dimension

    # Grid: (i‑blocks, j‑blocks, hidden‑channels)
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        H,
    )

    trimul_matmul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        N=N,
        H=H,
        B=B,
        stride_left_b=stride_left_b,
        stride_left_i=stride_left_i,
        stride_left_k=stride_left_k,
        stride_left_h=stride_left_h,
        stride_right_b=stride_right_b,
        stride_right_i=stride_right_i,
        stride_right_k=stride_right_k,
        stride_right_h=stride_right_h,
        stride_out_b=stride_out_b,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        stride_out_h=stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6) Output LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # Apply the output gate
    out_norm = out_norm * out_gate

    # ------------------------------------------------------------------
    # 7) Final linear projection back to original dim
    # ------------------------------------------------------------------
    out_final = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

    return out_final
