    stride_b_o, stride_i_o, stride_j_o, stride_d_o,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):

    pid_head = tl.program_id(0)
    total_heads = B * hidden_dim
    if pid_head >= total_heads:

    # Compute batch and hidden indices
    b = pid_head // hidden_dim
    d = pid_head % hidden_dim

    # Offsets for block rows and cols
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    input_tensor, mask, weights, config = data

    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # load weights

    eps = 1e-5
    # Use weight and bias
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Linear projections: left, right
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    import triton
    import triton.language as tl

    # define kernel (if not defined above)

"""
tri_mul_triton.py

Implementation of the “outgoing” TriMul operation from AlphaFold3 using a
mixed‑precision Triton kernel for the core pairwise multiplication.
The kernel computes, for every batch `b` and hidden channel `d`:

    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

All remaining linear layers, LayerNorms and gating are performed with
PyTorch. The tensors are cast to FP16 before the kernel to exploit the
Tensor‑Core units of an H100, while accumulation is done in FP32 to keep
numerical accuracy. The final output is returned in FP32.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication with the transposed second
# operand (i.e. left @ right^T) for each (batch, hidden) pair.
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr,            # *FP16   [B, N, N, H] left  matrix
    right_ptr,           # *FP16   [B, N, N, H] right matrix
    out_ptr,             # *FP16   [B, N, N, H] output
    B, N, H,             # scalar sizes
    stride_bl, stride_il, stride_kl, stride_dl,    # strides for left
    stride_br, stride_jr, stride_kr, stride_dr,    # strides for right
    stride_bo, stride_io, stride_jo, stride_do,    # strides for out
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """TriMul core: out = left @ right^T   (per batch & hidden channel)."""
    # ------------------------------------------------------------------
    # 1) Identify which (batch, hidden) head this program processes
    # ------------------------------------------------------------------
    pid_head = tl.program_id(0)                     # head index = b * H + d
    total_heads = B * H
    if pid_head >= total_heads:
        return

    b = pid_head // H                               # batch index
    d = pid_head % H                                # hidden channel

    # ------------------------------------------------------------------
    # 2) Identify the tile of the output matrix (i‑block, j‑block)
    # ------------------------------------------------------------------
    pid_m = tl.program_id(1)                        # row‑tile index
    pid_n = tl.program_id(2)                        # col‑tile index

    i_start = pid_m * BLOCK_M
    j_start = pid_n * BLOCK_N

    # base offsets for this head (batch * hidden stride)
    off_head_l = b * stride_bl + d * stride_dl
    off_head_r = b * stride_br + d * stride_dr
    off_head_o = b * stride_bo + d * stride_do

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 3) Loop over the reduction dimension k
    # ------------------------------------------------------------------
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_tile * BLOCK_K

        # offsets within the current tile
        i_off = i_start + tl.arange(0, BLOCK_M)
        j_off = j_start + tl.arange(0, BLOCK_N)
        k_off = k_start + tl.arange(0, BLOCK_K)

        # bounds masks
        mask_i = i_off < N
        mask_j = j_off < N
        mask_k = k_off < N

        # ----- load left[i, k, d] ------------------------------------
        # shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + off_head_l \
                    + i_off[:, None] * stride_il \
                    + k_off[None, :] * stride_kl
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # ----- load right_T[k, j, d] = right[j, k, d] ---------------
        # we load the transposed view so that a dot computes Σ_k left*right
        # shape (BLOCK_K, BLOCK_N)
        right_T_ptrs = right_ptr + off_head_r \
                       + k_off[:, None] * stride_kr \
                       + j_off[None, :] * stride_jr
        right_T = tl.load(right_T_ptrs,
                          mask=mask_k[:, None] & mask_j[None, :],
                          other=0.0)

        # ----- accumulate ------------------------------------------------
        acc += tl.dot(left, right_T)   # (M,K)·(K,N) → (M,N)

    # ------------------------------------------------------------------
    # 4) Write the result back (cast to FP16)
    # ------------------------------------------------------------------
    store_mask = mask_i[:, None] & mask_j[None, :]
    out_ptrs = out_ptr + off_head_o \
               + i_off[:, None] * stride_io \
               + j_off[None, :] * stride_jo
    tl.store(out_ptrs, acc.to(tl.float16), mask=store_mask)

    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    # Cast to half‑precision for the heavy computation
    left_h = left.to(torch.float16).contiguous()

    out_h = torch.empty_like(left_h, dtype=torch.float16).contiguous()

    stride_bl, stride_il, stride_kl, stride_dl = left_h.stride()
    stride_br, stride_jr, stride_kr, stride_dr = right_h.stride()
    stride_bo, stride_io, stride_jo, stride_do = out_h.stride()

    # Blocking parameters – chosen to fit H100 Tensor‑Core resources
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    total_heads = B * hidden_dim
    grid = (
        total_heads,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    trimul_kernel[grid](
        left_h,
        right_h,
        out_h,
        B,
        N,
        hidden_dim,
        stride_bl,
        stride_il,
        stride_kl,
        stride_dl,
        stride_br,
        stride_jr,
        stride_kr,
        stride_dr,
        stride_bo,
        stride_io,
        stride_jo,
        stride_do,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Synchronise before moving back to FP32
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
