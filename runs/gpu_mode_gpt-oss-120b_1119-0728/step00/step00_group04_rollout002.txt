def layernorm_kernel(x_ptr, y_ptr, weight_ptr, bias_ptr, M, N, eps, stride_xm, stride_xn, stride_ym, stride_yn):
    pid = tl.program_id(axis=0)
    # row offset (0 <= pid < M)
    offs_m = pid * stride_xm  # pointer to start of row in x

    # We'll iterate over n dimension using loops
    sum_val = tl.float32(0)
    sum_sq = tl.float32(0)

    # load in chunks
    for start_n in range(0, N, BLOCK_N):
        cur_n = start_n + tl.arange(0, BLOCK_N)  # shape BLOCK_N
        mask = cur_n < N
        ptrs = x_ptr + offs_m + cur_n * stride_xn
        vals = tl.load(ptrs, mask=mask, other=0.0)  # shape BLOCK_N
        sum_val += tl.sum(vals, mask=mask)
        sum_sq += tl.sum(vals * vals, mask=mask)

    mean = sum_val / N
    var = sum_sq / N - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    for start_n in range(0, N, BLOCK_N):
        cur_n = start_n + tl.arange(0, BLOCK_N)
        mask = cur_n < N

def layernorm_kernel(
    x_ptr, y_ptr, weight_ptr, bias_ptr,
    M, N, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    stride_wn, stride_bn,
    BLOCK_N: tl.constexpr):
    pid = tl.program_id(0)
    # compute row pointer offset
    row_offset = pid * stride_xm

    # compute sum and sum of squares across N
    sum_val = tl.float32(0.0)
    sum_sq = tl.float32(0.0)

    for i in range(0, N, BLOCK_N):
        cur_index = i + tl.arange(0, BLOCK_N)
        mask = cur_index < N
        x_ptrs = x_ptr + row_offset + cur_index * stride_xn
        x = tl.load(x_ptrs, mask=mask, other=0.0)
        sum_val += tl.sum(x, mask=mask)

                     stride_wn, stride_bn,

    weight_ptr,           # *float32   gamma  [C]
    bias_ptr,             # *float32   beta   [C]
    M,                    # int32      number of rows (=B*N*N)
    C,                    # int32      channel dimension (dim)
    eps,                  # float32    epsilon for stability
    stride_xm, stride_xn, # int32      strides of x (row, col)
    stride_ym, stride_yn, # int32      strides of y (row, col)
    stride_wn, stride_bn, # int32      strides of weight/bias (normally 1)
    BLOCK_N: tl.constexpr # compile‑time block size for the channel loop
):
    pid = tl.program_id(0)                     # one program per row (M)

    # ------------------------------------------------------------------
    # Compute row offset (in elements) for the current (b,i,j) entry
    # ------------------------------------------------------------------
    row_offset = pid * stride_xm

    # ------------------------------------------------------------------
    # First pass: accumulate sum and sum of squares over the channel dim
    # ------------------------------------------------------------------
    sum_val = tl.float32(0.0)
    sum_sq = tl.float32(0.0)

    for start_n in range(0, C, BLOCK_N):
        cur_n = start_n + tl.arange(0, BLOCK_N)               # [BLOCK_N]
        mask = cur_n < C
        x = tl.load(x_ptr + row_offset + cur_n * stride_xn,
                    mask=mask, other=0.0)                      # [BLOCK_N]
        sum_val += tl.sum(x, mask=mask)
        sum_sq  += tl.sum(x * x, mask=mask)

    # ------------------------------------------------------------------
    # Compute mean and inverse std
    # ------------------------------------------------------------------
    mean = sum_val / C
    var = sum_sq / C - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: write normalised + affine transformed values
    # ------------------------------------------------------------------
    for start_n in range(0, C, BLOCK_N):
        cur_n = start_n + tl.arange(0, BLOCK_N)
        mask = cur_n < C
        x = tl.load(x_ptr + row_offset + cur_n * stride_xn,
                    mask=mask, other=0.0)

        # (x - mean) / sqrt(var+eps)
        norm = (x - mean) * inv_std

        # affine
        w = tl.load(weight_ptr + cur_n * stride_wn,
                    mask=mask, other=0.0)
        b = tl.load(bias_ptr + cur_n * stride_bn,
                    mask=mask, other=0.0)
        y = norm * w + b

        tl.store(y_ptr + row_offset + cur_n * stride_yn,
                 y, mask=mask)


# ----------------------------------------------------------------------
# Entry point expected by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul block.
    Arguments:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------
    # Extract model weights (all are 2‑D weight matrices or 1‑D bias)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    left_proj_weight  = weights["left_proj.weight"]   # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]  # (hidden_dim, dim)

    left_gate_weight  = weights["left_gate.weight"]   # (hidden_dim, dim)
    right_gate_weight = weights["right_gate.weight"]  # (hidden_dim, dim)
    out_gate_weight   = weights["out_gate.weight"]    # (hidden_dim, dim)

    to_out_norm_weight = weights["to_out_norm.weight"]  # (hidden_dim,)
    to_out_norm_bias   = weights["to_out_norm.bias"]    # (hidden_dim,)

    to_out_weight = weights["to_out.weight"]            # (dim, hidden_dim)

    # --------------------------------------------------------------
    # 1) First LayerNorm (fused via Triton)
    # --------------------------------------------------------------
    B, N, _, _ = input_tensor.shape   # input shape = [B, N, N, dim]
    M = B * N * N

    # Flatten to [M, dim] – safe because dim is the last axis.
    x_flat = input_tensor.view(M, dim)

    # Allocate output for the normalised tensor.
    x_norm_flat = torch.empty_like(x_flat)

    # Launch Triton kernel.  BLOCK_N = 32 works for all supported dims (128/384).
    grid = (M,)
    layernorm_kernel[grid](
        x_flat, x_norm_flat,
        norm_weight, norm_bias,
        M, dim, eps,
        # strides for x and y (row-major)
        x_flat.stride(0), x_flat.stride(1),
        x_norm_flat.stride(0), x_norm_flat.stride(1),
        # weight / bias strides (typically 1)
        norm_weight.stride(0), norm_bias.stride(0),
        BLOCK_N=32,
        num_warps=4,
    )
    # Ensure the kernel has finished before we continue with torch ops.
    torch.cuda.synchronize(device)

    # Reshape back to [B, N, N, dim]
    x_norm = x_norm_flat.view(B, N, N, dim)

    # --------------------------------------------------------------
    # 2) Linear projections (no bias) and gating
    # --------------------------------------------------------------
    # left / right projections
    left = F.linear(x_norm, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x_norm, right_proj_weight) # [B, N, N, hidden_dim]

    # Optional masking (mask shape = [B, N, N])
    if not config.get("nomask", False):
        # Convert possible bool mask to float
        mask = mask_tensor.to(dtype=dtype)          # [B, N, N]
        mask = mask.unsqueeze(-1)                  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # Gates (sigmoid of linear projections)
    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_weight))

    # Apply element‑wise gates
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 3) Pairwise multiplicative update (einsum)
    #    out[b, i, j, d] = sum_k left[b, i, k, d] * right[b, j, k, d]
    # --------------------------------------------------------------
    out = torch.einsum('b i k d, b j k d -> b i j d', left, right)   # [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 4) Second LayerNorm (hidden_dim) – use PyTorch built‑in for simplicity
    # --------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=eps)

    # --------------------------------------------------------------
    # 5) Output gating and final linear projection back to `dim`
    # --------------------------------------------------------------
    out = out * out_gate                     # element‑wise gate
    output = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return output
