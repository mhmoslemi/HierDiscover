@triton.jit
def tr_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bd = tl.program_id(2)

    total_bd = B * H
    b = pid_bd // H
    h = pid_bd % H

    # compute start offsets
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    k_offset = tl.arange(0, BLOCK_K)
    # mask for boundaries
    mask_i = i_offset < N
    mask_j = j_offset < N
    mask_k = k_offset < N

    # compute base offsets
    base_left = b * stride_left_b + h * stride_left_h
    base_right = b * stride_right_b + h * stride_right_h
    base_out = b * stride_out_b + h * stride_out_h

    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute k tile offsets
        cur_k_offset = k * BLOCK_K + k_offset
        mask_k_cur = cur_k_offset < N

        # pointers
        left_ptrs = left_ptr + base_left + i_offset[:, None] * stride_left_i + cur_k_offset[None, :] * stride_left_k
        right_ptrs = right_ptr + base_right + cur_k_offset[:, None] * stride_right_k + j_offset[None, :] * stride_right_i

        # load tiles
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k_cur[None, :], other=0.0)
        right = tl.load(right_ptrs, mask=mask_k_cur[:, None] & mask_j[None, :], other=0.0)

        # dot
        acc += tl.dot(left, right)  # left: (M,K), right: (K,N)
    # store result
    out_ptrs = out_ptr + base_out + i_offset[:, None] * stride_out_i + j_offset[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float32), mask=mask_i[:, None] & mask_j[None, :])
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm on input
    eps = 1e-5
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)

    # Projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Mask if needed
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(x.dtype) # broadcast
        left = left * mask_exp
        right = right * mask_exp

    # Gating
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Cast to half for kernel
    left_h = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()
    # Output allocate
    out_h = torch.empty_like(left_h, dtype=torch.float16)

    # Launch Triton kernel
    B, N, _, H = left_h.shape  # B,N,N,H

    # Compute strides
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left_h.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right_h.stride()
    stride_out_b, stride_out_i, stride_out_k, stride_out_h = out_h.stride() # k is same as third N

    # Triton launch
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = lambda META: (triton.cdiv(N, META['BLOCK_M']), triton.cdiv(N, META['BLOCK_N']), B * H)

    tr_mul_kernel[grid](
        left_h,
        right_h,
        out_h,
        B,
        N,
        H,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_h,
        stride_right_b,
        stride_right_i,
        stride_right_k,
        stride_right_h,
        stride_out_b,
        stride_out_i,
        stride_out_k,
        stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Convert back to float32
    out = out_h.to(torch.float32)

@triton.jit
def tr_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bd = tl.program_id(2)

    total_bd = B * H
    b = pid_bd // H
    h = pid_bd % H

    # offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # base offset for this batch and channel
    base_left = b * stride_left_b + h * stride_left_h
    base_right = b * stride_right_b + h * stride_right_h
    base_out = b * stride_out_b + h * stride_out_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = k_tile * BLOCK_K + offs_k
        mask_k = cur_k < N

        # compute pointers for this tile
        left_ptrs = left_ptr + base_left + offs_i[:, None] * stride_left_i + cur_k[None, :] * stride_left_k
        right_ptrs = right_ptr + base_right + cur_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i

        # load
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # multiply-accumulate
        acc += tl.dot(left, right).to(tl.float32)

    # store output
    out_ptrs = out_ptr + base_out + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tr_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bd = tl.program_id(2)

    # derive batch and channel indices
    b = pid_bd // H
    h = pid_bd % H

    # compute start offsets for the i, j dimensions
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base address for the current (b, h) slice
    base_left = b * stride_left_b + h * stride_left_h
    base_right = b * stride_right_b + h * stride_right_h
    base_out = b * stride_out_b + h * stride_out_h

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # left tile: (M, K)
        left_ptrs = left_ptr + base_left \
            + offs_i[:, None] * stride_left_i \
            + cur_k[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # right tile: (K, N)  (note the transposition)
        right_ptrs = right_ptr + base_right \
            + cur_k[:, None] * stride_right_k \
            + offs_j[None, :] * stride_right_i
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # multiply‑accumulate
        acc += tl.dot(left, right).to(tl.float32)

    # write back the result
    out_ptrs = out_ptr + base_out \
        + offs_i[:, None] * stride_out_i \
        + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])
```

def custom_kernel(data):
    """
    Outgoing TriMul implementation.
    Performs LayerNorm, linear projections, gating, and the core
    pairwise multiplication (einsum) using a custom Triton kernel.
    """
    import torch
    import torch.nn.functional as F
    import triton

    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    eps = 1e-5

    # ---------- LayerNorm on the input ----------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"],
                     eps=eps)

    # ---------- Linear projections ----------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- Optional mask ----------
    if not nomask:
        # mask shape: [B, N, N]
        mask_exp = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # ---------- Gates ----------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # Core Tensor contraction: out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
    # -------------------------------------------------
    # Cast to half precision for the heavy kernel (memory & compute saving)
    left_h = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()

    B, N, _, H = left_h.shape  # shape == (B, N, N, hidden_dim)

    out_h = torch.empty_like(left_h, dtype=torch.float16)

    # strides in element units
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left_h.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right_h.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out_h.stride()

    # tile sizes (tuned for H100)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = lambda META: (
        triton.cdiv(N, META["BLOCK_M"]),
        triton.cdiv(N, META["BLOCK_N"]),
        B * H,
    )

    tr_mul_kernel[grid](
        left_h,
        right_h,
        out_h,
        B,
        N,
        H,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_h,
        stride_right_b,
        stride_right_i,
        stride_right_k,
        stride_right_h,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Back to FP32 for the remaining layers
    out = out_h.to(torch.float32)

    # ---------- Output LayerNorm ----------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"],
                       eps=eps)

    # ---------- Output gate ----------
    out = out * out_gate

    # ---------- Final linear ----------
    out = F.linear(out, weights["to_out.weight"])

    return out

"""
TriMul (outgoing) implementation.
---------------------------------

This function reproduces the forward pass of the TriMul module from
AlphaFold3.  All element‑wise operations (LayerNorm, linear projections,
gating, masks, the final LayerNorm and linear) are performed with PyTorch.
The computationally expensive tensor contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which corresponds to the einsum
`einsum('... i k d, ... j k d -> ... i j d', left, right)`,
is off‑loaded to a custom Triton kernel.  The kernel computes many
independent matrix multiplications (one per batch and hidden channel)
by tiling the i, j and k dimensions (BLOCK_M×BLOCK_N×BLOCK_K) and
accumulates in FP32 while the inputs are stored in FP16 to reduce
memory traffic.

The implementation works for the full range of test configurations
(B∈{1,2}, N∈{128,256,512,1024}, dim∈{128,384}, hidden_dim∈{128,384},
with or without a mask) and does not compute gradients.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tr_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    where left/right/out are stored as [B, N, N, H] in row‑major order.
    """
    pid_i = tl.program_id(0)        # i‑tile
    pid_j = tl.program_id(1)        # j‑tile
    pid_bd = tl.program_id(2)       # combined batch & hidden‑channel index

    # decode batch and hidden index
    b = pid_bd // H
    h = pid_bd % H

    # tile offsets for i, j and the reduction dimension k
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointer for the current (b, h) slice
    base_left = b * stride_left_b + h * stride_left_h
    base_right = b * stride_right_b + h * stride_right_h
    base_out = b * stride_out_b + h * stride_out_h

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # left tile: shape (M, K)
        left_ptrs = left_ptr + base_left \
            + offs_i[:, None] * stride_left_i \
            + cur_k[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # right tile: shape (K, N) – note the transposed layout
        right_ptrs = right_ptr + base_right \
            + cur_k[:, None] * stride_right_k \
            + offs_j[None, :] * stride_right_i
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # multiply‑accumulate (inputs are fp16, accumulation in fp32)
        acc += tl.dot(left, right).to(tl.float32)

    # write result (cast back to the output dtype, here fp16)
    out_ptrs = out_ptr + base_out \
        + offs_i[:, None] * stride_out_i \
        + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])

    hidden_dim = config["hidden_dim"]

        right = right * mask_exp

    right = right * right_gate

    # launch grid: (i_tiles, j_tiles, batch*hidden)
    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H,
    )

    tr_mul_kernel[grid](
        left_h,
        right_h,
        out_h,
        B,
        N,
        H,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_h,
        stride_right_b,
        stride_right_i,
        stride_right_k,
        stride_right_h,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # back to FP32 for the remaining layers
    out = out_h.to(torch.float32)

    # ------------------------------------------------------------------
    # 6) Output LayerNorm (over hidden dimension)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 7) Output gate (element‑wise)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8) Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
```
