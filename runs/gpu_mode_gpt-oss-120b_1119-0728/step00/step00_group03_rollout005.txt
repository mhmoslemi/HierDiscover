@triton.jit
def layernorm_kernel(
    x_ptr, out_ptr,
    weight_ptr, bias_ptr,
    stride_xb, stride_xi, stride_xj, stride_xc,
    stride_ob, stride_oi, stride_oj, stride_oc,
    B, N, C,
    epsilon,
    BLOCK_C: tl.constexpr
):
    pid = tl.program_id(0)

    # Compute indices
    total_seq = N * N
    b = pid // total_seq
    seq_index = pid % total_seq
    i = seq_index // N
    j = seq_index % N

    # Compute base pointers offsets for this row
    x_row_ptr = x_ptr + b*stride_xb + i*stride_xi + j*stride_xj
    out_row_ptr = out_ptr + b*stride_ob + i*stride_oi + j*stride_oj

    # Compute sum and sumsq across channel dimension C
    sum = tl.float32(0.0)
    sum_sq = tl.float32(0.0)

    # iterate over channel blocks
    c_offset = 0
    while c_offset < C:
        # offsets for this block
        offs = c_offset + tl.arange(0, BLOCK_C)
        mask = offs < C
        # load values
        x = tl.load(x_row_ptr + offs * stride_xc, mask=mask, other=0.0).to(tl.float32)
        sum += tl.sum(x)
        sum_sq += tl.sum(x * x)
        c_offset += BLOCK_C

    # compute mean and variance
    mean = sum / C
    var = sum_sq / C - mean * mean
    inv_std = 1.0 / tl.sqrt(var + epsilon)

    # second pass: write normalized values
    c_offset = 0
    while c_offset < C:
        offs = c_offset + tl.arange(0, BLOCK_C)
        mask = offs < C
        x = tl.load(x_row_ptr + offs * stride_xc, mask=mask, other=0.0).to(tl.float32)

        w = tl.load(weight_ptr + offs, mask=mask, other=0.0).to(tl.float32)
        b_val = tl.load(bias_ptr + offs, mask=mask, other=0.0).to(tl.float32)

        norm = (x - mean) * inv_std
        out = norm * w + b_val

        tl.store(out_row_ptr + offs * stride_oc, out, mask=mask)

    return

"""
custom_kernel for the “outgoing” Triangular Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑Norm the input tensor `x` (shape [B,N,N,C]) using a custom Triton kernel.
2. Project the normalized tensor to a hidden dimension `H` with two linear layers
   (`left_proj`, `right_proj`) and compute three gating vectors (`left_gate`,
   `right_gate`, `out_gate`).  Gates use a sigmoid non‑linearity.
3. Apply the optional pairwise mask (broadcast over the hidden dimension) and
   multiply the projections by their respective gates.
4. Fuse the expensive contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   into a batched matrix multiplication:
        - reshape `left` → (B·H, N, N)
        - reshape `right` → (B·H, N, N) and transpose the last two axes
        - compute `out_flat = torch.bmm(left_flat, right_flat)`.
   The GEMM is performed in FP16 inside an `amp.autocast` block for speed and
   the result is cast back to FP32.
5. Layer‑Norm the contraction result with a second Triton kernel
   (`to_out_norm`).
6. Multiply by `out_gate` and apply the final linear projection back to the
   original dimensionality `C`.
7. Return the tensor of shape [B,N,N,C].

At least one non‑trivial operation (the two LayerNorms) is implemented in
Triton, while the heavy O(N³) contraction uses an optimized batched GEMM.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
import math

# ---------------------------------------------------------------------------
# Triton kernel: layer‑norm over the last dimension.
# ---------------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                     # *const float32
    out_ptr,                   # *float32
    weight_ptr,                # *const float32
    bias_ptr,                  # *const float32
    stride_xb, stride_xi, stride_xj, stride_xc,
    stride_ob, stride_oi, stride_oj, stride_oc,
    B, N, C,                  # runtime shape parameters
    epsilon,                  # float32
    BLOCK_C: tl.constexpr,    # compile‑time tile size for channel dim
):
    pid = tl.program_id(0)                                 # 1‑D grid
    total_seq = N * N
    b = pid // total_seq                                   # batch index
    seq_id = pid % total_seq
    i = seq_id // N                                        # first spatial index
    j = seq_id % N                                         # second spatial index

    # -----------------------------------------------------------------------
    # 1) Compute mean & variance across the channel dimension C
    # -----------------------------------------------------------------------
    sum_val = tl.float32(0.0)
    sum_sq  = tl.float32(0.0)
    c_offset = 0
    while c_offset < C:
        offs = c_offset + tl.arange(0, BLOCK_C)            # [BLOCK_C]
        mask = offs < C
        # pointer to the start of (b,i,j,:) slice
        x_base = x_ptr + b * stride_xb + i * stride_xi + j * stride_xj
        x = tl.load(x_base + offs * stride_xc, mask=mask, other=0.0).to(tl.float32)  # [BLOCK_C]
        sum_val += tl.sum(x)                               # scalar
        sum_sq  += tl.sum(x * x)                           # scalar
        c_offset += BLOCK_C

    C_f = tl.float32(C)                                    # cast once
    mean = sum_val / C_f
    var  = sum_sq / C_f - mean * mean
    inv_std = 1.0 / tl.sqrt(var + epsilon)

    # -----------------------------------------------------------------------
    # 2) Write normalized (x - mean) / sqrt(var) * weight + bias
    # -----------------------------------------------------------------------
    c_offset = 0
    while c_offset < C:
        offs = c_offset + tl.arange(0, BLOCK_C)
        mask = offs < C
        x_base = x_ptr + b * stride_xb + i * stride_xi + j * stride_xj
        x = tl.load(x_base + offs * stride_xc, mask=mask, other=0.0).to(tl.float32)

        w = tl.load(weight_ptr + offs, mask=mask, other=0.0).to(tl.float32)
        b_val = tl.load(bias_ptr + offs, mask=mask, other=0.0).to(tl.float32)

        norm = (x - mean) * inv_std
        out = norm * w + b_val

        out_base = out_ptr + b * stride_ob + i * stride_oi + j * stride_oj
        tl.store(out_base + offs * stride_oc, out, mask=mask)

        c_offset += BLOCK_C


def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """
    Layer‑norm along the last dimension using the Triton kernel above.
    Input/weight/bias must be on the same CUDA device.
    """
    assert x.is_cuda and weight.is_cuda and bias.is_cuda
    B, N, N2, C = x.shape
    assert N == N2, "TriMul expects a square pairwise tensor."
    out = torch.empty_like(x)

    # Strides (in elements, not bytes)
    stride_xb, stride_xi, stride_xj, stride_xc = x.stride()
    stride_ob, stride_oi, stride_oj, stride_oc = out.stride()

    total = B * N * N
    grid = (total,)
    BLOCK_C = 128                         # enough for the max C (384) – processed in loops

    layernorm_kernel[grid](
        x, out,
        weight, bias,
        stride_xb, stride_xi, stride_xj, stride_xc,
        stride_ob, stride_oi, stride_oj, stride_oc,
        B, N, C,
        eps,
        BLOCK_C=BLOCK_C,
    )
    return out


# ---------------------------------------------------------------------------
# Main entry point required by the test harness
# ---------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.
    Arguments:
        data = (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("epsilon", 1e-5)

    # -------------------------------------------------------------------
    # 1. Layer‑norm the input (float32)
    # -------------------------------------------------------------------
    x_norm = triton_layernorm(
        input_tensor,
        weights["norm.weight"],
        weights["norm.bias"],
        eps,
    )   # [B,N,N,dim]

    # -------------------------------------------------------------------
    # 2. Linear projections and gates (use FP16 for speed inside autocast)
    # -------------------------------------------------------------------
    with torch.cuda.amp.autocast():
        left_proj  = F.linear(x_norm, weights["left_proj.weight"])
        right_proj = F.linear(x_norm, weights["right_proj.weight"])
        left_gate  = F.linear(x_norm, weights["left_gate.weight"]).sigmoid()
        right_gate = F.linear(x_norm, weights["right_gate.weight"]).sigmoid()

    # -------------------------------------------------------------------
    # 3. Apply optional mask (broadcast over hidden dim)
    # -------------------------------------------------------------------
    if mask is not None:
        mask_f = mask.unsqueeze(-1).to(left_proj.dtype)   # shape [B,N,N,1]
        left = left_proj * left_gate * mask_f
        right = right_proj * right_gate * mask_f
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # -------------------------------------------------------------------
    # 4. Compute the O(N³) contraction via batched GEMM.
    #    Cast to FP16 for the matrix multiplication, then back to FP32.
    # -------------------------------------------------------------------
    # Ensure contiguity for the view/transpose operations
    left = left.contiguous()
    right = right.contiguous()
    B, N, _, _ = left.shape

    # reshape to (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # transpose the second N dimension of right (K ↔ J)
    right_t = right_perm.transpose(2, 3)                # [B, H, N, N]

    # fuse batch & channel dimension for a single torch.bmm call
    left_flat = left_perm.view(B * hidden_dim, N, N).to(torch.float16)
    right_flat = right_t.view(B * hidden_dim, N, N).to(torch.float16)

    out_flat = torch.bmm(left_flat, right_flat)         # [B*H, N, N]  FP16
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)                         # back to FP32 for the rest

    # -------------------------------------------------------------------
    # 5. Second Layer‑Norm on the contraction output
    # -------------------------------------------------------------------
    out_norm = triton_layernorm(
        out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        eps,
    )   # [B,N,N,hidden_dim]

    # -------------------------------------------------------------------
    # 6. Out‑gate (computed from the normalized input, stays FP32)
    # -------------------------------------------------------------------
    out_gate = F.linear(x_norm, weights["out_gate.weight"]).sigmoid()
    out_norm = out_norm * out_gate

    # -------------------------------------------------------------------
    # 7. Final linear projection back to original dim
    # -------------------------------------------------------------------
    final = F.linear(out_norm, weights["to_out.weight"])   # [B,N,N,dim]

    return final
```
