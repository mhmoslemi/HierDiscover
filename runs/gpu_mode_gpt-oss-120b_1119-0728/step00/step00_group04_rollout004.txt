3. Apply optional mask and the gates:
   ``left  = left_proj(x)  * mask * left_gate``,
   ``right = right_proj(x) * mask * right_gate``.
   The tensors are cast to *float16* to save bandwidth.
4. **Triangular multiplicative update** – the heavy N³ part.
   For each batch and hidden channel we need

        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

   which is a batched matrix multiplication of
   ``left[b,:, :, h]`` × ``right[b,:, :, h]ᵀ``.
   This is implemented in a fused Triton kernel
   (`tri_mul_kernel`) that:
   * iterates over K‑tiles,
   * loads half‑precision tiles,
   * accumulates in FP32 with `tl.dot`,
   * writes the FP32 result.
5. Normalize the result with a second LayerNorm (hidden_dim),
   multiply by ``out_gate``, and finally linearly project back
   to ``dim``.
6. Return the tensor of shape ``[B, N, N, dim]`` (FP32).

The kernel only handles the O(N³) contraction; all other
operations are performed with PyTorch (which is negligible
compared to the matrix multiplication).  The implementation
supports both masked and un‑masked inputs and works for the
full range of shapes used in the test suite.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr,            # *tl.pointer* to left tensor (FP16)
    right_ptr,           # *tl.pointer* to right tensor (FP16)
    out_ptr,             # *tl.pointer* to output tensor (FP32)
    B, N, H,             # batch, sequence length, hidden_dim
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched triangular multiplication:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
       The kernel processes one (batch, hidden) pair per program
       (grid third dimension).  Each program computes a tile of
       the output matrix (i‑block × j‑block) with a K‑reduction.
    """
    pid_m = tl.program_id(0)          # block row (i)
    pid_n = tl.program_id(1)          # block col (j)
    pid_bh = tl.program_id(2)         # packed (batch, hidden)

    # -------------------- decode batch & hidden index --------------------
    h = pid_bh % H                    # hidden channel
    b = pid_bh // H                   # batch index

    # -------------------- compute global offsets --------------------
    i_base = pid_m * BLOCK_M
    j_base = pid_n * BLOCK_N

    i_index = i_base + tl.arange(0, BLOCK_M)
    j_index = j_base + tl.arange(0, BLOCK_N)

    # Masks for the edges of the matrix
    i_mask = i_index < N
    j_mask = j_index < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of K‑tiles
    num_k_tiles = tl.cdiv(N, BLOCK_K)

    for tk in range(num_k_tiles):
        k_base = tk * BLOCK_K
        k_index = k_base + tl.arange(0, BLOCK_K)
        k_mask = k_index < N

        # -------------------- load left tile (i × k) --------------------
        # left[b, i, k, h] – strides: (b,i,k,h)
        left_ptrs = (
            left_ptr
            + b * stride_l_b
            + h * stride_l_h
            + i_index[:, None] * stride_l_i
            + k_index[None, :] * stride_l_k
        )
        left_tile = tl.load(
            left_ptrs,
            mask=(i_mask[:, None] & k_mask[None, :]),
            other=0.0,
            dtype=tl.float16,
        ).to(tl.float32)

        # -------------------- load right tile (j × k) --------------------
        # right[b, j, k, h] – same stride pattern as left
        right_ptrs = (
            right_ptr
            + b * stride_r_b
            + h * stride_r_h
            + j_index[:, None] * stride_r_i
            + k_index[None, :] * stride_r_k
        )
        right_tile = tl.load(
            right_ptrs,
            mask=(j_mask[:, None] & k_mask[None, :]),
            other=0.0,
            dtype=tl.float16,
        ).to(tl.float32)

        # -------------------- dot‑product over K (sum_k) --------------------
        # tl.dot returns a (BLOCK_M, BLOCK_N) matrix
        acc += tl.dot(left_tile, right_tile)

    # -------------------- write the result --------------------
    out_ptrs = (
        out_ptr
        + b * stride_o_b
        + h * stride_o_h
        + i_index[:, None] * stride_o_i
        + j_index[None, :] * stride_o_j
    )
    tl.store(
        out_ptrs,
        acc,
        mask=(i_mask[:, None] & j_mask[None, :]),
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[B, N, N, dim] (float32)
        - mask         : torch.Tensor[B, N, N] or None
        - weights      : dict of torch.Tensors (see body for keys)
        - config       : dict with keys:
            * dim          – input feature dimension
            * hidden_dim   – inner projection dimension
            * nomask (bool, optional) – True if mask should be ignored

    Returns
    -------
    torch.Tensor[B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # weight shortcuts
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    device = input_tensor.device
    dtype = input_tensor.dtype  # expect float32

    # ------------------------------------------------------------------
    # 1) Input layer‑norm
    # ------------------------------------------------------------------
    # LayerNorm over the last dimension (dim)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # shape [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3) Mask (if present)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask shape [B, N, N] → broadcast on hidden dim
        mask_exp = mask.unsqueeze(-1).to(dtype)  # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid after linear)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight)) # [B,N,N,hidden_dim]
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))    # [B,N,N,hidden_dim]

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Cast to half for the expensive contraction
    left_h = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()

    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 5) Triangular multiplicative update (handled by Triton)
    # ------------------------------------------------------------------
    # Allocate output tensor (float32)
    out = torch.empty(B, N, N, hidden_dim, dtype=torch.float32, device=device)

    # Strides in number of elements (not bytes)
    slb, sli, slk, slh = left_h.stride()
    srb, sri, srk, srh = right_h.stride()
    sob, soi, soj, soh = out.stride()

    # Block sizes (tuned for H100 FP16 matmul)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    tri_mul_kernel[grid](
        left_h,
        right_h,
        out,
        B,
        N,
        hidden_dim,
        slb,
        sli,
        slk,
        slh,
        srb,
        sri,
        srk,
        srh,
        sob,
        soi,
        soj,
        soh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,                # good default on H100
    )
    # ------------------------------------------------------------------
    # 6) Output layer‑norm, gating and final linear projection
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )  # [B, N, N, hidden_dim]

    out = out_norm * out_gate  # apply output gate (broadcasted)

    # Final projection back to `dim`
    output = F.linear(out, to_out_weight)  # [B, N, N, dim]

    return output
```
