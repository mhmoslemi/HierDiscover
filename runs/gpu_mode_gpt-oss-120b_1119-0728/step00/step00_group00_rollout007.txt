def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_hb = tl.program_id(2)

    num_h_blocks = tl.cdiv(H, BLOCK_H)
    batch_idx = pid_hb // num_h_blocks
    hb = pid_hb % num_h_blocks

    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N
    h_start = hb * BLOCK_H

    # offsets
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)
    offs_h = h_start + tl.arange(0, BLOCK_H)

    # mask out-of-bounds
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_h = offs_h < H

    # compute base pointers
    left_batch_ptr = left_ptr + batch_idx * stride_left_b
    right_batch_ptr = right_ptr + batch_idx * stride_right_b
    out_batch_ptr = out_ptr + batch_idx * stride_out_b

    # allocate accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, N, BLOCK_K):
        # compute K tile size dynamically (maybe last tile smaller)
        cur_k = tl.minimum(BLOCK_K, N - k)

import triton
import triton.language as tl

@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    # compute block indices
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_z = tl.program_id(2)

    # compute batch and hidden block indices
    num_h_blocks = tl.cdiv(H, BLOCK_H)
    batch = pid_z // num_h_blocks
    hb = pid_z % num_h_blocks

    # offsets
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N
    h_start = hb * BLOCK_H

    # range for each dimension
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    offs_h = h_start + tl.arange(0, BLOCK_H)
    offs_k = tl.arange(0, BLOCK_K)

    # masks for out-of-bound indices
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_h = offs_h < H

    # base pointers for batch
    left_base = left_ptr + batch * stride_left_b
    right_base = right_ptr + batch * stride_right_b
    out_base = out_ptr + batch * stride_out_b

    # accumulator init
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, N, BLOCK_K):
        # compute k offsets
        k_start = k
        k_offs = k_start + offs_k
        mask_k = k_offs < N

        # left tile: shape (M, K, H)
        left_ptrs = left_base + \
            (offs_i[:, None, None] * stride_left_i) + \
            (k_offs[None, :, None] * stride_left_k) + \
            (offs_h[None, None, :] * stride_left_h)
        left_tile = tl.load(left_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
            other=0.0, dtype=tl.float16).to(tl.float32)

        # right tile transposed: we need right[k, j, h]
        right_ptrs = right_base + \
            (k_offs[:, None, None] * stride_right_k) + \
            (offs_j[None, :, None] * stride_right_j) + \
            (offs_h[None, None, :] * stride_right_h)
        right_tile = tl.load(right_ptrs,
            mask=mask_k[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0, dtype=tl.float16).to(tl.float32)

        # inner loop over hidden dim blocks
        for dh in range(0, BLOCK_H):
            # add contribution for each d
            a = left_tile[:, :, dh]   # (M, K)
            b = right_tile[:, :, dh]  # (K, N)
            acc[:, :, dh] += tl.dot(a, b)
    # write results to out
    out_ptrs = out_base + \
        (offs_i[:, None, None] * stride_out_i) + \
        (offs_j[None, :, None] * stride_out_j) + \
        (offs_h[None, None, :] * stride_out_h)
    # store with mask
    tl.store(out_ptrs, acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :])

"""
TriMul "outgoing" kernel (AlphaFold3 style) implemented with Triton.

Algorithm
---------
1. Layer‑norm the input tensor `x` over the channel dimension.
2. Compute two linear projections (left & right) and three sigmoid gates
   (left‑gate, right‑gate, out‑gate).  All heavy linear work is done in
   FP16 for bandwidth.
3. Apply the optional pair‑wise mask to the projected tensors and then
   multiply by the corresponding gates → `left` and `right`.
4. Core operation:
       out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   This is an N×N × N matrix‑multiply for each hidden channel `d`.
   A custom Triton kernel tiles the i‑, j‑, k‑ and hidden‑dimensions
   (M, N, K, H) and performs a batched GEMM with an explicit reduction
   over `k`.  The accumulator is float32 while the loads are FP16.
5. Layer‑norm the result over the hidden dimension, multiply by the
   `out_gate`, and finish with a final linear projection back to `dim`.
6. The function returns a tensor of shape [B, N, N, dim] (float32).

The kernel works for any batch size ≤2, sequence length up to 1024 and
hidden dimensions that are multiples of the BLOCK_H tile (default 8).
"""

import torch
import triton
import triton.language as tl


@triton.jit
def trimul_kernel(
    left_ptr,               # *fp16   [B, N, N, H]
    right_ptr,              # *fp16   [B, N, N, H] (will be read as right_T[k,j,h])
    out_ptr,                # *fp32   [B, N, N, H]
    B, N, H,                # int32 scalar sizes
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """
    Compute out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    with tiling over (i,j,k,h).  The inner product over k is performed
    with tl.dot for each hidden slice.
    """
    pid_i = tl.program_id(0)                     # block row (i)
    pid_j = tl.program_id(1)                     # block col (j)
    pid_z = tl.program_id(2)                     # batch * hidden‑block

    # decode batch index and hidden‑block index
    num_h_blocks = tl.cdiv(H, BLOCK_H)
    batch = pid_z // num_h_blocks
    hb = pid_z % num_h_blocks

    # ----- block start indices ------------------------------------------------
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N
    h_start = hb * BLOCK_H

    # offsets within the block
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    offs_h = h_start + tl.arange(0, BLOCK_H)
    offs_k = tl.arange(0, BLOCK_K)               # reused inside the k‑loop

    # out‑of‑bounds masks for the three static axes
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_h = offs_h < H

    # base pointers for the current batch
    left_base  = left_ptr  + batch * stride_left_b
    right_base = right_ptr + batch * stride_right_b
    out_base   = out_ptr   + batch * stride_out_b

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # ----- loop over the reduction dimension k --------------------------------
    num_k = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k):
        k_start = kk * BLOCK_K
        k_offs = k_start + offs_k                # absolute k indices
        mask_k = k_offs < N

        # --------------------------------------------------------------------
        # Load left[i,k,h]  (shape M x K x H)
        # --------------------------------------------------------------------
        left_ptrs = (
            left_base
            + (offs_i[:, None, None] * stride_left_i)
            + (k_offs[None, :, None] * stride_left_k)
            + (offs_h[None, None, :] * stride_left_h)
        )
        left_tile = tl.load(
            left_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
            other=0.0,
            dtype=tl.float16,
        ).to(tl.float32)

        # --------------------------------------------------------------------
        # Load right_T[k,j,h] = right[j,k,h]  (shape K x N x H)
        # --------------------------------------------------------------------
        right_ptrs = (
            right_base
            + (k_offs[:, None, None] * stride_right_k)   # k dimension first  → transposed view
            + (offs_j[None, :, None] * stride_right_i)   # j dimension (the original second N)
            + (offs_h[None, None, :] * stride_right_h)
        )
        right_tile = tl.load(
            right_ptrs,
            mask=mask_k[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
            dtype=tl.float16,
        ).to(tl.float32)

        # --------------------------------------------------------------------
        # Accumulate dot products for each hidden slice
        # --------------------------------------------------------------------
        for dh in range(0, BLOCK_H):
            # slice (M,K) and (K,N) for the current hidden channel
            a = left_tile[:, :, dh]      # (M, K)
            b = right_tile[:, :, dh]     # (K, N)
            acc[:, :, dh] += tl.dot(a, b)   # (M, N)

    # ----- write back ---------------------------------------------------------
    out_ptrs = (
        out_base
        + (offs_i[:, None, None] * stride_out_i)
        + (offs_j[None, :, None] * stride_out_j)
        + (offs_h[None, None, :] * stride_out_h)
    )
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
    )

    """

    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), norm_weight, norm_bias
    )  # [B,N,N,dim] float32

    # Cast to half for the projection/gate part (still cheap compared to the N³)
    x_h = x.to(compute_dtype)

    # ------------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(compute_dtype)
    right_proj_w = weights["right_proj.weight"].to(compute_dtype)

    left = torch.nn.functional.linear(x_h, left_proj_w)   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x_h, right_proj_w)

    # ------------------------------------------------------------------------
    # 3. Optional mask (applied before the gates)
    # ------------------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        mask_f = mask.to(compute_dtype).unsqueeze(-1)    # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------------
    # 4. Gates (sigmoid), also in FP16
    # ------------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(compute_dtype)
    right_gate_w = weights["right_gate.weight"].to(compute_dtype)
    out_gate_w   = weights["out_gate.weight"]           # keep FP32 for later

    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x_h, left_gate_w)
    ).to(compute_dtype)
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x_h, right_gate_w)
    ).to(compute_dtype)
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, out_gate_w)      # stay FP32
    )                                                 # [B,N,N,hidden]

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------------
    # 5. Core einsum over k using the Triton kernel
    # ------------------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Tiling parameters – chosen to fit H100 registers and shared memory
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    BLOCK_H = 8

    # strides in elements (torch gives strides in number of elements, not bytes)
    stride_l = left.stride()   # (B, N, N, H)
    stride_r = right.stride()
    stride_o = out.stride()

    grid = (
        ( (N + BLOCK_M - 1) // BLOCK_M,
          (N + BLOCK_N - 1) // BLOCK_N,
          B * ((hidden_dim + BLOCK_H - 1) // BLOCK_H) )
    )

    trimul_kernel[grid](
        left, right, out,
        B, N, hidden_dim,
        stride_l[0], stride_l[1], stride_l[2], stride_l[3],
        stride_r[0], stride_r[1], stride_r[2], stride_r[3],
        stride_o[0], stride_o[1], stride_o[2], stride_o[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )
    # ------------------------------------------------------------------------
    # 6. Post‑processing: layer‑norm, out‑gate, final linear
    # ------------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out, (hidden_dim,), to_out_norm_w, to_out_norm_b
    )  # still FP32

    out = out * out_gate

    to_out_w = weights["to_out.weight"]   # shape (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_w)  # final [B,N,N,dim]

    return out
```

The code above follows the reference PyTorch implementation, moving the dominant
`einsum('... i k d, ... j k d -> ... i j d')` into a fused Triton kernel while
keeping the cheap element‑wise parts in PyTorch.  All heavy memory traffic is
performed in FP16, the accumulator uses FP32 for numerical stability, and the
final output is returned as FP32.
