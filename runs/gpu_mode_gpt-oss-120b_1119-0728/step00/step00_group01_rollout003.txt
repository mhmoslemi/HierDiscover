    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B = input_tensor.shape[0]
    N = input_tensor.shape[1]
    D = dim

    # Convert mask to torch if None (maybe mask is torch Tensor with dtype bool or 0/1)
    if mask is None:

    mask = mask.to(input_tensor.dtype) # for multiplication

    # LayerNorm: compute mean and var over last dimension.
    # use torch.nn.functional.layer_norm: (input, normalized_shape, weight, bias, eps)
    # normalized_shape = (dim,)
    # outputs shape same as input.
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # Linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Apply mask if not nomask config (maybe config includes "nomask" flag) but we can always multiply.
    if not config.get("nomask", True):
        mask_unsqueeze = mask.unsqueeze(-1)  # shape B,N,N,1
        left = left * mask_unsqueeze
        right = right * mask_unsqueeze
    else:
        # no mask, just keep left/right unchanged.
        pass

    # Gating
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Reshape for Triton batched matmul
    # Permute to [B, hidden_dim, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # Flatten batch and hidden_dim
    B_h = B * hidden_dim
    left_flat = left_perm.view(B_h, N, N)
    right_flat = right_perm.view(B_h, N, N)
    # Transpose right for dot
    right_T = right_flat.transpose(1, 2).contiguous()  # shape (B_h, N, N)

    # Output placeholder
    out_flat = torch.empty_like(left_flat)  # shape (B_h, N, N)

    # Compute strides for Triton

    stride_batch_a = left_flat.stride(0)

    stride_ak = left_flat.stride(2)

    stride_bk = right_T.stride(1)
    stride_bn = right_T.stride(2)

    stride_batch_c = out_flat.stride(0)
    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)

    # Determine grid size
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    M = N
    K = N
    num_blocks_m = (M + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = B_h * num_blocks_m * num_blocks_n

    grid = (total_blocks,)

    # Launch kernel
    batched_matmul_kernel[grid](
        left_flat,
        right_T,
        out_flat,
        stride_batch_a, stride_am, stride_ak,
        stride_batch_b, stride_bk, stride_bn,
        stride_batch_c, stride_cm, stride_cn,
        B_h,
        M, N, K,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,

@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_batch_a, stride_am, stride_ak,
    stride_batch_b, stride_bk, stride_bn,
    stride_batch_c, stride_cm, stride_cn,
    BATCH: tl.constexpr,
    M: tl.constexpr, N: tl.constexpr, K: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    # compute batch index and block coordinates
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    batch = pid // (num_blocks_m * num_blocks_n)
    block_idx = pid % (num_blocks_m * num_blocks_n)
    pid_m = block_idx // num_blocks_n
    pid_n = block_idx % num_blocks_n

    # offsets for M and N
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # mask K
        mask_k = offs_k < K
        # Load A tile: shape BLOCK_M x BLOCK_K
        a_ptrs = A_ptr + batch * stride_batch_a + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & mask_k[None, :], other=0.0, dtype=tl.float16)
        # Load B tile: shape BLOCK_K x BLOCK_N (since B is KxN)
        b_ptrs = B_ptr + batch * stride_batch_b + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & (offs_n[None, :] < N), other=0.0, dtype=tl.float16)
        # Compute dot product
        acc += tl.dot(a, b)

    # Store result
    c_ptrs = C_ptr + batch * stride_batch_c + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_store = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_store)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # Determine values
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # Device
    device = input_tensor.device
    dtype = input_tensor.dtype

    # LayerNorm x
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    # Linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])
    # Mask if present
    if mask is not None:
        mask = mask.to(dtype).unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask
        right = right * mask
    # Gating
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()
    left = left * left_gate
    right = right * right_gate

    # Prepare for trig matmul
    # we will cast to float16 for kernel; left/right after gating may be float32; cast to half
    # Permute to [B, hidden_dim, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # Cast to half
    left_fp16 = left_perm.to(torch.float16)
    right_fp16 = right_perm.to(torch.float16)
    # flatten batch and hidden
    B = left_fp16.shape[0]
    H = left_fp16.shape[1]
    N = left_fp16.shape[2]
    BH = B * H
    left_flat = left_fp16.view(BH, N, N)
    right_flat = right_fp16.view(BH, N, N)

    # Transpose right for multiplication
    right_T = right_flat.transpose(1, 2).contiguous()

    # Output buffer in float32
    out_flat = torch.empty((BH, N, N), dtype=torch.float32, device=device)

    # Strides
    stride_batch_a = left_flat.stride(0)
    stride_am = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_batch_b = right_T.stride(0)
    stride_bk = right_T.stride(1)
    stride_bn = right_T.stride(2)

    stride_batch_c = out_flat.stride(0)
    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)

    # Block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Compute grid
    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = BH * num_blocks_m * num_blocks_n
    grid = (total_blocks,)

    batched_matmul_kernel[grid](
        left_flat,
        right_T,
        out_flat,
        stride_batch_a, stride_am, stride_ak,
        stride_batch_b, stride_bk, stride_bn,
        stride_batch_c, stride_cm, stride_cn,
        BH,
        N, N, N,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,  # maybe adjust
    )

    # Reshape output
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    # to_out_norm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])
    # out_gate already computed earlier (size [B,N,N,H]), but out_gate dtype may be float32; fine
    out = out * out_gate
    # final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out

"""
TriMul (outgoing) forward pass – Triton accelerated implementation.

The heavy part of the module is the 4‑D tensor einsum:
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

We rewrite it as a batched matrix multiplication for each hidden channel `h`:
    left  → (B·H, N, N)   with layout (i, k)
    right → (B·H, N, N)   with layout (j, k) → transpose → (B·H, K, N) where K = N
    out   → (B·H, N, N)   = left @ right_T

The kernel below performs this batched matmul in FP16 (accumulating in FP32) and
stores the result in FP32.  All other ops (LayerNorm, linear projections,
gating, mask handling, final LayerNorm and linear) are done with PyTorch
for simplicity.  The implementation supports all test configurations
mentioned in the prompt and runs efficiently on an H100.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ B_T)  where
#   A : (batch, M, K)  – layout (i, k)
#   B_T: (batch, K, N) – layout (k, j)  (the transposed right tensor)
#   C : (batch, M, N)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr,                       # *Pointer* to A matrix
    b_ptr,                       # *Pointer* to B_T matrix
    c_ptr,                       # *Pointer* to output matrix
    stride_batch_a, stride_am, stride_ak,   # Strides for A
    stride_batch_b, stride_bk, stride_bn,   # Strides for B_T
    stride_batch_c, stride_cm, stride_cn,   # Strides for C
    BATCH: tl.constexpr,                     # total batches = B * hidden_dim
    M: tl.constexpr, N: tl.constexpr, K: tl.constexpr,  # matrix sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # Compute batch index and block coordinates (i‑block, j‑block)
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    batch = pid // (num_blocks_m * num_blocks_n)
    block_idx = pid % (num_blocks_m * num_blocks_n)
    pid_m = block_idx // num_blocks_n
    pid_n = block_idx % num_blocks_n

    # Offsets for the current block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks to avoid out‑of‑bounds loads / stores
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Main K‑loop
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile: (BLOCK_M, BLOCK_K) – FP16 → cast to FP32 in the dot
        a_ptrs = (
            a_ptr
            + batch * stride_batch_a
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(a_ptrs,
                    mask=(mask_m[:, None] & mask_k[None, :]),
                    other=0.0,
                    dtype=tl.float16)

        # Load B_T tile: (BLOCK_K, BLOCK_N)
        b_ptrs = (
            b_ptr
            + batch * stride_batch_b
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(b_ptrs,
                    mask=(mask_k[:, None] & mask_n[None, :]),
                    other=0.0,
                    dtype=tl.float16)

        # FP16 dot‑product accumulated in FP32
        acc += tl.dot(a, b)

    # Store the result
    c_ptrs = (
        c_ptr
        + batch * stride_batch_c
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    tl.store(c_ptrs, acc, mask=(mask_m[:, None] & mask_n[None, :]))

# ----------------------------------------------------------------------
# Custom Triton‑accelerated TriMul forward
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Compute the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (dtype = float32)
    """
    # unpack arguments
    inp, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    device = inp.device
    dtype = inp.dtype

    # ---------- LayerNorm on input ----------
    x = F.layer_norm(
        inp,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]

    # ---------- Linear projections ----------
    left = F.linear(x, weights["left_proj.weight"])    # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])  # [B, N, N, hidden_dim]

    # ---------- Mask (if provided) ----------
    if mask is not None:
        # mask is [B, N, N] – broadcast on hidden dim
        mask = mask.to(dtype).unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ---------- Gating ----------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Prepare tensors for the batched matmul kernel.
    # Convert to FP16 for higher throughput; the kernel will accumulate in FP32.
    # ------------------------------------------------------------------
    # (B, N, N, H) -> (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous().to(torch.float16)
    right_perm = right.permute(0, 3, 1, 2).contiguous().to(torch.float16)

    B, H, N, _ = left_perm.shape
    batch_hidden = B * H

    # Collapse batch & hidden dim → (B·H, N, N)
    left_flat = left_perm.view(batch_hidden, N, N)
    right_flat = right_perm.view(batch_hidden, N, N)

    # Transpose the right operand so we compute A @ B_T
    right_T = right_flat.transpose(1, 2).contiguous()

    # Output buffer (FP32)
    out_flat = torch.empty((batch_hidden, N, N), dtype=torch.float32, device=device)

    # -------------------- Strides (in number of elements) --------------------
    stride_batch_a = left_flat.stride(0)
    stride_am = left_flat.stride(1)   # i‑stride (M)
    stride_ak = left_flat.stride(2)   # k‑stride (K)

    stride_batch_b = right_T.stride(0)
    stride_bk = right_T.stride(1)     # k‑stride (K)  – note: after transpose this is N
    stride_bn = right_T.stride(2)     # n‑stride (N)

    stride_batch_c = out_flat.stride(0)
    stride_cm = out_flat.stride(1)    # i‑stride (M)
    stride_cn = out_flat.stride(2)    # j‑stride (N)

    # -------------------- Kernel launch configuration --------------------
    BLOCK_M = 64          # rows per thread block
    BLOCK_N = 64          # cols per thread block
    BLOCK_K = 64          # reduction tile size

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = batch_hidden * num_blocks_m * num_blocks_n
    grid = (total_blocks,)

    # Launch the Triton kernel
    batched_matmul_kernel[grid](
        left_flat,
        right_T,
        out_flat,
        stride_batch_a, stride_am, stride_ak,
        stride_batch_b, stride_bk, stride_bn,
        stride_batch_c, stride_cm, stride_cn,
        batch_hidden,                # BATCH
        N, N, N,                    # M, N, K (all equal to seq_len)
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        # Tuning parameters – adjust for the target GPU
        num_warps=8,
        num_stages=3,
    )

    # ------------------------------------------------------------------
    # Reshape back to (B, N, N, hidden_dim) and apply the remaining layers.
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, hidden_dim]

    # Final LayerNorm (hidden_dim)
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # Apply the output gate (broadcasted over the last dim)
    out = out * out_gate

    # Linear projection back to `dim`
    out = F.linear(out, weights["to_out.weight"])

    return out
```

**Explanation of the workflow**

1. **LayerNorm + Linear Projections** – performed with PyTorch for simplicity.
2. **Mask & Gates** – element‑wise operations, still on the GPU in FP32.
3. **Permutation & FP16 conversion** – bring the tensors to layout `[B, H, N, N]`,
   flatten batch × hidden to a single leading dimension and transpose the right
   operand.
4. **Batched GEMM** – the Triton kernel computes, for each `(b, h)` pair,
   a matrix product `left[b,h] @ right[b,h].T`.  The kernel works on tiles
   (`BLOCK_M × BLOCK_N` output) and reduces over `K` in `BLOCK_K`‑ sized
   chunks, loading data as `float16` and accumulating in `float32`.
5. **Reshape & post‑processing** – after the kernel we reshape back, apply the
   second LayerNorm, multiply by the output gate, and finish with a linear
   projection to the original `dim`.

The kernel is fully statically typed (`tl.constexpr`) and uses no Python
control‑flow, satisfying Triton‑3.3.1 requirements and delivering a fast
implementation suitable for the target H100 hardware.
