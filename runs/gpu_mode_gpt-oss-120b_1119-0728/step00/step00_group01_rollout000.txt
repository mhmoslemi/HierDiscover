@triton.jit
def tri_mul_kernel(
        left_ptr, right_ptr, out_ptr,
        B, N, H,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_j, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bc = tl.program_id(2)

    # compute batch and channel index
    total_bc = B * H
    batch = pid_bc // H
    chan = pid_bc % H

    # offsets for output block
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask out-of-range
    mask_i = offs_i < N
    mask_j = offs_j < N

    # compute pointer base for left: left[batch, i, k, chan]
    # A index: (batch, i, k)
    # B index for right_T: (batch, k, j)
    # compute base address offsets
    batch_offset_left = batch * stride_left_b
    batch_offset_right = batch * stride_right_b
    batch_offset_out = batch * stride_out_b

    # offset for channel (H)
    chan_offset_left = chan * stride_left_h
    chan_offset_right = chan * stride_right_h
    chan_offset_out = chan * stride_out_h

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute k offset
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left tile (M,K)
        # left_ptrs = left_ptr + batch_offset_left + offs_i[:,None] * stride_left_i + offs_k[None,:] * stride_left_k + chan_offset_left
        # Use tl.load with mask for i, k
        left_tile = tl.load(
            left_ptr + batch_offset_left + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k + chan_offset_left,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0
        ).to(tl.float32)  # cast to fp32

        # load right tile transposed (K,N) i.e., using right_ptr base with swapped axes
        # right_T[k, j] = right[batch, j, k, chan]
        right_tile = tl.load(
            right_ptr + batch_offset_right + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_j + chan_offset_right,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0
        ).to(tl.float32)

        # accumulate dot
        acc += tl.dot(left_tile, right_tile)

        # increment pointers (not needed because we compute offsets each iter)
    # write output
    out_ptrs = out_ptr + batch_offset_out + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j + chan_offset_out

    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
custom_kernel
==============

Implements the *outgoing* TriMul operation (AlphaFold‑3) using a
Triton kernel for the heavy N³ contraction:

    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

where ``left`` and ``right`` are the linearly projected and gated
versions of the normalized input.  The kernel performs a batched
per‑channel matrix multiplication (one matmul for each hidden channel
``h``) with tiling over the i, j and k dimensions.  All other
operations (layer‑norm, linear projections, sigmoid gates, final
normalisation and projection) are carried out with regular PyTorch
functional calls, which keeps the code simple while still satisfying
the requirement that “part of the operations” be written in Triton.

The kernel works for any batch size (1‑2), sequence length up to 1024
and hidden dimensions that are a multiple of the block size.  It
handles optional masking and works in fp32 (the tensor layout is
contiguous (B,N,N,H) → strides (N·N·H, N·H, H, 1)).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched per‑channel matmul
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,          # const float*  (B,N,N,H) left tensor
    right_ptr,         # const float*  (B,N,N,H) right tensor
    out_ptr,           # float*        (B,N,N,H) output tensor
    B, N, H,           # int meta‑parameters
    # strides for the three tensors (contiguous layout)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute out = left @ right^T per hidden channel.

    left  : (B, i, k, H)
    right : (B, j, k, H)  (we read it as (B, k, j, H) to get a K×N tile)
    out   : (B, i, j, H)
    """
    pid_i = tl.program_id(0)          # block index over the i dimension
    pid_j = tl.program_id(1)          # block index over the j dimension
    pid_bc = tl.program_id(2)         # flattened index for (batch, channel)

    # --------------------------------------------------------------
    # Decode batch and hidden‑channel index
    # --------------------------------------------------------------
    batch = pid_bc // H
    chan  = pid_bc %  H

    # --------------------------------------------------------------
    # Offsets for the output tile (i, j)
    # --------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointers (including batch and channel offsets)
    batch_off_left = batch * stride_left_b
    batch_off_right = batch * stride_right_b
    batch_off_out = batch * stride_out_b

    chan_off_left = chan * stride_left_h
    chan_off_right = chan * stride_right_h
    chan_off_out = chan * stride_out_h

    # --------------------------------------------------------------
    # Accumulator for the tile
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over the reduction dimension k
    # --------------------------------------------------------------
    num_blocks_k = tl.cdiv(N, BLOCK_K)
    for k_iter in range(0, num_blocks_k):
        offs_k = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # -------- load left tile (M × K) ---------------------------------
        left_tile = tl.load(
            left_ptr
            + batch_off_left
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
            + chan_off_left,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0
        ).to(tl.float32)

        # -------- load right tile transposed (K × N) ----------------------
        # right is stored as (batch, j, k, chan);
        # we need (k, j) for the dot product.
        right_tile = tl.load(
            right_ptr
            + batch_off_right
            + offs_k[:, None] * stride_right_k   # row = k
            + offs_j[None, :] * stride_right_j   # col = j
            + chan_off_right,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0
        ).to(tl.float32)

        # -------- accumulate ------------------------------------------------
        acc += tl.dot(left_tile, right_tile)

    # --------------------------------------------------------------
    # Write out the computed tile
    # --------------------------------------------------------------
    out_ptrs = out_ptr \
        + batch_off_out \
        + offs_i[:, None] * stride_out_i \
        + offs_j[None, :] * stride_out_j \
        + chan_off_out

    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Entry point called by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] – the result of the TriMul forward
        pass (no gradients).
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]               # input / output channel dimension
    hidden_dim = config["hidden_dim"] # projection dimension
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # Retrieve weights (all are 2‑D or 1‑D tensors, stored on CUDA)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # --------------------------------------------------------------
    # 1.) LayerNorm over the last dimension (dim)
    # --------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias)

    # --------------------------------------------------------------
    # 2.) Linear projections (no bias)
    # --------------------------------------------------------------
    # shape after linear: (B, N, N, hidden_dim)
    left  = F.linear(x, left_proj_weight)   # left_proj_weight : (hidden_dim, dim)
    right = F.linear(x, right_proj_weight)

    # --------------------------------------------------------------
    # 3.) Optional mask (broadcasted on hidden dim)
    # --------------------------------------------------------------
    if not nomask:
        # mask shape: (B, N, N) -> (B, N, N, 1)
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)
        left  = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 4.) Gating values (sigmoid of linear projections of the *normalized* x)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    left  = left  * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5.) Core TriMul contraction via Triton kernel
    # --------------------------------------------------------------
    B, N, _, _ = left.shape  # left.shape == right.shape == (B,N,N,hidden_dim)

    # Allocate output buffer (same dtype as left/right)
    out = torch.empty_like(left, dtype=torch.float32, device=left.device)

    # Triton grid configuration
    #   grid[0] -> i‑blocks, grid[1] -> j‑blocks, grid[2] -> batch*hidden
    BLOCK_M = 64   # tile size over the i dimension
    BLOCK_N = 64   # tile size over the j dimension
    BLOCK_K = 64   # tile size over the reduction (k) dimension

    grid_i = triton.cdiv(N, BLOCK_M)
    grid_j = triton.cdiv(N, BLOCK_N)
    grid_bc = B * hidden_dim

    # Strides of the three tensors (contiguous layout)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Launch kernel
    tri_mul_kernel[
        (grid_i, grid_j, grid_bc)
    ](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B, N, hidden_dim,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_j, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,  # reasonable default for fp32 on H100
    )

    # --------------------------------------------------------------
    # 6.) Final LayerNorm + gating + output projection
    # --------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias)

    out = out * out_gate                     # broadcasted element‑wise

    # Linear projection back to original dimension (hidden_dim -> dim)
    out = F.linear(out, to_out_weight)      # to_out_weight : (dim, hidden_dim)

    return out
