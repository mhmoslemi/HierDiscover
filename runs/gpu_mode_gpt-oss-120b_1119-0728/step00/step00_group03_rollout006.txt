def batched_tri_mul_kernel(

    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    left_proj = F.linear(x_norm, weights["left_proj.weight"])
    right_proj = F.linear(x_norm, weights["right_proj.weight"])
    left_gate_pre = F.linear(x_norm, weights["left_gate.weight"])
    right_gate_pre = F.linear(x_norm, weights["right_gate.weight"])
    out_gate_pre = F.linear(x_norm, weights["out_gate.weight"])

    # gates
    left_gate = left_gate_pre.sigmoid()
    right_gate = right_gate_pre.sigmoid()
    out_gate = out_gate_pre.sigmoid()

    # mask handling
    if mask is None or config.get("nomask", False):
        mask_tensor = torch.ones((bs, seq_len, seq_len), dtype=x_norm.dtype, device=device)
    else:

    mask_tensor = mask_tensor.unsqueeze(-1)  # [B,N,N,1]

    # apply mask and gates
    left = left_proj * mask_tensor * left_gate
    right = right_proj * mask_tensor * right_gate

    # cast to half for Triton kernel
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # reshape for batched GEMM: (B, H, N, N) -> (B*H, N, N)
    left_h = left_h.permute(0, 3, 1, 2).contiguous()
    right_h = right_h.permute(0, 3, 1, 2).contiguous()
    batch_h = bs * hidden_dim
    left_h = left_h.view(batch_h, seq_len, seq_len)
    right_h = right_h.view(batch_h, seq_len, seq_len)

    # allocate output
    out_h = torch.empty_like(left_h, dtype=torch.float16)

    # get strides
    stride_ab, stride_am, stride_ak = left_h.stride()   # (batch, row, col)
    # for B (transposed)
    b_stride_batch, b_stride_row, b_stride_col = right_h.stride()
    # For transposed: row stride = original col stride, col stride = original row stride
    stride_bm = b_stride_col
    stride_bk = b_stride_row
    stride_bb = b_stride_batch

    # output strides are same as left
    stride_cb, stride_cm, stride_cn = out_h.stride()

    # launch Triton kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        batch_h,
        triton.cdiv(seq_len, BLOCK_M),
        triton.cdiv(seq_len, BLOCK_N),
    )
    batched_tri_mul_kernel[grid](
        left_h,
        right_h,
        out_h,
        seq_len,
        seq_len,
        seq_len,
        stride_am,
        stride_ak,
        stride_ab,
        stride_bm,
        stride_bk,
        stride_bb,
        stride_cm,
        stride_cn,
        stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B,N,N,H]
    out = out_h.view(bs, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1).contiguous()

    # cast back to float32 for remaining ops
    out = out.to(torch.float32)

    # to_out_norm
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # output gate
    out = out * out_gate

    # final linear
    out = F.linear(out, weights["to_out.weight"])

    return out

"""
TriMul (outgoing) forward pass with Triton acceleration.

The dominant operation in TriMul is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which can be seen as a batched matrix multiplication:
for each (b,d) pair we compute  out_{b,d} = left_{b,d} @ right_{b,d}ᵀ.
The kernel treats every hidden channel as an independent batch (B·D batches)
and performs the multiplication in FP16 with an FP32 accumulator.
All other operations (LayerNorms, linear projections and gating) are executed
with PyTorch’s highly‑optimised kernels.

The function `custom_kernel` receives:
    (input_tensor, mask, weights, config)
and returns the processed tensor of shape [B, N, N, dim] (float32).
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matmul computing C = A @ Bᵀ
# A, B and C are all [batch, M, K] tensors (M == N == K == seq_len).
# The kernel works on half‑precision inputs and accumulates in FP32.
# ----------------------------------------------------------------------
@triton.jit
def batch_tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix dimensions (M=N=K=seq_len)
    stride_am, stride_ak, stride_ab,   # strides for A: (row, col, batch)
    stride_bm, stride_bk, stride_bb,   # strides for Bᵀ: (col, row, batch)
    stride_cm, stride_cn, stride_cb,   # strides for C: (row, col, batch)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched matmul: for each batch `b`
        C[b,i,j] = Σ_k A[b,i,k] * B[b,j,k]
    A and B are half‑precision, accumulation is FP32.
    """
    batch_id = tl.program_id(0)            # batch index (B·H)
    pid_m    = tl.program_id(1)            # block row
    pid_n    = tl.program_id(2)            # block col

    # ------------------------------------------------------------------
    # Offsets for the output tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Pointers to the current tiles of A and Bᵀ
    # A : [batch, M, K] row‑major, stride_ak == 1
    # Bᵀ: we read B as transposed, i.e. we access B[j,k] (j is column of Bᵀ)
    # ------------------------------------------------------------------
    a_ptrs = a_ptr + batch_id * stride_ab + offs_m[:, None] * stride_am + (tl.arange(0, BLOCK_K)[None, :] * stride_ak)
    b_ptrs = b_ptr + batch_id * stride_bb + (tl.arange(0, BLOCK_K)[:, None] * stride_bk) + offs_n[None, :] * stride_bm

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k  = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)                # (BLOCK_M, BLOCK_K) half
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)                # (BLOCK_K, BLOCK_N) half

        # dot product, promote to FP32
        acc += tl.dot(a.to(tl.float32), b.to(tl.float32))

        # advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    # Store the result (cast back to FP16)
    # ------------------------------------------------------------------
    c_ptrs = c_ptr + batch_id * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul forward.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim]
        - mask         : torch.Tensor or None, shape [B, N, N]
        - weights      : dict of torch.Tensors (model parameters)
        - config       : dict with keys "dim", "hidden_dim", "nomask" ...

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype  = input_tensor.dtype

    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    batch_size, seq_len, _, _ = input_tensor.shape

    # --------------------------------------------------------------
    # 1. LayerNorm over the feature dimension
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim], float32

    # --------------------------------------------------------------
    # 2. Linear projections (no bias) and gating pre‑activations
    # --------------------------------------------------------------
    left_proj  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right_proj = F.linear(x, weights["right_proj.weight"])
    left_gate_pre  = F.linear(x, weights["left_gate.weight"])
    right_gate_pre = F.linear(x, weights["right_gate.weight"])
    out_gate_pre   = F.linear(x, weights["out_gate.weight"])

    # --------------------------------------------------------------
    # 3. Sigmoid gates
    # --------------------------------------------------------------
    left_gate  = left_gate_pre.sigmoid()
    right_gate = right_gate_pre.sigmoid()
    out_gate   = out_gate_pre.sigmoid()   # shape [B,N,N,hidden_dim]

    # --------------------------------------------------------------
    # 4. Optional mask (mask is 0/1)
    # --------------------------------------------------------------
    if mask is None or config.get("nomask", False):
        mask_tensor = torch.ones((batch_size, seq_len, seq_len),
                                 dtype=x.dtype, device=device)
    else:
        mask_tensor = mask.to(x.dtype)
    mask_tensor = mask_tensor.unsqueeze(-1)  # [B,N,N,1]

    # --------------------------------------------------------------
    # 5. Apply mask + gating
    # --------------------------------------------------------------
    left  = left_proj * mask_tensor * left_gate   # [B,N,N,hidden_dim]
    right = right_proj * mask_tensor * right_gate

    # --------------------------------------------------------------
    # 6. Cast to half‑precision for the massive N³ operation
    # --------------------------------------------------------------
    left_h  = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # --------------------------------------------------------------
    # 7. Reshape to (B·H, N, N) so each hidden channel becomes a batch
    # --------------------------------------------------------------
    left_h  = left_h.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_h = right_h.permute(0, 3, 1, 2).contiguous()
    batch_h = batch_size * hidden_dim
    left_h  = left_h.view(batch_h, seq_len, seq_len)
    right_h = right_h.view(batch_h, seq_len, seq_len)

    # --------------------------------------------------------------
    # 8. Allocate output buffer (half)
    # --------------------------------------------------------------
    out_h = torch.empty_like(left_h, dtype=torch.float16)   # (B·H,N,N)

    # --------------------------------------------------------------
    # 9. Extract strides for the kernel
    # --------------------------------------------------------------
    # A (left_h) strides: (batch, row, col)
    stride_ab, stride_am, stride_ak = left_h.stride()
    # B (right_h) original strides
    b_batch, b_row, b_col = right_h.stride()
    # For Bᵀ we swap row/col strides:
    stride_bb = b_batch          # batch stride stays the same
    stride_bk = b_col            # row stride of Bᵀ (k dimension) = original col stride (=1)
    stride_bm = b_row            # col stride of Bᵀ (j dimension) = original row stride (=N)

    # C strides are identical to A's layout
    stride_cb, stride_cm, stride_cn = out_h.stride()

    # --------------------------------------------------------------
    # 10. Launch Triton kernel
    # --------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        batch_h,
        triton.cdiv(seq_len, BLOCK_M),
        triton.cdiv(seq_len, BLOCK_N),
    )
    batch_tri_mul_kernel[grid](
        left_h,
        right_h,
        out_h,
        seq_len,
        seq_len,
        seq_len,
        stride_am,
        stride_ak,
        stride_ab,
        stride_bm,
        stride_bk,
        stride_bb,
        stride_cm,
        stride_cn,
        stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 11. Reshape back to [B, N, N, hidden_dim]
    # --------------------------------------------------------------
    out = out_h.view(batch_size, hidden_dim, seq_len, seq_len)
    out = out.permute(0, 2, 3, 1).contiguous()   # [B, N, N, hidden_dim]
    out = out.to(torch.float32)                  # back to FP32 for the rest

    # --------------------------------------------------------------
    # 12. Output LayerNorm (hidden_dim) + output gate
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate

    # --------------------------------------------------------------
    # 13. Final linear projection to original dimension
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
