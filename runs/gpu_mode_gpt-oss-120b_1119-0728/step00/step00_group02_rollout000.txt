"""
TriMul “outgoing” forward (AlphaFold3 style) – Triton‑accelerated elementwise
fusing of mask & gate, followed by batched GEMM for the O(N³) contraction.

Algorithm
---------
1. Layer‑norm the input tensor x ∈ ℝ^{B×N×N×C}.
2. Compute five linear projections:
   * left  = x·W_left          (→ H)
   * right = x·W_right         (→ H)
   * l_gate = σ(x·W_left_gate) (→ H)
   * r_gate = σ(x·W_right_gate)(→ H)
   * o_gate = σ(x·W_out_gate)  (→ H)
3. Fuse mask (if present), left_gate and right_gate with a tiny Triton kernel:
       left_f   = left   * l_gate * mask
       right_f  = right  * r_gate * mask
   The kernel works on the flattened index  (b,i,k,h)  and respects arbitrary
   strides, so it works for any contiguous / non‑contiguous layout.
4. Batched matrix multiplication over the “k’’ dimension:
       out_{b,i,j,h} = Σ_k left_f[b,i,k,h] * right_f[b,j,k,h]
   Implemented with a single torch.bmm on a view (B·H, N, N) → (B·H, N, N).
   The multiplication is performed in FP16 for speed and cast back to FP32.
5. Apply a second Layer‑norm on the hidden dimension, modulate by o_gate,
   and project back to the original channel dimension.

Only the elementwise mask‑gate fusion is written in Triton – the heavy
contraction uses cuBLAS via torch.bmm, which is already highly tuned on an
H100.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse mask with left/right projections and their gates.
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    # input pointers
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    # output pointers (same layout as left/right)
    out_left_ptr, out_right_ptr,
    # shape parameters
    B, N, H,
    # strides for left/right (assume same layout for gates)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    # stride for mask (B,N,N)
    stride_mask_b, stride_mask_i, stride_mask_k,
    # compile‑time block size
    BLOCK: tl.constexpr,
):
    """Fuse mask, left/right gates and the projections."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)                     # [0, BLOCK)
    total = B * N * N * H
    mask = offs < total

    # ----- decode linear index -----
    # idx = b * (N*N*H) + i * (N*H) + k * H + h
    idx = offs
    b = idx // (N * N * H)
    rem = idx % (N * N * H)
    i = rem // (N * H)
    rem = rem % (N * H)
    k = rem // H
    h = rem % H

    # ----- compute element offsets -----
    off_left = (
        b * stride_left_b
        + i * stride_left_i
        + k * stride_left_k
        + h * stride_left_h
    )
    off_right = (
        b * stride_right_b
        + i * stride_right_i
        + k * stride_right_k
        + h * stride_right_h
    )
    off_lgate = off_left   # same layout
    off_rgate = off_right  # same layout
    off_mask = b * stride_mask_b + i * stride_mask_i + k * stride_mask_k

    # ----- load values (masked) -----
    l_val   = tl.load(left_ptr   + off_left,   mask=mask, other=0.0)
    r_val   = tl.load(right_ptr  + off_right,  mask=mask, other=0.0)
    lg_val  = tl.load(left_gate_ptr  + off_lgate,  mask=mask, other=0.0)
    rg_val  = tl.load(right_gate_ptr + off_rgate, mask=mask, other=0.0)
    m_val   = tl.load(mask_ptr   + off_mask,   mask=mask, other=1.0)

    # ----- fuse -----
    out_l = l_val * lg_val * m_val
    out_r = r_val * rg_val * m_val

    # ----- store -----
    tl.store(out_left_ptr  + off_left,  out_l, mask=mask)
    tl.store(out_right_ptr + off_right, out_r, mask=mask)

# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul forward pass (outgoing version) with a Triton‑accelerated mask &
    gate fusion.  Arguments:
        data = (input_tensor, mask_tensor_or_None, weights_dict, config_dict)

    Returns:
        Tensor of shape [B, N, N, C]  (float32)
    """
    # ------------------------------------------------------------------
    # 1️⃣ unpack inputs & configuration
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    # input shape: [B, N, N, C]
    B, N, _, C = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # ``nomask`` flag – if True we ignore the supplied mask
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 2️⃣ load weights (all are FP32 tensors)
    # ------------------------------------------------------------------
    # LayerNorm on the input channel dimension C
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    # Linear projections (C → hidden_dim)
    left_proj_weight  = weights["left_proj.weight"]   # (hidden_dim, C)
    right_proj_weight = weights["right_proj.weight"]
    # Gates (C → hidden_dim)
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]
    # Output LayerNorm (hidden_dim)
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    # Final linear (hidden_dim → C)
    to_out_weight = weights["to_out.weight"]        # (C, hidden_dim)

    # ------------------------------------------------------------------
    # 3️⃣ Input LayerNorm
    # ------------------------------------------------------------------
    # torch.nn.functional.layer_norm expects normalized_shape = (C,)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(C,),
        weight=norm_weight,
        bias=norm_bias,
    )  # shape: [B, N, N, C]

    # ------------------------------------------------------------------
    # 4️⃣ Linear projections & gates (still FP32)
    # ------------------------------------------------------------------
    left      = torch.nn.functional.linear(x, left_proj_weight)      # [B,N,N,H]
    right     = torch.nn.functional.linear(x, right_proj_weight)     # [B,N,N,H]
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate= torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate  = torch.nn.functional.linear(x, out_gate_weight).sigmoid()  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 5️⃣ Prepare mask (float32, broadcastable)
    # ------------------------------------------------------------------
    if nomask or mask_tensor is None:
        # all‑ones mask – creates the smallest possible tensor to keep the
        # kernel simple (no need to branch on mask presence inside the kernel)
        mask = torch.ones((B, N, N), dtype=left.dtype, device=left.device)
    else:
        # Ensure mask dtype matches the others and add a singleton dim later
        mask = mask_tensor.to(dtype=left.dtype)

    # ------------------------------------------------------------------
    # 6️⃣ Fuse mask & gates with Triton
    # ------------------------------------------------------------------
    left_fused  = torch.empty_like(left)
    right_fused = torch.empty_like(right)

    total_elements = B * N * N * hidden_dim
    BLOCK = 1024  # typical Triton block size, works well for most sizes
    grid = (triton.cdiv(total_elements, BLOCK),)

    _gate_mask_fuse_kernel[grid](
        left, right,
        left_gate, right_gate,
        mask,
        left_fused, right_fused,
        B, N, hidden_dim,
        # strides for left
        left.stride(0), left.stride(1), left.stride(2), left.stride(3),
        # strides for right
        right.stride(0), right.stride(1), right.stride(2), right.stride(3),
        # strides for gates (identical to left/right)
        left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
        # strides for mask (B,N,N)
        mask.stride(0), mask.stride(1), mask.stride(2),
        BLOCK=BLOCK,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Batched GEMM:  out[b,i,j,h] = Σ_k left_fused[b,i,k,h] * right_fused[b,j,k,h]
    #    We perform the GEMM in FP16 for higher throughput and cast back.
    # ------------------------------------------------------------------
    left_fused = left_fused.to(torch.float16)
    right_fused = right_fused.to(torch.float16)

    # reshape to (B*H, N, N) -> treat each hidden channel as an independent batch
    left_mat  = left_fused.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_mat = right_fused.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # batched matrix multiply with the transpose on the second operand
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N) FP16

    # reshape back to [B, N, N, H] and cast to FP32 for the remaining ops
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm + out‑gate modulation
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9️⃣ Final linear projection back to original channel dimension C
    # ------------------------------------------------------------------
    output = torch.nn.functional.linear(out, to_out_weight)  # [B, N, N, C]

    return output
