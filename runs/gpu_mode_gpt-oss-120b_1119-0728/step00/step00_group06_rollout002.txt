"""
TriMul (outgoing) implementation.

The costly part of the TriMul forward pass is the cubic contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which can be seen as a batch of matrix‑matrix products:
    out_{b,:, :, d} = left_{b,:, :, d} @ right_{b,:, :, d}.T

We implement this contraction with a custom Triton kernel that
processes each (batch, hidden‑dim) pair as an independent tiled
matrix‑multiply (M×K)·(K×N) → (M×N).  All other operations
(layer‑norms, linear projections, gating, final projection) are
performed with PyTorch functional calls.  The kernel works for
arbitrary batch size, sequence length and hidden dimension and
handles masks and edge‑tiles via masking.

The final output has shape [B, N, N, dim] (float32) as required.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel for the N³ contraction
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    # pointers
    left_ptr, right_ptr, out_ptr,
    # problem size
    B, N, H,
    # strides for left (shape: [B, N, N, H])
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    # strides for right (same layout as left)
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    # strides for out (shape: [B, N, N, H])
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # tile sizes (compile‑time constants)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    by treating each (b, h) pair as an independent matrix multiply
    (N × N) @ (N × N)ᵀ.
    """
    pid_m = tl.program_id(0)          # tile row (i)
    pid_n = tl.program_id(1)          # tile col (j)
    pid_bh = tl.program_id(2)         # flat index for (batch, hidden)

    # --------------------------------------------------------------
    # decode batch and hidden index from pid_bh
    # --------------------------------------------------------------
    batch = pid_bh // H
    hidden = pid_bh % H

    # --------------------------------------------------------------
    # compute the absolute offsets for the current tile
    # --------------------------------------------------------------
    i_offset = pid_m * BLOCK_M
    j_offset = pid_n * BLOCK_N

    # range of row / column indices inside the tile
    offs_i = i_offset + tl.arange(0, BLOCK_M)
    offs_j = j_offset + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds rows / cols
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator for the tile
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of k‑tiles
    num_k = tl.cdiv(N, BLOCK_K)

    # ------------------------------------------------------------------
    # Loop over k dimension tiles
    # ------------------------------------------------------------------
    for k_iter in range(num_k):
        k_offset = k_iter * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # --------------------------------------------------------------
        # Load A block (left)   shape (BLOCK_M, BLOCK_K)
        # --------------------------------------------------------------
        a_ptrs = (
            left_ptr
            + batch * stride_left_b
            + hidden * stride_left_h
            + (offs_i[:, None] * stride_left_i)
            + (offs_k[None, :] * stride_left_k)
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # --------------------------------------------------------------
        # Load B block (right) transposed, shape (BLOCK_K, BLOCK_N)
        # We need right[b, j, k, h] but we load it as (k, j)
        # --------------------------------------------------------------
        b_ptrs = (
            right_ptr
            + batch * stride_right_b
            + hidden * stride_right_h
            + (offs_k[:, None] * stride_right_k)      # k dimension
            + (offs_j[None, :] * stride_right_i)      # j dimension
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )

        # --------------------------------------------------------------
        # Matrix‑multiply the two tiles and accumulate
        # --------------------------------------------------------------
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)

    # ------------------------------------------------------------------
    # Write the tile to global memory
    # ------------------------------------------------------------------
    c_ptrs = (
        out_ptr
        + batch * stride_out_b
        + hidden * stride_out_h
        + (offs_i[:, None] * stride_out_i)
        + (offs_j[None, :] * stride_out_j)
    )
    tl.store(
        c_ptrs,
        acc,
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
# Helper that launches the Triton kernel
# ----------------------------------------------------------------------
def _tri_mul_triton(left: torch.Tensor, right: torch.Tensor,
                    block_m: int = 64, block_n: int = 64, block_k: int = 64) -> torch.Tensor:
    """
    Batched matrix multiplication for the TriMul contraction.
    left / right: [B, N, N, H] (contiguous, float32/float16)
    Returns out of same shape.
    """
    assert left.shape == right.shape
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # ensure contiguous layout for correct stride interpretation
    left = left.contiguous()
    right = right.contiguous()
    out = out.contiguous()

    # strides (in elements, not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # number of tiles
    num_tiles_m = (N + block_m - 1) // block_m
    num_tiles_n = (N + block_n - 1) // block_n

    grid = (num_tiles_m, num_tiles_n, B * H)   # (i_tile, j_tile, batch*hidden)

    _tri_mul_kernel[grid](
        left, right, out,
        B, N, H,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_i, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=block_m, BLOCK_N=block_n, BLOCK_K=block_k,
    )
    return out

def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module.

    Arguments
    ---------
    data : tuple
        (input, mask, weights, config)
        - input : torch.Tensor, shape [B, N, N, dim]
        - mask  : torch.Tensor, shape [B, N, N] (may be None)
        - weights: dict of model parameters
        - config : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, on the same device as input)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5  # default eps used by torch.nn.LayerNorm

    # ------------------------------------------------------------------
    # Layer‑norm on the input (normalize over the last dimension)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )

    # ------------------------------------------------------------------
    # Linear projections (no bias) -> [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Optional mask (mask shape [B, N, N] → broadcast to hidden_dim)
    # ------------------------------------------------------------------
    if mask_tensor is not None and mask_tensor.numel() > 0:
        mask = mask_tensor.to(dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # Gating values (sigmoid) – computed from the *original* normalized x
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Core cubic contraction via Triton
    # ------------------------------------------------------------------
    # Select reasonable tile sizes; they work for N up to 1024.
    # For very small N we fallback to a smaller block to avoid excessive padding.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    if dim <= 128:   # not a strict rule; just a heuristic for the test sizes
        BLOCK_M = 64
        BLOCK_N = 64
        BLOCK_K = 64
    else:
        BLOCK_M = 32
        BLOCK_N = 32
        BLOCK_K = 32

    out_hidden = _tri_mul_triton(left, right, block_m=BLOCK_M, block_n=BLOCK_N, block_k=BLOCK_K)

    # ------------------------------------------------------------------
    # Post‑processing: layer‑norm over hidden_dim, gating, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
