```
def custom_kernel(data):
    # unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)

    # Linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Gating
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    # Apply mask and gates using Triton kernel
    if mask is not None:
        mask = mask.to(x.dtype)
        mask = mask.view(mask.shape[0], mask.shape[1], mask.shape[2], 1)  # broadcast

        # Compute left = left * left_gate * mask
        left = fuse_mul_gate_mask(left, left_gate, mask)  # using kernel
        # Compute right = right * right_gate * mask
        right = fuse_mul_gate_mask(right, right_gate, mask)  

    else:
        left = left * left_gate
        right = right * right_gate

    # Compute out = left @ right^T per hidden dim
    # Reshape for batched matmul
    left_perm = left.permute(0, 3, 1, 2)   # B, H, N, N
    right_perm = right.permute(0, 3, 1, 2)
    out = torch.matmul(left_perm, right_perm.transpose(-2, -1))  # B, H, N, N
    out = out.permute(0, 2, 3, 1)  # B, N, N, H

    # to_out_norm layernorm over hidden_dim
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=eps)

    # Multiply by out_gate
    out = out * out_gate

    # Final linear mapping hidden_dim -> dim
    output = torch.nn.functional.linear(out, weights["to_out.weight"])

    return output
```

import torch
import triton
import triton.language as tl

@triton.jit
def fuse_mul_gate_mask_kernel(a_ptr, gate_ptr, mask_ptr, out_ptr,
                              total_elements, hidden_dim,
                              BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # Compute index into mask: each hidden_dim elements share the same mask entry
    mask_idx = offs // hidden_dim

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)

    out = a * g * m
    tl.store(out_ptr + offs, out, mask=mask)

def _apply_gate_and_mask(a: torch.Tensor, gate: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Fuse element‑wise multiplication `a * gate * mask` using a Triton kernel.
    Arguments:
        a   – Tensor of shape [B, N, N, H]
        gate– Tensor of same shape as ``a``.
        mask– Tensor of shape [B, N, N] (broadcasted over ``H``).
    Returns:
        Tensor of same shape as ``a``.
    """
    # flatten tensors
    a_flat = a.contiguous().view(-1)
    gate_flat = gate.contiguous().view(-1)
    mask_flat = mask.contiguous().view(-1)          # [B*N*N]
    out_flat = torch.empty_like(a_flat)

    total = a_flat.numel()
    hidden_dim = a.shape[-1]
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)
    fuse_mul_gate_mask_kernel[grid](
        a_flat,
        gate_flat,
        mask_flat,
        out_flat,
        total,
        hidden_dim,
        BLOCK_SIZE=BLOCK,
    )
    return out_flat.view_as(a)

def custom_kernel(data):
    """
    Implements the “outgoing” TriMul operation.
    Steps:
    1. Layer‑norm the input.
    2. Two linear projections (left / right) and three gating linear layers.
    3. Fuse mask, gate and projection via a Triton kernel.
    4. For each hidden channel compute a batched matrix multiplication
       left @ rightᵀ (equivalent to the einsum in the reference implementation).
    5. Apply a second LayerNorm, multiply by the output gate and a final linear projection.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input layer‑norm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) for left/right and gating logits
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 3) Apply optional mask and element‑wise gates.
    #    The mask is broadcast over the hidden dimension.
    # ------------------------------------------------------------------
    if config.get("nomask", False) or mask is None:
        # No mask – just apply the gates.
        left = left * left_gate
        right = right * right_gate
    else:
        # mask shape: [B, N, N] or [B, N, N, 1]
        mask_tensor = mask.to(input_tensor.dtype)
        if mask_tensor.dim() == 4:
            mask_tensor = mask_tensor.squeeze(-1)          # [B, N, N]
        # Fuse multiplication with a Triton kernel.
        left = _apply_gate_and_mask(left, left_gate, mask_tensor)
        right = _apply_gate_and_mask(right, right_gate, mask_tensor)

    # ------------------------------------------------------------------
    # 4) The core multiplicative update:
    #    out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h]
    #    This is a batched matmul for each hidden channel.
    # ------------------------------------------------------------------
    # reshape to [B, H, N, N] to use torch.matmul which internally calls cuBLAS.
    left_perm = left.permute(0, 3, 1, 2)   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2) # [B, H, N, N]

    # Perform (left @ rightᵀ) for each (b, h) pair.
    out = torch.matmul(left_perm, right_perm.transpose(-2, -1))  # [B, H, N, N]
    out = out.permute(0, 2, 3, 1)  # back to [B, N, N, H]

    # ------------------------------------------------------------------
    # 5) Second layer‑norm, output gate and final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    output = torch.nn.functional.linear(out, weights["to_out.weight"])

    return output

    4. Batched matrix multiplication per hidden channel:

    5. Second LayerNorm, apply output gate and final linear projection.
    """

    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # 1) Input layer‑norm
    # --------------------------------------------------------------------

        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # 2) Linear projections (no bias) + gating logits
    # --------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))

    # --------------------------------------------------------------------
    if config.get("nomask", False) or mask is None:

        right = right * right_gate
