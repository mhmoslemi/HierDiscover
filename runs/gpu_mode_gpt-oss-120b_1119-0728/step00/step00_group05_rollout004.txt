    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,

        left_ptrs = left_ptr + b * stride_l_b \
                    + i_off[:, None] * stride_l_i \
                    + k_off[None, :] * stride_l_k \
                    + h * stride_l_h

                     + k_off[:, None] * stride_r_k \
                     + j_off[None, :] * stride_r_j \
                     + h * stride_r_h

        acc += tl.dot(left, right)

    # store result
    out_ptrs = out_ptr + b * stride_o_b \
               + i_off[:, None] * stride_o_i \
               + j_off[None, :] * stride_o_j \
               + h * stride_o_h
    tl.store(out_ptrs, acc, mask = i_mask[:, None] & j_mask[None, :])

def custom_kernel(data):
    """
    Triton-accelerated forward pass of the TriMul module (outgoing version).
    The function performs:
      1) LayerNorm on the input.
      2) Linear projections for left/right and three gates.
      3) Masking and gating.
      4) Core tri‑multiplicative update:
           out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
         implemented with a Triton kernel.
      5) LayerNorm on the hidden dimension, out‑gate multiplication and final linear projection.
    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        * input_tensor : torch.Tensor [B, N, N, dim]
        * mask         : torch.Tensor [B, N, N] (or None)
        * weights      : dict of tensors (see code)
        * config       : dict with keys "dim", "hidden_dim"
    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    # layer norm on input
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5
    )
    # linear projections (no bias)
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)
    # mask handling
    if mask is None or mask.numel() == 0:
        mask = torch.ones_like(input_tensor[..., 0])  # [B,N,N]
    mask = mask.to(dtype=x.dtype, device=device).unsqueeze(-1)  # [B,N,N,1]

    left = left * mask
    right = right * mask
    # gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ensure contiguous tensors for Triton
    left = left.contiguous()
    right = right.contiguous()

    # core TriMul using Triton
    out_hidden = trimul_einsum(left, right)   # shape [B,N,N,hidden_dim]

    # second layer norm on hidden dimension
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_hidden = torch.nn.functional.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5
    )
    # out gate
    out_hidden = out_hidden * out_gate

    # final linear projection to original dim
    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out_hidden, to_out_weight)  # bias=False

    return out

"""
TriMul (outgoing) forward pass with Triton acceleration.

The algorithm follows the reference PyTorch implementation:
1. Layer‑norm the input tensor.
2. Project the normalized tensor to a hidden dimension (left/right) and
   compute three sigmoid‑gates (left, right, out).
3. Apply the input mask and the two gates to the left/right tensors.
4. Core computation:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   (Einstein sum “... i k d, ... j k d -> ... i j d”).  This O(N³) reduction
   over the sequence axis is implemented as a Triton kernel that performs a
   blocked matrix‑multiply for each hidden dimension.
5. Layer‑norm the hidden result, apply the out‑gate and finally a linear
   projection back to the original feature dimension.

Only the expensive reduction (step 4) is written in Triton; all other
operations use native PyTorch for simplicity and correctness.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _trimul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr):
    """
    Triton kernel that computes:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

    Each program instance processes one hidden dimension ``h`` for one
    batch element ``b`` and a tile (i_block, j_block) of the output matrix.
    """
    pid0 = tl.program_id(0)      # encodes (batch, hidden) → combined index
    pid_i = tl.program_id(1)     # i‑tile
    pid_j = tl.program_id(2)     # j‑tile

    # ----- decode batch index and hidden index (BLOCK_H = 1) -----
    h = pid0 % H                 # hidden dimension this program works on
    b = pid0 // H                # batch index

    # ----- i / j offsets for this tile -----
    i_off = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j_off = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    i_mask = i_off < N
    j_mask = j_off < N

    # accumulator for the tile (float32 for numerical safety)
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # ----- loop over the reduction dimension k -----
    for k in range(0, N, BLOCK_K):
        cur_K = tl.minimum(N - k, BLOCK_K)
        k_off = k + tl.arange(0, BLOCK_K)
        k_mask = k_off < N

        # left[b,i,k,h]   → shape (BLOCK_I, cur_K)
        left_ptrs = left_ptr + b * stride_l_b \
                                 + i_off[:, None] * stride_l_i \
                                 + k_off[None, :] * stride_l_k \
                                 + h * stride_l_h
        left = tl.load(left_ptrs,
                       mask=i_mask[:, None] & k_mask[None, :],
                       other=0.0)

        # right[b,j,k,h] transposed to (k, j) → shape (cur_K, BLOCK_J)
        right_ptrs = right_ptr + b * stride_r_b \
                                   + k_off[:, None] * stride_r_k \
                                   + j_off[None, :] * stride_r_j \
                                   + h * stride_r_h
        right = tl.load(right_ptrs,
                        mask=k_mask[:, None] & j_mask[None, :],
                        other=0.0)

        # matrix‑multiply the tile (BLOCK_I × cur_K) @ (cur_K × BLOCK_J)
        acc += tl.dot(left, right)

    # ----- write the tile to global memory -----
    out_ptrs = out_ptr + b * stride_o_b \
                         + i_off[:, None] * stride_o_i \
                         + j_off[None, :] * stride_o_j \
                         + h * stride_o_h
    tl.store(out_ptrs, acc, mask=i_mask[:, None] & j_mask[None, :])


def _trimul_einsum(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Wrapper that launches the Triton kernel for the hidden‑dimensional reduction.
    Input tensors must be contiguous and have shape [B, N, N, H].
    Returns a tensor of the same shape and dtype.
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, N, _, H = left.shape

    out = torch.empty_like(left)

    # strides expressed in element counts (not bytes)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # block sizes – chosen to give good occupancy on H100
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32

    grid = (B * H,
            (N + BLOCK_I - 1) // BLOCK_I,
            (N + BLOCK_J - 1) // BLOCK_J)

    _trimul_einsum_kernel[grid](
        left, right, out,
        B, N, H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_j, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J, BLOCK_K=BLOCK_K,
    )
    return out


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask         : torch.Tensor of shape [B, N, N] (or ``None``)
        - weights      : dict of model tensors
        - config       : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32).
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm on the input (dim‑wise)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) for left/right and the three gates
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )

    # ------------------------------------------------------------------
    # 3️⃣ Apply mask and the two gates
    # ------------------------------------------------------------------
    if mask is None or mask.numel() == 0:
        # no mask -> treat as all‑ones
        mask = torch.ones(
            (input_tensor.shape[0], input_tensor.shape[1], input_tensor.shape[2]),
            dtype=dtype,
            device=device,
        )
    mask = mask.to(dtype=dtype, device=device).unsqueeze(-1)  # [B,N,N,1]

    left = left * mask * left_gate
    right = right * mask * right_gate

    # ------------------------------------------------------------------
    # 4️⃣ Core TriMul reduction (Triton)
    # ------------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    hidden = _trimul_einsum(left, right)                     # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 5️⃣ Second layer‑norm, out‑gate, and final projection back to dim
    # ------------------------------------------------------------------
    hidden = torch.nn.functional.layer_norm(
        hidden,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    hidden = hidden * out_gate

    out = torch.nn.functional.linear(hidden, weights["to_out.weight"])
    return out
