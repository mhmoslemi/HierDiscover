def custom_kernel(data):
   input_tensor, mask, weights, config = data
   device = input_tensor.device
   bs, seq_len, _, dim = input_tensor.shape
   hidden_dim = config["hidden_dim"]
   nomask = config.get("nomask", False)
   eps = 1e-5

   # LN
   x_norm = torch.nn.functional.layer_norm(...)
   # Cast to half for subsequent linear ops
   x_norm_h = x_norm.to(torch.float16)

   # Cast weights for linear projection and gate to half
   left_proj_w = weights["left_proj.weight"].to(torch.float16)
   right_proj_w = weights["right_proj.weight"].to(torch.float16)
   left_gate_w = weights["left_gate.weight"].to(torch.float16)
   right_gate_w = weights["right_gate.weight"].to(torch.float16)
   out_gate_w = weights["out_gate.weight"].to(torch.float16)

   # Compute projections & gates
   left = torch.nn.functional.linear(x_norm_h, left_proj_w)    # [B,N,N,hidden]
   right = torch.nn.functional.linear(x_norm_h, right_proj_w)
   left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm_h, left_gate_w))
   right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm_h, right_gate_w))
   out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm_h, out_gate_w))

   if not nomask and mask is not None:
       mask_exp = mask.unsqueeze(-1).to(x_norm_h.dtype)
       left = left * mask_exp
       right = right * mask_exp

   # Apply gates
   left = left * left_gate
   right = right * right_gate

   # Bmm multiplication
   left_flat = left.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, seq_len, seq_len)  # [B*H,N,N]
   right_flat = right.permute(0, 3, 2, 1).contiguous().view(bs * hidden_dim, seq_len, seq_len)   # [B*H,N,N]
   # Use torch.bmm in half precision
   out_flat = torch.bmm(left_flat, right_flat)     # FP16 result
   out = out_flat.view(bs, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1).contiguous()  # [B,N,N,hidden]
   out = out.to(torch.float32)   # cast for LN

   # LayerNorm+gate using Triton kernel
   # kernel takes out, out_gate (float32), to_out_norm weight/bias, produce out_norm_gated ...
   out_norm_gated = layernorm_gate_kernel(...)
   # final linear projection
   result = torch.nn.functional.linear(out_norm_gated, weights["to_out.weight"])

   return result

@triton.jit
def layernorm_gate_proj_kernel(
    out_ptr, gate_ptr, result_ptr,
    norm_weight_ptr, norm_bias_ptr,
    proj_weight_ptr,  # shape (dim, hidden)
    # strides
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_res_b, stride_res_i, stride_res_j, stride_res_d,
    stride_norm_w, stride_norm_b,
    stride_proj_d, stride_proj_h,
    hidden, dim, eps,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    b = tl.program_id(2)
    i = tl.program_id(0)
    j = tl.program_id(1)

    # compute base offsets
    out_base = out_ptr + b*stride_out_b + i*stride_out_i + j*stride_out_j
    gate_base = gate_ptr + b*stride_gate_b + i*stride_gate_i + j*stride_gate_j

    # First pass: compute mean, var
    sum_val = tl.float32(0.0)
    sumsq_val = tl.float32(0.0)
    # Iterate over hidden dimension
    for offset in range(0, hidden, BLOCK_H):
        h = tl.arange(0, BLOCK_H) + offset
        mask = h < hidden
        out_vec = tl.load(out_base + h*stride_out_h, mask=mask, other=0.0)  # FP32
        sum_val += tl.sum(out_vec)
        sumsq_val += tl.sum(out_vec * out_vec)
    mean = sum_val / hidden
    var = sumsq_val / hidden - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # Second pass: compute final projection and store result
    for d_off in range(0, dim, BLOCK_D):
        d = tl.arange(0, BLOCK_D) + d_off
        mask_d = d < dim
        acc = tl.zeros((BLOCK_D,), dtype=tl.float32)
        for offset in range(0, hidden, BLOCK_H):
            h = tl.arange(0, BLOCK_H) + offset
            mask_h = h < hidden
            out_vec = tl.load(out_base + h*stride_out_h, mask=mask_h, other=0.0)
            gate_vec = tl.load(gate_base + h*stride_gate_h, mask=mask_h, other=0.0)

            # layernorm scale and bias per hidden
            w_norm = tl.load(norm_weight_ptr + h*stride_norm_w, mask=mask_h, other=0.0)
            b_norm = tl.load(norm_bias_ptr + h*stride_norm_b, mask=mask_h, other=0.0)

            # normalized, gated
            norm = (out_vec - mean) * inv_std
            norm = norm * w_norm + b_norm
            gated = norm * gate_vec  # shape (BLOCK_H,)

            # load projection weight submatrix (hidden x dim)
            w_proj = tl.load(proj_weight_ptr + h*stride_proj_h + d*stride_proj_d,
                             mask=mask_h[:, None] & mask_d[None, :], other=0.0)  # shape (BLOCK_H,BLOCK_D)

            # contribution to output dimensions
            # gated is shape (BLOCK_H,), need to be 1xK
            # tl.dot expects (M,K) and (K,N)
            # Convert to (1,BLOCK_H)
            gated_mat = gated[None, :]  # shape (1, BLOCK_H)
            # w_proj is (BLOCK_H, BLOCK_D)
            partial = tl.dot(gated_mat, w_proj)  # shape (1, BLOCK_D)
            acc += partial[0]  # add vector
        # store acc
        result_ptrs = result_ptr + b*stride_res_b + i*stride_res_i + j*stride_res_j + d*stride_res_d
        tl.store(result_ptrs, acc, mask=mask_d)

    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # LayerNorm (float32)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # Cast to fp16 for linear ops
    x_fp16 = x.to(torch.float16)

    # Cast weights to fp16
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left = torch.nn.functional.linear(x_fp16, left_proj_w)  # [B,N,N,hidden]
    right = torch.nn.functional.linear(x_fp16, right_proj_w)
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_fp16, left_gate_w))

        right = right * mask_exp

def layernorm_gate_proj_kernel(
    out_ptr, gate_ptr, result_ptr,
    norm_w_ptr, norm_b_ptr, proj_w_ptr,
    # strides
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,

    BLOCK_H: tl.constexpr, BLOCK_D: tl.constexpr,
):

    i = tl.program_id(0)
    j = tl.program_id(1)

    out_base = out_ptr + b*stride_out_b + i*stride_out_i + j*stride_out_j
    gate_base = gate_ptr + b*stride_gate_b + i*stride_gate_i + j*stride_gate_j

    # First pass: compute mean and var

    sumsq_val = tl.float32(0.0)

        h = tl.arange(0, BLOCK_H) + off

"""
Highly‑optimized Triton‑accelerated forward pass for the *outgoing* TriMul
operator (AlphaFold‑3).  The implementation follows the reference PyTorch
module but fuses the expensive parts:

* Layer‑norm on the input (float32, PyTorch).
* Linear projections, gates and optional mask are performed in FP16 to
  exploit Tensor‑Core speed.
* The N³ “tri‑multiplicative” update
      out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
  is expressed as a batched GEMM and executed with `torch.bmm`,
  which on an H100 uses the highly tuned cuBLAS Tensor‑Core kernel.
* A tiny Triton kernel fuses the final LayerNorm, the output‑gate and
  the write‑back to the result buffer, removing an extra memory‑bound
  pass.

The overall data‑flow is:

    x_norm  = LayerNorm(x)                               # [B,N,N,D]
    left    = (x_norm @ W_left)      (FP16)               # [B,N,N,H]
    right   = (x_norm @ W_right)     (FP16)               # [B,N,N,H]
    left_g  = sigmoid(x_norm @ W_left_gate) (FP16)       # [B,N,N,H]
    right_g = sigmoid(x_norm @ W_right_gate)(FP16)       # [B,N,N,H]
    out_g   = sigmoid(x_norm @ W_out_gate)  (FP32)       # [B,N,N,H]

    left  = left  * left_g   (mask applied before gating)
    right = right * right_g

    # -------------------- N³ update --------------------
    # view as (B*H, N, N) and perform B*H independent matmuls:
    #   out_flat[b*h,i,j] = Σ_k left_flat[b*h,i,k] * right_flat[b*h,k,j]
    out = torch.bmm(left_flat, right_flat)                # FP16 → FP32
    out = out.view(B, H, N, N).permute(0,2,3,1)           # [B,N,N,H]

    # ------------------ final LN + out‑gate -----------------
    # Triton kernel: per (b,i,j) vector of length H
    #   out_norm = (out - μ) / sqrt(σ²+ε) * w_LN + b_LN
    #   out_norm = out_norm * out_gate
    # result   = out_norm @ W_out   (torch.nn.functional.linear)
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fused LayerNorm + output‑gate.
# ----------------------------------------------------------------------
@triton.jit
def _layernorm_gate_kernel(
    out_ptr,            # FP32  [B,N,N,H]
    gate_ptr,           # FP32  [B,N,N,H]
    out_norm_ptr,       # FP32  [B,N,N,H] (output of this kernel)
    norm_w_ptr,         # FP32  [H]          (LayerNorm weight)
    norm_b_ptr,         # FP32  [H]          (LayerNorm bias)

    # strides (in elements) for the three tensors
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_norm_b, stride_norm_i, stride_norm_j, stride_norm_h,

    # static parameters
    hidden: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    One program instance processes a single (b,i,j) position.
    It first reduces over the hidden dimension to obtain mean/variance,
    then applies the LayerNorm, the output‑gate and writes the result.
    """
    b = tl.program_id(2)          # batch axis
    i = tl.program_id(0)          # seq i
    j = tl.program_id(1)          # seq j

    # base pointers for this (b,i,j)
    out_base  = out_ptr  + b*stride_out_b  + i*stride_out_i  + j*stride_out_j
    gate_base = gate_ptr + b*stride_gate_b + i*stride_gate_i + j*stride_gate_j
    norm_base = out_norm_ptr + b*stride_norm_b + i*stride_norm_i + j*stride_norm_j

    # -------------------------------------------------
    # 1️⃣  Compute mean and variance over the hidden axis
    # -------------------------------------------------
    sum_val   = tl.float32(0.0)
    sumsq_val = tl.float32(0.0)

    for off in range(0, hidden, BLOCK_H):
        h = tl.arange(0, BLOCK_H) + off               # hidden indices in this tile
        mask_h = h < hidden
        # load a slice of the raw output
        out_vec = tl.load(out_base + h*stride_out_h,
                          mask=mask_h, other=0.0)     # FP32
        sum_val   += tl.sum(out_vec)                  # scalar reduction
        sumsq_val += tl.sum(out_vec * out_vec)

    mean = sum_val / hidden
    var  = sumsq_val / hidden - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)                # FP32 scalar

    # -------------------------------------------------
    # 2️⃣  Apply LayerNorm, multiply by output‑gate and store
    # -------------------------------------------------
    for off in range(0, hidden, BLOCK_H):
        h = tl.arange(0, BLOCK_H) + off
        mask_h = h < hidden

        out_vec  = tl.load(out_base  + h*stride_out_h,
                           mask=mask_h, other=0.0)  # raw value
        gate_vec = tl.load(gate_base + h*stride_gate_h,
                           mask=mask_h, other=0.0)  # out‑gate

        # LayerNorm parameters (weight & bias are 1‑D, stride == 1)
        w = tl.load(norm_w_ptr + h, mask=mask_h, other=0.0)
        b_ = tl.load(norm_b_ptr + h, mask=mask_h, other=0.0)

        # (out - μ) / √(σ²+ε) * w + b
        normed = (out_vec - mean) * inv_std
        normed = normed * w + b_

        # multiply by the output‑gate
        gated = normed * gate_vec

        # write back
        tl.store(norm_base + h*stride_norm_h, gated, mask=mask_h)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul module.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, D] (float32)
        - mask          : torch.Tensor [B, N, N] (bool/float) or None
        - weights       : dict of model weights (all on CUDA)
        - config        : dict with keys ['dim', 'hidden_dim', 'nomask']
    Returns
    -------
    torch.Tensor of shape [B, N, N, D] (float32)
    """
    # ----------------------- unpack -----------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # -------------------------------------------------
    # 0️⃣  Input LayerNorm (float32, PyTorch)
    # -------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                                   # [B, N, N, D]  float32

    # -------------------------------------------------
    # 1️⃣  FP16 linear projections & gates
    # -------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # cast weights once to FP16
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    left  = torch.nn.functional.linear(x_fp16, left_proj_w)   # [B,N,N,H] FP16
    right = torch.nn.functional.linear(x_fp16, right_proj_w)  # [B,N,N,H] FP16

    # optional mask (broadcast over the hidden dimension)
    if (not nomask) and (mask is not None):
        mask_exp = mask.unsqueeze(-1).to(x_fp16.dtype)        # [B,N,N,1]
        left  = left * mask_exp
        right = right * mask_exp

    left_gate  = torch.sigmoid(torch.nn.functional.linear(x_fp16, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_fp16, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x_fp16, out_gate_w))  # FP16

    # apply gates
    left  = left  * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 2️⃣  N³ batched GEMM via torch.bmm (FP16 → FP32)
    # -------------------------------------------------
    # bring tensors to layout [B, H, N, N] and flatten batch*hidden
    left_f = left.permute(0, 3, 1, 2).contiguous().view(bs * hidden, seq_len, seq_len)
    # right needs a transpose on the last two dims (k,j)
    right_f = right.permute(0, 3, 2, 1).contiguous().view(bs * hidden, seq_len, seq_len)

    # torch.bmm automatically uses Tensor‑Cores for FP16 on H100
    out_f = torch.bmm(left_f, right_f)        # FP16 result, shape [B*H, N, N]

    # reshape back to [B,N,N,H] and up‑cast to FP32 for the next LayerNorm
    out = out_f.view(bs, hidden, seq_len, seq_len).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)               # [B,N,N,H] float32

    # out_gate from Step 1 is still FP16 → convert to float32
    out_gate = out_gate.to(torch.float32)     # [B,N,N,H] float32

    # -------------------------------------------------
    # 3️⃣  Fuse output‑LayerNorm & out‑gate in Triton
    # -------------------------------------------------
    out_norm = torch.empty_like(out)          # buffer for LN‑gated tensor

    # strides for the three tensors (in elements)
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h = out_gate.stride()
    stride_norm_b, stride_norm_i, stride_norm_j, stride_norm_h = out_norm.stride()
    # LayerNorm parameters (contiguous → stride == 1)
    norm_w = weights["to_out_norm.weight"]
    norm_b = weights["to_out_norm.bias"]
    # launch kernel one program per (i,j,b) position
    grid = (seq_len, seq_len, bs)
    BLOCK_H = 128                                     # hidden axis tile size
    _layernorm_gate_kernel[grid](
        out,
        out_gate,
        out_norm,
        norm_w,
        norm_b,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
        stride_norm_b, stride_norm_i, stride_norm_j, stride_norm_h,
        hidden,
        eps,
        BLOCK_H=BLOCK_H,
    )

    # -------------------------------------------------
    # 4️⃣  Final projection back to `dim` (standard linear)
    # -------------------------------------------------
    result = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

    return result
