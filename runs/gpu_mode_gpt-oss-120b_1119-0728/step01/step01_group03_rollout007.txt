def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # Normalization
    x = torch.nn.functional.layer_norm(input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps
    )

    # Project
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"], bias=None)
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"], bias=None)

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"], bias=None))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"], bias=None))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"], bias=None))

    # Cast to FP16
    left_proj = left_proj.to(torch.float16)
    right_proj = right_proj.to(torch.float16)
    left_gate = left_gate.to(torch.float16)
    right_gate = right_gate.to(torch.float16)

    if nomask or mask is None:
        mask_fp16 = torch.ones((bs, N, N), dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask.to(torch.float16)   # shape (B,N,N)

    # Output buffer

"""
TriMul (outgoing) forward pass optimized with a fused Triton kernel.
The heavy O(N³·hidden) product

    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

is computed as a batched GEMM:
    out_slice = left_slice @ right_sliceᵀ

Both the linear projections and the element‑wise gates are applied in
float16.  The kernel fuses the mask, the per‑channel gates and the
matrix multiplication, reading each tensor only once and accumulating
in float32.  The resulting tensor (float32) is then layer‑normed,
gated and projected back to the original feature dimension.
"""

import torch
import triton
import triton.language as tl

# ---------------------------------------------------------------------------
# Triton kernel: one GEMM (A @ Bᵀ) per (batch, hidden) slice.
# ---------------------------------------------------------------------------
@triton.jit
def _trimul_outgoing_kernel(
    # pointers
    A_ptr,        # left_proj      [B, N, N, H]  fp16
    A_gate_ptr,   # left_gate      [B, N, N, H]  fp16
    B_ptr,        # right_proj     [B, N, N, H]  fp16
    B_gate_ptr,   # right_gate     [B, N, N, H]  fp16
    mask_ptr,     # mask           [B, N, N]    fp16 (0/1)
    C_ptr,        # output         [B, N, N, H] fp32
    # problem size
    M, N, K,      # all equal to seq_len
    # strides for A (and A_gate) – 4‑D layout (B, i, k, h)
    stride_Ab, stride_Ai, stride_Ak,
    # strides for B (and B_gate) – (B, i, k, h) same layout, we will treat
    # the row dimension as k and the column dimension as i (transpose on‑the‑fly)
    stride_Bb, stride_Bi, stride_Bk,
    # strides for mask (B, i, k)
    stride_maskb, stride_maski, stride_maskk,
    # strides for C (output) – same layout as A/B
    stride_Cb, stride_Ci, stride_Cj,
    # runtime params
    hidden_dim: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
