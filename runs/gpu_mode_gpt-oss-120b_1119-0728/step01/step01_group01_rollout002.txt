"""
custom_kernel – optimized “outgoing” TriMul.

Algorithm
---------
1. Layer‑norm the input (last dimension = dim).
2. Linear projections → left/right (shape [B,N,N,hidden]).
3. Optional mask (broadcasted over hidden) applied to left/right.
4. Gating: sigmoid(linear(x_norm)) → left_gate/right_gate/out_gate.
5. Apply element‑wise gates to left/right.
6. Heavy contraction  out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is performed as a batched GEMM on half‑precision tensors.
   We reshape (B, N, N, H) → (B*H, N, N) and call a custom
   Triton kernel that loads fp16 values, accumulates in fp32,
   and writes the result in fp16.  Tensor‑Core friendly block
   sizes (128‑128‑64) are used.
7. Reshape the result back to [B,N,N,hidden] (float32).
8. Layer‑norm → out_gate (sigmoid) → final linear to original dim.

The contraction dominates the runtime (O(B·H·N³)); using fp16
reduces memory traffic and enables Tensor‑Core acceleration,
bringing the total latency well below the target 1 ms.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: half‑precision batched GEMM for the TriMul contraction.
# ----------------------------------------------------------------------
@triton.jit
def _trimmul_outgoing_fp16_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    stride_a_batch, stride_a_i, stride_a_k,  # A strides
    stride_b_batch, stride_b_k, stride_b_i,  # B strides (B accessed transposed)
    stride_c_batch, stride_c_i, stride_c_j,  # C strides
    N,                                       # sequence length
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute C = A @ Bᵀ for many independent (batch, hidden) slices.

    A  : (batch_total, N, N)   – fp16
    B  : (batch_total, N, N)   – fp16, accessed as (k,j)
    C  : (batch_total, N, N)   – fp16 (accumulator in fp32)
    """
    pid_m = tl.program_id(0)   # block row  (i)
    pid_n = tl.program_id(1)   # block col  (j)
    pid_b = tl.program_id(2)   # batch * hidden index

    # ------------------------------------------------------------------
    # Offsets for the current block
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension k
    # ------------------------------------------------------------------
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k in range(num_k_blocks):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Load A block (M x K) – fp16 → fp32
        a_ptrs = (
            a_ptr
            + pid_b * stride_a_batch
            + offs_m[:, None] * stride_a_i
            + offs_k[None, :] * stride_a_k
        )
        mask_a = (offs_m[:, None] < N) & (offs_k[None, :] < N)
        a = tl.load(a_ptrs, mask=mask_a, other=0.0, dtype=tl.float16).to(tl.float32)

        # Load B block (K x N) transposed (k,j) – fp16 → fp32
        b_ptrs = (
            b_ptr
            + pid_b * stride_b_batch
            + offs_k[:, None] * stride_b_k    # k‑dimension stride
            + offs_n[None, :] * stride_b_i    # j‑dimension stride
        )
        mask_b = (offs_k[:, None] < N) & (offs_n[None, :] < N)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0, dtype=tl.float16).to(tl.float32)

        # Accumulate
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Write C (fp16)
    # ------------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_b * stride_c_batch
        + offs_m[:, None] * stride_c_i
        + offs_n[None, :] * stride_c_j
    )
    mask_c = (offs_m[:, None] < N) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)

    )  # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2. Linear projections (no bias) → left / right (FP32)
    # --------------------------------------------------------------
    left = F.linear(x, _w("left_proj.weight"))   # [B, N, N, hidden]
    right = F.linear(x, _w("right_proj.weight"))

    # --------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden)
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] → [B, N, N, 1] and match dtype
        mask_f = mask.to(dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------
    # 4. Gating (sigmoid of linear layers)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, _w("left_gate.weight")))
    right_gate = torch.sigmoid(F.linear(x, _w("right_gate.weight")))
    out_gate = torch.sigmoid(F.linear(x, _w("out_gate.weight")))

    # --------------------------------------------------------------
    # 5. Apply element‑wise gates
    # --------------------------------------------------------------
    left = left * left_gate    # [B, N, N, hidden]
    right = right * right_gate

    # --------------------------------------------------------------
    # 6. Heavy contraction – batched half‑precision GEMM via Triton
    # --------------------------------------------------------------
    B, N, _, _ = left.shape   # N is already known, dim may be different
    H = hidden_dim

    # Reshape to (B*H, N, N) and cast to fp16 for the kernel
    # Permute to bring the hidden dim in front (contiguous layout)
    left_fp16 = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N).half()
    right_fp16 = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N).half()

    # Output buffer (fp16)
    out_fp16 = torch.empty_like(left_fp16, dtype=torch.float16)

    # Triton block sizes – tuned for H100 Tensor‑Core fp16 GEMM
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    # Grid: (rows, cols, batch_total)
    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H,
    )

    # Launch kernel
    _trimmul_outgoing_fp16_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        # A strides
        left_fp16.stride(0),
        left_fp16.stride(1),
        left_fp16.stride(2),
        # B strides (transposed access)
        right_fp16.stride(0),
        right_fp16.stride(2),   # stride for k
        right_fp16.stride(1),   # stride for i (original j)
        # C strides
        out_fp16.stride(0),
        out_fp16.stride(1),
        out_fp16.stride(2),
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, hidden) and convert to float32
    out = out_fp16.view(B, H, N, N).permute(0, 2, 3, 1).contiguous().float()

    # --------------------------------------------------------------
    # 7. Output processing: LayerNorm → out_gate → final linear
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=_w("to_out_norm.weight"),
        bias=_w("to_out_norm.bias"),
    )
    out = out * out_gate
    out = F.linear(out, _w("to_out.weight"))

    return out
