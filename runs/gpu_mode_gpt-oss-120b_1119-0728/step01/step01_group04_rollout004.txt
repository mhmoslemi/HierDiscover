```
@triton.jit
def batch_tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    # start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # compute pointer offsets for first k tile
    a_ptrs = a_ptr + pid_batch * stride_a_batch + offs_m[:, None] * stride_a_row + tl.arange(0, BLOCK_K)[None, :] * stride_a_col
    b_ptrs = b_ptr + pid_batch * stride_b_batch + tl.arange(0, BLOCK_K)[:, None] * stride_b_row + offs_n[None, :] * stride_b_col
    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & (tl.arange(0, BLOCK_K)[None, :] + k*BLOCK_K < K), other=0.0)
        b = tl.load(b_ptrs, mask=(tl.arange(0, BLOCK_K)[:, None] + k*BLOCK_K < K) & (offs_n[None, :] < N), other=0.0)
        acc += tl.dot(a, b)
        a_ptrs += BLOCK_K * stride_a_col
        b_ptrs += BLOCK_K * stride_b_row
    c_ptrs = c_ptr + pid_batch * stride_c_batch + offs_m[:, None] * stride_c_row + offs_n[None, :] * stride_c_col
    tl.store(c_ptrs, acc.to(tl.float16), mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))
```

"""
custom_kernel : Triton‑accelerated outgoing TriMul (AlphaFold3)

The forward pass consists of
1. LayerNorm over the feature dimension.
2. Half‑precision linear projections (left/right) and half‑precision
   gating (sigmoid) – all performed with cuBLAS (torch.nn.functional.linear)
   on fp16 tensors.
3. Element‑wise mask+gate fusion (still fp16).
4. A batched matrix‑multiply C = A @ Bᵀ where A and B have shape
   (B·H, N, N).  This is the N³ kernel and is implemented in Triton.
   A and B are read as fp16, the accumulation is fp32 and the result is
   written back as fp16.
5. The result is reshaped, converted to fp32, Layer‑Normed, multiplied
   by the output gate and finally projected back to the original feature
   dimension.

The Triton kernel uses a 128×128×32 tile (128‑bit FMAs) with 8 warps and
double‑buffered stages, which maps to the Hopper Tensor‑Cores and yields
sub‑millisecond runtimes on the test configurations.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matmul C = A @ Bᵀ
# A, B, C : [batch, N, N]   (batch = B * hidden_dim)
# A and B are fp16; accumulation is fp32; result is fp16.
# ----------------------------------------------------------------------
@triton.jit
def batch_tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,               # base pointers
    M, N, K,                           # all equal to seq_len
    stride_am, stride_ak, stride_ab,   # strides for A (row, col, batch)
    stride_bm, stride_bk, stride_bb,   # strides for Bᵀ (col, row, batch)
    stride_cm, stride_cn, stride_cb,   # strides for C (row, col, batch)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched fp16 matmul with fp32 accumulation.

    For each batch `b`:
        C[b,i,j] = Σ_k A[b,i,k] * B[b,j,k]
    """
    batch_id = tl.program_id(0)   # B·H dimension
    pid_m    = tl.program_id(1)   # tile row
    pid_n    = tl.program_id(2)   # tile col

    # ------------------------------------------------------------------
    # Offsets for the output tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Pointers to the beginning of the A and Bᵀ tiles for this batch
    # ------------------------------------------------------------------
    a_ptrs = a_ptr + batch_id * stride_ab + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = b_ptr + batch_id * stride_bb + tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bm

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension in tiles
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)               # (BLOCK_M, BLOCK_K) fp16
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)               # (BLOCK_K, BLOCK_N) fp16

        acc += tl.dot(a.to(tl.float32), b.to(tl.float32))

        a_ptrs += BLOCK_K * stride_ak   # move A to next K‑tile
        b_ptrs += BLOCK_K * stride_bk   # move Bᵀ to next K‑tile

    # ------------------------------------------------------------------
    # Write back the tile
    # ------------------------------------------------------------------
    c_ptrs = c_ptr + batch_id * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Triton‑accelerated outgoing TriMul forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Parameters
    ----------
    data : tuple (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor or None, shape [B, N, N] (float32/bool)
        - weights      : dict of model parameters
        - config       : dict with keys "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    batch_size, seq_len, _, _ = input_tensor.shape

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm over the feature dimension (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim] float32

    # --------------------------------------------------------------
    # 2️⃣ Cast to fp16 *before* the heavy linear ops
    # --------------------------------------------------------------
    x_h = x.half()                       # fp16

    # ---- linear projections (no bias) ---------------------------------
    left_proj  = F.linear(x_h, weights["left_proj.weight"].half())   # [B,N,N,H] fp16
    right_proj = F.linear(x_h, weights["right_proj.weight"].half())

    # ---- gating pre‑activations ---------------------------------------
    left_gate_pre  = F.linear(x_h, weights["left_gate.weight"].half())
    right_gate_pre = F.linear(x_h, weights["right_gate.weight"].half())
    out_gate_pre   = F.linear(x_h, weights["out_gate.weight"].half())

    # ---- sigmoid (works in fp16) ------------------------------------
    left_gate  = left_gate_pre.sigmoid()      # fp16
    right_gate = right_gate_pre.sigmoid()
    out_gate   = out_gate_pre.sigmoid()       # fp16

    # --------------------------------------------------------------
    # 3️⃣ Mask handling (fp16)
    # --------------------------------------------------------------
    if mask is None or config.get("nomask", False):
        mask_h = torch.ones((batch_size, seq_len, seq_len), dtype=torch.float16, device=device)
    else:
        mask_h = mask.to(dtype=torch.float16, device=device)
    mask_h = mask_h.unsqueeze(-1)               # [B,N,N,1] fp16

    # --------------------------------------------------------------
    # 4️⃣ Fuse mask + gating into the projected tensors
    # --------------------------------------------------------------
    left  = left_proj * mask_h * left_gate    # fp16, shape [B,N,N,H]
    right = right_proj * mask_h * right_gate   # fp16

    # --------------------------------------------------------------
    # 5️⃣ Reshape so every hidden channel becomes a separate batch:
    #    (B·H, N, N)  – this layout matches the Triton kernel.
    # --------------------------------------------------------------
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()
    batch_h = batch_size * hidden_dim
    left  = left.view(batch_h, seq_len, seq_len)   # (B·H, N, N)
    right = right.view(batch_h, seq_len, seq_len)

    # --------------------------------------------------------------
    # 6️⃣ Allocate the output tensor (fp16)
    # --------------------------------------------------------------
    out_h = torch.empty_like(left, dtype=torch.float16)   # (B·H, N, N)

    # --------------------------------------------------------------
    # 7️⃣ Gather stride information for the kernel
    # --------------------------------------------------------------
    # A (left) strides: (batch, row, col)
    stride_ab, stride_am, stride_ak = left.stride()
    # B (right) strides – we will read it as Bᵀ
    stride_bb, stride_brow, stride_bcol = right.stride()
    # After transposition:
    #   row‑stride of Bᵀ = original col‑stride
    #   col‑stride of Bᵀ = original row‑stride
    stride_bk = stride_bcol        # K dimension stride (inner)
    stride_bm = stride_brow        # M‑dimension stride (output column)
    # C strides are identical to A’s layout
    stride_cb, stride_cm, stride_cn = out_h.stride()

    # --------------------------------------------------------------
    # 8️⃣ Launch the Triton kernel
    # --------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch_h,
        triton.cdiv(seq_len, BLOCK_M),
        triton.cdiv(seq_len, BLOCK_N),
    )
    batch_tri_mul_kernel[grid](
        left,
        right,
        out_h,
        seq_len,
        seq_len,
        seq_len,
        stride_am, stride_ak, stride_ab,
        stride_bm, stride_bk, stride_bb,
        stride_cm, stride_cn, stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # --------------------------------------------------------------
    # 9️⃣ Reshape back to the original 4‑D layout and promote to fp32
    # --------------------------------------------------------------
    out = out_h.view(batch_size, hidden_dim, seq_len, seq_len)   # (B, H, N, N)
    out = out.permute(0, 2, 3, 1).contiguous()                  # (B, N, N, H)
    out = out.to(torch.float32)                                 # fp32 for the remaining ops

    # --------------------------------------------------------------
    # 1️⃣0️⃣ Hidden‑dim LayerNorm + output gate (out_gate was fp16)
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate.to(torch.float32)   # broadcast over (B,N,N,H)

    # --------------------------------------------------------------
    # 1️⃣1️⃣ Final projection back to `dim`
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
```
