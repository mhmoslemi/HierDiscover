import torch.nn.functional as F

import triton.language as tl

"""
TriMul “outgoing” forward pass (AlphaFold3 style).

Algorithm
---------
1. Layer‑norm the input tensor `x` over the last dimension (dim).
2. Cast `x` to float16 and compute the four linear projections
   (left, right, left_gate, right_gate) and the output gate.
   All linear layers are bias‑free; weights are also cast to float16.
3. Fuse projection, gating and optional mask by element‑wise multiplies
   (still in float16).
4. Compute the core O(N³) operation
        out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]
   which is equivalent to a batched matrix multiplication
        out_d = left_d @ right_dᵀ
   for each hidden channel `d`.  This is implemented by a custom
   Triton kernel that works on half‑precision data, accumulates in fp32,
   and uses `BLOCK_M = 128`, `BLOCK_N = 128`, `BLOCK_K = 64` (good
   occupancy on H100).
5. Reshape the result to `[B, N, N, hidden_dim]` and cast back to fp32.
6. Apply a second LayerNorm (over `hidden_dim`), multiply by the
   previously computed output gate, and finally project back to `dim`
   with a bias‑free linear layer.
7. Return the tensor of shape `[B, N, N, dim]` (float32).

The most expensive part (step 4) is fully handled in Triton, while
the cheap linear / norm / sigmoid operations stay in PyTorch.
"""

# ----------------------------------------------------------------------
# Triton kernel: batched (B*H) matrix multiplication C = A @ Bᵀ
#   A : (batch, M, K)   (half)
#   B : (batch, K, N)   (half)
#   C : (batch, M, N)   (half, accumulation in fp32)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # global pointers
    M, N, K,                           # matrix dimensions
    stride_ab, stride_am, stride_ak,   # strides for A (batch, M, K)
    stride_bb, stride_bk, stride_bn,   # strides for B (batch, K, N)
    stride_cb, stride_cm, stride_cn,   # strides for C (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)          # batch index (B*H)
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile col

    # Offsets for the output block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Compute pointers for the current tiles of A and B
        a_ptrs = (
            a_ptr
            + pid_b * stride_ab
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak
        )
        b_ptrs = (
            b_ptr
            + pid_b * stride_bb
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )

        # Bounds checking
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)   # fp16
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)   # fp16

        # Multiply‑accumulate (fp16 inputs, fp32 accumulator)
        acc += tl.dot(a, b)

    # Write the result back to global memory
    c_ptrs = (
        c_ptr
        + pid_b * stride_cb
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=c_mask)


def _batched_matmul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication using the Triton kernel above.
    Arguments:
        a: Tensor of shape (B*H, M, K)  (dtype torch.float16)
        b: Tensor of shape (B*H, K, N)  (dtype torch.float16)
    Returns:
        c: Tensor of shape (B*H, M, N)  (dtype torch.float16)
    """
    assert a.shape[0] == b.shape[0]
    B_h, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2

    c = torch.empty((B_h, M, N), device=a.device, dtype=a.dtype)

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64          # larger K tile reduces loop overhead

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B_h,
    )

    _batched_matmul_kernel[grid](
        a,
        b,
        c,
        M,
        N,
        K,
        a.stride(0),
        a.stride(1),
        a.stride(2),
        b.stride(0),
        b.stride(1),
        b.stride(2),
        c.stride(0),
        c.stride(1),
        c.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,        # leverage tensor cores on H100
    )
    return c

def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” module.
    Arguments:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        output tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2) Cast to float16 for the heavy path
    # ------------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections (bias‑free) in float16
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left = F.linear(x_f16, left_proj_w)    # [B, N, N, hidden_dim]
    right = F.linear(x_f16, right_proj_w)

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid) in float16
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x_f16, left_gate_w))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x_f16, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_f16, out_gate_w))    # used later

    # ------------------------------------------------------------------
    # 5) Apply gates (and optional mask) – still in float16
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    if not nomask and mask_tensor is not None:
        mask_f16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f16
        right = right * mask_f16

    # ------------------------------------------------------------------
    # 6) Core pairwise multiplication via batched half‑precision matmul
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # B, N, N, hidden_dim

    # Rearrange to (B, hidden_dim, N, N) where the last two dims are (M,K) and (K,N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N) -> (B, H, M, K)
    right_perm_T = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, N)

    # Collapse batch & hidden dimensions
    BH = B * hidden_dim
    left_flat = left_perm.view(BH, N, N)          # (B*H, M, K)
    right_flat = right_perm_T.view(BH, N, N)     # (B*H, K, N)

    # Batched matmul (half precision, fp32 accumulation)
    out_flat = _batched_matmul(left_flat, right_flat)   # (B*H, N, N)  half

    # Reshape back to (B, N, N, hidden_dim) – still half
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Output LayerNorm over hidden_dim (float32)
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out_fp32 = F.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8) Apply output gate (float32)
    # ------------------------------------------------------------------
    out_gate_fp32 = out_gate.to(torch.float32)
    out_fp32 = out_fp32 * out_gate_fp32

    # ------------------------------------------------------------------
    # 9) Final linear projection back to `dim` (float32)
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"]
    result = F.linear(out_fp32, to_out_w)   # [B, N, N, dim]  float32

    return result
