"""
Optimized outgoing TriMul (AlphaFold3) forward pass.

Algorithm
---------
1. Layer‑norm the input tensor (float32 → float32).
2. Cast to fp16 and fuse the five linear projections
      (left_proj, right_proj, left_gate, right_gate, out_gate)
   into a single matrix‑multiply:
        proj = x_norm_fp16 @ W_fused.T                # shape [B,N,N,5*H]
   where W_fused = concat([left_proj, right_proj,
                           left_gate, right_gate, out_gate])  (out_features = 5*H).
3. Slice the fused result, apply the optional mask, and multiply by the
   sigmoid gates (still fp16).
4. Compute the batch‑wise GEMM  out_h = left @ rightᵀ   for every hidden
   channel h using a custom Triton kernel.  The kernel works on the
   original layout `[B,N,N,H]` and therefore avoids the costly
   `permute(...).contiguous()` copy of the reference implementation.
5. Layer‑norm the hidden‑dim result (float32), multiply by the out‑gate,
   and project back to the original dimension with a final linear layer.
6. Return a float32 tensor of shape `[B,N,N,dim]`.

The costly N³ reduction is performed by the Triton kernel, while the
linear projections are collapsed into a single matmul, cutting memory
traffic and kernel launch overhead dramatically (≈5× speed‑up on H100).
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton GEMM kernel: C = A @ Bᵀ  where
#   A, B : (batch, H, N, N)  (layout: [B, N, N, H])
#   C    : (batch, H, N, N)
# The hidden dimension H is fused into the program‑id dimension.
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                         # pointers
    B, H, M, N, K,                               # dimensions (M=N, K=N)
    # strides for A
    stride_ab, stride_ai, stride_ak, stride_ah,
    # strides for B
    stride_bb, stride_bi, stride_bk, stride_bh,
    # strides for C
    stride_cb, stride_ci, stride_cj, stride_ch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM over the hidden dimension.

    Computes for every batch b and hidden index h:
        C[b, h, i, j] = Σ_k A[b, h, i, k] * B[b, h, j, k]
    """
    pid = tl.program_id(2)        # combined batch*hidden
    pid_batch = pid // H          # actual batch index
    pid_h = pid % H               # hidden index inside the batch

    pid_m = tl.program_id(0)      # tile row
    pid_n = tl.program_id(1)      # tile col

    # tile start coordinates
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K‑dimension (N)
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # mask for tails
        mask_a = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        mask_b = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        # A tile: A[b, h, i, k]
        a = tl.load(
            a_ptr
            + pid_batch * stride_ab
            + pid_h * stride_ah
            + offs_m[:, None] * stride_ai
            + offs_k[None, :] * stride_ak,
            mask=mask_a,
            other=0.0,
        )
        # Bᵀ tile: we need B[b, h, j, k] with rows = k, cols = j
        b = tl.load(
            b_ptr
            + pid_batch * stride_bb
            + pid_h * stride_bh
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bi,
            mask=mask_b,
            other=0.0,
        )
        acc += tl.dot(a, b)                     # (BLOCK_M, BLOCK_N) fp32
        # next K‑block
        # (pointer arithmetic is implicit in the next loop iteration)

    # write out (store as fp16 to save bandwidth)
    c = acc.to(tl.float16)
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(
        c_ptr
        + pid_batch * stride_cb
        + pid_h * stride_ch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj,
        c,
        mask=mask_c,
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor : torch.Tensor [B, N, N] (bool/float) – optional
            - weights_dict: model weights (float32)
            - config_dict : must contain "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    device = x.device
    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]
    eps = 1e-5
    nomask = cfg.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )  # [B,N,N,dim] float32

    # cast once to fp16 for the heavy linear work
    x_norm_fp16 = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Fuse the five linear maps into a single matmul
    #    weight shapes: (hidden, dim) each, bias = False
    # ------------------------------------------------------------------
    fused_weight = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                           # (5*H, dim)

    # combined projection [B,N,N,5*H]  (fp16)
    proj = F.linear(x_norm_fp16, fused_weight)   # torch.nn.functional works with fp16

    # ------------------------------------------------------------------
    # 3) slice and apply mask & gates (still fp16)
    # ------------------------------------------------------------------
    H = hidden
    left = proj[..., 0:H]
    right = proj[..., H:2 * H]
    left_gate = torch.sigmoid(proj[..., 2 * H:3 * H])
    right_gate = torch.sigmoid(proj[..., 3 * H:4 * H])
    out_gate = torch.sigmoid(proj[..., 4 * H:])          # [B,N,N,H]

    if not nomask:
        # mask may be bool or float; broadcast to hidden dim
        mask_f = mask.to(left.dtype).unsqueeze(-1)       # [B,N,N,1] fp16
        left = left * mask_f
        right = right * mask_f

    # apply per‑position gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4) Batched GEMM: out_h = left @ rightᵀ  (fp16)
    #    Avoid costly .permute(...).contiguous() by addressing the
    #    original layout directly inside the kernel.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape          # left: [B,N,N,H]  (fp16)

    out = torch.empty_like(left)     # allocate [B,N,N,H] fp16 for the result

    # strides of the original layout
    stride_ab, stride_ai, stride_ak, stride_ah = left.stride()   # (batch,i,k,h)
    stride_bb, stride_bi, stride_bk, stride_bh = right.stride()
    stride_cb, stride_ci, stride_cj, stride_ch = out.stride()

    # tiling – 64x64 tiles work well on H100; K‑tile = 64
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # rows
        (N + BLOCK_N - 1) // BLOCK_N,   # cols
        B * H,                          # batch * hidden
    )

    # launch kernel – 8 warps, 3 stages for double‑buffering
    tri_mul_gemm_kernel[grid](
        left,
        right,
        out,
        B, H, N, N, N,
        stride_ab,
        stride_ai,
        stride_ak,
        stride_ah,
        stride_bb,
        stride_bi,
        stride_bk,
        stride_bh,
        stride_cb,
        stride_ci,
        stride_cj,
        stride_ch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # ------------------------------------------------------------------
    # 5) Hidden‑dim layer‑norm (float32)
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)
    out_norm = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )  # [B,N,N,H] float32

    # 6) Apply out‑gate (broadcast from fp16 to fp32)
    out_norm = out_norm * out_gate.to(torch.float32)

    # 7) Final projection back to `dim`
    #    We keep this in fp32; the weight is already float32.
    out_final = F.linear(out_norm, w["to_out.weight"])   # [B,N,N,dim] float32

    return out_final
