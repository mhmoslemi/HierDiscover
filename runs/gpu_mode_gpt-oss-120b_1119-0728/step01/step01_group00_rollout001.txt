"""
TriMul (outgoing) – Triton‑accelerated implementation
====================================================

The heavy part of the TriMul forward pass is the three‑dimensional
einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which can be seen as a batched GEMM for every hidden dimension *d*.
For a tensor of shape [B,N,N,H] this is equivalent to
performing `B*H` independent matrix multiplications of size N×N.

The kernel below executes this GEMM in pure Triton:

* A and B are stored in FP16, the accumulation is done in FP32.
* Tiling parameters (BLOCK_M, BLOCK_N, BLOCK_K) are chosen to
  fully exploit H100 Tensor‑Cores while staying inside the shared‑memory
  budget (≈32 KB per block for 128×128×64 tiles).
* The batch‑group dimension (B·H) is handled with a third program ID,
  meaning a single kernel launch covers the whole workload.
* Out‑of‑bounds loads are masked; the result is written back as FP16
  and later cast to FP32 for the remaining point‑wise operations.

All other operations (LayerNorm, linear projections, sigmoid‑gates,
masking, final LayerNorm/linear) are performed with PyTorch
(either FP32 or FP16 where appropriate) because they are negligible
compared to the O(N³) work of the GEMM.

The function `custom_kernel` follows the required signature:
    (input, mask, weights, config) → output tensor of shape [B,N,N,D] (float32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------
# Triton batched GEMM used for the “einsum” part of TriMul.
# -------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Computes, for each group g (g = batch * hidden_dim):
        C[g,i,j] = Σ_k A[g,i,k] * B[g,k,j]

    Shapes:
        A : (G, M, K)
        B : (G, K, N)
        C : (G, M, N)

    All tensors are stored in row‑major order.
    """

    pid_m = tl.program_id(0)          # tile row   (i)
    pid_n = tl.program_id(1)          # tile col   (j)
    pid_g = tl.program_id(2)          # group id   (batch*hidden)

    # -----------------------------------------------------------------
    # Tile offsets
    # -----------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # -----------------------------------------------------------------
    # Pointers to the current tile of A and B
    # -----------------------------------------------------------------
    a_ptrs = a_ptr + pid_g * stride_a_batch \
                   + offs_m[:, None] * stride_a_m \
                   + offs_k[None, :] * stride_a_k
    b_ptrs = b_ptr + pid_g * stride_b_batch \
                   + offs_k[:, None] * stride_b_k \
                   + offs_n[None, :] * stride_b_n

    # -----------------------------------------------------------------
    # Accumulator (float32)
    # -----------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of K‑tiles we have to iterate over
    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for _ in range(num_k_tiles):
        # Load a and b (fp16) with out‑of‑bounds masking
        a = tl.load(
            a_ptrs,
            mask=(offs_m < M)[:, None] & (offs_k < K)[None, :],
            other=0.0,
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k < K)[:, None] & (offs_n < N)[None, :],
            other=0.0,
        )
        # Accumulate dot product (fp16×fp16 → fp32)
        acc += tl.dot(a, b)

        # Move to the next K‑tile
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # -----------------------------------------------------------------
    # Write back C (cast back to fp16)
    # -----------------------------------------------------------------
    c_ptrs = c_ptr + pid_g * stride_c_batch \
                     + offs_m[:, None] * stride_c_m \
                     + offs_n[None, :] * stride_c_n
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=(offs_m < M)[:, None] & (offs_n < N)[None, :],
    )


# -------------------------------------------------------------
# Entry point: TriMul forward pass (outgoing variant)
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        * input_tensor : torch.Tensor, shape [B, N, N, D] (float32 or fp16)
        * mask_tensor  : torch.Tensor, shape [B, N, N] or None
        * weights_dict : mapping from weight name to torch.Tensor
        * config_dict  : contains ``dim`` (D) and ``hidden_dim`` (H) and
                         optional ``nomask`` (bool)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, D] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]               # D
    hidden_dim = config["hidden_dim"] # H
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # Helper to fetch weight tensors (already on correct device)
    # -----------------------------------------------------------------
    def _w(name):
        return weights[name].to(device)

    # Layer‑norm on the input (norm over the last dim D)
    norm_weight = _w("norm.weight")
    norm_bias   = _w("norm.bias")

    # Linear projection weights (no bias)
    left_proj_weight  = _w("left_proj.weight")
    right_proj_weight = _w("right_proj.weight")
    left_gate_weight  = _w("left_gate.weight")
    right_gate_weight = _w("right_gate.weight")
    out_gate_weight   = _w("out_gate.weight")
    to_out_norm_weight = _w("to_out_norm.weight")
    to_out_norm_bias   = _w("to_out_norm.bias")
    to_out_weight      = _w("to_out.weight")   # shape (dim, hidden_dim)

    # -----------------------------------------------------------------
    # 1) Input LayerNorm
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )   # (B, N, N, D)

    # -----------------------------------------------------------------
    # 2) Linear projections to hidden dimension (float32)
    # -----------------------------------------------------------------
    # Shape after projection: (B, N, N, H)
    left  = F.linear(x, left_proj_weight)   # left_proj: (D, H)
    right = F.linear(x, right_proj_weight)

    # -----------------------------------------------------------------
    # 3) Apply optional mask (broadcast over hidden dim)
    # -----------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # mask: (B, N, N) → (B, N, N, 1)
        mask_fp = mask_tensor.to(dtype).unsqueeze(-1)
        left  = left * mask_fp
        right = right * mask_fp

    # -----------------------------------------------------------------
    # 4) Compute sigmoid gates (float32)
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))   # (B,N,N,H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))    # (B,N,N,H)

    # -----------------------------------------------------------------
    # 5) Cast to fp16 and apply element‑wise gates
    # -----------------------------------------------------------------
    left  = left.to(torch.float16) * left_gate.to(torch.float16)
    right = right.to(torch.float16) * right_gate.to(torch.float16)

    # -----------------------------------------------------------------
    # 6) Rearrange tensors for the batched GEMM
    #    left  : (B, N, N, H) → (B, H, N, N)  (i,k)
    #    right : (B, N, N, H) → (B, H, N, N)  (k,j)
    # -----------------------------------------------------------------
    #   After permute+contiguous the layout is
    #   (B, H, N, N) with stride (H*N*N, N*N, N, 1)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N) – i,k
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B,H,N,N) – k,j

    B, H, N, _ = left_perm.shape
    groups = B * H                     # total number of independent matmuls

    # Collapse the leading dimensions → (G, N, N)
    A = left_perm.view(groups, N, N)
    B_mat = right_perm.view(groups, N, N)

    # Output buffer for the GEMM (fp16)
    C = torch.empty_like(A, dtype=torch.float16, device=device)

    # -----------------------------------------------------------------
    # 7) Launch Triton batched GEMM
    # -----------------------------------------------------------------
    BLOCK_M = 128   # rows per block
    BLOCK_N = 128   # cols per block
    BLOCK_K = 64    # reduction tile

    grid = (
        triton.cdiv(N, BLOCK_M),   # number of tiles along M
        triton.cdiv(N, BLOCK_N),   # number of tiles along N
        groups,                    # one program per group (batch*hidden)
    )

    _batched_matmul_kernel[grid](
        A, B_mat, C,
        N, N, N,
        A.stride(0), A.stride(1), A.stride(2),   # batch, m, k strides
        B_mat.stride(0), B_mat.stride(1), B_mat.stride(2),   # batch, k, n strides
        C.stride(0), C.stride(1), C.stride(2),   # batch, m, n strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 8) Reshape back to (B, N, N, H) and cast to fp32 for the rest
    # -----------------------------------------------------------------
    out_hidden = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)
    out_hidden = out_hidden.to(torch.float32)

    # -----------------------------------------------------------------
    # 9) Output LayerNorm, gating, final linear projection back to D
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate                       # element‑wise gating (fp32)
    out = F.linear(out, to_out_weight)         # (B,N,N,D)

    return out
```
