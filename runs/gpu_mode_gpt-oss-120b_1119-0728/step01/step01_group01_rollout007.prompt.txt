import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
custom_kernel – TriMul “outgoing” implementation.

The heavy tensor contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed by a dedicated Triton GEMM kernel.
All other operations (LayerNorm, linear projections, gating) are done with
standard PyTorch functional calls.  The kernel works on a flattened
(batch * hidden)-dimensional view of the tensors and computes a batched
matrix‑multiply A @ Bᵀ for each hidden slice.  The implementation keeps
the tensors in fp32 for exactness, but the Triton computation can be
switched to fp16 by casting before the kernel launch if desired.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul contraction.
# ----------------------------------------------------------------------
@triton.jit
def _trimmul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,                          # pointers
    stride_a_batch, stride_a_i, stride_a_k,       # A strides
    stride_b_batch, stride_b_k, stride_b_i,       # B strides (B is accessed transposed)
    stride_c_batch, stride_c_i, stride_c_j,       # C strides
    N,                                            # sequence length (square matrices)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block col
    pid_b = tl.program_id(2)          # batch * hidden index

    # ------------------------------------------------------------------
    # Offsets for the current block
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension k
    # ------------------------------------------------------------------
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k_blocks):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Masks for out‑of‑bounds loads
        mask_a = (offs_m[:, None] < N) & (offs_k[None, :] < N)
        mask_b = (offs_k[:, None] < N) & (offs_n[None, :] < N)

        # A block  (shape: BLOCK_M x BLOCK_K)
        a_ptrs = (
            a_ptr
            + pid_b * stride_a_batch
            + offs_m[:, None] * stride_a_i
            + offs_k[None, :] * stride_a_k
        )
        a = tl.load(a_ptrs, mask=mask_a, other=0.0).to(tl.float32)

        # B block, transposed (shape: BLOCK_K x BLOCK_N)
        b_ptrs = (
            b_ptr
            + pid_b * stride_b_batch
            + offs_k[:, None] * stride_b_k
            + offs_n[None, :] * stride_b_i
        )
        b = tl.load(b_ptrs, mask=mask_b, other=0.0).to(tl.float32)

        # Accumulate the dot product
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Write the result
    # ------------------------------------------------------------------
    offs_i = offs_m
    offs_j = offs_n
    mask_c = (offs_i[:, None] < N) & (offs_j[None, :] < N)

    c_ptrs = (
        c_ptr
        + pid_b * stride_c_batch
        + offs_i[:, None] * stride_c_i
        + offs_j[None, :] * stride_c_j
    )
    tl.store(c_ptrs, acc, mask=mask_c)


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operation.
    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        input_tensor : [B, N, N, dim]   (float32, CUDA)
        mask_tensor  : [B, N, N]       (float32/bool/None)
        weights_dict : mapping of weight names to tensors
        config_dict  : {"dim": int,
                        "hidden_dim": int,
                        "nomask": bool (optional)}
    Returns
    -------
    out : torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # Helper to fetch weights (they are already on the correct device)
    # --------------------------------------------------------------
    def _w(name):
        return weights[name]

    # --------------------------------------------------------------
    # LayerNorm over the last dimension
    # --------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=_w("norm.weight"),
        bias=_w("norm.bias"),
    )  # [B, N, N, dim]

    # --------------------------------------------------------------
    # Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x_norm, _w("left_proj.weight"))   # [B, N, N, hidden]
    right = F.linear(x_norm, _w("right_proj.weight"))

    # --------------------------------------------------------------
    # Optional mask (applied before gating)
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] -> broadcast over hidden dim
        mask_f = mask.to(dtype).unsqueeze(-1)        # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------
    # Gating (sigmoid of linear layers)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, _w("left_gate.weight")))
    right_gate = torch.sigmoid(F.linear(x_norm, _w("right_gate.weight")))
    out_gate = torch.sigmoid(F.linear(x_norm, _w("out_gate.weight")))

    left = left * left_gate      # element‑wise
    right = right * right_gate

    # --------------------------------------------------------------
    # Heavy contraction – implemented as a batched GEMM in Triton
    # --------------------------------------------------------------
    # Rearrange to (B, hidden, N, N) and flatten the first two dims:
    left_perm = left.permute(0, 3, 1, 2).contiguous()     # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_perm.shape
    batch_total = B * H

    left_flat = left_perm.view(batch_total, N, N)   # [B*H, N, N]
    right_flat = right_perm.view(batch_total, N, N)

    out_flat = torch.empty_like(left_flat)          # result buffer

    # Triton block sizes – tuned for H100 / FP32
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        batch_total,
    )

    _trimmul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        # strides for A
        left_flat.stride(0),
        left_flat.stride(1),
        left_flat.stride(2),
        # strides for B (transposed access)
        right_flat.stride(0),
        right_flat.stride(2),   # stride for k in B
        right_flat.stride(1),   # stride for i (original j) in B
        # strides for C
        out_flat.stride(0),
        out_flat.stride(1),
        out_flat.stride(2),
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, hidden)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # Output processing: LayerNorm → gating → final linear
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=_w("to_out_norm.weight"),
        bias=_w("to_out_norm.bias"),
    )
    out = out * out_gate                           # element‑wise gate
    out = F.linear(out, _w("to_out.weight"))       # [B, N, N, dim]

    return out
